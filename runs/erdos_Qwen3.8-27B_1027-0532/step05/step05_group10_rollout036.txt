- Since h is binary: = h[i-(m-n+1)] (the "1" case: h[...]=0 so 1-0=1; the "0" case: h[...]=1 so 1-1=0)
- Wait: 1-h[x] where h[x] is 0 or 1: if h[x]=0 then 1-h[x]=1, if h[x]=1 then 1-h[x]=0.
- So the decrease is (1-h[i-(m-n+1)]) which is 1 where h[i-(m-n+1)]=0 and 0 where h[i-(m-n+1)]=1.

- c[m] increases by 1*(1-h[j-(m-(n-1))]) for m ∈ [j, j+n-1]

- New c[m] = old c[m] + delta[m]
- We need max_m (old c[m] + delta[m]) < old_max

        top_k = set(np.argsort(corr)[-n_active:].tolist())

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.95, damping + 0.02)

        if it % 100 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.7)

    return best_h, best_c5

def binary_sa_search(h_binary, n, rng, time_limit, T_init=0.005, T_min=0.0005, cool=0.9995):
    start = time.time()
    h = h_binary.copy().astype(float)
    target = n // 2
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    T = T_init
    no_improve = 0
    half_n = n - 1

    while time.time() - start < time_limit and T > T_min:
        c5_cur = corr.max() * dx

        peak_idx = int(np.argmax(corr))
        peak_lag = peak_idx - half_n

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        candidates = []
        seen = set()

        if abs(peak_lag) >= 1:
            k = peak_lag
            if k > 0:
                for idx in range(len(zeros)):
                    i = zeros[idx]
                    j = i + k
                    if j < n and h[j] == 1:
                        key = (i, j)
                        if key not in seen:
                            seen.add(key)
                            candidates.append(key)
            else:
                k2 = -k
                for idx in range(len(zeros)):
                    i = zeros[idx]
                    j = i - k2
                    if j >= 0 and j < n and h[j] == 1:
                        key = (i, j)
                        if key not in seen:
                            seen.add(key)
                            candidates.append(key)

        if len(candidates) < 50:
            for idx in range(min(len(ones), 30)):
                i = ones[idx]
                for jdx in range(min(len(zeros), 30)):
                    j = zeros[jdx]
                    if abs(i - j) <= 15:
                        key = (i, j)
                        if key not in seen:
                            seen.add(key)
                            candidates.append(key)

        for _ in range(15):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            key = (i, j)
            if key not in seen:
                seen.add(key)
                candidates.append(key)

        rng.shuffle(candidates)

        found_improvement = False
        for (i, j) in candidates[:80]:
            if h[i] != 0 or h[j] != 1:
                continue
            h[i] = 1
            h[j] = 0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = new_corr.max() * dx
            delta = new_c5 - c5_cur
            if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
                corr = new_corr
                c5_cur = new_c5
                found_improvement = True
                break
            else:
                h[i] = 0
                h[j] = 1

        if not found_improvement:
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')

        T *= cool

        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

    return best_h, best_c5

def binary_hill_climb(h_binary, n, rng, time_limit, max_rounds=5000):
    start = time.time()
    h = h_binary.copy().astype(float)
    target = n // 2
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    stagnation = 0

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 100:
                break

        peak_idx = int(np.argmax(corr))
        peak_lag = peak_idx - half_n
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        improved_this_round = False
        tried = set()

        if abs(peak_lag) >= 1:
            k = peak_lag
            if k > 0:
                for i in zeros:
                    j = i + k
                    if j < n and h[j] == 1:
                        tried.add((i, j))
            else:
                k2 = -k
                for i in zeros:
                    j = i - k2
                    if 0 <= j < n and h[j] == 1:
                        tried.add((i, j))

            for (i, j) in list(tried):
                h[i] = 1
                h[j] = 0
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_c5 = new_corr.max() * dx
                if new_c5 < c5_cur - 1e-15:
                    corr = new_corr
                    c5_cur = new_c5
                    improved_this_round = True
                    break
                h[i] = 0
                h[j] = 1

        if not improved_this_round:
            tried2 = set()
            for i in ones[:80]:
                for j in zeros[:80]:
                    if abs(i - j) <= 25:
                        key = (i, j)
                        if key not in tried2:
                            tried2.add(key)
                            h[i] = 0
                            h[j] = 1
                            new_c5 = compute_c5(h)
                            if new_c5 < c5_cur - 1e-15:
                                corr = np.correlate(h, 1.0 - h, mode='full')
                                c5_cur = new_c5
                                improved_this_round = True
                                break
                            h[i] = 1
                            h[j] = 0
                if improved_this_round:
                    break

        if not improved_this_round:
            for _ in range(3):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def subgradient_refine(h, n, target, time_limit, max_steps=3000):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def periodic_init(n, period_samples, phase, target):
    x = np.arange(n)
    h = np.zeros(n)
    h[(x + phase) % period_samples < period_samples // 2] = 1.0
    return project_feasible(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    n = 400
    target = n // 2

    h_init = np.array(initial_h_values, dtype=float)
    h_init = project_feasible(h_init, target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to polish the parent
    if tleft() > 150:
        h_r, sc = slsqp_round(best_h, n, target, [200, 1000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    # Phase 2: Convert to binary and do simulated annealing
    if tleft() > 400:
        h_bin = to_binary_from_continuous(best_h, n, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

        sa_time = min(300, tleft() * 0.4)
        h_sa, c5_sa = binary_sa_search(best_h, n, rng, sa_time,
                                         T_init=0.003, T_min=0.0003, cool=0.9997)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 3: Hill climbing on binary
    if tleft() > 200:
        h_bin = to_binary_from_continuous(best_h, n, target)
        hc_time = min(250, tleft() * 0.35)
        h_hc, c5_hc = binary_hill_climb(h_bin, n, rng, hc_time)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc

    # Phase 4: SLSQP on the binary result (continuous refinement)
    if tleft() > 150:
        h_r, sc = slsqp_round(best_h, n, target, [500, 2000, 10000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    # Phase 5: Another binary round after continuous refinement
    if tleft() > 200:
        h_bin = to_binary_from_continuous(best_h, n, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

        hc_time = min(150, tleft() * 0.4)
        h_hc, c5_hc = binary_hill_climb(best_h, n, rng, hc_time)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc

    # Phase 6: LP-based refinement
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.5)
        h_lp, c5_lp = lp_constraint_generation(best_h, n, target,
                                                max_iter=10000, time_limit=lp_time,
                                                damping_init=0.85, reg=1e-5, n_active=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: SLSQP high alpha
    if tleft() > 100:
        h_r, sc = slsqp_round(best_h, n, target, [2000, 10000, 50000], 1200, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    # Phase 8: Subgradient
    if tleft() > 80:
        h_sg, c5_sg = subgradient_refine(best_h, n, target, time_limit=min(50, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 9: Multi-start binary search from different initializations
    if tleft() > 300:
        n_starts = 4
        start_time_each = min(50, tleft() / (2 * n_starts))
        for si in range(n_starts):
            if tleft() < 60:
                break
            if si == 0:
                h_pert = best_h + rng.randn(n) * 0.05
            elif si == 1:
                h_pert = periodic_init(n, 16, rng.randint(16), target)
            elif si == 2:
                h_pert = to_binary_from_continuous(1.0 - best_h, n, target)
            else:
                h_pert = to_binary_from_continuous(best_h, n, target)
                h_pert = h_pert + rng.randn(n) * 0.1
                h_pert = project_feasible(h_pert, target)
                h_pert = (h_pert > 0.5).astype(float)
                h_pert = project_feasible(h_pert, target)

            h_bin_try = to_binary_from_continuous(h_pert, n, target)
            h_hc_try, c5_try = binary_hill_climb(h_bin_try, n, rng, start_time_each)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_hc_try

    # Phase 10: Final binary hill climb with more time
    if tleft() > 150:
        h_bin = to_binary_from_continuous(best_h, n, target)
        hc_time = min(120, tleft() * 0.6)
        h_hc, c5_hc = binary_hill_climb(h_bin, n, rng, hc_time)
        if c5_hc < best_score:
            best_score = c5_hc
            best_h = h_hc

    # Phase 11: Final SLSQP polish
    if tleft() > 60:
        h_r, sc = slsqp_round(best_h, n, target, [5000, 20000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    # Phase 12: Final subgradient
    if tleft() > 30:
        h_sg, c5_sg = subgradient_refine(best_h, n, target, time_limit=min(20, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 13: Try higher resolution
    if tleft() > 200:
        n2 = 600
        target2 = n2 // 2
        h_interp = interpolate_to(best_h, n2, target2)
        h_bin2 = to_binary_from_continuous(h_interp, n2, target2)
        hc_time = min(100, tleft() * 0.5)
        h_hc2, c5_hc2 = binary_hill_climb(h_bin2, n2, rng, hc_time)
        if c5_hc2 < best_score:
            best_score = c5_hc2
            best_h = h_hc2
            if tleft() > 60:
                h_r2, sc2 = slsqp_round(best_h, n2, target2, [1000, 5000], 1000, 1e-15)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_r2

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
