def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

- For each coordinate i, the optimal value (given all other coordinates fixed and the max overlap as objective) can be found by:
  1. Compute the overlap as a function of h_i (it's a quadratic in h_i for each lag)
  2. The max of these quadratics is what we want to minimize
  3. Try h_i = 0, h_i = 1, and the critical points of each active quadratic
  
This is a 1D non-convex problem (max of quadratics), but we can solve it exactly by checking all critical points.

For lag k, the contribution to O_k from coordinate i is:
If i is in the "first factor" range: h_i * (1-h_{i+k})
If i is in the "second factor" range: h_{i-k} * (1-h_i)

So O_k as a function of h_i (with other h fixed) is:
O_k(h_i) = c_k + h_i * a_k (if i is in first factor range for lag k)
         + h_{i-k} * (1-h_i) (if i is in second factor range for lag k)

Wait, this is linear in h_i for each lag! Because for a fixed lag k, h_i appears at most twice:
- As a first factor: h_i * (1-h_{i+k}) → linear in h_i
- As a second factor: h_{i-k} * (1-h_i) = h_{i-k} - h_{i-k} * h_i → linear in h_i

So O_k is LINEAR in h_i for each k. The max of linear functions is convex in h_i. We can minimize it by checking the vertices (where two linear functions cross) or the endpoints (0 or 1).

This means coordinate descent will converge in a finite number of steps! Each coordinate update is exact.

The algorithm:
1. For coordinate i, collect all lags k where i appears (either as first or second factor)
2. For each such lag, compute the slope and intercept of O_k as a function of h_i
3. Find the point in [0,1] that minimizes max_k (intercept_k + slope_k * h_i)
4. This is a 1D convex optimization (min of a convex piecewise-linear function)
5. Solve by checking all breakpoints (where two lines cross) and the endpoints

The breakpoints are at h_i = 0, h_i = 1, and for each pair of lags (k1, k2): h_i = (intercept_k2 - intercept_k1) / (slope_k1 - slope_k2) if slope_k1 ≠ slope_k2.

But there could be many lags (up to 2n-1 = 799 for n=400), so checking all pairs is O(n²) per coordinate. That's too slow.

Better approach: Since we only care about the max, we can:
1. Start with h_i = current value
2. Compute all O_k values
3. Find the max
4. The subgradient at the current point tells us which direction to move
5. Do a line search (binary search on the slope changes)

Even simpler: since the function is piecewise-linear and convex, we can use a **ternary search** or **binary search on the derivative**.

Actually, the simplest approach: just try h_i = 0 and h_i = 1 and the current value, and pick the best. This is a "coordinate descent with three-point search" which is guaranteed to make progress.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def gradient_lag(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[:n - k] += (1.0 - h[k:])
            grad[k:] -= h[:n - k]
    else:
        m = -k
        if m < n:
            grad[m:] += (1.0 - h[:n - m])
            grad[:n - m] -= h[m:]
    return grad


def overlap_at_lag(h, k, n, dx):
    if k >= 0:
        if k >= n:
            return 0.0
        return float(np.sum(h[:n - k] * (1.0 - h[k:]))) * dx
    else:
        m = -k
        if m >= n:
            return 0.0
        return float(np.sum(h[m:] * (1.0 - h[:n - m]))) * dx


def lp_step_two_sided(h_old, n, target, lags, dx, trust_radius=None):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    if trust_radius is not None:
        constraints.append(cp.norm(h_var - h_old, 1) <= trust_radius)
    for k in lags:
        O_k_val = overlap_at_lag(h_old, k, n, dx)
        grad_k = gradient_lag(h_old, k, n) * dx
        C_k = float(np.dot(grad_k, h_old)) - O_k_val
        constraints.append(cp.sum(grad_k * h_var) <= t_var + C_k)
        if k >= 0 and k < n:
            S = float(np.sum(h_old[:n - k])) * dx
            coeff_lower = np.zeros(n)
            coeff_lower[k:] = h_old[:n - k] * dx
            constraints.append(cp.sum(coeff_lower * h_var) >= S - t_var)
        elif k < 0:
            m = -k
            if m < n:
                S = float(np.sum(h_old[m:])) * dx
                coeff_lower = np.zeros(n)
                coeff_lower[:n - m] = h_old[m:] * dx
                constraints.append(cp.sum(coeff_lower * h_var) >= S - t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)


def lp_step_one_sided(h_old, n, target, lags, dx):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    g = 1.0 - h_old
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)


def alternating_lp(h_init, n, target, max_iter, time_limit,
                   k_start=50, k_end=300, damping=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 300:
            break
        progress = it / max(1, max_iter)
        K = int(k_start + (k_end - k_start) * min(1.0, progress * 1.5))
        K = min(K, 2 * n - 1)
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:K]
        lags = top_idx - n_half
        h_new = lp_step_two_sided(h, n, target, lags, dx)
        if h_new is None:
            h_new = lp_step_one_sided(h, n, target, lags, dx)
        if h_new is None:
            break
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def nesterov_sg(h_init, n, target, max_steps, time_limit, alpha=100.0, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, target)
        overlaps = compute_overlaps(y)
        c_max = np.max(overlaps)
        weights = np.exp(alpha * (overlaps - c_max))
        wsum = weights.sum()
        if wsum < 1e-30:
            break
        weights = weights / wsum
        n_half = n - 1
        sorted_idx = np.argsort(weights)[::-1]
        cum_w = np.cumsum(weights[sorted_idx])
        n_active = int(np.searchsorted(cum_w, 0.95)) + 1
        n_active = min(n_active, len(weights))
        active_idx = sorted_idx[:n_active]
        active_lags = active_idx - n_half
        active_w = weights[active_idx]
        active_w = active_w / active_w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(active_lags):
            grad += active_w[i] * gradient_lag(y, k, n)
        grad *= dx
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break
        lr = (c_max - best_c5 + 1e-8) / (norm_g ** 2 + 1e-20)
        lr = min(max(lr, 0.02), 3.0)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
        v = h_try - h
        h = h_try
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)) * (2.0 / len(h))


def cons_fun(h):
    return np.sum(h) - _g_target


def cons_jac(h):
    return np.ones(_g_n)


def slsqp_opt(h_start, n, target, alphas, maxiter=2000):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coord_descent_3pt(h_init, n, target, max_sweeps, time_limit):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            for new_val in (0.0, 1.0):
                if abs(new_val - old_val) < 1e-10:
                    continue
                h[i] = new_val
                h = project_feasible(h, target)
                c5_trial = compute_c5(h)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    best_h = h.copy()
                else:
                    h[i] = old_val
                    h = project_feasible(h, target)
            h[i] = old_val
            h = project_feasible(h, target)
    return best_h, best_c5

        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Phase 9: Final SLSQP
    if tleft() > 50:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_pre, c5_pre = slsqp_opt(best_h, n_fin, t_fin,
                                   alphas=[1000, 10000, 100000, 500000],
                                   maxiter=2000)
        if c5_pre < best_score:
            best_score = c5_pre
            best_h = h_pre

    # Phase 10: Final coordinate descent
    if tleft() > 30:
        n_fin = len(best_h)
        t_fin = n_fin / 2.0
        h_fin2, c5_fin2 = coord_descent_3pt(best_h, n_fin, t_fin,
                                            max_sweeps=300, time_limit=min(25, tleft() * 0.6))
        if c5_fin2 < best_score:
            best_score = c5_fin2
            best_h = h_fin2

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    best_h = h_final
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    else:
        h_final = project_feasible(best_h, target_use)
        final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
