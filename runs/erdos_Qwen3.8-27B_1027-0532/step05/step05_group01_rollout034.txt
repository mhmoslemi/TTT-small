from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_logsumexp(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lbfgs_smooth(h0, n, target, alphas, maxiter, penalty_coeff):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_logsumexp(corr, alphas[-1]) * dx + penalty_coeff * (np.sum(h) - target)**2
    
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alphas[-1] * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        # Gradient of logsumexp wrt corr
        # d/dh_i of corr[k] for shift k:
        # corr[k] = sum_i h[i] * (1-h[i+k]) (over valid i)
        # d corr[k] / d h[i] = (1-h[i+k]) + ... (from the other term where i appears as i+k)
        # Actually: corr[k] = sum_j h[j] * (1-h[j+k])
        # d corr[k] / d h[i] = (1-h[i+k]) [from j=i term] + ... 
        # Also h[i] appears in (1-h[j+k]) when j+k = i, i.e., j = i-k
        # So d corr[k] / d h[i] = (1-h[i+k]) - h[i-k] (with boundary handling)
        
        g = np.zeros(n)
        mid = n - 1
        idxs = np.arange(n)
        for w_idx, (wi, k_idx) in enumerate(zip(weights, range(len(corr)))):
            k = k_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v1] += wi * (1.0 - h[idxs[v1] - k])
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g[v2] -= wi * h[idxs[v2] + k]
        
        g *= dx
        # Add penalty gradient
        g += 2 * penalty_coeff * (np.sum(h) - target)
        return g
    
    h_cur = h0.copy()
    for alpha in alphas:
        alphas = [alpha]  # update for inner function
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except:
            pass
    return h_cur

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad(h, alpha, n_top=None):
    """Compute gradient of smoothed max correlation. Returns (smoothed_value, gradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if n_top is not None and n_top < len(corr):
        # Only top n_top shifts
        top_idx = np.argsort(corr)[-n_top:]
        c_sub = corr[top_idx]
        c_max = np.max(c_sub)
        shifted = np.clip(alpha * (c_sub - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w /= np.sum(w)
        smooth_val = (c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w /= np.sum(w)
        top_idx = np.arange(len(corr))
        smooth_val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    grad = np.zeros(n)
    mid = n - 1
    idxs = np.arange(n)
    for wi, k_idx in zip(w, top_idx):
        if wi < 1e-10:
            continue
        k = int(k_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    
    grad *= dx
    return smooth_val, grad

def lbfgs_optimize(h0, n, target, alphas, maxiter_per_alpha, n_top=None, penalty=0.0):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    best_c5 = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj(h):
            val, _ = lse_grad(h, alpha, n_top)
            pen = penalty * (np.sum(h) - target)**2 if penalty > 0 else 0
            return val + pen
        
        def grad(h):
            _, g = lse_grad(h, alpha, n_top)
            if penalty > 0:
                g = g + 2 * penalty * (np.sum(h) - target)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter_per_alpha, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except:
            pass
    return h_cur, best_c5

def greedy_cd(h, n, target, time_limit, max_steps=10000):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1
    n_corr = 2 * n - 1
    
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break
        
        # Find active shift
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid
        
        # Compute gradient at active shift
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        # Greedy: decrease at max grad, increase at min grad
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if i == j:
            continue
        
        # Line search for optimal delta
        # h[i] - delta, h[j] + delta
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        
        # Bisection for optimal delta
        a, b = delta_lo, delta_hi
        for _ in range(30):
            mid_d = 0.5 * (a + b)
            h_try = h.copy()
            h_try[i] -= mid_d
            h_try[j] += mid_d
            corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
            if np.max(corr_try) < np.max(corr):
                a = mid_d  # can go higher
            else:
                b = mid_d
        delta_star = 0.5 * (a + b)
        if abs(delta_star) < 1e-12:
            continue
        
        h[i] -= delta_star
        h[j] += delta_star
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
    
    return best_h, best_c5
