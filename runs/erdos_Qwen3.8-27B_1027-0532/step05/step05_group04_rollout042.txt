Let me also think about the discrete coordinate descent more carefully. The key insight is that for this problem, the optimal solution is likely "nearly binary" (values close to 0 or 1). So snapping values to 0 or 1 (when they're already close) might help.

Here's my refined plan for the discrete CD:
1. For each j, if h[j] is close to 0 or 1 (say within 0.01), try snapping it to 0 or 1
2. If the score improves, accept
3. Also try random "flips" (0→1 or 1→0) with a small probability (SA-like)

This is cheap and might help with the final precision.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr[n-1:])) * dx

    n = len(h)
    return compute_corr(h)[n-1:]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.99999999
        else:
            mask = h > 1e-8
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    return grad * dx

def nesterov_adapt(h, n, target, n_steps, beta_init, beta_growth, lr_init, deadline):
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    beta = beta_init
    lr = lr_init
    moment = 0.9
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = get_over(h_cur)
        max_ov = float(np.max(over))
        
        # Adaptive top-K
        K = max(5, n // 20)
        top_idx = np.argsort(over)[::-1][:K]
        top_vals = over[top_idx]
        
        weights = np.zeros(n)
        sh = beta * (top_vals - max_ov)
        sh = np.clip(sh, -500.0, 0.0)
        w = np.exp(sh)
        w /= w.sum()

            h_cur = h_new
            cur_score = sc_new

def discrete_cd(h, n, target, n_sweeps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    over = get_over(h_cur)
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old_val = h_cur[j]
            # Try 0 and 1
            for cv in [0.0, 1.0]:
                delta = cv - old_val
                if abs(delta) < 1e-12:
                    continue

                for k in range(n):

                    else:

                        if j >= k:
