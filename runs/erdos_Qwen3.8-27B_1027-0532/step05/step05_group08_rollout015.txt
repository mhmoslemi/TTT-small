            h_new, sc_new = active_set_lbfgs(best_h, n_cur, target_cur, na, 5000, 1500, deadline - 15)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Subgradient
    if time.time() < deadline - 20:
        for temp in [0.003, 0.001]:
            if time.time() > deadline - 15:
                break
            h_new, sc_new = subgradient_momentum(best_h, n_cur, target_cur, 1500, temp, 0.005, deadline - 12)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Surgery
    h_new, sc_new = worst_lag_surgery(best_h, n_cur, target_cur, 400, 0.004, deadline - 5)
    if sc_new < best_score:
        best_score = sc_new
        best_h = h_new

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 8.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Optimize at original resolution
    t1_deadline = start_time + min(350, budget_s * 0.25)
    best_h, best_score = multi_resolve_optimize(best_h, t1_deadline, rng)

    # Phase 2: Try n=600
    n2 = 600
    target2 = n2 / 2.0
    t2_deadline = start_time + min(600, budget_s * 0.45)
    if time.time() < t2_deadline - 60:
        h2 = interpolate_to(best_h, n2, target2)
        best_h2, best_score2 = multi_resolve_optimize(h2, t2_deadline, rng)
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 3: Try n=800
    n3 = 800
    target3 = n3 / 2.0
    t3_deadline = deadline - 60
    if time.time() < t3_deadline - 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        best_h3, best_score3 = multi_resolve_optimize(h3, t3_deadline, rng)
        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 4: Random restarts at best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    t4_deadline = deadline - 40
    for attempt in range(6):
        if time.time() > t4_deadline:
            break
        scale = 0.05 + 0.03 * attempt
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, target_cur)
        h_ref = lbfgsb_round(h_pert, n_cur, target_cur, 500, 50.0, 200)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            h_ref2 = slsqp_round(best_h, n_cur, target_cur, 5000, 2000, 1e-15)
            sc2 = compute_c5(h_ref2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 200, 0.003, t4_deadline - 5)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 5: Final polish
    if time.time() < deadline - 10:
        h_new = slsqp_round(best_h, n_cur, target_cur, 50000, 3000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if time.time() < deadline - 5:
        h_new, sc_new = worst_lag_surgery(best_h, n_cur, target_cur, 200, 0.002, deadline - 3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
