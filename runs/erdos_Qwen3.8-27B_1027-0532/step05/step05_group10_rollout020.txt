import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def make_binary(h):
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones == target:
        return h_bin
    rng = np.random.RandomState(7)
    if n_ones > target:
        ones = np.where(h_bin == 1)[0]
        to_flip = n_ones - target
        idx = rng.choice(len(ones), to_flip, replace=False)
        h_bin[ones[idx]] = 0
    else:
        zeros = np.where(h_bin == 0)[0]
        to_flip = target - n_ones
        idx = rng.choice(len(zeros), to_flip, replace=False)
        h_bin[zeros[idx]] = 1
    return h_bin


def binary_local_search(h, time_limit=100.0, n_cand=25):
    n = len(h)
    half_n = n - 1
    dx = 2.0 / n
    h_cur = make_binary(h)
    C = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    best_h = h_cur.copy()
    best_c5 = float(np.max(C)) * dx
    start = time.time()
    no_improve = 0
    max_no_improve = 3000

    while time.time() - start < time_limit:
        if no_improve > max_no_improve:
            break

        k_max = int(np.argmax(C))
        k_shift = k_max - half_n
        current_max = float(C[k_max])

        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        a_ks = ones + k_shift
        a_km = ones - k_shift
        loss = np.zeros(len(ones))
        m1 = (a_ks >= 0) & (a_ks < n)
        loss[m1] -= (1.0 - h_cur[a_ks[m1]])
        m2 = (a_km >= 0) & (a_km < n)
        loss[m2] += h_cur[a_km[m2]]

        b_ks = zeros + k_shift
        b_km = zeros - k_shift
        gain = np.zeros(len(zeros))
        m3 = (b_ks >= 0) & (b_ks < n)
        gain[m3] += (1.0 - h_cur[b_ks[m3]])
        m4 = (b_km >= 0) & (b_km < n)
        gain[m4] -= h_cur[b_km[m4]]

        top_a = np.argsort(loss)[:n_cand]
        top_b = np.argsort(gain)[:n_cand]

        best_new = best_c5
        best_a = -1
        best_b = -1

        ks_arr = np.arange(-(n - 1), n)

        for ia in top_a:
            a = ones[ia]
            ra = np.zeros(2 * n - 1)
            idx1 = a + ks_arr
            m = (idx1 >= 0) & (idx1 < n)
            ra[m] -= (1.0 - h_cur[idx1[m]])
            idx2 = a - ks_arr
            m = (idx2 >= 0) & (idx2 < n)
            ra[m] += h_cur[idx2[m]]

            for ib in top_b:
                b = zeros[ib]
                ab = (b - a)
                ba = (a - b)
                add_k = b + k_shift
                sub_k = b - k_shift
                gc = 0.0
                if 0 <= add_k < n:
                    gc += (1.0 - h_cur[add_k])
                if 0 <= sub_k < n:
                    gc -= h_cur[sub_k]

                if loss[ia] + gc >= 0:
                    continue

                ab_delta = -ra + np.zeros(2 * n - 1)
                ib_vec = np.zeros(2 * n - 1)
                idx3 = b + ks_arr
                m = (idx3 >= 0) & (idx3 < n)
                ib_vec[m] += (1.0 - h_cur[idx3[m]])
                idx4 = b - ks_arr
                m = (idx4 >= 0) & (idx4 < n)

                dO = -ra + ib_vec
                new_max = float(np.max(C + dO)) * dx
                if new_max < best_new - 1e-14:
                    best_new = new_max
                    best_a = a
                    best_b = b

        if best_a >= 0:

            h_cur[best_b] = 1.0
            ks_arr = np.arange(-(n - 1), n)
            dO = np.zeros(2 * n - 1)
            idx1 = best_a + ks_arr
            m = (idx1 >= 0) & (idx1 < n)
            dO[m] -= (1.0 - h_cur[np.where(idx1[m] == best_b, best_b, idx1[m])])
            # Recompute properly using old h (before swap)
            # Actually let's just recompute C from scratch for safety
            C = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            best_c5 = best_new
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                rng2 = np.random.RandomState(no_improve)
                found = False
                for _ in range(30):

                        best_h = h_try.copy()
                        no_improve = 0
                        found = True
                        break
                if not found and no_improve > max_no_improve:
                    break

    return best_h, best_c5


def alternating_lp_sparse(h_init, n, target, max_iter, time_limit,
                          damping_init=0.8, reg=1e-4, n_active=40):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-5:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.95, damping + 0.03)

        if it % 100 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.75)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
