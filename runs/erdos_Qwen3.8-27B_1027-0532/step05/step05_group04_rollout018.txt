            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.03, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def fourier_perturb(h, n, target, rng, strength):
    h_f = np.fft.rfft(h - 0.5)
    n_harmonics = max(n // 4, 4)
    for k in range(1, n_harmonics):
        h_f[k] += strength * (rng.randn() - 0.5) * (1.0 / (1.0 + 0.1 * k))
    h_new = np.fft.irfft(h_f * 2 * n, n=n) - 0.5 + 0.5
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, target)


def physical_perturb(h, n, target, rng, strength):
    noise = rng.randn(n) * strength
    h_new = h + noise
    return project_feasible(h_new, target)


def ilsb_round(h, n, target, n_iters, rng, alpha, maxiter, deadline):
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    best_score = compute_c5(best_h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if it % 3 == 0:
            h_pert = fourier_perturb(best_h, n, target, rng, 0.05 + 0.1 * it / max(n_iters, 1))
        else:
            h_pert = physical_perturb(best_h, n, target, rng, 0.03 + 0.05 * it / max(n_iters, 1))
        h_opt, sc_opt = lbfgsb_round(h_pert, n, target, [alpha, alpha * 3], maxiter,
                                      min(time.time() + 60, deadline))
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt
    return best_h, best_score


def progressive_refine(h, n, target, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    alpha_schedule = [50, 150, 500, 1500, 5000, 15000, 50000, 150000]
    for i, a in enumerate(alpha_schedule):
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = max(100, int(remaining / max(len(alpha_schedule) - i, 1) / 2))
        maxit = min(maxit, 2000)
        h_cur, cur_score = lbfgsb_round(h_cur, n, target, [a], maxit,
                                         time.time() + remaining * 0.9)
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Diverse initialization via DE on coarse grids
    deadline_p1 = start_time + 70
    candidates = [h_init]
    for nc, pop, gens in [(20, 40, 150), (50, 30, 120), (100, 25, 80)]:
        remaining = deadline_p1 - time.time()
        if remaining < 10:
            break
        h_de, _ = de_search(nc, nc / 2.0, pop, gens, rng,
                            deadline=start_time + 70 - remaining, F=0.75, CR=0.85)
        h_fine = interpolate_to(h_de, n, target)
        candidates.append(h_fine)

    # Fourier-based structured candidates
    for freq in [3, 5, 7, 11]:
        x = np.linspace(0, 2, n)
        h_struct = 0.5 + 0.3 * np.sin(np.pi * freq * x) + 0.1 * np.sin(np.pi * 2 * freq * x)
        h_struct = project_feasible(h_struct, target)
        candidates.append(h_struct)

    # Pick best candidate from L-BFGS-B quick pass
    deadline_p2 = start_time + 200
    for cand in candidates:
        remaining = deadline_p2 - time.time()
        if remaining < 20:
            break
        sc = compute_c5(cand)
        if sc < best_score:
            best_h = cand.copy()
            best_score = sc
        h_opt, sc_opt = lbfgsb_round(cand, n, target, [100, 400], 300,
                                      min(time.time() + remaining * 0.4, deadline_p2))
        if sc_opt < best_score:
            best_score = sc_opt
            best_h = h_opt

    # Phase 2: Main progressive L-BFGS-B refinement
    deadline_p3 = start_time + 550
    best_h, best_score = progressive_refine(best_h, n, target, deadline_p3)

    # Phase 3: Iterated local search
    deadline_p4 = start_time + 850
    remaining = deadline_p4 - time.time()
    if remaining > 50:
        n_iters = max(5, int(remaining / 30))
        best_h, best_score = ilsb_round(best_h, n, target, n_iters, rng,
                                         5000, 400, deadline_p4)

    # Phase 4: Subgradient + SLSQP polish
    deadline_p5 = start_time + 1050
    remaining = deadline_p5 - time.time()
    if remaining > 40:
        sg_steps = max(100, int(remaining / 5))
        best_h, best_score = subgradient_round(best_h, n, target, sg_steps,
                                                n_active=4, lr_init=0.004,
                                                lr_decay=0.996, temp=0.02,
                                                deadline=start_time + 1000)
        remaining2 = start_time + 1100 - time.time()
        if remaining2 > 30:
            best_h, best_score = slsqp_round(best_h, n, target,
                                              [5000, 20000, 80000], 2000, 1e-17,
                                              deadline=start_time + 1100)

    # Phase 5: Final micro-polish
    deadline_final = start_time + budget_s - 20
    remaining = deadline_final - time.time()
    if remaining > 30:
        # L-BFGS-B with highest alpha
        best_h, best_score = lbfgsb_round(best_h, n, target,
                                           [30000, 100000, 300000], 1000,
                                           deadline_final - 10)
    remaining = deadline_final - time.time()
    if remaining > 20:
        best_h, best_score = subgradient_round(best_h, n, target,
                                                max(50, int(remaining - 5)),
                                                n_active=6, lr_init=0.002,
                                                lr_decay=0.998, temp=0.05,
                                                deadline=deadline_final)

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    if final_score > best_score:
        best_h = project_feasible(h_init, target)
        final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
