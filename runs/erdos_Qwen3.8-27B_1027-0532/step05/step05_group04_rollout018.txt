def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

def compute_grads(h, n):
    """Compute gradients of all f_k with respect to h. Returns array of shape (n, n)."""
    dx = 2.0 / n
    grads = np.zeros((n, n))
    for k in range(n):
        g = np.zeros(n)
        j_pos = np.arange(0, n - k)
        if len(j_pos) > 0:
            g[j_pos] += (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        if len(j_neg) > 0:
            g[j_neg] -= h[j_neg - k]
        grads[k] = dx * g
    return grads

    """Take one LP step to reduce the max overlap."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    
    # Select lags to include
    if all_lags:
        indices = np.arange(n)
    else:
        top_k = 100
        indices = np.argsort(over)[::-1][:top_k]
    
    # Build gradient matrix for selected lags
    G = np.zeros((len(indices), n))
    for idx, k in enumerate(indices):
        k = int(k)
        g = np.zeros(n)
        j_pos = np.arange(0, n - k)
        if len(j_pos) > 0:
            g[j_pos] += (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        if len(j_neg) > 0:
            g[j_neg] -= h[j_neg - k]
        G[idx] = dx * g
    
    # LP: min t
    # s.t. G @ d - t <= -over[indices] * dx  (linearized constraints)
    #      sum(d) = 0  (preserve sum)
    #      -h_i <= d_i <= 1-h_i  (feasibility)
    #      |d_i| <= eps  (step size)
    #      t >= 0
    
    m = len(indices)
    # A_ub: m rows for lag constraints
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = G
    A_ub[:, -1] = -1.0
    b_ub = -over[indices] * dx
    
    # A_eq: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        lo = max(-h[i], -eps)
        hi = min(1.0 - h[i], eps)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((0.0, None))
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n], res.x[-1]
    return None, None

                best_h = h_half.copy()
                h_cur = h_cur + 0.5 * d
            else:
                h_cur = best_h.copy()  # revert
    return best_h, best_score

    final_score = compute_c5(h_out)
