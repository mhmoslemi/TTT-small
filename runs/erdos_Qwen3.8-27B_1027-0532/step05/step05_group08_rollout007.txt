        h_eq, sc_eq = equalize_lags(best_h, n_cur, target_cur, top5, 0.005, 2000, deadline - 120)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        # Re-do high-alpha after equalization
        if time.time() < deadline - 100:
            corr = compute_corr(best_h)
            top3 = np.argsort(corr)[::-1][:3]
            h_new = slsqp_opt(best_h, n_cur, target_cur, 100000, top3, 2000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Alternating pair focus
    if time.time() < deadline - 120:
        for outer in range(15):
            if time.time() > deadline - 100:
                break
            corr = compute_corr(best_h)
            top6 = np.argsort(corr)[::-1][:6]
            # Focus on pair (0,1)
            h_new = slsqp_opt(best_h, n_cur, target_cur, 50000, [int(top6[0]), int(top6[1])], 1000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            # Focus on pair (2,3)
            if time.time() < deadline - 105:
                h_new = slsqp_opt(best_h, n_cur, target_cur, 50000, [int(top6[2]), int(top6[3])], 1000, 1e-16)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
            # Focus on all 4
            if time.time() < deadline - 108:
                top4 = np.argsort(compute_corr(best_h))[::-1][:4]
                h_new = slsqp_opt(best_h, n_cur, target_cur, 30000, top4, 800, 1e-16)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

    # Phase 8: Basin hopping
    if time.time() < deadline - 80:
        h_bh, sc_bh = basin_hopping_local(best_h, n_cur, target_cur, 6, rng, deadline - 50)
        if sc_bh < best_score:
            best_score = sc_bh
            best_h = h_bh

    # Phase 9: Final high-alpha polish
    if time.time() < deadline - 30:
        corr = compute_corr(best_h)
        top2 = np.argsort(corr)[::-1][:2]
        for alpha in [200000, 500000]:
            if time.time() > deadline - 20:
                break
            h_new = slsqp_opt(best_h, n_cur, target_cur, alpha, top2, 2000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 10: Final equalization + polish
    if time.time() < deadline - 15:
        corr = compute_corr(best_h)
        top4 = np.argsort(corr)[::-1][:4]
        h_eq, sc_eq = equalize_lags(best_h, n_cur, target_cur, top4, 0.003, 500, deadline - 10)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
