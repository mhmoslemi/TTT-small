def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):

    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)

    tgt_cur = n_cur / 2.0
    if tleft() > 200:
        h_tl, sc_tl = top_lag_gradient(best_h, tgt_cur, n_top=10, step=0.004,
                                        max_iter=500, deadline=dl())
        if sc_tl < best_score:
            best_score = sc_tl
            best_h = h_tl

    if tleft() > 150:
        h_sg, sc_sg = subgradient_optimize(best_h, tgt_cur, max_iter=1500,
                                            deadline=dl(), initial_step=0.004)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 200:
        for a in [100000, 500000, 1000000]:
            if tleft() < 80:
                break
            h_new = lbfgs_optimize(best_h, tgt_cur, a, mu, 400, dl())
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new

    # Phase 8: Try upscaling to n=1200 if time allows
    if tleft() > 400 and n_cur == n2:
        n3 = 3 * n
        target3 = n3 / 2.0
        h_up3 = upsample(best_h, n_cur, n3)
        h_up3 = project_feasible(h_up3, target3)
        for a in [50, 500, 5000, 20000]:
            if tleft() < 100:
                break
            h_opt = lbfgs_optimize(h_up3, target3, a, mu, 200, dl())
            sc = compute_c5(h_opt)
            if sc < best_score:
                best_score = sc
                best_h = h_opt
                n_cur = n3
                tgt_cur = target3
            else:
                h_up3 = h_opt

    # Phase 9: SLSQP polish
    if tleft() > 60:
        h_slq = slsqp_polish(best_h, tgt_cur, 20000, 500)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 10: Final coordinate descent
    if tleft() > 80:
        h_cd, sc_cd = coordinate_descent(best_h, tgt_cur, n_sweeps=2, deadline=dl())
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 11: Final subgradient sweep
    if tleft() > 50:
        h_sg, sc_sg = subgradient_optimize(best_h, tgt_cur, max_iter=800,
                                            deadline=dl(), initial_step=0.003)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 12: Final L-BFGS-B
    if tleft() > 40:
        h_new = lbfgs_optimize(best_h, tgt_cur, 500000, mu, 300, dl())
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, tgt_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
