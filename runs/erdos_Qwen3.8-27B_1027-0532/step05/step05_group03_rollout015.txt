def slsqp_refine(h_init, alphas, maxiter=2000, ftol=1e-15, deadline=None):
    """SLSQP with log-sum-exp smoothing."""
    global _g_n, _g_target, _g_alpha
    n = len(h_init)
    target = n / 2.0
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def _obj_soft_max(x):

    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha
