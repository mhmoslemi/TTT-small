                   damping=0.7, reg=5e-5):

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n, nonneg=True, up_bound=1.0)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS, max_iters=100000)
                except Exception:
                    break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:

        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_grad_step(h, alpha, dx, top_m=None, max_iters=500, start_time=None, time_limit=None):
    n = len(h)
    h = h.copy()

    for step in range(max_iters):
        if start_time is not None and time.time() - start_time > time_limit - 10:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)

        if top_m is not None and top_m < 2 * n - 1:
            top_indices = np.argsort(corr)[-top_m:]
            sub_corr = corr[top_indices]
        else:
            top_indices = np.arange(2 * n - 1)
            sub_corr = corr

        shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
        exp_shifted = np.exp(shifted)
        wsum = np.sum(exp_shifted)
        if wsum < 1e-300:
            break
        weights = exp_shifted / wsum

        grad = np.zeros(n)
        n_half = n - 1

        for idx_k, w in zip(top_indices, weights):
            if w < 1e-14:
                continue
            k = idx_k - n_half
            if k == 0:
                grad += w * (1.0 - 2.0 * h) * dx
            elif k > 0:
                if n - k > 0:
                    grad[k:] += w * (1.0 - h[:n - k]) * dx
                    grad[:n - k] -= w * h[k:] * dx
            else:
                m = -k
                if n - m > 0:
                    grad[:n - m] += w * (1.0 - h[m:]) * dx
                    grad[m:] -= w * h[:n - m] * dx

        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-12:
            break

        lr = 1.0 / (gnorm + 1e-10)
        lr = min(lr, 2.0 / alpha)

        improved = False
        for _ in range(30):
            h_try = h - lr * grad
            h_try = project_feasible(h_try, n / 2.0)
            score = compute_c5(h_try)
            if score < c_max - 1e-14:
                h = h_try
                improved = True
                break
            lr *= 0.5

        if not improved:
            break

    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Jacobi coordinate descent - fast structural optimization
    if tleft() > 300:
        h_cd, c5_cd = jacobi_coord_descent(best_h, target,
                                           max_iters=200,
                                           start_time=start_time,
                                           time_limit=budget_s,
                                           damping=1.0)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Alternating LP for structural changes
    if tleft() > 250:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=500,
                                     time_limit=budget_s,
                                     start_time=start_time,
                                     deadline_offset=150,
                                     damping=0.7, reg=3e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: More Jacobi CD after LP
    if tleft() > 200:
        h_cd, c5_cd = jacobi_coord_descent(best_h, target,
                                           max_iters=300,
                                           start_time=start_time,
                                           time_limit=budget_s,
                                           damping=0.8)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Smooth gradient descent
    if tleft() > 150:
        for alpha in [50, 200, 1000, 5000]:
            if tleft() < 60:
                break
            h_smooth = smooth_grad_step(best_h, alpha, 2.0 / n,
                                        top_m=50, max_iters=200,
                                        start_time=start_time,
                                        time_limit=budget_s)
            h_smooth = project_feasible(h_smooth, target)
            sc = compute_c5(h_smooth)
            if sc < best_score:
                best_score = sc
                best_h = h_smooth

    # Phase 5: SLSQP polish
    if tleft() > 100:
        h_slq = slsqp_optimize(best_h, n, target,
                                [100, 500, 2000, 10000],
                                maxiter=1500, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=50)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 6: Another round of Jacobi CD
    if tleft() > 120:
        h_cd, c5_cd = jacobi_coord_descent(best_h, target,
                                           max_iters=300,
                                           start_time=start_time,
                                           time_limit=budget_s,
                                           damping=0.9)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Try different resolutions
    if tleft() > 200:
        for n_try in [500, 600, 320]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)

            h_cd, c5_cd = jacobi_coord_descent(h_interp, target_try,
                                               max_iters=100,
                                               start_time=start_time,
                                               time_limit=budget_s,
                                               damping=0.9)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n = n_try
                target = target_try

            if tleft() < 80:
                continue

            h_lp2, c5_lp2 = alternating_lp(best_h, n, target,
                                           max_iter=200,
                                           time_limit=budget_s,
                                           start_time=start_time,
                                           deadline_offset=80,
                                           damping=0.7, reg=3e-5)
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2

            if tleft() < 60:
                continue

            h_cd2, c5_cd2 = jacobi_coord_descent(best_h, target,
                                                 max_iters=200,
                                                 start_time=start_time,
                                                 time_limit=budget_s,
                                                 damping=0.85)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 8: Random restarts with local optimization
    if tleft() > 80:
        for noise in [0.005, 0.01, 0.02]:
            if tleft() < 40:
                break
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_pert = best_h + rng.randn(n_use) * noise
            h_pert = project_feasible(h_pert, target_use)
            h_cd, c5_cd = jacobi_coord_descent(h_pert, target_use,
                                               max_iters=100,
                                               start_time=start_time,
                                               time_limit=budget_s,
                                               damping=0.9)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 9: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                                [2000, 10000, 50000],
                                maxiter=800, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
