    return compute_grad_weighted(h, weights) / alpha

import numpy as np

def compute_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:n - 1 + n]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlap(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[:n - k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n - k]
    return grad

    corr = compute_overlap(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)

                    best_score = sc0
                    best_h = h_cur.copy()
                    any_change = True
                elif sc1 < best_score:
                    h_cur = h_t1
                    best_score = sc1
                    best_h = h_cur.copy()
                    any_change = True
                else:
                    h_cur[i] = old_val
        if not any_change:
            break
    return best_h, best_score


def slsqp_polish(h, n, target, alpha_list, maxiter, ftol, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

    for a in alpha_list:
        if deadline is not None and time.time() > deadline:
            break

        def obj(x):
            corr = compute_overlap(x)
            cm = float(np.max(corr))
            sh = np.clip(a * (corr - cm), -700.0, 0.0)
            return (cm + float(np.log(np.sum(np.exp(sh)))) / a) * dx

        def gfun(x):
            corr = compute_overlap(x)
            cm = float(np.max(corr))
            sh = np.clip(a * (corr - cm), -700.0, 0.0)
            p = np.exp(sh) / np.sum(np.exp(sh))
            return compute_grad_weighted(x, p) * dx

        try:
            res = minimize(obj, h_cur, jac=gfun, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def perturbation_search(h, n, target, n_trials, rng, deadline=None):
    best_h = h.copy()
    best_score = compute_c5(h)
    for trial in range(n_trials):
        if deadline and time.time() > deadline:
            break
        h_pert = h + rng.randn(n) * 0.05
        h_pert = project_feasible(h_pert, target)
        sc = compute_c5(h_pert)
        if sc < best_score:
            best_score = sc
            best_h = h_pert.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA with increasing alpha (fast convergence)
    for alpha in [100, 300, 800]:
        if tleft() < 150:
            break
        dl = start_time + 120
        h_new, sc = fista_lse(best_h, n, target, alpha, 800, 0.5, dl)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Adam reweighting (adaptive per-coordinate)
    if tleft() > 400:
        dl = start_time + 350
        h_new, sc = adam_reweight(best_h, n, target, n_outer=20, beta_init=200,
                                  beta_growth=1.5, inner_iters=60, lr_init=0.01,
                                  deadline=dl)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: SLSQP with analytical gradient
    if tleft() > 250:
        h_new, sc = slsqp_polish(best_h, n, target, [500, 2000, 5000], 3000, 1e-15,
                                 deadline=start_time + budget_s * 0.45)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Rounding refinement (new technique)
    if tleft() > 250:
        h_new, sc = rounding_refine(best_h, n, target, threshold=0.03, n_rounds=5,
                                    deadline=start_time + budget_s * 0.55)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if tleft() > 150:
            h_new, sc = slsqp_polish(best_h, n, target, [2000, 8000], 2500, 1e-15,
                                     deadline=start_time + budget_s * 0.65)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 5: FISTA with very high alpha
    if tleft() > 200:
        h_new, sc = fista_lse(best_h, n, target, 2000, 1000, 0.1,
                              deadline=start_time + budget_s * 0.75)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Perturbation search
    if tleft() > 150:
        h_new, sc = perturbation_search(best_h, n, target, 20, rng,
                                        deadline=start_time + budget_s * 0.8)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if tleft() > 100:
            h_new, sc = slsqp_polish(best_h, n, target, [5000, 15000], 2500, 1e-16,
                                     deadline=start_time + budget_s * 0.88)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Final FISTA + SLSQP
    if tleft() > 80:
        h_new, sc = fista_lse(best_h, n, target, 5000, 500, 0.05,
                              deadline=start_time + budget_s - 60)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if tleft() > 30:
            h_new, sc = slsqp_polish(best_h, n, target, [20000, 80000], 3000, 1e-17,
                                     deadline=start_time + budget_s - 20)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Final rounding micro-pass
    if tleft() > 20:
        h_new, sc = rounding_refine(best_h, n, target, threshold=0.01, n_rounds=2,
                                    deadline=start_time + budget_s - 10)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
