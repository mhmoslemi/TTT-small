def project_box_affine(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

    return np.correlate(h, 1.0 - h, mode='full')

    return float(np.max(compute_corr(h))) * (2.0 / len(h))

def lse_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # Get top-k lags by value
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[idx]
    c_max = c_top.max()
    shifted = np.clip(alpha * (c_top - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time


def project_box_affine(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    return float(np.max(compute_corr(h))) * (2.0 / len(h))


def grad_single_lag(h, m):
    n = len(h)
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g


def lse_grad(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    c_top = corr[idx]
    c_max = c_top.max()
    shifted = np.clip(alpha * (c_top - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return np.zeros(n)
    w /= w_sum
    g = np.zeros(n)
    for j in range(top_k):
        wk = w[j]
        if wk < 1e-12:
            continue
        m = int(idx[j])
        g += wk * grad_single_lag(h, m)
    return g * dx


def sum_sq_grad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for j in range(top_k):
        m = int(idx[j])
        cm = corr[m]
        g += cm * grad_single_lag(h, m)
    return 2.0 * dx * dx * g


def sum_sq_value(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    return (dx * dx) * np.sum(corr[idx] ** 2)


def fista_smooth(h_init, target, grad_fn, max_iter, deadline, L_init, top_k=30):
    n = len(h_init)
    h = h_init.copy()
    y = h_init.copy()
    t = 1.0
    L = L_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stall_count = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = grad_fn(y)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        L = max(L * 1.05, 2.0 / (n * 0.01))
        h_new = project_box_affine(y - g / L, target)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = project_box_affine(y, target)
        h = h_new
        t = t_new
        if it % 30 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                stall_count = 0
            else:
                stall_count += 1
                if stall_count > 20:
                    L *= 1.5
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def fista_lse(h_init, target, alpha, max_iter, deadline, top_k=30):
    n = len(h_init)
    L_init = 2.0 * alpha + 1.0
    return fista_smooth(h_init, target, lambda x: lse_grad(x, alpha, top_k),
                        max_iter, deadline, L_init, top_k)


def fista_sum_sq(h_init, target, top_k, max_iter, deadline):
    n = len(h_init)
    L_init = 4.0 * top_k * (2.0 / n) ** 2 * n
    L_init = max(L_init, 1.0)
    return fista_smooth(h_init, target, lambda x: sum_sq_grad(x, top_k),
                        max_iter, deadline, L_init, top_k)


def direct_max_reduction(h_init, target, max_steps, deadline):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02
    for it in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        g = grad_single_lag(h, m_star) * dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = project_box_affine(h - step * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.005, 0.08)
        else:
            step *= 0.92
            if step < 1e-7:
                break
    return best_h, best_c5


def multi_lag_step(h_init, target, top_k, max_steps, deadline):
    n = len(h_init)

    h = h_init.copy()
    best_h = h.copy()

    step = 0.005
    for it in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        g = np.zeros(n)
        for m in idx:
            g += grad_single_lag(h, int(m))
        g *= dx / top_k
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = project_box_affine(h - step * g, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new

            h = h_new
            step = min(step * 1.008, 0.03)

            if step < 1e-8:
                break
    return best_h, best_c5


def simulated_annealing(h_init, target, max_iter, t_start, t_end, deadline, rng):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(1, max_iter))
        scale = 0.03 + 0.04 * (t / t_start)
        h_pert = h + rng.randn(n) * scale
        h_pert = project_box_affine(h_pert, target)
        c5_pert = compute_c5(h_pert)
        c5_cur = compute_c5(h)
        delta = c5_pert - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            h = h_pert
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h.copy()
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, len(h_src))
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_affine(h_out, target_dst)


def slsqp_refine(h_init, target, alpha, maxiter, deadline):
    n = len(h_init)
    top_k = 40

    def obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c_active = corr[idx]
        am = c_active[0]
        shifted = np.clip(alpha * (c_active - am), -500.0, 0.0)
        return am + np.log(np.sum(np.exp(shifted))) / alpha

    def objg(x):
        return lse_grad(x, alpha, top_k)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h_init, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_out = project_box_affine(res.x, target)
        return h_out, compute_c5(h_out)
    except Exception:
        return h_init, compute_c5(h_init)
