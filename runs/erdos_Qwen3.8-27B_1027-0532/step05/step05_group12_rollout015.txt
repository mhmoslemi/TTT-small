def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:

    If active_k is None, use all k with g_k close to max (within tolerance).

    h = project_feasible(h, target)

    best_score = compute_c5(h)

    t = 1.0

    y = h.copy()
    
    # Step size schedule: gamma / sqrt(k)
    gamma = 0.5  # tune this
    stagnation_count = 0

            break
        
        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max_y = np.max(corr_y)

            elif k > 0:

                grad[k:] -= y[:n-k] * dx
            else:
                m = -k
                grad[m:] += (1.0 - y[:n-m]) * dx
                grad[:n-m] -= y[m:] * dx
        
        grad /= n_active
        
        # Step size
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        step_size = gamma / (np.sqrt(step + 1) * gnorm)
        
        # Take step
        h_prev = h.copy()
        h = y - step_size * grad
        h = project_feasible(h, target)
        
        # Evaluate
        if step % 10 == 0:
            score = compute_c5(h)
            if score < best_score:

                best_h = h.copy()
                stagnation_count = 0
            else:

def compute_exact_subgrad(h, dx, n, tol_frac=0.05):
    """Compute subgradient of max_k g_k(h) where g_k = sum h[i]*(1-h[i+k])*dx.
    Uses the identity g_k = S - A_k where A_k is autocorrelation.
    """
    n_half = n - 1
    # Compute autocorrelation
    # A_k = sum_i h[i]*h[i+k] for k >= 0
    # This is the autocorrelation
    autocorr = np.correlate(h, h, mode='full')  # length 2n-1
    # autocorr[n_half + k] = A_k for k >= 0
    # autocorr[n_half - k] = A_k for k >= 0 (by symmetry)

    S = np.sum(h)
    g_values = S - autocorr  # this gives (S - A_k) for all k (positive and negative)
    
    c_max = np.max(g_values)
    tol = tol_frac * (c_max + 1e-10)

    n_active = np.sum(active)
    
    if n_active == 0:
        active[np.argmax(g_values)] = True
        n_active = 1

    # Gradient of g_k = -gradient of A_k (since S is constant)

    for idx in np.where(active)[0]:

            grad += 2.0 * h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def exact_subgrad(h, dx, top_k=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    n_half = n - 1

        top_indices = np.argsort(corr)[-top_k:]
    else:
        tol = 0.02 * (c_max + 1e-10)

        top_indices = np.array([np.argmax(corr)])
    weights = np.ones(len(top_indices)) / len(top_indices)

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30,
                   damping=0.7, reg=1e-4):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_cons = 2 * n - 1
    coeff_matrix = np.zeros((n_cons, n))
    n_half = n - 1
    for j in range(n_cons):
        k = j - n_half
        if k == 0:
            coeff_matrix[j, :] = 1.0
        elif k > 0:
            coeff_matrix[j, k:] = 1.0
            coeff_matrix[j, :n - k] = -1.0
        else:
            m = -k
            coeff_matrix[j, :n - m] = 1.0
            coeff_matrix[j, m:] = -1.0
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        g = 1.0 - h
        rhs = np.zeros(n_cons)
        for j in range(n_cons):
            k = j - n_half
            if k == 0:
                rhs[j] = np.sum(g) * dx
            elif k > 0:
                rhs[j] = np.sum(g[k:] * h[:n - k]) * dx
            else:
                m = -k
                rhs[j] = np.sum(g[:n - m] * h[m:]) * dx
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0,
            coeff_matrix @ h_var <= t_var + rhs
        ]
        reg_w = reg / (1.0 + it * 0.1)
        objective = t_var + reg_w * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(soft_obj, h_cur, args=(alpha, 10),
                           jac=lambda x: soft_obj_grad(x, alpha, 10),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP (30% budget)
    lp_budget = budget_s * 0.30
    if tleft() > lp_budget * 0.5:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000,
                                     time_limit=start_time + lp_budget,
                                     start_time=start_time,
                                     deadline_offset=budget_s - lp_budget,
                                     damping=0.7, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Momentum soft-max gradient (25% budget)
    sm_budget = budget_s * 0.25
    if tleft() > 60:
        h_sm, c5_sm = momentum_softmax(best_h, n, target,
                                       time_limit=start_time + sm_budget,
                                       start_time=start_time,
                                       deadline_offset=budget_s - sm_budget)
        if c5_sm < best_score:
            best_score = c5_sm
            best_h = h_sm

    # Phase 3: Nesterov exact subgradient (15% budget)
    ns_budget = budget_s * 0.15
    if tleft() > 60:
        h_ns, c5_ns = nesterov_subgrad(best_h, n, target,
                                       time_limit=start_time + ns_budget,
                                       start_time=start_time,
                                       deadline_offset=budget_s - ns_budget)
        if c5_ns < best_score:
            best_score = c5_ns
            best_h = h_ns

    # Phase 4: Binary local search (10% budget)
    bls_budget = budget_s * 0.10
    if tleft() > 30:
        h_bl, c5_bl = binary_local_search(best_h, n, target,
                                          time_limit=start_time + bls_budget,
                                          start_time=start_time,
                                          deadline_offset=budget_s - bls_budget,
                                          n_sweeps=2000)
        if c5_bl < best_score:
            best_score = c5_bl
            best_h = h_bl
        # Also try from a pure binary threshold
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        h_bl2, c5_bl2 = binary_local_search(h_bin, n, target,
                                            time_limit=start_time + bls_budget,
                                            start_time=start_time,
                                            deadline_offset=budget_s - bls_budget,
                                            n_sweeps=2000)
        if c5_bl2 < best_score:
            best_score = c5_bl2
            best_h = h_bl2

    # Phase 5: Local perturbation restarts (10% budget)
    pert_budget = budget_s * 0.10
    pert_scales = [0.005, 0.01, 0.02, 0.03, 0.05]
    for noise in pert_scales:
        if tleft() < 40:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_sm2, c5_sm2 = momentum_softmax(h_pert, n, target,
                                         time_limit=start_time + pert_budget,
                                         start_time=start_time,
                                         deadline_offset=budget_s - pert_budget)
        if c5_sm2 < best_score:
            best_score = c5_sm2
            best_h = h_sm2

    # Phase 6: Multi-resolution (15% budget)
    mr_budget = budget_s * 0.15
    for n_try in [600, 800]:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        time_sub = min(60, tleft() * 0.4)
        h_lp2, c5_lp2 = alternating_lp(h_interp, n_try, target_try,
                                       max_iter=20000,
                                       time_limit=start_time + time_sub,
                                       start_time=start_time,
                                       deadline_offset=budget_s - time_sub,
                                       damping=0.7, reg=1e-5)
        if tleft() > 60:
            time_sub2 = min(60, tleft() * 0.3)
            h_sm3, c5_sm3 = momentum_softmax(h_lp2, n_try, target_try,
                                             time_limit=start_time + time_sub2,
                                             start_time=start_time,
                                             deadline_offset=budget_s - time_sub2)
            if c5_sm3 < c5_lp2:
                h_lp2, c5_lp2 = h_sm3, c5_sm3
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2
            n = n_try
            target = target_try

    # Phase 7: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                               [100, 500, 2000, 10000],
                               maxiter=1000, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=15)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Phase 8: Final Nesterov
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ns2, c5_ns2 = nesterov_subgrad(best_h, n_use, target_use,
                                         time_limit=budget_s,
                                         start_time=start_time,
                                         deadline_offset=5)
        if c5_ns2 < best_score:
            best_score = c5_ns2
            best_h = h_ns2

    # Final projection
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
