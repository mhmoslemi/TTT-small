def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def get_top_shifts(h, m):
    """Get the indices of the m largest correlation values"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    top_idx = np.argsort(corr)[-m:]
    return top_idx, corr[top_idx]

def compute_grad_for_shifts(h, shifts, dx):
    """Compute gradient contributions for specific shifts"""
    n = len(h)
    grad = np.zeros(n)
    n_half = n - 1
    for idx in shifts:
        k = idx - n_half
        if k == 0:
            grad += (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += (1.0 - h[:n-k]) * dx
                grad[:n-k] -= h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += (1.0 - h[m:]) * dx
                grad[m:] -= h[:n-m] * dx
    return grad

def smooth_grad(h, alpha, dx, top_m=None):
    """Smooth max gradient using log-sum-exp"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
    else:
        top_indices = np.arange(2 * n - 1)
    
    sub_corr = corr[top_indices]
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    
    if wsum < 1e-300:
        return np.zeros(n), c_max
    
    weights = exp_shifted / wsum
    obj = float(c_max + np.log(wsum) / alpha)
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k]) * dx
                grad[:n-k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n-m] * dx
    
    return grad, obj

def adam_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30):
    """Adam-based projected gradient with smooth max"""
    dx = 2.0 / n
    h_cur = project_feasible(h_init.copy(), target)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    beta1 = 0.75
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.5
    t = 0
    
    # Phases with increasing alpha
    phases = [
        (20, 50, 3000),
        (30, 100, 3000),
        (40, 300, 3000),
        (50, 1000, 5000),
        (60, 3000, 5000),
        (80, 10000, 8000),
        (100, 30000, 10000),
        (200, 100000, 15000),
    ]
    
    total_budget = time_limit - deadline_offset
    total_steps = sum(p[2] for p in phases)
    time_per_step = total_budget / total_steps
    
    for top_m, alpha, max_steps in phases:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        # Reset momentum for new phase
        m_adam = np.zeros(n)
        v_adam = np.zeros(n)
        t = 0
        lr = base_lr
        
        stall = 0
        stall_patience = max(200, max_steps // 10)
        
        for step in range(max_steps):
            elapsed = time.time() - start_time
            if elapsed > time_limit - deadline_offset:
                break
            if elapsed > time_limit - 10:
                break
            
            grad, _ = smooth_grad(h_cur, alpha, dx, top_m=top_m)
            t += 1
            
            m_adam = beta1 * m_adam + (1 - beta1) * grad
            v_adam = beta2 * v_adam + (1 - beta2) * (grad * grad)
            m_hat = m_adam / (1 - beta1 ** t)
            v_hat = v_adam / (1 - beta2 ** t)
            
            step_size = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_try = project_feasible(h_cur - step_size, target)
            score = compute_c5(h_try)
            
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_try.copy()
                h_cur = h_try
                stall = 0
            else:
                h_cur = h_try
                stall += 1
            
            if stall > stall_patience:
                lr *= 0.7
                stall = 0
                if lr < 1e-5:
                    break
    
    return best_h, best_score

        # Line search

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    # Phase 1: Adam at original resolution
    if tleft() > 250:
        time_lim = min(300, tleft() * 0.35)
        h_new, sc = adam_optimize(best_h, n, target, time_lim, start_time,
                                  deadline_offset=max(60, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Targeted shift descent
    if tleft() > 200:
        time_lim = min(180, tleft() * 0.3)
        h_new, sc = targeted_shift_descent(best_h, n, target, time_lim, start_time,
                                           deadline_offset=max(60, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Momentum descent with high alpha
    if tleft() > 200:
        time_lim = min(150, tleft() * 0.3)
        h_new, sc = momentum_descent(best_h, n, target, time_lim, start_time,
                                     deadline_offset=max(60, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: SLSQP polish
    if tleft() > 120:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        time_lim = min(100, tleft() * 0.3)
        h_new, sc = slsqp_polish(best_h, n_cur, tgt_cur, time_lim, start_time,
                                 deadline_offset=max(40, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Second Adam round at original resolution
    if tleft() > 200:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        time_lim = min(200, tleft() * 0.35)
        h_new, sc = adam_optimize(best_h, n_cur, tgt_cur, time_lim, start_time,
                                  deadline_offset=max(50, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Higher resolution 600
    if tleft() > 250:
        n_high = 600
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        time_lim = min(200, tleft() * 0.35)
        h_new, sc = adam_optimize(h_interp, n_high, target_high, time_lim, start_time,
                                  deadline_offset=max(60, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new
            # Further polish at n=600
            if tleft() > 100:
                time_lim2 = min(80, tleft() * 0.3)
                h_new2, sc2 = targeted_shift_descent(best_h, n_high, target_high,
                                                     time_lim2, start_time,
                                                     deadline_offset=max(40, budget_s - time_lim2))
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new2

    # Phase 7: Random restarts
    for i in range(4):
        if tleft() < 80:
            break
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        noise = rng.randn(n_cur) * 0.015
        h_pert = project_feasible(best_h + noise, tgt_cur)
        time_lim = min(70, tleft() * 0.3)
        h_new, sc = adam_optimize(h_pert, n_cur, tgt_cur, time_lim, start_time,
                                  deadline_offset=max(30, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Resolution 800
    if tleft() > 200:
        n_high = 800
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        time_lim = min(150, tleft() * 0.4)
        h_new, sc = adam_optimize(h_interp, n_high, target_high, time_lim, start_time,
                                  deadline_offset=max(50, budget_s - time_lim))
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Final targeted descent
    if tleft() > 50:
        n_cur = len(best_h)
        tgt_cur = n_cur / 2.0
        time_lim = min(50, tleft() * 0.7)
        h_new, sc = targeted_shift_descent(best_h, n_cur, tgt_cur, time_lim, start_time,
                                           deadline_offset=15)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final feasibility
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
