Let me think about parallel tempering more carefully:
- Maintain K replicas at temperatures T_1 > T_2 > ... > T_K
- At each step, pick a random replica and do a swap
- Periodically try to swap configurations between adjacent replicas
- The swap is accepted with probability min(1, exp(-(E_high - E_low)(1/T_low - 1/T_high)))

This is well-known to help escape local minima.

For the block move:
- Pick a block of L consecutive 1s
- Pick a location to move it to (where all L positions are 0)
- Compute the delta in correlation
- Accept/reject based on SA criterion

The delta for a block move can be computed as the sum of deltas for individual position changes, but we need to be careful about interactions.

Actually, for a block move (moving L ones from positions {a, a+1, ..., a+L-1} to {b, b+1, ..., b+L-1}), the delta in correlation is:
delta = sum_{i in source} delta_i - sum_{j in target} delta_j

where delta_i is the "contribution" of position i to the correlation. But this isn't quite right because the deltas interact.

A simpler approach: compute the full correlation before and after the move, and take the difference. This is O(n log n) with FFT, which is acceptable for block moves done less frequently.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    
    return best_h, best_c5

def parallel_tempering(h_init, n_replicas, temps, n_iters, rng, time_limit, start_time):
    n = len(h_init)
    dx = 2.0 / n
    positions = np.arange(n)
    
    # Initialize replicas
    replicas = []
    for r in range(n_replicas):
        h_r = h_init.copy()
        if r > 0:
            # Perturb higher-temperature replicas
            n_pert = rng.randint(2, 5 + r * 2)
            for _ in range(n_pert):
                ones = np.where(h_r == 1.0)[0]
                zeros = np.where(h_r == 0.0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h_r[i] = 0.0
                    h_r[j] = 1.0
        full_r = full_corr_fast(h_r)
        c5_r = float(full_r.max()) * dx
        replicas.append({
            'h': h_r,
            'full': full_r,
            'c5': c5_r,
            'best_c5': c5_r,
            'best_h': h_r.copy()
        })
    
    best_overall_c5 = min(rep['best_c5'] for rep in replicas)
    best_overall_h = replicas[np.argmin([rep['best_c5'] for rep in replicas])]['best_h'].copy()
    
    swap_interval = max(100, n_iters // 50)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Pick a random replica to update
        r = rng.randint(n_replicas)
        rep = replicas[r]
        h = rep['h']
        full = rep['full']
        t = temps[r]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - rep['c5']
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            rep['c5'] = c5_new
            if c5_new < rep['best_c5']:
                rep['best_c5'] = c5_new
                rep['best_h'] = h.copy()
                if c5_new < best_overall_c5:
                    best_overall_c5 = c5_new
                    best_overall_h = h.copy()
        
        # Replica exchange
        if (it + 1) % swap_interval == 0:
            for r2 in range(n_replicas - 1):
                # Try to swap replicas r2 and r2+1
                c5_a = replicas[r2]['c5']
                c5_b = replicas[r2+1]['c5']
                t_a = temps[r2]
                t_b = temps[r2+1]
                # beta = 1/t
                delta_log = (1.0/t_b - 1.0/t_a) * (c5_a - c5_b)
                if delta_log > 0 or rng.random() < np.exp(delta_log):
                    # Swap
                    replicas[r2]['h'], replicas[r2+1]['h'] = replicas[r2+1]['h'], replicas[r2]['h']
                    replicas[r2]['full'], replicas[r2+1]['full'] = replicas[r2+1]['full'], replicas[r2]['full']
                    replicas[r2]['c5'], replicas[r2+1]['c5'] = replicas[r2+1]['c5'], replicas[r2]['c5']
    
    return best_overall_h, best_overall_c5
