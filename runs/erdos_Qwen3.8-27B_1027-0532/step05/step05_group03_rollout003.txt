def get_overlaps_fast(h):

    # Autocorrelation via FFT
    N = 2 * n  # zero-pad to avoid circular convolution
    H = np.fft.rfft(h, N)
    ac = np.fft.irfft(H * np.conj(H), N)[:n]
    # Prefix sum of h
    prefix = np.cumsum(h)
    # overlap[k] = sum_{i=0}^{n-1-k} h[i] - ac[k] = prefix[n-1-k] - ac[k]
    # prefix[n-1-k] is the sum of h[0:n-k]

    # sum_{i=0}^{n-1-k} h[i] = prefix[n-1-k]
    # So overlap[k] = prefix[n-1-k] - ac[k]
    # For k=0: overlap[0] = prefix[n-1] - ac[0] = sum(h) - sum(h^2) = n/2 - n/2 = 0 (for bang-bang)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def get_overlaps(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_bang_bang(h, target):
    n = len(h)
    sorted_idx = np.argsort(-h)
    h_b = np.zeros(n)
    h_b[sorted_idx[:int(target)]] = 1.0

def delta_swap(h, p, q):
    n = len(h)
    delta = np.zeros(n)
    if p < n:
        k1 = np.arange(0, n - p)
        delta[k1] -= (1.0 - h[p + k1])
    if p >= 0:
        k2 = np.arange(0, p + 1)
        delta[k2] += h[p - k2]
    if q < n:
        k3 = np.arange(0, n - q)
        delta[k3] += (1.0 - h[q + k3])
    if q >= 0:
        k4 = np.arange(0, q + 1)
        delta[k4] -= h[q - k4]
    return delta

def _obj_slsqp(x):
    h_p = project_feasible(x, _g_target)
    ov = get_overlaps(h_p)
    c_max = float(np.max(ov))
    shifted = _g_alpha * (ov - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h, alphas, maxiter, ftol, time_limit):
    global _g_n, _g_target, _g_alpha
    n = len(h)
    _g_n = n
    _g_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > time_limit:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj_slsqp, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        best_h = h_bb2

    # Phase 4: Two-swap search
    t4 = min(deadline - (budget_s - 15) * 0.15, start_time + 900)
    h_2s, sc_2s = two_swap_search(best_h, t4, rng, max_iters=50000, top_k=3)
    if sc_2s < best_score:
        best_score = sc_2s
        best_h = h_2s

    # Phase 5: Block swaps
    t5 = min(deadline - (budget_s - 15) * 0.10, start_time + 1050)
    h_bs, sc_bs = block_swap_search(best_h, t5, rng, block_size=3, max_iters=30000)
    if sc_bs < best_score:
        best_score = sc_bs
        best_h = h_bs
    h_bs2, sc_bs2 = block_swap_search(best_h, t5, rng, block_size=5, max_iters=20000)
    if sc_bs2 < best_score:
        best_score = sc_bs2
        best_h = h_bs2

    # Phase 6: Multi-resolution
    current_n = len(best_h)
    for n_try in [current_n + 2, current_n + 4, current_n + 6]:
        if time.time() > deadline - 120:
            break
        target_try = n_try / 2.0
        h_fine = interpolate_h(best_h, n_try, target_try)
        h_fine_bb = round_bang_bang(h_fine, target_try)
        t_fine = min(deadline - 60, start_time + (budget_s - 15) * 0.85)
        h_fine_bb, sc_fine = targeted_binary_search(h_fine_bb, t_fine, rng, max_iters=50000, top_k=5)
        if time.time() < deadline - 60 and sc_fine < best_score:
            h_fine_cont, sc_fine_c = slsqp_refine(h_fine_bb, [200, 1000], 1500, 1e-14, deadline - 40)
            if sc_fine_c < sc_fine:
                sc_fine = sc_fine_c
                h_fine_bb = h_fine_cont
            else:
                h_fine_bb = round_bang_bang(h_fine_cont, target_try)
                h_fine_bb, sc_fine2 = targeted_binary_search(h_fine_bb, deadline - 30, rng, max_iters=30000, top_k=5)
                if sc_fine2 < sc_fine:
                    sc_fine = sc_fine2
                    h_fine_bb = h_fine_bb
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine_bb.copy()
            best_n = n_try

    # Phase 7: Final targeted search from best
    t7 = min(deadline - 30, start_time + (budget_s - 15) * 0.92)
    h_final, sc_final = nearest_neighbor_swap(best_h, t7, rng, max_iters=200000, search_radius=8)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h_final

    # Phase 8: Final SA polish
    t8 = min(deadline - 15, start_time + (budget_s - 15) * 0.96)
    h_polish, sc_polish = sa_binary(best_h, t8, rng, T_init=0.002, T_min=0.00005, cooldown=0.9995)
    if sc_polish < best_score:
        best_score = sc_polish
        best_h = h_polish

    # Final: SLSQP on continuous if time remains
    if time.time() < deadline - 10:
        h_cont_final, sc_cf = slsqp_refine(best_h, [500, 2000], 500, 1e-15, deadline)
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cont_final.copy()
        else:
            # Try rounding the continuous result and one more binary pass
            h_r = round_bang_bang(sc_cf if isinstance(sc_cf, np.ndarray) else best_h, len(best_h) / 2.0)

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
