```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)
```

Wait, corr_k = sum_i h[i] * (1-h[i+k]) is NOT linear in h (it's bilinear). So it's not an LP.

For all k: #{i: h[i]=1, h[i+k]=0} <= floor(T * n / 2)

This is still not linear because of the product h[i] * (1-h[i+k]).

```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1

    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def fast_smooth_grad(h, alpha):
    """Compute gradient of the smooth max objective."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = corr - corr.max()
    exp_corr = np.exp(alpha * corr_shifted)
    w = exp_corr / exp_corr.sum()
    
    # term1[i] = sum_m w[m] * (1-h)[i+m-(n-1)] for valid indices
    # term2[i] = sum_m w[m] * h[i-m+(n-1)] for valid indices
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]
    
    j1 = i_idx + m_idx - (n-1)
    valid1 = (j1 >= 0) & (j1 < n)
    j2 = i_idx - m_idx + (n-1)
    valid2 = (j2 >= 0) & (j2 < n)
    
    # Use clipping to avoid out-of-bounds
    j1_clipped = np.clip(j1, 0, n-1)
    j2_clipped = np.clip(j2, 0, n-1)
    
    M1 = np.where(valid1, 1.0 - h[j1_clipped], 0.0)
    M2 = np.where(valid2, h[j2_clipped], 0.0)
    
    grad = (2.0 / n) * (M1 @ w + M2 @ w)
    return grad

def pgd_optimize(h_start, target_sum, alpha_schedule, lr_schedule, max_iters, time_limit, start_time):
    """Projected gradient descent with adaptive step size."""
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        alpha = min(alpha_schedule, alpha_schedule * (it + 1) / 10.0)  # ramp up
        grad = fast_smooth_grad(h, alpha)
        
        # Backtracking line search
        lr = lr_schedule
        for _ in range(20):
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target_sum)
            score = compute_c5(h_new)
            curr_score = compute_c5(h)
            if score < curr_score:
                break
            lr *= 0.5
        else:
            break
        
        h = h_new
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score
```

```python
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = corr - corr.max()
    exp_corr = np.exp(alpha * corr_shifted)
    w = exp_corr / exp_corr.sum()
    
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2*n-1)[None, :]

def greedy_multi_shift(h, max_iters, time_limit, start_time, top_k=5, scan_w=25):
    """Greedy that targets multiple worst shifts simultaneously."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 2000:
            break
        
        # Find top-k worst shifts
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = top_idx - (n - 1)
        top_vals = corr[top_idx].astype(float)
        max_val = float(corr.max())
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find candidates: 1s that overlap with 0s at to
        # Instead of a single worst shift, consider all top-k shifts
        i_cand = set()
        j_cand = set()
        for k in top_shifts:
            for i in ones:
                if 0 <= i + k < n and h[i + k] == 0:
                    i_cand.add(i)
            for j in zeros:
                if 0 <= j + k < n and h[j + k] == 1:
                    j_cand.add(j)
        
        i_cand = list(i_cand)[:scan_w]
        j_cand = list(j_cand)[:scan_w]
        
        if not i_cand or not j_cand:
            stall += 1
            continue
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for i in i_cand:
            for j in j_cand:
                # Compute new max considering all top shifts
                new_max = max_val
                improved = True
                for k_idx in top_idx:
                    k = k_idx - (n - 1)
                    delta = 0.0
                    if 0 <= i + k < n:
                        delta -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        delta += h[i - k]
                    if 0 <= j + k < n:
                        delta += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        delta -= h[j - k]
                    v = float(corr[k_idx]) + delta
                    if v > new_max:
                        new_max = v
                        # Early check: if this is worse than current max, skip
                        if new_max >= max_val - best_improve:
                            improved = False
                            break
                if improved:
                    imp = max_val - new_max
                    if imp > best_improve:
                        best_improve = imp
                        best_i = i
                        best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def memetic_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    """SA with periodic greedy local search."""
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    greedy_interval = 500  # Do greedy every 500 SA steps
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        h_new = h.copy()
        r = rng.random()
        
        if r < 0.4:
            # Single swap targeting worst shifts
            worst = np.argsort(corr)[-3:]
            k = worst[rng.randint(len(worst))] - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n and h[i+k] == 0]
            jp = [j for j in zeros if 0 <= j + k < n and h[j+k] == 1]
            if ip and jp:
                h_new[ip[rng.randint(ip)]] = 0
                h_new[jp[rng.randint(jp)]] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        elif r < 0.7:
            # Random single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
        elif r < 0.85:
            # Block move: flip a contiguous segment
            block_len = rng.randint(2, 6)
            start = rng.randint(0, n - block_len)
            # Find a block with roughly equal 1s and 0s to flip
            seg = h_new[start:start+block_len]
            n_ones = int(seg.sum())
            n_zeros = block_len - n_ones
            if abs(n_ones - n_zeros) <= 1:
                h_new[start:start+block_len] = 1.0 - seg
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        else:
            # Pair swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / max(t, 1e-20))):
            h = h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)


def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shifted = corr - corr.max()
    exp_corr = np.exp(alpha * corr_shifted)
    w = exp_corr / exp_corr.sum()
    i_idx = np.arange(n)[:, None]
    m_idx = np.arange(2 * n - 1)[None, :]
    j1 = i_idx + m_idx - (n - 1)
    valid1 = (j1 >= 0) & (j1 < n)
    j2 = i_idx - m_idx + (n - 1)
    valid2 = (j2 >= 0) & (j2 < n)
    j1c = np.clip(j1, 0, n - 1)
    j2c = np.clip(j2, 0, n - 1)
    M1 = np.where(valid1, 1.0 - h[j1c], 0.0)
    M2 = np.where(valid2, h[j2c], 0.0)
    return (2.0 / n) * (M1 @ w + M2 @ w)


def pgd_refine(h_start, target_sum, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    stages = [
        (5.0, 0.05, 800),
        (20.0, 0.03, 800),
        (80.0, 0.015, 800),
        (400.0, 0.008, 800),
        (2000.0, 0.004, 800),
        (10000.0, 0.002, 800),
        (40000.0, 0.001, 800),
    ]
    for alpha, lr, max_it in stages:
        if time.time() - start_time > time_limit:
            break
        for it in range(max_it):
            if time.time() - start_time > time_limit:
                break
            grad = smooth_grad(h, alpha)
            h_new = project_feasible(h - lr * grad, target_sum)
            score = compute_c5(h_new)
            if score < curr_score:
                h = h_new
                curr_score = score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            else:
                h_half = project_feasible(h - 0.4 * lr * grad, target_sum)
                score_h = compute_c5(h_half)
                if score_h < curr_score:
                    h = h_half
                    curr_score = score_h
                    if score_h < best_score:
                        best_score = score_h
                        best_h = h.copy()
    return best_h, best_score


def greedy_multi(h, max_iters, time_limit, start_time, top_k, scan_w):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 3000:
            break
        top_idx = np.argsort(corr)[-top_k:]
        max_val = float(corr.max())
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i_cand = []
        j_cand = []
        seen_i = set()
        seen_j = set()
        for ki in top_idx:
            k = int(ki - (n - 1))
            for i in ones:
                if i not in seen_i and 0 <= i + k < n and h[i + k] == 0:
                    i_cand.append(i)
                    seen_i.add(i)
            for j in zeros:
                if j not in seen_j and 0 <= j + k < n and h[j + k] == 1:
                    j_cand.append(j)
                    seen_j.add(j)
        i_cand = i_cand[:scan_w]
        j_cand = j_cand[:scan_w]
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            continue
        best_improve = 0.0
        best_i = -1
        best_j = -1
        for i in i_cand:
            for j in j_cand:
                new_max = -1.0
                for ki in top_idx:
                    k = int(ki - (n - 1))
                    delta = 0.0
                    if 0 <= i + k < n:
                        delta -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        delta += h[i - k]
                    if 0 <= j + k < n:
                        delta += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        delta -= h[j - k]
                    v = float(corr[ki]) + delta
                    if v > new_max:
                        new_max = v
                if new_max < max_val - best_improve:
                    best_improve = max_val - new_max
                    best_i = i
                    best_j = j
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c5


def memetic_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, greedy_every):
    n = len(h_start)
    h = h_start.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 15000:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        h_new = h.copy()
        r = rng.random()
        if r < 0.45:
            worst = np.argsort(corr)[-4:]
            ki = worst[rng.randint(len(worst))]
            k = int(ki - (n - 1))
            ip = [i for i in ones if 0 <= i + k < n and h[i + k] == 0]
            jp = [j for j in zeros if 0 <= j + k < n and h[j + k] == 1]
            if ip and jp:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        elif r < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
        elif r < 0.82:
            blen = rng.randint(2, 7)
            start = rng.randint(0, n - blen)
            seg = h_new[start:start + blen].copy()
            n_ones_seg = int(seg.sum())
            n_zeros_seg = blen - n_ones_seg
            if abs(n_ones_seg - n_zeros_seg) <= 1:
                h_new[start:start + blen] = 1.0 - seg
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_new[i] = 0
                h_new[j] = 1
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / max(t, 1e-20))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if greedy_every > 0 and (it + 1) % greedy_every == 0:
            elapsed = time.time() - start_time
            remaining = time_limit - elapsed
            if remaining > 15:
                g_lim = min(8.0, remaining * 0.3)
                h_g, c5_g = greedy_multi(h, max_iters=1000, time_limit=g_lim,
                                         start_time=time.time(), top_k=3, scan_w=12)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g.copy()
                    h = h_g.copy()
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    current_c5 = compute_c5(h)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2)

    elapsed = time.time() - start_time
    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from initial solution
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 30:
        alloc = min(80.0, remaining * 0.12)
        h_g, c5_g = greedy_multi(best_h, max_iters=50000, time_limit=alloc,
                                 start_time=start_time, top_k=5, scan_w=30)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Memetic SA (long run)
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 100:
        alloc = min(400.0, remaining * 0.38)
        max_it = int(min(600000, alloc * 2500))
        h_sa, c5_sa = memetic_sa(best_h, target, max_it, 0.015, 0.00001,
                                 rng, alloc, start_time, greedy_every=800)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: PGD continuous refinement
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 60:
        alloc = min(180.0, remaining * 0.22)
        h_pgd, c5_pgd = pgd_refine(best_h, float(target), alloc, start_time)
        if c5_pgd < best_score:
            best_score = c5_pgd
            best_h = h_pgd.copy()

    # Phase 4: Binary SA from PGD projection
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 60:
        h_bin = project_binary(best_h, target)
        alloc = min(200.0, remaining * 0.25)
        max_it = int(min(500000, alloc * 2500))
        h_sa2, c5_sa2 = memetic_sa(h_bin, target, max_it, 0.008, 0.000005,
                                   rng, alloc, start_time, greedy_every=600)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 5: Second PGD
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 40:
        alloc = min(120.0, remaining * 0.2)
        h_pgd2, c5_pgd2 = pgd_refine(best_h, float(target), alloc, start_time)
        if c5_pgd2 < best_score:
            best_score = c5_pgd2
            best_h = h_pgd2.copy()

    # Phase 6: Final focused SA
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 30:
        h_bin2 = project_binary(best_h, target)
        alloc = min(100.0, remaining * 0.3)
        max_it = int(min(300000, alloc * 3000))
        h_sa3, c5_sa3 = memetic_sa(h_bin2, target, max_it, 0.004, 0.000002,
                                   rng, alloc, start_time, greedy_every=500)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 7: Final PGD polish
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 15:
        alloc = max(5.0, min(60.0, remaining * 0.4))
        h_pgd3, c5_pgd3 = pgd_refine(best_h, float(target), alloc, start_time)
        if c5_pgd3 < best_score:
            best_score = c5_pgd3
            best_h = h_pgd3.copy()

    # Phase 8: One more short SA if time allows
    remaining = budget_s - 30 - (time.time() - start_time)
    if remaining > 20:
        h_bin3 = project_binary(best_h, target)
        alloc = max(5.0, min(40.0, remaining * 0.5))
        max_it = int(min(150000, alloc * 3000))
        h_sa4, c5_sa4 = memetic_sa(h_bin3, target, max_it, 0.003, 0.000001,
                                   rng, alloc, start_time, greedy_every=400)
        if c5_sa4 < best_score:
            best_score = c5_sa4
            best_h = h_sa4.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
```
