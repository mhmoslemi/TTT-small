```python
def project_onto_polytope(x, target_sum):
    """Project x onto {h : 0 ≤ h ≤ 1, sum(h) = target_sum}"""
    # Bisection on lambda: h_i = clip(x_i - lambda, 0, 1)
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2, 0, 1)
    # Final adjustment for numerical precision
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        # Adjust the element closest to 0.5
        mid_mask = (h > 0.01) & (h < 0.99)
        if mid_mask.any():
            h[mid_mask] += diff / mid_mask.sum()
            h = np.clip(h, 0, 1)
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_onto_polytope(x, target_sum):
    """Project x onto {h : 0 <= h <= 1, sum(h) = target_sum} via bisection."""
    if x.sum() == target_sum:
        return np.clip(x, 0.0, 1.0)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix numerical precision
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def gradient_lse(h, alpha):
    """Compute gradient of LSE(alpha, corr * dx) w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    c_max = over.max()
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(n) / n
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for k in range(n):
        wk = w[k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            # d(corr_k)/dh[j] = (1-h[j-k]) * 1_{j>=k} - h[j+k] * 1_{j<n-k}
            if k < n:
                # j in [k, n): (1-h[j-k])
                grad[k:] += wk * (1.0 - h[:n-k])
                # j in [0, n-k): -h[j+k]
                grad[:n-k] -= wk * h[k:]
    grad *= dx
    return grad, c_max * dx

def pgd_lse(h_init, alpha, n_iters, deadline, step_decay=0.9995):
    """PGD with LSE smoothing."""
    n = len(h_init)
    target = n / 2.0
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    step = 1.0 / (dx * (2.0 + alpha * n * 0.5))
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad, obj_val = gradient_lse(h_cur, alpha)
        h_new = h_cur - step * grad
        h_new = project_onto_polytope(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
        step *= step_decay
    
    return best_h, best_score
```

```python
    lse_val = (c_max + float(np.log(ws)) / alpha) * dx
    return grad, lse_val

def pgd_lse_optimize(h_init, alpha_list, iters_per_alpha, deadline):
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    n = len(h_cur)
    target = n / 2.0
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        step = 2.0 / (dx * (2.0 + alpha * n * 0.8))
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            grad, _ = lse_grad_and_val(h_cur, alpha)
            h_new = project_onto_polytope(h_cur - step * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_score

def fista_lse_optimize(h_init, alpha_list, iters_per_alpha, deadline):
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_cur)
    n = len(h_cur)
    target = n / 2.0
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        dx = 2.0 / n
        step = 2.0 / (dx * (2.0 + alpha * n * 0.8))
        h_prev = h_cur.copy()
        t = 1.0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y = h_cur + (t - 1.0) / t_new * (h_cur - h_prev)
            y = np.clip(y, 0.0, 1.0)
            grad, _ = lse_grad_and_val(y, alpha)
            h_new = project_onto_polytope(y - step * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_prev = h_cur
            h_cur = h_new
            t = t_new
    return best_h, best_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            extra = np.zeros((pop_size - pop.shape[0], n))
            for i in range(pop_size - pop.shape[0]):
                extra[i] = project_onto_polytope(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        for i in range(pop_size):
            pop[i] = project_onto_polytope(rng.rand(n) * 0.5 + 0.25, target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_onto_polytope(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_onto_polytope(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_onto_polytope(h_out, target_dst)

def slsqp_optimize(h, n, target, alphas, maxiter, deadline):
    from scipy.optimize import minimize
    dx = 2.0 / n

    def obj_fn(x, alpha):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(lambda x: obj_fn(x, float(a)), h_cur, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_onto_polytope(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 15.0

    def tleft():
        return hard_deadline - time.time()

    n_parent = len(initial_h_values)
    target_parent = n_parent / 2.0
    h_parent = project_onto_polytope(np.array(initial_h_values, dtype=float), target_parent)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)
    best_n = n_parent

    def check_update(h_new, n_new):
        nonlocal best_h, best_score, best_n
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            best_n = n_new

    # Phase 1: DE at n=40 for diverse basins
    n1 = 40
    t1 = n1 / 2.0
    h1_init = interpolate_to(h_parent, n1, t1)
    pop1 = [h1_init.copy()]
    for _ in range(19):
        pop1.append(project_onto_polytope(rng.rand(n1) * 0.5 + 0.25, t1))
    dl1 = min(start_time + 35, hard_deadline)
    h1, sc1 = de_optimize(n1, t1, 20, 600, 0.8, 0.7, dl1, rng, init_pop=pop1)
    h1_up = interpolate_to(h1, best_n, target_parent)
    check_update(h1_up, best_n)

    # Phase 2: PGD at n=400 (parent-based)
    n2 = 400
    t2 = n2 / 2.0
    h2_init = project_onto_polytope(best_h, t2) if best_n == 400 else interpolate_to(best_h, n2, t2)
    dl2 = min(start_time + 120, hard_deadline)
    h2 = pgd_lse_optimize(h2_init, [50, 200, 800], 3000, dl2)
    check_update(h2, n2)
    if best_n != 400:
        h2 = interpolate_to(h2, best_n, best_n / 2.0)

    # Phase 3: PGD at n=400 (DE-based)
    h3_init = interpolate_to(h1, 400, 200.0)
    dl3 = min(start_time + 220, hard_deadline)
    h3 = pgd_lse_optimize(h3_init, [50, 200, 800, 3000], 2500, dl3)
    check_update(h3, 400)

    # Phase 4: FISTA at n=400 from best
    dl4 = min(start_time + 350, hard_deadline)
    h4_init = project_onto_polytope(best_h, best_n / 2.0) if best_n == 400 else interpolate_to(best_h, 400, 200.0)
    h4 = fista_lse_optimize(h4_init, [200, 1000, 5000], 3000, dl4)
    check_update(h4, 400)

    # Phase 5: SLSQP polish at n=400
    if tleft() > 120:
        dl5 = min(start_time + 500, hard_deadline)
        h5_init = project_onto_polytope(best_h, best_n / 2.0) if best_n == 400 else interpolate_to(best_h, 400, 200.0)
        h5 = slsqp_optimize(h5_init, 400, 200.0, [500, 2000, 8000], 2000, dl5)
        check_update(h5, 400)

    # Phase 6: PGD at n=800
    if tleft() > 200:
        n6 = 800
        t6 = n6 / 2.0
        h6_init = interpolate_to(best_h, n6, t6)
        dl6 = min(start_time + 700, hard_deadline)
        h6 = pgd_lse_optimize(h6_init, [100, 500, 2000, 8000], 4000, dl6)
        check_update(h6, n6)

    # Phase 7: FISTA at n=800
    if tleft() > 150 and best_n >= 800:
        n7 = 800
        t7 = n7 / 2.0
        h7_init = project_onto_polytope(best_h, t7) if best_n == 800 else interpolate_to(best_h, n7, t7)
        dl7 = min(start_time + 850, hard_deadline)
        h7 = fista_lse_optimize(h7_init, [500, 2000, 10000], 4000, dl7)
        check_update(h7, n7)

    # Phase 8: PGD at n=1600
    if tleft() > 250:
        n8 = 1600
        t8 = n8 / 2.0
        h8_init = interpolate_to(best_h, n8, t8)
        dl8 = min(start_time + 1050, hard_deadline)
        h8 = pgd_lse_optimize(h8_init, [200, 1000, 5000], 3000, dl8)
        check_update(h8, n8)

    # Phase 9: Final high-alpha PGD polish at best_n
    if tleft() > 60:
        n9 = best_n
        t9 = n9 / 2.0
        h9_init = project_onto_polytope(best_h, t9)
        alphas_final = [2000, 10000, 50000]
        dl9 = hard_deadline - 5.0
        h9 = pgd_lse_optimize(h9_init, alphas_final, 5000, dl9)
        check_update(h9, n9)

    # Phase 10: Final FISTA polish
    if tleft() > 30:
        n10 = best_n
        t10 = n10 / 2.0
        h10_init = project_onto_polytope(best_h, t10)
        h10 = fista_lse_optimize(h10_init, [5000, 20000], 3000, hard_deadline - 3.0)
        check_update(h10, n10)

    best_h = project_onto_polytope(best_h, best_n / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, best_n
```
