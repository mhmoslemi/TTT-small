```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_weighted_grad(h, y, n):
    grad = np.zeros(n)
    y0 = y[n-1]
    grad += y0 * (1.0 - 2.0 * h)
    for s in range(1, n):
        ws = y[n-1+s]
        if ws < 1e-15:
            continue
        grad[:n-s] += ws * (1.0 - h[s:])
        grad[s:] -= ws * h[:n-s]
        wm = y[n-1-s]
        if wm < 1e-15:
            continue
        grad[s:] += wm * (1.0 - h[:n-s])
        grad[:n-s] -= wm * h[s:]
    return grad

def fista_pressure(h, n, target, alphas, max_iter, lr_init=0.1, momentum=True):
    """
    FISTA with pressure-based objective.
    alphas: list of alpha values to use (increasing)
    """
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        y = np.cumsum(np.zeros(2*n-1))  # will be computed
        h_prev = h_cur.copy()
        if momentum:
            t_k = 1.0
        
        for it in range(max_iter):
            # Compute weights
            corr = compute_corr(h_cur)
            c_max = np.max(corr)
            y = np.exp(alpha * (corr - c_max))
            y_sum = np.sum(y)
            if y_sum > 0:
                y /= y_sum
            
            # Compute gradient
            grad = compute_weighted_grad(h_cur, y, n)
            
            # FISTA step
            if momentum:
                t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
                y_point = h_cur + (t_k - 1.0) / t_next * (h_cur - h_prev)
                h_prev = h_cur.copy()  # will update below
                h_new = project_feasible(y_point - lr_init * grad, target)
                t_k = t_next
            else:
                h_new = project_feasible(h_cur - lr_init * grad, target)
            
            h_cur = h_new
            
            # Check improvement
            if it % 20 == 0:
                score = compute_c5(h_cur)
                if score < best_score:
                    best_score = score
                    best_h = h_cur.copy()
        
        h_cur = best_h.copy()  # reset to best for next alpha
        # Re-project to ensure feasibility
        h_cur = project_feasible(h_cur, target)
    
    return h_cur, compute_c5(h_cur)
```

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_wgrad(h, y, n):
    """Compute gradient of J(h) = sum_k y[k] * corr_full[k] * dx"""
    dx = 2.0 / n
    grad = np.zeros(n)
    # s = 0
    y0 = y[n-1]
    grad += y0 * (1.0 - 2.0 * h)
    # s > 0 (k = n-1+s)
    for s in range(1, n):
        wp = y[n-1+s]  # weight for positive shift s
        wn = y[n-1-s]  # weight for negative shift s (same as shift -s)
        if wp < 1e-15 and wn < 1e-15:
            continue
        # Positive shift s: (1-h[j+s])*1(j<=n-1-s) - h[j-s]*1(j>=s)
        if wp > 1e-15:
            grad[:n-s] += wp * (1.0 - h[s:])
            grad[s:] -= wp * h[:n-s]
        # Negative shift s: (1-h[j-s])*1(j>=s) - h[j+s]*1(j<=n-1-s)
        if wn > 1e-15:
            grad[s:] += wn * (1.0 - h[:n-s])
            grad[:n-s] -= wn * h[s:]
    return grad * dx  # include dx factor
