def smooth_max_objective_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)  # length 2n-1
    dx = 2.0 / n
    
    # Compute softmax weights
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp  # p[s] for s = 0, ..., 2n-2, corresponding to shift s' = s - (n-1)
    
    # Objective
    f = (np.log(sum_exp) + alpha * max_corr) / alpha
    
    # Gradient
    grad = np.zeros(n)
    # p is indexed by m = 0, ..., 2n-2, shift = m - (n-1)
    for i in range(n):
        for t in range(n):
            shift1 = i - t  # for the first term
            shift2 = t - i  # for the second term
            if -（n-1) <= shift1 <= n-1:
                m1 = shift1 + (n-1)
                grad[i] += p[m1] * (1 - h[t])
            if -(n-1) <= shift2 <= n-1:
                m2 = shift2 + (n-1)
                grad[i] -= p[m2] * h[t]

    second_term = p2 @ h

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(h, alpha, mu, target):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    f = (np.log(np.sum(np.exp(shifted))) + alpha * max_corr) / alpha * dx
    f += mu * (np.sum(h) - target) ** 2
    return f


def smooth_max_grad(h, alpha, mu, target):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    p = exp_vals / sum_exp
    i_idx = np.arange(n).reshape(-1, 1)
    t_idx = np.arange(n).reshape(1, -1)
    m1 = i_idx - t_idx + (n - 1)
    valid1 = (m1 >= 0) & (m1 <= 2 * n - 2)
    p1 = np.where(valid1, p[m1], 0.0)
    first_term = p1 @ (1.0 - h)
    m2 = t_idx - i_idx + (n - 1)
    valid2 = (m2 >= 0) & (m2 <= 2 * n - 2)
    p2 = np.where(valid2, p[m2], 0.0)
    second_term = p2 @ h
    grad = dx * (first_term - second_term) + 2.0 * mu * (np.sum(h) - target)
    return grad


def lbfgsb_phase(h0, n, target, alphas, mu, maxiter, deadline):
    bounds = [(0.0, 1.0)] * n
    h_cur = h0.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, jac=smooth_max_grad,
                           args=(alpha, mu, target), method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def find_optimal_t(corr, C, t_min, t_max):
    if t_max - t_min < 1e-13:
        return (t_min + t_max) * 0.5
    lo = 0.0
    has_pos = np.any(C > 1e-14)
    if has_pos:
        hi = float(np.max(corr[C > 1e-14] + C[C > 1e-14] * t_max))
    else:
        hi = float(np.max(corr + C * t_max))
    for _ in range(50):
        mid = (lo + hi) * 0.5
        pos = C > 1e-14
        neg = C < -1e-14
        flat = ~(pos | neg)
        if np.any(flat) and np.max(corr[flat]) > mid + 1e-13:
            lo = mid
            continue
        ub = t_max
        if has_pos:
            ub = min(ub, float(np.min((mid - corr[pos]) / C[pos])))
        lb = t_min
        if np.any(neg):
            lb = max(lb, float(np.max((mid - corr[neg]) / C[neg])))
        if lb <= ub + 1e-11:
            hi = mid
        else:
            lo = mid
    v = hi
    pos = C > 1e-14
    neg = C < -1e-14
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v - corr[pos]) / C[pos])))
    lb = t_min
    if np.any(neg):
        lb = max(lb, float(np.max((v - corr[neg]) / C[neg])))
    ub = min(ub, t_max)
    lb = max(lb, t_min)
    if lb > ub:
        t_star = (t_min + t_max) * 0.5
    else:
        t_star = lb + (ub - lb) * 0.5
    return t_star


def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=40):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur).astype(float)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    max_sweeps = n_sweeps if n_sweeps else 100000

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur).astype(float)

        rng.shuffle(pair_positions)
        max_delta = 0.0

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_vec(h_cur, i, n)
            Bj = compute_B_vec(h_cur, j, n)
            C = Bi - Bj
            t_star = find_optimal_t(corr, C, t_min, t_max)

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
                max_delta = max(max_delta, abs(delta))

        new_score = float(np.max(corr)) * dx
        if new_score < best_score:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 300 and sweep > 200:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur).astype(float)
            best_score = float(np.max(corr)) * dx
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, jac=smooth_max_grad,
                           args=(a, 0.0, target), method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
