import numpy as np

# Pre-imported helpers
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N, dtype=float)
    a[:n] = h
    b = np.zeros(N, dtype=float)
    b[:n] = (1.0 - h)[::-1]
    corr = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b), dtype=float).real
    return corr[:2 * n - 1][::-1]

def initial_h_values():

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h
