  = 1[j < n-k] - h[j+k]*1[j < n-k] - h[j-k]*1[j >= k]
  = (1-h[j+k])*1[j+k < n] - h[j-k]*1[j >= k]

So:
∂c_k/∂h[j] = (1-h[j+k]) if j+k < n, minus h[j-k] if j >= k

The gradient of the LSE objective:
grad[j] = dx * ∑_k w_k * ∂c_k/∂h[j]
= dx * ∑_k w_k * (1-h[j+k]) * 1[j+k < n] - dx * ∑_k w_k * h[j-k] * 1[j >= k]

First term: ∑_{k=0}^{n-1-j} w_k * (1-h[j+k])
This is a correlation: let v = 1-h. Then it's ∑_{k=0}^{n-1-j} w_k * v[j+k]
= (w ⊛ v)[j] where ⊛ denotes cross-correlation with the first argument indexed by lag
= ∑_k w[k] * v[j+k]

Using FFT with zero-padding to fsize:
Let W = FFT(w, fsize), V = FFT(v, fsize)
Then IFFT(conj(W) * V)[j] = ∑_k w[(j-k) mod fsize] * v[k]... no that's not right.

Let me use the standard formula:
Cross-correlation: (f ⊛ g)[m] = ∑_n f[n] * g[n+m]

Using FFT: f ⊛ g = IFFT(conj(FFT(f)) * FFT(g))

So (w ⊛ v)[j] = IFFT(conj(FFT(w)) * FFT(v))[j] = ∑_k w[k] * v[k+j]

Yes! That's what we want. So:
first_term = IFFT(conj(FFT(w, fsize)) * FFT(v, fsize))[:n]

Second term: ∑_{k=0}^{j} w_k * h[j-k]
= ∑_k w[k] * h[j-k] * 1[j-k >= 0]
Let l = j-k: = ∑_l h[l] * w[j-l] * 1[l <= j]
= (h * w)[j] where * denotes convolution... no.
= ∑_l h[l] * w[j-l] = (h ⊛ w)[j]... wait.

Convolution: (h * w)[j] = ∑_l h[l] * w[j-l]

Yes! So the second term is just (h * w)[j] (convolution).

Using FFT: (h * w) = IFFT(FFT(h) * FFT(w))

So:
second_term = IFFT(FFT(h, fsize) * FFT(w, fsize))[:n]

And grad = dx * (first_term - second_term)

So for binary h, the problem is purely combinatorial. The continuous relaxation allows fractional values, which can potentially reduce the max.

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    e = np.exp(shifted)
    s = float(e.sum())
    w = e / s

    term2 = np.fft.irfft(np.fft.rfft(h, fsize) * W, fsize)[:n]
    return dx * (term1 - term2)

def lse_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return (cmax + np.log(np.exp(shifted).sum()) / alpha) * dx

            quad = 0.5 / step * np.dot(xn - yc, xn - yc)

def active_grad(h, corr, top_k=5, temp=0.01):
    """Subgradient focusing on top_k active lags."""
    n = len(h)
    dx = 2.0 / n
    idx = np.argsort(corr)[::-1][:top_k]
    max_c = corr[idx[0]]
    w = np.exp((corr[idx] - max_c) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for w_i, k in zip(w, idx):
        k = int(k)
        grad[:n-k] += w_i * (1.0 - h[k:n])
        grad[k:] -= w_i * h[:n-k]
    return grad * dx

def active_refine(h, target, deadline, top_k=5, temp=0.01, lr_init=0.1):

def bit_flip_search(h, target, deadline, max_flips_per_round=50):
    """Try flipping near-binary values to 0 or 1."""
    n = len(h)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)

def active_grad_direct(h, corr, top_k, temp):
    n = len(h)
    dx = 2.0 / n
    idx = np.argsort(corr)[::-1][:top_k]
    mv = corr[idx[0]]
    w = np.exp((corr[idx] - mv) / temp)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        ki = int(ki)
        grad[:n - ki] += wi * (1.0 - h[ki:n])
        grad[ki:] -= wi * h[:n - ki]
    return grad * dx

                        break
                    else:
                        h_c[j] = old
                        h_c = project_feasible(h_c, target)
                else:
                    continue
                break
            if time.time() > deadline:
                break
        if not improved:
            break
        corr = get_corr(h_c)
        worst = np.argsort(corr)[::-1]
    return h_c, cur

def multi_restart_fista(h0, target, n_restarts, time_each, alpha, deadline):
    n = len(h0)
    best = h0.copy()
    best_sc = compute_c5(h0)
    rng = np.random.RandomState(777)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        dl = min(time.time() + time_each, deadline)
        noise_scale = 0.03 + 0.02 * r
        h_pert = h0 + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_new, sc_new = fista_opt(h_pert, target, alpha, dl)
        if sc_new < best_sc:
            best_sc = sc_new
            best = h_new
    return best, best_sc

def coarse_to_fine(h_fine, n, target, deadline):
    ratio = 4
    n_c = n // ratio
    t_c = n_c / 2.0
    x_c = np.linspace(0, 1, n_c)
    x_f = np.linspace(0, 1, n)
    h_c = np.interp(x_c, x_f, h_fine)
    h_c = project_feasible(h_c, t_c)
    sc_c = compute_c5(h_c)
    dl1 = min(time.time() + 60, deadline - 60)
    h_c, sc_c = fista_opt(h_c, t_c, 1000, dl1)
    h_up = np.interp(x_f, x_c, h_c)
    h_up = project_feasible(h_up, target)
    sc_up = compute_c5(h_up)
    return h_up, sc_up

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    deadline = start + budget_s - 12.0
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_sc = compute_c5(h_init)

    rng = np.random.RandomState(seed)

    cands = [h_init]
    tm = make_thue_morse(n)
    cands.append(project_feasible(tm, target))
    for ss in [3, 11, 42, 99]:
        rb = np.zeros(n)
        pos = np.random.RandomState(ss).choice(n, n // 2, replace=False)
        rb[pos] = 1.0
        cands.append(project_feasible(rb, target))
    cands.append(project_feasible(h_init + rng.randn(n) * 0.08, target))
    cands.append(project_feasible(h_init + rng.randn(n) * 0.15, target))

    for hc in cands:
        sc = compute_c5(hc)
        if sc < best_sc:
            best_sc = sc
            best_h = hc.copy()

    alphas = [80, 300, 1000, 3000, 8000, 20000]
    times = [70, 60, 70, 70, 70, 60]
    for a, td in zip(alphas, times):
        if time.time() > deadline - 250:
            break
        dl = min(time.time() + td, deadline - 200)
        h_new, sc_new = fista_opt(best_h, target, a, dl)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 120:
        dl = min(time.time() + 80, deadline - 80)
        h_new, sc_new = active_refine(best_h, target, dl, top_k=5, temp=0.008, lr0=0.04)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 150:
        dl = min(time.time() + 100, deadline - 100)
        h_new, sc_new = fista_scaled(best_h, target, 15000, dl)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 80:
        dl = min(time.time() + 60, deadline - 50)
        h_new, sc_new = active_refine(best_h, target, dl, top_k=3, temp=0.003, lr0=0.015)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 100:
        time_each = max(15, int((deadline - 100 - time.time()) / 4))
        h_new, sc_new = multi_restart_fista(best_h, target, 4, time_each, 5000, deadline - 60)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 50:
        dl = min(time.time() + 30, deadline - 30)
        h_up, sc_up = coarse_to_fine(best_h, n, target, dl)
        if sc_up < best_sc:
            best_sc = sc_up
            best_h = h_up
            if time.time() < deadline - 20:
                h_new, sc_new = fista_opt(best_h, target, 10000, min(time.time() + 20, deadline - 10))
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_h = h_new

    if time.time() < deadline - 40:
        dl = min(time.time() + 35, deadline - 10)
        h_new, sc_new = local_refine(best_h, target, dl)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    if time.time() < deadline - 15:
        dl = min(time.time() + 12, deadline - 5)
        h_new, sc_new = active_refine(best_h, target, dl, top_k=2, temp=0.001, lr0=0.005)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_sc = compute_c5(h_final)
    return h_final, final_sc, n
