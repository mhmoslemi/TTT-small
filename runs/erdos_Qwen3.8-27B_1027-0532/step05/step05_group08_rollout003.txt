import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, top_k=16):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stagnant = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
            stagnant = 0
        else:
            stagnant += 1
        top_idx = np.argsort(corr)[::-1][:top_k]
        base = corr[top_idx[0]]
        diffs = corr[top_idx] - base
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)

            m = int(m)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_projection(h, target):
    n = len(h)
    n_ones = int(round(target))
    idx = np.argsort(h)[::-1][:n_ones]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def simulated_annealing(h, n, target, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    score_cur = best_score
    temp = 0.005
    cooling = 0.997
    for step in range(20000):
        if time.time() > deadline:
            break
        if temp < 1e-6:
            break
        # Random perturbation
        scale = 0.02 * temp * 100  # scale with temperature
        h_pert = h_cur + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        score_new = compute_c5(h_pert)
        delta = score_new - score_cur
        if delta < 0 or rng.rand() < np.exp(-delta / (temp * 0.01)):
            h_cur = h_pert
            score_cur = score_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_cur.copy()
        temp *= cooling
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

    c_top = float(np.max(corr[top_m]))

def topk_lse_grad(h, alpha, k):
    n = len(h)
    corr = compute_corr(h)
    top_m = np.argsort(corr)[::-1][:k]
    c_top = float(np.max(corr[top_m]))

    w = np.exp(shifted)

def lbfgsb_round(h, n, target, alpha, mu, maxiter, topk=None):
    if topk is None or topk >= 2 * n - 1:
        def obj(x):
            return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
        def objg(x):
            return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    else:
        def obj(x):
            return topk_lse_value(x, alpha, topk) + mu * (np.sum(x) - target) ** 2
        def objg(x):
            return topk_lse_grad(x, alpha, topk) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})

        return np.sum(x) - target

        return np.ones(n)

        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)
        for alpha in [20, 100, 500, 2000, 8000, 30000]:
            if time.time() > deadline - 80:
                break
            for mu in [30.0, 150.0]:
                h3_new = lbfgsb_round(best_h3, n3, target3, alpha, mu, 200)
                sc3 = compute_c5(h3_new)
                if sc3 < best_score3:
                    best_score3 = sc3
                    best_h3 = h3_new
        if time.time() < deadline - 60:
            h_sg3, sc_sg3 = bb_subgradient(best_h3, n3, target3, 1500, deadline - 50, top_k=12, temp_init=0.002)
            if sc_sg3 < best_score3:
                best_score3 = sc_sg3
                best_h3 = h_sg3
        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 10: Final intensive polish on best solution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time.time() < deadline - 30:
        for alpha in [20000, 50000, 100000]:
            if time.time() > deadline - 20:
                break
            h_new = lbfgsb_round(best_h, n_cur, target_cur, alpha, 500.0, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
    if time.time() < deadline - 15:
        h_sg, sc_sg = bb_subgradient(best_h, n_cur, target_cur, 1000, deadline - 10, top_k=6, temp_init=0.001)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    if time.time() < deadline - 5:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.001, deadline - 3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
    if time.time() < deadline - 2:
        h_new = slsqp_round(best_h, n_cur, target_cur, 50000, 1000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
