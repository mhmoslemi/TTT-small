import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        n_free = np.sum(mask)
        if n_free == 0:
            break
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha, corr_top_m=50):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    if corr_top_m < 2*n-1:
        top_idx = np.argsort(corr)[-corr_top_m:]
        sub = corr[top_idx]
    else:
        sub = corr
    cmax = np.max(corr)
    shifted = np.clip(alpha * (sub - cmax), -700.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad_smoother(h, alpha, dx, top_m=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    total = 2*n - 1
    if top_m < total:
        top_idx = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(total)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - cmax), -700.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_s / wsum
    
    grad = np.zeros(n)
    nh = n - 1
    for idx, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k = idx - nh
        if k == 0:
            grad += w * (1.0 - 2.0*h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k]) * dx
                grad[:n-k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n-m] * dx
    return grad

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30, damping=0.8):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Build constraints efficiently
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, verbose=False, warm_start=True)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Check convergence
        if it % 50 == 0:
            change = np.max(np.abs(h_new - prev_h))
            prev_h = h_new.copy()

    total = 2 * n - 1
    if top_m < total:
        top_idx = np.argsort(corr)[-top_m:]
        sub = corr[top_idx]
    else:
        sub = corr
    cmax = np.max(corr)
    shifted = np.clip(alpha * (sub - cmax), -700.0, 0.0)
    return float(cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_grad(h, alpha, dx, top_m=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    total = 2 * n - 1
    if top_m < total:
        top_idx = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(total)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - cmax), -700.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_s / wsum
    grad = np.zeros(n)
    nh = n - 1
    for idx, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k = idx - nh
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset, damping):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_h = h.copy()
    stall_count = 0

    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > time_limit - deadline_offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                seg = g[lo - k:hi - k] * dx
                coeff[lo:hi] = seg
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        if it % 100 == 0 and it > 100:
            change = np.max(np.abs(h_new - prev_h))
            if change < 1e-7:
                stall_count += 1
                if stall_count > 5:
                    break
            else:
                stall_count = 0
            prev_h = h_new.copy()

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def adam_projected(h_init, n, target, max_iter, time_limit, start_time, deadline_offset):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    lr_init = 0.8
    lr_final = 0.005
    T = max(1, max_iter)

    for t in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        progress = t / T
        lr = lr_final + 0.5 * (lr_init - lr_final) * (1.0 + np.cos(np.pi * progress))
        top_m = int(10 + 190 * progress)
        alpha = 30.0 + 500.0 * progress

        grad = compute_grad(h, alpha, dx, top_m=top_m)

        m_vec = beta1 * m_vec + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m_vec / (1.0 - beta1 ** (t + 1))
        v_hat = v_vec / (1.0 - beta2 ** (t + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_try = project_feasible(h - step, target)
        score = compute_c5(h_try)

        if score < best_score - 1e-14:
            best_score = score
            best_h = h_try.copy()
            h = h_try
        else:
            h_try2 = project_feasible(h - 0.3 * step, target)
            score2 = compute_c5(h_try2)
            if score2 < best_score - 1e-14:
                best_score = score2
                best_h = h_try2.copy()
                h = h_try2

    return best_h, best_score

_g_n = 0
_g_target = 0.0

def _sobj(h, alpha, top_m):
    return smooth_obj(h, alpha, top_m)

def _cfun(h):
    return np.sum(h) - _g_target

def _cjac(h):
    return np.ones(_g_n)

def slsqp_opt(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = min(60, 2 * n - 1)
        try:
            res = minimize(_sobj, h_cur, args=(alpha, top_m),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
        except Exception:
            pass

    return h_cur

def interp_to_n(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def _tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    # Phase 1: Alternating LP
    if _tleft() > 250:
        lp_budget = min(budget_s * 0.42, _tleft())
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_budget,
                                     start_time=start_time,
                                     deadline_offset=budget_s - lp_budget,
                                     damping=0.75)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Adam projected gradient
    if _tleft() > 200:
        adam_budget = min(budget_s * 0.22, _tleft())
        h_adam, c5_adam = adam_projected(best_h, n, target,
                                         max_iter=30000, time_limit=adam_budget,
                                         start_time=start_time,
                                         deadline_offset=budget_s - adam_budget)
        if c5_adam < best_score:
            best_score = c5_adam
            best_h = h_adam

    # Phase 3: SLSQP polish
    if _tleft() > 100:
        h_slq = slsqp_opt(best_h, n, target,
                          [30, 100, 500, 2000, 8000],
                          maxiter=4000, ftol=1e-18,
                          time_limit=budget_s, start_time=start_time,
                          deadline_offset=50)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 4: Perturbation restarts
    for noise in [0.003, 0.008, 0.02, 0.04]:
        if _tleft() < 90:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_slq = slsqp_opt(h_pert, n, target,
                          [200, 1000, 5000],
                          maxiter=3000, ftol=1e-18,
                          time_limit=budget_s, start_time=start_time,
                          deadline_offset=35)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: Try higher resolution
    if _tleft() > 300:
        for n_try in [600, 800]:
            if _tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interp_to_n(best_h, n_try, target_try)
            lp_b = min(120, _tleft() * 0.4)
            h_lp2, c5_lp2 = alternating_lp(h_interp, n_try, target_try,
                                           max_iter=20000, time_limit=lp_b,
                                           start_time=start_time,
                                           deadline_offset=budget_s - lp_b,
                                           damping=0.8)
            pg_b = min(80, _tleft() * 0.25)
            if _tleft() > 60:
                h_ad2, c5_ad2 = adam_projected(h_lp2, n_try, target_try,
                                               max_iter=10000, time_limit=pg_b,
                                               start_time=start_time,
                                               deadline_offset=budget_s - pg_b)
                if c5_ad2 < c5_lp2:
                    h_lp2, c5_lp2 = h_ad2, c5_ad2
            if c5_lp2 < best_score:
                best_score = c5_lp2
                best_h = h_lp2
                n = n_try
                target = target_try

    # Phase 6: Final SLSQP
    if _tleft() > 35:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_fin = slsqp_opt(best_h, n_use, t_use,
                          [5000, 20000, 50000],
                          maxiter=3000, ftol=1e-18,
                          time_limit=budget_s, start_time=start_time,
                          deadline_offset=10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Check complement
    h_comp = 1.0 - best_h
    h_comp = project_feasible(h_comp, len(best_h) / 2.0)
    sc_comp = compute_c5(h_comp)
    if sc_comp < best_score:
        best_score = sc_comp
        best_h = h_comp

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
