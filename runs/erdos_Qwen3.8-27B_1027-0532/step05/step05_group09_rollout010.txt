```python
            vals = 1.0 - h[i_lo:i_hi+1]
            grad[i_lo:i_hi+1] += w[k] * vals
        
        # -h[i+off]: i+off in [0,n-1] => i in [-off, n-1-off] ∩ [0,n-1]
        j_lo = max(0, -off)
        j_hi = min(n-1, n-1-off)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] -= w[k] * h[j_lo:j_hi+1]
    
    return grad
```
