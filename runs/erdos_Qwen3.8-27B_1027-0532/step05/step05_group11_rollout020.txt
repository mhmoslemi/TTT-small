def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

        else:

def softmax_gradient(h, alpha, n_top=None):
    """Compute the gradient of the softmax approximation of max overlap."""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    n_half = n - 1
    
    if n_top is not None:
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        # Scale down the rest
        overlaps_scaled = np.minimum(overlaps, overlaps[top_idx[0]])
        overlaps_scaled[top_idx] = overlaps[top_idx]
    else:
        overlaps_scaled = overlaps
    
    # Softmax weights
    max_ov = np.max(overlaps_scaled)
    exp_vals = np.exp(alpha * (overlaps_scaled - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient

    grad = np.zeros(n)

        w = weights[k_offset]
        if w < 1e-15:

            idx2 = np.arange(0, n - kp)
            grad[idx2] += w * (-h[idx2 + kp])

def compute_c5(h):

def compute_all_overlaps(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

        else:

    t_var = cp.Variable()

        h_var >= 0.0,
        h_var <= 1.0

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()

            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    best_h = project_feasible(best_h, target)
    return best_h, compute_c5(best_h)


def alternating_lp_full(h_init, n, target, max_iter, time_limit,
                        damping_start, damping_end, reg_start, reg_end,
                        use_full_lag=True, n_top=200):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 300:
            break
        progress = min(1.0, it / max(1, max_iter * 0.5))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        if use_full_lag and n <= 300:
            h_new, t_val = full_lag_lp(h, n, target, reg)
        else:
            h_new, t_val = top_lag_lp(h, n, target, reg, n_top)
        if h_new is None or np.all(h_new == h):
            stuck += 1
            continue
        h_blend = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h_blend, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def perturb_h(h, rng, strength, n_perturb):
    h_new = h.copy()
    idx = rng.choice(len(h), size=n_perturb, replace=False)
    h_new[idx] += rng.randn(n_perturb) * strength
    return project_feasible(h_new, np.sum(h))


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 100:
        n_coarse = max(100, n // 4)
        target_c = n_coarse / 2.0
        h_c = interpolate_to(best_h, n_coarse, target_c)
        t_lp = min(80, tleft() * 0.25)
        h_c, c5_c = alternating_lp_full(h_c, n_coarse, target_c,
                                         max_iter=3000, time_limit=t_lp,
                                         damping_start=0.9, damping_end=0.7,
                                         reg_start=1e-3, reg_end=1e-8,
                                         use_full_lag=True, n_top=150)
        h_fine = interpolate_to(h_c, n, target)
        sc = compute_c5(h_fine)
        if sc < best_score:
            best_score = sc
            best_h = h_fine

    if tleft() > 120:
        t_lp = min(180, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_full(best_h, n, target,
                                           max_iter=5000, time_limit=t_lp,
                                           damping_start=0.95, damping_end=0.8,
                                           reg_start=1e-4, reg_end=1e-9,
                                           use_full_lag=False, n_top=250)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 100:
        t_sg = min(100, tleft() * 0.2)
        alphas = [100.0, 500.0, 2000.0, 10000.0]
        h_sg, c5_sg = softmax_proximal(best_h, n, target,
                                        alpha_list=alphas,
                                        n_steps=2000, lr_init=0.3,
                                        lr_decay=0.9999)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 120:
        h_ref, c5_ref = slsqp_optimize(best_h, n, target,
                                        [100.0, 500.0, 2000.0, 10000.0, 50000.0],
                                        maxiter=2000, ftol=1e-15)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref

    if tleft() > 100:
        t_lp2 = min(120, tleft() * 0.25)
        h_lp2, c5_lp2 = alternating_lp_full(best_h, n, target,
                                             max_iter=4000, time_limit=t_lp2,
                                             damping_start=0.9, damping_end=0.75,
                                             reg_start=1e-5, reg_end=1e-10,
                                             use_full_lag=False, n_top=300)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    n_attempts = 8
    for attempt in range(n_attempts):
        if tleft() < 60:
            break
        strengths = [0.05, 0.1, 0.15, 0.03, 0.08, 0.12, 0.06, 0.2]
        strength = strengths[attempt % len(strengths)]
        h_pert = perturb_h(best_h, rng, strength, n_perturb=max(20, n // 10))
        t_micro = min(40, tleft() * 0.15)
        h_micro, c5_micro = alternating_lp_full(h_pert, n, target,
                                                 max_iter=1500, time_limit=t_micro,
                                                 damping_start=0.9, damping_end=0.7,
                                                 reg_start=1e-3, reg_end=1e-6,
                                                 use_full_lag=False, n_top=200)
        if c5_micro < best_score:
            best_score = c5_micro
            best_h = h_micro
        else:
            if tleft() > 50:
                h_sl, c5_sl = slsqp_optimize(h_pert, n, target,
                                              [200.0, 1000.0, 5000.0],
                                              maxiter=1000, ftol=1e-14)
                if c5_sl < best_score:
                    best_score = c5_sl
                    best_h = h_sl

    if tleft() > 100:
        for n_high in [500, 600]:
            if tleft() < 80:
                break
            target_h = n_high / 2.0
            h_hi = interpolate_to(best_h, n_high, target_h)
            t_hi = min(60, tleft() * 0.25)
            h_hi, c5_hi = alternating_lp_full(h_hi, n_high, target_h,
                                               max_iter=2000, time_limit=t_hi,
                                               damping_start=0.9, damping_end=0.75,
                                               reg_start=1e-4, reg_end=1e-8,
                                               use_full_lag=(n_high <= 300), n_top=200)
            if tleft() > 60:
                h_hi, c5_hi2 = slsqp_optimize(h_hi, n_high, target_h,
                                               [100.0, 500.0, 2000.0, 10000.0],
                                               maxiter=1000, ftol=1e-14)
                c5_hi = c5_hi2
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_hi
                n = n_high
                target = n_high / 2.0

    if tleft() > 80:
        alphas2 = [500.0, 2000.0, 10000.0, 50000.0]
        t_sg2 = min(80, tleft() * 0.3)
        np_steps = max(1000, int(t_sg2 * 200))
        h_sg2, c5_sg2 = softmax_proximal(best_h, n, target,
                                          alpha_list=alphas2,
                                          n_steps=np_steps, lr_init=0.2,
                                          lr_decay=0.99995)
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    if tleft() > 60:
        h_final_try, c5_final = slsqp_optimize(best_h, n, target,
                                                [500.0, 2000.0, 10000.0, 50000.0, 200000.0],
                                                maxiter=2000, ftol=1e-15)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_try

    if tleft() > 40:
        t_final = min(35, tleft() * 0.8)
        h_fin, c5_fin = alternating_lp_full(best_h, n, target,
                                             max_iter=2000, time_limit=t_final,
                                             damping_start=0.95, damping_end=0.9,
                                             reg_start=1e-6, reg_end=1e-10,
                                             use_full_lag=(n <= 300), n_top=300)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
