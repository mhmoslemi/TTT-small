        return np.sum(x) - _g_target

        return np.ones(_g_n)

    h_cur = h.copy()

                h_cur[j] = h_new
        h_cur = project_feasible(h_cur, target)
        corr_full = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c = corr_full[n - 1:n - 1 + n].copy()
        new_score = float(np.max(c)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                alpha = max(alpha * 0.8, 0.05)
                if alpha < 0.06:
                    break
    return best_h, best_score


def _obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def reweighted_grad_descent(h, n, target, n_steps, beta_init, beta_growth, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        total_w = float(np.sum(weights))
        if total_w < 1e-30:
            break
        weights /= total_w
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-8:
                continue
            nfirst = n - k
            grad[:nfirst] += wk * (1.0 - h_cur[k:k + nfirst])
            grad[k:] -= wk * h_cur[:nfirst]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            if step % 50 == 0:
                beta = min(beta * beta_growth, 50000.0)
        else:
            lr *= lr_decay
            if lr < 1e-8:
                lr = lr_init * 0.1
                beta = max(beta * 0.8, 10)
    return h_cur, cur_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            nfirst = n - k
            grad[:nfirst] += wk * (1.0 - h_cur[k:k + nfirst])
            grad[k:] -= wk * h_cur[:nfirst]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def de_search(n, target, pop_size, max_generations, rng, deadline, F, CR):
    population = []
    for i in range(pop_size):
        hh = rng.rand(n)
        hh = project_feasible(hh, target)
        population.append(hh)
    scores = [compute_c5(hh) for hh in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
    return best, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Quick DE on coarse grids
    n_c1 = 80
    h_de1, _ = de_search(n_c1, n_c1 / 2.0, 30, 60, rng, start_time + 25, 0.8, 0.9)
    h_de1_fine = interpolate_to(h_de1, n, target)
    sc = compute_c5(h_de1_fine)
    if sc < best_score:
        best_score = sc
        best_h = h_de1_fine

    n_c2 = 150
    h_de2, _ = de_search(n_c2, n_c2 / 2.0, 25, 50, rng, start_time + 45, 0.7, 0.8)
    h_de2_fine = interpolate_to(h_de2, n, target)
    sc = compute_c5(h_de2_fine)
    if sc < best_score:
        best_score = sc
        best_h = h_de2_fine

    # Phase 2: Exact coordinate descent (main optimization)
    if tleft() > 150:
        h_ecd, sc_ecd = exact_coord_descent(best_h, n, target, 3000, 1.0, start_time + 300, rng)
        if sc_ecd < best_score:
            best_score = sc_ecd
            best_h = h_ecd

    # Phase 3: Reweighted gradient descent
    if tleft() > 150:
        h_rgd, sc_rgd = reweighted_grad_descent(best_h, n, target, 3000, 100, 1.15, 0.0008, 0.99, start_time + 500)
        if sc_rgd < best_score:
            best_score = sc_rgd
            best_h = h_rgd

    # Phase 4: More exact coordinate descent (finer alpha)
    if tleft() > 150:
        h_ecd2, sc_ecd2 = exact_coord_descent(best_h, n, target, 3000, 0.5, start_time + 700, rng)
        if sc_ecd2 < best_score:
            best_score = sc_ecd2
            best_h = h_ecd2

    # Phase 5: Subgradient polish
    if tleft() > 100:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, 1000, 5, 0.001, 0.998, start_time + 820)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_ref = slsqp_round(best_h, n, target, [2000, 8000, 20000], 3000, 1e-16, start_time + 1000)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final exact coordinate descent (very fine)
    if tleft() > 60:
        h_ecd3, sc_ecd3 = exact_coord_descent(best_h, n, target, 2000, 0.3, start_time + budget_s - 60, rng)
        if sc_ecd3 < best_score:
            best_score = sc_ecd3
            best_h = h_ecd3

    # Phase 8: Final subgradient micro-polish
    if tleft() > 30:
        h_sg2, sc_sg2 = subgradient_polish(best_h, n, target, 500, 8, 0.0005, 0.999, start_time + budget_s - 15)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
