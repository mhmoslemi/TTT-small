import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_vector(h):
    """Compute the correlation vector without the dx scaling."""
    return np.correlate(h, 1.0 - h, mode='full')

def lp_norm_obj(h, p, dx=None):
    """L_p norm of correlation vector, raised to 1/p to approximate max."""
    corr = corr_vector(h)
    if dx is not None:
        corr = corr * dx
    # Clip to avoid negative values (shouldn't happen but just in case)
    corr = np.maximum(corr, 0.0)
    if p == np.inf:
        return float(np.max(corr))
    return float(np.sum(corr ** p) ** (1.0 / p))

    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if compute_c5(h) <= t_val:
        return h, compute_c5(h)
    
    corr = corr_vector(h)
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)

def frank_wolfe_step(h, target, n, g_k):
    """Given gradient g_k, find the vertex of the feasible polytope that minimizes g_k · h."""
    # Sort by gradient value
    order = np.argsort(g_k)
    h_v = np.zeros(n)

    # The remaining n/2 get 0 (since we need sum = n/2 and exactly n/2 are 1)
    # Actually for sum = n/2, we set the smallest n/2 to 1
    return h_v

def subgradient_multi(h, n, target, max_steps, time_limit, n_active=5):

    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

        top_idx = np.argsort(corr)[-n_active:]
        weights = np.exp((corr[top_idx] - corr[top_idx].max()) * 50)

        for w, k_idx in zip(weights, top_idx):

                grad[:n-jk] -= w * dx * h_cur[jk:]

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lp_smooth(h, alpha):
    """Log-sum-exp smooth approximation of max."""
    corr = get_corr(h)
    c_max = np.max(corr)
    if alpha == 0 or alpha == np.inf:
        return float(c_max), corr
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(val), corr

def lp_smooth_grad(h, alpha):
    """Gradient of log-sum-exp smooth max."""
    corr = get_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    n = len(h)

            continue

    half_n = n - 1

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def constraint_gen_lp(h_init, n, target, max_iter, time_limit,
                      damping=0.7, n_active=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1

    corr = get_corr(h)
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    no_improve_count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)

        solved = False
        try:
            prob.solve(solver='CLARABEL', verbose=False)
            solved = (prob.status == 'optimal' or prob.status == 'optimal_inaccurate')
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver='SCS', eps=1e-10, max_iters=150000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        corr = get_corr(h)
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        active_k = top_k | active_k

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(2 * n - 1):
            if k_idx not in active_k and corr[k_idx] * dx > t_val + 1e-5:
                active_k.add(k_idx)
                break

        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())

        if no_improve_count > 20:
            damping = max(0.3, damping * 0.9)
            if no_improve_count > 60:
                break
        elif no_improve_count == 0:
            damping = min(0.95, damping + 0.01)

    return best_h, best_c5

def _lp_obj_and_grad(hf, p, n, dx):
    corr = get_corr(hf) * dx
    corr = np.maximum(corr, 1e-15)
    val = float(np.sum(corr ** p) ** (1.0 / p))
    if p <= 2:
        grad = (corr ** (p - 1)) * (val ** (1 - p)) / n
    else:
        norm_p = (np.sum(corr ** p)) ** (1.0 / p)
        weights = (corr ** (p - 1)) / (norm_p ** p) * val
        half_n = n - 1
        g = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for k_idx in active:
            w = weights[k_idx]
            if w < 1e-12:
                continue
            k = k_idx - half_n
            if k >= 0:
                g[:n - k] += w * dx * (1.0 - hf[k:n])
                g[k:] -= w * dx * hf[:n - k]
            else:
                jk = -k
                g[jk:] += w * dx * (1.0 - hf[0:n - jk])
                g[:n - jk] -= w * dx * hf[jk:]
        return val, g
    return val, grad

def lbfgs_lp_solve(h_init, n, target, p_list, maxiter_total, time_limit):

            break

            corr = np.maximum(corr, 1e-15)

        def grad_func(hf):
            corr = get_corr(hf) * dx
            corr = np.maximum(corr, 1e-15)
            norm_p = (np.sum(corr ** p)) ** (1.0 / p)
            val = norm_p
