import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter):
    def obj(x):
        return lse_value(x, alpha)
    def objg(x):
        return lse_grad(x, alpha)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h

def binarize(h, n, target):
    """Binarize h to 0-1 valued with exactly target ones."""
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:int(target)]] = 1.0
    return h_bin

def flip_local_search(h01, n, target, max_sweeps, deadline, rng, use_annealing=False, T_init=0.01):
    """1-flip local search on 0-1 valued h."""
    h = h01.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    current_score = best_score
    T = T_init
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            old = h[i]
            h[i] = 1.0 - old
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr)) * dx
            delta = new_max - current_score
            if delta < -1e-15:
                current_score = new_max
                corr = new_corr
                improved = True
                if current_score < best_score:
                    best_score = current_score
            elif use_annealing and delta < 0.05:
                # Accept with probability exp(-delta/T)
                if rng.random() < np.exp(-delta / max(T, 1e-10)):
                    current_score = new_max
                    corr = new_corr
                else:
                    h[i] = old
            else:
                h[i] = old
        if use_annealing:
            T *= 0.995
        if not improved and not use_annealing:
            break
    
    return h, best_score

def pair_flip_local_search(h01, n, target, max_rounds, deadline, rng):
    """2-flip local search on 0-1 valued h."""
    h = h01.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    current_score = best_score
    
    # For efficiency, precompute one-flip effects
    # But for simplicity, just try pairs
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Try random pairs
        n_trials = 2000
        for _ in range(n_trials):
            i, j = rng.choice(n, size=2, replace=False)
            if h[i] == h[j]:
                continue  # Flipping two same-valued positions
            old_i, old_j = h[i], h[j]
            h[i] = 1.0 - old_i
            h[j] = 1.0 - old_j
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr)) * dx
            if new_max < current_score - 1e-15:
                current_score = new_max
                corr = new_corr
                improved = True
                if current_score < best_score:
                    best_score = current_score
            else:
                h[i] = old_i
                h[j] = old_j
        if not improved:
            break
    
    return h, best_score

def worst_lag_guided_search(h01, n, target, max_rounds, deadline, rng):
    """Local search guided by the worst lag."""
    h = h01.astype(float).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    current_score = best_score
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        # Find worst lag
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        
        # Find positions that would reduce the worst lag
        # For k_star >= 0: corr = count of j where h[j]=1 and h[j+k_star]=0
        # To reduce: flip h[j] from 1 to 0 (where h[j]=1, h[j+k_star]=0)
        #              OR flip h[j+k_star] from 0 to 1
        candidates = []
        if k_star >= 0:
            for j in range(n - k_star):
                if h[j] == 1.0 and h[j + k_star] == 0.0:
                    candidates.append((j, 0.0))  # flip to 0
                    candidates.append((j + k_star, 1.0))  # flip to 1
        else:
            ell = -k_star
            for j in range(ell, n):
                if h[j] == 1.0 and h[j - ell] == 0.0:
                    candidates.append((j, 0.0))
                    candidates.append((j - ell, 1.0))
        
        if not candidates:
            break
        
        # Shuffle and try top candidates
        rng.shuffle(candidates)
        improved = False
        for pos, new_val in candidates[:50]:
            if time.time() > deadline:
                break
            old = h[pos]
            if old == new_val:
                continue
            h[pos] = new_val
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr)) * dx
            if new_max < current_score - 1e-15:
                current_score = new_max
                corr = new_corr
                improved = True
                if current_score < best_score:
                    best_score = current_score
            else:
                h[pos] = old
            if improved:
                break  # Restart with new worst lag
        
        if not improved:
            # Try random flips as escape
            i = rng.randint(n)
            h[i] = 1.0 - h[i]
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr)) * dx
            # Accept even if worse (annealing-style escape)
            current_score = new_max
            corr = new_corr
    
    return h, best_score

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 8
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter):
    def obj(x):
        return lse_value(x, alpha)
    def objg(x):
        return lse_grad(x, alpha)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return h


def binarize(h, n, target):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.int8)
    h_bin[idx[:int(target)]] = 1
    return h_bin


def flip_search_fast(h_init, n, deadline, rng, anneal=False):
    h = h_init.astype(np.int8).copy()
    hx = h.astype(float)
    one_minus = 1.0 - hx
    corr = compute_corr(hx)
    dx = 2.0 / n
    current = float(np.max(corr)) * dx
    best = current
    best_h = h.copy()
    T = 0.03
    stall = 0
    while time.time() < deadline:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            hx[i] = 1.0 - hx[i]
            one_minus[i] = 1.0 - hx[i]
            nc = compute_corr(hx)
            nm = float(np.max(nc)) * dx
            if nm < current - 1e-15:
                current = nm
                corr = nc
                h[i] = 1 - h[i]
                improved = True
                if nm < best:
                    best = nm
                    best_h = h.copy()
            elif anneal and nm - best < 0.02:
                prob = np.exp(-(nm - current) / max(T, 1e-9))
                if prob > 1.0:
                    prob = 1.0
                if rng.random() < prob:
                    current = nm
                    corr = nc
                    h[i] = 1 - h[i]
                else:
                    hx[i] = 1.0 - hx[i]
                    one_minus[i] = 1.0 - hx[i]
            else:
                hx[i] = 1.0 - hx[i]
                one_minus[i] = 1.0 - hx[i]
        if anneal:
            T *= 0.995
        if not improved:
            stall += 1
            if not anneal:
                break
            if stall > 80:
                n_flip = rng.randint(1, 4)
                for _ in range(n_flip):
                    j = rng.randint(n)
                    h[j] = 1 - h[j]
                    hx[j] = float(h[j])
                    one_minus[j] = 1.0 - hx[j]
                corr = compute_corr(hx)
                current = float(np.max(corr)) * dx
                stall = 0
            elif stall > 200:
                break
        else:
            stall = 0
    return best_h, best


def swap_search_fast(h_init, n, deadline, rng):
    h = h_init.astype(np.int8).copy()
    hx = h.astype(float)
    corr = compute_corr(hx)
    dx = 2.0 / n
    current = float(np.max(corr)) * dx
    best = current
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    while time.time() < deadline:
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_trials = 3000
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            hx[i] = 0.0
            hx[j] = 1.0
            nc = compute_corr(hx)
            nm = float(np.max(nc)) * dx
            if nm < current - 1e-15:
                current = nm
                corr = nc
                h[i] = 0
                h[j] = 1
                pos_i = np.where(ones == i)[0]
                if len(pos_i) > 0:
                    ones = np.delete(ones, pos_i[0])
                pos_j = np.where(zeros == j)[0]
                if len(pos_j) > 0:
                    zeros = np.delete(zeros, pos_j[0])
                ones = np.append(ones, [j])
                zeros = np.append(zeros, [i])
                improved = True
                if nm < best:
                    best = nm
                    best_h = h.copy()
            else:
                hx[i] = 1.0
                hx[j] = 0.0
        if not improved:
            stall += 1
            if stall > 5:
                break
        else:
            stall = 0
    return best_h, best


def greedy_worst_lag_flip(h_init, n, deadline, rng):
    h = h_init.astype(np.int8).copy()
    hx = h.astype(float)
    corr = compute_corr(hx)
    dx = 2.0 / n
    current = float(np.max(corr)) * dx
    best = current
    best_h = h.copy()
    stall = 0
    while time.time() < deadline:
        if stall > 30:

                    candidates.append((j, 0))
                    candidates.append((j + k_star, 1))
        else:
            ell = -k_star
            for j in range(ell, n):

                    candidates.append((j, 0))
                    candidates.append((j - ell, 1))
        if not candidates:
            break
        rng.shuffle(candidates)

                break

                continue
            h[pos] = new_val
            hx[pos] = float(new_val)
            nc = compute_corr(hx)

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score
