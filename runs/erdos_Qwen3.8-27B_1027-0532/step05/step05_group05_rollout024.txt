def compute_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)

from scipy.optimize import minimize
import time

_g_target = 0.0
_g_alpha = 0.0
_g_lse_obj = None

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    return np.sum(x) - _g_target

def _eq_jac(x):
    return np.ones(_g_n)

    corr = compute_corr(x)
    over = corr[_g_n-1:_g_n-1+_g_n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for i in range(len(alpha_list)):
        if time.time() > deadline:
            break
        _g_alpha = float(alpha_list[i])
        maxiter = maxiter_list[i] if i < len(maxiter_list) else maxiter_list[-1]
        try:
            res = minimize(_lsexp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v = np.zeros(n)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        # Get top n_active lags
        top_idx = np.argsort(over)[::-1][:n_active]
        # Compute weighted gradient
        weights = np.zeros(n)
        for k in top_idx:
            weights[k] = 1.0 / float(n_active)
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            w = float(weights[k])
            if k == 0:
                grad += w * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += w * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= w * h_cur[j_neg - k]
        grad *= dx
        # Momentum update

        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= 0.95
            v *= 0.5  # damp momentum on failure
            if lr < 1e-12:
                break
    return best_h, best_score

def fourier_perturb(h, n, target, rng, n_modes, amplitude):
    h_new = h.copy()

        a = rng.randn() * amplitude / np.sqrt(m)
        b = rng.randn() * amplitude / np.sqrt(m)
        x = np.arange(n)
        h_new += a * np.cos(2.0 * np.pi * m * x / n) + b * np.sin(2.0 * np.pi * m * x / n)
    return project_feasible(h_new, target)

            grad[j_neg] -= h_cur[j_neg - k_star]

            eps = min(eps * 1.03, 0.02)

    return h_cur, cur_score

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _eq_con_fun(x):
    return np.sum(x) - _g_target

def _eq_con_jac(x):
    return np.ones(_g_n)

def _lsexp_objective(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, n, target, alpha_list, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alpha_list:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lsexp_objective, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def momentum_subgrad(h, n, target, n_steps, momentum, lr_init, n_active, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    v = np.zeros(n)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        grad = np.zeros(n)
        w_each = 1.0 / float(n_active)
        for k in top_idx:
            k = int(k)
            if k == 0:
                grad += w_each * (1.0 - 2.0 * h_cur) * dx
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += w_each * (1.0 - h_cur[j_pos + k]) * dx
                j_neg = np.arange(k, n)
                grad[j_neg] -= w_each * h_cur[j_neg - k] * dx
        v = momentum * v + grad
        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.015, lr_init * 4.0)
        else:
            lr *= 0.93
            v *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init

        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)

            grad = (1.0 - 2.0 * h_cur) * dx

            j_pos = np.arange(0, n - k_star)

            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]

            eps = min(eps * 1.03, 0.02)

    h_new = h.copy()

        b = rng.randn() * amplitude / np.sqrt(float(m))
        h_new += a * np.cos(2.0 * np.pi * m * x / n) + b * np.sin(2.0 * np.pi * m * x / n)
    return project_feasible(h_new, target)

def de_quick(n, target, pop, n_gen, f_factor, cr, deadline, rng):

    scores = np.array([compute_c5(p) for p in pop])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        i = rng.randint(pop_size)
        mask = idx_range != i
        avail = idx_range[mask]
        a, b, c = rng.choice(avail, size=3, replace=False)
        v = pop[a] + f_factor * (pop[b] - pop[c])
        j_rand = rng.randint(n)
        trial = v.copy()
        r = rng.rand(n)
        trial[r > cr] = pop[i][r > cr]
        trial[j_rand] = v[j_rand]
        trial = project_feasible(trial, target)
        score = compute_c5(trial)
        if score <= scores[i]:
            pop[i] = trial
            scores[i] = score
            if score < best_score:
                best_score = score
                best_h = trial.copy()
    return best_h, best_score

def paired_lag_surgery(h, n, target, max_steps, eps_init, deadline, n_pairs):

    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:

        over = corr[n - 1:n - 1 + n]

        grad = np.zeros(n)
        for k in top_k:
            k = int(k)
            if k == 0:
                grad += (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= h_cur[j_neg - k]
        grad *= dx / float(n_pairs)
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.015)

            eps *= 0.6
            if eps < 1e-10:
                break
    return h_cur, cur_score
