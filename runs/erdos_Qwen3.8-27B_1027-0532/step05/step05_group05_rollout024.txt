import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_over(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:n-1+n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.99999
        else:
            mask = h > 1e-5
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_g_alpha = 100.0
_g_n = 400
_g_target = 200.0

def _obj(x):
    over = compute_over(x)
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _grad(x):
    n = _g_n
    over = compute_over(x)
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    dx = 2.0 / n
    grad = np.zeros(n)

        if w < 1e-18:

        if k == 0:

        else:
            grad[:n-k] += w * (1.0 - x[k:])
            grad[k:] += -w * x[:n-k]
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:

                          constraints=cons, jac=_grad,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def binary_local_search(h, n, target, n_rounds, n_trials_per_round, deadline, rng):
    dx = 2.0 / n
    h_cur = project_feasible(h, target)
    # Project to binary
    h_bin = (h_cur > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)

    best_score = compute_c5(h_cur)
    
    cur_score = compute_c5(h_bin)
    
    zeros = np.where(h_bin < 0.5)[0]
    ones = np.where(h_bin > 0.5)[0]
    
    for rnd in range(n_rounds):

        improved = False
        for trial in range(n_trials_per_round):

For 0 < k ≤ min(i, n-1-i): both indicators are 1
∂over_k/∂h[i] = dx * [(1-h[i+k]) - h[i-k]]

For i < k ≤ n-1-i (possible when i < n/2): only first indicator
∂over_k/∂h[i] = dx * (1-h[i+k])

For n-1-i < k ≤ i (possible when i ≥ n/2): only second indicator
∂over_k/∂h[i] = dx * (-h[i-k])

For k > max(i, n-1-i): ∂over_k/∂h[i] = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_over(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:n-1+n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.99999
        else:
            mask = h > 1e-5
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

_g_alpha = 100.0
_g_n = 400
_g_target = 200.0

def _obj(x):
    over = compute_over(x)
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _grad(x):
    n = _g_n
    over = compute_over(x)
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k] * dx
        if w < 1e-18:
            continue
        if k == 0:
            grad += w * (1.0 - 2.0 * x)
        else:
            grad[:n-k] += w * (1.0 - x[k:])
            grad[k:] += -w * x[:n-k]
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=_grad,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def accelerated_grad(h, n, target, alpha, n_steps, lr_init, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    h_prev = h.copy()
    x_mom = h.copy()
    t = 1.0
    lr = lr_init
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        momentum = (t - 1.0) / t_new
        y = x_mom + momentum * (x_mom - h_prev)
        y = np.clip(y, 0.0, 1.0)
        over_y = compute_over(y)
        c_max = float(np.max(over_y))
        shifted = alpha * (over_y - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        grad = np.zeros(n)
        for k in range(n):
            w = weights[k] * dx
            if w < 1e-18:
                continue
            if k == 0:
                grad += w * (1.0 - 2.0 * y)
            else:
                grad[:n-k] += w * (1.0 - y[k:])
                grad[k:] += -w * y[:n-k]
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)
        sc = float(np.max(compute_over(h_new))) * dx
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if step % 100 == 99:
            sc_cur = float(np.max(compute_over(h_cur))) * dx
            if sc < sc_cur:
                lr = min(lr * 1.05, 0.3)
            else:
                lr *= 0.8
        h_prev = h_cur
        h_cur = h_new
        x_mom = h_new
        t = t_new
    return best_h, best_score

def binary_local_search(h, n, target, n_rounds, n_trials, deadline, rng):
    h_cur = h.copy()
    h_bin = (h_cur > 0.5).astype(float)
    diff = target - np.sum(h_bin)
    if abs(diff) > 0.5:
        h_bin[0:50] = 0.5
        h_bin = project_feasible(h_bin, target)
    cur_score = compute_c5(h_bin)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        zeros = np.where(h_bin < 0.5)[0]
        ones = np.where(h_bin > 0.5)[0]
        for trial in range(n_trials):
            if len(zeros) == 0 or len(ones) == 0:
                break
            i = int(zeros[rng.randint(len(zeros))])
            j = int(ones[rng.randint(len(ones))])
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            sc = compute_c5(h_bin)
            if sc < cur_score - 1e-12:
                cur_score = sc
                improved = True
            else:
                h_bin[i] = 0.0
                h_bin[j] = 1.0
        if not improved:
            break
    if cur_score < best_score:
        best_score = cur_score
        best_h = h_bin.copy()
    if compute_c5(h_cur) < best_score:
        best_score = compute_c5(h_cur)
        best_h = h_cur.copy()
    return best_h, best_score

def coordinate_descent(h, n, target, n_sweeps, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = np.arange(n)
        if sweep % 2 == 1:
            order = order[::-1]
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            h_test = h_cur.copy()
            new_val = h_cur[i] + 0.1
            if new_val < 1.0:
                h_test[i] = new_val
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score - 1e-12:
                    h_cur = h_test
                    cur_score = sc
                    improved = True
                    continue
            new_val = h_cur[i] - 0.1
            if new_val > 0.0:
                h_test = h_cur.copy()
                h_test[i] = new_val
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score - 1e-12:
                    h_cur = h_test
                    cur_score = sc
                    improved = True
                    continue
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_score

def subgradient_active(h, n, target, n_steps, n_active, temp, lr_init, lr_decay, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = compute_over(h_cur)
        top_idx = np.argsort(over)[::-1][:n_active]
        top_vals = over[top_idx]
        diffs = top_vals - top_vals[0]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] += -wk * h_cur[:n-k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        c_curr = float(np.max(compute_over(h_cur))) * dx
        if sc < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 4.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                lr = lr_init * 0.1
    return best_h, best_score

            best_score = sc_sg
            best_h = h_sg

    if tleft() > 40:
        nf = len(best_h)
        tf = nf / 2.0
        hf, scf = slsqp_with_grad(best_h, nf, tf,
                                   [10000, 50000, 200000], 1500, 1e-16, hard_deadline)
        if scf < best_score:
            best_score = scf
            best_h = hf

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
