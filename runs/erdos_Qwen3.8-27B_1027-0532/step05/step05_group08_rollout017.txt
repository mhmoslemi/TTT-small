        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_round(h, n, target, alpha, mu, maxiter, deadline):
    def obj(x):
        corr = compute_corr(x)
        sm = smooth_max_value(corr, alpha)
        pen = mu * (np.sum(x) - target) ** 2
        return sm + pen

    def objg(x):
        g = smooth_max_grad_full(x, alpha)
        g += 2.0 * mu * (np.sum(x) - target) * np.ones(n)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter, ftol, deadline):
    def obj(x):
        corr = compute_corr(x)
        return smooth_max_value(corr, alpha)

    def objg(x):
        return smooth_max_grad_full(x, alpha)

    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def trustcon_round(h, n, target, alpha, maxiter, deadline):
    def obj(x):
        corr = compute_corr(x)
        return smooth_max_value(corr, alpha)

    def objg(x):
        return smooth_max_grad_full(x, alpha)

    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='trust-constr', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'gtol': 1e-10, 'xtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return h


def active_set_momentum(h, n, target, alpha, M, max_iter, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    patience = 0
    patience_max = 80

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * (2.0 / n)

        if c_curr < best_score - 1e-14:
            best_score = c_curr
            best_h = h_cur.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_max:
                break

        top_idx = np.argsort(corr)[::-1][:M]
        g = active_set_grad(h_cur, alpha, top_idx)

        v = beta * v + (1.0 - beta) * g
        g_norm = np.linalg.norm(v)
        if g_norm < 1e-14:
            break

        v_normed = v / g_norm

        found = False
        for factor in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            trial_lr = lr * factor
            h_trial = project_feasible(h_cur - trial_lr * v_normed * g_norm, target)
            sc_trial = compute_c5(h_trial)
            if sc_trial < c_curr - 1e-13:
                h_cur = h_trial
                lr = min(lr * 1.03, 0.02)
                found = True
                break

        if not found:
            lr *= 0.4
            if lr < 1e-11:
                break

    return best_h, best_score


def gradient_descent_line(h, n, target, alpha, max_iter, lr_init, deadline, M=6):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * (2.0 / n)
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()

        top_idx = np.argsort(corr)[::-1][:M]
        g = active_set_grad(h_cur, alpha, top_idx)

        h_new = project_feasible(h_cur - lr * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-13:
            h_cur = h_new
            lr = min(lr * 1.02, 0.015)
        else:
            lr *= 0.7
            if lr < 1e-11:
                break

    return best_h, best_score


def targeted_lag_reduction(h, n, target, max_steps, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    eps = 0.005

    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= 2.0 / n

        h_new = project_feasible(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.012)
        else:
            eps *= 0.8
            if eps < 1e-9:
                break

    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 12.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation with careful alpha schedule
    alphas_p1 = [30, 80, 250, 800, 3000, 10000, 30000, 80000]
    for alpha in alphas_p1:
        if time.time() > deadline - 350:
            break
        for mu in [5.0, 50.0, 500.0, 2000.0]:
            if time.time() > deadline - 300:
                break
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 200, deadline - 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Trust-constr at moderate alpha
    for alpha in [500, 2000, 10000]:
        if time.time() > deadline - 280:
            break
        h_new = trustcon_round(best_h, n, target, alpha, 400, deadline - 280)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Active set with momentum (key improvement over previous)
    configs_p3 = [
        (8, 600, 0.008, 2000),
        (12, 500, 0.006, 4000),
        (16, 400, 0.005, 8000),
    ]
    for M, maxit, lr, alp in configs_p3:
        if time.time() > deadline - 200:
            break
        h_new, sc = active_set_momentum(best_h, n, target, alp, M, maxit, lr, deadline - 200)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: SLSQP at high alpha
    for alpha in [5000, 20000, 80000]:
        if time.time() > deadline - 180:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 3000, 1e-15, deadline - 180)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Higher resolution n=500
    if time.time() < deadline - 250:
        n2 = 500
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)

        for alpha in [50, 300, 1500, 8000, 40000]:
            if time.time() > deadline - 100:
                break
            for mu in [10.0, 200.0, 2000.0]:
                if time.time() > deadline - 90:
                    break
                h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 200, deadline - 90)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

        if time.time() < deadline - 70:
            h2_new, sc2 = active_set_momentum(best_h2, n2, target2, 5000, 10, 300, 0.006, deadline - 70)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

        if time.time() < deadline - 50:
            h2_new = slsqp_round(best_h2, n2, target2, 20000, 1500, 1e-15, deadline - 50)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 6: Higher resolution n=600
    if time.time() < deadline - 180:
        n3 = 600
        target3 = n3 / 2.0
        n_cur = len(best_h)
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)

        for alpha in [50, 300, 2000, 10000, 50000]:
            if time.time() > deadline - 60:
                break
            for mu in [10.0, 200.0]:
                if time.time() > deadline - 55:
                    break
                h3_new = lbfgsb_round(best_h3, n3, target3, alpha, mu, 150, deadline - 55)
                sc3 = compute_c5(h3_new)
                if sc3 < best_score3:
                    best_score3 = sc3
                    best_h3 = h3_new

        if time.time() < deadline - 40:
            h3_new, sc3 = active_set_momentum(best_h3, n3, target3, 5000, 10, 200, 0.005, deadline - 40)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3_new

        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 7: Targeted lag surgery + active set fine-tuning
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_surg, sc_surg = targeted_lag_reduction(best_h, n_cur, target_cur, 300, deadline - 60)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

        if time.time() < deadline - 30:
            h_new, sc = active_set_momentum(best_h, n_cur, target_cur, 8000, 8, 200, 0.003, deadline - 25)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Random perturbations for escaping local optima
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        scales = [0.04, 0.08, 0.12, 0.06, 0.1, 0.15]
        for attempt in range(6):
            if time.time() > deadline - 30:
                break
            h_pert = best_h + rng.randn(n_cur) * scales[attempt]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = lbfgsb_round(h_pert, n_cur, target_cur, 2000, 500.0, 200, deadline - 30)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if time.time() < deadline - 20:
                    h_ref2 = slsqp_round(best_h, n_cur, target_cur, 20000, 1000, 1e-15, deadline - 15)
                    sc2 = compute_c5(h_ref2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_h = h_ref2

    # Phase 9: Final fine-tuning
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, sc = gradient_descent_line(best_h, n_cur, target_cur, 5000, 150, 0.002, deadline - 10, M=6)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Final SLSQP
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_round(best_h, n_cur, target_cur, 50000, 1500, 1e-16, deadline - 3)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
