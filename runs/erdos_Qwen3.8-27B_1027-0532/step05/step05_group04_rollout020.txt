Let me also think about whether I should provide analytical gradients to SLSQP. The objective is the LSE approximation:
f(h) = c_max + (1/alpha) * log(sum(exp(alpha * (corr - c_max))))

The gradient of this with respect to h[i] involves the derivative of corr with respect to h[i], which I already have. But the LSE complicates things because c_max depends on h too.

Actually, for the LSE:
f(h) = c_max + (1/alpha) * log(sum_k exp(alpha * (corr_k - c_max)))
     = c_max + (1/alpha) * log(sum_k exp(alpha * corr_k) * exp(-alpha * c_max))
     = c_max + (1/alpha) * log(sum_k exp(alpha * corr_k)) - c_max
     = (1/alpha) * log(sum_k exp(alpha * corr_k))

Wait, that's the standard log-sum-exp identity! So f(h) = (1/alpha) * log(sum_k exp(alpha * corr_k)).

The gradient is:
∂f/∂h[i] = (1/alpha) * (sum_k alpha * exp(alpha * corr_k) * ∂corr_k/∂h[i]) / (sum_k exp(alpha * corr_k))
         = sum_k w_k * ∂corr_k/∂h[i]

where w_k = exp(alpha * corr_k) / sum_j exp(alpha * corr_j) are the softmax weights.

This is exactly the reweighted gradient! So the SLSQP with LSE objective is equivalent to doing weighted gradient descent with softmax weights, but with the QP subproblem for step size selection.

This means I can provide the analytical gradient to SLSQP, which would be:
grad[i] = sum_k w[k] * ∂corr_k/∂h[i]

where ∂corr_k/∂h[i] is what I computed in _grad_single_lag (but without the dx factor, since corr_k is the raw sum, not the integral).

This is the same weighted gradient I've been computing, just with the weights computed from the full correlation (all 2n-1 lags) rather than just the positive lags.

Actually wait, I need to double-check. The full correlation has 2n-1 entries. The LSE is over ALL of them. But the positive and negative lags give the same overlaps (by the symmetry I discussed earlier, approximately). So the LSE over all 2n-1 is dominated by the same worst lag (appearing twice, at +k and -k).

For the gradient, I need ∂corr_k/∂h[i] for all k in the full correlation output. The full correlation at index m (0-indexed) corresponds to lag m-(n-1).

For lag k (positive, k ≥ 0): corr_{n-1+k} = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
For lag -k (negative): corr_{n-1-k} = sum_{j=k}^{n-1} h[j]*(1-h[j-k])

The gradient for lag k (positive):
∂/∂h[i] corr_{n-1+k} = (1-h[i+k]) if i ≤ n-1-k, minus h[i-k] if i ≥ k

The gradient for lag -k:
∂/∂h[i] corr_{n-1-k} = (1-h[i-k]) if i ≥ k, minus h[i+k] if i ≤ n-1-k

So the total gradient from both lag k and lag -k is:
(1-h[i+k])*1_{i≤n-1-k} - h[i-k]*1_{i≥k} + (1-h[i-k])*1_{i≥k} - h[i+k]*1_{i≤n-1-k}

For the LSE with high alpha, only the worst lag (and its negative) will have significant weight. So the gradient is dominated by the single worst lag pair.

_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _lse_objective(x):
    corr = compute_corr(x)
    # LSE = (1/alpha) * log(sum(exp(alpha * corr)))
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _lse_gradient(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    n = _g_n
    grad = np.zeros(n)

        wk = weights[k_full]
        if wk < 1e-15:

        lag = k_full - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            k = lag
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - x[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * x[j_neg - k]
        else:

            j_pos = np.arange(k, n)
            grad[j_pos] += wk * (1.0 - x[j_pos - k])
            j_neg = np.arange(0, n - k)
            grad[j_neg] -= wk * x[j_neg + k]
    
    return grad

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_lse_objective, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=_lse_gradient,
                           options={'maxiter': maxiter, 'ftol': ftol, ' disp': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur

def topk_gradient_step(h, n, target, k_top, lr, deadline=None, n_iter=1):
    """Take gradient steps on the top-k worst lags."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for _ in range(n_iter):

            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:k_top]
        grad = np.zeros(n)
        for k_full in top_idx:
            k_full = int(k_full)
            lag = k_full - (n - 1)
            if lag == 0:
                grad[:] += (1.0 - 2.0 * h_cur)
            elif lag > 0:
                k = lag
                grad[:n-k] += (1.0 - h_cur[k:])
                grad[k:] -= h_cur[:n-k]
            else:
                k = -lag
                grad[k:] += (1.0 - h_cur[:-k])
                grad[:n-k] -= h_cur[k:]
        
        grad /= k_top
        grad *= dx

        best_h = h_cur
        best_sc = cur_score

            h_try = project_feasible(h_try, target)

                best_sc = sc
                best_h = h_try
            step *= 0.5

    """Softmax-weighted gradient descent with backtracking."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):

            break

        c_max = float(np.max(corr))
        weights = np.exp(beta * (corr - c_max))
        weights /= weights.sum()

        for k_full in range(2 * n - 1):
            wk = weights[k_full]
            if wk < 1e-12:
                continue
            k_full = int(k_full)
            lag = k_full - (n - 1)
            if lag == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)

                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
            else:
                k = -lag
                grad[k:] += wk * (1.0 - h_cur[:-k])
                grad[:n-k] -= wk * h_cur[k:]
        
        grad *= dx
        
        # Backtracking
        best_h = h_cur
        best_sc = cur_score
        step_lr = lr
        for _ in range(10):
            h_try = h_cur - step_lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_sc:
                best_sc = sc
                best_h = h_try
            step_lr *= 0.5
            if step_lr < 1e-12:

            h_cur = best_h
            cur_score = best_sc
            lr = max(step_lr * 2.0, lr * 0.98)
        else:
            lr *= 0.8
            if lr < 1e-12:

    return h_cur, cur_score

def _lse_gradient(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    n = _g_n
    grad = np.zeros(n)
    
    for k_full in range(2 * n - 1):
        wk = weights[k_full]
        if wk < 1e-10:
            continue
        lag = k_full - (n - 1)
        if lag == 0:
            grad[:] += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            k = lag
            grad[:n-k] += wk * (1.0 - x[k:])
            grad[k:] -= wk * x[:n-k]
        else:
            k = -lag
            grad[k:] += wk * (1.0 - x[:-k])
            grad[:n-k] -= wk * x[k:]
    
    return grad

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_with_grad(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_lse_objective, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=_lse_gradient,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur

def softmax_gd(h, n, target, n_steps, beta, lr_init, deadline=None):
    """Efficient softmax-weighted gradient descent."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stalled = 0
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        weights = np.exp(beta * (corr - c_max))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k_full in range(2 * n - 1):
            wk = weights[k_full]
            if wk < 1e-12:
                continue
            lag = k_full - (n - 1)
            if lag == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)
            elif lag > 0:
                k = lag
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
            else:
                k = -lag
                grad[k:] += wk * (1.0 - h_cur[:-k])
                grad[:n-k] -= wk * h_cur[k:]
        
        grad *= dx
        
        # Backtracking line search
        best_h = h_cur
        best_sc = cur_score
        step_lr = lr
        for _ in range(8):
            h_try = h_cur - step_lr * grad
            h_try = project_feasible(h_try, target)
