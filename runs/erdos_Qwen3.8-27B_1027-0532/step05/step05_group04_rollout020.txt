    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    scaled = alpha * over * dx  # = alpha * G_k
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    for k in range(n):

def annealed_gradient_descent(h, n, target, n_steps=10000, T_start=0.001, T_end=1e-8,
                             p=3, lr_init=0.003, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    rng = np.random.RandomState(123)
    lr = lr_init
    
    for step in range(n_steps):
        if deadline and time.time() > deadline:
            break
        
        t_frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** t_frac
        
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]
        c_curr = float(np.max(over)) * dx
        
        top_idx = np.argsort(over)[::-1][:p]
        max_over = over[top_idx[0]]
        weights = np.exp((over[top_idx] - max_over) / 5.0)
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * dx * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * dx * h_cur[j_neg - k]
        
        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)
        
        delta = sc_new - c_curr
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
        else:
            lr *= 0.99
    
    return best_h, best_score
