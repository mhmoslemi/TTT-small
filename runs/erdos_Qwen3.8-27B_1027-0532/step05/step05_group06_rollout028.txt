import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def grad_lag_k(k, h, n):
    """Compute gradient of corr[n-1+k] w.r.t. h, vectorized."""
    grad = np.zeros(n)
    if k >= 0:
        # grad[i] = (1-h[i-k]) for i>=k  +  (-h[i+k]) for i<=n-1-k
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[0:n-k])
            grad[0:n-k] += (-h[k:n])
    else:
        m = -k
        # grad[i] = (1-h[i+m]) for i<=n-1-m  +  (-h[i-m]) for i>=m
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:] += (-h[0:n-m])
    return grad


def lse_obj_and_grad(h, alpha, target, mu):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -800.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(h))
    val += mu * (s - target) ** 2
    
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(2*n-1) / (2*n-1)
    
    # Find active lags
    active = np.where(weights > 1e-12)[0]
    grad = np.zeros(n)
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        grad += w * grad_lag_k(lag, h, n)
    
    grad += 2.0 * mu * (s - target)
    return val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999999
        else:
            mask = h > 1e-9
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def grad_lag_k(k, h, n):
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[0:n-k])
            grad[0:n-k] += (-h[k:n])
    else:
        m = -k
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:] += (-h[0:n-m])
    return grad


def lse_objective(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -800.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val


def lse_gradient(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -800.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(2*n-1) / (2*n-1)
    active = np.where(weights > 1e-12)[0]
    grad = np.zeros(n)
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        grad += w * grad_lag_k(lag, x, n)
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

                       options={'maxiter': maxiter, 'ftol': ftol,
                                'gtol': 1e-12, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h
