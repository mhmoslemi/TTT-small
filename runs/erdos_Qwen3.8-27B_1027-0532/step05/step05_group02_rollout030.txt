def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_cons_n = 0
_cons_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    stall = 0
    max_stall = 150000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        # 30% chance of 2-swap
        if rng.random() < 0.3:
            idx_i = rng.choice(len(ones), size=2, replace=False)
            idx_j = rng.choice(len(zeros), size=2, replace=False)
            i1, i2 = ones[idx_i[0]], ones[idx_i[1]]
            j1, j2 = zeros[idx_j[0]], zeros[idx_j[1]]
            delta = compute_delta_2swap(h, i1, i2, j1, j2)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i1 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i1, j1)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[j1] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def diversify(h, fraction, rng):
    n = len(h)
    h_new = h.copy()
    n_flip = max(1, int(n * fraction))
    positions = rng.choice(n, size=n_flip, replace=False)
    for pos in positions:
        h_new[pos] = 1.0 - h_new[pos]
    target = n // 2
    current = int(h_new.sum())
    if current > target:
        ones = np.where(h_new == 1)[0]
        excess = current - target
        if excess <= len(ones):
            chosen = rng.choice(len(ones), size=excess, replace=False)
            h_new[ones[chosen]] = 0
    elif current < target:
        zeros = np.where(h_new == 0)[0]
        deficit = target - current
        if deficit <= len(zeros):
            chosen = rng.choice(len(zeros), size=deficit, replace=False)
            h_new[zeros[chosen]] = 1
    return h_new

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy polish
    allowed = min(50, tleft() * 0.08)
    h_g, c5_g = fast_greedy(best_h, max_iters=100000, rng=rng,
                             time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Long weighted SA
    allowed = min(300, tleft() * 0.30)
    n_sa = int(min(8000000, allowed * 50000))
    h_sa1, c5_sa1 = sa_weighted(best_h, n_sa, 0.012, 0.00003, rng,
                                 time_limit=allowed, start_time=start_time)
    if c5_sa1 < best_score:
        best_score = c5_sa1
        best_h = h_sa1.copy()

    # Phase 3: 2-swap SA from current best
    if tleft() > 150:
        allowed = min(120, tleft() * 0.20)
        n_sa2 = int(min(4000000, allowed * 30000))
        h_sa2, c5_sa2 = sa_2swap(best_h, n_sa2, 0.010, 0.00003, rng,
                                  time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Iterated local search with diverse perturbations
    fracs = [0.04, 0.06, 0.03, 0.08, 0.05, 0.10, 0.07, 0.04, 0.06, 0.05]
    for frac in fracs:
        if tleft() < 70:
            break
        h_pert = diversify(best_h, frac, rng)
        h_pert = project_binary(h_pert, target_init)

        allowed_g = min(15, tleft() * 0.05)
        h_pg, c5_pg = fast_greedy(h_pert, max_iters=20000, rng=rng,
                                   time_limit=allowed_g, start_time=start_time)

        allowed_s = min(50, tleft() * 0.15)
        n_is = int(min(2000000, allowed_s * 40000))
        h_ps, c5_ps = sa_weighted(h_pg, n_is, 0.008, 0.00002, rng,
                                    time_limit=allowed_s, start_time=start_time)

        if c5_ps < best_score:
            best_score = c5_ps
            best_h = h_ps.copy()

    # Phase 5: Try different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8, n_init + 16]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.06)
        h_g4, c5_g4 = fast_greedy(h_bin_try, max_iters=15000, rng=rng,
                                    time_limit=allowed, start_time=start_time)
        allowed2 = min(25, tleft() * 0.07)
        n_sa4 = int(min(1500000, allowed2 * 40000))
        h_s4, c5_s4 = sa_weighted(h_g4, n_sa4, 0.006, 0.00002, rng,
                                    time_limit=allowed2, start_time=start_time)
        if c5_s4 < c5_g4:
            h_try = h_s4
            c5_try = c5_s4
        else:
            h_try = h_g4
            c5_try = c5_g4
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: SLSQP continuous refinement
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 60, 200, 600, 2000, 6000, 20000]
        allowed = min(tleft() - 40, 160)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-14,
                                  time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final binary SA burst
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(5000000, max(50000, (tleft() - 20) * 50000)))
        h_sa_f, c5_f = sa_weighted(h_bin_final, n_sa_f, 0.005, 0.00001,
                                     rng, time_limit=max(5.0, tleft() - 15), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 8: Final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                  [5000, 15000, 50000], 2000, 1e-15,
                                  time_limit=max(5.0, tleft() - 10), start_start=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
