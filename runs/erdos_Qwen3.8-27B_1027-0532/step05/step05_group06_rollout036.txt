def subgradient_step(h, target, n_top, n_steps, lr_init, deadline):
    """Projected subgradient method focusing on top lags."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = max(top_vals.std(), 1e-15)
        # Weighted subgradient
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / top_std
            lag = int(ki) - (n - 1)
            if lag == 0:
                grad += w * (1.0 - 2.0 * h_cur)
            elif lag > 0:
                grad[lag:] += w * (1.0 - h_cur[:n - lag])
                grad[:n - lag] -= w * h_cur[lag:]
            else:
                m = -lag
                grad[:n - m] += w * (1.0 - h_cur[m:])
                grad[m:] -= w * h_cur[:n - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.03, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return h_cur, cur_score


def coord_descent_1d(h, j, target, n_iter):
    """Golden section search for coordinate j."""
    n = len(h)
    a, b = 0.0, 1.0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    h_c = h.copy()
    h_c[j] = c
    h_c = project_feasible(h_c, target)
    fc = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
    h_d = h.copy()
    h_d[j] = d
    h_d = project_feasible(h_d, target)
    fd = float(np.max(np.correlate(h_d, 1.0 - h_d, mode='full')))
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h_c = h.copy()
            h_c[j] = c
            h_c = project_feasible(h_c, target)
            fc = float(np.max(np.correlate(h_c, 1.0 - h_c, mode='full')))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h_d = h.copy()
            h_d[j] = d
            h_d = project_feasible(h_d, target)
            fd = float(np.max(np.correlate(h_d, 1.0 - h_d, mode='full')))
    best_v = (a + b) / 2.0
    return best_v


def coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h_cur, cur_score
            best_v = coord_descent_1d(h_cur, j, target, 15)

                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-13:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def upsample(h, n_new):
    x_old = np.linspace(0.0, 1.0, len(h))
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def diverse_start(h_base, rng, idx):
    """Create diverse initial solutions."""
    n = len(h_base)
    target = n / 2.0
    if idx == 0:
        # Slight noise
        h = h_base + rng.randn(n) * 0.02
    elif idx == 1:
        # Smooth perturbation (low frequency)

        mask = (freqs > 0) & (freqs < 0.05)
        spec = np.zeros(n // 2 + 1)
        spec[mask] = rng.randn(mask.sum()) * 0.3
        pert = np.fft.irfft(spec, n)
        h = h_base + pert
    elif idx == 2:

        period = n // 6
        h = np.where((np.arange(n) % (2 * period)) < period, 0.7, 0.3)
    elif idx == 3:
        # Cosine wave
        h = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / (n // 4))
    elif idx == 4:
        # Binary random
        h = (rng.rand(n) < 0.5).astype(float)
    elif idx == 5:
        # Bump
        center = rng.randint(n // 4, 3 * n // 4)
        width = n // 8
        t = np.arange(n)
        h = h_base + 0.2 * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif idx == 6:
        # Valley
        center = rng.randint(n // 4, 3 * n // 4)

        t = np.arange(n)

        h = np.roll(h_base, shift)

def crossover(h1, h2, t):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(500):

        if abs(diff) < 1e-13:

            mask = h < 1.0
        else:

    hc[j] = c
    hc = project_feasible(hc, target)
    fc = float(np.max(np.correlate(hc, 1.0 - hc, mode='full')))
    hd = h.copy()
    hd[j] = d
    hd = project_feasible(hd, target)
    fd = float(np.max(np.correlate(hd, 1.0 - hd, mode='full')))
    for _ in range(n_iter):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            hc = h.copy()
            hc[j] = c
            hc = project_feasible(hc, target)
            fc = float(np.max(np.correlate(hc, 1.0 - hc, mode='full')))
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            hd = h.copy()
            hd[j] = d
            hd = project_feasible(hd, target)
            fd = float(np.max(np.correlate(hd, 1.0 - hd, mode='full')))
    return (a + b) / 2.0


def coord_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h_cur, cur_score
            best_v = _coord_1d(h_cur, j, target, 15)
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-13:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def upsample_h(h, n_new):
    x_old = np.linspace(0.0, 1.0, len(h))
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)


def make_diverse(h_base, rng, idx):
    n = len(h_base)
    target = n / 2.0
    if idx == 0:
        h = h_base + rng.randn(n) * 0.03
    elif idx == 1:
        freqs = np.fft.rfftfreq(n)
        mask = (freqs > 0) & (freqs < 0.05)
        spec = np.zeros(n // 2 + 1)
        spec[mask] = rng.randn(int(mask.sum())) * 0.4
        pert = np.fft.irfft(spec, n)
        h = h_base + pert
    elif idx == 2:
        period = max(n // 8, 2)
        h = np.where((np.arange(n) % (2 * period)) < period, 0.65, 0.35)
    elif idx == 3:
        h = 0.5 + 0.25 * np.cos(2.0 * np.pi * np.arange(n) / (n // 5))
    elif idx == 4:
        h = (rng.rand(n) < 0.5).astype(np.float64)
    elif idx == 5:
        center = rng.randint(n // 4, 3 * n // 4)
        width = max(n // 10, 3)
        t = np.arange(n, dtype=np.float64)
        h = h_base + 0.15 * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif idx == 6:
        center = rng.randint(n // 4, 3 * n // 4)
        width = max(n // 10, 3)
        t = np.arange(n, dtype=np.float64)
        h = h_base - 0.15 * np.exp(-0.5 * ((t - center) / width) ** 2)
    elif idx == 7:
        shift = rng.randint(1, n // 5)
        h = np.roll(h_base, shift)
        h = 0.7 * h + 0.3 * 0.5
    else:
        h = h_base + rng.randn(n) * 0.06
    return project_feasible(h, target)


def crossover_mix(h1, h2, t):
    return (1.0 - t) * h1 + t * h2


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_cur = project_feasible(np.array(initial_h_values, dtype=np.float64), target)
    best_h = h_cur.copy()
    best_score = compute_c5(best_h)
    mu = 200.0

    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if time.time() > deadline - 500:
            break
        h_new = lbfgs_step(h_cur, target, a, mu, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    if time.time() < deadline - 400:
        h_sg, sc_sg = subgradient_step(h_cur, target, 8, 400, 0.008, deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()
        h_cur = h_sg

    if time.time() < deadline - 350:
        h_new = lbfgs_step(h_cur, target, 500000, mu, 800)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    if time.time() < deadline - 300:
        h_new = slsqp_step(h_cur, target, 100000, 1e6, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    n2 = 2 * n
    target2 = n2 / 2.0
    if time.time() < deadline - 200:
        h_fine = upsample_h(h_cur, n2)
        h_fine = project_feasible(h_fine, target2)
        h_cur = h_fine
        fine_alphas = [100, 1000, 10000, 100000, 500000]
        for a in fine_alphas:
            if time.time() > deadline - 150:
                break
            h_new = lbfgs_step(h_cur, target2, a, mu, 600)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new

        if time.time() < deadline - 80:
            h_sg, sc_sg = subgradient_step(h_cur, target2, 10, 300, 0.005, deadline)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg.copy()
            h_cur = h_sg

        if time.time() < deadline - 60:
            h_new = lbfgs_step(h_cur, target2, 1000000, mu, 600)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new

    n_cur = len(best_h)
    tgt_cur = n_cur / 2.0

    if time.time() < deadline - 100:
        pop_size = 5
        population = [best_h.copy()]
        for i in range(1, pop_size):
            population.append(make_diverse(best_h, rng, i))
        for i in range(pop_size):
            if time.time() > deadline - 40:
                break
            h_opt = population[i]
            h_opt = lbfgs_step(h_opt, tgt_cur, 1000, mu, 400)
            h_opt = lbfgs_step(h_opt, tgt_cur, 20000, mu, 400)
            population[i] = h_opt
        pop_scores = [compute_c5(p) for p in population]
        for cr in range(4):
            if time.time() > deadline - 30:
                break
            order = np.argsort(pop_scores)
            i1 = int(order[0])
            i2 = int(order[1])
            t = rng.uniform(0.25, 0.75)
            child = crossover_mix(population[i1], population[i2], t)
            child = project_feasible(child, tgt_cur)
            child += rng.randn(n_cur) * 0.015
            child = project_feasible(child, tgt_cur)
            child = lbfgs_step(child, tgt_cur, 2000, mu, 300)
            child = lbfgs_step(child, tgt_cur, 50000, mu, 300)
            sc = compute_c5(child)
            if sc < best_score:
                best_score = sc
                best_h = child.copy()
            worst_idx = int(np.argmax(pop_scores))
            population[worst_idx] = child
            pop_scores[worst_idx] = sc

    if time.time() < deadline - 40:
        h_cd, sc_cd = coord_descent(best_h, tgt_cur, 2, deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    if time.time() < deadline - 25:
        h_new = lbfgs_step(best_h, tgt_cur, 2000000, mu, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    if time.time() < deadline - 10:
        h_new = slsqp_step(best_h, tgt_cur, 500000, 1e7, 200)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    h_final = project_feasible(best_h, tgt_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
