import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gradient(h, alpha):
    n = len(h)
    g = compute_corr(h)
    g_shift = g - np.max(g)
    w = np.exp(alpha * g_shift)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(2*n-1) / (2*n-1)
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for l in range(2*n-1):
        wl = w[l]
        if wl < 1e-14:
            continue
        lag = l - (n-1)
        # First term: (1-h[j-lag]) for valid j
        j_start = max(0, lag)
        j_end = min(n, n + lag)
        if j_start < j_end:
            idx = j_start - lag
            grad[j_start:j_end] += wl * (1.0 - h[idx:idx + (j_end-j_start)])
        # Second term: -h[j+lag] for valid j
        j_start2 = max(0, -lag)
        j_end2 = min(n, n - lag)
        if j_start2 < j_end2:
            idx2 = j_start2 + lag
            grad[j_start2:j_end2] -= wl * h[idx2:idx2 + (j_end2-j_start2)]
    
    return grad

    dx = 2.0 / n
    h_cur = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_h = h0.copy()
    best_score = compute_c5(h0)
    
    # Adaptive step size
    lr = 1.0
    n_iter = 0
    
    while time.time() < deadline:
        alpha = min(alpha_init * (1.05 ** n_iter), alpha_max)
        grad = smooth_max_gradient(h_cur, alpha)
        
        # Lipschitz estimate
        L = np.max(np.abs(grad)) * alpha + 1.0
        step = min(lr, 2.0 / L)

        y_new = y - step * grad

        score_new = compute_c5(h_new)
        score_old = compute_c5(h_cur)
        
        if score_new > score_old + 1e-12:
            lr *= 0.5
            n_iter += 1
            continue

        y = h_new + (t_k - 1.0) / t_new * (h_new - h_cur)
        y = np.clip(y, 0.0, 1.0)
        h_cur = h_new
        t_k = t_new
        n_iter += 1
        
        if score_new < best_score:
            best_score = score_new
            best_h = h_cur.copy()
        
        # If alpha is large enough and we've converged, increase alpha
        if n_iter % 100 == 0:
            alpha_init = min(alpha_init * 2.0, alpha_max)

            break

        h = np.clip(h, 0.0, 1.0)

def smooth_gradient(h, alpha):
    """Compute gradient of smooth max approximation of C5."""
    n = len(h)
    g = compute_corr(h)
    g_max = np.max(g)
    g_shift = g - g_max
    w = np.exp(alpha * g_shift)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum

        wl = w[l]
        if wl < 1e-14:
            continue
        lag = l - (n-1)

        js = max(0, lag)

        je2 = min(n, n - lag)
        if js2 < je2:

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    n = len(h)
    g = compute_corr(h)
    gm = np.max(g)
    w = np.exp(alpha * (g - gm))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for l in range(2*n-1):
        wl = w[l]
        if wl < 1e-13:
            continue
        lag = l - (n-1)
        js = max(0, lag)
        je = min(n, n + lag)
        if js < je:
            grad[js:je] += wl * (1.0 - h[js-lag:je-lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            grad[js2:je2] -= wl * h[js2+lag:je2+lag]
    return grad

def projected_gradient(h0, n, target, deadline, alpha, max_iter=5000):
    h_cur = h0.copy()
    best_h = h0.copy()
    best_score = compute_c5(h0)
    lr = 1.0 / (alpha + 2.0)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_grad(h_cur, alpha)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        # Backtracking
        if sc > compute_c5(h_cur) + 1e-12:
            lr *= 0.5
            if lr < 1e-8:
                break
            continue
        # Increase lr slowly
        lr = min(lr * 1.05, 1.0 / (alpha * 0.1 + 1.0))
        h_cur = h_new

def fista(h0, n, target, deadline, alpha, max_iter=5000):

    t_k = 1.0
    lr = 1.0 / (alpha + 2.0)
    best_h = h0.copy()
    best_score = compute_c5(h0)

            break

        y = y - lr * grad

        sc = compute_c5(h_new)

            # Reset momentum

            t_k = 1.0

def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=30):

    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    stall_count = 0
    
    for sweep in range(100000):

            break
        
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        idx = rng.permutation(n)
        for p in range(n // 2):
            i = int(idx[2*p])
            j = int(idx[2*p+1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:

            # corr_new = corr + delta * (B_i - B_j) where B_k is the "B" vector

            Bi = np.zeros(2*n-1)
            for l in range(2*n-1):
                lag = l - (n-1)
                val = 0.0

                if 0 <= i - lag < n:
                    val += (1.0 - h_cur[i - lag])
                # -h[i+lag] if 0 <= i+lag < n
                if 0 <= i + lag < n:
                    val -= h_cur[i + lag]
                Bi[l] = val
            
            # Similarly for Bj
            Bj = np.zeros(2*n-1)
            for l in range(2*n-1):
                lag = l - (n-1)
                val = 0.0
                if 0 <= j - lag < n:
                    val += (1.0 - h_cur[j - lag])
                if 0 <= j + lag < n:
                    val -= h_cur[j + lag]
                Bj[l] = val

            # delta ranges from (t_min - h_cur[i]) to (t_max - h_cur[i])
            
            d_min = t_min - h_cur[i]
            d_max = t_max - h_cur[i]
            
            # Binary search on the minimum possible max value
            lo = min(float(np.min(A + C * d_min)), float(np.min(A + C * d_max)))
            hi = max(float(np.max(A + C * d_min)), float(np.max(A + C * d_max)))
            
            for _ in range(40):
                mid = (lo + hi) * 0.5
                # Check if max_k (A[k] + mid_delta * C[k]) <= mid for some mid_delta in [d_min, d_max]
                # For each k: A[k] + delta*C[k] <= mid => delta <= (mid-A[k])/C[k] if C[k]>0
                #                              delta >= (mid-A[k])/C[k] if C[k]<0
                pos = C > 1e-14
                neg = C < -1e-14
                ub = d_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = d_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-12:
                    hi = mid
                else:
                    lo = mid
            
            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = d_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = d_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, d_max)
            lb = max(lb, d_min)
            if lb > ub:
                d_star = (d_min + d_max) * 0.5
            else:
                d_star = (lb + ub) * 0.5
            
            h_cur[i] += d_star
            h_cur[j] -= d_star
            corr += d_star * C
            
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-14:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 100:
            break
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    n = len(h)
    g = compute_corr(h)
    gm = np.max(g)
    w = np.exp(alpha * (g - gm))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for l in range(2 * n - 1):
        wl = w[l]
        if wl < 1e-13:
            continue
        lag = l - (n - 1)
        js = max(0, lag)
        je = min(n, n + lag)
        if js < je:
            grad[js:je] += wl * (1.0 - h[js - lag:je - lag])
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            grad[js2:je2] -= wl * h[js2 + lag:je2 + lag]
    return grad

def fista_run(h0, n, target, deadline, alpha, max_iter=20000):
    h_cur = h0.copy()
    y = h0.copy()
    t_k = 1.0
    lr = 1.0 / (0.5 * alpha + 2.0)
    best_h = h0.copy()
    best_score = compute_c5(h0)
    stall = 0
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        grad = smooth_grad(h_cur, alpha)
        y = y - lr * grad
        h_new = project_feasible(y, target)
        sc = compute_c5(h_new)
        sc_old = compute_c5(h_cur)

        if sc > sc_old + 1e-11:
            y = h_cur.copy()
            t_k = 1.0
            lr = max(lr * 0.5, 1e-7)
            stall += 1
            if stall > 100:
                break
            continue

        if sc < best_score - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        stall = 0

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h_cur)
        np.clip(y, 0.0, 1.0, out=y)
        h_cur = h_new
        t_k = t_new
        lr = min(lr * 1.02, 2.0 / (alpha + 1.0))

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

        if no_improve > 3000:
            break

    return best_h, best_score

def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    for l in range(nl1):
        lag = l - (n - 1)
        val = 0.0
        il = i - lag
        if 0 <= il < n:
            val += (1.0 - h[il])
        ir = i + lag
        if 0 <= ir < n:
            val -= h[ir]
        B[l] = val
    return B

def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=25, max_stall=150):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    stall_count = 0

    for sweep in range(200000):
        if time.time() > deadline:
            break
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        idx = rng.permutation(n)
        for p in range(n // 2):
            i = int(idx[2 * p])
            j = int(idx[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_vec(h_cur, i, n)
            Bj = compute_B_vec(h_cur, j, n)
            C = Bi - Bj
            A = corr.copy()

            d_min = t_min - h_cur[i]
            d_max = t_max - h_cur[i]

            lo = min(float(np.min(A + C * d_min)), float(np.min(A + C * d_max)))
            hi = max(float(np.max(A + C * d_min)), float(np.max(A + C * d_max)))
            if hi - lo < 1e-14:
                continue

            for _ in range(38):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                ub = d_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = d_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-12:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = d_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = d_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, d_max)
            lb = max(lb, d_min)
            if lb > ub:
                d_star = (d_min + d_max) * 0.5
            else:
                d_star = (lb + ub) * 0.5

            h_cur[i] += d_star
            h_cur[j] -= d_star
            corr += d_star * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-14:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > max_stall:
            break

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for a in alphas:
        def obj_func(x, alpha=a):
            g = compute_corr(x)
            gm = np.max(g)
            shifted = alpha * (g - gm)
            shifted = np.clip(shifted, -700.0, 0.0)
            return gm + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def cons_eq(x):
            return np.sum(x) - target

        def cons_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def fourier_init(n, seed, K=20):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a = rng.randn() * 0.15 / np.sqrt(k)
        b = rng.randn() * 0.15 / np.sqrt(k)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)
