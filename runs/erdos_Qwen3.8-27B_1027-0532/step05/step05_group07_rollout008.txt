def smooth_max_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)

def compute_grad_smooth(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    w = smooth_max_weights(corr, alpha)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * one_minus_h[i_lo + k:i_hi + k]
        i_lo2 = max(0, k)
        i_hi2 = min(n, n + k)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] -= wk * h[i_lo2 - k:i_hi2 - k]
    return grad


def slsqp_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_cons_eq(x):
    return np.sum(x) - _g_target


def slsqp_cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=5000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_eq, 'jac': slsqp_cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def gradient_descent(h, n, target, deadline, rng, alpha_start=100, alpha_end=10000, 
                     n_iters=5000, lr_init=0.01, momentum=0.9):
    """Nesterov accelerated projected gradient descent on smooth max."""
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    
    n_total = 2 * n - 1
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Progressive alpha
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Nesterov lookahead
        y = h_cur + momentum * v
        
        # Compute gradient at y (projected)
        y_proj = project_feasible(y, target)
        grad = compute_grad_smooth(y_proj, alpha)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.1 * it)
        
        # Gradient step
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        
        # Check improvement
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        
        h_cur = h_new
    
    return best_h, best_score
