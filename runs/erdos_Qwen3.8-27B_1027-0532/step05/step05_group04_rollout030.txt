def precompute_rem_add(h, n):
    """Precompute rem_p[k] for all p (where h[p]=1) and add_q[k] for all q (where h[q]=0)."""
    # rem_p[k]: change in c_k when h[p] flips 1→0
    # Only defined for p where h[p] == 1
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    rem = np.zeros((n, n))  # rem[p, k] for k=0..n-1
    add = np.zeros((n, n))  # add[q, k] for k=0..n-1
    
    for p in ones:
        # Case 1: k ≤ p, contribution is h[p-k] - 1
        for k in range(1, min(p, n-1) + 1):
            rem[p, k] += h[p-k] - 1
        # Case 2: k ≤ n-1-p, contribution is +h[p+k]
        for k in range(1, min(n-1-p, n-1) + 1):
            rem[p, k] += h[p+k]
    
    for q in zeros:
        # Case 1: k ≤ q, contribution is 1 - h[q-k]
        for k in range(1, min(q, n-1) + 1):
            add[q, k] += 1 - h[q-k]
        # Case 2: k ≤ n-1-q, contribution is -h[q+k]
        for k in range(1, min(n-1-q, n-1) + 1):
            add[q, k] -= h[q+k]
    
    return rem, add

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    """G[p, k] = ∂c_k/∂h[p] for k=1,...,n-1"""
    G = np.zeros((n, n))
    for k in range(1, n):
        # Case 1: p >= k, contribution (1-h[p-k])
        p1 = np.arange(k, n)
        G[p1, k] += (1.0 - h[p1 - k])
        # Case 2: p <= n-1-k, contribution -h[p+k]
        p2 = np.arange(0, n - k)
        G[p2, k] += (-h[p2 + k])
    return G

def get_overlapping_corr(corr, n):
    """Get the overlapping part of the correlation (lags 0 to n-1)."""
    return corr[n-1:n-1+n]

    # worst[p,q] = max_k (G[p,k] - G[q,k]) for k in active_lags
    G_active = G[:, active_lags]  # n × |active|

    n_active = len(active_lags)

        diff = g[:, None] - g[None, :]

    if worst_masked[flat_idx] >= 0:

    # 2. alpha <= h[q]

    diffs = G[p, :] - G[q, :]

        margins = margins[margins > 0]
        if len(margins) > 0:

    alpha = max(alpha, 1e-10)

def cg_optimize(h, n, target, max_iters, deadline):
    """Conditional gradient optimization."""
    best_h = h.copy()
    best_score = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        M = float(np.max(over))
        score = M * (2.0/n)

            best_score = score
            best_h = h_cur.copy()

        active = np.where(over >= M - tol)[0]
        active = active[active > 0]  # exclude k=0
        
        if len(active) == 0:
            active = np.array([np.argmax(over)])

        G = compute_gradient_matrix(h_cur, n)

            g = G[:, active[i]]
            diff = g[:, None] - g[None, :]
            worst_active = np.maximum(worst_active, diff)
