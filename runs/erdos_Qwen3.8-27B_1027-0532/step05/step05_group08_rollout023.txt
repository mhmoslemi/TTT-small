```python
    if active_indices is not None:

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_gradient(h, alpha, active_indices=None):

    corr = compute_corr(h)
    if active_indices is not None:
        corr_active = corr[active_indices]
        am = float(np.max(corr_active))
        shifted = np.clip(alpha * (corr_active - am), -700.0, 0.0)

        shifted = np.clip(alpha * (corr - am), -700.0, 0.0)

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def simulated_annealing(h, n, target, n_steps, t_init, t_decay, deadline, scale=0.05):

            break
```

**Idea: Use the dual problem**

The primal is: min max_k f_k(h) s.t. 0≤h≤1, ∑h=n/2

At the optimum, the optimal dual variables λ_k are concentrated on the active set. This suggests an alternating algorithm:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha, active=None):
    corr = compute_corr(h)
    if active is not None:
        corr = np.asarray(corr[active])
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha, active=None):
    n = len(h)
    corr = compute_corr(h)
    if active is not None:
        corr_a = np.asarray(corr[active])
        am = float(np.max(corr_a))
        shifted = np.clip(alpha * (corr_a - am), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        idx = np.asarray(active)
    else:
        am = float(np.max(corr))
        shifted = np.clip(alpha * (corr - am), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        idx = np.arange(len(corr))
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, idx):
        if wk < 1e-14:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def weighted_corr_grad(h, weights, indices):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(weights, indices):
        if wk < 1e-14:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter, active=None):
    def obj(x):
        return lse_value(x, alpha, active) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha, active) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol, active=None):
    def obj(x):
        return lse_value(x, alpha, active)
    def objg(x):
        return lse_grad(x, alpha, active)
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def pg_with_bb(h, n, target, alpha, max_steps, lr_init, deadline, m_start=10, m_end=30, m_ramp=200):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    h_prev = h_cur.copy()
    g_prev = None
    for step in range(max_steps):
        if time.time() > deadline:
            break
        m = min(m_end, m_start + (step * (m_end - m_start)) // max(m_ramp, 1))
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:m]
        g = lse_grad(h_cur, alpha, top_idx)
        if g_prev is not None:
            s = h_cur - h_prev
            y = g - g_prev
            sy = float(s @ y)
            yy = float(y @ y)
            if yy > 1e-14 and sy > 1e-14:
                lr_bb = sy / yy
                lr = 0.5 * lr + 0.5 * float(np.clip(lr_bb, 1e-7, 0.05))
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        if compute_c5(h_new) < best_score:
            best_score = compute_c5(h_new)
            best_h = h_new.copy()
        h_prev = h_cur.copy()
        g_prev = g.copy()
        h_cur = h_new
    return best_h, best_score

def dual_weights_optimize(h, n, target, max_iters, inner_maxiter, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    corr_len = 2 * n - 1
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        beta = 30.0 + 10.0 * it
        shifted = np.clip(beta * (corr - c_max), -700.0, 0.0)
        lam = np.exp(shifted)
        lam /= lam.sum()
        nonzero = lam > 1e-10
        indices = np.where(nonzero)[0]
        w = lam[indices]
        def obj(x):
            c = compute_corr(x)
            return float(w @ c[indices]) + 1e8 * (np.sum(x) - target) ** 2
        def objg(x):
            g = weighted_corr_grad(x, w, indices)
            g += 2e8 * (np.sum(x) - target)
            return g
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', jac=objg,
                           bounds=bounds, options={'maxiter': inner_maxiter, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            break
    return best_h, best_score

def sa_structured(h, n, target, n_steps, t_init, t_decay, deadline, scale=0.03, kernel_size=7):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    T = t_init
    rng = np.random.RandomState(999)
    kernel = np.ones(kernel_size) / kernel_size
    for step in range(n_steps):
        if time.time() > deadline:
            break
        noise = rng.randn(n) * scale
        noise = np.convolve(noise, kernel, mode='same')
        h_new = h_cur + noise
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        sc_cur = compute_c5(h_cur)
        if sc_new < sc_cur or (T > 1e-10 and rng.random() < np.exp(-(sc_new - sc_cur) / T)):
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
        T *= t_decay
    return best_h, best_score

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, m_top=15):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = min(m_top, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = weighted_corr_grad(h_cur, weights, top_idx)
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.03)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.02, 0.015)
        else:
            eps *= 0.85
            if eps < 1e-10:
                break
    return h_cur, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha, active=None):
    corr = compute_corr(h)
    if active is not None:
        corr = np.asarray(corr[active])
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha, active=None):
    n = len(h)
    corr = compute_corr(h)
    if active is not None:
        corr_a = np.asarray(corr[active])
        am = float(np.max(corr_a))
        shifted = np.clip(alpha * (corr_a - am), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        idx = np.asarray(active)
    else:
        am = float(np.max(corr))
        shifted = np.clip(alpha * (corr - am), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        idx = np.arange(len(corr))
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, idx):
        if wk < 1e-14:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def weighted_corr_grad(h, weights, indices):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(weights, indices):
        if wk < 1e-14:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])

        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_round(h, n, target, alpha, mu, maxiter, active=None):
    def obj(x):
        return lse_value(x, alpha, active) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha, active) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter, ftol, active=None):
    def obj(x):
        return lse_value(x, alpha, active)
    def objg(x):
        return lse_grad(x, alpha, active)
    def cf(x):
        return np.sum(x) - target
    def cj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

    h_prev = h_cur.copy()
    g_prev = None
    dx = 2.0 / n

            break
        m = min(m_end, m_start + (step * (m_end - m_start)) // max(m_ramp, 1))
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:m]
        g = lse_grad(h_cur, alpha, top_idx)
        if g_prev is not None:

            y = g - g_prev
            sy = float(s @ y)
            yy = float(y @ y)

                lr = 0.5 * lr + 0.5 * float(np.clip(lr_bb, 1e-7, 0.05))

            best_h = h_new.copy()

                h_new = lbfgsb_round(h, n, target, alpha, 300.0, 150, active=top_m)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h = h_new

    if time.time() < deadline - 300:
        for alpha in [500, 2000, 8000]:
            if time.time() > deadline - 300:
                break
            h_pg, sc_pg = pg_bb(h, n, target, alpha, 8000, 0.004,
                                 deadline - 300, 8, 30, 800)
            if sc_pg < best_score:
                best_score = sc_pg
                best_h = h_pg.copy()
                h = h_pg.copy()

    if time.time() < deadline - 200:
        h_dw, sc_dw = dual_weights_opt(h, n, target, 40, 80, deadline - 200)
        if sc_dw < best_score:
            best_score = sc_dw
            best_h = h_dw.copy()
            h = h_dw.copy()

    if time.time() < deadline - 140:
        for alpha in [5000, 20000]:
            if time.time() > deadline - 130:
                break
            h_new = slsqp_round(h, n, target, alpha, 3000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h = h_new

    if time.time() < deadline - 100:
        for scale_val in [0.04, 0.02, 0.01]:
            if time.time() > deadline - 95:
                break
            h_sa, sc_sa = sa_structured(h, n, target, 3000, 0.008, 0.9995,
                                        deadline - 95, scale_val, 7)
            if sc_sa < best_score:
                best_score = sc_sa
                best_h = h_sa.copy()
                h = h_sa.copy()

    if time.time() < deadline - 60:
        for temp in [0.01, 0.005, 0.002]:
            if time.time() > deadline - 55:
                break
            h_sg, sc_sg = subgrad_polish(h, n, target, 4000, temp, 0.003,
                                          0.997, deadline - 55, 20)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg.copy()
                h = h_sg.copy()

    if time.time() < deadline - 40:
        h_surg, sc_surg = worst_lag_surg(h, n, target, 600, 0.003, deadline - 40)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg.copy()
            h = h_surg.copy()

    if time.time() < deadline - 30:
        for attempt in range(8):
            if time.time() > deadline - 15:
                break
            scales = [0.04, 0.07, 0.03, 0.1, 0.05, 0.12, 0.06, 0.08]
            h_pert = h + rng.randn(n) * scales[attempt]
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfgsb_round(h_pert, n, target, 500, 200.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()
                h = h_ref.copy()
                h_surg2, sc_surg2 = worst_lag_surg(h, n, target, 200, 0.002, deadline - 12)
                if sc_surg2 < best_score:
                    best_score = sc_surg2
                    best_h = h_surg2.copy()

    if time.time() < deadline - 8:
        h_sg, sc_sg = subgrad_polish(best_h, n, target, 2000, 0.002,
                                      0.002, 0.998, deadline - 6, 15)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()

    if time.time() < deadline - 3:
        h_surg, sc_surg = worst_lag_surg(best_h, n, target, 300, 0.001, deadline - 2)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg.copy()

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    if len(initial_h_values) != n:
        h_back = interpolate_to(h_final, n0, target0)
        sc_back = compute_c5(h_back)
        if sc_back < final_score:
            h_final = h_back
            final_score = sc_back

    return h_final, final_score, len(h_final)
```
