15. **Let me think about Frank-Wolfe more carefully:**
    
    The feasible set P = {h : 0 ≤ h_i ≤ 1, sum(h) = n/2}.
    The extreme points of P are the 0/1 vectors with exactly n/2 ones.
    
    Frank-Wolfe algorithm:
    1. Start with h ∈ P (e.g., a 0/1 vector)
    2. Compute subgradient g of the objective at h
    3. Find s = argmin_{s ∈ P} <g, s> (the "linear oracle")
    4. Update: h ← (1-γ)h + γs, where γ is a step size
    
    The linear oracle problem: $\arg\min_{s \in P} \sum_i g_i s_i$ subject to $s_i \in \{0,1\}$, $\sum s_i = n/2$.
    This is solved by picking the n/2 indices with the smallest g_i and setting s=1 there.
    
    This is very efficient! And it maintains feasibility automatically.

Let me simplify my approach:

**Method 1: Frank-Wolfe with subgradient of the max**

For the objective $f(h) = \max_k c_k(h) \cdot dx$, a subgradient at h is:
$g = \nabla c_{k^*}(h) \cdot dx$ where $k^*$ is an argmax.

Frank-Wolfe:
- $s = \arg\min_{s \in P} \langle g, s \rangle$ → pick n/2 indices with smallest g_i, set s=1 there
- Update: $h \leftarrow (1-\gamma_t) h + \gamma_t s$

This is very fast per iteration! The only cost is computing the subgradient, which is O(n) for a single shift.

**Method 2: Smoothed Frank-Wolfe**

Use the log-sum-exp smoothing: $f_\alpha(h) = \frac{1}{\alpha} \log \sum_k \exp(\alpha c_k(h))$
The gradient is: $\nabla f_\alpha(h) = \sum_k p_k \nabla c_k(h)$ where $p_k = \frac{\exp(\alpha c_k)}{\sum_j \exp(\alpha c_j)}$

This requires computing the gradient for all k with non-negligible weight, but with large enough alpha, only a few matter.

**Method 3: Weighted sum SLSQP**

Fix weights $w_k$, minimize $\sum_k w_k c_k(h)$ using SLSQP. The objective is smooth (bilinear) so SLSQP should work well. Update weights after each round.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n

    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad *= dx
    return grad

def smooth_gradient(h, n, alpha):
    """Compute gradient of log-sum-exp smoothing."""
    dx = 2.0 / n
    corr = get_correlation(h)
    c_max = np.max(corr)
    log_w = alpha * (corr - c_max)
    w = np.exp(np.clip(log_w, -500, 0))
    w = w / w.sum()
    
    # Only compute gradient for shifts with significant weight
    threshold = w.max() * 1e-3
    active = np.where(w > threshold)[0]
    
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)

        wm = w[m]

        grad[v1] += wm * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wm * h[idxs[v2] + k]
    grad *= dx
    return grad

def frank_wolfe(h0, n, target, time_limit, max_iter=50000, alpha=50.0, use_smooth=True):
    """Frank-Wolfe algorithm for minimizing max overlap."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n // 2
    
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        if use_smooth:
            grad = smooth_gradient(h, n, alpha)
        else:
            grad = subgradient_max(h, n)
        
        # Linear oracle: find s in P that minimizes <grad, s>
        # P = {s: 0<=s_i<=1, sum(s)=n/2}
        # Optimal s: set s=1 for the n/2 indices with smallest grad values
        sort_idx = np.argsort(grad)
        s = np.zeros(n)
        s[sort_idx[:n_half]] = 1.0
        
        # Step size
        gamma = 2.0 / (t + 2)
        
        # Update
        h_new = (1 - gamma) * h + gamma * s
        
        # Check if improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        
        # Early stopping if gap is small
        if t > 100:
            gap = np.dot(grad, h - s)
            if gap < 1e-7:
                break
    
    return best_h, best_c5

def fw_with_restart(h0, n, target, time_limit, max_iter=20000, alpha=100.0):
    """FW with periodic restart to avoid stagnation."""
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for t in range(max_iter):

def weighted_slsqp(h0, n, target, alphas, maxiter, ftol=1e-15, time_limit=None):
    """SLSQP with weighted sum objective and adaptive weights."""
    global _g_n, _g_target, _g_weights
    start = time.time()
    _g_n = n
    _g_target = target
    
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        corr = get_correlation(h)
        dx = 2.0 / n
        return float(np.sum(_g_weights * corr * dx))
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        # Compute weights based on current h
        corr = get_correlation(h)
        c_max = np.max(corr)
        log_w = alpha * (corr - c_max)
        _g_weights = np.exp(np.clip(log_w, -500, 0))
        _g_weights = _g_weights / _g_weights.sum()
        
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)

            pass
    
    return h

def fw_line_search(h, s, n, gamma_max=1.0):
    """Find best gamma in [0, gamma_max] for h + gamma*(s-h)."""
    # Minimize max_k c_k(h + gamma*(s-h))
    # = max_k (c_k(h) + gamma * d_k)
    # where d_k = c_k(s) - c_k(h)
    # This is the max of linear functions in gamma, so it's convex in gamma.
    # We can use ternary search or just evaluate at a few points.
    
    corr_h = get_correlation(h)
    corr_s = get_correlation(s)
    d = corr_s - corr_h
    dx = 2.0 / n
    
    # The function f(gamma) = max_k (corr_h[k] + gamma * d[k]) * dx
    # is convex in gamma. We want the minimum in [0, gamma_max].
    # Try a few candidate values
    best_gamma = 0.0
    best_val = float(np.max(corr_h)) * dx
    
    for gamma in np.linspace(0, gamma_max, 20):
        val = float(np.max(corr_h + gamma * d)) * dx
        if val < best_val:
            best_val = val
            best_gamma = gamma
    
    # Refine with finer grid around best
    if 0 < best_gamma < gamma_max:
        for gamma in np.linspace(max(0, best_gamma - 0.05), min(1, best_gamma + 0.05), 20):
            val = float(np.max(corr_h + gamma * d)) * dx
            if val < best_val:
                best_val = val
                best_gamma = gamma
    
    return best_gamma

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_weights = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_gradient(h, n, alpha):
    dx = 2.0 / n
    corr = get_correlation(h)
    c_max = np.max(corr)
    log_w = alpha * (corr - c_max)
    w = np.exp(np.clip(log_w, -500.0, 0.0))
    w = w / w.sum()
    mid = n - 1
    idxs = np.arange(n)
    threshold = w.max() * 1e-4
    active = np.where(w > threshold)[0]
    grad = np.zeros(n)
    for m in active:
        k = int(m) - mid
        wm = w[m]
        if wm < 1e-12:
            continue
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wm * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wm * h[idxs[v2] + k]
    grad *= dx
    return grad

def fw_line_search_gamma(h, s, n):
    corr_h = get_correlation(h)
    corr_s = get_correlation(s)
    d = corr_s - corr_h
    dx = 2.0 / n
    best_gamma = 0.0
    best_val = float(np.max(corr_h))
    for gamma in np.linspace(0.0, 1.0, 30):
        val = float(np.max(corr_h + gamma * d))
        if val < best_val:
            best_val = val
            best_gamma = gamma
    lo = max(0.0, best_gamma - 0.04)
    hi = min(1.0, best_gamma + 0.04)
    for gamma in np.linspace(lo, hi, 20):
        val = float(np.max(corr_h + gamma * d))
        if val < best_val:
            best_val = val
            best_gamma = gamma
    return best_gamma

def frank_wolfe_ls(h0, n, target, time_limit, max_iter=30000, alpha=100.0):
    start = time.time()
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n // 2
    stall = 0
    for t in range(max_iter):
        if time.time() - start > time_limit:
            break
        alpha_t = alpha * (1.0 + 0.2 * t / max_iter)
        grad = smooth_gradient(h, n, alpha_t)
        sort_idx = np.argsort(grad)
        s = np.zeros(n)
        s[sort_idx[:n_half]] = 1.0
        gamma = fw_line_search_gamma(h, s, n)
        if gamma < 1e-6:
            stall += 1
            if stall > 500:
                break
            h = (1.0 - 1e-6) * h + 1e-6 * s
        else:
            stall = 0
            h = (1.0 - gamma) * h + gamma * s
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def weighted_slsqp(h0, n, target, alphas_list, maxiter, ftol=1e-15, time_limit=None):
    global _g_n, _g_target, _g_weights
    start = time.time()
    _g_n = n
    _g_target = target
    h = h0.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_fn(h):
        corr = get_correlation(h)
        dx = 2.0 / n
        return float(np.sum(_g_weights * corr * dx))

    def cons_fn(h):
        return np.sum(h) - target

    def cons_jac_fn(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac_fn}]
    for alpha in alphas_list:
        if time_limit is not None and time.time() - start > time_limit:
            break
        corr = get_correlation(h)
        c_max = np.max(corr)
        log_w = alpha * (corr - c_max)
        _g_weights = np.exp(np.clip(log_w, -500.0, 0.0))
        _g_weights = _g_weights / _g_weights.sum()
        try:
            res = minimize(obj_fn, h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        except Exception:
            pass
    return h

def pair_cd(h0, n, target, time_limit, max_sweeps=5000, n_cand=50):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = get_correlation(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        high_set = sorted_idx[-n_cand:]
        low_set = sorted_idx[:n_cand]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                delta_lo = max(h[i] - 1.0, -h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active_mask]) if active_mask.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)

def momentum_subgrad(h0, n, target, time_limit, max_steps=30000, mu=0.8, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = get_correlation(h)
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
