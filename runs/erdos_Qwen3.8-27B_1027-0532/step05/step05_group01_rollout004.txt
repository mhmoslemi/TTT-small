def lp_minmax(h_current, n, target, T):
    """
    Linearize the objective around h_current and solve the LP.
    Returns the solution of the LP.
    """
    dx = 2.0 / n
    corr = np.correlate(h_current, 1.0 - h_current, mode='full')
    n_corr = len(corr)
    mid = n - 1
    
    # Identify active shifts (top T)
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:T]
    
    # Build linearization
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    exprs = []
    for s_idx in active:
        k = int(s_idx) - mid
        # Linearize corr[k] around h_current
        # corr[k] = sum_i h[i] * (1 - h[i+k]) where out-of-bounds = 0
        # Linearization: corr[k] ≈ corr_current[k] + sum_i grad[i] * (h[i] - h_current[i])
        # where grad[i] = (1 - h_current[i+k if in bounds]) - h_current[i-k if in bounds]
        
        idxs = np.arange(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad = np.zeros(n)
        grad[v1] += (1.0 - h_current[idxs[v1] - k])
        grad[v2] -= h_current[idxs[v2] + k]
        
        # corr[k] ≈ corr_current[k] + grad · (h - h_current)
        #         = corr_current[k] - grad · h_current + grad · h
        #         = corr[k]_linear(h)
        linear_val = corr[s_idx] - np.dot(grad, h_current) + grad @ h_var
        exprs.append(linear_val <= t)
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target] + exprs
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS)  # or GLOP, CLARABEL
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1)
    return h_current

        k_star = int(np.argmax(corr)) - mid

        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_delta = 0.0
        best_approx = c5_val
        best_i = -1
        best_j = -1
        best_g = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                d_hi = min(h[i], 1.0 - h[j])
                if d_hi < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = 0.0
                b_d = d_hi
                for _ in range(24):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_approx:
                    best_approx = approx_max
                    best_delta = delta_star
                    best_i = i
                    best_j = j
                    best_g = g
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def lp_refine(h0, n, target, time_limit, n_active=20, max_rounds=50):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:n_active]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for s_idx in active:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            lin_val = corr[s_idx] + grad_k @ (h_var - h)
            constraints.append(lin_val <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.CVXOPT)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)
        if c5_lp < c5_val - 1e-10:
            h = h_lp
        else:
            mix = 0.5
            h_mix = project_feasible(mix * h + (1 - mix) * h_lp, target)
            c5_mix = compute_c5(h_mix)
            if c5_mix < c5_val - 1e-10:
                h = h_mix
            else:
                break
    return best_h, best_c5

def nesterov_smooth(h0, n, target, time_limit, max_steps=10000):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    t_nest = 1.0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        alpha = 50.0 * (1.0 + step * 0.05)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        grad = smooth_obj_grad(h, alpha, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        t_nest = t_new
        lr = 2.0 / (1.0 + t_nest)
        improved = False
        for lr_trial in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_try = project_feasible(h + beta * v - lr_trial * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                v = 0.9 * v + grad
                h = h_try
                improved = True
                break
        if not improved:
            v = 0.5 * v
            if step > 500:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_step(h, n, target, n_top=10, lr=0.02):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    sorted_idx = np.argsort(corr)[::-1][:n_top]
    alpha_w = 30.0
    w = np.exp(alpha_w * (corr[sorted_idx] - c_max))
    w = w / w.sum()
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    for wi, s_idx in zip(w, sorted_idx):
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    h_new = project_feasible(h - lr * grad, target)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: SLSQP with analytical gradient (key improvement)
    if tleft() > 200:
        h_p, s_p = slsqp_grad_round(best_h, n, target,
                                     [50, 200, 800, 3000, 10000, 30000],
                                     3000, 1e-14, tleft() * 0.4, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 2: Fast greedy CD
    if tleft() > 300:
        cd_time = min(300.0, tleft() * 0.35)
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=cd_time, max_sweeps=20000, n_cand=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP refinement after CD
    if tleft() > 150:
        h_p, s_p = slsqp_grad_round(best_h, n, target,
                                     [200, 1000, 5000, 20000, 80000],
                                     2000, 1e-14, tleft() * 0.3, start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p

    # Phase 4: LP refinement
    if tleft() > 100:
        h_lp, c5_lp = lp_refine(best_h, n, target,
                                 time_limit=min(120.0, tleft() * 0.3),
                                 n_active=15, max_rounds=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: Another CD round
    if tleft() > 150:
        cd_time = min(200.0, tleft() * 0.35)
        h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                      time_limit=cd_time, max_sweeps=15000, n_cand=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Nesterov smooth optimization
    if tleft() > 80:
        h_ne, c5_ne = nesterov_smooth(best_h, n, target,
                                       time_limit=min(80.0, tleft() * 0.3),
                                       max_steps=15000)
        if c5_ne < best_score:
            best_score = c5_ne
            best_h = h_ne

    # Phase 7: SLSQP + CD polish
    if tleft() > 100:
        h_p, s_p = slsqp_grad_round(best_h, n, target,
                                     [500, 2000, 10000, 50000, 200000],
                                     2000, 1e-15, min(60.0, tleft() * 0.3), start_time)
        if s_p < best_score:
            best_score = s_p
            best_h = h_p
        if tleft() > 50:
            h_cd, c5_cd = fast_greedy_cd(best_h, n, target,
                                          time_limit=min(40.0, tleft() * 0.4),
                                          max_sweeps=10000, n_cand=25)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Try higher resolution
    if tleft() > 200:
        for n_try in [600, 1000, 1600]:
            if tleft() < 150:
                break
            target_try = float(n_try) / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires, s_hires = slsqp_grad_round(h_interp, n_try, target_try,
                                                 [50, 200, 800, 3000, 10000],
                                                 2000, 1e-14,
                                                 min(100.0, tleft() * 0.35), start_time)
            if s_hires < best_score:
                best_score = s_hires
                best_h = h_hires
            if tleft() < 60:
                break
            h_cd2, c5_cd2 = fast_greedy_cd(h_hires if s_hires < best_score else best_h,
                                            len(h_hires) if s_hires < best_score else n,
                                            (len(h_hires) if s_hires < best_score else n) / 2.0,
                                            time_limit=min(60.0, tleft() * 0.3),
                                            max_sweeps=8000, n_cand=15)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 9: Final LP refinement at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_lp, c5_lp = lp_refine(best_h, n_use, target_use,
                                 time_limit=min(60.0, tleft() * 0.4),
                                 n_active=20, max_rounds=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_final, s_final = slsqp_grad_round(best_h, n_use, target_use,
                                             [2000, 10000, 50000, 200000],
                                             1500, 1e-15,
                                             min(25.0, tleft() * 0.7), start_time)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    # Phase 11: Final CD
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cd, c5_cd = fast_greedy_cd(best_h, n_use, target_use,
                                      time_limit=min(12.0, tleft() * 0.7),
                                      max_sweeps=5000, n_cand=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_out = project_feasible(best_h, target_final)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_final
