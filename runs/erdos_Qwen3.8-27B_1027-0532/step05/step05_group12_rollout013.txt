def lp_alternating(h_init, n, target, start_time, budget, deadline_offset=60,
                   max_iter=200, damping=0.7):

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()

            break

        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

            lo = max(0, k)
            hi = min(n, k + n)

                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(objective), constraints)

        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

        h = project_feasible(h, target)

        if c5_val < best_c5 - 1e-14:

def project_feasible(h, target_sum):

    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:

        else:

        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_c_at_k_vec(h, k_act, dx, n):
    grad = np.zeros(n)
    if k_act == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif k_act > 0:
        nv = n - k_act
        if nv > 0:
            grad[:nv] += (1.0 - h[k_act:k_act+nv]) * dx
            grad[k_act:] -= h[:nv] * dx

        m = -k_act
        nv = n - m
        if nv > 0:
            grad[m:] += (1.0 - h[:nv]) * dx
            grad[:nv] -= h[m:] * dx
    return grad

The dual of the min-max problem can be written as:
$$\min_h \max_k c_k(h) = \min_h \max_{\lambda \in \Delta} \sum_k \lambda_k c_k(h)$$
where $\Delta$ is the simplex (convex combination of shifts).

- Maintain a distribution $\lambda$ over shifts
- Update $h$ to minimize $\sum_k \lambda_k c_k(h)$ (this is a smooth objective)
- Update $\lambda$ to put more weight on the worst shifts

This is the **multiplicative weights** or **follow-the-regularized-leader** approach, and it has theoretical guarantees for min-max games.

def lp_round(h_init, n, target, start_time, budget, deadline_offset=60, max_iter=50, damping=0.7):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > budget - deadline_offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        obj = t_var + 1e-4 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def lbfgs_polish(h_init, n, target, start_time, budget, deadline_offset=30):
    global _g_alpha, _g_penalty, _g_n, _g_target, _g_dx
    dx = 2.0 / n
    _g_n = n
    _g_target = target
    _g_dx = dx
    h_cur = project_feasible(h_init.copy(), target)

    for alpha in [200, 500, 1500, 5000]:
        if time.time() - start_time > budget - deadline_offset:
            break
        _g_alpha = float(alpha)
        _g_penalty = 500.0
        try:
            res = minimize(_obj_lbfgs, h_cur, jac=_grad_lbfgs, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
