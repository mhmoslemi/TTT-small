        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        it += 1
    return best_h, best_c5

def full_lp(h_init, n, target, time_limit, max_iter=50000, solver_pref='GLPK'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it = 0
    while it < max_iter:
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for k in range(2 * n - 1):
            m = k - (n - 1)
            coef = np.zeros(n)
            if m >= 0:
                coef[:n - m] = g[m:] * dx
            else:
                coef[-m:] = g[:n + m] * dx
            if np.any(coef > 1e-15):
                constraints.append(cp.sum(cp.multiply(coef, h_var)) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        try:
            if solver_pref == 'GLPK':
                prob.solve(solver=cp.GLPK, verbose=False)
            elif solver_pref == 'CBC':
                prob.solve(solver=cp.CBC, verbose=False)
            else:
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=200000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            solved = False
        if not solved:
            try:
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=200000, verbose=False)
                solved = h_var.value is not None
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS, max_iter=200000)
                    solved = h_var.value is not None
                except Exception:
                    break
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = project_feasible(h_new, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        it += 1
    return best_h, best_c5

def slsqp_optimize(h_init, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _gn, _gt
    _gn = n
    _gt = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

_gn = 0
_gt = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.array([np.sum(h) - _gt])

def _cons_jac(h):
    return np.array([np.ones(_gn)])

def directed_refine(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    m = int(target)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        k = k_star - (n - 1)
        best_move = None
        best_val = best_c5
        for _ in range(200):
            if time.time() - start > time_limit:
                break
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[np.random.randint(len(ones))])
            j = int(zeros[np.random.randint(len(zeros))])
            h[i] = 0.0
            h[j] = 1.0
            val = compute_c5(h)
            if val < best_val:
                best_val = val
                best_move = (i, j)
            else:
                h[i] = 1.0
                h[j] = 0.0
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            if best_val < best_c5:
                best_c5 = best_val
                best_h = h.copy()
        else:
            if step > 200:
                break
    return best_h, best_c5

def try_resolution(h_best, n_best, target_best, n_try, time_limit, seed=0):
    rng = np.random.RandomState(seed)
    if n_try == n_best:
        h_start = h_best.copy()
    else:
        x_old = np.linspace(0.0, 1.0, n_best)
        x_new = np.linspace(0.0, 1.0, n_try)
        h_start = np.interp(x_new, x_old, h_best)
        h_start = project_feasible(h_start, n_try / 2.0)
    h_lp, c5_lp = full_lp(h_start, n_try, n_try / 2.0, min(time_limit * 0.6, 80), max_iter=5000)
    if time_limit * 0.6 > 40 and c5_lp < compute_c5(h_start):
        h_slq, c5_slq = slsqp_optimize(h_lp, n_try, n_try / 2.0, [500, 2000], 500, 1e-14, time_limit * 0.3)
        if c5_slq < c5_lp:
            return h_slq, c5_slq, n_try
    return h_lp, c5_lp, n_try

def run(seed=42, budget_s=1300, **kwargs):
    global _gn, _gt
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary SA from initial
    if tleft() > 250:
        sa_time = min(250, tleft() * 0.3)
        h_sa, c5_sa = binary_sa(h_init, n, target, sa_time, seed=seed)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa

    # Phase 2: Full LP on best
    if tleft() > 200:
        lp_time = min(200, tleft() * 0.35)
        h_lp, c5_lp = full_lp(best_h, n, target, lp_time, max_iter=100000, solver_pref='GLPK')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: SLSQP polish
    if tleft() > 100:
        slq_time = min(100, tleft() * 0.2)
        h_slq, c5_slq = slsqp_optimize(best_h, n, target, [100, 500, 2000], 1000, 1e-14, slq_time)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 4: Full LP round 2
    if tleft() > 150:
        lp_time = min(150, tleft() * 0.3)
        h_lp, c5_lp = full_lp(best_h, n, target, lp_time, max_iter=100000, solver_pref='GLPK')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: SLSQP with higher alpha
    if tleft() > 80:
        slq_time = min(80, tleft() * 0.3)
        h_slq, c5_slq = slsqp_optimize(best_h, n, target, [2000, 10000, 50000], 800, 1e-15, slq_time)
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 6: Directed binary refinement
    if tleft() > 100:
        dir_time = min(100, tleft() * 0.25)
        h_dir, c5_dir = directed_refine(best_h, n, target, dir_time)
        if c5_dir < best_score:
            best_score = c5_dir
            best_h = h_dir

    # Phase 7: Binary SA from random start
    if tleft() > 150:
        sa_time = min(120, tleft() * 0.3)
        h_rand = rng.randint(0, 2, size=n).astype(float)
        h_rand = project_feasible(h_rand, target)
        h_sa, c5_sa = binary_sa(h_rand, n, target, sa_time, seed=seed + 1)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
        if tleft() > 80:
            lp_time = min(80, tleft() * 0.4)
            h_lp, c5_lp = full_lp(h_sa, n, target, lp_time, max_iter=50000, solver_pref='GLPK')
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    # Phase 8: Binary SA from periodic pattern
    if tleft() > 150:
        period = 4
        h_per = np.zeros(n)
        for i in range(n):
            h_per[i] = 1.0 if (i % period) < period // 2 else 0.0
        h_per = project_feasible(h_per, target)
        sa_time = min(100, tleft() * 0.3)
        h_sa, c5_sa = binary_sa(h_per, n, target, sa_time, seed=seed + 2)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
        if tleft() > 60:
            lp_time = min(60, tleft() * 0.4)
            h_lp, c5_lp = full_lp(h_sa, n, target, lp_time, max_iter=50000, solver_pref='GLPK')
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    # Phase 9: Try different resolutions
    if tleft() > 180:
        for n_try in [n - 50, n + 50, n + 100]:
            if tleft() < 80:
                break
            if n_try < 100 or n_try > 800:
                continue
            if n_try % 2 != 0:
                n_try += 1
            try:
                h_t, c5_t, n_t = try_resolution(best_h, n, target, n_try, min(80, tleft() * 0.4), seed=seed + n_try)
                if c5_t < best_score:
                    best_score = c5_t
                    best_h = h_t
            except Exception:
                pass

    # Phase 10: Final LP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(50, tleft() * 0.6)
        h_lp, c5_lp = full_lp(best_h, n_use, target_use, lp_time, max_iter=50000, solver_pref='GLPK')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slq, c5_slq = slsqp_optimize(best_h, n_use, target_use, [5000, 20000], 500, 1e-16, min(30, tleft() * 0.8))
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 12: One more SA round if time
    if tleft() > 60:
        sa_time = min(50, tleft() * 0.7)
        h_sa, c5_sa = binary_sa(best_h, len(best_h), len(best_h) / 2.0, sa_time, seed=seed + 100)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
        if tleft() > 20:
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_lp, c5_lp = full_lp(best_h, n_use, target_use, min(20, tleft() * 0.8), max_iter=20000, solver_pref='GLPK')
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
