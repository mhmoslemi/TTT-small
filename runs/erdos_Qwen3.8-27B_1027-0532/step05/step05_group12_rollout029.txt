def project_feasible(h, target_sum):

def pc_overlap(params, p):
    """Compute max overlap for piecewise-constant h with p segments.
    params: array of length 2p: [a_1,...,a_p, v_1,...,v_p]
    a_i are log-parameters for segment lengths
    v_i are segment values
    Returns: (max_overlap, h_array_of_length_p, boundaries)
    """
    a = params[:p]
    v = params[p:]
    # Segment lengths
    exp_a = np.exp(a - np.max(a))  # for numerical stability
    w = 2.0 * exp_a / np.sum(exp_a)
    # Boundaries
    theta = np.concatenate([[0.0], np.cumsum(w)])
    # Compute O(m) at all breakpoints
    # Breakpoints: m = theta[i] - theta[j] for all i, j
    # Filter to [-2, 2]
    m_vals = np.sort(np.unique(np.concatenate([theta[:,None] - theta[None,:].ravel()]).ravel()))
    m_vals = m_vals[(m_vals >= -2.0) & (m_vals <= 2.0)]
    if len(m_vals) == 0:
        m_vals = np.array([0.0])
    
    max_overlap = 0.0
    for m in m_vals:
        # O(m) = integral_0^2 h(x+m)*(1-h(x)) dx
        # Break [0,2] at points where h(x) or h(x+m) changes
        # h(x) changes at theta[1:-1]
        # h(x+m) changes at theta[1:-1] - m
        breaks = np.concatenate([[0.0], theta[1:-1], theta[1:-1] - m, [2.0]])
        breaks = np.sort(breaks[(breaks >= 0.0) & (breaks <= 2.0)])
        if len(breaks) < 2:
            continue
        # For each subinterval, h(x) and h(x+m) are constant
        # Find which segment each point is in
        overlap = 0.0
        for idx in range(len(breaks)-1):
            x_lo = breaks[idx]
            x_hi = breaks[idx+1]
            length = x_hi - x_lo
            if length < 1e-15:
                continue
            x_mid = 0.5*(x_lo + x_hi)
            # Find segment for x_mid
            seg_i = np.searchsorted(theta, x_mid, side='right') - 1
            seg_i = max(0, min(p-1, seg_i))
            # Find segment for x_mid + m
            xm = x_mid + m
            if xm < 0 or xm > 2:
                h_xm = 0.0
            else:
                seg_j = np.searchsorted(theta, xm, side='right') - 1
                seg_j = max(0, min(p-1, seg_j))
                h_xm = v[seg_j]
            overlap += length * h_xm * (1.0 - v[seg_i])
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap, v, theta

def pc_objective(params, p):
    max_ov, _, _ = pc_overlap(params, p)
    return max_ov

def pc_constraint(params, p):
    a = params[:p]
    v = params[p:]
    exp_a = np.exp(a - np.max(a))
    w = 2.0 * exp_a / np.sum(exp_a)
    return np.sum(v * w) - 1.0

def pc_constraint_jac(params, p):
    a = params[:p]
    v = params[p:]
    exp_a = np.exp(a - np.max(a))
    s = np.sum(exp_a)
    w = 2.0 * exp_a / s
    # d(sum(v*w))/da_i = sum(v) * dw_i/da_i
    # dw_i/da_i = w_i * (1 - w_i/2) ... let me derive
    # w_i = 2*exp(a_i)/sum(exp(a_j))
    # dw_i/da_i = 2*exp(a_i)*(sum(exp) - exp(a_i))/sum(exp)^2 = w_i*(1 - exp(a_i)/sum(exp)) = w_i*(1 - w_i/2)
    # dw_j/da_i = -2*exp(a_j)*exp(a_i)/sum(exp)^2 = -w_j*w_i/2 for j≠i
    jacs = np.zeros(2*p)
    for i in range(p):
        dwi_da_i = w[i] * (1.0 - w[i]/2.0)
        jacs[i] = np.sum(v) * dwi_da_i

def pc_to_discrete(v, theta, n):
    """Convert piecewise-constant to discrete array of length n."""
    x = np.linspace(0, 2, n, endpoint=False) + 1.0/n  # midpoints
    h = np.interp(x, theta[1:-1], np.concatenate([[v[0]], v]), left=v[0], right=v[-1])
    # Actually, use step function
    h = np.zeros(n)
    for i in range(len(x)):
        seg = np.searchsorted(theta, x[i], side='right') - 1
        seg = max(0, min(len(v)-1, seg))
        h[i] = v[seg]
    return h

def smooth_obj_discrete(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad_discrete(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

def proj_grad_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    steps_per_alpha = 2000
    top_m_list = [5, 10, 20, 30, 50, 80, 120, 200, 2*n-1, 2*n-1]

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]

        momentum_v = np.zeros(n)
        beta = 0.9
        stalled = 0

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            grad = smooth_grad_discrete(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad

            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-14:
                break

            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 1.0)

            improved = False
            for _ in range(30):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5

            if not improved:
                stalled += 1
                if stalled > 30:
                    break

    return best_h, best_score

def slsqp_optimize_discrete(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(_global_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj_discrete, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur

def optimize_piecewise(p, h_init, n_init, time_limit, start_time):

        if np.sum(mask) > 0:

    a_init = np.log(w_init + 1e-10)

    for i in range(p):

    cons = [{'type': 'eq', 'fun': lambda params: pc_constraint(params, p),
             'jac': lambda params: pc_constraint_jac(params, p)}]
    
    best_params = params_init.copy()
    best_score = pc_objective(params_init, p)
    
    for maxiter in [100, 200, 500, 1000]:
        if time.time() - start_time > time_limit - 30:

            res = minimize(pc_objective, best_params, args=(p,),

            if res.x is not None:
                sc = pc_objective(res.x, p)
                if sc < best_score:

        except Exception:
            pass

Let me also add a **block coordinate descent** approach:
- Pick a random subset of indices (a "block")
- Optimize only those coordinates (holding others fixed)
- This is a 1D or low-dimensional subproblem that can be solved optimally

For a single coordinate h[i]:
O(k) = sum_j h[j](1-h[j+k])

The derivative of O(k) with respect to h[i] depends on which shifts k involve h[i]:
- For shift k: h[i] appears as h[i] (with coefficient (1-h[i+k])) or as h[i+k] (with coefficient -h[i-k])

So ∂O(k)/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate bounds checking)

The max over k of O(k) is minimized by setting h[i] to balance the top contributors.

For a block of m coordinates, we can solve a small QP or use gradient descent.

Actually, for a single coordinate, the problem is:
min_{h[i]} max_k [O(k) + (∂O(k)/∂h[i]) * (h[i] - h[i]_old)]

This is a 1D problem (piecewise linear), solvable by checking O(m) breakpoints where m is the number of "active" shifts.

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

def nesterov_grad_descent(h_init, n, target, time_limit, start_time, deadline_offset=30):
    """Nesterov accelerated gradient with projection."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Nesterov parameters
    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    top_m_list = [5, 10, 20, 30, 50, 80, 120, 200, 2*n-1, 2*n-1]
    steps_per_alpha = 1500
    
    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]
        
        y = h.copy()  # extrapolated point
        t_k = 1.0  # Nesterov momentum parameter
        stalled = 0
        
        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            
            grad_y = smooth_grad(y, alpha, dx, top_m=top_m)
            
            # Learning rate
            gnorm = np.max(np.abs(grad_y))
            if gnorm < 1e-14:
                break
            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 0.5)
            
            # Nesterov update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_new
            y_new = project_feasible(h - lr * grad_y, target)
            y = y_new + beta * (y_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = y_new
            t_k = t_new
            
            score = compute_c5(h)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h.copy()
                stalled = 0
            else:
                stalled += 1
            
            if stalled > 50:
                break
    
    return best_h, best_score

def generic_momentum(h_init, n, target, time_limit, start_time, deadline_offset=30):
    """Generic momentum method with adaptive parameters."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    steps_per_alpha = 1000
    top_m_list = [5, 8, 12, 20, 30, 50, 80, 120, 200, 2*n-1, 2*n-1, 2*n-1]

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]

        v = np.zeros(n)
        beta = 0.85 + 0.1 * (ai / len(alpha_schedule))
        stalled = 0
        lr = 0.5

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            v = beta * v + (1.0 - beta) * grad

            gnorm = np.max(np.abs(v))
            if gnorm < 1e-14:
                break

            # Adaptive learning rate
            lr = min(lr * 1.1, 2.0 / (alpha * gnorm + 1e-10))
            lr = min(lr, 1.0)

            improved = False
            for trial in range(25):
                h_try = h - lr * v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5

            if not improved:
                lr *= 2.0  # try larger next time
                lr = min(lr, 1.0)
                stalled += 1
                if stalled > 40:
                    break

    return best_h, best_score

def slsqp_polish(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(_global_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur

def block_coordinate_descent(h_init, n, target, time_limit, start_time, deadline_offset=30, block_size=10):
    """Optimize random blocks of coordinates."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(12345)
    
    alpha = 100
    
    for iteration in range(100000):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        # Pick a random block
        start_idx = rng.randint(0, n - block_size)
        block = np.arange(start_idx, start_idx + block_size)
        
        # Compute gradient for the block
        grad_full = smooth_grad(h, alpha, dx, top_m=20)
        grad_block = grad_full[block]
        
        # Line search within the block
        lr = 1.0 / (alpha * np.max(np.abs(grad_block)) + 1e-10)
        lr = min(lr, 0.5)
        
        improved = False
        for trial in range(10):
            h_try = h.copy()
            h_try[block] = np.clip(h_try[block] - lr * grad_block, 0, 1)
            h_try = project_feasible(h_try, target)
            score = compute_c5(h_try)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
            lr *= 0.5
        
        if not improved:
            # Try a larger step
            lr = 0.1
            h_try = h.copy()
            h_try[block] -= lr * grad_block
            h_try = project_feasible(h_try, target)
            score = compute_c5(h_try)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
