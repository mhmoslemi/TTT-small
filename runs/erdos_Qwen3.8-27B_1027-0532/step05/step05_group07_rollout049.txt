```python
def lbfgsb_lse(h, n, target, beta, max_iter=200):
    """Minimize log-sum-exp of correlation using L-BFGS-B with penalty for constraint."""
    
    def objective(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = np.clip(beta * (corr - c_max), -500, 0)
        lse = c_max + np.log(np.sum(np.exp(shifted))) / beta
        # Add penalty for constraint violation
        penalty = 0.5 * (np.sum(x) - target)**2
        return lse + penalty
    
    def gradient(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = np.clip(beta * (corr - c_max), -500, 0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        J = corr_jacobian(x)
        grad = J.T @ weights
        # Add penalty gradient
        grad += (np.sum(x) - target) * np.ones(n)
        return grad
    
    bounds = [(0, 1)] * n
    res = minimize(objective, h, method='L-BFGS-B', jac=gradient,
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15})
    return project_feasible(res.x, target)
```

```python
def corr_jacobian(x):
    n = len(x)
    J = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        m = n - 1 - s
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo > j_hi:
            continue
        J[m, j_lo:j_hi+1] += (1.0 - x[j_lo+s:j_hi+1+s])
        lo2 = max(j_lo + s, 0)
        hi2 = min(j_hi + s, n-1)
        if lo2 <= hi2:
            J[m, lo2:hi2+1] -= x[lo2-s:hi2+1-s]
    return J

def lse_objective_and_grad(x, beta, target, penalty=1e4):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(beta * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    lse_val = c_max + np.log(sum_exp) / beta
    
    weights = exp_shifted / sum_exp
    J = corr_jacobian(x)
    grad = J.T @ weights
    
    # Penalty for constraint
    constraint_viol = np.sum(x) - target
    lse_val += penalty * constraint_viol**2
    grad += 2.0 * penalty * constraint_viol * np.ones(len(x))
    
    return lse_val, grad

def lbfgsb_optimize(h, n, target, beta, maxiter=100, penalty=1e4):
    def obj(x):
        val, _ = lse_objective_and_grad(x, beta, target, penalty)
        return val
    
    def grad(x):
        _, g = lse_objective_and_grad(x, beta, target, penalty)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def compute_B_vec(h, i, n):
    """Compute the B vector for coordinate i."""
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    for k in range(n):
        # corr[n-1+k] involves h[i] when i is in the valid range
        # For shift s=k: corr[n-1+k] = sum_j h[j]*(1-h[j+k])
        # h[i] appears when j=i (first term) or j+k=i (second term)
        # First term: j=i, need i in [0, n-1-k], contribution (1-h[i+k])
        # Second term: j+k=i, j=i-k, need i-k in [0, n-1-k], contribution -h[i-k]
        if 0 <= i <= n-1-k:
            B[n-1+k] += (1.0 - h[i+k])
        if 0 <= i-k <= n-1-k:
            B[n-1+k] -= h[i-k]
        # For shift s=-k (k>0): corr[n-1-k] = sum_j h[j]*(1-h[j-k])
        # h[i] appears when j=i (first term, need i-k in [0,n-1]) or j-k=i (second term, need j=i+k in [0, n-1])
        if k > 0:
            if 0 <= i <= n-1 and 0 <= i-k <= n-1:
                B[n-1-k] += (1.0 - h[i-k])
            if 0 <= i+k <= n-1 and 0 <= i <= n-1:
                B[n-1-k] -= h[i+k]
    return B
```

```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h + (diff / cnt) * mask
        np.clip(h, 0.0, 1.0, out=h)
    return h

def build_jacobian(x):
    """Build the Jacobian of corr = correlate(x, 1-x, 'full') w.r.t. x."""
    n = len(x)
    J = np.zeros((2*n-1, n))
    ones = 1.0
    for s in range(-(n-1), n):
        m = n - 1 - s
        j_lo = max(0, -s)
        j_hi = min(n-1, n-1-s)
        if j_lo > j_hi:
            continue
        # First term: J[m, j_lo:j_hi+1] += (1 - x[j_lo+s : j_hi+1+s])
        J[m, j_lo:j_hi+1] += ones - x[j_lo+s:j_hi+1+s]
        # Second term: J[m, lo2:hi2+1] -= x[lo2-s:hi2+1-s]
        lo2 = max(j_lo + s, 0)
        hi2 = min(j_hi + s, n - 1)
        if lo2 <= hi2:
            J[m, lo2:hi2+1] -= x[lo2-s:hi2+1-s]
    return J

def lse_f_grad(x, beta, target, penalty):
    """Log-sum-exp objective and gradient."""
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(beta * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)

    J = build_jacobian(x)
    grad = J.T @ w

    f_val += penalty * viol * viol
    grad += 2.0 * penalty * viol
    
    return f_val, grad

        return lse_f_grad(x, beta, target, penalty)[1]

                      bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_beta = 1.0
_g_target = 0.0
_g_penalty = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h + (diff / cnt) * mask
        np.clip(h, 0.0, 1.0, out=h)
    return h


def build_jacobian(x):
    n = len(x)
    J = np.zeros((2 * n - 1, n))
    for s in range(-(n - 1), n):
        m = n - 1 - s
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo > j_hi:
            continue
        J[m, j_lo:j_hi + 1] += 1.0 - x[j_lo + s:j_hi + 1 + s]
        lo2 = max(j_lo + s, 0)
        hi2 = min(j_hi + s, n - 1)
        if lo2 <= hi2:
            J[m, lo2:hi2 + 1] -= x[lo2 - s:hi2 + 1 - s]
    return J


def lse_f_grad(x, beta, target, penalty):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(beta * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    f_val = c_max + np.log(sum_exp) / beta
    w = exp_s / sum_exp
    J = build_jacobian(x)
    grad = J.T @ w
    viol = float(np.sum(x)) - target
    f_val += penalty * viol * viol
    grad += 2.0 * penalty * viol
    return f_val, grad


def _lbf_obj(x):
    return lse_f_grad(x, _g_beta, _g_target, _g_penalty)[0]


def _lbf_grad(x):
    return lse_f_grad(x, _g_beta, _g_target, _g_penalty)[1]


def lbfgsb_pass(h, n, target, beta, maxiter, penalty):
    global _g_beta, _g_target, _g_penalty
    _g_beta = beta
    _g_target = target
    _g_penalty = penalty
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbf_obj, h, method='L-BFGS-B', jac=_lbf_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def paired_cd(h, n, target, deadline, rng, max_sweeps=2000):
    dx = 2.0 / n
    h_cur = h.copy()
    best = compute_c5(h_cur)
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        J = build_jacobian(h_cur)
        idx = np.arange(n)
        rng.shuffle(idx)
        half_n = n // 2
        for p in range(half_n):
            i = int(idx[2 * p])
            j = int(idx[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-12:
                continue
            C = J[:, i] - J[:, j]
            d = h_cur[i]
            pos = C > 1e-14
            neg = C < -1e-14
            zero = ~(pos | neg)
            lo = t_min
            hi = t_max
            for _ in range(35):
                mid = (lo + hi) * 0.5
                ok = True
                if np.any(zero) and np.max(corr[zero]) > mid + 1e-13:
                    ok = False
                else:
                    if np.any(pos):
                        tups = (mid - corr[pos]) / C[pos] + d
                        if np.min(tups) < t_min - 1e-12:
                            ok = False
                    if ok and np.any(neg):
                        tdns = (mid - corr[neg]) / C[neg] + d
                        if np.max(tdns) > t_max + 1e-12:
                            ok = False
                    if ok:
                        if np.any(pos):
                            ub = min(t_max, float(np.min(tups)))
                        else:
                            ub = t_max
                        if np.any(neg):
                            lb = max(t_min, float(np.max(tdns)))
                        else:
                            lb = t_min
                        if lb > ub + 1e-12:
                            ok = False
                if ok:
                    hi = mid
                else:
                    lo = mid
            v = hi
            if np.any(pos):
                tups = (v - corr[pos]) / C[pos] + d
                ub = min(t_max, float(np.min(tups)))
            else:
                ub = t_max
            if np.any(neg):
                tdns = (v - corr[neg]) / C[neg] + d
                lb = max(t_min, float(np.max(tdns)))
            else:
                lb = t_min
            lb = max(lb, t_min)
            ub = min(ub, t_max)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = (lb + ub) * 0.5
            delta = t_star - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        sc = float(np.max(corr)) * dx
        if sc < best:
            best = sc
            stall = 0
        else:
            stall += 1
            if stall > 120:
                break
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
