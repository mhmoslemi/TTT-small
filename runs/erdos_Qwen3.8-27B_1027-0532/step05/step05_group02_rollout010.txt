    scaled = beta * corr * dx
    max_s = float(scaled.max())
    weights = np.exp(scaled - max_s)
    weights = weights / weights.sum()  # length 2n-1
    
    # s1[j] = sum_u weights[n-1+j-u] * (1-h[u])
    # s2[j] = sum_v weights[n-1+v-j] * h[v]
    
    one_minus_h = 1.0 - h
    # Build index matrix
    # s1[j] uses weights[n-1+j-u] for u=0..n-1, j=0..n-1
    # indices: n-1+j-u for j=0..n-1, u=0..n-1 → range: n-1-(n-1)=0 to n-1+(n-1)-0=2n-2 ✓
    
    J, U = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    idx1 = (n-1) + J - U  # shape (n, n)
    idx2 = (n-1) + U - J  # shape (n, n)
    
    s1 = (weights[idx1] * one_minus_h[None, :]).sum(axis=1)
    s2 = (weights[idx2] * h[None, :]).sum(axis=1)
    
    return dx * (s1 - s2)

def gradient_lse(h, beta, dx):
