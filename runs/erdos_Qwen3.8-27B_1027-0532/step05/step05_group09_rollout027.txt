```python
def binary_local_search(h_binary, target, max_iters, deadline):
    """Local search on binary sequence with sum constraint."""
    n = len(h_binary)
    h_cur = h_binary.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Try flipping a pair of bits (one 0→1 and one 1→0) to maintain sum
        if h_cur.sum() != target:
            # Repair
            diff = int(h_cur.sum() - target)
            if diff > 0:
                # Need to flip |diff| ones to zeros
                ones = np.where(h_cur == 1)[0]
                for _ in range(abs(diff)):
                    idx = rng.choice(ones)
                    h_cur[idx] = 0
                    ones = np.where(h_cur == 1)[0]
            else:
                zeros = np.where(h_cur == 0)[0]
                for _ in range(abs(diff)):
                    idx = rng.choice(zeros)
                    h_cur[idx] = 1
                    zeros = np.where(h_cur == 0)[0]
        
        # Try a move: pick a position to flip 0→1 and another to flip 1→0
        zeros = np.where(h_cur == 0)[0]
        ones = np.where(h_cur == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = rng.choice(zeros)
        j = rng.choice(ones)
        h_cur[i] = 1
        h_cur[j] = 0
        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
        else:
            h_cur[i] = 0
            h_cur[j] = 1
    
    return h_cur, cur_score
```

    """Compute sum of top M c_k values and its gradient."""
    n = len(h)
    c_k = np.correlate(h, 1-h, mode='full')[n-1:] * dx
    # Find top M
    top_idx = np.argsort(c_k)[::-1][:M]
    obj = c_k[top_idx].sum()
    # Gradient: sum of gradients of top M lags
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        # ∂c_k/∂h[j] = dx * (1-h[j+k]) for j=0..n-1-k
        #             - dx * h[j-k] for j=k..n-1
        j_pos = np.arange(0, n-k)
        grad[j_pos] += dx * (1 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= dx * h[j_neg - k]
    return obj, grad

        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def c5_grad_single(h, dx, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:

        grad[jp] += dx * (1.0 - h[jp + k])
        jn = np.arange(k, n)
        grad[jn] -= dx * h[jn - k]
    return grad

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 0.0
    if lo >= hi:
        return np.zeros_like(h)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def c5_grad_single(h, dx, k):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        jp = np.arange(n - k)
        grad[jp] += dx * (1.0 - h[jp + k])
        jn = np.arange(k, n)
        grad[jn] -= dx * h[jn - k]
    return grad

def lse_obj_grad(h, alpha, dx):
    n = len(h)
    c_k = compute_all_ck(h)
    c_max = float(np.max(c_k))
    shifted = alpha * (c_k - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    ev = np.exp(shifted)
    se = float(ev.sum())
    w = ev / se
    obj = (c_max + np.log(se) / alpha) * dx
    grad = np.zeros(n)
    for k in range(n):
        if w[k] > 1e-18:
            grad += w[k] * c5_grad_single(h, dx, k)
    return obj, grad

def topm_obj_grad(h, M, dx):
    n = len(h)
    c_k = compute_all_ck(h)
    top_idx = np.argsort(c_k)[::-1][:M]
    obj = float(c_k[top_idx].sum())
    grad = np.zeros(n)
    for k in top_idx:
        grad += c5_grad_single(h, dx, int(k))
    return obj, grad

def fista_run(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    step = 1.0 / (alpha * 0.1 + 1.0)
    lr_max = 3.0 / (alpha * 0.1 + 1.0)
    stall = 0
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y, alpha, dx)
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-14:
            break
        # Backtracking
        found = False
        for _ in range(50):
            x_new = project_feasible(y - step * grad_y, target)
            d = x_new - y
            rhs = obj_y + np.dot(grad_y, d) + 0.5 / step * np.dot(d, d)
            obj_new, _ = lse_obj_grad(x_new, alpha, dx)
            if obj_new <= rhs + 1e-12:
                found = True
                break
            step *= 0.7
        if not found:
            step = max(step, 1e-20)
            x_new = project_feasible(y - step * grad_y, target)
        step = min(step / 0.7, lr_max)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score - 1e-12:
            best_score = sc
            best_x = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
    return best_x, best_score
