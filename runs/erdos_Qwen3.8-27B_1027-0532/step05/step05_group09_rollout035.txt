            lower = h_sorted[k - 1] - 1.0
            upper = h_sorted[k - 1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)

    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.03, 1.0))
    step = lr_init

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        # Adaptive step: estimate Lipschitz constant
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink_count = 0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * 0.9:
                break
            step *= 0.7
            shrink_count += 1
            x_new = project_feasible(y_clipped - step * grad_y, target)
        if shrink_count == 0:
            step = min(step * 1.2, lr_init * 5.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def active_set_polish(h, target, deadline, top_k=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    
    for it in range(20000):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        top_indices = np.argsort(corr)[-top_k:]
        
        grad = np.zeros(n)
        for ki in top_indices:
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= h_cur[jn - ki]
        grad *= dx / len(top_indices)
        grad -= np.mean(grad)
        
        # Adaptive step with backtracking
        base_step = 0.05 / (1.0 + 0.001 * it)
        for mult in [1.0, 0.5, 0.25, 0.1]:
            step = base_step * mult
            h_new = project_feasible(h_cur - step * grad, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc_new
                base_step = min(step * 1.1, 0.1)
                break
        else:
            # No improvement, reduce step
            base_step *= 0.8
            if base_step < 1e-6:
                break
    
    return h_cur, cur_score

def diversify_search(h_best, target, deadline, rng):
    """Try diverse starting points and optimize each briefly."""
    n = len(h_best)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    
    n_restarts = 12
    time_per = (deadline - time.time()) / n_restarts - 0.5
    
    for i in range(n_restarts):
        if time.time() > deadline:
            break
        
        # Generate diverse starting point
        method = i % 4
        if method == 0:
            h_pert = h_best + rng.randn(n) * (0.03 + 0.02 * i)
        elif method == 1:
            # Binary-like
            h_pert = (h_best + rng.randn(n) * 0.1 > 0.5).astype(float)
        elif method == 2:
            # Shifted
            shift = rng.randint(1, n // 4)
            h_pert = np.roll(h_best, shift)
            h_pert = np.clip(h_pert, 0, 1)
        else:
            # Piecewise smooth noise
            noise = rng.randn(n)
            kernel = np.exp(-0.5 * np.arange(-5, 6)**2 / 4)
            kernel /= kernel.sum()
            h_pert = h_best + np.convolve(noise, np.zeros(n) + kernel, 'same') * 0.05
        
        h_pert = project_feasible(h_pert, target)
        
        # Brief FISTA
        dl = time.time() + max(time_per, 5)
        h_new, sc_new = fista_optimize(h_pert, target, 5000, 5000, dl)
        
        # Brief active set polish
        if time.time() < deadline - 5:

            if sc_new2 < sc_new:
                h_new, sc_new = h_new2, sc_new2
        
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
            time_per = (deadline - time.time()) / max(n_restarts - i - 1, 1) - 0.5
    
    return best_h, best_score

def late_negative_step(h, target, deadline, rng):

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            lower = h_sorted[k]
            upper = h_sorted[k - 1]
        else:
            lower = h_sorted[k - 1] - 1.0
            upper = h_sorted[k - 1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = -1.0
        hi = 2.0
        for _ in range(100):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        if k < n:
            lower = h_sorted[k]
            upper = h_sorted[k - 1]
        else:
            lower = h_sorted[k - 1] - 1.0
            upper = h_sorted[k - 1]
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = -1.0
        hi = 2.0
        for _ in range(100):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

            h_new, sc_new = active_set_polish(best_h, target, time.time() + 25, top_k=5)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 6: Final FISTA + active set
    if tleft() > 60:
        h_new, sc_new = fista_optimize(best_h, target, 80000, 30000, time.time() + 35)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = active_set_polish(best_h, target, time.time() + 20, top_k=7)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Final hot spot + active set
    if tleft() > 30:
        h_new, sc_new = hotspot_search(best_h, target, time.time() + 15, rng)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = active_set_polish(best_h, target, time.time() + 12, top_k=5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
