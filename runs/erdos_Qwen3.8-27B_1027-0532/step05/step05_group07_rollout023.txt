```python
import cvxpy as cp

def active_set_lp(h0, n, target, active_lags, max_iter=50, tol=1e-12):
    """Solve the active-set LP to find a descent direction"""
    dx = 2.0 / n
    nl1 = 2 * n - 1
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Compute gradients for active lags
    # B[i, m] = d(corr[m])/d(h[i]) at h0
    # B[i, m] = (1-h0[i-k]) - h0[i+k] where k = m-(n-1)
    grad_matrix = np.zeros((n, len(active_lags)))
    for li, m in enumerate(active_lags):
        k = m - (n - 1)
        for i in range(n):
            val = 0.0
            idx1 = i - k
            if 0 <= idx1 < n:
                val += (1.0 - h0[idx1])
            idx2 = i + k
            if 0 <= idx2 < n:
                val -= h0[idx2]
            grad_matrix[i, li] = val * dx
    
    # LP: minimize t
    # subject to: corr0[m]*dx + grad_matrix[:,li] · h ≤ t  for each active lag li
    #             0 ≤ h ≤ 1
    #             sum(h) = target
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
    for li, m in enumerate(active_lags):
        constraints.append(corr0[m] * dx + grad_matrix[:, li] @ h_var <= t_var)
    
    objective = cp.Minimize(t_var)
```

```python
        else:

            lo2 = max(0, -k)

        h_var = cp.Variable(n)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def compute_grad_columns(h_cur, n, lag_indices):
    dx = 2.0 / n
    nl1 = 2 * n - 1
    G = np.zeros((n, len(lag_indices)))
    for li, m in enumerate(lag_indices):
        k = m - (n - 1)
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            G[idx, li] += (1.0 - h_cur[idx - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            G[idx, li] -= h_cur[idx + k]
    G *= dx
    return G


def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=40, max_bin_iter=30):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep = 0

    while time.time() < deadline:
        sweep += 1
        if sweep % recompute_interval == 0:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = c
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            lo = t_min
            hi = t_max
            for _ in range(max_bin_iter):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((float(np.max(A + C * mid)) - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((float(np.max(A + C * mid)) - A[neg]) / C[neg])))
                if lb <= ub + 1e-12:
                    hi = mid
                else:
                    lo = mid

            t_star = (lo + hi) * 0.5
            t_star = max(t_min, min(t_max, t_star))
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-15:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-16:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 80 and sweep > 50:
            scale = 0.015 + 0.01 * min(stall_count // 80, 5)
            h_cur = project_feasible(h_cur + rng.randn(n) * scale, target)
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            best_score = compute_c5(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq_fun(x):
    return np.sum(x) - _g_target


def cons_jac_fun(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=2000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_jac_fun}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def interpolate_to_n(h_src, n_dst, target_dst):

            best_score = sc_final
            best_h = h_final

    # Final SLSQP if time
    if tleft() > 30:
        n_f = len(best_h)
        h_f = slsqp_polish(best_h, n_f, n_f / 2.0, [5000, 20000, 50000], 1000, 1e-18)
        sc_f = compute_c5(h_f)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    if final_score > best_score:
        h_out = project_feasible(best_h, len(best_h) / 2.0)
        final_score = best_score
    return h_out, final_score, len(h_out)
```
