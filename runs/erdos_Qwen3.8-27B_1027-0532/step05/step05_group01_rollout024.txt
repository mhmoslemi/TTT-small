        if c_max * dx < best_c5 - 1e-13:
            best_c5 = c_max * dx
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
            if stuck_count > 10000:
                break
        
        # Find top-k active shifts
        top_indices = np.argsort(corr)[-top_k_shifts:][::-1]
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for ti in top_indices:
            s = int(ti) - mid
            # dcorr[ti]/dh[m] = (1-h[m+s])*I(0<=m+s<n) - h[m-s]*I(0<=m-s<n)
            ms_plus = idxs + s
            ms_minus = idxs - s
            valid_plus = (ms_plus >= 0) & (ms_plus < n)
            valid_minus = (ms_minus >= 0) & (ms_minus < n)
            grad += (1.0 - h[ms_plus[valid_plus]]) * valid_plus - h[ms_minus[valid_minus]] * valid_minus
        
        # Remove mean to preserve sum constraint
        grad -= np.mean(grad)
        
        # Select candidates
        # Decrease h where grad > 0 (high gradient → decreasing h reduces objective)
        # Increase h where grad < 0
        pos_mask = grad > 0
        neg_mask = grad < 0
        
        if not pos_mask.any() or not neg_mask.any():
            stuck_count += 100
            continue
        
        # Top candidates by gradient magnitude
        pos_idx = np.where(pos_mask)[0]
        neg_idx = np.where(neg_mask)[0]
        pos_idx = pos_idx[np.argsort(grad[pos_idx])[::-1]][:n_cand]
        neg_idx = neg_idx[np.argsort(grad[neg_idx])][:n_cand]
        
        # Try pairs
        best_improve = 0.0
        best_i, best_j, best_delta = -1, -1, 0.0
        
        for i in pos_idx:
            hi = h[i]
            if hi < 1e-10:
                continue
            # Precompute effect of decreasing h[i]
            ip = idxs + (i - mid + mid)  # this is just s + i where s = ti - mid... 
            # Actually, let me compute the effect vector for coordinate i
            # effect_decrease_i[s] = -dcorr[s]/dh[i] = -(1-h[i+s])*I(0<=i+s<n) + h[i-s]*I(0<=i-s<n)
            
            for j in neg_idx:
                hj = h[j]
                if hj > 1.0 - 1e-10:
                    continue
                
                # Compute d[s] = effect of (+delta to j, -delta to i)
                # d[s] = dcorr[s]/dh[j] - dcorr[s]/dh[i]
                # dcorr[s]/dh[m] = (1-h[m+s])*I(0<=m+s<n) - h[m-s]*I(0<=m-s<n)
                
                # For coordinate i:
                i_s = i + s_vals  # i + s for all s
                i_minus_s = i - s_vals
                valid_ip = (i_s >= 0) & (i_s < n)
                valid_im = (i_minus_s >= 0) & (i_minus_s < n)
                dci = np.zeros(n_corr)
                dci[valid_ip] += (1.0 - h[i_s[valid_ip]])
                dci[valid_im] -= h[i_minus_s[valid_im]]
                
                # For coordinate j:
                j_s = j + s_vals
                j_minus_s = j - s_vals
                valid_jp = (j_s >= 0) & (j_s < n)
                valid_jm = (j_minus_s >= 0) & (j_minus_s < n)
                dcj = np.zeros(n_corr)
                dcj[valid_jp] += (1.0 - h[j_s[valid_jp]])
                dcj[valid_jm] -= h[j_minus_s[valid_jm]]
                
                d = dcj - dci
                
                # Line search: minimize max(corr + delta * d)
                delta_max = min(hi, 1.0 - hj)
                if delta_max < 1e-12:
                    continue
                
                # Bisection
                a, b = 0.0, delta_max
                for _ in range(12):
                    mid_d = 0.5 * (a + b)
                    nc = corr + mid_d * d
                    fm = float(np.max(nc))
                    active = nc >= fm - 1e-10
                    fp = float(np.max(d[active])) if active.any() else 0.0
                    if fp > 1e-12:
                        a = mid_d
                    else:
                        b = mid_d
                
                delta = 0.5 * (a + b)
                if delta < 1e-12:
                    continue
                
                new_max = float(np.max(corr + delta * d))
                improve = c_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i, best_j, best_delta = i, j, delta
        
        if best_i >= 0 and best_improve > 1e-10:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            # Project to maintain sum
            h = project_feasible(h, target)
            corr = np.correlate(h, 1.0 - h, mode='full')
        else:
            stuck_count += 10
            # Occasionally try a random perturbation to escape
            if stuck_count % 5000 == 0:
                rng_i = np.random.randint(n)
                rng_j = np.random.randint(n)
                if h[rng_i] > 0.05 and h[rng_j] < 0.95:
                    delta = 0.05 * min(h[rng_i], 1.0 - h[rng_j])
                    h[rng_i] -= delta
                    h[rng_j] += delta
                    h = project_feasible(h, target)
                    corr = np.correlate(h, 1.0 - h, mode='full')
    
    return best_h, best_c5

For the binary local search:
- State: binary vector h ∈ {0,1}^n with Σh = n/2
- Move: pick i with h[i]=1 and j with h[j]=0, swap (h[i]→0, h[j]→1)
- Cost: max_k corr[k] (computed from the binary vector)
- Acceptance: greedy (only accept if cost decreases) or Metropolis (accept with probability exp(-Δ/T))

The cost evaluation is O(n²) = 160K per move. At 100M ops/sec, that's 1.6ms per move. In 100 seconds, about 60,000 moves. That's a lot!

But I can optimize: when flipping one bit, the correlation changes only at O(n) shifts (the ones affected by that bit). So I can update the correlation in O(n) instead of O(n²).

Wait, flipping h[i] from 1 to 0:
- corr[k] changes for all k where the term involving h[i] is non-zero
- h[i] appears in corr[k] as: h[i]*(1-h[i+k]) [when i is valid] and h[i-k]*(1-h[i]) [when i-k is valid]
- So corr[k] changes for k = 0 (the h[i]*(1-h[i]) term, which is 0 for binary) and for k = -s where i-s is valid (the h[i-s]*(1-h[i]) term)

Actually, let me just compute it directly. For a binary h, corr[k] = #{i : h[i]=1, h[i+s]=0} where s = k - mid.

When h[i] changes from 1 to 0:
- For each s where i+s is valid: the pair (i, i+s) contributes 1 to corr if h[i+s]=0, else 0. After the flip, it contributes 0. So corr[s+mid] decreases by (1-h[i+s]) = h[i+s] (since h[i+s] is 0 or 1, 1-h[i+s] is 1 if h[i+s]=0).
  Wait: before flip, contribution = 1*(1-h[i+s]) = 1-h[i+s]. After flip: 0. Change: -(1-h[i+s]).
  For binary: if h[i+s]=0, change = -1. If h[i+s]=1, change = 0.
  
- For each s where i-s is valid: the pair (i-s, i) contributes h[i-s]*(1-h[i]). Before flip: h[i-s]*0 = 0. After flip: h[i-s]*1 = h[i-s]. Change: +h[i-s].
  For binary: if h[i-s]=1, change = +1. If h[i-s]=0, change = 0.

So the total change is at most 2n in magnitude, affecting at most 2n shifts.

For a pair of flips (i: 1→0, j: 0→1), the total change affects at most 4n shifts, and can be computed in O(n).

This makes the binary local search very fast: O(n) per move instead of O(n²).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

_global_n = 0
_global_target = 0.0

def _obj_fn(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _con_fun(h):
    return np.sum(h) - _global_target

def _con_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    h_cur = h_start.copy()
    best_c = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_obj_fn, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def binary_local_search(h_cont, n, target, time_limit, max_iter=200000, n_try=100):
    """Convert to binary and do local search."""
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    mid = n - 1
    
    # Round to binary
    h_bin = (h_cont > 0.5).astype(float)
    # Fix sum
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    n_ones = len(ones)
    target_ones = n // 2
    if n_ones > target_ones:
        # Flip some ones to zeros
        excess = n_ones - target_ones
        flip_to_0 = ones[np.random.choice(n_ones, excess, replace=False)]
        h_bin[flip_to_0] = 0
    elif n_ones < target_ones:
        deficit = target_ones - n_ones
        flip_to_1 = zeros[np.random.choice(len(zeros), deficit, replace=False)]
        h_bin[flip_to_1] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    s_vals = np.arange(-(n-1), n)
    
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        c_max = float(np.max(corr))
        if c_max * dx < best_c5 - 1e-13:
            best_c5 = c_max * dx
            best_h = h_bin.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 5000:
                break
        
        # Find best pair flip
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        # Sample candidates
        n_sample = min(len(ones), n_try)
        n_sample_z = min(len(zeros), n_try)
        sample_ones = np.random.choice(ones, n_sample, replace=False)
        sample_zeros = np.random.choice(zeros, n_sample_z, replace=False)
        
        best_imp = 0.0
        best_i = best_j = -1
        
        for i in sample_ones:
            # Effect of flipping i: 1 -> 0
            ip = i + s_vals
            im = i - s_vals
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            # Change in corr: -(1-h[i+s]) for valid i+s, +h[i-s] for valid i-s
            # For binary: -(1-h_bin[i+s]) = -1 if h_bin[i+s]=0, 0 if h_bin[i+s]=1
            # +h_bin[i-s] = 1 if h_bin[i-s]=1, 0 if h_bin[i-s]=0
            eff_i = np.zeros(n_corr)
            eff_i[vp] -= (1.0 - h_bin[ip[vp]])
            eff_i[vm] += h_bin[im[vm]]
            
            for j in sample_zeros:
                # Effect of flipping j: 0 -> 1
                jp = j + s_vals
                jm = j - s_vals
                vjp = (jp >= 0) & (jp < n)
                vjm = (jm >= 0) & (jm < n)
                eff_j = np.zeros(n_corr)
                eff_j[vjp] += (1.0 - h_bin[jp[vjp]])
                eff_j[vjm] -= h_bin[jm[vjm]]
                
                d = eff_i + eff_j
                new_max = float(np.max(corr + d))
                imp = c_max - new_max
                if imp > best_imp:
                    best_imp = imp
                    best_i, best_j = i, j
        
        if best_i >= 0 and best_imp > 1e-12:
            h_bin[best_i] = 0
            h_bin[best_j] = 1
            # Update corr incrementally
            # (Just recompute for simplicity - O(n^2) but n=400 is fine)
            corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        else:
            stuck += 10
    
    return best_h, best_c5

def subgradient_method(h_init, n, target, time_limit, mu=0.8, base_lr=0.05, n_top=5):

            vm = (mm >= 0) & (mm < n)
            grad += w * ((1.0 - h[mp[vp]]) * vp.astype(float) - h[mm[vm]] * vm.astype(float))
        
        grad *= dx
        grad -= np.mean(grad)
        
        v = mu * v + grad
        # Adaptive step size
        lr = base_lr / (1.0 + 0.01 * step)
        h_new = project_feasible(h - lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < compute_c5(h) + 0.001:  # Allow small increases for exploration
            h = h_new
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 2.0, n_dst)  # Wait, should be (0,1) normalized?
    # Actually, the grid is [0, 2] with n points
    # x_old = np.linspace(0, 2, n_src) approximately (or 2/n * arange(n))
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return h_out

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    # Initialize from parent
    h_init = np.array(initial_h_values, dtype=float).copy()
    n = len(h_init)
    target = n / 2.0
    
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP
    if tleft() > 100:
        h_p = slsqp_round(best_h, n, target, [100, 500, 2000, 10000], 1000, 1e-14)
        s = compute_c5(h_p)
        if s < best_score:
            best_score = s
            best_h = h_p
    
    # Phase 2: Fast CD (main workhorse)
    if tleft() > 200:
        cd_time = min(400, tleft() * 0.4)
        h_cd, c5_cd = fast_cd(best_h, n, target, cd_time, n_cand=20, top_k=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    
    # Phase 3: SLSQP polish
    if tleft() > 80:
        h_p = slsqp_round(best_h, n, target, [2000, 10000, 50000], 1000, 1e-15)
        s = compute_c5(h_p)
        if s < best_score:
            best_score = s
            best_h = h_p
    
    # Phase 4: Binary local search
    if tleft() > 150:
        h_bin, c5_bin = binary_local_search(best_h, n, target, min(100, tleft() * 0.3))
        if c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin
            # Unfreeze: add small continuous perturbations around binary
            h_unfree = h_bin + rng.randn(n) * 0.02
            h_unfree = project_feasible(h_unfree, target)
            if tleft() > 60:
                h_p2 = slsqp_round(h_unfree, n, target, [500, 2000, 10000], 800, 1e-14)
                s2 = compute_c5(h_p2)
                if s2 < best_score:
                    best_score = s2
                    best_h = h_p2
    
    # Phase 5: More CD with different parameters
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.4)
        h_cd2, c5_cd2 = fast_cd(best_h, n, target, cd_time, n_cand=30, top_k=5)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
    
    # Phase 6: Subgradient with momentum
    if tleft() > 80:
        h_sg, c5_sg = subgradient_method(best_h, n, target, min(60, tleft() * 0.3))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    
    # Phase 7: SLSQP polish
    if tleft() > 60:
        h_p3 = slsqp_round(best_h, n, target, [5000, 20000], 800, 1e-15)
        s = compute_c5(h_p3)
