import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_gradients_for_shifts(h, shift_indices):
    """Compute gradients of f(m) for given shift indices.
    grad_m[i] = 1 - h[i+m] - h[i-m] (with boundary terms = 0)
    where m = shift_index - (n-1)
    """
    n = len(h)
    mid = n - 1
    grads = np.zeros((len(shift_indices), n))
    idxs = np.arange(n)
    for si, s_idx in enumerate(shift_indices):
        m = int(s_idx) - mid
        # grad[i] = 1 - h[i+m] - h[i-m]
        g = np.ones(n)
        vp = (idxs + m >= 0) & (idxs + m < n)
        vm = (idxs - m >= 0) & (idxs - m < n)
        g[vp] -= h[idxs[vp] + m]
        g[vm] -= h[idxs[vm] - m]
        grads[si] = g
    return grads

def lp_cutting_plane(h, n, target, time_limit, max_iters=500, K=20, lambda_max=1.0):
    """LP-based cutting plane method for the minimax problem."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h_cur)
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_cur.copy()
        
        # Get top K shifts
        top_k = np.argsort(corr)[-K:]
        top_k = top_k[np.argsort(corr[top_k])]  # sort by value
        
        f_vals = corr[top_k]
        grads = compute_gradients_for_shifts(h_cur, top_k)  # K x n
        
        # Set up LP: minimize t
        # subject to: grads[i] @ h <= t + c_i for i in range(K)
        #             sum(h) = target
        #             0 <= h <= 1
        # where c_i = -f_vals[i] + grads[i] @ h_cur
        
        c_vec = -f_vals + grads @ h_cur
        
        # cvxpy LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        for i in range(K):
            constraints.append(grads[i] @ h_var <= t_var + c_vec[i])
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS)
                except Exception:
                    break
        
        if h_var.value is None:
            break
        
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        
        # Line search: try fractional steps
        best_lambda = 0.0
        best_trial_c5 = c5_now
        
        for lam in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:

            h_try = project_feasible(h_try, target)

            if trial_c5 < best_trial_c5:

                best_lambda = lam

            # Use the average gradient of top shifts
            avg_grad = grads.mean(axis=0)
            avg_grad -= np.mean(avg_grad)
            for lr in [0.01, 0.005, 0.002]:

                trial_c5 = compute_c5(h_try)
                if trial_c5 < c5_now - 1e-14:

def pair_cd_v2(h, n, target, time_limit, max_sweeps=5000, n_cand_start=20, n_cand_max=80):
    """Improved pair coordinate descent with adaptive candidate count."""
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n-1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        # Adaptive n_cand: start small, grow when stuck
        frac = min(1.0, sweep / 500.0)
        n_cand = int(n_cand_start + (n_cand_max - n_cand_start) * frac)
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 300:
            break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Compute g vector: change in corr per unit delta
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                
                # Binary search for optimal delta
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    def smooth_obj(h, alpha):
        corr = compute_corr(h)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(h):
        return np.sum(h) - target
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def nesterov_smoothed(h_init, n, target, time_limit, alpha_init=10, alpha_final=500, max_steps=5000):
    """Nesterov accelerated gradient on smoothed objective."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    
    for k in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        # Adaptive alpha
        progress = min(1.0, k / max_steps)
        alpha = alpha_init * (alpha_final / alpha_init) ** progress
        
        corr = compute_corr(y)
        c_max = np.max(corr)
        
        # Softmax weights
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        w = np.exp(shifted)
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.zeros_like(corr)
            w[np.argmax(corr)] = 1.0
        else:
            w /= w_sum
        
        # Gradient of smoothed objective
        grad = np.zeros(n)
        top_k = np.argsort(w)[-20:]  # Only use top 20 for efficiency
        w_top = w[top_k]
        w_top = w_top / w_top.sum()
        
        for wi, s_idx in zip(w_top, top_k):
            m = int(s_idx) - mid
            vp = (idxs + m >= 0) & (idxs + m < n)
            vm = (idxs - m >= 0) & (idxs - m < n)
            grad[vp] += wi * (1.0 - h[idxs[vp] + m])
            grad[vm] -= wi * h[idxs[vm] - m]
        
        grad *= dx
        grad -= np.mean(grad)
        
        # Nesterov step
        t_k = 1 + np.sqrt(1 + 4 * k * k) / 2
        lr = 2.0 / (t_k + 1) * 100  # Tune this
        
        # Actually for constrained problem, use simpler approach
        # h_new = proj(h - step * grad)
        # y_new = h_new + (t_k - 1) / t_{k+1} * (h_new - h)
        
        h_new = project_feasible(h - lr * grad, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        t_next = 1 + np.sqrt(1 + 4 * (k+1) * (k+1)) / 2
        beta = (t_k - 1) / t_next
        y = h_new + beta * (h_new - h)
        y = project_feasible(y, target)
        h = h_new
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def qp_cutting_plane(h, n, target, time_limit, max_iters=500, K=20, mu_init=50.0, mu_min=5.0):
    """QP-based cutting plane method."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    idxs = np.arange(n)
    mid = n - 1
    
    # Pre-compute shift values for gradient computation
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h_cur)
        c5_now = float(np.max(corr)) * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_cur.copy()
        
        # Adaptive mu: decrease over time for larger steps
        mu = max(mu_min, mu_init * (1.0 - it / max_iters))
        
        # Get active shifts: top K or within 5% of max
        c_max = np.max(corr)
        threshold = c_max - 0.05 * c_max  # within 5%
        active = np.where(corr >= threshold)[0]
        if len(active) > K:
            active = np.argsort(corr)[-K:]
        
        # Compute gradients for active shifts
        K_act = len(active)
        A_rows = np.zeros((K_act, n))
        c_vec = np.zeros(K_act)
        
        for ai, a_idx in enumerate(active):
            m = int(a_idx) - mid
            g = np.ones(n)
            vp = (idxs + m >= 0) & (idxs + m < n)
            vm = (idxs - m >= 0) & (idxs - m < n)
            g[vp] -= h_cur[idxs[vp] + m]
            g[vm] -= h_cur[idxs[vm] - m]
            A_rows[ai] = g
            # c_k = -f(k) + g · h_cur where f(k) = corr[k] (the overlap at shift k)
            # Actually f(k) = corr[k] (the raw correlation value, not multiplied by dx)
            c_vec[ai] = -corr[a_idx] + g @ h_cur
        
        # Build QP using cvxopt
        # Variables: x = [h (n), t (1)]
        # min (1/2) x' P x + q' x
        # s.t. G x <= h_g, A_eq x = b
        
        N = n + 1
        # P = block_diag(mu * I_n, 0)
        P = cvxopt.matrix(np.zeros((N, N)))
        for i in range(n):
            P[i, i] = mu
        # P is already symmetric
        P = P + P.T  # ensure symmetry (it's already symmetric)
        
        # q = [-mu * h_cur; 1]
        q = cvxopt.matrix(np.zeros(N))
        q[:n] = -mu * h_cur
        q[n] = 1.0  # coefficient of t
        
        # Inequality constraints: G x <= h_g
        # 1) For each active shift: g_i' h - t <= c_i  →  [g_i, -1] [h; t] <= c_i
        # 2) -h <= 0
        # 3) h <= 1
        
        n_ineq = K_act + 2 * n
        G = np.zeros((n_ineq, N))
        h_g = np.zeros(n_ineq)
        
        for ai in range(K_act):
            G[ai, :n] = A_rows[ai]
            G[ai, n] = -1.0
            h_g[ai] = c_vec[ai]
        
        # -h <= 0
        G[K_act:K_act+n, :n] = -np.eye(n)
        h_g[K_act:K_act+n] = 0.0
        
        # h <= 1
        G[K_act+n:K_act+2*n, :n] = np.eye(n)
        h_g[K_act+n:K_act+2*n] = 1.0
        
        G = cvxopt.matrix(G)
        h_g = cvxopt.matrix(h_g)
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, N))
        A_eq[0, :n] = 1.0
        A_eq = cvxopt.matrix(A_eq)
        b_eq = cvxopt.matrix(target)
        
        # Solve
        try:
            sol = cvxopt.solvers.qp(P, q, G, h_g, A_eq, b_eq, verbose=0)
        except Exception:
            break
        
        if sol['status'] != 'optimal':
            # Try cvxpy as fallback
            try:
                h_var = cp.Variable(n)
                t_var = cp.Variable()
                constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
                for ai in range(K_act):
                    constraints.append(cp.sum(A_rows[ai] * h_var) - t_var <= c_vec[ai])
                obj = cp.Minimize(t_var + (mu/2) * cp.sum_squares(h_var - h_cur))
                prob = cp.Problem(obj, constraints)
                prob.solve(solver=cp.CVCXPY if hasattr(cp, 'CVCXPY') else cp.CLARABEL)
                if h_var.value is not None:
                    h_lp = project_feasible(h_var.value, target)
                else:
                    break
            except Exception:
                break
        else:
            h_lp = sol['x'][:n]
            h_lp = project_feasible(np.array(h_lp), target)
        
        # Check improvement
        c5_new = compute_c5(h_lp)
        if c5_new < c5_now - 1e-14:
            h_cur = h_lp
        else:
            # Try fractional step
            improved = False
            for lam in [0.5, 0.3, 0.1, 0.05]:
                h_try = project_feasible((1-lam)*h_cur + lam*h_lp, target)
                if compute_c5(h_try) < c5_now - 1e-14:
                    h_cur = h_try
                    improved = True
                    break
            if not improved and it > 30:
                break
    
    return best_h, best_c5

def pair_cd(h_init, n, target, time_limit, max_sweeps=5000, n_cand_start=15, n_cand_max=80):

    h = h_init.copy()

    idxs = np.arange(n)
    consecutive_no_improve = 0

        if time.time() - start > time_limit:
            break

        frac = min(1.0, sweep / 400.0)
        n_cand = int(n_cand_start + (n_cand_max - n_cand_start) * frac)

        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:

        else:

        if consecutive_no_improve > 250:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = float(np.max(corr + delta_star * g)) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    def smooth_obj(h, alpha):
        corr = compute_corr(h)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def nesterov_smoothed(h_init, n, target, time_limit, alpha_init=5, alpha_final=2000, max_steps=8000):

        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

        grad *= dx
        grad -= np.mean(grad)

        t_k = 1.0 + np.sqrt(1.0 + 4.0 * k * k) / 2.0
        lr = 2.0 / (t_k + 1.0) * 50.0

        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()

        t_next = 1.0 + np.sqrt(1.0 + 4.0 * (k + 1) * (k + 1)) / 2.0
        beta = (t_k - 1.0) / t_next
        y_new = project_feasible(h_try + beta * (h_try - h), target)
        h = h_try
        y = y_new

    return best_h, best_c5


def local_perturbation(h_init, n, target, max_tries, time_limit, seed_val=777):
    start = time.time()
    rng_local = np.random.RandomState(seed_val)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8, 13, 21]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.03, 0.05, 0.08, 0.12, 0.18]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 18.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 200:
        h_qp, c5_qp = qp_cutting_plane(best_h, n, target, time_limit=min(130, tleft() * 0.2),
                                        max_iters=500, K=20, mu_start=80.0, mu_end=10.0)
        if c5_qp < best_score:
            best_score = c5_qp
            best_h = h_qp

    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target, [200, 1000, 5000, 20000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                               max_sweeps=5000, n_cand_start=15, n_cand_max=80)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 150:
        h_qp, c5_qp = qp_cutting_plane(best_h, n, target, time_limit=min(100, tleft() * 0.2),
                                        max_iters=400, K=25, mu_start=50.0, mu_end=5.0)
        if c5_qp < best_score:
            best_score = c5_qp
            best_h = h_qp

    if tleft() > 100:
        nev_time = min(70, tleft() * 0.25)
        h_nev, c5_nev = nesterov_smoothed(best_h, n, target, time_limit=nev_time,
                                           alpha_init=5, alpha_final=2000, max_steps=5000)
        if c5_nev < best_score:
            best_score = c5_nev
            best_h = h_nev

    if tleft() > 150:
        cd_time = min(200, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                               max_sweeps=5000, n_cand_start=25, n_cand_max=100)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [2000, 8000, 30000, 80000], 600, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    rng = np.random.RandomState(seed)
    for noise_scale in [0.01, 0.02, 0.04, 0.07, 0.10, 0.15]:
        if tleft() < 70:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        if tleft() > 50:
            h_qp, c5_qp = qp_cutting_plane(h_pert, n, target, time_limit=min(25, tleft() * 0.3),
                                            max_iters=150, K=15, mu_start=60.0, mu_end=10.0)
            if c5_qp < best_score:
                best_score = c5_qp
                best_h = h_qp
        if tleft() > 40:
            h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=min(25, tleft() * 0.3),
                                   max_sweeps=1500, n_cand_start=15, n_cand_max=50)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    if tleft() > 40:
        h_p, c5_p = local_perturbation(best_h, n, target, max_tries=1000,
                                        time_limit=min(30, tleft() * 0.5), seed_val=12345)
        if c5_p < best_score:
            best_score = c5_p
            best_h = h_p

    if tleft() > 25:
        h_final = slsqp_round(best_h, n, target, [5000, 20000, 60000], 400, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
