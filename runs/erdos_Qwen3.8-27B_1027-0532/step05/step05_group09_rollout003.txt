import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    # Project h onto {0<=x<=1, sum(x)=target}
    h = h.astype(float).copy()
    n = len(h)
    # Use bisection on the threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient
    # d/dh_i [sum_k w_k * sum_j h[j+k](1-h[j])]
    # = sum_k w_k * [sum_{j: j+k=i} (1-h[j]) - sum_{j: j=i-k} h[j+k]]
    # For corr[k] = sum_{j=0}^{n-k-1} h[j+k]*(1-h[j])
    # d/dh_i corr[k] = (1-h[i-k]) if i >= k (from h[j+k] term with j=i-k)
    #                  - h[i+k] if i+k < n (from (1-h[j]) term with j=i)
    
    # Using FFT for efficiency
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_k = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + ((t_k - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking line search
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 2.0)
        x_prev = x
        x = x_new
        t_k = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def smart_coord_descent(h, target, deadline, max_sweeps=100):
    """Coordinate descent with golden-section search on each coordinate"""
    n = len(h)

def active_set_refine(h, target, deadline, max_rounds=500):

    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)

            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        threshold = corr.max() - 1e-6
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(corr)[::-1][:3]
        grad = np.zeros(n)
        for ki in active:
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= h_cur[jn - ki]
        grad *= dx / max(len(active), 1)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad_norm = grad / gnorm
        best_sc = cur_score
        best_h = h_cur.copy()
