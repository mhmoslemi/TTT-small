import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-16, 'maxls': 100, 'maxfun': 100000})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def block_cd(h, target, block_size=5, n_rounds=50, deadline=None):
    """Block coordinate descent: optimize blocks of consecutive variables."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Randomize block order each round
        start_offsets = np.arange(0, n, block_size)
        np.random.shuffle(start_offsets)
        
        for offset in start_offsets:
            if deadline is not None and time.time() > deadline:
                break
            # Define the block
            block_start = int(offset)
            block_end = min(block_start + block_size, n)
            block_idx = np.arange(block_start, block_end)
            
            # Fix other variables, optimize block
            # The objective as a function of the block is: max_k corr_k
            # where corr_k depends on the block variables
            # Use a local optimization on the block
            
            h_fixed = h_cur.copy()
            h_block = h_cur[block_idx].copy()
            
            def block_obj(v, h_fixed=h_fixed, block_idx=block_idx):
                h_test = h_fixed.copy()
                h_test[block_idx] = v
                # Adjust sum
                s = np.sum(h_test)
                if abs(s - target) > 1e-10:
                    # Simple projection: adjust the block to fix the sum
                    adj = target - s
                    h_test[block_idx] = np.clip(v + adj / len(v), 0, 1)
                corr = np.correlate(h_test, 1.0 - h_test, mode='full')
                return float(np.max(corr))
            
            # Use Nelder-Mead or a simple line search
            best_block = h_block.copy()
            best_val = block_obj(h_block)
            
            # Try small perturbations of the block
            for attempt in range(20):
                if deadline is not None and time.time() > deadline:
                    break
                pert = np.random.randn(len(h_block)) * 0.05
                candidate = np.clip(h_block + pert, 0, 1)
                val = block_obj(candidate)
                if val < best_val - 1e-12:
                    best_val = val
                    best_block = candidate
            
            if np.any(np.abs(best_block - h_cur[block_idx]) > 1e-10):
                h_cur[block_idx] = best_block
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-13:
                    cur_score = sc
                    improved = True
        
        if not improved:
            break
    
    return h_cur, cur_score

def fista_optimize(h, target, alpha, mu, maxiter, deadline):
    """FISTA (Fast Iterative Shrinkage-Thresholding) on the LSE objective."""
    n = len(h)
    dx = 2.0 / n
    h_y = h.copy()
    h_z = h.copy()
    t_k = 1.0
    cur_score = compute_c5(h_z)
    best_h = h_z.copy()
    best_score = cur_score
    
    # Step size (use a conservative one)
    # Lipschitz constant of gradient: roughly 2*n (worst case)
    # Use a backtracking line search instead
    L = 2.0 * n * dx  # conservative initial estimate
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        
        _, grad = lse_obj_and_grad(h_z, alpha, target, mu)
        
        # Gradient step
        h_new = h_z - grad / L
        
        # Project onto [0,1] with sum constraint
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Backtracking if needed
        f_new = lse_obj_and_grad(h_new, alpha, target, mu)[0]
        f_z = lse_obj_and_grad(h_z, alpha, target, mu)[0]
        backtracks = 0
        while (f_new > f_z + 0.5 * L * np.sum((h_new - h_z)**2) + 1e-10) and backtracks < 10:
            L *= 2.0
            h_new = h_z - grad / L
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = project_feasible(h_new, target)
            f_new = lse_obj_and_grad(h_new, alpha, target, mu)[0]
            backtracks += 1
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        h_y = h_new + ((t_k - 1.0) / t_new) * (h_new - h_z)
        h_z = h_new
        t_k = t_new
        
        # Track best
        sc = compute_c5(h_z)
        if sc < best_score:
            best_score = sc
            best_h = h_z.copy()
    
    return best_h, best_score

def simulated_annealing(h, target, n_iter=5000, deadline=None):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])

            n_flip = max(3, int(n_current * param))
            idx = rng.choice(n_current, size=n_flip, replace=False)
            h_pert[idx] = rng.uniform(0, 1, n_flip)
        elif strat == 'binary':
            h_pert = (best_h > 0.5).astype(float)
            h_pert += rng.randn(n_current) * 0.05
        elif strat == 'smooth':
            freq = np.fft.rfftfreq(n_current)
            n_freq = n_current // 2 + 1
            spec = np.zeros(n_freq)
            band = int(0.01 * n_freq)
            spec[1:1 + band] = rng.randn(band) * param
            pert = np.fft.irfft(spec, n_current)
            h_pert += pert
        h_pert = project_feasible(h_pert, target_current)
        h_new = lbfgs_optimize(h_pert, target_current, 200, mu, 300, deadline)
        h_new = lbfgs_optimize(h_new, target_current, 5000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        h_fine = upsample_and_optimize(best_h, n, target, n2, target2,
                                       deadline - 100, mu)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target_current = target2

            # Further optimize at higher resolution
            if tleft() > 200:
                h_eq, sc_eq = top_lag_equalize(best_h, target_current, n_top=6,
                                               n_steps=300, deadline=deadline)
                if sc_eq < best_score:
                    best_score = sc_eq
                    best_h = h_eq
                h_new = lbfgs_optimize(best_h, target_current, 20000, mu, 500, deadline)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

                if tleft() > 100:
                    h_bc, sc_bc = block_cd(best_h, target_current, block_size=5,
                                           n_rounds=40, deadline=deadline)
                    if sc_bc < best_score:
                        best_score = sc_bc
                        best_h = h_bc

                if tleft() > 80:
                    h_sa, sc_sa = simulated_annealing(best_h, target_current,
                                                      n_iter=5000, deadline=deadline, seed=555)
                    if sc_sa < best_score:
                        best_score = sc_sa
                        best_h = h_sa
                    h_new = lbfgs_optimize(best_h, target_current, 50000, mu, 400, deadline)
                    sc = compute_c5(h_new)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_new

    # Phase 8: Final equalization + L-BFGS
    if tleft() > 80:
        h_eq, sc_eq = top_lag_equalize(best_h, target_current, n_top=10,
                                       n_steps=300, deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target_current, 100000, mu, 600, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Block CD final polish
    if tleft() > 60:
        h_bc, sc_bc = block_cd(best_h, target_current, block_size=3,
                               n_rounds=30, deadline=deadline)
        if sc_bc < best_score:
            best_score = sc_bc
            best_h = h_bc

    # Phase 10: SLSQP final polish
    if tleft() > 60:
        h_slq = slsqp_polish(best_h, target_current, 20000, 5000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 11: Very fine L-BFGS
    if tleft() > 40:
        h_new = lbfgs_optimize(best_h, target_current, 500000, mu, 800, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final projection and evaluation
    h_final = project_feasible(best_h, target_current)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
