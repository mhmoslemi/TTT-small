import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def subgradient_of_top_lags(h, n_top=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:n_top]
    top_vals = corr[top_idx]
    top_max = top_vals[0]
    temp = max(top_max - top_vals, 1e-12)
    weights = np.exp(-(top_vals - top_max) / temp)
    weights = weights / weights.sum()
    grad = np.zeros(n)
    for ki, w in zip(top_idx, weights):
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
    return grad, corr, float(np.max(corr))


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)

            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:

                grad[j2] -= w * x[j2 + lag]
        else:

            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]

    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):

                        break
                    h_cur[j] = 0.0
                    h_cur[k] = 1.0
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def multi_resolution_refine(h_base, n_base, n_fine, deadline, alphas, mu):
    x_old = np.linspace(0.0, 1.0, n_base, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_fine, endpoint=False)
    h_fine = np.interp(x_new, x_old, h_base)
    target_fine = n_fine / 2.0
    h_fine = project_feasible(h_fine, target_fine)
    for a in alphas:
        if time.time() > deadline:
            break
        h_fine = lbfgs_optimize(h_fine, target_fine, a, mu, 300, deadline)
    return h_fine


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 20

    def tleft():
        return deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 50.0

    # Phase 1: L-BFGS-B with moderate alpha
    alphas_p1 = [10, 50, 200, 1000]
    for a in alphas_p1:
        if tleft() < 80:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Subgradient descent (direct max)
    if tleft() > 120:
        h_sg, sc_sg = subgradient_descent(best_h, target, n_iter=1500, n_top=8, deadline=deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 3: L-BFGS-B with high alpha
    if tleft() > 80:
        for a in [5000, 20000]:
            if tleft() < 40:
                break
            h_new = lbfgs_optimize(best_h, target, a, mu, 500, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Binary rounding + random local search
    if tleft() > 60:
        h_bin, sc_bin = binary_round_and_search(best_h, target, n_trials=3000, deadline=deadline)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        # Also try rounding at different thresholds
        for thresh in [0.4, 0.6]:
            if tleft() < 30:
                break
            h_t = np.where(best_h >= thresh, 1.0, 0.0)
            h_t = project_feasible(h_t, target)
            h_bin2, sc_bin2 = binary_round_and_search(h_t, target, n_trials=2000, deadline=deadline)
            if sc_bin2 < best_score:
                best_score = sc_bin2
                best_h = h_bin2

    # Phase 5: Subgradient again from best (possibly binary)
    if tleft() > 100:
        h_sg2, sc_sg2 = subgradient_descent(best_h, target, n_iter=1500, n_top=12, deadline=deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 6: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 300:
        alphas_fine = [50, 500, 5000]
        h_fine = multi_resolution_refine(best_h, n, n2, deadline, alphas_fine, mu)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2
        if tleft() > 100:
            h_sg3, sc_sg3 = subgradient_descent(best_h, target, n_iter=1000, n_top=10, deadline=deadline)
            if sc_sg3 < best_score:
                best_score = sc_sg3
                best_h = h_sg3
        if tleft() > 60:
            h_new = lbfgs_optimize(best_h, target, 10000, mu, 400, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Binary local search at current resolution
    if tleft() > 40:
        h_bin3, sc_bin3 = binary_round_and_search(best_h, target, n_trials=3000, deadline=deadline)
        if sc_bin3 < best_score:
            best_score = sc_bin3
            best_h = h_bin3

    # Phase 8: SLSQP polish
    if tleft() > 60:
        h_slq = slsqp_polish(best_h, target, 5000, 5000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 9: Final subgradient + binary
    if tleft() > 40:
        h_sg4, sc_sg4 = subgradient_descent(best_h, target, n_iter=800, n_top=15, deadline=deadline)
        if sc_sg4 < best_score:
            best_score = sc_sg4
            best_h = h_sg4
        if tleft() > 20:
            h_bin4, sc_bin4 = binary_round_and_search(best_h, target, n_trials=2000, deadline=deadline)
            if sc_bin4 < best_score:
                best_score = sc_bin4
                best_h = h_bin4

    # Phase 10: Final L-BFGS-B
    if tleft() > 30:
        h_final_lb = lbfgs_optimize(best_h, target, 50000, mu, 300, deadline)
        sc = compute_c5(h_final_lb)
        if sc < best_score:
            best_score = sc
            best_h = h_final_lb

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
