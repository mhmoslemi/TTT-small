The gradient of corr_k with respect to h has a very specific structure: it's a difference of two shifted versions of (1-h) or h. This means the gradient is "smooth" in some sense, and the Hessian has a simple structure too.

  Wait, let me redo. corr_k = sum_i h[i](1-h[i+k]).

  So ∂corr_k/∂h[i] = (1-h[i+k]) - h[i-k] (with appropriate bounds).

  So the Hessian is a sparse matrix with at most 2 nonzeros per row! This is a tridiagonal-like structure (actually, it's a "banded" matrix with bandwidth k).

This is great! The Hessian of each corr_k is very sparse. The Hessian of the LogSumExp would be a weighted sum of these sparse matrices, which is still relatively structured.

The first term is easy (sparse). The second term is a sum of rank-1 matrices, which for 2n-1 terms is expensive.

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def truncated_lse_val_grad(h, alpha, top_m):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_m]
    corr_top = corr[top_idx]
    c_max = float(np.max(corr_top))
    shifted = np.clip(alpha * (corr_top - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, top_idx):
        if wk < 1e-16:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return val, g


def full_lse_val_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, range(len(corr))):
        if wk < 1e-16:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return val, g


def slsqp_truncated(h, n, target, alpha, top_m, maxiter, ftol, deadline):
    def obj(x):
        if time.time() > deadline:
            return 1e10
        v, _ = truncated_lse_val_grad(x, alpha, top_m)
        return v
    def objg(x):
        if time.time() > deadline:
            return np.zeros(n)
        _, g = truncated_lse_val_grad(x, alpha, top_m)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_full(h, n, target, alpha, maxiter, ftol, deadline):
    def obj(x):
        if time.time() > deadline:
            return 1e10
        v, _ = full_lse_val_grad(x, alpha)
        return v
    def objg(x):
        if time.time() > deadline:
            return np.zeros(n)
        _, g = full_lse_val_grad(x, alpha)
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,

    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,

        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgsb_truncated(h, n, target, alpha, top_m, mu, maxiter, deadline):
    def obj(x):
        if time.time() > deadline:
            return 1e10
        v, _ = truncated_lse_val_grad(x, alpha, top_m)
        return v + mu * (np.sum(x) - target) ** 2
    def objg(x):
        if time.time() > deadline:
            return np.zeros(n)
        _, g = truncated_lse_val_grad(x, alpha, top_m)
        return g + 2.0 * mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lag_equalize(h, n, target, top_m, max_iter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        top_vals = corr[top_idx]
        c_max = top_vals[0]
        spread = c_max - top_vals[-1]
        if spread < 1e-14:
            break
        g = np.zeros(n)
        for m in top_idx:
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += (1.0 - h_cur[k:])
                g[k:] -= h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += (1.0 - h_cur[:n - ell])
                g[:n - ell] -= h_cur[ell:]
        g *= dx
        step = 0.002
        for attempt in range(20):
            h_try = h_cur - step * g
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                h_cur = h_try
                step = min(step * 1.1, 0.01)
                break
            step *= 0.7
            if step < 1e-10:

    return h_cur, best_score


def subgradient_top(h, n, target, n_steps, top_k, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        diffs = top_vals - top_vals[0]
        weights = np.exp(diffs / max(temp, 1e-10))
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = project_feasible(h_cur - lr * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-16:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-13:
                break
    return best_h, best_score


def newton_pair(h, n, target, k1, k2, max_iter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m1 = k1 + (n - 1)
        m2 = k2 + (n - 1)
        c1 = corr[m1]
        c2 = corr[m2]
        cmax_all = float(np.max(corr))
        if cmax_all < c1 and cmax_all < c2:
            return h_cur, best_score
        diff_c = c1 - c2
        if abs(diff_c) < 1e-15:
            break
        g1 = np.zeros(n)
        k1i = k1
        if k1i >= 0:
            g1[:n - k1i] += (1.0 - h_cur[k1i:])
            g1[k1i:] -= h_cur[:n - k1i]
        else:
            ei = -k1i
            g1[ei:] += (1.0 - h_cur[:n - ei])
            g1[:n - ei] -= h_cur[ei:]
        g2 = np.zeros(n)
        k2i = k2
        if k2i >= 0:
            g2[:n - k2i] += (1.0 - h_cur[k2i:])
            g2[k2i:] -= h_cur[:n - k2i]
        else:
            ei2 = -k2i
            g2[ei2:] += (1.0 - h_cur[:n - ei2])
            g2[:n - ei2] -= h_cur[ei2:]
        g_diff = g1 - g2
        g_diff *= dx
        g_mean = (g1 + g2) * 0.5
        g_mean *= dx
        c_max_now = max(c1, c2)
        step_mean = 0.001 * (c_max_now - target * 0 + 0.5)
        step_diff = 0.0005 * diff_c
        for sc_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = h_cur - sc_factor * (step_mean * g_mean + step_diff * 0.001 * g_diff)
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_score:
                best_score = sc_try
                h_cur = h_try
                break
    return h_cur, best_score

            k_worst = m_worst - (n_cur - 1)
            g_worst = np.zeros(n_cur)
            if k_worst >= 0:
                g_worst[:n_cur - k_worst] += (1.0 - best_h[k_worst:])
                g_worst[k_worst:] -= best_h[:n_cur - k_worst]
            else:
                ew = -k_worst
                g_worst[ew:] += (1.0 - best_h[:n_cur - ew])
                g_worst[:n_cur - ew] -= best_h[ew:]
            g_worst *= (2.0 / n_cur)
            scales = [0.05, 0.08, 0.12]
            h_pert = best_h - scales[attempt] * g_worst
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = lbfgsb_truncated(h_pert, n_cur, target_cur, 500, 5, 100.0, 200, deadline)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_ref2 = slsqp_truncated(best_h, n_cur, target_cur, 50000, 3, 2000, 1e-17, deadline)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Final projection and evaluation
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
