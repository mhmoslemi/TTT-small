The damping (h = (1-α)*h + α*h_new) is a standard technique to ensure convergence (Bailey's theorem / Black &amp; Lipman). With α < 1, the sequence converges to a fixed point of the best-response mapping.

The fixed point satisfies h* = BR(h*), meaning h* is a "Nash equilibrium" of the game where the "payoff" is the overlap. At this equilibrium, the LP value equals the true max overlap.

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -350.0, 0.0)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def adam_optimize(h_init, n, target, max_steps, time_limit, alpha_start=500.0):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    lr = 0.005
    alpha = alpha_start
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 500:
            break

        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        m = np.max(corr)
        shifted = np.clip(alpha * (corr - m), -350.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights = weights / wsum

        grad = np.zeros(n)
        n_half = n - 1
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-14:
                continue
            k = idx - n_half
            if k >= 0:
                js = np.arange(k, n)
            else:
                js = np.arange(0, n + k)
            if len(js) > 0:
                grad[js] += w * (1.0 - h[js - k])
            if k >= 0:
                js2 = np.arange(0, n - k)
            else:
                js2 = np.arange(-k, n)
            if len(js2) > 0:
                grad[js2] -= w * h[js2 + k]
        grad *= dx

        sum_err = np.sum(h) - target
        grad += 5.0 * sum_err / n

        t = step + 1
        m_v = 0.9 * m_v + 0.1 * grad
        v_v = 0.999 * v_v + 0.001 * grad ** 2
        m_hat = m_v / (1.0 - 0.9 ** t)
        v_hat = v_v / (1.0 - 0.999 ** t)

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            if step % 100 == 0 and step > 0:
                alpha = min(alpha * 1.5, 50000.0)
                lr = max(lr * 1.05, 0.001)
        else:
            no_improve += 1
            lr *= 0.97
            if no_improve % 100 == 0 and no_improve > 100:
                alpha = max(alpha * 0.7, 100.0)

    return best_h, best_c5

def shake(h, n, target, rng, noise_level, n_shake):
    h_sh = h.copy()
    for _ in range(n_shake):
        mask = rng.random(n) < 0.3
        if mask.sum() > 0:
            h_sh[mask] += rng.randn(mask.sum()) * noise_level
            h_sh = project_feasible(h_sh, target)
    return h_sh

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coarse-to-fine LP
    if tleft() > 200:
        for n_coarse in [100, 200]:
            if tleft() < 80:
                break
            target_c = n_coarse / 2.0
            h_c = interpolate_to(best_h, n_coarse, target_c)
            h_c, c5_c = full_alt_lp(h_c, n_coarse, target_c,
                                     max_iter=5000, time_limit=min(60, tleft() * 0.3))
            h_fine = interpolate_to(h_c, n, target)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = h_fine

    # Phase 2: Full LP at native resolution
    if tleft() > 150:
        lp_time = min(350, tleft() * 0.4)
        h_lp, c5_lp = full_alt_lp(best_h, n, target,
                                   max_iter=10000, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Adam polish
    if tleft() > 100:
        adam_time = min(180, tleft() * 0.25)
        h_adam, c5_adam = adam_optimize(best_h, n, target,
                                         max_steps=100000, time_limit=adam_time,
                                         alpha_start=500.0)
        if c5_adam < best_score:
            best_score = c5_adam
            best_h = h_adam

    # Phase 4: Second LP pass after Adam
    if tleft() > 120:
        lp_time = min(200, tleft() * 0.3)
        h_lp2, c5_lp2 = full_alt_lp(best_h, n, target,
                                     max_iter=10000, time_limit=lp_time)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 5: Multi-resolution attempts
    for n_try in [300, 500, 600]:
        if tleft() < 120:
            break
        target_t = n_try / 2.0
        h_try = interpolate_to(best_h, n_try, target_t)
        h_try, c5_try = full_alt_lp(h_try, n_try, target_t,
                                     max_iter=5000, time_limit=min(80, tleft() * 0.3))
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try
            n = n_try
            target = target_t
        if tleft() < 60:
            break

    # Phase 6: Shake and re-optimize
    for attempt in range(6):
        if tleft() < 80:
            break
        noise = 0.02 + attempt * 0.015
        h_sh = shake(best_h, n, target, rng, noise, n_shake=4)
        h_sh, c5_sh = full_alt_lp(h_sh, n, target,
                                   max_iter=3000, time_limit=min(40, tleft() * 0.35))
        if c5_sh < best_score:
            best_score = c5_sh
            best_h = h_sh

    # Phase 7: Adam from shaken points
    for attempt in range(3):
        if tleft() < 80:
            break
        h_sh = shake(best_h, n, target, rng, 0.03, n_shake=3)
        h_a, c5_a = adam_optimize(h_sh, n, target,
                                   max_steps=50000, time_limit=min(40, tleft() * 0.3),
                                   alpha_start=200.0)
        if c5_a < best_score:
            best_score = c5_a
            best_h = h_a

    # Phase 8: Final LP polish
    if tleft() > 50:
        h_final_lp, c5_final = full_alt_lp(best_h, n, target,
                                            max_iter=5000, time_limit=min(40, tleft() * 0.7))
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_lp

    # Phase 9: Final Adam polish
    if tleft() > 30:
        h_final_a, c5_fa = adam_optimize(best_h, n, target,
                                          max_steps=30000, time_limit=min(25, tleft() * 0.7),
                                          alpha_start=2000.0)
        if c5_fa < best_score:
            best_score = c5_fa
            best_h = h_final_a

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
