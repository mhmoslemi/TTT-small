from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_to_continuous(h_bin, target, spread=0.1):
    """Convert binary to continuous by spreading boundary values"""
    h = h_bin.astype(float).copy()
    # Find transitions and spread them
    for i in range(len(h)-1):
        if h[i] != h[i+1]:
            h[i] = 0.5 + spread * (0.5 - h[i])
            h[i+1] = 0.5 + spread * (0.5 - h[i+1])
    return project_feasible(h, target)

def binary_local_search(h_bin, n, target, max_iters, deadline, rng):
    """Local search on binary representation using pair flips"""
    cur = h_bin.copy()
    cur_score = compute_c5(cur)
    
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    T = 0.1
    T_min = 0.001
    cooling = 0.995
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Pick a random pair to flip
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Compute new score (approximate: only affected lags change)
        cur[i], cur[j] = 1.0 - cur[i], 1.0 - cur[j]
        # Actually for binary: cur[i] was 1, now 0; cur[j] was 0, now 1
        new_score = compute_c5(cur)
        
        delta = new_score - cur_score
        if delta < 0 or rng.rand() < np.exp(-delta / T if T > T_min else 0):
            cur_score = new_score
            # Update sets
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
        else:
            cur[i], cur[j] = 1.0 - cur[i], 1.0 - cur[j]
        
        T = max(T * cooling, T_min)
    
    return cur, cur_score

import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _minmax_1d(C, B, a, b):
    """Minimize max_k(C[k] + B[k]*t) over t in [a, b]."""
    lo, hi = a, b
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = C + B * mid
        v_max = vals.max()
        active = np.where(vals >= v_max - 1e-14)[0]
        B_min_act = B[active].min()
        B_max_act = B[active].max()
        if B_min_act >= -1e-14:
            hi = mid
        elif B_max_act <= 1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0


def coord_descend(h, n, target, n_sweeps, deadline, rng):
    """Coordinate descent with exact 1-D min-max subproblem."""
    h = h.copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    n_lags = 2 * n - 1
    best_score = compute_c5(h)
    best_h = h.copy()
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        # Random order
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            # Compute c_k for current h
            corr = np.correlate(h, 1.0 - h, mode='full')

                if 0 <= imk < n:
                    B[m] -= h[imk]
            
            C = corr - h[i] * B
            
            # Solve 1-D subproblem
            t_star = _minmax_1d(C, B, 0.0, 1.0)
            h[i] = t_star

        h = project_feasible(h, target_sum)
        sc = compute_c5(h)
        if sc < best_score:

            best_h = h.copy()

    B = np.zeros(n_lags)
    # First term: B[m] += 1 - h[i+k] where k = m-(n-1), i.e., B[m] += 1-h[m+i-n+1]
