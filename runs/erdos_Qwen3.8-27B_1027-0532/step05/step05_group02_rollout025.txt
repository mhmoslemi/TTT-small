    it = 0

    while time.time() - start_time < time_limit:
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_mom
        h_nest = h + beta * (h - h_prev)
        h_nest = np.clip(h_nest, 0.0, 1.0)

        need_recompute = (it % n_check == 0)
        if need_recompute:
            corr = full_corr_fast(h_nest)
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        grad = np.zeros(n)
        if k_star >= 0:
            idx = positions + k_star
            valid = (idx < n)
            grad[valid] -= h_nest[idx[valid]]
            idx2 = positions - k_star
            valid2 = (idx2 >= 0)
            grad[valid2] += (1.0 - h_nest[idx2[valid2]])
        else:
            idx = positions + k_star
            valid = (idx >= 0)
            grad[valid] -= h_nest[idx[valid]]
            idx2 = positions - k_star
            valid2 = (idx2 < n)
            grad[valid2] += (1.0 - h_nest[idx2[valid2]])

        grad = grad - grad.mean()

        h = h - eta * grad
        h = project_feasible(h, target)
        h_prev = h_nest + beta * (h - h_nest)
        h_prev = np.clip(h_prev, 0.0, 1.0)

        eta *= eta_decay

        if it % (n_check * 10) == 0:
            corr = full_corr_fast(h)
            curr_c5 = float(corr.max()) * (2.0 / n)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()

        it += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    target = n / 2.0
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    curr_score = compute_c5(h_cur)

    alphas = [50, 200, 1000, 5000, 20000, 100000]
    maxiter_per = 800

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_swap_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 50000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        r = rng.random()
        if r < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta_for_swap(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i2 == i1:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if j2 == j1 or j1 == i1 or j2 == i1 or j1 == i2 or j2 == i2:
                continue
            delta = (compute_delta_for_swap(h, i1, j1) + compute_delta_for_swap(h, i2, j2))
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy polish on initial binary (30s)
    allowed = min(30.0, tleft() * 0.08)
    h_g, c5_g = greedy_polish(best_h, max_iters=50000, rng=rng,
                              time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Focused SA on binary (120s)
    allowed = min(120.0, tleft() * 0.25)
    n_sa = int(min(3000000, allowed * 30000))
    h_sa, c5_sa = focused_sa(best_h, n_sa, 0.015, 0.00005, rng,
                             time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Multi-swap SA (100s)
    allowed = min(100.0, tleft() * 0.20)
    n_ms = int(min(2000000, allowed * 20000))
    h_ms, c5_ms = multi_swap_sa(best_h, n_ms, 0.012, 0.00003, rng,
                                time_limit=allowed, start_time=start_time)
    if c5_ms < best_score:
        best_score = c5_ms
        best_h = h_ms.copy()

    # Phase 4: Continuous subgradient at initial resolution (150s)
    allowed = min(150.0, tleft() * 0.25)
    h_cont, c5_cont = subgradient_continuous(best_h, time_limit=allowed,
                                             start_time=start_time, n_check=5)
    if c5_cont < best_score:
        best_score = c5_cont
        best_h = h_cont.copy()

    # Phase 5: SLSQP refinement (100s)
    if tleft() > 60:
        allowed = min(100.0, tleft() * 0.20)
        h_ref, c5_ref = slsqp_refine(best_h, time_limit=allowed,
                                     start_time=start_time)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref.copy()

    # Phase 6: Higher resolution binary search
    if tleft() > 150:
        n_hi = n_init + 400
        if n_hi % 2 != 0:
            n_hi += 1
        target_hi = n_hi // 2
        h_interp = interpolate_to(best_h, n_hi, float(target_hi))
        h_hi_bin = project_binary(h_interp, target_hi)

        allowed = min(40.0, tleft() * 0.12)
        h_hi_g, c5_hi_g = greedy_polish(h_hi_bin, max_iters=30000, rng=rng,
                                        time_limit=allowed, start_time=start_time)
        allowed2 = min(60.0, tleft() * 0.18)
        n_sa_hi = int(min(2000000, allowed2 * 20000))
        h_hi_sa, c5_hi_sa = focused_sa(h_hi_g, n_sa_hi, 0.01, 0.00003, rng,
                                       time_limit=allowed2, start_time=start_time)
        if c5_hi_sa < c5_hi_g:
            h_hi_best = h_hi_sa
            c5_hi_best = c5_hi_sa
        else:
            h_hi_best = h_hi_g
            c5_hi_best = c5_hi_g

        if c5_hi_best < best_score:
            best_score = c5_hi_best
            best_h = h_hi_best.copy()

    # Phase 7: Continuous subgradient at higher resolution
    if tleft() > 100:
        n_use = len(best_h)
        allowed = min(80.0, tleft() * 0.25)
        h_cg, c5_cg = subgradient_continuous(best_h, time_limit=allowed,
                                             start_time=start_time, n_check=8)
        if c5_cg < best_score:
            best_score = c5_cg
            best_h = h_cg.copy()

    # Phase 8: SLSQP final polish
    if tleft() > 40:
        allowed = min(60.0, tleft() * 0.30)
        h_ref, c5_ref = slsqp_refine(best_h, time_limit=allowed,
                                     start_time=start_time)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref.copy()

    # Phase 9: Final binary SA burst
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(500000, max(5000, (tleft() - 12) * 30000)))
        h_sa_f, c5_f = focused_sa(h_bin_final, n_sa_f, 0.006, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 10),
                                  start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Final: ensure feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
