    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_momentum(h, n, target, n_steps, n_active, lr_init, lr_decay,
                         momentum, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    v = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = int(np.min(n_active, n))
        top_idx = np.argsort(over)[::-1][:top_k]
        diffs = over[top_idx] - over[top_idx[0]]
        temperatures = [0.002, 0.003, 0.005, 0.008, 0.01, 0.015, 0.02, 0.03][step % 8]
        weights = np.exp(diffs / temperatures)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h_cur - lr * v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 2.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def coordinate_descent(h, n, target, max_rounds, top_k, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        idx_sorted = np.argsort(over)[::-1][:top_k]
        improved = False
        for _ in range(2):
            corr = compute_corr(h_cur)
            over = corr[n - 1:n - 1 + n]
            idx_sorted = np.argsort(over)[::-1][:top_k]
            for k in idx_sorted:
                k = int(k)
                if k == 0:
                    continue
                for i in range(n - k):
                    if time.time() > deadline:
                        break
                    delta = h_cur[i] * (1.0 - h_cur[i + k])
                    c_old = float(over[k])
                    trial = h_cur.copy()
                    trial[i] = 0.0
                    trial[i + k] = min(1.0, h_cur[i + k] + delta / max(1e-15, h_cur[i]) * 0.5)
                    trial = project_feasible(trial, target)
                    corr_t = compute_corr(trial)
                    c_new = float(np.max(corr_t[n - 1:n - 1 + n]))
                    if c_new < c_old - 1e-14:
                        h_cur = trial
                        cur_score = c_new * dx
                        improved = True
                        break
    return h_cur, cur_score


def spectral_surgery(h, n, target, n_iters, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            break
        h_fft = np.fft.rfft(h_cur)
        freqs = np.fft.rfftfreq(n) * n
        peak_mask = (np.abs(freqs) >= n * 0.15) & (np.abs(freqs) <= n * 0.45)
        phase_shift = 2.0 * np.pi * k_star * freqs / n
        h_fft[peak_mask] *= np.exp(-0.02 * phase_shift[peak_mask])
        h_new = np.fft.irfft(h_fft, n)
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            h_fft = np.fft.rfft(h_cur)
            peak_mask2 = (np.abs(freqs) >= n * 0.1) & (np.abs(freqs) <= n * 0.5)
            h_fft[peak_mask2] *= 0.98
            h_new = np.fft.irfft(h_fft, n)
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score:
                h_cur = h_new
                cur_score = sc_new
    return h_cur, cur_score


def worst_lag_surgery(h, n, target, max_steps, initial_eps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_eps
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            break
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        j_neg = np.arange(k_star, n)
        grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score


def multi_lag_surgery(h, n, target, max_steps, initial_eps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_eps
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top3_idx = np.argsort(over)[::-1][:3]
        top3_vals = over[top3_idx]
        diffs = top3_vals - top3_vals[0]
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, k in zip(w, top3_idx):
            k = int(k)
            if k == 0:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wi * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wi * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.015)
        else:
            eps *= 0.6
            if eps < 1e-9:
                break
    return h_cur, cur_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 6)
                amp = rng.uniform(0.1, 0.4)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.35, 0.05, 0.95), target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_block_h, _g_block_s, _g_block_e, _g_target, _g_alpha
    _g_target = target
    _g_alpha = float(alpha)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_block_h = h_cur.copy()
            _g_block_s = s
            _g_block_e = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur


def annealing_polish(h, n, target, n_steps, scale, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    T_init = 0.005
    for step in range(n_steps):
        if time.time() > deadline:
            break
        T = T_init * (0.995 ** step)
        if T < 1e-6:
            break
        idx = np.random.randint(n)
        delta = np.random.randn() * scale
        trial = h_cur.copy()
        trial[idx] = np.clip(trial[idx] + delta, 0.0, 1.0)
        trial = project_feasible(trial, target)
        sc_new = compute_c5(trial)
        sc_cur = compute_c5(h_cur)
        if sc_new < sc_cur or np.random.rand() < np.exp(-(sc_new - sc_cur) / T):
            h_cur = trial
            if sc_new < best_score:
                best_score = sc_new
                best_h = trial.copy()
    return best_h, best_score
