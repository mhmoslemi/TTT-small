import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:]
    return float(np.max(over)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        alpha_f = float(a)

        def _obj(x):
            hf = project_feasible(x, target)
            corr = compute_corr(hf)
            over = corr[len(hf)-1:]
            c_max = float(np.max(over))
            shifted = alpha_f * (over - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha_f

        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def nesterov_subgradient(h, n, target, n_steps, n_active, lr_init, lr_decay, temp_init, temp_decay, deadline):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    temp = temp_init
    st_count = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        y = h_cur + v
        y = np.clip(y, 0.0, 1.0)
        corr = compute_corr(y)
        over = corr[n-1:]
        c_curr = float(np.max(over)) * dx

        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k >= n:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - y[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * y[j_neg - k]
        grad *= dx

        v = 0.9 * v - lr * grad
        z = h_cur + v
        h_new = project_feasible(z, target)
        sc_new = compute_c5(h_new)

        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
            st_count = 0
        else:
            st_count += 1
            lr *= lr_decay
            v *= 0.5
            if lr < 1e-12 or st_count > 150:
                break

        if sc_new < best_score:
            best_score = sc_new
            best_h = h_cur.copy()
        temp = max(temp * temp_decay, 0.001)
    return best_h, best_score


def lag_focused_slsqp(h, n, target, deadline, n_focus, n_rounds, alpha):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        top_lags = np.argsort(over)[::-1][:n_focus].tolist()

        def _obj_focused(x):
            hf = project_feasible(x, target)
            corr_f = compute_corr(hf)
            over_f = corr_f[n-1:]
            c_max = float(np.max(over_f))
            focus_vals = over_f[top_lags]
            focus_max = float(np.max(focus_vals))
            shifted_all = alpha * (over_f - c_max)
            shifted_all = np.clip(shifted_all, -700.0, 0.0)
            full_lse = c_max + float(np.log(np.sum(np.exp(shifted_all)))) / alpha
            shifted_f = alpha * (focus_vals - focus_max)
            shifted_f = np.clip(shifted_f, -700.0, 0.0)
            focus_lse = focus_max + float(np.sum(np.exp(shifted_f))) / alpha
            return 2.0 * focus_lse + 0.5 * full_lse

        try:
            res = minimize(_obj_focused, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': 1500, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
        h_cur, cur_score = nesterov_subgradient(h_cur, n, target, 80, n_focus, 0.002, 0.995, 0.005, 0.99, deadline)
    return h_cur, cur_score


def alternating_lag_min(h, n, target, deadline, n_rounds):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        if k_star < n:
            j_pos = np.arange(0, n - k_star)
            grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        for eps in [0.01, 0.005, 0.002, 0.001, 0.02, 0.0005]:
            h_try = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_try)
            if sc < cur_score - 1e-14:
                h_cur = h_try
                cur_score = sc
                break
    return h_cur, cur_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 4
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            else:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.25, target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)

    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score
