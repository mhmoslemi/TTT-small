def project_to_binary(h, target):
    """Round h to binary, maintaining sum = target."""
    n = len(h)
    h_sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:int(target)]] = 1.0
    return h_bin

def golden_ratio_binary(n, target, seed=0):
    """Generate a low-discrepancy binary sequence using golden ratio."""
    phi = (1 + np.sqrt(5)) / 2
    # Place 1s at positions floor(k * n / target * phi) mod n
    positions = np.zeros(target, dtype=int)
    for k in range(target):
        positions[k] = int((k * phi * n / target)) % n
    # Remove duplicates (shouldn't happen for golden ratio but just in case)
    unique_positions = np.unique(positions)
    if len(unique_positions) < target:

        remaining = np.setdiff1d(np.arange(n), unique_positions)

        unique_positions = np.concatenate([unique_positions, extra])

    h[unique_positions] = 1.0
    return h
