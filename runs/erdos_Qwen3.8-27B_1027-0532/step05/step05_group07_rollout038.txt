def subgradient_descent(h, n, target, deadline, alpha_schedule, step_scale=0.5):
    """Log-sum-exp subgradient descent with projection."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t = 0
    
    while time.time() < deadline:
        t += 1
        # Choose alpha based on schedule
        if alpha_schedule == 'increasing':
            alpha = 50.0 * (1.0 + t * 0.01)
        elif alpha_schedule == 'constant':
            alpha = 500.0
        else:  # 'adaptive'
            alpha = min(50.0 * (1.0 + t * 0.005), 5000.0)
        
        # Compute gradient of LSE
        corr = compute_corr(h_cur)
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        weights = np.exp(np.clip(shifted, -700, 0))
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.ones(2*n-1) / (2*n-1)
        else:
            weights /= w_sum
        
        grad = lse_gradient(h_cur, alpha)  # This computes sum weights[m] * dcorr[m]/dh[j]
        
        # Actually, the gradient of LSE = (1/alpha) * sum exp(alpha*corr_m) * grad_corr_m / sum exp(alpha*corr_m)
        # = sum (exp(alpha*corr_m)/sum) * grad_corr_m = sum weights[m] * grad_corr_m
        # So grad above IS the gradient of LSE (times alpha), wait...
        # f_α = (1/α) log(Σ exp(α corr_m))
        # ∇f_α = (Σ exp(α corr_m) ∇corr_m) / (Σ exp(α corr_m)) = Σ w_m ∇corr_m
        # So what I computed is exactly ∇f_α. Good.
        
        # Step size: use normalized step
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # η = step_scale * f_α / ||grad||^2 (approximate)
        f_alpha = (1.0/alpha) * np.log(np.sum(np.exp(np.clip(alpha*(corr-c_max), -700, 0))) + c_max*alpha/alpha)
        # Actually f_alpha = c_max + (1/alpha)*log(sum(exp(alpha*(corr-c_max))))
        f_alpha = c_max + (1.0/alpha) * np.log(np.sum(np.exp(np.clip(alpha*(corr-c_max), -700, 0))))
        
        eta = step_scale * f_alpha / (grad_norm * grad_norm + 1e-20)
        eta = min(eta, 0.1)  # Cap the step
        
        h_new = h_cur - eta * grad
        h_new = project_feasible(h_new, target)
        
        new_score = compute_c5(h_new)
        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()
        
        # Accept if it improves or is close
        if new_score < compute_c5(h_cur) + 1e-10:
            h_cur = h_new
        else:

            h_new = h_cur - eta * grad
            h_new = project_feasible(h_new, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
            # If still not improving, break
            if compute_c5(h_new) >= compute_c5(h_cur) - 1e-12:
                if t > 100:
                    break
    
    return best_h, best_score

def compute_corr(h):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 1e-14:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad_fast(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)

    w_sum = float(weights.sum())

        return np.ones(n) / n, f
    weights /= w_sum
    
    g = np.zeros(n)
    # lag = 0
    g += weights[n-1] * (1.0 - 2.0 * h)
    # negative lags: m = 0 to n-2, lag = m-(n-1) = -(n-1-m)

        if w < 1e-15:
            continue
        l = n - 1 - m
        g[l:] += w * (1.0 - h[:n-l])
        g[:n-l] -= w * h[l:]
    # positive lags: m = n to 2n-2, lag = m-(n-1)

        w = weights[m]
        if w < 1e-15:

    _g_target = target
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = compute_corr(x)
        cmax = float(np.max(corr))
        shifted = _g_alpha * (corr - cmax)
        return cmax + float(np.log(np.sum(np.exp(np.clip(shifted, -700.0, 0.0))))) / _g_alpha
    
    def cons(x):
        return np.sum(x) - _g_target
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:

                          bounds=bounds,

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

import numpy as np

_g_target = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 1e-14:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_G_i(h, i, n):
    lags = np.arange(-(n - 1), n)
    ip = i + lags
    im = i - lags
    term1 = np.zeros(2 * n - 1)
    mask1 = (ip >= 0) & (ip < n)
    term1[mask1] = 1.0 - h[ip[mask1]]
    term2 = np.zeros(2 * n - 1)
    mask2 = (im >= 0) & (im < n) & (lags != 0)
    term2[mask2] = h[im[mask2]]
    return term1 - term2


def lse_grad_fast(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    weights = np.exp(np.clip(shifted, -700.0, 0.0))
    w_sum = float(weights.sum())
    if w_sum < 1e-300:
        f = c_max + np.log(2.0 * n - 1) / alpha
        return np.ones(n) / n, f, corr
    weights /= w_sum
    g = np.zeros(n)
    g += weights[n - 1] * (1.0 - 2.0 * h)
    for m in range(n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        l = n - 1 - m
        g[l:] += w * (1.0 - h[:n - l])
        g[:n - l] -= w * h[l:]
    for m in range(n, 2 * n - 1):
        w = weights[m]
        if w < 1e-15:
            continue
        l = m - n + 1
        g[:n - l] += w * (1.0 - h[l:])
        g[l:] -= w * h[:n - l]
    f = c_max + np.log(w_sum) / alpha
    return g, f, corr


def smooth_grad_desc(h, n, target, deadline, rng,
                     alpha_start=15.0, alpha_end=4000.0,
                     momentum=0.82, step_cap=0.5):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)

    stall = 0

        t += 1
        progress = min(t / 2500.0, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        grad, f_alpha, corr = lse_grad_fast(h_cur, alpha)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-13:
            break
        eta = step_cap * max(f_alpha, 0.003) / (gnorm * gnorm + 1e-12)
        eta = min(eta, 0.4)
        v = momentum * v - eta * grad
        h_new = project_feasible(h_cur + v, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-12:
            best_score = sc_new

            stall = 0

            h_cur = h_new
        else:

                    v = np.zeros(n)

        if stall > 4000 and t > 3000:

    return best_h, best_score


def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=40):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    indices = np.arange(n)
    stall = 0
    G_cache = {}
    for sweep in range(500000):
        if time.time() > deadline:
            break
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
            sc = float(np.max(corr)) * dx
            if sc < best_score - 1e-13:
                best_score = sc
                stall = 0
            else:
                stall += 1
            if stall > 500 and sweep > 300:
                break
        G_cache.clear()
        rng.shuffle(indices)
        for p in range(n // 2):
            i = int(indices[2 * p])
            j = int(indices[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            if i not in G_cache:
                G_cache[i] = compute_G_i(h_cur, i, n)
            if j not in G_cache:
                G_cache[j] = compute_G_i(h_cur, j, n)
            C = G_cache[i] - G_cache[j]
            A = corr
            lo = 0.0
            hi = float(np.max(A + np.abs(C) * t_max)) + 1.0

            def _feasible(v):
                pos = C > 1e-14
                neg = C < -1e-14
                zero = ~(pos | neg)
                if np.any(zero) and float(np.max(A[zero])) > v + 1e-13:
                    return None
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min(h_cur[i] + (v - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max(h_cur[i] + (v - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    return (lb, ub)
                return None

            for _ in range(40):
                mid = (lo + hi) * 0.5
                if _feasible(mid) is not None:
                    hi = mid
                else:
                    lo = mid
            v_star = hi
            fe = _feasible(v_star)
            if fe is None:
                continue

            else:

                h_cur[j] = c - t_star

def slsqp_polish(h, n, target, alphas, maxiter=2000):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = compute_corr(x)
        cmax = float(np.max(corr))
        shifted = _g_alpha * (corr - cmax)
        return cmax + float(np.log(np.sum(np.exp(np.clip(shifted, -700.0, 0.0))))) / _g_alpha

    def obj_grad(x):
        n_local = len(x)
        corr = compute_corr(x)
        cmax = float(np.max(corr))
        shifted = _g_alpha * (corr - cmax)
        weights = np.exp(np.clip(shifted, -700.0, 0.0))
        w_sum = float(weights.sum())
        if w_sum < 1e-300:
            return np.zeros(n_local)
        weights /= w_sum
        g = np.zeros(n_local)
        g += weights[n_local - 1] * (1.0 - 2.0 * x)
        for m in range(n_local - 1):
            w = weights[m]
            if w < 1e-15:
                continue
            l = n_local - 1 - m
            g[l:] += w * (1.0 - x[:n_local - l])
            g[:n_local - l] -= w * x[l:]
        for m in range(n_local, 2 * n_local - 1):
            w = weights[m]
            if w < 1e-15:
                continue
            l = m - n_local + 1
            g[:n_local - l] += w * (1.0 - x[l:])
            g[l:] -= w * x[:n_local - l]
        return g

    def cons_fun(x):
        return np.sum(x) - _g_target

    def cons_jac(x):
        return np.ones(_g_n)

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-17})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 6: Random restarts with diverse perturbations
    dl = start_time + (budget_s - 15.0) * 0.92
    restart_scales = [0.025, 0.04, 0.06, 0.015, 0.08, 0.035]
    n_restarts = min(6, int(max(0, tleft() / 100.0)))
    for attempt in range(n_restarts):
        if tleft() < 80:
            break
        ns = restart_scales[attempt % len(restart_scales)]
        h_pert = best_h + rng.randn(n) * ns
        h_pert = project_feasible(h_pert, target)
        h_res, sc_res = smooth_grad_desc(h_pert, n, target, dl, rng,
                                         alpha_start=30.0, alpha_end=3000.0,
                                         momentum=0.8, step_cap=0.35)
        if sc_res < best_score:
            h_res, sc_res = paired_coord_desc(h_res, n, target, dl, rng, recompute_interval=30)
            if sc_res < best_score:
                best_score = sc_res
                best_h = h_res

    # Phase 7: Final SLSQP polish
    if tleft() > 50:
        h_fin, sc_fin = slsqp_polish(best_h, n, target, [500, 3000, 12000], 2000)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Final coordinate descent
    if tleft() > 20:
        h_fin, sc_fin = paired_coord_desc(best_h, n, target,
                                          start_time + budget_s - 10.0, rng,
                                          recompute_interval=25)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
