            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        h_g2, c5_g2 = greedy_local_search(h_pert, max_steps=3000, rng=rng,
                                          time_limit=min(15, time_limit), start_time=start_time,
                                          n_cand=n_cand)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    return best_h, best_c5

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_periodic_initial(n, target, rng):
    h = np.zeros(n)
    pattern_len = rng.randint(3, 8)
    ones_per_pattern = max(1, round(pattern_len * target / n))
    pattern = np.zeros(pattern_len)
    pattern[:ones_per_pattern] = 1.0
    pattern = pattern[rng.permutation(pattern_len)]
    n_repeats = n // pattern_len
    h[:n_repeats * pattern_len] = np.tile(pattern, n_repeats)
    if n % pattern_len != 0:
        h[n_repeats * pattern_len:] = pattern[:n % pattern_len]
    return project_binary(h, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    if tleft() < 20:
        n_final = len(best_h)
        return best_h, compute_c5(best_h), n_final

    # Phase 1: Quick greedy from initial
    allowed = min(60, tleft() * 0.08)
    h_g, c5_g = greedy_local_search(best_h, max_steps=20000, rng=rng,
                                    time_limit=allowed, start_time=start_time, n_cand=15)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: ILS (main workhorse)
    allowed = min(450, tleft() * 0.35)
    max_rounds = int(allowed / 30) + 5
    h_il, c5_il = ils_search(best_h, max_rounds=max_rounds, rng=rng,
                             time_limit=allowed, start_time=start_time, n_cand=12)
    if c5_il < best_score:
        best_score = c5_il
        best_h = h_il.copy()

    # Phase 3: Long SA from best
    if tleft() > 100:
        allowed = min(300, tleft() * 0.30)
        n_sa = int(min(8000000, allowed * 60000))
        h_sa, c5_sa = sa_optimized(best_h, n_sa, 0.03, 0.00005, rng,
                                   time_limit=allowed, start_time=start_time, focus_frac=0.5)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: ILS with larger perturbations (escape deeper)
    if tleft() > 80:
        allowed = min(200, tleft() * 0.20)
        n_pert_large = 15
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert_large):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        h_g2, c5_g2 = greedy_local_search(h_pert, max_steps=5000, rng=rng,
                                          time_limit=min(30, allowed * 0.3), start_time=start_time,
                                          n_cand=15)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() > 60:
            allowed2 = min(100, tleft() * 0.15)
            n_sa2 = int(min(3000000, allowed2 * 50000))
            h_sa2, c5_sa2 = sa_optimized(h_pert, n_sa2, 0.02, 0.0001, rng,
                                         time_limit=allowed2, start_time=start_time, focus_frac=0.4)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

    # Phase 5: Try alternative n values
    for n_try in [n_init + 60, n_init - 60, n_init + 120]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(30, tleft() * 0.10)
        h_g3, c5_g3 = greedy_local_search(h_bin_try, max_steps=10000, rng=rng,
                                          time_limit=allowed, start_time=start_time, n_cand=12)
        if tleft() > 40:
            allowed2 = min(20, tleft() * 0.08)
            n_sa3 = int(min(1000000, allowed2 * 50000))
            h_s3, c5_s3 = sa_optimized(h_g3, n_sa3, 0.015, 0.0001, rng,
                                       time_limit=allowed2, start_time=start_time, focus_frac=0.4)
            if c5_s3 < c5_g3:
                h_try = h_s3
                c5_try = c5_s3
            else:
                h_try = h_g3
                c5_try = c5_g3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: Coarse-to-fine for fresh start
    if tleft() > 100:
        n_crse = 100
        target_crse = n_crse // 2
        x_new = np.linspace(0, 1, n_crse)
        x_old = np.linspace(0, 1, len(best_h))
        h_crse = np.interp(x_new, x_old, best_h)
        h_crse_bin = project_binary(h_crse, target_crse)

        allowed = min(20, tleft() * 0.08)
        h_cg, c5_cg = greedy_local_search(h_crse_bin, max_steps=5000, rng=rng,
                                          time_limit=allowed, start_time=start_time, n_cand=10)
        if tleft() > 60:
            allowed2 = min(15, tleft() * 0.06)
            h_cs, c5_cs = sa_optimized(h_cg, int(min(500000, allowed2 * 80000)),
                                       0.02, 0.001, rng,
                                       time_limit=allowed2, start_time=start_time, focus_frac=0.5)
            if c5_cs < c5_cg:
                h_crse_best = h_cs
            else:
                h_crse_best = h_cg
        else:
            h_crse_best = h_cg

        # Upscale
        h_up = project_binary(interpolate_to(h_crse_best, len(best_h), float(len(best_h) // 2)),
                              len(best_h) // 2)
        allowed3 = min(25, tleft() * 0.08)
        h_ug, c5_ug = greedy_local_search(h_up, max_steps=10000, rng=rng,
                                          time_limit=allowed3, start_time=start_time, n_cand=12)
        if c5_ug < best_score:
            best_score = c5_ug
            best_h = h_ug.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 600, 2000]
        allowed = min(tleft() - 20, 120)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1500, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final SA burst
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(3000000, max(50000, (tleft() - 10) * 60000)))
        h_sa_f, c5_f = sa_optimized(h_bin_final, n_sa_f, 0.01, 0.00001,
                                    rng, time_limit=max(5.0, tleft() - 8), start_time=start_time,
                                    focus_frac=0.5)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 9: Final SLSQP squeeze
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [1000, 3000, 8000], 1000, 1e-15,
                                 time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
