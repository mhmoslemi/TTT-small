            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def lin_grad(h, g, alpha, dx):
    """Gradient of log-sum-exp smoothed overlap with g = 1-h_old fixed."""
    n = len(h)
    corr = np.correlate(h, g, mode='full')
    ov = corr * dx
    c_max = np.max(ov)
    shifted = np.clip(alpha * (ov - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-16:
            continue
        m = idx - n_half
        if m >= 0:
            grad[m:] += w * dx * g[:n - m]
        else:
            p = -m
            grad[:n - p] += w * dx * g[p:]
    return grad

def lin_fista(h_init, g, alpha, target, n_inner, dx, start_time, deadline_offset=30, budget_s=1300):
    n = len(h_init)
    h = project_feasible(h_init.copy(), target)
    y = h.copy()
    t_k = 1.0
    step = alpha / (4.0 * n + 1)
    for _ in range(n_inner):
        if time.time() - start_time > budget_s - deadline_offset:
            break
        grad = lin_grad(h, g, alpha, dx)
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        t_k = t_new
        h = h_new
    return h

def alt_fista(h_init, n, target, alphas, outer_per_alpha, inner_per_alpha, start_time, budget_s, deadline_offset=60):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for alpha in alphas:
        for _ in range(outer_per_alpha):
            if time.time() - start_time > budget_s - deadline_offset:
                return best_h, best_c5
            g = 1.0 - h
            h = lin_fista(h, g, alpha, target, inner_per_alpha, dx, start_time, deadline_offset, budget_s)
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-15:
                best_c5 = c5_val
                best_h = h.copy()
    return best_h, best_c5

def smooth_grad_full(h, alpha, dx, top_m=None):
    """Gradient of log-sum-exp smoothed C5 (non-linearized, bilinear)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    ov = corr * dx
    c_max = np.max(ov)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(ov)[-top_m:]
        sub_ov = ov[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_ov = ov
    shifted = np.clip(alpha * (sub_ov - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-16:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[k:] += w * (1.0 - h[:n - k]) * dx
            grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            grad[:n - m] += w * (1.0 - h[m:]) * dx
            grad[m:] -= w * h[:n - m] * dx
    return grad

def projgrad_momentum(h_init, n, target, alpha, top_m, n_steps, momentum, start_time, budget_s, deadline_offset=30):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_steps):
        if time.time() - start_time > budget_s - deadline_offset:
            break
        grad = smooth_grad_full(h, alpha, dx, top_m=top_m)
        v = momentum * v + (1.0 - momentum) * grad
        gnorm = np.max(np.abs(v))
        if gnorm < 1e-15:
            break
        lr = min(1.0 / (alpha * gnorm + 1e-10), 2.0)
        improved = False
        for _ in range(30):
            h_try = h - lr * v
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
            lr *= 0.5
        if not improved:
            # Try a smaller step
            v *= 0.5
    return best_h, best_c5

def local_search(h_init, n, target, n_sweeps, start_time, budget_s, deadline_offset=30, rng=None):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() - start_time > budget_s - deadline_offset:
            break
        order = rng.permutation(n) if rng is not None else np.arange(n)
        improved = False
        for i in order:
            for target_val in [0.0, 1.0]:
                h_try = h.copy()
                h_try[i] = target_val
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-15:
                    h = h_try
                    best_c5 = sc
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ov = corr * dx
    c_max = np.max(ov)
    shifted = np.clip(alpha * (ov - c_max), -600.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_cons_fun(h):
    return np.sum(h) - _global_target

def slsqp_cons_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, start_time, budget_s, deadline_offset=30):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time.time() - start_time > budget_s - deadline_offset:
            break
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur
