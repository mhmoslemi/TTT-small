def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def build_grad_matrix(n):
    """Build the matrix M1 where M1[j,i] = r_index = j-i+n-1"""
    j_idx = np.arange(n)
    i_idx = np.arange(n)
    return j_idx[:, None] - i_idx[None, :] + (n - 1)  # shape (n, n)

def compute_softmax_grad(h, beta, grad_idx):
    """Compute soft-max objective and gradient.
    grad_idx[j,i] = index into p array for p[j-i+n-1]
    """
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    max_corr = np.max(corr)
    shifted = beta * (corr - max_corr)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()
    
    obj = (max_corr + np.log(exp_vals.sum()) / beta) * dx
    
    # First term: sum_i (1-h[i]) * p[j-i+n-1] = (1-h) @ grad_idx^T indexed... 
    # Actually: First_term[j] = sum_i (1-h[i]) * p[grad_idx[j,i]]
    # This is: take p with index grad_idx, giving matrix P[j,i] = p[grad_idx[j,i]]
    # Then First_term = P @ (1-h)
    
    P = np.take(p, grad_idx)  # shape (n, n)
    term1 = P @ (1.0 - h)
    term2 = P.T @ h  # P.T[j,i] = P[i,j] = p[grad_idx[i,j]] = p[i-j+n-1]
    
    grad = dx * (term1 - term2)
    
    return obj, grad

def gradient_deep(h, n, target, beta, max_iter, lr, deadline, rng, grad_idx):
    """Run projected gradient method with backtracking line search."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_obj = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        obj, grad = compute_softmax_grad(h_cur, beta, grad_idx)
        
        # Projected gradient step
        # We want to MINIMIZE obj, so step is -grad
        # After step, project to feasible set
        for attempt in range(20):
            step_size = lr / (1.0 + attempt * 0.5)
            h_new_raw = h_cur - step_size * grad
            h_new = project_feasible(h_new_raw, target)
            
            # Check if it improved
            c5_new = compute_c5(h_new)
            if c5_new < best_obj:
                best_obj = c5_new
                best_h = h_new.copy()
                h_cur = h_new
                break
            # Also check the soft-max objective for convergence
            # (the actual C5 might not decrease even if soft-max does)
        
        # If no improvement after several iterations, reduce learning rate
        if it % 100 == 99:
            lr *= 0.8
    
    return best_h, best_obj

def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=50):
    """Paired coordinate descent from previous program."""
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    
    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        rng.shuffle(pair_positions)
        
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            
            # Binary search for the best t
            lo = 0.0
            hi = float(np.max(A + C * t_max))
            
            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid
            
            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5
            
            delta = t_star - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 100 and sweep > 50:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def compute_B(h, i, n):
    """Compute B vector for coordinate descent."""
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B

def slsqp_polish(h, n, target, alphas, maxiter=2000, ftol=1e-16):
    """SLSQP polish using soft-max objective."""
    grad_idx = build_grad_matrix(n)
    
    for alpha in alphas:
        try:
            res = minimize(
                lambda x: _softmax_obj(x, alpha, n, grad_idx)[0],
                lambda x: _softmax_obj(x, alpha, n, grad_idx)[1],
                h,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < compute_c5(h):
                h = h_new
        except Exception:
            pass
    
    return h

def _softmax_obj(x, beta, n, grad_idx):
    return compute_softmax_grad(x, beta, grad_idx)

    max_c = np.max(corr)
    s = beta * (corr - max_c)
    s = np.clip(s, -700.0, 0.0)
    e = np.exp(s)
    return e / e.sum()

def get_weights_topk(corr, k):
    """Get uniform weights on top-k worst (largest) correlations."""
    idx = np.argsort(corr)[::-1][:k]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k
    return w

def compute_grad(h, weights, grad_idx):
    """Compute gradient given weights."""
    n = len(h)
    dx = 2.0 / n
    P = np.take(weights, grad_idx)
    return dx * (P.dot(1.0 - h) - P.T.dot(h))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_refine(h, n, target, deadline, grad_idx):
    _G_N = n
    _G_GI = grad_idx
    _G_TGT = target
    cur_c5 = compute_c5(h)

        _G_BETA = beta

                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}],
                          options={'maxiter': 800, 'ftol': 1e-18})
