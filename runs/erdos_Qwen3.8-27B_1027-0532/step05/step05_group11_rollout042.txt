def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_full_lp(h, n, target):
    dx = 2.0 / n
    g = 1.0 - h
    c = np.zeros(n + 1)
    c[-1] = 1.0
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    for k in range(-(n - 1), n):
        row_idx = k + n - 1
        if k >= 0:
            A_ub[row_idx, :n - k] = g[k:] * dx
        else:
            A_ub[row_idx, -k:] = g[:n + k] * dx
        A_ub[row_idx, n] = -1.0
    b_ub = np.zeros(n_lags)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0.0, 1.0)] * n + [(0.0, None)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.status != 0:
        return None, None
    return result.x[:n], float(result.x[n])

            break

            stall = 0

        h = h_new
    return best_h, best_c5

            best_h = h_lp4
        h_fin = slsqp_polish(best_h, n_use, t_use,
                              [2000, 10000, 50000],
                              maxiter=1500)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
