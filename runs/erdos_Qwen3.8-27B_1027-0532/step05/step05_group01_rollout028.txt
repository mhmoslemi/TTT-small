def pair_cd_optimize(h_start, n, target, time_limit, max_sweeps=3000, n_cand=60):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        c_max = np.max(corr)
        active_threshold = c_max - 0.03 * c_max
        active = np.where(corr >= active_threshold)[0]

        grad = np.zeros(n)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta_val = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx

                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta_val = delta_star

            h[best_i] -= best_delta_val
            h[best_j] += best_delta_val
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

def gradient_smooth_optimize(h_start, n, target, time_limit, max_steps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    alpha = 100.0
    lr = 0.01
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            alpha *= 1.5
            lr *= 0.5
            if lr < 1e-6:
                break
            no_improve = 0

        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            w_top = np.zeros_like(corr)
            top_idx = np.argmax(corr)
            w_top[top_idx] = 1.0
            weights = w_top
        else:
            weights /= w_sum

        grad = np.zeros(n)
        top_indices = np.argsort(corr)[-20:]
        top_weights = weights[top_indices]
        tw_sum = top_weights.sum()
        if tw_sum < 1e-30:
            top_weights = np.ones(len(top_indices)) / len(top_indices)
        else:
            top_weights /= tw_sum

        for wi, s_idx in zip(top_weights, top_indices):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr_try = lr * lr_mult
            h_try = h - lr_try * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            lr *= 0.7

    return best_h, best_c5

        elif strat == 'low_freq':
            freq = rng.randint(2, 5)
            phase = rng.rand() * 2 * np.pi
            x = np.linspace(0, 2 * np.pi * freq, n)
            h_pert += 0.05 * np.sin(x + phase)
        elif strat == 'high_freq':
            n_bumps = rng.randint(3, 8)
            for _ in range(n_bumps):
                center = rng.randint(0, n)
                width = rng.randint(3, 10)
                amp = rng.choice([-0.1, 0.1, -0.15, 0.15])
                h_pert += amp * np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
        elif strat == 'push_binary':
            mid_mask = (h_pert > 0.15) & (h_pert < 0.85)
            n_mid = np.sum(mid_mask)
            if n_mid > 0:
                push = rng.choice([-1, 1], size=int(n_mid))
                h_pert[mid_mask] += push * 0.3
        elif strat == 'hole_fill':
            zeros = h_pert < 0.05
            ones = h_pert > 0.95
            n_zeros = np.sum(zeros)
            n_ones = np.sum(ones)
            if n_zeros > 0 and n_ones > 0:
                move = min(n_zeros, n_ones) * 0.3
                h_pert[zeros] += move / max(n_zeros, 1)
                h_pert[ones] -= move / max(n_ones, 1)
        elif strat == 'sinusoid':
            freq = rng.uniform(1, 10)
            phase = rng.rand() * 2 * np.pi
            x = np.linspace(0, 2, n)
            h_pert += 0.1 * np.sin(2 * np.pi * freq * x + phase)

        h_pert = project_feasible(h_pert, target)
        c5_pert = compute_c5(h_pert)

        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            ref_h = h_pert

    return best_h, best_c5


def triplet_cd(h_start, n, target, time_limit, max_sweeps=2000, n_cand=30):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        c_max = np.max(corr)
        active_threshold = c_max - 0.05 * c_max
        active = np.where(corr >= active_threshold)[0]

        grad = np.zeros(n)
        for s in active:
            k = int(s) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low3 = sorted_idx[:n_cand]
        high3 = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_move = None

        for i in high3:
            for j in low3:
                if i == j:
                    continue
                delta_lo = max(0.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d, b_d = delta_lo, delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial - 1e-15:
                    best_trial = approx_max
                    best_move = (i, j, delta_star)

        if best_move is not None:
            i, j, d = best_move
            h[i] -= d
            h[j] += d
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 180:
        h_slp, c5_slp = slp_optimize(best_h, n, target, time_limit=min(120, tleft() * 0.5),
                                     max_iter=150, initial_delta=0.03, active_frac=0.05)
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    if tleft() > 150:
        h_cd, c5_cd = pair_cd_optimize(best_h, n, target, time_limit=min(100, tleft() * 0.4),
                                       max_sweeps=4000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_slp, c5_slp = slp_optimize(best_h, n, target, time_limit=min(60, tleft() * 0.4),
                                     max_iter=200, initial_delta=0.02, active_frac=0.03)
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    if tleft() > 200:
        h_ms, c5_ms = multi_start_optimize(best_h, n, target, time_limit=min(60, tleft() * 0.3),
                                           seed=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 150:
        cand_h = best_h
        if tleft() > 100:
            h_cd, c5_cd = pair_cd_optimize(cand_h, n, target, time_limit=min(80, tleft() * 0.4),
                                           max_sweeps=3000, n_cand=55)
            if c5_cd < compute_c5(cand_h):
                cand_h = h_cd
        if tleft() > 60:
            h_slp, c5_slp = slp_optimize(cand_h, n, target, time_limit=min(50, tleft() * 0.4),
                                         max_iter=200, initial_delta=0.015, active_frac=0.04)
            if c5_slp < best_score:
                best_score = c5_slp
                best_h = h_slp

    if tleft() > 120:
        h_tr, c5_tr = triplet_cd(best_h, n, target, time_limit=min(60, tleft() * 0.4),
                                 max_sweeps=2000, n_cand=35)
        if c5_tr < best_score:
            best_score = c5_tr
            best_h = h_tr

    if tleft() > 100:
        h_grad, c5_grad = gradient_smooth_optimize(best_h, n, target,
                                                   time_limit=min(50, tleft() * 0.4),
                                                   max_steps=50000)
        if c5_grad < best_score:
            best_score = c5_grad
            best_h = h_grad

    if tleft() > 150:
        n_try = 500
        target_try = n_try / 2.0
        x_src = np.linspace(0.0, 1.0, n)
        x_dst = np.linspace(0.0, 1.0, n_try)
        h_interp = np.interp(x_dst, x_src, best_h)
        h_interp = project_feasible(h_interp, target_try)
        h_hi, c5_hi = slp_optimize(h_interp, n_try, target_try,
                                   time_limit=min(60, tleft() * 0.3),
                                   max_iter=100, initial_delta=0.02, active_frac=0.04)
        if c5_hi < best_score:
            best_score = c5_hi
            best_h = h_hi
        if tleft() > 60:
            h_cd2, c5_cd2 = pair_cd_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                             time_limit=min(40, tleft() * 0.4),
                                             max_sweeps=2000, n_cand=40)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    if tleft() > 80:
        h_slp, c5_slp = slp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                     time_limit=min(40, tleft() * 0.4),
                                     max_iter=300, initial_delta=0.01, active_frac=0.02)
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    if tleft() > 50:
        nn = len(best_h)
        h_cd, c5_cd = pair_cd_optimize(best_h, nn, nn / 2.0,
                                       time_limit=min(30, tleft() * 0.5),
                                       max_sweeps=2000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        nn = len(best_h)
        h_slp, c5_slp = slp_optimize(best_h, nn, nn / 2.0,
                                     time_limit=min(20, tleft() * 0.5),
                                     max_iter=300, initial_delta=0.005, active_frac=0.01)
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp

    nn = len(best_h)
    h_final = project_feasible(best_h, nn / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
