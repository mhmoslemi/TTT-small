import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_and_grad(h, alpha):
    """Compute soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    c_max = np.max(corr)
    exp_vals = np.exp(alpha * (corr - c_max))
    w = exp_vals / np.sum(exp_vals)
    
    obj = (c_max + np.log(np.sum(exp_vals)) / alpha) * dx
    
    # Gradient
    grad = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - center
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] -= wk * (1.0 - h[k:])
            grad[k:] += wk * (1.0 - h[:n-k])
        else:
            k_abs = -k
            grad[:n-k_abs] += wk * (1.0 - h[k_abs:])
            grad[k_abs:] -= wk * h[:n-k_abs]
    
    grad *= dx
    return obj, grad

def nesterov_optimize(h, alphas, max_iters, ftol):
    """Nesterov accelerated projected gradient with adaptive alpha."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = compute_c5(h)
    
    x = h.copy()
    y = h.copy()  # Nesterov momentum position
    momentum = 0.9
    
    for alpha in alphas:
        # Reset Nesterov state for each alpha
        x = best_h.copy()
        y = best_h.copy()
        momentum = 0.9
        
        for it in range(max_iters):
            # Evaluate at Nesterov position
            y = project_feasible(y, target)
            obj_y, grad_y = soft_max_obj_and_grad(y, alpha)
            
            # Nesterov step
            t = (1.0 + np.sqrt(1.0 + 4.0 * momentum**2)) / 2.0
            beta = (t - 1.0) / t
            momentum = beta
            x_new = y - (1.0 / (alpha * 2.0 * t)) * grad_y
            # Actually, let me use a proper step size
            # The Lipschitz constant of the gradient is bounded, let's use a fixed step
            # L ≈ 2 * dx * 2 (rough bound)
            step = 1.0 / (alpha * np.max(np.abs(grad_y)) + 1e-10)
            step = min(step, 0.5 / (alpha + 1.0))
            
            x_new = y + beta * (x - y) - step * grad_y
            
            # Project
            x_new = project_feasible(x_new, target)
            
            # Evaluate
            obj_x = compute_c5(x_new)
            if obj_x < best_val:
                best_val = obj_x
                best_h = x_new.copy()
            
            x = x_new
        
    return best_h, best_val

    zeros = np.where(h_cur < 0.5)[0]

    rng = np.random.RandomState(42)

            break
        improved = False
        for _ in range(5000):

                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            val = compute_c5(h_cur)
            
            if val < best_val:

                best_h = h_cur.copy()
                improved = True
                # Update ones/zeros
                ones = np.where(h_cur > 0.5)[0]
                zeros = np.where(h_cur < 0.5)[0]
            else:
                h_cur[i] = 1.0
                h_cur[j] = 0.0
        
        if not improved:
            # Try a random restart
            n_flips = rng.randint(5, 20)
            flip_ones = rng.choice(ones, min(n_flips, len(ones)), replace=False)
            flip_zeros = rng.choice(zeros, min(n_flips, len(zeros)), replace=False)
            for idx in flip_ones:
                h_cur[idx] = 0.0
            for idx in flip_zeros:
                h_cur[idx] = 1.0
            h_cur = project_feasible(h_cur, target)

            zeros = np.where(h_cur < 0.5)[0]
    
    return best_h, best_val

```python
def topk_grad(h, k):
    """Gradient of the mean of top-k overlaps."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Get top-k indices
    top_idx = np.argsort(corr)[::-1][:k]
    top_vals = corr[top_idx]
    
    # Gradient is average of individual gradients
    grad = np.zeros(n)
    center = n - 1
    for idx in top_idx:
        k_off = idx - center
        if k_off == 0:
            grad += (1.0 - 2.0 * h) / k
        elif k_off > 0:
            grad[:n-k_off] -= (1.0 - h[k_off:]) / k
            grad[k_off:] += (1.0 - h[:n-k_off]) / k
        else:
            k_abs = -k_off
            grad[:n-k_abs] += (1.0 - h[k_abs:]) / k
            grad[k_abs:] -= h[:n-k_abs] / k
    
    return grad * dx
```

This is much faster to compute (only k terms vs. 2n-1 terms).
