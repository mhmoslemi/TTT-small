            else:
                dk = 0.0
                if j + k_star < n:
                    dk += h_cur[j + k_star]
                if j >= k_star:
                    dk -= 1.0 - h_cur[j - k_star]
                if dk < -0.5:
                    candidates.append((j, 0.0, dk))

        if not candidates:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        candidates.sort(key=lambda x: x[2])
        n_to_try = min(len(candidates), 15)
        best_hm = max_cur
        best_j = -1
        best_val = 0.0

        for ci in range(n_to_try):
            j, val, _ = candidates[ci]
            h_cur[j] = val
            over_new = compute_over(h_cur)
            hm = float(np.max(over_new))
            h_cur[j] = 1.0 - val
            if hm < best_hm - 1e-14:
                best_hm = hm
                best_j = j
                best_val = val

        if best_j >= 0:
            h_cur[best_j] = best_val
            over_cur = compute_over(h_cur)
            max_cur = float(np.max(over_cur))
            if max_cur < best_max - 1e-14:
                best_h = h_cur.copy()
                best_max = max_cur
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_h, best_max * (2.0 / n) / (2.0 / n)

def pair_flip_search(h, target, start_time=None, time_limit=30.0, max_pairs=500):
    n = len(h)
    h_cur = h.copy().astype(float)
    over_cur = compute_over(h_cur)
    max_cur = float(np.max(over_cur))
    best_h = h_cur.copy()
    best_max = max_cur

    k_star = int(np.argmax(over_cur))
    ones = np.where(h_cur == 1.0)[0]
    zeros = np.where(h_cur == 0.0)[0]

    if len(ones) < 2 or len(zeros) < 2:
        return best_h, best_max * (2.0 / n)

    np.random.seed(12345)
    ones_sample = np.random.choice(ones, min(30, len(ones)), replace=False)
    zeros_sample = np.random.choice(zeros, min(30, len(zeros)), replace=False)

    pairs_tried = 0
    for j1 in ones_sample:
        if start_time is not None and time.time() - start_time > time_limit:
            break
        for j2 in zeros_sample:
            if j1 == j2:
                continue
            pairs_tried += 1
            if pairs_tried > max_pairs:
                break
            if j1 + k_star < n:
                dk1 = h_cur[j1 + k_star]
            else:
                dk1 = 0.0
            if j1 >= k_star:
                dk1 -= (1.0 - h_cur[j1 - k_star])
            else:
                dk1 = 0.0
            if j2 + k_star < n:
                dk2 = -h_cur[j2 + k_star]
            else:
                dk2 = 0.0
            if j2 >= k_star:
                dk2 += (1.0 - h_cur[j2 - k_star])
            else:
                dk2 = 0.0
            if dk1 + dk2 >= -0.5:
                continue
            h_cur[j1] = 0.0
            h_cur[j2] = 1.0
            over_new = compute_over(h_cur)
            hm = float(np.max(over_new))
            if hm < best_max - 1e-14:
                best_max = hm
                best_h = h_cur.copy()
            else:
                h_cur[j1] = 1.0
                h_cur[j2] = 0.0

    return best_h, best_max * (2.0 / n)

_sg_alpha = 0.0
_sg_n = 0
_sg_target = 0.0

def soft_max_obj(x):
    h = project_feasible(x, _sg_target)
    corr = compute_corr(h)
    over = corr[_sg_n-1:_sg_n-1+_sg_n] * (2.0 / _sg_n)
    m = float(np.max(over))
    shifted = _sg_alpha * (over - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    return m + float(np.log(np.sum(np.exp(shifted)))) / _sg_alpha

def soft_max_grad(x):
    h = project_feasible(x, _sg_target)
    n = _sg_n
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n] * dx
    m = float(np.max(over))
    shifted = _sg_alpha * (over - m)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    val = m + float(np.log(np.sum(exp_vals))) / _sg_alpha
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad[:] += wk * dx * (1.0 - 2.0 * h)
        else:
            idx1 = np.arange(k, n)
            grad[idx1] += wk * dx * (1.0 - h[idx1 - k])
            idx2 = np.arange(0, n - k)
            grad[idx2] -= wk * dx * h[idx2 + k]
    return val, grad

def cons_eq_fun(x):
    return np.sum(x) - _sg_target

def cons_eq_jac(x):
    return np.ones(_sg_n)

def slsqp_optimize(h, alphas, maxiter=2000, ftol=1e-15, start_time=None, time_limit=100.0):
    global _sg_n, _sg_target, _sg_alpha
    _sg_n = len(h)
    _sg_target = _sg_n / 2.0
    bounds = [(0.0, 1.0)] * _sg_n
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if start_time is not None and time.time() - start_time > time_limit:
            break
        _sg_alpha = float(a)
        try:
            res = minimize(soft_max_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=soft_max_grad,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _sg_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp,
                          start_time=None, time_limit=100.0):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)

    for step in range(n_steps):
        if start_time is not None and time.time() - start_time > time_limit:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score - 1e-12:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.01, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def try_rounding_improve(h, target, n_thresholds=10):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(h)
    for i in range(n_thresholds):
        thresh = (i + 1) / (n_thresholds + 1)
        h_r = round_to_bangbang(h, target)
        if i > 0:
            noise = np.random.randn(n) * 0.01 * i
            h_r = round_to_bangbang(h + noise, target)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _sg_n, _sg_target, _sg_alpha
    start_time = time.time()
    deadline = start_time + budget_s - 20.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: Bang-bang round + local search
    h_bb = round_to_bangbang(h_init, target)
    sc_bb = compute_c5(h_bb)
    if sc_bb < best_score:
        best_score = sc_bb
        best_h = h_bb.copy()

    h_ls, sc_ls = bangbang_local_search(best_h, target, max_iter=2000,
                                         start_time=start_time, time_limit=80.0)
    sc_ls = compute_c5(h_ls)
    if sc_ls < best_score:
        best_score = sc_ls
        best_h = h_ls.copy()

    # Phase 2: SLSQP with analytical gradient
    if tleft() > 200:
        h_ref, sc = slsqp_optimize(best_h, [100, 500, 2000, 10000],
                                    maxiter=2000, ftol=1e-15,
                                    start_time=start_time, time_limit=150.0)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Bang-bang local search again
    if tleft() > 150:
        h_bb2 = round_to_bangbang(best_h, target)
        h_ls2, sc_ls2 = bangbang_local_search(h_bb2, target, max_iter=2000,
                                               start_time=start_time, time_limit=100.0)
        sc_ls2 = compute_c5(h_ls2)
        if sc_ls2 < best_score:
            best_score = sc_ls2
            best_h = h_ls2.copy()

    # Phase 4: Pair flip search
    if tleft() > 120:
        h_pf, sc_pf = pair_flip_search(best_h, target, start_time=start_time,
                                        time_limit=40.0, max_pairs=1000)
        sc_pf = compute_c5(h_pf)
        if sc_pf < best_score:
            best_score = sc_pf
            best_h = h_pf.copy()
        # Local search after pair flips
        h_ls3, sc_ls3 = bangbang_local_search(best_h, target, max_iter=1500,
                                               start_time=start_time, time_limit=60.0)
        sc_ls3 = compute_c5(h_ls3)
        if sc_ls3 < best_score:
            best_score = sc_ls3
            best_h = h_ls3.copy()

    # Phase 5: Subgradient
    if tleft() > 150:
        h_sg, sc_sg = subgradient_momentum(best_h, 400, 4, 0.003, 0.997, 0.3, 0.005,
                                            start_time=start_time, time_limit=60.0)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref2, sc2 = slsqp_optimize(best_h, [2000, 10000, 50000],
                                      maxiter=1500, ftol=1e-16,
                                      start_time=start_time, time_limit=80.0)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Phase 6: Multi-resolution - finer grid
    if tleft() > 300:
        for n_fine in [n + 100, n + 200, n + 400]:
            if tleft() < 200:
                break
            target_fine = n_fine / 2.0
            h_fine = interpolate_to(best_h, n_fine, target_fine)
            h_fine_bb = round_to_bangbang(h_fine, target_fine)
            hf, scf = bangbang_local_search(h_fine_bb, target_fine, max_iter=1500,
                                             start_time=start_time, time_limit=60.0)
            scf = compute_c5(hf)
            if tleft() < 100:
                if scf < best_score:
                    best_score = scf
                    best_h = hf
                continue
            h_ref_f, sc_rf = slsqp_optimize(hf, [500, 2000, 10000],
                                             maxiter=1500, ftol=1e-15,
                                             start_time=start_time, time_limit=80.0)
            if sc_rf < best_score:
                best_score = sc_rf
                best_h = h_ref_f
            if tleft() < 100:
                continue
            h_bb_f = round_to_bangbang(best_h, len(best_h) / 2.0)
            hf2, scf2 = bangbang_local_search(h_bb_f, len(best_h) / 2.0,
                                               max_iter=1500,
                                               start_time=start_time, time_limit=50.0)
            scf2 = compute_c5(hf2)
            if scf2 < best_score:
                best_score = scf2
                best_h = hf2

    # Phase 7: Random restarts
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    for attempt in range(8):
        if tleft() < 80:
            break
        if attempt % 3 == 0:
            h_pert = best_h + rng.randn(n_cur) * 0.05
        elif attempt % 3 == 1:
            block = max(5, n_cur // 30)
            h_pert = best_h.copy()
            for b in range(n_cur // block):
                if rng.random() < 0.3:
                    shift = rng.uniform(-0.1, 0.1)
                    s = b * block
                    e = min((b + 1) * block, n_cur)
                    h_pert[s:e] += shift
        else:
            h_pert = best_h.copy()
            n_flips = n_cur // 15
            flip_idx = rng.choice(n_cur, n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_feasible(h_pert, target_cur)
        h_pert_bb = round_to_bangbang(h_pert, target_cur)
        hr, scr = bangbang_local_search(h_pert_bb, target_cur, max_iter=1000,
                                         start_time=start_time, time_limit=40.0)
        scr = compute_c5(hr)
        if tleft() < 60:
            if scr < best_score:
                best_score = scr
                best_h = hr
            continue
        h_ref_r, sc_rr = slsqp_optimize(hr, [1000, 5000, 20000],
                                         maxiter=1000, ftol=1e-15,
                                         start_time=start_time, time_limit=50.0)
        if sc_rr < best_score:
            best_score = sc_rr
            best_h = h_ref_r
            n_cur = len(best_h)
            target_cur = n_cur / 2.0

    # Phase 8: Final intensive bang-bang search
    if tleft() > 60:
        h_final_bb = round_to_bangbang(best_h, len(best_h) / 2.0)
        h_fin, sc_fin = bangbang_local_search(h_final_bb, len(best_h) / 2.0,
                                               max_iter=3000,
                                               start_time=start_time, time_limit=40.0)
        sc_fin = compute_c5(h_fin)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 9: Pair flips on best
    if tleft() > 30:
        h_pf2, sc_pf2 = pair_flip_search(best_h, len(best_h) / 2.0,
                                          start_time=start_time, time_limit=20.0,
                                          max_pairs=500)
        sc_pf2 = compute_c5(h_pf2)
        if sc_pf2 < best_score:
            best_score = sc_pf2
            best_h = h_pf2
        h_ls_f, sc_ls_f = bangbang_local_search(best_h, len(best_h) / 2.0,
                                                 max_iter=1000,
                                                 start_time=start_time, time_limit=15.0)
        sc_ls_f = compute_c5(h_ls_f)
        if sc_ls_f < best_score:
            best_score = sc_ls_f
            best_h = h_ls_f

    # Phase 10: Final SLSQP
    if tleft() > 20:
        h_last, sc_last = slsqp_optimize(best_h, [50000, 200000],
                                          maxiter=500, ftol=1e-16,
                                          start_time=start_time, time_limit=15.0)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
