One more thought: the `project_feasible` function is called very frequently (every FISTA iteration). Let me make it faster by reducing the number of iterations and using a smarter approach.

Actually, the projection is: given h in [0,1]^n, find h' in [0,1]^n with sum(h') = target that minimizes ||h' - h||^2. This is a quadratic program. The exact solution can be found by:
- If sum(h) = target, return h
- If sum(h) < target: we need to increase some values. The optimal is to increase the values that are furthest from 1 (i.e., the smallest values). But we also need to respect the upper bound.
- Similarly for sum(h) > target.

A simple and fast approach: use a threshold. Find a lambda such that sum(clip(h + lambda, 0, 1)) = target. This can be found by bisection.

```python
def project_feasible_fast(h, target_sum):

    lo = -max(abs(s - target_sum), 1.0)

        mid = (lo + hi) / 2
        h_try = np.clip(h + mid, 0.0, 1.0)
        s_try = np.sum(h_try)
        if s_try < target_sum:
            lo = mid
        else:
            hi = mid
    
    h_out = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final correction
    diff = target_sum - np.sum(h_out)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_out < 1.0 - 1e-15
        else:
            mask = h_out > 1e-15
        if np.sum(mask) > 0:

            h_out[mask] += diff / np.sum(mask)

    return h_out
```

```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):

    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target_sum - s
    if abs(diff) < 1e-14:
        return h
    lo = -max(abs(diff), 2.0)
    hi = max(abs(diff), 2.0)

            h_out[mask] += diff2 / nm
            h_out = np.clip(h_out, 0.0, 1.0)
    return h_out


def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')


def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = corr_array(h)
    c_max = np.max(corr)
    total = len(corr)
    if top_m is not None and top_m < total:
        top_idx = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(total)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_s = np.exp(shifted)
    ws = np.sum(exp_s)
    if ws < 1e-300:
        return np.zeros(n), corr
    w = exp_s / ws
    grad = np.zeros(n)
    nh = n - 1
    for i in range(len(w)):

        if wi < 1e-12:

            if n - m > 0:
                grad[:n - m] += wi * (1.0 - h[m:]) * dx
                grad[m:] -= wi * h[:n - m] * dx
    return grad, corr

def alternating_lp(h_init, n, target, max_iter, deadline, damping=0.7, reg=1e-4):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = 1.0 - h
        hv = cp.Variable(n)
        tv = cp.Variable()
        cons = [cp.sum(hv) == target, hv >= 0, hv <= 1]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                c = np.zeros(n)
                c[lo:hi] = g[lo - k:hi - k] * dx
                cons.append(c @ hv <= tv)
        obj = tv + reg * cp.sum_squares(hv - h)
        prob = cp.Problem(cp.Minimize(obj), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-6, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
            except Exception:
                break
        if hv.value is None:
            break
        h_new = np.clip(np.asarray(hv.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def slsqp_refine(h_start, n, target, alphas, maxiter, deadline):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n)}]
    h_cur = project_feasible(h_start.copy(), target)
    h_best = h_cur.copy()
    c5_best = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        def obj_a(h, a=a):
            return smooth_max_val(corr_array(h), a)
        def grad_a(h, a=a):
            g, _ = smooth_grad(h, a, dx)
            return g
        try:
            res = minimize(obj_a, h_cur, jac=grad_a, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new
                    h_cur = h_new
        except Exception:
            pass
    return h_best, c5_best


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline_total = start_time + budget_s - 10.0

    def tleft():
        return max(0.0, deadline_total - time.time())

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP at initial resolution (n=400)
    if tleft() > 200:
        dl = start_time + min(150, tleft() * 0.5)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=8000, deadline=dl,
                                     damping=0.7, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: FISTA with increasing alpha at n=400
    if tleft() > 250:
        for i, alpha in enumerate([50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]):
            if tleft() < 50:
                break
            tm = 50 if alpha < 1000 else 30
            dl = start_time + min(30, tleft() * 0.12)
            h_f, c5_f = fista_smo(best_h, n, target, alpha,
                                   max_iter=5000, deadline=dl, top_m=tm)
            if c5_f < best_score:
                best_score = c5_f
                best_h = h_f

    # Phase 3: SLSQP polish at n=400
    if tleft() > 150:
        dl = start_time + min(80, tleft() * 0.4)
        h_s, c5_s = slsqp_refine(best_h, n, target,
                                  [500, 1000, 3000, 8000, 20000, 50000],
                                  maxiter=800, deadline=dl)
        if c5_s < best_score:
            best_score = c5_s
            best_h = h_s

    # Phase 4: Multi-resolution coarse-to-fine
    for n_try in [600, 800, 1000, 1200]:
        if tleft() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        c5_cur = compute_c5(h_interp)

        # FISTA at new resolution
        if tleft() > 80:
            for alpha in [100, 500, 2000, 8000, 20000]:
                if tleft() < 40:
                    break
                tm = 50 if alpha < 2000 else 30
                dl = start_time + min(25, tleft() * 0.15)
                h_f, c5_f = fista_smo(h_interp, n_try, target_try, alpha,
                                       max_iter=3000, deadline=dl, top_m=tm)
                if c5_f < c5_cur:
                    h_interp = h_f
                    c5_cur = c5_f

        # SLSQP at new resolution
        if tleft() > 60:
            dl = start_time + min(40, tleft() * 0.25)
            h_s, c5_s = slsqp_refine(h_interp, n_try, target_try,
                                      [1000, 3000, 8000, 20000, 50000],
                                      maxiter=600, deadline=dl)
            if c5_s < c5_cur:
                h_interp = h_s
                c5_cur = c5_s

        if c5_cur < best_score:
            best_score = c5_cur
            best_h = h_interp

    # Phase 5: Perturbation restarts
    n_use = len(best_h)
    target_use = n_use / 2.0
    for noise_scale in [0.02, 0.05, 0.1, 0.15]:
        if tleft() < 50:
            break
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert = project_feasible(h_pert, target_use)
        dl1 = start_time + min(15, tleft() * 0.3)
        h_f, c5_f = fista_smo(h_pert, n_use, target_use, 2000,
                               max_iter=2000, deadline=dl1, top_m=30)
        if tleft() > 30 and c5_f < best_score + 0.001:
            dl2 = start_time + min(20, tleft() * 0.4)
            h_s, c5_s = slsqp_refine(h_f, n_use, target_use,
                                      [2000, 8000, 20000, 50000],
                                      maxiter=500, deadline=dl2)
            if c5_s < c5_f:
                c5_f = c5_s
                h_f = h_s
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Phase 6: Final polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        dl = start_time + min(tleft() * 0.8, 40)
        h_fin, c5_fin = slsqp_refine(best_h, n_use, target_use,
                                      [5000, 20000, 50000, 100000],
                                      maxiter=400, deadline=dl)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
