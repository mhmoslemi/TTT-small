import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def lp_active_step(h, n, target, k_active, trust_radius, solver='GLOP'):
    mid = n - 1
    f_vals = np.zeros(len(k_active))
    grads = np.zeros((len(k_active), n))
    for idx, k_idx in enumerate(k_active):
        k = int(k_idx) - mid
        if k >= 0:
            m = np.arange(0, n - k)
        else:
            m = np.arange(-k, n)
        f_vals[idx] = np.sum(h[m] * (1.0 - h[m + k]))
        idxs = np.arange(n)
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grads[idx, v1] += (1.0 - h[idxs[v1] - k])
        grads[idx, v2] -= h[idxs[v2] + k]

    d = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        cp.sum(d) == 0,
        h + d >= 0,
        h + d <= 1,
        d >= -trust_radius,
        d <= trust_radius,
    ]
    for idx in range(len(k_active)):
        constraints.append(grads[idx] @ d <= t - f_vals[idx])

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=solver)
        if d.value is not None and t.value is not None:
            d_val = np.array(d.value).flatten()
            if np.all(np.isfinite(d_val)):
                h_new = project_feasible(h + d_val, target)
                return h_new, float(t.value)
    except Exception:
        pass
    try:
        prob.solve(solver='ECOS')
        if d.value is not None and t.value is not None:
            d_val = np.array(d.value).flatten()
            if np.all(np.isfinite(d_val)):
                h_new = project_feasible(h + d_val, target)
                return h_new, float(t.value)
    except Exception:
        pass
    return None, None


def active_set_method(h, n, target, time_limit, max_iter=200,
                      n_active_frac=0.03, trust_init=0.5, trust_min=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trust = trust_init

    for iteration in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx

        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()

        threshold = c_max - n_active_frac * c_max
        active_mask = corr >= threshold
        k_active = np.where(active_mask)[0]
        if len(k_active) == 0:
            k_active = np.array([int(np.argmax(corr))])
        if len(k_active) > 25:
            top_idx = np.argsort(corr)[-25:]
            k_active = top_idx

        h_new, t_val = lp_active_step(h, n, target, k_active, trust)

        if h_new is not None:
            c5_new = compute_c5(h_new)
            if c5_new < c5_now - 1e-12:
                h = h_new
                trust = min(trust * 1.3, 0.8)
            else:
                trust *= 0.5
                if trust < trust_min:
                    break
        else:
            break

    return best_h, best_c5


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 150:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)

                v_jm = (j - shifts >= 0) & (j - shifts < n)

            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

def binarize_step(h, target, thresh_hi, thresh_lo):
    h_new = h.copy()
    h_new[h_new > thresh_hi] = 1.0
    h_new[h_new < thresh_lo] = 0.0
    return project_feasible(h_new, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def momentum_subgradient(h, n, target, time_limit, max_steps, mu=0.85, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

            best_h = h_lp

    # Phase 7: Multi-resolution - try higher n
    if tleft() > 250:
        for n_try in [500, 600]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [500, 2000, 10000], 1000, 1e-15)
            sc = compute_c5(h_hires)
            if sc < best_score:
                sc2, _ = fast_pair_cd(h_hires, n_try, target_try,
                                      time_limit=min(60, tleft() * 0.3),
                                      max_sweeps=2000, n_cand=40)
                if sc2 < sc:
                    sc = sc2
                    h_hires = h_hires
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 8: Final pair CD at best resolution
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(80, tleft() * 0.5)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: LP Active-Set final
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(50, tleft() * 0.5)
        h_lp, c5_lp = active_set_method(best_h, n_use, target_use,
                                        time_limit=lp_time,
                                        max_iter=150,
                                        n_active_frac=0.02,
                                        trust_init=0.3)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Momentum subgradient
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_mom, c5_mom = momentum_subgradient(best_h, n_use, target_use,
                                             time_limit=min(30, tleft() * 0.4),
                                             max_steps=20000, mu=0.9, base_lr=0.015)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 80000], 800, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final pair CD squeeze
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=min(15, tleft() * 0.6),
                                   max_sweeps=1000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
