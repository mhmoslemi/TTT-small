def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

            # For a given lag k, the overlap is:
            # O_k = sum_{j: 0<=j<n and 0<=j+k<n} h[j] * (1-h[j+k]) * dx
            # 
            # Terms where j = i (if i+k < n): h[i] * (1-h[i+k]) * dx
            # Terms where j+k = i (if i-k >= 0): h[i-k] * (1-h[i]) * dx
            # 
            # So O_k = [O_k without these two terms] + h[i]*(1-h[i+k])*dx + h[i-k]*(1-h[i])*dx
            # Let C_k = O_k without these two terms (constant w.r.t. h[i])
            # O_k = C_k + dx*h[i] - dx*h[i]*h[i+k] + dx*h[i-k] - dx*h[i]*h[i-k]
            #     = (C_k + dx*h[i-k]) + dx*h[i]*(1 - h[i+k] - h[i-k])
            
            # This is linear in h[i]! The coefficient of h[i] is:
            #   a_k = dx*(1 - h[i+k] - h[i-k]) (with appropriate bounds)
            # And the constant is:
            #   b_k = C_k + dx*h[i-k]
            
            # The constraint is O_k <= t for all k in active set.
            # Each gives a linear constraint on h[i].
            # We want to minimize t = max_k O_k(h[i]), subject to 0<=h[i]<=1.
            # Also need to maintain sum(h) = target, which means:
            #   sum(h) - h[i] + h[i]_new = target
            #   h[i]_new = h[i] (since we're only changing h[i] and compensating)
            # Actually, the sum constraint means we can't freely change h[i] without
            # adjusting another coordinate. This makes coordinate descent more complex.
            # 
            # Alternative: use a penalty for the sum constraint.
            # Or: after changing h[i], project back to feasibility.
            
            # Let me just try small changes and accept if it improves.
            best_local = old_val
            best_local_c5 = best_c5
            for delta in [-0.05, -0.02, -0.01, 0.01, 0.02, 0.05, -0.005, 0.005]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h_try = h.copy()
                h_try[i] = new_val
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_local_c5 - 1e-12:
                    best_local_c5 = sc
                    best_local = new_val
            
            if best_local != old_val:
                h[i] = best_local
                h = project_feasible(h, target)
                if best_local_c5 < best_c5 - 1e-14:
                    best_c5 = best_local_c5
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def subgradient_refine(h_init, n, target, time_limit, n_top=30, momentum_init=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()

        y = np.clip(y, 0.0, 1.0)

        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half

        w = 0.85 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            if k >= 0:
                mask1 = idxs + k < n
                grad[idxs[mask1]] += w[i] * (1.0 - y[idxs[mask1] + k])

            else:

                grad[idxs[mask1]] += w[i] * (1.0 - y[idxs[mask1] - k])

def smooth_logsumexp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_smooth(h_start, n, target, alphas, maxiter):
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(
                lambda x: smooth_logsumexp_obj(x, alpha),
                h_cur,
                method='SLSQP',
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_local_search(h_init, n, target, time_limit, max_iters=200000, seed=42):
