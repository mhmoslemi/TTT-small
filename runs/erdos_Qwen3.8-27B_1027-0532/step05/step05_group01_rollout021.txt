def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

        except Exception:
            pass

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-15):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

            pass
    return h_cur


def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40, n_bisect=14):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 80:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi = h[i]
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g_i = np.zeros(n_corr)
            g_i[v_ip] -= (1.0 - h[i + shifts[v_ip]])
            g_i[v_im] += h[i - shifts[v_im]]

            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(hi, 1.0 - h[j])
                if delta_hi < 1e-12:
                    continue

                v_jp = (j + shifts >= 0) & (j + shifts < n)
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g = g_i.copy()
                g[v_jp] += (1.0 - h[j + shifts[v_jp]])
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = 0.0
                b_d = delta_hi
                for _ in range(n_bisect):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    f_prime = np.max(g[active]) if active.any() else 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 30:
                break

    return best_h, best_c5

def penalty_target_method(h_init, n, target, time_limit, T_init, T_target, max_steps=2000):
    start_time = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)
    mid = n - 1

    T_lo = T_target
    T_hi = T_init
    n_bisect_T = 8

    for T_iter in range(n_bisect_T):
        if time.time() - start_time > time_limit:
            break
        T_mid = 0.5 * (T_lo + T_hi)
        T_corr = T_mid / dx

        h_cur = h.copy()
        lr = 0.05
        for step in range(max_steps):
            if time.time() - start_time > time_limit:
                break
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            c5_now = float(np.max(corr)) * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_cur.copy()

            violations = np.maximum(corr - T_corr, 0.0)
            if violations.max() < 1e-8:
                break

            grad = np.zeros(n)
            active_idx = np.where(corr > T_corr + 1e-10)[0]
            for s_idx in active_idx:
                k = int(s_idx) - mid
                w = (corr[int(s_idx)] - T_corr)
                v1 = (idxs - k >= 0) & (idxs - k < n)
                v2 = (idxs + k >= 0) & (idxs + k < n)
                grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
                grad[v2] -= w * h_cur[idxs[v2] + k]

            grad *= 2.0 * dx
            grad -= np.mean(grad)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break

            improved = False
            for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
                lr_try = lr * lr_mult
                h_try = project_feasible(h_cur - lr_try * grad, target)
                corr_try = np.correlate(h_try, 1.0 - h_try, mode='full')
                pen_try = np.sum(np.maximum(corr_try - T_corr, 0.0) ** 2)
                pen_old = np.sum(violations ** 2)
                if pen_try < pen_old:
                    h_cur = h_try
                    lr = lr_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-6:
                    break

        c5_test = compute_c5(h_cur)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_cur.copy()

        if c5_test <= T_mid + 1e-5:
            T_lo = T_mid
            h = h_cur.copy()
        else:
            T_hi = T_mid

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation_search(h_init, n, target, max_tries, time_limit, seed_val=99999):
    start = time.time()
    rng_local = np.random.RandomState(seed_val)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.02, 0.05, 0.1]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to get a good starting point
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [500, 2000, 10000],
                            1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Aggressive pair coordinate descent (main workhorse)
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=6000, n_cand=45, n_bisect=14)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish after CD
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [5000, 20000, 80000],
                            1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Another round of pair CD with different params
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=5000, n_cand=55, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Random pair CD for diversity
    if tleft() > 100:
        rp_time = min(80, tleft() * 0.15)
        h_rp, c5_rp = random_pair_cd(best_h, n, target,
                                     time_limit=rp_time,
                                     max_sweeps=4000, n_bisect=12)
        if c5_rp < best_score:
            best_score = c5_rp
            best_h = h_rp

    # Phase 6: SLSQP with very high alpha
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target,
                            [20000, 100000, 200000],
                            1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Penalty target method
    if tleft() > 100:
        pen_time = min(120, tleft() * 0.2)
        T_init = best_score + 0.005
        T_target = best_score - 0.005
        h_pen, c5_pen = penalty_target_method(best_h, n, target,
                                              time_limit=pen_time,
                                              T_init=T_init,
                                              T_target=T_target,
                                              max_steps=1000)
        if c5_pen < best_score:
            best_score = c5_pen
            best_h = h_pen

    # Phase 8: Multi-restart with perturbations + pair CD
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.08, 0.15]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        # Quick pair CD from perturbed start
        cd_t = min(60, tleft() * 0.15)
        h_try, c5_try = fast_pair_cd(h_pert, n, target,
                                     time_limit=cd_t,
                                     max_sweeps=2000, n_cand=35, n_bisect=12)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try

    # Phase 9: Try slightly higher resolution
    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [500, 2000, 10000],
                                  1000, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 80 and sc < best_score + 0.001:
                cd_t = min(80, tleft() * 0.3)
                h_hires2, sc2 = fast_pair_cd(h_hires, n_try, target_try,
                                             time_limit=cd_t,
                                             max_sweeps=3000, n_cand=40, n_bisect=13)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_hires2
                    n = n_try
                    target = target_try

    # Phase 10: Final pair CD at best resolution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(60, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=50, n_bisect=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000, 200000],
                              800, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
