def compute_grad_max_c5(h, dx, top_k=5):
    """Compute subgradient of max_k overlap_k, weighted by top_k overlaps."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    # Get top k lags
    order = np.argsort(corr)[::-1][:top_k]
    # Weight: exponential of normalized difference
    max_val = corr[order[0]]
    if max_val == 0:
        return np.zeros(n)
    weights = np.exp((corr[order] - max_val) / (0.01 * max_val + 1e-10))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, order):
        k = int(k)
        if k == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        else:
            jp = np.arange(0, n - k)
            grad[jp] += w * (1.0 - h[jp + k])
            jn = np.arange(k, n)
            grad[jn] -= w * h[jn - k]
    grad *= dx
    return grad

        if obj_new < best_obj:

                accepted += 1
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
        
        T *= cooling
    
    return best_h, best_score

                cur_score = compute_c5(h_cur)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def subgrad_true_max(h, dx, top_k=5, temp_scale=0.02):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    max_val = float(np.max(corr))
    if max_val < 1e-15:
        return np.zeros(n)
    order = np.argsort(corr)[::-1][:top_k]
    w = np.exp((corr[order] - max_val) / max(temp_scale * max_val, 1e-12))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, order):
        ki = int(ki)
        if ki == 0:
            grad[:] += wi * (1.0 - 2.0 * h)
        else:
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h[jn - ki]
    grad *= dx
    return grad

def optimize_direct(h0, target, deadline, top_k=5, temp_scale=0.02):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_obj = compute_c5(x)
    lr = 2.0 / (np.linalg.norm(subgrad_true_max(x, dx, top_k, temp_scale)) + 1e-8)
    lr = min(lr, 10.0)
    prev_grad = None
    prev_x = x.copy()
    no_improve = 0
    
    for it in range(500000):
        if time.time() > deadline:
            break
        g = subgrad_true_max(x, dx, top_k, temp_scale)
        if prev_grad is not None:
            sy = np.dot(x - prev_x, g - prev_grad)
            yy = np.dot(g - prev_grad, g - prev_grad)
            if yy > 1e-15:
                lr_bb = sy / yy
                if lr_bb > 1e-6 and lr_bb < 50.0:
                    lr = 0.5 * lr + 0.5 * lr_bb
        x_prev = x.copy()
        x_new = project_feasible(x - lr * g, target)
        obj_new = compute_c5(x_new)
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                step_scale = 0.5 ** min(no_improve // 100, 5)
                x_new = project_feasible(x_prev - lr * step_scale * g, target)

                lr *= 0.8
                no_improve = 0
        v = 0.9 * v + 0.1 * (x_new - x)
        x = x_new
        prev_grad = g
        prev_x = x_prev
    
    return best_x, best_obj

def anneal_explore(h0, target, deadline, T_init=0.1, cool=0.965):
    n = len(h0)
    h_cur = h0.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    rng = np.random.RandomState(314)
    T = T_init
    T_min = 0.001
    
    while T > T_min and time.time() < deadline:
        block_len = rng.randint(2, 15)
        start = rng.randint(0, max(1, n - block_len))
        end = min(start + block_len, n)
        h_new = h_cur.copy()
        h_new[start:end] += rng.randn(end - start) * T
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T * 0.003, 1e-8)):
            h_cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        T *= cool
    return best_h, best_score

def coord_fine(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(555)
    improvement_streak = 0
    
    while time.time() < deadline:
        improvement_streak = 0
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old = h_cur[idx]
            if old < 5e-5:
                trials = [0.0, 0.01, 0.05]
            elif old > 1.0 - 5e-5:
                trials = [1.0, 0.99, 0.95]
            else:
                trials = [old * 0.2, old * 0.5, old * 0.8, old * 1.25, old * 1.5, 1.0 - old]
            best_trial = old
            best_sc = cur_score
            for tr in trials:
                tr = float(np.clip(tr, 0.0, 1.0))
                if abs(tr - old) < 1e-10:
                    continue
                h_cur[idx] = tr
                h_tmp = project_feasible(h_cur, target)
                sc = compute_c5(h_tmp)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_trial = tr
            if best_trial != old:
                h_cur[idx] = best_trial
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                improvement_streak += 1
            else:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
        if improvement_streak == 0:
            break
    return h_cur, cur_score

def multi_start(fit_func, h_best, target, deadline, n_starts, seed):
    n = len(h_best)
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_score = compute_c5(best_h)
    per_start_time = max(10, (deadline - time.time()) / n_starts)
    for i in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.04 * (i + 1) ** 0.7
        h_p = best_h + rng.randn(n) * scale
        h_p = project_feasible(h_p, target)
        d = time.time() + per_start_time
        h_new, sc = fit_func(h_p, target, d)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score
