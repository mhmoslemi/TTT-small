            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        overlaps = compute_overlaps(h_cur)
        c_curr = float(np.max(overlaps)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(overlaps)[::-1][:n_active]
        diffs = overlaps[top_idx] - overlaps[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if n - k > 0:
                grad[0:n-k] += wk * (1.0 - h_cur[k:n])
            if k > 0:
                grad[k:n] -= wk * h_cur[0:n-k]
        grad *= dx
        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_dx
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 20.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA at multiple temperatures
    for alpha in [50, 200, 800, 3000, 10000]:
        if tleft() < 60:
            break
        h_f, sc_f = fista_optimize(best_h, n_final, target_final, alpha,
                                   max_iter=2000, deadline=min(start_time + 60 + 40 * alpha / 50, hard_deadline))
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 2: SLSQP polishing
    if tleft() > 120:
        h_s, sc_s = slsqp_optimize(best_h, n_final, target_final,
                                   [100, 500, 2000, 10000], 3000, 1e-16,
                                   min(start_time + 200, hard_deadline))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 3: FISTA again with lower temperature
    if tleft() > 100:
        for alpha in [5000, 20000, 50000]:
            if tleft() < 40:
                break
            h_f, sc_f = fista_optimize(best_h, n_final, target_final, alpha,
                                       max_iter=1500, deadline=min(start_time + 300, hard_deadline))
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f

    # Phase 4: Coordinate descent
    if tleft() > 80:
        h_cd, sc_cd = coordinate_descent(best_h, n_final, target_final,
                                         n_sweeps=30, deadline=min(start_time + 380, hard_deadline))
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 5: SLSQP with very tight tolerances
    if tleft() > 100:
        h_s, sc_s = slsqp_optimize(best_h, n_final, target_final,
                                   [2000, 10000, 50000, 200000], 5000, 1e-18,
                                   min(start_time + 500, hard_deadline))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 6: Subgradient polish
    if tleft() > 80:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                         500, 5, 0.002, 0.997,
                                         min(start_time + 580, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if tleft() > 50:
            h_s, sc_s = slsqp_optimize(best_h, n_final, target_final,
                                       [5000, 20000, 100000], 3000, 1e-17,
                                       hard_deadline)
            if sc_s < best_score:
                best_score = sc_s
                best_h = h_s

    # Phase 7: Pair swap search
    if tleft() > 60:
        h_ps, sc_ps = pair_swap_search(best_h, n_final, target_final,
                                       n_iter=5000, deadline=min(start_time + 640, hard_deadline), rng=rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps

    # Phase 8: Multiple random restarts
    if tleft() > 100:
        for attempt in range(10):
            if tleft() < 50:
                break
            scale = 0.01 + 0.01 * attempt
            h_pert = project_feasible(best_h + rng.randn(n_final) * scale, target_final)
            # Quick FISTA
            h_f, sc_f = fista_optimize(h_pert, n_final, target_final, 5000,
                                       max_iter=800, deadline=hard_deadline)
            # Quick SLSQP
            if tleft() > 30:
                h_s, sc_s = slsqp_optimize(h_f, n_final, target_final,
                                           [2000, 10000, 50000], 2000, 1e-17, hard_deadline)
                if sc_s < best_score:
                    best_score = sc_s
                    best_h = h_s
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f

    # Phase 9: Final coordinate descent
    if tleft() > 40:
        h_cd, sc_cd = coordinate_descent(best_h, n_final, target_final,
                                         n_sweeps=20, deadline=hard_deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 20:
        h_s, sc_s = slsqp_optimize(best_h, n_final, target_final,
                                   [10000, 50000, 200000], 2000, 1e-18, hard_deadline)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
