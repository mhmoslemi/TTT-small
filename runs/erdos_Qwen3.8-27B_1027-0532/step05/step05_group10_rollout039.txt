def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, alpha, target, n):
    """Log-sum-exp objective and its gradient"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    max_scaled = np.max(scaled)
    log_sum_exp = max_scaled + np.log(np.sum(np.exp(scaled - max_scaled)))
    f = log_sum_exp / alpha
    
    # Softmax weights
    p = np.exp(scaled - max_scaled) / np.sum(np.exp(scaled - max_scaled))

    start = time.time()

    f += (rho / 2.0) * sum_diff ** 2
    grad += rho * sum_diff  # this adds rho * sum_diff to each component
    
    return f, grad

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_grad(x, alpha, n, rho, target):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * corr
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    sum_exp = np.sum(exp_s)
    f = (max_s + np.log(sum_exp)) / alpha
    p = exp_s / sum_exp
    half_n = n - 1
    grad = np.zeros(n)
    active = np.where(p > 1e-8)[0]
    for ki in active:
        m = int(ki) - half_n
        w = p[ki]
        if m >= 0:
            lo1, hi1 = m, n
            lo2, hi2 = 0, n - m
        else:
            lo1, hi1 = 0, n + m
            lo2, hi2 = -m, n
        if hi1 > lo1:
            grad[lo1:hi1] += w * dx * (1.0 - x[lo1-m:hi1-m])
        if hi2 > lo2:
            grad[lo2:hi2] -= w * dx * x[lo2+m:hi2+m]
    sd = np.sum(x) - target
    f += 0.5 * rho * sd * sd
    grad += rho * sd
    return f, grad

def lbfgs_opt(h0, n, target, alphas, maxiter, ftol, tlimit, rho=5000.0):
    ts = time.time()
    bounds = [(0.0, 1.0)] * n
    bh = h0.copy()
    bs = compute_c5(h0)
    for a in alphas:
        if time.time() - ts > tlimit:
            break
        res = minimize(lambda x: lse_obj_grad(x, a, n, rho, target),
                      bh, method='L-BFGS-B', jac=True, bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        hn = project_feasible(res.x, target)
        s = compute_c5(hn)
        if s < bs:
            bs = s
            bh = hn
    return bh, bs

def bcd_refine(h0, n, target, tlimit, max_rounds=300):
    ts = time.time()
    dx = 2.0 / n
    hc = h0.copy()
    bh = hc.copy()
    bs = compute_c5(hc)
    hn = n - 1
    for r in range(max_rounds):
        if time.time() - ts > tlimit:
            break
        imp = False
        for j in range(n):
            if time.time() - ts > tlimit:
                break
            corr = np.correlate(hc, 1.0 - hc, mode='full')
            mv = np.max(corr)
            oh = hc[j]
            # Try moving h[j]
            ks = int(np.argmax(corr))
            mk = ks - hn
            gk = 0.0
            if 0 <= j - mk < n:
                gk += 1.0 - hc[j - mk]
            if 0 <= j + mk < n:
                gk -= hc[j + mk]
            if abs(gk) < 1e-12:
                continue
            d = -np.sign(gk)
            best_t = oh
            best_mv = mv
            for st in [0.1, 0.3, 0.5, 1.0]:
                t = oh + d * st
                t = float(np.clip(t, 0.0, 1.0))
                if abs(t - oh) < 1e-10:
                    break
                dl = t - oh
                nms = 0.0
                for ki in range(len(corr)):
                    m = ki - hn
                    g = 0.0
                    if 0 <= j - m < n:
                        g += 1.0 - hc[j - m]
                    if 0 <= j + m < n:
                        g -= hc[j + m]
                    v = corr[ki] + dl * g
                    if v > nms:
                        nms = v
                if nms < best_mv:
                    best_mv = nms
                    best_t = t
            if best_t != oh:
                hc[j] = best_t
                imp = True
        hc = project_feasible(hc, target)
        s = compute_c5(hc)
        if s < bs - 1e-14:
            bs = s
            bh = hc.copy()
        elif r > 15 and not imp:
            break
    return bh, bs
