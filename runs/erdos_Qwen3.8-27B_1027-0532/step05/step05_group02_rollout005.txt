def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h, 0, 1).astype(float)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Water-filling projection
    if diff > 0:
        # Need to add mass
        order = np.argsort(1.0 - h)  # prioritize those closer to 1

        remaining = diff

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_binary(h, target_sum):
    h_bin = np.round(h).astype(int)
    h_bin = np.clip(h_bin, 0, 1)
    cur = int(h_bin.sum())
    if cur == target_sum:
        return h_bin.astype(float)
    if cur > target_sum:
        to_flip = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h value (least "committed")
        order = np.argsort(h[to_flip])
        h_bin[to_flip[order[:cur-target_sum]]] = 0
    else:
        to_flip = np.where(h_bin == 0)[0]
        order = np.argsort(-h[to_flip])
        h_bin[to_flip[order[:target_sum-cur]]] = 1
    return h_bin.astype(float)

def smooth_obj(h, alpha):
    c = full_corr(h)
    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp(alpha*(c - cmax)))))/alpha

def smooth_grad(h, alpha):
    n = len(h)
    c = full_corr(h)
    cmax = np.max(c)
    exp_c = np.exp(alpha * (c - cmax))
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        m = k - (n-1)
        pk = p[k]
        if pk < 1e-18:
            continue
        if m >= 0:
            # j from m to n-1: add pk*(1-h[j-m])
            if m < n:
                grad[m:] += pk * (1.0 - h[:n-m])
            # j from 0 to n-1-m: subtract pk*h[j+m]
            if n-1-m >= 0:
                grad[:n-m] -= pk * h[m:]
        else:
            ma = -m
            # j from 0 to n-1-ma: add pk*(1-h[j+ma])
            if n-1-ma >= 0:
                grad[:n-ma] += pk * (1.0 - h[ma:])
            # j from ma to n-1: subtract pk*h[j-ma]
            grad[ma:] -= pk * h[:n-ma]
    return grad

- Objective: smooth_obj(h, alpha)
- Gradient: smooth_grad(h, alpha)
- Constraint: sum(h) = n/2 (equality)
- Bounds: 0 ≤ h[i] ≤ 1

SLSQP can handle this. The issue is that SLSQP might be slow for n=600. Let me test with a time limit.

Alternative: I can use the projected gradient method (FISTA) which is simpler and might converge faster for this problem.

FISTA with projection:
1. Compute gradient g = smooth_grad(h, alpha)
2. Take a step: y = x - (1/L) * g where L is the Lipschitz constant
3. Project: x_new = project_to_sum(y, target)
4. Momentum update

The Lipschitz constant L can be estimated via power iteration or just set conservatively.

For the quadratic part of c_k, the Hessian is:
∂²c_k/∂h[j]∂h[l] = -δ(j,l+m) - δ(j,l-m) + ... (it's a banded matrix)

The eigenvalues are bounded by 2 (since each h[j] appears in at most 2 terms of R(m)). So for the smoothed objective with parameter α, the Lipschitz constant is bounded by 2α.

Let me use step size 1/(2α) for the gradient step.

Let me also consider using L-BFGS-B with a penalty for the constraint instead of SLSQP. The penalty method:
minimize obj(h) + μ * (sum(h) - target)²
subject to 0 ≤ h ≤ 1

This can be solved with L-BFGS-B (which is faster than SLSQP). I increase μ over time to enforce the constraint.

Or even better: I can use the fact that the constraint is linear. After each L-BFGS-B step, I project onto the constraint set (sum = target, 0 ≤ h ≤ 1). This is the "projected gradient" approach.

Let me use a combination:
1. L-BFGS-B on the smooth objective (ignoring the sum constraint, just using bounds)
2. After convergence (or time limit), project to satisfy the sum constraint
3. Repeat

Actually, this might not work well because the optimizer might drift far from the constraint.

Let me use a penalty approach:
obj_pen(h) = smooth_obj(h, alpha) + μ * (sum(h) - target)²

The gradient is:
∂obj_pen/∂h = smooth_grad(h, alpha) + 2μ * (sum(h) - target) * 1

This is easy to compute. And I can use L-BFGS-B with bounds [0,1] for each h[i].

I'll start with a small μ and increase it. This should work well.

        order = np.argsort(-h_work[zeros])

            zeros = np.where(h_bin == 0)[0]
            h_bin[zeros[:target_sum-final_sum]] = 1

def smooth_obj_and_grad(h, alpha, mu=0.0):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    obj = (cmax + np.log(np.sum(np.exp(alpha*(c - cmax)))))/alpha
    
    if mu > 0:
        s = np.sum(h)
        target = len(h) / 2.0
        obj += mu * (s - target)**2
    
    exp_c = np.exp(alpha * (c - cmax))
    p = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        m = k - (n-1)
        pk = p[k]
        if pk < 1e-18:
            continue
        if m >= 0:
            if m < n:
                grad[m:] += pk * (1.0 - h[:n-m])
            if n-1-m >= 0:
                grad[:n-m] -= pk * h[m:]
        else:
            ma = -m
            if n-1-ma >= 0:
                grad[:n-ma] += pk * (1.0 - h[ma:])
            grad[ma:] -= pk * h[:n-ma]
    
    if mu > 0:
        s = np.sum(h)
        target = len(h) / 2.0
        grad += 2*mu*(s - target)
    
    return obj, grad

def slsqp_refine(h_start, alpha, mu, maxiter, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x):
        val, _ = smooth_obj_and_grad(x, alpha, mu)
        return val
    
    def grad_func(x):
        _, g = smooth_obj_and_grad(x, alpha, mu)
        return g
    
    try:
        res = minimize(obj_func, h_start.copy(), jac=grad_func,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12,
                                'maxfun': maxiter*10})
        h_new = project_to_sum(res.x, target)
        return h_new
    except Exception:
        return h_start
