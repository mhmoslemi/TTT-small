def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)

def compute_delta(h, i, j):

    k = np.arange(2*n-1) - (n-1)

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_init(n, target, rng, style):
    h = np.zeros(n)
    if style == 0:
        # Alternating blocks
        block = max(4, n // 60)
        for i in range(n):
            block_idx = i // block
            h[i] = 1.0 if block_idx % 2 == 0 else 0.0
    elif style == 1:
        # Golden ratio based
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n):
            x = (i * phi) % 1.0
            h[i] = 1.0 if x < 0.5 else 0.0
    elif style == 2:
        # Smooth transition
        x = np.linspace(0, 1, n)
        h = (np.sin(2 * np.pi * x) > 0).astype(float)
    elif style == 3:
        # Periodic with period 3
        for i in range(n):
            h[i] = 1.0 if i % 3 < 2 else 0.0
    else:
        # Random
        h = (rng.random(n) > 0.5).astype(float)
    return project_binary(h, target)

    h_s, c5_s = sa_reheat(h_g, n_sa, 0.008, 0.00003, rng,
                           time_limit=allowed2, start_time=start_time)
    if c5_s < c5_g:
        return h_s, c5_s
    return h_g, c5_g

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2
    dx_init = 2.0 / n_init

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Initial greedy
    allowed = min(40.0, tleft() * 0.08)
    h_g, c5_g = greedy_improved(best_h, max_iters=30000, rng=rng,
                                 time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Long SA with reheat
    allowed = min(300.0, tleft() * 0.25)
    n_sa = int(min(8000000, allowed * 40000))
    h_sa, c5_sa = sa_reheat(best_h, n_sa, 0.025, 0.00008, rng,
                             time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Block-move SA
    allowed = min(150.0, tleft() * 0.15)
    n_sb = int(min(3000000, allowed * 25000))
    h_sb, c5_sb = sa_block_moves(best_h, n_sb, 0.015, 0.0001, rng,
                                  time_limit=allowed, start_time=start_time)
    if c5_sb < best_score:
        best_score = c5_sb
        best_h = h_sb.copy()

    # Phase 4: Iterated local search
    allowed = min(250.0, tleft() * 0.25)
    if allowed > 50:
        h_ils, c5_ils = iterated_local_search(best_h, rng, allowed, start_time, max_rounds=15)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

    # Phase 5: Multi-start with structured initializations
    if tleft() > 80:
        for style in range(4):
            if tleft() < 40:
                break
            h_struct = structured_init(n_init, target_init, rng, style)
            # Short greedy
            allowed_s = min(15.0, tleft() * 0.1)
            h_sg, c5_sg = greedy_improved(h_struct, max_iters=5000, rng=rng,
                                           time_limit=allowed_s, start_time=start_time)
            # Short SA
            if tleft() > 30:
                allowed_s2 = min(20.0, tleft() * 0.12)
                n_sas = int(min(300000, allowed_s2 * 20000))
                h_ss, c5_ss = sa_reheat(h_sg, n_sas, 0.015, 0.0001, rng,
                                         time_limit=allowed_s2, start_time=start_time)
                if c5_ss < c5_sg:
                    h_sg = h_ss
                    c5_sg = c5_ss
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg.copy()

    # Phase 6: Finer resolution
    if tleft() > 100:
        n_fine = n_init + 100
        if n_fine % 2 != 0:
            n_fine += 1
        target_fine = n_fine // 2
        h_f, c5_f = try_finer_resolution(best_h, n_fine, target_fine, rng,
                                          min(80.0, tleft()) - 10, start_time)
        if c5_f < best_score:
            best_score = c5_f
            # Map back to original resolution
            h_mapped = interpolate_to(h_f, n_init, target_init)
            h_bin_mapped = project_binary(h_mapped, target_init)
            allowed_p = min(20.0, tleft() * 0.08)
            h_mp, c5_mp = greedy_improved(h_bin_mapped, max_iters=5000, rng=rng,
                                           time_limit=allowed_p, start_time=start_time)
            if c5_mp < best_score:
                best_score = c5_mp
                best_h = h_mp.copy()
            else:
                best_h = h_mp.copy() if c5_mp < compute_c5(best_h) else best_h
                best_score = min(best_score, c5_mp)

    # Phase 7: Continuous refinement
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 100, 300, 1000, 3000, 10000]
        allowed = min(tleft() - 20, 200)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-14,
                                  time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final SA burst
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(2000000, max(20000, (tleft() - 12) * 50000)))
        h_sa_f, c5_f = sa_reheat(h_bin_final, n_sa_f, 0.006, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 9: Final greedy polish
    if tleft() > 10:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_f2 = project_binary(best_h, target_use)
        h_pol, c5_pol = greedy_improved(h_bin_f2, max_iters=10000, rng=rng,
                                         time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if c5_pol < best_score:
            best_score = c5_pol
            best_h = h_pol

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
