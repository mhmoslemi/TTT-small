**New idea: Use the dual problem**

The primal is: min_h max_s c_s(h) subject to h ∈ [0,1]^n, sum(h) = target.

By minimax duality (not exact since the problem is non-convex in h, but as a heuristic):
max_w min_h (sum_s w_s c_s(h)) subject to w ≥ 0, sum(w) = 1, h ∈ [0,1]^n, sum(h) = target.

For a fixed w, the inner problem is:
min_h sum_s w_s c_s(h) = min_h sum_s w_s [sum_i h[i](1-h[i+s])]
= min_h sum_i h[i] (1 - sum_s w_s h[i+s])

This is still bilinear (h[i] * h[i+s]). But if we fix the "other" h values, it's quadratic in each h[i] individually.

Alternatively, we can use an alternating approach:
- Fix w, minimize over h (inner problem)
- Update w based on which shifts are active (outer problem)

The inner problem min_h sum_i h[i](1 - sum_s w_s h[i+s]) is a quadratic program! (Since sum_s w_s h[i+s] is linear in h for fixed i, the product h[i] * (linear in h) is quadratic.)

Wait, it's not exactly quadratic because h[i] appears in multiple terms. Let me write it out:

sum_i h[i] (1 - sum_s w_s h[(i+s) mod n])
= sum_i h[i] - sum_i sum_s w_s h[i] h[(i+s) mod n]
= sum_i h[i] - sum_i sum_j w_{j-i} h[i] h[j]  (reindexing)

This is a quadratic form: h^T Q h + b^T h where Q is a circulant-like matrix.

Actually, this IS a quadratic program! We can solve it with cvxpy (if Q is PSD) or with a QP solver.

Is Q PSD? Q[i,j] = -w_{j-i} (with appropriate indexing). The eigenvalues of a circulant matrix are the DFT of its first row. The first row of Q is related to w. If w is a probability vector, the DFT of w is non-negative at frequency 0 but can be negative at other frequencies. So Q is not necessarily PSD.

If Q is not PSD, the QP is non-convex. But we can still try to solve it (e.g., with a local QP solver).

            g = np.zeros(n)
            v1 = (idxs + k >= 0) & (idxs + k < n)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            g[v1] += (1.0 - h_cur[idxs[v1] + k])
            g[v2] -= h_cur[idxs[v2] - k]
            grad += g
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        t_val = t_new

        lr = 2.0 / (t_new * t_new)
        y_new = project_feasible(y - lr * grad, target)
        h_new = y_new + beta * (y_new - h_cur)
        h_new = project_feasible(h_new, target)
        h_cur = y_new
        y = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def try_structured_init(n, target, seed):
    rng = np.random.RandomState(seed)
    initializations = []

    for freq in [3, 5, 7, 11, 13]:
        x = np.arange(n)
        h_sino = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
        h_sino = project_feasible(h_sino, target)
        initializations.append(h_sino)

    for block_size in [4, 8, 16]:
        h_block = np.zeros(n)
        for i in range(0, n, 2 * block_size):
            h_block[i:i + block_size] = 1.0
        h_block = project_feasible(h_block, target)
        h_block = h_block + rng.randn(n) * 0.1
        h_block = project_feasible(h_block, target)
        initializations.append(h_block)

    h_rand = project_feasible(0.5 + rng.randn(n) * 0.3, target)
    initializations.append(h_rand)

    return initializations


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to establish good local minimum
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [200, 1000, 5000], 800, 1e-14,
                            time_limit=min(80, tleft() - 20), start_time=start_time)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Active-set LP cutting plane
    if tleft() > 300:
        cp_time = min(350, tleft() * 0.35)
        h_cp, c5_cp = cutting_plane_with_linesearch(
            best_h, n, target, time_limit=cp_time,
            top_k_schedule=[5, 10, 15, 20, 25, 15, 10, 5])
        if c5_cp < best_score:
            best_score = c5_cp
            best_h = h_cp

    # Phase 3: SLSQP after cutting plane
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [500, 2000, 10000], 1000, 1e-14,
                            time_limit=min(80, tleft() - 20), start_time=start_time)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Pair CD for fine-tuning
    if tleft() > 150:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=8000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Another round of cutting plane
    if tleft() > 200:
        cp_time = min(180, tleft() * 0.3)
        h_cp2, c5_cp2 = cutting_plane_with_linesearch(
            best_h, n, target, time_limit=cp_time,
            top_k_schedule=[3, 5, 8, 12, 10, 5, 3])
        if c5_cp2 < best_score:
            best_score = c5_cp2
            best_h = h_cp2

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000], 1500, 1e-15,
                            time_limit=min(60, tleft() - 15), start_time=start_time)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [200, 1000, 5000], 800, 1e-14,
                                  time_limit=min(80, tleft() - 30), start_time=start_time)
            sc = compute_c5(h_hires)
            if tleft() > 60:
                cd_t = min(40, tleft() - 40)
                h_cd2, c5_cd2 = fast_pair_cd(h_hires, n_try, target_try,
                                             time_limit=cd_t, max_sweeps=3000, n_cand=40)
                if c5_cd2 < sc:
                    sc = c5_cd2
                    h_hires = h_cd2
            if sc < best_score:
                best_score = sc
                best_h = h_hires

    # Phase 8: Multi-start with structured initializations at best resolution
    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        rng = np.random.RandomState(seed)
        inits = try_structured_init(n_use, target_use, seed)
        for h_s in inits[:3]:
            if tleft() < 60:
                break
            h_sl = slsqp_round(h_s, n_use, target_use,
                               [200, 1000, 5000], 600, 1e-14,
                               time_limit=min(40, tleft() - 30), start_time=start_time)
            sc = compute_c5(h_sl)
            if sc < best_score:
                best_score = sc
                best_h = h_sl

    # Phase 9: Final cutting plane
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cp_time = min(60, tleft() * 0.5)
        h_cp3, c5_cp3 = cutting_plane_with_linesearch(
            best_h, n_use, target_use, time_limit=cp_time,
            top_k_schedule=[3, 5, 8, 5, 3])
        if c5_cp3 < best_score:
            best_score = c5_cp3
            best_h = h_cp3

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 100000], 1000, 1e-15,
                              time_limit=min(25, tleft() - 15), start_time=start_time)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
