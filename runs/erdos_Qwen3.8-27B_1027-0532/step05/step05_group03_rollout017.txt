And neg_k + common_k = #{m < n-k : h[m+k]=1}.

For the full correlation, the C₅ value is max(max_k pos_k, max_k neg_k, lag_0_value) * dx.
Where lag_0_value = #{m : h[m]=1, h[m]=0} = 0 for binary h.

So for binary h, C₅ = max(max_k pos_k, max_k neg_k) * dx.

            break

        else:

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
