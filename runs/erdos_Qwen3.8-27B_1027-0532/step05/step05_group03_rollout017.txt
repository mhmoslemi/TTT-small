def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:])) * dx

def get_lag_values(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, n_ones):
    h_float = h.astype(float)
    # Use the fractional part to break ties
    sorted_idx = np.argsort(-h_float, kind='stable')
    h_bin = np.zeros(len(h_float))
    h_bin[sorted_idx[:n_ones]] = 1.0
    return h_bin

def precompute_profiles(h):
    """Precompute profile_plus and profile_minus for all positions."""
    n = len(h)
    profile_plus = np.zeros((n, n))
    profile_minus = np.zeros((n, n))
    for m in range(n):
        # profile_plus[m][k] = 1 - h[m+k] for k < n-m
        if n - m > 0:
            profile_plus[m, :n-m] = 1.0 - h[m:n]
        # profile_minus[m][k] = h[m-k] for k <= m
        if m >= 0:
            profile_minus[m, :m+1] = h[0:m+1][::-1]
    return profile_plus, profile_minus
