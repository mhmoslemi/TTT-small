    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    fsize = 2 * n
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    H = np.fft.rfft(h, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    Wf = np.fft.rfft(w_flipped, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_lse(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = 1.0 / (max(alpha * 0.05, 1.0))
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), 2.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def equalize_top_lags(h, target, deadline, m=5, temp=0.005):
    """Focus optimization on equalizing the top m lags"""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for it in range(10000):

        w = np.exp(np.clip(shifted, -700, 0))
        w = w / w.sum()
        
        # Compute weighted subgradient
        fsize = 2 * n
        h_p = np.zeros(fsize)
        h_p[:n] = h_cur
        v_p = np.zeros(fsize)
        v_p[:n] = 1.0 - h_cur
        H = np.fft.rfft(h_p)
        V = np.fft.rfft(v_p)
        W = np.fft.rfft(w)
        w_flipped = w[::-1]
        Wf = np.fft.rfft(w_flipped)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (corr_pos - corr_neg)
        
        # Adaptive step size
        step = 0.1 / (np.linalg.norm(grad) + 1e-10)
        h_new = project_feasible(h_cur - step * grad, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
        else:
            # Try smaller step
            step *= 0.5
            h_new = project_feasible(h_cur - step * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
    
    return h_cur, cur_score

**Idea: Use the structure of the problem more directly.**

At the optimum, suppose the top m lags all achieve the same max value C. Then we have m constraints:
f_k(h) = C for k in active_set

This defines a system of m equations in n unknowns (with the additional constraint sum(h) = target and 0≤h≤1). We can try to solve this system using Newton's method.

The Jacobian of f_k w.r.t. h is:
∂f_k/∂h_i = (1-h_{i+k}) for i < n-k, -h_{i-k} for i ≥ k

This is a sparse matrix (banded). We can solve the linear system J δ = -residual using iterative methods.

But with n=400 and m=3-5, this should be feasible. Let me try:

Actually, the system is:
f_k(h) = C, k in active_set (m equations)
sum(h) = target (1 equation)
Total: m+1 equations, n unknowns. So we have n - m - 1 degrees of freedom.

We can use this to define a manifold and optimize on it. But that's complex.

**Simpler approach: Use the KKT conditions.**

At the optimum, for the active constraints (where h_i is at 0 or 1), we have:
∇(max f_k)(h) + λ * 1 + μ = 0 (KKT condition)

where λ is the Lagrange multiplier for the sum constraint, and μ_i ≥ 0 for the box constraints.

For the free variables (0 < h_i < 1), we need:
[∇(max f_k)]_i + λ = 0

This gives us a system we can solve.

Let me try a different approach: **alternating minimization with explicit equalization**.

Algorithm:
1. Compute corr and find top m lags
2. For the top m lags, compute the "excess" (corr_k - target_c)
3. Move h in the direction that reduces the excess for the top lags while maintaining the sum constraint
4. The direction is: -sum_k w_k * grad_k + (sum_k w_k * sum(grad_k) / n) * 1 (to maintain sum)

This is essentially what the subgradient method does, but with a more aggressive step.

Let me try yet another approach: **Direct iterative refinement using the correlation structure**.

Key observation: corr_k = sum_i h_i (1-h_{i+k}) * dx. If we increase h_i, it:
- Decreases corr_k for all k where h_{i+k} is large (because 1-h_{i+k} decreases)
- Increases corr_k for all k where h_{i-k} is large (because of the -h_{i-k} term)

Wait, that's not quite right. Let me re-derive:
∂corr_k/∂h_i = (1-h_{i+k})*dx for i < n-k (from the h_i(1-h_{i+k}) term)
∂corr_k/∂h_i = -h_{i-k}*dx for i ≥ k (from the (1-h_i)*h_{i-k} term... wait)

Actually, corr_k = sum_{j=0}^{n-k-1} h_j (1-h_{j+k}) * dx

So ∂corr_k/∂h_i:
- If i < n-k: appears in term j=i as h_i(1-h_{i+k}), derivative is (1-h_{i+k})*dx
- If i ≥ k: appears in term j=i-k as h_{i-k}(1-h_i), derivative is -h_{i-k}*dx
- Both if k ≤ i < n-k: derivative is (1-h_{i+k})*dx - h_{i-k}*dx

```python
from scipy.optimize import linprog

def blocking_direction(grads_active, n_active):
    """
    Find the direction that most reduces the max of the active correlations.
    grads_active: shape (n_active, n) - gradients of the active correlations
    Returns: (w, t) where w are the weights and t is the reduction amount
    """
    # We want: min_t s.t. w^T g_k ≤ t for all k, sum(w) = 1, w ≥ 0
    # Variables: [w_0, ..., w_{m-1}, t]
    # Constraints: g_k^T w - t ≤ 0 for each k
    #              sum(w) = 1
    #              w ≥ 0
    
    m = n_active
    n_vars = m + 1  # m weights + 1 for t
    
    # Minimize t
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # g_k^T w - t ≤ 0  =>  [g_k, -1] @ [w, t] ≤ 0
    A_ub = np.zeros((m, n_vars))
    for k in range(m):
        A_ub[k, :m] = grads_active[k]
        A_ub[k, m] = -1.0
    b_ub = np.zeros(m)
    
    # Equality: sum(w) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :m] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: w ≥ 0, t free
    bounds = [(0, None)] * m + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:m], res.x[-1]
    else:
        # Fallback: uniform weights
        return np.ones(m) / m, 0.0
```
