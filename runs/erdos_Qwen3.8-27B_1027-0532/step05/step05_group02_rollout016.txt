3. Compute g = sum(w) - convolve(h, w) - convolve(h, w[::-1]) → O(n log n)
4. h = h - step * g → O(n)
5. Project to [0,1], adjust sum → O(n)

This is much faster than SLSQP! We can run many more iterations.

    n = len(h)

    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):

    idx = i - k

def gradient_refine(h, n_iters, alphas, step_decay, rng, time_limit, start_time):
    """Projected gradient descent with soft-max objective."""
    n = len(h)
    h = h.copy().astype(float)
    dx = 2.0 / n
    target_sum = float(n) / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    for alpha in alphas:
        # Compute initial correlation
        a = np.zeros(N)
        a[:n] = h
        b = np.zeros(N)
        b[:n] = (1.0 - h)[::-1]
        
        for it in range(n_iters):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            
            # Compute correlation via FFT
            corr = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real[:2*n-1][::-1]
            
            # Softmax weights
            shifted = alpha * (corr - corr.max())
            w = np.exp(shifted)
            w_sum = w.sum()
            w = w / w_sum
            
            # Compute gradient: g[i] = sum_k w_k * ((1-h[i+k]) - h[i-k])
            # = w_sum - sum_k w_k * h[i+k] - sum_k w_k * h[i-k]
            # The first term is 1 (since w is normalized)
            # sum_k w_k * h[i+k] is a convolution
            # We need to be careful with the indexing
            
            # corr_k at index idx (0 to 2n-2) corresponds to shift k = idx - (n-1)
            # d(corr_k)/dh[i] = (1-h[i+k]) - h[i-k] where k = idx-(n-1)
            # So g[i] = sum_{idx} w[idx] * ((1-h[i + idx-(n-1)]) - h[i - (idx-(n-1))])
            # = sum_{idx} w[idx] * (1-h[i+k]) - sum_{idx} w[idx] * h[i-k]
            # where k = idx - (n-1)
            
            # Let's define: p[i] = sum_{idx} w[idx] * h[i + k(idx)] where k(idx) = idx-(n-1)
            # q[i] = sum_{idx} w[idx] * h[i - k(idx)]
            # Then g[i] = 1 - p[i] - q[i]
            
            # p[i] = sum_k w[k+(n-1)] * h[i+k] (when 0<=k+(n-1)<2n-1 and 0<=i+k<n)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):

    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:

                        best_h = h.copy()

                    break
            if found:
                break

        if not found:
            for _ in range(40):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

                    if new_c5 < best_c5:
                        best_c5 = new_c5

                    break

def sa_search(h_start, n_iters, t_init, t_final, rng, time_limit, start_time, focus=True):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 50000:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if focus and it % 3 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                if len(i_cands) == 0:
                    i_cands = ones
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(j_cands) == 0:
                    j_cands = zeros
                i = i_cands[rng.randint(len(i_cands))]
                j = j_cands[rng.randint(len(j_cands))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5
