def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

        if diff > 0:

        else:

        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    if active_indices is not None:

        c_max = float(np.max(c_sel))
        shifted = np.clip(alpha * (c_sel - c_max), -700.0, 0.0)

def grad_corr_lag(h, k, n):
    """Gradient of correlation at lag k (0-indexed from -n+1 to n-1) with respect to h."""
    # k is the actual lag: corr[n-1+k]
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    return g

def lse_top(h, alpha, top_m):
    corr = compute_corr(h)
    c_sel = corr[top_m]
    c_max = float(np.max(c_sel))
    shifted = np.clip(alpha * (c_sel - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_top_grad(h, alpha, top_m):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    c_sel = corr[top_m]
    c_max = float(np.max(c_sel))
    shifted = np.clip(alpha * (c_sel - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for wk, m in zip(w, top_m):
        if wk < 1e-12:
            continue
        k = int(m) - (n - 1)
        g += wk * grad_lag(h, k, n)
    g *= dx
    return g

def lbfgsb_optimize(h, n, target, alpha, top_m, mu, maxiter=300):
    def obj(x):
        return lse_top(x, alpha, top_m) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_top_grad(x, alpha, top_m) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def slsqp_optimize(h, n, target, alpha, top_m, maxiter=2000):
    def obj(x):
        return lse_top(x, alpha, top_m)
    def objg(x):
        return lse_top_grad(x, alpha, top_m)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def update_top_m(h, n, m):
    corr = compute_corr(h)
    return np.argsort(corr)[::-1][:m]

def symmetrize(h):
    n = len(h)
    return (h + h[::-1]) / 2.0

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, top_k=10):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            g += wk * grad_lag(h_cur, k, n)
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g


def lse_top_value(h, alpha, top_m):
    corr = compute_corr(h)
    c_sel = corr[top_m]
    c_max = float(np.max(c_sel))
    shifted = np.clip(alpha * (c_sel - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_top_grad(h, alpha, top_m):
    n = len(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    c_sel = corr[top_m]
    c_max = float(np.max(c_sel))
    shifted = np.clip(alpha * (c_sel - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for wk, m in zip(w, top_m):
        if wk < 1e-13:
            continue
        k = int(m) - (n - 1)
        g += wk * grad_lag(h, k, n)
    g *= dx
    return g


def update_top_m(h, n, m):
    corr = compute_corr(h)
    return np.argsort(corr)[::-1][:m]


def symmetrize(h):
    return (h + h[::-1]) / 2.0


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgsb_round(h, n, target, alpha, top_m, mu, maxiter=200):
    def obj(x):
        return lse_top_value(x, alpha, top_m) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_top_grad(x, alpha, top_m) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def slsqp_round(h, n, target, alpha, top_m, maxiter=2000):
    def obj(x):
        return lse_top_value(x, alpha, top_m)

    def objg(x):
        return lse_top_grad(x, alpha, top_m)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, top_k=10):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            g += wk * grad_lag(h_cur, k, n)
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):

        g = grad_lag(h_cur, k_star, n) * dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.02)
        else:
            eps *= 0.82
            if eps < 1e-10:
                break
    return h_cur, best_score


def multi_alpha_continuation(h, n, target, alphas, mu_list, maxiter_per, deadline, m_lags=25):

    best_score = compute_c5(h)
    for alpha in alphas:

            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
    return best_h, best_score
