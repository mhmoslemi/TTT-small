def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha) * (2.0 / _g_n)

    return np.sum(x) - _g_target

def _sum_jac(x):
    return np.ones(_g_n)

    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_obj, 'jac': _sum_jac}]

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_active(h, n, target, n_steps, n_active, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        # Focus on positive lags (indices n-1 to 2n-2) and negative lags (0 to n-2)
        # Actually, the problem uses max over ALL of the full correlation
        # Let's use the first n entries (lags 0 to n-1 in one direction)
        over = corr  # full correlation, all lags
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Find top n_active lags
        top_idx = np.argsort(over)[::-1][:n_active]
        # Softmax weights
        diffs = over[top_idx] - over[top_idx[0]]
        temperature = 0.01
        weights = np.exp(diffs / temperature)
        weights /= weights.sum()
        
        # Compute gradient for each active lag
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            # k is index in full correlation array
            # For full correlation of length 2n-1:
            # index 0 corresponds to lag -(n-1)
            # index n-1 corresponds to lag 0
            # index 2n-2 corresponds to lag (n-1)
            # So actual lag = k - (n-1)
            lag = k - (n - 1)
            if lag >= 0:
                # c[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
                # d/dh[i] = (1-h[i+lag]) for i < n-lag
                # d/dh[i] = -h[i-lag] for i > lag (when i-lag < n-lag, i.e., i < n)
                pos = np.arange(0, n - lag)
                grad[pos] += wk * (1.0 - h_cur[pos + lag])
                neg = np.arange(lag, n)
                grad[neg] -= wk * h_cur[neg - lag]
            else:
                # lag < 0, let l = -lag
                l = -lag
                # c[-l] = sum_{i=l}^{n-1} h[i]*(1-h[i-l])
                # d/dh[i] = (1-h[i-l]) for i >= l
                # d/dh[i] = -h[i+l] for i < n-l
                pos = np.arange(l, n)
                grad[pos] += wk * (1.0 - h_cur[pos - l])
                neg = np.arange(0, n - l)
                grad[neg] -= wk * h_cur[neg + l]
        
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.01, lr_init * 5.0)
        else:
            lr *= 0.9
            if lr < 1e-12:
                lr = lr_init * 0.1
    return best_h, best_score

def coordinate_descent(h, n, target, n_passes, eta, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        # Try each coordinate
        order = np.arange(n)
        for i in order:
            # Try increasing h[i]
            h_test = h_cur.copy()
            h_test[i] = min(1.0, h_cur[i] + eta)
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_score:
                h_cur = h_test
                best_score = sc
                improved = True
            else:
                # Try decreasing h[i]
                h_test = h_cur.copy()
                h_test[i] = max(0.0, h_cur[i] - eta)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < best_score:
                    h_cur = h_test
                    best_score = sc
                    improved = True
        if not improved:
            eta *= 0.9
            if eta < 1e-8:
                break
    return h_cur, best_score

def binary_project(h, n, target):
    """Project h toward a binary solution, then re-optimize."""
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(np.sum(h_bin))
    if n_ones > int(target):
        # Need to flip some 1s to 0s
        excess = n_ones - int(target)
        # Flip the ones with smallest h values
        one_indices = np.where(h_bin == 1)[0]
        sorted_by_h = one_indices[np.argsort(h[one_indices])]
        for i in range(excess):
            h_bin[sorted_by_h[i]] = 0
    elif n_ones < int(target):
        deficit = int(target) - n_ones
        zero_indices = np.where(h_bin == 0)[0]
        sorted_by_h = zero_indices[np.argsort(-h[zero_indices])]
        for i in range(deficit):
            h_bin[sorted_by_h[i]] = 1
    return h_bin

def interpolate_to(h_src, n_dst, target_dst):

    return project_feasible(h_out, target_dst)

        best_h = h_cur.copy()

    # Phase 2: Subgradient active lags
    t2 = min(start_time + 220, hard_deadline)
    h_sg, sc_sg = subgradient_active(best_h, n, target, 3000, 5, 0.004, t2)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg.copy()

    # Phase 3: SLSQP with high alphas
    t3 = min(start_time + 330, hard_deadline)
    h_cur, sc = slsqp_opt(best_h, n, target, [20000, 80000, 200000], 4000, 1e-16, t3)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()

    # Phase 4: Worst-lag descent
    t4 = min(start_time + 400, hard_deadline)
    h_wl, sc_wl = worst_lag_descend(best_h, n, target, 2000, 0.003, t4)
    if sc_wl < best_score:
        best_score = sc_wl
        best_h = h_wl.copy()

    # Phase 5: Coordinate descent
    t5 = min(start_time + 550, hard_deadline)
    h_cd, sc_cd = coordinate_descent(best_h, n, target, 80, 0.04, t5)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd.copy()

    # Phase 6: SLSQP after coordinate descent
    t6 = min(start_time + 650, hard_deadline)
    h_cur, sc = slsqp_opt(best_h, n, target, [50000, 200000, 500000], 3000, 1e-16, t6)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()

    # Phase 7: Binary blend + SLSQP
    t7 = min(start_time + 780, hard_deadline)
    h_bin = binary_project(best_h, n, target)
    for mix in [0.2, 0.4, 0.6]:
        if time.time() > t7:
            break
        h_mix = mix * h_bin + (1.0 - mix) * best_h
        h_mix = project_feasible(h_mix, target)
        h_cand, sc_cand = slsqp_opt(h_mix, n, target, [5000, 20000, 100000], 2000, 1e-15, t7)
        if sc_cand < best_score:
            best_score = sc_cand
            best_h = h_cand.copy()

    # Phase 8: Perturbation restarts
    t8 = min(start_time + 900, hard_deadline)
    for attempt in range(6):
        if time.time() > t8:
            break
        scale = 0.03 + 0.02 * attempt
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h_cand, sc_cand = slsqp_opt(h_pert, n, target, [2000, 10000, 50000], 2000, 1e-15, t8)
        if sc_cand < best_score:
            best_score = sc_cand
            best_h = h_cand.copy()

    # Phase 9: Subgradient + surgery
    t9 = min(start_time + 1050, hard_deadline)
    h_sg2, sc_sg2 = subgradient_active(best_h, n, target, 2000, 3, 0.002, t9)
    if sc_sg2 < best_score:
        best_score = sc_sg2
        best_h = h_sg2.copy()
    h_wl2, sc_wl2 = worst_lag_descend(best_h, n, target, 1000, 0.001, t9)
    if sc_wl2 < best_score:
        best_score = sc_wl2
        best_h = h_wl2.copy()

    # Phase 10: Try upscaling to n+200 with subgradient
    t10 = min(start_time + budget_s - 30, hard_deadline)
    n_up = n + 200
    t_up = n_up / 2.0
    h_up = interpolate_to(best_h, n_up, t_up)
    h_up, sc_up = subgradient_active(h_up, n_up, t_up, 3000, 5, 0.003, t10)
    if sc_up < best_score:
        best_score = sc_up
        best_h = h_up.copy()
        n = n_up
        target = t_up

    # Final SLSQP
    t_final = hard_deadline
    h_fin, sc_fin = slsqp_opt(best_h, n, target, [100000, 300000], 3000, 1e-17, t_final)
    if sc_fin < best_score:
        best_score = sc_fin
        best_h = h_fin.copy()

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
