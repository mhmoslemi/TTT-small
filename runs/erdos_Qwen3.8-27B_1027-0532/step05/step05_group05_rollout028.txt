**Approach: Min-max via bisection on t**

The problem is equivalent to: find the minimum t such that there exists h with:
- 0 ≤ h[i] ≤ 1 for all i
- Σ h[i] = n/2
- For all k: Σ_j h[j](1-h[j+k]) * dx ≤ t

This is a feasibility problem for a given t. The constraints Σ_j h[j](1-h[j+k]) * dx ≤ t are non-convex (because of the product). However, we can linearize them using the fact that for binary h, the product is the product itself, and for general h ∈ [0,1], we can use McCormick envelopes.

For the product h[j](1-h[j+k]), let p[j,k] = h[j](1-h[j+k]). The McCormick relaxation gives:
- p ≥ 0
- p ≥ (1-h[j+k]) - (1-h[j]) = h[j] - h[j+k]
- p ≥ h[j+k] - h[j]
- p ≤ 1 - h[j+k]
- p ≤ h[j]
- p ≤ h[j] + 1 - h[j+k] = 1 - (h[j+k] - h[j])

Wait, for product xy where x,y ∈ [0,1]:
McCormick: xy ≥ x + y - 1, xy ≤ x, xy ≤ y, xy ≥ 0

So for p = h[j](1-h[j+k]), let x = h[j], y = 1-h[j+k]:
p ≥ x + y - 1 = h[j] + 1 - h[j+k] - 1 = h[j] - h[j+k]
p ≥ 0
p ≤ x = h[j]
p ≤ y = 1-h[j+k]

The constraint becomes: Σ_{j valid} p[j,k] ≤ t/dx for all k.

This is a linear program with variables h[n] and p[n*(2n-1)] (roughly), which is about 400 + 400*799 ≈ 320,000 variables. That's too many for most LP solvers in reasonable time.

But wait, we don't need all p variables. For a given t, we can check feasibility using the McCormick relaxation as a relaxation (necessary but not sufficient condition). If the relaxation is infeasible, the original is infeasible. If feasible, the original might or might not be feasible.

This gives us a lower bound on t, which is useful for understanding the problem but not directly for finding a good h.

        except Exception:
            pass
    return h_cur, cur_score


def coordinate_descent(h, n, target, n_sweeps, n_eval, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_h = h_cur.copy()
    best_score = float(np.max(corr)) * dx
    k_vals = np.arange(-(n - 1), n)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        n_imp = 0
        for i in range(n):
            idx = int(order[i])
            v = h_cur[idx]
            u_min = -v
            u_max = 1.0 - v
            if u_max - u_min < 1e-12:
                continue
            valid_pos = (k_vals >= -idx) & (k_vals <= n - 1 - idx)
            valid_neg = (k_vals >= idx - n + 1) & (k_vals <= idx)
            a = np.zeros(2 * n - 1)
            if np.any(valid_pos):
                a[valid_pos] = 1.0 - h_cur[idx + k_vals[valid_pos]]
            if np.any(valid_neg):
                a[valid_neg] -= h_cur[idx - k_vals[valid_neg]]
            u_grid = np.linspace(u_min, u_max, n_eval)
            linear = corr[None, :] + u_grid[:, None] * a[None, :]
            linear[:, n - 1] = -np.inf
            linear_max = np.max(linear, axis=1)
            quad = corr[n - 1] + u_grid * (1.0 - 2.0 * v) - u_grid * u_grid
            f_vals = np.maximum(linear_max, quad)
            best_idx = int(np.argmin(f_vals))
            best_u = float(u_grid[best_idx])
            best_f = float(f_vals[best_idx])
            if 0 < best_idx < n_eval - 1:
                u_fine = np.linspace(float(u_grid[best_idx - 1]), float(u_grid[best_idx + 1]), 30)
                linear2 = corr[None, :] + u_fine[:, None] * a[None, :]
                linear2[:, n - 1] = -np.inf
                linear_max2 = np.max(linear2, axis=1)
                quad2 = corr[n - 1] + u_fine * (1.0 - 2.0 * v) - u_fine * u_fine
                f_fine = np.maximum(linear_max2, quad2)
                fi = int(np.argmin(f_fine))
                if f_fine[fi] < best_f:
                    best_u = float(u_fine[fi])
                    best_f = float(f_fine[fi])
            if best_f * dx < best_score:
                h_cur[idx] = v + best_u
                corr += best_u * a
                corr[n - 1] -= best_u * best_u
                new_max = float(np.max(corr)) * dx
                if new_max < best_score:
                    best_score = new_max
                    best_h = h_cur.copy()
                n_imp += 1
        h_cur = project_feasible(h_cur, target)
        corr = compute_corr(h_cur)
        sc = float(np.max(corr)) * dx
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        if n_imp == 0:
            break
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        j_neg = np.arange(k_star, n)
        grad[j_neg] -= h_cur[h_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.02)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    h_cd, sc_cd = coordinate_descent(best_h, n_final, target_final, 300, 150, min(start_time + 250, hard_deadline), rng)
    sc_cd = compute_c5(h_cd)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    if tleft() > 120:
        h_ref, sc_ref = slsqp_optimize(best_h, n_final, target_final, [500, 2000, 8000, 30000, 100000], 3000, 1e-16, min(start_time + 400, hard_deadline))
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    if tleft() > 200:
        h_cd2, sc_cd2 = coordinate_descent(best_h, n_final, target_final, 400, 200, min(start_time + 650, hard_deadline), rng)
        sc_cd2 = compute_c5(h_cd2)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    if tleft() > 150:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 500, 4, 0.003, 0.997, min(start_time + 750, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 200:
        x = np.linspace(0, 2, n_final)
        n_restarts = 4
        for r in range(n_restarts):
            if tleft() < 50:
                break
            if r == 0:
                h_pert = best_h + rng.randn(n_final) * 0.04
            elif r == 1:
                h_pert = best_h + rng.randn(n_final) * 0.08
            elif r == 2:
                freq = rng.uniform(2, 15)
                h_pert = best_h + 0.06 * np.sin(freq * x + rng.uniform(0, 2 * np.pi))
            else:
                h_pert = best_h + rng.randn(n_final) * 0.06
            h_pert = project_feasible(h_pert, target_final)
            dl = min(start_time + tleft() - 20, hard_deadline - 30)
            h_r, sc_r = coordinate_descent(h_pert, n_final, target_final, 100, 150, dl, rng)
            sc_r = compute_c5(h_r)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
            if tleft() > 60:
                h_ref2, sc_ref2 = slsqp_optimize(best_h, n_final, target_final, [2000, 10000, 50000], 2000, 1e-15, min(start_time + tleft() - 10, hard_deadline - 20))
                if sc_ref2 < best_score:
                    best_score = sc_ref2
                    best_h = h_ref2

    if tleft() > 80:
        h_cd3, sc_cd3 = coordinate_descent(best_h, n_final, target_final, 300, 250, min(start_time + tleft() - 20, hard_deadline - 30), rng)
        sc_cd3 = compute_c5(h_cd3)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    if tleft() > 50:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 200, 0.001, min(start_time + tleft() - 10, hard_deadline - 20))
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    if tleft() > 30:
        h_ref3, sc_ref3 = slsqp_optimize(best_h, n_final, target_final, [5000, 20000, 100000], 2000, 1e-17, hard_deadline - 5.0)
        if sc_ref3 < best_score:
            best_score = sc_ref3
            best_h = h_ref3

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
