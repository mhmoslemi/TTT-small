from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lse_val_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    val = c_max + np.log(w_sum) / alpha
    w_norm = w / w_sum
    g = np.zeros(n)
    n_corr = len(corr)
    for i in range(n_corr):
        wk = w_norm[i]
        if wk < 1e-14:
            continue
        k = i - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return val, g

    val, _ = lse_val_grad(x, alpha)
    s = np.sum(x) - target
    return val + 0.5 * mu * s * s

    _, g = lse_val_grad(x, alpha)
    s = np.sum(x) - target
    return g + mu * s * np.ones(len(x))


def slsqp_obj(x, alpha):
    val, _ = lse_val_grad(x, alpha)
    return val


def slsqp_grad(x, alpha):
    _, g = lse_val_grad(x, alpha)
    return g

    try:
        res = minimize(lbfgs_obj, h, method='L-BFGS-B', jac=lbfgs_grad,
                       args=(alpha, mu, target), bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-10})

    except Exception:

                break
            h_new = do_lbfgs(best_h, n, target, alpha, mu, 150)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    slsqp_alphas = [500, 2000, 8000, 30000, 100000]
    for alpha in slsqp_alphas:
        if time.time() > deadline - 200:
            break
        h_new = do_slsqp(best_h, n, target, alpha, 2000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if time.time() < deadline - 180:
        h_nl, sc_nl = multi_lag_slsqp(best_h, n, target, 8, 3000, 300, deadline - 180)
        if sc_nl < best_score:
            best_score = sc_nl
            best_h = h_nl

    if time.time() < deadline - 150:
        h_ns, sc_ns = nesterov_pg(best_h, n, target, 8000, 3000, 0.003, deadline - 150)
        if sc_ns < best_score:
            best_score = sc_ns
            best_h = h_ns

    if time.time() < deadline - 130:
        h_tl, sc_tl = targeted_lag_opt(best_h, n, target, 300, 0.005, deadline - 130)
        if sc_tl < best_score:
            best_score = sc_tl
            best_h = h_tl

    if time.time() < deadline - 100:
        temps = [0.005, 0.002, 0.001]
        for tp in temps:
            if time.time() > deadline - 80:
                break
            h_sg, sc_sg = subgradient_multi(best_h, n, target, 1000, 6, tp, 0.003, 0.997, deadline - 80)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    if time.time() < deadline - 80:
        h_surg, sc_surg = targeted_lag_opt(best_h, n, target, 200, 0.003, deadline - 80)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    if time.time() < deadline - 200:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        alphas_p2 = [30, 100, 400, 1500, 5000, 15000, 50000]
        for alpha in alphas_p2:
            if time.time() > deadline - 100:
                break
            for mu in [10.0, 100.0]:
                if time.time() > deadline - 100:
                    break
                h2_new = do_lbfgs(best_h2, n2, target2, alpha, mu, 120)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 80:
            for alpha in [5000, 20000, 60000]:
                if time.time() > deadline - 60:
                    break
                h2_new = do_slsqp(best_h2, n2, target2, alpha, 1500)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 60:
            h2_ml, sc2_ml = multi_lag_slsqp(best_h2, n2, target2, 8, 5000, 200, deadline - 60)
            if sc2_ml < best_score2:
                best_score2 = sc2_ml
                best_h2 = h2_ml
        if time.time() < deadline - 50:
            h2_ns, sc2_ns = nesterov_pg(best_h2, n2, target2, 10000, 2000, 0.002, deadline - 50)
            if sc2_ns < best_score2:
                best_score2 = sc2_ns
                best_h2 = h2_ns
        if time.time() < deadline - 40:
            h2_tl, sc2_tl = targeted_lag_opt(best_h2, n2, target2, 150, 0.003, deadline - 40)
            if sc2_tl < best_score2:
                best_score2 = sc2_tl
                best_h2 = h2_tl
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sa, sc_sa = hybrid_sa(best_h, n_cur, target_cur, 5000, 80, 0.15, 0.93, deadline - 40, seed + 1)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    if time.time() < deadline - 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(3):
            if time.time() > deadline - 25:
                break
            scales = [0.05, 0.10, 0.15]
            h_pert = best_h + rng.randn(n_cur) * scales[attempt]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = do_lbfgs(h_pert, n_cur, target_cur, 2000, 100.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_ref2 = do_slsqp(best_h, n_cur, target_cur, 50000, 1000)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    if time.time() < deadline - 15:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_surg, sc_surg = targeted_lag_opt(best_h, n_cur, target_cur, 150, 0.002, deadline - 10)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = do_slsqp(best_h, n_cur, target_cur, 100000, 1000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
