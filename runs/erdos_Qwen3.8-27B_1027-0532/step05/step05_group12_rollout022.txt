            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, alpha):
    """Smooth max of correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha, top_k=None):
    """Gradient of smooth max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if top_k is not None and top_k < 2*n - 1:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(2*n - 1)
    
    shifted = np.clip(alpha * (corr[top_idx] - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / wsum
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    
    return grad * dx

def fista_optimize(h_init, n, target, time_deadline, alphas=None, top_ks=None):
    """FISTA with smooth max objective."""
    if alphas is None:
        alphas = [50, 200, 1000, 5000, 20000, 100000]
    if top_ks is None:
        top_ks = [50, 50, 100, 100, 200, 400]
    
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    n_terms = 2*n - 1
    # Lipschitz constant estimate: each individual gradient has L2 norm <= sqrt(n) * 2
    # The smooth max has Lipschitz constant <= alpha * (max individual grad norm^2) / alpha 
    # Actually, for S_alpha(x) = (1/alpha) log sum exp(alpha x_i), 
    # the Hessian is diagonal with entries <= 1 (since it's a softmax), 
    # so Lipschitz of grad is bounded by the max eigenvalue of the Gram matrix of gradients.
    # Simplistically, L <= alpha * n (very rough upper bound)
    # In practice, use backtracking.
    
    for ai, alpha in enumerate(alphas):
        top_k = top_ks[min(ai, len(top_ks)-1)]
        if top_k > n_terms:
            top_k = n_terms
        
        t = 1.0  # momentum parameter
        y = h.copy()
        L = 1.0  # initial Lipschitz estimate
        stalled = 0
        
        while time.time() < time_deadline:

            # Gradient step with backtracking
            step = 1.0 / L

            x_new = project_feasible(x_new, target)
            
            # Check if step is good (sufficient decrease)
            obj_new = smooth_max_obj(x_new, alpha)
            obj_y = smooth_max_obj(y, alpha)

            # For simplicity, just accept and update L
            # Better: use the fact that if ||g|| is small, increase L
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-14:
                break
            
            # Update momentum
            y_new = x_new.copy()
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y_next = y_new + ((t - 1.0) / t_new) * (x_new - h)
            y_next = project_feasible(y_next, target)

            y = y_next

        if stalled > 500:

            h = best_h.copy()

def binary_local_search(h_init, n, target, time_deadline, max_rounds=100):

            sort_idx = np.argsort(h_flat)  # zeros first

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

  Let k = i - (n-1). Then corr_full[i] = sum_j h[j] * (1-h[j+k])
  
  When h[j] changes from 0 to 1:
  - In the sum for corr_full[i], the term for this j changes from 0 to (1-h[j+k]) if j+k is a valid index
  - Also, for the term where j' + k = j (i.e., j' = j-k), the term h[j-k] * (1-h[j]) changes from h[j-k] * 1 to h[j-k] * 0 = 0, so it changes by -h[j-k]
  
  So: delta_corr_full[i] = (1-h[j+k]) * I(0<=j+k<n) - h[j-k] * I(0<=j-k<n)
  
  This is the same for all i (it doesn't depend on i, only on j and k).
  
  Wait no, it DOES depend on k (which is determined by i). So for each i (each lag), the change is different.

OK so for each flip of position j, the change in each correlation value corr_full[i] is:
- Let k = i - (n-1)
- delta = (1-h[j+k]) if 0<=j+k<n, else 0, minus h[j-k] if 0<=j-k<n, else 0

This is O(1) per lag, so O(n) per flip, and O(n^2) per full pass. Same as before.

But I can vectorize it in numpy! For a given flip at position j:
- For all k in range(-n+1, n): delta[k] = (1-h[j+k]) if valid - h[j-k] if valid
- This is just a shifted version of h and 1-h

Actually, the change in the full correlation vector when flipping h[j] from a to b (change of delta = b-a) is:
- The new corr = old_corr + delta * v where v[i] = (1-h[i + (n-1) - j]) - h[j + i - (n-1) + j] ... hmm this is getting confusing.

Let me just use the direct approach: for each flip, recompute np.correlate. With numpy's C implementation, this should be fast enough.

For n=800: np.correlate takes about 0.1-0.5 ms. 800 flips per pass = 80-400ms per pass. 10 passes = 0.8-4 seconds. Very manageable.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_k is not None and top_k < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_k:]
    else:
        top_idx = np.arange(2 * n - 1)
    shifted = np.clip(alpha * (corr[top_idx] - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_vals / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n - k]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m])
            grad[:n - m] -= w * h[m:]
    return grad * dx

def fista_optimize(h_init, n, target, deadline, alphas, top_ks):
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for ai, alpha in enumerate(alphas):
        top_k = min(top_ks[ai], 2 * n - 1)
        t_mom = 1.0
        x_prev = h.copy()
        y = h.copy()
        base_step = alpha / (n * 0.5)
        base_step = min(base_step, 5.0)
        for it in range(8000):
            if time.time() > deadline:
                break
            g = smooth_max_grad(y, alpha, top_k=top_k)
            gnorm = np.linalg.norm(g)
            if gnorm < 1e-15:
                break
            step = base_step / (0.5 + 0.5 * gnorm * n / alpha)
            step = min(step, 3.0)
            x_new = y - step * g
            x_new = project_feasible(x_new, target)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - x_prev)
            y_new = project_feasible(y_new, target)
            x_prev = h.copy()
            h = x_new
            y = y_new
            t_mom = t_new
            if it % 50 == 0:
                score = compute_c5(h)
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        h = best_h.copy()
    return best_h, best_score

def projected_grad_optimize(h_init, n, target, deadline, alphas, top_ks):
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for ai, alpha in enumerate(alphas):
        top_k = min(top_ks[ai], 2 * n - 1)
        momentum = np.zeros(n)
        beta_mom = 0.8
        stalled = 0
        for it in range(20000):
            if time.time() > deadline:
                break
            g = smooth_max_grad(h, alpha, top_k=top_k)
            momentum = beta_mom * momentum + (1.0 - beta_mom) * g
            gnorm = np.max(np.abs(momentum))
            if gnorm < 1e-14:
                break
            lr = 1.0 / (gnorm + 1e-10)
            lr = min(lr, 3.0)
            improved = False
            for _ in range(30):
                h_try = h - lr * momentum
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 100:
                    break
    return best_h, best_score

def binary_local_search(h_init, n, target, deadline, max_rounds=200):
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        one_idx = np.where(h == 1.0)[0]
        order = np.argsort(h_init[one_idx])
        flip_n = n_ones - target_ones
        h[one_idx[order][:flip_n]] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(h == 0.0)[0]
        order = np.argsort(-h_init[zero_idx])
        flip_n = target_ones - n_ones
        h[zero_idx[order][:flip_n]] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            h[j] = 1.0 - h[j]
            score = compute_c5(h)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h.copy()
                improved = True
            else:
                h[j] = 1.0 - h[j]
        if not improved:
            break
    return best_h, best_score

def pair_swap_search(h_init, n, target, deadline, max_rounds=100):
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        one_idx = np.where(h == 1.0)[0]
        order = np.argsort(h_init[one_idx])
        flip_n = n_ones - target_ones
        h[one_idx[order][:flip_n]] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(h == 0.0)[0]
        order = np.argsort(-h_init[zero_idx])
        flip_n = target_ones - n_ones
        h[zero_idx[order][:flip_n]] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        one_positions = list(np.where(h == 1.0)[0])
        zero_positions = list(np.where(h == 0.0)[0])
        for a in one_positions:
            if time.time() > deadline:
                break
            for b in zero_positions:
                if time.time() > deadline:
                    break
                h[a] = 0.0
                h[b] = 1.0
                score = compute_c5(h)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h.copy()
                    improved = True
                    one_positions.remove(a)
                    one_positions.append(b)
                    zero_positions.remove(b)
                    zero_positions.append(a)
                else:
                    h[a] = 1.0
                    h[b] = 0.0
        if not improved:
            break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_obj(h, alpha):
    return smooth_max_obj(h, alpha)

def slsqp_grad(h, alpha):
    return smooth_max_grad(h, alpha, top_k=None)

def slsqp_cons_fun(h, target):
    return np.sum(h) - target

def slsqp_cons_jac(h, n):
    return np.ones(n)

def slsqp_polish(h_init, n, target, deadline, alphas):
    h = project_feasible(h_init.copy(), target)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, h, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           jac=lambda hh, aa=alpha: slsqp_grad(hh, aa),
                           constraints=[{'type': 'eq',
                                         'fun': lambda hh: np.sum(hh) - target,
                                         'jac': lambda hh: np.ones(n)}],
                           options={'maxiter': 1000, 'ftol': 1e-18})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h):
                    h = h_new
        except Exception:
            pass
    return h
