21. **Detailed coordinate descent**:
    
    For coordinate i:
    - S = sum(h) - h[i] (sum of all other coordinates)
    - For each shift k (offset from 0 to 2n-2):
      - The contribution of h[i] to the correlation at shift k:
        - h[i] appears in term (i, i+k-n+1) as h[i]*(1-h[i+k-n+1]) if valid
        - h[i] appears in term (i-k+n-1, i) as h[i-k+n+1]*(1-h[i]) if valid
      - So the change in corr[k] when h[i] changes to t:
        - If i+k-n+1 is valid (say j = i+k-n+1): corr[k] changes by (t - h[i])*(1-h[j])
        - If i-k+n-1 is valid (say j = i-k+n-1): corr[k] changes by -h[j]*(t - h[i])
      - So corr_new[k] = corr_old[k] + (t - h[i]) * [(1-h[j1]) - h[j2]] (where j1, j2 may not exist)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent_step(h, i, i_slopes, i_intercepts, mu, S, target, dx):
    """Find optimal h[i] given all other coordinates fixed.
    
    i_slopes: slope_m for each m (shift), shape (2n-1,)
    i_intercepts: corr_old[k] - h[i] * slope_{n-1-k} for each k, shape (2n-1,)
    mu: penalty parameter
    S: sum of all other coordinates
    target: target sum (n/2)
    dx: step size

    def obj(t):
        vals = i_intercepts + t * i_slopes
        return np.max(vals) * dx + mu * (S + t - target) ** 2
    
    res = minimize_scalar(obj, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-12})
    return res.x

        intercepts: array of shape (2n-1,) where corr_new[k] = intercepts[k] + t * slopes[k]

    best_c5 = compute_c5(h_cur)

    idxs = np.arange(n)

        if time.time() - start_time > time_limit:
            break
        
        order = rng.permutation(n) if random_order else idxs
        S = np.sum(h_cur) - h_cur[0]  # will update as we go
        
        for i in order:
            if time.time() - start_time > time_limit:

            S_other = np.sum(h_cur) - h_cur[i]
            slopes, intercepts = compute_slopes_and_intercepts(h_cur, i)

            h_cur[i] = t_new
        
        h_cur = project_feasible(h_cur, target)
        c5_val = compute_c5(h_cur)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_cur.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_slopes_vec(h, i):
    n = len(h)
    k = np.arange(2 * n - 1)
    idx1 = i - n + 1 + k
    idx2 = i + n - 1 - k
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    first_term = np.zeros(2 * n - 1)
    second_term = np.zeros(2 * n - 1)
    first_term[valid1] = 1.0 - h[idx1[valid1]]
    second_term[valid2] = h[idx2[valid2]]
    return first_term - second_term

def coord_obj(t, intercepts, slopes, mu, S, target, dx):
    vals = intercepts + t * slopes
    return float(np.max(vals)) * dx + mu * (S + t - target) ** 2

def coordinate_descent(h, target, mu, max_sweeps, time_limit, seed=42):
    n = len(h)
    dx = 2.0 / n
    start_time = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(seed)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        order = rng.permutation(n)
        for i in order:
            if time.time() - start_time > time_limit:
                break
            S_other = np.sum(h_cur) - h_cur[i]
            slopes = compute_slopes_vec(h_cur, i)
            intercepts = corr - h_cur[i] * slopes
            res = minimize_scalar(coord_obj, bounds=(0.0, 1.0), method='bounded',
                                  args=(intercepts, slopes, mu, S_other, target, dx),
                                  options={'xatol': 1e-10})
            t_new = res.x
            corr += (t_new - h_cur[i]) * slopes
            h_cur[i] = t_new

        h_cur = project_feasible(h_cur, target)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_val = compute_c5(h_cur)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_cur.copy()

    return best_h, best_c5

def smooth_max_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -350.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -350.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    half_n = n - 1
    grad = np.zeros(n)
    idxs = np.arange(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - half_n
        v1 = (idxs >= k) & (idxs < n + k)
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] - k]) * dx
        grad[v2] -= w * h[idxs[v2] - k] * dx
    return grad

def slsqp_polish(h, target, alphas, maxiter_per_alpha, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj_smooth(x):
            return smooth_max_val(x, alpha)
        def grad_smooth(x):
            return smooth_max_grad(x, alpha)
        def cons_fun(x):
            return np.array([np.sum(x) - target])
        def cons_jac(x):
            return np.array([np.ones(n)])
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_smooth, h_cur, jac=grad_smooth, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def top_m_subgradient(h, target, m, max_steps, time_limit, seed=0):

    dx = 2.0 / n
    start_time = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(seed)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = np.argsort(corr)[-m:]
        grad = np.zeros(n)
        for k_idx in top_k:
            k = k_idx - half_n
            v1 = (idxs >= k) & (idxs < n + k)
            v2 = (idxs - k >= 0) & (idxs - k < n)
            grad[v1] += (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= h_cur[idxs[v2] - k]
        grad *= dx / m

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.08, 0.15, 0.25, 0.4]:

            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_shake(h, target, frac, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    h_shake = h.copy()
    n_ones = int(np.sum(h > 0.5))
    n_shake = max(1, int(frac * n))
    one_idx = np.where(h_shake > 0.5)[0]
    zero_idx = np.where(h_shake <= 0.5)[0]
    if len(one_idx) >= n_shake:
        flip_to_zero = rng.choice(one_idx, size=n_shake, replace=False)
        h_shake[flip_to_zero] = 0.0
    if len(zero_idx) >= n_shake:
        flip_to_one = rng.choice(zero_idx, size=n_shake, replace=False)
        h_shake[flip_to_one] = 1.0
    return project_feasible(h_shake, target)

            best_c5 = c5_val
            best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.92)
        else:
            damping = min(0.95, damping + 0.02)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent with small penalty (fast exploration)
    if tleft() > 300:
        h_cd, c5_cd = coordinate_descent(best_h, target, mu=5.0,
                                         max_sweeps=500, time_limit=min(120, tleft() * 0.25),
                                         seed=seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Coordinate descent with larger penalty (tighter feasibility)
    if tleft() > 250:
        h_cd, c5_cd = coordinate_descent(best_h, target, mu=50.0,
                                         max_sweeps=500, time_limit=min(120, tleft() * 0.25),
                                         seed=seed + 1)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Coordinate descent with large penalty
    if tleft() > 200:
        h_cd, c5_cd = coordinate_descent(best_h, target, mu=200.0,
                                         max_sweeps=400, time_limit=min(100, tleft() * 0.25),
                                         seed=seed + 2)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Top-m subgradient for joint moves
    if tleft() > 150:
        h_sg, c5_sg = top_m_subgradient(best_h, target, m=5,
                                        max_steps=3000, time_limit=min(80, tleft() * 0.3),
                                        seed=seed)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: SLSQP polish with smooth max
    if tleft() > 100:
        h_pol = slsqp_polish(best_h, target, [100, 500, 2000], 1500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Another round of coordinate descent after SLSQP
    if tleft() > 200:
        h_cd, c5_cd = coordinate_descent(best_h, target, mu=100.0,
                                         max_sweeps=400, time_limit=min(100, tleft() * 0.3),
                                         seed=seed + 3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Top-m subgradient with different m
    if tleft() > 120:
        h_sg, c5_sg = top_m_subgradient(best_h, target, m=10,
                                        max_steps=2000, time_limit=min(60, tleft() * 0.3),
                                        seed=seed + 4)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: SLSQP with higher alpha
    if tleft() > 80:
        h_pol = slsqp_polish(best_h, target, [500, 2000, 10000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 9: Random restarts with coordinate descent
    for restart_seed in [seed + 10, seed + 20, seed + 30]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * 0.1
        h_pert = project_feasible(h_pert, target)
        h_cd, c5_cd = coordinate_descent(h_pert, target, mu=50.0,
                                         max_sweeps=300, time_limit=min(40, tleft() * 0.3),
                                         seed=restart_seed)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Binary shake and recovery
    for shake_frac in [0.1, 0.2, 0.35]:
        if tleft() < 60:
            break
        h_shake = binary_shake(best_h, target, shake_frac, seed=seed + 100)
        h_cd, c5_cd = coordinate_descent(h_shake, target, mu=50.0,
                                         max_sweeps=200, time_limit=min(30, tleft() * 0.4),
                                         seed=seed + 200)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: LP constraint generation
    if tleft() > 100:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=5000, time_limit=min(80, tleft() * 0.6),
                                     damping_init=0.85, n_active=30)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final coordinate descent polish
    if tleft() > 60:
        h_cd, c5_cd = coordinate_descent(best_h, target, mu=500.0,
                                         max_sweeps=300, time_limit=min(40, tleft() * 0.5),
                                         seed=seed + 300)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 13: Final SLSQP
    if tleft() > 40:
        h_pol = slsqp_polish(best_h, target, [2000, 10000, 50000], 800, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 14: Higher resolution attempt
    if tleft() > 150:
        n_try = n + 200
        if n_try % 2 == 0:
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_cd, c5_cd = coordinate_descent(h_interp, target_try, mu=50.0,
                                             max_sweeps=300, time_limit=min(80, tleft() * 0.5),
                                             seed=seed + 400)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n = n_try
                target = target_try
            if tleft() > 50:
                h_pol = slsqp_polish(best_h, target, [500, 3000], 800, 1e-15)
                sc = compute_c5(h_pol)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol

    # Phase 15: Final top-m subgradient
    if tleft() > 30:
        h_sg, c5_sg = top_m_subgradient(best_h, target, m=3,
                                        max_steps=2000, time_limit=min(20, tleft() * 0.7),
                                        seed=seed + 500)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
