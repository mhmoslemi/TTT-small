    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=2000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_cd(h, n, target, deadline, rng, block_size=3, n_bsearch=25):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    best_score = float(np.max(corr)) * dx
    stall_count = 0

    for sweep in range(1000000):
        if time.time() > deadline:
            break

        if sweep % 50 == 0 and sweep > 0:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')

        perm = rng.permutation(n)
        n_blocks = n // block_size

        for b in range(n_blocks):
            idxs = [int(perm[b * block_size + s]) for s in range(block_size)]
            c = sum(h_cur[ix] for ix in idxs)
            if c < 1e-14 or c > float(block_size) - 1e-14:
                continue

            B_total = np.zeros(2 * n - 1)
            for ix in idxs:
                B_total += compute_B(h_cur, ix, n)

            A = corr - sum(h_cur[ix] * compute_B(h_cur, ix, n) for ix in idxs)
            for ix in idxs:
                A += compute_B(h_cur, ix, n) * h_cur[ix]
            A = corr
            for ix in idxs:
                Bix = compute_B(h_cur, ix, n)
                A = A - h_cur[ix] * Bix
            for ix in idxs[1:]:
                Bix = compute_B(h_cur, ix, n)
                A = A + Bix * c

            lo = 0.0
            hi = float(np.max(A + B_total * (c - 1e-10))) if np.max(B_total) > 0 else float(np.max(A))

            for _ in range(n_bsearch):
                mid = (lo + hi) * 0.5
                if B_total[mid] if isinstance(B_total, dict) else 0:
                    pass
                pos = B_total > 1e-14
                neg = B_total < -1e-14
                ub = c - 1e-12
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / B_total[pos])))
                lb = max(0.0, c - float(block_size))
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / B_total[neg])))
                if lb <= ub + 1e-10:
                    hi = mid
                else:
                    lo = mid

            break

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent at original resolution (25% of effective time)
    deadline_p1 = start_time + (budget_s - 20.0) * 0.25
    h_cd, sc_cd = paired_cd(best_h, n, target, deadline_p1, rng,
                             recompute_interval=40, n_bsearch=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: SLSQP polish at original resolution
    if tleft() > 100:
        h_ref, sc_ref = slsqp_polish(best_h, n, target, [80, 200, 500, 1500], 2000, 1e-15)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 3: Resolution increase to 2x
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        h_2x = interpolate_to(best_h, n2, target2)
        deadline_p3a = start_time + (budget_s - 20.0) * 0.52
        h_cd2, sc_cd2 = paired_cd(h_2x, n2, target2, deadline_p3a, rng,
                                   recompute_interval=30, n_bsearch=25)
        if tleft() > 150:
            h_2x, sc_2x = slsqp_polish(h_cd2, n2, target2, [100, 400, 1000], 1500, 1e-14)
            sc_cd2 = sc_2x
        deadline_p3c = start_time + (budget_s - 20.0) * 0.72
        h_cd2, sc_cd2 = paired_cd(h_2x, n2, target2, deadline_p3c, rng,
                                   recompute_interval=30, n_bsearch=25)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2
    elif tleft() > 150:
        deadline_p3 = start_time + (budget_s - 20.0) * 0.75
        h_cd, sc_cd = paired_cd(best_h, n, target, deadline_p3, rng,
                                 recompute_interval=35, n_bsearch=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Refine at original resolution using knowledge from 2x
    best_n = len(best_h)
    best_target = best_n / 2.0
    if tleft() > 180 and best_n == n2:
        h_back = interpolate_to(best_h, n, target)
        deadline_p4 = start_time + (budget_s - 20.0) * 0.85
        h_cd, sc_cd = paired_cd(h_back, n, target, deadline_p4, rng,
                                 recompute_interval=30, n_bsearch=30)
        if tleft() > 80:
            h_cd, sc_cd = slsqp_polish(h_cd, n, target, [200, 800, 2000, 5000], 1500, 1e-15)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            best_n = n
            best_target = target

    # Phase 5: Random restarts at best resolution
    deadline_p5 = start_time + budget_s - 40.0
    noise_scales = [0.03, 0.05, 0.02, 0.08, 0.04, 0.06, 0.015, 0.055, 0.025, 0.045, 0.035, 0.01]
    for attempt in range(16):
        if tleft() < 50:
            break
        ns = noise_scales[attempt % len(noise_scales)]
        if attempt >= len(noise_scales):
            ns = ns * 0.7
        h_pert = best_h + rng.randn(best_n) * ns
        h_pert = project_feasible(h_pert, best_target)
        h_res, sc_res = paired_cd(h_pert, best_n, best_target, deadline_p5, rng,
                                   recompute_interval=25, n_bsearch=25)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    # Phase 6: Final SLSQP polish
    if tleft() > 40:
        fn = len(best_h)
        ft = fn / 2.0
        h_final, sc_final = slsqp_polish(best_h, fn, ft, [500, 2000, 5000, 10000], 1000, 1e-16)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 7: Last coordinate descent
    if tleft() > 20:
        fn = len(best_h)
        ft = fn / 2.0
        h_final, sc_final = paired_cd(best_h, fn, ft,
                                       start_time + budget_s - 10.0, rng,
                                       recompute_interval=20, n_bsearch=25)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
