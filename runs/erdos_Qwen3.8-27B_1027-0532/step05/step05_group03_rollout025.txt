        x = x ^ (x >> s)
    return (x & 1).astype(float)

def golden_sequence(n):
    phi = (np.sqrt(5.0) - 1.0) / 2.0
    idx = np.arange(n, dtype=float)
    return np.where((idx * phi) % 1.0 < 0.5, 1.0, 0.0)

def delta_all_lags(h, i, j, n):
    alt = np.zeros(4)
    alt[0] = i - 1
    alt[1] = i - n
    alt[2] = j - 1
    alt[3] = j - n
    ms = np.round(alt).astype(int)
    valid = (ms >= -1) & (ms <= n)
    alt[~valid] = -1
    base = h.copy()
    base[i] = 0.0
    base[j] = 1.0
    contrib_all = np.zeros(n)
    for kk in [1, n]:
        m = ms[:, kk - 1]
        vw = m >= 0
        if not np.any(vw):
            continue
        a = alt[vw, kk - 1]
        b = base[a]
        c = base[a + (kk - 1)]
        d = c * (1.0 - b)
        if kk == 1:
            contrib_all += d
        else:
            valid_k = (np.arange(n) - (kk - 1) <= a)
            contrib_all[valid_k] += d[valid_k]
    return contrib_all

def binary_swap_search(h_in, max_iter=20000, top_lags=5, n_cand=100, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    n = len(h_in)
    h = h_in.copy()
    over = compute_corr(h)[n-1:]
    best_max = float(np.max(over))
    best_h = h.copy()
    patience = 0
    for it in range(max_iter):
        patience += 1
        if patience > 2000:
            break
        top_idx = np.argsort(over)[::-1][:top_lags]
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_t = min(n_cand, len(ones) * len(zeros))
        ii = rng.choice(len(ones), n_t, replace=False)
        jj = rng.choice(len(zeros), n_t, replace=False)
        i_vals = ones[ii]
        j_vals = zeros[jj]
        best_improve = -1e-10
        best_i, best_j = -1, -1
        for ci in range(n_t):
            iv = int(i_vals[ci])
            jv = int(j_vals[ci])
            det = delta_all_lags(h, iv, jv, n)
            new_over_top = over[top_idx] + det[top_idx]
            new_worst = float(np.max(new_over_top))
            cur_worst = float(over[top_idx[0]])
            if new_worst < cur_worst + best_improve:
                best_improve = new_worst - cur_worst
                best_i, best_j = iv, jv
        if best_i >= 0 and best_improve < -1e-12:
            h[best_i] = 0.0
            h[best_j] = 1.0
            over = compute_corr(h)[n-1:]
            cur_max = float(np.max(over))
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = h.copy()
                patience = 0
    return best_h, best_max

def anneal_binary(h_in, start_temp=0.02, end_temp=0.001, n_iter=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    n = len(h_in)
    dx = 2.0 / n
    h = h_in.copy()
    over = compute_corr(h)[n-1:]
    cur_max = float(np.max(over))
    best_max = cur_max
    best_h = h.copy()
    n_t = n_iter
    for it in range(n_t):
        temp = start_temp * (end_temp / start_temp) ** (it / max(n_t, 1))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        ci = rng.randint(len(ones))
        cj = rng.randint(len(zeros))
        iv = int(ones[ci])
        jv = int(zeros[cj])
        det = delta_all_lags(h, iv, jv, n)
        new_over = over + det
        new_max = float(np.max(new_over))
        delta_energy = (new_max - cur_max) * dx
        if delta_energy < 0 or rng.rand() < np.exp(-delta_energy / max(temp, 1e-10)):
            h[iv] = 0.0
            h[jv] = 1.0
            over = new_over
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max * dx

def subspace_optimize(h_in, max_iter=500):
    n = len(h_in)
    dx = 2.0 / n
    target = n / 2.0
    h = h_in.copy()
    cur_score = compute_c5(h)
    for it in range(max_iter):
        over = compute_corr(h)[n-1:]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        if k_star == 0:
            grad[:] = dx * (1.0 - 2.0 * h)
        else:
            upper = n - 1 - k_star
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h[idx2 - k_star]
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        best_t = 0.0
        best_sc = cur_score
        for t in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
            h_cand = h - t * grad
            h_cand = project_feasible(h_cand, target)
            sc = compute_c5(h_cand)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_t = t
        if best_t > 0:
            h = h - best_t * grad
            h = project_feasible(h, target)
            cur_score = best_sc
        else:
            break
    return h, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = np.round(h_init)
    h_bin = project_feasible(h_bin, target)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    if tleft() > 400:
        h_ref, sc = slsqp_optimize(best_h, [100, 500, 2000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            h_bin = np.round(h_ref)
            h_bin = project_feasible(h_bin, target)
            if compute_c5(h_bin) < best_score:
                best_score = compute_c5(h_bin)
                best_h = h_bin.copy()

    cur_n = len(best_h)
    cur_target = cur_n / 2.0

    if tleft() > 300:
        h_bs, sc_bs = binary_swap_search(np.round(best_h), max_iter=30000, top_lags=5, n_cand=200, rng=rng)
        if sc_bs < best_score:
            best_score = sc_bs
            best_h = h_bs.copy()

    if tleft() > 250:
        h_an, sc_an = anneal_binary(np.round(best_h), start_temp=0.015, end_temp=0.0005, n_iter=8000, rng=rng)
        if sc_an < best_score:
            best_score = sc_an
            best_h = h_an.copy()

    if tleft() > 200:
        h_sub, sc_sub = subspace_optimize(best_h, max_iter=300)
        if sc_sub < best_score:
            best_score = sc_sub
            best_h = h_sub.copy()
        h_bin2 = np.round(h_sub)
        h_bin2 = project_feasible(h_bin2, cur_target)
        if compute_c5(h_bin2) < best_score:
            best_score = compute_c5(h_bin2)
            best_h = h_bin2.copy()

    if tleft() > 150:
        h_refs, sc_refs = slsqp_optimize(best_h, [5000, 20000, 100000], 1500, 1e-16)
        if sc_refs < best_score:
            best_score = sc_refs
            best_h = h_refs.copy()

    if tleft() > 120:
        h_bs2, sc_bs2 = binary_swap_search(np.round(best_h), max_iter=40000, top_lags=3, n_cand=300, rng=rng)
        if sc_bs2 < best_score:
            best_score = sc_bs2
            best_h = h_bs2.copy()

    if tleft() > 100:
        h_an2, sc_an2 = anneal_binary(np.round(best_h), start_temp=0.008, end_temp=0.0002, n_iter=6000, rng=rng)
        if sc_an2 < best_score:
            best_score = sc_an2
            best_h = h_an2.copy()

    if tleft() > 80:
        tm = thue_morse(n)
        tm = project_feasible(tm, target)
        sc_tm = compute_c5(tm)
        gr = golden_sequence(n)
        gr = project_feasible(gr, target)
        sc_gr = compute_c5(gr)
        for h_alt, sc_alt in [(tm, sc_tm), (gr, sc_gr)]:
            if sc_alt < best_score + 0.01:
                h_bsa, sc_bsa = binary_swap_search(h_alt, max_iter=20000, top_lags=5, n_cand=200, rng=rng)
                if sc_bsa < best_score:
                    best_score = sc_bsa
                    best_h = h_bsa.copy()
                h_ana, sc_ana = anneal_binary(h_bsa, start_temp=0.01, end_temp=0.0003, n_iter=5000, rng=rng)
                if sc_ana < best_score:
                    best_score = sc_ana
                    best_h = h_ana.copy()

    if tleft() > 60:
        h_fin, sc_fin = slsqp_optimize(best_h, [50000, 200000], 800, 1e-17)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin.copy()

    if tleft() > 40:
        h_bsf, sc_bsf = binary_swap_search(np.round(best_h), max_iter=30000, top_lags=4, n_cand=250, rng=rng)
        if sc_bsf < best_score:
            best_score = sc_bsf
            best_h = h_bsf.copy()
        h_subf, sc_subf = subspace_optimize(best_h, max_iter=200)
        if sc_subf < best_score:
            best_score = sc_subf
            best_h = h_subf.copy()

    if tleft() > 20:
        h_last, sc_last = slsqp_optimize(best_h, [100000], 500, 1e-17)
        if sc_last < best_score:
            best_score = sc_last
            best_h = h_last.copy()

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
