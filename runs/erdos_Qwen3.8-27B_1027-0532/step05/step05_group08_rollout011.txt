Let me flesh out the follow-the-leader approach:

The idea is:
1. Maintain a weight vector p over the lags
2. At each iteration:
   a. Optimize h to minimize sum_k p_k * corr[k] (a quadratic objective)
   b. Update p to put more weight on the currently worst lags

For step (a), the objective is:
Q(h) = sum_k p_k * corr[k] * dx = dx * sum_k p_k * sum_i h[i+k]*(1-h[i])

This is a quadratic in h. The gradient is:
∂Q/∂h[i] = dx * sum_k p_k * (∂corr[k]/∂h[i])

Which is the same form as the LogSumExp gradient but with p instead of w.

For step (b), we can use:
p_k = exp(β * corr_k(h)) / sum_j exp(β * corr_j(h))

with some temperature β.

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    ws = w.sum()
    w /= ws
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-14:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def weighted_obj(h, p):
    corr = compute_corr(h)
    return float(np.dot(p, corr))


def weighted_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        pk = p[n - 1 + k]
        if pk < 1e-15:
            continue
        g[:n - k] += pk * (1.0 - h[k:])
        g[k:] -= pk * h[:n - k]
    for ell in range(1, n):
        pk = p[n - 1 - ell]
        if pk < 1e-15:
            continue
        g[ell:] += pk * (1.0 - h[:n - ell])
        g[:n - ell] -= pk * h[ell:]
    g *= dx
    return g


def lbfgsb_lse(h, n, target, alpha, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_obj, h, method='L-BFGS-B',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lse_obj, h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

    best_h = h_cur.copy()

            return weighted_obj(x, p)

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):

    best_h = h_cur.copy()

    lr = lr_init

        weights /= weights.sum()

        h_surg, sc_surg = worst_lag_surgery(best_h1, n1, target1, 500, 0.004, deadline - 80)
        if sc_surg < best_score1:
            best_score1 = sc_surg
            best_h1 = h_surg

    # Phase 10: Second round SLSQP after surgery
    if time.time() < deadline - 60:
        for alpha in [50000, 100000, 200000]:
            if time.time() > deadline - 40:
                break
            h_new = slsqp_lse(best_h1, n1, target1, alpha, 2000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score1:
                best_score1 = sc
                best_h1 = h_new

    # Phase 11: Try n=1000 if time permits
    if time.time() < deadline - 250:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h1, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [50, 300, 1500, 6000, 20000]:
            if time.time() > deadline - 100:
                break
            h2_new = lbfgsb_lse(best_h2, n2, target2, alpha, 300)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 80:
            h2_new = slsqp_lse(best_h2, n2, target2, 30000, 2000, 1e-16)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 60:
            h_sg2, sc_sg2 = subgradient_polish(best_h2, n2, target2, 1500, 0.003, 0.002, 0.997, deadline - 60)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2
        if time.time() < deadline - 40:
            h_surg2, sc_surg2 = worst_lag_surgery(best_h2, n2, target2, 300, 0.003, deadline - 40)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2
        if best_score2 < best_score1:
            best_score1 = best_score2
            best_h1 = best_h2

    # Phase 12: Multi-start exploration
    if time.time() < deadline - 100:
        h_exp, sc_exp = multi_start_explore(best_h1, n1, target1, deadline - 100, rng)
        if sc_exp < best_score1:
            best_score1 = sc_exp
            best_h1 = h_exp

    # Phase 13: Final subgradient + surgery
    if time.time() < deadline - 30:
        n_cur = len(best_h1)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish(best_h1, n_cur, target_cur, 1000, 0.001, 0.001, 0.998, deadline - 20)
        if sc_sg < best_score1:
            best_score1 = sc_sg
            best_h1 = h_sg
        if time.time() < deadline - 10:
            h_surg, sc_surg = worst_lag_surgery(best_h1, n_cur, target_cur, 200, 0.001, deadline - 10)
            if sc_surg < best_score1:
                best_score1 = sc_surg
                best_h1 = h_surg

    # Phase 14: Final SLSQP
    if time.time() < deadline - 5:
        n_cur = len(best_h1)
        target_cur = n_cur / 2.0
        h_new = slsqp_lse(best_h1, n_cur, target_cur, 100000, 1500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score1:
            best_score1 = sc
            best_h1 = h_new

    h_final = project_feasible(best_h1, len(best_h1) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
