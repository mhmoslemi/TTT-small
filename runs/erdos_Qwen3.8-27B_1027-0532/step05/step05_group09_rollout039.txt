def project_feasible(h, target):

    n = len(h)

When h[i] changes from 1 to 0 and h[j] changes from 0 to 1:
For B[a, k] (which represents the change in corr[k] from turning on position a):
- B[a,k] = (1-h[a+k])*[a+k<n] - h[a-k]*[a>=k] for k>=1

  - a-k=i: a = i+k, k ranges over 1..n-1-i. B[i+k, k] changes by: the h[i] term changes from 1 to 0, so B[i+k, k] += 1 (since we had -h[i] = -1, now -0 = 0)
- For position j (0→1): affects B[a,k] where a+k=j or a-k=j
  - a+k=j: a = j-k, k ranges over 1..n-1-j. B[j-k, k] changes by: (1-h[j]) changes from 1 to 0, so B[j-k,k] -= 1
  - a-k=j: a = j+k, k ranges over 1..n-1-j. B[j+k, k] changes by: h[j] changes from 0 to 1, so B[j+k, k] -= 1

So the update is:
- For k=1 to n-1-i: B[i-k, k] += 1, B[i+k, k] += 1
- For k=1 to n-1-j: B[j-k, k] -= 1, B[j+k, k] -= 1

This is O(n) per move! Much better than O(n^2).

    smoothness_bound = 8.0 / (n * dx)
    lr_init = 1.0 / max(alpha * 0.01, smoothness_bound * 0.5)
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        corr = compute_corr_full(y_clipped)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha) * dx
        
        w = weights.copy()
        fsize = 2 * n
        Wf = np.fft.rfft(w[::-1], fsize)
        V = np.fft.rfft(1.0 - y_clipped, fsize)
        W = np.fft.rfft(w, fsize)
        H = np.fft.rfft(y_clipped, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = np.zeros(2*n-1)
        grad[n:] = corr_pos[1:]
        grad[:n-1] = corr_neg[1:][::-1]
        
        h_full = y_clipped
        v_full = 1.0 - y_clipped
        grad_h = np.zeros(n)
        for lag in range(n-1, 2*n-1):
            k = lag - (n-1)
            if k == 0:
                continue
            wi = grad[lag]
            if k >= 0:
                jp = np.arange(0, n-k)
                grad_h[jp] += wi * (-v_full[jp+k])
                jn = np.arange(k, n)
                grad_h[jn] += wi * h_full[jn-k]
            else:
                k_pos = -k
                jp = np.arange(0, n-k_pos)
                grad_h[jp+k_pos] += wi * (-v_full[jp])
                jn = np.arange(k_pos, n)
                grad_h[jn-k_pos] += wi * h_full[jn]
        grad_h *= dx
        
        x_new = project_feasible(y_clipped - step * grad_h, target)
        
        shrink = 0.0
        for _ in range(30):
            corr_new = compute_corr_full(x_new)
            c_max_new = float(np.max(corr_new))
            shifted_new = alpha * (corr_new - c_max_new)
            shifted_new = np.clip(shifted_new, -700.0, 0.0)
            obj_new = (c_max_new + np.log(float(np.exp(shifted_new).sum())) / alpha) * dx
            decrease = obj - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_h, target)
        step = min(step / max(1.0 - shrink, 0.01), lr_init * 3.0)
        
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5_full(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def arc_polish(h, target, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5_full(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr_full(h_cur)
        sorted_idx = np.argsort(corr)[::-1]
        num_active = 5
        active = sorted_idx[:num_active]
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            if ki >= n - 1:
                k = ki - (n-1)
                jp = np.arange(0, n-k)
                grad[jp] += wi * (-(1.0 - h_cur[jp+k]))
                jn = np.arange(k, n)
                grad[jn] += wi * h_cur[jn-k]
            else:
                k = (n-1) - ki
                jp = np.arange(0, n-k)
                grad[jp+k] += wi * (-(1.0 - h_cur[jp]))
                jn = np.arange(k, n)
                grad[jn-k] += wi * h_cur[jn]
        grad *= dx
        
        for eps_try in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            h_new = h_cur - eps_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5_full(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                break
    
    return h_cur, cur_score

def round_to_binary(h, target):
    n = len(h)
    h_bin = np.zeros(n)
    idx = np.argsort(h)[::-1]
    h_bin[idx[:int(target)]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 20.0
    rng = np.random.RandomState(seed)
    
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5_full(best_h)
    
    def elapsed():
        return time.time() - start_time
    
    def tremaining():
        return deadline_final - time.time()
    
    # Phase 1: Quick FISTA from initial
    if tremaining() > 60:
        h_new, sc_new = fista_optimize(best_h, target, 200, 5000, start_time + 40)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_new, sc_new = fista_optimize(best_h, target, 2000, 5000, start_time + 60)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 2: Binary local search from rounded solution
    if tremaining() > 200:
        h_bin = round_to_binary(best_h, target)
        h_bin, sc_bin = binary_local_search(h_bin, start_time + 120, max_sweeps=500)
        h_from_bin = h_bin.copy()
        h_from_bin = project_feasible(h_from_bin, target)
        sc_from_bin = compute_c5_full(h_from_bin)
        if sc_from_bin < best_score:
            best_score = sc_from_bin
            best_h = h_from_bin
        
        # FISTA from binary
        if tremaining() > 100:
            h_new, sc_new = fista_optimize(best_h, target, 500, 5000, start_time + 80)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            h_new, sc_new = fista_optimize(best_h, target, 5000, 5000, start_time + 100)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
    
    # Phase 3: More binary search cycles
    if tremaining() > 200:
        best_bin = round_to_binary(best_h, target)
        for cycle in range(3):
            if tremaining() < 120:
                break
            best_bin, sc_b = binary_local_search(best_bin, start_time + 60, max_sweeps=300)
            h_cont = project_feasible(best_bin.astype(float), target)
            sc_cont = compute_c5_full(h_cont)
            h_new, sc_new = fista_optimize(h_cont, target, 2000, 5000, start_time + 40)
            if sc_new < sc_cont:
                sc_cont = sc_new
                h_cont = h_new
            if sc_cont < best_score:
                best_score = sc_cont
                best_h = h_cont
                best_bin = round_to_binary(best_h, target)
    
    # Phase 4: Multi-start with perturbations
    if tremaining() > 300:
        scales = [0.05, 0.1, 0.15, 0.08, 0.12, 0.2]
        for s in scales:
            if tremaining() < 50:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            h_new, sc_new = fista_optimize(h_pert, target, 1000, 3000, start_time + 30)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if tremaining() < 50:
                break
            h_new, sc_new = fista_optimize(best_h, target, 10000, 3000, start_time + 25)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
    
    # Phase 5: Arc polish + high-alpha FISTA
    if tremaining() > 100:
        h_new, sc_new = arc_polish(best_h, target, 100, start_time + 40)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if tremaining() > 60:
            h_new, sc_new = fista_optimize(best_h, target, 20000, 5000, start_time + 50)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
    
    # Phase 6: Final binary + continuous cycle
    if tremaining() > 100:
        h_bin = round_to_binary(best_h, target)
        h_bin, _ = binary_local_search(h_bin, start_time + 40, max_sweeps=200)
        h_cont = project_feasible(h_bin.astype(float), target)
        if tremaining() > 60:
            h_new, sc_new = fista_optimize(h_cont, target, 5000, 5000, start_time + 50)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
        else:
            sc_cont = compute_c5_full(h_cont)
            if sc_cont < best_score:
                best_score = sc_cont
                best_h = h_cont
    
    # Phase 7: Final arc polish
    if tremaining() > 50:
        h_new, sc_new = arc_polish(best_h, target, 100, start_time + 40)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Final: ensure feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5_full(h_final)
    
    return h_final, final_score, len(h_final)
