def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_all_c(h, dx):
    n = len(h)
    corr = compute_corr(h)
    return corr * dx  # all c_k values

def compute_grad_c_k(h, k, n):
    """Compute gradient of c_k with respect to h.
    c_k = sum_i h[i] * (1-h[i+k]) for valid i
    """
    g = np.zeros(n)
    if k >= 0:
        # c_k = sum_{i=0}^{n-1-k} h[i] * (1 - h[i+k])
        # d/dh[j] = (1-h[j+k]) for j <= n-1-k (from first factor)
        #           - h[j-k] for j >= k (from second factor, where i = j-k)
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    else:
        ell = -k
        # c_{-ell} = sum_{i=ell}^{n-1} h[i] * (1 - h[i-ell])
        # = sum_{i=0}^{n-1-ell} h[i+ell] * (1 - h[i])
        # d/dh[j] = -h[j+ell] for j <= n-1-ell (from second factor)
        #           (1-h[j-ell]) for j >= ell (from first factor, where i = j-ell)
        for j in range(ell, n):
            g[j] += (1.0 - h[j - ell])
        for j in range(n - ell):
            g[j] -= h[j + ell]
    return g * dx

def compute_grad_c_k_fast(h, k, n, dx):
    """Vectorized gradient of c_k."""
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    return g * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    dx = 2.0 / len(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2*n-1):
        wk = w[m]
        if wk < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2*n-1):
        wk = w[m]
        if wk < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    return g

def grad_c_k_single(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n-ell])
        g[:n-ell] -= h[ell:]
    return g * dx

def lbfgsb_lse(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lp_cutting_plane(h, n, target, top_m, max_iter, deadline):
    """LP-based cutting plane method for min-max problem."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_vals = corr * dx
        c_max = float(np.max(c_vals))
        
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        
        # Select top_m worst shifts
        idx_sorted = np.argsort(c_vals)[::-1]
        active_idx = idx_sorted[:top_m]
        
        # Build LP: min t s.t. grad_k(h*)^T h - t <= c_k - grad_k(h*)^T h*
        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0
        
        n_cons = len(active_idx)
        A_ub = np.zeros((n_cons, n_vars))
        b_ub = np.zeros(n_cons)
        
        for i, m in enumerate(active_idx):
            k = int(m) - (n - 1)
            gk = grad_c_k_single(h_cur, k, n, dx)
            A_ub[i, :n] = gk
            A_ub[i, n] = -1.0
            b_ub[i] = c_vals[m] - gk.dot(h_cur)
        
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        bounds = [(0, 1)] * n + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[:n]
                t_lp = res.x[n]
                
                # Line search: try h_cur + alpha * (h_lp - h_cur)
                d = h_lp - h_cur
                found_improvement = False
                for alpha_ls in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                    h_try = h_cur + alpha_ls * d
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-12:
                        h_cur = h_try
                        best_score = sc
                        best_h = h_cur.copy()
                        found_improvement = True
                        break
                if not found_improvement:
                    # Try the LP point directly
                    h_try = project_feasible(h_lp, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-12:
                        h_cur = h_try
                        best_score = sc
                        best_h = h_cur.copy()
            else:
                break
        except Exception:
            break
    
    return best_h, best_score

def subgradient_multi(h, n, target, n_steps, top_k, lr_init, lr_decay, deadline):
    """Subgradient descent using top-k worst shifts."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_vals = corr * dx
        c_max = float(np.max(c_vals))
        
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        
        # Get top-k worst
        idx_sorted = np.argsort(c_vals)[::-1][:top_k]
        
        # Weighted gradient (equal weights for top-k)
        g = np.zeros(n)
        for m in idx_sorted:
            k = int(m) - (n - 1)
            g += grad_c_k_single(h_cur, k, n, dx)
        g /= top_k
        
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < c_max - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize, linprog
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    c_max = float(np.max(c))
    shifted = np.clip(alpha * (c - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wk = w[m]
        if wk < 1e-15:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    return g


def grad_c_k_single(h, k, n, dx):
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g * dx


def lbfgsb_lse(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lp_cutting_plane(h, n, target, top_m, max_iter, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_vals = corr * dx
        c_max = float(np.max(c_vals))

        if c_max < best_score - 1e-14:
            best_score = c_max
            best_h = h_cur.copy()

        idx_sorted = np.argsort(c_vals)[::-1]
        active_idx = idx_sorted[:top_m]

        n_vars = n + 1
        c_obj = np.zeros(n_vars)
        c_obj[n] = 1.0

        n_cons = len(active_idx)
        A_ub = np.zeros((n_cons, n_vars))
        b_ub = np.zeros(n_cons)

        for i, m in enumerate(active_idx):
            k = int(m) - (n - 1)
            gk = grad_c_k_single(h_cur, k, n, dx)
            A_ub[i, :n] = gk
            A_ub[i, n] = -1.0
            b_ub[i] = c_vals[m] - gk.dot(h_cur)

        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.success:
                h_lp = res.x[:n]
                d = h_lp - h_cur
                found_improvement = False
                for alpha_ls in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]:
                    h_try = h_cur + alpha_ls * d
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-14:
                        h_cur = h_try
                        best_score = sc
                        best_h = h_cur.copy()
                        found_improvement = True
                        break
                if not found_improvement:
                    h_try = project_feasible(h_lp, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-14:
                        h_cur = h_try
                        best_score = sc
                        best_h = h_cur.copy()
                    else:
                        return best_h, best_score
            else:
                break
        except Exception:
            break

    return best_h, best_score


def subgradient_multi(h, n, target, n_steps, top_k, lr_init, lr_decay, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init

    for step in range(n_steps):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_vals = corr * dx
        c_max = float(np.max(c_vals))

        if c_max < best_score - 1e-14:
            best_score = c_max
            best_h = h_cur.copy()

        idx_sorted = np.argsort(c_vals)[::-1][:top_k]

        g = np.zeros(n)
        for m in idx_sorted:
            k = int(m) - (n - 1)
            g += grad_c_k_single(h_cur, k, n, dx)
        g /= top_k

        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < c_max - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.005, 0.06)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break

    return best_h, best_score


def smooth_binary_search(h, n, target, max_iter, deadline):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    tgt = int(target)
    if n_ones > tgt:

        order = idx_ones[np.argsort(h[idx_ones])]
        h_bin[order[-(n_ones - tgt):]] = 0.0
    elif n_ones < tgt:
        idx_zeros = np.where(h_bin == 0.0)[0]
        order = idx_zeros[np.argsort(-h[idx_zeros])]
        h_bin[order[:tgt - n_ones]] = 1.0

    best_h = h_bin.copy()
    best_score = compute_c5(h_bin)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        best_delta = 1e-14
        best_pair = None
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        n_try = min(len(ones), 60)
        m_try = min(len(zeros), 60)
        for a in zeros[:m_try]:
            h_bin[a] = 1.0
            for b in ones[:n_try]:
                h_bin[b] = 0.0
                sc = compute_c5(h_bin)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h_bin.copy()
                    best_pair = (a, b)
                    break
                h_bin[b] = 1.0
            if best_pair is not None:
                break
            h_bin[a] = 0.0
        if best_pair is None:
            break

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
