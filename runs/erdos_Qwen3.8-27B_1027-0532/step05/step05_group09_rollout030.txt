```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_clipped = np.clip(h - mid, 0.0, 1.0)
        s = h_clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_weights(corr, K, temperature):
    n = len(corr)
    if K >= n:
        K = n
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    shifted = (top_vals - top_vals.max()) / temperature
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Build full weight vector
    w = np.zeros(n)
    w[top_idx] = weights
    return w

def weighted_grad_fft(h, w, n):
    # Compute gradient of sum_k w_k f_k(h) using FFT
    fsize = 2 * n
    H = np.fft.rfft(h, fsize)
    W = np.fft.rfft(w, fsize)
    
    # forward[j] = sum_{k=0}^{n-1-j} w[k] h[j+k]
    forward = np.fft.irfft(np.conj(W) * H, fsize)[:n]
    
    # backward[j] = sum_{k=0}^{j} w[k] h[j-k] = convolve(h, w)[j]
    backward = np.fft.irfft(H * W, fsize)[:n]
    
    # P[j] = sum_{k=0}^{n-1-j} w[k]
    P = np.zeros(n)
    cumsum = np.cumsum(w)
    P[0] = cumsum[-1]
    P[1:] = cumsum[:-1][::-1]
    # Wait: P[j] = sum_{k=0}^{n-1-j} w[k] = cumsum[n-1-j]
    P = cumsum[::-1].copy()
    # P[0] = cumsum[n-1] = sum of all w
    # P[j] = cumsum[n-1-j]
    # So P = cumsum[::-1]
    
    grad = P - forward - backward
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h_clipped = np.clip(h - mid, 0.0, 1.0)
        s = h_clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_weights(corr, K, temperature):
    n = len(corr)
    K = min(K, n)
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    shifted = (top_vals - top_vals.max()) / temperature
    weights = np.exp(shifted)
    weights /= weights.sum()
    w = np.zeros(n)
    w[top_idx] = weights
    return w

def weighted_grad_fft(h, w, n):
    fsize = 2 * n
    H = np.fft.rfft(h, fsize)
    W = np.fft.rfft(w, fsize)
    forward = np.fft.irfft(np.conj(W) * H, fsize)[:n]
    backward = np.fft.irfft(H * W, fsize)[:n]
    cumsum = np.cumsum(w)
    P = cumsum[::-1].copy()
    return P - forward - backward

def nesterov_topk(h0, target, n, deadline, K_init=5, T_init=0.005, lr_init=0.01):
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    t = 1.0
    K = K_init
    T = T_init
    lr = lr_init
    best_x = x.copy()
    best_score = compute_c5(x)
    stall = 0
    dx = 2.0 / n
    
    while time.time() < deadline:
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = x + momentum * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        corr = compute_corr(y_clipped)
        w = topk_weights(corr, K, T)
        grad = weighted_grad_fft(y_clipped, w, n) * dx
        
        x_new = project_feasible(y_clipped - lr * grad, target)
        
        x_prev = x
        x = x_new
        t = t_new
        
        if int(t_new) % 20 == 0:
            sc = compute_c5(x)
            if sc < best_score - 1e-12:
                best_score = sc
                best_x = x.copy()
                stall = 0
            else:
                stall += 1
                if stall > 5:
                    lr *= 0.7
                    K = min(K + 3, n // 2)
                    T = max(T * 0.85, 0.0005)
                    stall = 0
    
    return best_x, best_score

def block_swap(h, target, rng, num_swaps=10):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for _ in range(num_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i = h_cur[i]
        old_j = h_cur[j]
        h_cur[i] = old_j
        h_cur[j] = old_i
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc >= cur_score:
            h_cur[i] = old_i
            h_cur[j] = old_j
            h_cur = project_feasible(h_cur, target)
    return h_cur, cur_score

def coord_descent(h, target, deadline, max_sweeps=100):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old = h_cur[idx]
            # Try a few fractional values
            for trial in [min(1.0, old + 0.1), max(0.0, old - 0.1), 0.0, 1.0]:
                if abs(trial - old) < 1e-12:
                    continue
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    old = trial
                else:
                    h_cur[idx] = old
                    h_cur = project_feasible(h_cur, target)
                    break
    return h_cur, cur_score

def multi_res_warm_start(h_full, target, n, deadline):
    # Downsample, optimize, upsample
    h_cur = h_full.copy()
    for factor in [4, 2]:
        n_small = n // factor
        if n_small < 8:
            continue
        target_small = n_small / 2.0
        x_small = np.linspace(0, 1, n_small)
        x_full = np.linspace(0, 1, n)
        h_small = np.interp(x_small, x_full, h_cur)
        h_small = project_feasible(h_small, target_small)
        # Optimize at smaller resolution
        t_end = min(time.time() + 30, deadline)
        h_small, _ = nesterov_topk(h_small, target_small, n_small, t_end, K_init=3, T_init=0.01, lr_init=0.02)
        # Upsample back
        h_up = np.interp(x_full, x_small, h_small)
        h_up = project_feasible(h_up, target)
        if compute_c5(h_up) < compute_c5(h_cur):
            h_cur = h_up
    return h_cur
