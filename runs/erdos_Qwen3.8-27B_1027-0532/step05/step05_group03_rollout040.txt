```python
def frank_wolfe_bottleneck(h, n_steps, m_worst=3, gamma_init=0.5):
    n = len(h)
    target = n / 2
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for t in range(n_steps):
        # Compute all lags
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        lags = corr[n-1:n-1+n]  # lags 0 to n-1
        c_curr = float(np.max(lags)) * (2.0/n)
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Find worst lag(s)
        top_idx = np.argsort(lags)[::-1][:m_worst]
        
        # Compute vertex: minimize weighted sum of top lags
        # For each position i, cost = weighted sum of (1-h[i+k]) for k in top_idx
        costs = np.zeros(n)
        for k in top_idx:
            k = int(k)
            if k == 0:
                costs += (1.0 - h_cur)  # cost of setting h[i]=1 is (1-h[i]) when k=0
            else:
                for i in range(n - k):
                    costs[i] += (1.0 - h_cur[i+k])
                # positions i >= n-k have cost 0 for this lag
        
        # Vertex: set top 'target' positions (lowest cost) to 1
        order = np.argsort(costs)
        vertex = np.zeros(n)
        vertex[order[:int(target)]] = 1.0
        
        # FW step
        gamma = 2.0 / (t + 2)  # standard FW schedule
        # Or adaptive: line search
        h_new = (1.0 - gamma) * h_cur + gamma * vertex
        
        # Check improvement
        c_new = compute_c5(h_new)
        if c_new < c_curr:
            h_cur = h_new
        else:
            # Try smaller step
            gamma = 0.5 * gamma
            h_new = (1.0 - gamma) * h_cur + gamma * vertex
            c_new = compute_c5(h_new)
            if c_new < c_curr:
                h_cur = h_new
            else:
                # stagnation: try a larger diversification step
                h_cur = project_feasible(h_cur + 0.1 * np.random.randn(n), target)
    
    return best_h, best_score
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        if adj > 0 and h[mask].max() + adj > 1.0:
            adj = min(1.0 - h[mask].min(), adj) * 0.5
        if adj < 0 and h[mask].min() + adj < 0.0:
            adj = max(h[mask].max(), adj) * 0.5
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fw_vertex_for_lag(h, k, target):
    n = len(h)
    if k == 0:
        costs = 1.0 - h
    else:
        costs = np.zeros(n)
        valid = n - k
        costs[:valid] = 1.0 - h[k:k+n]
    order = np.argsort(costs, kind='mergesort')
    vertex = np.zeros(n)
    vertex[order[:int(target)]] = 1.0
    return vertex

def fw_vertex_combined(h, top_lags, weights, target):
    n = len(h)
    costs = np.zeros(n)
    for k, w in zip(top_lags, weights):
        k = int(k)
        if k == 0:
            costs += w * (1.0 - h)
        else:
            valid = n - k
            costs[:valid] += w * (1.0 - h[k:k+n])
    order = np.argsort(costs, kind='mergesort')
    vertex = np.zeros(n)
    vertex[order[:int(target)]] = 1.0
    return vertex

def bottleneck_fw(h, n_steps, m_worst=3, gamma_init=0.3, gamma_min=0.01):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    gamma = gamma_init
    momentum_h = h_cur.copy()
    for t in range(n_steps):
        corr = compute_corr(h_cur)
        lags = corr[n-1:n-1+n]
        c_curr = float(np.max(lags)) * (2.0 / n)
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(lags)[::-1][:m_worst]
        diffs = lags[top_idx] - lags[top_idx[0]]
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        vertex = fw_vertex_combined(h_cur, top_idx, weights, target)
        h_fw = (1.0 - gamma) * h_cur + gamma * vertex
        h_fw = project_feasible(h_fw, target)
        c_fw = compute_c5(h_fw)
        if c_fw < c_curr - 1e-15:
            momentum_h = 0.7 * momentum_h + 0.3 * h_fw
            h_cur = project_feasible(momentum_h, target)
            gamma = min(gamma * 1.02, 0.5)
        else:
            gamma = max(gamma * 0.9, gamma_min)
    return best_h, best_score

def binary_local_search(h_bin, n_steps, rng, max_k_check=10):
    n = len(h_bin)
    target = int(n / 2.0)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    h_cur = h_bin.copy()
    best_h = h_cur.copy()
    corr = compute_corr(h_cur)
    lags = corr[n-1:n-1+n]
    best_score = float(np.max(lags)) * (2.0 / n)
    temp = 0.02
    for step in range(n_steps):
        c_curr = float(np.max(lags)) * (2.0 / n)
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        k_star = int(np.argmax(lags))
        found = False
        n_trials = 20
        for _ in range(n_trials):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            dc = np.zeros(n)
            for k in range(max_k_check):
                if k == 0:
                    dc[k] = 0.0
                    continue
                va = a + k
                vb = b + k
                delta = 0.0
                if va < n:
                    delta -= (1.0 - h_cur[va])
                if vb < n:
                    delta += (1.0 - h_cur[vb])
                dc[k] = delta
            new_lags = lags.copy()
            for k in range(max_k_check):
                new_lags[k] += dc[k] * (2.0 / n)
            new_max = float(np.max(new_lags)) * (2.0 / n)
            if new_max < c_curr:
                h_cur[a] = 0.0
                h_cur[b] = 1.0
                ones[ones == a] = b
                zeros[zeros == b] = a
                corr = compute_corr(h_cur)
                lags = corr[n-1:n-1+n]
                found = True
                break
            elif rng.random() < np.exp((c_curr - new_max) / temp):
                h_cur[a] = 0.0
                h_cur[b] = 1.0
                ones[ones == a] = b
                zeros[zeros == b] = a
                corr = compute_corr(h_cur)
                lags = corr[n-1:n-1+n]
                found = True
                break
        if not found and step % 100 == 0:
            temp *= 0.95
    return best_h, best_score

def continuous_subgrad(h, n_steps, n_active=4, lr_init=0.004, lr_decay=0.998, temp=0.005):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    vel = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        lags = corr[n-1:n-1+n]
        c_curr = float(np.max(lags)) * (2.0 / n)
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(lags)[::-1][:n_active]
        diffs = lags[top_idx] - lags[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * (2.0 / n) * (1.0 - 2.0 * h_cur)
            else:
                valid = n - k
                idx1 = np.arange(0, valid)
                grad[idx1] += wk * (2.0 / n) * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * (2.0 / n) * h_cur[idx2 - k]
        vel = 0.4 * vel + 0.6 * grad
        h_new = h_cur - lr * vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(_obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def _obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def round_to_binary(h, target):
    n = len(h)
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:int(target)]] = 1.0
    return h_bin

def tleft(start_time, budget_s):
    return budget_s - 20.0 - (time.time() - start_time)
```
