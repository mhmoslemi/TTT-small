def adversarial_lp(h_init, n, target, time_limit, n_active=50, damping=0.8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(50000):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + 1e-5 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = h_var.value is not None
            except Exception:
                pass

        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-5:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

    return best_h, best_c5


_g_n = 0
_g_target = 0.0
_g_alpha = 100.0


def smooth_obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(_g_alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _g_target])


def cons_jac(h):
    return np.array([np.ones(_g_n)])


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_pass(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: Binary hill-climbing from rounded initial
    if tleft() > 150:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()

        h_climb, sc_climb = binary_hill_climb(h_bin, n, target,
                                               time_limit=min(120, tleft() - 30), rng=rng)
        if sc_climb < best_score:
            best_score = sc_climb
            best_h = h_climb

    # Phase 2: SLSQP from best (continuous)
    if tleft() > 150:
        h_ref, sc_ref = slsqp_refine(best_h, n, target,
                                      [100, 500, 2000, 10000], 3000, 1e-15)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 3: Adversarial LP
    if tleft() > 200:
        h_lp, sc_lp = adversarial_lp(best_h, n, target,
                                      time_limit=min(180, tleft() - 30),
                                      n_active=60, damping=0.85)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 4: SLSQP with very high alpha
    if tleft() > 120:
        h_ref, sc_ref = slsqp_refine(best_h, n, target,
                                      [5000, 20000, 100000, 500000], 2000, 1e-16)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 5: Binary round + simulated annealing
    if tleft() > 150:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, target)
        sc_b2 = compute_c5(h_bin2)
        if sc_b2 < best_score:
            best_score = sc_b2
            best_h = h_bin2.copy()

        h_sa, sc_sa = simulated_annealing(best_h, n, target,
                                           time_limit=min(100, tleft() - 30), rng=rng)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 6: SLSQP from binary result
    if tleft() > 100:
        h_ref, sc_ref = slsqp_refine(best_h, n, target,
                                      [200, 1000, 5000, 50000], 2000, 1e-15)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 7: Subgradient refinement
    if tleft() > 80:
        h_sg, sc_sg = subgradient_pass(best_h, n, target,
                                        max_steps=10000, time_limit=min(60, tleft() - 20))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 8: Higher resolution
    if tleft() > 250:
        n_try = n + 200
        if n_try % 2 == 0:
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_lp2, sc_lp2 = adversarial_lp(h_interp, n_try, target_try,
                                            time_limit=min(150, tleft() * 0.5),
                                            n_active=60, damping=0.8)
            if sc_lp2 < best_score:
                best_score = sc_lp2
                best_h = h_lp2
                n = n_try
                target = target_try
            if tleft() > 80:
                n_use = len(best_h)
                target_use = n_use / 2.0
                h_ref, sc_ref = slsqp_refine(best_h, n_use, target_use,
                                              [500, 5000, 50000], 1500, 1e-15)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 9: More binary search at current resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin3 = (best_h > 0.5).astype(float)
        h_bin3 = project_feasible(h_bin3, target_use)
        h_climb2, sc_climb2 = binary_hill_climb(h_bin3, n_use, target_use,
                                                  time_limit=min(80, tleft() - 30), rng=rng)
        if sc_climb2 < best_score:
            best_score = sc_climb2
            best_h = h_climb2

    # Phase 10: Final SLSQP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref, sc_ref = slsqp_refine(best_h, n_use, target_use,
                                      [10000, 100000, 500000], 1500, 1e-16)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 11: Final adversarial LP
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp3, sc_lp3 = adversarial_lp(best_h, n_use, target_use,
                                        time_limit=min(40, tleft() - 15),
                                        n_active=80, damping=0.9)
        if sc_lp3 < best_score:
            best_score = sc_lp3
            best_h = h_lp3

    # Phase 12: Final subgradient
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg2, sc_sg2 = subgradient_pass(best_h, n_use, target_use,
                                          max_steps=5000, time_limit=min(15, tleft() - 5))
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
