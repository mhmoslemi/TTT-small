```python
def multiplicative_weights(h_init, target, max_iter, beta_schedule, deadline):
    n = len(h_init)
    n_lags = 2 * n - 1
    h = h_init.copy()
    pi = np.ones(n_lags) / n_lags
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Step 1: Minimize weighted sum of correlations
        def f(x):
            return weighted_corr_objective_and_grad(x, pi, target, 1e6)[0]
        def g(x):
            return weighted_corr_objective_and_grad(x, pi, target, 1e6)[1]
        
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=[(0,1)]*n,
                       options={'maxiter': 200, 'ftol': 1e-15})
        h = project_feasible(res.x, target)
        
        # Step 2: Update weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        beta = beta_schedule[it]
        pi_new = np.exp(beta * corr / np.max(corr))
        pi_new /= pi_new.sum()
        pi = pi_new
        
    return h
```

```python
        h = h_new
    return h


def top_lag_equalize(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = float(top_vals.mean())
        top_std = float(top_vals.std())
        if top_std < 1e-13:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]

    def eq(x):
        return float(np.sum(x)) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 200.0
    bounds = [(0.0, 1.0)] * n

    # Phase 1: L-BFGS log-sum-exp warm-up
    alphas_p1 = [10, 50, 200, 1000, 5000, 20000]
    for a in alphas_p1:
        if time.time() > deadline - 400:
            break

        def f1(x):
            return lse_obj_and_grad(x, a, target, mu)[0]

        def g1(x):
            return lse_obj_and_grad(x, a, target, mu)[1]

        x_res = lbfgs_solve(best_h, f1, g1, bounds, 600, deadline)
        h_new = project_feasible(x_res, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Multiplicative weights refinement at current resolution
    mw_betas = [1.0, 3.0, 8.0, 20.0, 50.0, 100.0]
    for beta in mw_betas:
        if time.time() > deadline - 200:
            break
        n_outer = max(15, int(30 - 5 * mw_betas.index(beta)))
        h_mw = multiplicative_weights(best_h, target, n_outer, beta, mu, 120, deadline - 100)
        sc = compute_c5(h_mw)
        if sc < best_score:
            best_score = sc
            best_h = h_mw

    # Phase 3: Top-lag equalization
    if time.time() < deadline - 150:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=5, n_steps=200,
                                       deadline=deadline - 100)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        # Quick L-BFGS after equalization
        a = 50000

        def f3(x):
            return lse_obj_and_grad(x, a, target, mu)[0]

        def g3(x):
            return lse_obj_and_grad(x, a, target, mu)[1]

        x_res = lbfgs_solve(best_h, f3, g3, bounds, 300, deadline)
        h_new = project_feasible(x_res, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Multi-resolution upsample to 2n
    n2 = 2 * n
    target2 = n2 / 2.0
    if time.time() < deadline - 250:
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n2)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)
        bounds2 = [(0.0, 1.0)] * n2

        for a in [50, 500, 5000, 50000]:
            if time.time() > deadline - 150:
                break

            def f4(x):
                return lse_obj_and_grad(x, a, target2, mu)[0]

            def g4(x):
                return lse_obj_and_grad(x, a, target2, mu)[1]

            x_res = lbfgs_solve(h_fine, f4, g4, bounds2, 300, deadline)
            h_fine = project_feasible(x_res, target2)

        # Multiplicative weights at fine resolution
        for beta in [5.0, 20.0, 50.0]:
            if time.time() > deadline - 100:
                break
            h_fine = multiplicative_weights(h_fine, target2, 20, beta, mu, 80,
                                            deadline - 80)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2
            bounds = bounds2

    # Phase 5: Final multiplicative weights with high beta
    if time.time() < deadline - 60:
        for beta in [100.0, 200.0, 500.0]:
            if time.time() > deadline - 40:
                break
            h_mw = multiplicative_weights(best_h, target, 25, beta, mu, 100,
                                          deadline - 30)
            sc = compute_c5(h_mw)
            if sc < best_score:
                best_score = sc
                best_h = h_mw

    # Phase 6: SLSQP polish
    if time.time() < deadline - 40:
        h_slq = slsqp_polish(best_h, target, 10000, 2000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Final equalization + L-BFGS
    if time.time() < deadline - 30:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=3, n_steps=100,
                                       deadline=deadline - 20)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        cur_bounds = [(0.0, 1.0)] * len(best_h)
        cur_target = len(best_h) / 2.0
        a_final = 100000

        def f7(x):
            return lse_obj_and_grad(x, a_final, cur_target, mu)[0]

        def g7(x):
            return lse_obj_and_grad(x, a_final, cur_target, mu)[1]

        x_res = lbfgs_solve(best_h, f7, g7, cur_bounds, 300, deadline)
        h_new = project_feasible(x_res, cur_target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
