def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_c5_and_correlation(h):

def compute_grad_corr(h, lag):
    """Compute gradient of corr[lag] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        # corr[lag+n-1] = sum_j h[j] * (1-h[j+lag]) for j=0..n-1-lag
        # d/dh[j] = 1-h[j+lag] for j < n-lag
        # d/dh[j] = -h[j-lag] for j >= lag
        j1 = np.arange(0, n - lag)
        grad[j1] += (1.0 - h[j1 + lag])
        if lag > 0:
            j2 = np.arange(lag, n)
            grad[j2] -= h[j2 - lag]
    else:
        m = -lag
        j1 = np.arange(0, n - m)
        grad[j1] += (1.0 - h[j1 + m])
        if m > 0:
            j2 = np.arange(m, n)
            grad[j2] -= h[j2 - m]
    return grad

def perturb_structured(h, rng, mode, param):
    n = len(h)
    h_pert = h.copy()
    if mode == 'gauss':
        h_pert += rng.randn(n) * param
    elif mode == 'structured':
        # Add low-frequency perturbation

        n_low = max(1, int(n * param))
        for k in range(1, min(n_low, n // 2 + 1)):
            spectrum[k] = rng.randn() * 0.5
        pert = np.fft.irfft(spectrum, n)
        h_pert += pert
    elif mode == 'blocks':
        block_size = param
        n_blocks = n // block_size
        for i in range(n_blocks):
            val = rng.uniform(-0.3, 0.3)

        h_pert[idx] = 1.0 - h_pert[idx]
    else:
        h_pert += rng.randn(n) * param
    return h_pert

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        if w < 1e-15:
            continue
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f, h.copy(), jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h

def penalty_obj_and_grad(x, t_target, target_sum, penalty, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    excess = np.maximum(vals - t_target, 0.0)
    val = penalty * np.sum(excess ** 2) + mu * (np.sum(x) - target_sum) ** 2
    
    grad = np.zeros(n)
    active = np.where(excess > 1e-8)[0]
    for ki in active:
        w = 2.0 * penalty * excess[ki] * dx
        if w < 1e-15:
            continue
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])

            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (np.sum(x) - target_sum)
    return val, grad

def penalty_lbfgs(h, t_target, target, penalty, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return penalty_obj_and_grad(x, t_target, target, penalty, mu)[0]

def subgradient_nesterov(h0, target, n_iter, lr_init, momentum, deadline):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        # Nesterov
        h_nest = np.clip(h - v * momentum, 0.0, 1.0)
        
        corr = np.correlate(h_nest, 1.0 - h_nest, mode='full')
        c_max = np.max(corr)
        
        tol = max(1e-8, 1e-5 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(0, n - lag)
                grad[lag:] += (1.0 - h_nest[:n - lag])
                if lag > 0:
                    grad[:n - lag] -= h_nest[lag:]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[:n - m] += (1.0 - h_nest[m:])
                if m > 0:
                    grad[m:] -= h_nest[:n - m]
        grad /= len(active)
        
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h = h_new
        # Decay
        lr = lr_init / (1.0 + 0.01 * it)
    
    return best_h, best_score

def top_lag_equalize(h, target, n_top, n_steps, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        
        if top_std < 1e-12:
            break
        
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    
    return h_cur, cur_score

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h.copy(), jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h

def upsample(h, n_old, n_new, target):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, target)

def perturb(h, rng, mode, param):
    n = len(h)
    h_pert = h.copy()
    if mode == 'gauss':
        h_pert += rng.randn(n) * param
    elif mode == 'lowfreq':
        freqs = np.fft.rfftfreq(n)
        spectrum = np.zeros(n // 2 + 1)
        n_low = max(2, int(n * param))
        for k in range(1, min(n_low, n // 2 + 1)):
            spectrum[k] = rng.randn()
        pert = np.fft.irfft(spectrum, n)
        h_pert += pert * 0.3
    elif mode == 'flip':
        n_flip = int(n * param)
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = 1.0 - h_pert[idx]
    elif mode == 'block':
        bs = max(1, int(param))
        for i in range(0, n, bs):
            h_pert[i:i+bs] += rng.uniform(-0.2, 0.2)
    return h_pert

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        if w < 1e-15:
            continue
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h.copy(), jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h


def penalty_obj_and_grad(x, t_target, target_sum, penalty, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    excess = np.maximum(vals - t_target, 0.0)
    val = penalty * np.sum(excess ** 2) + mu * (np.sum(x) - target_sum) ** 2
    grad = np.zeros(n)
    active = np.where(excess > 1e-8)[0]
    for ki in active:
        w = 2.0 * penalty * excess[ki] * dx
        if w < 1e-15:
            continue
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (np.sum(x) - target_sum)
    return val, grad


def penalty_lbfgs(h, t_target, target, penalty, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return penalty_obj_and_grad(x, t_target, target, penalty, mu)[0]

    def g(x):
        return penalty_obj_and_grad(x, t_target, target, penalty, mu)[1]

    try:
        res = minimize(f, h.copy(), jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgradient_nesterov(h0, target, n_iter, lr_init, momentum, deadline):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    h_avg = np.zeros(n)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        h_nest = np.clip(h - v * momentum, 0.0, 1.0)
        corr = np.correlate(h_nest, 1.0 - h_nest, mode='full')
        c_max = float(np.max(corr))
        tol = max(1e-8, 1e-5 * c_max)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                grad[lag:n] += (1.0 - h_nest[:n - lag])
                if lag > 0:
                    grad[:n - lag] -= h_nest[lag:n]
            else:
                m = -lag
                grad[:n - m] += (1.0 - h_nest[m:n])
                if m > 0:
                    grad[m:n] -= h_nest[:n - m]
        grad /= len(active)
        
        lr = lr_init / (1.0 + 0.005 * it)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        h_avg = h_avg + h_new / (it + 1)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        if (it + 1) % 200 == 0:
            sc_avg = compute_c5(h_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_h = h_avg.copy()
        
        h = h_new
    
    sc_avg = compute_c5(h_avg)
    if sc_avg < best_score:
        best_score = sc_avg
        best_h = h_avg.copy()
    return best_h, best_score


def top_lag_equalize(h, target, n_top, n_steps, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        
        if top_std < 1e-13:
            break
        
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.04, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h.copy(), jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h


def upsample(h, n_old, n_new, target):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, target)


def perturb(h, rng, mode, param):
    n = len(h)
    h_pert = h.copy()
    if mode == 'gauss':
        h_pert += rng.randn(n) * param
    elif mode == 'lowfreq':
        spectrum = np.zeros(n // 2 + 1)
        n_low = max(2, int(n * param))
        for k in range(1, min(n_low, n // 2 + 1)):
            spectrum[k] = rng.randn()
        pert = np.fft.irfft(spectrum, n)
        h_pert += pert * 0.3
    elif mode == 'flip':
        n_flip = max(1, int(n * param))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = 1.0 - h_pert[idx]
    elif mode == 'block':
        bs = max(2, int(param))
        for i in range(0, n, bs):
            h_pert[i:i + bs] += rng.uniform(-0.15, 0.15)
    return h_pert
