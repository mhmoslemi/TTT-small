import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

# Global variables for SLSQP callbacks
_global_n = 0
_global_target = 0.0
_global_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit,
                   damping_init=0.8, reg=1e-4, n_active=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    
    stall_count = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0

            lo = max(0, k)

                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False

            solved = (h_var.value is not None)
        except Exception:
            pass

                solved = (h_var.value is not None)
            except Exception:
                pass

            break

            stall_count = 0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.array([np.sum(h) - _global_target])

def cons_jac(h):
    return np.array([np.ones(_global_n)])

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def subgradient_improve(h, n, target, max_steps, time_limit):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_periodic_pattern(n, periods, duty=0.5):
    """Create a periodic pattern with given number of periods and duty cycle"""
    h = np.zeros(n)
    period_len = n // periods
    on_len = int(period_len * duty)
    for p in range(periods):
        start = p * period_len

import numpy as np

import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx2 = np.where(h > 1e-10)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def alternating_lp(h_init, n, target, max_iter, time_limit,
                   damping_init=0.7, reg=1e-4, n_active=60):
    start = time.time()
    dx = 2.0 / n

    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())
    stall = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                j_lo = lo - k
                j_hi = hi - k
                coeff = np.zeros(n)
                coeff[lo:hi] = g[j_lo:j_hi] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000, verbose=False)
            solved = (h_var.value is not None)
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = (h_var.value is not None)
            except Exception:
                pass

        if not solved:
            break

        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0

            current_reg = max(1e-8, current_reg * 0.8)

            stall = 0

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def power_obj(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.sum(corr ** p) ** (1.0 / p))


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:

                cur_score = sc
        except Exception:
            pass

def subgradient_improve(h, n, target, max_steps, time_limit):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    return project_feasible(np.clip(h, 0, 1), target)

    # Phase 4: Pairwise + subgradient
    if tleft() > 60:
        h_pw, c5_pw = pairwise_refine(best_h, best_n, best_n / 2.0,
                                       n_pairs=20000, time_limit=min(30, tleft() * 0.4))
        if c5_pw < best_score:
            best_score = c5_pw
            best_h = h_pw

    if tleft() > 40:
        h_sg, c5_sg = subgradient_improve(best_h, best_n, best_n / 2.0,
                                           max_steps=3000, time_limit=min(25, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Higher resolution n=500
    if tleft() > 350:
        n2 = 500
        target2 = n2.0 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        lp_time = min(150, tleft() * 0.3)
        h2_lp, c5_2 = alternating_lp(h2, n2, target2,
                                      max_iter=30000, time_limit=lp_time,
                                      damping_init=0.7, reg=1e-4, n_active=70)
        if tleft() > 100:
            h2_r = slsqp_round(h2_lp, n2, target2,
                                [200, 1000, 5000], 2500, 1e-14)
            sc2 = compute_c5(h2_r)
            if sc2 < best_score - 1e-6:
                best_score = sc2
                best_h = h2_r
                best_n = n2

    # Phase 6: n=600
    if tleft() > 300:
        n3 = 600
        target3 = n3.0 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        lp_time = min(130, tleft() * 0.35)
        h3_lp, c5_3 = alternating_lp(h3, n3, target3,
                                      max_iter=30000, time_limit=lp_time,
                                      damping_init=0.7, reg=1e-4, n_active=70)
        if tleft() > 80:
            h3_r = slsqp_round(h3_lp, n3, target3,
                                [200, 1000, 5000], 2000, 1e-14)
            sc3 = compute_c5(h3_r)
            if sc3 < best_score - 1e-6:
                best_score = sc3
                best_h = h3_r
                best_n = n3

    # Phase 7: Diverse restarts
    for attempt in range(10):
        if tleft() < 100:
            break
        noise = 0.04 + 0.04 * attempt
        h_pert = best_h + rng.randn(best_n) * noise
        h_pert = project_feasible(h_pert, best_n / 2.0)

        if tleft() > 80:
            h_lp, c5_lp = alternating_lp(h_pert, best_n, best_n / 2.0,
                                          max_iter=15000, time_limit=min(45, tleft() * 0.5),
                                          damping_init=0.6, reg=1e-3, n_active=50)
            if tleft() > 50:
                h_r = slsqp_round(h_lp, best_n, best_n / 2.0,
                                   [100, 1000, 5000], 1500, 1e-14)
                sc = compute_c5(h_r)
                if sc < best_score:
                    best_score = sc
                    best_h = h_r

    # Phase 8: Binary exploration
    if tleft() > 200:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, best_n / 2.0)
        h_bc, c5_bc = binary_hill_climb(h_bin, best_n, best_n / 2.0,
                                         max_iter=50000, time_limit=min(60, tleft() * 0.3))
        if c5_bc < best_score + 0.01:
            h_ref = slsqp_round(h_bc, best_n, best_n / 2.0,
                                 [100, 500, 2000, 10000], 2000, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: Quasi-periodic structured starts
    if tleft() > 150:
        for period in [7, 9, 11, 13]:
            if tleft() < 60:
                break
            h_qp = make_quasi_periodic(best_n, period, 0.3, best_n / 2.0)
            h_qp_r = slsqp_round(h_qp, best_n, best_n / 2.0,
                                  [100, 500, 2000], 1500, 1e-14)
            sc = compute_c5(h_qp_r)
            if sc < best_score:
                best_score = sc
                best_h = h_qp_r

    # Phase 10: Final aggressive polish
    if tleft() > 80:
        h_r = slsqp_round(best_h, best_n, best_n / 2.0,
                          [2000, 10000, 50000, 200000], 3000, 1e-16)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best_h = h_r

    if tleft() > 40:
        h_pw, c5_pw = pairwise_refine(best_h, best_n, best_n / 2.0,
                                       n_pairs=30000, time_limit=min(25, tleft() * 0.5))
        if c5_pw < best_score:
            best_score = c5_pw
            best_h = h_pw

    if tleft() > 20:
        h_sg, c5_sg = subgradient_improve(best_h, best_n, best_n / 2.0,
                                           max_steps=5000, time_limit=min(15, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
