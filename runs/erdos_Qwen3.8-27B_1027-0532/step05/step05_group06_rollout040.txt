def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    weights = np.exp(shifted)

                elif np.min(slopes) >= 0:

                else:
                    break

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

    return np.correlate(h, 1.0 - h, mode='full')


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

    s = float(np.sum(x))

    weights = np.exp(shifted)

    if wsum > 1e-300:

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    mu = 200.0

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    # Phase 1: LSE L-BFGS-B warm-up
    alphas_p1 = [20, 100, 500, 2000, 10000]
    for a in alphas_p1:
        if tleft() < 100:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 2: Analytical CD (main workhorse)
    if tleft() > 120:
        h_cd, sc_cd = analytic_cd(best_h, target, n_sweeps=50)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    # Phase 3: LSE at higher alpha
    if tleft() > 100:
        for a in [50000, 200000]:
            if tleft() < 60:
                break
            h_new = lbfgs_optimize(best_h, target, a, mu, 800)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()

    # Phase 4: Top-lag equalization
    if tleft() > 80:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=5, n_steps=400)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq.copy()

    # Phase 5: More analytical CD
    if tleft() > 80:
        h_cd, sc_cd = analytic_cd(best_h, target, n_sweeps=60)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    # Phase 6: Projected gradient with adaptive LR
    if tleft() > 80:
        h_pg = best_h.copy()
        lr = 0.01
        for a in [100, 1000, 5000]:
            for it in range(200):
                if tleft() < 30:
                    break
                h_new = projected_gradient_step(h_pg, target, a, lr)
                sc = compute_c5(h_new)
                if sc < compute_c5(h_pg):
                    h_pg = h_new
                    lr = min(lr * 1.02, 0.05)
                else:
                    lr *= 0.85
                    if lr < 1e-10:
                        break
            h_cd, sc_cd = analytic_cd(h_pg, target, n_sweeps=30)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd.copy()
                h_pg = h_cd.copy()

    # Phase 7: LSE very high alpha
    if tleft() > 60:
        h_new = lbfgs_optimize(best_h, target, 500000, mu, 1000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 8: Final analytical CD
    if tleft() > 60:
        h_cd, sc_cd = analytic_cd(best_h, target, n_sweeps=40)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    # Phase 9: SLSQP polish
    if tleft() > 60:
        h_slq = slsqp_polish(best_h, target, 50000, 3000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq.copy()
        h_cd, sc_cd = analytic_cd(best_h, target, n_sweeps=20)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    # Phase 10: Try upsample to 2n if time permits
    if tleft() > 250:
        n2 = 2 * n
        target2 = n2 / 2.0
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n2)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)
        for a in [50, 500, 5000, 50000]:
            if tleft() < 60:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, mu, 400)
        if tleft() > 60:
            h_fine, sc_fine = analytic_cd(h_fine, target2, n_sweeps=30)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine.copy()

    # Phase 11: Final SLSQP + CD
    final_n = len(best_h)
    final_target = final_n / 2.0
    if tleft() > 30:
        h_final = slsqp_polish(best_h, final_target, 100000, 2000)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final.copy()
    if tleft() > 10:
        h_cd, sc_cd = analytic_cd(best_h, final_target, n_sweeps=20)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    h_final = project_feasible(best_h, final_target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
