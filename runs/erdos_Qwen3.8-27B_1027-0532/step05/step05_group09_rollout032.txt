        # Sum constraint penalty
        sum_h = h.sum()
        sum_target = n / 2.0

        # Adam-like update

        h_new = project_feasible(h_new, target)

            best_score = sc
            best_h = h_new.copy()
            z = np.log(np.clip(best_h, 1e-8, 1-1e-8) / (1 - np.clip(best_h, 1e-8, 1-1e-8)))

def pairwise_exchange(h, target, deadline, n_trials=5000):

            h_cur = project_feasible(h_cur, target)

                cur_score = sc
            else:
                h_cur[i] += eps
                h_cur[j] -= eps
                h_cur = project_feasible(h_cur, target)
                break
    
    return h_cur, cur_score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_grad_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_run(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = 1.0 / max(0.02 * alpha, 0.5)
    step_max = 2.0 / max(0.02 * alpha, 0.5)
    stagnation = 0

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_grad_fft(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        for _ in range(25):
            obj_new, _ = lse_grad_fft(x_new, alpha)
            if obj_new <= obj_y + 1e-4 * step * np.dot(grad_y, x_new - y_clipped):
                break
            step *= 0.5
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step * 1.3, step_max)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj - 1e-12:
            best_obj = sc
            best_x = x.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation % 200 == 0:
                t = 1.0
                x_prev = x.copy()
            if stagnation > 1500:
                break
    return best_x, best_obj

def subgrad_run(h0, target, max_iters, deadline, n_active):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = 0.003
    prev_grad = None
    prev_x = x.copy()
    stagnation = 0

    for k in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')[n-1:]
        c_max = float(np.max(corr))

        top_idx = np.argsort(corr)[::-1][:n_active]
        top_vals = corr[top_idx]
        gap = max(top_vals[0] - top_vals[-1], 1e-8)
        alpha_w = 20.0 / gap
        w_local = np.exp(alpha_w * (top_vals - top_vals[0]))
        w_local = w_local / w_local.sum()

        weights_full = np.zeros(len(corr))
        for wi, ki in zip(w_local, top_idx):
            weights_full[int(ki)] = wi

        w_flipped = weights_full[::-1].copy()
        v = 1.0 - x
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(v, fsize)
        W = np.fft.rfft(weights_full, fsize)
        H = np.fft.rfft(x, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (corr_pos - corr_neg)

        gnorm2 = np.dot(grad, grad)
        if gnorm2 > 1e-20:
            gap_to_best = c_max - best_obj
            if gap_to_best > 1e-12:
                step_polyak = gap_to_best / gnorm2
                step = min(max(step, 1e-6), min(step_polyak * 0.5, 0.03))

        if prev_grad is not None:
            s = x - prev_x
            r = grad - prev_grad
            rs = np.dot(r, s)
            rr = np.dot(r, r)
            if rr > 1e-20:
                step_bb = abs(rs) / rr
                step = 0.5 * step + 0.5 * min(max(step_bb, 1e-6), 0.03)

        x_new = project_feasible(x - step * grad, target)
        prev_x = x.copy()
        prev_grad = grad.copy()
        x = x_new

        sc = compute_c5(x)
        if sc < best_obj - 1e-13:
            best_obj = sc
            best_x = x.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 80:
                step *= 0.75
            if stagnation > 600:
                break
    return best_x, best_obj

def binary_sa(h0, target, max_iters, deadline, seed=0):
    n = len(h0)
    dx = 2.0 / n
    h = np.round(h0).copy()
    h = np.clip(h, 0, 1)
    num_ones = int(np.sum(h == 1))
    if num_ones > n // 2:
        one_pos = np.where(h == 1)[0]
        victim = one_pos[np.argsort(h0[one_pos])[:num_ones - n // 2]]
        h[victim] = 0
    elif num_ones < n // 2:
        zero_pos = np.where(h == 0)[0]
        victim = zero_pos[np.argsort(-h0[zero_pos])[:n // 2 - num_ones]]
        h[victim] = 1

    cur_max = compute_c5(h)
    best_h = h.copy()
    best_max = cur_max

    rng = np.random.RandomState(seed)
    T_init = 0.03
    T_min = 0.00005
    cool_rate = (T_min / T_init) ** (1.0 / max(max_iters, 1))
    T = T_init

    for k in range(max_iters):
        if time.time() > deadline:
            break
        T = T_init * (cool_rate ** k)

        one_pos = np.where(h == 1)[0]
        zero_pos = np.where(h == 0)[0]
        i = one_pos[rng.randint(len(one_pos))]
        j = zero_pos[rng.randint(len(zero_pos))]

        h[i] = 0
        h[j] = 1
        new_max = compute_c5(h)
        delta = new_max - cur_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0

    return best_h, best_max

def local_search(h, target, deadline, seed=0):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(seed)

    for trial in range(80000):
        if time.time() > deadline:
            break
        nnz = rng.randint(2, 10)
        idx = rng.choice(n, nnz, replace=False)
        dir_vec = np.zeros(n)
        dir_vec[idx] = rng.randn(nnz)
        dir_vec[idx] -= dir_vec[idx].mean()

        accepted = False
        for eps in [0.008, 0.004, 0.002, 0.012]:
            h_try = h_cur + eps * dir_vec
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                h_cur = h_try
                accepted = True
                break

    return h_cur, cur_score

def diverse_starts(h0, n, target, seed):

        h_bin[zp[np.argsort(-h0[zp])[:n // 2 - num_ones]]] = 1

    return starts
