def compute_corr(h):

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha, active_idx=None):
    corr = compute_corr(h)
    if active_idx is not None:
        c_active = corr[active_idx]
        c_max = float(np.max(c_active))
        shifted = np.clip(alpha * (c_active - c_max), -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    else:

        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)

def lse_grad_active(h, alpha, active_idx):

        c_max = float(np.max(corr))

        w /= w.sum()

        g = np.zeros(n)
        for m_idx in range(len(corr)):
            if w[m_idx] < 1e-13:
                continue

            if k >= 0:
                g[:n - k] += w[m_idx] * (1.0 - x[k:])
                g[k:] -= w[m_idx] * x[:n - k]
            else:
                ell = -k
                g[ell:] += w[m_idx] * (1.0 - x[:n - ell])
                g[:n - ell] -= w[m_idx] * x[ell:]
        g *= dx
        # L1 gradient: -1 where x < 0.5, +1 where x > 0.5
        l1_g = np.where(x < 0.5, -1.0, 1.0) * lambda_l1
        g += l1_g
        g += 2.0 * mu * (np.sum(x) - target)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    """Gradient of corr[m] w.r.t. h."""

        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]

        g[ell:] += (1.0 - h[:n - ell])

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def cons_jac(x):
    return np.ones(len(x))


def corr_grad_single(h, m, n):
    g = np.zeros(n)
    k = m - (n - 1)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g


def active_value(h, active_idx, alpha):
    corr = compute_corr(h)
    c_act = corr[active_idx]
    c_max = float(np.max(c_act))
    shifted = np.clip(alpha * (c_act - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def active_grad(h, active_idx, alpha, n):
    corr = compute_corr(h)
    c_act = corr[active_idx]
    c_max = float(np.max(c_act))
    shifted = np.clip(alpha * (c_act - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wi, mi in zip(w, active_idx):
        if wi < 1e-14:
            continue
        g += wi * corr_grad_single(h, int(mi), n)
    g *= dx
    return g


def full_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def full_grad(h, alpha, n):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)

def slsqp_active(h, n, target, active_idx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': cons_jac}]

    def obj_func(x):
        return active_value(x, active_idx, alpha)

    def grad_func(x):
        return active_grad(x, active_idx, alpha, n)

    try:
        res = minimize(obj_func, h, method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lbfgs_full(h, n, target, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n

        return full_value(x, alpha) + mu * (np.sum(x) - target) ** 2

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9})

    except Exception:

def interp_up(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src >= n_dst:
        return project_feasible(h_src[:n_dst].copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return project_feasible(np.interp(x_new, x_old, h_src), target_dst)

                if sc_new < best_sc:
                    best_sc = sc_new
                    best_h = h_new

    # Phase 3: Momentum subgradient + surgery
    if time.time() < deadline - 250:
        for temp in [0.03, 0.01, 0.003]:
            if time.time() > deadline - 200:
                break
            h_sg, sc_sg = momentum_subgrad(best_h, n, target, 3000, temp, 0.004, 0.997, 0.3, deadline - 200)
            if sc_sg < best_sc:
                best_sc = sc_sg
                best_h = h_sg
    if time.time() < deadline - 200:
        h_s, sc_s = surgery(best_h, n, target, 500, 0.005, deadline - 200)
        if sc_s < best_sc:
            best_sc = sc_s
            best_h = h_s

    # Phase 4: Resolution increase to n=800
    if time.time() < deadline - 350:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interp_up(best_h, n2, target2)
        best_h2 = h2.copy()
        best_sc2 = compute_c5(h2)

        for alpha in [50, 200, 800, 3000]:
            if time.time() > deadline - 250:
                break
            for m_top in [5, 10, 15]:
                if time.time() > deadline - 200:
                    break
                h2_new, sc2_new = active_set_optimize(best_h2, n2, target2, alpha, m_top, 5, deadline - 200)
                if sc2_new < best_sc2:
                    best_sc2 = sc2_new
                    best_h2 = h2_new

        if time.time() < deadline - 180:
            for alpha in [200, 1000]:
                if time.time() > deadline - 130:
                    break
                for mu in [50.0, 500.0]:
                    if time.time() > deadline - 120:
                        break
                    h2_new = lbfgs_full(best_h2, n2, target2, alpha, mu, 150)
                    sc2 = compute_c5(h2_new)
                    if sc2 < best_sc2:
                        best_sc2 = sc2
                        best_h2 = h2_new

        if time.time() < deadline - 100:
            for temp in [0.02, 0.005]:
                if time.time() > deadline - 70:
                    break
                h_sg2, sc_sg2 = momentum_subgrad(best_h2, n2, target2, 1500, temp, 0.003, 0.998, 0.3, deadline - 70)
                if sc_sg2 < best_sc2:
                    best_sc2 = sc_sg2
                    best_h2 = h_sg2

        if time.time() < deadline - 50:
            h_s2, sc_s2 = surgery(best_h2, n2, target2, 400, 0.003, deadline - 50)
            if sc_s2 < best_sc2:
                best_sc2 = sc_s2
                best_h2 = h_s2

        if best_sc2 < best_sc:
            best_sc = best_sc2
            best_h = best_h2

    # Phase 5: Random perturbations with re-optimization
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        scales = [0.03, 0.06, 0.1, 0.02, 0.08, 0.12, 0.05, 0.15]
        for attempt in range(len(scales)):
            if time.time() > deadline - 40:
                break
            h_pert = project_feasible(best_h + rng.randn(n_cur) * scales[attempt], target_cur)
            h_ref, sc_ref = active_set_optimize(h_pert, n_cur, target_cur, 500, 6, 4, deadline - 40)
            if sc_ref < best_sc:
                best_sc = sc_ref
                best_h = h_ref
                if time.time() < deadline - 30:
                    h_ref2, sc_ref2 = active_set_optimize(best_h, n_cur, target_cur, 3000, 5, 3, deadline - 30)
                    if sc_ref2 < best_sc:
                        best_sc = sc_ref2
                        best_h = h_ref2

    # Phase 6: Final momentum subgradient
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for temp in [0.005, 0.001]:
            if time.time() > deadline - 30:
                break
            h_sg, sc_sg = momentum_subgrad(best_h, n_cur, target_cur, 2000, temp, 0.002, 0.999, 0.2, deadline - 30)
            if sc_sg < best_sc:
                best_sc = sc_sg
                best_h = h_sg

    # Phase 7: Final surgery
    if time.time() < deadline - 25:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_s, sc_s = surgery(best_h, n_cur, target_cur, 400, 0.002, deadline - 25)
        if sc_s < best_sc:
            best_sc = sc_s
            best_h = h_s

    # Phase 8: Very final active-set with high alpha
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, sc_new = active_set_optimize(best_h, n_cur, target_cur, 50000, 3, 3, deadline - 10)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_sc = compute_c5(h_final)
    return h_final, final_sc, len(h_final)
