        weights = corr[top_idx]

        if w_sum < 1e-15:

        else:

        h_new = project_feasible(h_new, target)

def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=50):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    stall_count = 0
    
    for sweep in range(200000):
        if time.time() > deadline:
            break
        
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        # Get top correlations to guide pair selection
        top_mask = corr > 0.9 * np.max(corr)
        
        # Shuffle and try pairs
        indices = np.arange(n)
        rng.shuffle(indices)
        
        for p in range(n // 2):
            i = int(indices[2 * p])
            j = int(indices[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            
            # Binary search for best t
            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A[C > 0] + C[C > 0] * t_max))
            else:
                hi = float(np.max(A + C * t_max))
            
            for _ in range(30):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid
            
            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5
            
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score:
            best_score = new_score
            best_h = h_cur.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 300 and sweep > 200:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)
