import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_top_shifts(h, dx, alpha, top_m=None, threshold=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if threshold is not None:
        active = corr * dx >= (c_max * dx - threshold)
        top_indices = np.where(active)[0]
        if len(top_indices) > top_m:
            top_indices = np.argsort(corr)[-top_m:]
    elif top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
    else:
        top_indices = np.arange(2 * n - 1)
    sub_corr = corr[top_indices] * dx
    c_max_dx = c_max * dx
    shifted = np.clip(alpha * (sub_corr - c_max_dx), -500.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    if wsum < 1e-300:
        return np.zeros(n), c_max_dx
    weights = exp_s / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-12:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad, c_max_dx


def proj_gradient(h_init, n, target, start_time, time_limit, deadline_offset=20):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    threshold_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
    top_m_schedule = [200, 200, 150, 100, 80, 60, 50, 40, 30, 20, 15, 10, 10]

    for ai in range(len(alpha_schedule)):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        alpha = alpha_schedule[ai]
        thresh = threshold_schedule[ai]
        top_m = top_m_schedule[ai]
        n_steps = 2000

        momentum_v = np.zeros(n)
        beta = 0.8
        stalled = 0

        for step in range(n_steps):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            grad, c_cur = grad_top_shifts(h, dx, alpha, top_m=top_m, threshold=thresh)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-14:
                break
            lr = 1.0 / (alpha * gnorm * dx + 1e-12)
            lr = min(lr, 1.0)
            improved = False
            for _ls in range(30):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 80:
                    beta = max(0.5, beta - 0.05)
                    if stalled > 150:
                        break
    return best_h, best_score


def binary_snap(h, n, target):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    return h_bin


def greedy_binary_search(h_bin, n, target, start_time, time_limit, deadline_offset=30, max_rounds=5000):
    dx = 2.0 / n
    h = h_bin.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(12345)

    for r in range(max_rounds):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        i = rng.randint(n)
        if h[i] > 0.5:
            h[i] -= 1.0
        else:
            h[i] += 1.0
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        score = compute_c5(h)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
        else:

                h[i] += 1.0
            else:
                h[i] -= 1.0
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, start_time, time_limit, alphas, maxiter, deadline_offset=30):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break

        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            dx = 2.0 / len(x)
            shifted = np.clip(a * (corr * dx - c_max * dx), -400.0, 0.0)
            return float((c_max + np.log(np.sum(np.exp(shifted))) / a))

        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                sc = compute_c5(h_new)
                if sc < best_score - 1e-15:
                    best_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def alternating_lp(h_init, n, target, start_time, time_limit, max_iter=5000, deadline_offset=40, damping=0.7):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + 1e-4 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5
