```python
def compute_lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    # Objective: f = (1/alpha) * log(sum(exp(alpha * corr)))
    # (equivalently: c_max + log(sum(exp(shifted)))/alpha)
    f = c_max + np.log(exp_shifted.sum()) / alpha
    
    # Gradient
    first = np.convolve(1.0 - h, p, mode='full')[n-1:3*n-2]
    second = np.correlate(h, p, mode='full')[n-1:3*n-2]
    grad = first - second
    
    return f, grad, corr, p
```

What if I use a **custom quasi-Newton method** (like L-BFGS) with:

```python
def lbfgs_lse(h, target, alpha, max_iter, deadline, m=20):
    n = len(h)
    h_cur = h.astype(float).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    # L-BFGS memory
    s_list = []
    y_list = []
    
    prev_grad = None
    prev_h = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        f, grad, corr, p = compute_lse_grad(h_cur, alpha)
        
        # Project gradient: subtract mean (for equality constraint)
        grad_proj = grad - np.mean(grad)
        
        # L-BFGS two-loop recursion
        if prev_grad is not None:
            s = h_cur - prev_h
            y = grad_proj - (prev_grad - np.mean(prev_grad))
            sy = np.dot(s, y)
            if sy > 1e-10:
                s_list.append(s)
                y_list.append(y)
                if len(s_list) > m:
                    s_list.pop(0)
                    y_list.pop(0)
        
        # Two-loop recursion to compute r = H * grad_proj
        r = grad_proj.copy()
        alphas_buf = []
        betas_buf = []
        for i in range(len(s_list)-1, -1, -1):
            gamma = np.dot(y_list[i], s_list[i])
            beta = gamma / np.dot(y_list[i], r)
            r -= beta * y_list[i]
            alphas_buf.append(np.dot(s_list[i], r))
            betas_buf.append(beta)
        for i in range(len(s_list)):
            gamma = np.dot(y_list[i], s_list[i])
            r += (alphas_buf[i] - betas_buf[i] * np.dot(y_list[i], r)) * s_list[i] / gamma
        
        # Search direction (negate for minimization)
        d = -r
        # Project direction: remove mean component
        d -= np.mean(d)
        
        # Line search (backtracking)
        lr = 1.0
        for _ in range(30):
            h_trial = h_cur + lr * d
            h_trial = project_feasible(h_trial, target)
            f_trial = compute_c5(h_trial)  # Use actual C5, not LSE
            if f_trial < f or f_trial < best_score:
                break
            lr *= 0.5
        else:
            # Accept anyway if it's not too bad
            if compute_c5(project_feasible(h_cur + lr * d, target)) > best_score * 1.01:
```

    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lse_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    p = exp_shifted / sum_exp
    
    f = c_max + np.log(sum_exp) / alpha

    second = np.correlate(h, p, mode='full')
    grad = first[n-1:2*n-1] - second[n-1:2*n-1]
