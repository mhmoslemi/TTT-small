    """Apply swap h[i]:1->0, h[j]:0->1 to correlation array. Returns new corr."""
    n = len(h)
    corr = corr.copy()
    
    # For shift m, delta_c[m] = h[i+m] - 1 + h[i-m] + 1 - h[j+m] - h[j-m]
    # where out-of-bounds h values are 0
    
    # h[i+m] for m where 0<=i+m<n: m from -i to n-1-i
    # h[i-m] for m where 0<=i-m<n: m from i-(n-1) to i
    # h[j+m] for m where 0<=j+m<n: m from -j to n-1-j  
    # h[j-m] for m where 0<=j-m<n: m from j-(n-1) to j
    
    # Range for h[i+m]: corr indices from n-1-i to n-1-i+(n-i-1) = 2n-i-2
    # Values: h[0:n-i]
    # Plus contribution: -1 in this range
    
    # Range for h[i-m]: corr indices from i to i+n-1
    # Values: h[::-1] = h[n-1], h[n-2], ..., h[0]
    
    # Range for h[j+m]: corr indices from n-1-j to n-1-j+(n-j-1) = 2n-j-2
    # Values: h[0:n-j]

    corr[j : j+n] -= h[::-1]  # j-m term (length n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def apply_swap(corr, h, i, j):
    """Apply swap h[i]:1->0, h[j]:0->1 to corr in-place."""
    n = len(h)
    corr[n-1-i : 2*n-i-1] += h[0:n-i] - 1.0
    corr[i : i+n] += h[::-1]
    corr[n-1-j : 2*n-j-1] -= h[0:n-j]
    corr[j : j+n] -= h[::-1]

def undo_swap(corr, h, i, j):

    n = len(h)

    corr[i : i+n] -= h[::-1]
    corr[n-1-j : 2*n-j-1] += h[0:n-j]
    corr[j : j+n] += h[::-1]
