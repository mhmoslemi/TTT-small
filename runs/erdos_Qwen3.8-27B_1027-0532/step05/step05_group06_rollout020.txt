def topk_refine(h, target, k_top, alpha, mu, maxiter, deadline):

    h_cur = h.copy()

        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        topk = np.argsort(corr)[::-1][:k_top]
        h_new = lbfgs_topk(h_cur, target, topk, alpha, mu, maxiter)

        sc_cur = compute_c5(h_cur)
        if sc_new < sc_cur:
            h_cur = h_new
        else:
            break
    return h_cur

def upsample(h, n_new):

import numpy as np
from scipy.optimize import minimize
from functools import partial
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_gradient(h, ki):
    n = len(h)
    lag = int(ki) - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        j1 = np.arange(lag, n)
        grad[j1] += (1.0 - h[j1 - lag])
        if lag > 0:
            j2 = np.arange(0, n - lag)
            grad[j2] -= h[j2 + lag]
    else:
        m = -lag
        j1 = np.arange(0, n - m)
        grad[j1] += (1.0 - h[j1 + m])
        if m > 0:
            j2 = np.arange(m, n)
            grad[j2] -= h[j2 - m]
    return grad


def sg_optimize(h, target, n_iter, step_init, momentum, deadline):
    h_cur = h.copy()
    v = np.zeros_like(h_cur)
    best_h = h_cur.copy()
    best_score = compute_c5(best_h)
    for t in range(n_iter):
        if time.time() > deadline:
            break
        step = step_init / (1.0 + 0.05 * t)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_max = int(np.argmax(corr))
        grad = corr_gradient(h_cur, k_max)
        v = momentum * v - step * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score
