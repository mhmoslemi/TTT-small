                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_step(h, target, alpha, mu, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj_grad(x, alpha, target, mu)[0]
    
    def grad(x):
        return smooth_max_obj_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def subgradient_momentum(h, target, n_steps, eta0, beta, deadline):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    
    for t in range(n_steps):
        if time.time() > deadline:
            break
        eta = eta0 / (1.0 + t * 0.1)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        # Find active lags (within 1% of max)
        threshold = c_max * 0.99
        active_mask = corr >= threshold
        n_active = int(active_mask.sum())
        if n_active == 0:
            active_mask = np.zeros(len(corr), dtype=bool)
            active_mask[np.argmax(corr)] = True
            n_active = 1
        
        # Compute average subgradient
        grad = np.zeros(n)
        active_idx = np.where(active_mask)[0]
        for ki in active_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= n_active
        
        # Momentum update
        v = beta * v + (1.0 - beta) * grad
        h_new = h_cur - eta * v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            h_cur = h_new  # still update, momentum helps
    
    return best_h, best_score

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return smooth_max_obj_grad(x, alpha, target, 1e8)[0]
    
    def g(x):
        return smooth_max_obj_grad(x, alpha, target, 1e8)[1]
    
    def eq(x):
        return np.sum(x) - target
    
    def eq_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def top_lag_equalize(h, target, n_top=5, n_steps=200, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.7
            if lr < 1e-9:
                break
    return h_cur, cur_score

def selective_coord_descent(h, target, n_sweeps=3, n_trials=5, deadline=None):
    """Coordinate descent that only evaluates a few candidate values per coordinate."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break
            # Candidate values: current, 0, 1, and a few perturbations
            candidates = [h_cur[j], 0.0, 1.0]
            if 0 < h_cur[j] < 1:
                candidates.append(h_cur[j] * 0.5)
                candidates.append(h_cur[j] * 1.5 if h_cur[j] < 0.67 else h_cur[j] * 0.67)
            
            best_v = h_cur[j]
            best_sc = cur_score
            
            for v in candidates:
                v = np.clip(v, 0.0, 1.0)
                h_trial = h_cur.copy()
                h_trial[j] = v
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_v = v
            
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                improved = True
    
    return h_cur, cur_score

                h_trial[j] = v
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_sc - 1e-12:
                    best_sc = sc
                    best_v = v
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur = h_trial.copy()
                cur_score = best_sc
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 40.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    mu = 50.0

    # Phase 1: L-BFGS-B annealing at n=400
    alphas_p1 = [100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if deadline - time.time() < 80:
            break
        h_new = lbfgs_step(best_h, target, a, mu, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Subgradient with momentum
    if deadline - time.time() > 120:
        h_sg, sc_sg = subgradient_momentum(best_h, target, 1500, 0.8, 0.9,
                                           start_time + budget_s - 350)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 3: Top-lag equalization + L-BFGS
    if deadline - time.time() > 100:
        h_eq, sc_eq = top_lag_equalize(best_h, target, 5, 400,
                                       start_time + budget_s - 250)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_step(best_h, target, 200000, mu, 400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Upsample to n=800
    n2 = 2 * n
    target2 = n2 / 2.0
    if deadline - time.time() > 500:
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)

        alphas_fine = [500, 5000, 50000, 200000]
        for a in alphas_fine:
            if deadline - time.time() < 100:
                break
            h_fine = lbfgs_step(h_fine, target2, a, mu, 400)

        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 5: Subgradient at current resolution
    if deadline - time.time() > 200:
        h_sg, sc_sg = subgradient_momentum(best_h, target, 800, 0.5, 0.9,
                                           start_time + budget_s - 200)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 6: SLSQP polish
    if deadline - time.time() > 130:
        h_slq = slsqp_polish(best_h, target, 50000, 5000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Selective coordinate descent
    if deadline - time.time() > 100:
        h_cd, sc_cd = selective_coord_descent(best_h, target, 3,
                                              start_time + budget_s - 50)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 8: Final L-BFGS + SLSQP
    if deadline - time.time() > 60:
        h_new = lbfgs_step(best_h, target, 500000, mu, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if deadline - time.time() > 35:
        h_slq = slsqp_polish(best_h, target, 100000, 3000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 9: Final equalization + tight L-BFGS
    if deadline - time.time() > 35:
        h_eq, sc_eq = top_lag_equalize(best_h, target, 3, 200,
                                       start_time + budget_s - 5)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_step(best_h, target, 1000000, mu, 200)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
