import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            return np.clip(h - tau, 0.0, 1.0)
    # Fallback bisection
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)

    step = lr_init

        if time.time() > deadline:

        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / max(1.0 - shrink, 1e-10), lr_init * 3.0)
        
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def top_k_subgradient(h, target, k_top, deadline, temperature=0.01):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for _ in range(500):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:k_top]
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / temperature)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, active):

            jp = np.arange(0, n - ki)

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

            grad[lo1:hi1+1] += wi * (1.0 - h[lo1-k:hi1+1-k])
        lo2 = max(0, -k)
        hi2 = min(n - 1, n - 1 - k)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= wi * h[lo2+k:hi2+1+k]
    return obj, alpha * dx * grad

def fista(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.05, 1.0)
    step = lr_init
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(20):
            obj_new, _ = lse_obj_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quad = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quad * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / max(1.0 - shrink, 1e-10), lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    return best_x, best_score

def top_k_subgrad(h, target, k_top, deadline, temp_init=0.01):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    temp = temp_init
    for _ in range(5000):
        if time.time() > deadline:
            break
        C = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top = np.argsort(C)[::-1][:k_top]
        diffs = C[top] - C[top[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top):
            ki = int(ki)
            k = ki - (n - 1)
            lo1 = max(0, k)
            hi1 = min(n - 1, n - 1 + k)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += wi * (1.0 - h_cur[lo1-k:hi1+1-k])
            lo2 = max(0, -k)
            hi2 = min(n - 1, n - 1 - k)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= wi * h_cur[lo2+k:hi2+1+k]
        grad *= dx
        improved = False
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                improved = True
                break
        if not improved:
            temp *= 0.8
            if temp < 1e-6:
                break
    return h_cur, cur_score
