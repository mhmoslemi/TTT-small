```python
def compute_delta_correct(h, i, j, n):
    """Compute delta in full correlation vector when h[i] 1→0 and h[j] 0→1."""
    # c[m] = sum_a h[a] * (1-h)[a + m - (n-1)] for valid a, a+m-(n-1)
    # Let r = m - (n-1). c[n-1+r] = sum_a h[a] * (1-h[a+r])
    # 
    # Affected terms: a where h[a] changes (a=i, a=j) or (1-h[a+r]) changes (a+r=i, a+r=j)
    # So affected a values: i, j, i-r, j-r for each r.
    # 
    # For a given r, the affected set is {i, j, i-r, j-r} ∩ [0, n)
    # For each unique a in this set, delta at lag r = h_new[a]*(1-h_new[a+r]) - h_old[a]*(1-h_old[a+r])
    # 
    # This is O(1) per r, so O(n) total.
    
    full = np.zeros(2*n - 1)
    for r in range(-(n-1), n):
        m = n - 1 + r
        # Affected a values
        affected = set()
        for a in [i, j, i - r, j - r]:
            if 0 <= a < n and 0 <= a + r < n:
                affected.add(a)
        delta_r = 0.0
        for a in affected:
            # Old term: h[a] * (1-h[a+r])
            old_a = h[a]
            old_ar = h[a + r]
            # New term: h_new[a] * (1-h_new[a+r])
            new_a = 0.0 if a == i else (1.0 if a == j else h[a])
            new_ar = 0.0 if (a + r) == i else (1.0 if (a + r) == j else h[a + r])
            delta_r += new_a * (1 - new_ar) - old_a * (1 - old_ar)
        full[m] = delta_r
    return full
```

```python
def compute_delta_fast(h, i, j):
    n = len(h)
    r = np.arange(-(n-1), n)  # lag values, length 2n-1
    m = n - 1 + r  # indices into full correlation vector
    
    # General formula (valid when no overlaps)
    # delta[r] = -(1-h[i+r]) + (1-h[j+r]) - h[i-r] + h[j-r]
    
    # h[i+r] valid when 0 <= i+r < n
    idx1 = i + r
    mask1 = (idx1 >= 0) & (idx1 < n)
    
    # h[j+r] valid when 0 <= j+r < n
    idx2 = j + r
    mask2 = (idx2 >= 0) & (idx2 < n)
    
    # h[i-r] valid when 0 <= i-r < n
    idx3 = i - r
    mask3 = (idx3 >= 0) & (idx3 < n)
    
    # h[j-r] valid when 0 <= j-r < n
    idx4 = j - r
    mask4 = (idx4 >= 0) & (idx4 < n)
    
    result = np.zeros(2*n-1)
    result[mask1] -= (1 - h[idx1[mask1]])
    result[mask2] += (1 - h[idx2[mask2]])
    result[mask3] -= h[idx3[mask3]]
    result[mask4] += h[idx4[mask4]]
    
    # Fix overlaps at r = j-i and r = i-j
    # At r = j-i: i+r = j, so the term at a=i involves h[j] (which changes)
    # The term at a=j involves h[j] (which changes)
    # Actually let me just recompute the delta at those specific r values.
    
    for r_spec in [j - i, i - j]:
        if -(n-1) <= r_spec <= n-1:
            m_spec = n - 1 + r_spec
            # Recompute delta at this lag
            affected = set()
            for a in [i, j, i - r_spec, j - r_spec]:
                if 0 <= a < n and 0 <= a + r_spec < n:
                    affected.add(a)
            delta_r = 0.0
            for a in affected:
                old_a = h[a]
                old_ar = h[a + r_spec]
                new_a = 0.0 if a == i else (1.0 if a == j else h[a])
                new_ar = 0.0 if (a + r_spec) == i else (1.0 if (a + r_spec) == j else h[a + r_spec])
                delta_r += new_a * (1 - new_ar) - old_a * (1 - old_ar)
            result[m_spec] = delta_r
    
    return result
```
