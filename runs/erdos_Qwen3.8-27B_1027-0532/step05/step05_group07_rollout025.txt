    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum

    grad_h = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            idx1 = np.arange(n - k)
            idx2 = np.arange(k, n)
            grad_h[idx1] += w * (1.0 - x[idx1 + k])
            grad_h[idx2] += w * (-x[idx2 - k])
        else:
            k2 = -k
            idx1 = np.arange(k2, n)
            idx2 = np.arange(n - k2)
            grad_h[idx1] += w * (1.0 - x[idx1 - k2])
            grad_h[idx2] += w * (-x[idx2 + k2])

    if _g_mu > 0:
        grad_h += 2.0 * _g_mu * (np.sum(x) - _g_target)
    return grad_h


def lbfgs_polish(h, n, target, alphas, mus, maxiter=3000):
    global _g_n, _g_target, _g_alpha, _g_mu
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for a in alphas:
        for mu in mus:
            _g_alpha = a
            _g_mu = mu
            try:
                res = minimize(_lse_obj, h_cur, jac=_lse_grad, method='L-BFGS-B',
                               bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16,
                                                       'gtol': 1e-12, 'maxls': 50})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
            except Exception:
                pass

    h_cur = project_feasible(h_cur, target)
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_mu
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def elapsed_ok(frac):
        return (time.time() - start_time) < budget_s * frac - 15.0

    def remaining_ok(sec):
        return (budget_s - (time.time() - start_time)) > sec + 10.0

    # Phase 1: Coordinate descent at original resolution
    deadline1 = start_time + budget_s * 0.22
    h1, sc1 = paired_cd(best_h, n, target, deadline1, rng, bs_iters=30)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: L-BFGS polish at original resolution
    if remaining_ok(40):
        h2 = lbfgs_polish(best_h, n, target, [200, 1000, 5000], [0.0, 1.0, 10.0], maxiter=3000)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Multi-resolution to 2x
    n2 = n * 2
    target2 = n2 / 2.0
    if remaining_ok(150):
        h_interp = interpolate_to(best_h, n2, target2)
        # Quick polish at new resolution
        h_interp = lbfgs_polish(h_interp, n2, target2, [500, 2000], [0.0, 1.0], maxiter=2000)

        deadline3 = start_time + budget_s * 0.52
        h3, sc3 = paired_cd(h_interp, n2, target2, deadline3, rng, bs_iters=30)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: L-BFGS at 2x
    if remaining_ok(50):
        h4 = lbfgs_polish(best_h, n2, target2, [500, 2000, 8000], [0.0, 0.5, 5.0, 20.0], maxiter=4000)
        sc4 = compute_c5(h4)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: More CD at 2x
    if remaining_ok(120):
        deadline5 = start_time + budget_s * 0.75
        h5, sc5 = paired_cd(best_h, n2, target2, deadline5, rng, bs_iters=30)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: L-BFGS at 2x with higher alpha
    if remaining_ok(40):
        h6 = lbfgs_polish(best_h, n2, target2, [2000, 8000, 30000], [0.0, 0.1, 1.0], maxiter=5000)
        sc6 = compute_c5(h6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Random restarts at 2x
    if remaining_ok(100):
        restart_scales = [0.02, 0.03, 0.05, 0.01, 0.04, 0.06]
        for attempt in range(len(restart_scales)):
            if not remaining_ok(60):
                break
            h_pert = best_h + rng.randn(n2) * restart_scales[attempt]
            h_pert = project_feasible(h_pert, target2)
            dl = start_time + budget_s * 0.90 - attempt * 5.0
            hr, scr = paired_cd(h_pert, n2, target2, min(dl, time.time() + 60), rng, bs_iters=25)
            if scr < best_score:
                best_score = scr
                best_h = hr

    # Phase 8: Final L-BFGS polish
    if remaining_ok(20):
        h7 = lbfgs_polish(best_h, n2, target2, [5000, 20000, 80000], [0.0, 0.01, 0.1], maxiter=8000)
        sc7 = compute_c5(h7)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Phase 9: Final CD with remaining time
    if remaining_ok(15):
        deadline9 = start_time + budget_s - 8.0
        h8, sc8 = paired_cd(best_h, n2, target2, deadline9, rng, bs_iters=25)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
