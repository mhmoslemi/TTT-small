def project_feasible(h, target_sum):

def subgradient_polish(h, n, target, n_steps=500, deadline=None):

    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    for step_num in range(n_steps):

        dx = 2.0 / n
