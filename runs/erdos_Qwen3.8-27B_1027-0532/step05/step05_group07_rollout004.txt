def lse_lbfgsb(h, n, target, alpha, maxiter=500):

        # The projection of the gradient onto the null space of the constraint
        # (sum(h)=target) is: g - mean(g) * ones

        return g - g_mean

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * dx

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / float(np.sum(exp_s))
    grad = np.zeros(2 * n - 1)
    n_half = n - 1
    for m in range(2 * n - 1):
        wm = weights[m]
        if wm < 1e-12:
            continue
        lo1 = max(0, m)
        hi1 = min(n - 1, m + n - 1)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += wm * (1.0 - h[lo1 - m:hi1 - m + 1])
        lo2 = max(0, -m)
        hi2 = min(n - 1, n - 1 - m)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] += wm * (-h[lo2 + m:hi2 + m + 1])
    return grad * dx

    def fun(x):
        return lse_objective(x, alpha)

    def jac(x):
        return lse_gradient(x, alpha)

    bounds = [(0.0, 1.0)] * n

                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-6,

        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def projected_nesterov(h, n, target, alpha, steps=2000, lr=None):
    dx = 2.0 / n
    h_y = h.copy()
    h_x = h.copy()
    best_h = h_x.copy()
    best_f = lse_objective(h_x, alpha)
    momentum = 0.9
    if lr is None:
        lr = 1.0 / (alpha * dx * n)
    t = 1.0
    for s in range(steps):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (1.0 - t) / t_new
        h_y = h_x + beta * (h_x - h_y)
        h_y = project_feasible(h_y, target)

def block_perturb(h, n, target, rng, block_size=10):

    n_blocks = n // block_size
    for _ in range(n_blocks):

        lo = b * block_size
        hi = lo + block_size
        noise = rng.randn(block_size) * 0.08
        h_new[lo:hi] += noise
    return project_feasible(h_new, target)
