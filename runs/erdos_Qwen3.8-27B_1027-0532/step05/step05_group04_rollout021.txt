from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_gradient(h, n, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-8:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        if k > 0:
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    return grad

def gradient_descent(h, n, target, n_steps, lr, beta, deadline, rng):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    v = np.zeros(n)
    moment = 0.85
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        
        grad = weighted_gradient(h_cur, n, weights)
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
        else:
            v *= 0.7
            lr = max(lr * 0.9, 1e-7)
    
    return h_cur, cur_score

def slsqp_optimize(h, n, target, alpha, maxiter, deadline):
    def obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def cons_eq(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < 1.0:  # sanity check
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _slsqp_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _slsqp_cons_eq(x):
    return np.sum(x) - _g_target


def _slsqp_cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h, n, target, alpha, maxiter, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_eq, 'jac': _slsqp_cons_jac}]
    try:
        if time.time() > deadline:
            return h, compute_c5(h)
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < 1.0:
            return h_new, sc
    except Exception:
        pass
    return h, compute_c5(h)


def gradient_descent(h, n, target, n_steps, lr, beta, deadline, rng):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    v = np.zeros(n)
    moment = 0.85
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-8:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            if k > 0:
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
        else:
            v *= 0.7
            lr = max(lr * 0.9, 1e-7)
    return h_cur, cur_score


def simulated_annealing(h, n, target, n_steps, t_init, t_final, deadline, rng, restart_interval=400):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    x = np.arange(n)
    no_improve = 0

    for step in range(n_steps):
        if time.time() > deadline:

        if no_improve > restart_interval:

            no_improve = 0
            if rng.rand() < 0.3:
                h_cur = best_h.copy()
                cur_score = best_score
        else:
            progress = step / max(n_steps - 1, 1)
            T = t_init * (t_final / t_init) ** progress

        h_new = h_cur.copy()
        move_type = rng.rand()

        if move_type < 0.25:
            m = rng.randint(1, max(2, n // 4))
            phase = rng.uniform(0, 2.0 * np.pi)
            amplitude = rng.uniform(0.008, 0.05) * (1.0 if rng.rand() < 0.5 else -1.0)
            h_new = h_cur + amplitude * np.cos(2.0 * np.pi * m * x / n + phase)
        elif move_type < 0.45:

                no_improve += 1

            no_improve += 1

    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, deadline, rng):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            if k > 0:
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= 0.97
            if lr < 1e-9:
                break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def de_diverse(n, target, pop_size, max_gen, rng, deadline, F=0.7, CR=0.85):
    population = []
    for i in range(pop_size):
        base = np.full(n, 0.5)
        m1 = rng.randint(1, n // 3)
        m2 = rng.randint(1, n // 3)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.1, 0.3)
        a2 = rng.uniform(0.05, 0.2)
        x = np.arange(n)
        base = base + a1 * np.cos(2 * np.pi * m1 * x / n + p1) + a2 * np.sin(2 * np.pi * m2 * x / n + p2)
        base = project_feasible(base, target)
        population.append(base)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score
