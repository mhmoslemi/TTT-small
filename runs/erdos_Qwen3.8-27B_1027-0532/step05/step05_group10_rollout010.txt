def project_h(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target}"""
    n = len(h)
    # Binary search for tau
    lo, hi = -1.0, 1.0 + 1.0  # tau range
    for _ in range(50):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi)/2, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - np.sum(proj)
    if abs(diff) > 1e-12:
        # Distribute among non-bound points
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def nesterov_optimize(h0, target, n, alpha_schedule, max_iter, time_limit, lr_init=0.01, momentum_init=0.9):
    start = time.time()
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = momentum_init
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        alpha = alpha_schedule[min(it, len(alpha_schedule)-1)]
        lr = lr_init / (1.0 + 0.01 * it)
        
        # Compute gradient at y = h + momentum * v
        y = h + momentum * v
        y = project_h(y, target)
        
        grad = compute_grad_smin(y, alpha)
        
        # Nesterov update
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_h(h_new, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_refine(h0, n, target, alphas, maxiter, ftol, time_limit):

    h_cur = h0.copy()
    cur_score = compute_c5(h_cur)

            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -350.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

        def obj_grad(x, a=alpha):

            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -350.0, 0.0)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            dx = 2.0 / n
            grad = np.zeros(n)
            half_n = n - 1
            for k_idx in range(len(corr)):

                    continue

                    idx2 = np.arange(k, n)

                    idx2 = np.arange(0, n + k)

def lp_refine_sparse(h0, n, target, time_limit, n_active=25, damping=0.85, reg=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h0.copy()
    h = project_h(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(50000):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        n_cons = 0
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
                n_cons += 1

        if n_cons == 0:
            break

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = h_var.value is not None
            except Exception:
                pass

        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_h(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()

        top_k = set(np.argsort(corr)[-n_active:].tolist())
        active_k = top_k

    return best_h, best_c5

    # Phase 7: Nesterov final polish with very high alpha
    if tleft() > 50:
        h_opt, c5_opt = nesterov_optimize(
            best_h, target, n,
            alpha_schedule=[10000, 30000, 100000],
            iters_per_alpha=200,
            time_limit=min(35, tleft() * 0.35),
            lr_init=0.1, momentum=0.95, sa_temp=0.0
        )
        if c5_opt < best_score:
            best_score = c5_opt
            best_h = h_opt

    # Phase 8: Try higher resolution
    if tleft() > 150:
        n2 = n + 100
        if n2 % 2 == 0:
            target2 = n2 / 2.0
            h_interp = interpolate_to(best_h, n2, target2)
            h_opt2, c5_opt2 = nesterov_optimize(
                h_interp, target2, n2,
                alpha_schedule=alpha_schedule_coarse + alpha_schedule_fine,
                iters_per_alpha=300,
                time_limit=min(80, tleft() * 0.4),
                lr_init=0.3, momentum=0.9, sa_temp=0.0
            )
            if c5_opt2 < best_score:
                best_score = c5_opt2
                best_h = h_opt2
                n = n2
                target = target2

    # Phase 9: Final LP at current resolution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, c5_lp = lp_refine_sparse(
            best_h, n_use, target_use,
            time_limit=min(30, tleft() * 0.5),
            n_active=35, damping=0.9, reg=1e-6
        )
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_slq, c5_slq = slsqp_refine(
            best_h, n_use, target_use,
            alphas=[5000, 20000],
            maxiter=500, ftol=1e-15,
            time_limit=min(20, tleft() * 0.6)
        )
        if c5_slq < best_score:
            best_score = c5_slq
            best_h = h_slq

    # Phase 11: Final Nesterov micro-polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_opt, c5_opt = nesterov_optimize(
            best_h, target_use, n_use,
            alpha_schedule=[50000, 200000],
            iters_per_alpha=100,
            time_limit=min(12, tleft() * 0.7),
            lr_init=0.05, momentum=0.97, sa_temp=0.0
        )
        if c5_opt < best_score:
            best_score = c5_opt
            best_h = h_opt

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_h(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
