    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_grad(h, alpha, n):
    """Compute gradient of LSE-smoothed max correlation."""
    corr = compute_corr(h)
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        m = k - (n - 1)
        # First term: (1 - h[j-m]) for j where 0 ≤ j-m < n
        js = max(0, m)
        je = min(n, n + m)
        if js < je:
            idx = np.arange(js, je)
            grad[idx] += w * (1.0 - h[idx - m])
        # Second term: -h[j+m] for j where 0 ≤ j+m < n
        js2 = max(0, -m)
        je2 = min(n, n - m)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            grad[idx2] -= w * h[idx2 + m]
    
    return grad * dx

def lse_gradient_descent(h, n, target, deadline, alpha_start=100, alpha_end=5000, 
                         n_iters=2000, lr_start=0.005, lr_decay=0.999):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n

            break

        lr = lr_start * (lr_decay ** it)
        
        grad = lse_grad(h_cur, alpha, n)
        
        # Backtracking line search
        f_cur = compute_c5(h_cur)
        step = lr
        improved = False
        for bt in range(25):
            h_new = h_cur - step * grad
            h_new = project_feasible(h_new, target)
            f_new = compute_c5(h_new)
            if f_new < f_cur - 1e-14:

            h_new = project_feasible(h_new, target)
            f_new = compute_c5(h_new)
            if f_new < f_cur:

def simulated_annealing(h, n, target, deadline, n_iters=3000, T_start=0.05, T_end=0.001):

            idx2 = np.arange(js2, je2)
            grad[idx2] -= h_cur[idx2 + m]
        grad *= dx
        eps = 0.005
        for trial in range(20):
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
                break
            eps *= 0.7
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: LSE gradient descent (coarse alpha, long run)
    deadline_p1 = start_time + 250
    h1, s1 = lse_gradient_descent(best_h, n, target, deadline_p1,
                                  alpha_start=50, alpha_end=1000,
                                  n_iters=3000, lr_start=0.01, lr_decay=0.9995)
    if s1 < best_score:
        best_score = s1
        best_h = h1

    # Phase 2: SLSQP with increasing alpha
    if tleft() > 200:
        deadline_p2 = start_time + 450
        h2, s2 = slsqp_round(best_h, n, target, [200, 800, 3000], 3000, deadline_p2)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Weighted gradient descent focusing on worst lags
    if tleft() > 200:
        deadline_p3 = start_time + 650
        h3, s3 = weighted_grad_descent(best_h, n, target, deadline_p3,
                                       n_outer=30, beta_init=200,
                                       beta_growth=1.4, inner_iters=80,
                                       lr_init=0.002)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: SLSQP polish after WGD
    if tleft() > 150:
        deadline_p4 = start_time + 800
        h4, s4 = slsqp_round(best_h, n, target, [1000, 5000], 2500, deadline_p4)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Simulated annealing for exploration
    if tleft() > 200:
        deadline_p5 = start_time + 1000
        h5, s5 = simulated_annealing(best_h, n, target, deadline_p5,
                                     n_iters=8000, T_start=0.015,
                                     T_end=0.0003, seed=seed + 100)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: LSE gradient descent after SA (exploitation)
    if tleft() > 150:
        deadline_p6 = start_time + 1150
        h6, s6 = lse_gradient_descent(best_h, n, target, deadline_p6,
                                      alpha_start=500, alpha_end=10000,
                                      n_iters=2000, lr_start=0.003, lr_decay=0.999)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Alternating lag minimization
    if tleft() > 80:
        deadline_p7 = start_time + budget_s - 80
        h7, s7 = alternating_lag_min(best_h, n, target, deadline_p7, n_rounds=500)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 40:
        deadline_p8 = start_time + budget_s - 20
        h8, s8 = slsqp_round(best_h, n, target, [5000, 20000], 2000, deadline_p8)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Try higher resolution
    if tleft() > 60:
        n2 = int(n * 1.25)
        target2 = n2 / 2.0
        h_fine = interpolate_to(best_h, n2, target2)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2
        if tleft() > 30:
            deadline_p9 = start_time + budget_s - 10
            h9, s9 = lse_gradient_descent(best_h, n, target, deadline_p9,
                                          alpha_start=1000, alpha_end=10000,
                                          n_iters=500, lr_start=0.002, lr_decay=0.999)
            if s9 < best_score:
                best_score = s9
                best_h = h9

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
