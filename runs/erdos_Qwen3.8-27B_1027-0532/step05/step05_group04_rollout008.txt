    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def nesterov_lse(h, n, target, n_steps, alpha, lr_init, moment, deadline=None):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        full_corr = corr
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                j_pos = np.arange(0, n - lag)
                grad[j_pos + lag] += wk * (1.0 - h_cur[j_pos])
                grad[j_pos] -= wk * h_cur[j_pos + lag]
            else:
                j_neg = np.arange(-lag, n)
                grad[j_neg] += wk * (1.0 - h_cur[j_neg + (-lag)])
                grad[j_neg + (-lag)] -= wk * h_cur[j_neg]
        grad *= dx
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.01, 0.05)
        else:
            v *= 0.5
            lr *= 0.9
        if lr < 1e-7:
            break
    return best_h, best_score


def coordinate_descent(h, n, target, n_sweeps, n_trials, deadline=None):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    trial_vals = np.linspace(0.0, 1.0, n_trials)
    idx_range = np.arange(n)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        order = idx_range.copy()
        np.random.shuffle(order)
        for i in order:
            if deadline is not None and time.time() > deadline:
                break
            old_val = h_cur[i]
            best_val = old_val
            best_sc = best_score
            remaining = target - (np.sum(h_cur) - h_cur[i])
            for tv in trial_vals:
                if tv < max(0.0, remaining - (n - 1) * 1.0) - 1e-10:
                    continue
                if tv > min(1.0, remaining) + 1e-10:
                    continue
                h_test = h_cur.copy()
                h_test[i] = tv
                sc = compute_c5(h_test)
                if sc < best_sc:
                    best_sc = sc
                    best_val = tv
            if best_val != old_val:
                h_cur[i] = best_val
                best_score = best_sc
                improved = True
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def de_search(n, target, pop_size, max_generations, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        full_corr = corr
        c_curr = float(np.max(full_corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(full_corr)[::-1][:n_active]
        diffs = full_corr[top_idx] - full_corr[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            lag = k - (n - 1)
            if lag >= 0:
                j_pos = np.arange(0, n - lag)
                grad[j_pos + lag] += wk * (1.0 - h_cur[j_pos])
                grad[j_pos] -= wk * h_cur[j_pos + lag]
            else:
                j_neg = np.arange(-lag, n)
                grad[j_neg] += wk * (1.0 - h_cur[j_neg + (-lag)])
                grad[j_neg + (-lag)] -= wk * h_cur[j_neg]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score


def weighted_grad_hmc(h, n, target, n_steps, beta, lr, moment, deadline=None):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        max_corr = float(np.max(corr))
        weights = np.exp(beta * (corr - max_corr))
        weights /= weights.sum()
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = weights[k]
            if wk < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                j_pos = np.arange(0, n - lag)
                grad[j_pos + lag] += wk * (1.0 - h_cur[j_pos])
                grad[j_pos] -= wk * h_cur[j_pos + lag]
            else:
                j_neg = np.arange(-lag, n)
                grad[j_neg] += wk * (1.0 - h_cur[j_neg + (-lag)])
                grad[j_neg + (-lag)] -= wk * h_cur[j_neg]
        grad *= dx
        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        if sc_new < compute_c5(h_cur):
            h_cur = h_new
        else:
            v *= 0.7
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top3 = np.argsort(corr)[::-1][:3]
        improved = False
        for k_star in top3:
            k_star = int(k_star)
            lag = k_star - (n - 1)
            grad = np.zeros(n)
            if lag >= 0:
                j_pos = np.arange(0, n - lag)
                grad[j_pos + lag] += (1.0 - h_cur[j_pos])
                grad[j_pos] -= h_cur[j_pos + lag]
            else:
                j_neg = np.arange(-lag, n)
                grad[j_neg] += (1.0 - h_cur[j_neg + (-lag)])
                grad[j_neg + (-lag)] -= h_cur[j_neg]
            grad *= dx
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-12:
                h_cur = h_new
                cur_score = sc_new
                improved = True
        if improved:
            eps = min(eps * 1.04, 0.01)
        else:
            eps *= 0.7
            if eps < 1e-8:
                break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP from initial
    h_s, sc_s = slsqp_round(best_h, n, target, [50, 200, 1000], 2000, 1e-15,
                            deadline=start_time + 60)
    if sc_s < best_score:
        best_score = sc_s
        best_h = h_s

    # Phase 2: Coarse DE for diverse starts
    if tleft() > 300:
        n_c = 80
        h_c, _ = de_search(n_c, n_c / 2.0, 25, 120, rng, deadline=start_time + 120, F=0.8, CR=0.85)
        h_c_fine = interpolate_to(h_c, n, target)
        h_s2, sc_s2 = slsqp_round(h_c_fine, n, target, [200, 1000, 5000], 3000, 1e-15,
                                  deadline=start_time + 200)
        if sc_s2 < best_score:
            best_score = sc_s2
            best_h = h_s2

        n_c2 = 120
        h_c2, _ = de_search(n_c2, n_c2 / 2.0, 20, 80, rng, deadline=start_time + 260, F=0.7, CR=0.8)
        h_c2_fine = interpolate_to(h_c2, n, target)
        h_s3, sc_s3 = slsqp_round(h_c2_fine, n, target, [500, 2000, 8000], 3000, 1e-15,
                                  deadline=start_time + 340)
        if sc_s3 < best_score:
            best_score = sc_s3
            best_h = h_s3

    # Phase 3: Nesterov LSE
    if tleft() > 250:
        h_nes, sc_nes = nesterov_lse(best_h, n, target, n_steps=1000, alpha=2000,
                                     lr_init=0.005, moment=0.9,
                                     deadline=start_time + budget_s * 0.45)
        if sc_nes < best_score:
            best_score = sc_nes
            best_h = h_nes

    # Phase 4: Momentum weighted gradient
    if tleft() > 200:
        h_mom, sc_mom = weighted_grad_hmc(best_h, n, target, n_steps=800, beta=800,
                                          lr=0.003, moment=0.85,
                                          deadline=start_time + budget_s * 0.55)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom

    # Phase 5: SLSQP high alpha
    if tleft() > 200:
        h_s4, sc_s4 = slsqp_round(best_h, n, target, [5000, 20000, 50000], 5000, 1e-16,
                                  deadline=start_time + budget_s * 0.7)
        if sc_s4 < best_score:
            best_score = sc_s4
            best_h = h_s4

    # Phase 6: Subgradient polish
    if tleft() > 150:
        h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps=500, n_active=5,
                                         lr_init=0.005, lr_decay=0.997,
                                         deadline=start_time + budget_s * 0.78)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 7: Worst-lag surgery
    if tleft() > 120:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, max_steps=300,
                                            initial_epsilon=0.003,
                                            deadline=start_time + budget_s * 0.84)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 8: Coordinate descent (very effective for final polish)
    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent(best_h, n, target, n_sweeps=3, n_trials=9,
                                         deadline=start_time + budget_s * 0.92)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: SLSQP final high alpha
    if tleft() > 60:
        h_final, sc_final = slsqp_round(best_h, n, target, [50000, 200000], 5000, 1e-17,
                                        deadline=start_time + budget_s - 40)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 10: Final coordinate descent micro-polish
    if tleft() > 25:
        h_cd2, sc_cd2 = coordinate_descent(best_h, n, target, n_sweeps=2, n_trials=5,
                                           deadline=start_time + budget_s - 10)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 11: Try finer grid
    if tleft() > 60:
        n_fine = min(n * 2, 800)
        target_fine = n_fine / 2.0
        h_fine = interpolate_to(best_h, n_fine, target_fine)
        h_fine_s, sc_fine_s = slsqp_round(h_fine, n_fine, target_fine, [2000, 10000], 4000, 1e-16,
                                          deadline=start_time + budget_s - 30)
        if sc_fine_s < best_score:
            best_score = sc_fine_s
            best_h = h_fine_s

    # Final: coordinate descent on whatever is best
    if tleft() > 15:
        n_opt = len(best_h)
        target_opt = n_opt / 2.0
        h_cd3, sc_cd3 = coordinate_descent(best_h, n_opt, target_opt, n_sweeps=2, n_trials=7,
                                           deadline=start_time + budget_s - 5)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
