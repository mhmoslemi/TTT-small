import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_objective(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

def lse_gradient(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

def lse_fg(x, alpha, target, mu):
    f = lse_objective(x, alpha, target, mu)
    g = lse_gradient(x, alpha, target, mu)
    return f, g

def lbfgs_run(h0, target, alpha, mu, maxiter, deadline):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        return lse_objective(x, alpha, target, mu)
    def jac(x):
        return lse_gradient(x, alpha, target, mu)
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-15, 'maxls': 30})
        return project_feasible(res.x, target)
    except:
        return h0

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_fg(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_run(h0, target, alpha, mu, maxiter, deadline):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        return lse_fg(x, alpha, target, mu)[0]

    def jac(x):
        return lse_fg(x, alpha, target, mu)[1]

    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-15, 'maxls': 30})
        return project_feasible(res.x, target)
    except Exception:
        return h0


def newton_coord_descent(h_init, target, alpha, mu, n_sweeps, deadline):
    n = len(h_init)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    p = np.arange(-(n - 1), n)
    sw = 2 * n - 1

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            jp = j + p
            jp_valid = (jp >= 0) & (jp < n)
            jm = j - p
            jm_valid = (jm >= 0) & (jm < n)

            g = np.zeros(sw)
            g[jp_valid] += (1.0 - h[jp[jp_valid]])
            g[jm_valid] -= h[jm[jm_valid]]

            v0 = h[j]
            S = float(np.sum(h)) - v0
            vt = target - S

            u = 0.0
            for _ in range(12):
                vals = corr + u * g
                vm = float(np.max(vals))
                w = np.exp(alpha * (vals - vm))
                ws = float(np.sum(w))
                if ws < 1e-300:
                    break
                w /= ws
                sum_wg = float(np.sum(w * g))
                sum_wg2 = float(np.sum(w * g * g))
                fp = sum_wg + 2.0 * mu * (v0 + u - vt)
                fpp = alpha * (sum_wg2 - sum_wg * sum_wg) + 2.0 * mu
                if fpp < 1e-10:
                    fpp = 1e-10
                du = -fp / fpp
                u_new = u + du
                v_check = v0 + u_new
                if v_check < 0.0:
                    u_new = -v0
                elif v_check > 1.0:
                    u_new = 1.0 - v0
                u = u_new
                if abs(du) < 1e-12:
                    break

            v_new = v0 + u
            v_new = float(np.clip(v_new, 0.0, 1.0))
            if abs(v_new - v0) > 1e-13:
                delta_v = v_new - v0
                h[j] = v_new
                corr += delta_v * g

    h = project_feasible(h, target)
    return h, compute_c5(h)


def top_lag_equalize(h, target, n_top, n_steps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-13:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
    return h_cur, cur_score
