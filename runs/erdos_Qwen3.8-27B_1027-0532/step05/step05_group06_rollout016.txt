import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    """Project h onto the feasible set: 0<=h<=1, sum(h)=target_sum"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_objective(x, alpha, target, mu):
    """Log-sum-exp smoothed max of correlation values, plus penalty for sum constraint."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    return lse + mu * (s - target) ** 2

def lse_gradient(x, alpha, target, mu):
    """Gradient of the log-sum-exp objective."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])

            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)

def lbfgs_step(h, target, alpha, mu, maxiter):

    bounds = [(0.0, 1.0)] * n

        return lse_objective(x, alpha, target, mu)
    def g(x):
        return lse_gradient(x, alpha, target, mu)
    try:
        res = minimize(f, h.copy(), jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 40})
        return project_feasible(res.x, target)
    except:
        return h

def adam_optimize(h, target, alpha, mu, lr, n_iter, deadline, beta1=0.9, beta2=0.999):
    """Adam optimizer for the LSE objective."""
    n = len(h)
    h_cur = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    for i in range(n_iter):
        if i % 10 == 0 and time.time() > deadline:
            break
        t += 1
        g = lse_gradient(h_cur, alpha, target, mu)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_cur = h_cur - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_cur = project_feasible(h_cur, target)
    return h_cur

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_val(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    return lse + mu * (s - target) ** 2

def lse_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx
