```python
def compute_avg_grad(h, n, top_lags):
    """Compute average gradient of corr[m]*dx over m in top_lags."""
    dx = 2.0 / n
    nl = 2 * n - 1
    grad = np.zeros(n)
    
    for m in top_lags:
        shift = m - (n - 1)
        # (1-h) shifted by shift, masked to valid range
        # contributes (1-h)[j+shift] to grad[j] when 0<=j+shift<n
        # -h shifted by -shift, contributes -h[j-shift] to grad[j] when 0<=j-shift<n
        
        # For (1-h)[j+shift]: valid when -shift <= j < n-shift
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1+shift:hi1+shift])
        
        # For -h[j-shift]: valid when shift <= j < n+shift
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-shift:hi2-shift]
    
    return grad * dx / len(top_lags)
```

**Frank-Wolfe (Conditional Gradient) method:**
At each iteration:
1. Compute subgradient g of C5 at current h (gradient of a top-lag correlation)
2. Solve: min_{y ∈ P} g · y where P = {h: 0≤h≤1, ∑h_i=n/2}

   Actually, since P is a transportation polytope-like set, the linear min over P is:
   - Sort g, set y_i = 0 for the largest g_i values, y_i = 1 for the smallest, and one fractional value.
   - Specifically: let s = n/2. Sort indices by g value. Set y_i = 1 for the s smallest g_i (if s is integer, which it is here).
   - Wait, s = n/2 = 200 which is an integer. So y_i = 1 for the 200 indices with smallest g_i, 0 otherwise.
3. h_new = (1-γ) * h + γ * y
4. Decrease γ

This is very simple and fast! Each iteration is O(n log n) for the sort.

For a given threshold T, the feasibility problem is:

**Sequential Linear Programming (SLP) approach:**
1. Start with current h
2. Linearize c[m] around h: c[m] ≈ c[m]_current + ∇c[m]_current · (h - h_current)
3. Solve: min T subject to c[m]_current + ∇c[m]_current · (h - h_current) ≤ T for all m (or active m)
   with 0≤h≤1, ∑h=n/2
4. Update h, repeat

This is a linear program at each step (minimize T with linear constraints and bounds). I can use scipy.optimize.linprog or cvxpy.

The LP is:
min T
s.t. ∑_j ∂c[m]/∂h[j] * h[j] ≤ T - c[m]_current + ∑_j ∂c[m]/∂h[j] * h_current[j]
     for all m in active set
     0 ≤ h[j] ≤ 1
     ∑ h[j] = n/2

This is a LP with 2n-1 (or K) constraints + 2n bounds + 1 equality = about 2n + K variables and constraints. For n=400, K=50, that's 401 variables and ~450 constraints. Very manageable for a LP solver.

```python
def slp_step(h, n, target, K, max_slp_iters=100, deadline=None):
    from scipy.optimize import linprog
    dx = 2.0 / n
    h_cur = h.copy()
    best = compute_c5(h_cur)
    
    for it in range(max_slp_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        nl = len(corr)
        
        # Get top-K lags
        top_k = np.argsort(corr)[-K:]
        
        # Compute gradients for top lags
        A = np.zeros((K, n))
        b = np.zeros(K)
        for idx, m in enumerate(top_k):
            shift = m - (n - 1)
            lo1 = max(0, -shift)
            hi1 = min(n, n - shift)
            if lo1 < hi1:
                A[idx, lo1:hi1] += (1.0 - h_cur[lo1+shift:hi1+shift])
            lo2 = max(0, shift)
            hi2 = min(n, n + shift)
            if lo2 < hi2:
                A[idx, lo2:hi2] -= h_cur[lo2-shift:hi2-shift]
            b[idx] = corr[m] - A[idx] @ h_cur
        
        # LP: min T
        # s.t. A @ h + b <= T  =>  A @ h - T <= -b
        #      0 <= h <= 1
        #      sum(h) = n/2
        # Variables: [h_0, ..., h_{n-1}, T]
        # c = [0, ..., 0, 1] (minimize T)
        
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize T
        
        # Inequality constraints: A @ h - T <= -b
        # In matrix form: [A, -1] @ [h; T] <= -b
        A_ub = np.hstack([A, -np.ones((K, 1))])
        b_ub = -b
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: 0 <= h_i <= 1, T >= 0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        T_new = res.x[-1]
        
        # Project to feasible
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        
        if sc < best:
            best = sc
            h_cur = h_new
    
    return h_cur, best
```

```python
    dx = 2.0 / n

    best = compute_c5(h_cur)

            break
        
        # Compute autocorrelation
        R = np.correlate(h_cur, h_cur, mode='full')
        
        # Find lags with smallest R (these give largest C5)
        # C5 = max_m (n/2 - R[m]) * dx
        # So we want the m with smallest R[m]
        worst_lags = np.argsort(R)[:K]  # K worst lags

        # s.t. grad_R[m] · h >= R[m]_cur - grad_R[m] · h_cur + t  (linearized R[m] >= t)
        #      0 <= h <= 1, sum(h) = n/2, t is free

        # Constraint: R[m]_cur + grad[m]·h - grad[m]·h_cur >= t
        # => grad[m]·h - t >= grad[m]·h_cur - R[m]_cur

        # => -grad[m]·h + t <= R[m]_cur - grad[m]·h_cur
        # => [-grad[m], 1] @ [h; t] <= R[m]_cur - grad[m]·h_cur
        
        A = np.zeros((K, n))
        rhs = np.zeros(K)

            # dR[m]/dh[j] = h[j+shift] (if valid) + h[j-shift] (if valid)
            lo1 = max(0, -shift)
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
