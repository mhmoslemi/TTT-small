def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradients(h, n):
    """Compute gradients of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n) where grad[j, m] = dcorr[j]/dh[m]
    """
    mid = n - 1
    idxs = np.arange(n)
    grads = np.zeros((2*n-1, n))
    for j in range(2*n-1):
        k = j - mid
        for m in range(n):
            g = 0.0
            if 0 <= m + k < n:
                g += (1.0 - h[m + k])
            if 0 <= m - k < n:
                g -= h[m - k]
            grads[j, m] = g
    return grads

def slp_step_cvxpy(h, n, target, trust_radius=0.1):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_gradients_fast(h, n)
    
    # Variables: h_new (n), t (scalar)
    h_new = cp.Variable(n)
    t = cp.Variable()
    
    # Objective: minimize t
    objective = cp.Minimize(t)
    
    constraints = []
    # Linearized correlation constraints
    for j in range(2*n-1):
        corr_lin = corr[j] + grads[j, :] @ (h_new - h)
        constraints.append(corr_lin * dx <= t)
    
    # Box constraints
    constraints.append(h_new >= 0)
    constraints.append(h_new <= 1)
    
    # Sum constraint
    constraints.append(cp.sum(h_new) == target)
    
    # Trust region
    constraints.append(cp.norm(h_new - h, 2) <= trust_radius)
    
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=10000)
    except:
        try:
            prob.solve(solver=cp.ECOS)
        except:
            return None, None
    
    if h_new.value is None or t.value is None:
        return None, None
    
    h_new_val = np.clip(h_new.value, 0, 1)
    h_new_val = project_feasible(h_new_val, target)
    return h_new_val, t.value

def slp_step_linprog(h, n, target, damping=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    grads = compute_gradients_fast(h, n)  # shape (2n-1, n)
    
    # LP: min t
    # s.t. corr[j] + grads[j] @ h_new <= t  (for all j)
    #      sum(h_new) = target
    #      0 <= h_new <= 1
    #      h_new = h + delta (implicitly)
    #      
    # Rewrite: corr[j] + grads[j] @ (h_new) <= t
    #        => grads[j] @ h_new - t <= -corr[j]
    #        
    # Variables: x = [h_new (n), t (1)] => (n+1) variables
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    # 
    # Inequality constraints A_ub @ x <= b_ub:
    #   For each j: grads[j] @ h_new - t <= -corr[j]
    #   A_row_j = [grads[j, 0], ..., grads[j, n-1], -1]
    #   b_j = -corr[j]
    #
    #   -h_new <= 0 => A_row = [-I, 0], b = 0
    #
    # Equality constraints A_eq @ x = b_eq:
    #   sum(h_new) = target => A_row = [1, 1, ..., 1, 0], b = target
    #
    # Bounds: 0 <= h_new[i] <= 1, t is free (or t >= 0)
    
    n_vars = n + 1
    c_obj = np.zeros(n_vars)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints
    n_ineq = (2*n - 1) + n  # correlation constraints + non-negativity
    A_ub = np.zeros((n_ineq, n_vars))
    b_ub = np.zeros(n_ineq)
    
    for j in range(2*n-1):
        A_ub[j, :n] = grads[j, :]
        A_ub[j, n] = -1.0
        b_ub[j] = -corr[j]
    
    # -h_new <= 0
    A_ub[2*n-1:, :n] = -np.eye(n)
    b_ub[2*n-1:] = 0
    
    # Equality: sum(h_new) = target
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Bounds
    bounds = [(0, 1)] * n + [(0, None)]  # h_new in [0,1], t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        h_new, c5_new = slp_step_linprog(h_cur, n, target, damping=1.0)
        if h_new is None:
            break
        
        c5_actual = compute_c5(h_new)
        if c5_actual < best_c5 - 1e-14:
            best_c5 = c5_actual
            best_h = h_new.copy()
            h_cur = h_new
        elif c5_actual < best_c5:
            h_cur = h_new  # still use it for next iteration

            c5_damped = compute_c5(h_damped)

                best_c5 = c5_damped
                best_h = h_damped.copy()
                h_cur = h_damped
            else:
                break  # No improvement possible
    
    return best_h, best_c5

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

            pass

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=3000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: SLP + CD cycle
    if tleft() > 100:
        slp_time = min(80, tleft() * 0.15)
        h_slp, c5_slp = slp_loop(best_h, n, target, time_limit=slp_time, max_iters=300)
        if c5_slp < best_score:
            best_score = c5_slp
            best_h = h_slp
        if tleft() > 60:
            cd_time = min(60, tleft() * 0.15)
            h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                       max_sweeps=2000, n_cand=50)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 8: Multi-restart with perturbations + SLP
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.07, 0.12]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        slp_time = min(60, tleft() * 0.12)
        h_ref, c5_ref = slp_loop(h_pert, n, target, time_limit=slp_time, max_iters=300)
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref
        if tleft() < 60:
            break
        cd_time = min(40, tleft() * 0.1)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=1500, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 9: Try higher resolution
    for n_try in [500, 600, 800]:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_init = compute_c5(h_interp)
        if sc_init < best_score:
            best_score = sc_init
            best_h = h_interp
            n = n_try
            target = target_try
        slp_time = min(80, tleft() * 0.15)
        h_hires, c5_hires = slp_loop(h_interp, n_try, target_try,
                                     time_limit=slp_time, max_iters=300)
        if c5_hires < best_score:
            best_score = c5_hires
            best_h = h_hires
            n = n_try
            target = target_try
        if tleft() < 60:
            break
        cd_time = min(40, tleft() * 0.1)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target, time_limit=cd_time,
                                   max_sweeps=1500, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        slp_time = min(40, tleft() * 0.4)
        h_final_slp, c5_final = slp_loop(best_h, n_use, target_use,
                                         time_limit=slp_time, max_iters=200)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_slp

    # Phase 11: Final SLSQP + pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pol = slsqp_round(best_h, n_use, target_use,
                            [10000, 50000],
                            500, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(15, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use, time_limit=cd_time,
                                   max_sweeps=1000, n_cand=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
