            g[:n-L] -= wk * h[L:]
    g *= dx

        h_p = project_feasible(h_raw, n / 2.0)
        corr = compute_corr(h_p)
        cm = float(np.max(corr))
        sh = np.clip(alpha * (corr - cm), -700.0, 0.0)
        return (cm + float(np.log(np.sum(np.exp(sh)))) / alpha) * (2.0 / n)

    bounds = [(-0.48, 0.48)] * (2 * K)
    res = minimize(obj4, coefs_init, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    h_out = 0.5 + cos_terms @ res.x[0::2] + sin_terms @ res.x[1::2]
    return project_feasible(h_out, n / 2.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def active_set_polish(h, n, target, top_m, alpha, deadline, n_rounds=5):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        active_max = float(corr[top_idx[0]])

        def obj_as(x):
            c = compute_corr(x)
            c_act = c[top_idx]
            am = float(np.max(c_act))
            sh = np.clip(alpha * (c_act - am), -700.0, 0.0)
            return (am + float(np.log(np.sum(np.exp(sh)))) / alpha) * (2.0 / n)

        dx_ = 2.0 / n

        def grad_as(x):
            c = compute_corr(x)
            c_act = c[top_idx]
            am = float(np.max(c_act))
            sh = np.clip(alpha * (c_act - am), -700.0, 0.0)
            w = np.exp(sh)
            w /= w.sum()
            g = np.zeros(n)
            for j, m in enumerate(top_idx):
                wk = w[j]
                k = m - (n - 1)
                if k >= 0:
                    g[:n - k] += wk * (1.0 - x[k:])
                    g[k:] -= wk * x[:n - k]
                else:
                    L = -k
                    g[L:] += wk * (1.0 - x[:-L])
                    g[:n - L] -= wk * x[L:]
            g *= dx_
            return g

        bounds = [(0.0, 1.0)] * n

        def cons_fun(x):
            return np.sum(x) - target

        def cons_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_as, h_cur, method='SLSQP', jac=grad_as,
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15})
            h_cand = project_feasible(res.x, target)
            sc = compute_c5(h_cand)
            if sc < best_score:
                best_score = sc
                h_cur = h_cand
        except Exception:
            pass
    return h_cur, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(h)

    # Phase 1: L-BFGS-B with increasing alpha
    alphas_p1 = [30, 100, 500, 2000, 8000, 20000, 50000]
    for alpha in alphas_p1:
        if time.time() > deadline - 500:
            break
        h_new = lbfgsb_lse(best_h, n, target, alpha, maxiter=400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: SLSQP refinement
    for alpha in [5000, 20000, 50000]:
        if time.time() > deadline - 400:
            break
        h_new = slsqp_lse(best_h, n, target, alpha, maxiter=3000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Fourier reparameterization
    for K in [10, 15, 20, 25]:
        if time.time() > deadline - 350:
            break
        coefs = extract_fourier_coefs(best_h, n, K)
        h_f = optimize_fourier(coefs, n, K, 5000, 500, deadline - 350)
        sc = compute_c5(h_f)
        if sc < best_score:
            best_score = sc
            best_h = h_f
        # Polish the Fourier result back in full dimension
        h_pol = lbfgsb_lse(best_h, n, target, 10000, maxiter=300)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Nesterov projected gradient
    if time.time() < deadline - 250:
        h_ng = nesterov_pg(best_h, n, target, 20000, deadline - 250, lr_init=0.005, beta=0.92)
        sc = compute_c5(h_ng)
        if sc < best_score:
            best_score = sc
            best_h = h_ng

    # Phase 5: Active set polish
    if time.time() < deadline - 200:
        for top_m in [3, 5, 8]:
            if time.time() > deadline - 100:
                break
            h_as, sc_as = active_set_polish(best_h, n, target, top_m, 10000, deadline - 100, n_rounds=3)
            if sc_as < best_score:
                best_score = sc_as
                best_h = h_as

    # Phase 6: Random perturbations + re-optimization
    if time.time() < deadline - 200:
        scales = [0.04, 0.08, 0.12, 0.06, 0.15, 0.2]
        for scale in scales:
            if time.time() > deadline - 60:
                break
            h_pert = project_feasible(best_h + rng.randn(n) * scale, target)
            h_opt = lbfgsb_lse(h_pert, n, target, 5000, maxiter=300)
            sc = compute_c5(h_opt)
            if sc < best_score:
                best_score = sc
                best_h = h_opt
                h_opt2 = slsqp_lse(best_h, n, target, 20000, maxiter=2000)
                sc2 = compute_c5(h_opt2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_opt2

    # Phase 7: Higher resolution n=600
    if time.time() < deadline - 300:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [50, 200, 1000, 5000, 20000]:
            if time.time() > deadline - 100:
                break
            h2_new = lbfgsb_lse(best_h2, n2, target2, alpha, maxiter=300)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 80:
            h2_new = slsqp_lse(best_h2, n2, target2, 10000, maxiter=2000)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 50:
            h2_f = extract_fourier_coefs(best_h2, n2, 15)
            h2_f = optimize_fourier(h2_f, n2, 15, 5000, 300, deadline - 50)
            sc2 = compute_c5(h2_f)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_f
            h2_pol = lbfgsb_lse(best_h2, n2, target2, 20000, maxiter=200)
            sc2 = compute_c5(h2_pol)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_pol
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 8: Final nesterov + surgery
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_ng = nesterov_pg(best_h, n_cur, target_cur, 30000, deadline - 30, lr_init=0.003, beta=0.93)
        sc = compute_c5(h_ng)
        if sc < best_score:
            best_score = sc
            best_h = h_ng

    # Phase 9: Final SLSQP
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_lse(best_h, n_cur, target_cur, 50000, maxiter=3000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
