    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    return c_max + np.log(w_sum) / alpha

def slsqp_grad(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    w = w / w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-13:
            g[:n-k] += wk * (1.0 - x[k:])
            g[k:] -= wk * x[:n-k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk > 1e-13:
            g[ell:] += wk * (1.0 - x[:n-ell])
            g[:n-ell] -= wk * x[ell:]
    g *= dx
    return g

def cons_sum(x, target):
    return np.sum(x) - target

def cons_sum_jac(x, target):
    return np.ones(len(x))
