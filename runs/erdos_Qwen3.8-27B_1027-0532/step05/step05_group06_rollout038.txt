import numpy as np

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_and_grad(h, alpha):
    """Compute smoothed objective and gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    obj = c_max + float(np.log(wsum)) / alpha
    
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:

                grad[j2] -= w * h[j2 + lag]
        else:

            if m > 0:

    return obj, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f_and_g(x):
        obj, grad = compute_c5_and_grad(x, alpha)
        s = float(np.sum(x))
        obj += mu * (s - target) ** 2
        grad += 2.0 * mu * (s - target)
        return obj, grad
    
    def f(x):
        return f_and_g(x)[0]
    
    def g(x):
        return f_and_g(x)[1]
    
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25,
                                'gtol': 1e-16, 'maxls': 80, 'maxfun': maxiter * 5})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def fista_optimize(h, target, alpha, mu, n_iter, deadline):
    """FISTA for the smoothed problem with box constraints."""
    n = len(h)
    x = h.copy()
    y = h.copy()
    t = 1.0
    L = alpha * 2.0 + 2.0 * mu + 1.0  # Lipschitz estimate (conservative)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = compute_c5_and_grad(y, alpha)
        s = float(np.sum(y))
        grad += 2.0 * mu * (s - target)
        
        # Gradient step
        z = y - grad / L
        
        # Project onto [0,1] and adjust for sum constraint
        z = np.clip(z, 0.0, 1.0)
        z = project_feasible(z, target)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = z + (t - 1.0) / t_new * (z - x)
        y = np.clip(y, 0.0, 1.0)
        x = z
        t = t_new
    
    return project_feasible(x, target)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj_grad(h, alpha, mu, target):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    obj = c_max + float(np.log(wsum)) / alpha
    s = float(np.sum(h))
    obj += mu * (s - target) ** 2
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-12:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            idx = np.arange(lag, n)
            grad[idx] += w * (1.0 - h[idx - lag])
            if lag > 0:
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= w * h[idx2 + lag]
        else:
            m = -lag
            idx = np.arange(0, n - m)
            grad[idx] += w * (1.0 - h[idx + m])
            if m > 0:
                idx2 = np.arange(m, n)
                grad[idx2] -= w * h[idx2 - m]
    grad += 2.0 * mu * (s - target)
    return obj, grad

def lbfgs_opt(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return smooth_obj_grad(x, alpha, mu, target)[0]
    def g(x):
        return smooth_obj_grad(x, alpha, mu, target)[1]
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-25, 'gtol': 1e-17, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgrad_project(h, target, n_steps, lr, deadline):
    """Projected subgradient on the true max."""
    n = len(h)
    h_cur = h.copy()
    best = compute_c5(h_cur)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = float(np.max(corr))
        # Subgradient: average gradient over all maximizing lags
        mask = corr >= max_val - 1e-10
        n_active = int(mask.sum())
        grad = np.zeros(n)
        active_idx = np.where(mask)[0]
        for ki in active_idx:
            lag = ki - (n - 1)
            if lag >= 0:
                idx = np.arange(lag, n)
                grad[idx] += (1.0 - h_cur[idx - lag])
                if lag > 0:
                    idx2 = np.arange(0, n - lag)
                    grad[idx2] -= h_cur[idx2 + lag]
            else:
                m = -lag
                idx = np.arange(0, n - m)
                grad[idx] += (1.0 - h_cur[idx + m])
                if m > 0:
                    idx2 = np.arange(m, n)
                    grad[idx2] -= h_cur[idx2 - m]
        grad /= n_active
        # Projected subgradient step with backtracking
        step_size = lr
        improved = False
        for _ in range(30):
            h_new = h_cur - step_size * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best:
                h_cur = h_new
                best = sc
                improved = True
                break
            step_size *= 0.5
        if not improved and step_size < 1e-14:
            break
    return h_cur, best

    return np.interp(x_new, x_old, h)
