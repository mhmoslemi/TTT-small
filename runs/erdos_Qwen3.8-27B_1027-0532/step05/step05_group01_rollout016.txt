        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5


def multi_active_descent(h, n, target, dx, max_steps, n_active=5, base_lr=0.003):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    idxs = np.arange(n)
    for step in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        scaled = corr * dx
        c_max = np.max(scaled)
        n_corr = 2 * n - 1
        top_idx = np.argsort(scaled)[-n_active:]
        alpha_w = 200.0
        w = np.exp(alpha_w * (scaled[top_idx] - c_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                improved = True
                break
        if not improved and step > 50:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_dx, _g_alpha, _g_top_k, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    alphas_lbfgsb = [50, 200, 1000, 5000, 20000, 80000, 200000]
    top_k_schedule = [30, 25, 20, 15, 12, 10, 8]

    for i, (alpha, tk) in enumerate(zip(alphas_lbfgsb, top_k_schedule)):
        if tleft() < 60:
            break
        h_new = run_lbfgsb(best_h, n, target, dx, alpha, tk, maxiter=500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 80:
        h_sl = slsqp_round(best_h, n, target, [20000, 80000, 200000], 400)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    if tleft() > 40:
        h_sd, c5_sd = steepest_descent_max(best_h, n, target, dx, max_steps=500, base_lr=0.004)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd

    if tleft() > 40:
        h_ma, c5_ma = multi_active_descent(best_h, n, target, dx, max_steps=500, n_active=5, base_lr=0.002)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    rng = np.random.RandomState(seed)
    for noise_scale in [0.005, 0.01, 0.02, 0.05]:
        if tleft() < 100:
            break
        h_pert = project_feasible(best_h + rng.randn(n) * noise_scale, target)
        for alpha in [10000, 50000, 200000]:
            if tleft() < 40:
                break
            h_new = run_lbfgsb(h_pert, n, target, dx, alpha, 10, maxiter=300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_pert = h_new

    if tleft() > 80:
        h_sl = slsqp_round(best_h, n, target, [50000, 200000], 500)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    if tleft() > 60:
        h_sd, c5_sd = steepest_descent_max(best_h, n, target, dx, max_steps=300, base_lr=0.003)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd
        h_ma, c5_ma = multi_active_descent(best_h, n, target, dx, max_steps=300, n_active=3, base_lr=0.002)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    for n_try in [600, 800]:
        if tleft() < 120:
            break
        target_try = n_try / 2.0
        dx_try = 2.0 / n_try
        h_interp = interpolate_to(best_h, n_try, target_try)
        for alpha, tk in [(5000, 15), (50000, 10), (200000, 8)]:
            if tleft() < 50:
                break
            h_new = run_lbfgsb(h_interp, n_try, target_try, dx_try, alpha, tk, maxiter=400)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_interp = h_new
        if tleft() > 40:
            h_sl = slsqp_round(h_interp, n_try, n_try / 2.0, [100000], 300)
            sc = compute_c5(h_sl)
            if sc < best_score:
                best_score = sc
                best_h = h_sl

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        dx_use = 2.0 / n_use
        h_final = slsqp_round(best_h, n_use, target_use, [100000, 500000], 600)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        dx_use = 2.0 / n_use
        h_sd, c5_sd = steepest_descent_max(best_h, n_use, target_use, dx_use, max_steps=500, base_lr=0.003)
        if c5_sd < best_score:
            best_score = c5_sd
            best_h = h_sd
        h_ma, c5_ma = multi_active_descent(best_h, n_use, target_use, dx_use, max_steps=500, n_active=3, base_lr=0.002)
        if c5_ma < best_score:
            best_score = c5_ma
            best_h = h_ma

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
