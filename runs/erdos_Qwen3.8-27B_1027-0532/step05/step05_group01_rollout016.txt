def gradient_for_shift_vec(h, shift):

    shift is the actual array index shift (positive or negative)."""

def precompute_coord_gradients(h):
    """Precompute the effect of each coordinate on all correlation values.
    Returns a matrix G of shape (n, 2n-1) where G[i, k] = d corr[k] / d h[i]."""
    n = len(h)
    n_corr = 2 * n - 1
    mid = n - 1
    G = np.zeros((n, n_corr))
    
    for i in range(n):
        # d corr[k] / d h[i]
        # corr[k] = sum_j h[j] * (1-h[j+shift]) where shift = k - mid
        # d/dh[i]: (1-h[i+shift]) if i+shift valid - h[i-shift] if i-shift valid

            if 0 <= i + shift < n:
                g += (1.0 - h[i + shift])
            if 0 <= i - shift < n:

```python
def exact_line_search(corr, d, delta_lo, delta_hi):
    """Find delta in [delta_lo, delta_hi] that minimizes max(corr + delta * d).
    Return the optimal delta and the resulting max value."""
    # The function f(δ) = max(corr + δ*d) is the upper envelope of lines.
    # f is piecewise linear and convex.
    # Find the minimum by checking all breakpoints.
    
    # A breakpoint occurs where two lines (corr[i] + δ*d[i]) cross another
    # (corr[j] + δ*d[j]), i.e., δ = (corr[j] - corr[i]) / (d[i] - d[j]) for d[i] ≠ d[j]
    
    # But checking all pairs is O(n^2). For n=400, that's 80,000 pairs.
    # Alternative: use the fact that f(δ) is convex and use ternary search or
    # binary search on the derivative.
    
    # The subgradient of f at δ is: max_{i: corr[i]+δ*d[i] = f(δ)} d[i]
    # f is minimized where 0 is in the subgradient, i.e., there exist i,j with
    # corr[i]+δ*d[i] = corr[j]+δ*d[j] = f(δ) and d[i] ≤ 0 ≤ d[j]
    
    # Simple approach: binary search on δ
    # f'(δ) (right derivative) = max_{i in argmax} d[i]
    # f'(δ) (left derivative) = min_{i in argmax} d[i]
    # f is minimized where left_deriv ≤ 0 ≤ right_deriv
    
    a, b = delta_lo, delta_hi
    for _ in range(60):  # enough for double precision
        mid = 0.5 * (a + b)
        vals = corr + mid * d
        f_mid = vals.max()
        active = vals >= f_mid - 1e-14
        left_deriv = d[active].min()
        right_deriv = d[active].max()
        if left_deriv > 1e-14:
            # f is increasing at mid, go left
            b = mid
        elif right_deriv < -1e-14:
            # f is decreasing at mid, go right
            a = mid
        else:
            # 0 is in subgradient, found minimum
            break
    
    delta_star = 0.5 * (a + b)
    f_star = np.max(corr + delta_star * d)
    return delta_star, f_star
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_gradient_shift(h, shift):
    """Gradient of f_shift w.r.t. h, where shift is the array index offset."""
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        grad[:m] += (1.0 - h[shift:])
        grad[shift:] -= h[:m]
    else:
        s = -shift
        m = n - s
        grad[s:] += (1.0 - h[:m])
        grad[:m] -= h[s:]
    return grad

def smooth_max_topk(corr, alpha, top_k):
    """Top-k soft max of corr values."""
    n_corr = len(corr)
    if top_k >= n_corr:
        idx = np.arange(n_corr)
    else:
        idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[idx]
    c_max = vals.max()
    shifted = np.clip(alpha * (vals - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = c_max + log_sum / alpha
    weights = exp_s / exp_s.sum()
    return obj, idx, weights

def slsqp_optimize(h0, n, target, alphas, maxiter_per_alpha, ftol, top_k=50):
    """SLSQP with top-k soft max objective."""
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h0.copy()
    
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, top_k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def _smooth_obj(h, alpha, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj, idx, w = smooth_max_topk(corr, alpha, top_k)
    return float(obj)
