import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n, adjusted to have sum = n//2."""
    # Generate longer sequence
    m = 1
    while m < n:
        m <<= 1
    tm = np.zeros(m, dtype=int)
    for i in range(1, m):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Take first n
    h = tm[:n].astype(float)
    # Adjust to have exactly n//2 ones
    return project_binary(h, n // 2)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    dx = 2.0 / n
    positions = np.arange(n)
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        
        # Find top M shifts
        top_M_idx = np.argsort(full)[-M:][::-1]  # descending
        top_M_vals = full[top_M_idx]
        
        # For each top shift, compute A and B
        # A[i] = contribution of h[i]=1 to overlap at that shift
        # B[j] = contribution of h[j]=0 to overlap at that shift
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        # delta[k] = h[i+k] + h[i-k] - h[j-k] - h[j+k]
        # We want to minimize sum_{k in top_M} (full[k] + delta[k])^2
        # ≈ sum_{k in top_M} (full[k]^2 + 2*full[k]*delta[k]) [ignoring delta^2]
        # = const + 2 * sum_{k in top_M} full[k] * delta[k]

        # For fixed j: score_j = sum_{k in top_M} full[k] * (h[j-k] + h[j+k])
        # Total score = score_i - score_j (we want to minimize this)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2 * n - 1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:

            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1

def thue_morse_binary(n):
    m = 1
    while m < n:
        m <<= 1
    tm = np.zeros(m, dtype=np.int8)
    for i in range(1, m):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm[:n].astype(float)
    return project_binary(h, n // 2)

        for n_try in [n_init + 2, n_init - 2, n_init + 4, n_init + 6]:
            if tleft() < 30:
                break
            if n_try < 100 or n_try % 2 != 0:
                continue
            target_try = n_try // 2
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try)

            allowed = min(12, tleft() * 0.05)
            h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=8000, rng=rng,
                                      time_limit=allowed, start_time=start_time)
            if tleft() < 20:
                break
            allowed2 = min(10, tleft() * 0.04)
            h_s3, c5_s3 = sa_focused(h_g3, n_iters=300000, t_init=0.006,
                                      t_final=0.00002, rng=rng,
                                      time_limit=allowed2, start_time=start_time)
            if c5_s3 < c5_g3:
                h_try = h_s3
                c5_try = c5_s3
            else:
                h_try = h_g3
                c5_try = c5_g3

            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try.copy()

    if tleft() > 30:
        allowed = min(60, tleft() - 15)
        h_t2, c5_t2 = tabu_sa(best_h, max_iters=300000, tabu_len=40,
                              t_init=0.008, t_final=0.00003, rng=rng,
                              time_limit=allowed, start_time=start_time)
        if c5_t2 < best_score:
            best_score = c5_t2
            best_h = h_t2.copy()

    if tleft() > 20:
        allowed = min(10, tleft() - 5)
        h_f, c5_f = fast_greedy(best_h, max_iters=15000, rng=rng,
                                 time_limit=allowed, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f.copy()

    if tleft() > 10:
        allowed = min(5, tleft() - 3)
        h_m2, c5_m2 = greedy_multi_shift(best_h, max_iters=3000, M=4,
                                         rng=rng, time_limit=allowed, start_time=start_time)
        if c5_m2 < best_score:
            best_score = c5_m2
            best_h = h_m2.copy()

    n_final = len(best_h)
    target_final = n_final // 2
    h_final = project_binary(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
