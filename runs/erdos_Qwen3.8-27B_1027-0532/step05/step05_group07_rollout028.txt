_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    c_max = float(np.max(corr))

    #   h[i] * (1-h[j]) where j = i + lag

def compute_B_fast(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    # For lag = -(n-1) to (n-1), index m = lag + n - 1
    # B[m] = (1-h[i+lag]) if 0<=i+lag<n, minus h[i-lag] if 0<=i-lag<n
    
    # Vectorize:
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    # i + lags: which are in [0, n)?
    idx_plus = i + lags  # shape (2n-1,)
    mask_plus = (idx_plus >= 0) & (idx_plus < n)
    # i - lags: which are in [0, n)?
    idx_minus = i - lags
    mask_minus = (idx_minus >= 0) & (idx_minus < n)
    
    B[mask_plus] += 1.0 - h[idx_plus[mask_plus]]
    B[mask_minus] -= h[idx_minus[mask_minus]]
    return B
