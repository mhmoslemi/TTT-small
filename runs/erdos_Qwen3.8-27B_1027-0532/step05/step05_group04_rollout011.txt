                continue
            j_pos = np.arange(0, n - k)
            g[j_pos] += ck * (1.0 - x[j_pos + k])
            j_neg = np.arange(k, n)
            g[j_neg] -= ck * x[j_neg - k]
        g *= dx
        return g
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h.copy()

def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        h_new = slsqp_smooth_max(h_cur, n, target, a, maxiter, ftol)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
    return h_cur

def reweighted_descent(h, n, target, n_outer, beta_init, beta_growth,
                       n_inner, lr_init, deadline=None, top_k=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    beta = beta_init
    dx = 2.0 / n
    for it in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        if top_k is not None and top_k < n:
            top_idx = np.argsort(over)[::-1][:top_k]
            weights = np.zeros(n)
            max_top = float(np.max(over[top_idx]))
            for idx in top_idx:
                weights[int(idx)] = np.exp(beta * (over[int(idx)] - max_top))
            weights /= weights.sum()
        else:
            weights = get_weights(over, beta)
        for inner in range(n_inner):
            if deadline is not None and time.time() > deadline:
                break
            grad = weighted_gradient(h_cur, n, weights)
            h_new, sc_new, _ = backtracking_step(h_cur, grad, target, lr_init)
            if sc_new < cur_score:
                h_cur = h_new
                cur_score = sc_new
        if (it + 1) % 4 == 0:
            if deadline is not None and time.time() < deadline - 10:
                h_new = slsqp_smooth_max(h_cur, n, target, 300, 2000, 1e-14)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
        beta = min(beta * beta_growth, 200000.0)
    return h_cur, cur_score

def lag_equalization(h, n, target, max_steps=200, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(max_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top2 = np.argsort(over)[::-1][:2]
        k1, k2 = int(top2[0]), int(top2[1])
        diff_val = over[k1] - over[k2]
        if abs(diff_val) < 1e-12:
            break
        # Try to reduce the gap by moving gradient of worst lag
        grad_worst = gradient_for_lag(h_cur, k1, n)
        grad_second = gradient_for_lag(h_cur, k2, n)
        combined = grad_worst - 0.5 * grad_second
        for eps in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps * combined
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-13:
                h_cur = h_new
                cur_score = sc_new
                break
    return h_cur, cur_score

def subgradient_fine(h, n, target, n_steps=500, n_active=2, lr_init=0.001,
                     lr_decay=0.997, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multi_scale_optimize(h, n, target, deadline):
    coarse_sizes = [50, 100, 200]
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for cs in coarse_sizes:
        if time.time() > deadline:
            break
        h_c = interpolate_to(best_h, cs, cs / 2.0)
        sc_c = compute_c5(h_c)
        h_c = slsqp_round(h_c, cs, cs / 2.0, [50, 200, 800], 2000, 1e-13,
                          deadline=min(deadline, start_time + 30))
        sc_c = compute_c5(h_c)
        h_fine = interpolate_to(h_c, n, target)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_h = h_fine
            best_score = sc_fine
    return best_h, best_score

start_time = 0.0

def run(seed=42, budget_s=1300, **kwargs):
    global start_time
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Multi-scale initialization (0-60s)
    if tleft() > 60:
        deadline_ms = start_time + 50
        h_ms, sc_ms = multi_scale_optimize(best_h, n, target, deadline_ms)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms

    # Phase 2: Initial SLSQP cleanup (50-130s)
    if tleft() > 80:
        h_ref = slsqp_round(best_h, n, target, [30, 100, 400, 1200],
                            3000, 1e-15, deadline=start_time + 130)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Main reweighted descent with backtracking (130-450s)
    if tleft() > 300:
        h_rw, sc_rw = reweighted_descent(best_h, n, target,
                                         n_outer=40, beta_init=200, beta_growth=1.4,
                                         n_inner=50, lr_init=0.002,
                                         deadline=start_time + 450, top_k=5)
        if sc_rw < best_score:
            best_score = sc_rw
            best_h = h_rw

    # Phase 4: SLSQP midpoint refinement (450-550s)
    if tleft() > 200:
        h_ref = slsqp_round(best_h, n, target, [500, 2000, 6000],
                            4000, 1e-15, deadline=start_time + 550)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Lag equalization (550-700s)
    if tleft() > 150:
        h_gap, sc_gap = lag_equalization(best_h, n, target, max_steps=500,
                                         deadline=start_time + 700)
        if sc_gap < best_score:
            best_score = sc_gap
            best_h = h_gap
        # Quick SLSQP after equalization
        if tleft() > 100:
            h_ref = slsqp_round(best_h, n, target, [2000, 6000],
                                3000, 1e-15, deadline=start_time + 750)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 6: Aggressive reweighted descent (750-1000s)
    if tleft() > 250:
        h_rw2, sc_rw2 = reweighted_descent(best_h, n, target,
                                           n_outer=50, beta_init=800, beta_growth=1.5,
                                           n_inner=40, lr_init=0.001,
                                           deadline=start_time + 1000, top_k=3)
        if sc_rw2 < best_score:
            best_score = sc_rw2
            best_h = h_rw2

    # Phase 7: Subgradient fine polish (1000-1100s)
    if tleft() > 120:
        h_sg, sc_sg = subgradient_fine(best_h, n, target, n_steps=400, n_active=2,
                                       lr_init=0.0008, lr_decay=0.998,
                                       deadline=start_time + 1100)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 8: High-precision SLSQP (1100-1250s)
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [5000, 15000, 40000],
                            5000, 1e-16, deadline=start_time + 1250)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final micro-polish (1250-1280s)
    if tleft() > 30:
        h_sg2, sc_sg2 = subgradient_fine(best_h, n, target, n_steps=200, n_active=1,
                                         lr_init=0.0003, lr_decay=0.999,
                                         deadline=start_time + budget_s - 15)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2
        h_ref = slsqp_round(best_h, n, target, [50000], 3000, 1e-16,
                            deadline=start_time + budget_s - 5)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
