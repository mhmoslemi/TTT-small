import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _lse_grad(x):
    n = _g_n
    dx = 2.0 / n
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    s = float(np.sum(w))
    if s < 1e-300:
        return np.zeros(n)
    w = w / s
    grad = np.zeros(n)
    for k in range(n):
        wk = _g_alpha * w[k]
        if wk < 1e-12:
            continue
        jp = np.arange(0, n - k)
        grad[jp] += wk * (1.0 - x[jp + k])
        jn = np.arange(k, n)
        grad[jn] -= wk * x[jn - k]
    grad *= dx
    return grad


def lbfgsb_lse(h, alpha, maxiter, ftol, gtol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * _g_n
    try:
        res = minimize(_lse_obj, h, method='L-BFGS-B', jac=_lse_grad,
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return project_feasible(res.x, _g_target)
    except Exception:
        return h.copy()


def nesterov_search(h, n, target, n_steps, alpha_start, alpha_end, lr_init, beta, deadline):
    global _g_n, _g_alpha
    _g_n = n
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    n_actual = max(1, n_steps)
    for step in range(n_actual):
        if time.time() > deadline:
            break
        frac = step / max(1, n_actual - 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        _g_alpha = alpha
        y = h_cur + beta * v
        y = project_feasible(y, target)
        grad = _lse_grad(y)
        v_new = beta * v - lr_init * grad
        h_trial = project_feasible(h_cur + v_new, target)
        sc_trial = compute_c5(h_trial)
        lr = lr_init
        for _ in range(12):
            if sc_trial <= cur_score:
                break
            lr *= 0.4
            v_new = beta * v - lr * grad
            h_trial = project_feasible(h_cur + v_new, target)
            sc_trial = compute_c5(h_trial)
        if sc_trial <= cur_score:
            v = v_new
            h_cur = h_trial
            cur_score = sc_trial
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        else:
            v = beta * v
    return best_h, best_score


def topk_subgrad(h, n, target, n_steps, top_k, lr_init, decay, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top_idx = np.argsort(over)[::-1][:top_k]
        top_max = over[top_idx[0]]
        diffs = over[top_idx] - top_max
        w = np.exp(diffs / 0.005)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_h = h_cur.copy()
        else:
            lr *= decay
            if lr < 1e-10:
                break
    return best_h, best_score


def slsqp_lse(h, n, target, alpha, maxiter, ftol):
    global _g_n, _g_alpha
    _g_n = n
    _g_alpha = alpha

    def _cons_fun(x):
        return np.sum(x) - target

    def _cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_lse_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def multi_restart(h, n, target, n_restarts, scale, deadline, alpha):
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(12345)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        h_pert = best_h + scale * (0.7 ** r) * rng.randn(n)
        h_pert = project_feasible(h_pert, target)
        h_ref = lbfgsb_lse(h_pert, alpha, 500, 1e-16, 1e-10)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    return best_h, best_score


def pair_move(h, n, target, n_trials, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    rng = np.random.RandomState(777)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        worst_k = int(np.argmax(over))
        jp = np.arange(0, n - worst_k)
        contrib = h_cur[jp] * (1.0 - h_cur[jp + worst_k])
        top_c = np.argsort(contrib)[::-1][:8]
        improved = False
        for j_idx in top_c:
            j = int(j_idx)
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = h_cur[j] * frac
                if amt < 1e-7:
                    continue
                h_try = h_cur.copy()
                h_try[j] -= amt
                h_try[j + worst_k] += amt
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score:
                    best_score = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    cur_score = sc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for _ in range(3):
                j1 = rng.randint(n)
                j2 = rng.randint(n)
                if j1 == j2:
                    continue
                amt = 0.02 * min(h_cur[j1], 1.0 - h_cur[j2])
                if amt < 1e-7:
                    continue
                h_try = h_cur.copy()
                h_try[j1] -= amt
                h_try[j2] += amt
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_score:
                    best_score = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    cur_score = sc
                    break
    return best_h, best_score


def de_polish(h_best, n, target, pop_size, n_gens, F, CR, deadline):
    rng = np.random.RandomState(99)
    pop = [project_feasible(h_best + 0.01 * rng.randn(n), target) for _ in range(pop_size)]
    scores = [compute_c5(p) for p in pop]
    idx = np.argsort(scores)
    pop = [pop[i] for i in idx]
    scores = [scores[i] for i in idx]
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    if scores[0] < best_score:
        best_score = scores[0]
        best_h = pop[0].copy()
    for gen in range(n_gens):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            cands = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            trial = np.zeros(n)
            jr = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == jr:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]
            trial = project_feasible(trial, target)
            ts = compute_c5(trial)
            if ts <= scores[i]:
                pop[i] = trial
                scores[i] = ts
                if ts < best_score:
                    best_score = ts
                    best_h = trial.copy()
        idx = np.argsort(scores)
        pop = [pop[i] for i in idx]
        scores = [scores[i] for i in idx]
    return best_h, best_score


def alternating_lags(h, n, target, n_outer, lr, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    dx = 2.0 / n
    for outer in range(n_outer):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        worst_k = int(np.argmax(over))
        jp = np.arange(0, n - worst_k)
        jn = np.arange(worst_k, n)
        grad = np.zeros(n)
        grad[jp] += (1.0 - h_cur[jp + worst_k])
        grad[jn] -= h_cur[jn - worst_k]
        grad *= dx
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            if sc < best_score:
                best_score = sc
                best_h = h_cur.copy()
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    return best_h, best_score
