import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        # Add to elements that have room
        order = np.argsort(1.0 - h)  # prefer elements close to 1? No, prefer those with most room
        order = np.argsort(h)  # prefer elements close to 0 (most room to increase)
        for idx in order:
            add = min(1.0 - h[idx], diff)
            h[idx] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        diff = -diff
        order = np.argsort(-h)  # prefer elements close to 1 (most room to decrease)
        for idx in order:
            sub = min(h[idx], diff)
            h[idx] -= sub
            diff -= sub
            if diff < 1e-14:
                break
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        # Need to flip (current - target_sum) ones to zeros
        excess = current - target_sum
        ones = np.where(h_r == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])
        for idx in order[:excess]:
            h_r[ones[idx]] = 0
    else:
        deficit = target_sum - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:deficit]:
            h_r[zeros[idx]] = 1
    return h_r.astype(float)

            if diff < 1e-12:
                break
    return h

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    if current == target_sum:
        return h_r.astype(float)
    if current > target_sum:
        excess = current - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        h_r[ones[order[:excess]]] = 0
    else:
        deficit = target_sum - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        h_r[zeros[order[:deficit]]] = 1
    return h_r.astype(float)

def make_P_idx(n):
    j_idx = np.arange(n)[:, None]
    m_idx = np.arange(n)[None, :]
    return n - 1 + m_idx - j_idx

def compute_obj_grad(h, beta, P_idx):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_c = float(corr.max())
    shifted = np.clip(beta * (corr - max_c), -700.0, 700.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    obj = max_c + float(np.log(exp_shifted.sum())) / beta
    A = (p[P_idx] * one_minus_h[None, :]).sum(axis=1)
    B = (p[P_idx.T] * h[None, :]).sum(axis=1)
    grad = A - B
    return obj, grad

def slsqp_analytic(h_start, n, target, P_idx, alphas, max_time, start_time):
    global _n_global, _target_global
    _n_global = n
    _target_global = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        def obj_fn(h):
            o, _ = compute_obj_grad(h, alpha, P_idx)
            return o
        
        def grad_fn(h):
            _, g = compute_obj_grad(h, alpha, P_idx)
            return g
        
        try:
            res = minimize(obj_fn, h_cur, jac=grad_fn,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
        
        if time.time() - start_time > max_time:
            break
    
    return h_cur, curr_score

def binary_local_search(h, max_time, start_time, top_k=5, scan_width=30):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 2000
    
    while True:
        if time.time() - start_time > max_time:
            break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        top_indices = np.argsort(corr)[-top_k:]
        top_shifts = top_indices - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        for k in top_shifts:
            # Find i in ones where h[i+k]=0, and j in zeros where h[j+k]=1
            i_cand = []
            for i in ones:
                if 0 <= i + k < n and h[i + k] == 0:
                    i_cand.append(i)
            j_cand = []
            for j in zeros:
                if 0 <= j + k < n and h[j + k] == 1:
                    j_cand.append(j)
            
            if len(i_cand) == 0 or len(j_cand) == 0:
                continue
            
            i_cand = i_cand[:scan_width]
            j_cand = j_cand[:scan_width]
            
            for i in i_cand:
                for j in j_cand:
                    # Evaluate new max
                    new_max = -1.0
                    for t_idx in top_indices:
                        tk = t_idx - (n - 1)
                        d = 0.0
                        if 0 <= i + tk < n:
                            d -= (1.0 - h[i + tk])
                        if 0 <= i - tk < n:
                            d += h[i - tk]
                        if 0 <= j + tk < n:
                            d += (1.0 - h[j + tk])
                        if 0 <= j - tk < n:
                            d -= h[j - tk]
                        v = corr[t_idx] + d
                        if v > new_max:
                            new_max = v
                    # Also check if any other shift becomes the new max
                    # (approximate: check a few random shifts)
                    if new_max < max_val - best_improve:
                        best_improve = max_val - new_max
                        best_i = i
                        best_j = j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def perturb_binary(h, n_pert, rng):
    h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    return h

def sa_binary(h, target_sum, max_time, start_time, t_init, t_final, rng):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 10000
    it = 0
    total_iters = int((max_time - (time.time() - start_time)) * 2000)
    total_iters = max(total_iters, 1000)
    
    while True:
        if time.time() - start_time > max_time:
            break
        if stall > max_stall:
            break
        it += 1
        
        ratio = min(it / total_iters, 1.0)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        h_new = h.copy()
        r = rng.random()
        moved = False
        
        if r < 0.5:
            # Targeted: pick from worst shift
            worst_idx = int(np.argmax(corr))
            k = worst_idx - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1
                moved = True
        elif r < 0.8:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1
            moved = True
        else:
            # Double swap
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0
                h_new[ones[i2]] = 0
                h_new[zeros[j1]] = 1
                h_new[zeros[j2]] = 1
                moved = True
        
        if not moved:
            continue
        
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / max(t, 1e-15))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

        h_bin2 = project_binary(best_h, target_int)
        h_bin2, c5_bin2 = binary_local_search(h_bin2, max_time=min(80, tleft() - 50),
                                               start_time=start_time)
        if c5_bin2 < best_score:
            best_score = c5_bin2
            best_h = h_bin2.copy()
    
    # Phase 4: SA on binary
    if tleft() > 100:

        h_bin3, c5_sa = sa_binary(h_bin3, target_int, max_time=min(150, tleft() - 50),
                                   start_time=start_time, t_init=0.02, t_final=0.0001,

        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_bin3.copy()
    
    # Phase 5: Second round of continuous optimization

        alphas2 = [50, 200, 1000, 5000, 20000]
        h_cont2, c5_cont2 = slsqp_analytic(best_h, n, target, P_idx, alphas2,
                                            max_time=min(200, tleft() - 50),
                                            start_time=start_time)
        if c5_cont2 < best_score:
            best_score = c5_cont2
            best_h = h_cont2.copy()
    
    # Phase 6: ILS with continuous refinement

        n_pert = 3 + rng.randint(5)
        h_pert = perturb_binary(best_h, n_pert, rng)
        alphas3 = [50, 200, 1000, 5000]
        h_new, c5_new = slsqp_analytic(h_pert, n, target, P_idx, alphas3,
                                        max_time=min(60, tleft() - 40),
                                        start_time=start_time)
        if c5_new < best_score:
            best_score = c5_new
            best_h = h_new.copy()
    
    # Phase 7: Final binary polish
    if tleft() > 30:
        h_bin_f = project_binary(best_h, target_int)
        h_bin_f, c5_f = binary_local_search(h_bin_f, max_time=max(10, tleft() - 10),
                                             start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_bin_f.copy()
    
    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, n
