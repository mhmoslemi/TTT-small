def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def weighted_corr_grad(x, corr, weights, mu, target):
    """Gradient of sum_k weights[k]*corr[k] + mu*(sum(x)-target)^2"""
    n = len(x)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:

                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

def lse_obj_and_grad(x, alpha, target, mu):

def iterative_reweight(h_init, target, n_outer, K_schedule, mu, maxiter_inner, deadline):
    """Iteratively reweight the top-K lags and optimize."""
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_f = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for outer in range(n_outer):
        if time.time() > deadline - 5:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        K = K_schedule[min(outer, len(K_schedule)-1)]
        top_K = np.argsort(corr)[::-1][:K]
        
        # Compute weights: higher weight for higher lags
        top_vals = corr[top_K]
        # Use inverse-rank weighting with some spread
        weights = np.zeros(len(corr))
        for rank, ki in enumerate(top_K):
            weights[ki] = 1.0 / (1.0 + rank * 0.3)
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            val = float(np.dot(weights, c))
            s = float(np.sum(x))
            val += mu * (s - target) ** 2
            return val
        
        def grad(x):
            return weighted_corr_grad(x, None, weights, mu, target)
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',

            best_f = f_new
            best_h = h_new.copy()
        h_cur = h_new

def coordinate_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):

            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            
            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))
            
            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            new_score = f1d(best_v)
            if new_score < cur_score - 1e-14:
                h_cur = h_cur.copy()
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = new_score
                improved = True
        if not improved:

        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

def lse_val(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

def lbfgs_lse(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: lse_val(x, alpha, target, mu),
                      h, jac=lambda x: lse_grad(x, alpha, target, mu),
                      method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def top_k_reweight(h, target, K, mu, maxiter):
    """Optimize with weights on top-K lags."""
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_K = np.argsort(corr)[::-1][:K]
    
    # Weights: decay by rank
    weights = np.zeros(len(corr))
    for rank, ki in enumerate(top_K):
        weights[ki] = 1.0 / (1.0 + rank * 0.4)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        val = float(np.dot(weights, c))
        s = float(np.sum(x))
        return val + mu * (s - target) ** 2
    
    def grad(x):
        n_ = len(x)
        g = np.zeros(n_)
        for ki in top_K:
            w = weights[ki]
            lag = int(ki) - (n_ - 1)
            if lag >= 0:
                j1 = np.arange(lag, n_)
                g[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n_ - lag)
                    g[j2] -= w * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n_ - m)
                g[j1] += w * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n_)
                    g[j2] -= w * x[j2 - m]
        s = float(np.sum(x))
        g += 2.0 * mu * (s - target)
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()

def nesterov_max(h_init, target, n_iter, lr_init, K, deadline):
    """Nesterov-accelerated subgradient on top-K lags."""
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = 0.9
    best_x = x.copy()
    best_f = compute_c5(x)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        y = project_feasible(x + beta * v, target)
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_K = np.argsort(corr)[::-1][:K]
        
        g = np.zeros(n)
        for ki in top_K:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                g[j1] += (1.0 - y[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    g[j2] -= y[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                g[j1] += (1.0 - y[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    g[j2] -= y[j2 - m]
        g /= K
        g += 50.0 * (np.sum(y) - target)
        
        v = beta * v - lr * g
        x_new = project_feasible(x + v, target)
        f_new = compute_c5(x_new)
        
        if f_new < best_f:
            best_f = f_new
            best_x = x_new.copy()
        
        # Adaptive lr
        if f_new <= best_f + 1e-10:
            lr = min(lr * 1.01, 0.08)
        else:
            lr *= 0.92
        
        x = x_new
    
    return best_x, best_f

def coord_descent_fast(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            
            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))
            
            fc = f1d(c)
            fd = f1d(d)
            for _ in range(20):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            new_score = f1d(best_v)
            if new_score < cur_score - 1e-14:
                h_cur = h_cur.copy()
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                cur_score = new_score
                improved = True
        if not improved:
            break
    return h_cur, cur_score
