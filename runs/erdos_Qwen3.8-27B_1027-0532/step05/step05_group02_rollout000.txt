                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            stall += 1
    return best_h, best_c5

def double_swap_search(h, n_iters, time_limit, start_time, rng, top_k=12):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 100
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        top_shifts = np.argsort(full)[-min(5, 2*n-1):]
        top_shifts = top_shifts[top_shifts > n-1]
        if len(top_shifts) == 0:
            stall += 1
            continue
        
        best_delta_max = max_val
        best_pair = None
        
        for s in top_shifts:
            k_s = s - (n - 1)
            if k_s == 0:
                continue
            A = np.zeros(n)
            idx = positions + k_s
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_s
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            B = np.zeros(n)
            idx = positions - k_s
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_s
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            cands_i = ones[np.argsort(A[ones])[:min(top_k, len(ones))]]
            cands_j = zeros[np.argsort(-B[zeros])[:min(top_k, len(zeros))]]
            
            for i1 in cands_i:
                for j1 in cands_j:
                    remaining_ones = [x for x in cands_i if x != i1]
                    remaining_zeros = [x for x in cands_j if x != j1]
                    for i2 in remaining_ones[:min(5, len(remaining_ones))]:
                        for j2 in remaining_zeros[:min(5, len(remaining_zeros))]:
                            if i2 == i1 or j2 == j1:
                                continue
                            d1 = compute_delta(h, i1, j1)
                            d2 = compute_delta(h, i2, j2)
                            combined = d1 + d2
                            new_max = float((full + combined).max())
                            if new_max < best_delta_max:
                                best_delta_max = new_max
                                best_pair = (i1, j1, i2, j2, combined)
            
            if best_pair is not None and best_delta_max < max_val:
                break
        
        if best_pair is not None and best_delta_max < max_val:
            i1, j1, i2, j2, _ = best_pair
            h[i1] = 0.0
            h[j1] = 1.0
            h[i2] = 0.0
            h[j2] = 1.0
            full = full_corr_fast(h)
            stall = 0
            new_c5 = float(full.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stall += 1
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    def obj(h, alpha):
        corr = full_corr_fast(h)
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sa_binary(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick greedy
    allowed = min(60, tleft() * 0.08)
    h_g, c5_g = fast_greedy(best_h, max_iters=50000, time_limit=allowed, start_time=start_time, top_k=20)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()
    
    # Phase 2: Continuous gradient optimization
    alphas = [10, 30, 100, 300, 1000, 3000, 10000]
    allowed = min(400, tleft() * 0.30)
    h_cont, c5_cont = projected_gradient_optimize(best_h, target_init, alphas, max_steps=2000, time_limit=allowed, start_time=start_time)
    if c5_cont < best_score:
        best_score = c5_cont
        best_h = h_cont.copy()
    
    # Phase 3: SLSQP refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(120, tleft() * 0.12)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, [100, 300, 1000, 3000], 3000, 1e-15, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
    
    # Phase 4: Binary projection + double swap
    h_bin = project_binary(best_h, target_init)
    bin_score = compute_c5(h_bin)
    
    allowed = min(80, tleft() * 0.10)
    h_ds, c5_ds = double_swap_search(h_bin, n_iters=5000, time_limit=allowed, start_time=start_time, rng=rng, top_k=10)
    if c5_ds < best_score:
        best_score = c5_ds
        best_h = h_ds.copy()
    
    # Phase 5: Greedy polish on best
    allowed = min(40, tleft() * 0.05)
    h_bp = project_binary(best_h, target_init)
    h_g2, c5_g2 = fast_greedy(h_bp, max_iters=20000, time_limit=allowed, start_time=start_time, top_k=20)
    if c5_g2 < best_score:
        best_score = c5_g2
        best_h = h_g2.copy()
    
    # Phase 6: Multi-resolution
    n_tries = [400, 500, 550, 650, 700, 800]
    for n_try in n_tries:
        if tleft() < 40:
            break
        if n_try % 2 != 0 or n_try < 100:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.04)
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=8000, time_limit=allowed, start_time=start_time, top_k=15)
        allowed2 = min(30, tleft() * 0.05)
        h_c3, c5_c3 = projected_gradient_optimize(h_g3, target_try, [50, 200, 1000], max_steps=1000, time_limit=allowed2, start_time=start_time)
        if c5_c3 < c5_g3:
            h_try = h_c3
            c5_try = c5_c3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()
    
    # Phase 7: Fine-tuned continuous optimization on best
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_fine = [500, 2000, 5000, 15000, 50000]
        allowed = min(tleft() - 30, 150)
        h_fine, sc_fine = projected_gradient_optimize(best_h, target_use, alphas_fine, max_steps=3000, time_limit=allowed, start_time=start_time)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine.copy()
    
    # Phase 8: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(tleft() - 10, 60)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, [5000, 15000], 2000, 1e-16, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
    
    # Phase 9: Final binary SA burst
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa = int(min(500000, max(5000, (tleft() - 8) * 30000)))
        h_sa, c5_sa = sa_binary(h_bin_final, n_sa, 0.005, 0.00001, rng, time_limit=max(3.0, tleft() - 6), start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
    
    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, n_final
