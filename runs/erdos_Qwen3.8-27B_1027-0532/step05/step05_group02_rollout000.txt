```python
import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_descent(h, max_iters, rng, time_limit, start_time):

    h = h.copy().astype(float)
    full = full_corr_fast(h).copy()
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 150
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        top_idx = np.argpartition(full, -5)[-5:]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        found = False
        for k_idx in top_idx:
            k_idx = int(k_idx)
            k_star = k_idx - (n - 1)
            if k_star == 0:
                continue

            s_i = np.zeros(len(ones))
            for idx in range(len(ones)):
                i = ones[idx]
                v = 0.0
                if 0 <= i + k_star < n:
                    v -= (1.0 - h[i + k_star])
                if 0 <= i - k_star < n:
                    v -= h[i - k_star]
                s_i[idx] = v

            s_j = np.zeros(len(zeros))
            for idx in range(len(zeros)):
                j = zeros[idx]
                v = 0.0
                if 0 <= j + k_star < n:
                    v += (1.0 - h[j + k_star])
                if 0 <= j - k_star < n:
                    v -= h[j - k_star]
                s_j[idx] = v

            ncand = min(15, len(ones))
            i_cands = ones[np.argsort(s_i)[:ncand]]
            j_cands = zeros[np.argsort(s_j)[:ncand]]

            best_new_max = max_val
            best_i = -1
            best_j = -1
            best_delta = None

            if best_i >= 0:

def sa_search(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h).copy()
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = min(1.0, it / max(n_iters - 1, 1))
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break

        if it % 3 == 0:
            top_idx = np.argpartition(full, -1)[-1:]
            k_star = int(top_idx[0]) - (n - 1)
            if k_star != 0:
                s_i = np.zeros(len(ones))
                for idx in range(len(ones)):
                    i = ones[idx]
                    v = 0.0
                    if 0 <= i + k_star < n:
                        v -= (1.0 - h[i + k_star])
                    if 0 <= i - k_star < n:
                        v -= h[i - k_star]
                    s_i[idx] = v
                s_j = np.zeros(len(zeros))
                for idx in range(len(zeros)):
                    j = zeros[idx]
                    v = 0.0
                    if 0 <= j + k_star < n:
                        v += (1.0 - h[j + k_star])
                    if 0 <= j - k_star < n:
                        v -= h[j - k_star]
                    s_j[idx] = v
                i_w = np.exp(-s_i / max(t * 5, 0.01))
                i_w /= i_w.sum()
                j_w = np.exp(-s_j / max(t * 5, 0.01))
                j_w /= j_w.sum()
                i = ones[rng.choice(len(ones), p=i_w)]
                j = zeros[rng.choice(len(zeros), p=j_w)]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        nm = float((full + delta).max())
        c5_new = nm * dx
        dc5 = c5_new - current_c5

            current_c5 = c5_new

                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

            no_improve += 1

    # Phase 3: SA from best solution
    if tleft() > 50:
        sa_alloc = min(300.0, tleft() * 0.35)
        n_sa = int(min(20000000, sa_alloc * 50000))
        h_sa, sc_sa = sa_search(best_h, n_sa, 0.03, 0.00005, rng,
                                time.start() + sa_alloc - start_time, start_time)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa.copy()

    # Phase 4: Another round of iterated LS with what we have
    no_improve2 = 0
    while tleft() > 50 and no_improve2 < 20:
        k_pert = rng.choice([5, 8, 12, 15, 20, 25, 30, 40])
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        n_flip = min(int(k_pert), len(ones), len(zeros))
        if n_flip < 2:
            no_improve2 += 1
            continue
        i_flip = rng.choice(ones, size=n_flip, replace=False)
        j_flip = rng.choice(zeros, size=n_flip, replace=False)
        h_pert[i_flip] = 0.0
        h_pert[j_flip] = 1.0

        g_alloc = min(25.0, tleft() * 0.3)
        h_new, sc_new = greedy_descent(h_pert, 60000, rng, time.start() + g_alloc - start_time, start_time)

        if sc_new < best_score - 1e-10:
            best_h = h_new.copy()
            best_score = sc_new
            no_improve2 = 0
        else:
            no_improve2 += 1

    # Phase 5: Continuous optimization via SLSQP
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 500, 1500]
        c_alloc = min(120.0, tleft() * 0.4)
        h_cont, sc_cont = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-14,
                                        time.start() + c_alloc - start_time, start_time)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()
        h_bin = project_binary(h_cont, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin.copy()
            # Greedy polish the binary
            g_alloc2 = min(20.0, tleft() * 0.2)
            h_bp, sc_bp = greedy_descent(best_h, 30000, rng, time.start() + g_alloc2 - start_time, start_time)
            if sc_bp < best_score:
                best_score = sc_bp
                best_h = h_bp.copy()

    # Phase 6: Final SA burst
    if tleft() > 25:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, n_use // 2)
        sc_final = compute_c5(h_bin_final)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_bin_final.copy()
        sa_alloc2 = max(5.0, tleft() - 10)
        n_sa2 = int(min(5000000, sa_alloc2 * 40000))
        h_sa_f, sc_sa_f = sa_search(best_h, n_sa2, 0.01, 0.00002, rng,
                                     time.start() + sa_alloc2 - start_time, start_time)
        if sc_sa_f < best_score:
            best_score = sc_sa_f
            best_h = h_sa_f.copy()

    # Phase 7: Final greedy polish
    if tleft() > 10:
        h_bin_end = project_binary(best_h, len(best_h) // 2)
        g_alloc3 = max(5.0, tleft() - 5)
        h_final_g, sc_final_g = greedy_descent(h_bin_end, 20000, rng, time.start() + g_alloc3 - start_time, start_time)
        if sc_final_g < best_score:
            best_score = sc_final_g
            best_h = h_final_g.copy()

    # Final feasibility
    n_final = len(best_h)
    h_final = project_feasible(best_h, float(n_final) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
```
