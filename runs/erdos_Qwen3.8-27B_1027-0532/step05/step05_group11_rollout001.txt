import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_push(h, target_sum, alpha):
    h_st = np.clip((h - 0.5) * alpha + 0.5, 0.0, 1.0)
    return project_feasible(h_st, target_sum)

def smooth_max_grad(h, alpha, n_top=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    n_half = n - 1
    c_max = np.max(corr)
    log_w = alpha * (corr - c_max)
    log_w = np.clip(log_w, -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(log_w)))
    p = np.exp(log_w - log_sum)
    top_idx = np.argsort(p)[::-1][:n_top]
    p_top = p[top_idx]
    s = p_top.sum()
    if s < 1e-15:
        p_top = p_top / p_top.sum()
    else:
        p_top = p_top / s
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = top_idx[i] - n_half
        wi = p_top[i]
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += wi * (1.0 - h[k:])
                grad[k:] += wi * (-h[:nk])
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                grad[kk:] += wi * (1.0 - h[:nk])
                grad[:nk] += wi * (-h[kk:])
    return grad * dx, float(np.max(corr))

def gradient_descent_smooth(h_init, n, target, time_limit, alphas, n_top=80, base_lr=0.05, momentum=0.85):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        lr = base_lr
        for it in range(500):
            if time.time() - start > time_limit:
                break
            grad, c_val = smooth_max_grad(h, alpha, n_top)
            v = momentum * v + (1.0 - momentum) * grad
            h_try = h - lr * v
            h_try = project_feasible(h_try, target)
            c_try = compute_c5(h_try)
            if c_try < best_c5 - 1e-12:
                best_c5 = c_try
                best_h = h_try.copy()
                h = h_try
                lr = min(lr * 1.05, 0.5)
            else:
                lr = lr * 0.7
                if lr < 1e-6:
                    break
    return best_h, best_c5

def lp_refine(h_init, n, target, time_limit, n_top=200, damping=0.9, reg=1e-5, max_iter=3000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for k in top_lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter, n_top=80):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    def obj_and_grad(h_flat, alpha):
        h = h_flat.reshape(-1)
        grad, _ = smooth_max_grad(h, alpha, n_top)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        log_w = alpha * (corr - c_max)
        log_w = np.clip(log_w, -500.0, 0.0)
        val = c_max + np.log(np.sum(np.exp(log_w))) / alpha
        return val, np.concatenate([grad, [1.0]])
    def obj_only(h_flat, alpha):
        return obj_and_grad(h_flat, alpha)[0]
    def con_fun(h_flat):
        return np.sum(h_flat) - target
    def con_jac(h_flat):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            def full_obj(h_f):
                return obj_and_grad(h_f, alpha)[0]
            def full_grad(h_f):
                return obj_and_grad(h_f, alpha)[1]
            res = minimize(full_obj, h_cur, jac=full_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def local_pair_swap(h, target, n_iterations=10000, time_limit=20, step_size=0.05):
    start = time.time()
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(777)
    for _ in range(n_iterations):
        if time.time() - start > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        old_i, old_j = h[i], h[j]
        h[i] = np.clip(h[i] + step_size, 0, 1)
        h[j] = np.clip(h[j] - step_size, 0, 1)
        h = project_feasible(h, target)
        c5_try = compute_c5(h)
        if c5_try < best_c5 - 1e-10:
            best_c5 = c5_try
        else:
            h[i], h[j] = old_i, old_j
            h = project_feasible(h, target)
    return h, best_c5
