def slsqp_polish(h, target, alpha, maxiter):

    h_cur = h.copy()

        return lse_obj_and_grad(x, alpha, target, 1e6)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e6)[1]

        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def bit_flip_search(h, target, n_trials=2000):
    """Try flipping individual bits and pairs of bits"""
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for trial in range(n_trials):
        # Try flipping a random bit
        j = trial % n
        old_val = h_cur[j]
        # Try setting to 0 or 1
        for new_val in [0.0, 1.0]:
            if abs(new_val - old_val) < 1e-6:
                continue
            h_try = h_cur.copy()
            h_try[j] = new_val
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-12:
                h_cur = h_try
                best_score = sc
                break
    
    return h_cur, best_score

def pair_flip_search(h, target, n_trials=1000, rng=None):

    best_score = compute_c5(h_cur)

        j, k = rng.choice(n, 2, replace=False)
        old_j, old_k = h_cur[j], h_cur[k]
        # Try a few new values
        for new_j in [0.0, 1.0]:
            for new_k in [0.0, 1.0]:

                    h_cur = h_try
                    best_score = sc
                    break

                break

                break
    return h_cur, best_score

def subsample_flip_search(h, target, n_trials, rng):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for trial in range(n_trials):
        n_flip = int(rng.randint(1, 4))
        idx = rng.choice(n, size=n_flip, replace=False)
        old_vals = h_cur[idx].copy()
        for mask in range(2 ** n_flip):
            if mask == 0:
                continue
            h_try = h_cur.copy()
            for ii in range(n_flip):
                if mask & (1 << ii):
                    h_try[idx[ii]] = 1.0 - old_vals[ii]
                else:
                    h_try[idx[ii]] = old_vals[ii]
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-14:
                h_cur = h_try
                best_score = sc
                break
    return h_cur, best_score

def water_fill_step(h, target, k, lr):
    n = len(h)
    corr = get_corr(h)
    c_max = float(np.max(corr))
    threshold = c_max - 0.1 * c_max
    n_above = int(np.sum(corr >= threshold))
    k = max(k, n_above, 3)
    top_idx = np.argsort(corr)[::-1][:k]
    top_vals = corr[top_idx]
    top_mean = top_vals.mean()
    top_spread = top_vals.max() - top_vals.min() + 1e-10
    grad = np.zeros(n)
    for i, ki in enumerate(top_idx):
        w = (top_vals[i] - top_mean) / top_spread + 0.1
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new

def water_fill_optimize(h, target, n_iter, k, lr_init):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_iter):
        h_new = water_fill_step(h_cur, target, k, lr)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return h_cur, best_score

def coordinate_descent(h, target, n_sweeps, n_points):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n_points):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            h_save = h_cur.copy()
            def f1d(v):
                h_new = h_save.copy()
                h_new[j] = v
                diff = target - np.sum(h_new)
                h_new[j] = float(np.clip(v + diff, 0.0, 1.0))
                corr = get_corr(h_new)
                return float(np.max(corr))
            fc = f1d(c)
            fd = f1d(d)
            for _ in range(22):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            h_cur[j] = best_v
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < cur_score - 1e-15:
                cur_score = sc
                improved = True
        if not improved:
            break
    return h_cur, cur_score

def annealed_lbfgs(h, target, alphas, mu, maxiter_each, deadline):
    h_cur = h.copy()
    for a in alphas:
        if time.time() > deadline:
            break
        h_new = lbfgs_optimize(h_cur, target, a, mu, maxiter_each)
        sc = compute_c5(h_new)
        h_cur = h_new
    return h_cur

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 50.0

    def tleft():
        return deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Fine-grained alpha schedule with L-BFGS-B
    alphas_p1 = [3, 10, 30, 100, 300, 1000, 3000, 10000]
    for i, a in enumerate(alphas_p1):
        if tleft() < 80:
            break
        iters = 400 + 80 * i
        mu_val = 20.0 if i < 3 else 50.0
        h_new = lbfgs_optimize(best_h, target, a, mu_val, iters)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Water-fill + L-BFGS cycle
    if tleft() > 120:
        h_wf, sc_wf = water_fill_optimize(best_h, target, n_iter=250, k=8, lr_init=0.004)
        if sc_wf < best_score:
            best_score = sc_wf
            best_h = h_wf
        h_new = lbfgs_optimize(best_h, target, 15000, 50.0, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Bit-flip local search
    if tleft() > 100:
        h_bf, sc_bf = bit_flip_search(best_h, target, n_trials=n * 3, rng=rng)
        if sc_bf < best_score:
            best_score = sc_bf
            best_h = h_bf
        h_new = lbfgs_optimize(best_h, target, 10000, 50.0, 400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Pair-flip search
    if tleft() > 80:
        h_pf, sc_pf = pair_flip_search(best_h, target, n_trials=n, rng=rng)
        if sc_pf < best_score:
            best_score = sc_pf
            best_h = h_pf
        h_new = lbfgs_optimize(best_h, target, 10000, 50.0, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Multi-flip search (2-3 bits at once)
    if tleft() > 80:
        h_mf, sc_mf = subsample_flip_search(best_h, target, n_trials=n // 2, rng=rng)
        if sc_mf < best_score:
            best_score = sc_mf
            best_h = h_mf
        h_new = lbfgs_optimize(best_h, target, 20000, 50.0, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Higher resolution
    if tleft() > 250:
        n2 = 2 * n
        target2 = n2 / 2.0
        x_old = np.linspace(0.0, 1.0, n)
        x_new = np.linspace(0.0, 1.0, n2)
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = project_feasible(h_fine, target2)
        alphas_fine = [20, 100, 500, 3000]
        for a in alphas_fine:
            if tleft() < 60:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, 50.0, 250)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 7: Diverse restarts with structured perturbations
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 150:
        restart_defs = [
            ('gauss_small', 0.03), ('gauss_med', 0.08), ('gauss_large', 0.15),
            ('flip_sparse', 0.05), ('flip_dense', 0.15),
            ('binary_slight', 0.1), ('binary_heavy', 0.3),
        ]
        for strat, param in restart_defs:
            if tleft() < 70:
                break
            h_pert = best_h.copy()
            if strat == 'gauss_small' or strat == 'gauss_med' or strat == 'gauss_large':
                h_pert += rng.randn(n_cur) * param
            elif strat == 'flip_sparse' or strat == 'flip_dense':
                n_flip = max(1, int(n_cur * param))
                idx = rng.choice(n_cur, size=n_flip, replace=False)
                h_pert[idx] = 1.0 - h_pert[idx]
            elif strat == 'binary_slight' or strat == 'binary_heavy':
                h_bin = (h_pert > 0.5).astype(float)
                n_flip = max(1, int(n_cur * param))
                idx = rng.choice(n_cur, size=n_flip, replace=False)
                h_bin[idx] = 1.0 - h_bin[idx]
                h_pert = h_bin + rng.randn(n_cur) * 0.03
            h_pert = project_feasible(h_pert, target_cur)
            h_new = lbfgs_optimize(h_pert, target_cur, 100, 30.0, 200)
            h_new = lbfgs_optimize(h_new, target_cur, 3000, 50.0, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            # Quick bit-flip on the new candidate
            h_bf2, sc_bf2 = bit_flip_search(h_new, target_cur, n_trials=n_cur // 2, rng=rng)
            if sc_bf2 < best_score:
                best_score = sc_bf2
                best_h = h_bf2

    # Phase 8: Second water-fill + high-alpha L-BFGS
    if tleft() > 80:
        h_wf2, sc_wf2 = water_fill_optimize(best_h, target_cur, n_iter=200, k=5, lr_init=0.003)
        if sc_wf2 < best_score:
            best_score = sc_wf2
            best_h = h_wf2
        h_new = lbfgs_optimize(best_h, target_cur, 50000, 50.0, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        # Another bit-flip pass
        h_bf3, sc_bf3 = bit_flip_search(best_h, target_cur, n_trials=n_cur, rng=rng)
        if sc_bf3 < best_score:
            best_score = sc_bf3
            best_h = h_bf3

    # Phase 9: Coordinate descent
    if tleft() > 80:
        n_sweeps = 2 if n_cur <= 400 else 1
        h_cd, sc_cd = coordinate_descent(best_h, target_cur, n_sweeps, n_cur)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 10: SLSQP polish
    if tleft() > 30:
        h_slq = slsqp_polish(best_h, target_cur, 10000, 3000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Final
    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
