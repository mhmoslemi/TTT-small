def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    h = h.copy()
    if diff > 0:

            room = 1.0 - h[idx]
            add = min(room, diff - added)
            h[idx] += add
            added += add
    else:

        scores = h  # higher score = more to remove
        sorted_idx = np.argsort(-scores)
        removed = 0.0
        for idx in sorted_idx:
            if removed >= -diff:
                break
            avail = h[idx]
            rem = min(avail, -diff - removed)
            h[idx] -= rem
            removed += rem
    return h

corr[k] = sum_j h[j] * (1 - h[j+k]) (with appropriate index ranges)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_box_simplex(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        projected = np.clip(h - tau, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)

def smooth_gradient(h, alpha):
    """Compute gradient of log-sum-exp smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    corr_max = np.max(corr)
    shifted = alpha * (corr - corr_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    w = np.exp(shifted)
    w = w / w.sum()
    
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    
    for k_idx in range(len(w)):
        if w[k_idx] < 1e-15:
            continue
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[k_idx] * h[idxs[v2] + k]
    
    grad *= dx
    grad -= np.mean(grad)  # Remove mean to maintain sum constraint (approximately)
    return grad, corr, corr_max

    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    
    total_iters = len(alpha_schedule)
    for step in range(total_iters):
        if time.time() - start > time_limit:
            break
        
        alpha = alpha_schedule[step]
        # Nesterov step
        y = h + mu * v
        y = np.clip(y, 0.0, 1.0)
        y = project_box_simplex(y, target)
        
        grad, corr, corr_max = smooth_gradient(y, alpha)
        
        # Adaptive step size
        lr = base_lr
        for _ in range(10):
            h_new = y - lr * grad
            h_new = project_box_simplex(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 or c5_new < compute_c5(y):
                break
            lr *= 0.5
        
        v = h_new - y
        h = h_new
        
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

def binary_sa(h_init, n, target, time_limit, rng_seed=42):
    """Simulated annealing on binary sequences."""
    rng = np.random.RandomState(rng_seed)
    h = (h_init > 0.5).astype(float)
    
    # Fix to exactly target ones
    n_ones = int(np.sum(h))
    if n_ones != target:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if n_ones > target:
            to_remove = rng.choice(ones_idx, size=n_ones - target, replace=False)
            h[to_remove] = 0
        else:
            to_add = rng.choice(zeros_idx, size=target - n_ones, replace=False)
            h[to_add] = 1
    
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    
    T_init = 0.003
    T_min = 0.0003
    cooling = 0.9997
    
    T = T_init
    start = time.time()
    
    while time.time() - start < time_limit:
        # Pick a random swap
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        
        # Do the swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Metropolis
            delta = new_c5 - cur_c5
            if np.log(rng.rand()) < -delta / T:
                cur_c5 = new_c5
            else:
                # Revert
                h[i], h[j] = h[j], h[i]
        
        T = max(T_min, T * cooling)
    
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    for alpha in alphas:

            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_box_simplex(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_box_simplex(h, target_sum):
    h = h.astype(float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        tau = 0.5 * (lo + hi)
        projected = np.clip(h - tau, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    return np.clip(h - tau, 0.0, 1.0)


def smooth_obj(h, alpha):
    corr = compute_corr(h)
    corr_max = np.max(corr)
    shifted = np.clip(alpha * (corr - corr_max), -500.0, 0.0)
    return float(corr_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    corr_max = np.max(corr)
    shifted = np.clip(alpha * (corr - corr_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    val = corr_max + np.log(w_sum) / alpha
    w = w / w_sum
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k_idx in active:
        k = k_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[k_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= w[k_idx] * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return val, grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_box_simplex(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def nesterov_grad(h_init, n, target, time_limit, base_lr=0.008, mu=0.82):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    alphas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    steps_per_alpha = 2000
    no_improve = 0
    total_steps = len(alphas) * steps_per_alpha
    for step in range(total_steps):
        if time.time() - start > time_limit:
            break
        alpha = alphas[min(step // steps_per_alpha, len(alphas) - 1)]
        y = h + mu * v
        y = project_box_simplex(y, target)
        val, grad = smooth_obj_grad(y, alpha)
        lr = base_lr
        improved = False
        for _ in range(8):
            h_new = y - lr * grad
            h_new = project_box_simplex(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h) - 1e-15:
                h = h_new
                improved = True
                break
            lr *= 0.5
        if not improved:
            no_improve += 1
            if no_improve > 3000:
                break
        else:
            no_improve = 0
        v = h - y
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5


def binary_sa(h_init, n, target, time_limit, rng_seed=42):
    rng = np.random.RandomState(rng_seed)
    h = (h_init > 0.5).astype(float)
    n_ones = int(np.sum(h))
    if n_ones != target:
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if n_ones > target:
            to_remove = rng.choice(ones_idx, size=n_ones - target, replace=False)
            h[to_remove] = 0
        else:
            to_add = rng.choice(zeros_idx, size=target - n_ones, replace=False)
            h[to_add] = 1
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.004
    T_min = 0.0002
    cooling = 0.99975
    start = time.time()
    no_improve_count = 0
    patience = 50000
    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            delta = new_c5 - cur_c5
            if np.log(rng.rand() + 1e-15) < -delta / T:
                cur_c5 = new_c5
                no_improve_count += 1
            else:
                h[i], h[j] = h[j], h[i]
                no_improve_count += 1
        T = max(T_min, T * cooling)
        if no_improve_count > patience and T <= T_min * 1.1:
            break
    return best_h, best_c5


def pair_cd_fast(h_init, n, target, time_limit, n_cand=30, max_sweeps=2000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 500:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_box_simplex(h, target)
    return best_h, best_c5


def multi_top_subgradient(h_init, n, target, time_limit, n_top=4, base_lr=0.015, mu=0.8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    step = 0
    max_steps = 50000
    while step < max_steps and time.time() - start < time_limit:
        corr = compute_corr(h)
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = h - lr * v
            h_try = project_box_simplex(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 2000:
            break
        step += 1
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_box_simplex(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_box_simplex(h_out, target_dst)


def perturbation_search(h_init, n, target, max_tries, time_limit, rng_seed=777):
    rng = np.random.RandomState(rng_seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    start = time.time()
    for step in range(max_tries):
        if time.time() - step > time_limit or time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8, 13, 21]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.03, 0.08, 0.15, 0.25]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_box_simplex(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5
