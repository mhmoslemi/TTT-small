import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = int(current_sum - target_sum)
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = int(target_sum - current_sum)
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    sl = h_r.sum()
    while sl < target_sum:
        z = np.where(h_r == 0)[0]
        if len(z) == 0:
            break
        h_r[z[0]] = 1
        sl += 1
    while sl > target_sum:
        o = np.where(h_r == 1)[0]
        if len(o) == 0:
            break
        h_r[o[0]] = 0
        sl -= 1
    return h_r


def vectorized_local_search(h, max_iters, time_limit, start_time, top_k=3, sample_size=80):
    n = len(h)
    h = h.copy()
    corr = get_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 1000

    rng = np.random.RandomState(12345)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        top_indices = np.argsort(corr)[-top_k:]
        current_max = float(corr.max())
        k_star = int(top_indices[-1]) - (n - 1)

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i_delta = np.zeros(len(ones))
        src_i = (ones + k_star) < n
        tgt_i = (ones - k_star) >= 0
        i_delta[src_i] -= (1.0 - h[ones[src_i] + k_star])
        i_delta[tgt_i] -= h[ones[tgt_i] - k_star]

        j_delta = np.zeros(len(zeros))
        src_j = (zeros + k_star) < n
        tgt_j = (zeros - k_star) >= 0
        j_delta[src_j] += (1.0 - h[zeros[src_j] + k_star])
        j_delta[tgt_j] += h[zeros[tgt_j] - k_star]

        if rng.random() < 0.3:
            alt_k = int(top_indices[rng.randint(len(top_indices))]) - (n - 1)
            if alt_k != k_star:
                i_delta_alt = np.zeros(len(ones))
                src_i2 = (ones + alt_k) < n
                tgt_i2 = (ones - alt_k) >= 0
                i_delta_alt[src_i2] -= (1.0 - h[ones[src_i2] + alt_k])
                i_delta_alt[tgt_i2] -= h[ones[tgt_i2] - alt_k]
                j_delta_alt = np.zeros(len(zeros))
                src_j2 = (zeros + alt_k) < n
                tgt_j2 = (zeros - alt_k) >= 0
                j_delta_alt[src_j2] += (1.0 - h[zeros[src_j2] + alt_k])
                j_delta_alt[tgt_j2] += h[zeros[tgt_j2] - alt_k]
                i_delta = 0.5 * (i_delta + i_delta_alt)
                j_delta = 0.5 * (j_delta + j_delta_alt)

        i_order = np.argsort(i_delta)
        j_order = np.argsort(j_delta)

        best_total = np.inf
        best_a = -1
        best_b = -1

        ns = min(sample_size, len(i_order))
        nsj = min(sample_size, len(j_order))

        for a_idx in range(ns):
            a = i_order[a_idx]
            ia = ones[a]
            for b_idx in range(nsj):
                b = j_order[b_idx]
                if ia == zeros[b]:
                    continue
                total = i_delta[a] + j_delta[b]
                if total < best_total:
                    best_total = total
                    best_a = a
                    best_b = b
                if i_delta[a] > 0.5:
                    break

        if best_a >= 0 and best_total < -0.5:
            i = ones[best_a]
            j = zeros[best_b]
            h[i] = 0
            h[j] = 1
            corr = get_corr(h)
            new_max = float(corr.max())
            if new_max * dx < best_c5:
                best_c5 = new_max * dx
                best_h = h.copy()
            elif new_max > current_max + 1.0:
                h[i] = 1
                h[j] = 0
                corr = get_corr(h)
            stall = 0
        else:
            stall += 1

    return best_h, best_c5


def sa_refine(h, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time, move_probs=None):
    n = len(h)
    h = h.copy()
    corr = get_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 5000

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        r = rng.random()
        h_new = h.copy()

        if r < 0.45:
            worst_idx = np.argsort(corr)[-8:]
            k = worst_idx[rng.randint(len(worst_idx))] - (n - 1)
            ip = ones[(ones + k) < n]
            jp = zeros[(zeros + k) >= 0 & ((zeros + k) < n)]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0
                h_new[jp[rng.randint(len(jp))]] = 1

        elif r < 0.70:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0
            h_new[j] = 1

        elif r < 0.85:
            i1, i2 = rng.choice(len(ones), 2, replace=False)
            j1, j2 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[i1]] = 0
            h_new[ones[i2]] = 0
            h_new[zeros[j1]] = 1
            h_new[zeros[j2]] = 1

        else:
            block_len = rng.randint(2, 5)
            if len(ones) >= block_len:
                idx = rng.choice(len(ones), block_len, replace=False)
                new_positions = rng.randint(0, n, block_len)
                for a in range(block_len):
                    old_i = ones[idx[a]]
                    new_j = new_positions[a]
                    if h_new[old_i] == 1 and h_new[new_j] == 0:
                        h_new[old_i] = 0
                        h_new[new_j] = 1

        if np.all(h_new == h):
            continue

        corr_new = get_corr(h_new)
        c5_new = float(corr_new.max()) * dx
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-14 and rng.random() < np.exp(-delta / max(t, 1e-20))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = float(target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score - 1e-10:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_init(n, rng):
    target = n // 2
    h = np.zeros(n)
    stride = n // target
    for i in range(target):
        pos = min(i * stride, n - 1)
        h[pos] = 1.0
    h = project_binary(h, target)
    return h


def random_binary_init(n, rng):
    target = n // 2
    h = np.zeros(n)
    ones_pos = rng.choice(n, target, replace=False)
    h[ones_pos] = 1.0
    return h
