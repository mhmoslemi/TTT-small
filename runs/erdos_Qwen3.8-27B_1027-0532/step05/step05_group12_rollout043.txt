import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(800):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = int(np.sum(mask))
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_fh(h, alpha, top_m, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_m:]
    else:
        top_idx = np.arange(2 * n - 1)
    top_vals = corr[top_idx]
    c_max = np.max(top_vals)
    shifted = np.clip(alpha * (top_vals - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = float(np.sum(exp_s))
    obj_val = (c_max + np.log(wsum) / alpha) * dx
    if wsum < 1e-300:
        return obj_val, np.zeros(n)
    weights = exp_s / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_idx, weights):
        if w < 1e-15:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return obj_val, grad

    full[free_indices] = free_vals
    obj, _ = smooth_max_fh(full, alpha, top_m, dx)
    return obj


def _fv_grad(free_vals, full_template, free_indices, alpha, top_m, dx):
    full = full_template.copy()
    full[free_indices] = free_vals
    _, grad_full = smooth_max_fh(full, alpha, top_m, dx)
    return grad_full[free_indices]

    return np.sum(x) - target


def _sum_eq_jac(x, target):
    return np.ones(len(x))

    obj, _ = smooth_max_fh(h, alpha, top_m, dx)
    return obj


def _all_grad(h, alpha, top_m, dx):
    _, grad = smooth_max_fh(h, alpha, top_m, dx)
    return grad


def _pen_obj(h, alpha, top_m, dx, target, pen_coeff):
    obj, _ = smooth_max_fh(h, alpha, top_m, dx)
    diff = np.sum(h) - target
    return obj + pen_coeff * diff * diff


def _pen_grad(h, alpha, top_m, dx, target, pen_coeff):
    _, grad = smooth_max_fh(h, alpha, top_m, dx)
    diff = np.sum(h) - target
    grad = grad + 2.0 * pen_coeff * diff
    return grad


def free_var_slsqp(h, free_eps, alpha, top_m, n, target, maxiter, deadline):
    dx = 2.0 / n
    free_mask = (h > free_eps) & (h < 1.0 - free_eps)
    n_free = int(np.sum(free_mask))
    if n_free == 0:
        return h
    free_indices = np.where(free_mask)[0]
    sum_fixed = float(np.sum(h[~free_mask]))
    target_free = target - sum_fixed
    if target_free < -1e-12 or target_free > n_free + 1e-12:
        return h
    h_free_init = h[free_mask].copy()
    bounds = [(0.0, 1.0)] * n_free
    cons = [{'type': 'eq', 'fun': _sum_eq_fun, 'jac': _sum_eq_jac, 'args': (target_free,)}]
    args_obj = (h, free_indices, alpha, top_m, dx)
    try:
        res = minimize(_fv_obj, h_free_init, jac=_fv_grad, args=args_obj,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'iprint': 0})
        if time.time() > deadline:
            return h
        if res.x is not None:
            h_new = h.copy()
            h_new[free_mask] = np.clip(np.asarray(res.x).ravel(), 0.0, 1.0)
            h_new = project_feasible(h_new, target)

def alternating_lp(h_init, n, target, max_iter, deadline, damping=0.5, reg=1e-5):
    import cvxpy as cp
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val

    h = h.copy()
    best_h = h.copy()

    order = np.arange(n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        np.random.shuffle(order)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            old_val = h[i]
            for new_val in [0.0, 0.25, 0.5, 0.75, 1.0, old_val + 0.1, old_val - 0.1]:
                if new_val < 0.0 or new_val > 1.0:

                                maxiter=2000, deadline=start_time + budget_s - 20)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try
        best_h = free_var_slsqp(best_h, 0.005, 50000, 50, n, target,
                                maxiter=1000, deadline=start_time + budget_s - 10)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    h_final = project_feasible(best_h, n / 2.0)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
