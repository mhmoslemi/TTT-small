def lse_grad(h, alpha):

        w[int(np.argmax(corr))] = 1.0

            # d/dh[j] = (1-h[j-k])*I(k<=j<=n-1) + (-h[j+k])*I(0<=j<=n-1-k)
            # First term: for j in [k, n-1]: (1-h[j-k])

            g[:n-k] -= wk * h[k:]

    g *= dx
    return g

            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP with high alpha
    alphas_p2 = [2000, 10000, 50000, 200000]
    for alpha in alphas_p2:
        if time.time() > deadline - 400:
            break
        h_new = lse_slsqp(best_h, n, target, alpha, 5000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Top-m active set optimization
    if time.time() < deadline - 350:
        for m_active in [3, 5, 8]:
            if time.time() > deadline - 340:
                break
            for p_val in [10, 50, 200]:
                if time.time() > deadline - 330:
                    break
                for pen in [0.001, 0.01]:
                    if time.time() > deadline - 325:
                        break
                    h_new = topm_lbfgsb(best_h, n, target, m_active, p_val, pen, 100.0, 200)
                    sc = compute_c5(h_new)
                    if sc < best_score:
                        best_score = sc
                        best_h = h_new
                    if time.time() < deadline - 320:
                        h_new2 = topm_slsqp(best_h, n, target, m_active, p_val, pen, 2000)
                        sc2 = compute_c5(h_new2)
                        if sc2 < best_score:
                            best_score = sc2
                            best_h = h_new2

    # Phase 4: Binary relaxation + simulated annealing
    if time.time() < deadline - 300:
        h_bin, c5_bin = binary_sa(best_h, n, target, 50000, 0.3, 0.001, deadline - 280, seed)
        best_bin_cont = project_feasible(h_bin, target)
        sc_bin = compute_c5(best_bin_cont)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = best_bin_cont
        else:
            h_blend = 0.5 * best_h + 0.5 * best_bin_cont
            h_blend = project_feasible(h_blend, target)
            h_ref = lse_lbfgsb(h_blend, n, target, 500, 50.0, 200)
            sc_blend = compute_c5(h_ref)
            if sc_blend < best_score:
                best_score = sc_blend
                best_h = h_ref

    # Phase 5: Momentum subgradient at current resolution
    if time.time() < deadline - 250:
        for temp in [0.005, 0.002, 0.001]:
            if time.time() > deadline - 240:
                break
            h_sg, sc_sg = momentum_subgrad(best_h, n, target, 2000, temp, 0.005, 0.9, deadline - 235)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 6: Higher resolution n=600
    if time.time() < deadline - 300:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [20, 100, 500, 2000, 10000, 50000]:
            if time.time() > deadline - 150:
                break
            for mu in [10.0, 100.0]:
                if time.time() > deadline - 145:
                    break
                h2_new = lse_lbfgsb(best_h2, n2, target2, alpha, mu, 200)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 130:
            for alpha in [5000, 30000]:
                if time.time() > deadline - 125:
                    break
                h2_new = lse_slsqp(best_h2, n2, target2, alpha, 3000, 1e-15)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 120:
            for m_act in [3, 5]:
                if time.time() > deadline - 115:
                    break
                h2_new = topm_lbfgsb(best_h2, n2, target2, m_act, 50, 0.01, 100.0, 150)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 100:
            h_sg2, sc_sg2 = momentum_subgrad(best_h2, n2, target2, 1500, 0.003, 0.004, 0.9, deadline - 95)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2
        if time.time() < deadline - 90:
            h_surg2, sc_surg2 = worst_lag_surgery(best_h2, n2, target2, 300, 0.005, deadline - 85)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 7: Try n=800 if time permits
    if time.time() < deadline - 250:
        n3 = 800
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)
        for alpha in [20, 100, 500, 2000, 10000]:
            if time.time() > deadline - 130:
                break
            for mu in [10.0, 100.0]:
                if time.time() > deadline - 125:
                    break
                h3_new = lse_lbfgsb(best_h3, n3, target3, alpha, mu, 150)
                sc3 = compute_c5(h3_new)
                if sc3 < best_score3:
                    best_score3 = sc3
                    best_h3 = h3_new
        if time.time() < deadline - 115:
            h3_new = lse_slsqp(best_h3, n3, target3, 10000, 2000, 1e-15)
            sc3 = compute_c5(h3_new)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3_new
        if time.time() < deadline - 105:
            h_sg3, sc_sg3 = momentum_subgrad(best_h3, n3, target3, 1000, 0.003, 0.003, 0.9, deadline - 100)
            if sc_sg3 < best_score3:
                best_score3 = sc_sg3
                best_h3 = h_sg3
        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 8: Random perturbations + re-optimization
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(6):
            if time.time() > deadline - 40:
                break
            scales = [0.05, 0.08, 0.12, 0.03, 0.15, 0.06]
            h_pert = best_h + rng.randn(n_cur) * scales[attempt]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref = lse_lbfgsb(h_pert, n_cur, target_cur, 500, 50.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_ref2 = lse_slsqp(best_h, n_cur, target_cur, 10000, 2000, 1e-15)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 9: Final aggressive polish
    if time.time() < deadline - 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgradient_polish(best_h, n_cur, target_cur, 2000, 0.001, 0.003, 0.995, deadline - 20)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        if time.time() < deadline - 15:
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 500, 0.003, deadline - 10)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 10: Final SLSQP
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for alpha_final in [50000, 200000]:
            if time.time() > deadline - 3:
                break
            h_new = lse_slsqp(best_h, n_cur, target_cur, alpha_final, 2000, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
