        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def precompute_effects(h):
    """Precompute flip effect vectors for all positions."""
    n = len(h)
    nr = 2 * n - 1
    # effect_1[p, idx]: delta to corr[idx] when flipping h[p] from 1 to 0
    # corr[idx] corresponds to lag k = idx - (n-1)
    effect_1 = np.zeros((n, nr))
    effect_0 = np.zeros((n, nr))
    
    for p in range(n):
        for idx in range(nr):
            k = idx - (n - 1)
            d = 0.0
            # h[p] changes from 1 to 0
            # Term i=p: h[p+k]*(1-h[p]) -> h[p+k]*(1-0). Delta: +h[p+k]
            if 0 <= p + k < n:
                d += h[p + k]
            # Term i=p-k: h[p]*(1-h[p-k]) -> 0. Delta: -(1-h[p-k])
            if 0 <= p - k < n:
                d -= (1.0 - h[p - k])
            effect_1[p, idx] = d
            effect_0[p, idx] = -d  # flipping from 0 to 1 is the negative
    
    return effect_1, effect_0

def greedy_v2(h, max_iters, time_limit, start_time):
    """Fast greedy using precomputed effects."""
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    dx = 2.0 / n
    best_score = float(corr.max()) * dx
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if no_improve > 150:
            break
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Precompute effects (O(n²) each time - expensive)
        # Alternative: update effects incrementally
        # For now, let's use the direct approach with top-k lags
        
        max_val = float(corr.max())
        # Get top 50 lags
        top_k = min(50, 2*n-1)
        top_indices = np.argsort(corr)[-top_k:]
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_new_max = max_val
        
        # For each one, compute its "profile" on the top lags
        # effect at top lag for flipping h[p] 1->0:
        # d = h[p+k] - (1-h[p-k]) (with boundary)
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Compute profiles for ones and zeros on top lags
        lags = top_indices - (n - 1)
        
        # For one at position p, effect on lag k: h[p+k] - (1-h[p-k])
        # For zero at position q, effect on lag k: -h[q+k] + (1-h[q-k])
        
        # Vectorize over positions
        # ones_profiles[i, j] = effect of flipping ones[i] on lags[j]
        ones_profiles = np.zeros((n_ones, top_k))
        zeros_profiles = np.zeros((n_zeros, top_k))
        
        for ii in range(n_ones):
            p = ones[ii]
            for jj in range(top_k):
                k = lags[jj]
                d = 0.0
                if 0 <= p + k < n:
                    d += h[p + k]
                if 0 <= p - k < n:
                    d -= (1.0 - h[p - k])
                ones_profiles[ii, jj] = d
        
        for jj_idx in range(n_zeros):
            q = zeros[jj_idx]
            for jj in range(top_k):
                k = lags[jj]
                d = 0.0
                if 0 <= q + k < n:
                    d -= h[q + k]
                if 0 <= q - k < n:
                    d += (1.0 - h[q - k])
                zeros_profiles[jj_idx, jj] = d
        
        # Now for each pair (i, j): new_max ≈ max over top_k lags of (corr[top] + ones_profiles[i] + zeros_profiles[j] + interaction)
        # Ignore interaction for now (it's at most +1 at 2 specific lags)
        
        # Vectorize: for each i, compute max over j
        for ii in range(n_ones):
            # new_corr for this i, all j: shape (n_zeros, top_k)
            combined = zeros_profiles + ones_profiles[ii][:, None]
            new_maxes = np.max(corr[top_indices][None, :] + combined, axis=1)
            best_local = np.argmin(new_maxes)
            if new_maxes[best_local] < best_new_max:
                best_new_max = new_maxes[best_local]
                best_i = ones[ii]
                best_j = zeros[best_local]
        
        if best_i >= 0 and best_new_max < max_val - 0.5:
            h[best_i] = 0
            h[best_j] = 1
            corr = compute_corr(h)
            sc = float(corr.max()) * dx
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_score
