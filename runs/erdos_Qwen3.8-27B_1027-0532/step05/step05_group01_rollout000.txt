import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    # Second pass for residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def multi_shift_gradient(h, n, dx, alpha, top_k):
    """Compute weighted gradient from top-k shifts."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    top_indices = np.argsort(corr)[-top_k:]
    w = np.exp(alpha * (corr[top_indices] - c_max))
    w = w / w.sum()
    
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    
    for wi, s_idx in zip(w, top_indices):

        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    
    grad *= dx

def nesterov_multishift(h, n, target, dx, time_limit, max_steps, 
                         alpha_start, alpha_end, top_k, mu, lr0):
    """Nesterov-accelerated multi-shift subgradient method."""
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        # Adaptive alpha and learning rate
        progress = min(1.0, step / max(1, max_steps * 0.5))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr0 * (1.0 - 0.5 * progress)

        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def multi_shift_gradient(h, n, dx, alpha, top_k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_k >= len(corr):
        top_indices = np.arange(len(corr))
    else:
        top_indices = np.argsort(corr)[-top_k:]
    w = np.exp(alpha * (corr[top_indices] - c_max))
    w = w / w.sum()
    grad = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for wi, s_idx in zip(w, top_indices):
        if wi < 1e-12:
            continue
        k = int(s_idx) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += wi * (1.0 - h[idxs[v1] - k])
        grad[v2] -= wi * h[idxs[v2] + k]
    grad *= dx
    grad -= np.mean(grad)
    return grad


def nesterov_multishift(h, n, target, dx, time_limit, max_steps,
                        alpha_start, alpha_end, top_k, mu, lr0):
    start = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, step / max(1, int(max_steps * 0.6)))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        lr = lr0 * (1.0 - 0.7 * progress)
        h_look = project_feasible(h - mu * v, target)
        grad = multi_shift_gradient(h_look, n, dx, alpha, top_k)
        v = mu * v + lr * grad
        h_new = project_feasible(h - v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None, t_start=None):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    if t_start is None:
        t_start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time_limit is not None and time.time() - t_start > time_limit - 5:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def pair_cd_improved(h, n, target, dx, time_limit, max_sweeps, n_cand, alpha_grad, top_k):
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consec_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consec_no_improve = 0
        else:
            consec_no_improve += 1
        if consec_no_improve > 150:
            break

        grad = multi_shift_gradient(h, n, dx, alpha_grad, top_k)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j] + 1.0)
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-13:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-13
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-13:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-14:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5


def simulated_annealing(h_init, n, target, time_limit, max_iters, T0, T1):
    start = time.time()
    rng_sa = np.random.RandomState(777)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        progress = it / max(1, max_iters)
        T = T0 * (T1 / T0) ** progress
        n_pert = int(rng_sa.choice([2, 3, 5, 8, 12]))
        n_pert = min(n_pert, n)
        idxs_p = rng_sa.choice(n, size=n_pert, replace=False)
        scale = float(rng_sa.choice([0.03, 0.06, 0.1, 0.15, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs_p] += rng_sa.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        delta_c5 = c5_try - c5_cur
        if delta_c5 < 0 or rng_sa.rand() < np.exp(-delta_c5 / max(T, 1e-8)):
            h_cur = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
        else:
            c5_cur = compute_c5(h_cur)
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
