```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    denom = exp_vals.sum()
    weights = exp_vals / denom
    val = c_max + np.log(denom) / alpha
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-13:
            continue
        k = k_idx - mid
        v1 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w * (1.0 - h[idxs[v1] + k])
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v2] -= w * h[idxs[v2] - k]
    return val, grad

def slsqp_fast(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None, start_time=None):
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        def obj_func(x):
            return smooth_obj_grad(x, alpha)
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj_func, h_cur, jac=True, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def pair_cd(h, n, target, time_limit, max_sweeps=3000, n_cand=40):
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * (2.0 / n)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 300:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_hi = min(h[i], 1.0 - h[j])
                delta_lo = max(h[i] - 1.0, -h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[idxs[v_ip] + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[idxs[v_im] - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[idxs[v_jp] + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[idxs[v_jm] - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * (2.0 / n)
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def momentum_sg(h_start, n, target, time_limit, max_steps=50000, mu=0.8, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        lr = base_lr / (1.0 + 0.001 * step)
        v = mu * v + grad
        h_new = project_feasible(h - lr * v, target)
        h = h_new
    return best_h, best_c5

def topk_subgradient(h_start, n, target, time_limit, max_steps=30000, n_top=8, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            break
        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 5000:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with analytic gradient at current resolution
    if tleft() > 180:
        h_pol, sc = slsqp_fast(best_h, n, target,
                               [200, 800, 3000, 15000, 80000, 300000],
                               2000, 1e-16,
                               time_limit=min(150, tleft() * 0.6),
                               start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD at current resolution
    if tleft() > 120:
        cd_time = min(120, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time, max_sweeps=5000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: More SLSQP with higher alphas
    if tleft() > 150:
        h_pol, sc = slsqp_fast(best_h, n, target,
                               [5000, 20000, 100000, 500000],
                               3000, 1e-16,
                               time_limit=min(120, tleft() * 0.5),
                               start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Pair CD again
    if tleft() > 80:
        cd_time = min(80, tleft() * 0.3)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time, max_sweeps=4000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Interpolate to n=800 and optimize
    n_hires = 800
    target_hires = n_hires / 2.0
    if tleft() > 250:
        h_interp = interpolate_to(best_h, n_hires, target_hires)
        h_pol, sc = slsqp_fast(h_interp, n_hires, target_hires,
                               [200, 1000, 5000, 20000],
                               1500, 1e-15,
                               time_limit=min(150, tleft() * 0.5),
                               start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
            n = n_hires
            target = target_hires

    # Phase 6: Pair CD at higher resolution
    if tleft() > 150:
        n_use = len(best_h)
        t_use = n_use / 2.0
        cd_time = min(120, tleft() * 0.4)
        h_cd, c5_cd = pair_cd(best_h, n_use, t_use,
                              time_limit=cd_time, max_sweeps=4000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: SLSQP at higher resolution with aggressive alphas
    if tleft() > 150:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_pol, sc = slsqp_fast(best_h, n_use, t_use,
                               [10000, 50000, 200000, 1000000],
                               2000, 1e-16,
                               time_limit=min(120, tleft() * 0.5),
                               start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Momentum subgradient
    if tleft() > 80:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_mom, c5_mom = momentum_sg(best_h, n_use, t_use,
                                    time_limit=min(60, tleft() * 0.3),
                                    max_steps=50000, mu=0.85, base_lr=0.01)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 9: Top-k subgradient
    if tleft() > 80:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_sg, c5_sg = topk_subgradient(best_h, n_use, t_use,
                                       time_limit=min(50, tleft() * 0.3),
                                       max_steps=30000, n_top=8, base_lr=0.015)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: Try n=1200 if time permits
    if tleft() > 300:
        n_try = 1200
        target_try = n_try / 2.0
        h_interp2 = interpolate_to(best_h, n_try, target_try)
        h_pol2, sc2 = slsqp_fast(h_interp2, n_try, target_try,
                                 [200, 1000, 5000, 20000],
                                 1500, 1e-15,
                                 time_limit=min(200, tleft() * 0.5),
                                 start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_pol2
            n = n_try
            target = target_try
        if tleft() > 100:
            n_use = len(best_h)
            t_use = n_use / 2.0
            cd_time = min(80, tleft() * 0.4)
            h_cd2, c5_cd2 = pair_cd(best_h, n_use, t_use,
                                    time_limit=cd_time, max_sweeps=3000, n_cand=50)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_final, sc_final = slsqp_fast(best_h, n_use, t_use,
                                       [50000, 200000, 1000000],
                                       2000, 1e-16,
                                       time_limit=min(30, tleft() * 0.7),
                                       start_time=start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 12: Final pair CD
    if tleft() > 30:
        n_use = len(best_h)
        t_use = n_use / 2.0
        h_cd3, c5_cd3 = pair_cd(best_h, n_use, t_use,
                                time_limit=min(20, tleft() * 0.7),
                                max_sweeps=2000, n_cand=40)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
