def sa_search(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def legendre_init(n, rng):
    p = 3
    while p < n:
        p += 2
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p >= n:
            break
    if p < n:
        p = n + 1 if (n + 1) % 2 == 1 else n + 2
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2
    
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    
    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        remaining = n - p
        h[p:] = 0.5
    
    target = n // 2
    h = project_binary(h, target)
    return h

def msequence_init(n, rng):
    # Generate an m-sequence based initialization
    # Use a primitive polynomial for the LFSR
    # For simplicity, use a pattern-based approach
    p = n // 2
    # Create a sequence based on quadratic residues of a large prime
    prime = 2 * p + 1
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(prime)) + 1, 2):
            if prime % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        prime += 2
    
    seq = np.zeros(prime)
    for a in range(1, prime):
        if pow(a, (prime - 1) // 2, prime) == 1:
            seq[a] = 1.0
    seq[0] = 0.5  # neutral
    
    # Take first n elements and project
    h = seq[:n].copy()
    if len(h) < n:
        h = np.pad(h, (0, n - len(h)), mode='wrap')
    h = project_binary(h, p)
    return h

def sa_fast(h, n, target, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    h = h.astype(int)
    corr = np.correlate(h, 1-h, mode='full').astype(np.float64)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > 10000:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        r = rng.random()
        if r < 0.5:
            # Targeted move: pick from worst correlations
            worst_idx = np.argmax(corr)
            k = worst_idx - (n-1)
            if k != 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        # Make swap
        h[i] = 0
        h[j] = 1
        
        # Compute new correlation
        new_corr = np.correlate(h, 1-h, mode='full').astype(np.float64)
        new_max = np.max(new_corr)
        new_c5 = new_max * dx
        delta = new_c5 - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            corr = new_corr
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            # Update ones/zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # Revert
            h[i] = 1
            h[j] = 0
            stall += 1

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    current_sum = h_r.sum()
