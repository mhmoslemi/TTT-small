def greedy_full(h, max_iters, time_limit=None, start_time=None, scan_width=20):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    stall = 0
    max_stall = 500
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i_cand = [i for i in ones if 0 <= i + k_star < n]
        j_cand = [j for j in zeros if 0 <= j + k_star < n]
        
        if len(i_cand) == 0 or len(j_cand) == 0:
            stall += 1
            if stall > max_stall:
                break
            continue
        
        top_threshold = max_val * 0.9
        top_indices = np.where(corr >= top_threshold)[0]
        if len(top_indices) == 0:
            top_indices = np.array([max_idx])
        
        best_improve = 0.0
        best_i = -1
        best_j = -1
        
        n_i = min(len(i_cand), scan_width)
        n_j = min(len(j_cand), scan_width)
        
        for ii in range(n_i):
            i = i_cand[ii]
            for jj in range(n_j):
                j = j_cand[jj]
                if i == j:
                    continue
                nm = max_val
                for t_idx in top_indices:
                    k = t_idx - (n - 1)
                    d = 0.0
                    if 0 <= i + k < n:
                        d -= (1.0 - h[i + k])
                    if 0 <= i - k < n:
                        d += h[i - k]
                    if 0 <= j + k < n:
                        d += (1.0 - h[j + k])
                    if 0 <= j - k < n:
                        d -= h[j - k]
                    v = corr[t_idx] + d
                    if v > nm:
                        nm = v
                        break
                if nm < max_val - best_improve:
                    best_improve = max_val - nm
                    best_i = i
                    best_j = j
        
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = compute_corr(h)
            sc = float(corr.max()) * (2.0 / n)
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    
    return best_h, best_c5

def sa_optimize(h, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):

    best_h = h.copy()

        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        if stall > max_stall:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        h_new = h.copy()
        moved = False
        
        if r < 0.5:
            # Targeted swap: pick from worst shifts
            worst = np.argsort(corr)[-10:]

            ip = [i for i in ones if 0 <= i + k < n and h[i + k] == 0]
            jp = [j for j in zeros if 0 <= j + k < n and h[j + k] == 1]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0

        elif r < 0.8:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        elif r < 0.95:
            # 2-pair swap
            if len(ones) >= 2 and len(zeros) >= 2:

                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0

                zeros_in_region = [z for z in region if h[z] == 0.0]
                ones_in_region = [o for o in region if h[o] == 1.0]
                if len(zeros_in_region) > 0 and len(ones_in_region) > 0:
                    # Swap one from each
                    h_new[ones_in_region[rng.randint(len(ones_in_region))]] = 0.0
                    h_new[zeros_in_region[rng.randint(len(zeros_in_region))]] = 1.0
                    moved = True
        
        if not moved:
            continue
        
        corr_new = compute_corr(h_new)
        c5_new = float(corr_new.max()) * (2.0 / n)

            else:
                stall += 1
        else:
            stall += 1

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def legendre_init(n, rng):
    # Find smallest prime >= n
    p = max(n, 3)
    if p % 2 == 0:
        p += 1
    while True:
        is_prime = True
        if p < 2:
            is_prime = False
        elif p < 4:
            is_prime = True
        else:
            if p % 2 == 0:
                is_prime = False
            else:
                for d in range(3, int(np.sqrt(p)) + 1, 2):
                    if p % d == 0:
                        is_prime = False
                        break
        if is_prime:
            break
        p += 2
    
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    
    seq = (leg >= 0).astype(float)
    if p >= n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
    
    target = n // 2
    h = project_binary(h, target)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def lse_obj_and_grad(h, beta, target_sum):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = beta * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / beta
    obj = (2.0 / n) * lse
    
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    n2 = 2 * n - 1
    for j in range(n):
        i1_lo = j
        i1_hi = min(n2 - 1, j + n - 1)
        if i1_lo <= i1_hi:
            idx1 = np.arange(i1_lo, i1_hi + 1)

    grad *= (2.0 / n)

    grad -= np.mean(grad)

        best_improve = 0.0

        best_j = -1

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
