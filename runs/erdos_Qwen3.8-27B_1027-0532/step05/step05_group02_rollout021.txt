```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h_r = np.clip(h_r, 0, 1).astype(int)

    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:

        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]

            extra = final_sum - target_sum

            extra = target_sum - final_sum
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def project_bounded_simplex(h, target_sum):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum} using water-filling."""
    n = len(h)
    h = h.astype(float).copy()
    # Find t such that sum(clip(h - t, 0, 1)) = target_sum
    lo = h.min() - target_sum
    hi = h.max()
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2.0
    result = np.clip(h - t, 0.0, 1.0)
    # Exact fix for floating point
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        free = np.where((result > 0) & (result < 1))[0]
        if len(free) > 0:
            result[free] += diff / len(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_grad(h, alpha):
    """Compute gradient of (1/alpha)*log(sum(exp(alpha*corr))) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp  # softmax weights, shape (2n-1,)
    
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        ell = m - (n - 1)
        if ell >= 0:
            # First term: (1-h[j-ell]) for j in [ell, n-1]
            j_lo, j_hi = ell, n - 1
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += pm * (1.0 - h[:j_hi+1-j_lo])
            # Second term: -h[j+ell] for j in [0, n-1-ell]
            j_lo2, j_hi2 = 0, n - 1 - ell
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] -= pm * h[ell:ell + j_hi2 - j_lo2 + 1]
        else:
            L = -ell
            # First term: (1-h[j+L]) for j in [0, n-1-L]
            j_lo, j_hi = 0, n - 1 - L
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += pm * (1.0 - h[L:L + j_hi - j_lo + 1])
            # Second term: -h[j-L] for j in [L, n-1]
            j_lo2, j_hi2 = L, n - 1
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] -= pm * h[0:j_hi2 - j_lo2 + 1]
    
    return grad

def projected_gradient_descent(h_init, target, alpha_start, alpha_end, 
                                n_alpha_steps, steps_per_alpha, 
                                lr_init, lr_decay, time_limit, start_time,
                                check_interval=50):
    """Run projected gradient descent on the log-sum-exp surrogate."""
    n = len(h_init)
    h = h_init.copy().astype(float)
    h = project_bounded_simplex(h, target)
    
    dx = 2.0 / n
    best_score = compute_c5(h)
    best_h = h.copy()
    
    total_steps = n_alpha_steps * steps_per_alpha
    current_step = 0
    
    for alpha_idx in range(n_alpha_steps):
        # Linear interpolation of alpha
        frac = alpha_idx / max(n_alpha_steps - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        for step in range(steps_per_alpha):
            current_step += 1
            if time.time() - start_time > time_limit:
                return best_h, best_score
            
            lr = lr_init * (lr_decay ** (current_step / steps_per_alpha))
            
            grad = lse_grad(h, alpha)
            h_new = h - lr * grad
            h_new = project_bounded_simplex(h_new, target)
            
            if current_step % check_interval == 0:
                score = compute_c5(h_new)
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
            
            h = h_new
    
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h.copy()
    
    return best_h, best_score
```

```python
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-15:
            continue
        ell = m - (n - 1)
        if ell >= 0:
            j_lo, j_hi = ell, n - 1
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += pm * (1.0 - h[:j_hi+1-j_lo])
            j_lo2, j_hi2 = 0, n - 1 - ell
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] -= pm * h[ell:ell + j_hi2 - j_lo2 + 1]
        else:
            L = -ell
            j_lo, j_hi = 0, n - 1 - L
            if j_lo <= j_hi:
                grad[j_lo:j_hi+1] += pm * (1.0 - h[L:L + j_hi - j_lo + 1])
            j_lo2, j_hi2 = L, n - 1
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2+1] -= pm * h[0:j_hi2 - j_lo2 + 1]
    return dx * val, dx * grad

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def greedy_binary(h, max_iters, rng, time_limit, start_time, top_lags=3):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 400
    positions = np.arange(n)
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        max_val = float(full.max())
        sorted_idx = np.argsort(full)[::-1][:top_lags]
        k_star = int(sorted_idx[0]) - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        
        A = np.zeros(n)
        B = np.zeros(n)
        for li in sorted_idx:
            k = li - (n - 1)
            if k == 0:
                continue
            w = full[li]
            idx = positions + k
            valid = (idx >= 0) & (idx < n)
            A[valid] += w * h[idx[valid]]
            idx = positions - k
            valid = (idx >= 0) & (idx < n)
            A[valid] += w * h[idx[valid]]
            idx = positions - k
            valid = (idx >= 0) & (idx < n)
            B[valid] += w * h[idx[valid]]
            idx = positions + k
            valid = (idx >= 0) & (idx < n)
            B[valid] += w * h[idx[valid]]
        
        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]
        
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    c5_new = float(full.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        
        if not found:
            for _ in range(30):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    c5_new = float(full.max()) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    
    return best_h, best_c5

def sa_binary(h, n_iters, t_init, t_final, rng, time_limit, start_time, focused=True):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if focused and it % 4 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_c = ones[A[ones] <= 1.0]
                j_c = zeros[B[zeros] >= 1.0]
                if len(i_c) > 0 and len(j_c) > 0:
                    i = i_c[rng.randint(len(i_c))]
                    j = j_c[rng.randint(len(j_c))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = float(n_init) / 2.0
    target_int = int(n_init / 2.0)
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    def tleft():
        return budget_s - 8.0 - (time.time() - start_time)
    
    # Phase 1: Continuous projected gradient with continuation (150s)
    if tleft() > 30:
        allowed = min(150, tleft() * 0.20)
        alpha_sched_1 = np.logspace(np.log10(10), np.log10(200), 20)
        h_cont, c5_cont = projected_gradient(best_h, target_init, alpha_sched_1,
                                             lr_base=0.5, time_limit=allowed,
                                             start_time=start_time, momentum=0.85)
        if c5_cont < best_score:
            best_score = c5_cont
            best_h = h_cont.copy()
    
    # Phase 2: SLSQP refinement on continuous (80s)
    if tleft() > 80:
        allowed = min(80, tleft() * 0.12)
        alphas = [100, 300, 1000, 3000, 10000]
        h_ref, sc = slsqp_smooth(best_h, n_init, target_init, alphas, 500,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
    
    # Phase 3: Binary local search from best continuous (120s)
    if tleft() > 50:
        h_bin = project_binary(best_h, target_int)
        bin_score = compute_c5(h_bin)
        
        allowed = min(60, tleft() * 0.10)
        h_g, c5_g = greedy_binary(h_bin, max_iters=30000, rng=rng,
                                  time_limit=allowed, start_time=start_time, top_lags=3)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()
        elif c5_g < bin_score:
            pass
        
        if tleft() > 40:
            allowed2 = min(60, tleft() * 0.10)
            n_sa = int(min(3000000, allowed2 * 50000))
            h_sa, c5_sa = sa_binary(h_g, n_sa, 0.02, 0.0001, rng,
                                    time_limit=allowed2, start_time=start_time, focused=True)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()
    
    # Phase 4: More aggressive continuous optimization (200s)
    if tleft() > 80:
        allowed = min(200, tleft() * 0.25)
        alpha_sched_2 = np.logspace(np.log10(200), np.log10(5000), 25)
        h_cont2, c5_cont2 = projected_gradient(best_h, target_init, alpha_sched_2,
                                               lr_base=0.3, time_limit=allowed,
                                               start_time=start_time, momentum=0.8)
        if c5_cont2 < best_score:
            best_score = c5_cont2
            best_h = h_cont2.copy()
    
    # Phase 5: Iterated greedy+SA with perturbation (200s)
    for it in range(8):
        if tleft() < 60:
            break
        
        # Greedy polish
        h_bin_cur = project_binary(best_h, target_int)
        if tleft() > 30:
            allowed = min(25, tleft() * 0.05)
            h_g2, c5_g2 = greedy_binary(h_bin_cur, max_iters=15000, rng=rng,
                                        time_limit=allowed, start_time=start_time, top_lags=2)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()
        
        if tleft() < 40:
            break
        
        # Perturb
        n_pert = rng.randint(2, 6)
        h_pert = h_bin_cur.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
        
        # SA from perturbed
        if tleft() > 30:
            allowed2 = min(50, tleft() * 0.08)
            n_sa2 = int(min(2000000, allowed2 * 40000))
            h_sa2, c5_sa2 = sa_binary(h_pert, n_sa2, 0.012, 0.00005, rng,
                                      time_limit=allowed2, start_time=start_time, focused=True)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()
    
    # Phase 6: Multi-resolution (150s)
    if tleft() > 100:
        for n_try in [800, 1000, 720]:
            if tleft() < 60:
                break
            if n_try % 2 != 0:
                continue
            target_try = float(n_try) / 2.0
            target_try_int = n_try // 2
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_bin_try = project_binary(h_interp, target_try_int)
            
            # Quick greedy
            if tleft() > 40:
                allowed = min(25, tleft() * 0.08)
                h_g3, c5_g3 = greedy_binary(h_bin_try, max_iters=10000, rng=rng,
                                            time_limit=allowed, start_time=start_time, top_lags=3)
            else:
                h_g3, c5_g3 = h_bin_try, compute_c5(h_bin_try)
            
            # SA
            if tleft() > 30:
                allowed2 = min(30, tleft() * 0.08)
                n_sa3 = int(min(1000000, allowed2 * 40000))
                h_s3, c5_s3 = sa_binary(h_g3, n_sa3, 0.01, 0.0001, rng,
                                        time_limit=allowed2, start_time=start_time, focused=True)
                if c5_s3 < c5_g3:
                    h_try, c5_try = h_s3, c5_s3
                else:
                    h_try, c5_try = h_g3, c5_g3
            else:
                h_try, c5_try = h_g3, c5_g3
            
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try.copy()
            
            # Continuous refinement at this resolution
            if tleft() > 40 and n_try <= 1000:
                allowed3 = min(30, tleft() * 0.08)
                alpha_sched = np.logspace(np.log10(50), np.log10(2000), 10)
                h_c4, c5_c4 = projected_gradient(h_try, target_try, alpha_sched,
                                                 lr_base=0.4, time_limit=allowed3,
                                                 start_time=start_time, momentum=0.8)
                if c5_c4 < best_score:
                    best_score = c5_c4
                    best_h = h_c4.copy()
    
    # Phase 7: Final continuous optimization at best resolution (80s)
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(80, tleft() * 0.15)
        alpha_sched_3 = np.logspace(np.log10(500), np.log10(20000), 15)
        h_final_cont, c5_final = projected_gradient(best_h, target_use, alpha_sched_3,
                                                    lr_base=0.15, time_limit=allowed,
                                                    start_time=start_time, momentum=0.7)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_final_cont.copy()
    
    # Phase 8: Final SLSQP (30s)
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(30, tleft() * 0.15)
        h_ref2, sc2 = slsqp_smooth(best_h, n_use, target_use, [5000, 15000], 300,
                                   time_limit=allowed, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()
    
    # Phase 9: Final binary SA burst (remaining time)
    if tleft() > 20:
        n_use = len(best_h)
        target_use_int = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use_int)
        n_sa_f = int(min(500000, max(5000, (tleft() - 10) * 30000)))
        h_sa_f, c5_f = sa_binary(h_bin_final, n_sa_f, 0.005, 0.00001,
                                 rng, time_limit=max(5.0, tleft() - 6), start_time=start_time,
                                 focused=True)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
    
    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, n_final
```
