grad[k:n] += 1 - h[0:n-k]

This doesn't vectorize easily across lags because each lag has a different "shift".

def projected_gradient(h_init, n_iter, lr, momentum, alpha, n_active):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    vel = np.zeros(n)
    
    for i in range(n_iter):
        corr = compute_full_corr(h)
        # Get top lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        # Softmax weights
        v = corr[top_idx]
        w = np.exp(alpha * (v - v[0]))
        w /= w.sum()
        # Gradient
        grad = np.zeros(n)
        for wi, ii in zip(w, top_idx):
            grad += wi * grad_for_full_corr_index(h, int(ii), n)
        grad *= dx
        # Update
        vel = momentum * vel + (1 - momentum) * grad
        h_new = h - lr * vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if sc < compute_c5(h) + 1e-15:
            h = h_new
        else:
            lr *= 0.95
            if lr < 1e-8:
                break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_full_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_full_corr_index(h, idx, n):
    lag = idx - (n - 1)
    grad = np.zeros(n)
    k = abs(lag)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
        return grad
    if k >= n:
        return grad
    if lag > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    return grad

def slsqp_full(h_init, alphas, maxiter, ftol):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    for alpha in alphas:
        def obj(x):
            h = project_feasible(x, target)
            corr = compute_full_corr(h)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
        def obj_grad(x):
            h = project_feasible(x, target)
            corr = compute_full_corr(h)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            weights = np.exp(shifted)
            weights /= weights.sum()
            grad = np.zeros(n)
            for i in range(len(corr)):
                if weights[i] > 1e-10:
                    grad += weights[i] * grad_for_full_corr_index(h, i, n)
            grad *= dx
            return grad
        def cons_eq(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

def proj_grad_momentum(h_init, n_iter, lr_init, lr_decay, momentum, alpha, n_active, patience=50):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    vel = np.zeros(n)
    lr = lr_init
    no_improve = 0
    for i in range(n_iter):
        corr = compute_full_corr(h)
        top_idx = np.argsort(corr)[::-1][:n_active]
        v = corr[top_idx]
        w = np.exp(alpha * (v - v[0]))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ii in zip(w, top_idx):
            grad += wi * grad_for_full_corr_index(h, int(ii), n)
        grad *= dx
        vel = momentum * vel + (1.0 - momentum) * grad
        h_new = h - lr * vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break
        if sc <= compute_c5(h) + 1e-14:
            h = h_new
            lr = min(lr * 1.002, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def bangbang_local_search(h_bb, max_rounds=50, n_trials_per_round=3000, rng=None):
    n = len(h_bb)
    h = h_bb.copy()
    cur_score = compute_c5(h)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    h[ones] = 1.0
    h[zeros] = 0.0
    cur_score = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        for t in range(n_trials_per_round):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            sc = compute_c5(h)
            if sc < cur_score - 1e-14:
                cur_score = sc
                ones = np.delete(ones, np.where(ones == i)[0][0])
                ones = np.append(ones, j)
                zeros = np.delete(zeros, np.where(zeros == j)[0][0])
                zeros = np.append(zeros, i)
                n_ones = len(ones)
                n_zeros = len(zeros)
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, cur_score
