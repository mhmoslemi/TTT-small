    n = len(h_init)

    h = h_init.copy()

    dx = 2.0 / n
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    ones_set = np.where(h == 1)[0].copy()
    zeros_set = np.where(h == 0)[0].copy()
    n_ones = len(ones_set)
    n_zeros = len(zeros_set)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
        
        # Find worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        # lag in terms of shift: k_star - (n-1)
        lag = k_star - (n - 1)
        
        # Find bad positions for this lag
        if lag >= 0:
            valid_i = np.arange(0, n - lag)
            bad_ones = valid_i[(h[valid_i] == 1) & (h[valid_i + lag] == 0)]
        else:
            l = -lag
            valid_i = np.arange(l, n)
            bad_ones = valid_i[(h[valid_i] == 1) & (h[valid_i - l] == 0)]
        
        if len(bad_ones) == 0:
            # No bad positions, try random swap
            i = ones_set[rng.randint(n_ones)]
            j = zeros_set[rng.randint(n_zeros)]
            candidates = [(i, j)]
        else:
            # Try swapping each bad one with a random zero
            n_try = min(len(bad_ones), 20)
            try_idx = rng.choice(len(bad_ones), n_try, replace=False)
            candidates = []
            for ti in try_idx:
                i = bad_ones[ti]
                j = zeros_set[rng.randint(n_zeros)]
                if j != i:
                    candidates.append((i, j))
        
        # Try all candidates, pick the best
        best_swap_c5 = cur_c5
        best_swap = None
        for (i, j) in candidates:
            h[i] = 0.0
            h[j] = 1.0
