            b_vals = slopes + lam

            a_use = a_vals[:n_use]

            # All pairwise crossings (vectorized)
            da = a_use[:, None] - a_use[None, :]
            db = b_use[None, :] - b_use[:, None]

            with np.errstate(divide='ignore', invalid='ignore'):

            else:
                candidates = np.array([0.0, 1.0, h[i]])
            
            # Evaluate
            vals = a_use[None, :] + b_use[None, :] * candidates[:, None]
            max_vals = vals.max(axis=1)

        h = project_feasible(h, target)

            h[mask] += rng.randn(max(1, int(mask.sum()))) * 0.05

    return best_h, best_c5

import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp_refined(h_init, n, target, max_iter, time_limit,
                           n_top=150, damping_start=0.9, damping_end=0.8,
                           reg_start=1e-4, reg_end=1e-9, solver='SCS'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 300 == 0 and stuck > 600:
            break

        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argpartition(overlaps, -n_top)[-n_top:]
        lags = top_idx - n_half

        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5

def coordinate_descent(h_init, n, target, time_limit, n_active=40, n_lines=15,
                       seed=99, momentum=0.0, shake_interval=300,
                       shake_prob=0.1, shake_noise=0.05):
    start_t = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    lam = 0.0
    no_improve = 0
    n_half = n - 1

    for sweep in range(200000):
        if time.time() - start_t > time_limit:
            break
        if no_improve > 1500:
            break

        order = rng.permutation(n)
        for i in order:
            overlaps = compute_overlaps(h)
            n_act = min(n_active, 2 * n_half + 1)
            top_idx = np.argpartition(overlaps, -n_act)[-n_act:]
            top_lags = top_idx - n_half
            top_vals = overlaps[top_idx]

            ip = i + top_lags
            im = i - top_lags
            valid_p = (ip >= 0) & (ip < n)
            valid_m = (im >= 0) & (im < n)
            slopes = np.zeros(len(top_lags))
            if np.any(valid_p):
                slopes[valid_p] += (1.0 - h[ip[valid_p]])
            if np.any(valid_m):
                slopes[valid_m] -= h[im[valid_m]]
            slopes *= dx

            a_vals = top_vals - slopes * h[i]
            b_vals = slopes + lam

            n_use = min(n_lines, len(a_vals))
            a_use = a_vals[:n_use]
            b_use = b_vals[:n_use]

            da = a_use[:, None] - a_use[None, :]
            db = b_use[None, :] - b_use[:, None]
            valid = np.abs(db) > 1e-15
            with np.errstate(divide='ignore', invalid='ignore'):
                xc = np.where(valid, da / np.where(valid, db, 1.0), -1.0)
            xc = xc[(xc >= 0.0) & (xc <= 1.0)]

            if len(xc) > 0:
                candidates = np.unique(np.concatenate([xc, [0.0, 1.0, h[i]]]))
            else:
                candidates = np.array([0.0, 1.0, h[i]])

            vals = a_use[None, :] + b_use[None, :] * candidates[:, None]
            max_vals = vals.max(axis=1)
            best_x = float(candidates[max_vals.argmin()])

            if momentum > 0:
                best_x = (1.0 - momentum) * h[i] + momentum * best_x
            h[i] = best_x

        sum_err = np.sum(h) - target
        lam = lam * 0.85 + 0.5 * sum_err / n
        h = project_feasible(h, target)

        c5 = compute_c5(h)

            h[mask] += rng.randn(cnt) * shake_noise
            h = project_feasible(h, target)
            no_improve = 0
            lam = 0.0

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                                 n_active=35, n_lines=14,
                                                 seed=seed + n_try)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3
                n = n_try
                target = target_try

    if tleft() > 60:
        cd_time = min(40, tleft() - 50)
        if cd_time > 10:
            h_cd4, c5_cd4 = coordinate_descent(best_h, n, target, time_limit=cd_time,
                                                n_active=50, n_lines=20,
                                                seed=seed + 500, momentum=0.2,
                                                shake_interval=500, shake_prob=0.03,
                                                shake_noise=0.02)
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4

    if tleft() > 50:
        h_lp3, c5_lp3 = alternating_lp_refined(best_h, n, target,
                                                max_iter=2000, time_limit=min(35, tleft() - 40),
                                                n_top=100, damping_start=0.96,
                                                damping_end=0.93, reg_start=1e-5,
                                                reg_end=1e-10)
        if c5_lp3 < best_c5:
            best_c5 = c5_lp3
            best_h = h_lp3

    if tleft() > 30:
        cd_time = min(20, tleft() - 25)
        if cd_time > 5:
            h_cd5, c5_cd5 = coordinate_descent(best_h, n, target, time_limit=cd_time,
                                                n_active=80, n_lines=30,
                                                seed=seed + 999, momentum=0.1,
                                                shake_interval=9999)
            if c5_cd5 < best_c5:
                best_c5 = c5_cd5
                best_h = h_cd5

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_c5 = compute_c5(h_final)

    return h_final, final_c5, len(h_final)
