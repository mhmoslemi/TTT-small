        if abs(diff) < 1e-14:

        if cnt == 0:
            break

def slsqp_optimize(h, alphas, maxiter, ftol, start_time=None, time_limit=None):
    global _g_n, _g_alpha
    _g_n = len(h)
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_n / 2.0)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def binary_local_search(h, max_accepted=200, start_time=None, time_limit=None, seed=123):
    n = len(h)
    dx = 2.0 / n
    h_b = np.where(h >= 0.5, 1.0, 0.0).copy()
    target = n // 2
    n_ones = int(np.sum(h_b))
    rng = np.random.RandomState(seed)
    if n_ones > target:
        ones = np.where(h_b == 1.0)[0]
        remove = rng.choice(ones, n_ones - target, replace=False)
        h_b[remove] = 0.0
    elif n_ones < target:
        zeros = np.where(h_b == 0.0)[0]
        add = rng.choice(zeros, target - n_ones, replace=False)
        h_b[add] = 1.0
    
    corr = compute_full_corr(h_b)
    best_max = float(np.max(corr))
    best_h = h_b.copy()
    n_accepted = 0
    trials_per_check = 30
    
    while n_accepted < max_accepted:
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        if n_accepted % 50 == 0 and n_accepted > 0:
            # Periodically re-evaluate
            pass
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        found_in_batch = 0
        for _ in range(trials_per_check):
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            old_i, old_j = h_b[i], h_b[j]
            h_b[i] = 0.0
            h_b[j] = 1.0
            new_corr = compute_full_corr(h_b)
            new_max = float(np.max(new_corr))
            if new_max < best_max - 1e-12:
                corr = new_corr
                best_max = new_max
                best_h = h_b.copy()
                n_accepted += 1
                found_in_batch += 1
            else:
                h_b[i] = old_i
                h_b[j] = old_j
            if n_accepted >= max_accepted:
                break
        if found_in_batch == 0:
            trials_per_check = min(trials_per_check * 2, 200)
            if trials_per_check >= 200:
                # Try one more round with larger batch
                trials_per_check = 200
    
    return best_h, best_max * dx

def subgradient_step(h, n_active, temp, start_time=None, time_limit=None, n_steps=300, lr_init=0.003, lr_decay=0.997, momentum=0.3):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    rng = np.random.RandomState(77)
    
    for step in range(n_steps):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        corr = compute_full_corr(h_cur)
        over = corr[n-1:n-1+n]  # shifts 0 to n-1
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Sample active lags with softmax weighting
        top_idx = np.argsort(over)[::-1][:max(n_active, 1)]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    
    return best_h, best_score

def worst_lag_surgery(h, max_steps=200, initial_epsilon=0.001, start_time=None, time_limit=None):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    
    for step in range(max_steps):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        corr = compute_full_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            grad = dx * (1.0 - 2.0 * h_cur)
        else:
            grad = np.zeros(n)
            upper = n - 1 - k_star
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h_cur[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h_cur[idx2 - k_star]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.006)
        else:
            eps *= 0.8
            if eps < 1e-8:
                break
    
    return h_cur, cur_score

def top_lag_surgery(h, max_steps=150, initial_epsilon=0.0008, n_lags=2, start_time=None, time_limit=None):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    
    for step in range(max_steps):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        corr = compute_full_corr(h_cur)
        over = corr[n-1:n-1+n]
        top_lags = np.argsort(over)[::-1][:n_lags]
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= dx * h_cur[idx2 - k]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.005)
        else:
            eps *= 0.85
            if eps < 1e-8:
                break
    
    return h_cur, cur_score

def tleft(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: SLSQP refinement
    h_ref, sc = slsqp_optimize(best_h, [50, 200, 1000], 2000, 1e-15, start_time, budget_s - 200)
    if sc < best_score:
        best_score = sc
        best_h = h_ref
    
    # Phase 2: Binary local search
    if tleft(start_time, budget_s) > 200:
        h_bin, sc_bin = binary_local_search(best_h, max_accepted=300, start_time=start_time, time_limit=budget_s - 150, seed=1)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            # SLSQP after binary
            h_ref, sc = slsqp_optimize(best_h, [500, 2000, 10000], 2000, 1e-15, start_time, budget_s - 100)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    
    # Phase 3: Subgradient
    if tleft(start_time, budget_s) > 200:
        h_sg, sc_sg = subgradient_step(best_h, 4, 0.004, start_time, budget_s - 100, n_steps=500, lr_init=0.003, lr_decay=0.997, momentum=0.3)
