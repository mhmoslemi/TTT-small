def soft_max_obj(h, alpha):
    """Smooth approximation of max of correlation."""
    corr = get_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr.astype(float) - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / alpha


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(soft_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def greedy_targeted(h, max_iters, time_limit=None, start_time=None, top_k=5, scan_width=25):
    """Greedy that focuses on the worst shifts."""

        if stall > max_stall:
            break

        max_val = float(corr.max())
        top_idx = np.argsort(corr)[-top_k:]
        top_shifts = top_idx - (n - 1)
        top_vals = corr[top_idx]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improve = 0.0
        best_a = -1
        best_b = -1

        for t_idx in range(len(top_shifts)):
            k = top_shifts[t_idx]
            # Find a (1-position) with h[a+k]=0 → changing a reduces this shift
            # Find b (0-position) with h[b+k]=1 → changing b increases this shift
            # Net effect: h[a+k] - h[b+k]
            # For maximum reduction: a+k=0, b+k=1
            a_cand = ones[(ones + k) % n < n]  # need bounds check

            b_cand = [j for j in zeros if 0 <= j + k < n and h[j + k] == 1]

            n_a = min(len(a_cand), scan_width)
            n_b = min(len(b_cand), scan_width)

            for ii in range(n_a):
                a = a_cand[ii]
                for jj in range(n_b):
                    b = b_cand[jj]
                    # Compute new max considering all top shifts
                    new_max = max_val
                    for t2 in range(len(top_shifts)):
                        k2 = top_shifts[t2]
                        delta = 0.0

                        if 0 <= a - k2 < n:
                            delta += h[a - k2]

def legendre_init(n, rng):
    """Generate a Legendre sequence of length n (or closest prime)."""
    p = n
    while True:
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 1
    if p % 2 == 0 and p != 2:
        p += 1
        while True:
            is_prime = True
            for d in range(2, int(np.sqrt(p)) + 1):
                if p % d == 0:
                    is_prime = False
                    break
            if is_prime and p % 2 == 1:
                break
            p += 2

    leg = np.zeros(p)
    leg[0] = 1.0
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
        else:
            leg[a] = 1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(soft_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def legendre_init(n):
    p = n
    while True:
        is_prime = True
        for d in range(2, int(np.sqrt(p)) + 1):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p % 4 == 3:
            break
        p += 2
    leg = np.zeros(p)
    leg[0] = 1.0
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
        else:
            leg[a] = 1.0
    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5
    target = n // 2
    h = project_binary(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Fast greedy
    allowed = min(60, max(10, tleft() * 0.06))
    h_g, c5_g = greedy_fast(best_h, max_iters=200000, time_limit=allowed,
                            start_time=start_time, top_k=6, scan_width=25)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Main directed SA (long)
    if tleft() > 80:
        allowed = min(350, tleft() * 0.30)
        max_it = int(min(500000, allowed * 1500))
        h_sa, c5_sa = sa_directed(best_h, target_init, max_it, 0.012, 0.00002, rng,
                                  time_limit=allowed, start_time=start_time, top_k=10)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Cooler SA
    if tleft() > 60:
        allowed = min(200, tleft() * 0.20)
        max_it = int(min(350000, allowed * 1500))
        h_sa2, c5_sa2 = sa_directed(best_h, target_init, max_it, 0.005, 0.000005, rng,
                                    time_limit=allowed, start_time=start_time, top_k=14)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Try neighboring n values
    n_options = [n_init + 2, n_init + 4, n_init - 2, n_init + 6, n_init + 8]
    n_options = sorted(set([x for x in n_options if x >= 300 and x % 2 == 0 and x != n_init]))

    for n_try in n_options:
        if tleft() < 50:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(20, tleft() * 0.04)
        h_g2, c5_g2 = greedy_fast(h_bin, max_iters=100000, time_limit=allowed,
                                  start_time=start_time, top_k=5, scan_width=20)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 30:
            continue
        allowed2 = min(40, tleft() * 0.06)
        max_it2 = int(min(200000, allowed2 * 1500))
        h_sa3, c5_sa3 = sa_directed(h_g2, target_try, max_it2, 0.008, 0.00003, rng,
                                    time_limit=allowed2, start_time=start_time, top_k=8)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 5: Legendre restart
    if tleft() > 80:
        n_leg = n_init
        target_leg = int(n_leg / 2)
        h_leg = legendre_init(n_leg)
        allowed = min(25, tleft() * 0.04)
        h_g3, c5_g3 = greedy_fast(h_leg, max_iters=100000, time_limit=allowed,
                                  start_time=start_time, top_k=6, scan_width=25)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()
        if tleft() > 40:
            allowed2 = min(60, tleft() * 0.08)
            max_it3 = int(min(200000, allowed2 * 1500))
            h_sa4, c5_sa4 = sa_directed(h_g3, target_leg, max_it3, 0.01, 0.00003, rng,
                                        time_limit=allowed2, start_time=start_time, top_k=10)
            if c5_sa4 < best_score:
                best_score = c5_sa4
                best_h = h_sa4.copy()

    # Phase 6: ILS perturbations
    if tleft() > 60:
        for pert_iter in range(6):
            if tleft() < 25:
                break
            n_pert = 2 + pert_iter
            h_pert = best_h.copy()
            ones_p = np.where(h_pert == 1)[0]
            zeros_p = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones_p) == 0 or len(zeros_p) == 0:
                    break
                i = ones_p[rng.randint(len(ones_p))]
                j = zeros_p[rng.randint(len(zeros_p))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones_p = np.where(h_pert == 1)[0]
                zeros_p = np.where(h_pert == 0)[0]
            allowed = min(15, tleft() * 0.03)
            h_pg, c5_pg = greedy_fast(h_pert, max_iters=80000, time_limit=allowed,
                                      start_time=start_time, top_k=5, scan_width=20)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()
            if tleft() < 15:
                break
            allowed2 = min(15, tleft() * 0.03)
            max_it4 = int(min(80000, allowed2 * 1500))
            h_ps, c5_ps = sa_directed(h_pg, int(len(h_pg) / 2), max_it4,
                                      0.004, 0.00001, rng,
                                      time_limit=allowed2, start_time=start_time, top_k=8)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 7: Continuous SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 100, 300, 1000, 3000, 10000, 30000, 100000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-15,
                                 time_limit=max(5, tleft() - 25), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary SA on refined
    if tleft() > 25:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(30, tleft() - 10)
        max_it5 = int(min(150000, allowed * 1500))
        h_sf, c5_sf = sa_directed(h_bin_f, target_use, max_it5, 0.003, 0.000005,
                                  rng, time_limit=allowed, start_time=start_time, top_k=12)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final polish
    if tleft() > 10:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [50000, 150000, 500000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 1000, 1e-16,
                                   time_limit=max(3, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
