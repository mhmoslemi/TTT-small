    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_local_search(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if k_star == 0:
            stall += 1
            continue

        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def replica_exchange_sa(h_init, n_iters, rng, time_limit, start_time, n_repl=10):
    n = len(h_init)
    dx = 2.0 / n
    positions = np.arange(n)

    target_ratio = 0.5
    temps = np.geomspace(0.003, 0.12, n_repl)

    replicas_h = []
    replicas_corr = []
    replicas_energy = []
    for r in range(n_repl):
        h_r = h_init.copy()
        if r > 0:
            n_pert = rng.randint(2, 6 + r * 2)
            ones = np.where(h_r == 1.0)[0]
            zeros = np.where(h_r == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h_r[i] = 0.0
                    h_r[j] = 1.0
                    ones = np.where(h_r == 1.0)[0]
                    zeros = np.where(h_r == 0.0)[0]
        corr_r = full_corr_fast(h_r)
        energy_r = float(corr_r.max()) * dx
        replicas_h.append(h_r)
        replicas_corr.append(corr_r)
        replicas_energy.append(energy_r)

    best_h = h_init.copy()
    best_energy = float(full_corr_fast(h_init).max()) * dx
    stall_count = 0
    max_stall = 200000

    swap_interval = max(1, n_iters // (n_repl * 50))

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall_count > max_stall:
            break

        r = rng.randint(n_repl)
        t = temps[r]
        h_r = replicas_h[r]
        corr_r = replicas_corr[r]
        energy_r = replicas_energy[r]

        ones = np.where(h_r == 1.0)[0]
        zeros = np.where(h_r == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        move_type = rng.random()
        if move_type < 0.7:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h_r, i, j)
            new_corr = corr_r + delta
            new_energy = float(new_corr.max()) * dx
            delta_e = new_energy - energy_r

            if delta_e < 0 or rng.random() < np.exp(-delta_e / t):
                h_r[i] = 0.0
                h_r[j] = 1.0
                replicas_corr[r] = new_corr
                replicas_energy[r] = new_energy
                replicas_h[r] = h_r
                if new_energy < best_energy:
                    best_energy = new_energy
                    best_h = h_r.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1
        elif move_type < 0.9:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                i2_arr = np.delete(ones, np.where(ones == i1)[0][0])
                i2 = i2_arr[rng.randint(len(ones)-1)]
                j1 = zeros[rng.randint(len(zeros))]
                j2_arr = np.delete(zeros, np.where(zeros == j1)[0][0])
                j2 = j2_arr[rng.randint(len(zeros)-1)]
                if i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2 and i1 != i2 and j1 != j2:
                    delta1 = compute_delta(h_r, i1, j1)
                    delta2 = compute_delta(h_r, i2, j2)
                    delta = delta1 + delta2
                    new_corr = corr_r + delta
                    new_energy = float(new_corr.max()) * dx
                    delta_e = new_energy - energy_r
                    if delta_e < 0 or rng.random() < np.exp(-delta_e / t):
                        h_r[i1] = 0.0
                        h_r[j1] = 1.0
                        h_r[i2] = 0.0
                        h_r[j2] = 1.0
                        replicas_corr[r] = new_corr
                        replicas_energy[r] = new_energy
                        replicas_h[r] = h_r
                        if new_energy < best_energy:
                            best_energy = new_energy
                            best_h = h_r.copy()
                            stall_count = 0
                        else:
                            stall_count += 1
                    else:
                        stall_count += 1
                else:
                    stall_count += 1
            else:
                stall_count += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h_r, i, j)
            new_corr = corr_r + delta
            new_energy = float(new_corr.max()) * dx
            delta_e = new_energy - energy_r
            if delta_e < 0 or rng.random() < np.exp(-delta_e / t):
                h_r[i] = 0.0
                h_r[j] = 1.0
                replicas_corr[r] = new_corr
                replicas_energy[r] = new_energy
                replicas_h[r] = h_r
                if new_energy < best_energy:
                    best_energy = new_energy
                    best_h = h_r.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                stall_count += 1

        if (it + 1) % swap_interval == 0:
            for s in range(n_repl - 2, -1, -1):
                r1, r2 = s, s + 1
                e1, e2 = replicas_energy[r1], replicas_energy[r2]
                t1, t2 = temps[r1], temps[r2]
                if t1 > 0 and t2 > 0:
                    swap_prob = np.exp(-(e2 - e1) * (1.0/t1 - 1.0/t2))
                    if rng.random() < min(1.0, swap_prob):
                        replicas_h[r1], replicas_h[r2] = replicas_h[r2], replicas_h[r1]
                        replicas_corr[r1], replicas_corr[r2] = replicas_corr[r2], replicas_corr[r1]
                        replicas_energy[r1], replicas_energy[r2] = replicas_energy[r2], replicas_energy[r1]

    return best_h, best_energy

def focused_sa(h_init, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if it % 3 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]

                i_cands = ones[A[ones] <= 1.0]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        elif it % 3 == 1:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1 = ones[rng.randint(len(ones))]
                ii = np.where(ones != i1)[0]
                i2 = ones[ii[rng.randint(len(ii))]]
                j1 = zeros[rng.randint(len(zeros))]
                ji = np.where(zeros != j1)[0]
                j2 = zeros[ji[rng.randint(len(ji))]]
                if i1 != j1 and i1 != j2 and i2 != j1 and i2 != j2:
                    d1 = compute_delta(h, i1, j1)
                    d2 = compute_delta(h, i2, j2)
                    new_max = float((full + d1 + d2).max())
                    c5_new = new_max * dx
                    delta_c5 = c5_new - current_c5
                    if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                        h[i1] = 0.0; h[j1] = 1.0
                        h[i2] = 0.0; h[j2] = 1.0
                        full += d1 + d2
                        current_c5 = c5_new
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h.copy()
                            stall = 0
                        else:
                            stall += 1
                        continue
                stall += 1
                continue

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def generate_thue_morse(n, target):
    tm = np.zeros(n, dtype=int)
    tm[0] = 0
    i = 1
    while i < n:
        j = 0
        k = i
        while j < n and k < n:
            tm[k] = 1 - tm[j]
            j += 1
            k += 1
        i *= 2
    h = (1 - tm).astype(float)
    return project_binary(h, target)

                                             time_limit=allowed_rx, start_time=start_time,
                                             n_repl=8)
        if c5_rx2 < best_score:
            best_score = c5_rx2
            best_h = h_rx2.copy()

    if tleft() > 100:
        n_tm = n_init
        target_tm = n_init // 2
        h_tm = generate_thue_morse(n_tm, target_tm)
        allowed_tm = min(30, tleft() * 0.12)
        h_tmg, c5_tmg = greedy_local_search(h_tm, max_iters=20000, rng=rng,
                                             time_limit=allowed_tm * 0.5, start_time=start_time)
        allowed_tm2 = min(20, tleft() * 0.08)

    for n_try in [n_init - 4, n_init + 4, n_init - 8, n_init + 8, n_init + 12]:
        if tleft() < 45:

    if tleft() > 12:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = max(5.0, tleft() - 8)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,

                                  time_limit=allowed, start_time=start_time)

            best_score = sc
            best_h = h_ref

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
