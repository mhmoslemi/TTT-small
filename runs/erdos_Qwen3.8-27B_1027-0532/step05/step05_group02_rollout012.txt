def project_sum(h, target_sum):
    # Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}
    h = h.copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-15:
        return np.clip(h, 0, 1)
    # Water filling
    if diff > 0:
        # Need to add diff total, distribute to elements with most room
        order = np.argsort(1.0 - h)  # most room first
        n_add = len(order)
        for k in range(1, n_add + 1):
            add_each = diff / k
            if add_each <= 1.0 - h[order[k-1]]:
                h[order[:k]] += add_each
                break
        else:
            h[:] = 1.0  # shouldn't happen if target_sum <= n
    else:
        need = -diff
        order = np.argsort(h)  # least value first (most room to decrease)
        n_sub = len(order)
        for k in range(1, n_sub + 1):
            sub_each = need / k
            if sub_each <= h[order[k-1]]:
                h[order[:k]] -= sub_each
                break
        else:
            h[:] = 0.0
    return np.clip(h, 0, 1)
