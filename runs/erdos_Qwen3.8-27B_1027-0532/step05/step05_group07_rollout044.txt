def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smoothed_objective(h, alpha, target, penalty=1e6):
    corr = compute_corr(h)
    max_c = float(np.max(corr))
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = max_c + float(np.log(np.sum(np.exp(shifted)))) / alpha
    constraint_viol = np.sum(h) - target
    return lse + penalty * constraint_viol * constraint_viol

def compute_smoothed_gradient(h, alpha, target, penalty=1e6):
    n = len(h)
    corr = compute_corr(h)
    max_c = float(np.max(corr))

            grad[k:] += -p[j] * h[:n - k]
        else:
            grad[-k:] += p[j] * one_minus_h[:n + k]
            grad[:-k] += -p[j] * h[-k:]

    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, target, penalty):
    n = len(h)
    corr = compute_corr(h)
    max_c = float(np.max(corr))
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -700.0, 0.0)
    vals = np.exp(shifted)
    total = vals.sum()
    lse_val = max_c + np.log(total) / alpha
    
    # Gradient
    p = vals / total
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    
    # We need: grad[i] = sum_k p_k * [(1-h[i+k]) - h[i-k]] (with validity)
    # p has length 2n-1, p[j] corresponds to lag k = j - (n-1)
    for j in range(2 * n - 1):

        else:
            pk = -k
            grad[pk:] += p[j] * one_minus_h[:n-pk]
            grad[:n-pk] -= p[j] * h[pk:]
    
    viol = np.sum(h) - target
    grad += 2.0 * penalty * viol
    obj = lse_val + penalty * viol * viol
    return obj, grad

def lbfgsb_run(h, n, target, alphas, maxiter_per_alpha, tol, ftol):
    bounds = [(0.0, 1.0)] * n
    penalty = 1e5
    for a in alphas:
        f = make_lbfgsb_fun(a, target, penalty)
        g = make_lbfgsb_grad(a, target, penalty)
        try:
            res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': ftol, 'gtol': 1e-8,
                                    'maxfun': maxiter_per_alpha * 3})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                h = h_new
        except Exception:
            pass
    return h


def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def paired_cd(h, n, target, deadline, rng, n_sweeps, recompute_interval):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_vec(h_cur, i, n)
            Bj = compute_B_vec(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            lo = 0.0
            hi = float(np.max(A + C * t_max))
            if not np.isfinite(hi):
                continue

            for _ in range(50):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)):
                    if np.max(A[~(pos | neg)]) > mid + 1e-13:
                        lo = mid
                        continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 300 and sweep > 200:
            pert_scale = 0.01 + 0.01 * min(stall_count // 300, 5)
            h_cur = project_feasible(h_cur + rng.randn(n) * pert_scale, target)
            corr = compute_corr(h_cur)
            best_score = float(np.max(corr)) * dx
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def gradient_cd(h, n, target, deadline, rng, n_steps):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    stagnant = 0

    for step in range(n_steps):
        if time.time() > deadline:
            break

        alpha = 2000
        _, grad = smooth_obj_grad(h_cur, alpha, target, 0.0)
        grad_centered = grad - np.mean(grad)

        improved = False
        for lr in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            h_try = project_feasible(h_cur - lr * grad_centered, target)
            c_try = compute_c5(h_try)
            if c_try < best_score - 1e-15:
                h_cur = h_try
                best_score = c_try
                improved = True
                stagnant = 0
                break

        if not improved:
            stagnant += 1
            if stagnant > 50:
                noise = 0.005 + 0.005 * min(stagnant // 50, 4)
                h_cur = project_feasible(h_cur + rng.randn(n) * noise, target)
                best_score = min(best_score, compute_c5(h_cur))
                stagnant = 0

    return h_cur, compute_c5(h_cur)


def multi_resolution(h, n, target, deadline, rng):
    best_h = h.copy()
    best_score = compute_c5(h)
    coarse_sizes = [n // 2, n // 4, n // 8]
    for cs in coarse_sizes:
        if cs < 20 or (time.time() - (deadline - 300)) > 250:
            break
        target_cs = cs / 2.0
        x_new = np.linspace(0.0, 1.0, cs)
        x_old = np.linspace(0.0, 1.0, n)
        h_cs = np.interp(x_new, x_old, best_h)
        h_cs = project_feasible(h_cs, target_cs)
        h_cs = lbfgsb_run(h_cs, cs, target_cs, [500, 5000, 50000], 3000, 1e-12, 1e-14)
        h_up = np.interp(x_old, x_new, h_cs)
        h_up = project_feasible(h_up, target)
        h_up = lbfgsb_run(h_up, n, target, [5000, 50000], 3000, 1e-12, 1e-14)
        sc = compute_c5(h_up)
        if sc < best_score:
            best_score = sc
            best_h = h_up
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 80:
        h = lbfgsb_run(best_h, n, target, [50, 200], 3000, 1e-12, 1e-13)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h

    if tleft() > 150:
        h = lbfgsb_run(best_h, n, target, [1000, 5000], 5000, 1e-13, 1e-14)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h

    if tleft() > 200:
        deadline = start_time + budget_s * 0.35
        h_cd, sc_cd = paired_cd(best_h, n, target, deadline, rng, 50000, 50)
        if sc_cd < best_score:
            best_score, best_h = sc_cd, h_cd

    if tleft() > 200:
        h = lbfgsb_run(best_h, n, target, [20000, 100000], 5000, 1e-13, 1e-14)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h

    if tleft() > 300:
        deadline_mr = start_time + budget_s * 0.60
        best_h, best_score = multi_resolution(best_h, n, target, deadline_mr, rng)

    if tleft() > 300:
        noise_levels = [0.01, 0.02, 0.03, 0.05, 0.015, 0.025, 0.04, 0.01,
                        0.02, 0.035, 0.015, 0.05, 0.01, 0.025, 0.04, 0.03,
                        0.01, 0.02, 0.05, 0.015]
        for attempt in range(20):
            if tleft() < 80:
                break
            nl = noise_levels[attempt % len(noise_levels)]
            h_pert = project_feasible(best_h + rng.randn(n) * nl, target)
            h = lbfgsb_run(h_pert, n, target, [2000, 20000, 200000], 3000, 1e-13, 1e-14)
            sc = compute_c5(h)
            if sc < best_score:
                best_score, best_h = sc, h

    if tleft() > 150:
        deadline = start_time + budget_s * 0.88
        h_gc, sc_gc = gradient_cd(best_h, n, target, deadline, rng, 5000)
        if sc_gc < best_score:
            best_score, best_h = sc_gc, h_gc

    if tleft() > 100:
        deadline = start_time + budget_s * 0.94
        h_cd, sc_cd = paired_cd(best_h, n, target, deadline, rng, 80000, 30)
        if sc_cd < best_score:
            best_score, best_h = sc_cd, h_cd

    if tleft() > 40:
        h = lbfgsb_run(best_h, n, target, [500000, 2000000], 10000, 1e-14, 1e-15)
        sc = compute_c5(h)
        if sc < best_score:
            best_score, best_h = sc, h

    if tleft() > 15:
        deadline = start_time + budget_s - 5.0
        h_cd, sc_cd = paired_cd(best_h, n, target, deadline, rng, 50000, 20)
        if sc_cd < best_score:
            best_score, best_h = sc_cd, h_cd

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
