import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_overlap(h):
    """Compute I(k) for k=0..n-1 using FFT: I(k) = sum_j h[j+k]*(1-h[j])"""
    n = len(h)
    n_fft = 1
    while n_fft < 2 * n:
        n_fft *= 2
    H = fft(h, n_fft)
    G = fft(1.0 - h, n_fft)
    # Cross-correlation: (h cross-corr (1-h))[m] = sum_j h[j] * (1-h)[j-m]
    # = ifft(FFT(h) * conj(FFT(1-h)))
    cc = ifft(H * np.conj(G)).real
    # cc[m] = sum_j h[j] * (1-h[j-m mod n_fft])
    # For m in 0..n-1 with zero padding: cc[m] = sum_{j=m}^{n-1} h[j] * (1-h[j-m])
    # = sum_{j=0}^{n-1-m} h[j+m] * (1-h[j])  [substituting j->j+m]
    # This is I(k)/dx for k=m
    return cc[:n]

def fft_gradient(h, alpha, target_sum):
    """Compute gradient of the soft-max objective using FFT."""
    n = len(h)
    dx = 2.0 / n
    over = fft_overlap(h)  # I(k)/dx for k=0..n-1
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()  # softmax weights
    
    # Gradient of f = c_max + (1/alpha)*log(sum(exp(alpha*(over-c_max))))
    # = (1/alpha) * sum_k alpha * w_k * over_k ... 
    # Actually: f = c_max + (1/alpha)*log(sum_k exp(alpha*(over_k - c_max)))
    # df/d(over_k) = w_k (the softmax weight)
    # d(over_k)/d(h[m]) = (1-h[m-k])*1_{m>=k} - h[m+k]*1_{m+k<n}
    # 
    # So df/dh[m] = sum_k w_k * [(1-h[m-k])*1_{m>=k} - h[m+k]*1_{m+k<n}]
    # = sum_{k=0}^{m} w_k*(1-h[m-k]) - sum_{k=0}^{n-1-m} w_k*h[m+k]
    
    # First term: (w * (1-h))[m] = ifft(FFT(w) * FFT(1-h))[m]
    # Second term: (h cross-corr w)[m] = ifft(FFT(h) * conj(FFT(w)))[m]
    
    n_fft = 1
    while n_fft < 2 * n:
        n_fft *= 2
    
    W = fft(w, n_fft)
    H = fft(h, n_fft)
    G = fft(1.0 - h, n_fft)
    
    term1 = ifft(W * G).real[:n]  # (w * (1-h))[m]
    term2 = ifft(H * np.conj(W)).real[:n]  # (h cross-corr w)[m]
    
    grad = (term1 - term2) * dx
    return grad, c_max * dx, w

def lbfgsb_softmax(h, n, target, alpha, maxiter, deadline):
    """L-BFGS-B optimization with soft-max objective."""
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        over = fft_overlap(x)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
        grad, _, _ = fft_gradient(x, alpha, target)
        return val, grad
    
    # Use penalty for sum constraint
    def obj_pen(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        val, grad = obj_and_grad(x_clipped)
        pen = 100.0 * (np.sum(x_clipped) - target)**2
        grad_pen = grad + 200.0 * (np.sum(x_clipped) - target)
        return val + pen, grad_pen
    
    res = minimize(obj_pen, h, method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                           'maxls': 50})
    return project_feasible(res.x, target)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_overlap(h):
    n = len(h)
    n_fft = 1
    while n_fft < 2 * n:
        n_fft *= 2
    H = fft(h, n_fft)
    G = fft(1.0 - h, n_fft)
    cc = ifft(H * np.conj(G)).real
    return cc[:n]

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    over = fft_overlap(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    Z = float(np.sum(exp_s))
    val = (c_max + float(np.log(Z)) / alpha) * dx
    w = exp_s / Z
    n_fft = 1
    while n_fft < 2 * n:
        n_fft *= 2
    W = fft(w, n_fft)
    H = fft(h, n_fft)
    G = fft(1.0 - h, n_fft)
    term1 = ifft(W * G).real[:n]
    term2 = ifft(H * np.conj(W)).real[:n]
    grad = (term1 - term2) * dx
    return val, grad

def accelerate(h0, n, target, alpha, n_steps, deadline, lr_init=0.008):

    best_h = h0.copy()
    best_f = compute_c5(best_h)
    t_val = 1.0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        val, grad = lse_obj_grad(y, alpha)
        lr = lr_init / (1.0 + 0.002 * step)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))

            best_h = h.copy()

    _st = target

        return np.sum(x) - _st

        return np.ones(_sn)

def compute_all_overlaps(h):
    """Compute I(k) for all k = -(n-1), ..., n-1"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # All overlap values

def soft_max_obj(h, alpha):
    over = compute_all_overlaps(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    over = corr_raw * dx
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    Z = float(np.sum(exp_s))
    w = exp_s / Z  # weights for each lag
    
    # w has length 2n-1, indexed as w[m] for lag m-(n-1)
    # Lag k corresponds to index k + n - 1
    # w_pos[k] = w[k + n - 1] for k = -(n-1), ..., n-1
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        wk = float(w[k + n - 1])
        if wk < 1e-15:
            continue
        if k >= 0:
            # I(k) = dx * sum_j h[j+k](1-h[j])
            # dI(k)/dh[m] = dx * [(1-h[m-k])*1_{m>=k} - h[m+k]*1_{m+k<n}]
            for m in range(k, n):
                grad[m] += wk * (1.0 - h[m-k])
            for m in range(0, n-k):
                grad[m] -= wk * h[m+k]
        elif k < 0:
            # I(k) = I(-|k|) = dx * sum_j h[j](1-h[j+k]) = dx * sum_j h[j](1-h[j-|k|])
            # Wait: for k < 0, I(k) = sum_j h[j](1-h[j+k]) where j+k >= 0
            # = sum_{j=-k}^{n-1} h[j](1-h[j+k])
            # dI(k)/dh[m] = dx * [(1-h[m+k])*1_{m+k>=0, m+k<n} - h[m-k]*1_{m-k>=0}]
            # For k < 0, m+k >= 0 means m >= -k, and m-k >= 0 means m >= k (always true since k<0 and m>=0)
            k_abs = -k
            for m in range(k_abs, n):
                grad[m] += wk * (1.0 - h[m-k])  # h[m] appears as h[j] with j=m, and (1-h[j+k])
            for m in range(0, n-k_abs):
                grad[m] -= wk * h[m+k_abs]  # h[m] appears in (1-h[j+k]) where j+k=m, so j=m-k
        grad *= dx
    
    return grad, c_max * dx
