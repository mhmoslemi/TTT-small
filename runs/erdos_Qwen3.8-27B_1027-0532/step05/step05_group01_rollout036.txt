import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            def obj(x, a=alpha):
                return smooth_obj(x, a)
            def con_fun(x):
                return np.sum(x) - target
            def con_jac(x):
                return np.ones(n)
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5

_SLQ_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET

def _slq_con_jac(x):
    return np.ones(len(x))

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_N, _SLQ_TARGET
    _SLQ_N = n
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:

def _slq_con_fun(x):
    return np.sum(x) - _SLQ_TARGET


def _slq_con_jac(x):
    return np.ones(len(x))


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _SLQ_TARGET
    _SLQ_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_con_fun, 'jac': _slq_con_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_c5


def fast_pair_cd(h_init, n, target, time_limit, n_cand=25, max_sweeps=50000):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0
    idxs = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        sorted_by_grad = np.argsort(-grad)
        high_set = sorted_by_grad[:n_cand]
        low_set = sorted_by_grad[-n_cand:]

        all_cands = np.unique(np.concatenate([high_set, low_set]))
        n_c = len(all_cands)
        profiles = np.zeros((n_c, n_corr))
        for ci in range(n_c):
            m = int(all_cands[ci])
            ks = np.arange(-(n - 1), n)
            idx1 = m - ks
            idx2 = m + ks
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            if np.any(valid1):
                profiles[ci, valid1] += h[idx1[valid1]]
            if np.any(valid2):
                profiles[ci, valid2] += h[idx2[valid2]]

        cand_pos = {}
        for p in range(n_c):
            cand_pos[int(all_cands[p])] = p

        best_improve = 0.0
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            hi_val = float(h[i])
            if hi_val < 1e-12:
                continue
            ci = cand_pos[int(i)]
            for j in low_set:
                if int(i) == int(j):
                    continue
                hj_val = float(h[j])
                delta_max = min(hi_val, 1.0 - hj_val)
                if delta_max < 1e-12:
                    continue
                cj = cand_pos[int(j)]
                g = profiles[ci] - profiles[cj]

                lo = 0.0
                hi_d = delta_max
                for _ in range(15):
                    mid = 0.5 * (lo + hi_d)
                    vals = corr + mid * g
                    f_mid = float(np.max(vals))
                    active = vals >= (f_mid - 1e-14)
                    if not np.any(active):
                        break
                    f_prime = float(np.max(g[active]))
                    if f_prime > 1e-14:
                        hi_d = mid
                    else:
                        lo = mid

                delta_star = 0.5 * (lo + hi_d)
                if delta_star < 1e-10:
                    continue

                new_max = float(np.max(corr + delta_star * g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)
                    best_delta = delta_star

        if best_i >= 0 and best_improve > 1e-13:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def multi_shift_subgrad(h_init, n, target, time_limit, max_steps, n_shifts=3, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_now = cur_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_shifts:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - cur_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, si in zip(w, top_idx):
            k = int(si) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            if np.any(v1):
                grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            if np.any(v2):
                grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 500:
            break

    return best_h, best_c5


def discrete_cd(h_init, n, target, time_limit, max_sweeps=20000):
    start = time.time()
    dx = 2.0 / n
    h = np.round(h_init)
    h = project_feasible(h, target)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(np.max(corr))
        c5_val = cur_max * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

        k_star = int(np.argmax(corr)) - (n - 1)
        idxs = np.arange(n)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        if np.any(v1):
            grad[v1] += (1.0 - h[idxs[v1] - k_star])
        if np.any(v2):
            grad[v2] -= h[idxs[v2] + k_star]

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        grad_ones = grad[ones]
        grad_zeros = grad[zeros]
        i_cands = ones[np.argsort(-grad_ones)[:30]]
        j_cands = zeros[np.argsort(grad_zeros)[:30]]

        best_improve = 0.0
        best_i = -1
        best_j = -1

        for i in i_cands:
            for j in j_cands:
                if int(i) == int(j):
                    continue
                ks = np.arange(-(n - 1), n)
                g = np.zeros(n_corr)
                idx_ik = int(i) - ks
                valid_ik = (idx_ik >= 0) & (idx_ik < n)
                idx_iak = int(i) + ks
                valid_iak = (idx_iak >= 0) & (idx_iak < n)
                idx_jk = int(j) - ks
                valid_jk = (idx_jk >= 0) & (idx_jk < n)
                idx_jak = int(j) + ks
                valid_jak = (idx_jak >= 0) & (idx_jak < n)
                if np.any(valid_ik):
                    g[valid_ik] += h[idx_ik[valid_ik]]
                if np.any(valid_iak):
                    g[valid_iak] += h[idx_iak[valid_iak]]
                if np.any(valid_jk):
                    g[valid_jk] -= h[idx_jk[valid_jk]]
                if np.any(valid_jak):
                    g[valid_jak] -= h[idx_jak[valid_jak]]

                new_max = float(np.max(corr + g))
                improve = cur_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_i = int(i)
                    best_j = int(j)

        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP from initial
    if tleft() > 80:
        hp, sc = slsqp_round(best_h, n, target, [100, 500, 2000, 10000, 50000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 2: Pair CD (main workhorse)
    if tleft() > 150:
        cd_time = min(350, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=25, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 3: SLSQP polish
    if tleft() > 60:
        hp, sc = slsqp_round(best_h, n, target, [2000, 10000, 50000, 200000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 4: Pair CD with larger candidate set
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        hcd, c5cd = fast_pair_cd(best_h, n, target, time_limit=cd_time, n_cand=40, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 5: Multi-shift subgradient
    if tleft() > 60:
        hsg, c5sg = multi_shift_subgrad(best_h, n, target, time_limit=min(40, tleft() * 0.3),
                                        max_steps=50000, n_shifts=3, base_lr=0.02)
        if c5sg < best_score:
            best_score = c5sg
            best_h = hsg

    # Phase 6: SLSQP
    if tleft() > 50:
        hp, sc = slsqp_round(best_h, n, target, [5000, 20000, 100000], 1500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 7: Try higher resolution n=600
    if tleft() > 300:
        n_new = 600
        target_new = n_new / 2.0
        h_interp = interpolate_to(best_h, n_new, target_new)
        hp, sc = slsqp_round(h_interp, n_new, target_new, [100, 500, 2000, 10000], 1500, 1e-15)
        if tleft() > 150:
            hcd, c5cd = fast_pair_cd(hp, n_new, target_new, time_limit=min(150, tleft() * 0.4),
                                     n_cand=25, max_sweeps=50000)
            if c5cd < best_score:
                best_score = c5cd
                best_h = hcd
                n = n_new
                target = target_new
        elif sc < best_score:
            best_score = sc
            best_h = hp
            n = n_new
            target = target_new

    # Phase 8: More pair CD at current resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(120, tleft() * 0.4)
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=cd_time, n_cand=20, max_sweeps=100000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 9: SLSQP with very high alpha
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [50000, 200000, 500000], 1000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Phase 10: Final pair CD
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hcd, c5cd = fast_pair_cd(best_h, n_use, target_use, time_limit=min(35, tleft() * 0.5),
                                 n_cand=30, max_sweeps=50000)
        if c5cd < best_score:
            best_score = c5cd
            best_h = hcd

    # Phase 11: Final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        hp, sc = slsqp_round(best_h, n_use, target_use, [100000, 500000, 1000000], 500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = hp

    # Final feasibility and scoring
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
