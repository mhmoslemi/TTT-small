        h_pert = project_feasible(h_pert, target)
        per_time = min(40, tleft() * 0.15)
        h_ref, sc_ref = slsqp_round(h_pert, n, target,
                                    [100, 500, 2000], 800, 1e-14, per_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 3: Main SLSQP with increasing alpha (the workhorse)
    if tleft() > 150:
        alpha_schedule = [200, 500, 1000, 2000, 5000, 10000]
        for i, alpha in enumerate(alpha_schedule):
            if tleft() < 25:
                break
            remaining = tleft() / (len(alpha_schedule) - i)
            maxiter = int(min(3000, remaining * 2.5))
            maxiter = max(maxiter, 200)
            h_new, sc = slsqp_round(best_h, n, target,
                                    [alpha], maxiter, 1e-15, remaining * 0.85)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Projected subgradient with multi-active gradient
    if tleft() > 50:
        h_sg, sc_sg = fast_projected_grad(best_h, n, target,
                                          min(40, tleft() * 0.4), base_lr=0.05)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 5: High-alpha SLSQP
    if tleft() > 100:
        alpha_schedule2 = [10000, 30000, 100000]
        for alpha in alpha_schedule2:
            if tleft() < 25:
                break
            maxiter = int(min(2000, tleft() * 1.8))
            maxiter = max(maxiter, 100)
            h_new, sc = slsqp_round(best_h, n, target,
                                    [alpha], maxiter, 1e-16, tleft() * 0.3)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: LP polish
    if tleft() > 60:
        h_lp, sc_lp = alternating_lp(best_h, n, target, 3000,
                                     min(50, tleft() * 0.5),
                                     damping=0.9, reg=1e-5, n_active=30)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 7: SLSQP after LP
    if tleft() > 50:
        h_new, sc = slsqp_round(best_h, n, target,
                                [5000, 20000], 1000, 1e-16, tleft() * 0.5)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final subgradient
    if tleft() > 20:
        h_sg, sc_sg = projected_subgradient(best_h, n, target,
                                            min(15, tleft() * 0.6), base_lr=0.02)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 9: Tiny perturbation + SLSQP (escape last local min)
    if tleft() > 30:
        for eps in [0.005, 0.01, 0.02]:
            if tleft() < 15:
                break
            h_pert = best_h + rng.randn(n) * eps
            h_pert = project_feasible(h_pert, target)
            h_new, sc = slsqp_round(h_pert, n, target,
                                    [10000, 50000], 500, 1e-16, tleft() * 0.3)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Final
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)
    return best_h, best_score, n
