def binary_search_lp(h_current, n, target, time_limit, n_top_lags=200):
    start = time.time()
    h = h_current.copy()
    t_target = compute_c5(h)  # initial upper bound
    
    # We'll do a few rounds of:
    # 1. Fix g = 1-h
    # 2. Solve LP to find h' with overlaps ≤ t
    # 3. Update h = damping * h + (1-damping) * h'
    # 4. Decrease t
    
    t = t_target
    damping = 0.7
    
    while time.time() - start < time_limit and t > 0.3:
        g = 1.0 - h
        dx = 2.0 / n
        
        # Select top lags
        corr = np.correlate(h, g, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top_lags]
        
        # Build LP
        # Variables: h' (n dims)
        # Constraints: for each top lag s, sum h'[i] * g[i+s] ≤ t/dx
        # 0 ≤ h' ≤ 1
        # sum(h') = target
        
        h_var = cp.Variable(n)
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        
        for idx in top_idx:
            s = idx - (n - 1)
            # overlap_s = sum_{valid i} h'[i+s] * g[i]
            # = sum_{valid i} h'[i+s] * g[i]
            # Let j = i+s, then i = j-s
            # = sum_{valid j} h'[j] * g[j-s]
            # valid j: 0≤j<n and 0≤j-s<n
            lo = max(0, s)
            hi = min(n, n + s)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-s:hi-s]
                constraints.append(coeff @ h_var <= t / dx)
        
        prob = cp.Problem(cp.Minimize(cp.sum_squares(h_var - h)), constraints)
        prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h = damping * h + (1-damping) * h_new
        h = project_feasible(h, target)
        
        new_t = compute_c5(h)
        if new_t < t:
            t = new_t
            damping = max(0.3, damping * 0.95)
    
    return h, compute_c5(h)

        h_var = cp.Variable(n)
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for idx in top_idx:
            s = idx - (n - 1)
            lo = max(0, s)
            hi = min(n, n + s)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - s:hi - s] * dx
                constraints.append(coeff @ h_var <= compute_c5(h))
        objective = cp.max([constraints[-len(top_idx):][k].value for k in range(1)]) if False else None
        t_var = cp.Variable()
        for idx in top_idx:
            s = idx - (n - 1)
            lo = max(0, s)
            hi = min(n, n + s)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - s:hi - s] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter=1000, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def shake_and_restore(h, n, target, rng, noise_level, n_shake):
    h_shake = h.copy()
    for _ in range(n_shake):
        mask = rng.random(n) < 0.3
        if mask.sum() > 0:
            h_shake[mask] += rng.randn(mask.sum()) * noise_level
        h_shake = project_feasible(h_shake, target)
    return h_shake


def generate_structured_init(n, rng, style):
    h = np.zeros(n)
    if style == 0:
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0
    elif style == 1:
        h = np.tile(np.array([1.0, 0.0]), n // 2)
        if n % 2:
            h = np.concatenate([h, [1.0]])
        perm = rng.permutation(n)
        h_new = np.zeros(n)
        h_new[perm] = h
        h = h_new
    elif style == 2:
        third = n // 3
        h[:third] = 1.0
        h[2 * third:] = 1.0
        if n - 2 * third < 1:
            h[-1] = 0.0
        h += rng.randn(n) * 0.05
    elif style == 3:
        t = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.sin(2.0 * np.pi * t * rng.randint(2, 5))
    else:
        fifth = n // 5
        for i in range(0, n, 2 * fifth):
            h[i:i + fifth] = 1.0
        h += rng.randn(n) * 0.05
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial (main method)
    if tleft() > 150:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = coordinate_descent(best_h, n, target, cd_time,
                                         lam_lr=0.08, rng=np.random.RandomState(seed + 1))
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: LP refinement
    if tleft() > 100:
        lp_time = min(150, tleft() * 0.2)
        h_lp, c5_lp = lp_refined(best_h, n, target, max_iter=5000,
                                 time_limit=lp_time, n_top=250,
                                 damping_start=0.9, damping_end=0.75,
                                 reg_start=1e-4, reg_end=1e-9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Coordinate descent from LP result
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.25)
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, target, cd_time,
                                           lam_lr=0.05, rng=np.random.RandomState(seed + 2))
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 4: Multi-start with structured initializations
    for trial in range(8):
        if tleft() < 80:
            break
        style = trial % 5
        h_struct = generate_structured_init(n, rng, style)
        t_trial = min(50, max(20, tleft() / (8 - trial)))
        h_t, c5_t = coordinate_descent(h_struct, n, target, t_trial,
                                       lam_lr=0.1, rng=np.random.RandomState(seed + 100 + trial))
        if c5_t < best_score:
            best_score = c5_t
            best_h = h_t
            # Continue with LP from this new best
            if tleft() > 60:
                h_lp2, c5_lp2 = lp_refined(best_h, n, target, max_iter=2000,
                                           time_limit=min(40, tleft() * 0.5),
                                           n_top=200, damping_start=0.85,
                                           damping_end=0.7, reg_start=1e-3, reg_end=1e-7)
                if c5_lp2 < best_score:
                    best_score = c5_lp2
                    best_h = h_lp2

    # Phase 5: SLSQP polish
    if tleft() > 80:
        h_ref = slsqp_optimize(best_h, n, target,
                               [100, 500, 2000, 10000, 50000],
                               maxiter=2000, ftol=1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Shake + coordinate descent
    for attempt in range(4):
        if tleft() < 70:
            break
        noise = 0.05 + attempt * 0.03
        h_shake = shake_and_restore(best_h, n, target, rng, noise, n_shake=5)
        t_sh = min(50, max(20, tleft() / (4 - attempt)))
        h_sh, c5_sh = coordinate_descent(h_shake, n, target, t_sh,
                                         lam_lr=0.08, rng=np.random.RandomState(seed + 500 + attempt))
        if c5_sh < best_score:
            best_score = c5_sh
            best_h = h_sh

    # Phase 7: Final coordinate descent (long, from best)
    if tleft() > 60:
        cd_time = min(100, tleft() * 0.4)
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, target, cd_time,
                                           lam_lr=0.03, rng=np.random.RandomState(seed + 999))
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3

    # Phase 8: Try higher resolution
    if tleft() > 200:
        for n_high in [500, 600]:
            if tleft() < 120:
                break
            target_h = n_high / 2.0
            h_hi = interpolate_to(best_h, n_high, target_h)
            h_hi, c5_hi = coordinate_descent(h_hi, n_high, target_h,
                                             min(60, tleft() * 0.3),
                                             lam_lr=0.05, rng=np.random.RandomState(seed + 2000))
            if tleft() > 40:
                h_hi2, c5_hi2 = lp_refined(h_hi, n_high, target_h,
                                           max_iter=2000, time_limit=min(40, tleft() * 0.5),
                                           n_top=200, damping_start=0.85,
                                           damping_end=0.7, reg_start=1e-4, reg_end=1e-8)
                if c5_hi2 < c5_hi:
                    h_hi, c5_hi = h_hi2, c5_hi2
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_hi
                n = n_high
                target = n_high / 2.0

    # Phase 9: Final LP polish
    if tleft() > 50:
        h_lp3, c5_lp3 = lp_refined(best_h, n, target, max_iter=2000,
                                   time_limit=min(40, tleft() * 0.6),
                                   n_top=300, damping_start=0.95,
                                   damping_end=0.9, reg_start=1e-6, reg_end=1e-10)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 10: Final coordinate descent polish
    if tleft() > 30:
        h_cd4, c5_cd4 = coordinate_descent(best_h, n, target,
                                           min(25, tleft() * 0.5),
                                           lam_lr=0.02, rng=np.random.RandomState(seed + 7777))
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
