def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    return float(np.max(corr)) * dx

def random_flip_move(h):
    ones = np.arange(len(h))[h==1]  # Indices of ones
    zeros = np.arange(len(h))[h==0]  # Indices of zeros
