def compute_swap_delta(h, i, j, n):
    """Compute change in overlap array when h[i]:1->0 and h[j]:0->1."""
    # For each lag, the affected positions m are: i, i-lag, j, j-lag
    # (deduplicated, and within valid range)
    # overlap index: m maps to correlation index (n-1) + lag
    delta = np.zeros(2*n-1)
    
    for lag in range(-(n-1), n):
        # Valid range for m at this lag
        if lag >= 0:
            m_lo, m_hi = 0, n - 1 - lag
        else:
            m_lo, m_hi = -lag, n - 1
        
        # Affected m values
        affected = set()
        for m_candidate in [i, i - lag, j, j - lag]:
            if m_lo <= m_candidate <= m_hi:
                affected.add(m_candidate)
        
        if not affected:
            continue
        
        corr_idx = (n - 1) + lag
        for m in affected:
            m_plus = m + lag
            if 0 <= m_plus < n:
                # Before: h[m] * (1 - h[m_plus])
                before = h[m] * (1.0 - h[m_plus])
                # After: h'[m] * (1 - h'[m_plus])
                hm = 0.0 if m == i else (1.0 if m == j else h[m])
                hmp = 0.0 if m_plus == i else (1.0 if m_plus == j else h[m_plus])
                after = hm * (1.0 - hmp)
                delta[corr_idx] += after - before
    
    return delta

_g_target = 0.0


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def compute_c5(h):
    return float(np.max(compute_overlaps(h)))


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _cons_fun(h):
    return np.sum(h) - _g_target


def _cons_jac(h):
    return np.ones(_g_n)


def _smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def slsqp_polish(h0, n, target, alphas, maxiter_per_alpha=500):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h0.copy(), target)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def to_binary(h, target, n):
    h_bin = (h >= 0.5).astype(float)
    diff = int(round(target - h_bin.sum()))
    if diff > 0:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        h_bin[order[:diff]] = 1.0
    elif diff < 0:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        h_bin[order[:(-diff)]] = 0.0
    return h_bin

def lp_refine(h_init, n, target, time_limit, n_top=80, damping=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1

    while time.time() - start < time_limit:

        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_top]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for ti in top_idx:
            k = int(ti - n_half)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        reg = 1e-5
        prob = cp.Problem(cp.Minimize(t_var + reg * cp.sum_squares(h_var - h)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
