def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_A(h):
    """Compute A[i, m] = h[i + shifts[m]] + h[i - shifts[m]] with out-of-bounds = 0"""
    n = len(h)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)
    A = np.zeros((n, n_corr))
    for i in range(n):
        a_idx = i + shifts
        b_idx = i - shifts
        valid_a = (a_idx >= 0) & (a_idx < n)
        valid_b = (b_idx >= 0) & (b_idx < n)
        A[i, valid_a] = h[a_idx[valid_a]]
        A[i, valid_b] += h[b_idx[valid_b]]
    return A

        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target

    return np.ones(_global_n)

    _global_n = n
    _global_target = target

            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

def compute_B(h):
    n = len(h)
    n_corr = 2 * n - 1
    shifts = np.arange(-(n-1), n)

    a = i_idx + shifts[None, :]  # (n, n_corr)
    b = i_idx - shifts[None, :]  # (n, n_corr)
    va = (a >= 0) & (a < n)  # (n, n_corr)
    vb = (b >= 0) & (b < n)
    # h[a] where valid, else 0
    ha = np.where(va, h[np.clip(a, 0, n-1)], 0.0)
    hb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
    B = (ha - va.astype(float)) + hb
    return B

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h):

def vectorized_pair_cd(h_init, n, target, time_limit, max_sweeps=50000, n_cand=40, n_deltas=8, escape_iter=100):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = compute_B(h)
    no_improve = 0
    idxs = np.arange(n)
    rng_local = np.random.RandomState(777)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > escape_iter:
                n_pert = int(rng_local.choice([3, 5, 8]))
                pert_idx = rng_local.choice(n, size=min(n_pert, n), replace=False)
                h[pert_idx] += rng_local.randn(min(n_pert, n)) * 0.15
                h = project_feasible(h, target)
                corr = np.correlate(h, 1.0 - h, mode='full')
                B = compute_B(h)
                no_improve = 0
                continue
            if no_improve > escape_iter * 3:
                break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        current_max = float(np.max(corr))
        best_score = current_max
        best_i = -1
        best_j = -1
        best_delta = 0.0

        delta_vals = np.linspace(0.01, 0.35, n_deltas)

        for i in low_set:
            max_d = min(h[i], float(np.min(1.0 - h[high_set])))
            if max_d < 0.005:
                continue
            feasible_deltas = delta_vals[delta_vals <= max_d]
            if len(feasible_deltas) == 0:
                continue
            for delta in feasible_deltas:
                base = corr + delta * B[i]
                new_corrs = base[None, :] - delta * B[high_set, :]
                scores = np.max(new_corrs, axis=1)
                bj = int(np.argmin(scores))
                sb = float(scores[bj])
                if sb < best_score:
                    best_score = sb
                    best_i = i
                    best_j = int(high_set[bj])
                    best_delta = delta

        if best_i >= 0 and best_delta > 1e-10:
            g = B[best_i] - B[best_j]
            d_lo = 0.0
            d_hi = min(h[best_i], 1.0 - h[best_j])
            for _ in range(25):
                d_mid = 0.5 * (d_lo + d_hi)
                vals = corr + d_mid * g
                f_mid = float(np.max(vals))
                active = vals >= f_mid - 1e-14
                f_prime = float(np.max(g[active]))
                if f_prime > 1e-12:
                    d_hi = d_mid
                else:
                    d_lo = d_mid
            best_delta = 0.5 * (d_lo + d_hi)

            if best_delta > 1e-10:
                h[best_i] -= best_delta
                h[best_j] += best_delta
                h = np.clip(h, 0.0, 1.0)
                diff = target - np.sum(h)
                if abs(diff) > 1e-12:
                    mask = (h > 1e-10) & (h < 1.0 - 1e-10)
                    if mask.sum() > 0:
                        h[mask] += diff / mask.sum()
                        h = np.clip(h, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                B = compute_B(h)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_search(h_best, n, target, time_limit, n_starts=6, seed=42):
    start = time.time()
    rng = np.random.RandomState(seed)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)

    for s in range(n_starts):
        if time.time() - start > time_limit:
            break
        scale = 0.05 + 0.15 * s / max(n_starts - 1, 1)
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        cd_time = max(5.0, (time_limit - (time.time() - start)) / (n_starts - s) * 0.7)
        h_try, c5_try = vectorized_pair_cd(h_pert, n, target,
                                           time_limit=cd_time,
                                           max_sweeps=5000, n_cand=40, n_deltas=6,
                                           escape_iter=80)
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try
            if time.time() - start < time_limit - 30:
                h_pol = slsqp_round(best_h, n, target,
                                    [500, 2000, 8000], 500, 1e-15)
                sc = compute_c5(h_pol)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol

    return best_h, best_score

                                         max_sweeps=50000, n_cand=40, n_deltas=8,
                                         escape_iter=150)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [500, 2000, 10000, 50000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        cd_time = min(300, tleft() * 0.4)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=cd_time,
                                         max_sweeps=50000, n_cand=60, n_deltas=10,
                                         escape_iter=120)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        ms_time = min(250, tleft() * 0.4)
        h_ms, c5_ms = multi_start_search(best_h, n, target,
                                         time_limit=ms_time,
                                         n_starts=8, seed=seed)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 200:
        for n_try in [500, 600]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_try, c5_try = vectorized_pair_cd(h_interp, n_try, target_try,
                                               time_limit=min(80, tleft() * 0.4),
                                               max_sweeps=10000, n_cand=40, n_deltas=8,
                                               escape_iter=100)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_try
                n = n_try
                target = target_try

    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = vectorized_pair_cd(best_h, n_use, target_use,
                                         time_limit=min(60, tleft() * 0.5),
                                         max_sweeps=30000, n_cand=50, n_deltas=8,
                                         escape_iter=80)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000], 500, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
