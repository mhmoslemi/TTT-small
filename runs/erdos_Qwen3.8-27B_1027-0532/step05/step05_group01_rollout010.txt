import numpy as np
from scipy.optimize import minimize

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_obj(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha)) * dx


def smooth_max_grad(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    n = len(h)
    idxs = np.arange(n)
    mid = n - 1
    grad = np.zeros(n)
    # For each correlation value corr[k], the gradient w.r.t. h[i] is:
    # corr[k] = sum_j h[j] * (1 - h[j+k]) (with appropriate bounds)
    # d(corr[k])/d(h[i]) = sum_j [delta_{i,j}(1-h[j+k]) - delta_{i,j+k} * h[j]]
    #                     = (1 - h[i+k]) - h[i-k] (with bounds checking)
    for s_idx in range(len(corr)):
        if weights[s_idx] < 1e-15:
            continue
        k = s_idx - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += weights[s_idx] * (1.0 - h[idxs[v1] - k])
        grad[v2] -= weights[s_idx] * h[idxs[v2] + k]
    grad *= dx
    return grad


def frank_wolfe_step(grad, target_sum):
    n = len(grad)
    # min <grad, s> subject to 0 <= s <= 1, sum(s) = target_sum
    # Optimal: s[i] = 1 for target_sum indices with smallest grad[i]
    # (target_sum should be an integer n/2)
    s = np.zeros(n)
    k = int(round(target_sum))
    if k <= 0 or k >= n:
        # If not integer, need to handle fractional
        s[:] = target_sum / n
    else:
        sorted_idx = np.argsort(grad)
        s[sorted_idx[:k]] = 1.0
    return s


def frank_wolfe_alpha(h, start_alpha, dx, time_limit, max_steps):
    """Frank-Wolfe method with smooth max."""
    start = time.time()
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    alpha = start_alpha
    t = 1.0  # step size parameter
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        obj = smooth_max_obj(h_cur, alpha, dx)
        grad = smooth_max_grad(h_cur, alpha, dx)
        
        # Frank-Wolfe step
        s = frank_wolfe_step(grad, target)
        
        # Line search along h_cur to s
        best_t = 0.0
        best_val = obj
        candidates = np.array([t, 2.0/(step+2), 0.5*t, 0.25*t, 0.1*t, 0.05*t])
        candidates = np.clip(candidates, 0.0, 1.0)
        for tc in candidates:
            h_try = (1.0 - tc) * h_cur + tc * s
            h_try = project_feasible(h_try, target)
            val = smooth_max_obj(h_try, alpha, dx)
            if val < best_val:
                best_val = val
                best_t = tc
            elif val > best_val + 1e-10:
                break
        
        h_cur = (1.0 - best_t) * h_cur + best_t * s
        h_cur = project_feasible(h_cur, target)
        
        actual = compute_c5(h_cur)
        if actual < best_score - 1e-13:
            best_score = actual
            best_h = h_cur.copy()
        
        # Adapt alpha
        if step % 50 == 49:
            alpha = min(alpha * 1.5, 100000.0)
        
        # Gap check
        gap = np.dot(grad, h_cur - s)
        if gap < 1e-12:
            break
    
    return best_h, best_score


def mirrored_descent(h, alpha, dx, time_limit, max_steps, mu=0.1):
    """Mirror descent with entropic regularization."""
    start = time.time()
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        grad = smooth_max_grad(h_cur, alpha, dx)
        # Mirror step: h_new[i] ∝ h_cur[i] * exp(-mu * grad[i])
        # But we need 0 <= h_new <= 1, sum = target
        # Use exponentiated gradient
        log_h = np.log(np.maximum(h_cur, 1e-30)) - mu * grad
        log_h = log_h - np.max(log_h)  # numerical stability
        h_new = np.exp(log_h)
        h_new = np.clip(h_new, 1e-15, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Step size adaptation
        actual = compute_c5(h_new)
        if actual < best_score - 1e-13:
            best_score = actual
            best_h = h_new.copy()
            h_cur = h_new
        else:
            # Halve the step
            mu = max(mu * 0.5, 0.001)
            h_try = h_cur.copy()
            log_h = np.log(np.maximum(h_cur, 1e-30)) - mu * grad
            log_h = log_h - np.max(log_h)
            h_try = np.exp(log_h)
            h_try = np.clip(h_try, 1e-15, 1.0)
            h_try = project_feasible(h_try, target)
            if compute_c5(h_try) < compute_c5(h_cur):
                h_cur = h_try
    
    return best_h, best_score

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(_global_n)
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0/_global_n)
    
    def smooth_grad(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights = weights / np.sum(weights)
        idxs = np.arange(_global_n)
        mid = _global_n - 1
        grad = np.zeros(_global_n)
        for s_idx in range(len(corr)):
            if weights[s_idx] < 1e-15:
                continue
            k = s_idx - mid
            v1 = (idxs - k >= 0) & (idxs - k < _global_n)
            v2 = (idxs + k >= 0) & (idxs + k < _global_n)
            grad[v1] += weights[s_idx] * (1.0 - h[idxs[v1] - k])
            grad[v2] -= weights[s_idx] * h[idxs[v2] + k]
        grad *= (2.0 / _global_n)
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
        else:
            if consecutive_no_improve > 80:
                break
    return best_h, best_c5


def multi_pair_cd(h_start, n, target, time_limit, max_sweeps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        if consecutive_no_improve > 150:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        n_cand = 30
        sorted_idx = np.argsort(grad)
        low_list = sorted_idx[:n_cand].tolist()
        high_list = sorted_idx[-n_cand:].tolist()
        used = set()
        pairs = []
        for i in high_list:
            if i in used:
                continue
            for j in low_list:
                if j == i or j in used:
                    continue
                pairs.append((i, j))
                used.add(i)
                used.add(j)
                break
            if len(pairs) >= 5:
                break
        if len(pairs) == 0:
            continue
        deltas = []
        for (i, j) in pairs:
            delta_lo = max(h[i] - 1.0, -h[j])
            delta_hi = min(h[i], 1.0 - h[j])
            if delta_hi - delta_lo < 1e-12:
                deltas.append(0.0)
                continue
            g = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            g[v_ip] += h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g[v_im] += h[i - shifts[v_im]]
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            g[v_jp] -= h[j + shifts[v_jp]]
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g[v_jm] -= h[j - shifts[v_jm]]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(15):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            deltas.append(0.5 * (a_d + b_d))
        h_try = h.copy()
        for (i, j), d in zip(pairs, deltas):
            if abs(d) > 1e-14:
                h_try[i] -= d
                h_try[j] += d
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val - 1e-14:
            h = h_try
        else:
            h_try2 = h.copy()
            for (i, j), d in zip(pairs, deltas):
                if abs(d) > 1e-14:
                    h_try2[i] -= 0.5 * d
                    h_try2[j] += 0.5 * d
            h_try2 = np.clip(h_try2, 0.0, 1.0)
            h_try2 = project_feasible(h_try2, target)
            if compute_c5(h_try2) < c5_val - 1e-14:
                h = h_try2
    return best_h, best_c5


def annealing_cd(h_start, n, target, time_limit, max_sweeps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    shifts = np.arange(-(n - 1), n)
    rng = np.random.RandomState(77777)
    T = 0.005

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        T *= 0.997
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        n_cand = 35
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        i = int(rng.choice(high_set))
        j = int(rng.choice(low_set))
        if i == j:
            continue
        delta_lo = max(h[i] - 1.0, -h[j])
        delta_hi = min(h[i], 1.0 - h[j])
        if delta_hi - delta_lo < 1e-12:
            continue
        if rng.random() < 0.6:
            g = np.zeros(n_corr)
            v_ip = (i + shifts >= 0) & (i + shifts < n)
            g[v_ip] += h[i + shifts[v_ip]]
            v_im = (i - shifts >= 0) & (i - shifts < n)
            g[v_im] += h[i - shifts[v_im]]
            v_jp = (j + shifts >= 0) & (j + shifts < n)
            g[v_jp] -= h[j + shifts[v_jp]]
            v_jm = (j - shifts >= 0) & (j - shifts < n)
            g[v_jm] -= h[j - shifts[v_jm]]
            a_d = delta_lo
            b_d = delta_hi
            for _ in range(15):
                mid_d = 0.5 * (a_d + b_d)
                vals = corr + mid_d * g
                f_mid = np.max(vals)
                active_mask = vals >= f_mid - 1e-14
                if active_mask.any():
                    f_prime = np.max(g[active_mask])
                else:
                    f_prime = 0.0
                if f_prime > 1e-14:
                    b_d = mid_d
                else:
                    a_d = mid_d
            delta = 0.5 * (a_d + b_d)
        else:
            delta = float(rng.uniform(delta_lo, delta_hi))
        if abs(delta) < 1e-12:
            continue
        h_try = h.copy()
        h_try[i] -= delta
        h_try[j] += delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < c5_val:
            h = h_try
        elif T > 1e-9 and rng.random() < np.exp((c5_val - c5_try) / max(T, 1e-12)):
            h = h_try
    return best_h, best_c5


def momentum_subgradient(h_start, n, target, time_limit, max_steps, mu, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_dx
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    _global_n = n
    _global_target = target
    _global_dx = 2.0 / n

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Frank-Wolfe global optimization
    if tleft() > 180:
        fw_time = min(200, tleft() * 0.25)
        h_fw, c5_fw = frank_wolfe_optimize(best_h, start_alpha=200.0,
                                           time_limit=fw_time, max_steps=5000)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw

    # Phase 2: Pair coordinate descent
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_optimize(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=5000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 150:
        h_pol = slsqp_round(best_h, n, target,
                            [500, 2000, 10000, 50000],
                            1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Multi-pair CD
    if tleft() > 130:
        mp_time = min(180, tleft() * 0.3)
        h_mp, c5_mp = multi_pair_cd(best_h, n, target,
                                    time_limit=mp_time, max_sweeps=3000)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    # Phase 5: Annealing CD for escaping local optima
    if tleft() > 100:
        an_time = min(120, tleft() * 0.25)
        h_an, c5_an = annealing_cd(best_h, n, target,
                                   time_limit=an_time, max_sweeps=3000)
        if c5_an < best_score:
            best_score = c5_an
            best_h = h_an

    # Phase 6: SLSQP after multi-pair and annealing
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000],
                            1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 7: Momentum subgradient
    if tleft() > 80:
        mom_time = min(60, tleft() * 0.3)
        h_mom, c5_mom = momentum_subgradient(best_h, n, target,
                                             time_limit=mom_time,
                                             max_steps=20000, mu=0.85,
                                             base_lr=0.02)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 8: Second round of pair CD + SLSQP
    if tleft() > 120:
        cd_time = min(150, tleft() * 0.3)
        h_cd, c5_cd = pair_cd_optimize(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=4000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
        if tleft() > 60:
            h_pol = slsqp_round(best_h, n, target,
                                [5000, 20000, 80000],
                                1000, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 9: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.01, 0.03, 0.06, 0.1, 0.15]:
        if tleft() < 70:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        if tleft() > 70:
            h_fw2, c5_fw2 = frank_wolfe_optimize(h_pert, start_alpha=500.0,
                                                  time_limit=min(30, tleft() * 0.2),
                                                  max_steps=1500)
            if c5_fw2 < compute_c5(h_pert):
                h_pert = h_fw2
        if tleft() > 35:
            h_cd2, c5_cd2 = pair_cd_optimize(h_pert, n, target,
                                             time_limit=min(25, tleft() * 0.3),
                                             max_sweeps=1500, n_cand=40)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    # Phase 10: Final pair CD at best
    if tleft() > 60:
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = pair_cd_optimize(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=2000, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        h_final = slsqp_round(best_h, n, target,
                              [5000, 20000, 100000],
                              800, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
