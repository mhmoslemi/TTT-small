def binary_greedy_search(h_bin, n, target, time_limit, max_rounds=100):
    """Greedy pair swap local search on binary vectors."""
    start = time.time()
    h = h_bin.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for round_num in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star_idx = np.argmax(corr)
        k = k_star_idx - (n - 1)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0
        best_swap = None
        
        for i in ones:
            for j in zeros:
                # Compute change in corr at lag k
                delta = 0
                # m = i: change is h[i+k] (or 1 if i+k==j)
                if 0 <= i+k < n:
                    delta += 1 if (i+k == j) else h[i+k]
                # m = i-k: change is -(1-h[i-k]) = h[i-k]-1 (if i-k != j)
                if 0 <= i-k < n and i-k != i:
                    if i-k == j:
                        delta += 0 - 1  # old was h[i]*(1-h[j])=1*1=1, new is 0*(1-1)=0
                    else:
                        delta += 0 - (1 - h[i-k])
                # m = j: change is -h[j+k] (or -1 if j+k==i)
                if 0 <= j+k < n:
                    delta -= 1 if (j+k == i) else h[j+k]
                # m = j-k: change is (1-h[j-k]) (if j-k != i)
                if 0 <= j-k < n and j-k != j:
                    if j-k == i:
                        delta += 0 - 0  # old was 0*(1-h[i])=0, new is 1*(1-1)=0
                    else:
                        delta += (1 - h[j-k]) - 0
                
                if delta < best_delta:
                    best_delta = delta
                    best_swap = (i, j)
        
        if best_swap is None:
            break
        
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        c5 = compute_c5(h)
        
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            # Try a few random swaps
            improved = False
            for _ in range(50):
                i_r = np.random.choice(ones)
                j_r = np.random.choice(zeros)
                h[i_r] = 0.0
                h[j_r] = 1.0

                    best_h = h.copy()
                    improved = True
                    break
                h[i_r] = 1.0
                h[j_r] = 0.0
            if not improved:
                break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def project_to_binary(h, target_sum):
    t = int(round(target_sum))
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros_like(h)
    h_bin[sorted_idx[:t]] = 1.0
    return h_bin

def lp_constraint_gen(h_init, n, target, time_limit, max_active=60, eps=1e-10, max_iters=20000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    reg = 1e-4
    half_n = n - 1

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max_raw = np.max(corr)
        threshold = 0.85 * c_max_raw
        active_mask = corr > threshold
        active_k = np.where(active_mask)[0]
        if len(active_k) > max_active:
            active_k = np.argsort(corr)[-max_active:]
        elif len(active_k) < 5:
            active_k = np.argsort(corr)[-max_active:]

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * (2.0 / n)
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=eps, max_iters=100000, verbose=False)
            if h_var.value is not None:
                solved = True
        except Exception:
            pass

        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                if h_var.value is not None:
                    solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        h = h_new

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

def smooth_obj(h, alpha):
    n = _global_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_polish(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    cur_score = compute_c5(h_cur)

        _global_alpha = alpha

            no_improve = 0
