import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_lse(h, alpha, n):
    """Analytical gradient of the LSE-smoothed max correlation."""
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    return grad * dx

def slsqp_lse(h, n, target, alpha, maxiter=2000, ftol=1e-16, deadline=None):
    dx = 2.0 / n
    def obj(x):
        corr = compute_corr(x)
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    def jac(x):
        return _grad_lse(x, alpha, n)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cons_fun, 'jac': cons_jac},)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def exp_weights_opt(h, n, target, n_outer, beta, n_inner, lr_init, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    
    for outer in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(n) / n
        else:
            weights /= wsum
        
        for inner in range(n_inner):
            if deadline is not None and time.time() > deadline:
                break
            grad = np.zeros(n)
            for k in range(n):
                wk = weights[k]
                if wk < 1e-12:
                    continue
                if k == 0:
                    grad += wk * (1.0 - 2.0 * h_cur)
                else:
                    grad[:n-k] += wk * (1.0 - h_cur[k:])
                    grad[k:] -= wk * h_cur[:n-k]
            grad *= 2.0 / n
            
            h_new = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
    
    return best_h, best_score

def subgradient_topm(h, n, target, n_steps, m, lr_init, deadline=None, temp=0.01):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        sorted_idx = np.argsort(over)[::-1]
        top_idx = sorted_idx[:m]
        top_vals = over[top_idx].astype(float)
        
        shifted = (top_vals - top_vals[0]) / temp
        wm = np.exp(shifted)
        wm /= wm.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(wm, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
        grad *= dx
        
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    
    return best_h, best_score

def de_search(n, target, pop_size, max_gen, rng, deadline=None, F=0.8, CR=0.9):
    population = [project_feasible(rng.rand(n), target) for _ in range(pop_size)]
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    
    for gen in range(max_gen):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            cands = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            j_rand = rng.randint(n)
            trial = population[i].copy()
            mask = rng.rand(n) < CR
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = project_feasible(trial, target)
            sc = compute_c5(trial)
            if sc <= scores[i]:
                population[i] = trial
                scores[i] = sc
                if sc < best_score:
                    best_score = sc
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_lse(h, alpha, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-14:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    return grad * dx

def _lse_obj(h, alpha, n):
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_lse(h, n, target, alpha, maxiter=2000, ftol=1e-16, deadline=None):
    def obj(x):
        return _lse_obj(x, alpha, n)
    def jac(x):
        return _grad_lse(x, alpha, n)
    def cfun(x):
        return np.sum(x) - target
    def cjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': cfun, 'jac': cjac},)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def _fast_grad_weighted(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
    return grad * dx

def exp_weights_opt(h, n, target, n_outer, beta, n_inner, lr_init, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for outer in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(n) / n
        else:
            weights /= wsum
        for inner in range(n_inner):
            if deadline is not None and time.time() > deadline:
                break
            grad = _fast_grad_weighted(h_cur, weights, n)
            h_new = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_score

def subgradient_topm(h, n, target, n_steps, m, lr_init, deadline=None, temp=0.01):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        sorted_idx = np.argsort(over)[::-1]
        top_idx = sorted_idx[:m]
        top_vals = over[top_idx].astype(float)
        shifted = (top_vals - top_vals[0]) / temp
        wm = np.exp(shifted)
        wm /= wm.sum()
        grad = np.zeros(n)
        for wk, k in zip(wm, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
        grad *= dx
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.01, 0.02)
        else:
            lr *= 0.9
            if lr < 1e-10:
                break
    return best_h, best_score

def momentum_ew(h, n, target, n_steps, beta, lr, momentum, deadline=None):

            weights = np.ones(n) / n
        else:
            weights /= wsum
        grad = _fast_grad_weighted(h_cur, weights, n)
        v = momentum * v - lr * grad
        h_new = project_feasible(h_cur + v, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
        else:
            v *= 0.5
    return best_h, best_score

    population = [project_feasible(rng.rand(n), target) for _ in range(pop_size)]
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_gen):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            cands = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            j_rand = rng.randint(n)
            trial = population[i].copy()
            mask = rng.rand(n) < CR
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = project_feasible(trial, target)
            sc = compute_c5(trial)
            if sc <= scores[i]:
                population[i] = trial
                scores[i] = sc
                if sc < best_score:
                    best_score = sc
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def perturb_restart(h, n, target, rng, scale=0.05):
    h_pert = h + scale * rng.randn(n)
