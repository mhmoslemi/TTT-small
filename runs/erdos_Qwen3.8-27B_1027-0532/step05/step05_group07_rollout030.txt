def fft_gradient_descent(h, n, target, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    # Precompute FFT of h_ext (will update each iteration)
    size = 1  # next power of 2 >= 4n-3
    while size < 4*n - 3:
        size *= 2
    
    alphas = [2000, 1000, 500, 200, 100, 50, 20, 10, 5, 3, 2, 1]
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        # For each alpha, do several iterations
        n_iter = 500  # adjust based on time
        for it in range(n_iter):
            if time.time() > deadline:
                break
            
            # Compute correlation
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            
            # Compute softmax weights
            corr_shifted = corr - corr.max()
            p = np.exp(alpha * corr_shifted)
            p_sum = p.sum()
            if p_sum < 1e-300:
                p = np.zeros(2*n-1)
                p[n-1] = 1.0
            else:
                p = p / p_sum
            
            # Compute gradient via FFT
            h_ext = np.zeros(2*n-1)
            h_ext[:n] = h_cur
            P_rev = p[::-1]
            
            H = np.fft.rfft(h_ext, size)
            conv1 = np.fft.irfft(H * np.fft.rfft(p, size), size)
            conv2 = np.fft.irfft(H * np.fft.rfft(P_rev, size), size)
            
            term1 = conv1[n-1:2*n-1]
            term2 = conv2[n-1:2*n-1]
            grad = 1.0 - term1 - term2
            
            # Project gradient: remove mean to maintain sum constraint
            grad -= grad.mean()
            
            # Backtracking line search
            step = 0.1
            for _ in range(20):
                h_trial = np.clip(h_cur - step * grad, 0, 1)
                h_trial = project_feasible(h_trial, target)
                sc_trial = compute_c5(h_trial)
                if sc_trial < best_score or sc_trial < compute_c5(h_cur):
                    h_cur = h_trial
                    if sc_trial < best_score:
                        best_score = sc_trial
                        best = h_cur.copy()
                    break
                step *= 0.5
            
            # If no improvement, try smaller step
            if step < 1e-10:
                break
        
    return best, best_score
