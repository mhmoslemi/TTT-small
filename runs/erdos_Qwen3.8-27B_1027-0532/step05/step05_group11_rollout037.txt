def project_feasible(h, target_sum):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, target):
    """Compute smooth log-sum-exp objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    shifted = alpha * (corr - max_corr)
    shifted = np.clip(shifted, -500.0, 0.0)

    val = max_corr + np.log(sum_exp) / alpha

    n_half = n - 1

    alpha, beta: arrays of length m
    Returns: (optimal_t, optimal_value)
    """
    m = len(alpha)

    best_t = 0.0

    val_1 = np.max(beta + alpha)
    if val_1 < best_val:

                if actual < best_val:
                    best_val = actual
                    best_t = t_star

def compute_alpha_for_i(h, i, dx, n):
    """Compute alpha_s for all lags s, for coordinate i."""
    n_half = n - 1
    # s ranges from -n_half to n_half
    # alpha_s = dx*(1-h[i-s])*1[0≤i-s<n] - dx*h[i+s]*1[0≤i+s<n]
    
    # For the first term: 0 ≤ i-s < n → i-n < s ≤ i, i.e., s ∈ [max(-n_half, i-n+1), min(n_half, i)]
    #   At those s: contribution = dx*(1-h[i-s])
    # For the second term: 0 ≤ i+s < n → -i ≤ s < n-i, i.e., s ∈ [max(-n_half, -i), min(n_half, n-1-i)]
    #   At those s: contribution = -dx*h[i+s]
    
    alpha = np.zeros(2*n-1)
    
    # First term
    s_lo1 = max(-n_half, i - n + 1)  # i-s >= 0 → s <= i
    s_hi1 = min(n_half, i)           # i-s < n → s > i-n
    if s_hi1 >= s_lo1:
        s_vals = np.arange(s_lo1, s_hi1 + 1)
        idx = i - s_vals  # indices into h: 0 to n-1
        alpha[s_vals + n_half] += dx * (1.0 - h[idx])
    
    # Second term
    s_lo2 = max(-n_half, -i)         # i+s >= 0 → s >= -i
    s_hi2 = min(n_half, n - 1 - i)   # i+s < n → s < n-i
    if s_hi2 >= s_lo2:
        s_vals = np.arange(s_lo2, s_hi2 + 1)
        idx = i + s_vals  # indices into h: 0 to n-1
        alpha[s_vals + n_half] -= dx * h[idx]
    
    return alpha

            break
        
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        progress = min(1.0, it / max(1, max_iter * 0.5))
        reg = reg_start * (reg_end / reg_start) ** progress
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=20000, eps=1e-7, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        progress2 = min(1.0, it / max(1, max_iter * 0.5))
        damp = damping_start + (damping_end - damping_start) * progress2
        h = (1.0 - damp) * h + damp * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_global_n = 0
_global_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_optimize(h_init, n, target, max_steps, time_limit, n_top=15, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        
        w = 0.8 ** np.arange(n_top)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(top_lags):
            ck = w[idx]
            if k >= 0:
                i1 = idxs[idxs + k < n]
                grad[i1] += ck * dx * (1.0 - y[i1 + k])
                i2 = idxs[idxs >= k]
                grad[i2] += ck * dx * (-y[i2 - k])
            else:
                i1 = idxs[idxs - k < n]
                grad[i1] += ck * dx * (1.0 - y[i1 + k])
                i2 = idxs[idxs + k >= 0]
                grad[i2] += ck * dx * (-y[i2 + k])
        
        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-15:
            break
        
        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_g ** 2 + 1e-20)
        lr = min(max(0.01, polyak_lr), 0.5)
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Coordinate descent (main workhorse)
    if tleft() > 250:
        h_cd, c5_cd = coordinate_descent(best_h, n, target,
                                         max_passes=500, time_limit=min(250, tleft()*0.6),
                                         damping=1.0, rng=rng, max_stale=30)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd
    
    # Phase 2: LP refinement
    if tleft() > 150:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=3000, time_limit=min(150, tleft()*0.5),
                                      n_top=200, damping_start=0.95, damping_end=0.85,
                                      reg_start=1e-4, reg_end=1e-8)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
    
    # Phase 3: More coordinate descent after LP
    if tleft() > 200:
        h_cd2, c5_cd2 = coordinate_descent(best_h, n, target,
                                            max_passes=500, time_limit=min(200, tleft()*0.6),
                                            damping=0.9, rng=rng, max_stale=25)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2
    
    # Phase 4: SLSQP smooth optimization
    if tleft() > 100:
        h_sl = slsqp_optimize(best_h, n, target,
                               [100, 500, 2000, 10000, 50000],
                               maxiter=1500, ftol=1e-15)
        sc = compute_c5(h_sl)
        if sc < best_score:
            best_score = sc
            best_h = h_sl
    
    # Phase 5: Subgradient
    if tleft() > 100:
        h_sg, c5_sg = subgradient_optimize(best_h, n, target,
                                            max_steps=30000, time_limit=min(100, tleft()*0.7),
                                            n_top=20, momentum=0.9)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
    
    # Phase 6: Final coordinate descent
    if tleft() > 150:
        h_cd3, c5_cd3 = coordinate_descent(best_h, n, target,
                                            max_passes=500, time_limit=min(150, tleft()*0.7),
                                            damping=1.0, rng=rng, max_stale=20)
        if c5_cd3 < best_score:
            best_score = c5_cd3
            best_h = h_cd3
    
    # Phase 7: LP polish
    if tleft() > 80:
        h_lp2, c5_lp2 = alternating_lp(best_h, n, target,
                                        max_iter=2000, time_limit=min(80, tleft()*0.7),
                                        n_top=250, damping_start=0.97, damping_end=0.9,
                                        reg_start=1e-5, reg_end=1e-10)
        if c5_lp2 > 0 and c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2
    
    # Phase 8: Final coordinate descent
    if tleft() > 60:
        h_cd4, c5_cd4 = coordinate_descent(best_h, n, target,
                                            max_passes=300, time_limit=min(60, tleft()*0.8),
                                            damping=1.0, rng=None, max_stale=15)
        if c5_cd4 < best_score:
            best_score = c5_cd4
            best_h = h_cd4
