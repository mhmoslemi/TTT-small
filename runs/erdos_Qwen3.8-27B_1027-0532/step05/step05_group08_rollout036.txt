import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slp_iteration(h, n, target, active_idx, max_iters=50):
    """SLP with cutting plane approach.
    Returns improved h or None if no improvement."""
    dx = 2.0 / n
    corr = compute_corr(h)
    current_max = float(np.max(corr))
    
    for slit in range(max_iters):
        m_active = len(active_idx)
        # Build LP: min T s.t. grad_m^T h - T <= b_m for active m
        #            0 <= h_i <= 1, sum(h) = target
        # Variables: [h_0, ..., h_{n-1}, T]
        # c = [0, ..., 0, 1]
        
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_rows = []
        b_vals = []
        
        for m_idx in active_idx:
            k = int(m_idx) - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[:n-k] += (1.0 - h[k:])
                g[k:] -= h[:n-k]
            else:
                ell = -k
                g[ell:] += (1.0 - h[:n-ell])
                g[:n-ell] -= h[ell:]
            
            # Linearized: corr[m](h_new) ≈ corr[m](h) + g^T (h_new - h)
            # Constraint: corr[m](h) + g^T (h_new - h) <= T
            # => g^T h_new - T <= g^T h - corr[m](h)
            row = np.append(g, -1.0)
            A_rows.append(row)
            b_vals.append(g @ h - corr[m_idx])
        
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        
        # Bounds: 0 <= h_i <= 1, T >= 0
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                T_new = res.x[n]
                h_new = project_feasible(h_new, target)
                new_corr = compute_corr(h_new)
                new_max = float(np.max(new_corr))
                
                if new_max < current_max - 1e-12:
                    improvement_ok = True
                else:
                    improvement_ok = False
                
                # Update active set: add any violated constraints
                violations = new_corr - T_new
                if np.max(violations) > 1e-6:
                    new_active = set(active_idx)
                    for i in range(len(violations)):
                        if violations[i] > 1e-6:
                            new_active.add(i)
                    # Keep only top violations
                    if len(new_active) > 200:
                        sorted_v = np.argsort(violations)[::-1][:200]
                        active_idx = sorted_v
                    else:
                        active_idx = list(new_active)
                    
                    h = h_new
                    current_max = new_max
                    continue
                
                if improvement_ok:
                    h = h_new
                    current_max = new_max
                break
        except Exception:
            break
    
    return h

    """Compute gradients of all correlation values w.r.t. h.
    Returns array of shape (2n-1, n)."""
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):

        if k >= 0:
            grads[m, :n-k] += (1.0 - h[k:])
            grads[m, k:] -= h[:n-k]
        else:
            ell = -k
            grads[m, ell:] += (1.0 - h[:n-ell])
            grads[m, :n-ell] -= h[ell:]
    return grads

    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad_fn(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def smooth_slp(h, n, target, alpha, maxiter=2000, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value_fn(x, alpha), h, method='SLSQP',
                      jac=lambda x: lse_grad_fn(x, alpha),
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def binary_search_opt(h, n, target, deadline, budget):
    """Binary projection + local search."""
    h_bin = (h > 0.5).astype(float).copy()
    # Fix count
    n_target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > n_target:
        one_pos = np.where(h_bin == 1)[0]
        order = one_pos[np.argsort(h[one_pos])]
        for p in order[:n_ones - n_target]:
            h_bin[p] = 0
    elif n_ones < n_target:
        zero_pos = np.where(h_bin == 0)[0]
        order = zero_pos[np.argsort(-h[zero_pos])]
        for p in order[:n_target - n_ones]:
            h_bin[p] = 1
    
    corr = compute_corr(h_bin)
    c_max = float(np.max(corr))
    
    while time.time() < min(deadline, time.time() + budget):
        if time.time() > deadline:
            break
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        
        if k_star >= 0:
            contrib = np.where((h_bin[:n-k_star] == 1) & (h_bin[k_star:] == 0))[0]
        else:
            ell = -k_star
            contrib = np.where((h_bin[ell:] == 1) & (h_bin[:n-ell] == 0))[0] + ell
        
        if len(contrib) == 0:
            break
        
        zeros = np.where(h_bin == 0)[0]
        best_delta = 1e-10
        best_pair = None
        
        # Try all contrib × zeros (limit to top 40 each)
        c_sample = contrib[:min(40, len(contrib))]
        z_sample = zeros[:min(40, len(zeros))]
        
        for i in c_sample:
            for j in z_sample:
                if i == j:
                    continue
                h_bin[i] = 0
                h_bin[j] = 1
                new_val = float(compute_corr(h_bin)[m_star])
                h_bin[i] = 1
                h_bin[j] = 0
                delta = new_val - corr[m_star]
                if -delta > best_delta:
                    best_delta = -delta
                    best_pair = (i, j)
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h_bin[i] = 0
        h_bin[j] = 1
        corr = compute_corr(h_bin)
        new_max = float(np.max(corr))
        if new_max < c_max:
            c_max = new_max
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    
    return h_bin, c_max
