# Q[1] = w0*h1+w1*h0 - w0*h1 = w1*h0 ✓
# Q[2] = w0*h2+w1*h1+w2*h0 - w0*h2 = w1*h1+w2*h0 ✓
# Q[3] = w0*h3+w1*h2+w2*h1+w3*h0 - w0*h3 = w1*h2+w2*h1+w3*h0 ✓



def lse_obj_and_grad_v2(h, alpha):
    """Compute LSE objective and gradient using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    fsize = 2 * n
    v = 1.0 - h
    w_flipped = weights[::-1].copy()
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    
    P_full = np.fft.irfft(Wf * V, fsize)
    P = P_full[n-1:2*n-1]
    
    Q_full = np.fft.irfft(W * H, fsize)
    Q = Q_full[:n] - weights[0] * h
    
    grad = dx * (P - Q)
    return obj, grad

def fista_optimize_v2(h0, target, alpha, max_iters, deadline, step_init=None):
    """FISTA for LSE objective."""
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if step_init is None:
        # Estimate L from the problem
        step_init = 0.3 / (alpha * dx_est)  # rough estimate
        dx_est = 2.0 / n
        step_init = 0.5 / alpha * n / 2.0  # heuristic
    
    step = step_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        obj_y, grad_y = lse_obj_and_grad_v2(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad_v2(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-14)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step / (1.0 - shrink), step_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    
    return best_x, best_obj

def top_k_subgradient(h, target, k_active, epoch_length, deadline, lr=None):
    """Subgradient method focusing on top-k active constraints."""
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    
    if lr is None:
        lr = 0.01
    
    for it in range(epoch_length):
        if time.time() > deadline:
            break
        
        # Compute correlation
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        
        # Get top-k active constraints
        top_idx = np.argsort(corr)[::-1][:k_active]
        c_max = corr[top_idx[0]]
        
        # Soft weights among top-k
        diffs = corr[top_idx] - c_max
        w = np.exp(diffs / (0.01 + np.std(corr[top_idx])))
        w /= w.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            if ki == 0:
                # corr[0] = sum h[i]*(1-h[i]), grad = (1-2h)
                grad += wi * (1.0 - 2.0 * h_cur)
            else:
                jp = np.arange(0, n - ki)
                grad[jp] += wi * (1.0 - h_cur[jp + ki])
                jn = np.arange(ki, n)
                grad[jn] -= wi * h_cur[jn - ki]
        
        grad *= dx
        
        # Take step with backtracking
        accepted = False
        for _ in range(20):
            h_new = project_feasible(h_cur - lr * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score + 1e-15:
                h_cur = h_new
                cur_score = sc
                accepted = True
                break
            lr *= 0.7
        
        if not accepted:
            lr *= 0.5
            if lr < 1e-10:
                lr = 1e-10
    
    return h_cur, cur_score

def structured_perturbation(h, n, rng, mode):
    """Generate a structured perturbation."""
    if mode == 0:
        # Gaussian noise
        return h + rng.randn(n) * 0.05
    elif mode == 1:
        # Sinusoidal perturbation
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.randint(1, 10)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.02, 0.1)
        return h + amp * np.sin(freq * np.pi * x + phase)
    elif mode == 2:
        # Block permutation
        block_size = rng.choice([10, 20, 40, 80])
        n_blocks = n // block_size
        perm = rng.permutation(n_blocks)
        h_pert = h.copy()
        for i in range(n_blocks):
            start_old = perm[i] * block_size
            start_new = i * block_size
            h_pert[start_new:start_new+block_size] = h[start_old:start_old+block_size]
        return h_pert
    elif mode == 3:
        # Scale + shift
        scale = rng.uniform(0.9, 1.1)
        shift = rng.uniform(-0.05, 0.05)
        return np.clip(h * scale + shift, 0, 1)
    else:
        # Random block flip
        n_blocks = 8
        block_size = n // n_blocks
        h_pert = h.copy()
        for i in range(n_blocks):
            if rng.random() < 0.3:
                start = i * block_size
                h_pert[start:start+block_size] = 1.0 - h[start:start+block_size]
        return h_pert

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    w = exp_s / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    fsize = 2 * n
    v = 1.0 - h
    wf = w[::-1].copy()
    F = np.fft.rfft
    I = np.fft.irfft
    P = I(F(wf, fsize) * F(v, fsize), fsize)[n-1:2*n-1]
    Q = I(F(w, fsize) * F(h, fsize), fsize)[:n]
    grad = dx * (P - Q)
    return obj, grad

        xp, x, t = x, xn, tn

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    w = exp_s / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    fsize = 2 * n
    v = 1.0 - h
    wf = w[::-1].copy()
    F = np.fft.rfft
    I = np.fft.irfft
    P = I(F(wf, fsize) * F(v, fsize), fsize)[n-1:2*n-1]
    Q = I(F(w, fsize) * F(h, fsize), fsize)[:n]
    grad = dx * (P - Q)
    return obj, grad
