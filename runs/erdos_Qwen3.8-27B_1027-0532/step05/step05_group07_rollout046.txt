```python
def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    # Positive shifts and zero: B[n-1] to B[2n-2]
    ks = np.arange(n)
    mask1 = ks <= i  # ks ≤ i, i.e., the "h[i-ks]" term is valid
    mask2 = (i + ks) < n  # i+ks < n, i.e., the "h[i+ks]" term is valid
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:2 * n - 1] = b_pos
    
    # Negative shifts: B[0] to B[n-2]
    # B[n-1-m] for m=1..n-1 should be (1-h[i+m])*[i+m<n] - h[i-m]*[m≤i]

    # (1-h[i+(n-1-j)])*[i+n-1-j<n] - h[i-(n-1-j)]*[n-1-j≤i]
    # Equivalently: b_neg[j] = (1-h[i+(n-1-j)])*[i+n-1-j<n] - h[i-(n-1-j)]*[n-1-j≤i]
    # Or: let m = n-1-j, b_neg[n-1-m] = (1-h[i+m])*[i+m<n] - h[i-m]*[m≤i]
```

```python
def block_swap_perturb(h, rng, L_min=3, L_max=20):
    n = len(h)
    h_new = h.copy()
    L = rng.randint(L_min, L_max + 1)
    i = rng.randint(0, n - L)
    j = rng.randint(0, n - L)
    if i == j:
        return h_new
    # Ensure blocks don't overlap (or allow overlap? Let's disallow for simplicity)
    if not (i + L <= j or j + L <= i):
        # Overlapping, just return original
        return h_new
    h_new[i:i+L], h_new[j:j+L] = h[j:j+L].copy(), h[i:i+L].copy()
    return h_new
```

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    # Positive part: B[n-1 : 2n-1]
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:2 * n - 1] = b_pos
    
    # Negative part: B[0 : n-1]
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg[::-1]  # FIX: reverse the negative part
    return B
