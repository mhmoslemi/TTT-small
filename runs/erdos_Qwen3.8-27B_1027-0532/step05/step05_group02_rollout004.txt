import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on tau
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        proj = np.clip(h - tau, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_gradient_smooth(h, alpha):
    """Compute gradient of smooth max approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    # Numerical stability
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # grad[i] = dx * (1 - sum_j w[j]*h[i+j-(n-1)] - sum_j w[j]*h[i-j+(n-1)])
    shifts = np.arange(-(n-1), n)
    i_idx = np.arange(n)[:, None]
    m_idx = shifts[None, :]
    
    sum_idx1 = i_idx + m_idx
    valid1 = (sum_idx1 >= 0) & (sum_idx1 < n)
    h1 = np.where(valid1, h[np.clip(sum_idx1, 0, n-1)], 0.0)
    term1 = h1 @ w
    
    sum_idx2 = i_idx - m_idx
    valid2 = (sum_idx2 >= 0) & (sum_idx2 < n)
    h2 = np.where(valid2, h[np.clip(sum_idx2, 0, n-1)], 0.0)
    term2 = h2 @ w
    
    grad = (1.0 - term1 - term2) * dx
    return grad

def grad_descent_continuous(h_start, alphas, max_iter_per_alpha, time_limit, start_time):
    """Projected gradient descent on continuous relaxation."""
    h = h_start.copy().astype(float)
    h = project_sum(h, len(h)/2.0)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        step = 0.1
        for it in range(max_iter_per_alpha):
            if time.time() - start_time > time_limit:
                break
            grad = compute_gradient_smooth(h, alpha)
            # Backtracking line search
            for _ in range(20):
                h_new = project_sum(h - step * grad, len(h)/2.0)
                score = compute_c5(h_new)
                if score < best_score or it < 5:  # accept if improves best or early
                    # Actually, let's use proper line search
                    h_new_score = compute_c5(h_new)
                    if h_new_score < compute_c5(h) or h_new_score < best_score:
                        break
                step *= 0.5
            h = project_sum(h - step * grad, len(h)/2.0)
            score = compute_c5(h)
            if score < best_score:
                best_score = score
                best_h = h.copy()
        # Restep
        step = 0.1
    
    return best_h, best_score

            best_h = h_new.copy()

        # Decay learning rate
        lr *= 0.999

    dx = 2.0 / n

    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):

        s = np.clip(h - tau, 0.0, 1.0).sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def compute_gradient_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max) - alpha * (corr - c_max).max()
    w = np.exp(shifted)
    w /= w.sum()
    
    shifts = np.arange(-(n-1), n)
    i_idx = np.arange(n)[:, None]
    m_idx = shifts[None, :]
    
    s1 = i_idx + m_idx
    v1 = (s1 >= 0) & (s1 < n)
    t1 = np.where(v1, h[np.clip(s1, 0, n-1)], 0.0) @ w
    
    s2 = i_idx - m_idx
    v2 = (s2 >= 0) & (s2 < n)
    t2 = np.where(v2, h[np.clip(s2, 0, n-1)], 0.0) @ w
    
    return (1.0 - t1 - t2) * dx

    n = len(h)

    b = np.zeros(N)

    idx = j + k

        score = compute_c5(h)

            best_h = h

    grad = (1.0 - t1 - t2) * dx
    return score, grad, w

def continuous_grad_opt(h_start, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    h = project_sum(h_start.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    alphas = [5, 15, 50, 150, 500, 1500, 5000, 15000]
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        h = project_sum(best_h.copy(), target)
        v = np.zeros(n)
        lr = 0.03
        momentum = 0.85
        n_iters = 800
        for it in range(n_iters):
            if time.time() - start_time > time_limit:
                break
            score, grad, w = smooth_max_and_grad(h, alpha)
            v = momentum * v - lr * grad
            h_new = project_sum(h + v, target)
            h = h_new
            actual_score = compute_c5(h)
            if actual_score < best_score:
                best_score = actual_score
                best_h = h.copy()
            lr = max(lr * 0.998, 0.002)
    return best_h, best_score

def continuous_sa(h_start, time_limit, start_time):
    n = len(h_start)
    target = n / 2.0
    h = project_sum(h_start.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    current_score = best_score
    alpha = 100
    t_init = 0.02
    t_final = 0.001
    n_total = 3000
    stall = 0
    for it in range(n_total):
        if time.time() - start_time > time_limit:
            break
        if stall > 10000:
            break
        ratio = it / max(n_total - 1, 1)
        temp = t_init * (t_final / t_init) ** ratio
        score, grad, w = smooth_max_and_grad(h, alpha)
        noise = temp * 0.1 * np.random.randn(n)
        h_new = project_sum(h - 0.02 * grad + noise, target)
        new_score = compute_c5(h_new)
        delta = new_score - current_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(temp, 1e-10)):
            h = h_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_score

def binary_greedy(h_start, max_iters, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_gain = 0
        best_pair = None
        n_try = min(300, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            gain = max_val - new_max
            if gain > best_gain:
                best_gain = gain
                best_pair = (int(i), int(j))
        if best_pair is not None:
            i, j = best_pair
            delta = compute_delta(h, i, j)
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            stall = 0
            c5 = float(full.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
    return best_h, best_c5

def sa_binary(h_start, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 60000:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        d = c5_new - current_c5
        if d < 0 or (t > 1e-15 and rng.random() < np.exp(-d / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def multi_swap_search(h_start, n_rounds, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    for rnd in range(n_rounds):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        max_val = float(full.max())
        ones = np.where(h > 0.999)[0]
        zeros = np.where(h < 0.001)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        best_gain = 0
        best_quad = None
        n_try = min(100, len(ones) * len(zeros) // 4)
        for _ in range(n_try):
            i1 = ones[rng.randint(len(ones))]
            i2_idx = rng.randint(len(ones)-1)
            i2 = ones[i2_idx if i2_idx < i1 else i2_idx+1]
            if i1 == i2:
                continue
            j1 = zeros[rng.randint(len(zeros))]
            j2_idx = rng.randint(len(zeros)-1)
            j2 = zeros[j2_idx if j2_idx < j1 else j2_idx+1]
            if j1 == j2:
                continue
            d1 = compute_delta(h, i1, j1)
            h[i1] = 0.0
            h[j1] = 1.0
            full += d1
            d2 = compute_delta(h, i2, j2)
            new_max = float((full + d2).max())
            gain = max_val - new_max
            if gain > best_gain:
                best_gain = gain
                best_quad = (int(i1), int(j1), int(i2), int(j2))
            h[i2] = 1.0
            h[j2] = 0.0
            full += d2
            h[i1] = 1.0
            h[j1] = 0.0
            full += d1
            full -= 2.0 * d1
        if best_quad is not None:
            i1, j1, i2, j2 = best_quad
            d1 = compute_delta(h, i1, j1)
            h[i1] = 0.0
            h[j1] = 1.0
            full += d1
            d2 = compute_delta(h, i2, j2)
            h[i2] = 0.0
            h[j2] = 1.0
            full += d2
            stall = 0
            c5 = float(full.max()) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            stall += 1
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_sum(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_sum(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Continuous gradient optimization (main phase)
    if tleft() > 100:
        alloc = min(350, tleft() * 0.35)
        x = np.random.RandomState(seed + 1)
        np.random.seed(seed + 1)
        h_cont, sc_cont = continuous_grad_opt(best_h, alloc, start_time)
        np.random.seed(seed + 1 + (time.time() - start_time).astype(np.int64) if hasattr(time.time() - start_time, 'astype') else seed + 1)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont.copy()

    # Phase 2: Continuous SA for exploration
    if tleft() > 80:
        alloc = min(200, tleft() * 0.20)
        np.random.seed(seed + 2)
        h_cont2, sc_cont2 = continuous_sa(best_h, alloc, start_time)
        np.random.seed(seed + 3)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2.copy()

    # Phase 3: Binary local search from continuous best
    if tleft() > 80:
        h_bin = project_binary(best_h, int(n_init / 2.0))
        alloc_g = min(100, tleft() * 0.12)
        h_bg, c5_bg = binary_greedy(h_bin, 50000, rng, alloc_g, start_time)
        if c5_bg < best_score:
            best_score = c5_bg
            best_h = h_bg.copy()
        if tleft() > 50:
            alloc_s = min(120, tleft() * 0.12)
            n_sa = int(min(5000000, alloc_s * 50000))
            h_bs, c5_bs = sa_binary(h_bg, n_sa, 0.015, 0.00005, rng, alloc_s, start_time)
            if c5_bs < best_score:
                best_score = c5_bs
                best_h = h_bs.copy()

    # Phase 4: Multi-swap binary search
    if tleft() > 60:
        h_bin2 = project_binary(best_h, int(n_init / 2.0))
        alloc_ms = min(80, tleft() * 0.10)
        h_ms, c5_ms = multi_swap_search(h_bin2, 5000, rng, alloc_ms, start_time)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 5: Multi-resolution
    for n_try in [400, 800, 1000]:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        alloc_c = min(80, tleft() * 0.15)
        np.random.seed(seed + n_try)
        h_c, sc_c = continuous_grad_opt(h_interp, alloc_c, start_time)
        np.random.seed(seed + n_try + 1)
        if sc_c < best_score:
            best_score = sc_c
            best_h = h_c.copy()

        if tleft() > 40:
            h_b = project_binary(h_c, n_try // 2)
            alloc_b = min(40, tleft() * 0.10)
            h_bg2, c5_bg2 = binary_greedy(h_b, 30000, rng, alloc_b, start_time)
            if c5_bg2 < best_score:
                best_score = c5_bg2
                best_h = h_bg2.copy()

    # Phase 6: Final continuous polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alloc = min(tleft() - 15, 120)
        np.random.seed(seed + 99)
        h_ref, sc_ref = continuous_grad_opt(best_h, alloc, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 7: Final binary polish
    if tleft() > 25:
        n_use = len(best_h)
        h_final_bin = project_binary(best_h, n_use // 2)
        alloc = min(tleft() - 10, 60)
        h_fb, c5_fb = sa_binary(h_final_bin, int(min(3000000, alloc * 40000)), 0.005, 0.00002, rng, alloc, start_time)
        if c5_fb < best_score:
            best_score = c5_fb
            best_h = h_fb.copy()

    # Final
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_out = project_sum(best_h, target_final)
    final_score = compute_c5(h_out)
    if final_score > best_score:
        h_out = project_sum(best_h, target_final)
    return h_out, final_score, n_final
