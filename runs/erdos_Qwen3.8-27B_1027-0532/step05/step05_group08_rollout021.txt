import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    """Full linear cross-correlation of h and (1-h)."""
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def symmetrize(h):
    n = len(h)
    h_sym = (h + h[::-1]) / 2.0
    return h_sym

def project_feasible_sym(h, target):
    """Project to satisfy 0<=h<=1, sum(h)=target, and h symmetric."""
    h = symmetrize(h.copy())
    n = len(h)
    n_half = n // 2
    # Only project the first half, then mirror
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        h = symmetrize(h)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = symmetrize(h)
    h = np.clip(h, 0.0, 1.0)
    return h

def make_symmetric(x, n):
    """Given first n/2 values, create symmetric full vector of length n."""
    n_half = n // 2
    h = np.zeros(n)
    h[:n_half] = x
    h[n_half:] = x[::-1]
    return h

def lse_value_sym(x, n, alpha):
    """Soft-max objective for symmetric parameterization."""
    h = make_symmetric(x, n)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad_sym(x, n, alpha):
    """Gradient of soft-max with respect to the first n/2 variables."""
    h = make_symmetric(x, n)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    # Compute gradient w.r.t. full h
    g_full = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-14:

        g_full[:n - k] += wk * (1.0 - h[k:])
        g_full[k:] -= wk * h[:n - k]

        if wk < 1e-14:
            continue
        g_full[ell:] += wk * (1.0 - h[:n - ell])
        g_full[:n - ell] -= wk * h[ell:]
    g_full *= dx
    # Project gradient to symmetric subspace
    g_sym = (g_full + g_full[::-1]) / 2.0
    return g_sym[:n // 2]

            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break

def soft_max_obj(h, alpha, top_m=None):
    corr = compute_corr(h)
    if top_m is not None:
        corr_use = corr[top_m]
        c_max = float(np.max(corr_use))
    else:
        c_max = float(np.max(corr))
        corr_use = corr
    shifted = np.clip(alpha * (corr_use - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def soft_max_grad(h, alpha, top_m=None):
    n = len(h)
    corr = compute_corr(h)
    if top_m is not None:
        active_corr = corr[top_m]
        c_max = float(np.max(active_corr))
    else:
        c_max = float(np.max(corr))
        active_corr = corr
        top_m = np.arange(len(corr))
    shifted = np.clip(alpha * (active_corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, top_m):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgs_optimize(h, n, target, alpha, mu, maxiter, top_m=None):
    def obj(x):
        return soft_max_obj(x, alpha, top_m) + mu * (np.sum(x) - target)**2
    def objg(x):
        return soft_max_grad(x, alpha, top_m) + 2.0 * mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg, bounds=bounds,
                      options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e6})
        return project_onto(res.x, target)
    except:
        return h

def slsqp_optimize(h, n, target, alpha, maxiter, top_m=None):
    def obj(x):
        return soft_max_obj(x, alpha, top_m)
    def objg(x):
        return soft_max_grad(x, alpha, top_m)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)

        res = minimize(obj, h, method='SLSQP', jac=objg, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_onto(res.x, target)
    except:
        return h

def subgrad_step(h, n, target, temp, lr, top_k=10):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    base = corr[top_idx[0]]
    weights = np.exp((corr[top_idx] - base) / temp)
    weights /= weights.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(weights, top_idx):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])

        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    h_new = h - lr * g
    return project_onto(h_new, target)

    except Exception:
        return h


def do_trust_constr(h, n, target, alpha, maxiter, top_m):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_func, 'args': (target,), 'jac': cons_jac}]
    try:
        res = minimize(slsqp_obj, h, args=(alpha, target, top_m),
                       method='trust-constr', jac=slsqp_jac, bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'gtol': 1e-9, 'xtol': 1e-12})
        return project_onto(res.x, target)
    except Exception:
        return h


def subgrad_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, top_k):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:top_k]
        base = corr[top_idx[0]]
        weights = np.exp((corr[top_idx] - base) / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_onto(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= 0.985
            if lr < 1e-10:
                break
    return best_h, best_score


def fista_polish(h, n, target, n_steps, alpha, deadline, top_m):
    h_cur = h.copy()
    y = h_cur.copy()
    t = 1.0
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step = 1.0 / (alpha * np.max(compute_corr(h_cur)) + 1.0)
    step = min(step, 0.01)
    for k in range(n_steps):
        if time.time() > deadline:
            break
        g = soft_max_grad(y, alpha, target, top_m)
        g = (g + g[::-1]) / 2.0
        h_new = y - step * g
        h_new = project_onto(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t - 1.0) / t_new * (h_new - h_cur)
        y = project_onto(y, target)
        h_cur = h_new
        t = t_new
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_onto(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.03, 0.012)
        else:
            eps *= 0.82
            if eps < 1e-10:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_onto(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_onto(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 12.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h0 = np.array(initial_h_values, dtype=float)
    h0 = symmetrize(h0)
    h = project_onto(h0, target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation at n=400
    alphas_p1 = [15, 40, 80, 150, 300, 600, 1200, 2500, 5000, 10000, 20000]
    for alpha in alphas_p1:
        if time.time() > deadline - 400:
            break
        for mu in [3.0, 30.0, 200.0, 1000.0]:
            if time.time() > deadline - 350:
                break
            h_new = do_lbfgs(best_h, n, target, alpha, mu, 250, None)
            h_new = symmetrize(h_new)
            h_new = project_onto(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Multi-lag active set optimization
    for outer in range(12):
        if time.time() > deadline - 250:
            break
        corr = compute_corr(best_h)
        top_m = list(np.argsort(corr)[::-1][:10])
        for alpha in [300, 1500, 6000, 20000]:
            if time.time() > deadline - 200:
                break
            h_new = do_lbfgs(best_h, n, target, alpha, 500.0, 300, top_m)
            h_new = symmetrize(h_new)
            h_new = project_onto(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 3: SLSQP and trust-constr refinement
    for alpha in [2000, 8000, 30000, 80000]:
        if time.time() > deadline - 150:
            break
        h_new = do_slsqp(best_h, n, target, alpha, 800, None)
        h_new = symmetrize(h_new)
        h_new = project_onto(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if time.time() > deadline - 100:
            break
        h_new = do_trust_constr(best_h, n, target, alpha, 500, None)
        h_new = symmetrize(h_new)
        h_new = project_onto(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: FISTA polishing
    if time.time() < deadline - 100:
        for alpha in [500, 2000, 8000]:
            if time.time() > deadline - 80:
                break
            h_f, sc_f = fista_polish(best_h, n, target, 3000, alpha, deadline - 70, None)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_f

    # Phase 5: Subgradient polishing
    if time.time() < deadline - 80:
        for temp in [0.03, 0.01, 0.005, 0.002]:
            if time.time() > deadline - 60:
                break
            h_sg, sc_sg = subgrad_polish(best_h, n, target, 3000, temp, 0.004, 0.997, deadline - 50, 10)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 6: Worst-lag surgery
    if time.time() < deadline - 50:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 500, 0.004, deadline - 40)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 7: Higher resolution n=800
    if time.time() < deadline - 200:
        n2 = 800
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)

        alphas_p7 = [20, 60, 150, 400, 1000, 3000, 8000, 20000]
        for alpha in alphas_p7:
            if time.time() > deadline - 80:
                break
            for mu in [10.0, 100.0, 500.0]:
                if time.time() > deadline - 60:
                    break
                h2_new = do_lbfgs(best_h2, n2, target2, alpha, mu, 200, None)
                h2_new = symmetrize(h2_new)
                h2_new = project_onto(h2_new, target2)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

        if time.time() < deadline - 60:
            corr2 = compute_corr(best_h2)
            top_m2 = list(np.argsort(corr2)[::-1][:10])
            for alpha in [500, 2000, 8000]:
                if time.time() > deadline - 45:
                    break
                h2_new = do_lbfgs(best_h2, n2, target2, alpha, 500.0, 200, top_m2)
                h2_new = symmetrize(h2_new)
                h2_new = project_onto(h2_new, target2)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

        if time.time() < deadline - 40:
            h2_new = do_slsqp(best_h2, n2, target2, 5000, 500, None)
            h2_new = symmetrize(h2_new)
            h2_new = project_onto(h2_new, target2)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

        if time.time() < deadline - 30:
            h_sg2, sc_sg2 = subgrad_polish(best_h2, n2, target2, 1500, 0.005, 0.003, 0.997, deadline - 20, 10)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 8: Random restarts at best resolution
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(6):

            scale = [0.06, 0.10, 0.04, 0.12, 0.08, 0.15][attempt]
            h_pert = best_h + rng.randn(n_cur) * scale
            h_pert = symmetrize(h_pert)
            h_pert = project_onto(h_pert, target_cur)
            h_ref = do_lbfgs(h_pert, n_cur, target_cur, 500, 200.0, 200, None)
            h_ref = symmetrize(h_ref)
            h_ref = project_onto(h_ref, target_cur)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if time.time() < deadline - 25:
                    h_ref2 = do_slsqp(best_h, n_cur, target_cur, 5000, 500, None)
                    h_ref2 = symmetrize(h_ref2)
                    h_ref2 = project_onto(h_ref2, target_cur)
                    sc2 = compute_c5(h_ref2)
                    if sc2 < best_score:

                        best_h = h_ref2

    # Phase 9: Final subgradient + surgery
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_sg, sc_sg = subgrad_polish(best_h, n_cur, target_cur, 1000, 0.003, 0.002, 0.998, deadline - 15, 8)
        if sc_sg < best_score:

            best_h = h_sg
        if time.time() < deadline - 10:
            h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.002, deadline - 8)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = do_slsqp(best_h, n_cur, target_cur, 50000, 500, None)
        h_new = symmetrize(h_new)
        h_new = project_onto(h_new, target_cur)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_onto(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
