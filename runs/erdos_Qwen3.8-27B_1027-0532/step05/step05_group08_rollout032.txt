    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

def golden_section_min(f, a, b, tol=1e-12, maxiter=100):
    """Minimize convex univariate function f on [a, b] using golden section."""
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - invphi * (b - a)
    d = a + invphi * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + invphi * (b - a)
            fd = f(d)
    return (a + b) / 2.0

    order = np.arange(n)

    for i in order:

        # k = m - (n-1) ranges from -i to n-1-i
        k_vals = np.arange(-i, n - i)

        term2 = h[idx2]
        idx3 = i - k_vals  # 2i-n+1 to 2i

        term3 = np.zeros(n)
        term3[valid3] = h[idx3[valid3]]
        a_m = 1.0 - term2 - term3
        
        # b_m for active indices, b = corr[m] - a_m * t_old
        b_active = corr[m_vals] - a_m * t_old

        if f_0 <= f_star and f_0 <= f_1:
            t_star = 0.0
        elif f_1 <= f_star and f_1 <= f_0:

            h[i] = t_star
            corr[m_vals] += a_m * delta
    
    # Project to feasible
    h = project_feasible(h, target)
    corr = compute_corr(h)
    return h, corr
