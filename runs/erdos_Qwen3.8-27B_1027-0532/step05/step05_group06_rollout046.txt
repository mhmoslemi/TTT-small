    return topm_values(x, target, mu, m)[0]


def topm_grad(x, target, mu, m):
    return topm_values(x, target, mu, m)[1]


def slsqp_f(x, target, mu):
    return lse_values(x, 50000, target, mu)[0]


def slsqp_g(x, target, mu):
    return lse_values(x, 50000, target, mu)[1]


def slsqp_eq(x, target):
    return np.sum(x) - target


def slsqp_eq_jac(x, target):
    return np.ones(len(x))


def lbfgs_optimize(h0, target, alpha, mu, maxiter, deadline):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f_obj, h0, jac=f_grad, args=(alpha, target, mu),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50, 'disp': 0})
        return project_feasible(res.x, target)
    except Exception:
        return h0


def subgradient_topm(h0, target, mu, m, n_iter, lr_init, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    for it in range(n_iter):
        if time.time() > deadline:
            break
        val, grad = topm_values(h, target, mu, m)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.75
            if lr < 1e-11:
                break
    return best_h, best_score


def upsample_array(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new


def make_initial_binary(n, rng):
    h = np.zeros(n)
    target = n / 2.0
    first_block = int(n * 0.35)
    second_start = int(n * 0.55)
    second_block = int(n * 0.35)
    h[:first_block] = 1.0
    h[second_start:second_start + second_block] = 1.0
    h += rng.uniform(-0.1, 0.1, n)
    project_feasible(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 30.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 100.0

    # Phase 1: L-BFGS-B with increasing alpha at base resolution
    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 200000, 1000000]
    for a in alphas_p1:
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 30:
            break
        maxit = int(min(800, max(100, remaining * 80)))
        h_new = lbfgs_optimize(best_h, target, a, mu, maxit, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_eq, sc_eq = subgradient_topm(best_h, target, 1e6, 5, 100, 0.008, deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    # Phase 2: Top-m subgradient with different m values
    for m in [3, 5, 7, 10]:
        if time.time() > deadline:
            break
        h_eq, sc_eq = subgradient_topm(best_h, target, 1e8, m, 200, 0.005, deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 500000, mu, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Multi-resolution upsample to 2x
    n2 = 2 * n
    target2 = n2 / 2.0
    if deadline - time.time() > 350:
        h_fine = upsample_array(best_h, 2)
        h_fine = project_feasible(h_fine, target2)
        alphas_fine = [50, 500, 5000, 50000, 200000]
        for a in alphas_fine:
            if time.time() > deadline:
                break
            remaining = deadline - time.time()
            if remaining < 40:
                break
            maxit = int(min(600, max(80, remaining * 50)))
            h_new = lbfgs_optimize(h_fine, target2, a, 500.0, maxit, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_fine = h_new
        h_eq, sc_eq = subgradient_topm(h_fine, target2, 1e8, 5, 150, 0.004, deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
            n = n2
            target = target2

    # Phase 4: Diversification with different starting points
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    n_starts = 5
    for i in range(n_starts):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        if remaining < 100:
            break
        if i == 0:
            h_pert = best_h + rng.randn(n_cur) * 0.05
        elif i == 1:
            h_pert = best_h + rng.randn(n_cur) * 0.15
        elif i == 2:
            idx = rng.choice(n_cur, size=max(1, n_cur // 4), replace=False)
            h_pert = best_h.copy()
            h_pert[idx] = rng.uniform(0, 1, len(idx))
        elif i == 3:
            freq = np.fft.rfftfreq(n_cur)
            mask = (freq > 0) & (freq < 0.015)
            pert = np.fft.irfft(np.where(mask, rng.randn(n_cur // 2 + 1), 0), n_cur)
            h_pert = best_h + pert * 0.8
        else:
            h_pert = make_initial_binary(n_cur, rng)
        h_pert = project_feasible(h_pert, target_cur)
        h_new = lbfgs_optimize(h_pert, target_cur, 1000, mu, 300, deadline)
        h_new = lbfgs_optimize(h_new, target_cur, 50000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Final top-m polish
    for m in [3, 5]:
        if time.time() > deadline:
            break
        h_eq, sc_eq = subgradient_topm(best_h, target_cur, 1e8, m, 200, 0.003, deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    # Phase 6: High-alpha L-BFGS refinement
    if deadline - time.time() > 40:
        h_new = lbfgs_optimize(best_h, target_cur, 2000000, mu, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: SLSQP final polish
    if deadline - time.time() > 30:
        n_slq = len(best_h)
        bounds = [(0.0, 1.0)] * n_slq
        cons = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac, 'args': (target_cur,)}]
        try:
            res = minimize(slsqp_f, best_h, jac=slsqp_g,
                           args=(target_cur, 1e10),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-16, 'disp': 0})
            h_slq = project_feasible(res.x, target_cur)
            sc = compute_c5(h_slq)
            if sc < best_score:
                best_score = sc
                best_h = h_slq
        except Exception:
            pass

    # Phase 8: Final coordinate-descent-like sweep
    if deadline - time.time() > 30:
        h_eq, sc_eq = subgradient_topm(best_h, target_cur, 1e10, 3, 150, 0.002, deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
