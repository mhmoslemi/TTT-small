def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_for_lags(h_fix, lags, n, target, dx):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    g = 1.0 - h_fix
    n_half = n - 1
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-10, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8, verbose=False)
        except Exception:
            return None, None
    if h_var.value is None:
        return None, None
    h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
    return h_new, t_var.value


def cutting_plane_lp(h_init, n, target, time_limit, max_iter=2000, n_init_lags=50):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1

    overlaps = compute_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:n_init_lags]
    active_lags = list(set((top_idx - n_half).tolist()))

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        h_new, t_val = solve_lp_for_lags(h, active_lags, n, target, dx)
        if h_new is None:
            break

        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()

        overlaps_new = compute_overlaps(h_new)
        threshold = t_val - 1e-8 if t_val is not None else best_c5
        violated_idx = np.where(overlaps_new > threshold + 1e-6)[0]

        if len(violated_idx) == 0:
            break

        new_lags = set((violated_idx - n_half).tolist())
        existing = set(active_lags)
        new_lags_to_add = new_lags - existing
        active_lags = list(existing | new_lags_to_add)

        if len(active_lags) > 300:
            overlaps_check = compute_overlaps(h_new)
            top_k = np.argsort(overlaps_check)[::-1][:200]
            active_lags = list(set((top_k - n_half).tolist()))

    return best_h, best_c5


def coord_descent(h_init, n, target, time_limit, rng, max_sweeps=20000):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stag = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if stag > 300:
            break

        improved_this_sweep = False
        active = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        if len(active) == 0:
            break
        if len(active) > 200:
            active = rng.choice(active, size=200, replace=False)

        for i0 in active:
            if time.time() - start > time_limit:
                break
            old = h[i0]
            if old < 0.05:
                cands = [0.0, 0.05, 0.15, 0.3, 0.5]
            elif old > 0.95:
                cands = [1.0, 0.95, 0.85, 0.7, 0.5]
            else:
                cands = [0.0, 1.0, old * 0.4, min(1.0, old * 1.6),
                         max(0.0, old - 0.1), min(1.0, old + 0.1),
                         max(0.0, old - 0.25), min(1.0, old + 0.25)]

            best_val = old
            best_obj = compute_c5(h)
            for c in cands:
                if abs(c - old) < 1e-12:
                    continue
                h[i0] = c
                obj = compute_c5(h)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = c
            h[i0] = best_val
            if best_val != old:
                improved_this_sweep = True

        h = project_feasible(h, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stag = 0
        else:
            stag += 1

    return best_h, best_c5


def pair_ops(h_init, n, target, time_limit, rng, max_iters=50000):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stag = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if stag > 500:
            break

        idx_a, idx_b = rng.choice(n, size=2, replace=False)
        a_val = h[idx_a]
        b_val = h[idx_b]

        if a_val <= 1e-8 and b_val <= 1e-8:
            continue
        if a_val >= 1.0 - 1e-8 and b_val >= 1.0 - 1e-8:
            continue

        deltas = [0.05, 0.1, 0.2, 0.3, 0.5]
        best_d = 0.0
        best_obj = compute_c5(h)

        for d in deltas:
            if a_val + d > 1.0 or b_val - d < 0.0:
                continue
            h[idx_a] = a_val + d
            h[idx_b] = b_val - d
            obj = compute_c5(h)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_d = d
            h[idx_a] = a_val
            h[idx_b] = b_val
            if b_val + d > 1.0 or a_val - d < 0.0:
                continue
            h[idx_a] = a_val - d
            h[idx_b] = b_val + d
            obj = compute_c5(h)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_d = -d

        if best_d != 0.0:
            h[idx_a] = a_val + best_d
            h[idx_b] = b_val - best_d
            h = project_feasible(h, target)
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stag = 0
            else:
                stag += 1
        else:
            stag += 1

    return best_h, best_c5


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _global_target


def _cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass

    return best_h, best_c5


def subgradient_momentum(h_init, n, target, max_steps, time_limit,
                         n_top=20, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half

        w = 0.85 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            if k >= 0:
                idx1 = idxs[idxs + k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 + k])
                idx2 = idxs[idxs >= k]
                grad[idx2] += w[i] * (-y[idx2 - k])
            else:
                idx1 = idxs[idxs - k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 - k])
                idx2 = idxs[idxs + k >= 0]
                grad[idx2] += w[i] * (-y[idx2 + k])

        grad *= dx
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.01, polyak_lr), 0.5)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def make_block_init(n, target, rng, n_blocks=5):
    h = np.zeros(n)
    block_size = n // n_blocks
    positions = sorted(rng.choice(n, size=n_blocks, replace=False))
    for pos in positions:
        start = (pos * block_size) % n
        end = start + block_size
        if end <= n:
            h[start:end] = 1.0
        else:
            h[start:n] = 1.0
            h[0:end - n] = 1.0
    return project_feasible(h, target)


def make_periodic_init(n, target, rng):
    h = np.zeros(n)
    period = rng.choice([3, 4, 5, 6, 7, 8])
    fill = rng.random(period)
    for i in range(n):
        h[i] = fill[i % period]
    return project_feasible(h, target)


def make_gaussian_init(n, target, rng):
    h = np.zeros(n)
    x = np.linspace(0, 2, n)
    centers = rng.uniform(0.2, 1.8, size=3)
    widths = rng.uniform(0.1, 0.4, size=3)
    for c, w in zip(centers, widths):
        h += np.exp(-0.5 * ((x - c) / w) ** 2)
    h = h / h.max()
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick LP to improve initial
    if tleft() > 100:
        h_lp, c5_lp = cutting_plane_lp(best_h, n, target, time_limit=min(60, tleft() * 0.3))
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent polish
    if tleft() > 80:
        h_cd, c5_cd = coord_descent(best_h, n, target, time_limit=min(80, tleft() * 0.3), rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Subgradient momentum
    if tleft() > 80:
        h_sg, c5_sg = subgradient_momentum(best_h, n, target, max_steps=30000,
                                           time_limit=min(60, tleft() * 0.25), n_top=25, momentum=0.88)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: SLSQP with progressive smoothing
    if tleft() > 100:
        h_sl, c5_sl = slsqp_optimize(best_h, n, target,
                                     [50, 200, 800, 3000, 10000, 50000, 200000],
                                     maxiter=3000, ftol=1e-15)
        if c5_sl < best_score:
            best_score = c5_sl
            best_h = h_sl

    # Phase 5: Pair operations
    if tleft() > 80:
        h_po, c5_po = pair_ops(best_h, n, target, time_limit=min(60, tleft() * 0.3), rng=rng)
        if c5_po < best_score:
            best_score = c5_po
            best_h = h_po

    # Phase 6: LP cutting plane again
    if tleft() > 80:
        h_lp2, c5_lp2 = cutting_plane_lp(best_h, n, target, time_limit=min(60, tleft() * 0.3),
                                         max_iter=3000, n_init_lags=80)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 7: More coordinate descent
    if tleft() > 60:
        h_cd2, c5_cd2 = coord_descent(best_h, n, target, time_limit=min(60, tleft() * 0.4), rng=rng)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 8: Try diverse initializations at same resolution
    inits = [
        make_block_init(n, target, rng, n_blocks=3),
        make_block_init(n, target, rng, n_blocks=7),
        make_periodic_init(n, target, rng),
        make_gaussian_init(n, target, rng),
    ]
    for h_try_init in inits:
        if tleft() < 50:
            break
        h_t, c5_t = cutting_plane_lp(h_try_init, n, target, time_limit=20, max_iter=500, n_init_lags=40)
        if c5_t < best_score:
            best_score = c5_t
            best_h = h_t
        h_t2, c5_t2 = coord_descent(h_t, n, target, time_limit=15, rng=rng, max_sweeps=5000)
        if c5_t2 < best_score:
            best_score = c5_t2
            best_h = h_t2

    # Phase 9: SLSQP again from best
    if tleft() > 80:
        h_sl2, c5_sl2 = slsqp_optimize(best_h, n, target,
                                       [200, 1000, 5000, 20000, 100000, 500000],
                                       maxiter=5000, ftol=1e-16)
        if c5_sl2 < best_score:
            best_score = c5_sl2
            best_h = h_sl2

    # Phase 10: Try higher resolutions
    for n_high in [450, 500, 600]:
        if tleft() < 120:
            break
        target_h = n_high / 2.0
        h_hi = interpolate_to(best_h, n_high, target_h)

        # Quick LP at higher res
        h_hi_lp, c5_hi_lp = cutting_plane_lp(h_hi, n_high, target_h,
                                             time_limit=min(40, tleft() * 0.3),
                                             max_iter=1000, n_init_lags=60)
        if c5_hi_lp < best_score:
            best_score = c5_hi_lp
            best_h = h_hi_lp
            n = n_high
            target = target_h

        # SLSQP at higher res
        if tleft() > 60:
            h_hi_sl, c5_hi_sl = slsqp_optimize(h_hi_lp, n_high, target_h,
                                               [100, 500, 2000, 10000, 50000],
                                               maxiter=2000, ftol=1e-15)
            if c5_hi_sl < best_score:
                best_score = c5_hi_sl
                best_h = h_hi_sl
                n = n_high
                target = target_h

    # Phase 11: Final coordinate descent polish
    if tleft() > 40:
        n_fin = len(best_h)
        tgt_fin = n_fin / 2.0
        h_fin, c5_fin = coord_descent(best_h, n_fin, tgt_fin,
                                      time_limit=min(35, tleft() * 0.7), rng=rng, max_sweeps=30000)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Phase 12: Final LP polish
    if tleft() > 30:
        n_fin = len(best_h)
        tgt_fin = n_fin / 2.0
        h_fin2, c5_fin2 = cutting_plane_lp(best_h, n_fin, tgt_fin,
                                           time_limit=min(25, tleft() * 0.8),
                                           max_iter=2000, n_init_lags=100)
        if c5_fin2 < best_score:
            best_score = c5_fin2
            best_h = h_fin2

    # Final project and return
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
