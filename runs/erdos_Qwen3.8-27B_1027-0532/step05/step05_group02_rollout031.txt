def delta_multi(h, changes):

    changes: list of (pos, diff) where diff is +1 (0→1) or -1 (1→0)
    Uses the ORIGINAL h for all computations (simultaneous change)."""

    k = np.arange(2*n-1) - (n-1)
    for pos, diff in changes:
        idx = pos + k
        mask = (idx >= 0) & (idx < n)
        delta[mask] += diff * (1.0 - h[idx[mask]])
        idx = pos - k
        mask = (idx >= 0) & (idx < n)
        delta[mask] -= diff * h[idx[mask]]
    return delta

def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2 * n - 1][::-1]

def delta_single(h, i, j):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    m = (idx >= 0) & (idx < n)
    delta[m] += -(1.0 - h[idx[m]])
    idx = i - k
    m = (idx >= 0) & (idx < n)
    delta[m] += h[idx[m]]
    idx = j + k
    m = (idx >= 0) & (idx < n)
    delta[m] += (1.0 - h[idx[m]])
    idx = j - k
    m = (idx >= 0) & (idx < n)
    delta[m] -= h[idx[m]]
    return delta

def project_binary(h, target):

        return h_r

        ones = np.where(h_r == 1.0)[0]

            h_r[ones[idx]] = 0.0

        zeros = np.where(h_r == 0.0)[0]
        order = np.argsort(-h[zeros])

def _smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    logsumexp = c_max + float(np.log(np.sum(np.exp(shifted)) / alpha))
    return logsumexp + 100.0 * (np.sum(h) - target) ** 2

def _slsqp_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = project_feasible(h_start.copy(), target)
    curr_score = compute_c5(h_cur)

    dx = 2.0 / n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                lambda x: _smooth_obj(x, alpha, n, target),
                h_cur, method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
