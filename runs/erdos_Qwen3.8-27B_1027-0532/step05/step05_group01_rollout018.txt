    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        top_idx = np.argsort(corr)[-n_top:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_idx] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_idx):
            if wi < 1e-8:
                continue
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k])
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 500:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_optimize(best_h, n, target, [50, 200, 1000, 5000], 2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        cd_time = min(400, tleft() * 0.45)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target, cd_time, n_cand=50, max_sweeps=50000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol = slsqp_optimize(best_h, n, target, [500, 2000, 10000, 50000], 2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        cd_time = min(300, tleft() * 0.5)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target, cd_time, n_cand=60, max_sweeps=50000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 60:
        h_pol = slsqp_optimize(best_h, n, target, [1000, 5000, 20000, 100000], 1500, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 80:
        h_sg, c5_sg = focused_subgradient(best_h, n, target,
                                          time_limit=min(40, tleft() * 0.3),
                                          max_steps=30000, n_top=10, base_lr=0.01)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg
        if tleft() > 40:
            h_cd2, c5_cd2 = vectorized_pair_cd(best_h, n, target,
                                               min(30, tleft() * 0.4), n_cand=50, max_sweeps=10000)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2

    rng = np.random.RandomState(seed)
    for noise_scale in [0.01, 0.03, 0.05]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref = slsqp_optimize(h_pert, n, target, [100, 500, 2000, 10000], 1500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        if tleft() > 60:
            h_cd3, c5_cd3 = vectorized_pair_cd(best_h, n, target,
                                               min(40, tleft() * 0.3), n_cand=50, max_sweeps=10000)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    if tleft() > 150:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_optimize(h_interp, n_try, target_try, [100, 500, 2000, 10000], 1500, 1e-15)
            sc = compute_c5(h_hires)
            if tleft() > 80:
                h_hires2, c5_h2 = vectorized_pair_cd(h_hires, n_try, target_try,
                                                     min(40, tleft() * 0.4), n_cand=40, max_sweeps=5000)
                if c5_h2 < sc:
                    sc = c5_h2
                    h_hires = h_hires2
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd_final, c5_final = vectorized_pair_cd(best_h, n_use, target_use,
                                                  min(50, tleft() * 0.5), n_cand=50, max_sweeps=20000)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_optimize(best_h, n_use, target_use, [5000, 20000, 100000], 1000, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)

    return h_out, final_score, n_use
