```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def lse_obj(x, alpha, target, mu):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (np.sum(x) - target) ** 2
    return val


def lse_grad(x, alpha, target, mu):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (np.sum(x) - target)
    return grad


def lbfgs_opt(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj(x, alpha, target, mu)

    def g(x):
        return lse_grad(x, alpha, target, mu)

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h


def round_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def binary_sa(h_bin, n_evals=100000, deadline=None, seed=42):

    ones = list(np.where(h_cur == 1)[0].tolist())
    zeros = list(np.where(h_cur == 0)[0].tolist())
    rng = np.random.RandomState(seed)
    
    T = max_cur * 0.05
    T_min = max_cur * 0.001
    
    for step in range(n_evals):
        if deadline is not None and time.time() > deadline:
            break
        if step % 1000 == 0:
            T = max(T_min, T * 0.98)
        
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        a = ones[i]
        b = zeros[j]
        
        h_cur[a] = 0
        h_cur[b] = 1
        corr_new = compute_corr(h_cur)
        max_new = float(np.max(corr_new))
        
        if max_new < max_cur or rng.random() < np.exp(-(max_new - max_cur) / (T + 1e-15)):
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def lse_obj(x, alpha, target, mu):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    val += mu * (np.sum(x) - target) ** 2
    return val


def lse_grad(x, alpha, target, mu):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    thresh = 1e-6
    active = np.where(weights > thresh)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (np.sum(x) - target)
    return grad


def lbfgs_opt(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj(x, alpha, target, mu)

    def g(x):
        return lse_grad(x, alpha, target, mu)

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 60})
        return project_feasible(res.x, target)
    except Exception:
        return h


def round_binary(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin


def binary_sa(h_bin, n_evals=100000, deadline=None, seed=42):
    n = len(h_bin)
    dx = 2.0 / n
    h_cur = h_bin.copy()
    corr_cur = compute_corr(h_cur).copy()
    max_cur = float(np.max(corr_cur))
    best_h = h_cur.copy()
    best_max = max_cur
    ones = list(np.where(h_cur == 1)[0].tolist())
    zeros = list(np.where(h_cur == 0)[0].tolist())
    rng = np.random.RandomState(seed)
    T = max_cur * 0.05
    T_min = max_cur * 0.002
    for step in range(n_evals):
        if deadline is not None and time.time() > deadline:
            break
        if step % 500 == 0:
            T = max(T_min, T * 0.97)
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        a = ones[i]
        b = zeros[j]
        h_cur[a] = 0.0
        h_cur[b] = 1.0
        corr_new = compute_corr(h_cur)
        max_new = float(np.max(corr_new))
        if max_new < max_cur or rng.random() < np.exp(-(max_new - max_cur) / (T + 1e-15)):
            corr_cur = corr_new
            max_cur = max_new
            ones[i] = b
            zeros[j] = a
            if max_cur < best_max:
                best_max = max_cur
                best_h = h_cur.copy()
        else:
            h_cur[a] = 1.0
            h_cur[b] = 0.0

def binary_greedy(h_bin, n_sweeps=50, deadline=None):

            a = ones[ti]

def subgrad_descent(h, target, n_steps=5000, deadline=None):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.05
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        k = min(5, len(corr))
        top_k = np.argsort(corr)[::-1][:k]
        grad = np.zeros(n)
        for ki in top_k:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= k
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    return best_h, best_score

            best_score = sc_sub
            best_h = h_sub.copy()
            best_n = n0

    h_bin = round_binary(best_cont_h)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()
        best_n = n0

    h_bin_out, sc_bin_out = binary_sa(h_bin, n_evals=200000, deadline=deadline - 350, seed=seed)
    if sc_bin_out < best_score:
        best_score = sc_bin_out
        best_h = h_bin_out.copy()
        best_n = n0

    if time.time() < deadline - 300:
        h_g, sc_g = binary_greedy(best_h, n_sweeps=30, deadline=deadline - 300)
        if sc_g < best_score:
            best_score = sc_g
            best_h = h_g.copy()
            best_n = n0

    n1 = 2 * n0
    target1 = n1 // 2
    if time.time() < deadline - 250:
        x_old = np.linspace(0.0, 1.0, n0)
        x_new = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(x_new, x_old, best_cont_h)
        h_up = project_feasible(h_up, target1)

        alphas_p3 = [5000, 50000, 500000, 2000000]
        for a in alphas_p3:
            if time.time() > deadline - 120:
                break
            h_new = lbfgs_opt(h_up, target1, a, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_cont_score:
                best_cont_score = sc
                best_cont_h = h_new.copy()
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                best_n = n1
                h_up = h_new.copy()

        h_bin1 = round_binary(h_up)
        sc_b1 = compute_c5(h_bin1)
        if sc_b1 < best_score:
            best_score = sc_b1
            best_h = h_bin1.copy()
            best_n = n1

        if time.time() < deadline - 60:
            h_sa1, sc_sa1 = binary_sa(h_bin1, n_evals=150000, deadline=deadline - 40, seed=seed + 1)
            if sc_sa1 < best_score:
                best_score = sc_sa1
                best_h = h_sa1.copy()
                best_n = n1

    if time.time() < deadline - 80:
        h_ms, sc_ms = multi_start_binary(n0, seed + 100, deadline - 50)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms.copy()
            best_n = n0

    if time.time() < deadline - 30:
        target_cur = len(best_h) // 2
        h_sub2, sc_sub2 = subgrad_descent(best_h, target_cur, 2000, deadline - 10)
        if sc_sub2 < best_score:
            best_score = sc_sub2
            best_h = h_sub2.copy()

    best_h = project_feasible(best_h, len(best_h) // 2)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
```
