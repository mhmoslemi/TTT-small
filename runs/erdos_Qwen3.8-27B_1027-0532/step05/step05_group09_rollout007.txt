import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full')[n-1:] * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def grad_for_lag(h, k, n):
    dx = 2.0 / n
    h_fwd = np.roll(h, -k)
    h_bwd = np.roll(h, k)
    return dx * ((1.0 - h_fwd) - h_bwd)

def fista_optimize(h0, target, alpha, max_iters, deadline):

    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_param = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = 1.0 / max(alpha * 0.05, 1.0)

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y = x + ((t_param - 1.0) / t_new) * (x - x_prev)
        y_c = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y_c, 1.0 - y_c, mode='full')[n-1:] * dx
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        exp_s = np.exp(shifted)
        sum_exp = float(exp_s.sum())
        weights = exp_s / sum_exp
        w_flipped = weights[::-1].copy()
        v = 1.0 - y_c
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(v, fsize)
        W = np.fft.rfft(weights, fsize)
        H = np.fft.rfft(y_c, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (corr_pos - corr_neg)
        obj_y = (c_max + np.log(sum_exp) / alpha)
        x_new = project_feasible(y_c - step * grad, target)
        for _ in range(30):
            corr_n = np.correlate(x_new, 1.0 - x_new, mode='full')[n-1:] * dx
            cm_n = float(np.max(corr_n))
            sh_n = np.clip(alpha * (corr_n - cm_n), -700.0, 0.0)
            se_n = float(np.exp(sh_n).sum())
            obj_n = cm_n + np.log(se_n) / alpha

        t_param = t_new

            best_x = x.copy()
    return best_x, best_obj
