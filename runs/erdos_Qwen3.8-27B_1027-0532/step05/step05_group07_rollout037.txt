   ∂c[k]/∂h[j] = (1-h[j+d]) * [max(0,-d) ≤ j ≤ min(n-1, n-1-d)]
                - h[j-d] * [max(d,0) ≤ j ≤ min(n-1+d, n-1)]
   
   Let me verify: for d=0:
   First term: (1-h[j]) * [0 ≤ j ≤ n-1] = (1-h[j])
   Second term: -h[j] * [0 ≤ j ≤ n-1] = -h[j]
   Total: 1 - 2h[j]
   
   c[n-1] = sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]^2
   ∂/∂h[j] = 1 - 2h[j] ✓
   
   For d=1:
   First term: (1-h[j+1]) * [0 ≤ j ≤ n-2] (since max(0,-1)=0, min(n-1, n-2)=n-2)
   Second term: -h[j-1] * [1 ≤ j ≤ n-1] (since max(1,0)=1, min(n-1+1, n-1)=n-1)
   
   c[n] = sum_{i=0}^{n-2} h[i](1-h[i+1])
   ∂/∂h[0] = (1-h[1]) ✓ (first term, j=0)
   ∂/∂h[n-1] = -h[n-2] ✓ (second term, j=n-1)
   ∂/∂h[j] for 1≤j≤n-2 = (1-h[j+1]) - h[j-1] ✓

   grad[j] = alpha * dx * sum_k weights[k] * ∂c[k]/∂h[j]

   ```python
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       max_c = np.max(corr)
       shifted = alpha * (corr - max_c)
       shifted = np.clip(shifted, -700, 0)
       w = np.exp(shifted)
       w /= w.sum()
       
       # For each lag d (from -(n-1) to (n-1)), compute contribution to gradient
       # d = k - (n-1), so k = d + (n-1)
       # weights are w[d + (n-1)]
       
       grad = np.zeros(n)
       d = np.arange(-(n-1), n)  # length 2n-1
       
       for idx in range(len(d)):
           w_val = w[idx]
           if w_val < 1e-12:
               continue
           d_val = d[idx]
           # Term 1: (1-h[j+d_val]) for j in [max(0,-d_val), min(n-1, n-1-d_val)]
           j1_start = max(0, -d_val)
           j1_end = min(n, n - d_val)
           if j1_start < j1_end:
               grad[j1_start:j1_end] += w_val * (1.0 - h[j1_start+d_val:j1_end+d_val])
           # Term 2: -h[j-d_val] for j in [max(d_val,0), min(n-1+d_val, n-1)]
           j2_start = max(0, d_val)
           j2_end = min(n, n + d_val)
           if j2_start < j2_end:
               grad[j2_start:j2_end] -= w_val * h[j2_start-d_val:j2_end-d_val]
       
       return grad * alpha * dx
   ```

```python
        seg_len = n // 4
        for s in splits:
            start = int(s * n / 4)
            end = min(start + seg_len, n)
            h_new[start:end] += rng.uniform(0.02, 0.06) * rng.choice([-1, 1])
    elif mode == 1:
        mid = n // 2
        h_new[:mid] += rng.uniform(0.03, 0.08)
        h_new[mid:] -= rng.uniform(0.03, 0.08)
    elif mode == 2:
        freq = rng.randint(2, 6)
        phase = rng.uniform(0, 2 * np.pi)
        noise = rng.uniform(0.03, 0.07) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h_new += noise
    elif mode == 3:
        idx = rng.choice(n, size=n // 10, replace=False)
        h_new[idx] += rng.uniform(0.05, 0.15, size=len(idx)) * rng.choice([-1, 1], size=len(idx))
    return project_feasible(h_new, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)
    
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP to establish good local opt
    if tleft() > 200:
        h_p, sc_p = slsqp_polish(best_h, n, target, [200, 800, 3000], 2000, 1e-15)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
    
    # Phase 2: Extended coordinate descent (main workhorse)
    deadline_p2 = start_time + (budget_s - 25.0) * 0.30
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p2, rng,
                                     recompute_interval=35, bisect_iters=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd
    
    # Phase 3: L-BFGS-B with sequential penalty
    if tleft() > 300:
        alpha_steps = [100, 500, 2000, 8000]
        for a in alpha_steps:
            if tleft() < 100:
                break
            for pen in [1000.0, 50000.0]:
                if tleft() < 80:
                    break
                h_new = lfgb_run(best_h, n, target, a, pen, maxiter=600)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new
    
    # Phase 4: SLSQP polish after L-BFGS-B
    if tleft() > 200:
        h_p, sc_p = slsqp_polish(best_h, n, target, [500, 2000, 8000], 2500, 1e-16)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
    
    # Phase 5: More coordinate descent
    deadline_p5 = start_time + (budget_s - 25.0) * 0.60
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p5, rng,
                                     recompute_interval=30, bisect_iters=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd
    
    # Phase 6: Structured perturbations + optimization
    deadline_p6 = start_time + (budget_s - 25.0) * 0.85
    for attempt in range(12):
        if tleft() < 80:
            break
        mode = attempt % 4
        h_pert = structured_perturb(best_h, n, target, rng, mode)
        sc_pert = compute_c5(h_pert)
        
        # Quick coordinate descent from perturbed point
        dl = min(start_time + (budget_s - 25.0) * (0.85 - 0.02 * attempt), 
                 start_time + tleft() - 10)
        h_res, sc_res = paired_coord_desc(h_pert, n, target, dl, rng,
                                            recompute_interval=25, bisect_iters=25)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res
        
        # L-BFGS-B from perturbed point
        if tleft() > 100:
            h_lb = lfgb_run(h_pert, n, target, 2000, 10000.0, maxiter=500)
            sc_lb = compute_c5(h_lb)
            if sc_lb < best_score:
                best_score = sc_lb
                best_h = h_lb
    
    # Phase 7: Final SLSQP polish
    if tleft() > 60:
        h_p, sc_p = slsqp_polish(best_h, n, target, [1000, 5000, 20000], 3000, 1e-17)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
    
    # Phase 8: Final coordinate descent
    if tleft() > 30:
        h_final, sc_final = paired_coord_desc(best_h, n, target,
                                              start_time + budget_s - 15.0, rng,
                                              recompute_interval=20, bisect_iters=35)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final
    
    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
