import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        if time.time() > deadline:
            return h_cur
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def subgradient_attack(h, target, n_iters, n_top, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init

    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]

        grad = np.zeros(n)
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= n_top

        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)

        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.08)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break

    return h_cur, cur_score

    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    total_rounds = max_rounds
    for rnd in range(total_rounds):
        if time.time() - start > timeout_s:
            break
        if rnd % total_rounds * 0.1 == total_rounds * 0.1 or rnd * 10 % total_rounds == 0:
            pass
        frac = rnd / max(total_rounds - 1, 1)
        temperature = t_init * (t_final / t_init) ** frac

        n_ones = len(ones_list)
        n_zeros = len(zeros_list)
        n_tries = min(200, n_ones * n_zeros)
        improved_this_round = False

        for _ in range(n_tries):
            i = ones_list[rng.randint(n_ones)]
            j = zeros_list[rng.randint(n_zeros)]

            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(new_corr.max())
            new_score = new_max * (2.0 / n)
            delta = new_score - cur_score

            if delta < 0 or rng.rand() < np.exp(-delta / (temperature * 0.01)):
                cur_max = new_max
                corr = new_corr
                cur_score = new_score
                ones_list.remove(i)
                zeros_list.append(i)
                zeros_list.remove(j)
                ones_list.append(j)
                improved_this_round = True

                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0

        if not improved_this_round and temperature < 0.02:
            break

    return best_h, best_score


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)


def evaluate_1d(v, h, j, target):
    h_new = h.copy()
    h_new[j] = v
    h_new = project_feasible(h_new, target)
    corr = np.correlate(h_new, 1.0 - h_new, mode='full')
    return float(np.max(corr))


def coordinate_descent(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = evaluate_1d(c, h_cur, j, target)
            fd = evaluate_1d(d, h_cur, j, target)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = evaluate_1d(c, h_cur, j, target)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = evaluate_1d(d, h_cur, j, target)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-8:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def make_random_candidate(n, rng, style):
    target = n / 2.0
    if style == 0:
        h = rng.uniform(0, 1, n)
    elif style == 1:
        h = (rng.uniform(0, 1, n) < 0.5).astype(float)
    elif style == 2:
        period = rng.randint(max(3, n // 8), n // 3)
        duty = rng.uniform(0.35, 0.65)
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % period) < int(period * duty) else 0.0
        h += rng.randn(n) * 0.1
    elif style == 3:
        h = np.zeros(n)
        n_blocks = rng.randint(4, 16)
        block_len = n // n_blocks
        for bidx in range(n_blocks):
            start = bidx * block_len
            end = min(start + block_len, n)
            fill = rng.uniform(0.2, 0.8)
            h[start:end] = (rng.uniform(0, 1, end - start) < fill).astype(float)
    else:
        h = rng.uniform(0, 1, n)
        h = np.where(h > 0.5, 1.0, 0.0).astype(float)
        n_flip = int(n * 0.1)
        idx = rng.choice(n, n_flip, replace=False)
        h[idx] = rng.uniform(0.3, 0.7, n_flip)
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 50.0

    # Phase 1: L-BFGS annealing at base resolution
    alphas_p1 = [5, 20, 100, 500, 2000, 5000, 10000, 30000, 50000, 100000]
    for a in alphas_p1:
        if tleft() < 40:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 400, start_time + budget_s - 30)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Subgradient attack
    if tleft() > 80:
        h_sg, sc_sg = subgradient_attack(best_h, target, 400, 10, 0.004,
                                         start_time + budget_s - 30)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_new = lbfgs_optimize(best_h, target, 50000, mu, 300, start_time + budget_s - 30)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Top-lag equalization
    if tleft() > 60:
        h_eq, sc_eq = top_lag_equalize(best_h, target, 5, 300)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq

    # Phase 4: Binary rounding + hill climbing
    if tleft() > 200:
        h_bin = binary_round(best_h, target)
        base_score = compute_c5(h_bin)
        if base_score < best_score + 0.02:
            h_climb, sc_climb = binary_hill_climb(h_bin, target, 500,
                                                  min(120, tleft() // 2), rng)
            if sc_climb < best_score:
                best_score = sc_climb
                best_h = h_climb.copy()

            # Simulated annealing on binary
            if tleft() > 100:
                h_sa, sc_sa = binary_sa(h_climb, target, 300,
                                        min(80, tleft() // 3), rng)
                if sc_sa < best_score:
                    best_score = sc_sa
                    best_h = h_sa.copy()

            # Continuous polish on best binary solution
            h_cont = best_h.astype(float) + rng.randn(n) * 0.02
            h_cont = project_feasible(h_cont, target)
            for a2 in [5000, 20000, 50000]:
                if tleft() < 40:
                    break
                h_cont = lbfgs_optimize(h_cont, target, a2, mu, 300,
                                        start_time + budget_s - 30)
            sc = compute_c5(h_cont)
            if sc < best_score:
                best_score = sc
                best_h = h_cont

    # Phase 5: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 300:
        h_fine = upsample(best_h, n2)
        h_fine = project_feasible(h_fine, target2)
        alphas_fine = [200, 1000, 5000, 20000, 50000, 100000]
        for a in alphas_fine:
            if tleft() < 40:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, mu, 250,
                                    start_time + budget_s - 30)
        if tleft() > 60:
            h_fine, _ = subgradient_attack(h_fine, target2, 200, 10, 0.003,
                                           start_time + budget_s - 30)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 6: Population-based exploration at current resolution
    if tleft() > 200:
        pop_size = min(6, max(3, int(tleft() // 100)))
        candidates = [best_h.copy()]

        for i in range(pop_size - 1):
            style = i % 5
            h_cand = make_random_candidate(n, rng, style)
            # Blend with best
            alpha_mix = rng.uniform(0.3, 0.7)
            h_cand = alpha_mix * best_h + (1 - alpha_mix) * h_cand
            h_cand = project_feasible(h_cand, target)
            candidates.append(h_cand)

        deadline_pop = start_time + budget_s - 40
        for gen in range(3):
            if time.time() > deadline_pop or tleft() < 60:
                break
            for idx in range(1, len(candidates)):
                if tleft() < 40:
                    break
                h_child = 0.5 * candidates[0] + 0.5 * candidates[idx]
                h_child = project_feasible(h_child, target)
                h_opt = lbfgs_optimize(h_child, target, 5000, mu, 200,
                                       start_time + budget_s - 30)
                h_opt = lbfgs_optimize(h_opt, target, 30000, mu, 200,
                                       start_time + budget_s - 30)
                sc = compute_c5(h_opt)
                if sc < best_score:
                    best_score = sc
                    best_h = h_opt
                    candidates[0] = h_opt

    # Phase 7: Final subgradient + L-BFGS polish
    if tleft() > 40:
        h_sg, sc_sg = subgradient_attack(best_h, target, 300, 8, 0.003,
                                         start_time + budget_s - 20)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 30:
        h_new = lbfgs_optimize(best_h, target, 100000, mu, 400,
                               start_time + budget_s - 15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Coordinate descent (only if n is small enough)
    if tleft() > 120 and n <= 800:
        h_cd, sc_cd = coordinate_descent(best_h, target, 2,
                                         start_time + budget_s - 15)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: Final fine-tuning
    if tleft() > 20:
        h_final_sg, sc_final_sg = subgradient_attack(best_h, target, 150, 5,
                                                     0.002, start_time + budget_s - 10)
        if sc_final_sg < best_score:
            best_score = sc_final_sg
            best_h = h_final_sg
        h_new = lbfgs_optimize(best_h, target, 200000, mu, 300,
                               start_time + budget_s - 5)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
