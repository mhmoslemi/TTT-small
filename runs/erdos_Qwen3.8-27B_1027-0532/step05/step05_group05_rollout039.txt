import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_overlap(h):
    """Compute overlap for lags 0 to n-1"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:]  # lags 0 to n-1

def lse_objective_and_grad(h, alpha, top_m=None, penalty_coeff=None, target_sum=None):
    """
    Compute LSE-smoothed objective and gradient.
    h: array of length n, values in [0,1]
    alpha: smoothing parameter
    top_m: if specified, only consider top-m lags
    penalty_coeff: if specified, add penalty for sum constraint violation
    target_sum: target sum for penalty
    """
    n = len(h)
    dx = 2.0 / n
    over = compute_overlap(h)  # length n
    
    if top_m is not None and top_m < n:
        # Get indices of top-m lags
        top_idx = np.argsort(over)[::-1][:top_m]
        over_sel = over[top_idx]
    else:
        top_idx = np.arange(n)
        over_sel = over
    
    c_max = float(np.max(over_sel))
    shifted = alpha * (over_sel - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    lse_val = c_max + log_sum / alpha
    
    # Gradient of LSE: (1/alpha) * sum_k exp(alpha*(over_k - c_max)) * grad(over_k)
    # But we need to be careful: when top_m is used and c_max changes with h,
    # the derivative is more complex. For simplicity, when top_m is used,
    # we treat the selected set as fixed (valid when the set doesn't change).
    weights = exp_vals / np.sum(exp_vals)  # normalized weights
    
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki = int(ki)
        if ki == 0:
            # over_0 = sum_j h[j]*(1-h[j])
            # grad = (1-2*h[j]) * dx
            grad += wi * (1.0 - 2.0 * h) * dx
        else:
            # over_k = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k])
            # d/dh[i]: (1-h[i+k]) for i < n-k (from h[i]*(1-h[i+k]))
            #           -h[i-k] for i >= k (from (1-h[i])*h[i-k])
            j_pos = np.arange(0, n - ki)
            grad[j_pos] += wi * (1.0 - h[j_pos + ki]) * dx
            j_neg = np.arange(ki, n)
            grad[j_neg] -= wi * h[j_neg - ki] * dx
    
    total_val = lse_val
    if penalty_coeff is not None and target_sum is not None:
        violation = np.sum(h) - target_sum
        total_val += penalty_coeff * violation ** 2
        grad += 2.0 * penalty_coeff * violation
    
    return total_val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgs_optimize(h, n, target, alphas, top_m_schedule, maxiter, tol, deadline, 
                   penalty_start=1.0, penalty_mult=2.0):
    """
    L-BFGS-B optimization with LSE-smoothed objective.
    Uses penalty for sum constraint.
    """
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    pen_coeff = penalty_start
    
    for phase, (alpha, top_m) in enumerate(zip(alphas, top_m_schedule)):
        if time.time() > deadline:
            break
        
        def obj(x):
            val, _ = lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)
            return val
        
        def grad(x):
            _, g = lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-6,
                                  'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                # Reduce penalty (constraint is better satisfied)
                pen_coeff = max(pen_coeff * 0.5, 0.01)
            else:
                # Increase penalty
                pen_coeff = min(pen_coeff * penalty_mult, 1000.0)
        except Exception:
            pass
    
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = compute_overlap(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur) * dx
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k]) * dx
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k] * dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 4.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        over = compute_overlap(h_cur)
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        if k_star == 0:
            grad = (1.0 - 2.0 * h_cur) * dx
        else:
            j_pos = np.arange(0, n - k_star)
            grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
            grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.02)
        else:
            epsilon_temp = eps
            eps *= 0.5
            if eps < 1e-10:
                break
    return h_cur, cur_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    idx_range = np.arange(pop_size)

        if time.time() > deadline:
            break

                scores[i] = score

                    best_score = score

    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlap(h))) * dx

def lse_objective_and_grad(h, alpha, top_m=None, pen_coeff=0.0, target_sum=None):
    n = len(h)
    dx = 2.0 / n
    over = compute_overlap(h)
    
    if top_m is not None and top_m < n:
        top_idx = np.argsort(over)[::-1][:top_m]
        over_sel = over[top_idx]
    else:
        top_idx = np.arange(n)
        over_sel = over
    
    c_max = float(np.max(over_sel))
    shifted = alpha * (over_sel - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = np.sum(exp_vals)
    lse_val = c_max + float(np.log(exp_sum)) / alpha
    
    weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki = int(ki)
        if ki == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        else:
            n_k = n - ki
            grad[:n_k] += wi * (1.0 - h[ki:]) * dx
            grad[ki:] -= wi * h[:n_k] * dx
    
    total_val = lse_val
    if pen_coeff > 0 and target_sum is not None:
        violation = np.sum(h) - target_sum
        total_val += pen_coeff * violation * violation
        grad += 2.0 * pen_coeff * violation
    
    return total_val, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgs_step(h, n, target, alpha, top_m, maxiter=200, ftol=1e-15, pen_coeff=1.0):
    '''Single L-BFGS-B optimization step with given parameters'''
    def obj(x):
        return lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)[0]
    def grad(x):
        return lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)[1]
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                  bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8,
                          'maxfun': maxiter * 5})
    h_new = project_feasible(res.x, target)
    return h_new

def lbfgs_multi(h, n, target, phases, maxiter, deadline):
    '''L-BFGS-B with multiple (alpha, top_m, pen_coeff) phases'''
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha, top_m, pen in phases:
        if time.time() > deadline:
            break
        try:
            h_new = lbfgs_step(h_cur, n, target, alpha, top_m, maxiter, 1e-15, pen)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                # On success, reduce penalty
                pen = max(pen * 0.7, 0.001)
            else:
                # On failure, keep same penalty or increase slightly
                pen = min(pen * 1.3, 100.0)
        except Exception:
            pass
    return h_cur, cur_score

        top_idx = np.argsort(over)[::-1][:n_active]

        h_new = project_feasible(h_new, target)

import numpy as np
from scipy.optimize import minimize
import time


def compute_overlap(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_overlap(h))) * dx


def lse_objective_and_grad(h, alpha, top_m=None, pen_coeff=0.0, target_sum=None):
    n = len(h)
    dx = 2.0 / n
    over = compute_overlap(h)
    if top_m is not None and top_m < n:
        top_idx = np.argsort(over)[::-1][:top_m]
        over_sel = over[top_idx]
    else:
        top_idx = np.arange(n)
        over_sel = over
    c_max = float(np.max(over_sel))
    shifted = alpha * (over_sel - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    exp_sum = float(np.sum(exp_vals))
    lse_val = c_max + float(np.log(exp_sum)) / alpha
    weights = exp_vals / exp_sum
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        ki = int(ki)
        if ki == 0:
            grad += wi * (1.0 - 2.0 * h) * dx
        else:
            n_k = n - ki
            grad[:n_k] += wi * (1.0 - h[ki:]) * dx
            grad[ki:] -= wi * h[:n_k] * dx
    total_val = lse_val
    if pen_coeff > 0 and target_sum is not None:
        violation = np.sum(h) - target_sum
        total_val += pen_coeff * violation * violation
        grad += 2.0 * pen_coeff * violation
    return total_val, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgs_step(h, n, target, alpha, top_m, maxiter=200, pen_coeff=1.0):
    def _obj(x):
        return lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)[0]

    def _grad(x):
        return lse_objective_and_grad(x, alpha, top_m, pen_coeff, target)[1]
    res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9,
                            'maxfun': maxiter * 5})
    return project_feasible(res.x, target)


def lbfgs_multi(h, n, target, phases, maxiter, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha, top_m, pen in phases:
        if time.time() > deadline:
            break
        try:
            h_new = lbfgs_step(h_cur, n, target, alpha, top_m, maxiter, pen)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = compute_overlap(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.004)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (1.0 - 2.0 * h_cur)
            else:
                n_k = n - k
                grad[:n_k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n_k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.015, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

        phases_c = [(30, 40, 1.0), (100, 30, 1.0), (500, 20, 1.0), (3000, 10, 1.0)]
        h_c_opt, sc_c_opt = lbfgs_multi(h_c_best, n_coarse, n_coarse / 2.0, phases_c,
                                        300, min(start_time + 260, hard_deadline))
        h_up = interpolate_to(h_c_opt, n_final, target_final)
        sc_up = compute_c5(h_up)
        phases_up = [(200, 30, 1.0), (1000, 20, 1.0), (5000, 12, 1.0), (20000, 8, 1.0)]
        h_up_opt, sc_up_opt = lbfgs_multi(h_up, n_final, target_final, phases_up,
                                          300, min(start_time + 310, hard_deadline))
        if sc_up_opt < best_score:
            best_score = sc_up_opt
            best_h = h_up_opt

    # Phase 4: Medium resolution n=120
    if tleft() > 150:
        n4 = 120
        t4 = n4 / 2.0
        h4_init = interpolate_to(best_h, n4, t4)
        pop4 = [h4_init.copy()]
        for i in range(9):
            scale = rng.uniform(0.02, 0.1)
            p = h4_init + rng.randn(n4) * scale
            pop4.append(project_feasible(p, t4))
        h4, sc4 = de_optimize(n4, t4, 10, 200, 0.7, 0.6,
                              min(start_time + 380, hard_deadline), rng, init_pop=pop4)
        phases4 = [(50, 30, 1.0), (300, 20, 1.0), (1500, 15, 1.0), (6000, 10, 1.0)]
        h4_opt, sc4_opt = lbfgs_multi(h4, n4, t4, phases4, 250, min(start_time + 420, hard_deadline))
        h4_up = interpolate_to(h4_opt, n_final, target_final)
        sc4_up = compute_c5(h4_up)
        if sc4_up < best_score:
            best_score = sc4_up
            best_h = h4_up
        phases4b = [(500, 20, 1.0), (3000, 15, 1.0), (10000, 10, 1.0)]
        h4b, sc4b = lbfgs_multi(best_h, n_final, target_final, phases4b,
                                300, min(start_time + 470, hard_deadline))
        if sc4b < best_score:
            best_score = sc4b
            best_h = h4b

    # Phase 5: Multiple perturbations + L-BFGS at full res
    if tleft() > 200:
        scales = [0.012, 0.02, 0.03, 0.04, 0.015, 0.025, 0.035, 0.05, 0.018, 0.028]
        max_restarts = min(10, max(1, int(tleft() // 30)))
        for i in range(max_restarts):
            if tleft() < 35:
                break
            h_pert = best_h + rng.randn(n_final) * scales[i]
            h_pert = project_feasible(h_pert, target_final)
            phases_r = [(100, 40, 1.0), (500, 25, 1.0), (2000, 15, 1.0), (8000, 10, 1.0), (30000, 6, 1.0)]
            h_r, sc_r = lbfgs_multi(h_pert, n_final, target_final, phases_r,
                                    250, hard_deadline)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    # Phase 6: Subgradient + high-alpha L-BFGS
    if tleft() > 80:
        h_sg2, sc_sg2 = subgradient_polish(best_h, n_final, target_final,
                                           400, 3, 0.003, 0.996, min(start_time + 720, hard_deadline))
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2
        phases6 = [(5000, 10, 1.0), (20000, 7, 1.0), (60000, 5, 1.0)]
        h_p6, sc_p6 = lbfgs_multi(best_h, n_final, target_final, phases6,
                                  250, min(start_time + 770, hard_deadline))
        if sc_p6 < best_score:
            best_score = sc_p6
            best_h = h_p6

    # Phase 7: Another round of restarts with smaller perturbations
    if tleft() > 100:
        small_scales = [0.008, 0.012, 0.015, 0.02, 0.01, 0.018]
        max_r2 = min(6, max(1, int(tleft() // 30)))
        for i in range(max_r2):
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n_final) * small_scales[i]
            h_pert = project_feasible(h_pert, target_final)
            phases_r2 = [(200, 30, 1.0), (1000, 20, 1.0), (5000, 12, 1.0), (20000, 8, 1.0)]
            h_r2, sc_r2 = lbfgs_multi(h_pert, n_final, target_final, phases_r2,
                                      250, hard_deadline)
            if sc_r2 < best_score:
                best_score = sc_r2
                best_h = h_r2

    # Phase 8: Final subgradient + L-BFGS
    if tleft() > 50:
        h_sg3, sc_sg3 = subgradient_polish(best_h, n_final, target_final,
                                           300, 3, 0.002, 0.997, min(start_time + 850, hard_deadline))
        if sc_sg3 < best_score:
            best_score = sc_sg3
            best_h = h_sg3
        if tleft() > 25:
            phases7 = [(10000, 8, 1.0), (40000, 5, 1.0), (100000, 4, 1.0)]
            h_p7, sc_p7 = lbfgs_multi(best_h, n_final, target_final, phases7,
                                      200, hard_deadline)
            if sc_p7 < best_score:
                best_score = sc_p7
                best_h = h_p7

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
