import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def pnorm_value(h, p):
    corr = compute_corr(h)
    cm = float(np.max(corr))
    if cm < 1e-15:
        return 0.0
    r = corr / cm
    return cm * float(np.sum(r ** p)) ** (1.0 / p)


def pnorm_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cm = float(np.max(corr))
    if cm < 1e-15:
        return np.zeros(n)
    r = corr / cm
    w = r ** (p - 1)
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-15:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    scale = float(np.sum(r ** p)) ** (1.0 / p - 1.0)
    g *= scale * dx
    return g


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-15:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-15:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def pnorm_lbfgsb(h, n, target, p, mu, maxiter, deadline):
    def obj(x):
        return pnorm_value(x, p) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return pnorm_grad(x, p) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h


def lse_lbfgsb(h, n, target, alpha, mu, maxiter, deadline):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_lse(h, n, target, alpha, maxiter, ftol, deadline):
    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_pnorm(h, n, target, p, maxiter, ftol, deadline):
    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: pnorm_value(x, p), h, method='SLSQP',
                       jac=lambda x: pnorm_grad(x, p),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def balance_top_lags(h, n, target, top_m, max_rounds, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        top_vals = corr[top_idx]
        top_mean = float(np.mean(top_vals))
        top_max = float(top_vals[0])
        if top_max - top_mean < 1e-8:
            break
        alpha = 5000.0
        shifted = np.clip(alpha * (corr[top_idx] - top_max), -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        for wk, m in zip(w, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        epsilon = 0.005
        h_new = h_cur - epsilon * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
            epsilon = min(epsilon * 1.01, 0.02)
        else:
            epsilon *= 0.8
            if epsilon < 1e-8:
                break
    return h_cur, best_score


def coord_subgrad_descent(h, n, target, n_sweeps, lr_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
            lr = min(lr * 1.005, 0.01)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    return h_cur, best_score


def multi_restart(h_base, n, target, n_restarts, scales, deadline, seed):
    rng = np.random.RandomState(seed)
    best_h = h_base.copy()
    best_score = compute_c5(h_base)
    for r in range(n_restarts):
        if time.time() > deadline:
            break
        scale = scales[r % len(scales)]
        h_pert = h_base + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)

        sc = compute_c5(h_opt)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
            h_opt2 = slsqp_lse(best_h, n, target, 5000, 2000, 1e-15, deadline)
            sc2 = compute_c5(h_opt2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_opt2

            best_score = sc_final_sg
            best_h = h_final_sg

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
