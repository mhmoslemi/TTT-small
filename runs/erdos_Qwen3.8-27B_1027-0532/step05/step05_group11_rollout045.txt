            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_exact(h, target):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += diff / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def subgradient_at(h, k, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        m = n - k
        grad[:m] += (1.0 - h[k:]) * dx
        grad[k:] += (-h[:m]) * dx
    else:
        m = -k
        grad[m:] += (1.0 - h[:n-m]) * dx
        grad[:n-m] += (-h[m:]) * dx
    return grad
