import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def autocorr(h):
    """Compute autocorrelation: R_k = dx * sum_{j=0}^{n-1-k} h[j]*h[j+k]"""
    n = len(h)
    dx = 2.0 / n
    R = np.zeros(n)
    for k in range(n):
        R[k] = dx * np.dot(h[:n-k], h[k:])
    return R

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    best_R_min = float(np.min(autocorr(h_cur)))
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        
        R = autocorr(h_cur)

        # = minimize -min_k R_k (smoothed)

        neg_alpha_R -= np.max(neg_alpha_R)  # for numerical stability

def compute_corr(h):

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:

        else:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):

def autocorr_linear(h):
    n = len(h)
    full = np.correlate(h, h, mode='full')
    return full[n-1:n-1+n]


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break

            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def build_h_fourier(params, n):
    m = len(params) // 2
    j = np.arange(n)
    h = np.full(n, 0.5)
    for l in range(1, m + 1):
        a = params[2 * (l - 1)]
        b = params[2 * (l - 1) + 1]
        h += a * np.cos(2.0 * np.pi * l * j / n) + b * np.sin(2.0 * np.pi * l * j / n)
    return h

        h_four2, sc_four2 = fourier_de(n_final, target_final, 12, 25, 120,
                                       min(start_time + 90, hard_deadline), rng)
        if sc_four2 < best_score:
            best_score = sc_four2
            best_h = h_four2.copy()

    # Phase 2: Newton on best solution (120s)
    if tleft() > 130:
        h_new, sc_new = newton_method(best_h, n_final, target_final, 80, 1500.0,
                                      min(start_time + 210, hard_deadline))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    # Phase 3: BB momentum (100s)
    if tleft() > 110:
        h_bb, sc_bb = bb_momentum(best_h, n_final, target_final, 3000, 5,
                                  min(start_time + 310, hard_deadline))
        if sc_bb < best_score:
            best_score = sc_bb
            best_h = h_bb.copy()

    # Phase 4: SLSQP high-alpha polish (180s)
    if tleft() > 100:
        h_ref = slsqp_optimize(best_h, n_final, target_final,
                               [800, 3000, 10000, 30000], 2500,
                               min(start_time + 490, hard_deadline))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Multi-resolution exploration (150s)
    if tleft() > 160:
        for n_try in [300, 500]:
            if tleft() < 80:
                break
            t_try = n_try / 2.0
            x_old = np.linspace(0, 1, n_final)
            x_new = np.linspace(0, 1, n_try)
            h_try = project_feasible(np.interp(x_new, x_old, best_h), t_try)
            pop = [h_try.copy()]
            for _ in range(14):
                scale = rng.uniform(0.02, 0.08)
                p = h_try + rng.randn(n_try) * scale
                pop.append(project_feasible(p, t_try))
            scores = np.array([compute_c5(p) for p in pop])
            for gen in range(60):
                if time.time() > hard_deadline - 20:
                    break
                for i in range(len(pop)):
                    a, b, c = rng.choice([j for j in range(len(pop)) if j != i],
                                         size=3, replace=False)
                    v = pop[a] + 0.7 * (pop[b] - pop[c])
                    r = rng.rand(n_try)
                    trial = v.copy()
                    trial[r > 0.6] = pop[i][r > 0.6]
                    trial = project_feasible(trial, t_try)
                    sc_t = compute_c5(trial)
                    if sc_t < scores[i]:
                        pop[i] = trial
                        scores[i] = sc_t
                    if sc_t < best_score:
                        best_score = sc_t
                        h_up = project_feasible(
                            np.interp(x_old, x_new, pop[i]), target_final)
                        sc_up = compute_c5(h_up)
                        if sc_up < best_score:
                            best_score = sc_up
                            best_h = h_up.copy()
            bi = int(np.argmin(scores))
            h_up = project_feasible(np.interp(x_old, x_new, pop[bi]), target_final)
            sc_up = compute_c5(h_up)
            if sc_up < best_score:
                best_score = sc_up
                best_h = h_up.copy()

    # Phase 6: Newton again after multi-res (80s)
    if tleft() > 90:
        h_new, sc_new = newton_method(best_h, n_final, target_final, 120, 2000.0,
                                      min(start_time + 730, hard_deadline))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    # Phase 7: SLSQP final (80s)
    if tleft() > 70:
        h_ref = slsqp_optimize(best_h, n_final, target_final,
                               [5000, 15000, 40000], 2000,
                               min(start_time + 810, hard_deadline))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: BB + Newton interleaved (60s)
    if tleft() > 60:
        h_bb, sc_bb = bb_momentum(best_h, n_final, target_final, 2000, 3,
                                  min(start_time + 870, hard_deadline))
        if sc_bb < best_score:
            best_score = sc_bb
            best_h = h_bb.copy()
        h_new, sc_new = newton_method(best_h, n_final, target_final, 100, 3000.0,
                                      min(start_time + 930, hard_deadline))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    # Phase 9: Structured perturbation restart (60s)
    if tleft() > 60:
        for attempt in range(6):
            if tleft() < 20:
                break
            freq = rng.choice([3, 5, 7, 11, 13, 17])
            amp = rng.uniform(0.005, 0.02)
            phase = rng.uniform(0, 2 * np.pi)
            j = np.arange(n_final)
            mod = best_h + amp * np.cos(2 * np.pi * freq * j / n_final + phase)
            mod = project_feasible(mod, target_final)
            h_new, sc_new = newton_method(mod, n_final, target_final, 50, 2000.0,
                                          min(start_time + 990, hard_deadline))
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()

    # Phase 10: Final squeeze (remaining)
    if tleft() > 20:
        h_ref = slsqp_optimize(best_h, n_final, target_final,
                               [10000, 30000, 60000], 1500, hard_deadline - 5)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 10:
        h_new, sc_new = newton_method(best_h, n_final, target_final, 50, 50000.0,
                                      hard_deadline)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
