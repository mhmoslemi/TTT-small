def binary_sa(h_init, n, max_iters=100000, t_start=1.0, t_end=0.01, time_limit=60):
    h = h_init.copy()
    # Make binary
    h[h > 0.5] = 1
    h[h <= 0.5] = 0
    # Fix sum
    target = n // 2
    while np.sum(h) > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.randint(len(idx))]] = 0
    while np.sum(h) < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.randint(len(idx))]] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = np.max(corr)
    best_h = h.copy()
    best_c5 = current_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (it / max_iters)
        
        # Pick random swap
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        if i == j:
            continue
        
        # Compute delta
        old_corr_max = current_c5
        h[i] = 0
        h[j] = 1
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(corr_new)
        
        delta = new_c5 - old_corr_max
        if delta < 0 or rng.random() < np.exp(-delta / (t * dx)):
            # Accept
            current_c5 = new_c5
            corr = corr_new
            # Update ones and zeros lists
            ones[i_idx] = j
            zeros[j_idx] = i
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5 * dx

def full_gradient(h, alpha, dx, top_m=None):

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_m:]
    else:
        top_idx = np.arange(2 * n - 1)
    shifted = np.clip(alpha * (corr[top_idx] - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        return np.zeros(n), c_max * dx
    weights = exp_vals / wsum

def projected_gradient_descent(h_init, n, target, max_iters, time_limit, start_time,
                                top_m=50, alpha=100, momentum=0.9):

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    stale = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if top_m is not None:
            grad, _ = full_gradient(h, alpha, dx, top_m=top_m)
        else:
            grad, _ = full_gradient(h, alpha, dx)
        v = momentum * v + (1.0 - momentum) * grad
        gnorm = np.max(np.abs(v))
        if gnorm < 1e-14:
            break
        lr = 1.0 / (alpha * gnorm + 1e-8)
        lr = min(lr, 3.0)
        improved = False
        for _ in range(30):

            h_try = project_feasible(h_try, target)

                best_score = score
                best_h = h_try.copy()
                h = h_try
                improved = True
                stale = 0
                break
            lr *= 0.4
        if not improved:
            stale += 1
            if stale > 80:
                break
        alpha *= 1.002
        if alpha > 50000:
            alpha = 50000
    return best_h, best_score

def lbfgs_optimize(h_init, n, target, alphas, penalty=100.0, maxiter=500, time_limit=None, start_time=0):
    h = project_feasible(h_init.copy(), target)
    bounds = [(0.0, 1.0)] * n

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30,
                   damping=0.7, reg=1e-4):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5


def golden_ratio_pattern(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        val = (i * phi) % 1.0
        h[i] = 1.0 if val < 0.5 else 0.0
    return h


def qr_pattern(n):
    # Find smallest prime p >= n+1
    p = n + 1
    while not _is_prime(p):
        p += 1
    h = np.zeros(n)
    if p > n:
        for i in range(n):
            val = (i + 1) % p
            if val == 0:
                continue
            if _is_quadratic_residue(val, p):
                h[i] = 1.0
    s = np.sum(h)
    if s > n / 2:
        ones = np.where(h == 1)[0]
        np.random.shuffle(ones)
        for idx in ones[:int(s - n / 2)]:
            h[idx] = 0
    elif s < n / 2:
        zeros = np.where(h == 0)[0]
        np.random.shuffle(zeros)
        for idx in zeros[:int(n / 2 - s)]:
            h[idx] = 1
    return h


def _is_prime(n):
    if n < 2:
        return False
    if n < 4:
        return True
    if n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True


def _is_quadratic_residue(a, p):
    return pow(a, (p - 1) // 2, p) == 1


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        # Quick gradient
        h_quick, c5_quick = projected_gradient_descent(
            h_try, n, target, max_iters=2000,
            time_limit=start_time + 40, start_time=start_time,
            top_m=20, alpha=30, momentum=0.8)
        # Quick L-BFGS
        h_lq = lbfgs_optimize(h_quick, n, target,
                              alphas=[50, 200, 1000],
                              penalty=100.0, maxiter=200,
                              time_limit=start_time + 55, start_time=start_time)
        sc = compute_c5(h_lq)
        if sc < best_score:
            best_score = sc
            best_h = h_lq

    # Phase 5: Alternating LP (150s)
    if tleft() > 120:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=3000,
                                     time_limit=start_time + 130,
                                     start_time=start_time, deadline_offset=60,
                                     damping=0.6, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: Final gradient + L-BFGS polish
    if tleft() > 60:
        h_pg, c5_pg = projected_gradient_descent(
            best_h, n, target, max_iters=3000,
            time_limit=start_time + 40, start_time=start_time,
            top_m=20, alpha=100, momentum=0.85)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg
    if tleft() > 40:
        h_fin = lbfgs_optimize(best_h, n, target,
                               alphas=[200, 1000, 5000, 20000],
                               penalty=100.0, maxiter=300,
                               time_limit=start_time + 35, start_time=start_time)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Phase 7: Try slightly different resolution
    if tleft() > 100:
        for n_try in [n * 2, n + 50]:
            if tleft() < 50:
                break
            target_try = n_try // 1.0
            target_try = n_try / 2.0
            if target_try != int(target_try):
                continue
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_quick, c5_q = projected_gradient_descent(
                h_interp, n_try, target_try, max_iters=1000,
                time_limit=start_time + 30, start_time=start_time,
                top_m=20, alpha=50, momentum=0.8)
            h_lq = lbfgs_optimize(h_quick, n_try, target_try,
                                  alphas=[50, 200, 1000],
                                  penalty=50.0, maxiter=150,
                                  time_limit=start_time + 45, start_time=start_time)
            sc = compute_c5(h_lq)
            if sc < best_score:
                best_score = sc
                best_h = h_lq
                n = n_try
                target = target_try

    # Phase 8: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        global _global_n, _global_target
        _global_n = n_use
        _global_target = target_use
        bounds = [(0.0, 1.0)] * n_use

        def _con_fun(x):
            return np.sum(x) - _global_target

        def _con_jac(x):
            return np.ones(_global_n)

        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for alpha_slsqp in [100, 500, 2000]:
            if tleft() < 15:
                break
            try:
                res = minimize(smooth_max_val(np.correlate(best_h, 1.0 - best_h, mode='full'), alpha_slsqp),
                               best_h, args=(),
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 500, 'ftol': 1e-18})
                # This won't work well since we need h-dependent corr
                pass
            except Exception:
                pass

    # Use L-BFGS as final polish instead of SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = lbfgs_optimize(best_h, n_use, target_use,
                                     alphas=[1000, 5000, 20000],
                                     penalty=200.0, maxiter=500,
                                     time_limit=start_time + 15, start_time=start_time)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
