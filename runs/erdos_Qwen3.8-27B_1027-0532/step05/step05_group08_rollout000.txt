def project_polytope(v, target):
    lo = -float(np.max(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v + lam, 0.0, 1.0)

def lse_grad(h, alpha, top_m=None):
    n = len(h)
    corr = compute_corr(h)
    if top_m is not None and top_m < len(corr):
        order = np.argsort(corr)[::-1]
        active_idx = order[:top_m]
    else:
        active_idx = np.arange(len(corr))
    active_corr = corr[active_idx]
    am = float(np.max(active_corr))
    shifted = np.clip(alpha * (active_corr - am), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones_like(w) / len(w)
    else:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, active_idx):
        if wk < 1e-12:
            continue
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n-k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n-ell])
            g[:n-ell] -= wk * h[ell:]
    g *= dx
    return g

def lse_value(h, alpha, top_m=None):
    corr = compute_corr(h)
    if top_m is not None and top_m < len(corr):
        order = np.argsort(corr)[::-1]
        active = corr[order[:top_m]]
    else:
        active = corr
    am = float(np.max(active))
    shifted = np.clip(alpha * (active - am), -700.0, 0.0)
    return am + float(np.log(np.sum(np.exp(shifted)))) / alpha

def fista_optimize(h_init, n, target, alpha, maxiter, top_m=None, deadline=None, momentum_decay=False):
    h = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    t = 1.0
    gamma = 1.0 / (alpha * 8.0 / n)
    gamma = min(gamma, 0.5)
    stale = 0
    for k in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        if momentum_decay:
            beta = (t - 1.0) / t_new
            beta = min(beta, 0.95)
        else:
            beta = (t - 1.0) / t_new
        y = h + beta * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        g = lse_grad(y, alpha, top_m=top_m)
        h_new = project_polytope(y - gamma * g, target)
        f_new = lse_value(h_new, alpha, top_m=top_m)
        f_y = lse_value(y, alpha, top_m=top_m)
        if f_new > f_y + 1e-6:
            gamma *= 0.5
            if gamma < 1e-10:
                gamma = 1e-10
            h_new = project_polytope(y - gamma * g, target)
        h_prev = h.copy()
        h = h_new
        t = t_new
        sc = compute_c5(h)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
        if stale > 800:
            break
    return best_h, best_score

def slsqp_round(h, n, target, alpha, maxiter, ftol, top_m=None):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    def obj(x):
        return lse_value(x, alpha, top_m=top_m)
    def jac(x):
        return lse_grad(x, alpha, top_m=top_m)
    try:
        res = minimize(obj, h, method='SLSQP', jac=jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_polytope(res.x, target)
    except Exception:
        return h

def lbfgsb_round(h, n, target, alpha, mu, maxiter, top_m=None):
    def obj(x):
        return lse_value(x, alpha, top_m=top_m) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha, top_m=top_m) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9})
        return project_polytope(res.x, target)
    except Exception:
        return h

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n-k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n-k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n-ell])
            g[:n-ell] -= h_cur[ell:]
        g *= dx
        h_new = project_polytope(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.01)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score

def multi_top_surgery(h, n, target, max_steps, top_k, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for m in top_idx:
            k = int(m) - (n - 1)
            if k >= 0:
                g[:n-k] += (1.0 - h_cur[k:])
                g[k:] -= h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += (1.0 - h_cur[:n-ell])
                g[:n-ell] -= h_cur[ell:]
        g *= dx
        h_new = project_polytope(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.008)
        else:
            eps *= 0.85
            if eps < 1e-10:
                break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_polytope(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_polytope(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 10.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_polytope(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA with continuation at original size
    alphas_p1 = [20, 50, 100, 300, 1000, 3000, 8000, 20000, 50000]
    for alpha in alphas_p1:
        if time.time() > deadline - 400:
            break
        rem = deadline - 400 - time.time()
        maxiter = min(5000, max(500, int(rem * 200)))
        h_new, sc = fista_optimize(best_h, n, target, alpha, maxiter, top_m=None, deadline=deadline - 400)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: L-BFGS-B refinement
    if time.time() < deadline - 200:
        for alpha in [200, 1000, 5000]:
            if time.time() > deadline - 200:
                break
            for mu in [10.0, 100.0]:
                h_new = lbfgsb_round(best_h, n, target, alpha, mu, 300, top_m=10)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

    # Phase 3: SLSQP polish
    if time.time() < deadline - 150:
        for alpha in [5000, 20000]:
            if time.time() > deadline - 150:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 1500, 1e-15, top_m=8)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Multi-top surgery
    if time.time() < deadline - 100:
        for top_k in [2, 3, 5, 8]:
            if time.time() > deadline - 80:
                break
            h_surg, sc_surg = multi_top_surgery(best_h, n, target, 500, top_k, 0.002, deadline - 80)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 5: Interpolate to higher resolution n=800
    n2 = 800
    target2 = n2 / 2.0
    if time.time() < deadline - 300:
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)

        alphas_p5 = [20, 50, 200, 500, 2000, 8000, 30000]
        for alpha in alphas_p5:
            if time.time() > deadline - 150:
                break
            rem = deadline - 150 - time.time()
            maxiter = min(4000, max(300, int(rem * 150)))
            h2_new, sc2 = fista_optimize(best_h2, n2, target2, alpha, maxiter, top_m=None, deadline=deadline - 150)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

        if time.time() < deadline - 100:
            for alpha in [1000, 5000]:
                if time.time() > deadline - 100:
                    break
                for mu in [50.0, 200.0]:
                    h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 200, top_m=10)
                    sc2 = compute_c5(h2_new)
                    if sc2 < best_score2:
                        best_score2 = sc2
                        best_h2 = h2_new

        if time.time() < deadline - 80:
            h_surg2, sc_surg2 = multi_top_surgery(best_h2, n2, target2, 400, 3, 0.001, deadline - 80)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2

        if time.time() < deadline - 60:
            h_surg2, sc_surg2 = worst_lag_surgery(best_h2, n2, target2, 300, 0.001, deadline - 60)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 6: Structured perturbations at best resolution
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        # Try sinusoidal perturbations
        for freq in [3, 5, 7, 11, 13]:
            if time.time() > deadline - 60:
                break
            x = np.linspace(0, 2*np.pi*freq, n_cur)
            for amp in [0.03, 0.06, 0.1]:
                h_pert = best_h + amp * np.sin(x)
                h_pert = project_polytope(h_pert, target_cur)
                h_ref, sc_ref = fista_optimize(h_pert, n_cur, target_cur, 5000, 2000, top_m=8, deadline=deadline - 60)
                if sc_ref < best_score:
                    best_score = sc_ref
                    best_h = h_ref

    # Phase 7: Random restarts
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        for attempt in range(3):
            if time.time() > deadline - 30:
                break
            h_pert = best_h + rng.randn(n_cur) * 0.08
            h_pert = project_polytope(h_pert, target_cur)
            h_ref, sc_ref = fista_optimize(h_pert, n_cur, target_cur, 10000, 1500, top_m=10, deadline=deadline - 30)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref
                h_ref2 = slsqp_round(best_h, n_cur, target_cur, 20000, 1000, 1e-15, top_m=6)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2

    # Phase 8: Final FISTA with very high alpha + surgery
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, sc_new = fista_optimize(best_h, n_cur, target_cur, 100000, 2000, top_m=5, deadline=deadline - 15)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 9: Final surgery
    if time.time() < deadline - 5:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.0005, deadline - 5)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 10: Final SLSQP
    if time.time() < deadline - 2:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_round(best_h, n_cur, target_cur, 50000, 500, 1e-16, top_m=4)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_polytope(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
