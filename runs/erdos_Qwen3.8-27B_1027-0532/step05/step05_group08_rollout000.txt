import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    w = np.exp(shifted)
    w_sum = w.sum()
    w = w / w_sum
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2*n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n-k]
        else:
            ell = -k
            g[ell:] += wm * (1.0 - h[:n-ell])
            g[:n-ell] -= wm * h[ell:]
    g *= dx
    return val, g

def topk_sum_obj(h, k_top):

    top_vals = np.sort(corr)[::-1][:k_top]
    return float(np.sum(top_vals))

    top_idx = np.argsort(corr)[::-1][:k_top]

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def obj(x):
        return lse_objective_and_grad(x, alpha)[0]
    def grad(x):
        return lse_objective_and_grad(x, alpha)[1]
    bounds = [(0.0, 1.0)] * n
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        v, _ = lse_objective_and_grad(x, alpha)
        return v + 0.5 * mu * (np.sum(x) - target)**2
    def objg(x):
        _, g = lse_objective_and_grad(x, alpha)
        return g + mu * (np.sum(x) - target) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

        # Minimize: t (i.e., [0, 0, ..., 0, 1] @ x)

        #   For each active k: grad_k @ h + offset_k <= t
        #   i.e., grad_k @ h - t <= -offset_k
        #   0 <= h_i <= 1
        #   sum(h) = target

        # Actually: c_k(h) ≈ c_k(h_cur) + grad_k @ (h - h_cur)
        #          = grad_k @ h + (c_k(h_cur) - grad_k @ h_cur)
        # So: grad_k @ h + offset_k <= t
        #    grad_k @ h - t <= -offset_k
        
        A_ub_rows = []
        b_ub = []
        
        for m_idx in top_m:
            m_idx = int(m_idx)

            g = np.zeros(n)
            if k >= 0:
                g[:n-k] += (1.0 - h_cur[k:])

                ell = -k
                g[ell:] += (1.0 - h_cur[:n-ell])
                g[:n-ell] -= h_cur[ell:]
            g *= dx
            
            offset = corr[m_idx] - g @ h_cur
            
            # Constraint: g @ h - t <= -offset
            row = np.concatenate([g, [-1.0]])
            A_ub_rows.append(row)
            b_ub.append(-offset)

        A_eq = np.zeros((1, N))
        A_eq[0, :n] = 1.0

        bounds = [(0, 1)] * n + [(0, None)]

        c_obj = np.zeros(N)
        c_obj[n] = 1.0
        
        try:
