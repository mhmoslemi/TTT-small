def compute_lags(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]

def compute_c5(h):
    return float(np.max(compute_lags(h))) * (2.0 / len(h))

def lag_gradient(h, k):
    """Compute gradient of over[k] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # over[k] = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
    # d/dh[j] = (1-h[j+k]) if j <= n-1-k  (j is the "first" element)
    #         + (-h[j-k]) if j >= k         (j is the "second" element)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        # j is "first" element: j from 0 to n-1-k
        grad[0:n-k] += (1.0 - h[k:n])
        # j is "second" element: j from k to n-1
        grad[k:n] -= h[0:n-k]
    return grad

def subgradient_step(h, k_active, lr):
    """Take a subgradient step for the max of active lags."""
    n = len(h)
    grad = np.zeros(n)
    for k in k_active:
        grad += lag_gradient(h, k)
    grad /= len(k_active)
    return h - lr * grad

def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}."""
    h = h.copy()

        if cnt == 0:

    n = len(h_init)
    bounds = [(0, 1)] * n

        h = project_feasible(x, target_sum)
        lags = compute_lags(h)
        # Use the largest alpha for the final evaluation
        a = _state['alpha']
        return soft_smooth_max(lags, a)
    
    def obj_grad(x):
        h = project_feasible(x, target_sum)
        lags = compute_lags(h)
        a = _state['alpha']
        # Gradient of log-sum-exp
        shifted = a * (lags - np.max(lags))
        shifted = np.clip(shifted, -700, 0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for k, w in enumerate(weights):
            if w > 1e-10:
                grad += w * lag_gradient(h, k)
        return a * grad
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

        try:

    n = len(h_cont)

            # Undo
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            break

    """Gradient of corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) w.r.t. h[j]."""
