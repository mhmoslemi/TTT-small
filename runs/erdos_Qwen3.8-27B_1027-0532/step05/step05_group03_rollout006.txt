import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:])) * dx  # only relevant lags

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_qr_init(n, p):
    """Create initial binary vector based on quadratic residues mod p"""
    h = np.zeros(n)
    # QRs mod p
    qrs = set()
    for x in range(1, p):
        qrs.add((x*x) % p)
    # Map positions to QRs
    for i in range(n):
        val = (i+1) % p
        if val in qrs:
            h[i] = 1.0
    h = project_feasible(h, n/2.0)
    return h

def make_random_binary(n, rng):
    """Create random binary vector with n/2 ones"""
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def sa_bangbang(h_init, n_steps, T_init, T_final, alpha_cool, rng, verbose=False):
    """Simulated annealing on bang-bang vectors"""
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h[remove] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h[add] = 1
    
    dx = 2.0 / n
    corr = compute_corr(h)[n-1:]  # relevant part, length n
    best_h = h.copy()
    best_corr_max = np.max(corr)
    
    ones_idx = set(np.where(h == 1)[0].tolist())
    zeros_idx = set(np.where(h == 0)[0].tolist())
    
    T = T_init
    for step in range(n_steps):
        T = T_init * (T_final / T_init) ** (step / max(n_steps-1, 1))
        
        # Pick random swap
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = rng.choice(list(ones_idx))
        j = rng.choice(list(zeros_idx))
        i, j = int(i), int(j)
        
        # Compute change in corr for all k
        # h[i]: 1->0, h[j]: 0->1
        delta = np.zeros(n)
        for k in range(n):
            d = 0.0
            # Position i becomes 0:
            # - term at x=i: was 1*(1-h[i+k]), now 0. Change: -(1-h[i+k]) if i+k < n
            if i + k < n:
                d -= (1.0 - h[i+k])
            # - term at x=i-k (where x+k=i): was h[i-k]*(1-1)=0, now h[i-k]*(1-0)=h[i-k]. Change: +h[i-k] if i-k >= 0
            if i - k >= 0:
                d += h[i-k]
            # Position j becomes 1:
            # + term at x=j: was 0, now 1*(1-h[j+k]). Change: +(1-h[j+k]) if j+k < n
            if j + k < n:
                d += (1.0 - h[j+k])
            # + term at x=j-k: was h[j-k]*(1-0)=h[j-k], now h[j-k]*(1-1)=0. Change: -h[j-k] if j-k >= 0
            if j - k >= 0:
                d -= h[j-k]
            delta[k] = d
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        old_max = np.max(corr)
        delta_cost = new_max * dx - old_max * dx
        
        if delta_cost <= 0 or rng.random() < np.exp(-delta_cost / max(T, 1e-10)):
            # Accept
            h[i] = 0
            h[j] = 1
            ones_idx.remove(i)
            ones_idx.add(j)
            zeros_idx.add(i)
            zeros_idx.remove(j)
            corr = new_corr
            if new_max < best_corr_max:
                best_corr_max = new_max
                best_h = h.copy()
        
        if verbose and step % 100000 == 0:
            print(f"Step {step}, T={T:.6f}, best={best_corr_max*dx:.6f}")
    
    return best_h, best_corr_max * dx

    h = project_feasible(h, n / 2.0)
    return h

def sa_bangbang(h_init, n_steps, T_init, T_final, rng, time_limit=None, start_time=None, directed_frac=0.0, n_worst=3):
    """Simulated annealing on bang-bang vectors with swap moves"""
    n = len(h_init)
    h = (h_init > 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target_ones:
        ones_idx = np.where(h == 1)[0]
        remove = rng.choice(ones_idx, n_ones - target_ones, replace=False)
        h[remove] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h == 0)[0]
        add = rng.choice(zeros_idx, target_ones - n_ones, replace=False)
        h[add] = 1
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    best_h = h.copy()
    best_max = float(np.max(corr))
    
    ones_pos = np.where(h == 1)[0].copy()
    zeros_pos = np.where(h == 0)[0].copy()
    
    for step in range(n_steps):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        
        T = T_init * (T_final / T_init) ** (step / max(n_steps - 1, 1))
        
        # Choose swap
        if directed_frac > 0 and rng.random() < directed_frac:
            # Directed: focus on worst lag
            k_star = int(np.argmax(corr))
            # Find good i: h[i]=1 and h[i+k_star]=0
            # i+k_star must be a valid position with h=0
            valid_range = min(n - k_star, n)
            cand_i = np.arange(0, valid_range)
            cand_i = cand_i[(h[cand_i] == 1) & (h[cand_i + k_star] == 0)]
            if len(cand_i) > 0:
                i = int(cand_i[rng.randint(len(cand_i))])
            else:
                i = int(ones_pos[rng.randint(len(ones_pos))])
            # j: random from zeros (or directed)
            j = int(zeros_pos[rng.randint(len(zeros_pos))])
            if j == i:
                j = int(zeros_pos[rng.randint(len(zeros_pos))])
        else:
            i = int(ones_pos[rng.randint(len(ones_pos))])
            j = int(zeros_pos[rng.randint(len(zeros_pos))])
        
        # Compute delta
        delta = np.zeros(n)
        # h[i]: 1 -> 0
        ln_i = n - i  # number of k values where i+k < n
        if ln_i > 0:
            delta[:ln_i] -= (1.0 - h[i:i+ln_i])
        if i >= 0:
            delta[:i+1] += h[i::-1]
        # h[j]: 0 -> 1
        ln_j = n - j
        if ln_j > 0:
            delta[:ln_j] += (1.0 - h[j:j+ln_j])
        if j >= 0:
            delta[:j+1] -= h[j::-1]
        
        new_max = float(np.max(corr + delta))
        old_max = float(np.max(corr))
        dcost = (new_max - old_max) * dx
        
        if dcost <= 0 or rng.random() < np.exp(-dcost / max(T, 1e-12)):
            h[i] = 0
            h[j] = 1
            ones_pos = np.delete(ones_pos, np.where(ones_pos == i)[0][0])
            ones_pos = np.append(ones_pos, j)
            zeros_pos = np.delete(zeros_pos, np.where(zeros_pos == j)[0][0])
            zeros_pos = np.append(zeros_pos, i)
            corr = corr + delta
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
    
    return best_h, best_max * dx
