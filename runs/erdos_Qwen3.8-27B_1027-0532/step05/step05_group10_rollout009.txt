import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def alternating_lp_sparse(h_init, n, target, max_iter, time_limit,
                          damping_init=0.8, reg=1e-4, n_active=30):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=80000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=80000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.92)
        else:
            damping = min(0.95, damping + 0.02)

        if it % 80 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.8)

    return best_h, best_c5


def binary_to_feasible(h, n, target):
    n_ones = int(round(target))
    idx_sorted = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx_sorted[:n_ones]] = 1.0
    return h_bin


def binary_anneal(h, n, target, max_steps, time_limit, T_init=0.005, T_min=0.0001):
    start = time.time()
    rng_local = np.random.RandomState(12345)
    h_bin = binary_to_feasible(h, n, target)
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur_h = h_bin.copy()
    cur_c5 = best_c5

    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(1, max_steps))

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        T = max(T_min, T * cooling)

        ones_idx = np.where(cur_h == 1.0)[0]
        zeros_idx = np.where(cur_h == 0.0)[0]

        i = ones_idx[rng_local.randint(len(ones_idx))]
        j = zeros_idx[rng_local.randint(len(zeros_idx))]

        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)

        delta = new_c5 - cur_c5
        if delta < 0 or rng_local.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1.0
            cur_h[j] = 0.0

    return best_h, best_c5

            ones_idx = np.where(cur_h == 1.0)[0]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        idx_valid1 = (0 <= idxs - k_star) & ((idxs - k_star) < n)
        idx_valid2 = (0 <= idxs + k_star) & ((idxs + k_star) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k_star])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k_star]]
        grad *= dx

        improved = False
        for lr in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25, 0.4]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5
