```python
def compute_gradient_h(h, weights):
    """Compute gradient of sum_k weights[k] * c[k] w.r.t. h.
    weights has length 2n-1, where weights[k+n-1] corresponds to shift k.
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[k + n - 1]
        if w == 0:
            continue
        # ∂c[k]/∂h[i] = (1-h[i+k]) - h[i-k] for valid i
        # i+k valid: 0 <= i+k < n => -k <= i < n-k
        # i-k valid: 0 <= i-k < n => k <= i < n+k
        # i valid: 0 <= i < n
        i_min = max(0, -k, k)
        i_max = min(n, n-k, n+k)
        if i_min >= i_max:
            continue
        i_range = np.arange(i_min, i_max)
        # (1-h[i+k]) part: valid when 0<=i+k<n
        valid_plus = (i_range + k >= 0) & (i_range + k < n)
        valid_minus = (i_range - k >= 0) & (i_range - k < n)
        grad[i_range[valid_plus]] += w * (1 - h[i_range[valid_plus] + k])
        grad[i_range[valid_minus]] += -w * h[i_range[valid_minus] - k]
    return grad

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    value = c_max + np.log(exp_shifted.sum()) / alpha
    
    grad = np.zeros(n)
    # For each shift k (corresponding to index k + n - 1 in corr):
    for k in range(-(n-1), n):
        wk = w[k + n - 1]
        if wk < 1e-15:
            continue
        # ∂c[k]/∂h[i] = (1-h[i+k]) for valid i+k, minus h[i-k] for valid i-k
        # i+k valid: 0 <= i+k < n => max(0,-k) <= i < min(n, n-k)
        # i-k valid: 0 <= i-k < n => max(0,k) <= i < min(n, n+k)
        
        i_start_plus = max(0, -k)
        i_end_plus = min(n, n - k)
        if i_start_plus < i_end_plus:
            idx = np.arange(i_start_plus, i_end_plus)
            grad[idx] += wk * (1.0 - h[idx + k])
        
        i_start_minus = max(0, k)
        i_end_minus = min(n, n + k)
        if i_start_minus < i_end_minus:
            idx = np.arange(i_start_minus, i_end_minus)
            grad[idx] -= wk * h[idx - k]
    
    return value, grad
```

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:

def project_binary(h, target_sum):

    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0

            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:

            h_r[idx[0]] = 0
            final_sum -= 1
        else:
