import numpy as np
from scipy.optimize import minimize
import time
from numpy.fft import fft, ifft

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def compute_c5_from_autocorr(autocorr, n):
    # C5 = 1 - (2/n) * min_k autocorr[k]
    # But this is only for circular autocorrelation
    # The actual C5 uses linear correlation (mode='full')
    # So we need to be careful
    # For the circular case: f_k = n/2 - autocorr_circ[k]
    # C5_circ = (2/n) * max_k f_k = (2/n) * (n/2 - min_k autocorr_circ[k]) = 1 - 2*min_k autocorr_circ[k]/n
    return 1.0 - 2.0 * float(np.min(autocorr)) / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_bisect(h, target_sum):
    n = len(h)
    h = h.astype(float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def circular_autocorr(h):
    n = len(h)
    H = fft(h)
    return np.real(ifft(H * np.conj(H)))

def lse_obj_grad(h, alpha):
    n = len(h)
    g = circular_autocorr(h)
    g_min = np.min(g)
    shifted = g - g_min
    exp_vals = np.exp(-alpha * shifted)
    log_sum = np.log(np.sum(exp_vals))
    F = g_min - (1.0 / alpha) * log_sum
    p = exp_vals / np.sum(exp_vals)
    p_rev = np.zeros(n)
    p_rev[0] = p[0]
    p_rev[1:] = p[1:][::-1]
    grad = np.real(ifft(fft(p) * fft(h) + fft(p_rev) * fft(h)))
    return F, grad

def fista_optimize(h0, n, target, alphas, max_iters, deadline):
    best_h = h0.copy()
    best_score = compute_c5(h0)
    for alpha in alphas:
        h = best_h.copy()
        y = h.copy()
        t_k = 1.0
        lr = 0.1 / alpha
        stalled = 0
        for it in range(max_iters):
            if time.time() > deadline:
                break
            _, grad = lse_obj_grad(y, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-10:
                break
            # Step with backtracking
            step = lr
            y_F, _ = lse_obj_grad(y, alpha)
            for inner in range(20):
                x_new = project_bisect(y - step * grad, target)
                x_F, _ = lse_obj_grad(x_new, alpha)
                descent = y_F - step * np.dot(grad, x_new - y)
                if x_F <= descent + 1e-12:
                    break
                step *= 0.5
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - h)
            h = x_new
            t_k = t_new
            score = compute_c5(h)
            if score < best_score - 1e-10:
                best_score = score
                best_h = h.copy()
                stalled = 0
            else:
                stalled += 1
            if stalled > 200:
                break
        # After each alpha, project and check
        h = project_feasible(h, target)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    return best_h, best_score

def coord_desc_autocorr(h0, n, target, deadline, rng, max_sweeps=5000):
    """Coordinate descent using autocorrelation formulation."""
    h = h0.copy()
    autocorr = circular_autocorr(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    stall_count = 0
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        stall_in_sweep = 0
        pairs = rng.permutation(n // 2)
        
        for pi in range(n // 2):
            i = int(pairs[2 * pi])
            j = int(pairs[2 * pi + 1])
            s = h[i] + h[j]
            if s < 1e-10 or s > 2.0 - 1e-10:
                continue
            t_min = max(0.0, s - 1.0)
            t_max = min(1.0, s)
            if t_max - t_min < 1e-12:
                continue
            
            # d_k = (h[(j+k)%n] + h[(j-k)%n]) - (h[(i+k)%n] + h[(i-k)%n])
            # For k=0: d_0 = 2*(h[j]-h[i])
            ks = np.arange(n)
            d = (h[(j + ks) % n] + h[(j - ks) % n]) - (h[(i + ks) % n] + h[(i - ks) % n])
            
            # We want to maximize min_k (autocorr[k] + (t - h[i]) * d_k)
            # where t is the new value of h[i] (h[j] = s - t)
            # Let delta = t - h[i], so delta ∈ [t_min - h[i], t_max - h[i]]
            delta_min = t_min - h[i]
            delta_max = t_max - h[i]
            
            # Maximize min_k (autocorr[k] + delta * d_k) over delta ∈ [delta_min, delta_max]
            # This is a 1D problem: find delta that maximizes the minimum of n linear functions
            
            # Binary search on the value v: find the largest v such that
            # there exists delta with autocorr[k] + delta*d_k >= v for all k
            # For each k: if d_k > 0, delta >= (v - autocorr[k]) / d_k
            #              if d_k < 0, delta <= (v - autocorr[k]) / d_k
            #              if d_k = 0, need autocorr[k] >= v
            
            lo_v = float(np.min(autocorr + delta_min * d))
            hi_v = float(np.min(autocorr + delta_max * d))
            
            # Actually, let's binary search on v (the minimum value we can achieve)
            # For a given v, the feasible delta range is:
            #   lower_k = (v - autocorr[k])/d_k for d_k > 0
            #   upper_k = (v - autocorr[k])/d_k for d_k < 0
            #   For d_k = 0: need autocorr[k] >= v
            #   delta must be in [max(lower_k, delta_min), min(upper_k, delta_max)]
            
            pos = d > 1e-14
            neg = d < -1e-14
            zero = ~(pos | neg)
            
            if np.any(zero) and np.any(autocorr[zero] < lo_v - 1e-14):
                # Can't achieve lo_v, but this shouldn't happen
                pass
            
            # Binary search on v
            for _ in range(30):
                mid_v = (lo_v + hi_v) * 0.5
                # Check feasibility
                lb = delta_min
                ub = delta_max
                if np.any(pos):
                    lb = max(lb, float(np.min((mid_v - autocorr[pos]) / d[pos])))
                if np.any(neg):
                    ub = min(ub, float(np.max((mid_v - autocorr[neg]) / d[neg])))
                if np.any(zero):
                    if np.min(autocorr[zero]) < mid_v - 1e-14:
                        hi_v = mid_v  # infeasible
                        continue
                if lb <= ub + 1e-13:
                    lo_v = mid_v  # feasible
                else:
                    hi_v = mid_v  # infeasible
            
            v = lo_v
            # Find delta
            lb = delta_min
            ub = delta_max
            if np.any(pos):
                lb = max(lb, float(np.min((v - autocorr[pos]) / d[pos])))
            if np.any(neg):
                ub = min(ub, float(np.max((v - autocorr[neg]) / d[neg])))
            if lb > ub:
                delta_star = (delta_min + delta_max) * 0.5
            else:
                delta_star = (lb + ub) * 0.5
            
            if abs(delta_star) < 1e-14:
                continue
            
            h[i] += delta_star
            h[j] -= delta_star
            autocorr += delta_star * d
            stall_in_sweep += 1
        
        # Recompute autocorr periodically for accuracy
        if (sweep + 1) % 50 == 0:
            autocorr = circular_autocorr(h)
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 100:
                # Perturb
                h = project_feasible(h + rng.randn(n) * 0.005, target)
                autocorr = circular_autocorr(h)
                stall_count = 0
    
    h = project_feasible(h, target)
    return h, compute_c5(h)

def slsqp_polish(h, n, target, alphas, deadline):
    """SLSQP polish using the smooth log-sum-exp objective."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            F, _ = lse_obj_grad(x, alpha)
            penalty = 1000.0 * (np.sum(x) - target) ** 2
            return F + penalty
        
        def obj_grad(x):
            _, g = lse_obj_grad(x, alpha)
            g = g + 2000.0 * (np.sum(x) - target) * np.ones(n)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 500, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, compute_c5(h_cur)

import numpy as np
from scipy.optimize import minimize
import time
from numpy.fft import fft, ifft


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def project_bisect(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)


def circular_autocorr(h):
    n = len(h)
    H = fft(h)
    return np.real(ifft(H * np.conj(H)))


def lse_obj_grad(h, alpha):
    n = len(h)
    g = circular_autocorr(h)
    g_min = np.min(g)
    shifted = g - g_min
    if alpha * shifted.max() > 50:
        shifted = shifted / shifted.max() * 50.0
        alpha_eff = alpha * shifted.max() / 50.0
    else:
        alpha_eff = alpha
    exp_vals = np.exp(-alpha_eff * shifted)
    total = np.sum(exp_vals)
    log_sum = np.log(total)
    F = g_min - (1.0 / alpha_eff) * log_sum
    p = exp_vals / total
    p_rev = np.zeros(n)
    p_rev[0] = p[0]
    p_rev[1:] = p[1:][::-1]
    grad = np.real(ifft(fft(p) * fft(h) + fft(p_rev) * fft(h)))
    return F, grad


def fista_optimize(h0, n, target, alphas, max_iters, deadline):
    best_h = h0.copy()
    best_score = compute_c5(h0)
    for alpha in alphas:
        h = best_h.copy()
        y = h.copy()
        t_k = 1.0
        lr = 0.5 / max(alpha, 1.0)
        stalled = 0
        for it in range(max_iters):
            if time.time() > deadline:
                break
            _, grad = lse_obj_grad(y, alpha)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-12:
                break
            step = lr
            y_F, _ = lse_obj_grad(y, alpha)
            for _inner in range(25):
                x_new = project_bisect(y - step * grad, target)
                x_F, _ = lse_obj_grad(x_new, alpha)
                descent = y_F - step * np.dot(grad, x_new - y)
                if x_F <= descent + 1e-10:
                    break
                step *= 0.5
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            y = x_new + ((t_k - 1.0) / t_new) * (x_new - h)
            h = x_new
            t_k = t_new
            if it % 50 == 0:
                score = compute_c5(h)
                if score < best_score - 1e-12:

                    best_h = h.copy()
                    stalled = 0

        h = project_feasible(h, target)

def coord_desc_autocorr(h0, n, target, deadline, rng, max_sweeps=8000, anneal_temp=0.0):
    h = h0.copy()
    autocorr = circular_autocorr(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    stall_count = 0
    ks = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        pairs = rng.permutation(n // 2)
        improved = 0

        for pi in range(n // 2):

            j = int(pairs[2 * pi + 1])
            s = h[i] + h[j]
            if s < 1e-10 or s > 2.0 - 1e-10:
                continue
            t_min = max(0.0, s - 1.0)
            t_max = min(1.0, s)
            if t_max - t_min < 1e-12:

            d = (h[(j + ks) % n] + h[(j - ks) % n]) - (h[(i + ks) % n] + h[(i - ks) % n])
            delta_min = t_min - h[i]

                    lb = lb2
            if np.any(neg):

                if ub2 < ub:
                    ub = ub2
            if lb > ub:
                delta_star = (delta_min + delta_max) * 0.5
            else:

                continue

            h[i] += delta_star
            h[j] -= delta_star
            autocorr += delta_star * d
            improved += 1

        if (sweep + 1) % 40 == 0:
            autocorr = circular_autocorr(h)

        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            stall_count = 0
        else:

                h = project_feasible(h + rng.randn(n) * pert_scale, target)
                autocorr = circular_autocorr(h)
                stall_count = 0

    h = project_feasible(h, target)
    return best_h, best_score


def lbfgs_polish(h, n, target, alphas, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() > deadline:
            break

        def _obj(x):
            F, _ = lse_obj_grad(x, alpha)
            pen = 500.0 * (np.sum(x) - target) ** 2
            return F + pen

        def _obj_grad(x):
            _, g = lse_obj_grad(x, alpha)
            g = g + 1000.0 * (np.sum(x) - target) * np.ones(n)
            return g

        try:
            res = minimize(_obj, h_cur, jac=_obj_grad, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass

    return h_cur, compute_c5(h_cur)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
