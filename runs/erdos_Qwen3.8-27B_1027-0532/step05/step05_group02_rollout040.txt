    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_greedy(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 200
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if k_star == 0:
            stall += 1
            continue

        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        top_i = ones[np.argsort(A[ones])[:min(30, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(30, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val - 1e-10:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(50):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val - 1e-10:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def sa_adaptive(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    positions = np.arange(n)
    tabu = np.zeros(n, dtype=int)
    tabu_counter = 0
    tabu_len = 50

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = min(it / max(n_iters * 0.8, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio
        if t < 1e-8:
            t = 1e-8

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if it % 3 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_score = A.copy()
                j_score = B.copy()
                for p in range(tabu_len):
                    tc = (tabu_counter - p) % n
                    if tabu[tc] > 0:
                        i_score[tabu[tc]] = -1e9
                        j_score[tabu[tc]] = -1e9
                i_valid = np.where((i_score > -1e8) & (h == 1.0))[0]
                j_valid = np.where((j_score > -1e8) & (h == 0.0))[0]
                if len(i_valid) > 0 and len(j_valid) > 0:
                    i = i_valid[rng.randint(len(i_valid))]
                    j = j_valid[rng.randint(len(j_valid))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            tabu_counter = (tabu_counter + 1) % n
            tabu[tabu_counter] = i
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    best_n = n_init

    # Phase 1: Quick greedy at n_init
    allowed = min(60, tleft() * 0.10)
    h_g, c5_g = fast_greedy(best_h, max_iters=50000, rng=rng,
                             time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA at n_init
    allowed = min(200, tleft() * 0.25)
    n_sa = int(min(4000000, allowed * 50000))
    h_sa, c5_sa = sa_adaptive(best_h, n_sa, 0.02, 0.00005, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: SLSQP continuous refinement at n_init
    if tleft() > 80:
        allowed = min(tleft() * 0.15, 120)
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 150, 500, 1500, 5000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            best_n = n_use

    # Phase 4: Try multiple n values with interpolation
    n_candidates = [602, 604, 606, 608, 610, 612, 614, 616, 620, 624]
    n_candidates = [n for n in n_candidates if n != n_init and n % 2 == 0 and n >= 100]

    h_continuous = best_h.copy() if best_n == n_init else interpolate_to(best_h, n_init, target_init)

    for n_try in n_candidates:
        if tleft() < 60:
            break
        target_try = n_try // 2

        h_interp = interpolate_to(h_continuous, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        sc_bin = compute_c5(h_bin_try)

        allowed_g = min(25, tleft() * 0.06)
        h_g2, c5_g2 = fast_greedy(h_bin_try, max_iters=10000, rng=rng,
                                   time_limit=allowed_g, start_time=start_time)
        if c5_g2 < sc_bin:
            h_bin_try = h_g2
            sc_bin = c5_g2

        if tleft() < 40:
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin_try.copy()
                best_n = n_try
            continue

        allowed_sa = min(35, tleft() * 0.07)
        n_sa2 = int(min(1500000, allowed_sa * 40000))
        h_sa2, c5_sa2 = sa_adaptive(h_bin_try, n_sa2, 0.015, 0.00005, rng,
                                     time_limit=allowed_sa, start_time=start_time)
        if c5_sa2 < sc_bin:
            h_bin_try = h_sa2
            sc_bin = c5_sa2

        if tleft() < 25:
            if sc_bin < best_score:
                best_score = sc_bin
                best_h = h_bin_try.copy()
                best_n = n_try
            continue

        if tleft() > 40:
            allowed_sl = min(20, tleft() * 0.04)
            h_ref2, sc2 = slsqp_refine(h_bin_try, n_try, float(n_try)/2.0,
                                        [50, 200, 1000, 3000], 800, 1e-13,
                                        time_limit=allowed_sl, start_time=start_time)
            if sc2 < sc_bin:
                sc_bin = sc2
                h_bin_try = h_ref2

        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin_try.copy()
            best_n = n_try

    # Phase 5: Final refinement on best solution
    if tleft() > 50:
        n_use = best_n
        target_use = float(n_use) / 2.0
        h_bin_final = project_binary(best_h, int(n_use / 2.0))
        sc_final = compute_c5(h_bin_final)

        allowed_g3 = min(30, tleft() * 0.15)
        h_g3, c5_g3 = fast_greedy(h_bin_final, max_iters=20000, rng=rng,
                                   time_limit=allowed_g3, start_time=start_time)
        if c5_g3 < sc_final:
            h_bin_final = h_g3
            sc_final = c5_g3

        if tleft() > 30:
            allowed_sa3 = min(40, tleft() * 0.20)
            n_sa3 = int(min(1000000, allowed_sa3 * 40000))
            h_sa3, c5_sa3 = sa_adaptive(h_bin_final, n_sa3, 0.01, 0.00002, rng,
                                         time_limit=allowed_sa3, start_time=start_time)
            if c5_sa3 < sc_final:
                h_bin_final = h_sa3
                sc_final = c5_sa3

        if tleft() > 20:
            allowed_sl2 = min(tleft() - 10, 40)
            h_ref3, sc3 = slsqp_refine(h_bin_final, n_use, target_use,
                                        [100, 500, 2000, 5000, 10000], 1000, 1e-14,
                                        time_limit=allowed_sl2, start_time=start_time)
            if sc3 < sc_final:
                h_bin_final = h_ref3
                sc_final = sc3
            if sc_final < best_score:
                best_score = sc_final
                best_h = h_bin_final.copy()
                best_n = n_use

    # Final feasibility check
    n_final = best_n
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    # Ensure it's valid
    h_final = np.clip(h_final, 0.0, 1.0)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_binary(best_h, int(n_final / 2.0))
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
