**Proximal Gradient Method for the Boolean Constraint**

Since h ∈ [0,1] and sum(h) = n/2, the feasible set is a hypercube intersected with a hyperplane. The projection onto this set is exactly what `project_feasible` does.

The key is to get a good gradient. For the true max:
∇C₅(h) = ∇f_{k*}(h) where k* = argmax_k f_k(h)

This is a subgradient (not gradient since max is not differentiable). But we can use it in a subgradient method.

The subgradient method with Polyak step size:
h_{k+1} = proj(h_k - α_k * g_k) where α_k = (C₅(h_k) - C₅*) / ||g_k||²

But we don't know C₅* (the optimal value). We can use an upper bound.

Alternatively, use the "normalized" step size: α_k = c / √k for some constant c.

Let me try this:

```python
def subgradient_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    c = 0.1  # step size parameter
    for k in range(100000):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        # Subgradient of f_{lag}
        sg = np.zeros(n)
        if lag == 0:
            sg = (1.0 - 2.0 * h) * dx
        elif lag > 0:
            sg[:n-lag] += (1.0 - h[lag:]) * dx
            sg[lag:] -= h[:n-lag] * dx
        else:
            m = -lag
            sg[m:] += (1.0 - h[:n-m]) * dx
            sg[:n-m] -= h[m:] * dx
        
        lr = c / np.sqrt(k + 1)
        h_new = h - lr * sg
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_score
```

The subgradient method is slow to converge (O(1/√k) rate), but it's simple and doesn't require smooth approximation. For our small problem (n=400), each iteration is O(n²) for the correlate computation, so 100000 iterations would take 100000 * 400² / 1e9 ≈ 16 seconds. That's feasible!

But O(1/√k) convergence means we need ~10^8 iterations to get 10^-4 accuracy, which is too many.

A better approach: use the "bundle method" or "cutting plane method" for the max of linear (approximately) functions. But this requires maintaining a growing set of linearizations.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            grad[:n - k] += w * (1.0 - h[k:]) * dx
            grad[k:] -= w * h[:n - k] * dx
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m]) * dx
            grad[:n - m] -= w * h[m:] * dx
    return grad

def _g_fun(h, target):
    return np.sum(h) - target

def _g_jac(h, target):
    return np.ones(len(h))

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        g = 1.0 - h
        for k in range(-(n - 1), n):
            coeff = np.zeros(n)
            if k >= 0:
                coeff[:n - k] = g[k:] * dx
            else:
                m = -k
                coeff[m:] = g[:n - m] * dx
            constraints.append(coeff @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = project_feasible(h_new, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        if it > 10 and abs(compute_c5(h) - best_c5) < 1e-12:
            break
    return best_h, best_c5

def nesterov_smooth(h_init, n, target, alphas, steps_per_alpha, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        t = 1.0
        top_m = 2 * n - 1
        if alpha < 200:
            top_m = min(60, 2 * n - 1)
        elif alpha < 1000:
            top_m = min(150, 2 * n - 1)

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            mu = (t - 1.0) / t_new
            y = h + mu * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            grad = smooth_grad(y, alpha, dx, top_m=top_m)
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-14:
                break
            lr = 0.5 / gnorm
            lr = min(lr, 5.0)
            improved = False
            for bt in range(50):
                h_new = h - lr * grad
                h_new = project_feasible(h_new, target)
                score = compute_c5(h_new)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_new.copy()
                    h_prev = h
                    h = h_new
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                h_new = h - 0.005 * grad
                h_new = project_feasible(h_new, target)
                h_prev = h
                h = h_new
            t = t_new
    return best_h, best_score

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _g_fun(x, target), 'jac': lambda x: _g_jac(x, target)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-14:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
