                j_neg = np.arange(0, n - k)
                grad[j_neg] -= h_cur[j_neg + k]
        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        best_h = h_cur
        best_sc = cur_score
        for step in [0.008, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0002]:
            h_test = h_cur - step * grad
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_sc:
                best_sc = sc
                best_h = h_test.copy()
        if best_sc < cur_score - 1e-14:
            h_cur = best_h
            cur_score = best_sc
        else:
            break
    return h_cur, cur_score

def fista_step(h, n, target, max_iter, deadline, alpha_sm=200.0):
    dx = 2.0 / n
    h_cur = h.copy()
    y = h_cur.copy()
    t_mom = 1.0
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    L = 4.0 * np.sqrt(n)
    step = 1.0 / (L * dx)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        c_max = float(np.max(corr))
        shifted = alpha_sm * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for j in range(len(weights)):
            w = weights[j]
            if w < 1e-10:
                continue
            if j >= n - 1:
                k = j - (n - 1)
                j_pos = np.arange(0, n - k)
                grad[j_pos] += w * (1.0 - y[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= w * y[j_neg - k]
            else:
                k = (n - 1) - j
                j_pos = np.arange(k, n)
                grad[j_pos] += w * (1.0 - y[j_pos - k])
                j_neg = np.arange(0, n - k)
                grad[j_neg] -= w * y[j_neg + k]
        grad *= dx
        x_new = y - step * grad
        x_new = project_feasible(x_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - h_cur)
        y = np.clip(y, 0.0, 1.0)
        t_mom = t_new
        h_cur = x_new
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 20.0

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_cur = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_cur.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return hard_deadline - time.time()

    # Phase 1: Quick settle SLSQP
    for alpha in [50, 200, 1000]:
        if tleft() < 40:
            break
        h_new = slsqp_pass(h_cur, n_final, target_final, alpha, 800, 1e-13, start_time + 55)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 2: Medium alpha SLSQP
    for alpha in [2000, 5000]:
        if tleft() < 80:
            break
        h_new = slsqp_pass(h_cur, n_final, target_final, alpha, 2000, 1e-14, min(start_time + 200, hard_deadline - 100))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 3: Active set descent
    if tleft() > 80:
        h_as, sc_as = active_set_descent(h_cur, n_final, target_final, 500, min(start_time + 280, hard_deadline - 80), frac=0.005)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as.copy()
        h_cur = h_as

    # Phase 4: High alpha SLSQP
    for alpha in [10000, 30000]:
        if tleft() < 100:
            break
        h_new = slsqp_pass(h_cur, n_final, target_final, alpha, 3000, 1e-15, min(start_time + 500, hard_deadline - 100))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 5: FISTA for diversification
    if tleft() > 120:
        h_fista, sc_f = fista_step(h_cur, n_final, target_final, 300, min(start_time + 620, hard_deadline - 120))
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_fista.copy()
        h_cur = h_fista
        h_new = slsqp_pass(h_cur, n_final, target_final, 10000, 2000, 1e-15, min(start_time + 700, hard_deadline - 100))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 6: Very high alpha SLSQP
    for alpha in [50000, 100000]:
        if tleft() < 100:
            break
        h_new = slsqp_pass(h_cur, n_final, target_final, alpha, 4000, 1e-16, min(start_time + 900, hard_deadline - 100))
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new

    # Phase 7: Active set descent (tighter)
    if tleft() > 60:
        h_as, sc_as = active_set_descent(h_cur, n_final, target_final, 400, min(start_time + 980, hard_deadline - 60), frac=0.002)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as.copy()
        h_cur = h_as

    # Phase 8: Perturbation restarts
    if tleft() > 100:
        for attempt in range(4):
            if tleft() < 50:
                break
            scales = [0.02, 0.04, 0.01, 0.03]
            h_pert = h_cur + rng.randn(n_final) * scales[attempt]
            h_pert = project_feasible(h_pert, target_final)
            h_new = slsqp_pass(h_pert, n_final, target_final, 5000, 1500, 1e-15, hard_deadline - 80)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            else:
                h_new = slsqp_pass(h_cur, n_final, target_final, 20000, 1500, 1e-15, hard_deadline - 80)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                    h_cur = h_new

    # Phase 9: Ultra high alpha final SLSQP
    if tleft() > 60:
        h_new = slsqp_pass(best_h, n_final, target_final, 200000, 3000, 1e-16, hard_deadline - 30)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 10: Final active set polish
    if tleft() > 30:
        h_as, sc_as = active_set_descent(best_h, n_final, target_final, 200, hard_deadline, frac=0.001)
        if sc_as < best_score:
            best_score = sc_as
            best_h = h_as.copy()

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
