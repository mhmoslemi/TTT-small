                sc = compute_c5(h_test)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_block = new_block
                    improved = True
            if not np.allclose(best_block, h_cur[j0:j1]):
                h_cur[j0:j1] = best_block
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
        if not improved:
            break
    return h_cur, cur_score


def coord_desc(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(20):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj(x, alpha, target, 1e6)

    def g(x):
        return lse_grad(x, alpha, target, 1e6)

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h


def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0
    rng = np.random.RandomState(seed)

    n0 = len(initial_h_values)
    target0 = n0 / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target0)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 50.0

    def tleft():
        return deadline - time.time()

    # Phase 1: Optimize at base resolution
    alphas_p1 = [10, 50, 200, 1000, 5000, 20000]
    h1, sc1 = multi_alpha_optimize(best_h, target0, alphas_p1, mu, deadline)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: Top-lag refinement at base resolution
    if tleft() > 120:
        h2 = top_lag_refine(best_h, target0, n_top=5, alpha=5000, mu=mu, maxiter=100, deadline=deadline)
        sc2 = compute_c5(h2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Upsample to 2x and optimize
    n2 = 2 * n0
    target2 = n2 / 2.0
    if tleft() > 300:
        h_up = upsample(best_h, n2)
        h_up = project_feasible(h_up, target2)
        alphas_p3 = [10, 50, 200, 1000, 5000]
        h3, sc3 = multi_alpha_optimize(h_up, target2, alphas_p3, mu, deadline)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
            n_cur = n2
            target_cur = target2
        else:
            n_cur = n0
            target_cur = target0

    # Phase 4: Upsample to 4x original and optimize
    if tleft() > 300:
        n4 = 4 * n0
        target4 = n4 / 2.0
        h_up4 = upsample(best_h, n4)
        h_up4 = project_feasible(h_up4, target4)
        alphas_p4 = [10, 50, 200, 1000]
        h4, sc4 = multi_alpha_optimize(h_up4, target4, alphas_p4, mu, deadline)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
            n_cur = n4
            target_cur = target4

    # Phase 5: Top-lag refinement at current resolution
    if tleft() > 150:
        h5 = top_lag_refine(best_h, target_cur, n_top=4, alpha=10000, mu=mu, maxiter=80, deadline=deadline)
        sc5 = compute_c5(h5)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: Diversified restarts at current resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    restart_strategies = [
        ('gauss', 0.1), ('gauss', 0.2), ('sparse', 0.05),
        ('gauss', 0.05), ('smooth', 0.2)
    ]
    for strat, param in restart_strategies:
        if tleft() < 80:
            break
        h_pert = best_h.copy()
        if strat == 'gauss':
            h_pert += rng.randn(n_cur) * param
        elif strat == 'sparse':
            n_flip = max(1, int(n_cur * param))
            idx = rng.choice(n_cur, size=n_flip, replace=False)
            h_pert[idx] = rng.uniform(0, 1, n_flip)
        elif strat == 'smooth':
            freq = np.fft.rfftfreq(n_cur)
            cutoff = int(0.03 * n_cur)
            spec = np.fft.rfft(h_pert) + rng.randn(n_cur // 2 + 1) * param
            spec[cutoff:] = np.fft.rfft(h_pert)[cutoff:]
            h_pert = np.fft.irfft(spec, n_cur)
        h_pert = project_feasible(h_pert, target_cur)
        h_new = lbfgs_lse(h_pert, target_cur, 500, mu, 200, deadline)
        h_new = lbfgs_lse(h_new, target_cur, 5000, mu, 200, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Block coordinate descent
    if tleft() > 150:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h7, sc7 = block_cd(best_h, target_cur, block_size=4, n_sweeps=5, deadline=deadline)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        h7b, sc7b = block_cd(best_h, target_cur, block_size=8, n_sweeps=3, deadline=deadline)
        if sc7b < best_score:
            best_score = sc7b
            best_h = h7b

    # Phase 8: Top-lag refinement again
    if tleft() > 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h8 = top_lag_refine(best_h, target_cur, n_top=3, alpha=20000, mu=mu, maxiter=60, deadline=deadline)
        sc8 = compute_c5(h8)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    # Phase 9: SLSQP polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h9 = slsqp_polish(best_h, target_cur, alpha=5000, maxiter=2000)
        sc9 = compute_c5(h9)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Coordinate descent (only if n is manageable)
    if tleft() > 120 and len(best_h) <= 800:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h10, sc10 = coord_desc(best_h, target_cur, n_sweeps=3, deadline=deadline)
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    # Phase 11: Final L-BFGS with high alpha
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h11 = lbfgs_lse(best_h, target_cur, 50000, mu, 500, deadline)
        sc11 = compute_c5(h11)
        if sc11 < best_score:
            best_score = sc11
            best_h = h11

    # Phase 12: Final SLSQP
    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h12 = slsqp_polish(best_h, target_cur, alpha=10000, maxiter=5000)
        sc12 = compute_c5(h12)
        if sc12 < best_score:
            best_score = sc12
            best_h = h12

    # Final projection and evaluation
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
