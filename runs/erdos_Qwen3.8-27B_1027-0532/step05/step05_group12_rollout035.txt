def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, deadline):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    global _global_n, _global_target
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    def time_left():
        return deadline - time.time()

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Hot coordinate descent on initial resolution
    if time_left() > 60:
        h_cd, c5_cd = hot_coord_descent(best_h, n, target,
                                        deadline=start_time + min(60, time_left()),
                                        n_grid=8, patience=20000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 2: Nesterov gradient descent
    if time_left() > 120:
        h_ng, c5_ng = nesterov_grad_descend(best_h, n, target,
                                            deadline=start_time + min(150, time_left()),
                                            alpha_start=50, alpha_end=30000,
                                            total_iters=80000, top_m=300)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    # Phase 3: Binary local search to find alternative structure
    if time_left() > 120:
        h_b, c5_b = binary_local_search(best_h, n, target,
                                        deadline=start_time + min(60, time_left()),
                                        n_flips=100000)
        if c5_b < best_score:
            best_score = c5_b
            best_h = h_b
        else:
            # Try refining the binary solution with fractional values
            h_frac = h_b + 0.1 * rng.randn(n)
            h_frac = project_feasible(h_frac, target)
            h_ng2, c5_ng2 = nesterov_grad_descend(h_frac, n, target,
                                                   deadline=start_time + min(60, time_left()),
                                                   alpha_start=100, alpha_end=20000,
                                                   total_iters=40000, top_m=200)
            if c5_ng2 < best_score:
                best_score = c5_ng2
                best_h = h_ng2

    # Phase 4: More coordinate descent
    if time_left() > 80:
        h_cd, c5_cd = hot_coord_descent(best_h, n, target,
                                        deadline=start_time + min(80, time_left()),
                                        n_grid=10, patience=30000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: SLSQP polish
    if time_left() > 60:
        h_slq = slsqp_optimize(best_h, n, target,
                               [200, 1000, 5000, 20000],
                               maxiter=2000, ftol=1e-16,
                               deadline=start_time + min(80, time_left()))
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 6: Upsample to higher resolution
    for n_try in [600, 800, 1200]:
        if time_left() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        # Coordinate descent at higher resolution
        if time_left() > 100:
            h_cd, c5_cd = hot_coord_descent(h_interp, n_try, target_try,
                                            deadline=start_time + min(80, time_left()),
                                            n_grid=8, patience=20000)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n = n_try
                target = target_try
            else:
                h_interp = h_cd

        # Gradient at higher resolution
        if time_left() > 80:
            h_ng, c5_ng = nesterov_grad_descend(h_interp, n_try, target_try,
                                                deadline=start_time + min(100, time_left()),
                                                alpha_start=50, alpha_end=20000,
                                                total_iters=40000, top_m=200)
            if c5_ng < best_score:
                best_score = c5_ng
                best_h = h_ng
                n = n_try
                target = target_try

    # Phase 7: Final coordinate descent on best solution
    if time_left() > 40:
        h_cd, c5_cd = hot_coord_descent(best_h, n, target,
                                        deadline=start_time + min(50, time_left()),
                                        n_grid=10, patience=10000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: Final SLSQP
    if time_left() > 30:
        h_fin = slsqp_optimize(best_h, n, target,
                               [5000, 20000, 50000],
                               maxiter=1000, ftol=1e-16,
                               deadline=start_time + min(40, time_left()))
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    if final_score > best_score + 1e-6:
        h_final = project_feasible(best_h, n_use / 2.0)
        final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
