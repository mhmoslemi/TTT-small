import numpy as np
from scipy.optimize import minimize
import time

def project_feasible_exact(h, target):
    """Project h onto {0 <= h <= 1, sum(h) = target} using bisection on lambda."""
    h = h.astype(float).copy()
    # Bisection for lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    # First check if simply clipping works
    h_clipped = np.clip(h, 0, 1)
    if abs(h_clipped.sum() - target) < 1e-12:
        return h_clipped
    # Need to shift
    for _ in range(60):
        mid = (lo + hi) / 2
        h_mid = np.clip(h - mid, 0, 1)
        s = h_mid.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_result = np.clip(h - (lo + hi) / 2, 0, 1)
    # Final correction
    diff = target - h_result.sum()
    if diff != 0:
        if diff > 0:
            mask = h_result < 1.0
        else:
            mask = h_result > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h_result[mask] += diff / cnt
            h_result = np.clip(h_result, 0, 1)
    return h_result

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_grad(x, alpha, target, mu):
    """Log-sum-exp objective and gradient."""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def nesterov_optimize(h_init, target, alpha, mu, maxiter, deadline, lr_init=0.01):
    """Nesterov-accelerated gradient method with projection."""
    n = len(h_init)
    x = h_init.copy()
    y = x.copy()
    momentum = 0.9
    lr = lr_init
    best_val = float('inf')
    best_x = x.copy()
    
    for it in range(maxiter):
        if time.time() > deadline:
            break
        val, grad = lse_obj_grad(y, alpha, target, mu)
        if val < best_val:
            best_val = val
            best_x = y.copy()
        x_new = y - lr * grad
        y_new = x_new + momentum * (x_new - x)
        x = project_feasible_exact(x_new, target)
        y = project_feasible_exact(y_new, target)
        # Adaptive learning rate
        if it % 100 == 99:
            if best_val < 0.38:
                lr *= 0.95
            elif it > 200:
                lr *= 0.98
    
    return best_x

def top_lag_subgrad(h, target, n_top=5, n_steps=500, deadline=None, lr_init=0.01):
    """Direct subgradient method focusing on top lags."""
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        # Weights proportional to excess over mean
        top_mean = top_vals.mean()
        top_range = top_vals.max() - top_vals.min()
        if top_range < 1e-12:
            weights = np.ones(n_top) / n_top
        else:
            weights = (top_vals - top_mean) / top_range
            weights = np.clip(weights, 0, None)
            wsum = weights.sum()
            if wsum > 1e-10:
                weights = weights / wsum
            else:
                weights = np.ones(n_top) / n_top
        
        grad = np.zeros(n)
        for i, ki in enumerate(top_idx):
            w = weights[i]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible_exact(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h_cur = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
            # Try a step in the opposite direction to escape
            h_alt = h_cur + lr * 0.5 * grad
            h_alt = project_feasible_exact(h_alt, target)
            sc_alt = compute_c5(h_alt)
            if sc_alt < best_score:
                best_score = sc_alt
                best_h = h_alt
                h_cur = h_alt
    
    return best_h, best_score

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 50})
        return project_feasible_exact(res.x, target)
    except Exception:
        return h_cur

def coordinate_descent(h, target, n_sweeps=5):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible_exact(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible_exact(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt

def project_exact(h, target_sum):

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def project_exact(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - result.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def nesterov_opt(h_init, target, alpha, mu, maxiter, deadline, lr_init=0.005):
    n = len(h_init)
    x = h_init.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = lr_init
    best_val = compute_c5(x)
    best_x = x.copy()
    for it in range(maxiter):
        if time.time() > deadline:
            break
        y = x + momentum * v
        val, grad = lse_obj_grad(y, alpha, target, mu)
        v = momentum * v - lr * grad
        x_new = project_feasible(y + v, target)
        sc = compute_c5(x_new)
        if sc < best_val:
            best_val = sc
            best_x = x_new.copy()
        x = x_new
        if it % 500 == 499:
            momentum = min(momentum + 0.005, 0.95)
        if it % 2000 == 1999:
            lr *= 0.9
    return best_x


def lbfgs_opt(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def top_lag_opt(h, target, n_top, n_steps, deadline, lr_init):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_min = top_vals.min()
        top_range = top_vals.max() - top_min
        if top_range < 1e-12:
            weights = np.ones(n_top) / n_top
        else:
            weights = (top_vals - top_min) / top_range
            ws = weights.sum()
            if ws < 1e-10:
                weights = np.ones(n_top) / n_top
            else:
                weights = weights / ws
        grad = np.zeros(n)
        for i, ki in enumerate(top_idx):
            w = weights[i]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = project_feasible(h_cur - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h_cur = h_new
            lr = min(lr * 1.03, 0.08)
        else:
            lr *= 0.82
            if lr < 1e-9:
                break
    return best_h, best_score


def binary_sa(h_start, target, n_iter, deadline, t_init=0.05, t_end=0.001):
    n = len(h_start)
    h = (h_start > 0.5).astype(float)
    if h.sum() > target:
        idx_ones = np.where(h == 1.0)[0]
        np.random.shuffle(idx_ones)
        h[idx_ones[int(target):]] = 0.0
    else:
        idx_zeros = np.where(h == 0.0)[0]
        np.random.shuffle(idx_zeros)
        h[idx_zeros[:int(target) - int(h.sum())]] = 1.0
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = t_init
    cooling = (t_end / t_init) ** (1.0 / max(n_iter, 1))
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t *= cooling
        idx_ones = np.where(h == 1.0)[0]
        idx_zeros = np.where(h == 0.0)[0]
        if len(idx_ones) == 0 or len(idx_zeros) == 0:
            break
        i = idx_ones[np.random.randint(len(idx_ones))]
        j = idx_zeros[np.random.randint(len(idx_zeros))]
        h[i] = 0.0
        h[j] = 1.0
        sc_new = compute_c5(h)
        delta = sc_new - compute_c5(h)
        if sc_new < best_score or np.random.rand() < np.exp(-(sc_new - best_score) / max(t, 1e-10)):
            if sc_new < best_score:
                best_score = sc_new
                best_h = h.copy()
        else:
            h[j] = 0.0
            h[i] = 1.0
    return project_feasible(best_h, target), best_score


def coord_descent(h, target, n_sweeps):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c_val = b - gr * (b - a)
            d_val = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c_val)
            fd = f1d(d_val)
            for _ in range(20):
                if fc < fd:
                    b = d_val
                    d_val = c_val
                    fd = fc
                    c_val = b - gr * (b - a)
                    fc = f1d(c_val)
                else:
                    a = c_val
                    c_val = d_val
                    fc = fd
                    d_val = a + gr * (b - a)
                    fd = f1d(d_val)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n_cur = len(h)
    bounds = [(0.0, 1.0)] * n_cur
    h_cur = h.copy()

    def f_slq(x):
        return lse_obj_grad(x, alpha, target, 1e10)[0]

    def g_slq(x):
        return lse_obj_grad(x, alpha, target, 1e10)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_j(x):
        return np.ones(n_cur)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_j}]
    try:
        res = minimize(f_slq, h_cur, jac=g_slq, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur
