RHS = grads[idx] @ h_cur - over[idx]

The solution will give a new h that (in the linearized model) has all active R(k) equal to t (the new minimum). Then we evaluate the actual R(k) at the new h and see if it improved.

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_over(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1 : 2*n-1].copy()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = len(h)
    g = np.zeros(n)
    g[0:n-k] = (1.0 - h[k:n])
    g[k:n] += -h[0:n-k]
    return g

def subgradient_optimize(h, target, n_steps_per_temp, temps, lr_init, lr_decay, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    for temp in temps:
        for _ in range(n_steps_per_temp):
            if time.time() > deadline:
                return best_h, best_score
            over = compute_over(h_cur)
            max_over = float(np.max(over))
            sc = max_over * dx
            if sc < best_score:
                best_score = sc
                best_h = h_cur.copy()
            weights = np.exp((over - max_over) / temp)
            weights /= weights.sum()
            grad = np.zeros(n)
            for k in range(n):
                wk = weights[k]
                if wk < 1e-12:
                    continue
                k = int(k)
                grad[0:n-k] += wk * (1.0 - h_cur[k:n])
                grad[k:n] += -wk * h_cur[0:n-k]
            grad *= dx
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < sc:
                h_cur = h_new
            else:
                lr *= lr_decay
                if lr < 1e-12:
                    lr = lr_init * 0.5
    return best_h, best_score

def slsqp_smooth(h, target, alpha, maxiter, deadline):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        over = compute_over(x)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    def obj_grad(x):
        over = compute_over(x)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
        grad = np.zeros(n)
        for k in range(n):
            wk = w[k]
            if wk < 1e-12:
                continue
            k = int(k)
            grad[0:n-k] += wk * (1.0 - x[k:n])
            grad[k:n] += -wk * x[0:n-k]
        grad *= dx
        return grad
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def lbfgsb_smooth(h, target, alpha, maxiter, penalty_coeff, deadline):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        over = compute_over(x)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        pen = penalty_coeff * (np.sum(x) - target) ** 2
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha + pen
    def obj_grad(x):
        over = compute_over(x)
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
        grad = np.zeros(n)
        for k in range(n):
            wk = w[k]
            if wk < 1e-12:
                continue
            k = int(k)
            grad[0:n-k] += wk * (1.0 - x[k:n])
            grad[k:n] += -wk * x[0:n-k]
        grad *= dx
        grad += 2.0 * penalty_coeff * (np.sum(x) - target)
        return grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=obj_grad,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def cutting_plane_step(h, target, n_active, deadline):
    n = len(h)
    dx = 2.0 / n
    over = compute_over(h)
    max_over = np.max(over)
    sorted_idx = np.argsort(over)[::-1][:n_active]
    active = sorted_idx[:n_active]
    grads = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        k = int(k)
        grads[idx, 0:n-k] = (1.0 - h[k:n])
        grads[idx, k:n] += -h[0:n-k]
    grads *= dx
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for idx in range(len(active)):
        A_ub[idx, :n] = grads[idx]
        A_ub[idx, n] = -1.0
        b_ub[idx] = np.dot(grads[idx], h) - over[int(active[idx])]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds_list = [(0.0, 1.0)] * n + [(0.0, 10.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs')
        if res.success:
            h_new = res.x[:n]
            h_new = project_feasible(h_new, target)
            return h_new
    except Exception:
        pass
    return h

            h_cur = h_new

                max_ov = np.max(over)
                if np.max(over) - np.min(over) < 1e-6:
                    break
    return best_h, best_score

            best_h = h_new.copy()
            h_cur = h_new
        else:
