def compute_c5(h):

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

def solve_full_lp(h_cur, n, target, reg=0.0, ref=None):
    dx = 2.0 / n
    g = 1.0 - h_cur
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    half_n = n - 1
    for k_idx in range(1, 2 * n - 1):
        k = k_idx - half_n
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    if reg > 0 and ref is not None:
        objective = t_var + reg * cp.sum_squares(h_var - ref)
    else:
        objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    for solver in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            if solver == cp.SCS:

        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                           max_steps=2000, time_limit=min(12, tleft() * 0.7))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
