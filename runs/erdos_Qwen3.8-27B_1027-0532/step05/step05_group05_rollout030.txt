def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        h[mask] += diff / cnt

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fourier_to_h(params, n):
    n_modes = len(params) // 2
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(n_modes):
        a = params[2 * k]
        phi = params[2 * k + 1]
        h += a * np.cos(np.pi * (k + 1) * x + phi)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)

def fourier_obj(params, n):
    h = fourier_to_h(params, n)
    return compute_c5(h)

_g_target = 0.0
_g_alpha = 0.0

    return np.ones(_g_n)
