```python
import cvxpy as cp

def scp_solve(h, target, max_iters=100):
    n = len(h)
    h_cv = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_cv >= 0, h_cv <= 1, cp.sum(h_cv) == target]
    
    h_cur = h.copy()
    for iter in range(max_iters):
        # Compute linearization
        cons = [h_cv >= 0, h_cv <= 1, cp.sum(h_cv) == target]
        for k in range(n):
            # f_k(h) ≈ f_k(h_cur) + grad_k^T (h - h_cur)
            # grad_k[j] = (1-h[j+k]) * [j+k<n] - h[j-k] * [j-k>=0]
            f_k_val = np.sum(h_cur[:n-k] * (1 - h_cur[k:]))
            grad_k = np.zeros(n)
            if k == 0:
                grad_k = 1 - 2*h_cur
            else:
                grad_k[:n-k] += (1 - h_cur[k:])
                grad_k[k:] -= h_cur[:n-k]
            cons.append(f_k_val + grad_k @ (h_cv - h_cur) <= t)
        
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, cons)
        prob.solve(solver=cp.SCS)  # or another solver
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_cv.value
        if h_new is None:
            break
        
        if np.max(np.abs(h_new - h_cur)) < 1e-8:
            h_cur = h_new
            break
        h_cur = h_new
    
    return h_cur
```
