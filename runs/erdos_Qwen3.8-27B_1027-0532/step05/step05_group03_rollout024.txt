```python
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def compute_grad_weighted(h, weights, dx):

    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-12:
            continue
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:

            grad[idx1] += w * dx * (1.0 - h[idx1 + k])
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr[n-1:n-1+n]
    return float(np.max(overlaps)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, k, dx):
    """Gradient of overlap at lag k."""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        upper = n - 1 - k
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += dx * (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= dx * h[idx2 - k]
    return grad

def compute_grad_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-12:
            continue
        grad += w * compute_grad_lag(h, k, dx)
    return grad

def iterative_reweight(h, n_iters, lr0, lr_decay, target, temp0, temp_decay):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    # Initialize weights uniformly
    weights = np.ones(n) / n
    
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        overlaps = corr[n-1:n-1+n] * dx
        c_max = float(np.max(overlaps))
        
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        
        # Update weights: softmax emphasizing worst lags
        temp = temp0 * (temp_decay ** it)
        temp = max(temp, 0.001)
        shifted = (overlaps - c_max) / temp
        shifted = np.clip(shifted, -50, 0)
        new_weights = np.exp(shifted)
        sw = new_weights.sum()
        if sw < 1e-30:
            new_weights = np.ones(n) / n
        else:
            new_weights /= sw
        weights = 0.6 * weights + 0.4 * new_weights
        weights /= weights.sum()
        
        lr = lr0 * (lr_decay ** it)
        grad = compute_grad_weighted(h_cur, weights, dx)
        
        # Try step
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new <= c_max + 1e-12:
            h_cur = h_new
        else:
            # Halve learning rate
            h_new = h_cur - 0.5 * lr * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new <= c_max + 1e-12:
                h_cur = h_new
            else:
                lr0 *= 0.9
    
    return best_h, best_score

def lag_surgery(h, lag_indices, max_steps, eps0, target):
    """Focused gradient descent on specific lags."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    eps = eps0
    
    for step in range(max_steps):
        # Combined gradient of selected lags
        grad = np.zeros(n)
        for k in lag_indices:
            grad += compute_grad_lag(h_cur, k, dx)
        grad /= len(lag_indices)
        
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new
            eps = min(eps * 1.02, 0.005)
        elif sc_new < best_score:
            h_cur = h_new
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    
    return best_h, best_score

def top_lags_surgery(h, n_top, max_steps, eps0, target):
    """Surgery on the top n_lags worst lags."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    eps = eps0
    
    for step in range(max_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        overlaps = corr[n-1:n-1+n]
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            grad += compute_grad_lag(h_cur, k, dx)
        grad /= n_top
        
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < best_score - 1e-14:

            h_cur = h_new
            eps = min(eps * 1.015, 0.004)
        else:
            eps *= 0.85

                break
    
    return best_h, best_score

def thue_morse_init(n):
    """Create initial h based on Thue-Morse sequence, scaled to have sum = n/2."""
    # Thue-Morse: t(0)=0, t(n) = popcount(n) mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Scale to have sum = n/2
    # thue-morse already has approximately n/2 ones
    target = n / 2.0
    h = project_feasible(tm, target)
    return h

def periodic_init(n, n_periods, duty=0.5, offset=0.0):
    """Create periodic initial h with n_periods blocks, given duty cycle."""
    h = np.zeros(n)
    period_len = n / n_periods
    block_len = int(period_len * duty)
    offset_idx = int(offset * period_len)
    for p in range(n_periods):
        start = int(p * period_len) + offset_idx
        end = start + block_len
        # Handle wrapping
        for i in range(start, min(end, n)):
            h[i] = 1.0
        if end > n:
            for i in range(0, end - n):
                h[i] = 1.0
    target = n / 2.0
    return project_feasible(h, target)

def slsqp_refine(h, alphas, maxiter, ftol):
    """SLSQP with soft-max smoothing."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(alpha):
        def _obj(x):
            h_local = project_feasible(x, target)
            corr = np.correlate(h_local, 1.0 - h_local, mode='full')
            overlaps = corr[n-1:n-1+n] * dx
            c_max = float(np.max(overlaps))
            shifted = alpha * (overlaps - c_max)
            shifted = np.clip(shifted, -700, 0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        return _obj
    
    def cons_eq(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for alpha in alphas:
        try:
            res = minimize(obj(alpha), h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def binary_project(h, target):
    """Project to binary (bang-bang) keeping the top 'target' values as 1."""
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(len(h))
    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin

def try_binary_improve(h, target):
    """Try binary version and see if it's better."""
    h_bin = binary_project(h, target)
    sc_bin = compute_c5(h_bin)
    return h_bin, sc_bin

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_n = 0
_g_target = 0.0
_g_dx = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr[n-1:n-1+n]
    return float(np.max(overlaps)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_lag(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    else:
        upper = n - 1 - k
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += dx * (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= dx * h[idx2 - k]
    return grad

def compute_grad_weighted(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-14:
            continue
        grad += w * compute_grad_lag(h, k, dx)
    return grad

def _obj_slsqp(x):
    h_local = project_feasible(x, _g_target)
    corr = np.correlate(h_local, 1.0 - h_local, mode='full')
    overlaps = corr[_g_n - 1:_g_n - 1 + _g_n] * _g_dx
    c_max = float(np.max(overlaps))
    shifted = _g_alpha * (overlaps - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_dx, _g_alpha
    n = len(h)
    target = n / 2.0
    _g_n = n
    _g_target = target
    _g_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _g_alpha = float(alpha)
        try:
            res = minimize(_obj_slsqp, h_cur, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def iterative_reweight(h, n_iters, lr0, lr_decay, target, temp0, temp_decay):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    weights = np.ones(n)
    for it in range(n_iters):
        overlaps = compute_overlaps(h_cur)
        c_max = float(np.max(overlaps))
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        temp = temp0 * (temp_decay ** it)
        temp = max(temp, 0.002)
        shifted = (overlaps - c_max) / temp
        shifted = np.clip(shifted, -50.0, 0.0)
        new_weights = np.exp(shifted)
        sw = new_weights.sum()
        if sw < 1e-30:
            new_weights = np.ones(n)
        else:
            new_weights /= sw
        weights = 0.5 * weights + 0.5 * new_weights
        weights /= weights.sum()
        lr = lr0 * (lr_decay ** it)
        grad = compute_grad_weighted(h_cur, weights, dx)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_max + 1e-12:
            h_cur = h_new
        else:
            h_new = h_cur - 0.5 * lr * grad

            sc_new = compute_c5(h_new)
            if sc_new <= c_max + 1e-12:

def top_lags_surgery(h, n_top, max_steps, eps0, target):

    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    eps = eps0
    for step in range(max_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        overlaps = corr[n-1:n-1+n]
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            grad += compute_grad_lag(h_cur, k, dx)
        grad /= float(n_top)
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new
            eps = min(eps * 1.02, 0.005)
        else:
            eps *= 0.82
            if eps < 1e-11:
                break
    return best_h, best_score

    h_bin = np.zeros(len(h))
    n_target = int(round(target))
    h_bin[sorted_idx[:n_target]] = 1.0
    return h_bin

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def downsample_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    block = n_src // n_dst
    h_out = np.zeros(n_dst)
    for i in range(n_dst):
        start = i * block
        end = min(start + block, n_src)
        h_out[i] = np.mean(h_src[start:end])
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_dx, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 100:
        h_ref, sc = slsqp_refine(best_h, [200, 1000, 5000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Iterative reweighting (main new technique)
    if tleft() > 300:
        h_wr, sc_wr = iterative_reweight(best_h, 1000, 0.004, 0.9995, target, 0.015, 0.9997)
        if sc_wr < best_score:
            best_score = sc_wr
            best_h = h_wr

    # Phase 3: Surgery + SLSQP
    if tleft() > 250:
        h_surg, sc_surg = top_lags_surgery(best_h, 3, 300, 0.0012, target)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref, sc = slsqp_refine(best_h, [2000, 10000, 50000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Anti-symmetric optimization
    if tleft() > 200:
        h_anti, sc_anti = antisymmetric_optimize(best_h, 800, 0.003, target)
        h_anti = project_feasible(h_anti, target)
        sc_anti = compute_c5(h_anti)
        if sc_anti < best_score:
            best_score = sc_anti
            best_h = h_anti
            h_ref, sc = slsqp_refine(best_h, [1000, 5000], 1500, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 5: Coarse-to-fine
    if tleft() > 400:
        coarse_n = n // 4
        target_coarse = coarse_n / 2.0
        h_coarse = downsample_to(best_h, coarse_n, target_coarse)
        h_coarse, sc_c = slsqp_refine(h_coarse, [100, 500, 2000], 1500, 1e-13)
        h_coarse, sc_c = iterative_reweight(h_coarse, 400, 0.005, 0.999, target_coarse, 0.02, 0.999)
        if sc_c < best_score * 1.01:
            h_up = interpolate_to(h_coarse, n, target)
            h_up, sc_up = slsqp_refine(h_up, [500, 2000, 10000], 1500, 1e-14)
            h_up, sc_up = iterative_reweight(h_up, 400, 0.003, 0.999, target, 0.01, 0.9995)
            if sc_up < best_score:
                best_score = sc_up
                best_h = h_up

    # Phase 6: Higher resolution
    if tleft() > 350:
        for n_fine in [n + 100, n + 200, n + 400]:
            if tleft() < 150:
                break
            target_fine = n_fine / 2.0
            h_fine = interpolate_to(best_h, n_fine, target_fine)
            h_fine, sc_fine = slsqp_refine(h_fine, [100, 500, 2000], 1200, 1e-13)
            h_fine, sc_fine = iterative_reweight(h_fine, 500, 0.003, 0.999, target_fine, 0.012, 0.9995)
            h_surg_f, sc_sf = top_lags_surgery(h_fine, 3, 200, 0.001, target_fine)
            if sc_sf < sc_fine:
                h_fine, sc_fine = h_surg_f, sc_sf
            h_ref_f, sc_rf = slsqp_refine(h_fine, [5000, 20000], 1000, 1e-14)
            if sc_rf < sc_fine:
                h_fine, sc_fine = h_ref_f, sc_rf
            if sc_fine < best_score:
                best_score = sc_fine
                best_h = h_fine
                n = n_fine
                target = target_fine

    # Phase 7: Binary projection attempt
    if tleft() > 150:
        h_bin = binary_project(best_h, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            h_ref, sc = slsqp_refine(best_h, [500, 2000, 10000], 1500, 1e-14)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
            h_wr, sc_wr = iterative_reweight(best_h, 400, 0.002, 0.999, target, 0.008, 0.999)
            if sc_wr < best_score:
                best_score = sc_wr
                best_h = h_wr

    # Phase 8: More reweighting with different schedule
    if tleft() > 120:
        h_wr, sc_wr = iterative_reweight(best_h, 800, 0.002, 0.9995, target, 0.006, 0.9992)
        if sc_wr < best_score:
            best_score = sc_wr
            best_h = h_wr
        h_surg, sc_surg = top_lags_surgery(best_h, 5, 200, 0.0006, target)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 9: Structured inits + optimize
    deadline_p9 = start_time + (budget_s - 30.0) * 0.82
    if tleft() > 100:
        for init_type in range(4):
            if time.time() > deadline_p9 or tleft() < 80:
                break
            if init_type == 0:
                h_try = thue_morse_init(n)
            elif init_type == 1:
                h_try = periodic_init(n, 3, 0.5, 0.0)
            elif init_type == 2:
                h_try = periodic_init(n, 5, 0.5, 0.1)
            else:
                h_try = periodic_init(n, 4, 0.45, 0.05)
            h_try, sc_try = slsqp_refine(h_try, [500, 2000], 1000, 1e-13)
            h_try, sc_try = iterative_reweight(h_try, 300, 0.003, 0.999, target, 0.01, 0.9995)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try

    # Phase 10: Random restarts
    if tleft() > 80:
        for attempt in range(6):
            if tleft() < 50:
                break
            h_pert = best_h.copy()
            if attempt % 3 == 0:
                h_pert += rng.randn(n) * 0.03
            elif attempt % 3 == 1:
                block = max(5, n // 50)
                for b in range(n // block):
                    if rng.random() < 0.3:
                        h_pert[b*block:min((b+1)*block, n)] += rng.uniform(-0.08, 0.08)
            else:
                tm = thue_morse_init(n)
                h_pert = 0.6 * best_h + 0.4 * tm
            h_pert = project_feasible(h_pert, target)
            h_ref, sc = slsqp_refine(h_pert, [1000, 5000], 1000, 1e-14)
            h_wr, sc_wr = iterative_reweight(h_ref, 300, 0.002, 0.999, target, 0.008, 0.999)
            if sc_wr < sc:
                h_ref, sc = h_wr, sc_wr
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 11: Final polish
    if tleft() > 50:
        h_surg, sc_surg = top_lags_surgery(best_h, 2, 200, 0.0004, target)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref, sc = slsqp_refine(best_h, [10000, 50000, 200000], 1000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 25:
        h_wr, sc_wr = iterative_reweight(best_h, 600, 0.001, 0.9997, target, 0.004, 0.999)
        if sc_wr < best_score:
            best_score = sc_wr
            best_h = h_wr

    if tleft() > 10:
        h_surg, sc_surg = top_lags_surgery(best_h, 3, 100, 0.0003, target)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
