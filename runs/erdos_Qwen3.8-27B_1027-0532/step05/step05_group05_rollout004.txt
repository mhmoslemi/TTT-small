        c_curr = c_max * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        sorted_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[sorted_idx] - corr[sorted_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, sorted_idx):
            k = int(k)
            if k >= 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
            else:
                kk = -k
                j_pos = np.arange(kk, n)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos - kk])
                j_neg = np.arange(0, n - kk)
                grad[j_neg] -= wk * h_cur[j_neg + kk]
        grad *= dx
        
        eps_init = 0.01
        eps = eps_init
        for ls in range(10):
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < c_curr - 1e-15:
                h_cur = h_new
                temp *= 0.995
                for ls2 in range(5):
                    h_try = h_cur - eps * 1.1 * grad
                    h_try = project_feasible(h_try, target)
                    sc_try = compute_c5(h_try)
                    if sc_try < compute_c5(h_cur):
                        h_cur = h_try
                        eps *= 1.1
                    else:
                        break
                break
            else:
                eps *= 0.6
                if eps < 1e-10:
                    break
        else:
            temp *= 0.95
    return best_h, best_score

def block_nm_optimize(h, n, target, block_size, n_rounds, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            h_template = h_cur.copy()
            
            def block_obj(x):
                ht = h_template.copy()
                ht[s:e] = np.clip(x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                return compute_c5(ht)
            
            try:
                res = minimize(block_obj, h_cur[s:e].copy(), method='Nelder-Mead',
                               options={'maxiter': 2000, 'xatol': 1e-10, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score

def coordinate_descent_lse(h, n, target, alpha, n_sweeps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            h_lo = h_cur.copy()
            h_lo[i] = 0.0
            h_hi = h_cur.copy()
            h_hi[i] = 1.0
            f_lo = lse_obj(project_feasible(h_lo, target), alpha)
            f_hi = lse_obj(project_feasible(h_hi, target), alpha)
            f_mid = lse_obj(h_cur, alpha)
            best_v = h_cur[i]
            best_f = f_mid
            for _ in range(30):
                mid = (h_cur[i] + best_v) / 2 if f_mid < best_f else (h_cur[i] + best_v) / 2
                lo = min(h_cur[i], best_v)
                hi = max(h_cur[i], best_v)
                mid = (lo + hi) / 2
                h_test = h_cur.copy()
                h_test[i] = mid
                h_test = project_feasible(h_test, target)
                f_test = lse_obj(h_test, alpha)
                if f_test < best_f:
                    best_v = mid
                    best_f = f_test
                else:
                    h_test2 = h_cur.copy()
                    h_test2[i] = (lo + mid) / 2 if mid > lo else (mid + hi) / 2
                    lo2 = min(mid, h_cur[i])
                    hi2 = max(mid, h_cur[i])
                    mid2 = (lo2 + hi2) / 2
                    h_test2[i] = mid2
                    h_test2 = project_feasible(h_test2, target)
                    f_test2 = lse_obj(h_test2, alpha)
                    if f_test2 < best_f:
                        best_v = mid2
                        best_f = f_test2
            if best_f < f_mid + 1e-15:
                h_cur[i] = best_v
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < cur_score:
            cur_score = sc
    return h_cur, cur_score

def generate_alternatives(n, target, rng, base_h, n_alt=5):
    alts = [base_h.copy()]
    x = np.linspace(0, 2, n)
    for i in range(n_alt):
        if i == 0:
            freq = rng.uniform(1.5, 4.5)
            amp = rng.uniform(0.2, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            h_new = 0.5 + amp * np.cos(2 * np.pi * freq * x + phase)
            h_new += rng.randn(n) * 0.05
        elif i == 1:
            block_size = rng.randint(3, 8)
            h_new = np.zeros(n)
            for b in range(0, n, block_size):
                val = rng.uniform(0.2, 0.8)
                h_new[b:min(b+block_size, n)] = val
        elif i == 2:
            h_new = 0.5 + rng.uniform(0.25, 0.4) * np.sin(2 * np.pi * rng.uniform(2, 6) * x)
            h_new += rng.randn(n) * 0.08
        elif i == 3:
            h_new = base_h.copy()
            for _ in range(20):
                pos = rng.randint(0, n - 10)
                h_new[pos:pos+10] = rng.uniform(0.1, 0.9, 10)
        else:
            h_new = base_h + rng.randn(n) * rng.uniform(0.05, 0.15)
        alts.append(project_feasible(np.clip(h_new, 0.0, 1.0), target))
    return alts

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 8.0

    def tleft():
        return hard_deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_start = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_start.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to smooth out the starting point
    h1, sc1 = slsqp_optimize(h_start, n, target, [50, 200, 1000], 1500, 1e-14, start_time + 25)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: Try alternatives and pick the best
    if tleft() > 200:
        alts = generate_alternatives(n, target, rng, best_h, 4)
        best_alt = best_h
        best_alt_score = best_score
        for alt in alts[1:]:
            if tleft() < 60:
                break
            h_a, sc_a = slsqp_optimize(alt, n, target, [200, 1000, 5000], 1000, 1e-14, min(start_time + 50, hard_deadline - 100))
            if sc_a < best_alt_score:
                best_alt_score = sc_a
                best_alt = h_a
        if best_alt_score < best_score:
            best_score = best_alt_score
            best_h = best_alt

    # Phase 3: High-alpha SLSQP - the main workhorse
    if tleft() > 200:
        h3, sc3 = slsqp_optimize(best_h, n, target, [2000, 5000, 10000, 30000], 2000, 1e-15, min(start_time + 200, hard_deadline - 150))
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Subgradient with line search
    if tleft() > 150:
        h4, sc4 = subgradient_line_search(best_h, n, target, 800, 5, 0.01, min(start_time + 300, hard_deadline - 100))
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
        h4b, sc4b = slsqp_optimize(best_h, n, target, [5000, 20000], 1500, 1e-15, min(start_time + 340, hard_deadline - 80))
        if sc4b < best_score:
            best_score = sc4b
            best_h = h4b

    # Phase 5: Block Nelder-Mead
    if tleft() > 150:
        for bs in [4, 5, 6]:
            if tleft() < 60:
                break
            h5, sc5 = block_nm_optimize(best_h, n, target, bs, 3, min(start_time + 450, hard_deadline - 80))
            if sc5 < best_score:
                best_score = sc5
                best_h = h5

    # Phase 6: More SLSQP with very high alpha
    if tleft() > 150:
        h6, sc6 = slsqp_optimize(best_h, n, target, [10000, 50000, 100000, 200000], 2500, 1e-16, min(start_time + 550, hard_deadline - 60))
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Perturbation + SLSQP cycles
    if tleft() > 120:
        scales = [0.02, 0.04, 0.06, 0.03, 0.05, 0.015, 0.08, 0.01]
        for att in range(min(8, int(tleft() // 20))):
            if tleft() < 30:
                break
            if att % 2 == 0:
                h_pert = best_h + rng.randn(n) * scales[att]
            else:
                h_pert = best_h.copy()
                n_flips = rng.randint(5, 15)
                for _ in range(n_flips):
                    pos = rng.randint(0, n - 3)
                    h_pert[pos:pos+3] = rng.uniform(0.1, 0.9, 3)
            h_pert = project_feasible(h_pert, target)
            h_p, sc_p = slsqp_optimize(h_pert, n, target, [5000, 20000, 100000], 1500, 1e-15, min(start_time + 700, hard_deadline - 30))
            if sc_p < best_score:
                best_score = sc_p
                best_h = h_p

    # Phase 8: Subgradient again with smaller temperature
    if tleft() > 80:
        h8, sc8 = subgradient_line_search(best_h, n, target, 500, 3, 0.005, min(start_time + 800, hard_deadline - 40))
        if sc8 < best_score:
            best_score = sc8
            best_h = h8
        h8b, sc8b = slsqp_optimize(best_h, n, target, [50000, 200000, 500000], 2000, 1e-16, min(start_time + 850, hard_deadline - 25))
        if sc8b < best_score:
            best_score = sc8b
            best_h = h8b

    # Phase 9: Final coordinate descent polish
    if tleft() > 60:
        h9, sc9 = coordinate_descent_lse(best_h, n, target, 20000, 20, min(start_time + 920, hard_deadline - 20))
        if sc9 < best_score:
            best_score = sc9
            best_h = h9
        h9b, sc9b = slsqp_optimize(best_h, n, target, [100000, 500000], 1500, 1e-16, hard_deadline)
        if sc9b < best_score:
            best_score = sc9b
            best_h = h9b

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
