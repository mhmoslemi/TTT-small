So we want to minimize the maximum of the inverse Fourier transform of F(h) * F(1-h) = F(h) * (F(1) - F(h)) = F(h) * (0 - F(h)) = -|F(h)|^2 (for the zero frequency, F(1) = integral of 1 over [0,2] = 2, so F(1-h) = 2*δ_0 - F(h)... this isn't quite right for a non-periodic function).

Actually, for a non-periodic function on [0,2], the Fourier transform is:
F(h)(ω) = ∫_0^2 h(x) e^{-iωx} dx

The correlation (for the infinite line, with h supported on [0,2]) would be:
(R_h)(τ) = ∫ h(x) h(x+τ) dx = (1/2π) ∫ |F(h)(ω)|^2 e^{iωτ} dω

But our overlap is ∫ h(x)(1-h(x+τ)) dx = ∫ h(x) dx - ∫ h(x)h(x+τ) dx = 1 - (R_h)(τ) (for τ where the integral is well-defined, i.e., 0 ≤ τ ≤ 2 assuming h is zero outside [0,2]).

where (R_h)(τ) = ∫_0^{2-τ} h(x)h(x+τ) dx is the autocorrelation.

So we want to maximize the minimum of (R_h)(τ) over τ! Or equivalently, minimize the maximum of 1 - (R_h)(τ), which is the same as minimizing the maximum overlap.

Actually no: we want to minimize max_τ (1 - (R_h)(τ)) = 1 - min_τ (R_h)(τ).

So we want to maximize min_τ (R_h)(τ). We want the autocorrelation to be as uniformly high as possible!

The first term is the partial sum of h, and the second is the partial autocorrelation.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lse_obj_and_grad(h, alpha):
    """Compute LSE smoothed max of correlate(h, 1-h) and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    lse_val = c_max + float(np.log(sum_exp)) / alpha
    weights = exp_s / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    lags = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-12:
            continue
        lag = int(lags[j])
        # dcorr_j/dh[m] = (1-h[m-lag])[0<=m-lag<n] - h[m+lag][0<=m+lag<n]
        ms = max(0, lag)
        me = min(n, n + lag)
        if me > ms:
            grad[ms:me] += w * (1.0 - h[ms - lag:me - lag])
        ms2 = max(0, -lag)
        me2 = min(n, n - lag)
        if me2 > ms2:
            grad[ms2:me2] -= w * h[ms2 + lag:me2 + lag]
    
    return lse_val, grad

def slsqp_with_grad(h, n, target, alphas, maxiter, ftol, deadline):
    """SLSQP with analytic gradient."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if time.time() > deadline:
            break
        alpha = float(a)
        
        def obj(x):
            val, _ = lse_obj_and_grad(x, alpha)
            return val
        
        def grad(x):
            _, g = lse_obj_and_grad(x, alpha)
            return g
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def active_set_subgradient(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    """Subgradient method focusing on active (worst) lags."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Pick top n_active lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        # Exponential weighting
        diffs = corr[top_idx] - corr[top_idx[0]]
        temperatures = [0.001, 0.003, 0.01, 0.03]
        temp = temperatures[min(step // 100, 3)]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        # Compute gradient from active lags
        grad = np.zeros(n)
        for wk, k_idx in zip(weights, top_idx):
            lag = int(k_idx) - (n - 1)
            ms = max(0, lag)
            me = min(n, n + lag)
            if me > ms:
                grad[ms:me] += wk * (1.0 - h_cur[ms - lag:me - lag])
            ms2 = max(0, -lag)
            me2 = min(n, n - lag)
            if me2 > ms2:
                grad[ms2:me2] -= wk * h_cur[ms2 + lag:me2 + lag]
        
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    
    return best_h, best_score
