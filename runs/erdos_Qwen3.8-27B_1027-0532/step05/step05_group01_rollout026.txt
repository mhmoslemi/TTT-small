```python
            best_h = h.copy()
        
        # Compute gradient of log-sum-exp
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        w = np.exp(alpha_param * (top_vals - c_max))
        w /= w.sum()
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, alpha):
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, -k)
    hi1 = min(n, n - k)
    if hi1 > lo1:
        js = np.arange(lo1, hi1)
        g[js] += (1.0 - h[js + k])
    lo2 = max(0, k)
    hi2 = min(n, n + k)
    if hi2 > lo2:
        js2 = np.arange(lo2, hi2)
        g[js2] -= h[js2 - k]
    return g

def nesterov_optimize(h_init, n, target, time_limit, alpha=80, n_top=8, lr_init=0.005, beta=0.9):
    """Fast Nesterov-accelerated gradient descent on smooth max of overlaps."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h_y = h.copy()  # Nesterov's y
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0
    
    for step in range(200000):
        if time.time() - start > time_limit:
            break
        
        # Compute objective and gradient at y
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
        
        # Gradient of log-sum-exp
        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        w = np.exp(alpha * (top_vals - c_max))
        w /= w.sum()
        
        g = np.zeros(n)
        mid = n - 1
        for wi, m in zip(w, top_idx):
            k = int(m) - mid
            g += wi * grad_at_shift(h, k)
        
        g *= dx
        g -= np.mean(g)  # maintain sum constraint
        
        # Nesterov step
        y = h + beta * (h - h_y)
        h_y = h.copy()
        
        # Gradient at y for the step (or at h for simplicity)
        # Let's use gradient at h for simplicity (standard Nesterov uses grad at y)
        # Compute grad at y
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max_y = np.max(corr_y)
        top_idx_y = np.argsort(corr_y)[-n_top:]
        top_vals_y = corr_y[top_idx_y]
        w_y = np.exp(alpha * (top_vals_y - c_max_y))
        w_y /= w_y.sum()
        
        g_y = np.zeros(n)
        for wi, m in zip(w_y, top_idx_y):
            k = int(m) - mid
            g_y += wi * grad_at_shift(y, k)
        g_y *= dx
        g_y -= np.mean(g_y)
        
        h_new = y - lr * g_y
        h_new = project_feasible(h_new, target)
        
        # Check if improved
        c5_new = compute_c5(h_new)
        if c5_new > c5_now + 0.01:
            # Step was too large, reduce lr
            lr *= 0.5
            h_new = h  # stay put
        else:
            h = h_new
        
    return best_h, best_c5
```

Wait, I want to also try one more idea: **the "dual" approach**.

The problem is:
min_h max_k O_k(h)
s.t. 0 ≤ h ≤ 1, sum(h) = target

By minimax duality (Sion's theorem), if the feasible set is convex and O_k is convex in h (which it's not, since O_k is quadratic with indefinite Hessian), we can't directly swap min and max.

However, we can use a **penalty/dual** approach:
min_h max_k [O_k(h) + λ_k * (sum(h) - target)]

This is a saddle point problem. We can solve it with a **primal-dual** method:
- Inner: min_h ∑_k p_k * O_k(h) + λ * (sum(h) - target)² (for fixed p, λ)
- Outer: update p_k ∝ exp(η * O_k(h)) and λ

The inner problem is a weighted sum of quadratics, which is non-convex but can be solved with gradient descent.

This is essentially what the log-sum-exp approach does (with p_k = softmax(O_k/α)). So my plan is already implementing this.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    lo1 = max(0, -k)
    hi1 = min(n, n - k)
    if hi1 > lo1:
        js = np.arange(lo1, hi1)
        g[js] += (1.0 - h[js + k])
    lo2 = max(0, k)
    hi2 = min(n, n + k)
    if hi2 > lo2:
        js2 = np.arange(lo2, hi2)
        g[js2] -= h[js2 - k]
    return g

def smooth_obj_slsqp(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_slsqp_n = 0
_slsqp_target = 0.0

def _slsqp_cons_fun(h):
    return np.sum(h) - _slsqp_target

def _slsqp_cons_jac(h):
    return np.ones(_slsqp_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha):
    global _slsqp_n, _slsqp_target
    _slsqp_n = n
    _slsqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_slsqp, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=30):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 150:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(2 * n - 1)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(g[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def nesterov_opt(h_init, n, target, time_limit, alpha_init=100, alpha_min=30,
                 n_top=8, lr_init=0.003, beta=0.9, anneal_rate=0.0005):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    alpha = alpha_init
    mid = n - 1
    no_improve = 0

    for step in range(500000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 8000:
            break

        c_max = np.max(corr)
        top_idx = np.argsort(corr)[-n_top:]
        top_vals = corr[top_idx]
        w = np.exp(alpha * (top_vals - c_max))
        w /= w.sum()

        g = np.zeros(n)
        for wi, m in zip(w, top_idx):
            k = int(m) - mid
            g += wi * grad_at_shift(h, k)
        g *= dx
        g -= np.mean(g)

        y = h + beta * (h - h_prev)
        h_prev = h.copy()

        corr_y = np.correlate(y, 1.0 - y, mode='full')
        c_max_y = np.max(corr_y)
        top_idx_y = np.argsort(corr_y)[-n_top:]
        top_vals_y = corr_y[top_idx_y]
        w_y = np.exp(alpha * (top_vals_y - c_max_y))
        w_y /= w_y.sum()

        g_y = np.zeros(n)
        for wi, m in zip(w_y, top_idx_y):
            k = int(m) - mid
            g_y += wi * grad_at_shift(y, k)
        g_y *= dx
        g_y -= np.mean(g_y)

        h_new = y - lr * g_y
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)

        if c5_new > c5_now + 0.005:
            lr *= 0.5
        elif c5_new < c5_now - 1e-12:
            lr = min(lr * 1.02, 0.05)
        h = h_new

        alpha = max(alpha_min, alpha * (1.0 - anneal_rate))

    return best_h, best_c5

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def topk_cd_refine(h, n, target, time_limit, n_cand=20, n_top_shifts=3):

        n_use = len(best_h)
        target_use = n_use / 2.0
        h_last, c5_last = fast_pair_cd(best_h, n_use, target_use,
                                        time_limit=min(25, tleft() * 0.6),
                                        max_sweeps=1500, n_cand=25)
        if c5_last < best_score:
            best_score = c5_last
            best_h = h_last

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
```
