_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

# Smooth max objective for SLSQP
def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):

    start_t = time.time()
    dx = 2.0 / n
    idxs = np.arange(n)
    mid = n - 1
    
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    num_active = num_active_start
    stagnation = 0
    
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = np.max(corr)
        c5_cur = c_max * dx
        
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_cur.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > 50:
            if num_active < num_active_max:
                num_active = min(num_active * 2, num_active_max)
                stagnation = 0
            else:
                break
        
        # Identify active set: top num_active
        top_idx = np.argsort(corr)[-num_active:]
        
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0  # minimize t
        
        A_ineq_rows = []
        b_ineq_vals = []
        
        for s_idx in top_idx:
            k = int(s_idx) - mid
            g_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            g_k[v1] += (1.0 - h_cur[idxs[v1] - k])
            g_k[v2] -= h_cur[idxs[v2] + k]
            
            rhs = np.dot(g_k, h_cur) - corr[s_idx]
            row = np.concatenate([g_k, [-1.0]])
            A_ineq_rows.append(row)
            b_ineq_vals.append(rhs)
        
        if A_ineq_rows:
            A_ineq = np.array(A_ineq_rows)
            b_ineq = np.array(b_ineq_vals)
        else:
            # No constraints, skip
            break
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 <= h_i <= 1, t is free
        bounds_lp = [(0, 1)] * n + [(None, None)]
        
        try:
            result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq,
                             A_eq=A_eq, b_eq=b_eq,

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
