                alpha = rng.uniform(0.3, 0.7)
                pop[i] = project_feasible(((np.arange(n) * alpha) % 1.0 < 0.5).astype(float), target)
    
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline, top_k=1):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = initial_epsilon
    for step in range(max_steps):
        if time.time() > deadline:
            break
        grad, c_curr = compute_grad_max(h_cur, n, top_k)
        if c_curr < cur_score:
            cur_score = c_curr
        
        accepted = False
        for backtrack in range(15):
            step_size = eps / (2.0 ** backtrack)
            h_new = project_feasible(h_cur - step_size * grad, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc_new
                accepted = True
                break
        if accepted:
            eps = min(eps * 1.08, 0.05)
        else:
            eps *= 0.4
            if eps < 1e-10:
                break
    return h_cur, cur_score

def binary_search_h(n, target, deadline, rng):
    """Find a good binary pattern via local search."""
    # Start with a structured pattern
    best_bin = None
    best_score = float('inf')
    
    # Try several structured patterns
    patterns = []
    for period in [3, 4, 5, 6, 7, 8, 10, 12, 13, 16, 20, 25, 30]:
        p = (np.arange(n) % period) < period / 2.0
        p = p.astype(float)
        # Adjust to get exact target
        p = project_feasible(p, target)
        # Round to binary
        sorted_idx = np.argsort(p)[::-1]
        bin_p = np.zeros(n)
        bin_p[sorted_idx[:int(target)]] = 1.0
        patterns.append(bin_p)
    
    # Also try sturmian sequences
    for alpha in [0.38, 0.42, 0.47, 0.53, 0.62, 0.65]:
        s = ((np.arange(n) * alpha) % 1.0 < 0.5).astype(float)
        sorted_idx = np.argsort(s)[::-1]
        bin_s = np.zeros(n)
        bin_s[sorted_idx[:int(target)]] = 1.0
        patterns.append(bin_s)
    
    # Try random binary
    for _ in range(5):
        r = (rng.rand(n) < 0.5).astype(float)
        sorted_idx = np.argsort(r)[::-1]
        bin_r = np.zeros(n)
        bin_r[sorted_idx[:int(target)]] = 1.0
        patterns.append(bin_r)
    
    # Evaluate all patterns
    for p in patterns:
        sc = compute_c5(p)
        if sc < best_score:
            best_score = sc
            best_bin = p.copy()
    
    # Local search: try random swaps
    for it in range(2000):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(best_bin == 1)[0]
        zeros = np.where(best_bin == 0)[0]
        for _ in range(50):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_bin[i] = 0
            best_bin[j] = 1
            sc = compute_c5(best_bin)
            if sc < best_score:
                best_score = sc
                improved = True
                ones = np.where(best_bin == 1)[0]
                zeros = np.where(best_bin == 0)[0]
            else:
                best_bin[i] = 1
                best_bin[j] = 0
        if not improved:
            # Try larger moves: block swaps
            block_size = rng.randint(2, 8)
            improved2 = False
            for _ in range(20):
                i = rng.randint(0, n - block_size)
                j = rng.randint(0, n - block_size)
                if abs(i - j) < block_size:
                    continue
                # Check if blocks have same pattern (same number of 1s)
                block_i = best_bin[i:i+block_size].sum()
                block_j = best_bin[j:j+block_size].sum()
                if block_i != block_j:
                    continue
                temp = best_bin[i:i+block_size].copy()
                best_bin[i:i+block_size] = best_bin[j:j+block_size]
                best_bin[j:j+block_size] = temp
                sc = compute_c5(best_bin)
                if sc < best_score:
                    best_score = sc
                    improved2 = True
                else:
                    best_bin[j:j+block_size] = temp
                    best_bin[i:i+block_size] = best_bin[j:j+block_size]
                    best_bin[j:j+block_size] = temp  # This is wrong, let me fix
            if not improved2 and not improved:
                break
    
    return best_bin, best_score
