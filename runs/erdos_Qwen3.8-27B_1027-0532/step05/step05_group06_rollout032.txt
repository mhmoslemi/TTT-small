def project_feasible(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    return float(np.max(corr)) * dx

def lse_objective(x, alpha, target, mu):

    s = float(np.sum(x))

    return val

    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)

    def elapsed():
        return time.time() - start_time

    def tleft():
        return budget_s - 45.0 - elapsed()

    # Phase 1: Optimize at base resolution
    alphas_p1 = [20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if tleft() < 200:
            break
        h_new = lbfgs_opt(h, target_base, a, mu, 800)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new

    # Phase 2: Upsample to 2x and do extended optimization
    n_fine = 2 * n_base
    target_fine = n_fine / 2.0
    h_fine = upsample(best_h, n_fine)
    h_fine = project_feasible(h_fine, target_fine)
    sc = compute_c5(h_fine)
    if sc < best_score:
        best_score = sc
        best_h = h_fine.copy()

    alphas_p2 = [5, 20, 50, 200, 1000, 5000, 20000, 100000, 500000, 2000000]
    max_iters_p2 = [400, 400, 500, 500, 600, 600, 600, 500, 500, 400]
    for i, a in enumerate(alphas_p2):
        if tleft() < 150:
            break
        h_fine = lbfgs_opt(h_fine, target_fine, a, mu, max_iters_p2[i])
        sc = compute_c5(h_fine)
        if sc < best_score:
            best_score = sc
            best_h = h_fine.copy()

    # Phase 3: Top-k focused optimization at fine resolution
    if tleft() > 150:
        for k in [3, 5, 7, 10]:
            if tleft() < 60:
                break
            for at in [30, 80]:
                if tleft() < 40:
                    break
                h_topk = topk_opt(best_h, target_fine, k, mu, at, 300)
                sc = compute_c5(h_topk)
                if sc < best_score:
                    best_score = sc
                    best_h = h_topk
                h_topk = lbfgs_opt(h_topk, target_fine, 100000, mu, 400)
                sc = compute_c5(h_topk)
                if sc < best_score:
                    best_score = sc
                    best_h = h_topk

    # Phase 4: Multi-start with structured perturbations at fine resolution
    if tleft() > 200:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        attempts = 0
        max_attempts = 12
        while tleft() > 80 and attempts < max_attempts:
            attempts += 1
            mode = ['phase', 'amplitude', 'shift'][attempts % 3]
            strength = 0.15 + 0.1 * (attempts % 4)
            h_pert = fourier_perturb(best_h, seed=seed + attempts * 7, mode=mode, strength=strength)
            h_pert = project_feasible(h_pert, target_cur)
            h_pert = lbfgs_opt(h_pert, target_cur, 500, mu, 300)
            h_pert = lbfgs_opt(h_pert, target_cur, 20000, mu, 300)
            h_pert = lbfgs_opt(h_pert, target_cur, 200000, mu, 200)
            sc = compute_c5(h_pert)
            if sc < best_score:
                best_score = sc
                best_h = h_pert.copy()

    # Phase 5: Coordinate descent
    if tleft() > 120:
        h_cd, sc_cd = coord_descent(best_h, target_fine, n_sweeps=2)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: SLSQP polish
    if tleft() > 80:
        h_slq = slsqp_polish(best_h, target_fine, alpha=5000, maxiter=300)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Another round of top-k + L-BFGS
    if tleft() > 60:
        for k in [3, 5]:
            if tleft() < 30:
                break
            h_tk = topk_opt(best_h, target_fine, k, mu, 50, 250)
            sc = compute_c5(h_tk)
            if sc < best_score:
                best_score = sc
                best_h = h_tk
            h_new = lbfgs_opt(best_h, target_fine, 1000000, mu, 400)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Final coordinate descent pass
    if tleft() > 60:
        h_cd, sc_cd = coord_descent(best_h, target_fine, n_sweeps=1)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: Final SLSQP
    if tleft() > 30:
        h_slq = slsqp_polish(best_h, target_fine, alpha=100000, maxiter=200)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_final = project_feasible(best_h, target_fine)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, best_score, len(h_final)
