    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    total = 2 * n - 1
    if top_m < total:
        idx = np.argpartition(corr, -top_m)[-top_m:]
        sub = corr[idx]
    else:
        idx = np.arange(total)
        sub = corr
    c_max = np.max(sub)
    shifted = np.clip(alpha * (sub - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    obj = (c_max + np.log(wsum) / alpha) * dx
    if wsum < 1e-300:
        return obj, np.zeros(n)
    weights = exp_s / wsum
    grad = np.zeros(n)
    half = n - 1
    for i in range(len(weights)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = idx[i] - half
        if k >= 0:
            L = n - k
            if L > 0:
                grad[:L] += w * (1.0 - h[k:]) * dx
            if k > 0 and L > 0:
                grad[k:] -= w * h[:L] * dx
        else:
            m = -k
            L = n - m
            if L > 0:
                grad[m:] += w * (1.0 - h[:L]) * dx
                grad[:L] -= w * h[m:] * dx
    return obj, grad

def _smooth_obj_wrapper(h, alpha, top_m):
    return smooth_obj_and_grad(h, alpha, top_m)[0]

def _smooth_grad_wrapper(h, alpha, top_m):
    return smooth_obj_and_grad(h, alpha, top_m)[1]

def slsqp_phase(h_init, n, target, alphas, top_ms, maxiter_list, time_limit, start_time, deadline_off):
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    for i in range(len(alphas)):
        if time.time() - start_time > time_limit - deadline_off:
            break
        alpha = alphas[i]
        top_m = top_ms[min(i, len(top_ms) - 1)]
        maxiter = maxiter_list[min(i, len(maxiter_list) - 1)]
        try:
            res = minimize(_smooth_obj_wrapper, h, args=(alpha, top_m),
                           jac=_smooth_grad_wrapper,
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: np.sum(x) - target,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h) - 1e-15:
                    h = h_new
        except Exception:
            pass
    return h

def momentum_grad(h_init, n, target, time_limit, start_time, deadline_off,
                  alphas=None, top_ms=None, steps_per_alpha=2000, beta=0.8):
    if alphas is None:
        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    if top_ms is None:
        top_ms = [30, 30, 25, 20, 15, 15, 10, 10, 10]
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit - deadline_off:
            break
        top_m = top_ms[min(ai, len(top_ms) - 1)]
        v = np.zeros(n)
        stalled = 0
        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_off:
                break
            obj, grad = smooth_obj_and_grad(h, alpha, top_m)
            v = beta * v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(v))
            if gnorm < 1e-14:
                break
            lr = min(1.0 / (alpha * gnorm + 1e-10), 1.0)
            improved = False
            for _ in range(30):
                h_try = project_feasible(h - lr * v, target)
                sc = compute_c5(h_try)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 80:
                    break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_perturb(h, rng, frac_flip):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h_bin))
    n_zeros = n - n_ones
    n_flip = max(1, int(n * frac_flip))
    if n_ones > 0 and n_zeros > 0:
        n_flip_1 = min(n_ones, n_flip // 2 + n_flip % 2)
        n_flip_0 = min(n_zeros, n_flip - n_flip_1)
        one_idx = np.where(h_bin == 1)[0]
        zero_idx = np.where(h_bin == 0)[0]
        flip_to_0 = rng.choice(one_idx, size=n_flip_1, replace=False)
        flip_to_1 = rng.choice(zero_idx, size=n_flip_0, replace=False)
        h_bin[flip_to_0] = 0.0
        h_bin[flip_to_1] = 1.0
    return h_bin

def _cons_fun(x, target):
    return np.sum(x) - target

def _cons_jac(x, n):
    return np.ones(n)

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    # Phase 1: SLSQP warm-up with moderate alpha
    if tleft() > 300:
        h1 = slsqp_phase(best_h, n, target,
                         alphas=[30, 80, 200, 500],
                         top_ms=[40, 30, 25, 20],
                         maxiter_list=[300, 500, 800, 1000],
                         time_limit=budget_s, start_time=start_time, deadline_off=1000)
        sc = compute_c5(h1)
        if sc < best_score:
            best_score = sc
            best_h = h1

    # Phase 2: Momentum gradient with increasing alpha
    if tleft() > 300:
        h2, sc2 = momentum_grad(best_h, n, target,
                                time_limit=budget_s, start_time=start_time, deadline_off=900,
                                alphas=[100, 300, 1000, 3000],
                                top_ms=[25, 20, 15, 10],
                                steps_per_alpha=3000)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: SLSQP high alpha polish
    if tleft() > 250:
        h3 = slsqp_phase(best_h, n, target,
                         alphas=[1000, 3000, 10000, 30000],
                         top_ms=[15, 10, 10, 8],
                         maxiter_list=[1000, 1500, 2000, 2000],
                         time_limit=budget_s, start_time=start_time, deadline_off=600)
        sc = compute_c5(h3)
        if sc < best_score:
            best_score = sc
            best_h = h3

    # Phase 4: Multiple diverse restarts
    restart_strategies = [
        ('gauss', 0.02), ('gauss', 0.04), ('gauss', 0.08),
        ('binary', 0.05), ('binary', 0.10),
        ('block', 0.03),
    ]
    for strategy, param in restart_strategies:
        if tleft() < 120:
            break
        if strategy == 'gauss':
            h_pert = best_h + rng.randn(n) * param
        elif strategy == 'binary':
            h_pert = binary_perturb(best_h, rng, param)
        else:  # block
            h_pert = best_h.copy()
            block_size = max(3, n // 50)
            start_pos = rng.randint(0, n - block_size)
            shift = rng.randint(-3, 4)
            h_pert[start_pos:start_pos+block_size] = np.roll(
                best_h[start_pos:start_pos+block_size], shift)
        h_pert = project_feasible(h_pert, target)

        h_r = slsqp_phase(h_pert, n, target,
                          alphas=[200, 1000, 5000],
                          top_ms=[15, 10, 8],
                          maxiter_list=[500, 800, 1000],
                          time_limit=budget_s, start_time=start_time, deadline_off=300)
        sc_r = compute_c5(h_r)
        if sc_r < best_score:
            best_score = sc_r
            best_h = h_r

        # Quick momentum polish on the restart result
        if tleft() > 100:
            h_mp, sc_mp = momentum_grad(h_r, n, target,
                                        time_limit=budget_s, start_time=start_time,
                                        deadline_off=budget_s - 200,
                                        alphas=[2000, 10000],
                                        top_ms=[10, 8],
                                        steps_per_alpha=2000)
            if sc_mp < best_score:
                best_score = sc_mp
                best_h = h_mp

    # Phase 5: Try higher resolution
    for n_try in [500, 640, 800]:
        if tleft() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        h_t = slsqp_phase(h_interp, n_try, target_try,
                          alphas=[50, 200, 800, 3000],
                          top_ms=[30, 25, 15, 10],
                          maxiter_list=[500, 800, 1000, 1500],
                          time_limit=budget_s, start_time=start_time, deadline_off=600)
        sc_t = compute_c5(h_t)

        if tleft() > 80:
            h_mp, sc_mp = momentum_grad(h_t, n_try, target_try,
                                        time_limit=budget_s, start_time=start_time,
                                        deadline_off=budget_s - 300,
                                        alphas=[1000, 5000],
                                        top_ms=[15, 10],
                                        steps_per_alpha=1500)
            if sc_mp < sc_t:
                h_t, sc_t = h_mp, sc_mp

        if sc_t < best_score:
            best_score = sc_t
            best_h = h_t
            n = n_try
            target = target_try

    # Phase 6: Final high-alpha SLSQP on best solution
    if tleft() > 40:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_fin = slsqp_phase(best_h, n_final, target_final,
                            alphas=[5000, 20000, 100000],
                            top_ms=[8, 5, 5],
                            maxiter_list=[1500, 2000, 2000],
                            time_limit=budget_s, start_time=start_time, deadline_off=20)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
