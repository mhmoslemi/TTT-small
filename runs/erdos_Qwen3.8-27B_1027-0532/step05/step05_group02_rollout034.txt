def projected_grad_descent(h, n_iters, alpha, lr_init, lr_end, time_limit, start_time):
    """Project onto [0,1]^n with sum = n/2, using smoothed gradient of max corr."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Current C5
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        # Learn rate (geometric decay)
        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio
        
        # Compute smoothed gradient
        max_corr = corr.max()
        shifted = alpha * (corr - max_corr)
        # Clip for numerical stability
        shifted = np.clip(shifted, -50, 0)
        exp_w = np.exp(shifted)
        W = exp_w / exp_w.sum()
        
        # Compute gradient: grad[i] = sum_s W_s * ((1-h[i+s]) - h[i-s])
        grad = np.full(n, W.sum())
        shifts = np.arange(2*n-1) - (n-1)
        
        for k in range(2*n-1):
            wk = W[k]
            if wk < 1e-12:
                continue
            s = shifts[k]
            if s >= 0:
                # h[i+s] valid for i = 0..n-1-s
                grad[:n-s] -= wk * h[s:]
                # h[i-s] valid for i = s..n-1
                grad[s:] -= wk * h[:n-s]
            else:
                abs_s = -s
                # h[i+s] = h[i-abs_s], valid for i = abs_s..n-1
                grad[abs_s:] -= wk * h[:n-abs_s]
                # h[i-s] = h[i+abs_s], valid for i = 0..n-1-abs_s
                grad[:n-abs_s] -= wk * h[abs_s:]
        
        # Update
        h = h - lr * grad
        
        # Project to [0,1] with sum = n/2
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        diff = target - h.sum()
        if diff > 0:
            # Need to increase sum: add to entries with room
            margins = 1.0 - h
            total_margin = margins.sum()
            if total_margin > 0:
                h += (diff / total_margin) * margins
        elif diff < 0:
            # Need to decrease sum: remove from entries with room
            total_headroom = h.sum()
            if total_headroom > 0:
                h += (diff / total_headroom) * h
        
        h = np.clip(h, 0.0, 1.0)
        # Final adjustment
        diff = target - h.sum()
        if abs(diff) > 1e-10:

            else:
                free = np.where(h > 0.0)[0]
            if len(free) > 0:
                h[free] += diff / len(free)
                h = np.clip(h, 0.0, 1.0)
    
    return best_h, best_c5

def compute_c5(h):

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def projected_grad_descent(h, n_iters, topk, lr_init, lr_end, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy().astype(float)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = float(corr.max()) * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    stall = 0
    max_stall = 5000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        lr = lr_init * (lr_end / lr_init) ** ratio

        topk_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k in topk_idx:
            s = shifts[k]
            if s >= 0:
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
            else:
                abs_s = -s
                grad[abs_s:] += (1.0 - h[:n - abs_s])
                grad[:n - abs_s] += (-h[abs_s:])
        grad /= topk

        h = h - lr * grad
        h = project_box_sum(h, target)

        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(corr.max()) * dx
        if curr_c5 < best_c5 - 1e-12:
            best_c5 = curr_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def projected_grad_softmax(h, n_iters, alpha, lr_init, lr_end, time_limit, start_time):

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n

    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_fun(h):
        return np.sum(h) - target

    def cons_jac(h):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(lambda x: smooth_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 4: Projected gradient descent (softmax)
    allowed = min(120.0, tleft() * 0.15)
    if allowed > 20:
        n_iters = int(min(2000000, allowed * 15000))
        h_pgs, c5_pgs = projected_grad_softmax(best_cont_h, n_iters, alpha=50.0,
                                                lr_init=0.03, lr_end=0.0005,
                                                time_limit=allowed,
                                                start_time=start_time)
        if c5_pgs < best_cont_score:
            best_cont_score = c5_pgs
            best_cont_h = h_pgs.copy()

    # Phase 5: SLSQP polish
    allowed = min(tleft() - 20, 120.0)
    if allowed > 30:
        h_ref, sc_ref = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                       alphas=[50, 200, 1000, 5000],
                                       maxiter=3000, ftol=1e-14,
                                       time_limit=allowed, start_time=start_time)
        if sc_ref < best_cont_score:
            best_cont_score = sc_ref
            best_cont_h = h_ref.copy()

    # Phase 6: Iterative improvement loop
    for iteration in range(6):
        if tleft() < 80:
            break

        # Binary SA from best binary
        allowed = min(40.0, tleft() * 0.10)
        if allowed > 15:
            n_pert = rng.randint(5, 12)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]

            n_sa2 = int(min(2000000, allowed * 50000))
            h_sa2, c5_sa2 = sa_improved(h_pert, n_sa2, 0.01, 0.00003, rng,
                                         time_limit=allowed, start_time=start_time)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

        if tleft() < 60:
            break

        # Continuous gradient from best binary
        allowed = min(50.0, tleft() * 0.12)
        if allowed > 20:
            n_iters = int(min(2000000, allowed * 30000))
            h_pg2, c5_pg2 = projected_grad_descent(best_h, n_iters, topk=3,
                                                    lr_init=0.04, lr_end=0.0005,
                                                    time_limit=allowed,
                                                    start_time=start_time)
            if c5_pg2 < best_cont_score:
                best_cont_score = c5_pg2
                best_cont_h = h_pg2.copy()

        if tleft() < 50:
            break

        # SLSQP
        allowed = min(40.0, tleft() * 0.10)
        if allowed > 15:
            h_ref2, sc_ref2 = slsqp_refine(best_cont_h, n_init, float(n_init) / 2.0,
                                            alphas=[200, 1000, 5000],
                                            maxiter=2000, ftol=1e-14,
                                            time_limit=allowed, start_time=start_time)
            if sc_ref2 < best_cont_score:
                best_cont_score = sc_ref2
                best_cont_h = h_ref2.copy()

    # Phase 7: Try different resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8]:
        if tleft() < 70:
            break
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_cont_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)

        allowed = min(20.0, tleft() * 0.08)
        if allowed < 10:
            continue
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=10000, rng=rng,
                                   time_limit=allowed * 0.5, start_time=start_time)

        allowed2 = min(25.0, tleft() * 0.10)
        if allowed2 < 15:
            continue
        n_iters3 = int(min(1000000, allowed2 * 30000))
        h_pg3, c5_pg3 = projected_grad_descent(h_g3, n_iters3, topk=4,
                                                lr_init=0.04, lr_end=0.0005,
                                                time_limit=allowed2,
                                                start_time=start_time)

        if c5_pg3 < best_cont_score:
            best_cont_score = c5_pg3
            best_cont_h = h_pg3.copy()

    # Phase 8: Final SLSQP burst
    if tleft() > 30:
        n_use = len(best_cont_h)
        h_ref3, sc_ref3 = slsqp_refine(best_cont_h, n_use, float(n_use) / 2.0,
                                        alphas=[500, 2000, 10000],
                                        maxiter=5000, ftol=1e-15,
                                        time_limit=max(10.0, tleft() - 10),
                                        start_time=start_time)
        if sc_ref3 < best_cont_score:
            best_cont_score = sc_ref3
            best_cont_h = h_ref3.copy()

    # Phase 9: Final topk gradient
    if tleft() > 20:
        n_use = len(best_cont_h)
        n_iters_f = int(min(1000000, max(50000, (tleft() - 10) * 40000)))
        h_pgf, c5_pgf = projected_grad_descent(best_cont_h, n_iters_f, topk=3,
                                                lr_init=0.02, lr_end=0.0001,
                                                time_limit=max(5.0, tleft() - 5),
                                                start_time=start_time)
        if c5_pgf < best_cont_score:
            best_cont_score = c5_pgf
            best_cont_h = h_pgf.copy()

    # Final: choose best between binary and continuous
    n_final = len(best_cont_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_cont_h, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_h, int(n_final / 2.0))
    bin_score = compute_c5(h_bin_final)

    if bin_score < final_score:
        return h_bin_final, bin_score, n_final
    else:
        return h_final, final_score, n_final
