This can be computed efficiently using the correlation structure.

Let me define:
For each k, ∂c_k/∂h[j] = dx * [(1-h[j+k]) * 1_{0≤j<n, 0≤j+k<n} - h[j-k] * 1_{0≤j-k<n, 0≤j<n}]

Let's define A_k[j] = ∂c_k/∂h[j] / dx = (1-h[j+k]) * 1_{valid1} - h[j-k] * 1_{valid2}

Then ∇f_α(h)[j] = dx * Σ_k w_k * A_k[j]

We can compute this as:
∇f_α(h)[j] = dx * [(Σ_k w_k * (1-h[j+k]) * 1_{valid1}) - (Σ_k w_k * h[j-k] * 1_{valid2})]

The first term: for fixed j, the sum over k where j+k is valid (i.e., 0 ≤ j+k < n, so -j ≤ k < n-j):

The second term: for fixed j, the sum over k where j-k is valid (i.e., 0 ≤ j-k < n, so j-n+1 ≤ k ≤ j):
Σ_{k=j-n+1}^{j} w_k * h[j-k]

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # c_k = corr[k] * dx for k = 0, ..., 2n-2
    c = corr * dx
    # Shifted for numerical stability
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum  # softmax weights
    
    # Now compute gradient
    # corr has length 2n-1, index k corresponds to lag k-(n-1)
    half_n = n - 1
    grad = np.zeros(n)
    
    for j in range(n):
        # First term: (1-h[j+k]) for valid k where 0<=j+k<n
        # k ranges from 0 to 2n-2 in corr indexing
        # lag = k - half_n
        # We need 0 <= j + lag < n, i.e., -j <= lag <= n-1-j
        # So corr index k = lag + half_n ranges from half_n - j to half_n + n - 1 - j
        
        # Second term: h[j-k] for valid k where 0<=j-k<n
        # lag = k - half_n
        # We need 0 <= j - lag < n, i.e., j-n+1 <= lag <= j
        # So corr index k ranges from half_n + j - n + 1 to half_n + j
        
        # First term
        k_lo1 = max(0, half_n - j)
        k_hi1 = min(2*n-2, half_n + n - 1 - j)
        if k_hi1 >= k_lo1:
            lags1 = np.arange(k_lo1, k_hi1+1) - half_n
            vals1 = 1.0 - h[j + lags1]
            grad[j] += np.dot(w[k_lo1:k_hi1+1], vals1)
        
        # Second term
        k_lo2 = max(0, half_n + j - n + 1)
        k_hi2 = min(2*n-2, half_n + j)
        if k_hi2 >= k_lo2:
            lags2 = np.arange(k_lo2, k_hi2+1) - half_n
            vals2 = h[j - lags2]
            grad[j] -= np.dot(w[k_lo2:k_hi2+1], vals2)
    
    grad *= dx
    return grad

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500, 0)
    w = np.exp(shifted)
    w = w / np.sum(w)
    
    half_n = n - 1
    grad = np.zeros(n)
    
    # First term: for each j, sum over valid k of w[k] * (1-h[j+k-half_n])
    # j+k-half_n ranges from j-half_n to j-half_n+2n-2
    # We need 0 <= j+k-half_n < n
    
    # Create index matrix
    j_vals = np.arange(n)
    k_vals = np.arange(2*n-1)
    # indices[i,j] = j + k_vals[j] - half_n (the index into h for the (1-h) term)
    # valid1[i,j] = 0 <= j + k_vals[j] - half_n < n
    
    # This is a (n, 2n-1) matrix which might be large
    # For n=400, it's 400*799 = 319,600 elements - manageable
    
    idx1 = j_vals[:, None] + k_vals[None, :] - half_n  # (n, 2n-1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    vals1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    grad += vals1.T @ w  # (n,) = (2n-1, n) @ (2n-1,)
    
    # Second term: for each j, sum over valid k of w[k] * h[j-k+half_n]
    idx2 = j_vals[:, None] - (k_vals[None, :] - half_n)  # = j_vals + half_n - k_vals
    valid2 = (idx2 >= 0) & (idx2 < n)
    vals2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    grad -= vals2.T @ w
    
    grad *= dx
    return grad

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = 1e6 * (np.sum(h) - target) ** 2
    return val + penalty


def smooth_obj_no_pen(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_grad(h, alpha, n, target):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum

    half_n = n - 1
    j_vals = np.arange(n)
    k_vals = np.arange(2 * n - 1)

    idx1 = j_vals[:, None] + k_vals[None, :] - half_n
    valid1 = (idx1 >= 0) & (idx1 < n)
    safe_idx1 = np.clip(idx1, 0, n - 1)
    vals1 = np.where(valid1, 1.0 - h[safe_idx1], 0.0)
    grad = vals1.T @ w

    idx2 = j_vals[:, None] - (k_vals[None, :] - half_n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    safe_idx2 = np.clip(idx2, 0, n - 1)
    vals2 = np.where(valid2, h[safe_idx2], 0.0)
    grad -= vals2.T @ w

    grad *= dx
    return grad


def sum_constraint(h, n, target):
    return np.array([np.sum(h) - target])


def sum_jacobian(h, n, target):
    return np.array([np.ones(n)])


def slsqp_with_grad(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, n, target),
             'jac': lambda x: sum_jacobian(x, n, target)}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start > time_limit:

        try:

                           jac=lambda x: smooth_grad(x, alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})

    return h_cur, cur_score

    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.ECOS, max_iter=100000, absltol=1e-10, reltol=1e-10)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=80000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())

        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.9)
        else:
            damping = min(0.95, damping + 0.03)

        if it % 50 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.7)

    return best_h, best_c5


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5, 1.0]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved:
            if step > 100:
                break
            h_cur = h_best_perturb(h_cur, target, 0.02)

    return best_h, best_c5


def h_best_perturb(h, target, noise_level):
    rng_local = np.random.RandomState(int(np.sum(h * 7919)) % 10000)
    h_pert = h + rng_local.randn(len(h)) * noise_level
    return project_feasible(h_pert, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def all_constr_lp(h_init, n, target, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    half_n = n - 1
    all_k = list(range(1, 2 * n - 1))

    for it in range(50):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k_idx in all_k:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + 1e-5 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.ECOS, max_iter=200000, absltol=1e-10, reltol=1e-10)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000)
                solved = True
            except Exception:
                break

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.7 * h + 0.3 * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: SLSQP with analytical gradient - moderate alpha
    if tleft() > 300:
        h_new, sc = slsqp_with_grad(best_h, n, target,
                                     [30, 100, 300], 3000, 1e-14, tleft() * 0.4)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: LP constraint generation
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_time,
                                     damping_init=0.7, reg=1e-4, n_active=40)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: SLSQP with high alpha after LP
    if tleft() > 200:
        h_new, sc = slsqp_with_grad(best_h, n, target,
                                     [300, 1000, 3000], 4000, 1e-15, tleft() * 0.35)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Another LP round with tighter settings
    if tleft() > 150:
        lp_time = min(150, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_time,
                                     damping_init=0.85, reg=1e-5, n_active=50)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: Subgradient with backtracking
    if tleft() > 80:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=10000, time_limit=min(40, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: SLSQP polish with very high alpha
    if tleft() > 100:
        h_new, sc = slsqp_with_grad(best_h, n, target,
                                     [1000, 5000, 20000], 3000, 1e-16, tleft() * 0.4)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Trust-constr for different landscape exploration
    if tleft() > 150:
        h_new, sc = trust_constr_opt(best_h, n, target,
                                     [200, 1000, 5000], 2000, tleft() * 0.3)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: LP with all constraints (smaller n for speed)
    if tleft() > 120:
        n_small = 200
        target_small = n_small / 2.0
        h_small = interpolate_to(best_h, n_small, target_small)
        h_lp, c5_lp = all_constr_lp(h_small, n_small, target_small,
                                    time_limit=min(60, tleft() * 0.3))
        if c5_lp < best_score:
            # Interpolate back to full resolution and refine
            h_back = interpolate_to(h_lp, n, target)
            h_new, sc = slsqp_with_grad(h_back, n, target,
                                         [100, 500], 2000, 1e-15, min(30, tleft() * 0.2))
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 9: Multi-start perturbation
    for noise_level in [0.05, 0.1, 0.2]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_level
        h_pert = project_feasible(h_pert, target)
        if tleft() > 60:
            h_new, sc = slsqp_with_grad(h_pert, n, target,
                                         [50, 300, 1000], 2000, 1e-15, min(40, tleft() * 0.5))
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 10: Final LP polish
    if tleft() > 80:
        lp_time = min(60, tleft() * 0.5)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=50000, time_limit=lp_time,
                                     damping_init=0.9, reg=1e-6, n_active=50)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final SLSQP with very high alpha
    if tleft() > 40:
        h_new, sc = slsqp_with_grad(best_h, n, target,
                                     [5000, 20000, 50000], 2000, 1e-16, tleft() * 0.7)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 12: Final subgradient polish
    if tleft() > 20:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                          max_steps=5000, time_limit=min(15, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
