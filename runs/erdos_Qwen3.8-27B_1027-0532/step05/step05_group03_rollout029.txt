```python
def compute_corr(h):

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx  # overlaps for lags 0 to n-1

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, target_sum):

            break

            mask = h < 1.0
        else:

        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
```

Actually, for a single swap (turn off position i, turn on position j), the change in the overlap vector can be computed analytically:
- overlap_k = ∑_{t=0}^{n-1-k} h[t](1-h[t+k])

- Δoverlap_k = (change due to h[i]) + (change due to h[j])

- For each position i (currently 1) and each lag k: contrib_i[k] = (1-h[i+k]) + h[i-k] (with boundary handling)
- For each position j (currently 0) and each lag k: contrib_j[k] = (1-h[j+k]) + h[j-k]

Then Δoverlap_{i,j}[k] = -contrib_i[k] + contrib_j[k]

I can precompute contrib_i for all i in O(n²) and contrib_j for all j in O(n²).
Then for each (i,j) pair, Δoverlap = -contrib_i + contrib_j, and new_overlaps = old_overlaps + Δoverlap.
The new C₅ is max(new_overlaps) = max(old_overlaps - contrib_i + contrib_j).

I can compute this for all (i,j) pairs at once using broadcasting:
- contrib_i: shape (n_ones, n_lags)
- contrib_j: shape (n_zeros, n_lags)
- new_c5[i,j] = max(old_overlaps - contrib_i[i,:] + contrib_j[j,:])

This is O(n_ones * n_zeros * n_lags) = O(n³/4) which for n=400 is about 16M operations. With numpy vectorization, this should be fast (maybe 1-2 seconds per round).

```python
def fast_swap_descent(h, max_rounds=200, time_limit=None, start_time=None):
    n = len(h)
    h_cur = h.copy().astype(float)
    target = n // 2
    
    # Precompute overlap vector
    def get_overlaps(h_vec):
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return corr[n-1:n-1+n] * (2.0/n)
    
    over = get_overlaps(h_cur)
    cur_c5 = float(np.max(over))
    best_h = h_cur.copy()
    best_c5 = cur_c5
    
    for round_num in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_lags = n
        
        # Precompute contributions
        # For each one position i and each lag k:
        # contrib_i[k] = (1-h[i+k]) if i+k < n else 0, plus h[i-k] if i-k >= 0 else 0
        # (these are the terms removed from overlap_k when we turn off h[i])
        
        # Vectorize: for each i in ones, compute the vector of contributions
        # contrib[i_idx, k] = (1 - h[i+k]) * (i+k < n) + h[i-k] * (i-k >= 0)
        
        # Actually, let's be more careful. The overlap is:
        # over[k] = sum_{t=0}^{n-1-k} h[t] * (1 - h[t+k])
        # 
        # When we set h[i] = 0 (was 1):
        #   - Terms where t = i: h[i](1-h[i+k]) = 1*(1-h[i+k]) = 1-h[i+k], present if i <= n-1-k i.e. i+k < n
        #   - Terms where t+k = i, i.e., t = i-k: h[i-k](1-h[i]) = h[i-k]*1 = h[i-k], present if i-k >= 0 and i-k <= n-1-k i.e. i <= n-1
        #     (the second condition is always true since i < n)
        #   So removal_i[k] = (1-h[i+k])*(i+k<n) + h[i-k]*(i-k>=0)
        #
        # When we set h[j] = 1 (was 0):
        #   - Terms where t = j: h[j](1-h[j+k]) = 0*(1-h[j+k]) = 0 → 1*(1-h[j+k]) = 1-h[j+k], present if j+k < n
        #     So addition from t=j: (1-h[j+k])*(j+k<n) - 0 = (1-h[j+k])*(j+k<n)
        #   - Terms where t+k = j, i.e., t = j-k: h[j-k](1-h[j]) = h[j-k]*1 = h[j-k] → h[j-k]*0 = 0
        #     So addition from t=j-k: 0 - h[j-k]*(j-k>=0) = -h[j-k]*(j-k>=0)
        #   So addition_j[k] = (1-h[j+k])*(j+k<n) - h[j-k]*(j-k>=0)
        #
        # Net change: Δover[k] = -removal_i[k] + addition_j[k]
        #                    = -(1-h[i+k])*(i+k<n) - h[i-k]*(i-k>=0) + (1-h[j+k])*(j+k<n) - h[j-k]*(j-k>=0)
        
        # Vectorized computation
        lags = np.arange(n_lags)
        
        # For ones: I[i_idx, k] = (1-h_cur[ones[i_idx]+k]) * (ones[i_idx]+k < n)
        ones_pos = ones[:, None]  # (n_ones, 1)
        lag_idx = lags[None, :]   # (1, n_lags)
        
        # indices for h[ones + k]
        sum_idx = ones_pos + lag_idx  # (n_ones, n_lags)
        mask_sum = sum_idx < n
        # Safe index
        safe_sum_idx = np.minimum(sum_idx, n-1)
        rem1 = (1.0 - h_cur[safe_sum_idx]) * mask_sum  # (n_ones, n_lags)
        
        # indices for h[ones - k]
        diff_idx = ones_pos - lag_idx
        mask_diff = diff_idx >= 0
        safe_diff_idx = np.maximum(diff_idx, 0)
        rem2 = h_cur[safe_diff_idx] * mask_diff  # (n_ones, n_lags)
        
        removal = rem1 + rem2  # (n_ones, n_lags)
        
        # For zeros:
        zeros_pos = zeros[:, None]
        sum_idx_z = zeros_pos + lag_idx
        mask_sum_z = sum_idx_z < n
        safe_sum_z = np.minimum(sum_idx_z, n-1)
        add1 = (1.0 - h_cur[safe_sum_z]) * mask_sum_z
        
        diff_idx_z = zeros_pos - lag_idx
        mask_diff_z = diff_idx_z >= 0
        safe_diff_z = np.maximum(diff_idx_z, 0)
        add2 = h_cur[safe_diff_z] * mask_diff_z
        
        addition = add1 - add2  # (n_zeros, n_lags)
        
        # new_overlaps[i,j,k] = over[k] - removal[i,k] + addition[j,k]
        # new_c5[i,j] = max_k new_overlaps[i,j,k]
        
        # Shape: (n_ones, n_zeros, n_lags)
        # This might be too large: 200 * 200 * 400 = 16M floats = 128MB. Should be OK.
        
        new_over = over[None, None, :] - removal[:, None, :] + addition[None, :, :]
        new_c5 = np.max(new_over, axis=2)  # (n_ones, n_zeros)
        
        best_idx = np.unravel_index(np.argmin(new_c5), new_c5.shape)
        min_val = new_c5[best_idx]
        
        if min_val < cur_c5 - 1e-15:
            i_pos = int(ones[best_idx[0]])
            j_pos = int(zeros[best_idx[1]])
            h_cur[i_pos] = 0
            h_cur[j_pos] = 1
            cur_c5 = float(min_val)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
        else:
            break
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    excess = int(np.sum(h_bin)) - int(target)
    if excess > 0:
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        h_bin[ones_sorted[:excess]] = 0
    elif excess < 0:
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        h_bin[zeros_sorted[:(-excess)]] = 1
    return h_bin

def fast_swap_descent(h, max_rounds=300, time_limit=None, start_time=None):
    n = len(h)
    h_cur = h.copy().astype(float)
    
    over = compute_overlaps(h_cur)
    cur_c5 = float(np.max(over))
    best_h = h_cur.copy()
    best_c5 = cur_c5
    
    lags = np.arange(n)
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        ones_pos = ones[:, None]
        zeros_pos = zeros[:, None]
        lag_idx = lags[None, :]
        
        # removal[i, k] for ones
        sum_idx = ones_pos + lag_idx
        mask_s = sum_idx < n
        ss = np.minimum(sum_idx, n-1)
        rem1 = (1.0 - h_cur[ss]) * mask_s
        
        diff_idx = ones_pos - lag_idx
        mask_d = diff_idx >= 0
        sd = np.maximum(diff_idx, 0)
        rem2 = h_cur[sd] * mask_d
        
        removal = rem1 + rem2
        
        # addition[j, k] for zeros
        sum_idx_z = zeros_pos + lag_idx
        mask_s_z = sum_idx_z < n
        ss_z = np.minimum(sum_idx_z, n-1)
        add1 = (1.0 - h_cur[ss_z]) * mask_s_z
        
        diff_idx_z = zeros_pos - lag_idx
        mask_d_z = diff_idx_z >= 0
        sd_z = np.maximum(diff_idx_z, 0)
        add2 = h_cur[sd_z] * mask_d_z
        
        addition = add1 - add2
        
        # Compute new c5 for all (i,j) pairs
        # new_c5[i,j] = max_k (over[k] - removal[i,k] + addition[j,k])
        # Use broadcasting: (n_ones, 1, n) + (1, n_zeros, n)
        new_over = over[None, None, :] - removal[:, None, :] + addition[None, :, :]
        new_c5 = np.max(new_over, axis=2)
        
        flat_idx = np.argmin(new_c5)
        best_i, best_j = np.unravel_index(flat_idx, new_c5.shape)
        min_val = float(new_c5[best_i, best_j])
        
        if min_val < cur_c5 - 1e-14:
            h_cur[int(ones[best_i])] = 0
            h_cur[int(zeros[best_j])] = 1
            cur_c5 = min_val
            over = compute_overlaps(h_cur)  # Recompute to avoid drift
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
        else:
            break
    
    return best_h, best_c5

def simulated_annealing(h, n_iters=5000, T_start=0.01, T_end=0.0001, time_limit=None, start_time=None):
    n = len(h)
    h_cur = h.copy().astype(float)
    cur_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(42)
    
    n_iters = min(n_iters, 5000)
    T_schedule = np.geomspace(T_start, T_end, n_iters)
    
    for it in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        T = T_schedule[it]
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h_test = h_cur.copy()
        h_test[i] = 0
        h_test[j] = 1
        new_c5 = compute_c5(h_test)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h_cur = h_test
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
    
    return best_h, best_c5

def sa_block_move(h, n_iters=3000, T_start=0.005, T_end=0.00005, block_size=4, time_limit=None, start_time=None, seed=42):
    """SA with block moves (swap entire blocks)."""
    n = len(h)
    h_cur = h.copy().astype(float)
    cur_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    rng = np.random.RandomState(seed)
    
    T_schedule = np.geomspace(T_start, T_end, n_iters)
    
    for it in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        T = T_schedule[it]
        # Pick a random block
        start = rng.randint(0, n - block_size + 1)
        block = h_cur[start:start+block_size]
        n_ones_in_block = int(np.sum(block))
        
        if n_ones_in_block == 0 or n_ones_in_block == block_size:
            continue
        
        # Generate a new random arrangement of the block with same number of ones
        new_block = np.zeros(block_size)
        positions = rng.choice(block_size, n_ones_in_block, replace=False)
        new_block[positions] = 1.0
        
        if np.array_equal(new_block, block):
            continue
        
        h_test = h_cur.copy()
        h_test[start:start+block_size] = new_block
        new_c5 = compute_c5(h_test)
        delta = new_c5 - cur_c5
        
        if delta < 0 or rng.random() < np.exp(-abs(delta) / T):
            h_cur = h_test
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
    
    return best_h, best_c5

def multi_scale_search(h_init, n_target, budget_s, start_time):
    """Try multiple scales and pick the best."""
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Try finer scales
    for n_fine in [n_target * 2, int(n_target * 1.5)]:
        if time.time() - start_time > budget_s * 0.7:
            break
        
        x_fine = np.linspace(0, 1, n_fine)
        x_coarse = np.linspace(0, 1, len(h_init))
        h_fine = np.interp(x_fine, x_coarse, h_init)
        h_fine = project_feasible(h_fine, n_fine / 2.0)
        h_bin = round_to_binary(h_fine, n_fine / 2.0)
        
        t_left = budget_s * 0.7 - (time.time() - start_time)
        if t_left < 30:
            break
        
        h_new, sc = fast_swap_descent(h_bin, max_rounds=100, time_limit=t_left * 0.6, start_time=start_time)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_new
    
    return best_h, best_c5

def slsqp_refine(h, target, maxiter=1000, alpha=1000):
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        hx = project_feasible(x, target)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha * dx
    
    def cons_eq(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except:
        return h, compute_c5(h)
```

```python
def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def compute_c5(h):
    return float(np.max(compute_overlaps(h)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, target):

    h_bin = (h >= 0.5).astype(float)
    excess = int(np.sum(h_bin)) - int(target)
    if excess > 0:
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]

        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        h_bin[zeros_sorted[:(-excess)]] = 1

def slsqp_refine(h, target, maxiter=800, alpha=5000):

    dx = 2.0 / n
    def obj(x):
        hx = project_feasible(x, target)
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        over = corr[n-1:n-1+n]
        c_max = float(np.max(over))
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * dx
    def cons_eq(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:

    if abs(np.sum(h_out) - target_out) > 1e-10:
        h_out = project_feasible(h_out, target_out)
    final_score = compute_c5(h_out)
    return h_out, final_score, n_out
```
