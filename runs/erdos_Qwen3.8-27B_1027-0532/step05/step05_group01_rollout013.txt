```python
def linearized_minimax(h, n, target, n_active, time_limit, max_iter=100):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr) * dx
        
        # Select active shifts (top n_active)
        top_idx = np.argsort(corr)[-n_active:]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target
        ]
        
        for s_idx in top_idx:
            k = int(s_idx) - mid
            # f_k(h) ≈ f_k(h) + grad_k · (h - h_current)
            # = corr[s_idx] * dx + sum_i grad_k[i] * (h[i] - h[i])
            # We want: corr[s_idx]*dx + sum_i grad_k[i]*(h_var[i] - h[i]) <= t_var
            
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]

            # c_max_current + grad_k · (h_var - h) <= t_var

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        # Solve
        try:
            prob.solve(solver=cp.GLOP)  # or ECOS, SCS
        except:
            try:
                prob.solve(solver=cp.ECOS)
            except:
                break

            break
```

```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def linearized_minimax_trust(h, n, target, time_limit, max_iter=200, 
                               n_active_max=50, delta_init=0.5, delta_min=0.01):
    """
    Linearized minimax with trust region LP subproblems.
    At each step:
    1. Compute all f_k values
    2. Select top n_active shifts
    3. Solve LP: min t s.t. linearized f_k <= t, ||h-h_curr||_inf <= delta
    4. Accept or reject based on actual improvement
    """
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    delta = delta_init
    consec_reject = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max_corr = float(np.max(corr))
        c5_now = c_max_corr * dx
        
        # Select active shifts
        n_active = min(n_active_max, 2*n-1)
        top_idx = np.argsort(corr)[-n_active:]
        
        # Build LP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target
        ]
        
        # Trust region
        constraints.append(h_var >= h - delta)
        constraints.append(h_var <= h + delta)
        
        # Linearized constraints
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            
            # f_k_lin(h_new) = corr[s_idx] + grad_k · (h_new - h) <= t_var (in corr units)
            # grad_k · h_new - t_var <= t_var - corr[s_idx] + grad_k · h
            # => grad_k · h_new - t_var <= (grad_k · h) - corr[s_idx]
            rhs = float(grad_k @ h) - float(corr[s_idx])
            constraints.append(grad_k @ h_var - t_var <= rhs)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        h_lp = np.array(h_var.value).flatten()
        h_lp = project_feasible(h_lp, target)
        
        c5_lp = compute_c5(h_lp)
        
        if c5_lp < c5_now - 1e-13:
            h = h_lp
            consec_reject = 0
            # Can grow trust region
            delta = min(1.0, delta * 1.2)
        else:
            consec_reject += 1
            delta *= 0.5
            if delta < delta_min or consec_reject > 10:
                break
    
    return h, compute_c5(h)

def binary_local_search(h, n, target, time_limit, max_rounds=200):

            best_h[best_i] = 0
            best_h[best_j] = 1

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def fast_pair_cd(h, n, target, time_limit, max_sweeps=3000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 200:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(22):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_optimization(h, n, target, time_limit, block_size=10, max_rounds=500):
    """
    Optimize blocks of consecutive positions simultaneously.
    For each block, we can redistribute mass within it.
    """
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    
    rng = np.random.RandomState(12345)
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Pick a random block
        block_start = rng.randint(0, n - block_size)
        block = slice(block_start, block_start + block_size)
        
        # Compute gradient at worst shift for this block
        k_star = int(np.argmax(corr)) - mid
        grad_block = np.zeros(block_size)
        for local_i, global_i in enumerate(range(block_start, block_start + block_size)):
            gi = global_i
            contrib = 0.0
            if gi - k_star >= 0 and gi - k_star < n:
                contrib += (1.0 - h[gi - k_star])
            if gi + k_star >= 0 and gi + k_star < n:
                contrib -= h[gi + k_star]
            grad_block[local_i] = contrib
        
        # Also consider contribution to objective from these positions
        # We want to minimize the max, so we should reduce values where grad > mean
        # and increase where grad < mean
        
        # Simple heuristic: move mass from high-grad to low-grad within block
        total_mass = h[block].sum()
        avg_grad = grad_block.mean()
        
        # New allocation: push mass towards low-grad positions
        weights = np.exp(-grad_block / (avg_grad + 0.1))
        weights /= weights.sum()
        h_new_block = weights * total_mass
        h_new_block = np.clip(h_new_block, 0.0, 1.0)
        
        # Fix sum constraint within block
        diff = h[block].sum() - h_new_block.sum()
        h_new_block += diff / block_size
        h_new_block = np.clip(h_new_block, 0.0, 1.0)
        
        h_try = h.copy()
        h_try[block] = h_new_block
        h_try = project_feasible(h_try, target)
        
        if compute_c5(h_try) < c5_val - 1e-14:
            h = h_try
    
    return best_h, best_c5
```

```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def linearized_minimax_trust(h, n, target, time_limit, max_iter=300,
                              n_active_max=50, delta_init=0.4, delta_min=0.005):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    mid = n - 1
    idxs = np.arange(n)
    delta = delta_init
    consec_reject = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        n_active = min(n_active_max, 2 * n - 1)
        top_idx = np.argsort(corr)[-n_active:]

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target,
            h_var >= h - delta,
            h_var <= h + delta
        ]

        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad_k = np.zeros(n)
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k[v1] += (1.0 - h[idxs[v1] - k])
            grad_k[v2] -= h[idxs[v2] + k]
            rhs = float(grad_k @ h) - float(corr[s_idx])
            constraints.append(grad_k @ h_var - t_var <= rhs)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CVXOPT]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break

        h_lp = np.array(h_var.value).flatten()
        if h_lp is None or np.any(np.isnan(h_lp)):
            break
        h_lp = project_feasible(h_lp, target)
        c5_lp = compute_c5(h_lp)

        if c5_lp < c5_now - 1e-13:
            h = h_lp
            consec_reject = 0
            delta = min(0.8, delta * 1.15)
        else:
            consec_reject += 1
            delta *= 0.6
            if delta < delta_min or consec_reject > 15:
                break

    return best_h, best_c5


def fast_pair_cd(h, n, target, time_limit, max_sweeps=3000, n_cand=50):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 300:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(24):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def binary_local_search(h, n, target, time_limit, max_rounds=200, n_sample=150):
    start = time.time()
    rng = np.random.RandomState(777)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        idx_ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx_ones])
        for i in range(n_ones - n // 2):
            h_bin[idx_ones[order[i]]] = 0
    elif n_ones < n // 2:
        idx_zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx_zeros])
        for i in range(n // 2 - n_ones):
            h_bin[idx_zeros[order[i]]] = 1

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    dx = 2.0 / n
    mid = n - 1
    idxs = np.arange(n)

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h_bin.copy()

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - best_h[idxs[v1] - k_star])
        grad[v2] -= best_h[idxs[v2] + k_star]

        ones_idx = np.where(best_h == 1)[0]
        zeros_idx = np.where(best_h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break

        ones_sorted = ones_idx[np.argsort(-grad[ones_idx])][:n_sample]
        zeros_sorted = zeros_idx[np.argsort(grad[zeros_idx])][:n_sample]

        best_improve = 0.0
        best_i = -1
        best_j = -1

        for i in ones_sorted:
            for j in zeros_sorted:
                if i == j:
                    continue
                shift_i = np.arange(-(n-1), n)
                ki = (i + shift_i >= 0) & (i + shift_i < n)
                kj = (j + shift_i >= 0) & (j + shift_i < n)
                new_corr_arr = corr.copy()
                delta_i = np.zeros(2*n-1)
                delta_i[ki] = -(1.0 - best_h[i + shift_i[ki]])
                delta_j = np.zeros(2*n-1)
                delta_j[kj] = (1.0 - best_h[j + shift_i[kj]])
                total_delta = delta_i + delta_j
                new_c5 = float(np.max(corr + total_delta)) * dx
                if new_c5 < best_c5 - 1e-13:
                    improve = best_c5 - new_c5
                    if improve > best_improve:
                        best_improve = improve
                        best_i = i
                        best_j = j

        if best_i >= 0:
            best_h[best_i] = 0.0
            best_h[best_j] = 1.0
        else:
            break

    return best_h, best_c5


def multi_shift_smooth(h, n, target, time_limit, max_steps=5000, n_top=8, base_lr=0.02, mu=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()

        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 80.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]

        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1]:
            lr = base_lr * lr_mult
            h_try = project_feasible(h - lr * v, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break

        if not improved and step > 500:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
