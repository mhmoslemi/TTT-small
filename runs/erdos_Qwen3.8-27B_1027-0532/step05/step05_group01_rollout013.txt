        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    
    for sweep in range(max_sweeps):

            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 500:
            # Try to restart with perturbation
            break
        
        # Find the worst shift
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        idxs = np.arange(n)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not mask.any():
            step = diff / n
            h = np.clip(h + step, 0.0, 1.0)
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj_func(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol=1e-14):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj_func, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_change_matrix(h, n):
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    change = np.zeros((n, n_corr))
    for p in range(n):
        pos_mask = (p + shifts >= 0) & (p + shifts < n)
        neg_mask = (p - shifts >= 0) & (p - shifts < n)
        change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
        change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def update_change_row(change, p, h, n, shifts):
    change[p, :] = 0.0
    pos_mask = (p + shifts >= 0) & (p + shifts < n)
    neg_mask = (p - shifts >= 0) & (p - shifts < n)
    change[p, pos_mask] += (1.0 - h[p + shifts[pos_mask]])
    change[p, neg_mask] -= h[p - shifts[neg_mask]]
    return change


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=50000,
                 n_cand_start=20, n_cand_end=70, max_no_improve=300):
    start = time.time()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    mid = n - 1

    h = h_init.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    consecutive_no_improve = 0

    change = compute_change_matrix(h, n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx

        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > max_no_improve:
            break

        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - mid

        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        progress = min(float(sweep) / max_sweeps, 1.0)
        current_ncand = int(n_cand_start + (n_cand_end - n_cand_start) * progress)
        current_ncand = min(current_ncand, n // 2)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:current_ncand]
        high_set = sorted_idx[-current_ncand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

                j = int(low_set[jj])

                delta_hi = min(h[i], 1.0 - h[j])

                    continue

                g = change[j] - change[i]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(14):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g

                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0

                    else:
                        a_d = mid_d

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)

            update_change_row(change, best_i, h, n, shifts)
            update_change_row(change, best_j, h, n, shifts)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    start = time.time()
    rng = np.random.RandomState(12345)
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if no_improve > 5000:
            break

        scale = base_scale * max(0.1, 1.0 - step / max_steps)
        n_pert = int(rng.choice([1, 2, 3, 5, 7, 10]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < cur_c5 - 1e-14:
            h = h_try
            cur_c5 = sc
            no_improve = 0
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 100:
                h = best_h.copy()
                cur_c5 = best_c5
                no_improve = 0

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    target0 = n0 / 2.0

    best_h = project_feasible(h_init, target0)
    best_score = compute_c5(best_h)

    if tleft() > 60:
        h_pol = slsqp_round(best_h, n0, target0,
                            [50, 200, 1000, 5000, 20000],
                            800, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 200:
        n1 = 800
        target1 = n1 / 2.0
        h1 = interpolate_to(best_h, n1, target1)
        sc1 = compute_c5(h1)
        if tleft() > 150:
            h_cd, c5_cd = fast_pair_cd(h1, n1, target1,
                                       time_limit=min(300, tleft() * 0.5),
                                       max_sweeps=30000,
                                       n_cand_start=15, n_cand_end=60,
                                       max_no_improve=400)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd
                n0 = n1
                target0 = target1

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0,
                            [1000, 5000, 20000],
                            1500, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 150:
        h_cd, c5_cd = fast_pair_cd(best_h, n0, target0,
                                   time_limit=min(250, tleft() * 0.5),
                                   max_sweeps=25000,
                                   n_cand_start=20, n_cand_end=80,
                                   max_no_improve=500)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    if tleft() > 100:
        h_ms, c5_ms = multi_shift_pair_cd(best_h, n0, target0,
                                          time_limit=min(120, tleft() * 0.5),
                                          max_sweeps=15000, n_cand=35, n_shifts=3)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    if tleft() > 100:
        h_pol = slsqp_round(best_h, n0, target0,
                            [5000, 20000, 80000],
                            1000, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 250:
        n2 = 1200
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        if tleft() > 200:
            h_cd2, c5_cd2 = fast_pair_cd(h2, n2, target2,
                                         time_limit=min(200, tleft() * 0.6),
                                         max_sweeps=20000,
                                         n_cand_start=15, n_cand_end=50,
                                         max_no_improve=300)
            if c5_cd2 < best_score:
                best_score = c5_cd2
                best_h = h_cd2
                n0 = n2
                target0 = target2
            if tleft() > 60:
                h_pol2 = slsqp_round(best_h, n0, target0,
                                     [2000, 10000],
                                     800, 1e-14)
                sc2 = compute_c5(h_pol2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_pol2

    if tleft() > 80:
        h_sls, c5_sls = stochastic_local_search(best_h, n0, target0,
                                                time_limit=min(40, tleft() * 0.5),
                                                max_steps=30000, base_scale=0.03)
        if c5_sls < best_score:
            best_score = c5_sls
            best_h = h_sls

    if tleft() > 80:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.03, 0.08, 0.15]:
            if tleft() < 50:
                break
            h_pert = best_h + rng.randn(n0) * noise_scale
            h_pert = project_feasible(h_pert, target0)
            h_ref = slsqp_round(h_pert, n0, target0,
                                [100, 1000, 5000, 20000],
                                600, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 60:
        h_cd_final, c5_final = fast_pair_cd(best_h, n0, target0,
                                            time_limit=min(50, tleft() * 0.6),
                                            max_sweeps=10000,
                                            n_cand_start=30, n_cand_end=80,
                                            max_no_improve=200)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    if tleft() > 30:
        h_final_pol = slsqp_round(best_h, n0, target0,
                                  [10000, 50000],
                                  600, 1e-14)
        sc = compute_c5(h_final_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_final_pol

    final_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(final_h)

    return final_h, final_score, len(final_h)
