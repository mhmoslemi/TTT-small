import time

_g_n = 0
_g_target = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_power(corr, p):
    """Power-mean smooth max: (mean(corr^p))^(1/p)"""
    # Only consider the "overlap" part (lags 0 to n-1)
    n = len(corr)  # corr has length 2n-1
    # Actually corr from np.correlate(h, 1-h, mode='full') has length 2n-1
    # The "overlap" lags are corr[n-1 : 2n-1] (lags 0 to n-1)
    vals = corr[n - 1:]
    vals = np.maximum(vals, 1e-10)
    return float((np.mean(vals ** p)) ** (1.0 / p))

def smooth_max_lse(corr, alpha):
    """Log-sum-exp smooth max"""
    vals = corr  # use all lags
    c_max = float(np.max(vals))
    shifted = alpha * (vals - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def gradient_of_corr(h, weights):
    """Compute weighted gradient of correlation with respect to h.
    
    weights[k] is the weight for lag k (k = 0 to n-1)
    The correlation at lag k is: sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
    
    d(corr_k)/d(h[j]) = (1-h[j+k]) if j+k < n  (j is the first factor)
                       - h[j-k] if j-k >= 0     (j is the second factor, h[j] = h[(j-k)+k])
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-12:
            continue
        # First factor: h[i] for i = 0 to n-1-k
        # d/d(h[i]) = (1-h[i+k])
        for i in range(n - k):
            grad[i] += w * (1.0 - h[i + k])
        # Second factor: h[i+k] = h[j] for j = k to n-1
        # d/d(h[j]) = -h[j-k]
        for j in range(k, n):
            grad[j] -= w * h[j - k]
    return grad * dx

```python
def compute_gradient(h, weights):
    """
    h: array of length n
    weights: array of length n, weights[k] for lag k
    Returns: gradient of weighted sum of correlations w.r.t. h, shape (n,)
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        # First factor: h[i] for i in [0, n-1-k]
        # d/dh[j] = (1-h[j+k]) for j in [0, n-1-k]
        grad[:n-k] += w * (1.0 - h[k:n])
        # Second factor: h[i+k] = h[j] for j in [k, n-1]
        # d/dh[j] = -h[j-k] for j in [k, n-1]
        grad[k:] -= w * h[:n-k]
    return grad * dx
```

This is O(n²) but uses vectorized numpy operations for each k, so it should be fast enough.

Actually, I think the key is to make h(1-h(x+k)) small for ALL k simultaneously. This is a very strong constraint.

Let me think about it differently. We want:
max_k Σ_i h[i](1-h[i+k]) ≤ T (for some target T)

This means for every k, the "weighted sum" of h[i] at positions where h[i+k] is small should be bounded.

One strategy: make h(i) and h(i+k) "anti-correlated" for every k. But this is impossible for all k simultaneously (unless h is constant, which gives the worst case).

The best we can do is to make the anti-correlation as "uniform" as possible across all k, so that no single k stands out.

This is exactly the "flat autocorrelation" problem I identified earlier. The solution is a function whose autocorrelation is as flat as possible.

In signal processing, this is related to "low PAPR sequences" or "radar waveforms with low sidelobes". The FEEDBACK, GPS, and other practical sequences achieve near-flat autocorrelation.

For our problem, the optimal continuous function might look like a "chirp" (linear frequency modulation):
h(x) = 0.5 + A * sin(2π(f0*x + B*x²/2))

This has a nearly flat autocorrelation (the "thumbtack" property). Let me try this as a starting point!

For a chirp: h(x) = 0.5 + 0.5*sin(2π(x + x²/2)) on [0,2]
The autocorrelation of a chirp is approximately a triangular function peaked at 0 and decaying linearly. This is not ideal for our purpose (we want flat, not peaked).

Actually, what we want is the OPPOSITE: we want the autocorrelation of h to be as FLAT as possible (not peaked). A chirp has a peaked autocorrelation (good for radar, bad for us).

What has a flat autocorrelation? A "white noise" sequence! But white noise in [0,1] with mean 0.5 would give h(1-h) ≈ 0.25 on average, and the correlation would be roughly 0.25 * (n-k) * dx ≈ 0.5 for all k. That's the same as the constant case.

So we need something between "structured" (peaked autocorrelation) and "random" (flat but high autocorrelation). The optimal solution minimizes the PEAK of h(1-h(x+k)), which is equivalent to making the "complement correlation" as flat and low as possible.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_softmax_weights(corr, n, temperature):
    over = corr[n - 1:2 * n - 1]
    max_val = float(np.max(over))
    exponents = (over - max_val) / temperature
    exponents = np.clip(exponents, -700.0, 0.0)
    weights = np.exp(exponents)
    s = weights.sum()
    if s < 1e-30:
        weights = np.ones(n) / n
    else:
        weights /= s
    return weights

def get_topk_weights(corr, n, k, temperature=0.001):
    over = corr[n - 1:2 * n - 1]
    weights = np.zeros(n)
    top_idx = np.argsort(over)[::-1][:k]
    top_vals = over[top_idx]
    max_v = top_vals[0]
    exponents = (top_vals - max_v) / temperature
    exponents = np.clip(exponents, -700.0, 0.0)
    w = np.exp(exponents)
    s = w.sum()
    if s > 1e-30:
        w /= s
    weights[top_idx] = w
    return weights

def compute_grad_full(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-14:
            continue
        grad[:n - k] += w * (1.0 - h[k:n])
        grad[k:] -= w * h[:n - k]
    return grad * dx

def compute_grad_topk(h, k_indices):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in k_indices:
        k = int(k)
        grad[:n - k] += (1.0 - h[k:n])
        grad[k:] -= h[:n - k]
    return grad * dx / len(k_indices)

def annealed_momentum(h, n, target, deadline, temp_init, temp_final,
                      lr_init, momentum, n_max_iter, topk_mode=False, k_init=5, k_max=50):
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    lr = lr_init
    n_iter = 0
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        frac = min(n_iter / max(n_max_iter, 1), 1.0)
        temp = temp_init * (temp_final / temp_init) ** frac
        corr = compute_corr(h_cur)
        over = corr[n - 1:2 * n - 1]
        if topk_mode:
            k_current = min(k_init + int(frac * (k_max - k_init)), n)
            top_idx = np.argsort(over)[::-1][:k_current]
            grad = compute_grad_topk(h_cur, top_idx)
        else:
            weights = get_softmax_weights(corr, n, temp)
            grad = compute_grad_full(h_cur, weights)
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            n_iter += 1
        else:
            v *= 0.5
            lr *= 0.92
            if lr < 1e-9:
                lr = lr_init
                v = np.zeros(n)
                h_new2 = h_cur - lr * grad
                h_new2 = project_feasible(h_new2, target)
                sc2 = compute_c5(h_new2)
                if sc2 < cur_score:
                    h_cur = h_new2
                    cur_score = sc2
                    n_iter += 1
    return h_cur, cur_score

def _slsqp_obj(x, n, target, alpha):
    h = project_feasible(x, target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _slsqp_cons_fun(x, n, target):
    return np.sum(x) - target

def _slsqp_cons_jac(x, n, target):
    return np.ones(n)

def slsqp_optimize(h, n, target, deadline, alphas, maxiter=2000, ftol=1e-15):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        cons = ({'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac},)
        try:
            res = minimize(_slsqp_obj, h_cur, args=(n, target, alpha),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def coarse_nm_opt(h, n, target, n_coarse, deadline):
    idx = np.linspace(0, n, n_coarse, endpoint=False).astype(int)
    h_c = h[idx].copy()
    h_c = project_feasible(h_c, n_coarse / 2.0)
    x_coarse = np.linspace(0, 1, n_coarse)
    x_fine = np.linspace(0, 1, n)
    def coarse_obj(x):
        h_f = np.interp(x_fine, x_coarse, x)
        h_f = project_feasible(h_f, target)
        return compute_c5(h_f)
    best_c = h_c.copy()
    best_sc = coarse_obj(h_c)
    try:
        res = minimize(coarse_obj, h_c, method='Nelder-Mead',
                      options={'maxiter': 8000, 'xatol': 1e-9, 'fatol': 1e-12})
        h_test = project_feasible(np.clip(res.x, 0, 1), n_coarse / 2.0)
        sc_test = coarse_obj(h_test)
        if sc_test < best_sc:
            best_c = h_test
            best_sc = sc_test
    except Exception:
        pass
    h_up = np.interp(x_fine, x_coarse, best_c)
    h_up = project_feasible(h_up, target)
    return h_up

def structured_perturb(h, n, target, rng, mode='smooth'):
    noise = np.zeros(n)
    if mode == 'smooth':
        k = rng.randint(5, 20)
        center = rng.randint(0, n)
        width = rng.randint(n // 10, n // 3)
        for i in range(n):
            d = abs(i - center) if abs(i - center) < n // 2 else n - abs(i - center)
            noise[i] = np.exp(-d ** 2 / (2.0 * width ** 2))
        noise *= rng.uniform(0.02, 0.08) * rng.choice([-1, 1])
    elif mode == 'osc':

        noise = rng.uniform(0.01, 0.05) * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    else:
        noise = rng.normal(0, 0.01, n)
    h_new = h + noise
    h_new = project_feasible(h_new, target)
    return h_new
