import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_topk(h, alpha, k):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(corr)
    if k < n:
        top_idx = np.argpartition(corr, -k)[-k:]
        sub = corr[top_idx]
    else:
        sub = corr
    c_max = np.max(sub)
    shifted = np.clip(alpha * (sub - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad_topk(h, alpha, k, dx):
    n = len(h)
    n_corr = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    if k < n_corr:
        top_idx = np.argpartition(corr, -k)[-k:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        sub_corr = corr
    
    c_max = np.max(sub_corr)
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx_k, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k_val = idx_k - n_half
        if k_val == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k_val > 0:
            if n - k_val > 0:
                grad[k_val:] += w * (1.0 - h[:n - k_val]) * dx
                grad[:n - k_val] -= w * h[k_val:] * dx
        else:
            m = -k_val
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

def adam_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Adam parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    # Schedule of (alpha, k, steps)
    schedule = [
        (50, 5, 2000),
        (100, 10, 2000),
        (200, 15, 2000),
        (500, 20, 2000),
        (1000, 30, 2000),
        (2000, 50, 2000),
        (5000, 80, 2000),
        (10000, 200, 2000),
        (20000, 500, 2000),
        (50000, 2*n-1, 2000),
    ]
    
    for alpha, k_val, steps in schedule:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        for step in range(steps):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            t += 1
            grad = smooth_grad_topk(h, alpha, k_val, dx)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_try = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_try = project_feasible(h_try, target)
            
            score = compute_c5(h_try)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h_try.copy()
            h = h_try
    
    return best_h, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def get_topk_and_weights(corr, alpha, k):
    n_corr = len(corr)
    if k < n_corr:
        top_idx = np.argpartition(corr, -k)[-k:]
        sub = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        sub = corr
    c_max = np.max(sub)
    shifted = np.clip(alpha * (sub - c_max), -600.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    if wsum < 1e-300:
        weights = np.ones(len(sub)) / len(sub)
    else:
        weights = exp_s / wsum
    return top_idx, weights

def grad_topk(h, top_idx, weights, dx):

            return h
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def get_topk_weights(corr, alpha, k):
    n_corr = len(corr)
    if k < n_corr:
        top_idx = np.argpartition(corr, -k)[-k:]
        sub = corr[top_idx]
    else:
        top_idx = np.arange(n_corr)
        sub = corr
    c_max = np.max(sub)
    shifted = np.clip(alpha * (sub - c_max), -600.0, 0.0)
    exp_s = np.exp(shifted)
    wsum = np.sum(exp_s)
    if wsum < 1e-300:
        weights = np.ones(len(sub)) / len(sub)
    else:
        weights = exp_s / wsum
    return top_idx, weights

def grad_topk(h, top_idx, weights, dx):
    n = len(h)
    n_half = n - 1
    grad = np.zeros(n)
    for idx_k, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k_val = int(idx_k) - n_half
        if k_val == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k_val > 0:
            m = k_val
            if m < n:
                grad[m:] += w * (1.0 - h[:n - m]) * dx
            if n - m > 0:
                grad[:n - m] -= w * h[m:] * dx
        else:
            m = -k_val
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
            if m < n:
                grad[m:] -= w * h[:n - m] * dx
    return grad

def adam_optimize(h_init, target, deadline):
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 0.005
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    schedule = [
        (30, 4, 4000),
        (60, 6, 4000),
        (100, 10, 4000),
        (200, 15, 4000),
        (400, 20, 4000),
        (800, 30, 4000),
        (1500, 50, 4000),
        (3000, 80, 4000),
        (6000, 120, 4000),
        (12000, 200, 4000),
        (25000, 400, 4000),
        (50000, 2 * n - 1, 4000),
    ]
    for alpha, k_val, steps in schedule:
        if time.time() > deadline:
            break
        for step in range(steps):
            if time.time() > deadline:
                break
            t += 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            top_idx, weights = get_topk_weights(corr, alpha, k_val)
            grad = grad_topk(h, top_idx, weights, dx)
            m_v = beta1 * m_v + (1.0 - beta1) * grad
            v_v = beta2 * v_v + (1.0 - beta2) * grad * grad
            m_hat = m_v / (1.0 - beta1 ** t)
            v_hat = v_v / (1.0 - beta2 ** t)
            update = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_try = h - update
            h_try = project_feasible(h_try, target)
            score = compute_c5(h_try)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h_try.copy()
            h = h_try
    return best_h, best_score

def slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_eq_fun(x):
    return np.sum(x) - x_sf_target

def slsqp_eq_jac(x):
    return np.ones(x_sf_n)

x_sf_n = 0
x_sf_target = 0.0

def slsqp_polish(h_init, target, alphas, maxiter, deadline):
    global x_sf_n, x_sf_target
    n = len(h_init)
    x_sf_n = n
    x_sf_target = target
    h_cur = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    # Phase 1: Adam from initial point
    if tleft() > 300:
        dl = start_time + min(350, tleft() * 0.45)
        h_a, sc = adam_optimize(best_h, target, dl)
        if sc < best_score:
            best_score, best_h = sc, h_a

    # Phase 2: SLSQP polish
    if tleft() > 120:
        dl = start_time + min(80, tleft() * 0.4)
        h_s = slsqp_polish(best_h, target, [200, 1000, 5000], 3000, dl)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score, best_h = sc, h_s

    # Phase 3: Multi-restart with diverse perturbations
    n_restarts = 8
    for i in range(n_restarts):
        if tleft() < 80:
            break
        strategy = i % 4
        if strategy == 0:
            h_pert = best_h + rng.randn(n) * 0.03
        elif strategy == 1:
            h_pert = best_h.copy()
            mask = rng.rand(n) < 0.1
            h_pert[mask] = rng.rand(int(mask.sum()))
        elif strategy == 2:
            h_pert = best_h.copy()
            n_blocks = 4
            for _ in range(n_blocks):
                bs = rng.randint(0, max(1, n - 15))
                bl = rng.randint(5, 15)
                h_pert[bs:bs + bl] = np.clip(h_pert[bs:bs + bl] + rng.randn(bl) * 0.1, 0, 1)
        else:
            h_pert = best_h.copy()
            idx = rng.choice(n, size=n // 5, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        h_pert = project_feasible(h_pert, target)
        dl = start_time + min(90, tleft() * 0.2)
        h_a, sc = adam_optimize(h_pert, target, dl)
        if sc < best_score:
            best_score, best_h = sc, h_a

    # Phase 4: Try higher resolution
    if tleft() > 400:
        for n_try in [500, 600]:
            if tleft() < 300:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            dl = start_time + min(200, tleft() * 0.4)
            h_a, sc = adam_optimize(h_interp, target_try, dl)
            if sc < best_score:
                best_score, best_h = sc, h_a
                n = n_try
                target = target_try

    # Phase 5: More Adam rounds on best
    if tleft() > 120:
        dl = start_time + min(120, tleft() * 0.5)
        h_a, sc = adam_optimize(best_h, target, dl)
        if sc < best_score:
            best_score, best_h = sc, h_a

    # Phase 6: Final SLSQP
    if tleft() > 50:
        n_use = len(best_h)
        t_use = n_use / 2.0
        dl = start_time + min(40, tleft() * 0.5)
        h_s = slsqp_polish(best_h, t_use, [5000, 20000], 2000, dl)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score, best_h = sc, h_s

    # Phase 7: Tiny Adam final kick
    if tleft() > 20:
        n_use = len(best_h)
        t_use = n_use / 2.0
        dl = start_time + max(0, tleft() * 0.8)
        h_a, sc = adam_optimize(best_h, t_use, dl)
        if sc < best_score:
            best_score, best_h = sc, h_a

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
