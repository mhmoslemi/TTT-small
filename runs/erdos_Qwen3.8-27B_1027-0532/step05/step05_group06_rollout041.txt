def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-14:
            break

            mask = h < 1.0
        else:

        cnt = int(mask.sum())
        if cnt == 0:

        h = h.copy()
        h[mask] += diff / cnt

def compute_grad_corr(h, k, n):
    """Compute gradient of correlation at lag k (index in full corr) w.r.t. h"""
    lag = int(k) - (n - 1)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def grad_corr_at_lag(h, k_idx, n):
    lag = int(k_idx) - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        j1 = np.arange(lag, n)
        grad[j1] += (1.0 - h[j1 - lag])
        if lag > 0:
            j2 = np.arange(0, n - lag)
            grad[j2] -= h[j2 + lag]
    else:
        m = -lag
        j1 = np.arange(0, n - m)
        grad[j1] += (1.0 - h[j1 + m])
        if m > 0:
            j2 = np.arange(m, n)
            grad[j2] -= h[j2 - m]
    return grad

def lse_obj_grad(x, alpha, target, mu, n_active_max=100):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    
    # Only compute gradient for lags with significant weight
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = weights / wsum
    
    # Sort and take top n_active_max
    if len(corr) > n_active_max:
        top_idx = np.argpartition(weights, -n_active_max)[-n_active_max:]
        grad = np.zeros(n)
        for ki in top_idx:
            w = weights[ki]
            if w > 1e-12:
                grad += w * grad_corr_at_lag(x, ki, n)
    else:
        grad = np.zeros(n)
        for ki in range(len(corr)):
            w = weights[ki]
            if w > 1e-12:
                grad += w * grad_corr_at_lag(x, ki, n)
    
    grad += 2.0 * mu * (s - target) * np.ones(n)
    return val, grad

def lse_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50,
                                'maxfun': maxiter * 3})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def subgradient_method(h, target, n_iter, deadline, n_top=10, verbose=False):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    
    step_size = 0.05
    no_improve = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        corr_dx = corr * dx
        c_max = float(np.max(corr_dx))
        
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top lags
        top_idx = np.argsort(corr_dx)[::-1][:n_top]
        top_vals = corr_dx[top_idx]
        
        # Soft weights based on proximity to max
        scale = max(best_score - top_vals[-1], 0.005) + 0.005
        weights = np.exp(-(best_score - top_vals) / scale)
        weights = weights / weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for i in range(n_top):
            w = weights[i]
            if w > 1e-10:
                grad += w * grad_corr_at_lag(h_cur, int(top_idx[i]), n)
        
        # Add constraint gradient
        sum_err = np.sum(h_cur) - target
        grad += 5.0 * sum_err * np.ones(n)
        
        # Step
        h_new = h_cur - step_size * grad
        h_new = project_feasible(h_new, target)
        h_cur = h_new
        
        # Adaptive step size
        if no_improve > 30:
            step_size *= 0.9
            if step_size < 0.001:
                break
        elif no_improve == 0:
            step_size = min(step_size * 1.01, 0.2)
    
    return best_h, best_score

def make_low_autocorr_binary(n, rng):
    """Create a binary sequence with n/2 ones using a Golay-like construction"""
    h = np.zeros(n)
    # Use a simple method: split into two halves, put 1s in one
    # Then apply a permutation to mix them
    half = n // 2
    h[:half] = 1.0
    # Apply a pseudo-random permutation that preserves the count
    perm = rng.permutation(n)
    h = h[perm]
    return h

def make_structural_start(n, n_ones, pattern_type, rng):
    """Create various structural starting points"""
    h = np.zeros(n)
    if pattern_type == 'alternating':
        h[::2] = 1.0
        # Adjust to get exactly n_ones
        excess = h.sum() - n_ones
        if excess > 0:
            idx = np.where(h == 1)[0]
            to_flip = idx[:int(excess)]
            h[to_flip] = 0
        elif excess < 0:
            idx = np.where(h == 0)[0]
            to_flip = idx[:int(-excess)]
            h[to_flip] = 1
    elif pattern_type == 'blocks':
        block_size = max(2, n // 20)
        for i in range(0, n, block_size):
            if (i // block_size) % 2 == 0:
                h[i:i+block_size] = 1.0
        h = project_feasible(h, n_ones)
    elif pattern_type == 'triangular':
        t = np.linspace(0, 1, n)
        h = (t < 0.5).astype(float) * (1 - 2*abs(t-0.5))
        h = np.clip(h, 0, 1)
        h = project_feasible(h, n_ones)
    elif pattern_type == 'random_binary':
        h = make_low_autocorr_binary(n, rng)
    elif pattern_type == 'sine':
        freq = rng.uniform(0.5, 5.0)
        h = 0.5 + 0.5 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n))
        h = project_feasible(h, n_ones)
    return project_feasible(h, n_ones)
