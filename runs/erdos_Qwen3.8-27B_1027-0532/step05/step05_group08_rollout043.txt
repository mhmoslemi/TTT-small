```python
    n = len(h)
    one_minus_h = 1.0 - h
    # For each lag k from -(n-1) to n-1, compute the gradient vector
    # We return a list of gradient vectors
    grads = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            # (1-h[j+k]) for j in [0, n-1-k)
            if n-1-k > 0:
                grads[m, :n-k] += one_minus_h[k:]
            # -h[j-k] for j in [k, n)
            if n-k > 0:
                grads[m, k:] -= h[:n-k]
        else:

            if n-ell > 0:
                grads[m, ell:] += one_minus_h[:n-ell]
```

```python
def lse_grad_fast(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    # Only iterate over lags with significant weight
    threshold = 1e-10
    for m in range(2*n-1):
        wm = w[m]
        if wm < threshold:
            continue
        k = m - (n-1)
        if k >= 0:
            if n-k > 0:
                g[:n-k] += wm * one_minus_h[k:]
                g[k:] -= wm * h[:n-k]
        else:
            ell = -k
            if n-ell > 0:
                g[ell:] += wm * one_minus_h[:n-ell]
                g[:n-ell] -= wm * h[ell:]
    g *= dx
    return g
```

This is the same as before but skips zero-weight lags. With alpha large enough, only the top 5-20 lags have significant weight, so this is O(n * n_active) which is much faster.

```python
def swap_delta(h, a, b, n):
    """Compute the change in full correlation when h[a]: 1->0, h[b]: 0->1."""
    k = np.arange(-(n-1), n)  # all lags
    # For a: -(1-h[a+k]) where valid
    ak = a + k
    term1 = np.zeros(2*n-1)
    valid1 = (ak >= 0) & (ak < n)
    term1[valid1] = -(1.0 - h[ak[valid1]])
    
    # +h[a-k] where valid
    ak2 = a - k
    term2 = np.zeros(2*n-1)
    valid2 = (ak2 >= 0) & (ak2 < n)
    term2[valid2] = h[ak2[valid2]]
    
    # +(1-h[b+k]) where valid
    bk = b + k
    term3 = np.zeros(2*n-1)
    valid3 = (bk >= 0) & (bk < n)
    term3[valid3] = (1.0 - h[bk[valid3]])
    
    # -h[b-k] where valid
    bk2 = b - k
    term4 = np.zeros(2*n-1)
    valid4 = (bk2 >= 0) & (bk2 < n)
    term4[valid4] = -h[bk2[valid4]]
    
    return term1 + term2 + term3 + term4
```

```python
def lse_grad_fast(h, alpha, n_active=20):

    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
```

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    if alpha < 1:
        return c_max
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha, n_top=30):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    # Process only top-weighted lags
    top = np.argsort(w)[::-1]
    for m in top:
        wm = w[m]
        if wm < 1e-12:
            break
        k = m - (n-1)
        if k >= 0:
            if n - k > 0:
                g[:n-k] += wm * one_minus_h[k:]
                g[k:] -= wm * h[:n-k]
        else:
            ell = -k
            if n - ell > 0:
                g[ell:] += wm * one_minus_h[:n-ell]
                g[:n-ell] -= wm * h[ell:]
    g *= dx
    return g

        return lse_grad(x, alpha) + 200.0 * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg, bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except:
        return h
