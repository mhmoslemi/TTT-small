    """Local search on binary patterns: flip single bits to reduce max overlap."""
    start = time.time()
    h = h_bin.copy()
    assert np.all((h == 0) | (h == 1))
    assert abs(np.sum(h) - target) < 1e-10
    
    overlaps = compute_overlaps_raw(h)
    current_max = float(np.max(overlaps))
    best_h = h.copy()
    best_max = current_max
    no_improve = 0
    
    n_full = 2 * n - 1
    n_half = n - 1
    
    while time.time() - start < time_limit:
        improved = False
        best_flip = -1
        best_new_max = current_max
        
        # Try flipping each 0 to 1 and each 1 to 0
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        # For efficiency, sample if there are too many candidates
        candidates = np.concatenate([zeros, ones])
        
        for i in candidates:
            h[i] = 1.0 - h[i]
            new_overlaps = compute_overlaps_raw(h)
            new_max = float(np.max(new_overlaps))
            h[i] = 1.0 - h[i]  # revert
            
            if new_max < best_new_max - 1e-10:
                best_new_max = new_max
                best_flip = i
        
        if best_flip >= 0:
            h[best_flip] = 1.0 - h[best_flip]
            overlaps = compute_overlaps_raw(h)
            current_max = best_new_max
            no_improve = 0
            improved = True
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break
    
    return best_h, best_max * (2.0 / n)

    """For each position i, compute the change in all overlap values if h[i] is flipped 0→1.

        # For k = 1 to n-1:
        # Δc(k) = (1-h[i-k])*I[k<=i<=n-1] - h[i+k]*I[0<=i<=n-1-k]

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def sine_binary(n, freq, target_int, rng=None):
    x = np.linspace(0, 2, n, endpoint=False)

def thue_morse_binary(n, target_int):
    h = np.zeros(n)
    for i in range(n):
        c = bin(i).count('1')
        h[i] = 1.0 if c % 2 == 0 else 0.0
    ones = int(np.sum(h))

        h[idx[:ones - target_int]] = 0.0

        idx = np.where(h == 0)[0]

    return h

def to_binary_from_real(h, target_int):
    order = np.argsort(h)[::-1]

    h_bin[order[:target_int]] = 1.0

def alternating_lp_refined(h_init, n, target, max_iter, time_limit, n_top=200, damping=0.9, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    n_half = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 300:
            break
        
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def momentum_subgradient(h_init, n, target, max_steps, time_limit, n_top=20, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        overlaps = compute_overlaps_raw(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        
        w = 0.85 ** np.arange(n_top)
        w = w / w.sum()
        
        grad = np.zeros(n)
        for i in range(n_top):
            k = top_lags[i]
            if k >= 0:
                idx1 = idxs[idxs + k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 + k])
                idx2 = idxs[idxs >= k]
                grad[idx2] += w[i] * (-y[idx2 - k])
            else:
                idx1 = idxs[idxs - k < n]
                grad[idx1] += w[i] * (1.0 - y[idx1 - k])
                idx2 = idxs[idxs + k >= 0]
                grad[idx2] += w[i] * (-y[idx2 + k])
        
        grad *= dx
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        
        c5_y = float(np.max(overlaps)) * dx
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.01, polyak_lr), 0.5)
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()
    
    return best_h, best_c5

def smooth_obj(h, alpha, target, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)

def slsqp_optimize(h_start, n, target, alphas, maxiter=1000):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x, alpha):
        return smooth_obj(x, alpha, target, n)
    
    def cons_eq(x):
        return np.sum(x) - target
    
    h_cur = h_start.copy()
    constraints = [{'type': 'eq', 'fun': cons_eq, 'jac': lambda x: np.ones(n)}]
    
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

                                                n_top=350, damping=0.88, reg=1e-5)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2.copy()
    
    # Phase 8: SLSQP polish
    if tleft() > 80:
        h_slsqp = slsqp_optimize(best_h.copy(), n, target,
                                  [100, 500, 2000, 10000, 50000], maxiter=800)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp.copy()
    
    # Phase 9: Final LP polish
    if tleft() > 50:
        h_lp3, c5_lp3 = alternating_lp_refined(best_h.copy(), n, target,
                                                max_iter=2000, time_limit=min(40, tleft() * 0.7),
                                                n_top=400, damping=0.95, reg=1e-6)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3.copy()
    
    # Phase 10: Final binary local search
    if tleft() > 30:
        h_bin3 = to_binary_from_real(best_h, target_int)
        h_ls3, c5_ls3 = fast_flip_search(h_bin3, n, target,
                                         time_limit=min(25, tleft() * 0.6),
                                         max_stuck=2000, kick_size=3, rng=rng)
        sc = compute_c5(h_ls3)
        if sc < best_score:
            best_score = sc
            best_h = h_ls3.copy()
    
    # Final feasibility check
    h_final = project_feasible(best_h, n / 2.0)
    final_score = compute_c5(h_final)
    
    return h_final, final_score, len(h_final)
