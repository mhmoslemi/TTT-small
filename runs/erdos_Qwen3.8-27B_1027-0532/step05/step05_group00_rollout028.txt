def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    diff = int(target_sum) - int(h_r.sum())
    if diff == 0:
        return h_r
    frac = h - np.round(h)
    if diff > 0:
        zeros = np.where(h_r == 0)[0]
        if len(zeros) >= diff:
            top_idx = zeros[np.argsort(-frac[zeros])[:diff]]
            h_r[top_idx] = 1.0
        else:
            h_r[zeros] = 1.0
    else:
        ones = np.where(h_r == 1)[0]
        if len(ones) >= -diff:
            bot_idx = ones[np.argsort(frac[ones])[:(-diff)]]
            h_r[bot_idx] = 0.0
        else:
            h_r[ones] = 0.0
    s = int(h_r.sum())
    while s > target_sum:
        ones = np.where(h_r == 1)[0]
        idx = ones[np.argmin(frac[ones])]
        h_r[idx] = 0.0
        s -= 1
    while s < target_sum:
        zeros = np.where(h_r == 0)[0]
        idx = zeros[np.argmax(frac[zeros])]

        s += 1

def swap_delta_full(h, i0, j0, n):
    delta = np.zeros(2 * n - 1)
    s_lo = -i0
    s_hi = n - 1 - i0
    m_lo = n - 1 + s_lo
    m_hi = n - 1 + s_hi
    if m_lo >= 0 and m_hi < 2 * n - 1:
        idx = np.arange(m_lo, m_hi + 1)
        delta[idx] += h[i0 + idx - (n - 1)] - 1.0
    s_lo = i0 - (n - 1)
    s_hi = i0
    m_lo = n - 1 + s_lo
    m_hi = n - 1 + s_hi
    if m_lo >= 0 and m_hi < 2 * n - 1:
        idx = np.arange(m_lo, m_hi + 1)
        delta[idx] += h[i0 - (idx - (n - 1))]
    s_lo = -j0
    s_hi = n - 1 - j0
    m_lo = n - 1 + s_lo
    m_hi = n - 1 + s_hi
    if m_lo >= 0 and m_hi < 2 * n - 1:
        idx = np.arange(m_lo, m_hi + 1)
        delta[idx] += 1.0 - h[j0 + idx - (n - 1)]
    s_lo = j0 - (n - 1)
    s_hi = j0
    m_lo = n - 1 + s_lo
    m_hi = n - 1 + s_hi
    if m_lo >= 0 and m_hi < 2 * n - 1:
        idx = np.arange(m_lo, m_hi + 1)
        delta[idx] -= h[j0 - (idx - (n - 1))]
    return delta

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-12,
                                 time_limit=min(tleft() - 20, 120), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary SA on refined solution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(50, tleft() - 10)
        max_it5 = int(min(800000, allowed * 15000))
        h_sf, c5_sf = sa_incr(h_bin_f, target_use, max_it5, 0.005, 0.00001,
                              rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final SLSQP polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [5000, 15000, 40000, 100000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-14,
                                   time_limit=max(5.0, tleft() - 5), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Phase 10: Final SA polish
    if tleft() > 15:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_final = project_binary(best_h, target_use)
        allowed = max(5.0, tleft() - 3)
        max_it6 = int(min(300000, allowed * 15000))
        h_fin, c5_fin = sa_incr(h_bin_final, target_use, max_it6, 0.003, 0.000005,
                                rng, time_limit=allowed, start_time=start_time)
        if c5_fin < best_score:
            best_score = c5_fin
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
