    """Simulated annealing on binary sequence."""
    n = len(h_init)
    h = h_init.copy().astype(float)
    # Ensure exactly target ones
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    if len(ones) > target:
        # Remove excess ones
        excess = len(ones) - int(target)
        remove = rng.choice(ones, excess, replace=False)
        h[remove] = 0.0
    elif len(ones) < target:
        need = int(target) - len(ones)
        add = rng.choice(zeros, need, replace=False)
        h[add] = 1.0

    corr = compute_corr(h)[n-1:n-1+n]
    cur_score = float(np.max(corr)) * dx
    best_h = h.copy()
    best_score = cur_score

    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())

            break
        if step % 1000 == 0:
            T = max(T * cool, T_final)
        
        # Pick random swap
        if not ones_set:
            break
        i = rng.choice(list(ones_set))  # This is slow for large sets
        j = rng.choice(list(zeros_set))

        h[i] = 0.0
        h[j] = 1.0
        ones_set.remove(i)
        ones_set.add(j)
        zeros_set.add(i)
        zeros_set.remove(j)

        new_score = float(np.max(corr_new)) * dx

            cur_score = new_score

    n = len(h)
    out = np.correlate(h, 1.0 - h, mode='full')
    return out[n-1:]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            h = np.minimum(h, 1.0)
        else:
            mask = h > 0.0
            if not mask.any():
                break

            h = np.maximum(h, 0.0)

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
        else:
            mask = h > 0.0
            if not mask.any():
                break
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)

    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
