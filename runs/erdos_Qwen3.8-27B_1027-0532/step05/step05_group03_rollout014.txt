I need to update A incrementally. When h[i] changes from 1 to 0 and h[j] changes from 0 to 1:
- A[p][k] = -(1-h[p+k])*dx + h[p-k]*dx (with bounds)
- The change in A[p] comes from h[p+k] or h[p-k] changing
- h[i] changed from 1 to 0: affects A[p] where p+k=i (p=i-k) or p-k=i (p=i+k)

  - A[i+k][k]: the term h[p-k] becomes h[i] = 0 instead of 1, delta = -dx
- h[j] changed from 0 to 1: affects A[p] where p+k=j or p-k=j
  - A[j-k][k]: delta = +dx (term goes from -1 to 0)
  - A[j+k][k]: delta = +dx (term goes from 0 to 1)

So the update to A is sparse! Only O(n) entries change (the entries on the anti-diagonals through i and j).

def update_A_after_swap(A, h, i_from_1, j_from_0, n, dx):
    """Update A after h[i] goes 1->0 and h[j] goes 0->1"""
    # h[i]: 1 -> 0
    # A[p][k] is affected when p+k = i (p = i-k) or p-k = i (p = i+k)
    for k in range(min(i+1, n)):  # p = i-k, valid when i-k >= 0
        p = i - k
        A[p, k] -= dx  # Because h[p+k] = h[i] went from 1 to 0, so (1-h[p+k]) went from 0 to 1, delta = -dx
    for k in range(min(n-1-i, n)):  # p = i+k, valid when i+k < n
        p = i + k
        A[p, k] -= dx  # Because h[p-k] = h[i] went from 1 to 0, so h[p-k] went from 1 to 0, delta = -dx
    
    # h[j]: 0 -> 1
    for k in range(min(j+1, n)):  # p = j-k
        p = j - k
        A[p, k] += dx  # h[p+k] = h[j] went from 0 to 1, so (1-h[p+k]) went from 1 to 0, delta = +dx
    for k in range(min(n-1-j, n)):  # p = j+k
        p = j + k
        A[p, k] += dx  # h[p-k] = h[j] went from 0 to 1, so delta = +dx
    
    return A

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_overlap_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_A(h):
    n = len(h)
    dx = 2.0 / n
    A = np.zeros((n, n))
    for i in range(n):
        k_max1 = n - 1 - i
        if k_max1 >= 0:
            A[i, 0:k_max1+1] += -(1.0 - h[i:i+k_max1+1]) * dx
        k_max2 = i
        if k_max2 >= 0:
            A[i, 0:k_max2+1] += h[i-k_max2:i+1][::-1] * dx
    return A

def update_A_swap(A, i_from_1, j_from_0, n, dx):
    for k in range(i + 1):
        p = i - k
        A[p, k] -= dx
    for k in range(n - i):
        p = i + k
        if p < n:
            A[p, k] -= dx
    for k in range(j + 1):
        p = j - k
        A[p, k] += dx
    for k in range(n - j):
        p = j + k
        if p < n:
            A[p, k] += dx
    return A

    h_b = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_b.sum())
    if n_ones > target:
        idx = np.where(h_b > 0.5)[0]
        h_b[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h_b <= 0.5)[0]
        h_b[idx[:target - n_ones]] = 1.0
    return h_b

def binary_greedy_swaps(h_b, max_rounds=500):
    n = len(h_b)
    dx = 2.0 / n
    h_b = h_b.copy()
    r = compute_overlap_vec(h_b)
    cur_score = float(r.max())
    best_h = h_b.copy()
    best_score = cur_score
    A = precompute_A(h_b)
    ones = np.where(h_b > 0.5)[0]
    zeros = np.where(h_b <= 0.5)[0]
    for rnd in range(max_rounds):
        best_imp = 1e-30
        best_i = -1
        best_j = -1
        for i in ones:
            r_partial = r + A[i]
            deltas = r_partial[None, :] - A[zeros, :]
            new_maxes = deltas.max(axis=1)
            bi = int(np.argmin(new_maxes))
            imp = cur_score - new_maxes[bi]
            if imp > best_imp:
                best_imp = imp
                best_i = int(i)
                best_j = int(zeros[bi])
        if best_i < 0:
            break
        h_b[best_i] = 0.0
        h_b[best_j] = 1.0
        A = update_A_swap(A, best_i, best_j, n, dx)
        ones = np.where(h_b > 0.5)[0]
        zeros = np.where(h_b <= 0.5)[0]
        r = compute_overlap_vec(h_b)
        cur_score = float(r.max())
        if cur_score < best_score:
            best_score = cur_score
            best_h = h_b.copy()

def binary_sa_swaps(h_b, n_steps=50000, t_start=0.008, t_end=0.0003, seed=42):
    n = len(h_b)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h_b = h_b.copy()
    r = compute_overlap_vec(h_b)
    cur_score = float(r.max())
    best_h = h_b.copy()
    best_score = cur_score
    A = precompute_A(h_b)
    ones = np.where(h_b > 0.5)[0]
    zeros = np.where(h_b <= 0.5)[0]
    for step in range(n_steps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        new_r = r + A[i] - A[j]
        new_score = float(new_r.max())
        delta = new_score - cur_score
        t = t_start * (t_end / t_start) ** (step / n_steps)
        if delta < 0 or rng.random() < np.exp(-delta / t):
            h_b[i] = 0.0
            h_b[j] = 1.0
            A = update_A_swap(A, i, j, n, dx)
            ones = np.where(h_b > 0.5)[0]
            zeros = np.where(h_b <= 0.5)[0]

            cur_score = new_score

                best_h = h_b.copy()

def obj_soft_max(x):
    n = _g_n
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr[n-1:2*n-1]))
    sub = corr[n-1:2*n-1] - c_max
    _g_alpha_val = _g_alpha
    if _g_alpha_val < 1:
        return c_max
    shifted = np.clip(_g_alpha_val * sub, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha_val

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:2*n-1]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

            best_score = sc_sg2
            best_h = h_sg2

    # Phase 8: High-alpha SLSQP
    if tleft(start_time, budget_s) > 120:
        h_ref, sc = slsqp_optimize(best_h, [2000, 10000, 50000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Surgery
    if tleft(start_time, budget_s) > 80:
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=200, initial_epsilon=0.0008, n_lags=2)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, max_steps=200, initial_epsilon=0.001)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    # Phase 10: Final binary round
    if tleft(start_time, budget_s) > 60:
        h_bin = binarize_and_project(best_h)
        h_greedy, _ = binary_greedy_swaps(h_bin, max_rounds=300)
        h_sa, sc_sa = binary_sa_swaps(h_greedy, n_steps=40000, t_start=0.003, t_end=0.0001, seed=seed+2000)
        sc_sa_actual = compute_c5(h_sa)
        if sc_sa_actual < best_score:
            best_h = project_feasible(h_sa, target)
            best_score = sc_sa_actual

    # Phase 11: Final SLSQP
    if tleft(start_time, budget_s) > 30:
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000], 1000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 12: Final surgery
    if tleft(start_time, budget_s) > 15:
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=100, initial_epsilon=0.0005, n_lags=3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
