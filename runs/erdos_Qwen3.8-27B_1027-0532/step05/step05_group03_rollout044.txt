    return h_bin

def steered_local_search(h, max_rounds, rng, max_candidates):
    n = len(h)
    h = h.copy()
    best = h.copy()
    best_score = compute_c5(h)
    over = get_overlaps(h)
    
    for rd in range(max_rounds):
        c_curr = float(np.max(over))
        if c_curr < best_score:
            best_score = c_curr
            best = h.copy()
        
        k_star = int(np.argmax(over))
        
        if k_star == 0:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            over = get_overlaps(h)
            continue
        
        candidates = []
        for m in range(n - k_star):
            if h[m] == 1 and h[m + k_star] == 0:
                candidates.append((m, m + k_star))
        
        if not candidates:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
            over = get_overlaps(h)
            continue
        
        if len(candidates) > max_candidates:
            chosen = rng.choice(len(candidates), max_candidates, replace=False)
            candidates = [candidates[i] for i in chosen]
        
        best_swap = None
        best_swap_score = c_curr
        
        for (i, j) in candidates:
            h[i] = 0
            h[j] = 1
            new_over = get_overlaps(h)
            sc = float(np.max(new_over))
            h[i] = 1
            h[j] = 0
            if sc < best_swap_score - 1e-12:
                best_swap = (i, j)
                best_swap_score = sc
        
        if best_swap is not None:
            h[best_swap[0]] = 0
            h[best_swap[1]] = 1
            over = get_overlaps(h)
        else:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                over = get_overlaps(h)
            else:
                break
    
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best = h.copy()
    return best, best_score

def sa_binary(h, rng, n_iter, T0, Tend, steer_ratio):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    over = get_overlaps(h)
    
    if n_iter <= 0:
        return best, best_score
    
    T = T0
    cooling = (Tend / T0) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        T *= cooling
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if rng.random() < steer_ratio:
            k_star = int(np.argmax(over))
            if k_star > 0:
                found = False
                for _ in range(8):
                    m = rng.randint(0, n - k_star)
                    if h[m] == 1 and h[m + k_star] == 0:
                        i, j = m, m + k_star
                        found = True
                        break
                if not found:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        over = get_overlaps(h)
        new_score = float(np.max(over)) * dx
        delta = new_score - curr_score
        
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best = h.copy()
        else:
            h[i] = 1
            h[j] = 0
            over = get_overlaps(h)
    
    return best, best_score

def sa_block_moves(h, rng, n_iter, T0, Tend, block_size):
    """SA with block moves (swap a block of 1s with a block of 0s)."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best = h.copy()
    best_score = compute_c5(h)
    curr_score = best_score
    over = get_overlaps(h)
    
    if n_iter <= 0:
        return best, best_score
    
    T = T0
    cooling = (Tend / T0) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        T *= cooling
        
        # Pick two blocks of same size to swap
        bs = block_size + rng.randint(-block_size//3, block_size//3 + 1)
        bs = max(2, min(bs, n // 4))
        
        pos1 = rng.randint(0, n - bs)
        pos2 = rng.randint(0, n - bs)
        if pos2 <= pos1 and pos2 + bs > pos1:
            continue  # Overlapping, skip
        
        # Perform block swap
        block1 = h[pos1:pos1+bs].copy()
        block2 = h[pos2:pos2+bs].copy()
        h[pos1:pos1+bs] = block2
        h[pos2:pos2+bs] = block1
        
        over = get_overlaps(h)
        new_score = float(np.max(over)) * dx
        delta = new_score - curr_score
        
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best = h.copy()
        else:
            h[pos1:pos1+bs] = block1
            h[pos2:pos2+bs] = block2
            over = get_overlaps(h)
    
    return best, best_score

def slsqp_obj(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    n = len(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    soft = float(np.log(np.sum(np.exp(shifted)))) / _g_alpha
    return c_max + soft

def slsqp_eq_fun(x):
    return np.sum(x) - _g_target

def slsqp_eq_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_step(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    
    for step in range(n_steps):
        over = get_overlaps(h_cur)
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        c_curr = compute_c5(h_cur)
        if sc_new < c_curr:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    
    return best_h, best_score

def time_left(start_time, budget_s):
    return budget_s - 5.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    h_bin1 = binarize_greedy(h_init, int(target))
    h_bin2 = binarize_threshold(h_init, int(target))
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: Steered local search on binary solutions
    for h_b in [h_bin1, h_bin2]:
        if time_left(start_time, budget_s) < 60:
            break
        h_s, sc = steered_local_search(h_b, 3000, rng, 25)
        if sc < best_score:
            best_score = sc
            best_h = h_s
    
    # Check if we found a binary with C5 <= 0.38
    # If so, continuous refinement should get us below 0.3808 easily
    
    # Phase 2: SA runs on binary
    sa_configs = [
        (15000, 0.04, 0.001, 0.6),
        (15000, 0.06, 0.002, 0.5),
        (10000, 0.03, 0.0005, 0.7),
        (20000, 0.05, 0.003, 0.4),
        (10000, 0.08, 0.005, 0.3),
    ]
    for (n_iter, T0, Tend, sr) in sa_configs:
        if time_left(start_time, budget_s) < 60:
            break
        h_sa, sc_sa = sa_binary(best_h, rng, n_iter, T0, Tend, sr)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

        if sc < best_score:
            best_score = sc
            best_h = h_s

    if time_left(start_time, budget_s) > 200:

            if sc_bsa < best_score:
                best_score = sc_bsa
                best_h = h_bsa
            h_s, sc = steered_local_search(h_bsa, 1500, rng, 20)
            if sc < best_score:
                best_score = sc
                best_h = h_s

    if time_left(start_time, budget_s) > 150:

            best_h = h_ref
