def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta_single(h, i, j, positions):
    """Compute correlation change for swapping h[i] (1->0) and h[j] (0->1)."""
    n = len(h)
    k = np.arange(2*n-1) - (n-1)
    delta = np.zeros(2*n-1)
    
    # Effect of h[i]: 1 -> 0
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1.0 - h[idx[mask]])
    
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += h[idx[mask]]
    
    # Effect of h[j]: 0 -> 1
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    delta[mask] += (1.0 - h[idx[mask]])
    
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= h[idx[mask]]
    
    return delta

def compute_delta_double(h, i1, i2, j1, j2, positions):
    """Compute correlation change for 2-swap: h[i1],h[i2] 1->0, h[j1],h[j2] 0->1."""
    delta1 = compute_delta_single(h, i1, j1, positions)
    delta2 = compute_delta_single(h, i2, j2, positions)
    return delta1 + delta2

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def top_k_pressure(full, k=3):
    """Compute pressure: max + weighted sum of top-k values."""
    sorted_vals = np.sort(full)[::-1]
    return float(sorted_vals[0] + 0.3 * (sorted_vals[1] + sorted_vals[2]) / 2)

def greedy_pressure(h, n_top, max_iters, rng, time_limit, start_time):
    """Greedy that minimizes pressure on top-n_top shifts."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    max_stall = 200
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        # Find worst shifts
        top_shifts = np.argsort(full)[::-1][:n_top]
        max_val = float(full.max())
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score each 1-position by how "bad" it is (contributes to worst shifts)
        scores_i = np.zeros(len(ones))
        for t in range(n_top):
            k = top_shifts[t] - (n-1)
            if k == 0:
                continue
            idx = ones + k
            valid = (idx >= 0) & (idx < n)
            # h[ones]=1, contributes (1-h[ones+k]) = 1-h[idx] where valid
            contrib = np.zeros(len(ones))
            contrib[valid] = 1.0 - h[idx[valid]]
            idx2 = ones - k
            valid2 = (idx2 >= 0) & (idx2 < n)
            contrib[valid2] += h[idx2[valid2]]
            scores_i += contrib
        
        # Score each 0-position by how "good" it would be
        scores_j = np.zeros(len(zeros))
        for t in range(n_top):
            k = top_shifts[t] - (n-1)
            if k == 0:
                continue
            idx = zeros + k
            valid = (idx >= 0) & (idx < n)
            contrib = np.zeros(len(zeros))
            contrib[valid] = 1.0 - h[idx[valid]]
            idx2 = zeros - k
            valid2 = (idx2 >= 0) & (idx2 < n)
            contrib[valid2] += h[idx2[valid2]]
            scores_j += contrib
        
        top_i = ones[np.argsort(-scores_i)[:min(15, len(ones))]]
        top_j = zeros[np.argsort(-scores_j)[:min(15, len(zeros))]]
        
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta_single(h, i, j, positions)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        
        if not found:
            stall += 1
    
    return best_h, best_c5

def sa_mixed(h, n_iters, t_init, t_final, p_double, rng, time_limit, start_time):
    """SA with mixed single and double swaps."""
    n = len(h)

    max_stall = 50000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        
        use_double = (rng.random() < p_double and len(ones) >= 2 and len(zeros) >= 2)
        
        if use_double:
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            if i1 == i2:
                i2 = ones[(rng.randint(len(ones))) % len(ones)]
                if i1 == i2:
                    use_double = False
            if use_double:
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2 or j1 in (i1, i2) or j2 in (i1, i2):
                    use_double = False
                if use_double:
                    # Check no overlap
                    if len({i1, i2, j1, j2}) < 4:
                        use_double = False
        
        if use_double:
            delta = compute_delta_double(h, i1, i2, j1, j2, positions)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            delta = compute_delta_single(h, i, j, positions)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5

                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    
    return best_h, best_c5

def lp_objective(h, p=20):
    """Lp norm of correlation as smooth approximation to max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Shift to positive
    max_c = float(np.max(corr))
    # Lp norm approximation to max
    positive = np.maximum(corr - max_c + 1.0, 0.0)  # shift so max is at 1
    lp = float(np.sum(positive ** p)) ** (1.0 / p)
    return max_c - 1.0 + lp  # approximate max with Lp

def continuous_refine_lp(h_start, n, target, ps, maxiter, time_limit, start_time):
    """Continuous refinement using Lp norm objective."""
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for p in ps:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                max_c = float(np.max(corr))
                shifted = np.maximum(corr - max_c + 1.0, 1e-10)
                return max_c - 1.0 + float(np.sum(shifted ** p)) ** (1.0/p)
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def embed_binary(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    ratio = n_fine // n_coarse
    if ratio < 1:
        # Downsample
        src = h_coarse[:n_fine]
        return project_binary(src, target_fine)
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = min(start + ratio, n_fine)
        if h_coarse[i] > 0.5:
            h_fine[start:end] = 1.0
        else:
            h_fine[start:end] = 0.0
    return project_binary(h_fine, target_fine)

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j, k_arr):
    n = len(h)
    delta = np.zeros(2*n-1)
    idx = i + k_arr
    mask = (idx >= 0) & (idx < n)
    delta[mask] -= (1.0 - h[idx[mask]])
    idx2 = i - k_arr
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta[mask2] += h[idx2[mask2]]
    idx3 = j + k_arr
    mask3 = (idx3 >= 0) & (idx3 < n)
    delta[mask3] += (1.0 - h[idx3[mask3]])
    idx4 = j - k_arr
    mask4 = (idx4 >= 0) & (idx4 < n)
    delta[mask4] -= h[idx4[mask4]]
    return delta

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            o2 = np.where(h_r == 1)[0]
            h_r[o2[:final_sum - target_sum]] = 0
        else:
            z2 = np.where(h_r == 0)[0]
            h_r[z2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def greedy_topk(h, n_top, max_iters, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    k_arr = np.arange(2*n-1) - (n-1)
    stall = 0
    max_stall = 150

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        top_shifts = np.argsort(full)[::-1][:n_top]
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        scores_i = np.zeros(n)
        scores_j = np.zeros(n)
        for t in range(n_top):
            k_val = int(top_shifts[t]) - (n-1)
            if k_val == 0:
                continue
            idx_p = np.arange(n) + k_val
            valid_p = (idx_p >= 0) & (idx_p < n)
            idx_m = np.arange(n) - k_val
            valid_m = (idx_m >= 0) & (idx_m < n)
            scores_i[valid_p] += (1.0 - h[idx_p[valid_p]])
            scores_i[valid_m] += h[idx_m[valid_m]]
            scores_j[valid_p] += (1.0 - h[idx_p[valid_p]])
            scores_j[valid_m] += h[idx_m[valid_m]]

        top_i = ones[np.argsort(-scores_i[ones])[:min(20, len(ones))]]
        top_j = zeros[np.argsort(-scores_j[zeros])[:min(20, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            stall += 1

    return best_h, best_c5

def sa_mixed(h, n_iters, t_init, t_final, p_double, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    k_arr = np.arange(2*n-1) - (n-1)
    pos = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        use_double = (rng.random() < p_double and len(ones) >= 2 and len(zeros) >= 2)

        if use_double and it % 7 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n-1)
            if k_star != 0:
                idx_a = pos + k_star
                va = (idx_a >= 0) & (idx_a < n)
                idx_b = pos - k_star
                vb = (idx_b >= 0) & (idx_b < n)
                bad_i = np.zeros(n)
                bad_i[va] += (1.0 - h[idx_a[va]])
                bad_i[vb] += h[idx_b[vb]]
                i_cands = ones[bad_i[ones] >= 1.0]
                if len(i_cands) >= 2:
                    i1 = i_cands[rng.randint(len(i_cands))]
                    i2 = i_cands[rng.randint(len(i_cands))]
                    if i1 == i2:
                        i2 = i_cands[(rng.randint(len(i_cands)) + 1) % len(i_cands)]
                        if i1 == i2:
                            use_double = False
                else:
                    use_double = False
                if use_double:
                    good_j = np.zeros(n)
                    good_j[va] += (1.0 - h[idx_a[va]])
                    good_j[vb] += h[idx_b[vb]]
                    j_cands = zeros[good_j[zeros] >= 1.0]
                    j_cands = j_cands[j_cands != i1]
                    j_cands = j_cands[j_cands != i2]
                    if len(j_cands) >= 2:
                        j1 = j_cands[rng.randint(len(j_cands))]
                        j2 = j_cands[rng.randint(len(j_cands))]
                        if j1 == j2 or j1 == i1 or j1 == i2 or j2 == i1 or j2 == i2:
                            j2 = j_cands[(rng.randint(len(j_cands)) + 1) % len(j_cands)]
                            if j1 == j2 or j1 in (i1,i2) or j2 in (i1,i2):
                                use_double = False
                    else:
                        use_double = False

        if use_double:
            delta = compute_delta(h, i1, j1, k_arr) + compute_delta(h, i2, j2, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i1] = 0.0
                h[i2] = 0.0
                h[j1] = 1.0
                h[j2] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def refine_lp(h_start, n, target, ps, maxiter, time_limit, start_time):
    global _g_n, _g_target, _g_p, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for p in ps:
        if time.time() - start_time > time_limit:
            break
        _g_p = p
        try:
            res = minimize(_lp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def refine_lse(h_start, n, target, alphas, maxiter, time_limit, start_time):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        _g_alpha = alpha
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s - 12.0

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    allowed = min(50, budget_s * 0.07)
    h_g, c5_g = greedy_topk(best_h, 3, 60000, allowed, start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    remaining = deadline - time.time()
    allowed = min(280, remaining * 0.35)
    n_sa = int(min(4000000, allowed * 25000))
    h_sa, c5_sa = sa_mixed(best_h, n_sa, 0.012, 0.00003, 0.35, rng, allowed, start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    for it in range(20):
        if time.time() > deadline - 80:
            break
        n_pert = rng.randint(4, 10)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        allowed1 = min(20, (deadline - time.time()) * 0.08)
        h_p2, c5_p2 = greedy_topk(h_pert, 3, 20000, allowed1, start_time)
        if c5_p2 < best_score:
            best_score = c5_p2
            best_h = h_p2.copy()

        if time.time() > deadline - 50:
            break
        allowed2 = min(45, (deadline - time.time()) * 0.12)
        n_sa2 = int(min(1200000, allowed2 * 20000))
        h_s2, c5_s2 = sa_mixed(h_p2, n_sa2, 0.008, 0.00002, 0.3, rng, allowed2, start_time)
        if c5_s2 < best_score:
            best_score = c5_s2
            best_h = h_s2.copy()

    offsets = [-200, -100, -50, 50, 100, 200, 300]
    for off in offsets:
        if time.time() > deadline - 70:
            break
        n_try = n_init + off
        if n_try < 200 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin = project_binary(h_interp, target_try)
        allowed1 = min(15, (deadline - time.time()) * 0.05)
        h_g2, c5_g2 = greedy_topk(h_bin, 3, 8000, allowed1, start_time)
        allowed2 = min(25, (deadline - time.time()) * 0.06)
        n_sa3 = int(min(600000, allowed2 * 18000))
        h_s3, c5_s3 = sa_mixed(h_g2, n_sa3, 0.008, 0.00008, 0.3, rng, allowed2, start_time)
        if c5_s3 < c5_g2:
            h_bt, c5_bt = h_s3, c5_s3
        else:
            h_bt, c5_bt = h_g2, c5_g2
        if c5_bt < best_score:
            best_score = c5_bt
            best_h = h_bt.copy()

    if time.time() < deadline - 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(120, deadline - time.time() - 40)
        h_ref, sc = refine_lp(best_h, n_use, target_use, [8, 16, 32, 64, 128, 256], 1500, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if time.time() < deadline - 25:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        rem = deadline - 20 - time.time()
        n_sa_f = int(min(800000, max(20000, rem * 25000)))
        h_sa_f, c5_f = sa_mixed(h_bin_final, n_sa_f, 0.004, 0.00001, 0.25, rng, max(5.0, rem), start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    if time.time() < deadline - 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        rem = deadline - 12 - time.time()
        h_ref2, sc2 = refine_lse(best_h, n_use, target_use, [300, 800, 2000], 1000, max(3.0, rem), start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
