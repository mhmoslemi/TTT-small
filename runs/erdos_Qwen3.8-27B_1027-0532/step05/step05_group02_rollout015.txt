    dx = 2.0 / n

    best_h = h.copy()
    stall = 0

        if stall > max_stall:

        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        if k_star == 0:

            k_star = top_k[len(top_k)//2] - (n-1)

        zeros = np.where(h == 0.0)[0]

        idx = positions + k_star

        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def parallel_tempering_sa(h_init, n_chains, t_high, t_low, total_time, start_time, rng, use_2swap=False):
    """Parallel tempering (replica exchange) simulated annealing"""
    n = len(h_init)
    dx = 2.0 / n
    positions = np.arange(n)
    
    # Initialize chains
    chains = []
    for c in range(n_chains):
        h_c = h_init.copy()
        full_c = full_corr_fast(h_c)
        c5_c = float(full_c.max()) * dx
        chains.append({'h': h_c, 'full': full_c, 'c5': c5_c})
    
    # Temperature schedule: geometric
    temps = np.geomspace(t_high, t_low, n_chains)
    
    # Track best
    best_h = h_init.copy()
    best_c5 = float(compute_c5(h_init))
    
    # Exchange interval
    exchange_interval = 200
    
    step = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > total_time:
            break
        
        for c in range(n_chains):
            if time.time() - start_time > total_time:
                break
            
            t = temps[c]
            h_c = chains[c]['h']
            full_c = chains[c]['full']
            c5_c = chains[c]['c5']
            
            ones = np.where(h_c == 1.0)[0]
            zeros = np.where(h_c == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            
            if use_2swap and len(ones) >= 2 and len(zeros) >= 2 and rng.random() < 0.3:
                # 2-swap move
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                if i1 == i2:
                    i2 = ones[(rng.randint(len(ones)) + 1) % len(ones)]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if j1 == j2:
                    j2 = zeros[(rng.randint(len(zeros)) + 1) % len(zeros)]
                if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                    # Fall back to 1-swap
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    delta = compute_delta(h_c, i, j)
                    new_full = full_c + delta
                    new_c5 = float(new_full.max()) * dx
                    delta_c5 = new_c5 - c5_c
                    if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / t):
                        h_c[i] = 0.0
                        h_c[j] = 1.0
                        chains[c]['full'] = new_full
                        chains[c]['c5'] = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_c.copy()
                else:
                    delta = compute_delta2(h_c, i1, i2, j1, j2)
                    new_full = full_c + delta
                    new_c5 = float(new_full.max()) * dx
                    delta_c5 = new_c5 - c5_c
                    if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / t):
                        h_c[i1] = 0.0
                        h_c[i2] = 0.0
                        h_c[j1] = 1.0
                        h_c[j2] = 1.0
                        chains[c]['full'] = new_full
                        chains[c]['c5'] = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_c.copy()
            else:
                # 1-swap move, with bias toward reducing max
                if step % 3 == 0 and len(zeros) > 0:
                    # Focused move: target the current max
                    max_idx = int(np.argmax(full_c))
                    k_star = max_idx - (n - 1)
                    if k_star != 0:
                        A = np.zeros(n)
                        idx = positions + k_star
                        valid = (idx >= 0) & (idx < n)
                        A[valid] += h_c[idx[valid]]
                        idx = positions - k_star
                        valid = (idx >= 0) & (idx < n)
                        A[valid] += h_c[idx[valid]]
                        # Prefer i where A[i] is small (1->0 reduces correlation)
                        i_scores = A[ones]
                        # Sample with probability proportional to (1 - A[i])
                        i_weights = np.clip(1.0 - i_scores / 2.0, 0.1, 1.0)
                        i_weights /= i_weights.sum()
                        i = ones[rng.choice(len(ones), p=i_weights)]
                    else:
                        i = ones[rng.randint(len(ones))]
                else:
                    i = ones[rng.randint(len(ones))]
                
                j = zeros[rng.randint(len(zeros))]
                
                delta = compute_delta(h_c, i, j)
                new_full = full_c + delta
                new_c5 = float(new_full.max()) * dx
                delta_c5 = new_c5 - c5_c
                
                if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                    h_c[i] = 0.0
                    h_c[j] = 1.0
                    chains[c]['full'] = new_full
                    chains[c]['c5'] = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_c.copy()
            
            step += 1
        
        # Exchange between adjacent chains
        if step % exchange_interval == 0:
            for c in range(n_chains - 1):
                if time.time() - start_time > total_time:
                    break
                c1, c2 = c, c + 1
                dE = chains[c2]['c5'] - chains[c1]['c5']
                dT = temps[c1] - temps[c2]
                if dT > 1e-15:
                    # Try to swap: lower temp chain gets worse solution
                    # Accept with prob min(1, exp(-dE/dT)) where dE = E_hot - E_cold
                    # If dE < 0 (hot chain is better), always accept
                    # If dE > 0, accept with prob exp(-dE/dT)
                    if dE < 0 or rng.random() < np.exp(-dE / dT):
                        # Swap
                        chains[c1]['h'], chains[c2]['h'] = chains[c2]['h'].copy(), chains[c1]['h'].copy()
                        chains[c1]['full'], chains[c2]['full'] = chains[c2]['full'].copy(), chains[c1]['full'].copy()
                        chains[c1]['c5'], chains[c2]['c5'] = chains[c2]['c5'], chains[c1]['c5']
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    cons_n_val = n
    cons_target_val = target
    
    def _smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    
    def _cons_fun(h):
        return np.sum(h) - cons_target_val
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(80):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full = full + delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def pt_sa(h_init, n_chains, t_high, t_low, total_time, start_time, rng, use_2swap):
    n = len(h_init)
    dx = 2.0 / n
    positions = np.arange(n)

    chain_hs = []
    chain_fulls = []
    chain_c5s = []
    for c in range(n_chains):
        h_c = h_init.copy()
        full_c = full_corr_fast(h_c)
        c5_c = float(full_c.max()) * dx
        chain_hs.append(h_c)
        chain_fulls.append(full_c)
        chain_c5s.append(c5_c)

    temps = np.geomspace(t_high, t_low, n_chains)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    exchange_interval = 150
    step = 0

    while time.time() - start_time < total_time:
        for c in range(n_chains):
            if time.time() - start_time > total_time:
                break

            t = temps[c]
            h_c = chain_hs[c]
            full_c = chain_fulls[c]
            c5_c = chain_c5s[c]

            ones = np.where(h_c == 1.0)[0]
            zeros = np.where(h_c == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue

            do_2swap = use_2swap and len(ones) >= 2 and len(zeros) >= 2 and (step % 3 == 0)

            if do_2swap:
                i1 = int(ones[rng.randint(len(ones))])
                i2 = int(ones[rng.randint(len(ones))])
                while i2 == i1 and len(ones) > 1:
                    i2 = int(ones[rng.randint(len(ones))])
                j1 = int(zeros[rng.randint(len(zeros))])
                j2 = int(zeros[rng.randint(len(zeros))])
                while j2 == j1 and len(zeros) > 1:
                    j2 = int(zeros[rng.randint(len(zeros))])
                if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                    pass
                else:
                    delta = compute_delta2(h_c, i1, i2, j1, j2)
                    new_full = full_c + delta
                    new_c5 = float(new_full.max()) * dx
                    delta_c5 = new_c5 - c5_c
                    accept = False
                    if delta_c5 < 0:
                        accept = True
                    elif t > 1e-15:
                        r = rng.random()
                        if r < np.exp(-delta_c5 / t):
                            accept = True
                    if accept:
                        h_c[i1] = 0.0
                        h_c[i2] = 0.0
                        h_c[j1] = 1.0
                        h_c[j2] = 1.0
                        chain_fulls[c] = new_full
                        chain_c5s[c] = new_c5
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_c.copy()
                    step += 1
                    continue

            if step % 4 == 0:
                max_idx = int(np.argmax(full_c))
                k_star = max_idx - (n - 1)
                if k_star != 0:
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h_c[idx[valid]]
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h_c[idx[valid]]
                    i_scores = A[ones]
                    i_weights = np.clip(1.5 - i_scores, 0.1, 2.0)
                    i_weights = i_weights / i_weights.sum()
                    i = int(ones[rng.choice(len(ones), p=i_weights)])
                else:
                    i = int(ones[rng.randint(len(ones))])
            else:
                i = int(ones[rng.randint(len(ones))])

            j = int(zeros[rng.randint(len(zeros))])

            delta = compute_delta(h_c, i, j)
            new_full = full_c + delta
            new_c5 = float(new_full.max()) * dx
            delta_c5 = new_c5 - c5_c

            accept = False
            if delta_c5 < 0:
                accept = True
            elif t > 1e-15:
                r = rng.random()
                if r < np.exp(-delta_c5 / t):
                    accept = True

            if accept:
                h_c[i] = 0.0
                h_c[j] = 1.0
                chain_fulls[c] = new_full
                chain_c5s[c] = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_c.copy()

            step += 1

        if step % exchange_interval == 0:
            for c in range(n_chains - 1):
                if time.time() - start_time > total_time:
                    break
                c1 = c
                c2 = c + 1
                dE = chain_c5s[c2] - chain_c5s[c1]
                dT = temps[c1] - temps[c2]
                if dT > 1e-15:
                    if dE < 0 or rng.random() < np.exp(-dE / dT):
                        chain_hs[c1], chain_hs[c2] = chain_hs[c2].copy(), chain_hs[c1].copy()
                        chain_fulls[c1], chain_fulls[c2] = chain_fulls[c2], chain_fulls[c1]
                        chain_c5s[c1], chain_c5s[c2] = chain_c5s[c2], chain_c5s[c1]

    return best_h, best_c5

def ily_search(h_init, rng, time_limit, start_time):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        remaining = time_limit - elapsed
        if remaining < 40:
            break

        k_pert = min(5 + iteration // 10, 20)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(k_pert):
            if len(ones) < 1 or len(zeros) < 1:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        t_local = max(8.0, min(50.0, remaining * 0.35))
        h_ls, c5_ls = pt_sa(h_pert, 4, 0.012, 0.00005, t_local, time.time(), rng, True)

        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

        iteration += 1

    return best_h, best_c5

def _slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    logsum = float(np.log(np.sum(np.exp(shifted))))
    return c_max + logsum / alpha

def _slsqp_cons(h):
    return np.sum(h) - _slslqp_target

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _slslqp_n, _slslqp_target, _slslqp_alpha
    _slslqp_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def run(seed=42, budget_s=1300, **kwargs):
    global _slsqp_n, _slsqp_target, _slsqp_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    # Phase 1: Greedy
    allowed = min(45, tleft() * 0.04)
    h_g, c5_g = greedy_optimize(best_h, rng, allowed, start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Main parallel tempering SA
    allowed = min(500, tleft() * 0.40)
    h_pt, c5_pt = pt_sa(best_h, 8, 0.012, 0.00003, allowed, time.time(), rng, True)
    if c5_pt < best_score:
        best_score = c5_pt
        best_h = h_pt.copy()

    # Phase 3: Second PT round with different params
    if tleft() > 150:
        allowed = min(350, tleft() * 0.30)
        h_pt2, c5_pt2 = pt_sa(best_h, 6, 0.006, 0.00001, allowed, time.time(), rng, True)
        if c5_pt2 < best_score:
            best_score = c5_pt2
            best_h = h_pt2.copy()

    # Phase 4: ILS
    if tleft() > 100:
        allowed = min(250, tleft() * 0.35)
        h_il, c5_il = ily_search(best_h, rng, allowed, start_time)
        if c5_il < best_score:
            best_score = c5_il
            best_h = h_il.copy()

    # Phase 5: Greedy polish
    if tleft() > 30:
        allowed = min(25, tleft() * 0.2)
        h_gp, c5_gp = greedy_optimize(best_h, rng, allowed, start_time)
        if c5_gp < best_score:
            best_score = c5_gp
            best_h = h_gp.copy()

    # Phase 6: SLSQP
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [50, 200, 800, 3000]
        allowed = min(tleft() - 20, 100)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1000, 1e-14, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final SA
    if tleft() > 15:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        allowed = max(5.0, tleft() - 8)
        h_fa, c5_fa = pt_sa(h_bin_final, 4, 0.004, 0.000005, allowed, time.time(), rng, False)
        if c5_fa < best_score:
            best_score = c5_fa
            best_h = h_fa

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
