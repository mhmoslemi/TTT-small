def lse_obj_grad(h, alpha):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(h)
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    obj = c_max + np.log(sum_exp) / alpha

    # Gradient computation
    # g_f[i] = sum_k p[k_idx] * h[i+k] (valid indices only)
    # g_b[i] = sum_k p[k_idx] * h[i-k] (valid indices only)
    n_corr = 2 * n - 1
    g_f = np.zeros(n)
    g_b = np.zeros(n)
    
    for k_idx in range(n_corr):
        pv = p[k_idx]
        if pv < 1e-14:
            continue

        if k >= 0:
            # h[i+k] valid for i in [0, n-k-1]
            g_f[:n-k] += pv * h[k:]
            # h[i-k] valid for i in [k, n-1]
            g_b[k:] += pv * h[:n-k]

            g_f[kk:] += pv * h[:n-kk]

            g_b[:n-kk] += pv * h[kk:]
    
    grad = 1.0 - g_f - g_b
    return obj, grad

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k

    idx = j - k

    result[mask] -= h[idx[mask]]
    idx = j + k

    phase1_budget = min(180, budget_s * 0.15)
    h_g, c5_g = fast_greedy(best_h, max_iters=80000, rng=rng,
                            time_limit=phase1_budget * 0.5, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    allowed_sa = min(120, tleft() * 0.15)
    n_sa = int(min(3000000, allowed_sa * 25000))
    h_sa, c5_sa = sa_focused(best_h, n_sa, 0.015, 0.00005, rng,
                             time_limit=allowed_sa, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    h_g2, c5_g2 = fast_greedy(best_h, max_iters=50000, rng=rng,
                              time_limit=min(60, tleft() * 0.06), start_time=start_time)
    if c5_g2 < best_score:
        best_score = c5_g2
        best_h = h_g2.copy()

    # Phase 2: Continuous optimization (the key improvement)
    alphas_coarse = [10, 30, 100, 300]
    cont_budget = min(400, tleft() * 0.35)
    h_cont, sc_cont = continuous_optimize(best_h, alphas_coarse, 300,
                                          time_limit=cont_budget, start_time=start_time)
    if sc_cont < best_score:
        best_score = sc_cont
        best_h = h_cont.copy()

    # Iterated continuous optimization with increasing alpha
    alphas_fine = [500, 1000, 2000, 5000]
    for alpha in alphas_fine:
        if tleft() < 60:
            break
        allowed = min(80, tleft() * 0.08)
        h_try = slsqp_smoothed(best_h, alpha, 500)
        h_try = iterated_project(h_try, float(len(h_try)) / 2.0)
        sc_try = compute_c5(h_try)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try.copy()

    # Phase 3: Binary refinement from continuous solution
    h_bin = project_binary(best_h, int(len(best_h) / 2.0))
    h_bg, c5_bg = fast_greedy(h_bin, max_iters=30000, rng=rng,
                              time_limit=min(60, tleft() * 0.06), start_time=start_time)
    if c5_bg < best_score:
        best_score = c5_bg
        best_h = h_bg.copy()
        h_cont_for_next = h_bg.copy()
    else:
        h_cont_for_next = best_h.copy()

    n_sa2 = int(min(2000000, max(50000, tleft() * 0.12 * 25000)))
    h_sa2, c5_sa2 = sa_focused(h_bin, n_sa2, 0.01, 0.00002, rng,
                               time_limit=min(100, tleft() * 0.12), start_time=start_time)
    if c5_sa2 < best_score:
        best_score = c5_sa2
        best_h = h_sa2.copy()

    # Phase 4: Multi-swap SA for escaping local optima
    if tleft() > 80:
        h_ms, c5_ms = sa_multi_swap(best_h, int(min(1500000, tleft() * 0.10 * 20000)),
                                    0.012, 0.00003, rng,
                                    time_limit=min(80, tleft() * 0.10), start_time=start_time)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 5: More continuous refinement
    if tleft() > 80:
        h_bin_now = project_binary(best_h, int(len(best_h) / 2.0))
        alphas_final = [3000, 8000, 15000, 30000]
        for alpha in alphas_final:
            if tleft() < 30:
                break
            try:
                h_try = slsqp_smoothed(best_h, alpha, 800)
                h_try = iterated_project(h_try, float(len(h_try)) / 2.0)
                sc_try = compute_c5(h_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 6: Try nearby resolutions
    for n_try in [n_init + 4, n_init - 4, n_init + 8, n_init - 8]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=8000, rng=rng,
                                  time_limit=min(20, tleft() * 0.05), start_time=start_time)
        if tleft() > 30:
            h_try = iterated_project(h_g3, float(n_try) / 2.0)
            try:
                h_ref = slsqp_smoothed(h_try, 3000, 300)
                h_ref = iterated_project(h_ref, float(n_try) / 2.0)
                sc_ref = compute_c5(h_ref)
                if sc_ref < c5_g3:
                    h_try = h_ref
                    c5_g3 = sc_ref
            except Exception:
                pass
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_try.copy()

    # Phase 7: Final continuous polish
    if tleft() > 30:
        alphas_ultimate = [10000, 20000, 50000]
        for alpha in alphas_ultimate:
            if tleft() < 10:
                break
            try:
                h_try = slsqp_smoothed(best_h, alpha, 1000)
                h_try = iterated_project(h_try, float(len(h_try)) / 2.0)
                sc_try = compute_c5(h_try)
                if sc_try < best_score:
                    best_score = sc_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 8: Final binary SA burst
    if tleft() > 20:
        n_use = len(best_h)
        h_bin_final = project_binary(best_h, int(n_use / 2.0))
        n_sa_f = int(min(500000, max(10000, (tleft() - 10) * 30000)))
        h_sa_f, c5_f = sa_focused(h_bin_final, n_sa_f, 0.005, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Final check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = iterated_project(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
