        val = (2.0 / p) * np.sum(h_period * (1.0 - np.roll(h_period, -k)))
        max_val = max(max_val, val)
    return max_val

def find_best_binary_periodic(p):

    best_pattern = None

def expand_periodic(h_period, n, p):
    """Expand a period-p pattern to length n (p must divide n)."""
    return np.tile(h_period, n // p)

def continuous_periodic_optimize(h_period_init, p, time_limit, start_time, deadline_offset=30):
    """Optimize continuous periodic pattern of period p."""
    # Objective: min max_k (2/p) * sum_r h[r] * (1-h[(r+k mod p)])
    # Subject to: sum(h) = p/2, 0 <= h <= 1
    
    def obj(h_vec):
        h_p = np.clip(h_vec, 0, 1)
        max_val = 0
        for k in range(p):
            val = (2.0 / p) * np.sum(h_p * (1.0 - np.roll(h_p, -k)))
            max_val = max(max_val, val)
        return max_val
    
    def grad_obj(h_vec):
        h_p = np.clip(h_vec, 0, 1)
        g = np.zeros(p)
        # Find the active constraint (max k)
        vals = np.array([(2.0 / p) * np.sum(h_p * (1.0 - np.roll(h_p, -k))) for k in range(p)])
        active_k = np.argmax(vals)
        # Gradient of the active constraint
        for r in range(p):
            # d/dh[r] of sum_r' h[r'] * (1-h[(r'+k) mod p])
            # = (1-h[(r+k) mod p]) - h[(r-k) mod p]
            g[r] = (1.0 - h_p[(r + active_k) % p]) - h_p[(r - active_k) % p]
        g *= (2.0 / p)
        return g
    
    # Use SLSQP
    bounds = [(0.0, 1.0)] * p
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - p/2.0, 'jac': lambda x: np.ones(p)}]
    
    try:
        res = minimize(obj, h_period_init, jac=grad_obj, method='SLSQP',
                       bounds=bounds, constraints=cons, options={'maxiter': 1000, 'ftol': 1e-14})
        if res.success or res.fun < obj(h_period_init):
            return np.clip(res.x, 0, 1)
    except Exception:
        pass
    
    return h_period_init

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time, deadline_offset=30,
                   damping=0.6, reg=1e-5):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

def project_grad_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000, 100000]
    steps_per_alpha = 2000
    top_m_list = [10, 20, 30, 50, 80, 120, 200, 300, 400, 2 * n - 1, 2 * n - 1]

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]

        momentum_v = np.zeros(n)
        beta = 0.85
        stalled = 0

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad

            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-12:
                break

            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 2.0)

            improved = False
            for _ in range(40):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5

            if not improved:
                stalled += 1
                if stalled > 50:
                    break

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
