Standard cross-correlation: (a ⊘ b)[i] = sum_j a[j+i] * conj(b[j]) = sum_j a[j] * conj(b[j-i])

So (w ⊛ h)[i] = sum_k w[k] * h[i+k]

This equals the convolution of w_flipped with h, shifted:
(w_flipped * h)[n-1-i]... let me verify.

(w_flipped * h)[m] = sum_j w_flipped[j] * h[m-j] = sum_j w[n-1-j] * h[m-j]

Let k = n-1-j, so j = n-1-k:
= sum_k w[k] * h[m-(n-1-k)] = sum_k w[k] * h[m-n+1+k]

For this to equal (w ⊛ h)[i] = sum_k w[k] * h[i+k], we need m-n+1 = i, so m = i+n-1.

So (w ⊛ h)[i] = (w_flipped * h)[i+n-1]

We can compute w_flipped * h via FFT.

Part1[i] = (w * (1-h))[i] (standard convolution, first n elements)
Part2[i] = (w_flipped * h)[i+n-1] (standard convolution, elements n-1 to 2n-2)

Both can be computed via FFT in O(n log n)!

```python
def compute_grad_fft(h, w, dx):
    """Compute gradient using FFT."""
    n = len(h)
    # Part 1: conv(w, 1-h), first n elements
    N = 2 * n
    W = np.fft.rfft(w, N)
    one_minus_h = 1.0 - h
    OMH = np.fft.rfft(one_minus_h, N)
    part1 = np.fft.irfft(W * OMH, N)[:n]
    
    # Part 2: conv(w_flipped, h), elements n-1 to 2n-2
    w_flipped = w[::-1]
    WF = np.fft.rfft(w_flipped, N)
    H = np.fft.rfft(h, N)
    conv_wh = np.fft.irfft(WF * H, N)
    part2 = conv_wh[n-1:n-1+n]
    
    grad = dx * (part1 - part2)
    return grad
```

def project_sum_box(v, target):
    """Project v onto {x: sum(x)=target, 0<=x[i]<=1}."""
    # This is equivalent to: min ||x-v||^2 s.t. sum(x)=target, 0<=x<=1
    # Solution: x[i] = clip(v[i] - lambda, 0, 1) for some lambda
    # Find lambda by bisection
    lo, hi = v.min() - target, v.max() + target
    # Actually, lambda is such that sum(clip(v - lambda, 0, 1)) = target
    # sum(clip(v - lambda, 0, 1)) is decreasing in lambda
    # When lambda -> -inf: sum -> n (all 1s)
    # When lambda -> +inf: sum -> 0 (all 0s)
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0, 1)

def balancing_surgery(h, target, n_steps=100, n_lags=3):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = get_c5(h_cur)
    eps = 0.01
    
    for step in range(n_steps):
        corr = compute_corr_full(h_cur)[n-1:]
        top_k = np.argsort(corr)[::-1][:n_lags]
        
        # Build gradient for the weighted combination of top lags
        # Weight the top lags to try to equalize them
        top_corr = corr[top_k]
        max_top = top_corr[0]
        # Weights: give more weight to lags that are above average
        weights = np.ones(n_lags)
        for i in range(n_lags):
            if top_corr[i] > max_top * 0.99:
                weights[i] = 2.0
        
        w = np.zeros(n)
        for i, k in enumerate(top_k):
            w[int(k)] = weights[i]
        
        grad = fft_gradient(h_cur, w, dx)
        grad_proj = grad - np.mean(grad)
        
        h_new = h_cur - eps * grad_proj
        h_new = project_sum_box(h_new, target)
        sc_new = get_c5(h_new)

            eps = min(eps * 1.01, 0.05)
        else:
            eps *= 0.85

                break

            best_c5 = c5_x
            best_x = x.copy()
