        for lag_idx in top_lags_idx[::-1]:
            k = int(lag_idx - (n - 1))
            if k == 0:
                continue
            i_pos = ones
            vi_plus = (i_pos + k >= 0) & (i_pos + k < n)
            vi_minus = (i_pos - k >= 0) & (i_pos - k < n)
            i_plus = np.zeros(len(ones))
            i_plus[vi_plus] = h[i_pos[vi_plus] + k]
            i_minus = np.zeros(len(ones))
            i_minus[vi_minus] = h[i_pos[vi_minus] - k]
            i_sc = i_plus + i_minus

            j_pos = zeros
            vj_plus = (j_pos + k >= 0) & (j_pos + k < n)
            vj_minus = (j_pos - k >= 0) & (j_pos - k < n)
            j_plus = np.zeros(len(zeros))
            j_plus[vj_plus] = h[j_pos[vj_plus] + k]
            j_minus = np.zeros(len(zeros))
            j_minus[vj_minus] = h[j_pos[vj_minus] - k]
            j_sc = j_plus + j_minus

            ti = np.argsort(i_sc)[:min(15, len(ones))]
            tj = np.argsort(-j_sc)[:min(15, len(zeros))]

            for oi in ti:
                for oj in tj:
                    i_val = int(ones[oi])
                    j_val = int(zeros[oj])
                    if i_val == j_val:
                        continue
                    delta = compute_delta(h, i_val, j_val)
                    new_max = float((full + delta).max())
                    if new_max < max_val:
                        h[i_val] = 0.0
                        h[j_val] = 1.0
                        full += delta
                        stall = 0
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            stall += 1

    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy with lag awareness
    allowed = min(60, tleft() * 0.10)
    h_g, c5_g = greedy_lag(best_h, max_iters=30000, rng=rng,
                           time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Main lag-aware SA (long)
    allowed = min(300, tleft() * 0.30)
    n_sa = int(min(3000000, allowed * 30000))
    h_sa, c5_sa = lag_aware_sa(best_h, n_sa, 0.03, 0.0005, rng,
                               time_limit=allowed, start_time=start_time,
                               n_hot=3, n_per_lag=6, n_random=3, top_k=10)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Iterated local search with restarts
    for it in range(15):
        if tleft() < 80:
            break

        # Greedy polish
        allowed = min(25, tleft() * 0.06)
        h_g2, c5_g2 = greedy_lag(best_h, max_iters=15000, rng=rng,
                                 time_limit=allowed, start_time=start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 50:
            break

        # Perturb with varying intensity
        n_pert = rng.randint(2, 6 + it)
        h_pert = best_h.copy()
        for _ in range(n_pert):
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0

        # SA from perturbed
        allowed = min(60, tleft() * 0.10)
        n_sa2 = int(min(1500000, allowed * 25000))
        h_sa2, c5_sa2 = lag_aware_sa(h_pert, n_sa2, 0.02, 0.0002, rng,
                                     time_limit=allowed, start_time=start_time,
                                     n_hot=3, n_per_lag=5, n_random=5, top_k=8)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

        # Also try greedy from perturbed
        if tleft() > 30:
            allowed = min(15, tleft() * 0.04)
            h_g3, c5_g3 = greedy_lag(h_pert, max_iters=10000, rng=rng,
                                     time_limit=allowed, start_time=start_time)
            if c5_g3 < best_score:
                best_score = c5_g3
                best_h = h_g3.copy()

    # Phase 4: Coarse-to-fine
    if tleft() > 150:
        n_coarse = 120
        target_coarse = n_coarse // 2
        x_new = np.linspace(0, 1, n_coarse)
        x_old = np.linspace(0, 1, n_init)
        h_coarse = np.interp(x_new, x_old, best_h)
        h_coarse_bin = project_binary(h_coarse, target_coarse)

        allowed = min(30, tleft() * 0.12)
        h_cg, c5_cg = greedy_lag(h_coarse_bin, max_iters=8000, rng=rng,
                                 time_limit=allowed * 0.5, start_time=start_time)
        allowed2 = min(25, tleft() * 0.08)
        n_sac = int(min(300000, allowed2 * 20000))
        h_cs, c5_cs = lag_aware_sa(h_cg, n_sac, 0.015, 0.0003, rng,
                                   time_limit=allowed2, start_time=start_time,
                                   n_hot=3, n_per_lag=4, n_random=3, top_k=8)
        if c5_cs < c5_cg:
            h_coarse_best = h_cs
        else:
            h_coarse_best = h_cg

        h_fine = interpolate_to(h_coarse_best, n_init, target_init)
        h_fine_bin = project_binary(h_fine, target_init)
        allowed3 = min(25, tleft() * 0.08)
        h_fg, c5_fg = greedy_lag(h_fine_bin, max_iters=12000, rng=rng,
                                 time_limit=allowed3 * 0.5, start_time=start_time)
        allowed4 = min(20, tleft() * 0.06)
        n_saf = int(min(300000, allowed4 * 20000))
        h_fs, c5_fs = lag_aware_sa(h_fg, n_saf, 0.01, 0.0002, rng,
                                   time_limit=allowed4, start_time=start_time,
                                   n_hot=3, n_per_lag=5, n_random=3, top_k=8)
        if c5_fs < c5_fg:
            h_embed = h_fs
            c5_embed = c5_fs
        else:
            h_embed = h_fg
            c5_embed = c5_fg
        if c5_embed < best_score:
            best_score = c5_embed
            best_h = h_embed.copy()

    # Phase 5: Try nearby resolutions
    for n_try in [n_init + 2, n_init + 4, n_init + 6, n_init - 2, n_init - 4]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.06)
        h_g4, c5_g4 = greedy_lag(h_bin_try, max_iters=8000, rng=rng,
                                 time_limit=allowed * 0.5, start_time=start_time)
        allowed2 = min(15, tleft() * 0.04)
        n_sa3 = int(min(200000, allowed2 * 15000))
        h_s3, c5_s3 = lag_aware_sa(h_g4, n_sa3, 0.01, 0.0002, rng,
                                   time_limit=allowed2, start_time=start_time,
                                   n_hot=3, n_per_lag=4, n_random=3, top_k=6)
        if c5_s3 < c5_g4:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g4
            c5_try = c5_g4
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: SLSQP refinement on continuous
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 600, 2000, 6000]
        allowed = min(tleft() - 30, 200)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1500, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final SA burst on binary projection
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        # Quick greedy first
        allowed_q = min(15, max(5.0, tleft() * 0.1))
        h_qg, c5_qg = greedy_lag(h_bin_final, max_iters=10000, rng=rng,
                                 time_limit=allowed_q, start_time=start_time)
        if c5_qg < compute_c5(h_bin_final):
            h_bin_final = h_qg
        n_sa_f = int(min(800000, max(50000, (tleft() - 15) * 25000)))
        h_sa_f, c5_f = lag_aware_sa(h_bin_final, n_sa_f, 0.008, 0.0001,
                                    rng, time_limit=max(5.0, tleft() - 12),
                                    start_time=start_time,
                                    n_hot=3, n_per_lag=5, n_random=2, top_k=8)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 8: Final SLSQP if time
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [3000, 8000, 20000], 1000, 1e-15,
                                 time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
