def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_lag_vectors(g, dx, n):
    vectors = []
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            v = np.zeros(n)
            v[lo:hi] = g[lo - k:hi - k] * dx
            vectors.append(v)
    return vectors


def slp_full(h_init, n, target, max_iter, time_limit, eps_sc=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        t0 = time.time()

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=eps_sc, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                accepted = True
                break

        if not accepted:
            if it > 20:
                break
            h = h_new

    return best_h, best_c5


def slp_top_k(h_init, n, target, max_iter, time_limit, n_top=300, eps_sc=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=eps_sc, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        accepted = False
        for alpha in [1.0, 0.6, 0.4, 0.3, 0.2, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                accepted = True
                break

        if not accepted:
            if it > 30:
                break

    return best_h, best_c5


def smooth_max_obj(h, alpha):
    overlaps = compute_overlaps(h)
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_grad(h, alpha):
    overlaps = compute_overlaps(h)
    n = len(h)
    dx = 2.0 / n
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    weights = np.exp(shiftted) if False else np.exp(shifted)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(len(weights)):
        if abs(weights[idx]) < 1e-15:
            continue
        k = idx - n_half
        w = weights[idx]
        lo = max(0, k)
        hi = min(n, k + n)
        if hi <= lo:
            continue
        i_arr = np.arange(lo, hi)
        j_arr = i_arr - k
        grad[i_arr] += w * (1.0 - h[j_arr])
        if k >= 0:
            i2 = np.arange(0, n - k)
            j2 = i2 + k
        else:
            i2 = np.arange(-k, n)
            j2 = i2 + k
        grad[i2] += w * (-h[j2])
    grad *= dx
    return grad


_smooth_n = 0
_smooth_target = 0


def _smooth_cons_fun(h):
    return np.sum(h) - _smooth_target


def _smooth_cons_jac(h):
    return np.ones(_smooth_n)


def smooth_refine(h_start, n, target, alphas, maxiter, time_limit):
    global _smooth_n, _smooth_target
    _smooth_n = n
    _smooth_target = target
    start = time.time()
    h_cur = h_start.copy()
    h_cur = project_feasible(h_cur, target)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _smooth_cons_fun, 'jac': _smooth_cons_jac}]

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
            h_try = project_feasible(res.x, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try
        except Exception:
            pass
    return best_h, best_c5


def mirror_descent(h_init, n, target, max_steps, time_limit, n_top=30):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_prev = h.copy()
    momentum = 0.9
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step % 500 == 400:
            momentum = max(0.5, momentum * 0.95)

        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        top_vals = overlaps[top_idx]

        w = 0.8 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi <= lo:
                continue
            i1 = np.arange(lo, hi)
            j1 = i1 - k
            grad[i1] += w[i] * (1.0 - y[j1])
            if k >= 0:
                i2 = np.arange(0, n - k)
                j2 = i2 + k
            else:
                i2 = np.arange(-k, n)
                j2 = i2 + k
            grad[i2] += w[i] * (-y[j2])
        grad *= dx

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-16:
            break

        c5_y = float(np.max(overlaps))
        polyak = (c5_y - best_c5) / (norm_g ** 2 + 1e-20)
        lr = min(max(0.005, polyak), 0.3)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def shake(h, n, target, rng, noise_level, n_rounds):
    h_shake = h.copy()
    for _ in range(n_rounds):
        mask = rng.random(n) < 0.4
        if mask.sum() > 0:
            h_shake[mask] += rng.randn(mask.sum()) * noise_level
        h_shake = project_feasible(h_shake, target)
    return h_shake


def run(seed=42, budget_s=1300, **kwargs):
    global _smooth_n, _smooth_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLP with all lags (most important improvement)
    if tleft() > 300:
        t1 = min(250, tleft() * 0.35)
        h1, c5_1 = slp_full(best_h, n, target, max_iter=500, time_limit=t1, eps_sc=1e-10)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1

    # Phase 2: SLP with top-K for more iterations
    if tleft() > 150:
        t2 = min(120, tleft() * 0.2)
        h2, c5_2 = slp_top_k(best_h, n, target, max_iter=1000, time_limit=t2, n_top=400, eps_sc=1e-10)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2

    # Phase 3: Mirror descent / subgradient
    if tleft() > 100:
        t3 = min(80, tleft() * 0.15)
        h3, c5_3 = mirror_descent(best_h, n, target, max_steps=50000, time_limit=t3, n_top=40)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3

    # Phase 4: Smooth refinement with L-BFGS-B
    if tleft() > 80:
        t4 = min(60, tleft() * 0.15)
        h4, c5_4 = smooth_refine(best_h, n, target,
                                  alphas=[50, 200, 1000, 5000, 20000],
                                  maxiter=500, time_limit=t4)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4

    # Phase 5: Another round of SLP full after smooth refine
    if tleft() > 80:
        t5 = min(70, tleft() * 0.2)
        h5, c5_5 = slp_full(best_h, n, target, max_iter=300, time_limit=t5, eps_sc=1e-11)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5

    # Phase 6: Shake and re-optimize
    for attempt in range(4):
        if tleft() < 60:
            break
        noise = 0.02 + attempt * 0.015
        h_sh = shake(best_h, n, target, rng, noise, n_rounds=2)
        t6 = min(35, tleft() * 0.3)
        h6, c5_6 = slp_full(h_sh, n, target, max_iter=200, time_limit=t6, eps_sc=1e-9)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6

    # Phase 7: Try higher resolution
    if tleft() > 120:
        for n_try in [500, 600, 800]:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            h_hi = interpolate_to(best_h, n_try, target_try)
            t7 = min(50, tleft() * 0.3)
            h_hi2, c5_hi = slp_full(h_hi, n_try, target_try, max_iter=200, time_limit=t7, eps_sc=1e-9)
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_hi2
                break

    # Phase 8: Final SLP polish
    if tleft() > 50:
        n_f = len(best_h)
        target_f = n_f / 2.0
        t8 = min(40, tleft() * 0.7)
        h8, c5_8 = slp_full(best_h, n_f, target_f, max_iter=200, time_limit=t8, eps_sc=1e-11)
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8

    # Phase 9: Final smooth polish
    if tleft() > 30:
        n_f = len(best_h)
        target_f = n_f / 2.0
        t9 = min(25, tleft() * 0.8)
        h9, c5_9 = smooth_refine(best_h, n_f, target_f,
                                  alphas=[200, 1000, 5000, 20000, 100000],
                                  maxiter=300, time_limit=t9)
        if c5_9 < best_score:
            best_score = c5_9
            best_h = h9

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
