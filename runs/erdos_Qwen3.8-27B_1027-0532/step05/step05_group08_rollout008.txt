```python
                h_new[j] -= delta

            else:
                h_new[j] += delta
                h_new[best_cand] -= delta

                best_score = sc
                improved = True
```

```python
        return np.ones(n)
```

```python
                delta = min(1.0 - best_h[j], best_h[jp])
                if delta < 1e-12:
                    continue
                gain = (g[jp] - g[j]) * delta
                if gain > best_gain:
                    best_gain = gain
                    best_move = (int(j), int(jp), delta)
        
        if best_move is None:
            break
        
        j, jp, delta = best_move
        h_new = best_h.copy()
        h_new[j] += delta
        h_new[jp] -= delta
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score - 1e-14:
            best_h = h_new
            best_score = sc
        else:
            # Try smaller steps
            found = False
            for frac in [0.5, 0.25, 0.1]:
                h_new = best_h.copy()
                h_new[j] += delta * frac
                h_new[jp] -= delta * frac
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_score - 1e-14:
                    best_h = h_new
                    best_score = sc
                    found = True
                    break
            if not found:
                break
    
    return best_h, best_score
```

One more thought: I should also try a "dual" approach. Define g(x) = 1 - h(x). Then:
∫h(x)(1-h(x+k))dx = ∫h(x)*g(x+k)dx

And the constraint becomes ∫g(x)dx = ∫(1-h(x))dx = 2 - 1 = 1.
So g also has integral 1. And the objective is ∫h(x)*g(x+k)dx.

We want to minimize max_k ∫h(x)*g(x+k)dx where ∫h = ∫g = 1, 0 ≤ h, g ≤ 1, and g = 1-h.

This is interesting: we want two "bumps" h and g (each of integral 1) to be as "disjoint" as possible under all shifts. Since g = 1-h, this means h should be "spread out" so that its complement is also "spread out" in a way that they don't align under any shift.

This reformulation doesn't immediately give a new algorithm, but it confirms the intuition that the optimal h is "spread out" (not concentrated).

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def swap_local_search_lse(h, n, target, alpha, max_steps, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in range(max_steps):
        if time.time() > deadline:
            break
        g = lse_grad(best_h, alpha)
        neg_mask = (g < -1e-12) & (best_h < 1.0 - 1e-10)
        pos_mask = (g > 1e-12) & (best_h > 1e-10)
        if not neg_mask.any() or not pos_mask.any():
            break
        neg_idx = np.where(neg_mask)[0]
        pos_idx = np.where(pos_mask)[0]
        neg_sorted = neg_idx[np.argsort(g[neg_idx])[:25]]
        pos_sorted = pos_idx[np.argsort(-g[pos_idx])[:25]]
        best_gain = 0.0
        best_move = None
        for j in neg_sorted:
            for jp in pos_sorted:
                delta = min(1.0 - best_h[j], best_h[jp])
                if delta < 1e-12:
                    continue
                gain = (g[jp] - g[j]) * delta
                if gain > best_gain:
                    best_gain = gain
                    best_move = (int(j), int(jp), delta)
        if best_move is None:
            break
        j, jp, delta = best_move
        found = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
            d = delta * frac
            if d < 1e-12:
                continue
            h_new = best_h.copy()
            h_new[j] += d
            h_new[jp] -= d
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-14:
                best_h = h_new
                best_score = sc
                found = True
                break
        if not found:
            break
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.02)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.astype(float).copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def optimize_resolution(h, n, target, deadline):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    alphas = [5, 8, 12, 20, 35, 60, 100, 180, 320, 560, 1000, 1800, 3200, 6000, 10000, 18000]
    for alpha in alphas:
        if time.time() > deadline - 60:
            break
        for mu in [10.0, 100.0]:
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 250)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
    for alpha in [500, 2000, 8000, 25000]:
        if time.time() > deadline - 50:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 2500, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if time.time() < deadline - 40:
        h_sw, sc_sw = swap_local_search_lse(best_h, n, target, 2000, 400, deadline - 20)
        if sc_sw < best_score:
            best_score = sc_sw
            best_h = h_sw
    if time.time() < deadline - 30:
        for temp in [0.02, 0.01, 0.005, 0.002]:
            if time.time() > deadline - 15:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n, target, 1500, temp, 0.004, 0.997, deadline - 10)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg
    if time.time() < deadline - 15:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 500, 0.004, deadline - 5)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
    if time.time() < deadline - 5:
        h_new = slsqp_round(best_h, n, target, 15000, 1500, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score
```
