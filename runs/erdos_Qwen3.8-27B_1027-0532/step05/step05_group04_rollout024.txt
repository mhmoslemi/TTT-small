Actually, the direct computation of the effect of a swap is:
If we flip h[i] from 1 to 0 and h[j] from 0 to 1:

For overlap at lag k: O(k) = ∑_{m=0}^{n-1-k} h[m+k](1-h[m])

The change due to flipping position p:
- If p appears as h[m+k] (i.e., m+k = p, so m = p-k): contribution changes from h[p](1-h[p-k]) to (1-h[p])(1-h[p-k])
  - Delta = (1-h[p])(1-h[p-k]) - h[p](1-h[p-k]) = (1-h[p] - h[p])(1-h[p-k]) = (1-2h[p])(1-h[p-k])
  - For h[p]=1: delta = -1*(1-h[p-k]) = -(1-h[p-k])
  - For h[p]=0: delta = 1*(1-h[p]) = (1-0) = 1... wait, that's the change when going from 0 to 1

Let me redo this. Let's say we change h[p] from a to b (a∈{0,1}, b=1-a).

Effect on O(k):
- p appears as h[m+k] when m = p-k (if 0 ≤ p-k ≤ n-1-k, i.e., k ≤ p ≤ n-1):
  Old contribution: h[p-k+k] * (1-h[p-k]) = a * (1-h[p-k])
  New contribution: b * (1-h[p-k])
  Delta: (b-a) * (1-h[p-k])

- p appears as h[m] when m = p (if 0 ≤ p ≤ n-1-k):
  Old contribution: h[p+k] * (1-a)
  New contribution: h[p+k] * (1-b)
  Delta: h[p+k] * ((1-b) - (1-a)) = h[p+k] * (a-b)

Total delta for O(k) when flipping h[p] from a to b:
delta_k = (b-a) * [(1-h[p-k]) * 1_{k≤p≤n-1} - h[p+k] * 1_{p≤n-1-k}]

For a swap (flip i from 1 to 0, and j from 0 to 1):
delta_k = -1 * [(1-h[i-k]) * 1_{k≤i≤n-1} - h[i+k] * 1_{i≤n-1-k}] + 1 * [(1-h[j-k]) * 1_{k≤j≤n-1} - h[j+k] * 1_{j≤n-1-k}]

This can be computed in O(n) for a given swap (i,j). But we need to check all n lags, so it's O(n) per swap.

For a full search over all O(n²) possible swaps, that's O(n³) per iteration, which is too slow for n=400.

Better approach: Precompute the "delta vectors" for each position flip.

For position p with current value h[p]:
delta_flip[p][k] = (1-2*h[p]) * [(1-h[p-k])*1_{k≤p} - h[p+k]*1_{p+k<n}]

Wait, let me be more careful. If h[p] changes by amount δ (from h[p] to h[p]+δ):
- As h[m+k] where m+k=p: contribution δ*(1-h[p-k]) for k such that 0≤p-k and p-k≤n-1-k, i.e., k≤p and p-k≤n-1-k → p≤n-1 (always true)
  So: k ranges from 0 to min(p, n-1)
- As h[m] where m=p: contribution -δ*h[p+k] for k such that 0≤p≤n-1-k, i.e., k≤n-1-p
  So: k ranges from 0 to min(n-1-p, n-1)

So the delta vector for a unit change at position p is:
d[k] = (1-h[p-k]) * 1_{k≤p} - h[p+k] * 1_{k≤n-1-p} for k=0,...,n-1

For a swap (i goes 1→0, j goes 0→1):
total_delta[k] = -d_i[k] + d_j[k]

And we want to find the swap that minimizes max_k (O(k) + total_delta[k]) * dx.

We can precompute d[p] for all p in O(n²) time, then for each pair (i,j) the delta is d[j] - d[i], and we need to check if max_k(O + d[j]-d[i]) < max_k(O).

This is still O(n³) for all pairs. But we can be smarter:

For the current max M = max_k O(k), we only need to consider lags where O(k) is close to M. Let S = {k : O(k) > M - ε} for some small ε. Then we only need to check that max_{k∈S} (O(k) + d[j][k] - d[i][k]) ≤ M - δ for some δ > 0.

This reduces the check to O(|S|) per swap.

import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    over = compute_corr(h)[n-1:n-1+n]
    return float(np.max(over)) * dx

def compute_overlaps(h):
    corr = compute_corr(h)
    n = len(h)
    return corr[n-1:n-1+n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(x):
    corr = compute_corr(x)
    n = _g_n
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_refine(h, n, target, alpha, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_lse, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def weighted_grad(h, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    return grad * dx

def anneal_minimax(h, n, target, beta_list, n_iters_per_beta, lr_init, deadline):
    """Weighted gradient descent with annealing on beta."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for beta in beta_list:
        for it in range(n_iters_per_beta):
            if time.time() > deadline:
                return h_cur, cur_score
            over = compute_overlaps(h_cur)
            max_over = float(np.max(over))
            weights = np.exp(beta * (over - max_over))
            weights /= weights.sum()
            grad = weighted_grad(h_cur, weights, dx)
            # Line search
            for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.01]:
                h_new = h_cur - trial_lr * grad
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
                    lr = trial_lr
                    break
            else:
                lr *= 0.7
                if lr < 1e-8:
                    break
    return h_cur, cur_score

def binary_sim_anneal(h, n, target, max_iters, t_init, t_final, deadline):
    """Simulated annealing on binary sequence via swaps."""
    dx = 2.0 / n
    # Start from current h, snap to binary
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    if n_ones > n_target:
        # Remove excess ones (those with lowest h values)
        one_idx = np.where(h_bin == 1.0)[0]
        # Sort by h value ascending, remove the ones with lowest h
        order = np.argsort(h[one_idx])
        to_remove = one_idx[order[:n_ones - n_target]]
        h_bin[to_remove] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_idx])
        to_add = zero_idx[order[:n_target - n_ones]]
        h_bin[to_add] = 1.0
    
    cur_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_score = cur_score
    cur_h = h_bin.copy()
    
    T = t_init
    cool = (t_final / t_init) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cool
        
        one_positions = np.where(cur_h == 1.0)[0]
        zero_positions = np.where(cur_h == 0.0)[0]
        
        # Random swap
        i = one_positions[rng.randint(len(one_positions))]
        j = zero_positions[rng.randint(len(zero_positions))]
        
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = compute_c5(cur_h)
        
        delta = new_score - cur_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # Revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
    
    return best_h, best_score

def frequency_perturb(h, n, target, rng, n_modes=10, amplitude=0.05):
    """Perturb h in frequency domain."""
    H = np.fft.rfft(h)
    # Perturb a few random frequency modes
    freqs = np.arange(len(H))
    selected = rng.choice(freqs[1:], size=min(n_modes, len(H)-1), replace=False)
    for f in selected:
        H[f] += amplitude * (rng.rand() - 0.5) * (1+2j) * abs(H[f] + 1e-10)
    h_new = np.fft.irfft(H, n=n)
    return project_feasible(h_new, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def de_search(n, target, pop_size, max_gen, rng, deadline, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            cands = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(cands, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            ts = compute_c5(trial)
            if ts <= scores[i]:
                population[i] = trial
                scores[i] = ts
                if ts < best_score:
                    best_score = ts
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        over = compute_overlaps(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def compute_c5(h):

    dx = 2.0 / n
    over = compute_corr(h)[n - 1:n - 1 + n]
    return float(np.max(over)) * dx

def compute_overlaps(h):

    n = len(h)
    return corr[n - 1:n - 1 + n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-14:

        if cnt == 0:

    _g_target = target

            weights = np.exp(beta * (over - max_over))
            weights /= weights.sum()
            grad = weighted_grad(h_cur, weights, dx)
            improved = False
            for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.01]:
                h_new = h_cur - trial_lr * grad
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
                    lr = max(trial_lr, 1e-6)
                    improved = True
                    break
            if not improved:
                lr *= 0.7
                if lr < 1e-9:
                    break
    return h_cur, cur_score

def binary_sim_anneal(h, n, target, max_iters, t_init, t_final, deadline, rng_local):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    if n_ones > n_target:
        one_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_idx])
        to_remove = one_idx[order[:n_ones - n_target]]
        h_bin[to_remove] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_idx])
        to_add = zero_idx[order[:n_target - n_ones]]
        h_bin[to_add] = 1.0

    over = compute_overlaps(h_bin)
    cur_max = float(np.max(over))
    best_h = h_bin.copy()
    best_score = cur_max * dx
    cur_h = h_bin.copy()

    T = t_init
    if max_iters > 1:
        cool = (t_final / t_init) ** (1.0 / max_iters)
    else:
        cool = 1.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cool
        one_positions = np.where(cur_h == 1.0)[0]
        zero_positions = np.where(cur_h == 0.0)[0]
        if len(one_positions) == 0 or len(zero_positions) == 0:
            break
        i = int(one_positions[rng_local.randint(len(one_positions))])
        j = int(zero_positions[rng_local.randint(len(zero_positions))])

        delta = np.zeros(n)
        ki = min(i, n - 1)
        if ki >= 0:
            delta[:ki + 1] -= (1.0 - cur_h[i - np.arange(ki + 1)])
        ki2 = min(n - 1 - i, n - 1)
        if ki2 >= 0:
            delta[:ki2 + 1] += cur_h[i + np.arange(ki2 + 1)]
        kj = min(j, n - 1)
        if kj >= 0:
            delta[:kj + 1] += (1.0 - cur_h[j - np.arange(kj + 1)])
        kj2 = min(n - 1 - j, n - 1)
        if kj2 >= 0:
            delta[:kj2 + 1] -= cur_h[j + np.arange(kj2 + 1)]

        new_over = over + delta
        new_max = float(np.max(new_over))
        new_score = new_max * dx
        delta_score = new_score - cur_max * dx

        if new_score < best_score or rng_local.rand() < np.exp(-delta_score / max(T, 1e-12)):
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            over = new_over
            if new_max < float(np.max(over)):
                pass
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()

    return best_h, best_score

def binary_steepest(h, n, target, max_iters, deadline, rng_local):
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    if n_ones > n_target:
        one_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[one_idx])
        to_remove = one_idx[order[:n_ones - n_target]]
        h_bin[to_remove] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_idx])
        to_add = zero_idx[order[:n_target - n_ones]]
        h_bin[to_add] = 1.0

    over = compute_overlaps(h_bin)
    cur_max = float(np.max(over))
    best_h = h_bin.copy()
    best_score = cur_max * dx
    cur_h = h_bin.copy()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        one_positions = np.where(cur_h == 1.0)[0]
        zero_positions = np.where(cur_h == 0.0)[0]
        n_sample = min(50, len(one_positions))
        z_sample = min(50, len(zero_positions))
        s_ones = rng_local.choice(len(one_positions), n_sample, replace=False)
        s_zeros = rng_local.choice(len(zero_positions), z_sample, replace=False)

        best_delta_max = 0.0
        best_i = -1
        best_j = -1

        for si in s_ones:
            i = int(one_positions[si])
            for sj in s_zeros:
                j = int(zero_positions[sj])
                delta = np.zeros(n)
                ki = min(i, n - 1)
                if ki >= 0:
                    delta[:ki + 1] -= (1.0 - cur_h[i - np.arange(ki + 1)])
                ki2 = min(n - 1 - i, n - 1)
                if ki2 >= 0:
                    delta[:ki2 + 1] += cur_h[i + np.arange(ki2 + 1)]
                kj = min(j, n - 1)
                if kj >= 0:
                    delta[:kj + 1] += (1.0 - cur_h[j - np.arange(kj + 1)])
                kj2 = min(n - 1 - j, n - 1)
                if kj2 >= 0:
                    delta[:kj2 + 1] -= cur_h[j + np.arange(kj2 + 1)]
                new_max = float(np.max(over + delta))
                improvement = cur_max - new_max
                if improvement > best_delta_max:
                    best_delta_max = improvement
                    best_i = i
                    best_j = j

        if best_i >= 0:
            cur_h[best_i] = 0.0
            cur_h[best_j] = 1.0
            over = over + (lambda d: d)(np.zeros(n))
            cur_h[best_i] = 0.0
            ki = min(best_i, n - 1)
            ki2 = min(n - 1 - best_i, n - 1)
            kj = min(best_j, n - 1)
            kj2 = min(n - 1 - best_j, n - 1)
            # Recompute over from scratch for correctness
            over = compute_overlaps(cur_h)
            cur_max = float(np.max(over))
            if cur_max * dx < best_score:
                best_score = cur_max * dx
                best_h = cur_h.copy()

    return best_h, best_score

def frequency_perturb(h, n, target, rng_local, n_modes, amplitude):
    H = np.fft.rfft(h)
    freqs = np.arange(1, len(H))
    selected = rng_local.choice(freqs, size=min(n_modes, len(freqs)), replace=False)
    for f in selected:
        H[f] += amplitude * (rng_local.rand() - 0.5) * (1 + 2j) * (abs(H[f]) + 0.01)
    h_new = np.fft.irfft(H, n=n)
    return project_feasible(h_new, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
