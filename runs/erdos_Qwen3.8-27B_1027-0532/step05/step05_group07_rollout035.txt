Let me also think about the projection. The feasible set is {h : 0 <= h[i] <= 1, sum(h) = target}.

- Find lambda such that sum(clip(y - lambda, 0, 1)) = target

Actually, let me be more careful. The Euclidean projection onto {0 <= x <= 1, sum(x) = target} is:
x* = argmin ||x - y||^2 subject to 0 <= x <= 1, sum(x) = target.

f(-inf) = n * 1 = n (all clipped to 1)
f(+inf) = 0 (all clipped to 0)

So if target in [0, n], there exists a lambda with f(lambda) = target.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    y = h.copy()
    # Bisection on lambda
    lo = -max(y) - 1e-10
    hi = max(y) + 1 - 1e-10 + 1e-10
    # f(lambda) = sum(clip(y - lambda, 0, 1)) is non-increasing in lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(val)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-15:
        # Adjust slightly
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        if mask.sum() > 0:
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_smooth_gradient(h, alpha):
    """Compute gradient of (1/alpha)*log(sum exp(alpha*c_k))"""
    corr = compute_corr(h)
    # Softmax weights
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    w = exp_vals / np.sum(exp_vals)
    n = len(h)
    # U has length 2n-1
    U = w  # already length 2n-1 (same as corr)
    term1 = np.correlate(U, 1.0 - h, 'full')[n-1:2*n-1]
    term2 = np.correlate(U[::-1], h, 'full')[n-1:2*n-1]
    return term1 - term2, corr, w

def projected_gradient_descent(h, n, target, alpha, n_iters, eta0, momentum=True, deadline=None):
    """Projected gradient descent on the smooth max objective"""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    if momentum:
        h_prev = h_cur.copy()
        beta = 1.0
    else:
        h_prev = None
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute gradient
        grad, corr, w = compute_smooth_gradient(h_cur, alpha)
        
        # Adaptive step size
        eta = eta0 / (1.0 + 0.01 * it)
        
        # Nesterov momentum
        if momentum:
            beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
