def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:

            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(overlaps, alpha):
    """Compute smooth max of overlaps using log-sum-exp."""
    c = np.max(overlaps)
    return c + np.log(np.sum(np.exp(alpha * (overlaps - c)))) / alpha

def smooth_max_grad(overlaps, alpha):
    """Compute softmax weights for the smooth max."""
    c = np.max(overlaps)
    logits = alpha * (overlaps - c)
    weights = np.exp(logits)
    weights /= np.sum(weights)
    return weights

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:

            idx2 = np.where(h < 1.0)[0]
        else:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_weights(overlaps, alpha):
    c = np.max(overlaps)
    logits = alpha * (overlaps - c)
    # Clip for numerical stability
    logits = np.clip(logits, -500, 0)
    weights = np.exp(logits)
    s = np.sum(weights)
    if s < 1e-300:
        weights = np.zeros_like(overlaps)
        weights[np.argmax(overlaps)] = 1.0
    else:
        weights /= s
    return weights

def compute_grad_topk(h, overlaps, n, top_k=50, alpha=100):
    """Compute gradient of smooth max using only top-k lags."""
    dx = 2.0 / n
    # Get top-k lags
    top_idx = np.argpartition(overlaps, -top_k)[-top_k:]
    top_vals = overlaps[top_idx]
    
    # Compute softmax weights over top-k
    c = np.max(top_vals)
    logits = alpha * (top_vals - c)
    logits = np.clip(logits, -500, 0)
    w = np.exp(logits)
    ws = np.sum(w)
    if ws < 1e-300:
        w = np.zeros(top_k)
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= ws
    
    # Also compute the overall smooth max value
    all_logits = alpha * (overlaps - c)
    all_logits = np.clip(all_logits, -500, 0)
    all_w = np.exp(all_logits)
    all_ws = np.sum(all_w)
    smooth_val = c + np.log(all_ws) / alpha if all_ws > 1e-300 else c

            if L > 0:

                grad[:L] += wi * dx * (-h[k2:])

            if stuck > 500:

                m = np.zeros(n)
                v = np.zeros(n)
                stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        overlaps = compute_overlaps(h)
        grad, smooth_val = compute_grad_smooth(h, overlaps, n, alpha, top_k)
        
        norm_grad = np.linalg.norm(grad)

def build_lp_constraints(h, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    n_half = n - 1
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def alternating_lp_refined(h_init, n, target, max_iter, time_limit,
                           n_top=100, damping=0.9, reg=1e-4, solver='SCS'):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 500 == 0 and stuck > 200:
            break
        
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var, t_var, constraints = build_lp_constraints(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            if solver == 'SCS':
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
            else:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter=2000):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    
    def smooth_obj(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr) * dx_global
        shifted = np.clip(alpha * (corr * dx_global - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    dx_global = 2.0 / n
    
    def smooth_obj_wrapper(h):
        return smooth_obj(h, _global_alpha)
    
    def cons_fun(h):
        return np.sum(h) - _global_target
    
    def cons_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = project_feasible(h_start.copy(), target)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj_wrapper, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def sinusoidal_perturbation(h, n, target, rng, n_modes=5, amplitude=0.1):
    """Add sinusoidal perturbations to escape local minima."""
    h_new = h.copy()
    x = np.arange(n)
    for _ in range(n_modes):
        freq = rng.randint(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, amplitude)
        h_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, target)

def random_perturbation(h, n, target, rng, noise_level=0.05):
    h_new = h + rng.randn(n) * noise_level
    return project_feasible(h_new, target)

For a "random" set A of measure 1 in [0,2], the expected |A ∩ (A-k)| ≈ 1/2 (since both sets have measure 1 and the space has measure 2, the expected overlap is 1*1/2 = 1/2). So the expected overlap is 1 - 1/2 = 1/2.

But we can do better than random. A structured set (like a union of carefully placed intervals) can have more uniform autocorrelation.

For the continuous-valued h, we can do even better by spreading the "mass" more smoothly.

I think the key insight is that the optimal h should be "as flat as possible" where the overlap is high, and "as peaked as possible" where the overlap is low. This is exactly what the optimization is trying to find.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 1.0
_global_dx = 1.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_grad_smooth(h, overlaps, n, alpha, top_k=100):
    dx = 2.0 / n
    n_full = len(overlaps)
    n_half = n - 1
    c = np.max(overlaps)
    logits = alpha * (overlaps - c)
    logits = np.clip(logits, -500, 0)
    exp_logits = np.exp(logits)
    total = np.sum(exp_logits)
    if total < 1e-300:
        smooth_val = c
        weights = np.zeros(n_full)
        weights[np.argmax(overlaps)] = 1.0
    else:
        smooth_val = c + np.log(total) / alpha
        weights = exp_logits / total
    threshold = 1e-12
    significant = np.where(weights > threshold)[0]
    if len(significant) > top_k:
        top_vals = weights[significant]
        top_local = np.argpartition(top_vals, -top_k)[-top_k:]
        significant = significant[top_local]
    grad = np.zeros(n)
    for idx in significant:
        w = weights[idx]
        k = idx - n_half
        if k >= 0:
            L = n - k
            if L > 0:
                grad[:L] += w * dx * (1.0 - h[k:])
                grad[k:] += w * dx * (-h[:L])
        else:
            k2 = -k
            L = n - k2
            if L > 0:
                grad[k2:] += w * dx * (1.0 - h[:L])
                grad[:L] += w * dx * (-h[k2:])
    return grad, smooth_val


def adam_smooth_max(h_init, n, target, time_limit,
                    alpha_start=20, alpha_end=500,
                    top_k=100, lr=0.02, beta1=0.9, beta2=0.999,
                    eps=1e-8, max_iter=200000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        t += 1
        progress = min(1.0, it / max(1, int(max_iter * 0.8)))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        overlaps = compute_overlaps(h)
        grad, smooth_val = compute_grad_smooth(h, overlaps, n, alpha, top_k)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 800:
                m = np.zeros(n)
                v = np.zeros(n)
                stuck = 0
                h = h_new
                continue
        h = h_new
    return best_h, best_c5


def momentum_smooth_max(h_init, n, target, time_limit,
                        alpha_start=50, alpha_end=1000,
                        top_k=80, lr=0.01, momentum=0.9, max_iter=100000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_prev = h.copy()
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, it / max(1, int(max_iter * 0.8)))
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        overlaps = compute_overlaps(h)
        grad, smooth_val = compute_grad_smooth(h, overlaps, n, alpha, top_k)
        h_new = h - lr * grad + momentum * (h - h_prev)
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            h_prev = h.copy()
            h = h_new
            stuck = 0
        else:
            h_prev = h.copy()
            stuck += 1
            if stuck > 1000:
                lr = lr * 0.5
                stuck = 0
    return best_h, best_c5


def polyak_smooth_max(h_init, n, target, time_limit,
                      alpha=500, top_k=50, max_iter=50000):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        overlaps = compute_overlaps(h)
        grad, smooth_val = compute_grad_smooth(h, overlaps, n, alpha, top_k)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        gap = smooth_val - best_c5
        if gap < 0:
            gap = 0.0
        lr = gap / (norm_grad ** 2 + 1e-20)
        lr = min(lr, 0.3)
        lr = max(lr, 1e-8)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c5


def build_lp_constraints(h_ref, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def alternating_lp_refined(h_init, n, target, max_iter, time_limit,
                           n_top=100, damping=0.9, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 300 == 0 and stuck > 150:
            break
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        h_var, t_var, constraints = build_lp_constraints(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def slsqp_smooth_obj(h):
    n = _global_n
    dx = _global_dx
    alpha = _global_alpha
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def slsqp_cons_fun(h):
    return np.sum(h) - _global_target


def slsqp_cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000):
    global _global_n, _global_target, _global_alpha, _global_dx
    _global_n = n
    _global_target = target
    _global_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        if time.time() > 0:
            pass
        try:
            res = minimize(slsqp_smooth_obj, h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                cur_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def sinusoidal_perturbation(h, n, target, rng, n_modes, amplitude):
    h_new = h.copy()
    x = np.arange(n)
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 3))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, amplitude)
        h_new += amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return project_feasible(h_new, target)


def random_perturbation(h, n, target, rng, noise_level):
    h_new = h + rng.randn(n) * noise_level
    return project_feasible(h_new, target)


def binary_flip_perturbation(h, n, target, rng, flip_frac):
    h_new = h.copy()
    mask = rng.random(n) < flip_frac
    h_new[mask] = 1.0 - h_new[mask]
    return project_feasible(h_new, target)


def block_perturbation(h, n, target, rng):
    h_new = h.copy()
    n_blocks = rng.randint(2, 6)
    block_size = n // n_blocks
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - block_size))
        end = min(n, start + block_size)
        shift = rng.uniform(-0.3, 0.3)
        h_new[start:end] += shift
    return project_feasible(h_new, target)
