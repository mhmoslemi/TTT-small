def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    # Project h onto {0 <= h <= 1, sum(h) = target}
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    if abs(h.sum() - target) < 1e-12:
        return h
    # Use bisection on the threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def coord_descent_polish(h, target, deadline, max_sweeps=500):

    rng = np.random.RandomState(42)

            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            # Binary search for the best value of h[idx]
            lo, hi = 0.0, 1.0
            best_local = old_val
            best_local_score = cur_score
            for _ in range(8):
                mid = (lo + hi) / 2.0
                h_cur[idx] = mid
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = mid
                    hi = mid
                else:
                    lo = mid
            # Also try endpoints
            for trial in [0.0, 1.0, old_val]:
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = trial
            h_cur[idx] = best_local
            h_cur = project_feasible(h_cur, target)
            cur_score = max(compute_c5(h_cur), best_local_score)
    
    return h_cur, cur_score

def structured_restart(h, n, target, mode, seed):
    """Create structured perturbations for restarts"""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    if mode == 0:  # Gaussian noise
        return project_feasible(h + rng.randn(n) * 0.05, target)
    elif mode == 1:  # Sine perturbation
        freq = rng.randint(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        return project_feasible(h + 0.03 * np.sin(2*np.pi*freq*x/2.0 + phase), target)
    elif mode == 2:  # Two-bump structure
        c1 = rng.uniform(0.2, 0.8)
        c2 = rng.uniform(0.5, 1.5)
        bump = np.exp(-((x-c1)/0.2)**2) + np.exp(-((x-c2)/0.2)**2)
        bump = bump / bump.sum() * n / 2.0
        return project_feasible(0.7 * h + 0.3 * bump, target)
    elif mode == 3:  # Smoother version
        from scipy.ndimage import gaussian_filter1d
        return project_feasible(gaussian_filter1d(h, sigma=3), target)
    else:
        return project_feasible(h + rng.randn(n) * 0.03, target)

from scipy.ndimage import gaussian_filter1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    if abs(h.sum() - target) < 1e-14:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def penalty_obj_and_grad(h, C_target):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    val = corr * dx
    violations = np.maximum(0.0, val - C_target)
    obj = 0.5 * float(np.sum(violations**2))
    mask = violations > 0.0
    if not np.any(mask):
        return obj, np.zeros(n)
    w = violations[mask]
    w_full = np.zeros(n)
    w_full[mask] = w
    w_flipped = w_full[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n

    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

    x_prev = x.copy()
    y = x.copy()

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.5
        for _ in range(25):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            dist2 = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= dist2 * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), 3.0 / L)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def fista_penalty(h0, target, C_target, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    _, g0 = penalty_obj_and_grad(y, C_target)
    L = max(np.linalg.norm(g0), 1e-8)
    step = 1.0 / max(L * 2.0, 1e-6)
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = penalty_obj_and_grad(y_clipped, C_target)
        if obj_y < 1e-15:
            break
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.5
        for _ in range(20):
            obj_new, _ = penalty_obj_and_grad(x_new, C_target)
            decrease = obj_y - obj_new
            dist2 = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= dist2 * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), 4.0 / L)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
    return best_x, best_score

def active_set_polish(h, target, deadline, max_rounds=2000):
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h, target).copy()
    cur_score = compute_c5(h_cur)
    best_x = h_cur.copy()
    best_score = cur_score
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = float(np.max(corr))
        tol = max(0.02 * c_max, 1e-6)
        active = np.where(corr >= c_max - tol)[0]
        if len(active) > 15:
            active = np.argsort(corr)[::-1][:10]
        w = np.zeros(n)
        w[active] = 1.0 / len(active)
        w_flipped = w[::-1].copy()
        v = 1.0 - h_cur
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(v, fsize)
        W = np.fft.rfft(w, fsize)
        H = np.fft.rfft(h_cur, fsize)
        pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        neg = np.fft.irfft(W * H, fsize)[:n]
        grad = dx * (pos - neg)
        found = False
        for eps in [0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                if sc < best_score:
                    best_score = sc
                    best_x = h_cur.copy()
                found = True
                break
        if not found:
            top3 = np.argsort(corr)[::-1][:3]
            w2 = np.zeros(n)
            w2[top3] = 1.0 / 3.0
            w2_flipped = w2[::-1].copy()
            Wf2 = np.fft.rfft(w2_flipped, fsize)
            W2 = np.fft.rfft(w2, fsize)
            pos2 = np.fft.irfft(Wf2 * V, fsize)[n-1:2*n-1]
            neg2 = np.fft.irfft(W2 * H, fsize)[:n]
            grad2 = dx * (pos2 - neg2)
            for eps in [0.002, 0.001, 0.0005, 0.0002]:
                h_new = project_feasible(h_cur - eps * grad2, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
                    if sc < best_score:
                        best_score = sc
                        best_x = h_cur.copy()
                    break
    return best_x, best_score

def bisection_on_C(h_init, target, C_achievable, deadline, num_iters=15):
    C_low = C_achievable * 0.95
    C_high = C_achievable
    best_x = h_init.copy()
    best_score = C_achievable
    for it in range(num_iters):
        if time.time() > deadline:
            break
        C_mid = (C_low + C_high) / 2.0
        inner_deadline = min(deadline, time.time() + 15.0)
        h_try, sc_try = fista_penalty(best_x, target, C_mid, 10000, inner_deadline)
        max_corr = compute_c5(h_try)
        if max_corr <= C_mid + 1e-7:
            C_high = C_mid
            if sc_try < best_score:
                best_score = sc_try
                best_x = h_try
        else:
            C_low = C_mid
            if sc_try < best_score:
                best_score = sc_try
                best_x = h_try
    return best_x, best_score

def coord_descent_polish(h, target, deadline, max_sweeps=200):
    n = len(h)
    h_cur = project_feasible(h, target).copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            best_local = old_val
            best_local_score = cur_score
            for _ in range(6):
                lo, hi = 0.0, 1.0
                mid = (lo + hi) / 2.0
                h_cur[idx] = mid
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = mid
                    hi = mid
                else:
                    lo = mid
            for trial in [0.0, 1.0]:
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = trial
            h_cur[idx] = best_local
            h_cur = project_feasible(h_cur, target)
            cur_score = compute_c5(h_cur)
    return h_cur, cur_score

def make_structured_perturbation(h, n, target, mode, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    if mode == 0:
        return project_feasible(h + rng.randn(n) * 0.04, target)
    elif mode == 1:
        freq = rng.randint(1, 6)
        phase = rng.uniform(0, 2 * np.pi)
        pert = 0.02 * np.sin(2 * np.pi * freq * x / 2.0 + phase)
        return project_feasible(h + pert, target)
    elif mode == 2:
        c1 = rng.uniform(0.3, 0.9)
        c2 = rng.uniform(1.0, 1.7)
        bump = np.exp(-((x - c1) / 0.25)**2) + np.exp(-((x - c2) / 0.25)**2)
        bump = bump / bump.sum() * (n / 2.0)
        return project_feasible(0.6 * h + 0.4 * bump, target)
    elif mode == 3:
        smoothed = gaussian_filter1d(h, sigma=5)
        return project_feasible(0.8 * h + 0.2 * smoothed, target)
    elif mode == 4:
        x2 = np.linspace(0, 1, n)
        sq = np.minimum(x2, 1.0 - x2) * 4.0
        return project_feasible(h + 0.03 * (sq - 0.5), target)
    else:
        return project_feasible(h + rng.randn(n) * 0.06, target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 8.0
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline - time.time()

    # Phase 1: FISTA continuation (aim ~250s)
    alphas_p1 = [30, 100, 300, 800, 2000, 5000]
    alloc_p1 = [40, 35, 35, 40, 40, 40]
    for i, (a, td) in enumerate(zip(alphas_p1, alloc_p1)):
        if tleft() < td + 5:
            break
        h_new, sc_new = fista_optimize(best_h, target, a, 200000, start_time + td)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: Active set polish (aim ~60s)
    if tleft() > 60:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 55, max_rounds=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: High-alpha FISTA + penalty bisection (aim ~200s)
    if tleft() > 200:
        for a in [10000, 30000, 80000]:
            if tleft() < 60:
                break
            h_new, sc_new = fista_optimize(best_h, target, a, 200000, start_time + 35)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if tleft() > 60:
                h_new, sc_new = active_set_polish(best_h, target, start_time + 20, max_rounds=1000)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new
        if tleft() > 100:
            h_new, sc_new = bisection_on_C(best_h, target, best_score, start_time + 95, num_iters=12)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 4: More bisection iterations (aim ~120s)
    if tleft() > 120:
        h_new, sc_new = bisection_on_C(best_h, target, best_score, start_time + 110, num_iters=10)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if tleft() > 40:
            h_new, sc_new = active_set_polish(best_h, target, start_time + 35, max_rounds=2000)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 5: Structured restarts (aim ~180s)
    if tleft() > 180:
        for mode in range(6):
            if tleft() < 40:
                break
            h_restart = make_structured_perturbation(best_h, n, target, mode, seed + mode * 73)
            h_new, sc_new = fista_optimize(h_restart, target, 3000, 100000, start_time + 22)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            if tleft() > 20:
                h_new, sc_new = active_set_polish(h_new, target, start_time + 15, max_rounds=1000)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new

    # Phase 6: Coarse-to-fine restart (aim ~100s)
    if tleft() > 100:
        n_coarse = n // 4
        target_coarse = n_coarse / 2.0
        x_coarse = np.linspace(0, 2, n_coarse)
        x_fine = np.linspace(0, 2, n)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, target_coarse)
        for a in [500, 5000]:
            if tleft() < 30:
                break
            h_coarse, _ = fista_optimize(h_coarse, target_coarse, a, 100000, start_time + 25)
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, target)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up
            if tleft() > 30:
                h_new, sc_new = fista_optimize(best_h, target, 10000, 100000, start_time + 25)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new
                if tleft() > 20:
                    h_new, sc_new = active_set_polish(best_h, target, start_time + 15, max_rounds=500)
                    if sc_new < best_score:
                        best_score = sc_new
                        best_h = h_new

    # Phase 7: Final bisection push (aim ~60s)
    if tleft() > 60:
        h_new, sc_new = bisection_on_C(best_h, target, best_score, start_time + 50, num_iters=6)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 8: Final active set polish + coordinate descent
    if tleft() > 30:
        h_new, sc_new = active_set_polish(best_h, target, start_time + 20, max_rounds=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    if tleft() > 15:
        cd_deadline = start_time + max(tleft() - 5, 5)
        h_new, sc_new = coord_descent_polish(best_h, target, cd_deadline, max_sweeps=100)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
