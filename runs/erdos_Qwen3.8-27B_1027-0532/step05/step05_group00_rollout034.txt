import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def smooth_max(corr, alpha):
    """Compute smooth max of corr using log-sum-exp."""
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(corr, alpha):

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h >= 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        r = rng.random()
        h_new = h.copy()
        moved = False
        if r < 0.5:
            worst = np.argsort(corr)[-8:]
            k = int(worst[rng.randint(len(worst))]) - (n - 1)
            ip = [i for i in ones if 0 <= i + k < n]
            jp = [j for j in zeros if 0 <= j + k < n]
            if len(ip) > 0 and len(jp) > 0:
                h_new[ip[rng.randint(len(ip))]] = 0.0
                h_new[jp[rng.randint(len(jp))]] = 1.0
                moved = True
        elif r < 0.85:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_new[i] = 0.0
            h_new[j] = 1.0
            moved = True
        else:
            if len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.choice(len(ones), 2, replace=False)
                j1, j2 = rng.choice(len(zeros), 2, replace=False)
                h_new[ones[i1]] = 0.0
                h_new[ones[i2]] = 0.0
                h_new[zeros[j1]] = 1.0
                h_new[zeros[j2]] = 1.0
                moved = True
        if not moved:
            continue
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        c5_new = float(corr_new.max()) * (2.0 / n)
        delta = c5_new - current_c5
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / max(t, 1e-30))):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5


def _smooth_slsqp_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return float(np.sum(h)) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_slsqp_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 25.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = float(n) / 2.0

    best_h = project_box_sum(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy on binary projection
    h_bin = project_binary(best_h, int(target))
    if tleft() > 150:
        h_g, c5_g = greedy_binary(h_bin, int(target), max_iters=20000,
                                  time_limit=min(80, tleft() * 0.12),
                                  start_time=start_time, scan_width=25)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Adam continuous optimization (main phase)
    if tleft() > 200:
        adam_time = min(600, tleft() - 200)
        h_a, s_a = adam_continuous(
            best_h, target, start_time, adam_time,
            alphas=[30, 60, 120, 250, 500, 1000, 2500, 6000, 15000, 40000],
            lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_a < best_score:
            best_score = s_a
            best_h = h_a.copy()

    # Phase 3: Multi-restart Adam from perturbed points
    if tleft() > 250:
        for restart_i in range(4):
            if tleft() < 100:
                break
            jitter_scale = 0.005 + 0.003 * restart_i
            h_pert = best_h + rng.randn(n) * jitter_scale
            h_pert = project_box_sum(h_pert, target)
            rt = min(150, (tleft() - 100) / (4 - restart_i))
            h_ar, s_ar = adam_continuous(
                h_pert, target, start_time, rt,
                alphas=[100, 500, 2000, 8000, 30000],
                lr=0.002, beta1=0.9, beta2=0.999, eps=1e-8
            )
            if s_ar < best_score:
                best_score = s_ar
                best_h = h_ar.copy()

    # Phase 4: Binary SA from best solution's projection
    if tleft() > 120:
        h_bin2 = project_binary(best_h, int(target))
        sa_time = min(200, tleft() - 80)
        max_it = int(min(200000, sa_time * 500))
        h_sa, c5_sa = sa_binary(h_bin2, int(target), max_it,
                                0.008, 0.00001, rng, sa_time, start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 5: SLSQP polish
    if tleft() > 60:
        slsqp_time = min(200, tleft() - 30)
        alphas_sl = [2000, 5000, 10000, 30000, 80000]
        h_ref, sc_ref = slsqp_refine(best_h, n, target, alphas_sl,
                                     3000, 1e-16, slsqp_time, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 6: Final short Adam polish
    if tleft() > 30:
        polish_time = max(10.0, tleft() - 10)
        h_p, s_p = adam_continuous(
            best_h, target, start_time, polish_time,
            alphas=[20000, 50000, 100000],
            lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8
        )
        if s_p < best_score:
            best_score = s_p
            best_h = h_p.copy()

    # Final feasibility check
    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)

    if final_score >= best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n
