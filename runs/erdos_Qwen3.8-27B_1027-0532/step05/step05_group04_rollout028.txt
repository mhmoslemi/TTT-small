        else:
            momentum = (k - 1) / float(k + 1)
            y = h + momentum * (h - x_prev)
            y = np.clip(y, 0.0, 1.0)
        grad = lse_gradient(y, alpha)
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            best_score = sc_new
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        obj_new = lse_value(h_new, alpha)
        obj_curr = lse_value(h, alpha)
        if obj_new > obj_curr + 1e-12:
            step *= 0.8
            h_new = h - step * grad
            h_new = project_feasible(h_new, target)
        else:
            step = min(step * 1.02, 5.0)
        x_prev = h
        h = h_new
        if stalled > 500:
            step *= 0.7
            stalled = 0
    return best_h,
