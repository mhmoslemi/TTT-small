        mask = (idx >= 0) & (idx < n)
        delta[mask] -= h[idx[mask]]
        idx = j + k_arr
        mask = (idx >= 0) & (idx < n)
        delta[mask] -= h[idx[mask]]
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    target_float = float(n_init) / 2.0

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick binary local search
    allowed = min(60, tleft() * 0.10)
    h_g, c5_g = fast_greedy(best_h, max_iters=30000, rng=rng,
                            time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Continuous gradient descent (main optimization)
    allowed = min(400, tleft() * 0.35)
    alphas_round1 = [5, 10, 20, 50, 100, 200, 500, 1000]
    h_cont, sc_cont = grad_descent(best_h, alphas_round1, max_iters=200,
                                   time_limit=allowed, start_time=start_time, momentum=0.3)
    if sc_cont < best_score:
        best_score = sc_cont
        best_h = h_cont.copy()

    # Phase 3: Binary projection + local search
    h_bin = project_binary(best_h, target_init)
    allowed = min(50, tleft() * 0.08)
    h_bg, c5_bg = fast_greedy(h_bin, max_iters=20000, rng=rng,
                              time_limit=allowed * 0.5, start_time=start_time)
    if tleft() > 20:
        n_sa = int(min(500000, max(10000, (tleft() - 10) * 20000)))
        allowed2 = min(30, tleft() * 0.08)
        h_bs, c5_bs = sa_focused(h_bg, n_sa, 0.01, 0.0001, rng,
                                 time_limit=allowed2, start_time=start_time)
        if c5_bs < c5_bg:
            h_bg = h_bs
            c5_bg = c5_bs
    if c5_bg < best_score:
        best_score = c5_bg
        best_h = h_bg.copy()

    # Phase 4: Second round of continuous GD from improved binary
    if tleft() > 100:
        allowed = min(300, tleft() * 0.30)
        alphas_round2 = [20, 50, 100, 200, 500, 1000, 2000, 5000]
        h_cont2, sc_cont2 = grad_descent(best_h, alphas_round2, max_iters=300,
                                         time_limit=allowed, start_time=start_time, momentum=0.4)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2.copy()

    # Phase 5: Binary + local search again
    if tleft() > 60:
        h_bin2 = project_binary(best_h, target_init)
        allowed = min(40, tleft() * 0.12)
        h_bg2, c5_bg2 = fast_greedy(h_bin2, max_iters=15000, rng=rng,
                                    time_limit=allowed * 0.5, start_time=start_time)
        if tleft() > 20:
            n_sa2 = int(min(300000, max(5000, (tleft() - 10) * 15000)))
            allowed2 = min(25, tleft() * 0.10)
            h_bs2, c5_bs2 = sa_focused(h_bg2, n_sa2, 0.008, 0.00005, rng,
                                       time_limit=allowed2, start_time=start_time)
            if c5_bs2 < c5_bg2:
                h_bg2 = h_bs2
                c5_bg2 = c5_bs2
        if c5_bg2 < best_score:
            best_score = c5_bg2
            best_h = h_bg2.copy()

    # Phase 6: Third round continuous GD with higher alphas
    if tleft() > 80:
        allowed = min(200, tleft() * 0.25)
        alphas_round3 = [100, 300, 1000, 3000, 10000]
        h_cont3, sc_cont3 = grad_descent(best_h, alphas_round3, max_iters=500,
                                         time_limit=allowed, start_time=start_time, momentum=0.5)
        if sc_cont3 < best_score:
            best_score = sc_cont3
            best_h = h_cont3.copy()

    # Phase 7: Multi-start continuous GD
    if tleft() > 100:
        for trial in range(3):
            if tleft() < 30:
                break
            h_pert = best_h.copy()
            noise_scale = 0.1 / (trial + 1)
            h_pert += noise_scale * rng.randn(n_init)
            h_pert = project_polytope(h_pert, target_float)
            allowed = min(60, tleft() * 0.20)
            alphas_ms = [10, 30, 100, 300, 1000]
            h_ms, sc_ms = grad_descent(h_pert, alphas_ms, max_iters=200,
                                       time_limit=allowed, start_time=start_time, momentum=0.3)
            if sc_ms < best_score:
                best_score = sc_ms
                best_h = h_ms.copy()

    # Phase 8: Final binary polish
    if tleft() > 30:
        h_bin_final = project_binary(best_h, target_init)
        n_sa_f = int(min(500000, max(10000, (tleft() - 10) * 20000)))
        h_sa_f, c5_f = sa_focused(h_bin_final, n_sa_f, 0.005, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 9: Final continuous refinement
    if tleft() > 20:
        alphas_final = [500, 2000, 10000]
        h_ref, sc_ref = grad_descent(best_h, alphas_final, max_iters=300,
                                     time_limit=max(5.0, tleft() - 5), start_time=start_time,
                                     momentum=0.5)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_polytope(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
