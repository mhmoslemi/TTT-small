def replica_sa(h, n_chains, n_iters_per_chain, t_values, rng, time_limit, start_time):
    """Replica exchange (parallel tempering) SA."""
    n = len(h)
    target = int(np.sum(h))
    
    # Initialize chains
    chains = [h.copy() for _ in range(n_chains)]
    corrs = [full_corr_fast(chains[c]) for c in range(n_chains)]
    dx = 2.0 / n
    current_c5 = [float(corrs[c].max()) * dx for c in range(n_chains)]
    best_c5 = min(current_c5)
    best_h = chains[np.argmin(current_c5)].copy()
    
    positions = np.arange(n)
    
    for it in range(n_iters_per_chain * n_chains):
        if time.time() - start_time > time_limit:
            break
        
        # Pick a random chain to update
        c = rng.randint(n_chains)
        h_c = chains[c]
        corr_c = corrs[c]
        t = t_values[c]
        
        ones = np.where(h_c == 1.0)[0]
        zeros = np.where(h_c == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Propose a move
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = compute_delta(h_c, i, j)
        new_corr = corr_c + delta
        new_c5 = float(new_corr.max()) * dx
        delta_c5 = new_c5 - current_c5[c]
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-abs(delta_c5) / t)):
            h_c[i] = 0.0
            h_c[j] = 1.0
            corr_c[:] = new_corr
            current_c5[c] = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_c.copy()
        
        # Periodically try to swap adjacent temperature chains
        if it % 10 == 9:
            # Try to swap chain c with c+1 or c-1
            c_other = (c + 1) % n_chains
            if current_c5[c] < current_c5[c_other]:
                # Swap if the hotter chain has a better solution
                chains[c], chains[c_other] = chains[c_other], chains[c]
                corrs[c], corrs[c_other] = corrs[c_other], corrs[c]
                current_c5[c], current_c5[c_other] = current_c5[c_other], current_c5[c]
            else:
                # Accept swap with probability exp((T_hot - T_cold) * (F_hot - F_cold))
                # Simplified: accept with probability exp(-delta_c5 / T_avg)
                t_avg = (t_values[c] + t_values[c_other]) / 2
                delta = current_c5[c_other] - current_c5[c]
                if rng.random() < np.exp(-max(0, delta) / max(t_avg, 1e-10)):
                    chains[c], chains[c_other] = chains[c_other], chains[c]
                    corrs[c], corrs[c_other] = corrs[c_other], corrs[c]
                    current_c5[c], current_c5[c_other] = current_c5[c_other], current_c5[c]
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
