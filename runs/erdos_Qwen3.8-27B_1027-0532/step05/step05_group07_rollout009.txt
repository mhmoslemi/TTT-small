def project_feasible(h, target_sum):

        else:

        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n-1:n-1+n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n-1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n-1] = b_neg
    return B

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def subgradient_polish(h, n, target, deadline, rng, max_iters=5000):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    lr = 0.5
    decay = 0.9995

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        tol = 0.05 * c_max
        active = corr >= c_max - tol
        active_idx = np.where(active)[0]
        if len(active_idx) == 0:
            break

        grad = np.zeros(n)
        for k in active_idx[:30]:
            s = int(k) - (n - 1)
            for i in range(n):
                contrib = 0.0
                if 0 <= i + s < n:
                    contrib += (1.0 - h_cur[i + s])
                if 0 <= i - s < n:
                    contrib -= h_cur[i - s]
                grad[i] += contrib
        grad /= max(1, len(active_idx))

        norm_g = np.linalg.norm(grad)
        if norm_g < 1e-12:
            break

        h_new = h_cur - lr * grad / norm_g
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
        lr *= decay

    return h_cur, compute_c5(h_cur)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n_src = len(initial_h_values)
    target_src = n_src / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target_src)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick CD at original resolution
    deadline_p1 = start_time + (budget_s - 30.0) * 0.12
    h_cd, sc_cd = paired_coord_desc(best_h, n_src, target_src, deadline_p1, rng,
                                     recompute_interval=30, bs_iters=30)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: Upsample to n=800
    n_hi = 2 * n_src
    target_hi = n_hi / 2.0
    h_hi = interpolate_to(best_h, n_hi, target_hi)

    # Phase 3: Main CD at high resolution
    deadline_p3 = start_time + (budget_s - 30.0) * 0.50
    h_hi, sc_hi = paired_coord_desc(h_hi, n_hi, target_hi, deadline_p3, rng,
                                     recompute_interval=50, bs_iters=30)
    if sc_hi < best_score:
        best_score = sc_hi
        best_h = h_hi

    # Phase 4: SLSQP polish at high resolution
    if tleft() > 150:
        h_ref, sc_ref = slsqp_polish(best_h, n_hi, target_hi,
                                      [100, 400, 1500, 5000], 1500, 1e-15)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 5: Subgradient polish
    if tleft() > 120:
        deadline_sg = start_time + (budget_s - 30.0) * 0.68
        h_sg, sc_sg = subgradient_polish(best_h, n_hi, target_hi, deadline_sg, rng)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 6: More CD after subgradient
    if tleft() > 200:
        deadline_p6 = start_time + (budget_s - 30.0) * 0.78
        h_cd, sc_cd = paired_coord_desc(best_h, n_hi, target_hi, deadline_p6, rng,
                                         recompute_interval=30, bs_iters=30)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 7: Random restarts
    deadline_p7 = start_time + (budget_s - 30.0) * 0.90
    for attempt in range(5):
        if tleft() < 80:
            break
        noise_scale = [0.04, 0.07, 0.03, 0.09, 0.05][attempt]
        h_pert = best_h + rng.randn(n_hi) * noise_scale
        h_pert = project_feasible(h_pert, target_hi)
        h_res, sc_res = paired_coord_desc(h_pert, n_hi, target_hi, deadline_p7, rng,
                                          recompute_interval=25, bs_iters=28)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    # Phase 8: Final SLSQP
    if tleft() > 60:
        h_final, sc_final = slsqp_polish(best_h, n_hi, target_hi,
                                          [500, 2000, 8000, 20000], 1500, 1e-16)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 9: Final CD
    if tleft() > 20:
        h_final, sc_final = paired_coord_desc(best_h, n_hi, target_hi,
                                              start_time + budget_s - 20.0, rng,
                                              recompute_interval=20, bs_iters=28)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target_hi)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
