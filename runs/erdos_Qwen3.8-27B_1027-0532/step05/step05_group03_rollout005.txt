def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # The relevant lags are indices n-1 through n-1+n (lags 0 through n-1)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

            best_h = h_cur.copy()
        
        # Update lower bound
        lb = max(lb, c_curr - (c_curr - lb) * 0.1)

def block_coordinate(h, n_blocks, target, max_inner_iter=200):
    """Block coordinate descent: optimize each block of variables"""
    global _g_n, _g_target
    n = len(h)
    _g_n = n
    _g_target = target
    block_size = n // n_blocks
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for it in range(max_inner_iter):
        improved = False
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n
            h_fixed = h_cur.copy()
            
            # Optimize block[start:end] while keeping rest fixed
            # The objective is max_k sum_i h[i](1-h[i+k])*dx
            # For a fixed block, this is still non-linear in the block variables
            # So we just use a subgradient step on the block
            over = get_over(h_cur)
            top_idx = np.argsort(over)[::-1][:3]
            
            grad_block = np.zeros(end - start)
            for k in top_idx:
                k = int(k)
                dx_val = 2.0 / n
                if k == 0:
                    g = dx_val * (1.0 - 2.0 * h_cur[start:end])
                    grad_block += g
                else:
                    full_grad = np.zeros(n)
                    upper = n - 1 - k
                    idx1 = np.arange(0, upper + 1)
                    full_grad[idx1] += dx_val * (1.0 - h_cur[idx1 + k])
                    idx2 = np.arange(k, n)
                    full_grad[idx2] -= dx_val * h_cur[idx2 - k]
                    grad_block += full_grad[start:end]
            
            # Update block
            eps = 0.005
            for sub in range(20):
                new_block = h_cur[start:end] - eps * grad_block
                new_block = np.clip(new_block, 0.0, 1.0)
                h_trial = h_fixed.copy()
                h_trial[start:end] = new_block
                h_trial = project_feasible(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < cur_score - 1e-13:
                    h_cur = h_trial
                    cur_score = sc
                    improved = True
                else:
                    eps *= 0.7
                    break
        
        if not improved:
            break
    
    return h_cur, cur_score

    for step in range(n_steps):
        over = get_over(h_cur)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score - 1e-13:
            best_score = c_curr
            best_h = h_cur.copy()

        top_idx = np.argsort(over)[::-1][:n_active]
        top_vals = over[top_idx]
        mean_top = np.mean(top_vals)

        # Gradient: push top lags toward their mean (balancing) + push mean down
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            weight = 1.0 / n_active
            if k == 0:
                grad[:] += weight * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += weight * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= weight * dx * h_cur[idx2 - k]

        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-14:
            h_cur = h_new
            lr = min(lr * 1.002, 0.005)
        else:
            lr *= 0.95
            if lr < 1e-10:
                break
    return best_h, best_score

def structured_starts(n, target, rng):
    x = np.linspace(0.0, 2.0, n)
    starts = []

    for m in [2, 3, 4, 5, 6, 8, 10, 12]:
        for phase in [0.0, 0.3, 0.5, 0.7]:
            h_s = 0.5 + 0.5 * np.sin(np.pi * m * x / 2.0 + phase * 2 * np.pi)
            h_s = project_feasible(h_s, target)
            starts.append(h_s)

    for n_blocks in [4, 6, 8, 10, 14, 20]:
        block_len = max(1, n // n_blocks)
        h_b = np.zeros(n)
        for i in range(0, n, 2):
            start_i = i * block_len
            end_i = min(start_i + block_len, n)
            h_b[start_i:end_i] = 1.0
        h_b = project_feasible(h_b, target)
        h_b += rng.randn(n) * 0.05
        h_b = project_feasible(h_b, target)
        starts.append(h_b)

    for period in [0.2, 0.25, 0.33, 0.4, 0.5, 0.66, 1.0]:
        h_t = np.zeros(n)
        for i in range(n):
            frac = (x[i] / period) % 1.0
            h_t[i] = 1.0 if frac < 0.5 else 0.0
        h_t = project_feasible(h_t, target)
        h_t += rng.randn(n) * 0.08
        h_t = project_feasible(h_t, target)
        starts.append(h_t)

    n_bumps = rng.randint(3, 10)
    centers = np.sort(rng.uniform(0.1, 1.9, n_bumps))
    widths = rng.uniform(0.08, 0.35, n_bumps)
    h_g = np.zeros(n)
    for c, w in zip(centers, widths):
        h_g += (1.0 / (w * np.sqrt(2.0 * np.pi))) * np.exp(-0.5 * ((x - c) / w) ** 2)
    h_g = h_g / (h_g.sum() + 1e-10) * n / 2.0
    h_g = project_feasible(h_g, target)
    starts.append(h_g)

    return starts

def try_start(h_s, budget_deadline, max_slsqp_iter=800):
    sc = compute_c5(h_s)
    if sc > 0.42:
        return h_s, sc
    h_opt, sc_opt = slsqp_run(h_s, 'active_sum', n_active_list=[3, 5], maxiter=max_slsqp_iter)
    h_opt, sc_opt = slsqp_run(h_opt, 'softmax', alpha_list=[200, 2000], maxiter=max_slsqp_iter, ftol=1e-15)
    h_opt, sc_opt2 = lag_surgery(h_opt, max_steps=150, initial_eps=0.001, n_lags=2)
    if sc_opt2 < sc_opt:
        sc_opt = sc_opt2
    return h_opt, sc_opt

def rleft(start_time, budget_s):
    return max(0.0, budget_s - 25.0 - (time.time() - start_time))

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_n_active
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Active-set SLSQP (sum formulation)
    if rleft(start_time, budget_s) > 500:
        h_r, sc = slsqp_run(best_h, 'active_sum', n_active_list=[3, 5, 8], maxiter=2000, ftol=1e-14)
        if sc < best_score:
            best_score, best_h = sc, h_r

    # Phase 2: Softmax SLSQP with increasing alpha
    if rleft(start_time, budget_s) > 450:
        h_r, sc = slsqp_run(best_h, 'softmax', alpha_list=[100, 500, 2000], maxiter=1500, ftol=1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_r

    # Phase 3: Lag surgery (top 2 and top 3)
    if rleft(start_time, budget_s) > 400:
        h_s, sc = lag_surgery(best_h, max_steps=300, initial_eps=0.0012, n_lags=2)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_s, sc = lag_surgery(best_h, max_steps=250, initial_eps=0.0008, n_lags=3)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_s, sc = lag_surgery(best_h, max_steps=200, initial_eps=0.001, n_lags=1)
        if sc < best_score:
            best_score, best_h = sc, h_s

    # Phase 4: SLSQP after surgery
    if rleft(start_time, budget_s) > 350:
        h_r, sc = slsqp_run(best_h, 'active_sum', n_active_list=[3, 5], maxiter=1500, ftol=1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_r
        h_r, sc = slsqp_run(best_h, 'active_max', n_active_list=[3, 5], maxiter=1000, ftol=1e-15)
        if sc < best_score:
            best_score, best_h = sc, h_r

    # Phase 5: Nesterov subgradient with varying parameters
    if rleft(start_time, budget_s) > 300:
        for temp, lr_i, n_act in [(0.003, 0.003, 4), (0.001, 0.002, 3), (0.005, 0.002, 5), (0.002, 0.001, 4)]:
            if rleft(start_time, budget_s) < 40:
                break
            h_sg, sc = subgradient_optimized(best_h, 600, n_act, lr_i, 0.998, 0.3, temp, use_nesterov=True)
            if sc < best_score:
                best_score, best_h = sc, h_sg

    # Phase 6: Balance subgradient
    if rleft(start_time, budget_s) > 250:
        h_bal, sc = balance_subgradient(best_h, 500, n_active=4, temp=0.003, lr_init=0.002)
        if sc < best_score:
            best_score, best_h = sc, h_bal
        h_bal, sc = balance_subgradient(best_h, 400, n_active=6, temp=0.005, lr_init=0.001)
        if sc < best_score:
            best_score, best_h = sc, h_bal

    # Phase 7: SLSQP + surgery cycle
    if rleft(start_time, budget_s) > 200:
        h_r, sc = slsqp_run(best_h, 'softmax', alpha_list=[5000, 20000], maxiter=1000, ftol=1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_r
        h_s, sc = lag_surgery(best_h, max_steps=200, initial_eps=0.0006, n_lags=2)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_r, sc = slsqp_run(best_h, 'active_sum', n_active_list=[3], maxiter=1000, ftol=1e-16)
        if sc < best_score:
            best_score, best_h = sc, h_r

    # Phase 8: Diverse structured starts
    if rleft(start_time, budget_s) > 150:
        starts = structured_starts(n, target, rng)
        for h_s in starts:
            if rleft(start_time, budget_s) < 30:
                break
            sc = compute_c5(h_s)
            if sc < best_score - 0.005:
                h_opt, sc_opt = try_start(h_s, start_time + budget_s - 25, max_slsqp_iter=600)
                if sc_opt < best_score:
                    best_score, best_h = sc_opt, h_opt
                elif sc_opt < best_score + 0.001:
                    h_opt, sc_opt = try_start(h_opt, start_time + budget_s - 25, max_slsqp_iter=400)
                    if sc_opt < best_score:
                        best_score, best_h = sc_opt, h_opt

    # Phase 9: Perturbation restarts from best
    if rleft(start_time, budget_s) > 100:
        for attempt in range(6):
            if rleft(start_time, budget_s) < 25:
                break
            h_p = best_h.copy()
            n_cur = len(h_p)
            t_cur = n_cur / 2.0
            strategy = attempt % 4
            if strategy == 0:
                h_p += rng.randn(n_cur) * 0.04
            elif strategy == 1:
                block = max(8, n_cur // 25)
                for b in range(n_cur // block):
                    if rng.random() < 0.35:
                        h_p[b*block:min((b+1)*block, n_cur)] += rng.uniform(-0.12, 0.12)
            elif strategy == 2:
                h_p += 0.03 * np.sin(np.arange(n_cur) * 0.12 + rng.rand() * 6.28)
            else:
                n_flip = n_cur // 15
                idx = rng.choice(n_cur, n_flip, replace=False)
                h_p[idx] = 1.0 - h_p[idx]
            h_p = project_feasible(h_p, t_cur)
            h_r, sc = slsqp_run(h_p, 'active_sum', n_active_list=[3, 5], maxiter=800, ftol=1e-14)
            h_s, sc2 = lag_surgery(h_r, max_steps=100, initial_eps=0.0008, n_lags=2)
            if sc2 < sc:
                sc, h_r = sc2, h_s
            if sc < best_score:
                best_score, best_h = sc, h_r

    # Phase 10: Multi-resolution (try finer grid)
    if rleft(start_time, budget_s) > 200:
        for n_fine in [n + 100, n + 200]:
            if rleft(start_time, budget_s) < 100:
                break
            t_fine = n_fine / 2.0
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n_fine)
            h_fine = np.interp(x_new, x_old, best_h)
            h_fine = project_feasible(h_fine, t_fine)
            h_fine, sc_f = slsqp_run(h_fine, 'active_sum', n_active_list=[3, 5], maxiter=1000, ftol=1e-14)
            h_fine, sc_f = lag_surgery(h_fine, max_steps=150, initial_eps=0.001, n_lags=2)
            h_fine, sc_f2 = slsqp_run(h_fine, 'softmax', alpha_list=[1000, 5000], maxiter=800, ftol=1e-15)
            if sc_f2 < sc_f:
                sc_f = sc_f2
            h_sg_f, sc_f3 = subgradient_optimized(h_fine, 300, 4, 0.002, 0.998, 0.3, 0.003, use_nesterov=True)
            if sc_f3 < sc_f:
                sc_f = sc_f3
                h_fine, sc_f = h_sg_f, sc_f3
            if sc_f < best_score:
                best_score, best_h = sc_f, h_fine

    # Phase 11: Final high-precision polish
    if rleft(start_time, budget_s) > 50:
        h_r, sc = slsqp_run(best_h, 'active_max', n_active_list=[3, 5], maxiter=2000, ftol=1e-17)
        if sc < best_score:
            best_score, best_h = sc, h_r
        h_s, sc = lag_surgery(best_h, max_steps=200, initial_eps=0.0004, n_lags=2)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_r, sc = slsqp_run(best_h, 'softmax', alpha_list=[50000, 100000], maxiter=1000, ftol=1e-17)
        if sc < best_score:
            best_score, best_h = sc, h_r

    # Phase 12: Final surgery
    if rleft(start_time, budget_s) > 20:
        h_s, sc = lag_surgery(best_h, max_steps=150, initial_eps=0.0003, n_lags=1)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_s, sc = lag_surgery(best_h, max_steps=100, initial_eps=0.0003, n_lags=3)
        if sc < best_score:
            best_score, best_h = sc, h_s
        h_r, sc = slsqp_run(best_h, 'active_sum', n_active_list=[3], maxiter=500, ftol=1e-17)
        if sc < best_score:
            best_score, best_h = sc, h_r

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
