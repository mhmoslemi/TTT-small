def project_feasible(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}"""
    h = np.clip(h.copy(), 0.0, 1.0)
    # Water filling to achieve target sum
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

        grad = np.zeros(n)
        sorted_active = sorted(active)
        for idx, k_idx in enumerate(sorted_active):
            pk = p_active[idx]
            if pk < 1e-13:
                continue
            k = k_idx - half
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            if hi1 > lo1:
                grad[lo1:hi1] += pk * (1.0 - y[lo1 + k:hi1 + k]) * dx
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if hi2 > lo2:
                grad[lo2:hi2] -= pk * y[lo2 - k:hi2 - k] * dx
        h_new = project_to_box_sum(y - lr * grad, target)
        h_prev = h
        h = h_new
        t = t_new
        if step % 200 == 0:
            corr_new = np.correlate(h, 1.0 - h, mode='full')
            top_k = set(np.argsort(corr_new)[-topk:].tolist())
            current_max = corr_new.max() * dx
            for k_idx in range(1, len(corr_new) - 1):
                if k_idx not in active and corr_new[k_idx] * dx > current_max - 0.005:
                    active.add(k_idx)
                    break
            active = top_k | active
            if len(active) > 3 * topk:
                active = set(np.argsort(corr_new)[-3 * topk:].tolist())
            obj = compute_c5(h)
            if obj < best_obj:
                best_obj = obj
                best_h = h.copy()
    obj = compute_c5(h)
    if obj < best_obj:
        best_obj = obj
        best_h = h.copy()
    return best_h, best_obj


def subgradient_polish(h_init, max_steps, target, time_limit):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    half = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half
        grad = np.zeros(n)
        v1 = (idxs >= k_star) & (idxs < n - k_star) if k_star >= 0 else (idxs >= -k_star) & (idxs + k_star < n)
        if k_star >= 0:
            grad[:n - k_star] += (1.0 - h[k_star:n]) * dx
            grad[k_star:] -= h[:n - k_star] * dx
        else:
            m = -k_star
            grad[m:] += (1.0 - h[:n - m]) * dx
            grad[:n - m] -= h[m:] * dx
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.05, 0.1, 0.2]:
            h_try = h - lr * grad
            h_try = project_to_box_sum(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_obj - 1e-15:
                best_obj = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 20:
                break
    return best_h, best_obj


def coordinate_polish(h_init, target, time_limit, max_rounds=50):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[j]
            for dv in [0.05, -0.05, 0.1, -0.1, 0.15, -0.15, 0.2, -0.2]:
                new_val = old_val + dv - (np.sum(h) - target) / n
                if new_val < 0.0 or new_val > 1.0:
                    continue
                h[j] = new_val
                h = project_to_box_sum(h, target)
                sc = compute_c5(h)
                if sc < best_obj - 1e-14:
                    best_obj = sc
                    best_h = h.copy()
                    improved = True
                else:
                    break
        if not improved:
            break
    return best_h, best_obj


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha, _global_lam, _global_rho
    start_time = time.time()

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_to_box_sum(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: FISTA with increasing alpha (global-ish optimization)
    alphas_fista = [5, 15, 50, 150, 500]
    for alpha in alphas_fista:
        if tleft() < 120:
            break
        h_new, score = fista(best_h, alpha, 4000, target)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: Augmented Lagrangian + L-BFGS-B
    if tleft() > 150:
        for alpha in [500, 1500, 5000]:
            if tleft() < 50:
                break
            h_new, score = aug_lag_lbfgs(best_h, alpha, n_outer=12, n_inner=300, target=target, rho_init=1.0)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 3: FISTA with top-k constraint generation
    if tleft() > 150:
        for alpha in [500, 1500, 5000, 15000]:
            if tleft() < 40:
                break
            h_new, score = fista_topk(best_h, alpha, 3000, target, topk=40)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 4: More AL + L-BFGS at high alpha
    if tleft() > 150:
        for alpha in [5000, 15000, 50000]:
            if tleft() < 40:
                break
            h_new, score = aug_lag_lbfgs(best_h, alpha, n_outer=15, n_inner=400, target=target, rho_init=5.0)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 5: High-alpha FISTA for local refinement
    if tleft() > 120:
        for alpha in [10000, 30000, 100000]:
            if tleft() < 30:
                break
            h_new, score = fista(best_h, alpha, 5000, target)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 6: Subgradient polish
    if tleft() > 60:
        h_new, score = subgradient_polish(best_h, max_steps=3000, target=target,
                                          time_limit=min(30, tleft() * 0.5))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 7: AL + L-BFGS after subgradient
    if tleft() > 80:
        for alpha in [10000, 30000]:
            if tleft() < 30:
                break
            h_new, score = aug_lag_lbfgs(best_h, alpha, n_outer=15, n_inner=500, target=target, rho_init=10.0)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 8: Perturbation restarts
    rng = np.random.RandomState(seed)
    for noise in [0.03, 0.07, 0.12, 0.2]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_to_box_sum(h_pert, target)
        h_new, score = fista(h_pert, 100, 2000, target)
        if tleft() > 30:
            h_new2, score2 = aug_lag_lbfgs(h_new, 5000, n_outer=10, n_inner=200, target=target)
            if score2 < score:
                h_new = h_new2
                score = score2
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 9: Coordinate polish
    if tleft() > 40:
        h_new, score = coordinate_polish(best_h, target, time_limit=min(20, tleft() * 0.6), max_rounds=30)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 10: Final AL polish
    if tleft() > 30:
        h_new, score = aug_lag_lbfgs(best_h, 50000, n_outer=20, n_inner=500, target=target, rho_init=50.0)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 11: Final subgradient
    if tleft() > 15:
        h_new, score = subgradient_polish(best_h, max_steps=2000, target=target,
                                          time_limit=min(12, tleft() * 0.7))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Final feasibility
    best_h = project_to_box_sum(best_h, target)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
