    n = len(h)

    c_max = float(np.max(corr))

            h_new = slsqp_lse(h_best, n, target, 5000, 300, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_best = h_new

    # Phase 6: Main exploration - random perturbations + SLSQP
    n_perturb = 35
    for attempt in range(n_perturb):
        if tleft() < 50:
            break
        r = rng.rand()
        scale = 0.01 + 0.06 * rng.rand()
        if r < 0.35:
            h_pert = h_best + rng.randn(n) * scale
        elif r < 0.6:
            n_pert_coords = rng.randint(10, 60)
            idx = rng.choice(n, n_pert_coords, replace=False)
            h_pert = h_best.copy()
            h_pert[idx] += rng.randn(n_pert_coords) * scale * 2.5
        elif r < 0.8:
            x = np.linspace(0, 1, n)
            freq = rng.choice([2, 3, 4, 5, 7])
            phase = rng.uniform(0, 2 * np.pi)
            h_pert = h_best + scale * np.cos(freq * 2 * np.pi * x + phase)
        else:
            # Two-segment perturbation
            mid = rng.randint(n // 3, 2 * n // 3)
            h_pert = h_best.copy()
            h_pert[:mid] += rng.uniform(-scale, scale)
            h_pert[mid:] += rng.uniform(-scale, scale)

        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)
        if sc_pert > best_score * 1.5:
            continue

        h_pert = slsqp_lse(h_pert, n, target, 2000, 250, 1e-14)
        sc = compute_c5(h_pert)
        if sc < best_score:
            best_score = sc
            h_best = h_pert
            if tleft() > 25:
                h_new = slsqp_lse(h_best, n, target, 10000, 350, 1e-16)
                sc2 = compute_c5(h_new)
                if sc2 < best_score:
                    best_score = sc2
                    h_best = h_new
        elif sc_pert < best_score:
            h_best = h_pert
            best_score = sc_pert

    # Phase 7: Block coordinate descent
    if tleft() > 60:
        for bs in [4, 5, 6]:
            if tleft() < 30:
                break
            h_bd, sc_bd = block_lfgs(h_best, n, target, bs, 8, 3000.0, min(start_time + 700, hard_deadline - 30), rng)
            if sc_bd < best_score:
                best_score = sc_bd
                h_best = h_bd

    # Phase 8: Final SLSQP polish
    if tleft() > 25:
        h_new = slsqp_lse(h_best, n, target, 50000, 500, 1e-17)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_best = h_new

    # Phase 9: Final swaps
    if tleft() > 20:
        h_swap, sc_swap = swap_optimize(h_best, n, target, 200, hard_deadline - 10, rng)
        if sc_swap < best_score:
            best_score = sc_swap
            h_best = h_swap

    # Phase 10: Last SLSQP
    if tleft() > 15:
        h_new = slsqp_lse(h_best, n, target, 100000, 300, 1e-17)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_best = h_new

    h_final = project_feasible(h_best, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
