        # f_k(h) = sum_{i=0}^{n-k-1} h[i](1-h[i+k]) * dx
        # grad wrt h[j]:
        #   from h[i] term: dx*(1-h[i+k]) for i=j (if j < n-k)
        #   from h[i+k] term: -dx*h[i] for i+k=j (if j>=k, i<j, i < n-k)
        if k == 0:
            grad += w * dx * (1 - 2*h)
        else:
            # Positive part: j in [0, n-k-1], contribution dx*(1-h[j+k])
            pos = np.arange(0, n-k)
            grad[pos] += w * dx * (1 - h[pos + k])
            # Negative part: j in [k, n-1], i = j-k, contribution -dx*h[j-k]
            # But i must be < n-k, so j-k < n-k, i.e., j < n. Always true.
            neg = np.arange(k, n)
            grad[neg] -= w * dx * h[neg - k]
    
    return grad, overlaps * dx

def nesterov_subgrad(h, n, target, alpha, top_m, n_iter, deadline):
    """Nesterov accelerated subgradient method."""
    dx = 2.0 / n
    h_curr = h.copy()
    h_prev = h.copy()
    y = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t = 1.0
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        
        # Nesterov: y = h + (t-1)/(t+2) * (h - h_prev)
        momentum = (t - 1) / (t + 2)
        y = h_curr + momentum * (h_curr - h_prev)
        y = project_feasible(y, target)
        
        # Compute gradient at y
        grad, overlaps = subgradient_step(y, n, target, alpha, top_m)
        
        # Step size: t / (alpha * sqrt(sum(overlaps^2))) approximately
        # Use standard: step = 2 / (alpha * (i+1))
        step = 2.0 / (alpha * (i + 1))
        # Cap the step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            step = min(step, 0.1 / grad_norm)
        
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        h_prev = h_curr
        h_curr = h_new
        t += 1
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_of_top(h, n, alpha, top_m):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    ov = corr[n-1:n-1+n]
    max_ov = np.max(ov)
    shifted = alpha * (ov - max_ov)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    sw = weights.sum()
    if sw < 1e-300:
        weights = np.zeros(n)
        weights[int(np.argmax(ov))] = 1.0
    else:
        weights /= sw
    if top_m < n:
        top_idx = np.argsort(ov)[::-1][:top_m]
        w_sub = weights[top_idx]
        ws = w_sub.sum()
        if ws > 0:
            w_sub /= ws
        else:
            w_sub = np.ones(top_m) / top_m
        weights = np.zeros(n)
        weights[top_idx] = w_sub
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-10:
            continue
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        else:
            pos = np.arange(0, n - k)
            grad[pos] += w * dx * (1.0 - h[pos + k])
            neg = np.arange(k, n)
            grad[neg] -= w * dx * h[neg - k]
    return grad, ov * dx

def nesterov_subgrad(h, n, target, alpha, top_m, n_iter, deadline):
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum_t = 0.0
    for i in range(n_iter):
        if time.time() > deadline:
            break
        t = i + 1
        momentum = (momentum_t - 1.0) / (momentum_t + 2.0) if momentum_t > 1 else 0.0
        momentum_t = t
        y = h_curr + momentum * (h_curr - h_prev)
        y = project_feasible(y, target)
        grad, ov = subgradient_of_top(y, n, alpha, top_m)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        step = 2.0 / (alpha * t ** 0.5)
        step = min(step, 0.5 / gn)
        h_new = y - step * grad
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h_prev = h_curr.copy()
        h_curr = h_new
    return best_h, best_score

def backtracking_subgrad(h, n, target, alpha, top_m, n_iter, deadline):
    h_curr = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 0.01
    for i in range(n_iter):
        if time.time() > deadline:
            break
        grad, ov = subgradient_of_top(h_curr, n, alpha, top_m)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        h_try = h_curr - lr * grad
        h_try = project_feasible(h_try, target)
        sc_try = compute_c5(h_try)
        cur_sc = compute_c5(h_curr)
        if sc_try < cur_sc - 1e-12:
            h_curr = h_try
            lr = min(lr * 1.05, 0.1)
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_try.copy()
        else:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_score

def slsqp_lse(h, n, target, alphas, maxiter, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    for a in alphas:
        if time.time() > deadline:
            break
        def obj(x, _a=float(a)):
            corr = np.correlate(x, 1.0 - x, mode='full')
            ov = corr[n-1:n-1+n]
            mx = np.max(ov)
            sh = _a * (ov - mx)
            sh = np.clip(sh, -700.0, 0.0)
            return (mx + np.log(np.sum(np.exp(sh))) / _a)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def fourier_perturb_search(h, n, target, n_modes, n_restarts, deadline, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    best_h = h.copy()
    best_score = compute_c5(h)
    for r in range(n_restarts):

        mode = rng.randint(1, n_modes + 1)

        amp = rng.uniform(0.02, 0.12)
        phase = rng.uniform(0, 2 * np.pi)
        perturb = amp * np.cos(np.pi * freq * x + phase)
        if rng.rand() < 0.5:
            perturb = amp * np.sin(np.pi * freq * x + phase)
        h_try = project_feasible(h + perturb, target)
        h_opt, sc = slsqp_lse(h_try, n, target, [200, 800], 800, min(deadline, time.time() + 30))
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()
    return best_h, best_score

    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        r = rng.rand()
        if r < 0.3:

        elif r < 0.5:
            freq = rng.randint(1, 8)

        elif r < 0.7:
            pop[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
        elif r < 0.85:
            blocks = rng.randint(3, 10)
            edges = np.sort(rng.choice(np.arange(1, n), size=blocks, replace=False))
            pop[i] = np.zeros(n)
            for j in range(len(edges) + 1):
                s = 0 if j == 0 else edges[j-1]
                e = n if j == len(edges) else edges[j]
                pop[i][s:e] = rng.uniform(0.1, 0.9)
        else:
            pop[i] = 0.5 + rng.uniform(0.1, 0.3) * np.cos(2 * np.pi * x / 2.0 * rng.randint(1, 5))
        pop[i] = project_feasible(pop[i], target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

                scores[i] = sc
                if sc < best_score:
                    best_score = sc
                    best_h = trial.copy()
    return best_h, best_score
