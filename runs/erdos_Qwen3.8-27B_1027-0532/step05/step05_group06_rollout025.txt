    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

import numpy as np
from scipy.optimize import minimize
import time


def project_fast(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_and_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    return val, weights, corr


def lse_grad(h, weights, n):
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
    return grad


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    val, weights, corr = corr_and_weights(x, alpha)
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    grad = lse_grad(x, weights, n)
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_run(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxls': 60, 'maxfun': maxiter * 10})
        return project_fast(res.x, target)
    except Exception:
        return h_cur


def top_lag_descent(h, target, n_top=5, n_steps=500, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.003
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = float(top_vals.mean())
        top_std = float(top_vals.std())
        if top_std < 1e-14:
            break
        weights_arr = np.zeros(len(corr))
        for ki in top_idx:
            weights_arr[ki] = (corr[ki] - top_mean) / (top_std + 1e-15)
        grad = lse_grad(h_cur, weights_arr, n)
        s = float(np.sum(h_cur))
        grad += 2.0 * 1.0 * (s - target)
        h_new = h_cur - lr * grad
        h_new = project_fast(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-15:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return h_cur, cur_score

    lr = 0.01

        norm_g = np.linalg.norm(grad)

        y_new = h_cur - step_size * v
        y_new = project_fast(y_new, target)

def slsqp_polish(h, target, alpha, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

        return lse_obj_and_grad(x, alpha, target, 1e6)[0]

        return lse_obj_and_grad(x, alpha, target, 1e6)[1]

    def eq(x):
        return np.sum(x) - target

        return np.ones(n)

    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 100})
        return project_fast(res.x, target)

        return h_cur


def generate_perturbation(base, rng, n, target, strategy, param):
    h_pert = base.copy()
    if strategy == 'gauss':
        h_pert += rng.randn(n) * param
    elif strategy == 'sparse':
        n_flip = max(1, int(n * param))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = rng.uniform(0, 1, n_flip)
    elif strategy == 'block':
        block_size = max(2, n // 20)
        for _ in range(int(param * 10)):
            start = rng.randint(0, max(1, n - block_size))
            val = rng.uniform(0, 1)
            h_pert[start:start + block_size] = val
    elif strategy == 'freq':
        freqs = np.fft.rfftfreq(n)
        mask = freqs < param
        pert = np.fft.irfft(np.where(mask, rng.randn(n // 2 + 1), 0), n)
        h_pert += pert * 0.5
    elif strategy == 'binary_mix':
        binary = (base > 0.5).astype(float)
        h_pert = param * h_pert + (1.0 - param) * binary
        h_pert += rng.randn(n) * 0.05
    h_pert = project_fast(h_pert, target)
    return h_pert

    return h_fine
