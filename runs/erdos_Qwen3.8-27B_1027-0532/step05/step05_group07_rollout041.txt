    """Compute gradient of sum of correlations at active lags w.r.t. h."""
    grad = np.zeros(n)

        if d >= 0:
            # c_d = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            for i in range(n - d):

            for i in range(n - e):

    return grad

    Returns array of shape (2n-1, n) where row m is gradient of corr[m]."""

                grads[m, idx2] -= h[idx2 - d]

            if n-1-e >= 0:
                idx = np.arange(e, n)

            idx2 = np.arange(n-e)

    best_h = h0.copy()

        corr = compute_corr(h)
        c_max = np.max(corr)
        
        # Identify active lags (within 1% of max)
        active = np.where(corr >= c_max * 0.99)[0]

        # Actually, for the LP we need all lags. Let's use all.
        grads = compute_all_lag_gradients(h, n)  # shape (2n-1, n)
        
        # Setup LP
        # Variables: [h[0], ..., h[n-1], t]
        # Minimize: t
        c_obj = np.zeros(n+1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: grads @ h - t <= b
        # where b = grads @ h0 - corr  (for each row)
        b = grads @ h - corr
        A_ub = np.hstack([grads, -np.ones((2*n-1, 1))])
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds
        bounds = [(0, 1)] * n + [(-np.inf, np.inf)]
        
        res = linprog(c_obj, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        h_new = res.x[:n]
        
        # Trust region / line search
        direction = h_new - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-12:
            break
        
        # Try full step, then bisection
        found = False
        alphas = [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]
        for alpha in alphas:
            h_try = h + alpha * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score - 1e-15:
                h = h_try
                best_h = h_try.copy()
                best_score = sc
                found = True
                break

            small_alpha = 0.01
            h_try = h + small_alpha * direction

            if sc < best_score:
                h = h_try
                best_h = h_try.copy()
                best_score = sc
            else:
                break
    
    return best_h, best_score

    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-17):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_step(h, n, target, n_active=10, step_decay=0.95):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:n_active]
    grads = compute_all_lag_gradients(h, n)
    g = np.sum(grads[active], axis=0) / len(active)
    return g


def projected_subgradient(h_init, n, target, deadline, n_active=10, lr_start=0.02,
                          lr_decay=0.98, momentum=0.9):
    h = h_init.copy()
    best_h = h_init.copy()
    best_score = compute_c5(h_init)
    lr = lr_start
    v = np.zeros(n)
    it = 0
    stall = 0
    while time.time() < deadline:
        g = subgradient_step(h, n, target, n_active)
        v = momentum * v - lr * g
        v_sum = np.sum(v)
        v = v - v_sum / n
        h_try = h + v
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_score - 1e-16:
            h = h_try
            best_h = h_try.copy()
            best_score = sc
            stall = 0
        else:
            stall += 1
            lr *= lr_decay
            if stall > 200:
                break
        it += 1
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: Quick coordinate descent to get close
    deadline_p1 = start_time + (budget_s - 25.0) * 0.15
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p1, rng,
                                    recompute_interval=20)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: LP linearization (main innovation)
    if tleft() > 120:
        deadline_p2 = start_time + (budget_s - 25.0) * 0.40
        h_lp, sc_lp = lp_phased_optimize(best_h, n, target, deadline_p2,
                                         n_active_start=30, n_active_end=150)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 3: Coordinate descent refinement after LP
    if tleft() > 100:
        deadline_p3 = start_time + (budget_s - 25.0) * 0.55
        h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p3, rng,
                                        recompute_interval=15)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Second LP round
    if tleft() > 150:
        deadline_p4 = start_time + (budget_s - 25.0) * 0.72
        h_lp, sc_lp = lp_phased_optimize(best_h, n, target, deadline_p4,
                                         n_active_start=50, n_active_end=200)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    # Phase 5: Subgradient refinement
    if tleft() > 80:
        deadline_p5 = start_time + (budget_s - 25.0) * 0.80
        h_sg, sc_sg = projected_subgradient(best_h, n, target, deadline_p5,
                                            n_active=15, lr_start=0.01)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 6: Final coordinate descent
    if tleft() > 50:
        deadline_p6 = start_time + (budget_s - 25.0) * 0.90
        h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p6, rng,
                                        recompute_interval=10)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 7: SLSQP polish
    if tleft() > 40:
        h_ref = slsqp_polish(best_h, n, target, [300, 1000, 5000, 15000], 2000, 1e-17)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Very fine coordinate descent at the end
    if tleft() > 20:
        deadline_p8 = start_time + budget_s - 20.0
        h_final, sc_final = paired_coord_desc(best_h, n, target, deadline_p8, rng,
                                              recompute_interval=5)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 9: Final LP micro-step
    if tleft() > 5:
        h_lp, sc_lp = lp_phased_optimize(best_h, n, target, start_time + budget_s - 15.0,
                                         n_active_start=80, n_active_end=200, max_iter=200)
        if sc_lp < best_score:
            best_score = sc_lp
            best_h = h_lp

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
