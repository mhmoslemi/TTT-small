```python
# Term 3: +h[i-k] where 0<=i-k<n

```

This is O(n) with a constant number of vector operations. In practice with NumPy, this might take ~5-20μs for n=400.

But actually, we can be even smarter. For binary h, (1-h[x]) is just the complement. So:
- Term 1: -(1-h[i+k]) = h[i+k] - 1 (for binary h)
- Term 2: +(1-h[j+k]) = 1 - h[j+k]
- Term 3: +h[i-k]
- Term 4: -h[j-k]

So Δc[n-1+k] = (h[i+k]-1) + (1-h[j+k]) + h[i-k] - h[j-k]
= h[i+k] - h[j+k] + h[i-k] - h[j-k]

Where terms with out-of-bounds indices are 0.

For binary h, each h value is 0 or 1, so each term is -1, 0, or 1, and the total delta is in {-4, -3, ..., 3, 4}.

Let me also think about the continuous refinement. The SLSQP approach from before is good but might be slow for n=400. An alternative is to use a projected gradient method:

f(h) = max_k g_k(h) where g_k(h) = ∑_i h[i](1-h[i+k])*dx

∇g_k(h)_i = dx * [(1-h[i+k]) - h[i-k]] (for valid indices)

∇f(h) = ∇g_{k*}(h) where k* = argmax_k g_k(h)

Then: h_new = project(h - lr * ∇f(h))

This is the subgradient method, which was in the previous code. It's simple but slow to converge.

A better approach: use the log-sum-exp smoothing:
f_α(h) = (1/α) * log(∑_k exp(α * g_k(h)))

∇f_α(h)_i = (1/∑_k exp(α*g_k)) * ∑_k exp(α*g_k) * ∇g_k(h)_i

This gives a smooth approximation to the max, and as α→∞, it approaches the true max. The gradient can be computed in O(n²) (for all k, compute ∇g_k and weight by softmax).

```python
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h > 0.5).astype(float)
    s = h_bin.sum()
    if s == n // 2:
        return h_bin
    if s > n // 2:
        ones = np.where(h_bin == 1.0)[0]
        # Remove the ones with lowest h values
        excess = int(s - n // 2)
        # Sort ones by h value (ascending) and remove the lowest
        h_at_ones = h[ones]
        remove_idx = ones[np.argsort(h_at_ones)[:excess]]
        h_bin[remove_idx] = 0.0
    else:
        zeros = np.where(h_bin == 0.0)[0]
        deficit = int(n // 2 - s)
        h_at_zeros = h[zeros]
        add_idx = zeros[np.argsort(h_at_zeros)[::-1][:deficit]]
        h_bin[add_idx] = 1.0
    return h_bin

def binary_sa(h_init, n, time_limit, rng):
    """Fast binary simulated annealing with incremental correlation update."""
    start = time.time()
    target = n // 2
    half_n = n - 1
    corr_len = 2 * n - 1
    
    h = to_binary(h_init, n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c[half_n] = 0.0  # Ensure k=0 is 0 for binary
    
    best_c5 = np.max(c)
    best_h = h.copy()
    cur_c5 = best_c5
    
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    
    # k_range for correlation indices
    # k in c[n-1+k] corresponds to shift k
    # k ranges from -(n-1) to (n-1)
    
    n_steps = 500000
    T_start = 0.02
    T_end = 0.001
    
    n_reject = 0
    n_plateau = 0
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        frac = step / float(n_steps)
        T = T_start * (T_end / T_start) ** frac
        
        # Pick random 1 and 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta correlation
        # dc[k_offset] where k_offset = k + half_n, k is the shift
        # dc[shift] = h[i+shift] - h[j+shift] + h[i-shift] - h[j-shift] (with bounds)
        dc = np.zeros(corr_len)
        
        # h[i+k] for valid i+k: contributes to dc at position (k + half_n)
        # k ranges such that 0 <= i+k < n, i.e., -i <= k < n-i
        k_lo = max(-i, -half_n)
        k_hi = min(n - 1 - i, half_n)
        if k_hi >= k_lo:
            shifts = np.arange(k_lo, k_hi + 1)
            dc[shifts + half_n] += h[i + shifts]
        
        # -h[j+k]
        k_lo = max(-j, -half_n)
        k_hi = min(n - 1 - j, half_n)
        if k_hi >= k_lo:
            shifts = np.arange(k_lo, k_hi + 1)
            dc[shifts + half_n] -= h[j + shifts]
        
        # +h[i-k]
        # i-k valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
        k_lo = max(i - n + 1, -half_n)
        k_hi = min(i, half_n)
        if k_hi >= k_lo:
            shifts = np.arange(k_lo, k_hi + 1)
            dc[shifts + half_n] += h[i - shifts]
        
        # -h[j-k]
        k_lo = max(j - n + 1, -half_n)
        k_hi = min(j, half_n)
        if k_hi >= k_lo:
            shifts = np.arange(k_lo, k_hi + 1)
            dc[shifts + half_n] -= h[j - shifts]
        
        # Fix k=0
        dc[half_n] = 0.0
        
        # New correlation
        c_new = c + dc
        new_c5 = np.max(c_new)
        
        delta = new_c5 - cur_c5
        
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            c = c_new
            cur_c5 = new_c5
            
            # Update ones and zeros
            ones[ones == i] = j  # Replace i with j in ones
            zeros[zeros == j] = i  # Replace j with i in zeros
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                n_plateau = 0
            else:
                n_plateau += 1
            
            n_reject = 0
        else:
            n_reject += 1
        
        # Reset plateau counter periodically
        if step % 100000 == 0 and step > 0:
            n_plateau = 0
    
    return best_h, best_c5
```
