from scipy.optimize import minimize
import time

_c_n = 0
_c_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
    return h

def project_binary(h, target_sum):
    idx = np.argsort(-h)[:int(target_sum)]
    h_r = np.zeros(len(h))
    h_r[idx] = 1.0
    return h_r

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return (c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha) * (2.0 / len(h))

def cons_fun(h):
    return np.sum(h) - _c_target

def cons_jac(h):
    return np.ones(_c_n)

def slsqp_phase(h_start, n, target, alphas, maxiter, ftol, tlimit, t0):
    global _c_n, _c_target
    _c_n = n
    _c_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - t0 > tlimit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                cur = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, cur
