import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]  # positive lags
    return float(np.max(over)) * dx

def grad_f_k(k, h, dx):
    """Gradient of f_k(h) = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])*dx"""
    n = len(h)
    g = np.zeros(n)
    # From h[i] term: for i in 0..n-1-k, contributes (1-h[i+k])*dx to g[i]
    pos = np.arange(0, n-k)
    g[pos] += (1.0 - h[pos + k]) * dx
    # From (1-h[i+k]) term: for i in 0..n-1-k, contributes -h[i]*dx to g[i+k]
    neg = np.arange(k, n)
    g[neg] -= h[neg - k] * dx
    return g

def project_box_sum(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum} using bisection."""
    h = h.copy()
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0.0, 1.0)

def proximal_gradient(h, n, target, n_steps=2000, K_top=10, 
                      temp_init=0.01, temp_decay=0.999,
                      lr_init=0.001, deadline=None):
    """Proximal gradient with softmax-weighted subgradient."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    temp = temp_init
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Softmax weights on top-K lags
        top_idx = np.argsort(over)[::-1][:K_top]
        diffs = over[top_idx] - np.max(over[top_idx])
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            grad += wk * grad_f_k(int(k), h_cur, dx)
        
        # Damped step with project
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.9
            if lr < 1e-10:
                # Reset with perturbation
                h_cur = best_h + 0.001 * np.random.randn(n)
                h_cur = project_box_sum(h_cur, target)
                lr = lr_init * 0.5
        
        temp *= temp_decay
        temp = max(temp, 1e-6)
    
    return best_h, best_score

    h = np.asarray(h, dtype=float)
    if abs(h.sum() - target_sum) < 1e-15:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2
    return np.clip(h - mid, 0.0, 1.0)

def proximal_grad_optimize(h, n, target, n_steps, K, temp_init, temp_decay,
                           lr_init, lr_max, deadline=None):
    """Proximal gradient with exact projection and softmax subgradient."""
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    temp = temp_init
    lr = lr_init
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Softmax weights on top-K lags
        top_idx = np.argsort(over)[::-1][:K]
        max_over = np.max(over[top_idx])
        diffs = over[top_idx] - max_over
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            if wi > 1e-10:
                grad += wi * grad_f_k(int(ki), h_cur, dx)
        
        # Step
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, lr_max)
        else:
            lr *= 0.85
            if lr < 1e-12:
                break
        
        temp = max(temp * temp_decay, 1e-8)
    
    return best_h, best_score

    pos = np.arange(0, n - k)
    g[pos] += (1.0 - h[pos + k]) * dx
    neg = np.arange(k, n)
    g[neg] -= h[neg - k] * dx
    return g

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    mid = (lo + hi) / 2.0
    return np.clip(h - mid, 0.0, 1.0)

def proximal_grad_optimize(h, n, target, n_steps, K, temp_init, temp_decay,
                           lr_init, lr_max, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    temp = temp_init
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:K]
        max_ov = float(np.max(over[top_idx]))
        diffs = over[top_idx] - max_ov
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wi, ki in zip(weights, top_idx):
            if wi > 1e-10:
                grad += wi * grad_f_k(int(ki), h_cur, dx)
        h_new = h_cur - lr * grad
        h_new = project_box_sum(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-16:
            h_cur = h_new
            lr = min(lr * 1.01, lr_max)
        else:
            lr *= 0.88
            if lr < 1e-12:
                break
        temp = max(temp * temp_decay, 1e-9)
    return best_h, best_score

def fista_optimize(h, n, target, n_steps, deadline=None):
    dx = 2.0 / n
    x = h.copy()
    y = h.copy()
    t_val = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 0.01
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr_y = compute_corr(y)
        over_y = corr_y[n-1:n-1+n]
        k_star = int(np.argmax(over_y))
        g = grad_f_k(k_star, y, dx)
        x_new = y - lr * g
        x_new = project_box_sum(x_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        y_new = x_new + ((t_val - 1.0) / t_new) * (x_new - x)
        y_new = project_box_sum(y_new, target)
        x = x_new
        y = y_new
        t_val = t_new
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_h = x.copy()
    return best_h, best_score

def slsqp_lse(h, n, target, alphas, maxiter, ftol, deadline=None):
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break

        def obj(x):
            xc = np.clip(x, 0.0, 1.0)
            corr = np.correlate(xc, 1.0 - xc, mode='full')
            over = corr[n-1:n-1+n]
            c_max = float(np.max(over))
            shifted = alpha * (over - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        except Exception:
            pass
    return best_h, best_score

def lp_guided_step(h, n, target, K, alpha_blend, deadline=None):
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    top_idx = np.argsort(over)[::-1][:K]
    max_ov = float(np.max(over[top_idx]))
    grad = np.zeros(n)
    for ki in top_idx:
        grad += grad_f_k(int(ki), h, dx)
    grad /= K
    h_new = h - alpha_blend * grad
    h_new = project_box_sum(h_new, target)
    return h_new

def multi_restart(h, n, target, n_restarts, scales, pg_steps, K,
                  temp_init, deadline=None):
    best_h = h.copy()
    best_score = compute_c5(h)
    rng = np.random.RandomState(777)
    for r in range(n_restarts):
        if deadline is not None and time.time() > deadline:
            break
        scale = scales[r % len(scales)]
        h_pert = best_h + scale * rng.randn(n)
        h_pert = project_box_sum(h_pert, target)
        h_opt, sc = proximal_grad_optimize(
            h_pert, n, target, n_steps=pg_steps, K=K,
            temp_init=temp_init, temp_decay=0.998,
            lr_init=0.003, lr_max=0.03, deadline=deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_opt
    return best_h, best_score

def coarse_to_fine(h, n, target, deadline=None):
    n_coarse = max(n // 4, 50)
    if n_coarse * 2 < n:
        x_coarse = np.linspace(0, 1, n_coarse)
        x_fine = np.linspace(0, 1, n)
        h_coarse = np.interp(x_coarse, x_fine, h)
        h_coarse = project_box_sum(h_coarse, n_coarse / 2.0)
        h_coarse_opt, _ = proximal_grad_optimize(
            h_coarse, n_coarse, n_coarse / 2.0, n_steps=1500, K=5,
            temp_init=0.01, temp_decay=0.999,
            lr_init=0.005, lr_max=0.05, deadline=deadline)
        h_up = np.interp(x_fine, x_coarse, h_coarse_opt)
        h_up = project_box_sum(h_up, target)
        return h_up
    return h.copy()

def top2_equalize(h, n, target, max_iters=200, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top2 = np.argsort(over)[::-1][:2]
        k1, k2 = int(top2[0]), int(top2[1])
        if abs(over[k1] - over[k2]) < 1e-10:
            g = grad_f_k(k1, h_cur, dx)
        else:
            g1 = grad_f_k(k1, h_cur, dx)
            g2 = grad_f_k(k2, h_cur, dx)
            diff_factor = (over[k1] - over[k2]) / max(abs(over[k1]) + abs(over[k2]), 1e-15)
            g = g1 + diff_factor * g2
        for alpha in [0.005, 0.002, 0.001, 0.0005]:
            h_test = h_cur - alpha * g
            h_test = project_box_sum(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_score - 1e-15:
                h_cur = h_test
                best_score = sc
                break
    return h_cur, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_box_sum(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA (fast initial convergence)
    dl1 = start_time + 120
    h_fista, sc = fista_optimize(best_h, n, target, n_steps=2000, deadline=dl1)
    if sc < best_score:
        best_score = sc
        best_h = h_fista

    # Phase 2: Proximal gradient with top-5
    if tleft() > 150:
        dl2 = start_time + 300
        best_h, best_score = proximal_grad_optimize(
            best_h, n, target, n_steps=5000, K=5,
            temp_init=0.008, temp_decay=0.9995,
            lr_init=0.005, lr_max=0.05, deadline=dl2)

    # Phase 3: SLSQP polish
    if tleft() > 120:
        dl3 = start_time + 500
        best_h, best_score = slsqp_lse(
            best_h, n, target, [200, 1000, 4000], 2500, 1e-15, deadline=dl3)

    # Phase 4: Coarse-to-fine restart
    if tleft() > 200:
        h_cf = coarse_to_fine(best_h, n, target, deadline=start_time + 600)
        sc_cf = compute_c5(h_cf)
        if sc_cf < best_score:
            best_score = sc_cf
            best_h = h_cf
        dl4 = start_time + 700
        best_h, best_score = proximal_grad_optimize(
            best_h, n, target, n_steps=3000, K=3,
            temp_init=0.003, temp_decay=0.999,
            lr_init=0.003, lr_max=0.03, deadline=dl4)

    # Phase 5: SLSQP with higher alpha
    if tleft() > 120:
        dl5 = start_time + 850
        best_h, best_score = slsqp_lse(
            best_h, n, target, [5000, 15000], 3000, 1e-16, deadline=dl5)

    # Phase 6: Top-2 equalization
    if tleft() > 100:
        dl6 = start_time + 950
        best_h, best_score = top2_equalize(
            best_h, n, target, max_iters=500, deadline=dl6)

    # Phase 7: Multi-restart with diverse scales
    if tleft() > 100:
        scales = [0.003, 0.005, 0.01, 0.002, 0.008, 0.004]
        dl7 = start_time + 1100
        best_h, best_score = multi_restart(
            best_h, n, target, n_restarts=30, scales=scales,
            pg_steps=400, K=3, temp_init=0.005, deadline=dl7)

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 60:
        dl8 = start_time + budget_s - 50
        best_h, best_score = slsqp_lse(
            best_h, n, target, [20000, 80000], 3000, 1e-17, deadline=dl8)

    # Phase 9: Final proximal gradient micro-steps
    if tleft() > 25:
        dl9 = start_time + budget_s - 15
        best_h, best_score = proximal_grad_optimize(
            best_h, n, target, n_steps=2000, K=2,
            temp_init=0.001, temp_decay=0.999,
            lr_init=0.0005, lr_max=0.005, deadline=dl9)

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
