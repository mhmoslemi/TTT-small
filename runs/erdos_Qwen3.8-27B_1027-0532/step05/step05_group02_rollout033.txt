def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fft(h):

    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def continuous_refine(h_start, n, time_limit, start_time):
    """Use L-BFGS-B on a smooth approximation of the max."""
    dx = 2.0 / n
    target = float(n) / 2.0
    
    # Smooth max using log-sum-exp
    alphas = [50, 200, 1000, 5000]
    h_cur = h_start.copy()
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def obj(x):
            corr = full_corr_fft(x) * dx
            # Log-sum-exp for numerical stability
            c_max = np.max(corr)
            if c_max * alpha > 500:
                return c_max + 500.0 / alpha
            shifted = alpha * (corr - c_max)
            return c_max + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / alpha + 0.01 * (np.sum(x) - target)**2
        
        def grad(x):
            corr = full_corr_fft(x) * dx
            c_max = np.max(corr)
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -500, 0)
            weights = np.exp(shifted)
            w_sum = np.sum(weights)
            if w_sum == 0:
                return np.zeros(n)
            p = weights / w_sum
            
            # Gradient of c(k) w.r.t. h[j]
            # c(k) = sum_i h[i]*(1-h[i+k]) * dx (approximately, ignoring boundary effects for simplicity)
            # Using FFT structure:
            # grad of full_corr_fft(x) w.r.t. x[j]
            # The correlation is: result[m] = sum_i x[i] * (1-x[i+m])
            # d result[m] / d x[j] = (1-x[j+m]) * I(j+m valid) - x[j-m] * I(j-m valid)
            # (for the appropriate range)
            
            g = np.zeros(2*n-1)
            m_range = np.arange(-(n-1), n)
            
            # For each j, compute the gradient contribution
            # This is O(n^2) but for n=600 it's fine
            for j in range(n):
                # First factor: x[j] contributes to c(m) when 0<=j<n and 0<=j+m<n
                valid_f = (j + m_range >= 0) & (j + m_range < n)
                # (1-x[j+m])
                jm = j + m_range
                val_f = np.zeros(2*n-1)
                val_f[valid_f] = 1.0 - x[jm[valid_f]]
                
                # Second factor: x[j] appears as (1-x[j]) when j = i+m, i = j-m
                valid_s = (j - m_range >= 0) & (j - m_range < n)
                jp = j - m_range
                val_s = np.zeros(2*n-1)
                val_s[valid_s] = x[jp[valid_s]]
                
                g += (val_f - val_s) * dx
            
            # p is the softmax weight
            return np.sum(p[:, None] * g, axis=1) + 0.02 * (x - target/n)
        
        # This gradient is O(n^2) which is 360000 for n=600
        # Each evaluation of obj is O(n log n)
        # L-BFGS-B will need maybe 100-500 iterations
        
        try:
            res = fmin_l_bfgs_b(obj, h_cur, fprime=grad, 
                                  factr=1e7, pgtol=1e-12,
                                  iprint=0, maxiter=200)
            h_new = project_feasible(res, target)
        except Exception:
            h_new = h_cur
        
        cur_score = compute_c5(h_cur)
        new_score = compute_c5(h_new)
        if new_score < cur_score:
            h_cur = h_new
    
    return h_cur, compute_c5(h_cur)
