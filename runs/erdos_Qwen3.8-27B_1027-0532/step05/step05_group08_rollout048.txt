  - The solution is: v[i] = 1 for the n/2 indices with smallest g[i], v[i] = 0 otherwise

- Update: h_new = (1-γ) * h + γ * v, where γ is a step size (can use γ = 2/(t+2) for the standard Fogel-Tracey step)

from scipy.optimize import minimize
import time


def compute_corr_fft(h):
    n = len(h)
    c = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(c, N)
    corr_raw = np.fft.irfft(np.conj(H) * C, N)
    result = np.empty(2 * n - 1)
    result[n - 1:] = corr_raw[:n]
    result[:n - 1] = corr_raw[N - n + 1:N]
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fft(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def topk_gradient(h, w, n, K=20):
    """Compute gradient using only top K lags."""
    dx = 2.0 / n
    g = np.zeros(n)
    # Find top K indices of w
    top_idx = np.argsort(w)[::-1][:K]
    for m in top_idx:
        wk = w[m]
        if wk < 1e-14:
            continue
        k = int(m) - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def frank_wolfe_step(h, g, target):
    """Frank-Wolfe linear minimization oracle.

def compute_corr_fft(h):
    n = len(h)
    c = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    C = np.fft.rfft(c, N)
    corr_raw = np.fft.irfft(np.conj(H) * C, N)
    result = np.empty(2 * n - 1)
    result[n - 1:] = corr_raw[:n]
    result[:n - 1] = corr_raw[N - n + 1:N]
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fft(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value_func(h, alpha):
    corr = compute_corr_fft(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _get_weights(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    return w


def topk_gradient(h, w, n, K):
    dx = 2.0 / n
    g = np.zeros(n)
    order = np.argsort(w)[::-1][:K]
    for m in order:
        wk = w[m]
        if wk < 1e-15:
            continue
        k = int(m) - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def lse_grad_func(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    w = _get_weights(corr, alpha)
    K = min(40, n)

            h_pg2, sc_pg2 = pg_round(best_h2, n2, target2, 5000, 200, deadline - 40)
            if sc_pg2 < best_score2:
                best_score2 = sc_pg2
                best_h2 = h_pg2

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 7: Final SLSQP
    if time.time() < deadline - 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new = slsqp_polish(best_h, n_cur, target_cur, 20000, 800)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 8: Final binary search
    if time.time() < deadline - 10:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bin, sc_bin = binary_local_search(best_h, n_cur, target_cur, 20, deadline - 5)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
