from scipy.optimize import minimize
import time

# Global variables for callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # We only care about the "overlap" part (lags 0 to n-1)
    # np.correlate(h, 1-h, 'full') has length 2n-1
    # The overlap part is indices n-1 to 2n-2 (lags 0 to n-1)
    over = corr[n-1:]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _lse_obj(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:

            break
        _g_alpha = float(a)

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def active_set_descent(h, n, target, n_steps, top_k, lr_init, momentum, deadline):
    """Gradient descent with momentum on the active set (top-k lags)."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top-k lags with softmax weighting
        top_idx = np.argsort(over)[::-1][:top_k]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / (0.01 * over[top_idx[0]] + 1e-10))
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        # Momentum step
        vel = momentum * vel - lr * grad
        h_new = h_cur + vel
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 10)
        else:
            lr *= 0.7
            vel *= 0.5
            if lr < 1e-12:
                break
    
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):

    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    over = corr[n - 1:]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.99999
        else:
            mask = h > 1e-5
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _lse_obj(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def _lse_grad(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    dx = 2.0 / _g_n
    g = np.zeros(_g_n)
    for wk, k in zip(w, range(_g_n)):
        k = int(k)
        jp = np.arange(0, _g_n - k)
        g[jp] += wk * (1.0 - x[jp + k])
        jn = np.arange(k, _g_n)
        g[jn] -= wk * x[jn - k]
    return g * dx

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def lbfgsb_optimize(h, n, target, alphas, penalties, maxiter, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a, pen in zip(alphas, penalties):
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        def obj_pen(x):
            corr = compute_corr(x)
            over = corr[_g_n - 1:]
            cm = float(np.max(over))
            sh = _g_alpha * (over - cm)
            sh = np.clip(sh, -700.0, 0.0)
            lse = cm + float(np.log(np.sum(np.exp(sh)))) / _g_alpha
            lse += pen * (np.sum(x) - _g_target) ** 2
            return lse
        def grad_pen(x):
            g = _lse_grad(x)
            g += 2.0 * pen * (np.sum(x) - _g_target)
            return g
        try:
            res = minimize(obj_pen, h_cur, jac=grad_pen, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def nesterov_active(h, n, target, n_steps, top_k, lr_init, mu, bt_max, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    h_prev = h_cur.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:top_k]
        mn = over[top_idx[0]]
        diffs = (over[top_idx] - mn) / 0.003
        weights = np.exp(diffs)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            jp = np.arange(0, n - k)
            grad[jp] += wk * (1.0 - h_cur[jp + k])
            jn = np.arange(k, n)
            grad[jn] -= wk * h_cur[jn - k]
        grad *= dx
        h_la = h_cur + mu * (h_cur - h_prev)
        h_prev = h_cur.copy()
        accepted = False
        for bt in range(bt_max):
            h_try = h_la - lr * grad
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < c_curr - 1e-15:
                h_cur = h_try
                accepted = True
                break
            lr *= 0.5
        if not accepted:
            h_cur = h_la
            lr *= 0.8
            if lr < 1e-14:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coordinated_perturb(h, n, target, rng):
    mode = rng.randint(0, 6)
    h_new = h.copy()
    if mode == 0:
        h_new += rng.randn(n) * 0.025
    elif mode == 1:
        x = np.linspace(0, 2, n)
        f = rng.uniform(1.0, 5.0)
        h_new += 0.06 * np.sin(f * np.pi * x + rng.uniform(0, 2 * np.pi))
    elif mode == 2:
        idx = rng.choice(n, rng.randint(8, 25), replace=False)
        h_new[idx] += rng.randn(len(idx)) * 0.12
    elif mode == 3:
        bs = rng.randint(8, 25)
        s1 = rng.randint(0, n - 2 * bs)
        s2 = s1 + rng.randint(bs + 1, n - bs - s1)
        h_new[s1:s1 + bs], h_new[s2:s2 + bs] = h_new[s2:s2 + bs].copy(), h_new[s1:s1 + bs].copy()
    elif mode == 4:
        x = np.linspace(0, 2, n)
        h_new += 0.03 * np.cos(rng.uniform(3, 8) * np.pi * x + rng.uniform(0, 2 * np.pi))
        idx = rng.choice(n, 5, replace=False)
        h_new[idx] += rng.randn(5) * 0.08
    else:
        h_new += rng.randn(n) * 0.015
        for _ in range(3):
            blk = rng.randint(5, 15)
            s = rng.randint(0, n - blk)
            h_new[s:s + blk] += rng.uniform(-0.05, 0.05)
    return project_feasible(h_new, target)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 8.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP initial refinement
    deadline1 = start_time + 100
    h1, sc1 = slsqp_optimize(best_h, n_final, target_final,
                              [200, 1000, 5000, 20000], 2500, 1e-16, deadline1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: L-BFGS-B with gradient (new, should be faster)
    deadline2 = min(start_time + 220, hard_deadline - 200)
    h2, sc2 = lbfgsb_optimize(best_h, n_final, target_final,
                               [5000, 20000, 100000, 500000],
                               [100.0, 1e6, 1e8, 1e10],
                               500, deadline2)
    if sc2 < best_score:
        best_score = sc2
        best_h = h2

    # Phase 3: Nesterov active set
    deadline3 = min(start_time + 350, hard_deadline - 150)
    for top_k in [3, 5, 8]:
        if tleft() < 40:
            break
        h3, sc3 = nesterov_active(best_h, n_final, target_final,
                                   300, top_k, 0.008, 0.9, 12,
                                   min(start_time + 350, deadline3))
        if sc3 < best_score:
            best_score = sc3
            best_h = h3
        h3, sc3 = lbfgsb_optimize(best_h, n_final, target_final,
                                   [50000, 200000, 1000000],
                                   [1e6, 1e8, 1e10],
                                   400, min(start_time + 370, hard_deadline - 130))
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Coarse resolution search
    if tleft() > 200:
        n_coarse = 200
        t_coarse = n_coarse / 2.0
        h_c = interpolate_to(best_h, n_coarse, t_coarse)
        h_c, sc_c = slsqp_optimize(h_c, n_coarse, t_coarse,
                                    [100, 500, 2000, 10000, 50000], 2000, 1e-15,
                                    min(start_time + 500, hard_deadline - 180))
        h_c, sc_c = nesterov_active(h_c, n_coarse, t_coarse,
                                     200, 5, 0.01, 0.9, 10,
                                     min(start_time + 540, hard_deadline - 160))
        h_fine = interpolate_to(h_c, n_final, target_final)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
        h_fine, sc_fine = lbfgsb_optimize(best_h, n_final, target_final,
                                           [20000, 100000, 500000],
                                           [1e5, 1e7, 1e9],
                                           400, min(start_time + 600, hard_deadline - 140))
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine

    # Phase 5: Structured perturbations + re-optimization
    if tleft() > 180:
        n_restarts = min(14, max(3, int(tleft() // 22)))
        for r in range(n_restarts):
            if tleft() < 30:
                break
            h_pert = coordinated_perturb(best_h, n_final, target_final, rng)
            h_r, sc_r = nesterov_active(h_pert, n_final, target_final,
                                         120, 5, 0.005, 0.9, 8,
                                         min(start_time + 600 + r * 22, hard_deadline - 25))
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
            h_r, sc_r = lbfgsb_optimize(best_h, n_final, target_final,
                                         [50000, 200000],
                                         [1e7, 1e9],
                                         300, min(start_time + 620 + r * 22, hard_deadline - 15))
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r

    # Phase 6: SLSQP with very high alpha
    if tleft() > 80:
        h6, sc6 = slsqp_optimize(best_h, n_final, target_final,
                                  [100000, 500000, 2000000], 2000, 1e-17,
                                  min(start_time + budget_s - 50, hard_deadline))
        if sc6 < best_score:
            best_score = sc6
            best_h = h6

    # Phase 7: Final Nesterov + L-BFGS-B
    if tleft() > 50:
        h7, sc7 = nesterov_active(best_h, n_final, target_final,
                                   200, 3, 0.003, 0.95, 15,
                                   min(start_time + budget_s - 30, hard_deadline))
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        h7, sc7 = lbfgsb_optimize(best_h, n_final, target_final,
                                   [500000, 2000000],
                                   [1e9, 1e11],
                                   300, min(start_time + budget_s - 12, hard_deadline))
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Final: one more SLSQP if time allows
    if tleft() > 20:
        h8, sc8 = slsqp_optimize(best_h, n_final, target_final,
                                  [1000000], 1000, 1e-17, hard_deadline)
        if sc8 < best_score:
            best_score = sc8
            best_h = h8

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        final_score = best_score
    return h_final, final_score, len(h_final)
