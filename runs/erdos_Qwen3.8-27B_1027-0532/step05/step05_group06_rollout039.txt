def binary_round(h, target, rng):
    """Round to binary and repair"""
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    # Adjust to meet target
    ones = int(h_bin.sum())
    target_int = int(target)
    if ones > target_int:
        # Turn off some ones
        one_idx = np.where(h_bin == 1.0)[0]
        excess = ones - target_int
        # Turn off the ones that are "least needed"
        # Use the original h values as a guide
        vals = h[one_idx]
        order = np.argsort(vals)[:excess]
        h_bin[one_idx[order]] = 0.0
    elif ones < target_int:
        # Turn on some zeros
        zero_idx = np.where(h_bin == 0.0)[0]
        deficit = target_int - ones
        vals = h[zero_idx]
        order = np.argsort(vals)[::-1][:deficit]
        h_bin[zero_idx[order]] = 1.0
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def weighted_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
    return grad

def lse_combined(x, alpha, target, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

def lse_grad_only(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = float(np.sum(x))
    grad += 2.0 * mu * (s - target)
    return grad

def lbfgs_step(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_combined, h, jac=lse_grad_only, method='L-BFGS-B',
                       bounds=bounds, args=(alpha, target, mu),
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h

Each trial involves copying the array and computing the correlation, which is O(n) = 400. So each sweep is O(n^3) = 64M operations. That's maybe 0.5-2 seconds per sweep. We can do maybe 50-200 sweeps in the time budget.

But wait, I can optimize: instead of computing the full correlation for each trial, I can compute just the change in the correlation. For a move (remove 1 from position i, add 1 at position j):

The change in corr_k is:
- For the term involving position i losing its 1: -(1 - h_{i+k}) for each valid k
- For the term involving position j gaining a 1: +(1 - h_{j+k}) for each valid k

But this is still O(n) per trial (need to check all k). However, we only care about the MAXIMUM correlation. So we can be smarter:
- Compute the change for each k
- Check if any k's correlation increases above the current max
- If the new max is lower, accept

This is still O(n) per trial but with lower constant.

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)

    h = np.zeros(n)
    fill = int(target) / n
    period = max(2, int(1.0 / fill))
    ones_per_period = int(round(fill * period))
    ones_per_period = max(1, min(ones_per_period, period - 1))
    for start in range(0, n, period):
        end = min(start + ones_per_period, n)
        h[start:end] = 1.0
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS warmup
    t1 = start_time + 120
    h1, sc1 = multi_start_lse(h_init, target, [20, 100, 500, 2000],
                              250, t1, mu=50.0)
    if sc1 < best_score:
        best_h = h1
        best_score = sc1

    # Phase 2: Binary rounding + hill climbing
    t2 = start_time + 350
    h_bin = binary_round_keep_sum(best_h, target)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_h = h_bin.copy()
        best_score = sc_bin
    h_bc, sc_bc = binary_hill_climb(h_bin, target, max_sweeps=200, deadline=t2)
    if sc_bc < best_score:
        best_h = h_bc.copy()
        best_score = sc_bc

    # Phase 3: More aggressive binary search with random moves
    t3 = start_time + 500
    for rs in [1, 7, 13, 42]:
        if time.time() > t3:
            break
        h_rb, sc_rb = binary_hill_climb_random(best_h, target, max_trials=20000,
                                               deadline=t3, rng_seed=rs)
        if sc_rb < best_score:
            best_h = h_rb.copy()
            best_score = sc_rb

    # Phase 4: Continuous optimization from best binary
    t4 = start_time + 650
    h4, sc4 = multi_start_lse(best_h, target, [100, 1000, 10000, 50000],
                              200, t4, mu=100.0)
    if sc4 < best_score:
        best_h = h4
        best_score = sc4

    # Phase 5: Subgradient with adaptive focus
    t5 = start_time + 850
    h5, sc5 = subgradient_optimize(best_h, target, t5, beta_start=8.0, step_start=0.015)
    if sc5 < best_score:
        best_h = h5
        best_score = sc5

    # Phase 6: Top-lag equalization
    t6 = start_time + 950
    h6, sc6 = top_lag_subgradient(best_h, target, n_top=5, n_steps=300,
                                  deadline=t6, lr=0.004)
    if sc6 < best_score:
        best_h = h6
        best_score = sc6
    h6b, sc6b = multi_start_lse(best_h, target, [5000, 50000], 150, t6, mu=50.0)
    if sc6b < best_score:
        best_h = h6b
        best_score = sc6b

    # Phase 7: Diverse restarts
    t7 = start_time + 1050
    restart_starts = [
        make_binary_start(n, target, np.random.RandomState(99)),
        periodic_start(n, target),
        make_binary_start(n, target, np.random.RandomState(123)),
    ]
    for rs_h in restart_starts:
        if time.time() > t7:
            break
        rs_h = project_feasible(rs_h, target)
        rs_opt, rs_sc = multi_start_lse(rs_h, target, [100, 2000, 20000],
                                        150, t7, mu=50.0)
        if rs_sc < best_score:
            best_h = rs_opt
            best_score = rs_sc

    # Phase 8: Upsample to 2x
    n2 = 2 * n
    target2 = n2 / 2.0
    t8 = start_time + budget_s - 80
    if time.time() < t8 - 30:
        h_fine = upsample(best_h, n2)
        h_fine = project_feasible(h_fine, target2)
        h8, sc8 = multi_start_lse(h_fine, target2, [50, 500, 5000, 20000],
                                  150, t8, mu=50.0)
        if sc8 < best_score:
            best_h = h8
            best_score = sc8
            n = n2
            target = target2

    # Phase 9: Final subgradient at current resolution
    if time.time() < deadline - 50:
        h9, sc9 = subgradient_optimize(best_h, target, deadline - 50,
                                       beta_start=15.0, step_start=0.01)
        if sc9 < best_score:
            best_h = h9
            best_score = sc9

    # Phase 10: Final coordinate descent
    if time.time() < deadline - 20:
        h10, sc10 = coordinate_descent_focused(best_h, target, n_sweeps=2,
                                               top_frac=0.5, deadline=deadline)
        if sc10 < best_score:
            best_h = h10
            best_score = sc10

    # Final: one more L-BFGS pass
    if time.time() < deadline - 5:
        h_fin, sc_fin = multi_start_lse(best_h, target, [10000], 100,
                                        deadline, mu=200.0)
        if sc_fin < best_score:
            best_h = h_fin
            best_score = sc_fin

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
