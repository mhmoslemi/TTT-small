Wait, let me be more careful. c(k) = Σ_i h(i)(1-h(i+k)) where the sum is over valid i.

∂c(k)/∂h[i] = (1-h(i+k)) [from the i-th term] + (-h(i-k)) [from the (i-k)-th term, where h(i-k)(1-h(i)) contributes -h(i-k) when h(i) changes]

So ∂c(k)/∂h[i] = (1-h(i+k)) - h(i-k) (with boundary terms being 0).

This can be computed for all k simultaneously:
∂c(k)/∂h[i] = (1-h(i+k)) - h(i-k)

For a fixed i, as k varies, this is:
- (1-h(i+k)) is a shifted version of (1-h)
- -h(i-k) is a reversed shifted version of h

So the gradient of f_α with respect to h[i] is:
∇_i f_α = Σ_k w_k * [(1-h(i+k)) - h(i-k)]
where w_k = exp(α*c(k)) / Σ_j exp(α*c(j))

This is a convolution-like operation. Specifically:
∇_i f_α = Σ_k w_k * (1-h(i+k)) - Σ_k w_k * h(i-k)
         = (w * (1-h))_i - (w * h)_{i-reversed}

where * denotes convolution. This can be computed in O(n log n) via FFT!

Let me define:
- w[k] = exp(α * c(k)) / Σ exp(α*c(j)) for k = -(n-1) to n-1
- G1[i] = Σ_k w[k] * (1-h[i+k]) = (w * (1-h))[i] (cross-correlation)
- G2[i] = Σ_k w[k] * h[i-k] = (w_rev * h)[i] where w_rev[k] = w[-k]

Then ∇_i f_α = G1[i] - G2[i]

Both G1 and G2 can be computed via FFT in O(n log n).

This makes the SLSQP much faster since we can provide the analytic gradient.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta_single(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    result[mask] -= 1.0
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    result[mask] += 1.0
    return result

def compute_delta_multi(h, pos_1, pos_0):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def greedy_polish(h, max_iters, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    positions = np.arange(n)
    for it in range(max_iters):

        max_val = float(full.max())

        k_star = max_idx - (n-1)
        if k_star == 0:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(80, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(80, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta_single(h, i, j)
                if (full + delta).max() < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta

                    if new_c5 < best_c5:

                    found = True
                    break
            if found:
                break
        if not found:
            break
    return h, best_c5

def _soft_obj(h, alpha):
    corr = full_corr_fast(h)
    mc = corr.max()
    shifted = alpha * (corr - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / alpha

def _soft_grad(h, alpha):
    n = len(h)
    corr = full_corr_fast(h)
    mc = corr.max()
    shifted = alpha * (corr - mc)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / exp_shifted.sum()
    i_idx = np.arange(n)
    k_idx = np.arange(-(n-1), n)
    I, K = np.meshgrid(i_idx, k_idx, indexing='ij')
    ik = I + K
    ikm = I - K
    valid = (ik >= 0) & (ik < n) & (ikm >= 0) & (ikm < n)
    diff = np.zeros((n, 2*n-1))
    diff[valid] = (1.0 - h[ik[valid]]) - h[ikm[valid]]
    return (weights[None, :] * diff).sum(axis=1)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            def obj_wrapper(x, a):
                return _soft_obj(x, a)
            def jac_wrapper(x, a):
                return _soft_grad(x, a)
            res = minimize(obj_wrapper, h_cur, args=(alpha,), jac=jac_wrapper,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
