        _g['active'] = None
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', jac=_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def lbfgs_opt(h, n, target, alpha, mu, maxiter=300):
    _g['alpha'] = alpha
    _g['target'] = target
    _g['n'] = n
    _g['mu'] = mu
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_obj_pen, h, method='L-BFGS-B', jac=_grad_pen,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient(h, n, target, n_steps, temp, lr_init, lr_decay, deadline, top_k=6):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stagnation = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
            stagnation = 0
        else:
            stagnation += 1
        if stagnation > 300:
            lr *= 0.85
            if lr < 1e-13:
                break
            stagnation = 0
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            break
        weights /= w_sum
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n-k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n-ell])
                g[:n-ell] -= wk * h_cur[ell:]
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new <= c_curr + 1e-14:
            h_cur = h_new
        else:
            lr *= lr_decay
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    no_improve = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n-k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n-k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n-ell])
            g[:n-ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            no_improve = 0
            eps = min(eps * 1.03, 0.012)
        else:
            eps *= 0.88
            no_improve += 1
            if no_improve > 200 or eps < 1e-10:
                break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lag_cycling(h, n, target, alpha, top_m, n_rounds, maxiter_per, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_m]
        for skip in range(min(top_m, 5)):
            if time.time() > deadline:
                break
            active = np.array(top_idx)
            active = np.delete(active, skip)
            h_new = slsqp_opt(h_cur, n, target, alpha, active=active,
                              maxiter=maxiter_per, ftol=1e-13)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
    return h_cur, best_score

def structured_perturb(h, n, target, rng, scale=0.1):
    h_new = h.copy()
    n_blocks = rng.randint(2, 6)
    for _ in range(n_blocks):
        start = rng.randint(0, n - 20)
        length = rng.randint(10, min(40, n - start))
        shift = rng.uniform(-scale, scale)
        h_new[start:start+length] += shift
    h_new = project_feasible(h_new, target)
    return h_new

def smooth_perturb(h, n, target, rng, n_modes=3):
    freqs = rng.randint(1, n // 4, size=n_modes)
    phases = rng.uniform(0, 2 * np.pi, size=n_modes)
    amps = rng.uniform(0.05, 0.15, size=n_modes)
    x = np.arange(n) / n
    pert = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        pert += a * np.sin(2 * np.pi * f * x + p)
    h_new = h + pert
    h_new = project_feasible(h_new, target)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B warmup
    alphas = [20, 60, 200, 600]
    for alpha in alphas:
        if time.time() > deadline - 500:
            break
        for mu in [10.0, 100.0]:
            if time.time() > deadline - 500:
                break
            h_new = lbfgs_opt(best_h, n, target, alpha, mu, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP refinement
    for alpha in [300, 2000]:
        if time.time() > deadline - 420:
            break
        h_new = slsqp_opt(best_h, n, target, alpha, maxiter=3000, ftol=1e-14)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Subgradient polish (multiple temperatures)
    if time.time() < deadline - 380:
        for temp, lri in [(0.005, 0.003), (0.002, 0.002), (0.001, 0.001)]:
            if time.time() > deadline - 370:
                break
            h_sg, sc_sg = subgradient(best_h, n, target, 3000, temp, lri, 0.995, deadline - 370, top_k=6)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 4: Worst lag surgery
    if time.time() < deadline - 340:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 600, 0.004, deadline - 340)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 5: Lag cycling to break degeneracy
    if time.time() < deadline - 280:
        h_cyc, sc_cyc = lag_cycling(best_h, n, target, 2000, 10, 3, 600, deadline - 280)
        if sc_cyc < best_score:
            best_score = sc_cyc
            best_h = h_cyc
        if time.time() < deadline - 260:
            h_sg, sc_sg = subgradient(best_h, n, target, 2000, 0.001, 0.0015, 0.995, deadline - 260, top_k=5)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 6: Structured perturbations + re-optimization
    if time.time() < deadline - 200:
        for attempt in range(5):
            if time.time() > deadline - 160:
                break
            if attempt % 2 == 0:
                h_pert = smoothed_perturb(best_h, n, target, rng)
            else:
                h_pert = structured_perturb(best_h, n, target, rng, scale=0.12)
            h_ref = lbfgs_opt(h_pert, n, target, 500, 50.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                h_ref2 = slsqp_opt(best_h, n, target, 3000, maxiter=2000, ftol=1e-14)
                sc2 = compute_c5(h_ref2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_ref2
                h_sg, sc_sg = subgradient(best_h, n, target, 1000, 0.001, 0.001, 0.995, deadline - 155, top_k=5)
                if sc_sg < best_score:
                    best_score = sc_sg
                    best_h = h_sg

    # Phase 7: Resolution increase to n=600
    if time.time() < deadline - 350:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)

        for alpha in [30, 100, 500, 2000]:
            if time.time() > deadline - 130:
                break
            for mu in [10.0, 100.0]:
                if time.time() > deadline - 130:
                    break
                h2_new = lbfgs_opt(best_h2, n2, target2, alpha, mu, 200)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

        if time.time() < deadline - 100:
            h2_new = slsqp_opt(best_h2, n2, target2, 2000, maxiter=2000, ftol=1e-14)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new

        if time.time() < deadline - 80:
            h2_sg, sc2_sg = subgradient(best_h2, n2, target2, 2000, 0.002, 0.002, 0.995, deadline - 80, top_k=6)
            if sc2_sg < best_score2:
                best_score2 = sc2_sg
                best_h2 = h2_sg

        if time.time() < deadline - 60:
            h2_surg, sc2_surg = worst_lag_surgery(best_h2, n2, target2, 400, 0.003, deadline - 60)
            if sc2_surg < best_score2:
                best_score2 = sc2_surg
                best_h2 = h2_surg

        if time.time() < deadline - 40:
            h2_cyc, sc2_cyc = lag_cycling(best_h2, n2, target2, 2000, 8, 2, 500, deadline - 40)
            if sc2_cyc < best_score2:
                best_score2 = sc2_cyc
                best_h2 = h2_cyc

        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2

    # Phase 8: Try n=800 if time permits
    if time.time() < deadline - 300:
        n3 = 800
        target3 = n3 / 2.0
        h3 = interpolate_to(best_h, n3, target3)
        best_h3 = h3.copy()
        best_score3 = compute_c5(h3)

        for alpha in [20, 80, 400]:
            if time.time() > deadline - 80:
                break
            for mu in [10.0, 100.0]:
                if time.time() > deadline - 80:
                    break
                h3_new = lbfgs_opt(best_h3, n3, target3, alpha, mu, 150)
                sc3 = compute_c5(h3_new)
                if sc3 < best_score3:
                    best_score3 = sc3
                    best_h3 = h3_new

        if time.time() < deadline - 60:
            h3_new = slsqp_opt(best_h3, n3, target3, 1500, maxiter=1500, ftol=1e-13)
            sc3 = compute_c5(h3_new)
            if sc3 < best_score3:
                best_score3 = sc3
                best_h3 = h3_new

        if time.time() < deadline - 40:
            h3_sg, sc3_sg = subgradient(best_h3, n3, target3, 1500, 0.002, 0.002, 0.995, deadline - 40, top_k=6)
            if sc3_sg < best_score3:
                best_score3 = sc3_sg
                best_h3 = h3_sg

        if best_score3 < best_score:
            best_score = best_score3
            best_h = best_h3

    # Phase 9: Final intensive subgradient + surgery on best solution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time.time() < deadline - 40:
        h_sg, sc_sg = subgradient(best_h, n_cur, target_cur, 3000, 0.0008, 0.001, 0.998, deadline - 30, top_k=4)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if time.time() < deadline - 15:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_cur, target_cur, 300, 0.002, deadline - 15)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 10: Final SLSQP polish
    if time.time() < deadline - 5:
        h_new = slsqp_opt(best_h, n_cur, target_cur, 5000, maxiter=1500, ftol=1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
