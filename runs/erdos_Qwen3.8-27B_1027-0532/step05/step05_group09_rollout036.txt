```python
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)

        improved = False

            # Compute A_k and B_k for all k
            # g_k = sum_{j=0}^{n-1-k} h[j]*(1-h[j+k]) * dx
            # Term involving h[i]:
            # If i <= n-1-k: h[i]*(1-h[i+k]) -> A_k += (1-h[i+k]), B_k unchanged
            # If k <= i <= n-1 (i.e., i-k >= 0 and i-k <= n-1-k): -h[i-k]*h[i] -> A_k += -h[i-k]

            # For k where i <= n-1-k (i.e., k <= n-1-i): contribution is h[i]*(1-h[i+k])
            # For k where k <= i (and i-k <= n-1-k, i.e., i-k+k = i <= n-1 which is always true): contribution is -h[i-k]*h[i]

            # B_k = g_k - h[i]*A_k  (the part of g_k not involving h[i])

            A[:n-i] = (1.0 - h_cur[i:n])  # length n-i

                else:

                    break

                    improved = True

        if not improved and sweep > 10:
```

```python
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:].astype(float)
    best_h = h_cur.copy()
    best_score = float(np.max(corr)) * dx
    cur_score = best_score
    updates_since_proj = 0
    
    for sweep in range(max_sweeps):
        for i in range(n):
            # Compute A for coordinate i
            A = np.zeros(n)
            n_fwd = n - i
            A[:min(n_fwd, n)] = 1.0 - h_cur[i:i+n_fwd] if n_fwd > 0 else 0
            n_bwd = i + 1
            if n_bwd > 0:
                A[:n_bwd] -= h_cur[:i+1][::-1]
            
            B = corr - h_cur[i] * A
            
            # Bisection to find optimal t
            lo_t, hi_t = 0.0, 1.0
            for _ in range(40):
                mid_t = (lo_t + hi_t) * 0.5
                vals = A * mid_t + B
                slope = A[np.argmax(vals)]
                if slope > 1e-12:
                    hi_t = mid_t
                elif slope < -1e-12:
                    lo_t = mid_t
                else:
                    break
            t_opt = (lo_t + hi_t) * 0.5
            
            delta = t_opt - h_cur[i]
            if abs(delta) < 1e-10:
                continue
            
            corr += delta * A
            h_cur[i] = t_opt
            updates_since_proj += 1
            
            if updates_since_proj >= project_every:
                h_cur = project_feasible(h_cur, target)
                corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:].astype(float)
                updates_since_proj = 0
                sc = float(np.max(corr)) * dx
                if sc < best_score:
                    best_score = sc
                    best_h = h_cur.copy()
                cur_score = sc

    h_cur = project_feasible(h_cur, target)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:].astype(float)

    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()
```

```python
def coord_gradient(h, j, full_corr):
    """Compute d(Full[m])/d(h[j]) for all m, and the base values B[m] = Full[m] - h[j]*d(Full[m])/d(h[j])"""
    n = len(h)
    m_values = np.arange(2*n-1)
    # First term: I(j ≤ m ≤ j+n-1) * (1 - h[(n-1)+j-m])
    # m ranges from j to j+n-1
    A = np.zeros(2*n-1)
    if j < 2*n-1:
        m_lo = max(j, 0)
        m_hi = min(j+n-1, 2*n-2)
        if m_lo <= m_hi:
            indices = np.arange(m_lo, m_hi+1)
            h_indices = (n-1) + j - indices  # k1 values
            A[indices] += (1.0 - h[h_indices])
    
    # Second term: -I((n-1)-j ≤ m ≤ 2(n-1)-j) * h[j+m-(n-1)]
    m_lo2 = max((n-1)-j, 0)
    m_hi2 = min(2*(n-1)-j, 2*n-2)
    if m_lo2 <= m_hi2:
        indices2 = np.arange(m_lo2, m_hi2+1)
        h_indices2 = j + indices2 - (n-1)  # k2 values
        A[indices2] -= h[h_indices2]
    
    B = full_corr - h[j] * A
    return A, B
```

```python
def min_max_linear(A, B, lo=0.0, hi=1.0):
    """Find t in [lo,hi] that minimizes max_m (A[m]*t + B[m])"""
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = A * mid + B
        slope = A[np.argmax(vals)]
        if slope > 1e-14:
            hi = mid
        elif slope < -1e-14:
            lo = mid
        else:
            return mid
    return (lo + hi) * 0.5
```
