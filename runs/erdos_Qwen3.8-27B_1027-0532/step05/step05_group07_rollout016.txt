```python
def find_optimal_t(A_top, B_top, t_min, t_max):
    # A_top, B_top: arrays of coefficients for top correlations
    # f(t) = max_k (A_top[k] + B_top[k] * t)
    # Find t in [t_min, t_max] that minimizes f(t)
    
    candidates = [t_min, t_max]
    n_top = len(A_top)
    for i in range(n_top):
        for j in range(i+1, n_top):
            if abs(B_top[i] - B_top[j]) > 1e-14:
                t_cross = (A_top[j] - A_top[i]) / (B_top[i] - B_top[j])
                if t_min <= t_cross <= t_max:
                    candidates.append(t_cross)
    
    best_t = t_min
    best_val = float('inf')
    for t in candidates:
        val = max(A_top + B_top * t)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val
```

Actually, I can be smarter. Instead of evaluating at ALL crossing points, I can use the fact that the function f(t) = max_k (D_k + E_k * t) is piecewise linear and convex (as a function of t). So I can use a ternary search or binary search on the derivative.

f(t) = max_k (D_k + E_k * t)
f'(t) (subgradient) = {E_k : D_k + E_k * t = f(t)}

The minimum is where 0 is in the subgradient, i.e., where there exist k1, k2 with E_{k1} < 0 < E_{k2} and D_{k1} + E_{k1}*t = D_{k2} + E_{k2}*t.

This is equivalent to the crossing point method. But I can find it more efficiently:

1. Sort the curves by their value at t_min
2. Walk from t_min to t_max, maintaining the upper envelope
3. The minimum is where the slope changes from negative to positive

Actually, the simplest efficient method:
1. At t_min, find which k achieves the max (call it k*)
2. If E_{k*} < 0, the function is decreasing, so move right
   Find where k* is no longer the max: solve D_{k*} + E_{k*}*t = D_k + E_k*t for the first k that overtakes
3. If E_{k*} > 0, the function is increasing, so the minimum is at t_min
4. If E_{k*} = 0, the function is flat (locally), minimum is anywhere

This gives an O(log K * K) algorithm per pair, which is much faster.

Let me implement this "upper envelope" method:

```python
def find_min_of_max_linear(D_sub, E_sub, t_min, t_max):
    """Find t in [t_min, t_max] that minimizes max_i (D_sub[i] + E_sub[i]*t).
    D_sub, E_sub are arrays for the subset of functions we're considering."""
    
    n_sub = len(D_sub)
    if n_sub == 0:
        return t_min, 0
    
    # Evaluate at t_min
    vals_tmin = D_sub + E_sub * t_min
    # Evaluate at t_max
    vals_tmax = D_sub + E_sub * t_max
    
    # If all slopes are >= 0, minimum is at t_min
    if np.all(E_sub >= -1e-15):
        return t_min, float(np.max(vals_tmin))
    # If all slopes are <= 0, minimum is at t_max
    if np.all(E_sub <= 1e-15):
        return t_max, float(np.max(vals_tmax))
    
    # Walk the upper envelope from t_min
    t = t_min
    for iteration in range(n_sub):
        vals = D_sub + E_sub * t
        k_star = int(np.argmax(vals))
        if E_sub[k_star] >= -1e-14:
            # Function is non-decreasing at t, so minimum is at t
            return t, float(vals[k_star])
        # Function is decreasing, find where another curve overtakes k_star
        # Solve D_{k*} + E_{k*}*t' = D_j + E_j*t' for the smallest t' > t
        # t' = (D_j - D_{k*}) / (E_{k*} - E_j)
        other = np.delete(np.arange(n_sub), k_star)
        denom = E_sub[k_star] - E_sub[other]
        # Only consider where denom > 0 (i.e., E_j > E_{k*})
        valid = denom > 1e-14
        if not np.any(valid):
            return t_max, float(np.max(D_sub + E_sub * t_max))
        num = D_sub[other[valid]] - D_sub[k_star]
        t_cands = num / denom[valid]
        t_cands = t_cands[t_cands > t + 1e-15]
        if len(t_cands) == 0:
            return t_max, float(np.max(D_sub + E_sub * t_max))
        t_new = float(np.min(t_cands))
        if t_new >= t_max:
            return t_max, float(np.max(D_sub + E_sub * t_max))
        t = min(t_new + 1e-12, t_max)  # Slightly past the crossing
    
    return t, float(np.max(D_sub + E_sub * t))
```

```python
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        def obj(x):
            corr_x = compute_corr(x)
            c_max = float(np.max(corr_x))
            shifted = alpha * (corr_x - c_max)
            shifted = np.clip(shifted, -600.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        def cons_eq(x):
            return np.sum(x) - target

        def cons_jac(x):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Fast coordinate descent
    deadline_p1 = start_time + (budget_s - 25.0) * 0.25
    h_cd, sc_cd = fast_coord_desc(best_h, n, target, deadline_p1, rng, top_K=40)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: SLSQP polish
    if tleft() > 100:
        h_ref, sc_ref = slsqp_polish(best_h, n, target, [200, 800, 3000], 2000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 3: More coordinate descent with subgradient-guided perturbation
    if tleft() > 200:
        deadline_p3 = start_time + (budget_s - 25.0) * 0.45
        h_cd, sc_cd = fast_coord_desc(best_h, n, target, deadline_p3, rng, top_K=50, recompute_every=20)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 4: Subgradient-guided moves
    if tleft() > 250:
        corr = compute_corr(best_h)
        for step in range(30):
            if time.time() > start_time + (budget_s - 25.0) * 0.55:
                break
            sg = subgradient_direction(best_h, corr, n, top_k=20)
            sg_norm = np.linalg.norm(sg)
            if sg_norm < 1e-10:
                break
            lr = 0.005 / (1.0 + step * 0.1)
            h_trial = project_feasible(best_h - lr * sg, target)
            h_trial, sc_trial = fast_coord_desc(h_trial, n, target,
                                               start_time + (budget_s - 25.0) * 0.55 + (time.time() - start_time - (budget_s - 25.0) * 0.55) * 0.3,
                                               rng, top_K=30, recompute_every=30)
            if sc_trial < best_score:
                best_score = sc_trial
                best_h = h_trial
                corr = compute_corr(best_h)
            else:
                lr *= 0.5

    # Phase 5: Multi-scale
    if tleft() > 300:
        n_coarse = max(n // 5, 40)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_coarse = start_time + (budget_s - 25.0) * 0.65
        h_coarse, sc_coarse = fast_coord_desc(h_coarse, n_coarse, target_coarse, deadline_coarse, rng, top_K=30)
        h_up = interpolate_to(h_coarse, n, target)
        h_up, sc_up = slsqp_polish(h_up, n, target, [300, 1500], 1500)
        deadline_up = start_time + (budget_s - 25.0) * 0.73
        h_up, sc_up = fast_coord_desc(h_up, n, target, deadline_up, rng, top_K=40)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 6: Diverse multi-start
    deadline_p6 = start_time + (budget_s - 25.0) * 0.85
    x = np.linspace(0, 2, n)
    for attempt in range(10):
        if tleft() < 60:
            break
        if attempt == 0:
            h_pert = best_h + rng.randn(n) * 0.04
        elif attempt == 1:
            h_pert = best_h + 0.05 * np.sin(3 * np.pi * x / 2) + rng.randn(n) * 0.01
        elif attempt == 2:
            h_pert = best_h - 0.04 * np.sin(5 * np.pi * x / 2) + rng.randn(n) * 0.01
        elif attempt == 3:
            h_pert = best_h + rng.randn(n) * 0.07
        elif attempt == 4:
            h_pert = np.clip(best_h * 0.7 + 0.3 * 0.5, 0, 1)
        elif attempt == 5:
            h_pert = best_h + rng.randn(n) * 0.03
        elif attempt == 6:
            h_pert = best_h + 0.03 * np.cos(7 * np.pi * x / 2) + rng.randn(n) * 0.02
        elif attempt == 7:
            h_pert = best_h + rng.randn(n) * 0.05
        elif attempt == 8:
            h_pert = best_h + 0.04 * np.sin(2 * np.pi * x) * rng.randn(n)
        else:
            h_pert = best_h + rng.randn(n) * 0.06
        h_pert = project_feasible(h_pert, target)
        sub_deadline = min(deadline_p6, start_time + (budget_s - 25.0) * 0.85 - (time.time() - start_time))
        if sub_deadline > time.time() + 30:
            h_res, sc_res = fast_coord_desc(h_pert, n, target, sub_deadline, rng, top_K=35, recompute_every=25)
            if sc_res < best_score:
                best_score = sc_res
                best_h = h_res

    # Phase 7: Final SLSQP polish
    if tleft() > 50:
        h_final, sc_final = slsqp_polish(best_h, n, target, [500, 2000, 8000, 20000], 3000)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Phase 8: Final coordinate descent
    if tleft() > 20:
        h_final, sc_final = fast_coord_desc(best_h, n, target, start_time + budget_s - 12.0, rng, top_K=50, recompute_every=15)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
```
