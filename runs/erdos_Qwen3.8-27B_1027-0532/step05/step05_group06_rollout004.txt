def perturb_and_optimize(h_base, target, rng, strat, param, alpha, mu, deadline):
    n = len(h_base)
    h_pert = h_base.copy()
    if strat == 0:
        h_pert += rng.randn(n) * param
    elif strat == 1:
        n_flip = max(1, int(n * param))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = rng.uniform(0, 1, n_flip)
    elif strat == 2:
        freq = np.fft.rfftfreq(n)
        mask = np.zeros(len(freq), dtype=bool)
        mask[1:4] = True
        pert = np.fft.irfft(np.where(mask, rng.randn(len(freq)), 0), n)
        h_pert += pert * param
    elif strat == 3:
        blk = max(2, n // 16)
        n_blocks = n // blk
        block_vals = rng.uniform(0, 1, n_blocks)
        h_pert = np.repeat(block_vals, blk)[:n]
    h_pert = project_feasible(h_pert, target)
    h_new = lbfgs_pass(h_pert, target, alpha, mu, 400, deadline)
    h_new = lbfgs_pass(h_new, target, min(alpha * 10, 200000), mu, 300, deadline)
    sc = compute_c5(h_new)
    return h_new, sc

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    def tleft():
        return deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    mu = 100.0
    alphas_main = [2, 6, 18, 54, 162, 486, 1458, 4374, 13122, 39366, 118098, 354294]

    for a in alphas_main:
        if tleft() < 70:
            break
        h_new = lbfgs_pass(best_h, target, a, mu, 800, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 150:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=12, n_steps=600, deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        for a in [50000, 200000]:
            if tleft() < 60:
                break
            h_new = lbfgs_pass(best_h, target, a, mu, 500, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if tleft() > 150:
        h_cd, sc_cd = coordinate_descent(best_h, target, n_sweeps=3, deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    strat_params = [
        (0, 0.1), (0, 0.2), (1, 0.05), (2, 0.3), (3, 0.0),
        (0, 0.05), (1, 0.1), (0, 0.15)
    ]
    for i, (strat, param) in enumerate(strat_params):
        if tleft() < 80:
            break
        h_p, sc_p = perturb_and_optimize(best_h, target, rng, strat, param, 500, mu, deadline)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
            h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=10, n_steps=300, deadline=deadline)
            if sc_eq < best_score:
                best_score = sc_eq
                best_h = h_eq
            h_new = lbfgs_pass(best_h, target, 50000, mu, 400, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        h_fine = upsample(best_h, n2, target2)
        alphas_fine = [5, 20, 80, 320, 1280, 5120, 20480, 81920]
        for a in alphas_fine:
            if tleft() < 70:
                break
            h_fine = lbfgs_pass(h_fine, target2, a, mu, 500, deadline)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    if tleft() > 150:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=15, n_steps=500, deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        for a in [100000, 500000]:
            if tleft() < 60:
                break
            h_new = lbfgs_pass(best_h, target, a, mu, 500, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if tleft() > 180:
        h_cd, sc_cd = coordinate_descent(best_h, target, n_sweeps=5, deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 60:
        h_slq = slsqp_polish(best_h, target, 100000, 3000, deadline)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    if tleft() > 120:
        h_eq, sc_eq = top_lag_equalize(best_h, target, n_top=8, n_steps=400, deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_cd, sc_cd = coordinate_descent(best_h, target, n_sweeps=3, deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    if tleft() > 50:
        h_final_slq = slsqp_polish(best_h, target, 200000, 2000, deadline)
        sc = compute_c5(h_final_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_final_slq

    if tleft() > 60:
        h_cd, sc_cd = coordinate_descent(best_h, target, n_sweeps=2, deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
