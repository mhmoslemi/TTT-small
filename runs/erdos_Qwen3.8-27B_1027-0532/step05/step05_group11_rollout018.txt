def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def overlap_grad_single(h, k, n):
    """Gradient of overlap at lag k w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k >= n:
            return grad
        # valid i: 0 <= i < n-k
        vi = np.arange(0, n - k)
        grad[vi] += (1.0 - h[vi + k]) * dx
        grad[k:] += -h[:n-k] * dx
    else:
        m = -k
        if m >= n:
            return grad
        # valid i: m <= i < n
        vi = np.arange(m, n)
        grad[vi] += (1.0 - h[vi - m]) * dx
        grad[:n-m] += -h[m:] * dx
    return grad

def fast_overlap_grad_top(h, n, n_top):
    """Compute weighted gradient from top-N active lags."""
    overlaps = compute_overlaps(h)
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    top_lags = top_idx - n_half
    top_vals = overlaps[top_idx]
    
    # Exponential weighting
    w = 0.88 ** np.arange(n_top)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_lags):
        grad += w[i] * overlap_grad_single(h, k, n)
    
    return grad, top_vals, top_lags, w

    """Nesterov-accelerated subgradient with adaptive step size."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_val = 1.0
    step = 0
    
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Nesterov update
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        beta = (t_val - 1.0) / t_new
        y = h + beta * (h - y)
        y = np.clip(y, 0.0, 1.0)
        
        grad, top_vals, _, _ = fast_overlap_grad_top(y, n, n_top)

def slsqp_refine(h_start, n, target, alphas, maxiter=1000):
    """SLSQP on smooth approximation."""
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    
    def obj_and_grad(h, alpha):
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        c = corr * dx
        c_max = np.max(c)
        shifted = np.clip(alpha * (c - c_max), -500, 0)
        exp_c = np.exp(shifted)
        log_sum = c_max + np.log(np.sum(exp_c)) / alpha
        weights = exp_c / np.sum(exp_c)
        
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for idx in active:
            k = idx - (n - 1)
            grad += weights[idx] * alpha * overlap_grad_single(h, k, n)
        
        return float(log_sum), grad
    
    def con_fun(h):
        return np.sum(h) - target
    
    def con_jac(h):
        return np.ones(n)
    
    for alpha in alphas:
        try:
            res = minimize(lambda x: obj_and_grad(x, alpha)[0],
                         h_cur,
                         jac=lambda x: obj_and_grad(x, alpha)[1],
                         method='SLSQP', bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                         options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h
