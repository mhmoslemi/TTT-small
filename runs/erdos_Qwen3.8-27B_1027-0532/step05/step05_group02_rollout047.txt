    return np.clip(h - lam, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def fast_greedy(h, max_iters, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 400
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(25, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(25, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(40):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j, k_arr)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1
    return best_h, best_c5

def sa_block(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr, block_frac=0.2, max_block_len=10):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 60000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        use_block = rng.random() < block_frac
        if use_block:
            blen = rng.randint(3, min(max_block_len, n // 10) + 1)
            a = rng.randint(0, n - blen + 1)
            b = a + blen - 1
            h_new = h.copy()
            h_new[a:b+1] = h[a:b+1][::-1]
            s_old = h_new.sum()
            s_new = h_new.sum()
            if abs(s_old - s_new) > 1e-10:
                continue
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            delta = compute_delta(h, i, j, k_arr)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
    return best_h, best_c5

def sa_focused(h, n_iters, t_init, t_final, rng, time_limit, start_time, k_arr):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 4 == 0:
            top_k = 5
            top_indices = np.argsort(full)[::-1][:top_k]
            best_move_found = False
            for ti in top_indices:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                    best_move_found = True
                    break
            if not best_move_found:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j, k_arr)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def compute_smooth_gradient(h, alpha, corr_full, N_fft_cache=None):
    n = len(h)
    max_c = corr_full.max()
    exp_c = np.exp(alpha * (corr_full - max_c))
    w = exp_c / exp_c.sum()
    N_fft = 1
    while N_fft < 3 * n - 2:
        N_fft <<= 1
    W_fft = np.fft.fft(w, N_fft)
    h_pad = np.zeros(N_fft)
    h_pad[:n] = h
    h_fft = np.fft.fft(h_pad)
    conv_full = np.fft.ifft(W_fft * h_fft).real
    conv = conv_full[:n]
    grad = 1.0 - 2.0 * conv
    return grad

def continuous_refine(h0, target, start_time, time_limit, alphas):
    n = len(h0)
    h = h0.astype(float).copy()
    h = project_box_sum(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        phase_time = min(60.0, time_limit - (time.time() - start_time))
        lr = 2.0 / (alpha + 1.0)
        momentum = 0.7
        v = np.zeros(n)
        n_iter = int(phase_time * 200)
        n_iter = max(n_iter, 50)

        for it in range(n_iter):
            if time.time() - start_time > time_limit:
                break
            corr = full_corr_fast(h)
            grad = compute_smooth_gradient(h, alpha, corr)
            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_box_sum(h_new, target)
            h = h_new
            if it % 50 == 0 or it == n_iter - 1:
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    return best_h, best_score

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def embed_search(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    ratio = n_fine // n_coarse
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = min(start + ratio, n_fine)
        if ratio == 1:
            h_fine[i] = h_coarse[i]
        elif h_coarse[i] > 0.5:
            h_fine[start:end] = 1.0
        else:
            h_fine[start:end] = 0.0
    return project_binary(h_fine, target_fine)

def perturb_diverse(h, rng, n_pert):
    n = len(h)
    h_pert = h.copy()
    ones = np.where(h_pert == 1.0)[0]
    zeros = np.where(h_pert == 0.0)[0]
    for _ in range(n_pert):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_pert[i] = 0.0
        h_pert[j] = 1.0
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
    return h_pert

def perturb_block(h, rng, max_block_len):
    n = len(h)
    h_pert = h.copy()
    blen = rng.randint(3, min(max_block_len, n // 8) + 1)
    a = rng.randint(0, n - blen + 1)
    h_pert[a:a+blen] = h[a:a+blen][::-1]
    return h_pert

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    k_arr = np.arange(2 * n_init - 1) - (n_init - 1)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy
    allowed = min(60, tleft() * 0.05)
    h_g, c5_g = fast_greedy(best_h, max_iters=30000, rng=rng,
                            time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA with block moves
    allowed = min(200, tleft() * 0.18)
    n_sa = int(min(4000000, allowed * 60000))
    h_sa, c5_sa = sa_block(best_h, n_sa, 0.025, 0.0001, rng,
                           time_limit=allowed, start_time=start_time, k_arr=k_arr,
                           block_frac=0.25, max_block_len=12)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Focused SA
    allowed = min(200, tleft() * 0.18)
    n_sa2 = int(min(4000000, allowed * 60000))
    h_sa2, c5_sa2 = sa_focused(best_h, n_sa2, 0.02, 0.00005, rng,
                               time_limit=allowed, start_time=start_time, k_arr=k_arr)
    if c5_sa2 < best_score:
        best_score = c5_sa2
        best_h = h_sa2.copy()

    # Phase 4: Continuous refinement with gradient
    if tleft() > 100:
        target_use = float(n_init) / 2.0
        alphas = [5, 15, 50, 150, 500, 2000, 8000]
        allowed = min(300, tleft() * 0.3)
        h_ref, sc = continuous_refine(best_h, target_use, start_time, allowed, alphas)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: Project to binary and do focused SA
    if tleft() > 60:
        h_bin = project_binary(best_h, target_init)
        allowed = min(80, tleft() * 0.12)
        n_sa3 = int(min(2000000, allowed * 50000))
        h_sa3, c5_sa3 = sa_focused(h_bin, n_sa3, 0.015, 0.00003, rng,
                                   time_limit=allowed, start_time=start_time, k_arr=k_arr)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 6: Iterated perturb + SA (multi-start)
    for restart in range(8):
        if tleft() < 80:
            break
        # Diverse perturbation
        if restart % 3 == 0:
            h_pert = perturb_diverse(best_h, rng, rng.randint(5, 12))
        elif restart % 3 == 1:
            h_pert = perturb_block(best_h, rng, 15)
        else:
            h_pert = perturb_diverse(best_h, rng, 3)
            h_pert = perturb_block(h_pert, rng, 8)

        # SA from perturbed
        allowed = min(60, tleft() * 0.08)
        n_sa_r = int(min(1500000, allowed * 50000))
        h_sr, c5_sr = sa_block(h_pert, n_sa_r, 0.012, 0.00003, rng,
                               time_limit=allowed, start_time=start_time, k_arr=k_arr,
                               block_frac=0.2, max_block_len=8)
        if c5_sr < best_score:
            best_score = c5_sr
            best_h = h_sr.copy()

    # Phase 7: Coarse-to-fine
    if tleft() > 120:
        n_coarse = 100
        target_coarse = n_coarse // 2
        k_arr_c = np.arange(2 * n_coarse - 1) - (n_coarse - 1)
        x_new = np.linspace(0, 1, n_coarse)
        x_old = np.linspace(0, 1, n_init)
        h_coarse = np.interp(x_new, x_old, best_h)
        h_coarse_bin = project_binary(h_coarse, target_coarse)

        allowed = min(30, tleft() * 0.1)
        h_cg, c5_cg = fast_greedy(h_coarse_bin, max_iters=8000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr_c)
        allowed2 = min(30, tleft() * 0.08)
        n_sac = int(min(400000, allowed2 * 30000))
        h_cs, c5_cs = sa_block(h_cg, n_sac, 0.01, 0.0001, rng,
                               time_limit=allowed2, start_time=start_time, k_arr=k_arr_c,
                               block_frac=0.3, max_block_len=8)
        h_coarse_best = h_cs if c5_cs < c5_cg else h_cg

        h_fine = embed_search(h_coarse_best, n_init, target_init)
        allowed3 = min(30, tleft() * 0.08)
        h_fg, c5_fg = fast_greedy(h_fine, max_iters=12000, rng=rng,
                                  time_limit=allowed3, start_time=start_time, k_arr=k_arr)
        allowed4 = min(30, tleft() * 0.06)
        n_saf = int(min(400000, allowed4 * 30000))
        h_fs, c5_fs = sa_focused(h_fg, n_saf, 0.008, 0.00003, rng,
                                 time_limit=allowed4, start_time=start_time, k_arr=k_arr)
        h_embed = h_fs if c5_fs < c5_fg else h_fg
        c5_embed = c5_fs if c5_fs < c5_fg else c5_fg
        if c5_embed < best_score:
            best_score = c5_embed
            best_h = h_embed.copy()

    # Phase 8: Try different resolutions
    for n_try in [n_init + 4, n_init + 8, n_init - 4, n_init + 12, n_init - 8]:
        if tleft() < 50:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        k_arr_try = np.arange(2 * n_try - 1) - (n_try - 1)
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.06)
        h_g3, c5_g3 = fast_greedy(h_bin_try, max_iters=8000, rng=rng,
                                  time_limit=allowed, start_time=start_time, k_arr=k_arr_try)
        allowed2 = min(20, tleft() * 0.05)
        n_sa4 = int(min(200000, allowed2 * 30000))
        h_s4, c5_s4 = sa_focused(h_g3, n_sa4, 0.008, 0.00005, rng,
                                 time_limit=allowed2, start_time=start_time, k_arr=k_arr_try)
        c5_try = c5_s4 if c5_s4 < c5_g3 else c5_g3
        h_try = h_s4 if c5_s4 < c5_g3 else h_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 9: SLSQP refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_slsqp = [20, 80, 300, 1000, 4000, 12000]
        allowed = min(tleft() - 30, 200)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas_slsqp, 3000, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 10: Another gradient-based refinement from SLSQP result
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [30, 100, 400, 1500, 6000, 20000]
        allowed = min(tleft() - 20, 120)
        h_ref2, sc2 = continuous_refine(best_h, target_use, start_time, allowed, alphas2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 11: Final binary SA burst
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        k_arr_f = np.arange(2 * n_use - 1) - (n_use - 1)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(800000, max(50000, (tleft() - 15) * 40000)))
        h_sa_f, c5_f = sa_focused(h_bin_final, n_sa_f, 0.006, 0.00001,
                                  rng, time_limit=max(5.0, tleft() - 12),
                                  start_time=start_time, k_arr=k_arr_f)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 12: Final SLSQP polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [5000, 15000, 40000], 2000, 1e-15,
                                 time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_box_sum(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
