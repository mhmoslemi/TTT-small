import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_over_k(h, k):
    """Compute gradient of over[k] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        upper = n - 1 - k
        idx1 = np.arange(0, upper + 1)
        grad[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        grad[idx2] -= h[idx2 - k]
    return grad

def pair_coord_descend(h, n_pairs, rng):
    """Pair coordinate descent: swap mass between pairs to minimize max overlap."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_score = compute_c5(h)
    
    for _ in range(n_pairs):
        # Choose a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Compute over for all k
        corr = compute_corr(h)
        over = corr[n-1:]  # shape (n,)
        
        # Compute gradients with respect to h[i] and h[j]
        # ∂over[k]/∂h[i] and ∂over[k]/∂h[j]
        # For over[k] = sum_{t=0}^{n-1-k} h[t](1-h[t+k]):
        # ∂over[k]/∂h[x] = (1-h[x+k])*I(x <= n-1-k) - h[x-k]*I(x >= k)
        
        diff_grad = np.zeros(n)
        for k in range(n):
            # Gradient contribution for h[i]
            gi = 0.0
            gj = 0.0
            if i <= n - 1 - k:
                gi += (1.0 - h[i + k])
            if i >= k:
                gi -= h[i - k]
            if j <= n - 1 - k:
                gj += (1.0 - h[j + k])
            if j >= k:
                gj -= h[j - k]
            diff_grad[k] = gi - gj
        
        # Minimize max_k (over[k] + delta * diff_grad[k])
        # subject to max(0, -h[i]) <= delta <= min(1-h[i], h[j])
        lo = max(0.0, -h[i])  # = 0 since h[i] >= 0
        hi = min(1.0 - h[i], h[j])
        
        if hi - lo < 1e-15:
            continue
        
        def f_delta(d):
            return np.max(over + d * diff_grad)
        
        result = minimize_scalar(f_delta, bounds=(lo, hi), method='bounded', options={'xatol': 1e-12})
        delta = result.x
        
        if abs(delta) > 1e-10:
            h[i] += delta
            h[j] -= delta
            # Check improvement
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
            else:
                # Revert
                h[i] -= delta
                h[j] += delta
    
    return h, best_score
