```python
def compute_delta(h, a, b):
    n = len(h)
    delta = np.zeros(n)
    
    # a: 1 -> 0
    # -[k < n-a]*(1-h[a+k]) for k=1..n-a-1
    # +[k <= a]*h[a-k] for k=1..a
    
    k1 = np.arange(1, min(n-a, n))
    if len(k1) > 0:
        delta[k1] -= (1 - h[a + k1])
    
    k2 = np.arange(1, min(a+1, n))
    if len(k2) > 0:
        delta[k2] += h[a - k2]
    
    # b: 0 -> 1
    # +[k < n-b]*(1-h[b+k]) for k=1..n-b-1
    # -[k <= b]*h[b-k] for k=1..b
    
    k3 = np.arange(1, min(n-b, n))
    if len(k3) > 0:
        delta[k3] += (1 - h[b + k3])
    
    k4 = np.arange(1, min(b+1, n))
    if len(k4) > 0:
        delta[k4] -= h[b - k4]
    
    return delta
```

```python
def eval_piecewise(v, n, m):
    """Compute C5 for piecewise-constant h."""
    h = np.zeros(n)
    for j in range(m):
        start = j * n // m
        end = (j+1) * n // m
        h[start:end] = v[j]
    corr = np.correlate(h, 1-h, mode='full')
    dx = 2.0/n
    return np.max(corr[n-1:n-1+n]) * dx, h
```

Actually, the simplest approach: the function f(t) = max_k (A_k*t + B_k) is convex piecewise-linear. Its minimum is where the subgradient contains 0, i.e., where there exist k1, k2 with A_{k1} < 0 < A_{k2} (or A_{k1} = 0) and f(t*) = A_{k1}*t* + B_{k1} = A_{k2}*t* + B_{k2}.

I can find this by:
1. Sorting the lines by slope A_k
2. Computing the upper envelope
3. Finding the minimum of the upper envelope

This is the classical "lower envelope of lines" problem (or upper, depending on perspective), which can be solved in O(n log n) using a convex hull trick.

But implementing this correctly is a bit involved. Let me use a simpler approach: since f(t) is convex, I can use a ternary search or even a simple bisection on the derivative.

The subgradient of f at t is the set of A_k for k that achieve the maximum at t. f is minimized where 0 is in the subgradient, i.e., where the maximum is achieved by at least one k with A_k ≤ 0 and at least one k with A_k ≥ 0 (or exactly one with A_k = 0).

Algorithm:
1. Find t* by bisection: at each step, evaluate f and its subgradient at the midpoint
2. If all A_k at the maximizers are > 0, decrease t
3. If all A_k at the maximizers are < 0, increase t
4. If there's a mix, we're at the optimum (or very close)

This converges in O(log(1/ε)) iterations, each O(n). So total O(n log(1/ε)) per coordinate.

```python
def minimize_1d_convex(A, B):
    """Find t in [0,1] minimizing max_k (A[k]*t + B[k])."""
    n = len(A)
    
    # The function f(t) = max_k (A[k]*t + B[k]) is convex piecewise-linear
    # Minimum is where 0 is in the subgradient
    
    # Check boundaries
    f_0 = np.max(B)  # f(0)
    f_1 = np.max(A + B)  # f(1)
    
    # Subgradient at 0: set of A[k] where B[k] = f_0
    mask_0 = np.abs(B - f_0) < 1e-14
    if np.all(A[mask_0] >= 0):
        return 0.0  # minimum at t=0
    
    # Subgradient at 1: set of A[k] where A[k]+B[k] = f_1
    mask_1 = np.abs((A + B) - f_1) < 1e-14
    if np.all(A[mask_1] <= 0):
        return 1.0  # minimum at t=1
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(80):
        t = (lo + hi) / 2
        vals = A * t + B
        k_max = np.argmax(vals)
        f_t = vals[k_max]
        
        # Check if 0 is in subgradient
        mask = np.abs(vals - f_t) < 1e-12
        if np.any(A[mask] < 0) and np.any(A[mask] > 0):
            return t  # 0 is in subgradient
        elif np.all(A[mask] < 0):
            hi = t  # need to increase t
        elif np.all(A[mask] > 0):
            lo = t  # need to decrease t
        else:
            # A[k_max] == 0, so t is optimal
            return t
    
    return (lo + hi) / 2
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:2*n-1])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def minimize_1d_convex(A, B):
    """Find t in [0,1] minimizing max_k (A[k]*t + B[k])."""
    n = len(A)
    
    # Check boundaries
    f_0 = float(np.max(B))
    mask_0 = np.abs(B - f_0) < 1e-13
    if np.all(A[mask_0] >= -1e-15):
        return 0.0

            else:

        else:

def sa_binary(h0, n_steps, T_init, T_final, seed):

    rng = np.random.RandomState(seed)
    n = len(h0)

    if s != n / 2:

        zeros_idx = np.where(h == 0)[0]
        if s > n / 2:

            idx = rng.choice(ones_idx, to_flip, replace=False)
            h[idx] = 0.0
        else:
            to_flip = int(n / 2 - s)
            idx = rng.choice(zeros_idx, to_flip, replace=False)
            h[idx] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr[n-1:2*n-1].astype(float)
    curr_max = float(np.max(O[1:]))
    best_h = h.copy()
    best_max = curr_max
    
    ones_list = np.where(h == 1)[0].tolist()
    zeros_list = np.where(h == 0)[0].tolist()
    
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T = T_init * (T_final / T_init) ** frac
        
        a = ones_list[rng.randint(len(ones_list))]
        b = zeros_list[rng.randint(len(zeros_list))]

        if np.any(mask_second_b):
            kk = k_arr[mask_second_b]
            delta[kk] -= h[b - kk]

        new_max = float(np.max(new_O[1:]))
        
        if new_max <= curr_max or (T > 0 and rng.rand() < np.exp((curr_max - new_max) / T)):
            h[a], h[b] = 0.0, 1.0

    n = len(h)
```

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def minimize_1d(A, B):
    """Find t in [0,1] minimizing max_k (A[k]*t + B[k])."""
    f0 = float(np.max(B))
    m0 = B >= f0 - 1e-13
    if np.all(A[m0] >= -1e-14):
        return 0.0
    f1 = float(np.max(A + B))
    m1 = (A + B) >= f1 - 1e-13
    if np.all(A[m1] <= 1e-14):
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(100):
        t = 0.5 * (lo + hi)
        vals = A * t + B
        mv = float(np.max(vals))
        m = vals >= mv - 1e-12
        if np.any(A[m] < -1e-14) and np.any(A[m] > 1e-14):
            return t
        elif np.all(A[m] < 0):
            hi = t
        elif np.all(A[m] > 0):
            lo = t
        else:

    return 0.5 * (lo + hi)

def coord_descent(h, n_pass=50):
    n = len(h)
    target = n * 0.5

            k_arr = np.arange(n)
