```python
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_k(h):
    """Compute all correlation values (not just max)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        # Need to increase sum; add to smallest values
        idx = np.argsort(h)
        added = 0
        for i in idx:
            if added >= diff:
                break
            space = 1.0 - h[i]
            add = min(space, diff - added)
            h[i] += add
            added += add
    else:
        # Need to decrease sum; remove from largest values
        idx = np.argsort(h)[::-1]
        removed = 0
        for i in idx:
            if removed >= -diff:
                break
            space = h[i]
            rem = min(space, -diff - removed)
            h[i] -= rem
            removed += rem
    return np.clip(h, 0.0, 1.0)

def lp_with_cutting_plane(h_init, n, target, time_limit, max_iters=5000):
    """Alternating LP with proper cutting plane constraint generation."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    half_n = n - 1
    active_k = set()
    # Start with top constraints
    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-50:].tolist())
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                # c[k_idx] = sum_i h[i] * (1-h[i+k]) for valid i
                # = sum_i h[i] * g[i+k] where i ranges over [0,n) and i+k ranges over [0,n)
                # i in [max(0,-k), min(n, n-k))
                lo_i = max(0, -k)
                hi_i = min(n, n - k)
                coeff[lo_i:hi_i] = g[lo_i + k:hi_i + k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + 1e-6 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.5 * h + 0.5 * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:

            best_h = h.copy()

        # Keep active set manageable
        if len(active_k) > 100:
            active_k = set(np.argsort(corr)[-80:].tolist())
    
    return best_h, best_c5
```

```python
def binary_local_search(h_binary, n, time_limit, max_rounds=1000):
    start = time.time()
    h = h_binary.copy().astype(float)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = float(np.max(c)) * dx
    best_h = h.copy()
    half_n = n - 1
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    no_improve = 0
    round_num = 0
    
    while round_num < max_rounds:
        if time.time() - start > time_limit:
            break
        
        improved = False
        # Shuffle for diversity
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        
        for a in zeros:
            for b in ones:
                # Compute Δc for swap (a: 0→1, b: 1→0)
                k = np.arange(2*n - 1) - half_n  # actual lags
                
                # f1: (1-h[a+k]) for valid a+k
                f1 = np.zeros(2*n - 1)
                valid1 = (a + k >= 0) & (a + k < n)
                f1[valid1] = 1.0 - h[a + k[valid1]]
                
                # f2: (1-h[b+k]) for valid b+k
                f2 = np.zeros(2*n - 1)
                valid2 = (b + k >= 0) & (b + k < n)
                f2[valid2] = 1.0 - h[b + k[valid2]]
                
                # f3: h[a-k] for valid a-k
                f3 = np.zeros(2*n - 1)
                valid3 = (a - k >= 0) & (a - k < n)
                f3[valid3] = h[a - k[valid3]]
                
                # f4: h[b-k] for valid b-k
                f4 = np.zeros(2*n - 1)
                valid4 = (b - k >= 0) & (b - k < n)
                f4[valid4] = h[b - k[valid4]]
                
                dc = f1 - f2 - f3 + f4
                new_c = c + dc
                new_max = float(np.max(new_c))
                
                if new_max < np.max(c) - 1e-10:
                    # Accept the swap
                    h[a] = 1.0
                    h[b] = 0.0
                    c = new_c
                    improved = True
                    # Update ones and zeros
                    ones = ones[ones != b]
                    zeros = zeros[zeros != a]
                    break
            
            if improved:
                break
        
        if improved:
            no_improve = 0
            c5 = float(np.max(c)) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 10:
                break
        
        round_num += 1
    
    return best_h, best_c5
```

```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Distribute to smallest entries
        order = np.argsort(h)
        for i in order:
            if diff <= 0:
                break
            add = min(1.0 - h[i], diff)
            h[i] += add
            diff -= add
    else:
        # Remove from largest entries
        order = np.argsort(h)[::-1]
        for i in order:
            if diff >= 0:
                break
            rem = min(h[i], -diff)
            h[i] -= rem
            diff += rem
    return np.clip(h, 0.0, 1.0)

def binary_local_search(h_binary, n, time_limit, max_rounds=500, mode='first'):
    """Efficient binary local search with pairwise swaps."""
    start = time.time()
    h = h_binary.copy().astype(float)
    dx = 2.0 / n
    c = compute_corr(h).astype(float)
    best_c5 = float(np.max(c)) * dx
    best_h = h.copy()
    half_n = n - 1
    lags = np.arange(2*n - 1) - half_n  # actual lag values
    
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    
    no_improve = 0
    for round_num in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        improved = False
        rng_local = np.random.RandomState(round_num + 1)
        rng_local.shuffle(ones)
        rng_local.shuffle(zeros)
        
        for ai, a in enumerate(zeros):
            if time.time() - start > time_limit:
                break
            # Precompute const_a = f1_a - f3_a
            const_a = np.zeros(2*n - 1)
            v1 = (a + lags >= 0) & (a + lags < n)
            const_a[v1] = 1.0 - h[a + lags[v1]]
            v3 = (a - lags >= 0) & (a - lags < n)
            const_a[v3] -= h[a - lags[v3]]
            
            for bi, b in enumerate(ones):
                # Compute Δc = const_a - f2_b + f4_b
                f2 = np.zeros(2*n - 1)
                v2 = (b + lags >= 0) & (b + lags < n)

                f4[v4] = h[b - lags[v4]]
                
                dc = const_a - f2 + f4

                new_max = float(np.max(new_c))

                if new_max < cur_max - 1e-10:
                    h[a] = 1.0
                    h[b] = 0.0
                    c = new_c
                    improved = True
                    ones = np.delete(ones, bi)
                    zeros = np.delete(zeros, ai)
                    cur_c5 = float(np.max(c)) * dx
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = h.copy()
                    break
            
            if improved:
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
```

For continuous refinement, I'll use a different smooth max:
- Power-sum: f_p(h) = (sum_k max(0, c_k)^p)^(1/p) for large p
- This approximates max_k c_k as p → ∞
- For SLSQP, p=50 to 200 should work well

The gradient of f_p with respect to h_i:
∂f_p/∂h_i = (1/p) * (sum max(0,c_k)^p)^(1/p-1) * sum_k max(0,c_k)^(p-1) * ∂c_k/∂h_i

And ∂c_k/∂h_i can be computed from the correlation structure.

Actually, SLSQP computes gradients numerically by default, which is slow for 400 variables. Let me provide analytical gradients.

c[m] = sum_i h[i] * g[i+k] where k = m-(n-1) and g = 1-h
= sum_i h[i] * g[i+k]

∂c[m]/∂h[j] = g[j+k] - h[j-k] (with appropriate boundary handling)

Wait, let me be more careful:
c[m] = sum_i h[i] * g[i + (m-(n-1))]

where the sum is over i such that 0 ≤ i < n and 0 ≤ i + (m-(n-1)) < n.

Let k = m - (n-1).
c[m] = sum_{i: 0≤i<n, 0≤i+k<n} h[i] * g[i+k]

∂c[m]/∂h[j]:
- From h[j] in the first factor: g[j+k] if 0≤j+k<n
- From g[i+k] = 1-h[i+k], and i+k=j → i=j-k: -h[j-k] if 0≤j-k<n

So ∂c[m]/∂h[j] = g[j+k]·[0≤j+k<n] - h[j-k]·[0≤j-k<n]

This is the same formula I used for the binary search! Good.

For the power-sum smooth max:
F = (sum_m max(0, c[m])^p)^(1/p)
∂F/∂h[j] = (1/p) * F^(1-p) * sum_m max(0,c[m])^(p-1) * [c[m] > 0] * ∂c[m]/∂h[j]

For very large p, this concentrates on the term(s) where c[m] is maximum, recovering the gradient of the true max.

Let me now think about the continuous_refine function more carefully. I want to use SLSQP with:
- Objective: power-sum smooth max of the correlation values
- Constraint: sum(h) = target
- Bounds: 0 ≤ h[i] ≤ 1
- Analytical gradient

The power-sum approach:
F(h) = (sum_m c[m]^p)^(1/p) where c[m] = correlation at shift m

But we need max(0, c[m]) to avoid issues with negative values (though correlations should be positive for our problem).

Actually, for our problem, c[m] = sum h[i]*(1-h[i+k]) ≥ 0 always (since h ∈ [0,1]), so we don't need the max(0,·).

Gradient of F:
∂F/∂h[j] = (1/p) * F^(1-p) * sum_m c[m]^(p-1) * ∂c[m]/∂h[j]

Where ∂c[m]/∂h[j] = g[j+k_m]·[0≤j+k_m<n] - h[j-k_m]·[0≤j-k_m<n]
with k_m = m - (n-1).

For SLSQP with n=400 variables, providing an analytical gradient should make it much faster than numerical differentiation.

Let me also consider: should I use all 2n-1 correlation values, or just the "important" ones? The max is typically achieved at a shift close to n/2 (the middle of the full correlate output) or at small shifts. Using all values is safer but slower.

For the power-sum with p=100, the gradient is dominated by the top few values, so it's effectively a "soft" version of the max. This should work well.

Let me also add a "freezing" heuristic:
- After the binary search, identify h values that are 0 or 1

If we freeze 370 variables and only optimize 30, the QP is 31×31, which is much faster.
