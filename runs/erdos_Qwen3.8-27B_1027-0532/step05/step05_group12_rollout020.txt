**Approach: Bisection on C with feasibility check**
- Guess a value C
- Check if there exists h with max_k corr_k * dx <= C and constraints
- This is a feasibility problem: find h such that for all k, sum_i h[i]*(1-h[i+k]) * dx <= C

But this is non-convex (quadratic constraints), so bisection doesn't directly help.

**Approach: Reformulation as SOCP or SDP**
The constraint sum_i h[i]*(1-h[i+k]) <= C/dx can be written as:

The term sum_i h[i]*h[i+k] is quadratic. If we relax h[i]*h[i+k] >= 2h[i]h[i+k] - h[i]² - h[i+k]² (which is always true by AM-GM... wait, that's not a useful relaxation).

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h_init, n, target, max_sweeps, time_limit, start_time, deadline_offset):
    """Coordinate descent using linearity of corr in each h[i]."""
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        n_improved = 0
        for i in range(n):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            
            hi_save = h[i]
            
            # Compute correlation with h[i]=0 and h[i]=1
            h[i] = 0.0
            corr0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            corr1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = hi_save
            
            D = corr1 - corr0
            
            # Find t in [0,1] minimizing max(corr0 + t*D)
            # Try initial candidates
            best_t = hi_save
            best_val = float(np.max(corr0 + hi_save * D)) * dx
            
            trials = np.array([0.0, 0.2, 0.5, 0.8, 1.0])
            for t in trials:
                val = float(np.max(corr0 + t * D)) * dx
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine
            if 0.0 < best_t < 1.0:
                lo = max(0.0, best_t - 0.15)
                hi = min(1.0, best_t + 0.15)
                for _ in range(4):
                    cands = np.linspace(lo, hi, 15)
                    for t in cands:
                        val = float(np.max(corr0 + t * D)) * dx
                        if val < best_val:
                            best_val = val
                            best_t = t
                    mid = (lo + hi) / 2
                    lo = best_t - (hi - lo) * 0.3
                    hi = best_t + (hi - lo) * 0.3
                    lo = max(0.0, lo)
                    hi = min(1.0, hi)
            
            if abs(best_t - hi_save) > 1e-10:
                h[i] = best_t
                n_improved += 1
        
        # Project to maintain sum constraint
        h = project_feasible(h, target)
        score = compute_c5(h)
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
        
        if n_improved == 0:
            break
    
    return best_h, best_score
