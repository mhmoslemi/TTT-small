def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(x, target_sum, alpha, top_m, penalty_coeff):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    # Use top-m LSE for efficiency
    if top_m < len(over):
        idx = np.argpartition(over, -top_m)[-top_m:]
        vals = over[idx]
    else:
        vals = over
    c_max = float(np.max(vals))
    shifted = alpha * (vals - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    pen = penalty_coeff * (np.sum(h) - target_sum) ** 2
    return lse * (2.0/n) + pen

                shifted_b = alpha * (over_b - c_max_b)
                shifted_b = np.clip(shifted_b, -700.0, 0.0)
                lse_b = c_max_b + float(np.log(np.sum(np.exp(shifted_b)))) / alpha
                pen_b = pen_coeff * (np.sum(hf) - _bt) ** 2
                return lse_b * (2.0 / n) + pen_b

            try:
                res = minimize(_block_obj, x0, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * block_size,
                               options={'maxiter': 100, 'ftol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_top_m, _g_pen
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 12.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Fast L-BFGS-B polish (multiple alpha levels)
    if tleft() > 150:
        h_ref, sc = lbfgs_optimize(best_h, n_final, target_final,
                                   alphas=[100, 500, 2000, 8000],
                                   top_ms=[n_final, n_final, n_final, n_final],
                                   pen_coeffs=[5, 5, 10, 10],
                                   maxiter=500,
                                   deadline=min(start_time + 180, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Multi-lag descent + L-BFGS-B follow-up
    if tleft() > 200:
        for n_lags in [3, 5, 8]:
            if tleft() < 50:
                break
            h_md, sc_md = multi_lag_descent(best_h, n_final, target_final,
                                            max_steps=300, n_lags=n_lags,
                                            eps_init=0.003,
                                            deadline=min(start_time + 300, hard_deadline))
            if sc_md < best_score:
                best_score = sc_md
                best_h = h_md
        h_ref, sc = lbfgs_optimize(best_h, n_final, target_final,
                                   alphas=[2000, 8000, 20000],
                                   top_ms=[n_final, n_final, n_final],
                                   pen_coeffs=[10, 10, 20],
                                   maxiter=800,
                                   deadline=min(start_time + 420, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Subgradient with varying focus
    if tleft() > 150:
        for n_lags in [2, 3, 5, 10]:
            if tleft() < 40:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                             n_steps=300, n_lags=n_lags,
                                             lr_init=0.002, lr_decay=0.995,
                                             deadline=min(start_time + 500, hard_deadline))
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg
        h_ref, sc = lbfgs_optimize(best_h, n_final, target_final,
                                   alphas=[5000, 20000],
                                   top_ms=[n_final, n_final],
                                   pen_coeffs=[20, 20],
                                   maxiter=1000,
                                   deadline=min(start_time + 580, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Worst-lag surgery
    if tleft() > 150:
        for _ in range(3):
            if tleft() < 40:
                break
            h_surg, sc_surg = worst_lag_descent(best_h, n_final, target_final,
                                                max_steps=400, eps_init=0.001,
                                                deadline=min(start_time + 680, hard_deadline))
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg
        h_ref, sc = lbfgs_optimize(best_h, n_final, target_final,
                                   alphas=[10000, 50000],
                                   top_ms=[n_final, n_final],
                                   pen_coeffs=[50, 50],
                                   maxiter=1500,
                                   deadline=min(start_time + 780, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Multi-resolution approach
    if tleft() > 200:
        for n_coarse in [100, 200]:
            if tleft() < 80:
                break
            x_coarse = np.linspace(0, 1, n_coarse)
            x_fine = np.linspace(0, 1, n_final)
            h_coarse = np.interp(x_coarse, x_fine, best_h)
            h_coarse = project_feasible(h_coarse, n_coarse / 2.0)
            h_c_opt, sc_c = lbfgs_optimize(h_coarse, n_coarse, n_coarse / 2.0,
                                           alphas=[100, 500, 2000, 10000],
                                           top_ms=[n_coarse] * 4,
                                           pen_coeffs=[5, 5, 10, 10],
                                           maxiter=500,
                                           deadline=min(start_time + 860, hard_deadline))
            h_up = np.interp(x_fine, x_coarse, h_c_opt)
            h_up = project_feasible(h_up, target_final)
            h_ref, sc = lbfgs_optimize(h_up, n_final, target_final,
                                       alphas=[2000, 10000],
                                       top_ms=[n_final, n_final],
                                       pen_coeffs=[10, 20],
                                       maxiter=500,
                                       deadline=min(start_time + 920, hard_deadline))
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 6: Random perturbation restarts
    if tleft() > 150:
        scales = [0.015, 0.03, 0.008, 0.025, 0.02, 0.04]
        for attempt in range(len(scales)):
            if tleft() < 40:
                break
            h_pert = best_h + rng.randn(n_final) * scales[attempt]
            h_pert = project_feasible(h_pert, target_final)
            h_ref, sc = lbfgs_optimize(h_pert, n_final, target_final,
                                       alphas=[500, 2000, 10000, 50000],
                                       top_ms=[n_final] * 4,
                                       pen_coeffs=[10, 10, 20, 50],
                                       maxiter=800,
                                       deadline=min(start_time + 1080, hard_deadline))
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 7: Block coordinate descent
    if tleft() > 80:
        for bs in [10, 20]:
            if tleft() < 30:
                break
            h_bd = block_lbfgs(best_h, n_final, target_final, bs, 3, 5000.0, 20.0,
                               deadline=min(start_time + 1150, hard_deadline))
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    # Phase 8: Final ultra-sharp polish
    if tleft() > 40:
        h_ref, sc = lbfgs_optimize(best_h, n_final, target_final,
                                   alphas=[50000, 100000, 200000],
                                   top_ms=[n_final, n_final, n_final],
                                   pen_coeffs=[100, 100, 100],
                                   maxiter=2000,
                                   deadline=min(start_time + 1230, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final tiny subgradient + surgery
    if tleft() > 25:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final,
                                         n_steps=200, n_lags=2,
                                         lr_init=0.0008, lr_decay=0.998,
                                         deadline=min(start_time + 1260, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft() > 15:
        h_surg, sc_surg = worst_lag_descent(best_h, n_final, target_final,
                                            max_steps=150, eps_init=0.0003,
                                            deadline=hard_deadline)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    parent_score = compute_c5(h_parent)
    if final_score > parent_score:
        h_final = h_parent
        final_score = parent_score
    return h_final, final_score, len(h_final)
