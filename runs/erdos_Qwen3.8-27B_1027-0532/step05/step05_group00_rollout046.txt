    """Generate a binary sequence inspired by Legendre symbol for length n."""
    # Find closest odd number >= n
    p = n if n % 2 == 1 else n + 1

                    is_prime = False
                    break

For a fixed h[-i], the contribution to R(k) from h[i] is:
- R(k) includes terms h[i](1-h[i+k]) and h[i-k](1-h[i])

def compute_c5(h):
    n = len(h)

def corr_array(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:current_sum - target_sum]:
            h_r[ones[idx]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:target_sum - current_sum]:
            h_r[zeros[idx]] = 1
    while int(h_r.sum()) > target_sum:
        h_r[np.where(h_r == 1)[0][0]] = 0
    while int(h_r.sum()) < target_sum:
        h_r[np.where(h_r == 0)[0][0]] = 1
    return h_r

def legendre_binary(n, rng):
    p = n if n % 2 == 1 else n + 1
    while True:
        is_prime = p > 1
        if p > 1:
            for d in range(2, int(np.sqrt(p)) + 1):
                if p % d == 0:
                    is_prime = False
                    break
        if is_prime:
            break
        p += 2
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = 0.0
    if p >= n:
        h = leg[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = leg
        for i in range(p, n):
            h[i] = 0.5
    target = n // 2
    return project_binary(h, target)

def smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)) / alpha))
    return val * (2.0 / n)

                best_n = len(h_ps)

    # Phase 5: Continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alloc_ref = min(250, tleft() * 0.5)
        h_ref, sc_ref = continuous_refine(best_h, n_use, target_use,
                                          alloc_ref, start_time)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref
            best_n = n_use

    # Phase 6: SA on binary projection of continuous solution
    if tleft() > 50:
        n_use = len(best_h)
        target_bin = int(n_use / 2)
        h_bin = project_binary(best_h, target_bin)
        alloc_sa = min(100, tleft() * 0.4)
        max_it5 = int(min(300000, alloc_sa * 2000))
        h_sf, c5_sf = sa_opt(h_bin, target_bin, max_it5, 0.006, 0.00001,
                             0.28, rng, time_limit=alloc_sa, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf
            best_n = n_use
        # Continuous refine again
        if tleft() > 30:
            alloc_ref2 = min(tleft() - 10, 100)
            h_ref2, sc_ref2 = continuous_refine(best_h, n_use, target_bin,
                                                alloc_ref2, start_time)
            if sc_ref2 < best_score:
                best_score = sc_ref2
                best_h = h_ref2
                best_n = n_use

    # Phase 7: Final polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alloc = max(5.0, tleft() - 5)
        h_final_cont, sc_final = continuous_refine(best_h, n_use, target_use,
                                                   alloc, start_time)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cont
            best_n = n_use

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
