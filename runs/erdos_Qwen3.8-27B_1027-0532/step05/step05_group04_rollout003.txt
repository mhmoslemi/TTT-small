            if sc < best_sc:
                best_sc = sc
                best = h_new.copy()
            if sc < compute_c5(h_cur):
                h_cur = h_new
                lr = min(lr * 1.03, 0.03)
            else:
                lr *= 0.6
                v *= 0.5
            if lr < 1e-10:
                break
    return best, best_sc

def topk_subgradient(h, n, target, dx, k_list, steps, lr_init, deadline):
    h_cur = h.copy()
    best = h_cur.copy()
    best_sc = compute_c5(h_cur)
    for k_active in k_list:
        lr = lr_init
        for step in range(steps):
            if time.time() > deadline:
                return best, best_sc
            grad = compute_grad_topk(h_cur, n, dx, k_active)
            h_new = h_cur - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best = h_new.copy()
            if sc < compute_c5(h_cur):
                h_cur = h_new
                lr = min(lr * 1.02, 0.015)
            else:
                lr *= 0.93
            if lr < 1e-10:
                break
    return best, best_sc

def交替_random_perturb(h, n, target, n_trials, deadline):
    h_cur = h.copy()
    best = h_cur.copy()
    best_sc = compute_c5(h_cur)
    rng = np.random.RandomState(12345)
    for t in range(n_trials):
        if time.time() > deadline:
            break
        h_test = h_cur.copy()
        idx = rng.choice(n, size=2, replace=False)
        h_test[idx[0]], h_test[idx[1]] = h_cur[idx[1]], h_cur[idx[0]]
        h_test = project_feasible(h_test, target)
        sc = compute_c5(h_test)
        if sc < best_sc:
            best_sc = sc
            best = h_test.copy()
            h_cur = h_test.copy()
    return best, best_sc
