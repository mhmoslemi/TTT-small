def compute_delta(i, j, h):
    n_local = len(h)
    delta = np.zeros(2 * n_local - 1)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def greedy_optimize(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 300

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n - 1)

        if k_star == 0:
            sorted_idx = np.argsort(corr)[::-1]
            k_star = 0
            for si in sorted_idx[:20]:
                kk = si - (n - 1)
                if kk != 0:
                    k_star = kk
                    max_idx = si
                    break
            if k_star == 0:
                stall += 1
                continue

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # For lag k_star, find swaps that reduce it
        # corr[n-1+k_star] = sum_t h[t+k_star]*(1-h[t]) for k_star > 0
        # To reduce: set h[t+k_star] to 0 (if it's 1) or set h[t] to 1 (if it's 0)
        # Swap: i in ones (source), j in zeros (dest)
        # Effect on lag k_star:
        #   If k_star > 0: corr = sum_t h[t+k]*(1-h[t])
        #   Removing h[i]=1: loses h[i] from h[t+k] when t+k=i => t=i-k; 
        #     also (1-h[i]) was 0, becomes 1, gains h[i+k]
        #   Adding h[j]=1: gains h[j] in h[t+k] when t=j-k;
        #     (1-h[j]) was 1, becomes 0, loses h[j+k]
        
        # Quick score: which swaps reduce corr at k_star the most?
        # Delta at k_star from swap (i,j):
        #   - (1-h[i-k_star]) if i-k_star in [0,n-1]  [removing h[i] from h[t+k] term]
        #   + h[i+k_star] if i+k_star in [0,n-1]     [adding to (1-h[t]) term]
        #   + (1-h[j-k_star]) if j-k_star in [0,n-1] [adding h[j] to h[t+k] term]
        #   - h[j+k_star] if j+k_star in [0,n-1]     [removing from (1-h[t]) term]
        
        k = k_star
        # Score for each one i: how much it contributes to corr at lag k
        # i contributes h[i]*(1-h[i-k]) to corr at lag k (as h[t+k] part, t=i-k)
        # AND (1-h[i])=0 blocks h[i-k]... wait
        # corr[n-1+k] = sum_t h[t+k]*(1-h[t])
        # i contributes when t+k = i (t=i-k): h[i]*(1-h[i-k])
        # AND when t = i: h[i+k]*(1-h[i]) = h[i+k]*0 = 0 (since h[i]=1)
        # So i only contributes h[i]*(1-h[i-k]) = 1-h[i-k] to corr at lag k (if i-k valid)
        
        # j contributes when t+k = j: h[j]*(1-h[j-k]) = 0 (since h[j]=0)
        # AND when t = j: h[j+k]*(1-h[j]) = h[j+k]*1 = h[j+k]
        # So j contributes h[j+k] to corr at lag k (if j+k valid)
        
        # After swap (i->0, j->1):
        # i no longer contributes 1-h[i-k] (good, reduces corr)
        # i now contributes h[i+k]*(1-0) = h[i+k] to (1-h[t]) term (bad, increases corr)
        # j now contributes 1-h[j-k] to h[t+k] term (bad, increases corr)  
        # j no longer contributes h[j+k] to (1-h[t]) term (good, reduces corr)
        
        # Net change at lag k:
        # = -(1-h[i-k])_valid + h[i+k]_valid + (1-h[j-k])_valid - h[j+k]_valid
        
        i_score = np.zeros(n)
        j_score = np.zeros(n)
        
        idx = np.arange(n)
        valid_i1 = (idx - k >= 0) & (idx - k < n)
        i_score[valid_i1] = -(1.0 - h[idx[valid_i1] - k])
        valid_i2 = (idx + k >= 0) & (idx + k < n)
        i_score[valid_i2] += h[idx[valid_i2] + k]
        
        valid_j1 = (idx - k >= 0) & (idx - k < n)
        j_score[valid_j1] = (1.0 - h[idx[valid_j1] - k])
        valid_j2 = (idx + k >= 0) & (idx + k < n)
        j_score[valid_j2] -= h[idx[valid_j2] + k]
        
        top_i = ones[np.argsort(i_score[ones])[:min(15, len(ones))]]
        top_j = zeros[np.argsort(j_score[zeros])[:min(15, len(ones))]]
        
        found = False

                    continue
                delta = compute_delta(int(i), int(j), h)
                new_max = float((corr + delta).max())
                if new_max < best_delta_max:

                    best_pair = (int(i), int(j))
        
        if best_pair is not None:
            i_sw, j_sw = best_pair
            delta = compute_delta(i_sw, j_sw, h)
            h[i_sw] = 0.0

            corr += delta
            stall = 0
            new_c5 = float(corr.max()) * dx
            if new_c5 < best_c5:

                best_h = h.copy()

                    break

def sa_optimize(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Focused move: bias toward critical lags
        if it % 3 == 0:
            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                idx = np.arange(n)
                valid_i1 = (idx - k_star >= 0) & (idx - k_star < n)
                i_pref = np.zeros(n)
                i_pref[valid_i1] = h[idx[valid_i1] - k_star]
                valid_i2 = (idx + k_star >= 0) & (idx + k_star < n)
                i_pref[valid_i2] -= h[idx[valid_i2] + k_star]
                valid_j1 = (idx - k_star >= 0) & (idx - k_star < n)
                j_pref = np.zeros(n)
                j_pref[valid_j1] = -h[idx[valid_j1] - k_star]
                valid_j2 = (idx + k_star >= 0) & (idx + k_star < n)
                j_pref[valid_j2] += h[idx[valid_j2] + k_star]
                
                i_weights = np.maximum(i_pref[ones] - i_pref[ones].min() + 1.0, 0.1)
                j_weights = np.maximum(j_pref[zeros] - j_pref[zeros].min() + 1.0, 0.1)
                i_probs = i_weights / i_weights.sum()
                j_probs = j_weights / j_weights.sum()
                i = int(ones[rng.choice(len(ones), p=i_probs)])
                j = int(zeros[rng.choice(len(zeros), p=j_probs)])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

        delta = compute_delta(i, j, h)
        new_max = float((corr + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        accept = False
        if delta_c5 < 0:
            accept = True
        elif t > 1e-15:
            if rng.random() < np.exp(-delta_c5 / t):
                accept = True

        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

        else:
            top_i = ones[:min(8, len(ones))]
            top_j = zeros[:min(8, len(zeros))]

        found = False
        for i1 in top_i:
            for i2 in top_i:
                if i1 >= i2:
                    continue
                for j1 in top_j:
                    for j2 in top_j:
                        if j1 >= j2:
                            continue
                        if i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                            continue
                        # Compute combined delta
                        d1 = compute_delta(int(i1), int(j1), h)
                        # Now h is modified for delta computation of second swap
                        # But we need to be careful: after first swap, h[i1]=0, h[j1]=1
                        # Let's compute the combined effect directly
                        # Actually let's just compute the full new corr
                        h_test = h.copy()
                        h_test[i1] = 0.0
                        h_test[i2] = 0.0
                        h_test[j1] = 1.0
                        h_test[j2] = 1.0
                        new_corr = np.correlate(h_test, 1.0 - h_test, mode='full')
                        new_max = float(new_corr.max())
                        if new_max < max_val:
                            h = h_test
                            corr = new_corr
                            found = True
                            new_c5 = new_max * dx
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_h = h.copy()
                            break
                    if found:
                        break
                if found:
                    break
            if found:
                break

        if found:
            stall = 0
        else:
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def make_periodic_h(n, period, ones_in_period):
    """Create a periodic binary vector with given period and ones pattern."""
    h = np.zeros(n)
    for i in range(n):
        if i % period in ones_in_period:
            h[i] = 1.0
    # Adjust to have exactly n/2 ones
    current = int(h.sum())
    target = n // 2
    if current > target:
        excess = current - target
        ones = np.where(h == 1)[0]
        h[ones[:excess]] = 0
    elif current < target:
        deficit = target - current
        zeros = np.where(h == 0)[0]
        h[zeros[:deficit]] = 1
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Aggressive greedy
    allowed = min(60, tleft() * 0.10)
    h_g, c5_g = greedy_optimize(best_h, max_iters=100000, rng=rng,
                                time_limit=allowed, start_time=start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA with focused moves
    allowed = min(200, tleft() * 0.25)
    n_sa = int(min(8000000, allowed * 60000))
    h_sa, c5_sa = sa_optimize(best_h, n_sa, 0.025, 0.00005, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Greedy again after SA
    allowed = min(40, tleft() * 0.06)
    h_g2, c5_g2 = greedy_optimize(best_h, max_iters=50000, rng=rng,
                                  time_limit=allowed, start_time=start_time)
    if c5_g2 < best_score:
        best_score = c5_g2
        best_h = h_g2.copy()

    # Phase 4: 2-swap moves
    if tleft() > 80:
        allowed = min(60, tleft() * 0.10)
        h_2s, c5_2s = two_swap_greedy(best_h, max_iters=20000, rng=rng,
                                       time_limit=allowed, start_time=start_time)
        if c5_2s < best_score:
            best_score = c5_2s
            best_h = h_2s.copy()
        # SA after 2-swap
        if tleft() > 60:
            allowed = min(80, tleft() * 0.12)
            n_sa2 = int(min(3000000, allowed * 50000))
            h_sa2, c5_sa2 = sa_optimize(best_h, n_sa2, 0.01, 0.00003, rng,
                                        time_limit=allowed, start_time=start_time)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

    # Phase 5: Iterated local search
    for ils_it in range(15):
        if tleft() < 80:
            break
        # Perturb
        n_pert = rng.randint(4, 10)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        # Short greedy
        allowed = min(20, tleft() * 0.04)
        h_pg, c5_pg = greedy_optimize(h_pert, max_iters=20000, rng=rng,
                                       time_limit=allowed, start_time=start_time)
        # Short SA
        if tleft() > 30:
            allowed = min(40, tleft() * 0.06)
            n_sap = int(min(1000000, allowed * 40000))
            h_ps, c5_ps = sa_optimize(h_pg, n_sap, 0.012, 0.00002, rng,
                                       time_limit=allowed, start_time=start_time)
            if c5_ps < c5_pg:
                h_local = h_ps
                c5_local = c5_ps
            else:
                h_local = h_pg
                c5_local = c5_pg
        else:
            h_local = h_pg
            c5_local = c5_pg

        if c5_local < best_score:
            best_score = c5_local
            best_h = h_local.copy()

    # Phase 6: Try different n values
    for n_try in [n_init - 4, n_init + 4, n_init - 8, n_init + 8]:
        if tleft() < 40:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.06)
        h_g3, c5_g3 = greedy_optimize(h_bin_try, max_iters=15000, rng=rng,
                                       time_limit=allowed, start_time=start_time)
        allowed2 = min(15, tleft() * 0.05)
        n_sa3 = int(min(500000, allowed2 * 40000))
        h_s3, c5_s3 = sa_optimize(h_g3, n_sa3, 0.01, 0.00005, rng,
                                   time_limit=allowed2, start_time=start_time)
        if c5_s3 < c5_g3:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 100, 300, 1000, 3000, 8000]
        allowed = min(tleft() - 20, 200)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-14,
                                  time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final binary SA polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(5000000, max(50000, (tleft() - 15) * 50000)))
        h_sa_f, c5_f = sa_optimize(h_bin_final, n_sa_f, 0.008, 0.00001,
                                    rng, time_limit=max(5.0, tleft() - 10),
                                    start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 9: Final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                  [3000, 8000, 20000], 2000, 1e-15,
                                  time_limit=max(5.0, tleft() - 5),
                                  start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
