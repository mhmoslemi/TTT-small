```python
def binary_local_search(h, n, target, time_limit, initial_temp=1.0, cooling=0.999):
    """Binary local search with simulated annealing"""
    start = time.time()
    dx = 2.0 / n
    h = (h > 0.5).astype(float)
    # Ensure correct number of 1s
    n_ones = int(np.sum(h))
    target_ones = int(target)
    
    # Adjust to have exactly target_ones ones
    if n_ones > target_ones:
        one_pos = np.where(h == 1)[0]
        to_remove = rng.choice(one_pos, n_ones - target_ones, replace=False)
        h[to_remove] = 0
    elif n_ones < target_ones:
        zero_pos = np.where(h == 0)[0]
        to_add = rng.choice(zero_pos, target_ones - n_ones, replace=False)
        h[to_add] = 1
    
    corr = np.correlate(h, 1-h, mode='full')
    c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = c5
    
    temp = initial_temp
    no_improve = 0
    
    while time.time() - start < time_limit:
        # Compute correlation
        corr = np.correlate(h, 1-h, mode='full')
        c5 = float(np.max(corr)) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            temp *= cooling
            no_improve = 0
            if temp < 1e-10:
                break
        
        # Try a random swap
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        
        # Perform swap
        h[i], h[j] = h[j], h[i]
        new_corr = np.correlate(h, 1-h, mode='full')
        new_c5 = float(np.max(new_corr)) * dx
        
        delta = new_c5 - c5
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-15)):
            c5 = new_c5
        else:
            # Revert
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

    n_ones = int(np.sum(h))
    if n_ones > target_ones:
        one_idx = np.where(h == 1)[0]
        to_remove = rng.choice(one_idx, n_ones - target_ones, replace=False)
        h[to_remove] = 0
    elif n_ones < target_ones:
        zero_idx = np.where(h == 0)[0]
        to_add = rng.choice(zero_idx, target_ones - n_ones, replace=False)
        h[to_add] = 1
    start = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = c5
    one_pos = np.where(h == 1)[0]
    zero_pos = np.where(h == 0)[0]

    step = 0

def build_periodic_pattern(pattern, n):
    p = len(pattern)
    reps = n // p
    h = np.tile(pattern, reps)
    if n % p != 0:
        h = np.concatenate([h, pattern[:n % p]])
    return h

            no_improve = 0
            improved = True

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
