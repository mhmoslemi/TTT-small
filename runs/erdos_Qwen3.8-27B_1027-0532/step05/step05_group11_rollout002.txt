    one_minus_h = 1.0 - h

    phi = corr * dx
    n_half = n - 1

    log_p = alpha * (phi - phi_max)
    p = np.exp(log_p)
    p = p / p.sum()
    
    F = phi_max + np.log(np.sum(np.exp(log_p))) / alpha
    
    # Compute gradient using FFT
    # For lag k (shift = k, where k ranges from -(n-1) to n-1):
    # corr index = k + (n-1)
    
    # p_pos[k] for k = 0, 1, ..., n-1: p[k + n - 1] * dx
    # p_neg[m] for m = 1, ..., n-1: p[n - 1 - m] * dx (this is lag -m)
    
    # Grad from positive lags (k >= 0):
    #   Term A: sum_{k=0}^{n-1-j} p_pos[k] * (1-h[j+k])  -- this is cross-corr of p_pos with (1-h)
    #   Term B: sum_{k=0}^{j} p_pos[k] * h[j-k]  -- this is cross-corr of p_pos[::-1] with h
    #   (actually it's the same as correlation)

    # Similarly for negative lags...
    
    # Let me just do it with np.correlate which is O(n * len(g)) for each
    # Actually for n=400, let me just do a direct loop over the top lags.
    
    # Find top lags
    top_n = 30

The key constraint is that the total "mass" of h is 1 (out of 2). So on average, h = 0.5.

If h were a square wave with period 2p and duty cycle 1/2, then:
- At lag = p, the overlap would be 0 (perfect anti-correlation)
- At lag = 0, the overlap would be ∫ h(1-h) = ∫ h - ∫ h² = 1 - ∫h². For a 0-1 function with mean 0.5, ∫h² = 1, so overlap at lag 0 = 0.
- At other lags, it depends on the structure.

Wait, for a 0-1 function, ∫h(1-h) = 0 always (since h(1-h) = 0 for h ∈ {0,1}). So lag 0 is never a problem for 0-1 functions!

For continuous-valued h, the lag 0 overlap is ∫ h(1-h) dx, which is maximized when h = 0.5 everywhere (giving 0.5). So continuous values make lag 0 potentially problematic.

This suggests that for the optimal solution, h should be close to 0-1 valued (to avoid the lag 0 issue), but with some smoothing to handle other lags better.

This is an important insight! The optimal solution might be a "nearly" 0-1 function with some transition regions.

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def compute_smooth_max_grad(h, alpha, dx, top_n=40):
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    phi = corr * dx
    n_half = n - 1

    phi_max = np.max(phi)
    log_p = alpha * (phi - phi_max)
    p = np.exp(log_p)
    p_sum = np.sum(p)
    p = p / p_sum

    F = phi_max + np.log(p_sum) / alpha

    top_idx = np.argsort(phi)[::-1][:top_n]
    top_p = p[top_idx]
    top_p_sum = np.sum(top_p)
    if top_p_sum < 1e-15:
        return F, np.zeros(n)
    top_p = top_p / top_p_sum

    grad = np.zeros(n)
    for i in range(len(top_idx)):
        idx = top_idx[i]
        k = idx - n_half
        w = top_p[i] * dx
        if k >= 0:
            if n - 1 - k >= 0:
                grad[:n - k] += w * (1.0 - h[k:n])
            if k < n:
                grad[k:] -= w * h[:n - k]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n - m])
            if n - m > 0:
                grad[:n - m] -= w * h[m:]

    return F, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def project_onto_simplex_box(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = np.zeros(len(h))
    hi = np.ones(len(h))
    active = (h > lo) & (h < hi)
    h_result = h.copy()
    if not np.any(active):
        return project_feasible(h, target_sum)
    current_sum = np.sum(h[active])
    needed = target_sum - (np.sum(h[~active]))
    n_active = np.sum(active)
    if needed <= 0:
        h_result[active] = 0.0
    elif needed >= n_active:
        h_result[active] = 1.0
    else:
        shift = (needed - current_sum) / n_active
        h_result[active] = h[active] + shift
    return project_feasible(h_result, target_sum)


def nesterov_smooth_gradient(h_init, n, target, dx, alphaSchedule,
                              max_iter, time_limit, momentum=0.9,
                              top_n=40, backtrack_max=30):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    y = h.copy()
    h_prev = h.copy()

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_smooth = np.inf

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        alpha = alphaSchedule[it % len(alphaSchedule)]
        F_y, grad_y = compute_smooth_max_grad(y, alpha, dx, top_n=top_n)

        t_k = 1.0 / (1.0 + momentum)
        new_t = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        y_new = h + ((t_k - 1.0) / new_t) * (h - h_prev)
        y_new = np.clip(y_new, 0.0, 1.0)

        F_y2, grad_y2 = compute_smooth_max_grad(y_new, alpha, dx, top_n=top_n)

        lr = 1.0
        h_try = h - lr * grad_y2
        h_try = project_feasible(h_try, target)

        for bt in range(backtrack_max):
            F_try, _ = compute_smooth_max_grad(h_try, alpha, dx, top_n=top_n)
            decrease = F_y2 - F_try - 0.5 * lr * np.dot(grad_y2, h_try - y_new)
            if F_try <= F_y2 + 1e-10:
                break
            lr *= 0.5
            h_try = h - lr * grad_y2
            h_try = project_feasible(h_try, target)

        h_prev = h.copy()
        h = h_try
        y = y_new

        F_h, _ = compute_smooth_max_grad(h, alpha, dx, top_n=top_n)
        if F_h < best_smooth - 1e-12:
            best_smooth = F_h
            actual_c5 = compute_c5(h)
            if actual_c5 < best_c5 - 1e-14:
                best_c5 = actual_c5
                best_h = h.copy()

    return best_h, best_c5


def alternating_lp(h_init, n, target, dx, max_iter, time_limit,
                   n_top=200, damping=0.9, reg=1e-5, solver='SCS'):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 300:
            break

        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=solver, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-8, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5


def sharpen_h(h, target, strength=0.5):
    """Push values towards 0 or 1 while maintaining the constraint."""
    midpoint = 0.5
    h_sharp = np.where(h > midpoint,
                       midpoint + strength * (h - midpoint) * 2,
                       midpoint + strength * (h - midpoint) * 2)
    return project_feasible(h_sharp, target)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _global_target


def _cons_jac(h):
    return np.ones(_global_n)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
