import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    d = np.arange(-(n - 1), n)
    idx_p = i + d
    idx_m = i - d
    B = np.zeros(nl1)
    valid_p = (idx_p >= 0) & (idx_p < n)
    valid_m = (idx_m >= 0) & (idx_m < n)
    B[valid_p] -= h[idx_p[valid_p]]
    B[valid_m] += (1.0 - h[idx_m[valid_m]])
    return B

def grad_for_lag(h, d, n):
    g = np.zeros(n)
    idx1 = np.arange(n) - d
    valid1 = (idx1 >= 0) & (idx1 < n)
    g[valid1] += (1.0 - h[idx1[valid1]])
    idx2 = np.arange(n) + d
    valid2 = (idx2 >= 0) & (idx2 < n)
    g[valid2] -= h[idx2[valid2]]
    return g

def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=40, max_stall=250):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0

    for sweep in range(200000):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_vec(h_cur, i, n)
            Bj = compute_B_vec(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            pos = C > 1e-14
            neg = C < -1e-14
            flat = ~(pos | neg)

            if np.any(pos) and np.any(neg):
                t_lo = t_min
                t_hi = t_max
                for _ in range(65):
                    t_mid = (t_lo + t_hi) * 0.5
                    max_pos = float(np.max(A[pos] + C[pos] * t_mid))
                    max_neg = float(np.max(A[neg] + C[neg] * t_mid))
                    max_flat = float(np.max(A[flat])) if np.any(flat) else -1e18
                    if max_pos >= max_neg and max_pos >= max_flat:
                        t_lo = t_mid
                    else:
                        t_hi = t_mid
                t_star = (t_lo + t_hi) * 0.5
            elif np.any(pos) and not np.any(neg):
                t_star = t_min
            elif np.any(neg) and not np.any(pos):
                t_star = t_max
            else:
                t_star = (t_min + t_max) * 0.5

            t_star = max(t_min, min(t_max, t_star))
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
                corr[np.abs(corr) < 1e-15] = 0.0

            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

            break

    h_final = project_feasible(h_cur, target)

            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

    # Phase 7: Final SLSQP with very high alpha
    if tleft() > 50:
        h_final = slsqp_polish(best_h, n, target, [20000, 60000, 150000], 3000, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 8: Final coordinate descent
    if tleft() > 20:
        h_final, sc_final = paired_coord_desc(best_h, n, target,
                                              start_time + budget_s - 8.0, rng,
                                              recompute_interval=20)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
