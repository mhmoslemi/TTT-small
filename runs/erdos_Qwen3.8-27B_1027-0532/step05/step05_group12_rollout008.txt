```python
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n-m]
    
    return grad * dx

def adam_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30):
    """Adam optimizer for the smooth max C5 problem."""
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    # Schedule: increase alpha over time, top_k gradually increases
    schedule = [
        # (alpha, top_k, steps, lr)
        (20, 20, 500, 0.5),
        (50, 30, 500, 0.3),
        (100, 50, 500, 0.2),
        (200, 100, 500, 0.15),
        (500, 200, 500, 0.1),
        (1000, 400, 500, 0.08),
        (2000, 800, 500, 0.05),
        (5000, 2*n-1, 500, 0.03),
        (10000, 2*n-1, 800, 0.02),
        (20000, 2*n-1, 800, 0.01),
    ]
    
    for alpha, top_k, steps, lr_base in schedule:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        
        stall_count = 0
        for step in range(steps):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            
            t += 1
            lr = lr_base / (1.0 + 0.001 * t)
            
            g = grad_c5_smooth(h, alpha, top_k)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_trial = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_trial = project_feasible(h_trial, target)
            
            score = compute_c5(h_trial)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h_trial.copy()
                h = h_trial
                stall_count = 0
            else:
                # Try smaller step
                for factor in [0.5, 0.25, 0.1]:
                    h_trial = h - lr * factor * m_hat / (np.sqrt(v_hat) + eps)
                    h_trial = project_feasible(h_trial, target)
                    score = compute_c5(h_trial)
                    if score < best_score - 1e-14:
                        best_score = score
                        best_h = h_trial.copy()
                        h = h_trial
                        stall_count = 0
                        break
                else:
                    stall_count += 1
                    if stall_count > 100:
                        break
    
    return best_h, best_score

def structured_perturbation(h, n, target, rng, mode=0):
    """Apply a structured perturbation to escape local optima."""
    h = h.copy()
    if mode == 0:
        # Shift a random segment
        seg_len = rng.randint(n//8, n//3)
        start = rng.randint(0, n - seg_len)
        shift = rng.randint(-seg_len//2, seg_len//2 + 1)
        h[start:start+seg_len] = np.roll(h[start:start+seg_len], shift)
    elif mode == 1:
        # Invert a random segment
        seg_len = rng.randint(n//10, n//4)
        start = rng.randint(0, n - seg_len)
        h[start:start+seg_len] = 1.0 - h[start:start+seg_len]
    elif mode == 2:
        # Add structured noise (sine wave)
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, 2*np.pi)
        x = np.arange(n) / n * 2 * np.pi
        h = h + amp * np.sin(freq * x + phase)
    elif mode == 3:
        # Random binary perturbation
        flip_mask = rng.rand(n) < 0.1
        h[flip_mask] = 1.0 - h[flip_mask]
    
    return project_feasible(h, target)

def interpolate_to(h_src, n_dst, target_dst):

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

For the p-norm:
f(h) = (sum_k c_k^p)^(1/p)

∂f/∂h[j] = (1/p) * (sum_k c_k^p)^(1/p - 1) * sum_k p * c_k^(p-1) * ∂c_k/∂h[j]
= (sum_k c_k^p)^(1/p - 1) * sum_k c_k^(p-1) * ∂c_k/∂h[j]

The weights are proportional to c_k^(p-1) / (sum_l c_l^p)^(1-1/p)

For large p, this concentrates on the maximum values.

Let me implement this properly. Actually, for numerical stability, I'll just use the log-sum-exp approach with adaptive alpha, since it's well-behaved.

Now, let me think about the Adam optimizer more carefully. The key issue is that the projection onto the feasible set (0 <= h <= 1, sum(h) = target) can create issues with Adam's momentum. When we project, the effective step is different from what Adam "thinks" it took.

A better approach might be to use a penalty method for the sum constraint:
minimize smooth_max(c) + mu * (sum(h) - target)^2
subject to 0 <= h <= 1

Then use simple projection for the box constraint and Adam for the rest. Increase mu over time.

But this might not be as accurate. Let me stick with the projection approach but be more careful.

Actually, another idea: use the fact that the problem is separable in a certain sense. The box constraint [0,1] is separable, and the sum constraint is a single linear constraint. I could use a proximal gradient method where the proximal step handles both constraints simultaneously.

The proximal operator for the constraint set {h: 0 <= h <= 1, sum(h) = target} is:
prox_{C}(v) = argmin_h 0.5*||h-v||^2 s.t. 0 <= h <= 1, sum(h) = target

This can be computed as: find tau such that sum_i clip(v_i - tau, 0, 1) = target. This is a 1D root-finding problem (bisection on tau).

This is more efficient than my iterative projection! Let me implement this.

```python
def prox_constraint(v, target_sum):
    """Proximal operator for {h: 0<=h<=1, sum(h)=target_sum}."""
    # Find tau such that sum_i clip(v_i - tau, 0, 1) = target_sum
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    
    for _ in range(100):
        tau = (lo + hi) / 2
        h = np.clip(v - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = tau
        elif s < target_sum:
            hi = tau
        else:
            break
        if hi - lo < 1e-12:
            break
    
    h = np.clip(v - (lo + hi) / 2, 0.0, 1.0)
    # Final correction for any tiny residual
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h
```

This is much faster than the iterative projection! The bisection converges in ~40 iterations regardless of n.

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_c(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def prox_constraint(v, target_sum):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        tau = (lo + hi) / 2.0
        h = np.clip(v - tau, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            lo = tau
        elif s < target_sum:
            hi = tau
        else:
            break
        if hi - lo < 1e-14:
            break
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        nf = np.sum(mask)
        if nf > 0:
            h = h.copy()
            h[mask] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    return prox_constraint(h, target_sum)

def grad_smooth_max(h, alpha, top_k, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    total_lags = 2 * n - 1
    if top_k < total_lags:
        top_idx = np.argsort(corr)[-top_k:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(total_lags)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    wsum = np.sum(exp_vals)
    if wsum < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights = exp_vals / wsum
    n_half = n - 1
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad * dx

def grad_pnorm(h, p, top_k, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    total_lags = 2 * n - 1
    if top_k < total_lags:
        top_idx = np.argsort(corr)[-top_k:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(total_lags)
        sub_corr = corr
    safe = np.maximum(sub_corr, 1e-12)
    weights = safe ** (p - 1)
    wmax = np.max(weights)
    if wmax > 0:
        weights = weights / wmax
    n_half = n - 1
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        k = idx - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def adam_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30, use_pnorm=False):
    dx = 2.0 / n
    h = prox_constraint(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1 = 0.75
    beta2 = 0.999
    eps = 1e-8
    t = 0
    total_lags = 2 * n - 1
    schedule = [
        (15, 10, 600, 0.8),
        (30, 20, 600, 0.6),
        (60, 40, 600, 0.45),
        (120, 80, 600, 0.35),
        (250, 150, 600, 0.25),
        (500, 300, 600, 0.2),
        (1000, 500, 800, 0.12),
        (2000, total_lags, 800, 0.08),
        (5000, total_lags, 800, 0.05),
        (10000, total_lags, 1000, 0.03),
        (20000, total_lags, 1000, 0.02),
        (50000, total_lags, 1200, 0.01),
    ]
    for phase_i, (alpha, top_k, steps, lr_base) in enumerate(schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        p_val = alpha if not use_pnorm else min(4.0 + phase_i * 2.0, 50.0)
        stall = 0
        for step in range(steps):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            t += 1
            lr = lr_base / (1.0 + 0.0005 * t)
            if use_pnorm:
                g = grad_pnorm(h, p_val, top_k, dx)
            else:
                g = grad_smooth_max(h, alpha, top_k, dx)
            m_buf = beta1 * m_buf + (1.0 - beta1) * g
            v_buf = beta2 * v_buf + (1.0 - beta2) * g * g
            m_hat = m_buf / (1.0 - beta1 ** t)
            v_hat = v_buf / (1.0 - beta2 ** t)
            step_dir = m_hat / (np.sqrt(v_hat) + eps)
            improved = False
            for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
                h_trial = h - lr * factor * step_dir
                h_trial = prox_constraint(h_trial, target)
                score = compute_c5(h_trial)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_trial.copy()
                    h = h_trial
                    improved = True
                    stall = 0
                    break
            if not improved:
                stall += 1
                if stall > 150:
                    break
    return best_h, best_score

def binary_refine(h_init, n, target, time_limit, start_time, deadline_offset=20):
    sorted_idx = np.argsort(-h_init)
    h = np.zeros(n)
    h[sorted_idx[:int(round(target))]] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    n_ones = int(round(target))
    rng_local = np.random.RandomState(12345)
    max_iter = 50000
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        if it % 5 == 0:
            i = rng_local.randint(n)
            if h[i] > 0.5:
                zeros = np.where(h == 0.0)[0]
                if len(zeros) == 0:
                    continue
                j = zeros[rng_local.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                score = compute_c5(h)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h.copy()
                else:
                    h[i] = 1.0
                    h[j] = 0.0
            else:
                ones = np.where(h == 1.0)[0]
                if len(ones) == 0:
                    continue
                j = ones[rng_local.randint(len(ones))]
                h[i] = 1.0
                h[j] = 0.0
                score = compute_c5(h)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h.copy()
                else:
                    h[i] = 0.0
                    h[j] = 1.0
        else:
            i, j = rng_local.choice(n, 2, replace=False)
            if h[i] == h[j]:
                continue
            h[i], h[j] = h[j], h[i]
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
    return best_h, best_score

def perturb_h(h, n, target, rng, mode, strength=0.05):
    h = h.copy()
    if mode == 0:
        h += rng.randn(n) * strength
    elif mode == 1:
        seg = rng.randint(n // 6, n // 3 + 1)
        start = rng.randint(0, n - seg)
        shift = rng.randint(-seg // 3, seg // 3 + 1)
        seg_copy = h[start:start + seg].copy()
        h[start:start + seg] = np.roll(seg_copy, shift)
    elif mode == 2:
        seg = rng.randint(n // 8, n // 4 + 1)
        start = rng.randint(0, n - seg)
        h[start:start + seg] = 1.0 - h[start:start + seg]
    elif mode == 3:
        freq = rng.uniform(1.0, 4.0)
        amp = strength * 2.0
        phase = rng.uniform(0, 2 * np.pi)
        x = np.arange(n) / n * 2.0 * np.pi
        h += amp * np.sin(freq * x + phase)
    elif mode == 4:
        flip_frac = min(0.15, strength * 3.0)
        mask = rng.rand(n) < flip_frac
        h[mask] = 1.0 - h[mask]
    return prox_constraint(h, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return prox_constraint(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return prox_constraint(h_out, target_dst)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha, 2 * n - 1),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            if res.x is not None:
                h_new = prox_constraint(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

def _cons_fun(h):
    return np.sum(h) - _global_target

def _cons_jac(h):
    return np.ones(_global_n)

def _smooth_obj(h, alpha, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_k < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_k:]
        sub = corr[top_idx]
    else:
        sub = corr
    shifted = np.clip(alpha * (sub - c_max), -700.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

_global_n = 0
_global_target = 0.0
```
