def project_onto_box_and_sum(y, S):
    # Find λ such that sum(clip(y - λ, 0, 1)) = S
    # clip(y - λ, 0, 1) = y - λ when 0 < y - λ < 1, i.e., y-1 < λ < y
    # This is a piecewise linear function of λ, decreasing.

        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2

import numpy as np
from scipy.optimize import minimize
import time


def project_box_sum(y, S):
    n = len(y)
    if S <= 0:
        return np.zeros(n)
    if S >= n:
        return np.ones(n)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx


def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= wsum
    w_full = w * dx
    conv1 = np.convolve(w_full, 1.0 - h, mode='full')
    conv2 = np.convolve(w_full[::-1], h, mode='full')
    return (conv1 - conv2)[n - 1: 2 * n - 1]


def projected_gradient_lse(h0, n, target, alpha, max_steps, lr_init, deadline):
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    best_h = h.copy()
    best_score = compute_c5(h)
    patience = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        g = lse_gradient(h, alpha)
        g_proj = g - np.mean(g)
        v = beta * v + (1.0 - beta) * g_proj
        h_new = h - lr * v
        h_new = project_box_sum(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            patience = 0
            lr = min(lr * 1.01, lr_init * 5.0)
        else:
            patience += 1
            lr *= 0.95
        h = h_new
        if patience > 2000:
            break
        if lr < 1e-14:
            break
    return best_h


def slsqp_polish(h, n, target, alphas, maxiter, deadline):
    best_h = h.copy()
    best_score = compute_c5(h)
    for a in alphas:
        if time.time() > deadline:
            break
        alpha = float(a)

            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

        try:

                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_box_sum(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
        except Exception:
            pass
    return best_h


def make_fourier_start(n, target, rng, n_terms_range=(3, 10)):
    x = np.linspace(0, 2, n)
    nt = rng.randint(n_terms_range[0], n_terms_range[1] + 1)
    freqs = rng.choice(np.arange(1, n // 2 + 1), size=nt, replace=False)
    amps = rng.uniform(0.05, 0.35, size=nt)
    phases = rng.uniform(0, 2 * np.pi, size=nt)
    h = 0.5 * np.ones(n)
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.cos(2.0 * np.pi * f * x / 2.0 + p)
    return project_box_sum(h, target)


def make_binary_start(n, target, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = int(h.sum())
    half = n // 2
    if n_ones > half:
        idx = np.where(h == 1)[0]
        remove = rng.choice(idx, size=n_ones - half, replace=False)
        h[remove] = 0.0
    elif n_ones < half:
        idx = np.where(h == 0)[0]
        add = rng.choice(idx, size=half - n_ones, replace=False)
        h[add] = 1.0
    h += rng.randn(n) * 0.08
    return project_box_sum(h, target)


def make_perturbed_start(h_base, n, target, rng, scale):
    return project_box_sum(h_base + rng.randn(n) * scale, target)


def optimize_single(h0, n, target, deadline):
    h = h0.copy()
    best_score = compute_c5(h)
    alpha_schedule = [20, 100, 500, 2000]
    for a in alpha_schedule:
        if time.time() > deadline - 30:
            break
        h_pg = projected_gradient_lse(h, n, target, a, 8000, 0.01, time.time())
        sc = compute_c5(h_pg)
        if sc < best_score:
            best_score = sc
            h = h_pg
    if time.time() < deadline - 15:
        h = slsqp_polish(h, n, target, [500, 2000, 8000], 2000, time.time())
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
    return h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    hard_deadline = start_time + budget_s - 15.0

    h_parent = project_box_sum(np.array(initial_h_values, dtype=float), target)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    starts = [h_parent.copy()]
    for _ in range(3):
        starts.append(make_fourier_start(n, target, rng))
    for scale in [0.05, 0.1, 0.15]:
        starts.append(make_perturbed_start(h_parent, n, target, rng, scale))
    for _ in range(2):
        starts.append(make_binary_start(n, target, rng))

    n_starts = len(starts)
    time_per_start = max(30, (hard_deadline - start_time - 60) / n_starts)

    for i, h0 in enumerate(starts):
        if time.time() > hard_deadline - 40:
            break
        t_dead = min(start_time + (i + 1) * time_per_start, hard_deadline - 30)
        h_opt, sc = optimize_single(h0, n, target, t_dead)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()

    if time.time() < hard_deadline - 60:
        h_final = slsqp_polish(best_h, n, target, [2000, 8000, 20000], 3000, hard_deadline - 20)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if time.time() < hard_deadline - 30:
        h_pg_final = projected_gradient_lse(best_h, n, target, 5000, 5000, 0.005, hard_deadline - 15)
        sc = compute_c5(h_pg_final)
        if sc < best_score:
            best_score = sc
            best_h = h_pg_final

    h_final = project_box_sum(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
