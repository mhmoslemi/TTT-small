    dx = 2.0 / n

    mu = 1.0
    
    for outer in range(20):
        if time.time() > deadline:
            break
        
        def aug_obj(h_vec):
            h = np.clip(h_vec, 0, 1)
            corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
            c = corr * dx
            alpha = 5000
            c_max = np.max(c)
            shifted = alpha * (c - c_max)
            shifted = np.clip(shifted, -700, 0)
            f = (c_max + np.log(np.exp(shifted).sum()) / alpha)
            penalty = mu * (np.sum(h) - target) ** 2 + lam * (np.sum(h) - target)
            return f + penalty
        
        def aug_grad(h_vec):
            h = np.clip(h_vec, 0, 1)
            # gradient of log-sum-exp part
            corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
            c = corr * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c = corr * dx
    c_max = float(np.max(c))
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted[-1] if shifted[-1] > -700 else -700, -700, 0)
    # Actually let me just clip properly
    shifted = np.maximum(shifted, -700)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    w_norm = w / w_sum
    obj = c_max + np.log(w_sum) / alpha
    
    u = 1.0 - h
    first_term = np.zeros(n)
    second_term = np.zeros(n)
    for j in range(n):
        if n - j > 0:
            first_term[j] = np.dot(w_norm[:n-j], u[j:])
        second_term[j] = np.dot(w_norm[:j+1], h[j::-1])
    grad = dx * (first_term - second_term)
    return obj, grad

def fista_step(h, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h)
    x = project_feasible(h, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 2.0 / (alpha * dx + 1)  # Hmm, need Lipschitz constant
    # Actually, let's use a fixed step or backtracking
    step = 0.01 / max(alpha * 0.01, 1.0)
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def lbfgs_optimize(h_init, target, deadline, alpha=5000, max_outer=10):
    """L-BFGS-B with penalty for equality constraint."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    mu = 100.0
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        def obj(h_vec):
            h_clip = np.clip(h_vec, 0.0, 1.0)
            corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')[n-1:]
            c = corr * dx
            c_max = float(np.max(c))
            shifted = alpha * (c - c_max)
            shifted = np.maximum(shifted, -700)
            w = np.exp(shifted)
            w_sum = w.sum()
            lse_val = c_max + np.log(max(w_sum, 1e-300)) / alpha
            penalty = mu * 0.5 * (np.sum(h_vec) - target) ** 2
            return lse_val + penalty
        
        def grad(h_vec):
            h_clip = np.clip(h_vec, 0.0, 1.0)
            corr = np.correlate(h_clip, 1.0 - h_clip, mode='full')[n-1:]
            c = corr * dx
            c_max = float(np.max(c))
            shifted = alpha * (c - c_max)
            shifted = np.maximum(shifted, -700)
            w = np.exp(shifted)
            w_sum = w.sum()
            w_norm = w / w_sum
            
            u = 1.0 - h_clip
            first_term = np.zeros(n)
            second_term = np.zeros(n)
            for j in range(n):
                if n - j > 0:
                    first_term[j] = np.dot(w_norm[:n-j], u[j:])
                second_term[j] = np.dot(w_norm[:j+1], h_clip[j::-1])
            grad_lse = dx * (first_term - second_term)
            
            # Gradient of penalty (only for non-clipped components)
            mask = (h_vec > 0) & (h_vec < 1)
            grad_pen = mu * (np.sum(h_vec) - target) * mask
            return grad_lse * mask + grad_pen
        
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=bounds, options={'maxiter': 300, 'gtol': 1e-8, 'ftol': 1e-12})
        h = res.x
        h = project_feasible(h, target)
        
        viol = abs(np.sum(h) - target)
        if viol > 1e-4:

    return h

def active_set_polish(h, target, deadline, num_active=5, max_iters=5000):
    """Polish by equalizing the top lags."""
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h, target).copy()
    cur_score = compute_c5(h_cur)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c = corr * dx
        c_max = np.max(c)
        # Get weights for active lags (uniform among top num_active)
        top_idx = np.argsort(c)[::-1][:num_active]
        diff = c[top_idx] - c[top_idx[0]]
        w = np.exp(diff / 0.01)
        w /= w.sum()
        
        # Subgradient step
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]

            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                improved = True
                break
        if not improved:

    return h_cur, cur_score

def lse_params(c, alpha):
    """Compute LSE value, weights, and gradient of LSE w.r.t. c_k."""
    c_max = float(np.max(c))
    shifted = alpha * (c - c_max)
    shifted = np.maximum(shifted, -700.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())

    obj = c_max + np.log(w_sum) / alpha

    u = 1.0 - h
    first = np.zeros(n)
    second = np.zeros(n)

            first[j] = np.dot(w_norm[:n-j], u[j:])
        second[j] = np.dot(w_norm[:j+1], h[j::-1])

    # Get top m lags

    u = 1.0 - h
    
    for ki, wi in zip(top_idx, top_w):
        ki = int(ki)
        # ∇c_ki

        jp = np.arange(0, n - ki)
        gc[jp] = (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        gc[jn] -= h[jn - ki]
        gc *= dx

        gc_minus_g = gc - g
        proj = np.dot(gc_minus_g, v)
        result += alpha * wi * proj * gc_minus_g

        for i in range(n - ki):
            hv[i] -= dx * v[i + ki]
            hv[i + ki] -= dx * v[i]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_weights(c, alpha):
    c_max = float(np.max(c))
    shifted = alpha * (c - c_max)
    shifted = np.maximum(shifted, -700.0)
    w = np.exp(shifted)
    w_sum = float(w.sum())
    w_norm = w / w_sum
    obj = c_max + np.log(w_sum) / alpha
    return obj, w_norm

def grad_lse_from_w(h, w_norm, dx):
    n = len(h)
    u = 1.0 - h
    first = np.zeros(n)
    second = np.zeros(n)
    for j in range(n):
        if n - j > 0:
            first[j] = np.dot(w_norm[:n-j], u[j:])
        second[j] = np.dot(w_norm[:j+1], h[j::-1])
    return dx * (first - second)

def corr_values(h, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr * dx

def hess_apply(h, v, w_norm, alpha, dx, top_m=4):
    n = len(h)
    c = corr_values(h, dx)
    g = grad_lse_from_w(h, w_norm, dx)
    top_idx = np.argsort(w_norm)[::-1][:top_m]
    tw = w_norm[top_idx].copy()
    tw_sum = tw.sum()
    if tw_sum < 1e-30:
        return np.zeros(n)
    tw /= tw_sum
    result = np.zeros(n)
    for ki, wi in zip(top_idx, tw):
        ki = int(ki)
        gc = np.zeros(n)
        jp = np.arange(0, n - ki)
        gc[jp] = (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        gc[jn] -= h[jn - ki]
        gc *= dx
        diff = gc - g
        proj = float(np.dot(diff, v))
        result += alpha * wi * proj * diff
        hv = np.zeros(n)
        if n - ki > 0:
            hv[:n-ki] -= dx * v[ki:]
            hv[ki:] -= dx * v[:n-ki]
        result += wi * hv
    return result

def cg_solve(A_func, b, tol=1e-10, max_it=80, v0=None):
    n = len(b)
    if v0 is not None:
        x = v0.copy()
    else:
        x = np.zeros(n)
    r = b - A_func(x)
    p = r.copy()
    rs = float(np.dot(r, r))
    nrm_b = float(np.linalg.norm(b))
    if nrm_b < 1e-30:
        return x
    for _ in range(max_it):
        if np.sqrt(rs) < tol * max(nrm_b, 1.0):
            break
        Ap = A_func(p)
        pAp = float(np.dot(p, Ap))
        if abs(pAp) < 1e-30:
            break
        alpha_cg = rs / pAp
        x += alpha_cg * p
        r_new = r - alpha_cg * Ap
        rs_new = float(np.dot(r_new, r_new))
        if rs_new >= rs:
            x -= alpha_cg * p
            r = r_new if rs_new < rs else r
            p = r.copy()
            rs = float(np.dot(r, r))
            if np.sqrt(rs) < tol * max(nrm_b, 1.0):
                break
            continue
        r = r_new
        beta = rs_new / rs
        p = r_new + beta * p
        rs = rs_new
    return x

            old_val = h_cur[idx]

def lbfgs_pass(h0, target, deadline, alpha=3000, max_outer=4):

    dx = 2.0 / n
    h = project_feasible(h0, target)
    mu = 50.0

            return val + pen

            c = corr_values(hc, dx)

            gl = grad_lse_from_w(hc, wn, dx)
            gp = np.full(n, mu * (np.sum(hv) - target))

        bounds = [(0.0, 1.0)] * n
        res = minimize(obj_fn, h, method='L-BFGS-B', jac=grad_fn,
                       bounds=bounds, options={'maxiter': 200, 'gtol': 1e-9, 'ftol': 1e-12})
        h = res.x
        h = project_feasible(h, target)
        viol = abs(np.sum(h) - target)
        if viol > 1e-4:

    best_h = project_feasible(h_init_arr, target)
    best_score = compute_c5(best_h)

    def tleft():
        return deadline_final - time.time()

    # Phase 1: Quick FISTA continuation (50s)
    if tleft() > 50:
        for a in [50, 300, 1500]:
            if tleft() < 18:
                break
            h_new, sc_new = fista_pass(best_h, target, a, 10000, start_time + 16)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 2: Gauss-Newton with continuation (150s)
    if tleft() > 140:
        for a in [500, 2000, 8000]:
            if tleft() < 40:
                break
            h_new, sc_new = gauss_newton(best_h, target, a, max_iters=40,
                                         deadline=start_time + 38, top_m=4)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 3: Subgradient polish (60s)
    if tleft() > 50:
        h_new, sc_new = subgrad_polish(best_h, target, start_time + 45, num_active=4, temp=0.002)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: More GN at high alpha (120s)
    if tleft() > 100:
        for a in [15000, 50000]:
            if tleft() < 50:
                break
            h_new, sc_new = gauss_newton(best_h, target, a, max_iters=30,
                                         deadline=start_time + 48, top_m=3)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 5: L-BFGS-B pass (80s)
    if tleft() > 70:
        h_new = lbfgs_pass(best_h, target, start_time + 60, alpha=10000, max_outer=3)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 6: Fine subgradient (50s)
    if tleft() > 40:
        h_new, sc_new = subgrad_polish(best_h, target, start_time + 35, num_active=3, temp=0.001)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: GN polish at very high alpha (80s)
    if tleft() > 70:
        h_new, sc_new = gauss_newton(best_h, target, 100000, max_iters=30,
                                     deadline=start_time + 65, top_m=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 8: Coord descent (remaining)
    if tleft() > 20:
        cd_deadline = min(start_time + max(tleft() - 5, 10), deadline_final - 3)
        h_new = coord_polish(best_h, target, cd_deadline, max_sweeps=50)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Final projection
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
