import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    h = h.copy()
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def alternating_lp(h_init, n, target, dx, time_limit, n_lags=400,
                   damping=0.95, max_iter=300, solver_pref='GLOP'):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    n_half = n - 1
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 80:
            break

        g = 1.0 - h
        overlaps = compute_all_overlaps(h)

        if n_lags == 'all':
            lag_idxs = np.arange(n_full)
        else:
            top_idx = np.argsort(overlaps)[::-1][:n_lags]
            lag_idxs = top_idx

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for idx in lag_idxs:
            k = idx - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solvers = [solver_pref, 'ECOS', 'SCS', 'CLARABEL', 'GLPK']
        solved = False
        for solver in solvers:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck % 20 == 0:

def smooth_obj(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    overlaps = corr * dx
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx):
    n = len(h)
    n_full = 2 * n - 1
    n_half = n - 1
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx

    shifted = alpha * (overlaps - np.max(overlaps))
    shifted = np.clip(shifted, -100.0, 0.0)
    exp_vals = np.exp(shifted)
    p = exp_vals / exp_vals.sum()

    grad = np.zeros(n)
    threshold = p.max() * 1e-8
    active = np.where(p > threshold)[0]
    for idx in active:
        k = idx - n_half
        w = p[idx] * dx
        if k >= 0:
            if n > k:
                m1 = np.arange(k, n)
                grad[m1] += w * (1.0 - h[m1 - k])
                m2 = np.arange(0, n - k)
                grad[m2] += w * (-h[m2 + k])
        else:
            kp = -k
            if n > kp:
                m1 = np.arange(kp, n)
                grad[m1] += w * (1.0 - h[m1 - kp])
                m2 = np.arange(0, n - kp)
                grad[m2] += w * (-h[m2 + kp])

    return grad

                     momentum=0.9, lr_init=0.08, max_steps=300):

    best_score = compute_c5(h)
    v = np.zeros(n)

        for it in range(max_steps):

            y = project_feasible(y, target)

            v = momentum * v - lr * grad
            h_new = h + v
            h_new = project_feasible(h_new, target)

            sc_new = compute_c5(h_new)
            sc_cur = best_score

            if sc_new < sc_cur:
                h = h_new
                lr = min(lr * 1.05, 1.0)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new.copy()
            else:
                lr *= 0.7
                h_new2 = h - lr * grad
                h_new2 = project_feasible(h_new2, target)
                sc2 = compute_c5(h_new2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new2.copy()
                    h = h_new2
                    v = np.zeros(n)
                else:
                    v *= 0.5

    return best_h, best_score


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_polish(h_init, n, target, dx, alphas, maxiter=2000):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    h_cur = h_init.copy()
    h_cur = project_feasible(h_cur, target)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, dx),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def generate_diverse_start(n, target, rng, mode):
    if mode == 0:
        h = np.zeros(n)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0
    elif mode == 1:
        freq = rng.uniform(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.4 * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        h = np.clip(h, 0, 1)
    elif mode == 2:
        h = 0.5 + rng.randn(n) * 0.3
        h = np.clip(h, 0, 1)
    elif mode == 3:
        period = n // (rng.randint(2, 8)) + 1
        h = np.tile(np.array([1.0, 0.0] * (period + 1)), n // (2 * (period + 1)) + 1)[:n]
        h = np.clip(h + rng.randn(n) * 0.1, 0, 1)
    else:
        h = np.full(n, 0.5)
        h[rng.choice(n, n // 4, replace=False)] = 1.0
        h[rng.choice(n, n // 4, replace=False)] = 0.0
    return project_feasible(h, target)
