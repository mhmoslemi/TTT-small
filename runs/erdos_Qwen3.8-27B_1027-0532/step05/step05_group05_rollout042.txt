def compute_over(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2*n-1]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:

            mask = h < 0.99999
        else:

    best_h = h.copy()
    best_score = compute_c5(h)
    
    for t in range(max_iter):
        if time.time() > deadline:
            break
        over = compute_over(h)
        c_max = float(np.max(over))
        score = c_max * dx
        
        if score < best_score:
            best_score = score
            best_h = h.copy()
        
        # Temperature schedule
        frac = t / max(max_iter - 1, 1)
        temp = temp_init * (temp_final / temp_init) ** frac
        
        # Softmax weights
        shifted = over - c_max
        weights = np.exp(shifted / max(temp, 1e-10))
        weights /= weights.sum()
        
        # Gradient
        g = np.zeros(n)
        for k in range(n):
            if weights[k] < 1e-12:
                continue
            g[k:] += weights[k] * (1.0 - h[:n-k])
            g[:n-k] -= weights[k] * h[k:]
        g *= dx
        
        # FW step
        order = np.argsort(g)
        s = np.zeros(n)
