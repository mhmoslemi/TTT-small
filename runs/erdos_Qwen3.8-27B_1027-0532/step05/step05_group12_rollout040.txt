def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_values(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def alternating_lp_adaptive(h_init, n, target, time_limit, start_time, deadline_offset=30,
                            M_start=8, M_max=300, damping=0.5, max_outer=500, reg=1e-5):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0

    for outer in range(max_outer):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        c_max = np.max(corr)
        threshold = c_max - 0.02 * c_max
        n_active = int(np.sum(corr >= threshold))
        M = max(M_start, min(n_active + 3, M_max))

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        top_idx = np.argsort(corr)[-M:]
        n_half = n - 1
        for idx in top_idx:
            k = idx - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                a = np.zeros(n)
                a[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(a @ h_var <= t_var)

        obj = cp.Minimize(t_var + reg * cp.sum_squares(h_var - h))
        prob = cp.Problem(obj, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iter=200000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break
        if no_improve % 10 == 0:
            damping = max(0.3, damping * 0.95)

    return best_h, best_c5

def proj_grad_method(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    top_m_list = [20, 30, 50, 80, 120, 200, 400, 2 * n - 1, 2 * n - 1, 2 * n - 1]

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]

        momentum_v = np.zeros(n)
        beta = 0.8
        stalled = 0
        max_steps = 5000

        for step in range(max_steps):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            corr = np.correlate(h, 1.0 - h, mode='full')
            c_max = np.max(corr)
            if top_m < 2 * n - 1:
                top_indices = np.argsort(corr)[-top_m:]
                sub_corr = corr[top_indices]
            else:
                top_indices = np.arange(2 * n - 1)
                sub_corr = corr
            shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
            exp_shifted = np.exp(shifted)
            wsum = np.sum(exp_shifted)
            if wsum < 1e-300:
                break
            weights = exp_shifted / wsum

            grad = np.zeros(n)
            n_half = n - 1
            for idx_k, w in zip(top_indices, weights):
                if w < 1e-14:
                    continue
                k = idx_k - n_half
                if k == 0:
                    grad += w * (1.0 - 2.0 * h) * dx
                elif k > 0:
                    if n - k > 0:
                        grad[k:] += w * (1.0 - h[:n - k]) * dx
                        grad[:n - k] -= w * h[k:] * dx
                else:
                    m = -k
                    if n - m > 0:
                        grad[:n - m] += w * (1.0 - h[m:]) * dx
                        grad[m:] -= w * h[:n - m] * dx

            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-13:
                break

            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 1.0)

            improved = False
            for _ in range(30):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5

            if not improved:
                stalled += 1
                if stalled > 80:
                    break

    return best_h, best_score


def block_coordinate_descent(h_init, n, target, time_limit, start_time, deadline_offset=30,
                             block_size=20, max_sweeps=200):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit - deadline_offset:
            break

        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            mask = np.zeros(n, dtype=bool)
            mask[start:end] = True

            g = 1.0 - h
            corr = np.correlate(h, g, mode='full')

            h_var = cp.Variable(end - start)
            t_var = cp.Variable()

            constraints_blk = []
            fixed_sum = np.sum(h[~mask])
            constraints_blk.append(cp.sum(h_var) == target - fixed_sum)
            constraints_blk.append(h_var >= 0.0)
            constraints_blk.append(h_var <= 1.0)

            top_idx = np.argsort(corr)[-50:]
            n_half = n - 1
            for idx in top_idx:
                k = idx - n_half
                lo = max(0, k)
                hi = min(n, k + n)
                if hi > lo:
                    a_full = np.zeros(n)
                    a_full[lo:hi] = g[lo - k:hi - k] * dx
                    a_blk = a_full[start:end]
                    rest = a_full.copy()
                    rest[start:end] = 0
                    const = np.dot(rest, h)
                    constraints_blk.append(a_blk @ h_var + const <= t_var)

            obj = cp.Minimize(t_var + 0.1 * cp.sum_squares(h_var - h[start:end]))
            prob = cp.Problem(obj, constraints_blk)
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                continue

            if h_var.value is None:
                continue

            h_block_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
            h[start:end] = h_block_new

        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP with adaptive active set
    if tleft() > 200:
        lp_time = min(250, tleft() * 0.35)
        h_lp, c5_lp = alternating_lp_adaptive(h_init, n, target,
                                              time_limit=lp_time,
                                              start_time=start_time,
                                              deadline_offset=100,
                                              M_start=8, M_max=200,
                                              damping=0.5, max_outer=500,
                                              reg=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Full LP polish with smaller active set threshold
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.3)
        h_lp2, c5_lp2 = full_lp_polish(best_h, n, target,
                                       time_limit=lp_time,
                                       start_time=start_time,
                                       deadline_offset=80,
                                       M_start=30, M_max=500,
                                       damping=0.7, max_outer=200,
                                       reg=1e-6)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 3: Projected gradient with high alpha
    if tleft() > 150:
        pg_time = min(300, tleft() * 0.4)
        h_pg, c5_pg = proj_grad_method(best_h, n, target,
                                       time_limit=pg_time,
                                       start_time=start_time,
                                       deadline_offset=60)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_slq = slsqp_optimize(best_h, n, target,
                               [100, 500, 2000, 10000, 50000],
                               maxiter=3000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=50)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: Block coordinate descent
    if tleft() > 100:
        bcd_time = min(150, tleft() * 0.3)
        h_bcd, c5_bcd = block_coordinate_descent(best_h, n, target,
                                                 time_limit=bcd_time,
                                                 start_time=start_time,
                                                 deadline_offset=50,
                                                 block_size=15, max_sweeps=500)
        if c5_bcd < best_score:
            best_score = c5_bcd
            best_h = h_bcd

    # Phase 6: Another alternating LP round
    if tleft() > 100:
        lp_time = min(150, tleft() * 0.3)
        h_lp3, c5_lp3 = alternating_lp_adaptive(best_h, n, target,
                                                time_limit=lp_time,
                                                start_time=start_time,
                                                deadline_offset=60,
                                                M_start=10, M_max=300,
                                                damping=0.6, max_outer=400,
                                                reg=5e-6)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 7: Try higher resolution
    if tleft() > 200:
        n_try = n + 100
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        if tleft() > 150:
            lp_time = min(120, tleft() * 0.35)
            h_lp_h, c5_lp_h = alternating_lp_adaptive(h_interp, n_try, target_try,
                                                      time_limit=lp_time,
                                                      start_time=start_time,
                                                      deadline_offset=80,
                                                      M_start=10, M_max=200,
                                                      damping=0.5, max_outer=300,
                                                      reg=1e-5)
            if c5_lp_h < c5_lp3 if tleft() > 0 else False:
                pass
            if c5_lp_h < best_score:
                best_score = c5_lp_h
                best_h = h_lp_h
                n = n_try
                target = target_try

        if tleft() > 100:
            pg_time = min(100, tleft() * 0.3)
            h_pg_h, c5_pg_h = proj_grad_method(best_h, n, target,
                                               time_limit=pg_time,
                                               start_time=start_time,
                                               deadline_offset=50)
            if c5_pg_h < best_score:
                best_score = c5_pg_h
                best_h = h_pg_h

        if tleft() > 60:
            h_slq_h = slsqp_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                     [500, 2000, 10000],
                                     maxiter=2000, ftol=1e-18,
                                     time_limit=budget_s, start_time=start_time,
                                     deadline_offset=30)
            sc = compute_c5(h_slq_h)
            if sc < best_score:
                best_score = sc
                best_h = h_slq_h

    # Phase 8: Diverse restarts
    for noise_scale in [0.005, 0.01]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise_scale
        h_pert = project_feasible(h_pert, len(h_pert) / 2.0)
        h_slq_r = slsqp_optimize(h_pert, len(h_pert), len(h_pert) / 2.0,
                                 [500, 2000, 10000, 50000],
                                 maxiter=2000, ftol=1e-18,
                                 time_limit=budget_s, start_time=start_time,
                                 deadline_offset=30)
        sc = compute_c5(h_slq_r)
        if sc < best_score:
            best_score = sc
            best_h = h_slq_r

    # Phase 9: Final alternating LP + SLSQP
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(60, tleft() * 0.4)
        h_fin_lp, c5_fin_lp = alternating_lp_adaptive(best_h, n_use, target_use,
                                                      time_limit=lp_time,
                                                      start_time=start_time,
                                                      deadline_offset=20,
                                                      M_start=15, M_max=400,
                                                      damping=0.6, max_outer=200,
                                                      reg=5e-6)
        if c5_fin_lp < best_score:
            best_score = c5_fin_lp
            best_h = h_fin_lp

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                               [2000, 10000, 50000, 200000],
                               maxiter=3000, ftol=1e-18,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=5)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
