```python
def find_t_star(C, B):
    """Find t in [0,1] minimizing max_k(C_k + B_k*t)"""
    # Initial estimate
    k0 = int(np.argmax(C))
    k1 = int(np.argmax(C + B))
    
    if k0 == k1:
        # Single line dominates at both ends
        if C[k0] <= C[k0] + B[k0]:
            return 0.0  # decreasing, min at 0
        else:
            return 1.0  # wait, if C[k0] > C[k0]+B[k0], B[k0] < 0, so g(1) < g(0), min at 1
    
    if B[k0] >= 0 and B[k1] <= 0:
        # Lines cross
        if B[k0] != B[k1]:
            t = (C[k1] - C[k0]) / (B[k0] - B[k1])
            return np.clip(t, 0.0, 1.0)
    
    # Fallback: bisection
    lo, hi = 0.0, 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        # Subgradient at mid
        vals = C + B * mid
        mx = vals.max()
        active = np.where(np.abs(vals - mx) < 1e-10 * (1 + abs(mx)))[0]
        sg_min = B[active].min()
        sg_max = B[active].max()
        if sg_min > 0:
            hi = mid
        elif sg_max < 0:
            lo = mid
        else:
            break
    return (lo + hi) / 2
```

One more consideration: the "sum constraint" issue. In coordinate descent, when I change h[i], the sum changes. I have two options:
1. Use a penalty term μ*(sum - target)² 
2. After each sweep, project back to sum = target

Option 2 is simpler and probably works well in practice. The projection is: h += (target - sum(h))/n for all i, then clip to [0,1] and iterate.

A better approach: incorporate the penalty into the coordinate optimization. When optimizing h[i], minimize:

where D = (sum of other h_j) - target.

This is convex in t (max of linear + quadratic). I can find the minimum efficiently.

The max term has subgradient {B_k : k in argmax(C_k + B_k*t)}.

At the optimum: 0 ∈ {B_k + 2μ(D+t) : k in argmax}

This means: -2μ(D+t) ∈ {B_k : k in argmax}

So: D + t = -B_k*/(2μ) for some active k*.

This gives t* = -D - B_k*/(2μ) for some active k*.

    c_max = float(over.max())
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_cons(x):
    return np.sum(x) - _g_target

def slsqp_cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}]
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n-1:n-1+n]
        c_curr = float(over.max()) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Coordinate descent (exact per-coordinate optimization)
    if tleft() > 400:
        h_cd, sc = coord_descent(h_init, n_sweeps=400, mu=50.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc

    # Phase 2: SLSQP polish
    if tleft() > 400:
        h_sl, sc = slsqp_optimize(best_h, [500, 2000, 10000], 3000, 1e-15)
        if sc < best_score:
            best_h, best_score = h_sl, sc

    # Phase 3: Another coordinate descent pass with different mu
    if tleft() > 300:
        h_cd, sc = coord_descent(best_h, n_sweeps=500, mu=20.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc

    # Phase 4: Subgradient with momentum
    if tleft() > 250:
        h_sg, sc = subgradient_momentum(best_h, 500, 5, 0.003, 0.997, 0.3, 0.004)
        if sc < best_score:
            best_h, best_score = h_sg, sc
        h_cd, sc = coord_descent(best_h, n_sweeps=300, mu=30.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc

    # Phase 5: Random restarts with coordinate descent
    for attempt in range(40):
        if tleft() < 40:
            break
        h_pert = best_h.copy()
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        strategy = attempt % 6
        if strategy == 0:
            h_pert += rng.randn(n_cur) * 0.02
        elif strategy == 1:
            h_pert += rng.randn(n_cur) * 0.05
        elif strategy == 2:
            block = max(5, n_cur // 30)
            for b in range(n_cur // block):
                if rng.random() < 0.5:
                    h_pert[b*block:min((b+1)*block, n_cur)] += rng.uniform(-0.08, 0.08)
        elif strategy == 3:
            freq = 2.0 * np.pi * rng.randint(2, 15) / n_cur
            phase = rng.rand() * 2.0 * np.pi
            h_pert += 0.03 * np.sin(np.arange(n_cur) * freq + phase)
        elif strategy == 4:
            n_flip = max(5, n_cur // 25)
            idx = rng.choice(n_cur, n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            # Antisymmetric perturbation
            h_pert += 0.04 * np.sin(np.pi * np.arange(n_cur) / n_cur * rng.randint(1, 8))

        h_pert = project_feasible(h_pert, target_cur)
        h_cd, sc = coord_descent(h_pert, n_sweeps=200, mu=50.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc
            if tleft() > 120:
                h_sl, sc2 = slsqp_optimize(h_cd, [2000, 10000], 1500, 1e-15)
                if sc2 < best_score:
                    best_h, best_score = h_sl, sc2
                h_cd2, sc3 = coord_descent(best_h, n_sweeps=200, mu=20.0)
                if sc3 < best_score:
                    best_h, best_score = h_cd2, sc3

    # Phase 6: Final SLSQP high precision
    if tleft() > 80:
        h_sl, sc = slsqp_optimize(best_h, [5000, 20000, 100000], 2000, 1e-16)
        if sc < best_score:
            best_h, best_score = h_sl, sc

    # Phase 7: Final coordinate descent
    if tleft() > 40:
        h_cd, sc = coord_descent(best_h, n_sweeps=500, mu=10.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc

    # Phase 8: Very fine coordinate descent
    if tleft() > 25:
        h_cd, sc = coord_descent(best_h, n_sweeps=300, mu=5.0)
        if sc < best_score:
            best_h, best_score = h_cd, sc

    # Final one-shot SLSQP
    if tleft() > 15:
        global _g_n, _g_target, _g_alpha
        _g_n = len(best_h)
        _g_target = _g_n / 2.0
        _g_alpha = 50000.0
        bounds = [(0.0, 1.0)] * _g_n
        cons = [{'type': 'eq', 'fun': slsqp_cons, 'jac': slsqp_cons_jac}]
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 500, 'ftol': 1e-17})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_h, best_score = h_new, sc
        except Exception:
            pass

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
