```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_gradients(h, lags):
    """Compute gradients of correlation w.r.t. h for given lags."""
    n = len(h)
    grads = np.zeros((len(lags), n))
    for m, k in enumerate(lags):
        # c[k] = sum_i h[i] * (1-h[i+k-n+1])
        # dc[k]/dh[i] = (1-h[i+k-n+1]) - h[i-k+n-1] (when indices valid)
        offset = k - n + 1
        # h[i] appears in two places:
        # As the first element: h[i] * (1-h[i+offset]) when i+offset in [0,n-1]
        # As the second element: h[i-offset] * (1-h[i]) when i-offset in [0,n-1]
        for i in range(n):
            g = 0.0
            # First occurrence: h[i] * (1-h[i+offset])
            j = i + offset
            if 0 <= j < n:
                g += (1.0 - h[j])
            # Second occurrence: h[i-offset] * (1-h[i])
            j2 = i - offset
            if 0 <= j2 < n:
                g -= h[j2]
            grads[m, i] = g
    return grads
```

```python
def compute_gradient_for_lag(h, k):
    n = len(h)
    offset = k - n + 1
    grad = np.zeros(n)
    # v[i+offset] where v = 1-h
    if offset >= 0:
        grad[:n-offset] += (1.0 - h[offset:])
    else:
        grad[-offset:] += (1.0 - h[:n+offset])
    # -h[i-offset]
    if offset >= 0:
        grad[offset:] -= h[:n-offset]
    else:
        grad[:n+offset] -= h[-offset:]
    return grad
```

```python
def project_feasible(h, target_sum):
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_gradient_for_lag(h, k):
    n = len(h)
    offset = k - n + 1
    grad = np.zeros(n)
    if offset >= 0:
        grad[:n - offset] += (1.0 - h[offset:])
        grad[offset:] -= h[:n - offset]
    else:
        grad[-offset:] += (1.0 - h[:n + offset])
        grad[:n + offset] -= h[-offset:]
    return grad

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    if c_max == 0:
        return 0.0
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def lp_guided_steps(h, target, max_iters, M, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:M]
        c_active = corr[top_idx]
        G = np.array([get_gradient_for_lag(h_cur, k) for k in top_idx])
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ineq = np.hstack([G, -np.eye(M)])
        b_ineq = -c_active
        A_neg = np.zeros((n, n + 1))
        A_neg[:, :n] = -np.eye(n)
        A_pos = np.zeros((n, n + 1))
        A_pos[:, :n] = np.eye(n)
        A_ub = np.vstack([A_ineq, A_neg, A_pos])
        b_ub = np.concatenate([b_ineq, np.zeros(n), 1.0 - h_cur])
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
        except Exception:
            break
        if not res.success:
            break
        delta = res.x[:n]
        max_abs = float(np.max(np.abs(delta)))
        if max_abs < 1e-12:
            break
        direction = delta / max_abs
        best_h = h_cur
        best_sc = cur_score
        steps = [min(max_abs, 0.05), min(max_abs, 0.02), min(max_abs, 0.01),
                 min(max_abs, 0.005), min(max_abs, 0.002), min(max_abs, 0.001)]
        for alpha in steps:
            if alpha < 1e-8:
                continue
            h_try = h_cur + alpha * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_h = h_try
        if best_sc >= cur_score - 1e-13:
            break
        h_cur = best_h
        cur_score = best_sc
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.004)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            grad += wk * get_gradient_for_lag(h_cur, int(k))
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score:
            h_cur = h_new
            best_h = h_new.copy()
            best_score = sc_new
            lr = min(lr * 1.02, lr_init * 2.5)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

def flat_top_equalize(h, target, n_lags, n_iters, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_lags]
        c_vals = corr[top_idx]
        c_mean = c_vals.mean()
        G = np.array([get_gradient_for_lag(h_cur, k) for k in top_idx])
        target_adj = c_mean - c_vals
        delta = np.zeros(n)
        for m in range(n_lags):
            delta += target_adj[m] * G[m]
        delta_sum = np.sum(delta)
        delta -= delta_sum / n
        max_abs = float(np.max(np.abs(delta)))
        if max_abs < 1e-14:
            break
        direction = delta / max_abs
        best_h = h_cur
        best_sc = cur_score
        for alpha in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_try = h_cur + alpha * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_h = h_try
        if best_sc >= cur_score - 1e-13:
            break
        h_cur = best_h
        cur_score = best_sc
    return h_cur, cur_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 6)
                amp = rng.uniform(0.1, 0.5)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.35, 0.05, 0.95), target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score
```
