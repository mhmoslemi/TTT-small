            return np.sum(x) - target

            return np.ones(n)

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

                c[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(c @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5


def _smooth_obj(x):
    n = _global_n
    alpha = _global_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    dx = 2.0 / n
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha)) * dx


def _cons_fun(x):
    return np.sum(x) - _global_target


def _cons_jac(x):
    return np.ones(_global_n)


def slsqp_smooth(h_start, n, target, alphas, maxiter):
    global _global_n, _global_alpha, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()

    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def perturb_targeted(h, target, n, rng, strength=0.005, n_lags=20):
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    max_val = np.max(overlaps)
    n_half = n - 1
    sorted_idx = np.argsort(overlaps)[::-1]
    secondary_idx = sorted_idx[3:3 + n_lags]
    h_new = h.copy()
    for idx in secondary_idx:
        k = idx - n_half
        grad_k = np.zeros(n)
        if k >= 0:
            if k == 0:
                grad_k[:] = 2.0 * (1.0 - h)
            else:
                grad_k[:n - k] = (1.0 - h[k:])
                grad_k[k:] = (-h[:n - k])
        else:
            k2 = -k
            if k2 == 0:
                grad_k[:] = 2.0 * (1.0 - h)
            else:
                grad_k[:n - k2] = (1.0 - h[k2:])
                grad_k[k2:] = (-h[:n - k2])
        grad_k *= dx
        h_new = h_new - strength * grad_k
        h_new = project_feasible(h_new, target)
    return h_new


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_alpha, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 150:
        ft = min(100, tleft() * 0.22)
        h_f, c5_f = fista_exact(best_h, n, target, max_steps=50000, time_limit=ft, epsilon_frac=0.02)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    if tleft() > 150:
        lt = min(180, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target, max_iter=5000, time_limit=lt,
                                      n_top=200, damping_start=0.9, damping_end=0.7,
                                      reg_start=1e-5, reg_end=1e-9)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 100:
        h_s = slsqp_smooth(best_h, n, target,
                            [100, 500, 2000, 10000, 50000, 200000], maxiter=2000)
        sc = compute_c5(h_s)
        if sc < best_score:
            best_score = sc
            best_h = h_s

    if tleft() > 100:
        ft2 = min(80, tleft() * 0.2)
        h_f2, c5_f2 = fista_exact(best_h, n, target, max_steps=50000, time_limit=ft2, epsilon_frac=0.01)
        if c5_f2 < best_score:
            best_score = c5_f2
            best_h = h_f2

    if tleft() > 100:
        lt2 = min(90, tleft() * 0.25)
        h_lp2, c5_lp2 = alternating_lp(best_h, n, target, max_iter=3000, time_limit=lt2,
                                        n_top=250, damping_start=0.85, damping_end=0.6,
                                        reg_start=1e-6, reg_end=1e-10)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    if tleft() > 80:
        h_s2 = slsqp_smooth(best_h, n, target,
                             [2000, 10000, 50000, 200000, 500000], maxiter=1500)
        sc = compute_c5(h_s2)
        if sc < best_score:
            best_score = sc
            best_h = h_s2

    for attempt in range(5):
        if tleft() < 70:
            break
        h_pert = perturb_targeted(best_h, target, n, rng, strength=0.003 + attempt * 0.002, n_lags=15)
        h_pf, c5_pf = fista_exact(h_pert, n, target, max_steps=20000,
                                   time_limit=min(30, tleft() * 0.3), epsilon_frac=0.02)
        if c5_pf < best_score:
            best_score = c5_pf
            best_h = h_pf
        else:
            if tleft() > 40:
                h_pl, c5_pl = alternating_lp(h_pert, n, target, max_iter=1000, time_limit=25,
                                              n_top=150, damping_start=0.8, damping_end=0.5,
                                              reg_start=1e-4, reg_end=1e-7)
                if c5_pl < best_score:
                    best_score = c5_pl
                    best_h = h_pl

    if tleft() > 140:
        for n_high in [500, 600]:
            if tleft() < 90:
                break
            target_h = n_high / 2.0
            h_hi = interpolate_to(best_h, n_high, target_h)
            h_hi, c5_hi = alternating_lp(h_hi, n_high, target_h, max_iter=2000,
                                          time_limit=min(60, tleft() * 0.3),
                                          n_top=200, damping_start=0.85, damping_end=0.6,
                                          reg_start=1e-5, reg_end=1e-8)
            h_hi, c5_hi = fista_exact(h_hi, n_high, target_h, max_steps=30000,
                                       time_limit=min(30, tleft() * 0.2), epsilon_frac=0.02)
            sc = compute_c5(h_hi)
            if sc < best_score:
                best_score = sc
                best_h = h_hi
                n = n_high
                target = n_high / 2.0

    if tleft() > 50:
        h_fin = slsqp_smooth(best_h, n, target,
                              [1000, 5000, 20000, 100000, 500000], maxiter=2000)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    if tleft() > 30:
        h_ff, c5_ff = fista_exact(best_h, n, target, max_steps=30000,
                                   time_limit=min(25, tleft() * 0.8), epsilon_frac=0.01)
        if c5_ff < best_score:
            best_score = c5_ff
            best_h = h_ff

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
