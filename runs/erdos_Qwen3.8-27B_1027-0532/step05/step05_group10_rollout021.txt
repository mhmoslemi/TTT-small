def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    """Returns (smooth_max_value, gradient) of smooth max of corr*dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c = corr * dx
    
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum
    F = c_max + np.log(w_sum) / alpha
    
    # Gradient computation
    v = 1.0 - 2.0 * h  # length n
    if n <= 500:
        # Direct
        grad = np.array([np.dot(w[j:j+n], v) for j in range(n)])
    else:

        while nfft < 3 * n:
            nfft *= 2
        v_ext = np.zeros(nfft)

        grad = conv[n-1:2*n-1][::-1]
    
    grad *= dx
    return F, grad

def lbfpbs_optimize(h_init, n, target, alphas, maxiter, time_limit, mu=1e4):
    """L-BFGS-B optimization with penalty for sum constraint."""
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    def objective(x, alpha):
        F, _ = smooth_max_grad(x, alpha)
        penalty = 0.5 * mu * (np.sum(x) - target) ** 2
        return F + penalty
    
    def gradient(x, alpha):
        F, g = smooth_max_grad(x, alpha)
        g += mu * (np.sum(x) - target) * np.ones(n)
        return g
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(objective, h_cur, args=(alpha,),
                          method='L-BFGS-B', jac=gradient, bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    
    return best_h, best_score

def projected_gradient(h_init, n, target, alpha, n_steps, time_limit, lr=None):
    """Projected gradient with backtracking line search."""
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    if lr is None:
        lr = 0.01
    
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        
        F, g = smooth_max_grad(h_cur, alpha)
        # Add Lagrange multiplier for sum constraint (use penalty-like approach)
        mu = 1e3
        g += mu * (np.sum(h_cur) - target) * np.ones(n)
        
        # Backtracking line search

        improved = False

                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
            step_size *= 0.8
        
        if not improved and step > 200:
            # Try smaller learning rate
            lr *= 0.5
        
    return best_h, best_score

            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    w_sum = np.sum(exp_shifted)
    w = exp_shifted / w_sum
    F = c_max + np.log(w_sum) / alpha
    v = 1.0 - 2.0 * h
    if n <= 600:
        grad = np.zeros(n)
        for j in range(n):
            grad[j] = np.dot(w[j:j + n], v)
    else:
        nfft = 1
        while nfft < 3 * n:
            nfft *= 2
        v_ext = np.zeros(nfft)
        v_ext[:n] = v
        w_r_ext = np.zeros(nfft)
        w_r_ext[:2 * n - 1] = w[::-1]
        conv = np.fft.irfft(np.fft.rfft(v_ext) * np.fft.rfft(w_r_ext), nfft)
        grad = conv[n - 1:2 * n - 1][::-1]
    grad *= dx
    return F, grad


def penalty_objective(x, alpha, mu):
    n = _global_n
    target = _global_target
    F, _ = smooth_max_value_and_grad(x, alpha)
    s = np.sum(x) - target
    return F + 0.5 * mu * s * s


def penalty_gradient(x, alpha, mu):
    n = _global_n
    target = _global_target
    F, g = smooth_max_value_and_grad(x, alpha)
    s = np.sum(x) - target
    g = g + mu * s * np.ones(n)
    return g


def lbfbs_round(h_init, n, target, alphas, maxiter_per_alpha, time_limit, mu=1e4):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(penalty_objective, h_cur, args=(alpha, mu),
                           method='L-BFGS-B', jac=penalty_gradient,
                           bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': 1e-20, 'gtol': 1e-16,
                                    'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_score


def nesterov_projected_grad(h_init, n, target, alpha, n_steps, time_limit, lr_init=0.01):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stuck = 0
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        F, g = smooth_max_value_and_grad(h, alpha)
        s = np.sum(h) - target
        g = g + 1e3 * s * np.ones(n)
        v = 0.9 * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
            stuck = 0
            lr = min(lr * 1.01, 0.1)
        else:
            stuck += 1
            if stuck > 50:
                lr *= 0.7
                v *= 0.5
                if lr < 1e-6:
                    break
    return best_h, best_score


def build_zadoff_like(n, freq_scale):
    idx = np.arange(n)
    h = 0.5 + 0.5 * np.cos(freq_scale * np.pi * idx * (idx + 1) / n)
    return h


def build_fourier_init(n, k_modes, rng):
    h = np.zeros(n)
    idx = np.arange(n)
    for k in k_modes:
        amp = rng.uniform(0.2, 0.5)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * idx / n + phase)
    h = np.clip(h, 0, 1)
    return h


def active_set_lp(h_init, n, target, K, max_iter, time_limit):
    import cvxpy as cp
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        top_m = np.argsort(c)[-K:]
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for m_idx in sorted(top_m):
            m = int(m_idx)
            k = m - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + 1e-5 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.7 * h + 0.3 * h_new
        h = project_feasible(h, target)
        sc = compute_c5(h)
        if sc < best_c5:
            best_c5 = sc
            best_h = h.copy()
    return best_h, best_c5
