import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target_sum - 1] if target_sum < n else 0
    h_bin = (h > threshold).astype(float)
    # Handle ties
    tied = np.where(np.abs(h - threshold) < 1e-10)[0]
    current = h_bin.sum()
    if current != target_sum:
        if current < target_sum:
            need = int(target_sum - current)
            tied_zeros = [i for i in tied if h_bin[i] == 0]
            for i in tied_zeros[:need]:
                h_bin[i] = 1
        else:
            need = int(current - target_sum)
            tied_ones = [i for i in tied if h_bin[i] == 1]
            for i in tied_ones[:need]:
                h_bin[i] = 0
    # Fix if still not exactly target_sum
    while h_bin.sum() > target_sum:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[np.argmin(h[idx])]] = 0
    while h_bin.sum() < target_sum:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[np.argmax(h[idx])]] = 1
    return h_bin

            if res.x is not None:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def legendre_init(n, rng):
    p = n if n % 2 == 1 else n + 1
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2

    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0

    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
        h[p:] = 0.5

    target = n // 2
    h = project_binary(h, target)
    return h


def ilt_sine_init(n, rng):
    target = n // 2
    freq = rng.uniform(1.0, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 1, n)
    h = (np.sin(2 * np.pi * freq * x + phase) > 0).astype(float)
    h = project_binary(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(budget_s - 15.0 - (time.time() - start_time), 0.0)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy from initial
    allowed = min(80, tleft() * 0.10)
    h_g, c5_g = greedy_improve(best_h, max_iters=80000, time_limit=allowed,
                               start_time=start_time, scan_width=40)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: SA from greedy
    allowed = min(250, tleft() * 0.22)
    max_it = int(min(400000, allowed * 1200))
    h_sa, c5_sa = sa_optimize(best_h, target_init, max_it, 0.015, 0.00002, rng,
                              time_limit=allowed, start_time=start_time)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Second SA with cooler schedule
    if tleft() > 150:
        allowed = min(180, tleft() * 0.18)
        max_it = int(min(300000, allowed * 1200))
        h_sa2, c5_sa2 = sa_optimize(best_h, target_init, max_it, 0.008, 0.000005, rng,
                                    time_limit=allowed, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: L-BFGS-B continuous optimization
    if tleft() > 100:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_lb = [30, 80, 200, 500, 1500, 5000, 15000, 50000]
        allowed = min(200, tleft() * 0.20)
        h_ref, sc = lbfgsb_refine(best_h, n_use, target_use, alphas_lb, 2000,
                                   time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 5: SLSQP refinement
    if tleft() > 80:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_sls = [100, 400, 1500, 5000, 20000, 80000]
        allowed = min(150, tleft() * 0.15)
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas_sls, 3000, 1e-15,
                                   time_limit=allowed, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2.copy()

    # Phase 6: Binary projection + more SA
    if tleft() > 80:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(120, tleft() * 0.15)
        max_it5 = int(min(200000, allowed * 1200))
        h_sf, c5_sf = sa_optimize(h_bin_f, target_use, max_it5, 0.008, 0.00001,
                                  rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Phase 7: ILS perturbations from best
    if tleft() > 100:
        n_best = len(best_h)
        t_best = int(n_best / 2.0)
        for pert_iter in range(6):
            if tleft() < 50:
                break
            n_pert = 2 + pert_iter
            h_pert = best_h.copy()
            h_pert_bin = project_binary(h_pert, t_best)
            ones = np.where(h_pert_bin == 1)[0]
            zeros = np.where(h_pert_bin == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert_bin[i] = 0
                h_pert_bin[j] = 1
                ones = np.where(h_pert_bin == 1)[0]
                zeros = np.where(h_pert_bin == 0)[0]

            allowed = min(40, tleft() * 0.06)
            h_pg, c5_pg = greedy_improve(h_pert_bin, max_iters=30000, time_limit=allowed,
                                         start_time=start_time, scan_width=30)
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

            if tleft() < 40:
                break
            allowed2 = min(50, tleft() * 0.08)
            max_it4 = int(min(80000, allowed2 * 1200))
            h_ps, c5_ps = sa_optimize(h_pg, t_best, max_it4, 0.006, 0.00001,
                                      rng, time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 8: Try alternative n values
    n_options = [n_init + 2, n_init + 4, n_init - 2, n_init + 6, n_init - 4]
    n_options = [x for x in n_options if x >= 400 and x % 2 == 0 and x != n_init]

    for n_try in n_options:
        if tleft() < 60:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin = project_binary(h_interp, target_try)

        allowed = min(25, tleft() * 0.05)
        h_g2, c5_g2 = greedy_improve(h_bin, max_iters=20000, time_limit=allowed,
                                     start_time=start_time, scan_width=25)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 40:
            continue
        allowed2 = min(30, tleft() * 0.06)
        max_it2 = int(min(60000, allowed2 * 1500))
        h_sa3, c5_sa3 = sa_optimize(h_g2, target_try, max_it2, 0.01, 0.00005, rng,
                                    time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 9: Legendre and sine-based initializations
    if tleft() > 80:
        for init_fn in [lambda n_, r_: legendre_init(n_, r_),
                        lambda n_, r_: ilt_sine_init(n_, r_)]:
            if tleft() < 50:
                break
            h_alt = init_fn(n_init, rng)
            allowed = min(25, tleft() * 0.05)
            h_ag, c5_ag = greedy_improve(h_alt, max_iters=20000, time_limit=allowed,
                                         start_time=start_time, scan_width=30)
            if c5_ag < best_score:
                best_score = c5_ag
                best_h = h_ag.copy()
            if tleft() > 40:
                allowed2 = min(35, tleft() * 0.07)
                max_it3 = int(min(60000, allowed2 * 1200))
                h_as, c5_as = sa_optimize(h_ag, target_init, max_it3, 0.01, 0.00003,
                                          rng, time_limit=allowed2, start_time=start_time)
                if c5_as < best_score:
                    best_score = c5_as
                    best_h = h_as.copy()

    # Phase 10: Final L-BFGS-B + SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(60, tleft() * 0.4)
        alphas_final = [2000, 10000, 50000, 200000]
        h_ref3, sc3 = lbfgsb_refine(best_h, n_use, target_use, alphas_final, 1500,
                                     time_limit=allowed * 0.6, start_time=start_time)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_ref3.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_sls2 = [5000, 20000, 100000, 500000]
        allowed = min(30, max(5.0, tleft() - 5))
        h_ref4, sc4 = slsqp_refine(best_h, n_use, target_use, alphas_sls2, 2000, 1e-16,
                                   time_limit=allowed, start_time=start_time)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_ref4.copy()

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
