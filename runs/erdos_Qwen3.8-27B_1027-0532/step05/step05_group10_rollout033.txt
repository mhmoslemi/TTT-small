```python
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    h = h.copy()
    h = project_feasible(h, target)
    start = time.time()
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    half = n - 1
    m = 2 * n - 1  # length of corr
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        any_improved = False
        # Randomize order for diversity
        order = np.random.permutation(n) if sweep > 0 else np.arange(n)
        
        for i in order:
            old_hi = h[i]
            if old_hi < 1e-10 or old_hi > 1.0 - 1e-10:
                # Already binary, try the other value
                candidates = [1.0] if old_hi < 0.5 else [0.0]
            else:
                candidates = [0.0, 1.0]

            best_local_val = old_hi

                if abs(delta) < 1e-12:
                    continue
                
                # Compute f_i (the effect on corr)
                # f_i[idx] for idx in [0, m-1], corresponding to shift s = idx - half
                # f_i[idx] = (1-h[i+s])*1_{0<=i+s<n} - h[i-s]*1_{0<=i-s<n}
                
                # i+s in [0,n-1]: s in [-i, n-1-i], idx in [half-i, half+n-1-i]
                lo1 = max(0, half - i)

                lo2 = max(0, half + i - (n - 1))
                hi2 = min(m - 1, half + i)
                
                new_corr = corr.copy()  # This is O(n) copy
                if hi1 >= lo1:
                    new_corr[lo1:hi1+1] += delta * (1.0 - h[lo1 - half + i : hi1 - half + i + 1])
                    # Wait, i+s where s = idx - half, so i+s = i + idx - half
                    # For idx in [lo1, hi1]: i+s in [i+lo1-half, i+hi1-half]
                    # = [i + half - i - half, i + half + n - 1 - i - half] = [0, n-1] ✓
                    # Actually: lo1 = max(0, half-i), so i+lo1-half = i + max(0,half-i) - half
                    # If half >= i: i + (half-i) - half = 0 ✓
                    # If half < i: i + 0 - half = i - half. But lo1 = 0 means half-i <= 0, i.e., i >= half.
                    # Then i+s = i + 0 - half = i - half which should be >= 0. Since i <= n-1 and half = n-1, i-half <= 0. Contradiction unless i = half = n-1.
                    # Hmm, let me just compute directly.
                
                # Let me redo this more carefully.
                # For idx in [0, m-1], shift s = idx - half.
                # Term 1: (1-h[i+s]) is valid when 0 <= i+s < n, i.e., -i <= s <= n-1-i
                #   => half-i <= idx <= half+n-1-i
                #   The h index is i+s = i + (idx-half) = idx + (i-half)
                #   Range: when idx = half-i: i+s = i + (half-i-half) = 0
                #          when idx = half+n-1-i: i+s = i + (half+n-1-i-half) = n-1
                #   So h indices range from 0 to n-1. ✓
                
                # Term 2: -h[i-s] is valid when 0 <= i-s < n, i.e., i-n+1 <= s <= i
                #   => half+i-(n-1) <= idx <= half+i
                #   The h index is i-s = i - (idx-half) = half + i - idx
                #   Range: when idx = half+i-(n-1): i-s = half+i-(half+i-(n-1)) = n-1
                #          when idx = half+i: i-s = half+i-(half+i) = 0
                #   So h indices range from n-1 down to 0. ✓

                    new_corr[lo1:hi1+1] += delta * (1.0 - h[h_idx])

                    new_corr[lo2:hi2+1] -= delta * h[h_idx]

                if new_max * dx < best_local_c5 - 1e-15:
                    best_local_c5 = new_max * dx
                    best_local_val = new_val
                    best_local_corr = new_corr
            
            if best_local_val != old_hi:
                h[i] = best_local_val
                corr = best_local_corr
                best_c5 = best_local_c5
                best_h = h.copy()
                any_improved = True
        
        if not any_improved:
            break
    
    return best_h, best_c5
```

One more idea: **Nesterov's smoothing**. The problem is min_h max_k J_k(h) s.t. constraints. We can smooth the max with a log-sum-exp or a quadratic penalty, and use a first-order method. The key is to use a backtracking line search to ensure convergence.

But SLSQP already does something similar. The issue is that SLSQP is slow for n=400 variables.

**Alternative: Use a projected gradient method with the smooth objective.**

The smooth objective is f(h) = (1/alpha) * log(sum_k exp(alpha * J_k(h))).
The gradient is: ∂f/∂h[i] = (1/alpha) * sum_k exp(alpha*J_k) * ∂J_k/∂h[i] / sum_k exp(alpha*J_k)
= sum_k p_k * ∂J_k/∂h[i] where p_k = exp(alpha*J_k) / sum_j exp(alpha*J_j)

This is a weighted average of the gradients of the individual J_k, with weights concentrated on the "active" (largest) J_k.

The projected gradient step is:
h_new = project(h - lr * grad, {0 ≤ h ≤ 1, sum(h) = target})

This is simple and efficient! Each iteration costs O(n) for the gradient and O(n) for the projection.

Let me implement this:

```python
def projected_gradient_smooth(h, target, alphas=[100, 500, 2000, 10000], 
                                max_iters_per_alpha=500, time_limit=120):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    h = project_feasible(h, target)
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    half = n - 1
    
    for alpha in alphas:
        # Backtracking line search
        lr = 1.0 / (alpha * dx)  # Initial learning rate (Lipschitz constant estimate)
        
        for it in range(max_iters_per_alpha):
            if time.time() - start > time_limit:
                return best_h, best_c5
            
            # Compute J_k for all k
            corr = np.correlate(h, 1.0 - h, mode='full')
            J = corr * dx  # This is J_k for each k
            
            # Soft-max weights
            J_shifted = alpha * (J - np.max(J))
            p = np.exp(J_shifted)
            p /= np.sum(p)
            
            # Gradient of the smooth objective
            # ∂f/∂h[i] = sum_k p_k * ∂J_k/∂h[i]
            # ∂J_k/∂h[i] = dx * [(1-h[i+k]) - h[i-k]] (with appropriate bounds)
            # Actually, let me compute this more carefully.
            # J_k = sum_j h[j] * (1-h[j+k]) * dx (for j, j+k in range)
            # ∂J_k/∂h[i] = dx * (1-h[i+k]) * 1_{i+k<n} + (-h[i-k]) * dx * 1_{i-k>=0}
            # Wait, the second term: J_k has term h[i-k] * (1-h[i]) when j = i-k
            # ∂/∂h[i] of h[i-k]*(1-h[i]) = -h[i-k]
            # So ∂J_k/∂h[i] = dx * [(1-h[i+k])*1_{i+k<n} - h[i-k]*1_{i-k>=0}]
            
            grad = np.zeros(n)
            # For each i, the gradient contribution is:
            # sum_k p_k * dx * [(1-h[i+k])*1_{i+k<n} - h[i-k]*1_{i-k>=0}]
            
            # This can be computed efficiently using correlations:
            # sum_k p_k * (1-h[i+k]) = sum_k p_k - sum_k p_k * h[i+k]
            # = sum(p) - (p circled with h)[i]... not quite, because of the boundary.
            
            # For the non-circular case, let me just compute it directly:
            # grad[i] = dx * [sum_k p_k * (1-h[i+k])*1_{i+k<n} - sum_k p_k * h[i-k]*1_{i-k>=0}]
            
            # The first term: sum_k p_k * (1-h[i+k]) for k such that i+k < n
            # = sum_{k=-(n-1)}^{n-1-i} p_k * (1-h[i+k])
            # Let m = i+k, then k = m-i, and m ranges from i-(n-1) to n-1, clipped to [0,n-1]
            # = sum_{m=max(0,i-n+1)}^{n-1} p_{m-i} * (1-h[m])
            # This is a convolution of p with (1-h), evaluated at position i.
            
            # Similarly, the second term: sum_k p_k * h[i-k] for k such that i-k >= 0
            # = sum_{k=-(n-1)}^{i} p_k * h[i-k]
            # Let m = i-k, then k = i-m, m ranges from 0 to i (and also i-(n-1) to i, clipped)
            # = sum_{m=max(0,i-n+1)}^{i} p_{i-m} * h[m]
            # This is a convolution of p (reversed) with h.
            
            # Using numpy: np.convolve or np.correlate
            # First term: correlate(p, 1-h, 'full') gives sum_k p[k] * (1-h[k+m-(n_p-1)])
            # Hmm, the indexing is getting complicated. Let me just use direct computation.
            
            # For n=400, direct O(n²) computation is 160K operations, which is fine.
            for i in range(n):
                acc = 0.0
                # Term 1: sum_k p_k * (1-h[i+k]) for valid k
                k_min = max(-(n-1), -i)  # k >= -i ensures i+k >= 0
                k_max = min(n-1, n-1-i)  # k <= n-1-i ensures i+k < n
                for k in range(int(k_min), int(k_max) + 1):
                    idx_p = k + half  # index into p (p has length 2n-1, indexed by k+half)
                    acc += p[idx_p] * (1.0 - h[i + k])
                # Term 2: -sum_k p_k * h[i-k] for valid k
                k_min2 = max(-(n-1), i - (n-1))  # i-k >= 0 means k <= i
                k_max2 = min(n-1, i)
                for k in range(int(k_min2), int(k_max2) + 1):
                    idx_p = k + half
                    acc -= p[idx_p] * h[i - k]
                grad[i] = acc * dx
            
            # Backtracking line search
            step = lr
            improved = False
            for _ in range(20):
                h_try = h - step * grad
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14 or step < 1e-8:
                    h = h_try
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h.copy()
                    improved = True
                    break
                step *= 0.5
            
            if not improved:
                break
        
        # Increase alpha (reduce smoothing)
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def coord_descent_fast(h, target, max_sweeps=500, time_limit=120):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    start = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    half = n - 1
    m = 2 * n - 1

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        any_improved = False
        order = np.arange(n)
        for i in order:
            old_hi = h[i]
            candidates = [0.0, 1.0]
            if 0.0 < old_hi < 1.0:
                candidates.append(0.5)
            best_local_c5 = best_c5
            best_local_val = old_hi
            best_local_corr = None
            for new_val in candidates:
                delta = new_val - old_hi
                if abs(delta) < 1e-12:
                    continue
                new_corr = corr.copy()
                lo1 = max(0, half - i)
                hi1 = min(m - 1, half + n - 1 - i)
                if hi1 >= lo1:
                    h_idx = np.arange(lo1, hi1 + 1) + (i - half)
                    new_corr[lo1:hi1 + 1] += delta * (1.0 - h[h_idx])
                lo2 = max(0, half + i - (n - 1))
                hi2 = min(m - 1, half + i)
                if hi2 >= lo2:
                    h_idx = (half + i) - np.arange(lo2, hi2 + 1)
                    new_corr[lo2:hi2 + 1] -= delta * h[h_idx]
                new_max = float(np.max(new_corr))
                if new_max * dx < best_local_c5 - 1e-15:
                    best_local_c5 = new_max * dx
                    best_local_val = new_val
                    best_local_corr = new_corr
            if best_local_corr is not None:
                h[i] = best_local_val
                corr = best_local_corr
                if best_local_c5 < best_c5 - 1e-15:
                    best_c5 = best_local_c5
                    best_h = h.copy()
                any_improved = True
        if not any_improved:
            break
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=300):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    start_t = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_t > time_limit:
            break
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def alternating_lp(h_init, n, target, max_iter, time_limit, n_active=40,
                   damping_init=0.85, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=80000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=80000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())
        if c5_val >= best_c5 - 1e-14:
            damping = max(0.5, damping * 0.95)
        else:
            damping = min(0.95, damping + 0.03)
        if it % 50 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.85)
    return best_h, best_c5


def projected_gradient_smooth(h, target, alphas, max_iters_per_alpha, time_limit):

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        if not improved and step > 100:
            break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
