            h_sa3, sc_sa3 = sa_binary(h_g3, target_init, max_it6, 0.006, 0.00001, rng, allowed6, start_time, top_k=5)
            if sc_sa3 < best_score:
                best_score = sc_sa3
                best_h = h_sa3.copy()

    # Phase 4: ILS with diverse restarts
    if tleft() > 100:
        for pert_iter in range(8):
            if tleft() < 50:
                break
            n_pert = 3 + pert_iter * 2
            h_pert = perturb_binary(best_h, n_pert, rng)
            allowed7 = min(25, tleft() * 0.04)
            h_pg, sc_pg = greedy_worst_shift(h_pert, 10000, allowed7, start_time, scan_width=20)
            if tleft() < 40:
                break
            allowed8 = min(40, tleft() * 0.05)
            max_it8 = int(min(100000, allowed8 * 1200))
            h_ps, sc_ps = sa_binary(h_pg, target_init, max_it8, 0.008, 0.00002, rng, allowed8, start_time, top_k=5)
            if sc_ps < best_score:
                best_score = sc_ps
                best_h = h_ps.copy()

    # Phase 5: Try different n values
    n_options = [n_init + 2, n_init + 4, n_init + 6, n_init + 8, n_init - 2, n_init + 10, n_init + 12]
    n_options = [x for x in n_options if x >= 200 and x % 2 == 0 and x != n_init]
    for n_try in n_options:
        if tleft() < 60:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_t = project_binary(h_interp, target_try)
        allowed9 = min(25, tleft() * 0.05)
        h_gt, sc_gt = greedy_worst_shift(h_bin_t, 15000, allowed9, start_time, scan_width=20)
        if sc_gt < best_score:
            best_score = sc_gt
            best_h = h_gt.copy()
        if tleft() < 40:
            continue
        allowed10 = min(40, tleft() * 0.05)
        max_it10 = int(min(80000, allowed10 * 1200))
        h_st, sc_st = sa_binary(h_gt, target_try, max_it10, 0.008, 0.00002, rng, allowed10, start_time, top_k=5)
        if sc_st < best_score:
            best_score = sc_st
            best_h = h_st.copy()

    # Phase 6: Final continuous polish on best solution
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        betas_final = [3000, 8000, 20000, 50000]
        allowed11 = max(10.0, min(120, tleft() - 10))
        h_ref, sc_ref = continuous_optimize(best_h, target_use, betas_final, allowed11, start_time, maxiter_per_beta=1000)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref.copy()

    # Phase 7: Final binary SA polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed12 = max(5.0, min(80, tleft() - 10))
        max_it12 = int(min(300000, allowed12 * 1200))
        h_sf, sc_sf = sa_binary(h_bin_f, target_use, max_it12, 0.004, 0.000005, rng, allowed12, start_time, top_k=5)
        if sc_sf < best_score:
            best_score = sc_sf
            best_h = h_sf.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, final_score, n_final
