def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, alpha):

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.sum(exp_shifted)
    if log_sum < 1e-300:
        obj = c_max
        weights = np.zeros(2 * n - 1)
        weights[np.argmax(overlaps)] = 1.0
    else:
        obj = c_max + np.log(log_sum) / alpha
        weights = exp_shifted / log_sum

    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - n_half
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w * dx * (1.0 - h[k:n])
            grad[k:] -= w * dx * h[:n - k]
        else:
            m = -k
            grad[m:] += w * dx * (1.0 - h[:n - m])
            grad[:n - m] -= w * dx * h[m:]

    return obj, grad

            val, _ = lse_obj_and_grad(x, alpha)
            c = np.sum(x) - target
            return val + lam * c + 0.5 * mu * c * c

        def grad_fn(x):
            _, g = lse_obj_and_grad(x, alpha)
            c = np.sum(x) - target
            g = g + (lam + mu * c) * np.ones(n)
            return g

            h = h_new

    bounds = [(0.0, 1.0)] * n

    def obj_fn(x):
        val, _ = lse_obj_and_grad(x, alpha)
        return val

        _, g = lse_obj_and_grad(x, alpha)
        return g

        return np.ones(n)

                       options={'maxiter': maxiter, 'ftol': 1e-15})

    except Exception:

def subgradient_step(h, n, target, n_top=30, lr=0.3):
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:n_top]
    top_vals = overlaps[top_idx]
    c_max = top_vals[0]
    w = np.exp((top_vals - c_max) / 0.05)
    w = w / w.sum()

    grad = np.zeros(n)
    for i, k_idx in enumerate(top_idx):
        k = k_idx - n_half
        if k == 0:
            grad += w[i] * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n - k] += w[i] * dx * (1.0 - h[k:n])
            grad[k:] -= w[i] * dx * h[:n - k]
        else:
            m = -k
            grad[m:] += w[i] * dx * (1.0 - h[:n - m])
            grad[:n - m] -= w[i] * dx * h[m:]

    h_new = h - lr * grad
    h_new = project_feasible(h_new, target)
    return h_new

        h_shake = project_feasible(h_shake, target)
        h_shake = slsqp_smooth(h_shake, n, target, 10000, maxiter=200)
        h_shake = project_feasible(h_shake, target)
        sc = compute_c5(h_shake)
        if sc < best_score:
            best_score = sc
            best_h = h_shake

    # Phase 8: Final L-BFGS at very high alpha
    if tleft() > 30:
        h_new = augmented_lbfgsb(best_h, n, target, 50000, time_budget=min(40, tleft() * 0.6))
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Final SLSQP
    if tleft() > 25:
        h_new = slsqp_smooth(best_h, n, target, 100000, maxiter=200)
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Final local search
    if tleft() > 15:
        h_ls, c5_ls = local_pair_search(best_h, n, target,
                                        max_rounds=100, rng=rng,
                                        time_limit=tleft() - 3, start_time=start_time)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
