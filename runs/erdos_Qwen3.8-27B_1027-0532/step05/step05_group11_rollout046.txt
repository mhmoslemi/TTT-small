import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_adaptive_lags(h, px=0.90):
    """Get lags where overlap is above px fraction of max"""
    overlaps = compute_overlaps(h)
    n_half = len(h) - 1
    max_overlap = np.max(overlaps)
    threshold = px * max_overlap
    above = np.where(overlaps >= threshold)[0]
    if len(above) > 200:
        # Take top 200
        top_idx = np.argsort(overlaps)[::-1][:200]
        return top_idx - n_half
    return above - n_half

def build_lp(h, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints

def lp_solve(h_init, n, target, max_iter, time_limit, px=0.90, lags_cap=200, damping=0.9, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        lags = get_adaptive_lags(h, px)
        if len(lags) > lags_cap:
            lags = np.argsort(compute_overlaps(h))[::-1][:lags_cap]
            lags = lags - (n - 1)
        
        h_var, t_var, constraints = build_lp(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

    return above - n_half

        y = project_feasible(y, target)

        val, grad = smooth_value_and_grad(y, alpha)
        h_new = y - lr * grad
        h_new = project_feasible(h_new, target)

        if it % 200 == 0:
            val_new, _ = smooth_value_and_grad(h_new, alpha)
            if val_new > val + 1e-8:
                lr *= 0.5
            elif lr < 0.05:
                lr *= 1.1

        h = h_new
        t_k = t_new
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5


def slsqp_phase(h_init, n, target, alphas, maxiter=2000):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

    def obj_func(h_arr, a):
        corr = np.correlate(h_arr, 1.0 - h_arr, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
        return float((c_max + np.log(np.sum(np.exp(shifted))) / a)) * (2.0 / n)

    def cons_fun(h_arr):
        return np.sum(h_arr) - target

    def cons_jac(h_arr):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_init.copy()
    for a in alphas:
        try:
            res = minimize(obj_func, h_cur, args=(a,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def subgradient_momentum(h_init, n, target, time_limit, n_top=30, momentum=0.85):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)

    for step in range(80000):
        if time.time() - start > time_limit:
            break
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        overlaps = compute_overlaps(y)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        w = 0.8 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            if k >= 0:
                j1 = idxs[idxs + k < n]
                grad[j1] += w[i] * (1.0 - y[j1 + k])
                j2 = idxs[idxs >= k]
                grad[j2] += w[i] * (-y[j2 - k])
            else:
                kp = -k
                j1 = idxs[idxs >= kp]
                grad[j1] += w[i] * (1.0 - y[j1 - kp])
                j2 = idxs[idxs + kp < n]
                grad[j2] += w[i] * (-y[j2 + kp])
        grad *= dx

        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.005, polyak_lr), 0.3)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: LP with adaptive lags (primary)
    if tleft() > 250:
        h1, c5_1 = lp_adaptive(best_h, n, target, time_limit=min(180, tleft() * 0.45),
                               n_lags_start=280, n_lags_end=180,
                               reg_start=1e-3, reg_end=1e-8, damping=0.90, frac=0.82)
        if c5_1 < best_score:
            best_score = c5_1
            best_h = h1

    # Phase 2: Nesterov on smooth objective
    if tleft() > 150:
        h2, c5_2 = nesterov_phase(best_h, n, target, time_limit=min(100, tleft() * 0.35), alpha=300.0)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2

    # Phase 3: All-lags LP
    if tleft() > 200:
        h3, c5_3 = lp_all_lags(best_h, n, target, time_limit=min(150, tleft() * 0.4),
                               reg_start=1e-4, reg_end=1e-10, damping=0.93)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3

    # Phase 4: SLSQP with high alpha
    if tleft() > 150:
        h4 = slsqp_phase(best_h, n, target, [300, 2000, 10000, 50000, 200000], maxiter=3000)
        sc4 = compute_c5(h4)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Subgradient momentum
    if tleft() > 100:
        h5, c5_5 = subgradient_momentum(best_h, n, target, time_limit=min(80, tleft() * 0.35),
                                        n_top=35, momentum=0.88)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5

    # Phase 6: Second all-lags LP
    if tleft() > 120:
        h6, c5_6 = lp_all_lags(best_h, n, target, time_limit=min(90, tleft() * 0.4),
                               reg_start=1e-5, reg_end=1e-11, damping=0.95)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6

    # Phase 7: Nesterov with higher alpha
    if tleft() > 80:
        h7, c5_7 = nesterov_phase(best_h, n, target, time_limit=min(60, tleft() * 0.5), alpha=1000.0)
        if c5_7 < best_score:
            best_score = c5_7
            best_h = h7

    # Phase 8: Adaptive LP again
    if tleft() > 80:
        h8, c5_8 = lp_adaptive(best_h, n, target, time_limit=min(60, tleft() * 0.5),
                               n_lags_start=350, n_lags_end=250,
                               reg_start=1e-5, reg_end=1e-10, damping=0.95, frac=0.92)
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8

    # Phase 9: Final SLSQP polish
    if tleft() > 50:
        h9 = slsqp_phase(best_h, n, target, [5000, 20000, 100000, 300000], maxiter=2500)
        sc9 = compute_c5(h9)
        if sc9 < best_score:
            best_score = sc9
            best_h = h9

    # Phase 10: Final LP polish
    if tleft() > 40:
        h10, c5_10 = lp_adaptive(best_h, n, target, time_limit=min(30, tleft() * 0.6),
                                 n_lags_start=200, n_lags_end=150,
                                 reg_start=1e-6, reg_end=1e-11, damping=0.97, frac=0.95)
        if c5_10 < best_score:
            best_score = c5_10
            best_h = h10

    # Final feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
