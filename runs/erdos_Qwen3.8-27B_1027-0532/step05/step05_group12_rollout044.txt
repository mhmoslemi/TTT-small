            break

        if ptype == 'gauss':
            h_pert = best_h + rng.randn(n) * amp
        elif ptype == 'structured':
            freq = rng.randint(2, 8)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2 * np.pi * freq, n)
            h_pert = best_h + amp * np.sin(x + phase)
        else:
            mask = rng.rand(n) < 0.3
            h_pert = best_h.copy()
            h_pert[mask] += rng.choice([-1, 1], size=np.sum(mask)) * amp

        h_pert = project_feasible(h_pert, target)

        h_cd, score_cd = coordinate_descent(h_pert, n, target,
                                            max_cycles=50000,
                                            time_limit=budget_s,
                                            start_time=start_time,
                                            deadline_offset=budget_s - 80,
                                            rho=0.5, seed_offset=200 + pi * 7)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd

        # Quick SLSQP on perturbed result
        if tleft() > 60:
            h_slq = slsqp_optimize(h_cd, n, target,
                                   [1000, 5000, 20000],
                                   maxiter=500, ftol=1e-15,
                                   time_limit=budget_s, start_time=start_time,
                                   deadline_offset=40)
            sc = compute_c5(h_slq)
            if sc < best_score:
                best_score = sc
                best_h = h_slq

    # Phase 5: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800, 1000]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)

            h_cd, score_cd = coordinate_descent(h_interp, n_try, target_try,
                                                max_cycles=50000,
                                                time_limit=budget_s,
                                                start_time=start_time,
                                                deadline_offset=80,
                                                rho=0.5, seed_offset=500 + n_try)
            if tleft() > 60:
                h_slq = slsqp_optimize(h_cd, n_try, target_try,
                                       [500, 2000, 10000],
                                       maxiter=500, ftol=1e-15,
                                       time_limit=budget_s, start_time=start_time,
                                       deadline_offset=40)
                sc = compute_c5(h_slq)
                if sc < score_cd:
                    h_cd, score_cd = h_slq, sc

            if score_cd < best_score:
                best_score = score_cd
                best_h = h_cd
                n = n_try
                target = target_try

    # Phase 6: Final coordinate descent + SLSQP
    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 60:
        h_cd, score_cd = coordinate_descent(best_h, n_use, target_use,
                                            max_cycles=50000,
                                            time_limit=budget_s,
                                            start_time=start_time,
                                            deadline_offset=30,
                                            rho=0.5, seed_offset=999)
        if score_cd < best_score:
            best_score = score_cd
            best_h = h_cd

    if tleft() > 30:
        h_slq = slsqp_optimize(best_h, n_use, target_use,
                               [2000, 10000, 50000],
                               maxiter=1000, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=10)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Final feasibility check
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    # Also check the pre-projection version (projection might slightly increase C5)
    pre_score = compute_c5(best_h)
    if pre_score < final_score:
        # Try to find a feasible point near best_h with lower score
        h_alt = project_feasible(best_h, n_final / 2.0)
        final_score = compute_c5(h_alt)
        h_final = h_alt

    return h_final, final_score, len(h_final)
