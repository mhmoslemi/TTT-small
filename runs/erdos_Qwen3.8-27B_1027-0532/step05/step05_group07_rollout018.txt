import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def solve_1d_minimax(A, C, t_min, t_max, n_bisect=60):
    lo = 0.0
    hi = float(np.max(A + C * t_max))
    if not np.any(C > 0):
        hi = float(np.max(A + C * t_max))
    else:
        candidates = [float(np.max(A))]
        pos = C > 1e-14
        if np.any(pos):
            candidates.append(float(np.max(A[pos] + C[pos] * t_max)))
        hi = max(candidates)
    for _ in range(n_bisect):
        mid = (lo + hi) * 0.5
        pos = C > 1e-14
        neg = C < -1e-14
        if np.any(~(pos | neg)) and float(np.max(A[~(pos | neg)])) > mid + 1e-14:
            lo = mid
            continue
        ub = t_max
        if np.any(pos):
            ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
        lb = t_min
        if np.any(neg):
            lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    v = (lo + hi) * 0.5
    pos = C > 1e-14
    neg = C < -1e-14
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
    lb = t_min
    if np.any(neg):
        lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
    ub = min(ub, t_max)
    lb = max(lb, t_min)
    if lb > ub:
        t_star = (t_min + t_max) * 0.5
    else:
        t_star = (lb + ub) * 0.5
    return t_star


def paired_cd(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=30):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0

    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            t_star = solve_1d_minimax(A, C, t_min, t_max, 55)

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:

            corr = compute_corr(h_cur)

    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    if _g_alpha < 1e-6:
        return c_max
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=5000, ftol=1e-18):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def triple_cd_step(h, i, j, k, n, corr):
    s = h[i] + h[j] + h[k]
    if s < 2e-14 or s > 3.0 - 2e-14:
        return h, corr
    Bi = compute_B(h, i, n)
    Bj = compute_B(h, j, n)
    Bk = compute_B(h, k, n)
    C1 = Bi - Bk
    C2 = Bj - Bk
    A0 = corr - h[i] * Bi - h[j] * Bj - h[k] * Bk + Bk * s

    best_t = (h[i], h[j])
    best_val = float(np.max(corr))
    n_grid = 8
    for a in range(n_grid + 1):
        for b in range(n_grid + 1 - a):
            u = a / n_grid
            v = b / n_grid
            nu = 1.0 - u - v
            if nu < -1e-12:
                continue
            nu = max(0.0, nu)
            val = float(np.max(A0 + C1 * (nu * s - h[i]) + C2 * (u * s - h[j])))
            if val < best_val:
                best_val = val
                best_t = (nu * s, u * s)
    if best_t[0] != h[i] or best_t[1] != h[j]:
        h[i] = best_t[0]
        h[j] = best_t[1]
        h[k] = s - best_t[0] - best_t[1]
        h = np.clip(h, 0.0, 1.0)
        corr = compute_corr(h)
    return h, corr
