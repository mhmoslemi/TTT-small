import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower - 1e-12 <= tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)


def obj_grad_active(h, alpha, active_lags):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_active = corr[active_lags]
    c_max = float(np.max(c_active))
    shifted = alpha * (c_active - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    n_act = len(active_lags)
    w_full = np.zeros(n)
    for i in range(n_act):
        w_full[int(active_lags[i])] = weights[i]
    w_flipped = w_full[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(w_full, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad


def find_active_lags(h, M, random_lag_prob=0.0):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    active = np.argsort(corr)[::-1][:M]
    if random_lag_prob > 0 and M < n - 1:
        rng_local = np.random.RandomState(int(corr.max() * 10000) % 100000)
        if rng_local.rand() < random_lag_prob:
            k_rand = rng_local.randint(0, n - 1)
            active = np.append(active, k_rand)
    return active


def fista_active(h0, target, alpha, active_lags, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_k = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.05, 1.0)
    step = lr_init
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = x + ((t_k - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = obj_grad_active(y_clipped, alpha, active_lags)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = obj_grad_active(x_new, alpha, active_lags)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            if step < 1e-15:

            x_new = project_feasible(y_clipped - step * grad_y, target)

def slsqp_step(h0, target, alpha, active_lags, maxiter=50):
    n = len(h0)
    x0 = project_feasible(h0, target).copy()

    def obj_func(x):
        return make_slsqp_obj(x, alpha, active_lags)

    def grad_func(x):
        return make_slsqp_grad(x, alpha, active_lags)

    def eq_fun(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_func, x0, method='SLSQP', jac=grad_func,
                          constraints=cons, bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-10})
        h_out = project_feasible(result.x, target)
    except Exception:
        h_out = x0
    return h_out

            h_new, sc_new = fista_active(best_h, target, 10000, active, 30000,
                                          time.time() + 20)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
    return best_h, best_score


def final_polish(h0, target, deadline):
    n = len(h0)
    h_cur = h0.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for alpha in [20000, 50000, 100000]:
        if time.time() > deadline:
            break
        active = find_active_lags(h_cur, 3)
        h_new, sc_new = fista_active(h_cur, target, alpha, active, 50000,
                                      time.time() + 25)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new.copy()
        active5 = find_active_lags(h_cur, 5)
        h_new, sc_new = fista_active(h_cur, target, alpha // 2, active5, 30000,
                                      time.time() + 20)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 20.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Active-set continuation (main optimization)
    dd1 = start_time + 450
    h1, sc1 = active_set_continuation(best_h, target, dd1)
    if sc1 < best_score:
        best_score = sc1
        best_h = h1

    # Phase 2: Coarse-to-fine refinement
    if time.time() < deadline_final - 250:
        dd2 = time.time() + 150
        h2, sc2 = coarse_fine_refine(best_h, target, dd2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h2

    # Phase 3: Multi-start perturbation
    if time.time() < deadline_final - 150:
        dd3 = time.time() + 120
        h3, sc3 = multi_start_optimize(best_h, target, dd3, n_starts=5, seed=seed)
        if sc3 < best_score:
            best_score = sc3
            best_h = h3

    # Phase 4: Structured restarts (different periodic structures)
    if time.time() < deadline_final - 120:
        dd4 = time.time() + 100
        h4, sc4 = structured_restart(best_h, target, dd4, seed=seed + 1)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4
        # Quick follow-up optimization on best structured result
        if time.time() < deadline_final - 60:
            active = find_active_lags(best_h, 5)
            h4b, sc4b = fista_active(best_h, target, 10000, active, 30000,
                                      time.time() + 50)
            if sc4b < best_score:
                best_score = sc4b
                best_h = h4b

    # Phase 5: Another active-set continuation pass on the best
    if time.time() < deadline_final - 100:
        dd5 = time.time() + 80
        h5, sc5 = active_set_continuation(best_h, target, dd5)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5

    # Phase 6: SLSQP refinement
    if time.time() < deadline_final - 40:
        active = find_active_lags(best_h, 5)
        h6 = slsqp_step(best_h, target, 10000, active, maxiter=50)
        sc6 = compute_c5(h6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6
        active3 = find_active_lags(best_h, 3)
        h6b = slsqp_step(best_h, target, 50000, active3, maxiter=50)
        sc6b = compute_c5(h6b)
        if sc6b < best_score:
            best_score = sc6b
            best_h = h6b

    # Phase 7: Final high-alpha polish
    if time.time() < deadline_final - 20:
        dd7 = deadline_final
        h7, sc7 = final_polish(best_h, target, dd7)
        if sc7 < best_score:
            best_score = sc7
            best_h = h7

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
