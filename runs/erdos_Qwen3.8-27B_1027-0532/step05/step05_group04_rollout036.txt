import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_penalty = 0.0
_g_adam_top_k = 5
_g_adam_focus = 200.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_lag(h, k, dx):
    n = len(h)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        j_pos = np.arange(0, n - k)
        grad[j_pos] += (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= h[j_neg - k]
    return grad * dx


def softmax_obj_and_grad(h, alpha, target, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n - 1:n - 1 + n]
    max_over = float(np.max(over))
    logits = alpha * (over - max_over)
    logits = np.clip(logits, -700.0, 0.0)
    exp_logits = np.exp(logits)
    sum_exp = float(exp_logits.sum())
    weights = exp_logits / sum_exp
    obj = max_over + float(np.log(sum_exp)) / alpha
    integral_diff = float(np.sum(h) - target)
    obj += penalty * integral_diff * integral_diff
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk > 1e-10:
            grad += wk * grad_lag(h, k, dx)
    grad += 2.0 * penalty * integral_diff
    return obj, grad


def _lbfgs_obj(x):
    o, g = softmax_obj_and_grad(x, _g_alpha, _g_target, _g_penalty)
    return o


def _lbfgs_jac(x):
    o, g = softmax_obj_and_grad(x, _g_alpha, _g_target, _g_penalty)
    return g


def lbfgs_round(h, n, target, alpha, maxiter, deadline, penalty=50000.0):
    global _g_alpha, _g_target, _g_penalty
    _g_alpha = alpha
    _g_target = target
    _g_penalty = penalty
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbfgs_obj, h, method='L-BFGS-B', jac=_lbfgs_jac,
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxfun': maxiter * 5})
        return project_feasible(res.x, target)
    except Exception:
        return h


def adam_optimize(h, n, target, n_steps, lr, top_k, deadline, focus_beta=200.0):
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()

        top_idx = np.argsort(over)[::-1][:top_k]
        over_top = over[top_idx]
        max_top = over_top[0]
        spread = max(max_top - over_top[-1], 1e-12)
        logits = focus_beta * (over_top - max_top) / spread
        logits = np.clip(logits, -100.0, 0.0)
        w = np.exp(logits)
        w /= w.sum()

        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            if wi > 1e-10:
                grad += wi * grad_lag(h_cur, int(ki), dx)

        integral_diff = float(np.sum(h_cur) - target)
        grad += 50.0 * integral_diff

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h_cur - update
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < c_curr:
            h_cur = h_new
        else:
            for frac in [0.5, 0.25, 0.1]:
                h_try = h_cur - frac * update
                h_try = project_feasible(h_try, target)
                sc_try = compute_c5(h_try)
                if sc_try < c_curr:
                    h_cur = h_try
                    break

    return best_h, best_score


def surgical_top1(h, n, target, n_steps, deadline):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    lr = 0.003

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        c_curr = over[k_star] * dx

        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()

        grad = grad_lag(h_cur, k_star, dx)
        top3 = np.argsort(over)[::-1][:3]
        for ki in top3[1:]:
            grad += 0.2 * grad_lag(h_cur, int(ki), dx)

        integral_diff = float(np.sum(h_cur) - target)
        grad += 50.0 * integral_diff

        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if sc < c_curr * 0.9995:
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.85

    return best_h, best_score


def structured_init(n, target, rng, init_type):
    if init_type == 0:
        p = rng.randint(max(n // 10, 2), n // 3)
        fill = max(p // 2, 1)
        h = np.zeros(n)
        for block in range(n // p + 1):
            start = block * p
            end = min(start + fill, n)
            h[start:end] = 1.0
        return project_feasible(h, target)
    elif init_type == 1:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        h = np.zeros(n)
        for i in range(n):
            if (i * phi) % 1.0 < 0.5:
                h[i] = 1.0
        return project_feasible(h, target)
    elif init_type == 2:
        freq = rng.randint(3, 12)
        h = 0.5 + 0.4 * np.sin(2.0 * np.pi * freq * np.arange(n) / n + rng.rand() * 6.28)
        return project_feasible(h, target)
    elif init_type == 3:
        h = rng.rand(n)
        return project_feasible(h, target)
    elif init_type == 4:
        p1 = rng.randint(max(n // 20, 2), n // 6)
        p2 = rng.randint(max(n // 4, p1 + 1), n // 2)
        h = np.zeros(n)
        for i in range(n):
            if (i % p1) < p1 // 3 and (i % p2) < p2 // 2:
                h[i] = 1.0
            elif (i % p1) >= 2 * p1 // 3 and (i % p2) >= p2 // 2:
                h[i] = 1.0
        return project_feasible(h, target)
    else:
        h = 0.5 + 0.3 * np.cos(2.0 * np.pi * np.arange(n) / n * 7.0)
        h += 0.2 * np.sin(2.0 * np.pi * np.arange(n) / n * 13.0 + 1.0)
        return project_feasible(h, target)


def multi_adam_hillclimb(h, n, target, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    phases = [
        (5, 300.0, 0.003, 1500),
        (3, 500.0, 0.002, 1500),
        (8, 200.0, 0.002, 1000),
        (3, 800.0, 0.001, 1000),
        (5, 400.0, 0.0008, 1500),
        (2, 1000.0, 0.0005, 1000),
    ]
    for top_k, focus, lr, steps in phases:
        if time.time() > deadline:
            break
        h_new, sc = adam_optimize(h_cur, n, target, n_steps=steps, lr=lr,
                                  top_k=top_k, deadline=deadline, focus_beta=focus)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_cur = h_new
    return best_h, best_score
