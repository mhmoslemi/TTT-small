```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_smooth_max(h, alpha, m_top=20):
    """Compute gradient of smooth max of overlaps using top-m lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    half_n = n - 1
    
    # Use top-m lags for efficiency
    top_idx = np.argsort(corr)[-m_top:]
    
    # Softmax weights over top-m
    shifted = alpha * (corr[top_idx] - corr[top_idx].max())
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    idxs = np.arange(n)
    
    for j, k_idx in enumerate(top_idx):
        pk = w[j]
        if pk < 1e-12:
            continue
        k = k_idx - half_n
        if k > 0:
            # ∂f_k/∂h[i] = (1-h[i+k]) for i < n-k, minus h[i-k] for i >= k
            valid1 = idxs < (n - k)
            grad[valid1] += pk * (1.0 - h[valid1 + k]) * dx
            valid2 = idxs >= k
            grad[valid2] += pk * (-h[valid2 - k]) * dx
        elif k < 0:
            kk = -k
            valid1 = idxs >= kk
            grad[valid1] += pk * (1.0 - h[valid1 - kk]) * dx
            valid2 = idxs < (n - kk)
            grad[valid2] += pk * (-h[valid2 + kk]) * dx
        else:
            # k=0: f_0 = sum h[i]*(1-h[i]), grad = 1-2*h[i]
            grad += pk * (1.0 - 2.0 * h) * dx
    
    return grad

def smooth_max_descent(h, n, target, time_limit, alphas=None):
    if alphas is None:
        alphas = [50, 100, 200, 500, 1000, 2000, 5000]
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        for it in range(5000):
            if time.time() - start > time_limit:
                break
            grad = compute_grad_smooth_max(h_cur, alpha, m_top=30)
            improved = False
            for lr in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                h_try = h_cur - lr * grad
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-15:
                    best_c5 = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    improved = True
                    break
            if not improved:
                break
    
    return best_h, best_c5

def slsqp_optimize(h, n, target, alpha, maxiter):
    """SLSQP with smooth max."""
    dx = 2.0 / n
    global_target = target
    global_n = n
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    def constraint_fun(x):
        return np.sum(x) - target
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new
    except:
        return h.copy()
```

For the gradient descent approach, we can use the TRUE max (not smoothed) and the subgradient, which is unbiased. The issue is that the subgradient is only defined at the max point, so we need to be at the max.

Here's a better approach: **Alternating maximization with exact max tracking**.

1. Compute the exact max and its location (argmax lag)
2. Compute the subgradient at that point
3. Take a small step in the -subgradient direction
4. Project
5. Repeat

The subgradient at h for the max of f_k is:
∇f_{k*}(h) where k* = argmax f_k

This is a single vector (not a weighted sum), so it's exact.

The issue is that if multiple lags tie for the max, the subgradient is in the convex hull of their gradients. Using just one of them might not decrease the max.

A better approach: use the **Euclidean norm of the top-m gradients** as a descent direction:
d = -∑_{k∈top_m} p_k * ∇f_k(h) where p_k is proportional to f_k(h) (or uniform)

This ensures we're decreasing all top-m overlaps simultaneously.

```python
def fine_descent(h, target, time_limit, alpha=1000, n_top=10):
    """Very fine descent using top-m smoothed gradient."""
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    half_n = n - 1
    idxs = np.arange(n)
    
    while time.time() - start < time_limit:
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        M = np.max(corr)
        
        # Get top n_top lags
        top_idx = np.argsort(corr)[-n_top:]
        
        # Softmax weights
        shifted = alpha * (corr[top_idx] - M)
        shifted = np.clip(shifted, -300, 0)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for j, k_idx in enumerate(top_idx):
            pk = w[j]
            k = k_idx - half_n
            if abs(pk) < 1e-10:
                continue
            if k > 0:
                v1 = idxs < (n - k)
                grad[v1] += pk * (1.0 - h_cur[v1 + k]) * dx
                v2 = idxs >= k
                grad[v2] += pk * (-h_cur[v2 - k]) * dx
            elif k < 0:
                kk = -k
                v1 = idxs >= kk
                grad[v1] += pk * (1.0 - h_cur[v1 - kk]) * dx
                v2 = idxs < (n - kk)
                grad[v2] += pk * (-h_cur[v2 + kk]) * dx
            else:
                grad += pk * (1.0 - 2.0 * h_cur) * dx
        
        # Line search with very small steps
        improved = False
        for lr in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-16:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break
        
        if not improved:
            # Try even smaller steps
            for lr in [0.0001, 0.0002, 0.0003, 0.0005]:
                h_try = h_cur - lr * grad
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-16:
                    best_c5 = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    improved = True
                    break
        
        if not improved:
            # Increase alpha for sharper focus
            alpha = min(alpha * 1.5, 100000)
    
    return best_h, best_c5
```

For continuous h, a "pair move" is: pick (i,j), change h[i] → h[i]+δ, h[j] → h[j]-δ (maintaining sum). Choose δ to minimize C₅.

For a given pair (i,j) and a given δ:
- The change in f_k is: δ*(1-h[i+k]) - δ*h[i-k] - δ*(1-h[j+k]) + δ*h[j-k] (with appropriate sign)

- From h[i] term: +δ*(1-h[i+k]) - δ*h[i-k] (approximately, for the terms involving h[i])

Actually, this is just the directional derivative in the direction e_i - e_j:
D_{e_i - e_j} f_k = (1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]

So for a small step δ:
f_k(h + δ(e_i - e_j)) ≈ f_k(h) + δ * [(1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k]]

We want to choose δ > 0 (or < 0) to minimize max_k f_k.

This is a 1D optimization: minimize max_k (f_k + δ * g_k(i,j)) where g_k(i,j) = (1-h[i+k]) - h[i-k] - (1-h[j+k]) + h[j-k].

The optimal δ is where the max is minimized, which is at the "crossover" point where the best and worst constrained lags cross.

This is exact (no approximation) and can compute the exact improvement for each pair.

For n=400, there are C(400,2) ≈ 80,000 pairs. For each pair, computing g_k for all k is O(n) = 400. Total: 32 million operations. Feasible in a few seconds!
