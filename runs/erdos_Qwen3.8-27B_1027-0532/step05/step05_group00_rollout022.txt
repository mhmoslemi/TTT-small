    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    exp_corr = np.exp(alpha * (corr - corr.max()))
    p = exp_corr / exp_corr.sum()
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h

        # term1: sum_j p[j-i+n-1] * (1-h[j]) for valid j
        # j-i+n-1 must be in [0, 2n-2], i.e., i-n+1 <= j <= i+n-1
        # But j is in [0, n-1], so j is in [max(0, i-n+1), min(n-1, i+n-1)] = [0, n-1] (as shown earlier)

        term2 = np.dot(p[idxs2], h)

    return grad

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def compute_grad(h, alpha, top_m=200):
    """Compute gradient of log-sum-exp smooth objective."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top-M shifts for efficiency
    if top_m < 2 * n - 1:
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_idx]
        exp_top = np.exp(alpha * (top_vals - top_vals.max()))
        p_full = np.zeros(2 * n - 1)
        p_full[top_idx] = exp_top / exp_top.sum()
    else:
        exp_corr = np.exp(alpha * (corr - corr.max()))
        p_full = exp_corr / exp_corr.sum()
    
    grad = np.zeros(n)
    # For efficiency, only process shifts with non-negligible weight
    active = p_full > 1e-10
    active_idx = np.where(active)[0]
    
    for idx in active_idx:
        k = int(idx - (n - 1))
        pm = p_full[idx]
        if k > 0:
            grad[0:n-k] += pm * (1.0 - h[k:n])
            grad[k:n] -= pm * h[0:n-k]
        elif k < 0:
            kp = -k
            grad[kp:n] += pm * (1.0 - h[0:n-kp])
            grad[0:n-kp] -= pm * h[kp:n]
        else:
            grad += pm * (1.0 - 2.0 * h)
    
    return grad
```
