from scipy.optimize import minimize, minimize_scalar
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)[n-1:])) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        delta = diff / cnt
        h[mask] = np.clip(h[mask] + delta, 0.0, 1.0)
    return h

def get_overlapping_corr(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:n-1+n]

def compute_logsumexp_obj(h, alpha):
    over = get_overlapping_corr(h)
    m = np.max(over)
    return m + np.log(np.sum(np.exp(alpha * (over - m)))) / alpha

def compute_logsumexp_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    over = get_overlapping_corr(h)
    m = np.max(over)
    weights = np.exp(alpha * (over - m))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-10:
            continue
        # ∂corr_k/∂h[j] = (1-h[j+k])*1{j<=n-1-k} - h[j-k]*1{j>=k}
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    grad *= dx
    return grad

def slsqp_smooth(h, n, target, alpha, maxiter, ftol, deadline):
    def obj(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        return compute_logsumexp_obj(x_clipped, alpha)
    
    def jac(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        return compute_logsumexp_grad(x_clipped, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    if time.time() > deadline:
        return h
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

    m = np.max(over)
    lse = m + np.log(np.sum(np.exp(alpha * (over - m)))) / alpha
    var = np.var(over)
    return lse + var_weight * var

def variance_aware_grad(h, alpha, var_weight):
    n = len(h)

    m = np.max(over)

        j_neg = np.arange(k, n)
        var_grad[j_neg] -= coef * h[j_neg - k]
    var_grad *= dx

    cur_sc = best_sc
    stuck = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t += 1
        over = get_over(h_cur)
        idx = np.argsort(over)[::-1][:top_k]
        w = np.zeros(n)
        w[idx] = 1.0 / top_k
        grad = np.zeros(n)
        for k in idx:
            k = int(k)
            grad[:n-k] += w[k] * (1.0 - h_cur[k:])
            grad[k:] -= w[k] * h_cur[:n-k]
        grad *= dx
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        m_hat = m_v / (1.0 - b1**t)
        v_hat = v_v / (1.0 - b2**t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_cur - step_vec, target)
        sc = compute_c5(h_new)
        if sc < cur_sc - 1e-15:
            h_cur = h_new
            cur_sc = sc
            stuck = 0
            if sc < best_sc:
                best_sc = sc
                best_h = h_cur.copy()
        else:
            stuck += 1
            if stuck > 100:
                lr *= 0.9
                if lr < 1e-8:
                    break
                stuck = 0
    return best_h, best_sc


def perturb_reopt(h, n, target, alpha, n_perturb, pert_scale, deadline):
    best_h = h.copy()
    best_sc = compute_c5(h)
    for p in range(n_perturb):
        if time.time() > deadline:
            break
        h_p = h + pert_scale * np.random.randn(n)
        h_p = project_feasible(h_p, target)
        h_opt, sc = adam_run(h_p, n, target, alpha, 200, 0.002, 0.9, 0.999, 1e-8,
                             deadline=start_time + 8)
        if sc < best_sc:
            best_sc = sc
            best_h = h_opt
    return best_h, best_sc


def multi_scale_optimize(h, n, target, deadline):
    coarse_sizes = [40, 80, 160]
    x_fine = np.linspace(0.0, 1.0, n)
    best = h.copy()
    best_sc = compute_c5(h)
    for nc in coarse_sizes:
        if time.time() > deadline:
            break
        x_c = np.linspace(0.0, 1.0, nc)
        h_c = np.interp(x_c, x_fine, best)
        h_c = project_feasible(h_c, nc / 2.0)
        for alpha in [50, 200, 1000]:
            if time.time() > deadline:
                break
            h_c, _ = adam_run(h_c, nc, nc / 2.0, alpha, 400, 0.005,
                              0.9, 0.999, 1e-8, deadline=deadline)
        h_f = np.interp(x_fine, x_c, h_c)
        h_f = project_feasible(h_f, target)
        h_f, sc = adam_run(h_f, n, target, 500, 300, 0.002,
                           0.9, 0.999, 1e-8, deadline=deadline)
        if sc < best_sc:
            best_sc = sc
            best = h_f
    return best, best_sc

def run(seed=42, budget_s=1300, **kwargs):
    global start_time
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 50.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Adam with increasing alpha from initial
    alphas_p1 = [30, 80, 200, 500]
    for alpha in alphas_p1:
        if tleft() < 80:
            break
        h_new, sc = adam_run(best_h, n, target, alpha, 600, 0.003,
                             0.9, 0.999, 1e-8, deadline=start_time + 60)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Variance-aware Adam to flatten profile
    if tleft() > 200:
        for vw in [0.05, 0.2, 0.8]:
            if tleft() < 80:
                break
            h_new, sc = var_adam_run(best_h, n, target, 300, vw, 500, 0.003,
                                     deadline=start_time + 50)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            if tleft() > 60:
                h_new2, sc2 = adam_run(best_h, n, target, 800, 300, 0.002,
                                       0.9, 0.999, 1e-8,
                                       deadline=start_time + 25)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_new2

    # Phase 3: SLSQP smooth optimization
    if tleft() > 150:
        for alpha in [200, 800, 3000, 10000]:
            if tleft() < 50:
                break
            h_new = slsqp_smooth_run(best_h, n, target, alpha, 2000, 1e-15,
                                     deadline=start_time + 30)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Heavy lag Adam (focus on worst lags)
    if tleft() > 120:
        for tk in [1, 2, 3, 5]:
            if tleft() < 60:
                break
            h_new, sc = heavy_lag_adam(best_h, n, target, tk, 600, 0.004,
                                       deadline=start_time + 40)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        if tleft() > 40:
            h_new = slsqp_smooth_run(best_h, n, target, 5000, 2000, 1e-16,
                                     deadline=start_time + 25)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 5: Multi-scale optimization
    if tleft() > 200:
        h_new, sc = multi_scale_optimize(best_h, n, target,
                                         deadline=start_time + 100)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if tleft() > 60:
            h_final2, sc2 = adam_run(best_h, n, target, 2000, 500, 0.001,
                                     0.9, 0.999, 1e-8,
                                     deadline=start_time + 40)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_final2

    # Phase 6: Second round Adam with high alpha
    if tleft() > 150:
        for alpha in [2000, 5000, 10000, 20000]:
            if tleft() < 60:
                break
            h_new, sc = adam_run(best_h, n, target, alpha, 400, 0.001,
                                 0.9, 0.999, 1e-8, deadline=start_time + 45)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Perturbation and re-optimization
    if tleft() > 200:
        for scale in [0.02, 0.05, 0.1]:
            if tleft() < 80:
                break
            h_new, sc = perturb_reopt(best_h, n, target, 1000, 5, scale,
                                      deadline=start_time + 50)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 8: Mixed alpha Adam (oscillate alpha to explore)
    if tleft() > 150:
        for alpha in [100, 500, 2000, 8000, 30000]:
            if tleft() < 50:
                break
            h_new, sc = adam_run(best_h, n, target, alpha, 300, 0.0015,
                                 0.9, 0.999, 1e-8, deadline=start_time + 30)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 9: SLSQP final polish
    if tleft() > 80:
        for alpha in [3000, 10000, 30000, 100000]:
            if tleft() < 30:
                break
            h_new = slsqp_smooth_run(best_h, n, target, alpha, 3000, 1e-17,
                                     deadline=start_time + 20)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 10: Final micro-polish
    if tleft() > 30:
        h_cur = best_h.copy()
        cur_sc = best_score
        dx = 2.0 / n
        for step in range(1000):
            if time.time() > start_time + 10:
                break
            over = get_over(h_cur)
            top5 = np.argsort(over)[::-1][:5]
            grad = np.zeros(n)
            for k in top5:
                k = int(k)
                grad[:n-k] += (1.0 - h_cur[k:])
                grad[k:] -= h_cur[:n-k]
            grad *= dx / 5.0
            improved = False
            for lr in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
                h_new = h_cur - lr * grad
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_sc - 1e-15:
                    h_cur = h_new
                    cur_sc = sc
                    improved = True
                    break
            if not improved:
                break
        if cur_sc < best_score:
            best_score = cur_sc
            best_h = h_cur

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
