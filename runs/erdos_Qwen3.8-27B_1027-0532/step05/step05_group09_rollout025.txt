import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def all_correlations(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return corr

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = projected.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def binary_projection(h, target):
    """Project to binary with exactly 'target' ones."""
    n = len(h)
    h_bin = np.zeros(n)
    indices = np.argsort(h)[::-1][:int(target)]
    h_bin[indices] = 1.0
    return h_bin

def lse_objective_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = all_correlations(h)
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    w = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx

    P = np.correlate(v, w, mode='full')[:n]
    N = np.convolve(w, h, mode='full')[:n]
    grad = dx * (P - N)
    
    return obj, grad

def arc_polish(h, target, max_rounds, deadline, top_k=5):
    """Polish by focusing on the top-k worst lags."""
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = all_correlations(h_cur)
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        # Weights: exponential of normalized differences
        diffs = corr[sorted_idx] - corr.min()
        w = np.exp(diffs / 0.01)
        w /= w.sum()
        
        # Gradient as weighted sum
        grad = np.zeros(n)
        v = 1.0 - h_cur
        for wi, ki in zip(w, sorted_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * v[jp + ki]
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Line search
        improved = False
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]:
            h_try = project_feasible(h_cur - eps * grad, target)
            sc = compute_c5(h_try)
            if sc < cur_score - 1e-15:
                h_cur = h_try
                cur_score = sc
                improved = True
                break
        if not improved:
            # Try smaller steps
            for eps in [5e-5, 1e-5, 5e-6, 1e-6]:

                sc = compute_c5(h_try)
                if sc < cur_score - 1e-15:
                    h_cur = h_try
                    cur_score = sc
                    break
            else:

def lse_obj_grad(h, alpha):

    grad = dx * (P - N)

def project_feasible(h, target):

def lse_obj_grad(h, alpha):
    n = len(h)

    corr = all_correlations(h)

    P = np.correlate(v, w, mode='full')[:n]

    grad = dx * (P - N)
    return obj, grad
