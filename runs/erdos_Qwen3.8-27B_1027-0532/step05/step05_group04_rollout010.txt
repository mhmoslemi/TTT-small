import numpy as np
from scipy.optimize import minimize, linprog
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_weights = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def get_overlaps(corr, n):
    return corr[n - 1:n - 1 + n]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(get_overlaps(compute_corr(h), n))) * dx


def project_exact(h, target_sum):
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    projected = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - projected.sum()
    if abs(diff) > 1e-10:
        mask = (projected > 1e-8) & (projected < 1.0 - 1e-8)
        cnt = mask.sum()
        if cnt > 0:
            projected[mask] += diff / cnt
            projected = np.clip(projected, 0.0, 1.0)
    return projected


def f_k_value(h, k):
    n = len(h)
    dx = 2.0 / n
    return float(np.sum(h[:n - k] * (1.0 - h[k:n]))) * dx


def f_k_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if k == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    else:
        g[:n - k] += (1.0 - h[k:n]) * dx
        g[k:] -= (h[:n - k]) * dx
    return g


def _obj_lse(x):
    n = _g_n
    corr = compute_corr(x)
    over = get_overlaps(corr, n)
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_exact(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def fista_lse(h, n, target, n_iter, alpha, deadline=None):
    h_cur = h.copy()
    y = h_cur.copy()
    t_moment = 1.0
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr_y = compute_corr(y)
        over_y = get_overlaps(corr_y, n)
        max_v = over_y.max()
        weights = np.exp(np.clip(alpha * (over_y - max_v), -500, 0))
        weights /= weights.sum()
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            if k == 0:
                grad += wk * (1.0 - 2.0 * y)
            else:
                grad[:n - k] += wk * (1.0 - y[k:n])
                grad[k:] -= wk * y[:n - k]

        y = project_exact(y, target)

        h_cur = h_new
        sc = compute_c5(h_cur)

            best_h = h_cur.copy()
    return best_h, best_score


def solve_lp_direction(h_cur, active_k, n, target):

    m = len(active_k)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)

        k = int(k)
        g = f_k_grad(h_cur, k)
        val = f_k_value(h_cur, k)
        A[i, :n] = g
        A[i, n] = -1.0
        c_k = val - np.dot(g, h_cur)
        b[i] = -c_k
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([target])

    try:

                      options={'presolve': True})

            return h_new
    except Exception:
        pass
    return None
