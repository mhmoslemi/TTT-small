def find_optimal_t(A, C, t_min, t_max, v_target):
    """Find t that minimizes max(A + C*t) subject to t_min <= t <= t_max,
    trying to get max <= v_target. Returns the best t."""
    # Binary search on v = max(A + C*t)
    lo_v = float(np.min(A))  # lower bound on achievable v
    hi_v = float(np.max(A + C * t_max)) if t_max > 0 else float(np.max(A + C * t_min))
    
    if hi_v - lo_v < 1e-13:
        return (t_min + t_max) * 0.5
    
    for _ in range(40):
        mid_v = (lo_v + hi_v) * 0.5
        pos = C > 1e-14
        neg = C < -1e-14
        ub = t_max
        if np.any(pos):
            ub = min(ub, float(np.min((mid_v - A[pos]) / C[pos])))
        lb = t_min
        if np.any(neg):
            lb = max(lb, float(np.max((mid_v - A[neg]) / C[neg])))
        if lb <= ub + 1e-12:
            hi_v = mid_v
        else:
            lo_v = mid_v
    
    v_best = hi_v
    pos = C > 1e-14
    neg = C < -1e-14
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v_best - A[pos]) / C[pos])))
    lb = t_min
    if np.any(neg):
        lb = max(lb, float(np.max((v_best - A[neg]) / C[neg])))
    ub = min(ub, t_max)
    lb = max(lb, t_min)
    if lb > ub:
        return (t_min + t_max) * 0.5
    return lb + (ub - lb) * 0.5

def fast_coord_desc(h, n, target, deadline, rng, recompute_interval=50):
    """Fast paired coordinate descent without SA."""
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    
    for sweep in range(100000):
        if time.time() > deadline:
            break
        
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        
        rng.shuffle(pair_positions)
        
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            
            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            
            t_star = find_optimal_t(A, C, t_min, t_max, best_score / dx)
            
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 300 and sweep > 200:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            best_score = float(np.max(corr)) * dx
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)

def block_swap(h, n, target, rng, block_size=10):
    """Swap two blocks of the h vector."""
    h_new = h.copy()
    i = rng.randint(0, n - block_size)
    j = rng.randint(0, n - block_size)
    while j == i or abs(j - i) < block_size:

    h_new[i:i+block_size], h_new[j:j+block_size] = h_new[j:j+block_size].copy(), h_new[i:i+block_size].copy()

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B

def find_optimal_t(A, C, t_min, t_max):

    hi = float(np.max(A + C * t_max)) if np.max(C) > 0 else float(np.max(A + C * t_min))

    # We want an upper bound on min_t max(A + C*t)
    # The maximum of A + C*t_max gives an upper bound
    hi = float(np.max(A + C * np.clip(t_max, t_min, t_max)))
    # And a lower bound is min(A)
    lo = float(np.min(A))
    
    if hi - lo < 1e-13:
        return 0.5 * (t_min + t_max)
    
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        pos = C > 1e-14
        neg = C < -1e-14
        ub = t_max
        if np.any(pos):
            ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
        lb = t_min
        if np.any(neg):
            lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
        if lb <= ub + 1e-12:
            hi = mid
        else:
            lo = mid
    
    v_best = hi
    pos = C > 1e-14
    neg = C < -1e-14
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v_best - A[pos]) / C[pos])))
    lb = t_min
    if np.any(neg):

    ub = min(ub, t_max)
    lb = max(lb, t_min)
    if lb > ub:
        return 0.5 * (t_min + t_max)
    return lb + 0.5 * (ub - lb)

            Bj = compute_B(h_cur, j, n)
            C_vec = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c_sum
            
            t_star = find_optimal_t(A, C_vec, t_min, t_max)
            delta = t_star - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c_sum - t_star
                corr += delta * C_vec
        
        sweep += 1
        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 200 and sweep > 100:
            pert = rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            best_score = float(np.max(corr)) * dx
            stall_count = 0
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def gradient_polish(h, n, target, deadline, alpha=1000.0):
    """Gradient descent on smoothed objective with projections."""
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.005
    patience = 0
    
    while time.time() < deadline and patience < 2000:
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum
        
        # Compute gradient
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if w[k] < 1e-10:
                continue
            s = k - (n - 1)
            if s >= 0:
                g[s:] += w[k] * (1.0 - h_cur[:n - s])
                g[:n - s] -= w[k] * h_cur[s:]
            else:
                sa = -s
                g[:n - sa] += w[k] * (1.0 - h_cur[sa:])
                g[sa:] -= w[k] * h_cur[:n - sa]
        
        # Project gradient onto sum=0 subspace
        g = g - np.mean(g)
        
        # Update with backtracking
        for _ in range(20):
            h_new = h_cur + lr * g
            h_new = project_feasible(h_new, target)
            new_score = compute_c5(h_new)
            if new_score < cur_score:
                h_cur = h_new
                cur_score = new_score
                patience = 0
                break
            lr *= 0.5
        
        else:
            patience += 1
        if lr < 1e-12:
            break
    
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def _obj_lse_alpha(x, alpha, n, target):
    x = project_feasible(x, target)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def _cons_eq(x, target):
    return np.sum(x) - target

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_polish(h, n, target, alphas, maxiter=2000, ftol=1e-15):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for a in alphas:
        def obj(x):
            return _obj_lse_alpha(x, a, n, target)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac_fn(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac_fn}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_swap(h, n, target, rng, block_size=10):
    h_new = h.copy()
    i = rng.randint(0, n - block_size)
    j = rng.randint(0, n - block_size)
    attempts = 0
    while abs(j - i) < block_size and attempts < 100:
        j = rng.randint(0, n - block_size)
        attempts += 1
    if abs(j - i) < block_size:
        return h_new
    tmp = h_new[i:i+block_size].copy()
    h_new[i:i+block_size] = h_new[j:j+block_size]
    h_new[j:j+block_size] = tmp
    return project_feasible(h_new, target)
