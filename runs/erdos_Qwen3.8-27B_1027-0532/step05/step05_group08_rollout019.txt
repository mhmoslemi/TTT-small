def logse_grad_wrap(h, alpha):

def active_logse_obj(h, alpha, active_idx, n):

    c_active = corr[active_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    return am + float(np.log(np.sum(np.exp(shifted)))) / alpha


def active_logse_grad(h, alpha, active_idx, n):
    corr = compute_corr(h)
    c_active = corr[active_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()

def slsqp_active(h, n, target, active_idx, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return active_logse_obj(x, alpha, active_idx, n)
    
    def objg(x):
        return active_logse_grad(x, alpha, active_idx, n)

        return np.sum(x) - target
    
    def con_jac(x):
        return np.ones(n)

    try:

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m_idx in zip(weights, top_idx):
            m_idx = int(m_idx)
            k = m_idx - (n - 1)
            if k >= 0:
                g[:n-k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n-ell])
                g[:n-ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)

        else:

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def equalize_active_lags(h, n, target, active_idx, max_iter, deadline):

    shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
    val = cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha
    val += mu * (np.sum(h) - target) ** 2
    return val


def penalized_grad(h, alpha, mu, top_k):
    n = len(h)
    target = n / 2.0
    g = logse_grad_sparse(h, alpha, top_k)
    g += 2.0 * mu * (np.sum(h) - target)
    return g


def active_obj(h, alpha, active_idx):
    n = len(h)
    corr = compute_corr(h)
    c_active = corr[active_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    return am + float(np.log(np.sum(np.exp(shifted)))) / alpha


def active_grad(h, alpha, active_idx):
    n = len(h)
    corr = compute_corr(h)
    c_active = corr[active_idx]
    am = float(np.max(c_active))
    shifted = np.clip(alpha * (c_active - am), -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.ones(len(active_idx)) / len(active_idx)
    else:
        w /= wsum
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, active_idx):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def constraint_fun(x):
    return np.sum(x) - len(x) / 2.0


def constraint_jac(x):
    return np.ones(len(x))


def lbfgsb_optimize(h, n, alpha, mu, top_k, maxiter):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(penalized_obj, h, method='L-BFGS-B', jac=penalized_grad,
                       args=(alpha, mu, top_k), bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-12, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slsqp_active_optimize(h, n, active_idx, alpha, maxiter, ftol):
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(active_obj, h, method='SLSQP', jac=active_grad,
                       args=(alpha, active_idx), bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(top_k) / top_k
        else:
            weights /= wsum
        g = np.zeros(n)
        for wk, m_idx in zip(weights, top_idx):
            m_idx = int(m_idx)
            k = m_idx - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = project_feasible(h_cur - lr * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= lr_decay
            if lr < 1e-13:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = project_feasible(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.015, 0.012)
        else:
            eps *= 0.82
            if eps < 1e-11:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multistart_refine(h, n, target, n_starts, deadline):
    rng = np.random.RandomState(12345)
    best_h = h.copy()
    best_score = compute_c5(h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        scale = 0.03 + 0.04 * s
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        top_k = 30
        for alpha in [500, 5000]:
            if time.time() > deadline:
                break
            h_new = lbfgsb_optimize(h_pert, n, alpha, 100.0, top_k, 150)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_pert = h_new
        corr = compute_corr(best_h)
        active = np.argsort(corr)[::-1][:5]
        h_new = slsqp_active_optimize(best_h, n, active, 10000, 1000, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B continuation with fine-grained alpha sweep
    alphas_p1 = [8, 15, 30, 60, 120, 250, 500, 1000, 2500, 5000, 10000, 25000]
    for alpha in alphas_p1:
        if time.time() > deadline - 300:
            break
        top_k = min(40, n // 5)
        for mu in [5.0, 50.0, 500.0]:
            if time.time() > deadline - 400:
                break
            h_new = lbfgsb_optimize(best_h, n, alpha, mu, top_k, 250)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Active-set SLSQP (multiple rounds with growing alpha)
    if time.time() < deadline - 200:
        corr = compute_corr(best_h)
        active = np.argsort(corr)[::-1][:8]
        for alpha in [2000, 10000, 40000]:
            if time.time() > deadline - 150:
                break
            h_new = slsqp_active_optimize(best_h, n, active, alpha, 3000, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            corr = compute_corr(best_h)
            active = np.argsort(corr)[::-1][:8]

    # Phase 3: Subgradient polish with adaptive parameters
    if time.time() < deadline - 150:
        for temp in [0.01, 0.004, 0.0015, 0.0005]:
            if time.time() > deadline - 100:
                break
            n_steps = 1500
            h_sg, sc_sg = subgradient_polish(best_h, n, target, n_steps, temp, 0.004, 0.997, deadline - 100)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 4: Worst-lag surgery (multiple passes)
    if time.time() < deadline - 100:
        for eps_init in [0.005, 0.002, 0.001]:
            if time.time() > deadline - 70:
                break
            h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 400, eps_init, deadline - 70)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 5: Another round of SLSQP + L-BFGS-B
    if time.time() < deadline - 80:
        top_k = 30
        for alpha in [20000, 60000]:
            if time.time() > deadline - 60:
                break
            h_new = lbfgsb_optimize(best_h, n, alpha, 500.0, top_k, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        corr = compute_corr(best_h)
        active = np.argsort(corr)[::-1][:5]
        h_new = slsqp_active_optimize(best_h, n, active, 50000, 2000, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Multi-resolution n=600
    if time.time() < deadline - 250:
        n2 = 600
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)

                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new

        if time.time() < deadline - 100:
            corr2 = compute_corr(best_h2)
            active2 = np.argsort(corr2)[::-1][:7]
            for alpha in [5000, 25000]:
                if time.time() > deadline - 80:
                    break
                h2_new = slsqp_active_optimize(best_h2, n2, active2, alpha, 2000, 1e-15)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
                corr2 = compute_corr(best_h2)
                active2 = np.argsort(corr2)[::-1][:7]

        if time.time() < deadline - 60:
            h_sg2, sc_sg2 = subgradient_polish(best_h2, n2, target2, 1200, 0.003, 0.003, 0.998, deadline - 50)

    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
