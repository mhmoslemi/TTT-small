import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        over = corr[n-1:]  # overlap part

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:

        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    return project_feasible(h_out, target_dst)

def compute_over_and_G(h, n):
    """Compute overlap values and gradient matrix."""
    dx = 2.0 / n
    over = np.zeros(n)
    G = np.zeros((n, n))
    for k in range(n):
        if k == 0:
            over[k] = np.sum(h * (1.0 - h)) * dx
            G[:, k] = dx * (1.0 - 2.0 * h)
        else:
            over[k] = np.sum(h[:n-k] * (1.0 - h[k:])) * dx
            g = np.zeros(n)
            g[:n-k] = 1.0 - h[k:]
            g[k:] -= h[:n-k]
            G[:, k] = dx * g
    return over, G

def pair_cd(h, n, target, n_rounds, deadline):
    """Pair coordinate descent with exact 1D subproblem solution."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        over, G = compute_over_and_G(h_cur, n)
        improved = False
        # Randomize pair order
        pairs = [(i, i+1) for i in range(0, n-1, 2)]
        np.random.shuffle(pairs)
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            D = G[i, :] - G[j, :]
            lo = max(-h_cur[i], h_cur[j] - 1.0)
            hi = min(1.0 - h_cur[i], 1.0 - h_cur[j])
            if hi - lo < 1e-13:
                continue
            # Minimize max_k(over[k] + delta * D[k]) for delta in [lo, hi]
            cur_lo, cur_hi = lo, hi
            a = cur_hi - gr * (cur_hi - cur_lo)
            b = cur_lo + gr * (cur_hi - cur_lo)
            fa = np.max(over + a * D)
            fb = np.max(over + b * D)
            for _ in range(50):
                if fa < fb:
                    cur_hi = b
                    b = a
                    fb = fa
                    a = cur_hi - gr * (cur_hi - cur_lo)
                    fa = np.max(over + a * D)
                else:
                    cur_lo = a
                    a = b
                    fa = fb
                    b = cur_lo + gr * (cur_hi - cur_lo)
                    fb = np.max(over + b * D)
            best_delta = (cur_lo + cur_hi) / 2.0
            new_val = h_cur[i] + best_delta
            if abs(best_delta) < 1e-13:
                continue
            h_new = h_cur.copy()
            h_new[i] = new_val
            h_new[j] = h_cur[j] - best_delta
            h_new = np.clip(h_new, 0.0, 1.0)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                h_cur = h_new
                best_score = sc
                improved = True
        if not improved:
            break
    return h_cur, best_score

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    n_global = n
    target_global = target
    alpha_global = [0.0]
    
    def _cons_eq(x):
        return np.sum(x) - target_global
    
    def _cons_jac(x):
        return np.ones(n_global)
    
    def _lse_obj(x):
        corr = compute_corr(x)
        c_max = float(np.max(corr))
        shifted = alpha_global[0] * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha_global[0]
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        alpha_global[0] = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, compute_c5(h_cur)

def subgradient_momentum(h, n, target, n_steps, n_active, lr_init, momentum, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        temp = max(over[top_idx[0]] * 0.005, 1e-6)
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        # Remove mean to maintain constraint
        grad -= np.mean(grad)
        
        vel = momentum * vel + (1.0 - momentum) * grad
        h_new = h_cur - lr * vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.05, lr_init * 10.0)
        else:
            lr *= 0.8
            vel *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_score

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        over, G = compute_over_and_G(h_cur, n)
        improved = False
        order = np.arange(0, n - 1, 2)
        rng.shuffle(order)
        for i in order:
            if time.time() > deadline:
                break
            j = i + 1
            D = G[i, :] - G[j, :]
            lo = max(-h_cur[i], h_cur[j] - 1.0)
            hi = min(1.0 - h_cur[i], 1.0 - h_cur[j])
            if hi - lo < 1e-13:
                continue
            cur_lo, cur_hi = lo, hi
            a = cur_hi - gr * (cur_hi - cur_lo)
            b = cur_lo + gr * (cur_hi - cur_lo)
            fa = np.max(over + a * D)
            fb = np.max(over + b * D)
            for _ in range(55):
                if fa < fb:
                    cur_hi = b
                    b = a
                    fb = fa
                    a = cur_hi - gr * (cur_hi - cur_lo)
                    fa = np.max(over + a * D)
                else:
                    cur_lo = a
                    a = b
                    fa = fb
                    b = cur_lo + gr * (cur_hi - cur_lo)
                    fb = np.max(over + b * D)
            best_delta = (cur_lo + cur_hi) / 2.0
            if abs(best_delta) < 1e-13:
                continue
            h_new = h_cur.copy()
            h_new[i] = h_cur[i] + best_delta
            h_new[j] = h_cur[j] - best_delta
            h_new = np.clip(h_new, 0.0, 1.0)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                h_cur = h_new
                best_score = sc
                improved = True
        if not improved:
            break
    return h_cur, best_score


def subgradient_momentum(h, n, target, n_steps, n_active, lr_init, momentum, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        temp = max(over[top_idx[0]] * 0.005, 1e-6)
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * (1.0 - 2.0 * h_cur)
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        grad -= np.mean(grad)
        vel = momentum * vel + (1.0 - momentum) * grad
        h_new = h_cur - lr * vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.05, lr_init * 10.0)
        else:
            lr *= 0.8
            vel *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_descent(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:]
        k_star = int(np.argmax(over))
        if k_star == 0:
            grad = dx * (1.0 - 2.0 * h_cur)
        else:
            grad = np.zeros(n)
            j_pos = np.arange(0, n - k_star)
            grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
            grad *= dx
        grad -= np.mean(grad)
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.03)
        else:
            eps *= 0.6
            if eps < 1e-10:
                break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    hard_deadline = start_time + budget_s - 25.0
    rng = np.random.RandomState(seed)

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Pair CD (fast exploitation)
    dl = min(start_time + 80, hard_deadline)
    h_p, sc_p = pair_cd(best_h, n_final, target_final, 40, dl, seed=seed)
    if sc_p < best_score:
        best_score = sc_p
        best_h = h_p

    # Phase 2: SLSQP moderate alpha
    dl = min(start_time + 200, hard_deadline)
    h_s, sc_s = slsqp_optimize(best_h, n_final, target_final, [200, 1000, 5000], 3000, 1e-14, dl)
    if sc_s < best_score:
        best_score = sc_s
        best_h = h_s

    # Phase 3: Pair CD again
    dl = min(start_time + 300, hard_deadline)
    h_p, sc_p = pair_cd(best_h, n_final, target_final, 40, dl, seed=seed + 1)
    if sc_p < best_score:
        best_score = sc_p
        best_h = h_p

    # Phase 4: SLSQP high alpha
    dl = min(start_time + 500, hard_deadline)
    h_s, sc_s = slsqp_optimize(best_h, n_final, target_final, [5000, 30000, 100000], 4000, 1e-15, dl)
    if sc_s < best_score:
        best_score = sc_s
        best_h = h_s

    # Phase 5: Subgradient with momentum
    dl = min(start_time + 650, hard_deadline)
    h_sg, sc_sg = subgradient_momentum(best_h, n_final, target_final, 2000, 5, 0.003, 0.9, dl)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg

    # Phase 6: Pair CD + worst-lag descent
    dl = min(start_time + 780, hard_deadline)
    h_p, sc_p = pair_cd(best_h, n_final, target_final, 30, dl, seed=seed + 2)
    if sc_p < best_score:
        best_score = sc_p
        best_h = h_p
    h_w, sc_w = worst_lag_descent(best_h, n_final, target_final, 300, 0.003, dl)
    if sc_w < best_score:
        best_score = sc_w
        best_h = h_w

    # Phase 7: SLSQP very high alpha
    dl = min(start_time + 950, hard_deadline)
    h_s, sc_s = slsqp_optimize(best_h, n_final, target_final, [50000, 200000, 500000], 3000, 1e-16, dl)
    if sc_s < best_score:
        best_score = sc_s
        best_h = h_s

    # Phase 8: Random restarts with pair CD + SLSQP
    dl = min(start_time + 1150, hard_deadline)
    restart_scales = [0.02, 0.04, 0.06, 0.03, 0.05, 0.01]
    for ri in range(len(restart_scales)):
        if time.time() > dl:
            break
        h_pert = best_h + rng.randn(n_final) * restart_scales[ri]
        h_pert = project_feasible(h_pert, target_final)
        sub_dl = min(time.time() + 25, dl)
        h_p, sc_p = pair_cd(h_pert, n_final, target_final, 8, sub_dl, seed=seed + 10 + ri)
        sub_dl2 = min(time.time() + 30, dl)
        h_s, sc_s = slsqp_optimize(h_p, n_final, target_final, [1000, 20000, 100000], 1500, 1e-14, sub_dl2)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 9: Final fine polishing
    if time.time() < hard_deadline - 40:
        dl = min(start_time + budget_s - 50, hard_deadline)
        h_p, sc_p = pair_cd(best_h, n_final, target_final, 25, dl, seed=seed + 100)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    if time.time() < hard_deadline - 20:
        dl = hard_deadline
        h_s, sc_s = slsqp_optimize(best_h, n_final, target_final, [100000, 500000, 1000000], 2000, 1e-17, dl)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
