from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(len(w)):

        if wk < 1e-15:

        k = idx - (n - 1)

        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_round(h, n, target, alpha, maxiter, ftol):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)

        res = minimize(lambda x: lse_value(x, alpha), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

# Callback functions for scipy optimize (using globals)
def _obj_lse(x):
    return lse_value(x, _g_alpha) + _g_mu * (np.sum(x) - _g_target) ** 2

def _grad_lse(x):
    return lse_grad(x, _g_alpha) + 2.0 * _g_mu * (np.sum(x) - _g_target)

def _obj_topk(x):
    return topk_value(x, _g_k_top) + _g_mu * (np.sum(x) - _g_target) ** 2

def _grad_topk(x):
    return topk_grad(x, _g_k_top) + 2.0 * _g_mu * (np.sum(x) - _g_target)

def _cons_eq_fun(x):
    return np.sum(x) - _g_target

def _cons_eq_jac(x):
    return np.ones(_g_n)

def lbfgsb_optimize(h, n, target, obj_func, grad_func, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h, method='L-BFGS-B', jac=grad_func,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_optimize(h, n, target, obj_func, grad_func, maxiter=5000, ftol=1e-16):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq_fun, 'jac': _cons_eq_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def surgery_backtrack(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
        # Compute gradient of worst lag
        m_star = int(np.argmax(corr))
        k = m_star - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n - k] += (1.0 - h_cur[k:])
            g[k:] -= h_cur[:n - k]
        else:
            ell = -k
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        g = -g / g_norm  # descent direction
        # Backtracking
        accepted = False
        for bt in range(25):
            ss = eps * (0.6 ** bt)
            h_new = h_cur + ss * g  # g is already negated
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < c_curr - 1e-16:
                h_cur = h_new
                accepted = True
                break
        if accepted:
            eps = min(eps * 1.04, 0.1)
        else:
            eps *= 0.6
            if eps < 1e-13:
                break
    return h_cur, best_score

def equalize_top3(h, n, target, max_iters, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top3 = np.argsort(corr)[::-1][:3]
        top_vals = corr[top3]
        c_max = float(np.max(corr)) * dx
        if c_max < best_score:
            best_score = c_max
        # Gradient: weighted by how much each top lag exceeds the min of top3
        min_top = float(np.min(top_vals))
        weights = top_vals - min_top
        if float(np.sum(weights)) < 1e-14:
            weights = np.ones(3) / 3.0
        weights = weights / float(np.sum(weights))
        g = np.zeros(n)
        for w, idx in zip(weights, top3):
            kk = int(idx) - (n - 1)
            if kk >= 0:
                g[:n - kk] += w * (1.0 - h_cur[kk:])
                g[kk:] -= w * h_cur[:n - kk]
            else:
                ell = -kk
                g[ell:] += w * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= w * h_cur[ell:]
        g *= dx
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break
        g = -g / g_norm
        eps = 0.02
        accepted = False
        for bt in range(20):
            ss = eps * (0.5 ** bt)
            h_new = h_cur + ss * g
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < best_score - 1e-16:
                h_cur = h_new
                best_score = sc_new
                accepted = True
                break
        if not accepted:
            break
    return h_cur, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
