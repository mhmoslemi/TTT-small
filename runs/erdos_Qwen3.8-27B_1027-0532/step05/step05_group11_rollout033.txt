def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def full_lp_solve(h, n, dx, target, solver='SCS'):
    """Solve the full LP with all lag constraints to get a new h that minimizes the max overlap."""
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    # Build all lag constraints
    # For lag k (from -(n-1) to n-1):
    # overlap_k = sum_{i} h[i] * g[i+k] * dx where both i and i+k are in [0,n)
    # i ranges from max(0, -k) to min(n-1, n-1-k)

        lo = max(0, -k)
        hi = min(n, n - k)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo+k:hi+k] * dx
            constraints.append(cp.sum(coeff * h_var) <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:

def build_A_matrix(n, dx):
    """Build the matrix A where row k corresponds to lag k-(n-1) and column i is the coefficient."""
    # A has shape (2n-1, n)
    # A[k, i] = dx if the overlap at lag (k-(n-1)) includes h[i]
    # For lag k' = k - (n-1): overlap = sum_i h[i] * g[i+k'] * dx
    # h[i] appears in the constraint for lag k' if 0 <= i+k' < n, i.e., -k' <= i < n-k'
    # So A[k, i] = dx * g[i+k'] = dx * (1-h[i+k'])
    # But g depends on h, so we can't precompute A fully.
    # We can precompute the structure (which entries are non-zero).
    
    A_structure = np.zeros((2*n-1, n), dtype=bool)
    for k in range(2*n-1):
        k_prime = k - (n-1)
        lo = max(0, -k_prime)
        hi = min(n, n - k_prime)
        A_structure[k, lo:hi] = True
    return A_structure

    g = 1.0 - h

        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi > lo:

    return A

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights = weights / np.sum(weights)
    
    # Gradient of corr[k] with respect to h
    # corr[k] = sum_i h[i] * (1-h)[i + k - (n-1)]
    # d corr[k] / d h[j] = (1-h)[j + k - (n-1)] - h[j + k - (n-1)] (if valid)
    #                   + h[j] * (-1) ... wait, let me be more careful
    
    # corr[k] = sum_i h[i] * (1 - h[i + lag]) where lag = k - (n-1)
    #         = sum_i h[i] - sum_i h[i] * h[i+lag]
    # d corr[k] / d h[j]:
    #   From first term: +1 if j is in range
    #   From second term: -h[j+lag] (if j+lag valid) - h[j-lag] (if j-lag valid)
    #   (the derivative of h[i]*h[i+lag] w.r.t. h[j] is h[j+lag] when i=j, and h[j-lag] when i+lag=j)
    
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        lag = k_idx - (n - 1)
        w = weights[k_idx]
        if w < 1e-15:
            continue
        lo = max(0, -lag)
        hi = min(n, n - lag)
        # d/dh[j] of sum_{i=lo}^{hi-1} h[i]*(1-h[i+lag])
        # = sum_{i=lo}^{hi-1} [delta(i,j)*(1-h[i+lag]) - h[i]*delta(i+lag,j)]
        # = (1-h[j+lag]) [if j+lag valid and j in [lo,hi)) 
        #   - h[j-lag] [if j-lag valid and j-lag in [lo,hi)]
        
        # First part: j in [lo, hi) and j+lag in [0, n)
        # j in [lo, hi) means the first term h[j] is included
        # j+lag in [0,n) means the second factor (1-h[j+lag]) is defined
        # lo = max(0,-lag), hi = min(n, n-lag)
        # j in [lo, hi) and j+lag in [0,n): 
        #   j+lag >= 0 => j >= -lag (satisfied since j >= lo >= -lag)
        #   j+lag < n => j < n-lag (satisfied since j < hi <= n-lag)
        # So first part: for j in [lo, hi), contribution is (1-h[j+lag])
        
        if hi > lo:
            grad[lo:hi] += w * (1.0 - h[lo+lag:hi+lag])
        
        # Second part: i+lag = j, so i = j-lag
        # i in [lo, hi) => j-lag in [lo, hi) => j in [lo+lag, hi+lag)
        # Also i >= 0, i < n: i = j-lag, so 0 <= j-lag < n => lag <= j < n+lag
        # Combined: j in [max(lo+lag, lag), min(hi+lag, n+lag))
        # = [max(0, 0), min(n, n)) ... let me just compute directly
        
        j_lo = max(0, lo + lag)
        j_hi = min(n, hi + lag)
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w * (-h[j_lo - lag:j_hi - lag])
    
    grad *= dx
    return grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_max_val_and_weights(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        weights = np.ones_like(weights) / len(weights)
    else:
        weights = weights / total
    return c_max + float(np.log(total) / alpha), weights

def _slsqp_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _slsqp_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    total = np.sum(weights)
    if total < 1e-300:
        weights = np.ones_like(weights) / len(weights)
    else:
        weights = weights / total
    n_half = n - 1
    grad = np.zeros(n)
    active = weights > 1e-8
    for k_idx in np.where(active)[0]:
        w = weights[k_idx]
        lag = k_idx - n_half
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi > lo:
            grad[lo:hi] += w * (1.0 - x[lo + lag:hi + lag])
            j_lo = max(0, lo + lag)
            j_hi = min(n, hi + lag)
            if j_hi > j_lo:
                grad[j_lo:j_hi] += w * (-x[j_lo - lag:j_hi - lag])
    grad *= dx
    return grad


def slsqp_polish(h_start, n, target, alphas, maxiter=500):
    bounds = [(0.0, 1.0)] * n
    h_cur = project_feasible(h_start.copy(), target)
    for alpha in alphas:
        try:
            res = minimize(
                _slsqp_obj, h_cur, args=(alpha,),
                jac=_slsqp_grad,
                method='SLSQP', bounds=bounds,
                constraints=[{'type': 'eq',
                              'fun': lambda x: np.sum(x) - target,
                              'jac': lambda x: np.ones(n)}],
                options={'maxiter': maxiter, 'ftol': 1e-14}
            )
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            h_opt, c5_opt = alternating_full_lp(h_try, n_try, target_try,
                                                time_limit=min(60, tleft() * 0.4),
                                                max_iter=800,
                                                alpha_start=0.8, alpha_end=0.15)
            if tleft() > 30:
                h_opt2 = slsqp_polish(h_opt, n_try, target_try,
                                      alphas=[5000, 30000], maxiter=300)
                if compute_c5(h_opt2) < compute_c5(h_opt):
                    h_opt = h_opt2
                c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                n = n_try
                target = target_try
                dx = 2.0 / n
                break

    # Phase 7: Final targeted LP polish
    if tleft() > 50:
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f
        hp7, c5p7 = alternating_targeted_lp(best_h, n_f, target_f,
                                            time_limit=min(35, tleft() * 0.6),
                                            max_iter=300, n_worst=50,
                                            alpha_start=0.5, alpha_end=0.1)
        if c5p7 < best_c5:
            best_c5 = c5p7
            best_h = hp7

    # Phase 8: Multi-start with perturbation
    if tleft() > 60:
        for attempt in range(3):
            if tleft() < 40:
                break
            noise = 0.02 + attempt * 0.015
            h_pert = best_h + rng.randn(len(best_h)) * noise
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            n_p = len(h_pert)
            target_p = n_p / 2.0
            dx_p = 2.0 / n_p
            h_opt, c5_opt = alternating_full_lp(h_pert, n_p, target_p,
                                                time_limit=min(20, tleft() * 0.4),
                                                max_iter=300,
                                                alpha_start=0.7, alpha_end=0.2)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 9: Final SLSQP
    if tleft() > 30:
        n_f = len(best_h)
        target_f = n_f / 2.0
        hp9 = slsqp_polish(best_h, n_f, target_f,
                           alphas=[10000, 50000, 200000], maxiter=300)
        sc9 = compute_c5(hp9)
        if sc9 < best_c5:
            best_c5 = sc9
            best_h = hp9

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
