import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)

    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(40):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)

        x = x_new
        t = t_new
        sc = compute_c5(x)

def top_lag_gradient(h, target, num_active):
    """Compute gradient as weighted combination of top lag gradients."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:num_active]
    # Use softmax weights based on how close to max
    max_corr = corr[active[0]]
    diffs = corr[active] - max_corr
    w = np.exp(diffs / 0.01)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, ki in zip(w, active):
        ki = int(ki)
        if ki == 0:
            # corr_0 = sum h[i]*(1-h[i])
            grad += wi * (1.0 - 2.0 * h)
        else:
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h[jn - ki]
    grad *= dx
    return grad, num_active

def top_lag_polish(h, target, deadline, max_rounds=500):
    """Polish by targeting the top lag(s) with adaptive steps."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        grad, _ = top_lag_gradient(h_cur, target, 3)
        # Backtracking line search
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                break
    return h_cur, cur_score

def swap_descent(h, target, deadline, max_sweeps=100):
    """Coordinate descent using swap moves (preserves sum exactly)."""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n * 3):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Try swapping i and j
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            sc = compute_c5(h_cur)
            if sc < cur_score - 1e-14:
                cur_score = sc
                improved = True
            else:
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        if not improved:
            break
    return h_cur, cur_score

    c_max = corr.max()
    shifted = alpha * (corr - c_max)

    grad = np.zeros(n)
