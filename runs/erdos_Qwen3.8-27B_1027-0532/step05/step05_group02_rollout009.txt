```python
def compute_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Compute weights (softmax)
    max_corr = corr.max()
    weights = np.exp(alpha * (corr - max_corr))
    weights /= weights.sum()
    # weights[k] corresponds to shift k = k - (n-1)
    
    # Compute gradient
    grad = np.zeros(n)
    for j in range(n):
        for k in range(2*n-1):
            shift = k - (n-1)
            # (1-h[j+shift]) if 0<=j+shift<n
            if 0 <= j + shift < n:
                grad[j] += weights[k] * (1.0 - h[j + shift])
            # -h[j-shift] if 0<=j-shift<n
            if 0 <= j - shift < n:
                grad[j] -= weights[k] * h[j - shift]
    return grad * dx
```
