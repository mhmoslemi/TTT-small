import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h_init, n, target, n_sweeps=50, verbose=False):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        O = np.correlate(h, 1.0 - h, mode='full') * dx
        improved = False

            lo = n - 1 - j
            hi = 2*n - 2 - j
            
            idx_range = np.arange(lo, hi + 1)
            jk = j + idx_range - (n - 1)
            jmk = j - idx_range + (n - 1)
            
            P = 1.0 - h[jk]

            A = P - Q

            else:

                candidates.append(np.clip(old * 1.3, 0, 1))
                candidates.append(np.clip(old - 0.2, 0, 1))

        if c5_now < best_c5 - 1e-12:

def ale_lp(h_init, n, target, max_iter=2000, time_limit=60.0, n_top=180,
           damping_start=0.95, damping_end=0.80, reg_start=1e-4, reg_end=1e-8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 300:
            break
        
        g = 1.0 - h
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        n_half = n - 1
        top_idx = np.argsort(corr)[::-1][:n_top]
        lags = top_idx - n_half
        
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo_idx = max(0, k)
            hi_idx = min(n, k + n)
            if hi_idx > lo_idx:
                coeff = np.zeros(n)
                coeff[lo_idx:hi_idx] = g[lo_idx - k:hi_idx - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    
    return best_h, best_c5

def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-15):
    dx = 2.0 / n
    
    def obj(params):
        h_p = np.clip(params, 0, 1)
        corr = np.correlate(h_p, 1.0 - h_p, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(obj, h_cur, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
