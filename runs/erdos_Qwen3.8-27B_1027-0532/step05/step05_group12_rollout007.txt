    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)

For the swap h[i]: 1→0, h[j]: 0→1:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            break
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, alpha, dx, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cMax = float(np.max(corr))
    n_corr = 2 * n - 1
    if top_m is not None and top_m < n_corr:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(n_corr)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - cMax), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return cMax, np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for i in range(len(top_indices)):
        w = weights[i]
        if w < 1e-15:
            continue
        idx_k = top_indices[i]
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            m = n - k
            if m > 0:
                grad[k:] += w * (1.0 - h[:m]) * dx
                grad[:m] -= w * h[k:] * dx
        else:
            m = -k
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return cMax, grad

    _global_n = n
    _global_target_sum = target
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]

    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        cMax = float(np.max(corr))
        shifted = np.clip(alpha * (corr - cMax), -500.0, 0.0)
        return cMax + float(np.log(np.sum(np.exp(shifted)))) / alpha

    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,

    return h_start

    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = 2 * n - 1
        if ai < 3:
            top_m = 20
        elif ai < 6:
            top_m = 50
        elif ai < 9:
            top_m = 100
        elif ai < 12:
            top_m = 200
        momentum = np.zeros(n)
        beta = 0.7
        lr_base = 0.5
        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            cMax, grad = smooth_max_and_grad(h, alpha, dx, top_m)
            momentum = beta * momentum + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum))
            if gnorm < 1e-14:

            lr = lr_base / (np.sqrt(alpha) * gnorm + 1e-8)

                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
                lr *= 0.5
            if not improved:
                break
    return best_h, best_score


def slsqp_multistage(h_start, n, target, time_limit, start_time,

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    if tleft() > 30:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_fin, sc_fin = slsqp_multistage(best_h, n_cur, t_cur,
                                         time_limit=budget_s,
                                         start_time=start_time,
                                         deadline_offset=8)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
