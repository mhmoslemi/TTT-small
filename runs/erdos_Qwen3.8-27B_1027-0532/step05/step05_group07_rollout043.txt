```python
def compute_lse_gradient(h, p, n):
    """Compute gradient of LSE objective."""
    nl1 = 2*n - 1
    # p is the softmax weights over the 2n-1 correlations
    # grad[i] = sum_p p_p * (1-h[i-p+n-1]) * 1_{valid} - h[i+p-n+1] * 1_{valid}
    
    # Use the formula: grad[i] = sum_q (1-h[q]) * p[i+n-1-q] - sum_r h[r] * p[r-n+1-i]
    
    # First term: convolution of (1-h) with reversed p
    # p indexed from 0 to 2n-2
    # p[i+n-1-q] for q=0..n-1 means we need p[n-1..2n-2] reversed, then convolve with (1-h)
    
    # Let me just do it naively for clarity (n=400 is small enough)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(n):
        # First sum
        total = 0.0
        for q in range(n):
            idx = i + n - 1 - q
            if 0 <= idx < nl1:
                total += one_minus_h[q] * p[idx]
        # Second sum
        for r in range(n):
            idx = r - n + 1 - i
            if 0 <= idx < nl1:
                total -= h[r] * p[idx]
        grad[i] = total
    return grad
```

```python
def compute_lse_gradient_vec(h, p, n):
    nl1 = 2*n - 1
    one_minus_h = 1.0 - h
    # Build index matrix
    # For grad[i], first sum: p[i+n-1-q] for q=0..n-1
    # indices = [[i+n-1-0, i+n-1-1, ..., i+n-1-(n-1)] for i in range(n)]
    i_idx = np.arange(n)[:, None]
    q_idx = np.arange(n)[None, :]
    p_idx1 = i_idx + n - 1 - q_idx  # shape (n, n)
    p_idx2 = q_idx - n + 1 - i_idx  # shape (n, n)
    
    # Clip to valid range and mask
    mask1 = (p_idx1 >= 0) & (p_idx1 < nl1)
    mask2 = (p_idx2 >= 0) & (p_idx2 < nl1)
    
    p_idx1_clipped = np.clip(p_idx1, 0, nl1-1)
    p_idx2_clipped = np.clip(p_idx2, 0, nl1-1)
    
    # First term: sum over q of one_minus_h[q] * p[p_idx1[i,q]]
    term1 = np.sum(one_minus_h[None, :] * p[p_idx1_clipped] * mask1, axis=1)
    # Second term: sum over r of h[r] * p[p_idx2[i,r]]
    term2 = np.sum(h[None, :] * p[p_idx2_clipped] * mask2, axis=1)
    
    return term1 - term2
```

```python
def lse_subgradient_method(h, n, target, deadline, initial_alpha=100, final_alpha=10000, n_iter=5000):
    """Nesterov-accelerated subgradient method on LSE approximation."""
    h_cur = h.copy()
    y = h_cur.copy()  # Nesterov momentum
    beta = 0.0  # momentum coefficient
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for t in range(n_iter):
        if time.time() > deadline:
            break
        alpha = initial_alpha * (final_alpha / initial_alpha) ** (t / max(n_iter-1, 1))
        
        corr = compute_corr(y)
        # Compute softmax weights
        shifted = alpha * (corr - np.max(corr))
        shifted = np.clip(shifted, -700, 0)
        weights = np.exp(shifted)
        p = weights / np.sum(weights)
        
        # Compute gradient
        grad = compute_lse_gradient_vec(y, p, n)
        
        # Nesterov: take step from y
        step = 2.0 / (np.sqrt(t+1) * max(1, np.linalg.norm(grad)))
        y_new = y - step * grad
        
        # Project to feasible set
        y_new = project_feasible(y_new, target)
        
        # Update momentum
        beta_new = (1 + np.sqrt(1 + 4*beta**2)) / 2
        h_new = (1 - beta_new/beta) * y_new + (beta_new/beta - 1) * h_cur if beta > 0 else y_new
        # Actually, standard Nesterov:
        # y_{k+1} = x_{k+1} + (beta_k - 1)/beta_{k+1} * (x_{k+1} - x_k)
        # x_{k+1} = proj(y_k - step * grad(y_k))
        # beta_{k+1} = (1 + sqrt(1 + 4*beta_k^2)) / 2
        
        h_cur = y_new  # simplified: just use the projected point
        y = h_new if beta > 1e-10 else y_new
        
        score = compute_c5(h_cur)
        if score < best_score:
            best_score = score
            best_h = h_cur.copy()
    
    return best_h, best_score
```

For a given target C₀, we can formulate:

The bilinear terms make this non-convex. But we can use CVXPY's DCP framework with a **successive linearization** approach:

At iteration t:
1. Linearize corr_k at h_t: corr_k(h_t + δ) ≈ corr_k(h_t) + ∇_h corr_k(h_t) · δ
2. Solve the linearized problem:
   min C
   s.t. corr_k(h_t) + ∇_h corr_k(h_t) · (h - h_t) ≤ C·n for all k
        0 ≤ h ≤ 1
        Σh = n/2
3. Update h_t ← h

This is a **cutting plane** / **outer approximation** method. Each subproblem is an LP (or can be solved efficiently).

The advantage is that it can make larger moves than gradient methods. The disadvantage is that it might not converge to the global optimum (it converges to a local optimum or a stationary point).

Let me implement this as one of the phases.

Actually, for the linearized problem, since corr_k(h) is bilinear:
corr_k(h) = Σ_i h[i](1-h[i+k]) = Σ_i h[i] - Σ_i h[i]h[i+k]

The gradient with respect to h is:
∂corr_k/∂h[j] = (1-h[j+k]) - h[j-k] (with appropriate boundary handling)

So the linearization at h_t is:
corr_k(h_t + δ) ≈ corr_k(h_t) + Σ_j ∂corr_k(h_t)/∂h[j] · δ[j]

And the subproblem is:
min C
s.t. corr_k(h_t) + B_k(h_t) · (h - h_t) ≤ C·n for all k
     0 ≤ h ≤ 1
     Σh = n/2

Where B_k(h_t) is the gradient row for correlation k.

This is an LP with 2n-1 constraints (one per k) plus the bound and sum constraints. With n=400, that's 799 + 400 + 1 = 1200 constraints and 401 variables (400 h's + 1 C). Very manageable.

```python
def linearized_subproblem(h_t, n, target):
    """Solve one iteration of the cutting plane method."""
    h = cp.Variable(n)
    C = cp.Variable()
    corr_t = compute_corr(h_t)
    
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target
    ]
    
    # Linearize each corr_k
    # corr_k(h) ≈ corr_t[k] + grad_k · (h - h_t)
    for k in range(len(corr_t)):
```
