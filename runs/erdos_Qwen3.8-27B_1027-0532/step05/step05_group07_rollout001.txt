            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_score

def simulated_annealing(h, n, target, deadline, rng, initial_temp=0.05, cooling=0.995, min_temp=0.001):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    temp = initial_temp
    stall = 0
    max_stall = 200
    while time.time() < deadline and temp > min_temp:
        move_type = rng.random()
        if move_type < 0.4:
            block_size = rng.randint(3, 12)
            start = rng.randint(0, max(1, n - block_size))
            sl = slice(start, start + block_size)
            old = h_cur[sl].copy()
            noise_scale = temp * 0.5
            new = old + rng.randn(block_size) * noise_scale
            new = np.clip(new, 0.0, 1.0)
            h_cur[sl] = new
            h_cur = project_feasible(h_cur, target)
        elif move_type < 0.7:
            idx = rng.randint(0, n)
            step_size = temp * 0.3
            delta = rng.uniform(-step_size, step_size)
            h_cur[idx] = np.clip(h_cur[idx] + delta, 0.0, 1.0)
            h_cur = project_feasible(h_cur, target)
        else:
            i = rng.randint(0, n - 1)
            j = rng.randint(0, n - 1)
            if i == j:
                j = (i + 1) % n
            amount = rng.uniform(0.0, temp * 0.4)
            if rng.random() < 0.5:
                h_cur[i] = np.clip(h_cur[i] - amount, 0.0, 1.0)
                h_cur[j] = np.clip(h_cur[j] + amount, 0.0, 1.0)
            else:
                h_cur[i] = np.clip(h_cur[i] + amount, 0.0, 1.0)
                h_cur[j] = np.clip(h_cur[j] - amount, 0.0, 1.0)
            h_cur = project_feasible(h_cur, target)
        new_score = compute_c5(h_cur)
        if new_score < best_score:
            accept = True
        else:
            delta_e = new_score - best_score
            accept_prob = np.exp(-delta_e / max(temp, 1e-10))
            accept = rng.random() < accept_prob
        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
                stall = 0
            else:
                stall += 1
            if stall > max_stall:
                temp = max(min_temp, temp * 0.9)
                stall = 0
        else:
            h_cur = best_h.copy()
        temp = max(min_temp, temp * cooling)
    return best_h, best_score

def basis_optimize(h, n, target, deadline, rng, n_modes=40):
    x = np.linspace(0.0, 2.0, n)
    basis = np.zeros((n, n_modes))
    basis[:, 0] = 1.0 / np.sqrt(n)
    for mode in range(1, n_modes):
        basis[:, mode] = np.cos(np.pi * mode * x / 2.0)
    beta_init = np.zeros(n_modes)
    beta_init[0] = 0.5
    best_h = h.copy()
    best_score = compute_c5(best_h)
    for trial in range(5):
        if time.time() >= deadline:
            break
        beta = beta_init + rng.randn(n_modes) * 0.3
        for it in range(2000):
            if time.time() >= deadline:
                break
            h_basis = 1.0 / (1.0 + np.exp(-basis @ beta))
            h_basis = project_feasible(h_basis, target)
            sc = compute_c5(h_basis)
            if sc < best_score:
                best_score = sc
                best_h = h_basis.copy()
            beta += rng.randn(n_modes) * 0.05
    return best_h, best_score

def paired_coord_desc(h, n, target, deadline, rng, n_sweeps=None, recompute_interval=50):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
    sweep_count = 0
    nl1 = 2 * n - 1
    for sweep in range(n_sweeps if n_sweeps else 100000):
        if time.time() > deadline:
            break
        sweep_count += 1
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
        rng.shuffle(pair_positions)
        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            B = np.zeros(nl1)
            ks = np.arange(n)
            mask1 = ks <= i
            mask2 = (i + ks) < n
            b_pos = np.zeros(n)
            if np.any(mask1):
                b_pos[mask1] += (1.0 - h_cur[i - ks[mask1]])
            if np.any(mask2):
                b_pos[mask2] -= h_cur[i + ks[mask2]]
            B[n - 1:n - 1 + n] = b_pos
            ms = np.arange(1, n)
            mask3 = (i + ms) < n
            mask4 = ms <= i
            b_neg = np.zeros(n - 1)
            if np.any(mask3):
                b_neg[mask3] += (1.0 - h_cur[i + ms[mask3]])
            if np.any(mask4):
                b_neg[mask4] -= h_cur[i - ms[mask4]]
            B[0:n - 1] = b_neg
            Bi = B
            B = np.zeros(nl1)
            mask1 = ks <= j
            mask2 = (j + ks) < n
            b_pos = np.zeros(n)
            if np.any(mask1):
                b_pos[mask1] += (1.0 - h_cur[j - ks[mask1]])
            if np.any(mask2):
                b_pos[mask2] -= h_cur[j + ks[mask2]]
            B[n - 1:n - 1 + n] = b_pos
            mask3 = (j + ms) < n
            mask4 = ms <= j
            b_neg = np.zeros(n - 1)
            if np.any(mask3):
                b_neg[mask3] += (1.0 - h_cur[j + ms[mask3]])
            if np.any(mask4):
                b_neg[mask4] -= h_cur[j - ms[mask4]]
            B[0:n - 1] = b_neg
            Bj = B
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c
            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A[C > 0] + C[C > 0] * t_max))
            else:
                hi = float(np.max(A + C * t_max))
            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid
            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5
            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
        new_score = float(np.max(corr)) * dx
        if new_score < best_score:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 150 and sweep > 50:
            pert = rng.randn(n) * 0.015
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            best_score = compute_c5(h_cur)
            stall_count = 0
    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)

def slsqp_polish(h, n, target, alphas, maxiter=2000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted))) / _g_alpha

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick coordinate descent to establish strong baseline
    deadline_p1 = start_time + (budget_s - 15.0) * 0.12
    h_cd, sc_cd = paired_coord_desc(best_h, n, target, deadline_p1, rng)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 2: Basis function optimization for global exploration
    if tleft() > 300:
        deadline_p2 = start_time + (budget_s - 15.0) * 0.30
        h_bas, sc_bas = basis_optimize(best_h, n, target, deadline_p2, rng, n_modes=35)
        if sc_bas < best_score:
            best_score = sc_bas
            best_h = h_bas

    # Phase 3: Adam gradient optimization on full space
    if tleft() > 250:
        deadline_p3 = start_time + (budget_s - 15.0) * 0.50
        alphas = [50, 100, 200, 500, 1000, 2000]
        h_adam, sc_adam = adam_optimize(best_h, n, target, deadline_p3, alphas, lr=0.005)
        if sc_adam < best_score:
            best_score = sc_adam
            best_h = h_adam

    # Phase 4: Simulated annealing with structured moves
    if tleft() > 200:
        deadline_p4 = start_time + (budget_s - 15.0) * 0.68
        h_sa, sc_sa = simulated_annealing(best_h, n, target, deadline_p4, rng,
                                          initial_temp=0.04, cooling=0.997, min_temp=0.002)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 5: Coordinate descent after exploration
    if tleft() > 150:
        deadline_p5 = start_time + (budget_s - 15.0) * 0.78
        h_cd2, sc_cd2 = paired_coord_desc(best_h, n, target, deadline_p5, rng)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 6: Second round of Adam with higher alpha
    if tleft() > 120:
        deadline_p6 = start_time + (budget_s - 15.0) * 0.86
        alphas2 = [500, 1000, 3000, 8000]
        h_adam2, sc_adam2 = adam_optimize(best_h, n, target, deadline_p6, alphas2, lr=0.002)
        if sc_adam2 < best_score:
            best_score = sc_adam2
            best_h = h_adam2

    # Phase 7: SLSQP polish
    if tleft() > 60:
        h_slq = slsqp_polish(best_h, n, target, [1000, 5000, 15000], 2000, 1e-17)
        sc_slq = compute_c5(h_slq)
        if sc_slq < best_score:
            best_score = sc_slq
            best_h = h_slq

    # Phase 8: Final coordinate descent
    if tleft() > 30:
        h_final, sc_final = paired_coord_desc(best_h, n, target,
                                              start_time + budget_s - 10.0, rng,
                                              recompute_interval=20)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
