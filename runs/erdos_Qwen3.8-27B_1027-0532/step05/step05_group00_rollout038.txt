def precompute_grads(h, n):
    grad = np.zeros((n, 2*n-1))
    
    # For p = n-1+s, s = 0, 1, ..., n-1 (shift s >= 0):
    # grad[j, p] = (1-h[j-s])*[s<=j] - h[j+s]*[j+s<n]
    # For all j simultaneously:
    for s in range(n):
        p = n - 1 + s
        # (1-h[j-s])*[s<=j]: for j >= s, this is (1-h[j-s])
        # h[j+s]*[j+s<n]: for j < n-s, this is h[j+s]
        idx1 = range(s, n)  # j from s to n-1
        idx2 = range(0, n-s)  # j from 0 to n-s-1
        grad[idx1, p] += (1.0 - h[np.array(idx1) - s])
        grad[idx2, p] -= h[np.array(idx2) + s]

        grad[idx2, p] -= h[np.array(idx2) - s]

Let me recalculate. For shift k* ≥ 0:
a_{k*}(j) = (1-h[j-k*])*[k*≤j] - h[j+k*]*[j≤n-1-k*]

If h[j]≈1, h[j+k*]≈0, and j is in the valid range:
a_{k*}(j) ≈ (1-h[j-k*]) - 0 = 1-h[j-k*]

If j-k* is also "bad" (h[j-k*]≈0): a_{k*}(j) ≈ 1
If j-k* is "good" (h[j-k*]≈1): a_{k*}(j) ≈ 0

For l with h[l]≈0, h[l+k*]≈1:
a_{k*}(l) = (1-h[l-k*])*[k*≤l] - h[l+k*]*[l≤n-1-k*]
≈ (1-h[l-k*]) - 1 = -h[l-k*]

If h[l-k*]≈0: a_{k*}(l) ≈ 0
If h[l-k*]≈1: a_{k*}(l) ≈ -1

So the change in c_{k*} from swap (h[j]-=δ, h[l]+=δ) is:
δ * (a_{k*}(l) - a_{k*}(j)) ≈ δ * (-h[l-k*] - (1-h[j-k*]))

For this to be negative (reducing c_{k*}), we need a_{k*}(l) < a_{k*}(j), i.e., the "slope" difference is negative.

If a_{k*}(j) ≈ 1 and a_{k*}(l) ≈ 0: change ≈ δ * (0-1) = -δ < 0. Good!
If a_{k*}(j) ≈ 1 and a_{k*}(l) ≈ -1: change ≈ δ * (-1-1) = -2δ < 0. Even better!

So the focused strategy is:
- j from "bad ones" (h[j] high, h[j+k*] low, and h[j-k*] low → a_{k*}(j) high)
- l from "good zeros" (h[l] low, h[l+k*] high, and h[l-k*] high → a_{k*}(l) very low)

This maximizes the reduction in c_{k*}.

def coord_desc_focused(h, n, target, rng, time_budget, start_time):
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    M = 30  # number of top shifts to track
    batch_size = 400
    
    # Precompute grads
    grad = precompute_grads(h, n)
    batches_since_grad = 0
    batches_since_corr = 0
    stall = 0
    
    while time.time() - start_time < start_time + time_budget:
        # Update top shifts
        top_idx = np.argsort(corr)[-M:]
        corr_top = corr[top_idx].copy()
        
        # Determine worst shift
        worst_p = top_idx[-1]  # the one with largest corr
        worst_s = worst_p - (n-1)
        
        # Generate focused pairs
        # For worst shift s:
        #   j: h[j] high, h[j+s] low (for s≥0) or h[j] high, h[j+s] low (for s<0)
        #   l: h[l] low, h[l+s] high
        
        j_batch = np.empty(batch_size, dtype=int)
        l_batch = np.empty(batch_size, dtype=int)
        
        n_focused = batch_size * 3 // 5  # 60% focused
        n_random = batch_size - n_focused
        
        # Focused pairs
        if worst_s >= 0:
            # c_s = ∑ h[i+s]*(1-h[i]), i from 0 to n-1-s
            # "Bad" positions: h[i+s] high, h[i] low → these DON'T contribute
            # "Good" positions: h[i+s] low, h[i] high → these DO contribute (we want to fix them)
            # Wait, I want to REDUCE c_s. c_s = ∑ h[i+s]*(1-h[i])
            # To reduce c_s, I want to find i where h[i+s] is high and h[i] is low
            # and either reduce h[i+s] or increase h[i]
            
            # Option A: reduce h[i+s] (set it toward 0) → j = i+s (the "1" that should go to 0)
            # Option B: increase h[i] (set it toward 1) → l = i (the "0" that should go to 1)
            
            # For Option A: j = i+s, where h[j] is high and h[j-s] is low
            #   Paired with l where h[l] is low (to receive the "mass")
            # For Option B: l = i, where h[l] is low and h[l+s] is high  
            #   Paired with j where h[j] is high (to donate the "mass")
            
            # Let me just use: j from high-h positions, l from low-h positions
            # But biased toward the worst shift
            
            high_idx = np.where(h > 0.6)[0]
            low_idx = np.where(h < 0.4)[0]
            
            if len(high_idx) > 0 and len(low_idx) > 0:
                # Focused: prefer j where h[j] high and h[j-s] low (for s≥0)
                if len(high_idx) > 1:
                    scores_j = np.zeros(len(high_idx))
                    valid_j = []
                    for idx, j in enumerate(high_idx):
                        if 0 <= j - worst_s < n and 0 <= j < n:
                            scores_j[idx] = h[j] * (1 - h[j - worst_s])
                        else:
                            scores_j[idx] = 0
                    # Pick j with highest "offending" score
                    top_j_scores = np.argsort(scores_j)[-min(n_focused, len(high_idx)):]
                    j_focused = high_idx[top_j_scores]
                else:
                    j_focused = high_idx[:n_focused] if len(high_idx) >= n_focused else high_idx
                
                # l: prefer where h[l] low and h[l+s] high
                if len(low_idx) > 1:
                    scores_l = np.zeros(len(low_idx))
                    for idx, l in enumerate(low_idx):
                        if 0 <= l + worst_s < n:
                            scores_l[idx] = (1 - h[l]) * h[l + worst_s]
                        else:
                            scores_l[idx] = 0
                    top_l_scores = np.argsort(scores_l)[-min(n_focused, len(low_idx)):]
                    l_focused = low_idx[top_l_scores]
                else:
                    l_focused = low_idx[:n_focused] if len(low_idx) >= n_focused else low_idx
                
                # Pair them (shuffle l)
                nf = len(j_focused)
                nl = len(l_focused)
                nn = min(nf, nl, n_focused)
                rng.shuffle(l_focused)
                j_batch[:nn] = j_focused[:nn]
                l_batch[:nn] = l_focused[:nn]
            else:
                nn = 0
            
            # Random pairs
            nr = min(n_random, len(high_idx), len(low_idx)) if len(high_idx) > 0 and len(low_idx) > 0 else 0
            if nr > 0:
                j_r = high_idx[rng.randint(0, len(high_idx), nr)]
                l_r = low_idx[rng.randint(0, len(low_idx), nr)]
                j_batch[nn:nn+nr] = j_r
                l_batch[nn:nn+nr] = l_r
                nn += nr
        else:
            # Similar for s < 0
            # ... (symmetric)
            high_idx = np.where(h > 0.6)[0]
            low_idx = np.where(h < 0.4)[0]
            nr = min(batch_size, len(high_idx), len(low_idx))
            if nr > 0:
                j_batch[:nr] = high_idx[rng.randint(0, len(high_idx), nr)]
                l_batch[:nr] = low_idx[rng.randint(0, len(low_idx), nr)]
            nn = nr
        
        # Fill remaining with random
        if nn < batch_size:
            j_batch[nn:] = rng.randint(0, n, batch_size - nn)
            l_batch[nn:] = rng.randint(0, n, batch_size - nn)
        
        # Avoid j == l
        same = j_batch == l_batch
        if same.any():
            l_batch[same] = (l_batch[same] + 1) % n
        
        # Compute dgrad for top_M shifts
        dgrad = grad[j_batch, top_idx] - grad[l_batch, top_idx]  # (K, M)
        
        # Feasible range for δ (h[j] += δ, h[l] -= δ)
        δ_min = np.maximum(-h[j_batch], h[l_batch] - 1.0)
        δ_max = np.minimum(1.0 - h[j_batch], h[l_batch])
        feasible = δ_min < δ_max - 1e-12
        
        if not feasible.any():
            batches_since_grad += 1
            batches_since_corr += 1
            if batches_since_grad >= 100:
                grad = precompute_grads(h, n)
                batches_since_grad = 0
            if batches_since_corr >= 50:
                corr = np.correlate(h, 1.0 - h, mode='full')
                batches_since_corr = 0
            continue
        
        dgrad_f = dgrad[feasible]
        δ_min_f = δ_min[feasible]
        δ_max_f = δ_max[feasible]
        j_f = j_batch[feasible]
        l_f = l_batch[feasible]
        K = len(j_f)
        
        if K == 0:
            continue
        
        # Bisection for 1D optimum
        lo = δ_min_f.copy()
        hi = δ_max_f.copy()
        for _ in range(10):
            mid = (lo + hi) * 0.5
            f_vals = corr_top[None, :] + mid[:, None] * dgrad_f
            idx_max = f_vals.argmax(axis=1)
            slopes = dgrad_f[np.arange(K), idx_max]
            lo = np.where(slopes < -1e-10, mid, lo)
            hi = np.where(slopes > 1e-10, mid, hi)
        
        mid = (lo + hi) * 0.5
        # Evaluate at candidate points
        candidates = np.stack([mid, lo, hi], axis=1)  # (K, 3)
        f_cand = np.zeros((K, 3))
        for ci in range(3):
            f_cand[:, ci] = (corr_top[None, :] + candidates[:, ci, None] * dgrad_f).max(axis=1)
        
        best_ci = f_cand.argmin(axis=1)
        δ_best = candidates[np.arange(K), best_ci]
        f_best = f_cand[np.arange(K), best_ci]
        
        # Accept if improvement (reduces the current max)
        current_max = float(corr.max())
        improved = f_best < current_max - 1e-12
        n_improved = improved.sum()
        
        if n_improved > 0:
            # Apply updates
            idx_imp = np.where(improved)[0]
            for i in idx_imp:
                δ = δ_best[i]
                h[j_f[i]] += δ
                h[l_f[i]] -= δ
                # Update corr approximately
                corr[top_idx] += δ * dgrad_f[i]
            
            stall = 0
        else:
            stall += 1
        
        # Periodic exact recomputation
        batches_since_grad += 1
        batches_since_corr += 1
        if batches_since_corr >= 50:
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = float(corr.max()) * dx
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            batches_since_corr = 0
        
        if batches_since_grad >= 100:
            grad = precompute_grads(h, n)
            batches_since_grad = 0
        
        # Stall handling: perturb if stuck
        if stall > 5000:
            # Random perturbation
            n_pert = rng.randint(3, 10)
            for _ in range(n_pert):
                j = rng.randint(n)
                l = rng.randint(n)
                if j != l:
                    δ = rng.uniform(-0.3, 0.3)
                    if 0 <= h[j] + δ <= 1 and 0 <= h[l] - δ <= 1:
                        h[j] += δ
                        h[l] -= δ
            corr = np.correlate(h, 1.0 - h, mode='full')
            grad = precompute_grads(h, n)
            stall = 0
    
    # Final
    corr = np.correlate(h, 1.0 - h, mode='full')
    final_c5 = float(corr.max()) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return best_h, best_c5

Let me double-check the gradient formula:

For shift s ≥ 0, p = n-1+s:
corr[p] = ∑_{i=0}^{n-1-s} h[i+s] * (1-h[i])

∂corr[p]/∂h[j]:
- h[j] appears as h[i+s] when i = j-s (valid if 0 ≤ j-s ≤ n-1-s, i.e., s ≤ j ≤ n-1)
  Contribution: (1-h[j-s])
- h[j] appears in (1-h[i]) when i = j (valid if 0 ≤ j ≤ n-1-s, i.e., j+s ≤ n-1)
  Contribution: -h[j+s]

So ∂corr[p]/∂h[j] = (1-h[j-s])*[s ≤ j ≤ n-1] + (-h[j+s])*[j+s ≤ n-1]

For shift s < 0 (let t = -s > 0), p = n-1-t:
corr[p] = ∑_{i=0}^{n-1-t} h[i] * (1-h[i+t])

∂corr[p]/∂h[j]:
- h[j] appears as h[i] when i = j (valid if 0 ≤ j ≤ n-1-t, i.e., j+t ≤ n-1)
  Contribution: (1-h[j+t])
- h[j] appears in (1-h[i+t]) when i+t = j, i.e., i = j-t (valid if 0 ≤ j-t ≤ n-1-t, i.e., t ≤ j ≤ n-1)
  Contribution: -h[j-t]

So ∂corr[p]/∂h[j] = (1-h[j+t])*[j+t ≤ n-1] + (-h[j-t])*[t ≤ j ≤ n-1]
