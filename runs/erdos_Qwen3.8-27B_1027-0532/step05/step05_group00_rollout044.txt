import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

def apply_pair_flip(corr, h, i, j):
    n = len(h)
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    corr[n-1-i : 2*n-1-i] -= one_minus_h
    corr[i : i+n] += h_rev
    corr[n-1-j : 2*n-1-j] += one_minus_h
    corr[j : j+n] -= h_rev
    h[i] = 0.0
    h[j] = 1.0

def apply_pair_flip_undo(corr, h, i, j):
    """Undo the pair flip: h[i] 0->1, h[j] 1->0"""
    n = len(h)
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    corr[n-1-i : 2*n-1-i] += one_minus_h
    corr[i : i+n] -= h_rev
    corr[n-1-j : 2*n-1-j] -= one_minus_h
    corr[j : j+n] += h_rev
    h[i] = 1.0
    h[j] = 0.0

    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = h.copy()
    current_c5 = best_c5
    n_corr = len(corr)
    
    stall = 0
    max_stall = 50000
    
    for it in range(n_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        r = rng.random()
        new_c5 = current_c5
        accepted = False
        i_used = -1
        j_used = -1

            max_idx = int(np.argmax(corr))
            k_star = max_idx - (n - 1)
            i_cands = [i for i in ones if 0 <= i + k_star < n and h[i + k_star] == 0]
            j_cands = [j for j in zeros if 0 <= j + k_star < n and h[j + k_star] == 1]
            if len(i_cands) > 0 and len(j_cands) > 0:
                i_used = i_cands[rng.randint(len(i_cands))]
                j_used = j_cands[rng.randint(len(j_cands))]
        elif r < 0.7:

            i_used = ones[rng.randint(len(ones))]
            j_used = zeros[rng.randint(len(zeros))]
        elif r < 0.85:
            # Block swap of length 1 (same as random pair but different selection)
            i_used = ones[rng.randint(len(ones))]
            j_used = zeros[rng.randint(len(zeros))]
        else:

            blen = rng.randint(1, 5)
            if len(ones) >= blen and len(zeros) >= blen:

                j_start = zeros[rng.randint(len(zeros))]

                valid_i = True
                valid_j = True
                for b in range(blen):

                        valid_i = False
                        break
                    if j_start + b >= n or h[j_start + b] != 0:
                        valid_j = False
                        break
                if valid_i and valid_j:

                    old_corr = corr.copy()
                    old_h = h.copy()
                    for b in range(blen):
                        apply_pair_flip(corr, h, i_start + b, j_start + b)
                    new_c5 = float(corr.max()) * (2.0 / n)

                        corr[:] = old_corr
                        i_used = -1
                        j_used = -1
            if not accepted and i_used == -1:
                i_used = ones[rng.randint(len(ones))]
                j_used = zeros[rng.randint(len(zeros))]
        
        if i_used >= 0 and j_used >= 0 and not accepted:
            # Apply the flip
            apply_pair_flip(corr, h, i_used, j_used)
            new_c5 = float(corr.max()) * (2.0 / n)
            delta = new_c5 - current_c5

            else:

                apply_pair_flip_undo(corr, h, i_used, j_used)
        
        if accepted:

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
