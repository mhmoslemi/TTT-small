import numpy as np
from scipy.optimize import minimize
import time


def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def compute_smooth_and_grad(h, alpha, m_active):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    c = corr * dx
    active = np.argsort(c)[::-1][:m_active]
    c_act = c[active]
    c_max = float(np.max(c_act))
    shifted = (c_act - c_max) * alpha
    shifted = np.clip(shifted, -50.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, active):
        shift = int(k) - (n - 1)
        if shift >= 0:
            if n - shift > 0:
                grad[:n - shift] += w * dx * (1.0 - h[shift:n])
                grad[shift:] -= w * dx * h[:n - shift]
        else:
            s = -shift
            if s < n:
                grad[s:] += w * dx * (1.0 - h[:n - s])
                grad[:n - s] -= w * dx * h[s:]
    return lse, grad


def worst_lag_grad(h, n, dx):
    corr = compute_corr_full(h)
    k_star = int(np.argmax(corr))
    shift = k_star - (n - 1)
    grad = np.zeros(n)
    if shift >= 0:
        if n - shift > 0:
            grad[:n - shift] += dx * (1.0 - h[shift:n])
            grad[shift:] -= dx * h[:n - shift]
    else:
        s = -shift
        if s < n:
            grad[s:] += dx * (1.0 - h[:n - s])
            grad[:n - s] -= dx * h[s:]
    return grad


def active_set_opt(h, target, deadline, alphas, m_active, n_steps, lr_init):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        lr = lr_init
        stall = 0
        for _ in range(n_steps):
            if time.time() > deadline:
                return best_h, best_c5
            _, grad = compute_smooth_and_grad(h_cur, float(alpha), m_active)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-13:
                break
            improved = False
            for _ in range(25):
                h_try = h_cur - lr * grad
                h_try = project_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h_cur = h_try
                    best_c5 = c5_try
                    best_h = h_cur.copy()
                    improved = True
                    lr = min(lr * 1.04, lr_init * 8.0)
                    stall = 0
                    break
                lr *= 0.5
            if not improved:
                stall += 1
                if stall > 40:
                    break
    return best_h, best_c5


def worst_lag_surgery(h, target, deadline, max_steps, eps_init):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    eps = eps_init
    stall = 0

        if time.time() > deadline:

        improved = False

def slsqp_smooth(h, n, target, alpha, maxiter, deadline):
    def obj(x):
        corr = compute_corr_full(x)
        c_max = float(np.max(corr))
        shifted = (corr - c_max) * alpha
        shifted = np.clip(shifted, -50.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    def obj_grad(x):
        n_inner = len(x)
        dx_inner = 2.0 / n_inner
        corr = compute_corr_full(x)
        c = corr * dx_inner
        c_max = float(np.max(c))
        shifted = (c - c_max) * alpha
        shifted = np.clip(shifted, -50.0, 0.0)
        weights = np.exp(shifted)
        wsum = float(np.sum(weights))
        if wsum < 1e-30:
            return np.zeros(n_inner)
        weights = weights / wsum
        gdx = np.zeros(n_inner)
        for w, k in zip(weights, range(len(corr))):
            if w < 1e-10:
                continue
            shift = int(k) - (n_inner - 1)
            if shift >= 0:
                if n_inner - shift > 0:
                    gdx[:n_inner - shift] += w * dx_inner * (1.0 - x[shift:n_inner])
                    gdx[shift:] -= w * dx_inner * x[:n_inner - shift]
            else:
                s = -shift
                if s < n_inner:
                    gdx[s:] += w * dx_inner * (1.0 - x[:n_inner - s])
                    gdx[:n_inner - s] -= w * dx_inner * x[s:]
        return gdx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return h


def multi_start_deopt(h_init, n, target, deadline, rng, pop_size=10, n_gen=100):
    dx = 2.0 / n
    pop = np.zeros((pop_size, n))
    pop[0] = h_init.copy()
    for i in range(1, pop_size):
        if i % 3 == 0:
            pop[i] = project_feasible(h_init + rng.randn(n) * 0.05, target)
        elif i % 3 == 1:
            pop[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
        else:
            pop[i] = project_feasible(np.clip(h_init + rng.randn(n) * 0.15, 0.05, 0.95), target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    f = 0.7
    cr = 0.6
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score
