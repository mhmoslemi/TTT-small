        if c_curr < best_score:

        # Get top n_active lags

        weights /= weights.sum()

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lse_obj_and_grad(h, alpha):
    n = len(h)
    fc = np.correlate(h, 1.0 - h, mode='full')
    n_fc = len(fc)
    c_max = float(np.max(fc))
    shifted = alpha * (fc - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    vals = np.exp(shifted)
    total = float(np.sum(vals))
    weights = vals / total
    log_sum = c_max + np.log(total) / alpha
    grad = np.zeros(n)
    for i in range(n_fc):
        w = weights[i]
        if w < 1e-12:
            continue
        offset = i + 1 - n
        lo1 = max(0, -offset)
        hi1 = min(n - 1, n - 1 - offset)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * (1.0 - h[lo1+offset:hi1+1+offset])
        lo2 = max(0, offset)
        hi2 = min(n - 1, n - 1 + offset)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * h[lo2-offset:hi2+1-offset]
    return log_sum, grad


def gd_momentum(h0, n, target, alpha, n_steps, lr, momentum, deadline):
    h_cur = h0.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_val = lse_obj_and_grad(h_cur, alpha)[0]
    for step in range(n_steps):
        if time.time() > deadline:
            break
        val, grad = lse_obj_and_grad(h_cur, alpha)
        v = momentum * v - lr * grad
        h_new = project_feasible(h_cur + v, target)
        new_val = lse_obj_and_grad(h_new, alpha)[0]
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
        h_cur = h_new
        if step % 200 == 199:
            lr = max(lr * 0.97, 1e-7)
    return best_h


def _cons_fun(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_grad(h0, n, target, alpha, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    _g_alpha = alpha

    def obj(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    bounds = [(0.0, 1.0)] * n

SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h0


def subgradient_polish(h0, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h0.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        fc = compute_corr(h_cur)
        c_curr = float(np.max(fc)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(fc)[::-1][:n_active]
        diffs = fc[top_idx] - fc[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            offset = k + 1 - n
            lo1 = max(0, -offset)
            hi1 = min(n - 1, n - 1 - offset)
            if lo1 <= hi1:
                grad[lo1:hi1+1] += wk * (1.0 - h_cur[lo1+offset:hi1+1+offset])
            lo2 = max(0, offset)
            hi2 = min(n - 1, n - 1 + offset)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= wk * h_cur[lo2-offset:hi2+1-offset]
        grad *= dx
        h_new = project_feasible(h_cur - lr * grad, target)
        sc_new = compute_c5(h_new)

            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)

            lr *= lr_decay

    return best_h, best_score

    x50 = np.linspace(0, 2, n1)
    for i in range(19):
        r = i % 6
        if r == 0:
            c = project_feasible(rng.rand(n1) * 0.6 + 0.2, t1)
        elif r == 1:
            freq = rng.randint(1, 6)
            amp = rng.uniform(0.1, 0.4)
            phase = rng.uniform(0, 2 * np.pi)
            c = project_feasible(0.5 + amp * np.cos(freq * np.pi * x50 + phase), t1)
        elif r == 2:
            c = project_feasible(h1_base + rng.randn(n1) * 0.12, t1)
        elif r == 3:
            c = project_feasible(rng.choice([0.0, 1.0], n1).astype(float) * 0.7 + rng.rand(n1) * 0.3, t1)
        elif r == 4:
            c = project_feasible(0.5 + rng.randn(n1) * 0.25, t1)
        else:
            c = project_feasible(np.clip(0.5 + rng.randn(n1) * 0.5, 0.02, 0.98), t1)
        candidates_50.append(c)

    best_50 = h1_base.copy()
    best_50_sc = compute_c5(h1_base)
    d1 = start_time + 55
    for c in candidates_50:
        if time.time() > d1:
            break
        h_opt = gd_momentum(c, n1, t1, alpha=30, n_steps=3000, lr=0.025, momentum=0.9, deadline=d1)
        sc = compute_c5(h_opt)
        if sc < best_50_sc:
            best_50_sc = sc
            best_50 = h_opt

    if time.time() < start_time + 85:
        h_pol = slsqp_grad(best_50, n1, t1, alpha=300, maxiter=5000, ftol=1e-15, deadline=start_time + 85)
        sc = compute_c5(h_pol)
        if sc < best_50_sc:
            best_50_sc = sc
            best_50 = h_pol

    h100_best = h_parent.copy()
    h100_best_sc = compute_c5(interpolate_to(h_parent, 100, 50.0))
    if time.time() < start_time + 220:
        n2 = 100
        t2 = n2 / 2.0
        h2_a = interpolate_to(best_50, n2, t2)
        h2_b = interpolate_to(h_parent, n2, t2)
        h2_gd1 = gd_momentum(h2_a, n2, t2, alpha=50, n_steps=2000, lr=0.012, momentum=0.9, deadline=start_time + 150)
        h2_gd2 = gd_momentum(h2_b, n2, t2, alpha=50, n_steps=1500, lr=0.012, momentum=0.9, deadline=start_time + 190)
        h2_s1 = slsqp_grad(h2_gd1, n2, t2, alpha=300, maxiter=4000, ftol=1e-15, deadline=start_time + 210)
        h2_s2 = slsqp_grad(h2_gd2, n2, t2, alpha=300, maxiter=3000, ftol=1e-15, deadline=start_time + 220)
        for hc in [h2_a, h2_gd1, h2_gd2, h2_s1, h2_s2]:
            sc = compute_c5(hc)
            if sc < h100_best_sc:
                h100_best_sc = sc
                h100_best = hc

    if time.time() < start_time + 500:
        n3 = 200
        t3 = n3 / 2.0
        h3_a = interpolate_to(h100_best, n3, t3)
        h3_b = interpolate_to(h_parent, n3, t3)
        h3_gd1 = gd_momentum(h3_a, n3, t3, alpha=50, n_steps=1200, lr=0.008, momentum=0.9, deadline=start_time + 350)
        h3_gd2 = gd_momentum(h3_b, n3, t3, alpha=50, n_steps=1000, lr=0.008, momentum=0.9, deadline=start_time + 420)
        h3_s1 = slsqp_grad(h3_gd1, n3, t3, alpha=400, maxiter=3000, ftol=1e-15, deadline=start_time + 460)
        h3_s2 = slsqp_grad(h3_gd2, n3, t3, alpha=400, maxiter=2500, ftol=1e-15, deadline=start_time + 500)
        best_200 = h3_a
        best_200_sc = compute_c5(h3_a)
        for hc in [h3_gd1, h3_gd2, h3_s1, h3_s2]:
            sc = compute_c5(hc)
            if sc < best_200_sc:
                best_200_sc = sc
                best_200 = hc
        h3_up = interpolate_to(best_200, n_final, target_final)
        sc_up = compute_c5(h3_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h3_up

    h4_candidates = [best_h.copy()]
    if time.time() < start_time + 400:
        h4_candidates.append(interpolate_to(h100_best, n_final, target_final))

    d4 = start_time + 750
    for h_cand in h4_candidates:
        if time.time() > d4:
            break
        h_s1 = slsqp_grad(h_cand, n_final, target_final, alpha=500, maxiter=4000, ftol=1e-16, deadline=d4)
        sc = compute_c5(h_s1)
        if sc < best_score:
            best_score = sc
            best_h = h_s1
        h_s2 = slsqp_grad(h_s1, n_final, target_final, alpha=2000, maxiter=3000, ftol=1e-16, deadline=d4)
        sc2 = compute_c5(h_s2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_s2

    if time.time() < start_time + 900:
        h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 600, 5, 0.004, 0.996, start_time + 900)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_s3 = slsqp_grad(best_h, n_final, target_final, alpha=5000, maxiter=2500, ftol=1e-16, deadline=start_time + 950)
        sc = compute_c5(h_s3)
        if sc < best_score:
            best_score = sc
            best_h = h_s3

    if time.time() < start_time + 1080:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 400, 0.004, start_time + 1080)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_s4 = slsqp_grad(best_h, n_final, target_final, alpha=10000, maxiter=2000, ftol=1e-16, deadline=start_time + 1120)
        sc = compute_c5(h_s4)
        if sc < best_score:
            best_score = sc
            best_h = h_s4

    if time.time() < start_time + 1200:
        for bs in [4, 6]:
            if time.time() > start_time + 1200:
                break
            h_bd = block_descent(best_h, n_final, target_final, alpha=3000, block_size=bs, n_rounds=3, deadline=start_time + 1200)
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    if time.time() < hard_deadline - 20:
        h_gd = gd_momentum(best_h, n_final, target_final, alpha=500, n_steps=800, lr=0.003, momentum=0.9, deadline=hard_deadline - 20)
        sc = compute_c5(h_gd)
        if sc < best_score:
            best_score = sc
            best_h = h_gd

    if time.time() < hard_deadline - 10:
        h_s5 = slsqp_grad(best_h, n_final, target_final, alpha=30000, maxiter=1500, ftol=1e-17, deadline=hard_deadline - 10)
        sc = compute_c5(h_s5)
        if sc < best_score:
            best_score = sc
            best_h = h_s5

    if time.time() < hard_deadline - 3:
        h_sg2, sc_sg2 = subgradient_polish(best_h, n_final, target_final, 300, 4, 0.002, 0.997, hard_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
