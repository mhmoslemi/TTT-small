    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e8)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def top_lag_gradient_step(h, target, n_top, lr, deadline, n_steps):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.85
            if lr < 1e-9:
                break
    return h_cur, cur_score

def upsample_function(h_old, n_old, n_new, target_new):
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_new, target_new)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 30.0
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 100.0

    # Phase 1: L-BFGS-B warm start at n
    alphas_p1 = [5, 20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if time.time() > deadline - 900:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 500, deadline - 800)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Coordinate descent at n (heavy)
    n_cd_sweeps = 800
    h_cd, sc_cd = coord_descend(best_h, target, n_cd_sweeps, deadline - 700)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 3: L-BFGS polish at n
    for a in [10000, 50000, 200000, 1000000]:
        if time.time() > deadline - 600:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 400, deadline - 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: More coordinate descent
    h_cd, sc_cd = coord_descend(best_h, target, 500, deadline - 450)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 5: Top-lag equalization
    if time.time() < deadline - 300:
        h_eq, sc_eq = top_lag_gradient_step(best_h, target, n_top=5, lr=0.01,
                                            deadline=deadline - 250, n_steps=300)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_cd, sc_cd = coord_descend(best_h, target, 300, deadline - 200)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 6: Upsample to 2n
    n2 = 2 * n
    target2 = float(n2) / 2.0
    if time.time() < deadline - 200:
        h_fine = upsample_function(best_h, n, n2, target2)
        sc_fine = compute_c5(h_fine)
        # Coordinate descent at finer resolution
        h_fine, sc_fine = coord_descend(h_fine, target2, 400, deadline - 150)
        # L-BFGS at finer
        for a in [100, 1000, 10000, 100000]:
            if time.time() > deadline - 100:
                break
            h_fine = lbfgs_optimize(h_fine, target2, a, mu, 300, deadline - 80)
        sc_fine = compute_c5(h_fine)
        # More coordinate descent
        h_fine, sc_fine = coord_descend(h_fine, target2, 300, deadline - 60)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 7: Final coordinate descent
    h_cd, sc_cd = coord_descend(best_h, target, 400, deadline - 40)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 8: SLSQP polish
    if time.time() < deadline - 20:
        h_slq = slsqp_polish(best_h, target, 5000, 300)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 9: Final coordinate descent
    h_cd, sc_cd = coord_descend(best_h, target, 200, deadline)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
