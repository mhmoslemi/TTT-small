import numpy as np

import time

def project_feasible(h, target_sum):
    """Project h onto the feasible set: 0 <= h <= 1, sum(h) = target_sum"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not mask.any():
            break
        h = h.copy()
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_objective(x, alpha, target, mu):
    """Log-sum-exp smoothed max of correlations + constraint penalty"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

def lse_grad(x, alpha, target, mu):
    """Gradient of the log-sum-exp smoothed objective"""
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()

    else:

    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]

def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return lse_objective(x, alpha, target, mu)
    def g(x):
        return lse_grad(x, alpha, target, mu)
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def block_coordinate_descent(h, target, n, block_size, n_sweeps, alpha, deadline):
    """Optimize blocks of adjacent coordinates using L-BFGS-B on the block"""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            blk = h_cur[start:end].copy()
            # Fix all other coordinates, optimize the block
            # The objective restricted to the block is still a max of quadratics
            # Use the log-sum-exp trick
            fixed = h_cur.copy()
            def block_f(blk_x):
                test = fixed.copy()
                test[start:end] = blk_x

                c_max = float(np.max(corr))
                shifted = alpha * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
            def block_g(blk_x):
                test = fixed.copy()
                test[start:end] = blk_x
                corr = np.correlate(test, 1.0 - test, mode='full')
                c_max = float(np.max(corr))
                shifted = alpha * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                weights = np.exp(shifted)
                wsum = weights.sum()
                if wsum > 1e-300:
                    weights = weights / wsum
                else:
                    weights = np.ones(len(corr)) / len(corr)
                grad_full = np.zeros(n)
                active = np.where(weights > 1e-10)[0]
                for ki in active:
                    w = weights[ki]
                    lag = int(ki) - (n - 1)
                    if lag >= 0:
                        j1 = np.arange(lag, n)

def adaptive_subgradient(h, target, n_steps, deadline):
    """Adaptive subgradient method with step size decay"""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    lr = 0.01
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        # Find lags within 2% of max
        threshold = c_max * 0.98
        top_idx = np.where(corr >= threshold)[0]
        # Subgradient: average gradient over top lags
        grad = np.zeros(n)
        n_top = len(top_idx)
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= n_top
        # Step
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
        # Decay
        lr *= 0.999
    return best_h, best_score

            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_step(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f_cb(x):
        return lse_obj_grad(x, alpha, target, mu)[0]

    def g_cb(x):
        return lse_obj_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f_cb, h, jac=g_cb, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def subgradient_polish(h, target, n_steps, deadline, k_frac=0.01):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.008
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        threshold = c_max * (1.0 - k_frac)
        top_idx = np.where(corr >= threshold)[0]
        if len(top_idx) == 0:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= len(top_idx)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if sc < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.75
        lr *= 0.998
        if lr < 1e-7:
            break
    return best_h, best_score


def upsample(h, n_target):
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n)
    x_new = np.linspace(0.0, 2.0, n_target)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_target / 2.0)


def slsqp_polish(h, target, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f_cb(x):
        return lse_obj_grad(x, 10000, target, 1e5)[0]

    def g_cb(x):
        return lse_obj_grad(x, 10000, target, 1e5)[1]

    def eq_cb(x):
        return np.sum(x) - target

    def eq_jac_cb(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq_cb, 'jac': eq_jac_cb}]
    try:
        res = minimize(f_cb, h, jac=g_cb, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s

    def tleft():
        return deadline - time.time() - 25.0

    n = len(initial_h_values)
    target = n / 2.0
    h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    alpha_schedule_1 = [10, 50, 200, 1000, 5000, 20000, 100000]
    for alpha in alpha_schedule_1:
        if tleft() < 70:
            break
        h_new = lbfgs_step(best_h, target, alpha, 50.0, 600)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_sg, sc_sg = subgradient_polish(best_h, target, 200, deadline, k_frac=0.01)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg

    if tleft() > 50:
        h_new = lbfgs_step(best_h, target, 50000, 100.0, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        h_fine = upsample(best_h, n2)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2
            h_fine = best_h

        alpha_schedule_2 = [20, 100, 500, 2000, 10000, 50000, 200000]
        for alpha in alpha_schedule_2:
            if tleft() < 200:
                break
            h_fine = lbfgs_step(h_fine, target2, alpha, 50.0, 500)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = h_fine

        h_sg2, sc_sg2 = subgradient_polish(best_h, target, 300, deadline, k_frac=0.005)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

        if tleft() > 100:
            h_new = lbfgs_step(best_h, target, 100000, 100.0, 500)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    n3 = 4 * len(initial_h_values)
    target3 = n3 / 2.0
    if tleft() > 300 and n3 > len(best_h):
        h_fine3 = upsample(best_h, n3)
        sc_fine3 = compute_c5(h_fine3)
        if sc_fine3 < best_score:
            best_score = sc_fine3
            best_h = h_fine3
            n = n3
            target = target3

        for alpha in [50, 500, 5000, 50000]:
            if tleft() < 100:
                break
            h_fine3 = lbfgs_step(h_fine3, target3, alpha, 50.0, 400)
            sc = compute_c5(h_fine3)
            if sc < best_score:
                best_score = sc
                best_h = h_fine3

        h_sg3, sc_sg3 = subgradient_polish(best_h, target, 200, deadline, k_frac=0.005)
        if sc_sg3 < best_score:
            best_score = sc_sg3
            best_h = h_sg3

    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    restart_scales = [0.01, 0.02, 0.04, 0.08, 0.015, 0.03, 0.05, 0.10]
    for i, scale in enumerate(restart_scales):
        if tleft() < 70:
            break
        h_pert = best_h + rng.randn(n_cur) * scale
        h_pert = project_feasible(h_pert, target_cur)
        h_new = lbfgs_step(h_pert, target_cur, 200, 50.0, 200)
        h_new = lbfgs_step(h_new, target_cur, 5000, 50.0, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    for alpha in [20000, 100000, 500000]:
        if tleft() < 50:
            break
        h_new = lbfgs_step(best_h, target_cur, alpha, 100.0, 400)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_sg_final, sc_sg_final = subgradient_polish(best_h, target_cur, 200, deadline, k_frac=0.003)
    if sc_sg_final < best_score:
        best_score = sc_sg_final
        best_h = h_sg_final

    if tleft() > 50:
        h_new = lbfgs_step(best_h, target_cur, 200000, 200.0, 500)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 40:
        h_slq = slsqp_polish(best_h, target_cur, 300)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, len(best_h)
