def project_feasible(h, target_sum):

    if abs(diff) < 1e-14:
        return h

        idx = np.where(h > 1e-10)[0]

    h = np.clip(h, 0.0, 1.0)

            h = np.clip(h, 0.0, 1.0)

    h = h_init.copy()
    h = project_feasible(h, target)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    phase = 0
    fracs = [0.95, 0.92, 0.88, 0.85, 0.82, 0.80]
    pens = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    while time.time() - start < time_limit and phase < len(fracs):
        tc = pens[phase]
        tf = fracs[phase]
        this_limit = min(120, time_limit - (time.time() - start))
        if this_limit < 30:
            break
        h_new, c5_new = active_lp(h, n, target, this_limit,
                                   max_active=max_active, max_iter=300,
                                   start_penalty=tc, decay=0.96,
                                   top_frac=tf)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        phase += 1

    return best_h, best_c5


def targeted_local_search(h_init, n, target, time_limit, n_top=30):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    idxs = np.arange(n)

    for step in range(100000):
        if time.time() - start > time_limit:
            break

        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        max_ov = float(overlaps[top_idx[0]])

        best_move = None
        best_move_val = best_c5

        for k in top_lags:
            if k >= 0:
                valid_i = idxs[:n - k]
            else:
                j = -k
                valid_i = idxs[j:]

            for i_pos in range(0, len(valid_i), max(1, len(valid_i) // 50)):
                i = valid_i[i_pos]
                if k >= 0:
                    j_idx = i + k
                    if h[i] <= 1e-8 or h[j_idx] >= 1.0 - 1e-8:
                        continue
                    for delta in [0.01, 0.005, 0.02, 0.001]:
                        h_try = h.copy()
                        h_try[i] = min(1.0, h[i] + delta)
                        h_try[j_idx] = max(0.0, h[j_idx] - delta)
                        h_try = project_feasible(h_try, target)
                        v = compute_c5(h_try)
                        if v < best_move_val:
                            best_move_val = v
                            best_move = h_try.copy()
                else:
                    j_idx = i - k
                    if h[i] <= 1e-8 or h[j_idx] >= 1.0 - 1e-8:
                        continue
                    for delta in [0.01, 0.005, 0.02, 0.001]:
                        h_try = h.copy()
                        h_try[i] = min(1.0, h[i] + delta)
                        h_try[j_idx] = max(0.0, h[j_idx] - delta)
                        h_try = project_feasible(h_try, target)
                        v = compute_c5(h_try)
                        if v < best_move_val:
                            best_move_val = v
                            best_move = h_try.copy()

            if k < 0:
                j_abs = -k
                valid_i2 = idxs[:n - j_abs]
                for i_pos in range(0, len(valid_i2), max(1, len(valid_i2) // 50)):
                    i2 = valid_i2[i_pos]
                    j_idx2 = i2 + j_abs
                    if h[j_idx2] <= 1e-8 or h[i2] >= 1.0 - 1e-8:
                        continue
                    for delta in [0.01, 0.005, 0.02, 0.001]:
                        h_try = h.copy()
                        h_try[j_idx2] = min(1.0, h[j_idx2] + delta)
                        h_try[i2] = max(0.0, h[i2] - delta)
                        h_try = project_feasible(h_try, target)
                        v = compute_c5(h_try)
                        if v < best_move_val:
                            best_move_val = v
                            best_move = h_try.copy()

        if best_move is not None:
            h = best_move
            if best_move_val < best_c5:
                best_c5 = best_move_val
                best_h = h.copy()
        else:
            if step > 100:
                break

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def momentum_subgradient(h_init, n, target, max_steps, time_limit,
                         n_top=20, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    n_half = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        overlaps = compute_overlaps(y)
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        top_lags = top_idx - n_half
        top_vals = overlaps[top_idx]

        w = 0.85 ** np.arange(n_top)
        w = w / w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_lags):
            gk = grad_overlap_vec(y, n, k)
            grad += w[i] * gk

        grad *= dx

        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break

        c5_y = float(np.max(overlaps))
        polyak_lr = (c5_y - best_c5) / (norm_grad ** 2 + 1e-20)
        lr = min(max(0.01, polyak_lr), 0.5)

        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_prev = h.copy()
            h = h_try
        else:
            h_prev = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Active LP with moderate constraints
    if tleft() > 150:
        lp_time = min(100, tleft() * 0.3)
        h_lp, c5_lp = active_lp(best_h, n, target, lp_time,
                                 max_active=200, max_iter=400,
                                 start_penalty=0.05, decay=0.97,
                                 top_frac=0.93)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Multi-start LP (progressive relaxation)
    if tleft() > 200:
        ms_time = min(200, tleft() * 0.35)
        h_ms, c5_ms = multi_start_lp(best_h, n, target, ms_time,
                                      max_active=250, top_frac=0.90)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 3: Momentum subgradient
    if tleft() > 120:
        sg_time = min(100, tleft() * 0.25)
        h_sg, c5_sg = momentum_subgradient(best_h, n, target,
                                            max_steps=50000, time_limit=sg_time,
                                            n_top=25, momentum=0.9)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: Targeted local search
    if tleft() > 100:
        ls_time = min(80, tleft() * 0.25)
        h_ls, c5_ls = targeted_local_search(best_h, n, target, ls_time, n_top=25)
        if c5_ls < best_score:
            best_score = c5_ls
            best_h = h_ls

    # Phase 5: SLSQP polish
    if tleft() > 120:
        h_ref = slsqp_optimize(best_h, n, target,
                                [50, 200, 1000, 5000, 20000],
                                maxiter=2000, ftol=1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: LP after SLSQP
    if tleft() > 120:
        lp_time = min(100, tleft() * 0.3)
        h_lp2, c5_lp2 = active_lp(best_h, n, target, lp_time,
                                   max_active=200, max_iter=400,
                                   start_penalty=0.02, decay=0.96,
                                   top_frac=0.90)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 7: Try different starting points
    if tleft() > 200:
        # Binary structure
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        h_bin = project_feasible(h_bin, target)
        sc = compute_c5(h_bin)
        h_candidate = h_bin

        # Smoothed structure
        from scipy.ndimage import gaussian_filter
        h_smooth = gaussian_filter(best_h, sigma=3.0)
        h_smooth = project_feasible(h_smooth, target)
        sc_smooth = compute_c5(h_smooth)

        # Perturbed structure
        h_pert = best_h + rng.randn(n) * 0.05
        h_pert = project_feasible(h_pert, target)
        sc_pert = compute_c5(h_pert)

        candidates = [(h_bin, sc), (h_smooth, sc_smooth), (h_pert, sc_pert)]
        candidates.sort(key=lambda x: x[1])

        for h_cand, sc_cand in candidates[:2]:
            if tleft() < 80:
                break
            h_opt, c5_opt = active_lp(h_cand, n, target, 60,
                                       max_active=200, max_iter=300,
                                       start_penalty=0.1, decay=0.97,
                                       top_frac=0.90)
            if c5_opt < best_score:
                best_score = c5_opt
                best_h = h_opt

    # Phase 8: Final local search
    if tleft() > 60:
        ls_time = min(50, tleft() * 0.5)
        h_ls2, c5_ls2 = targeted_local_search(best_h, n, target, ls_time, n_top=30)
        if c5_ls2 < best_score:
            best_score = c5_ls2
            best_h = h_ls2

    # Phase 9: Final LP polish
    if tleft() > 50:
        lp_time = min(40, tleft() * 0.7)
        h_lp3, c5_lp3 = active_lp(best_h, n, target, lp_time,
                                   max_active=150, max_iter=300,
                                   start_penalty=0.01, decay=0.95,
                                   top_frac=0.85)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Phase 10: Final SLSQP
    if tleft() > 40:
        h_final_try = slsqp_optimize(best_h, n, target,
                                      [200, 1000, 5000, 20000],
                                      maxiter=1000, ftol=1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_use
