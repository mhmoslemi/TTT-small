```python
def outer_approximation_lp(h_curr, n, target, all_lags=False):
    """Solve the LP approximation of the problem."""
    dx = 2.0 / n
    h_t = h_curr  # current point
    idxs = np.arange(n)
    
    # Determine which lags to include
    if all_lags:
        lags = list(range(-(n-1), n))
    else:
        corr = np.correlate(h_t, 1.0 - h_t, mode='full')
        # Take top lags
        top_idx = np.argsort(corr)[-50:]
        lags = [int(i) - (n-1) for i in top_idx]
    
    var_h = cp.Variable(n)
    var_t = cp.Variable()
    
    constraints = [
        cp.sum(var_h) == target,
        var_h >= 0,
        var_h <= 1,
        var_t >= 0
    ]
    
    for k in lags:
        # Valid i: max(0, -k) <= i < min(n, n-k)
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        valid_i = np.arange(i_lo, i_hi)
        valid_j = valid_i + k  # these are the i+k values
        
        # Linearized constraint:
        # sum_i [(1 - h_j^(t)) * h_i - h_i^(t) * h_j] <= t/dx - C_k
        # where C_k = sum_i h_i^(t) * h_j^(t)
        
        lhs_coeff = np.zeros(n)
        rhs_const = 0.0
        for i in valid_i:
            j = i + k
            lhs_coeff[i] += (1.0 - h_t[j])  # coefficient of h_i
            lhs_coeff[j] += (-h_t[i])       # coefficient of h_j
            rhs_const += h_t[i] * h_t[j]
        
        constraints.append(cp.sum(lhs_coeff * var_h) <= var_t / dx - rhs_const)
    
    objective = cp.Minimize(var_t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS)  # or CBC, GLPK
    except:
        try:
            prob.solve(solver=cp.CBC)
        except:
            try:
                prob.solve(solver=cp.GLPK)
            except:
                return h_curr, float('inf')
    
    if var_h.value is not None:
        h_new = np.clip(var_h.value, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        return h_new, compute_c5(h_new)
    else:
        return h_curr, compute_c5(h_curr)
```

```python
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

    return np.ones(_global_n)

            if compute_c5(h_new) < compute_c5(h_cur):
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    start = time.time()
    for alpha in alphas:
        if time_limit is not None and time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def solve_oa_lp(h_curr, n, target, n_top_lags, solver_list=None):
    dx = 2.0 / n
    h_t = h_curr.copy()
    corr = np.correlate(h_t, 1.0 - h_t, mode='full')
    mid = n - 1
    total_lags = 2 * n - 1
    m = min(n_top_lags, total_lags)
    lag_indices = np.argsort(corr)[-m:]

    var_h = cp.Variable(n)
    var_t = cp.Variable()

    constraints = [
        cp.sum(var_h) == target,
        var_h >= 0,
        var_h <= 1,
        var_t >= 0
    ]

    for k_idx in lag_indices:
        k = int(k_idx) - mid
        i_lo = max(0, -k)
        i_hi = min(n, n - k)
        if i_hi <= i_lo:
            continue
        valid_i = np.arange(i_lo, i_hi)
        valid_j = valid_i + k
        coeff = np.zeros(n)
        coeff[valid_i] += (1.0 - h_t[valid_j])
        coeff[valid_j] += (-h_t[valid_i])
        C_k = float(np.sum(h_t[valid_i] * h_t[valid_j]))
        constraints.append(cp.sum(coeff * var_h) <= var_t / dx - C_k)

    # k=0 constraint: sum h_i(1-h_i) <= t/dx
    # Linearized: sum (1-2*h_i^t)*h_i + sum h_i^t^2 <= t/dx
    lhs0 = 1.0 - 2.0 * h_t
    C0 = float(np.sum(h_t * h_t))
    constraints.append(cp.sum(lhs0 * var_h) <= var_t / dx - C0)

    prob = cp.Problem(cp.Minimize(var_t), constraints)

    if solver_list is None:
        solver_list = [cp.ECOS, cp.CBC, cp.GLOP, cp.GLPK]

    solved = False
    for solver in solver_list:
        try:
            prob.solve(solver=solver, verbose=False)
            if var_h.value is not None and var_h.value.size == n:
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return h_curr.copy(), compute_c5(h_curr), False

    h_lp = np.clip(var_h.value, 0.0, 1.0)
    h_lp = project_feasible(h_lp, target)
    return h_lp, compute_c5(h_lp), True


def oa_line_search(h_curr, h_lp, n, target, alphas=None):
    if alphas is None:
        alphas = [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9, 1.0]
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    for a in alphas:
        h_try = h_curr + a * (h_lp - h_curr)
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def oa_loop(h_start, n, target, time_limit, max_iters=500):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    stagnant = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        n_lags = min(50 + it * 2, 2 * n - 1)
        h_lp, lp_c5, solved = solve_oa_lp(h_cur, n, target, n_lags)

        if not solved:
            stagnant += 1
            if stagnant > 20:
                break
            continue

        h_new, c5_new = oa_line_search(h_cur, h_lp, n, target)

        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            stagnant = 0
        else:
            h_cur = h_lp
            stagnant += 1
            if stagnant > 40:
                break
        if it % 30 == 29:
            h_slq = slsqp_round(h_cur, n, target,
                                [500, 2000, 10000], 800, 1e-14,
                                time_limit=min(15, time_limit - (time.time() - start)))
            sc = compute_c5(h_slq)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_slq.copy()
                h_cur = h_slq
                stagnant = 0

    return best_h, best_c5


def fast_pair_cd(h_init, n, target, time_limit, max_sweeps=3000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if consecutive_no_improve > 150:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j + shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def momentum_subgradient(h, n, target, time_limit, max_steps, mu=0.85, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_perturbation_search(h_init, n, target, max_tries, time_limit):
    start = time.time()
    rng_local = np.random.RandomState(77777)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8, 13]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.03, 0.05, 0.1, 0.15, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP quick polish
    if tleft() > 200:
        h_pol = slsqp_round(best_h, n, target,
                            [200, 1000, 5000], 1200, 1e-14,
                            time_limit=min(80, tleft() * 0.25))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Outer Approximation LP loop (main new component)
    if tleft() > 300:
        oa_time = min(400, tleft() * 0.35)
        h_oa, c5_oa = oa_loop(best_h, n, target, time_limit=oa_time, max_iters=400)
        if c5_oa < best_score:
            best_score = c5_oa
            best_h = h_oa

    # Phase 3: SLSQP polish after OA
    if tleft() > 150:
        h_pol = slsqp_round(best_h, n, target,
                            [1000, 5000, 20000], 1500, 1e-15,
                            time_limit=min(80, tleft() * 0.25))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Pair CD
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.25)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=5000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: OA again (different lags now)
    if tleft() > 200:
        oa_time = min(200, tleft() * 0.25)
        h_oa2, c5_oa2 = oa_loop(best_h, n, target, time_limit=oa_time, max_iters=300)
        if c5_oa2 < best_score:
            best_score = c5_oa2
            best_h = h_oa2

    # Phase 6: SLSQP + Pair CD
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000], 1200, 1e-15,
                            time_limit=min(60, tleft() * 0.3))
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    if tleft() > 100:
        cd_time = min(80, tleft() * 0.3)
        h_cd, c5_cd = fast_pair_cd(best_h, n, target,
                                   time_limit=cd_time,
                                   max_sweeps=3000, n_cand=60)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Momentum subgradient
    if tleft() > 80:
        h_mom, c5_mom = momentum_subgradient(best_h, n, target,
                                             time_limit=min(40, tleft() * 0.3),
                                             max_steps=20000, mu=0.85, base_lr=0.015)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 8: Multi-restart with OA
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.08, 0.15]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_pert = slsqp_round(h_pert, n, target,
                             [200, 1000, 5000], 800, 1e-14,
                             time_limit=min(30, tleft() * 0.15))
        if tleft() < 80:
            sc = compute_c5(h_pert)
            if sc < best_score:
                best_score = sc
                best_h = h_pert
            continue
        h_oa3, c5_oa3 = oa_loop(h_pert, n, target, time_limit=min(40, tleft() * 0.2), max_iters=100)
        sc = compute_c5(h_oa3)
        if sc < best_score:
            best_score = sc
            best_h = h_oa3

    # Phase 9: Local perturbation + SLSQP
    if tleft() > 80:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                    max_tries=3000,
                                                    time_limit=min(40, tleft() * 0.3))
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if tleft() > 40:
            h_final_slq = slsqp_round(best_h, n, target,
                                      [5000, 20000, 80000], 1000, 1e-15,
                                      time_limit=min(40, tleft() * 0.5))
            sc = compute_c5(h_final_slq)
            if sc < best_score:
                best_score = sc
                best_h = h_final_slq

    # Phase 10: Try higher resolution
    if tleft() > 250:
        for n_try in [500, 600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hi = slsqp_round(h_interp, n_try, target_try,
                               [200, 1000, 5000, 20000], 1000, 1e-14,
                               time_limit=min(60, tleft() * 0.2))
            if tleft() > 80:
                h_hi2, sc2 = oa_loop(h_hi, n_try, target_try,
                                     time_limit=min(40, tleft() * 0.25), max_iters=100)
                if sc2 < compute_c5(h_hi):
                    h_hi = h_hi2
            sc = compute_c5(h_hi)
            if sc < best_score:
                best_score = sc
                best_h = h_hi
                n = n_try
                target = target_try

    # Phase 11: Final pair CD at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.4)
        h_cd, c5_cd = fast_pair_cd(best_h, n_use, target_use,
                                   time_limit=cd_time,
                                   max_sweeps=2000, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 100000], 800, 1e-15,
                              time_limit=max(20, tleft() * 0.6))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
