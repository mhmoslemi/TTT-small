def project_feasible(h, target_sum):

        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        n_blocks = n // block_size
        start_positions = np.arange(0, n - block_size + 1, block_size // 2)
        rng_local = np.random.RandomState(sweep % 10000)
        rng_local.shuffle(start_positions)

        improved_sweep = False
        for s in start_positions[:50]:
            if time.time() - start > time_limit:
                break
            block = slice(s, s + block_size)
            g_block = np.zeros(n_corr)
            for i in range(s, s + block_size):
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g_block[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g_block[v_im] += h[i - shifts[v_im]]
                v_jp = (i + shifts >= 0) & (i + shifts < n)
                g_block[v_jp] -= h[i + shifts[v_jp]]
                v_jm = (i - shifts >= 0) & (i - shifts < n)
                g_block[v_jm] -= h[i - shifts[v_jm]]

            avg_grad = np.mean(grad[s:s + block_size])
            if abs(avg_grad) < 1e-10:
                continue

            direction = -np.sign(avg_grad)
            h_try = h.copy()
            h_try[block] += direction * 0.05
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_val - 1e-14:
                h = h_try
                improved_sweep = True
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5_val = float(np.max(corr)) * dx
                break

    return best_h, best_c5


def upsample(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP at initial resolution
    if tleft() > 120:
        h_pol, sc = slsqp_round_multi(best_h, n_init, target_init,
                                      [100, 500, 2000, 10000], 800)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD at initial resolution
    if tleft() > 150:
        cd_time = min(120, tleft() * 0.2)
        h_cd, c5_cd = pair_cd(best_h, n_init, target_init,
                              time_limit=cd_time, n_cand=40, max_sweeps=3000)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Upsample to higher resolution and optimize
    for n_target in [600, 800]:
        if tleft() < 250:
            break
        h_up = upsample(best_h, n_target)
        target_up = n_target / 2.0
        h_opt, sc = slsqp_round_multi(h_up, n_target, target_up,
                                      [50, 200, 1000, 5000], 600)
        if tleft() > 150:
            h_cd2, c5_cd2 = pair_cd(h_opt, n_target, target_up,
                                    time_limit=min(80, tleft() * 0.2),
                                    n_cand=50, max_sweeps=3000)
            if c5_cd2 < sc:
                sc = c5_cd2
                h_opt = h_cd2
        if tleft() > 100:
            h_pol2, sc2 = slsqp_round_multi(h_opt, n_target, target_up,
                                            [500, 2000, 10000], 800)
            if sc2 < sc:
                sc = sc2
                h_opt = h_pol2
        if sc < best_score:
            best_score = sc
            best_h = h_opt

    # Phase 4: Spectral exploration at best resolution
    n_best = len(best_h)
    target_best = n_best / 2.0
    if tleft() > 100:
        h_spec, sc_spec = spectral_exploration(best_h, n_best, target_best,
                                               rng, time_limit=min(80, tleft() * 0.3),
                                               n_trials=100)
        if sc_spec < best_score:
            best_score = sc_spec
            best_h = h_spec

    # Phase 5: Block CD
    if tleft() > 100:
        h_blk, sc_blk = block_cd(best_h, n_best, target_best,
                                 time_limit=min(60, tleft() * 0.3),
                                 block_size=min(8, n_best // 50), max_sweeps=2000)
        if sc_blk < best_score:
            best_score = sc_blk
            best_h = h_blk

    # Phase 6: Another round of SLSQP + pair CD
    if tleft() > 150:
        h_pol3, sc3 = slsqp_round_multi(best_h, n_best, target_best,
                                        [200, 1000, 5000, 20000], 1000)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_pol3
        if tleft() > 80:
            h_cd3, c5_cd3 = pair_cd(best_h, n_best, target_best,
                                    time_limit=min(60, tleft() * 0.3),
                                    n_cand=60, max_sweeps=4000)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Phase 7: Try even higher resolution if time allows
    n_best = len(best_h)
    if tleft() > 300 and n_best < 1000:
        n_try = min(1000, n_best + 200)
        h_up2 = upsample(best_h, n_try)
        target_try = n_try / 2.0
        h_opt2, sc4 = slsqp_round_multi(h_up2, n_try, target_try,
                                        [100, 500, 2000, 8000], 600)
        if tleft() > 100:
            h_cd4, c5_cd4 = pair_cd(h_opt2, n_try, target_try,
                                    time_limit=min(60, tleft() * 0.25),
                                    n_cand=50, max_sweeps=2000)
            if c5_cd4 < sc4:
                sc4 = c5_cd4
                h_opt2 = h_cd4
        if sc4 < best_score:
            best_score = sc4
            best_h = h_opt2

    # Phase 8: Final SLSQP polish with high alpha
    n_final = len(best_h)
    target_final = n_final / 2.0
    if tleft() > 60:
        h_final_pol, sc_final = slsqp_round_multi(best_h, n_final, target_final,
                                                  [1000, 5000, 20000, 50000], 1000)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_pol

    # Phase 9: Final pair CD
    if tleft() > 40:
        h_cd_final, c5_final = pair_cd(best_h, n_final, target_final,
                                       time_limit=min(30, tleft() * 0.5),
                                       n_cand=50, max_sweeps=2000)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_cd_final

    # Final feasibility and evaluation
    n_out = len(best_h)
    target_out = n_out / 2.0
    best_h = project_feasible(best_h, target_out)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_out
