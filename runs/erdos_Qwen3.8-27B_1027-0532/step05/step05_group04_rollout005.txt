import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize, linprog


def compute_overlaps(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n - 1:]


def compute_c5(h):
    return float(np.max(compute_overlaps(h))) * (2.0 / len(h))


def grad_fk(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        for i in range(n):
            g[i] += (1.0 - 2.0 * h[i])
        return g
    g[:n - k] += (1.0 - h[k:])
    g[k:] -= h[:n - k]
    return g


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def slp_minmax(h, n, target, max_iter=300, active_tol=0.002, deadline=None,
               use_full_set=False, step_frac=1.0):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break

        over = compute_overlaps(h_cur)
        t_cur = float(np.max(over))

        if t_cur * dx < best_score:
            best_score = t_cur * dx
            best_h = h_cur.copy()

        if use_full_set:
            active = np.arange(n)
        else:
            thresh = t_cur - active_tol
            active = np.where(over >= thresh)[0]
            if len(active) < 3:
                active = np.argsort(over)[::-1][:5]

        cone = []
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        b_ub = []
        A_rows = []

        for k in active:
            k = int(k)
            g = grad_fk(h_cur, k)
            rhs = g @ h_cur - over[k]
            row = np.zeros(n + 1)
            row[:n] = g
            row[-1] = -1.0
            A_rows.append(row)
            b_ub.append(rhs)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])

        bounds = [(0.0, 1.0)] * n + [(None, None)]

        res = linprog(c_obj, A_ub=np.array(A_rows), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

        if not res.success:
            break

        h_new = res.x[:n]
        T_new = res.x[-1]

        if T_new > t_cur + 1e-10:
            h_new = h_cur * (1 - step_frac) + h_new * step_frac
        else:
            h_new = h_cur * (1 - step_frac) + h_new * step_frac

        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)

        if new_score < best_score - 1e-14:
            best_score = new_score
            best_h = h_new.copy()

        h_cur = h_new

        if t_cur - T_new < 1e-12:
            break

    return best_h, best_score


def subgradient_momentum(h, n, target, n_steps=2000, beta=300, lr=0.0005,
                         moment=0.9, n_active=None, deadline=None):
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr_cur = lr

    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break

        over = compute_overlaps(h_cur)
        max_over = float(np.max(over))
        c_curr = max_over * dx

        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()

        if n_active is not None and n_active < n:
            top_idx = np.argsort(over)[::-1][:n_active]
            diffs = over[top_idx] - over[top_idx[0]]
            weights = np.exp(beta * diffs)
            weights /= weights.sum()
            grad = np.zeros(n)
            for wk, k in zip(weights, top_idx):
                grad += wk * grad_fk(h_cur, int(k))
        else:
            weights = np.exp(beta * (over - max_over))
            weights /= weights.sum()
            grad = np.zeros(n)
            for k in range(n):
                if weights[k] < 1e-10:
                    continue
                grad += weights[k] * grad_fk(h_cur, k)

        grad *= dx
        v = moment * v - lr_cur * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

        if sc < c_curr:
            h_cur = h_new
        else:
            lr_cur = max(lr_cur * 0.9, 1e-8)
            v *= 0.5

    return best_h, best_score


def socp_init(n, target, deadline=None):
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()

    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]

    step = max(1, n // 80)
    for k in range(0, n, step):
        u = h_var[:n - k] + 1.0 - h_var[k:]
        constraints.append(cp.norm(u, 2) ** 2 <= 4.0 * t_var)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=50000, abstol=1e-10, reltol=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000)
        except Exception:
            return project_feasible(np.full(n, 0.5), target), compute_c5(np.full(n, 0.5))

    if h_var.value is None:
        return project_feasible(np.full(n, 0.5), target), compute_c5(np.full(n, 0.5))

    h_out = project_feasible(h_var.value, target)
    return h_out, compute_c5(h_out)


def lse_minimize(h, n, target, alphas, maxiter_each=2000, deadline=None):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()

    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break

        def obj(x):
            over = compute_overlaps(x)
            m = float(np.max(over))
            shifted = alpha * (over - m)
            shifted = np.clip(shifted, -700.0, 0.0)
            return m + float(np.log(np.sum(np.exp(shifted)))) / alpha

        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass

    return best_h, best_score


def trust_constr_polish(h, n, target, deadline=None, alpha=5000):
    dx = 2.0 / n

    def obj(x):
        over = compute_overlaps(x)
        m = float(np.max(over))
        shifted = alpha * (over - m)
        shifted = np.clip(shifted, -700.0, 0.0)
        return m + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def obj_grad(x):
        over = compute_overlaps(x)
        m = float(np.max(over))
        shifted = alpha * (over - m)
        shifted = np.clip(shifted, -700.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        g = np.zeros(n)
        for k in range(n):
            if w[k] < 1e-12:
                continue
            g += w[k] * grad_fk(x, k)
        g *= dx
        return g

    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target,
         'jac': lambda x: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, jac=obj_grad, method='trust-constr',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 500, 'gtol': 1e-12, 'xtol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)
