            if time.time() > deadline - 100:
                break
            h_sg, sc_sg = nesterov_subgradient(best_h, n, target, 3000, temp, 0.003,
                                               deadline - 80)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 6: Cutting-plane again (second round)
    if time.time() < deadline - 80:
        h_cp2, sc_cp2 = cutting_plane_refine(best_h, n, target, deadline - 60,
                                             max_rounds=300, max_active=80)
        if sc_cp2 < best_score:
            best_score = sc_cp2
            best_h = h_cp2

    # Phase 7: Worst-lag surgery
    if time.time() < deadline - 30:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 500, 0.003, deadline - 15)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 8: Final cutting-plane polish
    if time.time() < deadline - 10:
        h_cp3, sc_cp3 = cutting_plane_refine(best_h, n, target, deadline - 5,
                                             max_rounds=200, max_active=40)
        if sc_cp3 < best_score:
            best_score = sc_cp3
            best_h = h_cp3

    # Phase 9: Final L-BFGS-B with very high alpha
    if time.time() < deadline - 5:
        h_new = lbfgsb_round(best_h, n, target, 100000, 500.0, 300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 10: Final surgery
    if time.time() < deadline - 3:
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, n, target, 200, 0.001, deadline - 1)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n
