```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_sum(h, target):
    """Project h to have sum = target, clipping to [0,1]"""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    # Distribute the difference
    if diff > 0:
        # Need to increase sum; add to elements not at 1
        mask = h < 1.0
        count = mask.sum()
        if count == 0:
            return h
        h[mask] += diff / count
    else:
        # Need to decrease sum; subtract from elements not at 0
        mask = h > 0.0
        count = mask.sum()
        if count == 0:
            return h
        h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def continuous_corr_delta(h, i, delta):
    """Compute the change in correlation when h[i] changes by delta"""
    n = len(h)
    k = np.arange(2*n-1) - (n-1)
    result = np.zeros(2*n-1)
    # Contribution from j=i term: delta * (1 - h[i+k])
    idx = i + k
    valid = (idx >= 0) & (idx < n)
    result[valid] += delta * (1.0 - h[idx[valid]])
    # Contribution from j=i-k term (where j+k=i): -delta * h[i-k]
    idx = i - k
    valid = (idx >= 0) & (idx < n)
    result[valid] -= delta * h[idx[valid]]
    return result

def sa_continuous(h_start, n_iters, t_init, t_final, rng, time_limit, start_time):
    """Simulated annealing in the continuous domain"""
    n = len(h_start)
    target = float(n) / 2.0
    dx = 2.0 / n
    h = project_sum(h_start.copy(), target)
    corr = full_corr(h)
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 50000
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        if t < 1e-12:
            t = 1e-12
        
        # Pick a random index to modify
        i = rng.randint(n)
        # Step size proportional to temperature
        step = t * rng.uniform(-1, 1) * 0.5
        new_val = h[i] + step
        if new_val < 0 or new_val > 1:
            # Try a smaller step
            step = t * rng.uniform(-0.1, 0.1)
            new_val = h[i] + step
            if new_val < 0 or new_val > 1:
                stall += 1
                continue
        
        delta = new_val - h[i]
        # To maintain the sum constraint, adjust another element
        j = rng.randint(n)
        if j == i:
            j = (i + 1) % n
        # Adjust h[j] to compensate
        adjust = -delta
        new_j_val = h[j] + adjust
        if new_j_val < 0 or new_j_val > 1:
            # Can't make this move
            stall += 1
            continue
        
        delta_j = adjust
        corr_delta = continuous_corr_delta(h, i, delta) + continuous_corr_delta(h, j, delta_j)
        new_corr_max = float((corr + corr_delta).max())
        new_c5 = new_corr_max * dx
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or (rng.random() < np.exp(-delta_c5 / t)):
            h[i] = new_val
            h[j] = new_j_val
            corr += corr_delta
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    
    return best_h, best_c5
```

This is a weighted average of the individual gradients, weighted by soft attention on the maximum.

def smooth_max_obj(h, alpha, corr_cache=None):
    """Compute smooth max of correlation"""
    n = len(h)
    if corr_cache is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    else:
        corr = corr_cache
    max_c = corr.max()
    # Shift for numerical stability
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    log_sum = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    return log_sum

def smooth_max_grad(h, alpha):
    """Gradient of smooth max"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient of corr[k] with respect to h[i]
    # corr[k] = sum_m h[m](1-h[m+k])
    # d(corr[k])/d(h[i]) = (1-h[i+k]) - h[i-k] (for valid indices)
    
    # We need sum_k weights[k] * d(corr[k])/d(h[i])
    # = sum_k weights[k] * ((1-h[i+k]) - h[i-k])
    
    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)
    
    # For each i, sum over k: weights[k] * (1-h[i+k]) - weights[k] * h[i-k]
    # = sum_k weights[k] - sum_k weights[k]*h[i+k] - sum_k weights[k]*h[i-k]
    
    # sum_k weights[k] = 1 (normalized)
    # sum_k weights[k]*h[i+k]: for each i, this is a weighted sum of h at shifted positions
    # sum_k weights[k]*h[i-k]: similar
    
    # This is O(n²) per gradient evaluation, which is fine for n=600
    
    for i in range(n):
        idx1 = i + k_vals
        valid1 = (idx1 >= 0) & (idx1 < n)
        sum1 = np.sum(weights[valid1] * h[idx1[valid1]])
        
        idx2 = i - k_vals
        valid2 = (idx2 >= 0) & (idx2 < n)
        sum2 = np.sum(weights[valid2] * h[idx2[valid2]])
        
        grad[i] = 1.0 - sum1 - sum2
    
    return grad

    weights = np.exp(shifted)
    weights /= weights.sum()

    # = 1 - sum_k w_k * h[i+k] - sum_k w_k * h[i-k]
    # (the sum_k w_k = 1 term)

    grad = np.zeros(n)

            grad[valid1] += w * (1.0 - h[idx1[valid1]])
            grad[valid2] -= w * h[idx2[valid2]]

            break

            res = minimize(_slsqp_obj, h_cur, args=(alpha,),

            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new

def compute_c5(h):

    return float(np.max(corr)) * dx

def full_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:target_sum - final_sum]] = 1

    idx = i - k

    result[valid] += h[idx[valid]]

    result[valid] -= h[idx[valid]]
    idx = j - k
    valid = (idx >= 0) & (idx < n)
    result[valid] -= h[idx[valid]]
    return result
