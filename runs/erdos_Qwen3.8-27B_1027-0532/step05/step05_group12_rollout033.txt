from scipy.optimize import minimize
import cvxpy as cp

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

            mask = h < 1.0 - 1e-14

            mask = h > 1e-14

    """Get all correlation values with their lag indices."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(2*n-1) - (n-1)
    return corr, lags

def solve_lp(h_old, n, target, k_list=None, delta=0.0):
    """Solve the linearized LP.
    If k_list is None, use all lags.
    delta: small regularization to avoid degenerate LP.
    """
    dx = 2.0 / n
    n_half = n - 1
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    if k_list is None:
        k_list = np.arange(-(n-1), n)
    
    # Build coefficient matrix
    # For lag L: coeff[i] = (1 - h_old[i+L])*dx if 0<=i+L<n, else 0
    A_rows = []
    for L in k_list:
        coeff = np.zeros(n)
        if L >= 0:
            # i+L valid for i = 0, ..., n-1-L
            coeff[:n-L] = (1.0 - h_old[L:]) * dx
        else:
            # i+L valid for i = -L, ..., n-1
            coeff[-L:] = (1.0 - h_old[:n+L]) * dx
        A_rows.append(coeff)
    
    A = np.array(A_rows)  # shape (|k_list|, n)
    constraints.append(A @ h_var <= t_var)
    
    if delta > 0:
        objective = t_var + delta * cp.sum_squares(h_var - h_old)
    else:
        objective = t_var
    
    prob = cp.Problem(cp.Minimize(objective), constraints)
    
    try:

    except:

        except:

- For L ≥ 0: O_L(h) = sum_{i=0}^{n-1-L} h[i] * (1-h[i+L]) * dx

The gradient dO_L/dh[j]:
- For L ≥ 0:

  - If L ≤ j ≤ n-1: dO_L/dh[j] = -h[j-L]*dx (from the (1-h[i+L]) term where i+L=j, i=j-L)
  - If 0 ≤ j ≤ n-1-L AND L ≤ j ≤ n-1 (i.e., L ≤ j ≤ n-1-L): both terms apply

  - If j > n-1-L but j < L: neither applies (only possible if n-1-L < L, i.e., L > (n-1)/2)

  - For j in [0, n-1-L]: the h[j] term contributes (1-h[j+L])*dx
  - For j in [L, n-1]: the (1-h[j-L]) term contributes -h[j-L]*dx
  - So dO_L/dh[j] = (1-h[j+L])*dx * [0≤j≤n-1-L] - h[j-L]*dx * [L≤j≤n-1]

- For L < 0 (let M = -L > 0): O_{-M}(h) = sum_{i=M}^{n-1} h[i] * (1-h[i-M]) * dx

  - So dO_{-M}/dh[j] = (1-h[j-M])*dx * [M≤j≤n-1] - h[j+M]*dx * [0≤j≤n-1-M]

def grad_lag(h, L, dx):

    n = len(h)
    grad = np.zeros(n)

        # h[j] term: j in [0, n-1-L]

        # (1-h) term: j in [0, n-1-M]

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    if top_m_max is None:
        top_m_max = 2 * n - 1
    it = 0
    stall = 0
    top_m = top_m_init
    while time.time() - start_time < time_limit - deadline_offset:
        it += 1
        h_lp = solve_lp_direct(h, n, target, top_m=top_m)
        if h_lp is None:
            break
        h_try, sc, alpha_used = line_search_update(h, h_lp, target, best_c5)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try
            stall = 0
        else:
            h = (1.0 - 0.5) * h + 0.5 * h_lp
            h = project_feasible(h, target)
            stall += 1
            if stall % 50 == 0:
                top_m = min(top_m_max, top_m + 20)
            if stall > 300:
                break
        if it % full_every == 0 and top_m < 2 * n - 1:
            h_full = solve_lp_direct(h, n, target, k_list=np.arange(-(n-1), n))
            if h_full is not None:
                h_try2, sc2, _ = line_search_update(h, h_full, target, best_c5)
                if sc2 < best_c5:
                    best_c5 = sc2
                    best_h = h_try2.copy()
                    h = h_try2
                    stall = 0
    return best_h, best_c5

def gradient_polish(h_init, n, target, time_limit, start_time, deadline_offset=20):
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alphas = [50, 200, 1000, 5000, 20000, 100000]
    top_m_list = [20, 50, 100, 200, 400, 2 * n - 1]
    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]
        momentum = np.zeros(n)
        beta = 0.85
        lr = 2.0 / (alpha + 1.0)
        stalled = 0
        for step in range(5000):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            grad = smooth_max_grad(h, alpha, dx, top_m=top_m)
            momentum = beta * momentum + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum))
            if gnorm < 1e-13:
                break
            step_lr = lr / (gnorm + 1e-10)
            step_lr = min(step_lr, 1.0)
            improved = False
            for trial in range(30):
                h_try = h - step_lr * momentum
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-14:
                    best_c5 = sc
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                step_lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 40:
                    break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_polish(h_start, n, target, alphas, maxiter, time_limit, start_time, deadline_offset=20):
    dx = 2.0 / n
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        for trial_range in range(3):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            lo_h = np.maximum(h_cur - 0.1, 0.0)
            hi_h = np.minimum(h_cur + 0.1, 1.0)
            bounds = list(zip(lo_h, hi_h))
            try:
                res = minimize(
                    lambda x, a=alpha: _smooth_obj_local(x, a, dx),
                    h_cur, method='SLSQP', bounds=bounds,
                    constraints=[{'type': 'eq', 'fun': lambda x, t=target: np.sum(x) - t,
                                  'jac': lambda x, n=n: np.ones(n)}],
                    options={'maxiter': maxiter, 'ftol': 1e-18})
                if res.x is not None:
                    h_new = project_feasible(np.asarray(res.x).ravel(), target)
                    sc = compute_c5(h_new)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        h_cur = h_new
            except Exception:
                pass
    return h_cur, best_score

def _smooth_obj_local(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    # Phase 1: Alternating LP at initial resolution
    if tleft() > 250:
        t1 = min(350, tleft() * 0.35)
        h1, s1 = alternating_lp(best_h, n, target, time_limit=t1,
                                start_time=start_time, deadline_offset=120,
                                top_m_init=30, full_every=100)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Gradient polish at initial resolution
    if tleft() > 150:
        t2 = min(200, tleft() * 0.2)
        h2, s2 = gradient_polish(best_h, n, target, time_limit=t2,
                                 start_time=start_time, deadline_offset=80)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Alternating LP again with more lags
    if tleft() > 200:
        t3 = min(250, tleft() * 0.25)
        h3, s3 = alternating_lp(best_h, n, target, time_limit=t3,
                                start_time=start_time, deadline_offset=100,
                                top_m_init=100, full_every=150)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Interpolate to higher resolution and optimize
    if tleft() > 400:
        n2 = 800
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        t4 = min(350, tleft() * 0.35)
        h4, s4 = alternating_lp(h_interp, n2, target2, time_limit=t4,
                                start_time=start_time, deadline_offset=120,
                                top_m_init=30, full_every=100)
        if tleft() > 100:
            h4b, s4b = gradient_polish(h4, n2, target2, time_limit=min(150, tleft()),
                                       start_time=start_time, deadline_offset=60)
            if s4b < s4:
                h4, s4 = h4b, s4b
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SLSQP polish at best resolution
    if tleft() > 100:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h5, s5 = slsqp_polish(best_h, n_use, target_use,
                              alphas=[200, 1000, 5000, 20000],
                              maxiter=500, time_limit=budget_s,
                              start_time=start_time, deadline_offset=40)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Try n=1000 if time permits
    if tleft() > 500:
        n3 = 1000
        target3 = n3 / 2.0
        h_interp3 = interpolate_to(best_h, n3, target3)
        t6 = min(350, tleft() * 0.4)
        h6, s6 = alternating_lp(h_interp3, n3, target3, time_limit=t6,
                                start_time=start_time, deadline_offset=100,
                                top_m_init=30, full_every=100)
        if tleft() > 100:
            h6b, s6b = gradient_polish(h6, n3, target3, time_limit=min(150, tleft()),
                                       start_time=start_time, deadline_offset=50)
            if s6b < s6:
                h6, s6 = h6b, s6b
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final gradient polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h7, s7 = gradient_polish(best_h, n_use, target_use,
                                 time_limit=min(100, tleft()),
                                 start_time=start_time, deadline_offset=20)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final LP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h8, s8 = alternating_lp(best_h, n_use, target_use,
                                time_limit=min(50, tleft()),
                                start_time=start_time, deadline_offset=10,
                                top_m_init=200, full_every=50)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Final feasibility check
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
