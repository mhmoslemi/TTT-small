```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_and_weights(corr, alpha, idx=None):
    if idx is not None:
        c = corr[idx]
    else:
        c = corr
    cmax = float(np.max(c))
    shifted = np.clip(alpha * (c - cmax), -700.0, 0.0)
    w = np.exp(shifted)
    wsum = w.sum()
    w = w / wsum
    val = cmax + np.log(wsum) / alpha
    return val, w

def grad_from_weights(w, idx, h, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for i in range(len(w)):
        wi = w[i]
        if wi < 1e-10:
            continue
        mi = int(idx[i])
        k = mi - (n - 1)
        if k >= 0:
            g[:n-k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n-k]
        else:
            e = -k
            g[e:] += wi * (1.0 - h[:n-e])
            g[:n-e] -= wi * h[e:]
    g *= dx
    return g

def fista_opt(h0, n, target, alpha, max_iter, deadline, active_idx=None):
    h_y = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    best_sc = compute_c5(h0)
    best_h = h0.copy()
    L = 2.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if h_y is h_prev:
            break
        beta = (t - 1.0) / (t + 1.0)
        y = h_y + beta * (h_y - h_prev)
        
        corr = compute_corr(y)
        val, w = lse_and_weights(corr, alpha, active_idx)
        idx = active_idx if active_idx is not None else np.arange(len(corr))
        g = grad_from_weights(w, idx, y, n)
        
        # Backtracking line search for L
        # (simplified: just use fixed L with occasional adjustment)
        z = y - g / L
        h_new = project_feasible(z, target)
        
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new.copy()
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h_y
        h_y = h_new
        t = t_new
        
        # Adaptive L every 100 steps
        if it % 100 == 99:
            step_norm = np.linalg.norm(h_y - h_prev)
            if step_norm > 1e-10:
                corr2 = compute_corr(h_y)
                val2, w2 = lse_and_weights(corr2, alpha, active_idx)
                g2 = grad_from_weights(w2, idx, h_y, n)
                g_diff = np.linalg.norm(g2 - g)
                ratio = g_diff / step_norm
                if ratio < L * 0.5:
                    L = max(L * 0.6, 0.5)
                elif ratio > L * 2.0:
                    L = L * 1.8
    
    return best_h, best_sc

def slsqp_opt(h0, n, target, alpha, max_iter, deadline, active_idx=None):
    def obj(x):
        c = compute_corr(x)
        val, _ = lse_and_weights(c, alpha, active_idx)
        return val
    
    def objg(x):
        c = compute_corr(x)
        _, w = lse_and_weights(c, alpha, active_idx)
        idx = active_idx if active_idx is not None else np.arange(len(c))
        return grad_from_weights(w, idx, x, n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=objg,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h0

def subgrad_polish(h0, n, target, n_steps, temp, lr_init, lr_decay, deadline):
    h_cur = h0.copy()
    best_h = h0.copy()
    best_sc = compute_c5(h0)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        sc_cur = float(np.max(corr)) * (2.0 / n)
        if sc_cur < best_sc:
            best_sc = sc_cur
            best_h = h_cur.copy()
        top_k = 6
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        g = grad_from_weights(w, top_idx, h_cur, n)
        h_new = project_feasible(h_cur - lr * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < sc_cur:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_sc

def worst_lag_surgery(h0, n, target, max_steps, eps_init, deadline):
    h_cur = h0.copy()
    best_h = h0.copy()
    best_sc = compute_c5(h0)
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        g = grad_from_weights(np.array([1.0]), np.array([m_star]), h_cur, n)
        h_new = project_feasible(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best_h = h_new.copy()
            h_cur = h_new
            eps = min(eps * 1.05, 0.02)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return best_h, best_sc

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
