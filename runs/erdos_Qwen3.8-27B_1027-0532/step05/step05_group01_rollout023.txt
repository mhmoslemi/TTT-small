```python
def compute_g(i0, j0, h, n):
    shifts = np.arange(-(n-1), n)
    # h[i0 + shift] for valid indices
    idx_p = i0 + shifts  # = i0 - ℓ when shifts = -ℓ... 
    
    # Wait, I need to be careful about indexing.
    # ℓ ranges from -(n-1) to (n-1)
    # shifts array = [-(n-1), -(n-2), ..., 0, ..., n-2, n-1]
    # For each ℓ:
    #   h[i0-ℓ] is valid if 0 ≤ i0-ℓ < n
    #   h[i0+ℓ] is valid if 0 ≤ i0+ℓ < n
    #   h[j0-ℓ] is valid if 0 ≤ j0-ℓ < n
    #   h[j0+ℓ] is valid if 0 ≤ j0+ℓ < n
    
    i0_minus = i0 - shifts  # = i0 - ℓ for each ℓ in shifts
    i0_plus = i0 + shifts   # = i0 + ℓ
    j0_minus = j0 - shifts
    j0_plus = j0 + shifts
    
    mask_i0m = (i0_minus >= 0) & (i0_minus < n)
    mask_i0p = (i0_plus >= 0) & (i0_plus < n)
    mask_j0m = (j0_minus >= 0) & (j0_minus < n)
    mask_j0p = (j0_plus >= 0) & (j0_plus < n)
    
    g = np.zeros(2*n-1)
    g[mask_i0m] += (h[i0_minus[mask_i0m]] - 1.0)
    g[mask_i0p] += h[i0_plus[mask_i0p]]
    g[mask_j0m] -= (h[j0_minus[mask_j0m]] - 1.0)
    g[mask_j0p] -= h[j0_plus[mask_j0p]]
    
    return g
```

Now, for the binary search: given corr and g, find δ* that minimizes max(corr + δ*g).
The function f(δ) = max_k (corr[k] + δ*g[k]) is a convex piecewise-linear function in δ.
We want to find the minimum of f over δ ∈ [δ_lo, δ_hi].
The minimum occurs where the subgradient contains 0, i.e., where the max is achieved at a point where g[k] ≤ 0 for all active k (or equivalently, 0 is in the convex hull of {g[k] : k active}).

For binary search: f'(δ) (right derivative) = max_{k active at δ} g[k]. If this is > 0, f is increasing, so we should decrease δ. If < 0, f is decreasing, so we should increase δ.

Wait, actually f is convex, so:
- If the right derivative at δ is > 0, the minimum is to the left
- If the left derivative at δ is < 0, the minimum is to the right
- If 0 is between left and right derivatives, δ is optimal

For a piecewise linear convex function, we can use:
- f'(δ+) = max_{k: corr[k]+δ*g[k] = f(δ)} g[k]
- If f'(δ+) > 0: δ is too large, move left (b = δ)
- If f'(δ+) ≤ 0: δ is too small or optimal, move right (a = δ)

This is what the previous code does. Good.

Let me also think about a totally different approach: **Benders decomposition** or **cutting plane method**.

The problem is:
min_{h} max_{k} c_k(h)
s.t. 0 ≤ h ≤ 1, Σh = n/2

This is equivalent to:
min_{h, t} t
s.t. c_k(h) ≤ t for all k
     0 ≤ h ≤ 1
     Σh = n/2

The constraints c_k(h) ≤ t are non-convex (quadratic). But we can use a cutting plane approach:
1. Start with a subset of constraints
2. Solve the relaxed problem
3. Add the violated constraint
4. Repeat

But the subproblem is still non-convex. So this might not help directly.

Alternative: **Sequential Quadratic Programming (SQP)** approach. At each iteration:
1. Linearize c_k(h) around current h₀: c_k(h) ≈ c_k(h₀) + ∇c_k(h₀)·(h-h₀)
2. Solve: min t s.t. c_k(h₀) + ∇c_k(h₀)·(h-h₀) ≤ t, constraints
3. Update h

This is like a cutting plane method but with linear approximations. The subproblem is a linear program (LP)!

Let me think about this more carefully. At h₀, the active constraints are those k where c_k(h₀) is close to the maximum. We only need to include those in the LP.

Actually, a simpler approach: at each iteration, identify the top K shifts (those with largest c_k), and solve the LP:
min t
s.t. c_k(h₀) + ∇c_k(h₀)·(h-h₀) ≤ t for k in top K
     0 ≤ h ≤ 1
     Σh = n/2

This is a linear program with 2n+1 variables (h₀...h_{n-1}, t) and n+K+1 constraints. For n=400 and K=20, this is very fast with CVXPY/Gurobi.

Then update h₀ and repeat. This is essentially a **linearized Benders/cutting plane** method!

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit=None):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time_limit is not None and time.time() > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_g_for_pair(i0, j0, h, n, shifts):
    i0m = i0 - shifts
    i0p = i0 + shifts
    j0m = j0 - shifts
    j0p = j0 + shifts
    m1 = (i0m >= 0) & (i0m < n)
    m2 = (i0p >= 0) & (i0p < n)
    m3 = (j0m >= 0) & (j0m < n)
    m4 = (j0p >= 0) & (j0p < n)
    g = np.zeros(2 * n - 1)
    g[m1] += (h[i0m[m1]] - 1.0)
    g[m2] += h[i0p[m2]]
    g[m3] -= (h[j0m[m3]] - 1.0)
    g[m4] -= h[j0p[m4]]
    return g


def pair_cd_improved(h, n, target, time_limit, max_sweeps=10000, n_cand_low=40, n_cand_high=40):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand_low]
        high_set = sorted_idx[-n_cand_high:]

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = compute_g_for_pair(i, j, h, n, shifts)
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def benders_lp_round(h, n, target, time_limit, max_iters=200, K=15):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    idxs = np.arange(n)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()

        top_idx = np.argsort(corr)[-K:]

        vars_h = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [vars_h >= 0, vars_h <= 1, cp.sum(vars_h) == target]

        for k_idx in top_idx:
            k = int(k_idx) - mid
            m1 = (idxs - k >= 0) & (idxs - k < n)
            m2 = (idxs + k >= 0) & (idxs + k < n)
            grad_k = np.zeros(n)
            grad_k[m1] = (1.0 - h[idxs[m1] - k])
            grad_k[m2] -= h[idxs[m2] + k]
            rhs = t_var - corr[k_idx] + np.dot(grad_k, h)
            constraints.append(cp.dot(grad_k, vars_h) <= rhs)

        objective = cp.Minimize(t_var)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=10000, verbose=False)
            if vars_h.value is not None and not np.any(np.isnan(vars_h.value)):
                h_new = project_feasible(vars_h.value, target)
                c5_new = compute_c5(h_new)
                if c5_new < c5_val - 1e-15:
                    h = h_new
                else:
                    h_blend = 0.5 * h + 0.5 * h_new
                    h_blend = project_feasible(h_blend, target)
                    if compute_c5(h_blend) < c5_val - 1e-15:
                        h = h_blend
                    else:
                        break
        except Exception:
            break

    return best_h, best_c5


def soft_min_subgradient(h, n, target, time_limit, max_steps=5000, beta=30.0, base_lr=0.02):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    v = np.zeros(n)
    mu = 0.8

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()

        shifted = np.clip(beta * (corr - c_max), -300.0, 0.0)
        w = np.exp(shifted)
        w = w / w.sum()

        grad = np.zeros(n)
        active_shifts = np.where(w > 1e-6)[0]
        for s in active_shifts:
            k = int(s) - mid
            wi = w[s]
            m1 = (idxs - k >= 0) & (idxs - k < n)
            m2 = (idxs + k >= 0) & (idxs + k < n)
            grad[m1] += wi * (1.0 - h[idxs[m1] - k])
            grad[m2] -= wi * h[idxs[m2] + k]

        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h = h_new
        else:
            h_new2 = project_feasible(h - 0.5 * base_lr * v, target)
            if compute_c5(h_new2) < c5_now - 1e-15:
                h = h_new2
            elif step > 500:
                break

    return best_h, best_c5


def structured_init(n, target, seed):
    rng = np.random.RandomState(seed)
    candidates = []

    # Binary random
    h_bin = np.zeros(n)
    ones_pos = rng.choice(n, size=n // 2, replace=False)
    h_bin[ones_pos] = 1.0
    candidates.append(h_bin)

    # Smoothed random
    h_rand = rng.uniform(0, 1, n)
    from scipy.ndimage import gaussian_filter1d
    h_smooth = gaussian_filter1d(h_rand, sigma=3.0)
    h_smooth = (h_smooth - h_smooth.min()) / (h_smooth.max() - h_smooth.min() + 1e-10)
    candidates.append(h_smooth)

    # Cosine-based
    x = np.linspace(0, 2, n, endpoint=False)
    for freq in [1, 2, 3]:
        h_cos = 0.5 + 0.3 * np.cos(2 * np.pi * freq * x / 2.0) + 0.1 * np.cos(2 * np.pi * (freq + 1) * x / 2.0)
        h_cos = np.clip(h_cos, 0, 1)
        candidates.append(h_cos)

    # Triangle wave
    t = (x / 2.0) % 1.0
    h_tri = 0.5 + 0.4 * (1.0 - 4.0 * np.abs(t - 0.5))
    h_tri = np.clip(h_tri, 0, 1)
    candidates.append(h_tri)

    best_h = None
    best_score = np.inf
    for cand in candidates:
        h_p = project_feasible(cand, target)
        sc = compute_c5(h_p)
        if sc < best_score:
            best_score = sc
            best_h = h_p
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    # Phase 8: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            cd_time = min(60, tleft() * 0.2)
            h_hires, c5_hires = pair_cd_improved(h_interp, n_try, target_try,
                                                 time_limit=cd_time,
                                                 max_sweeps=3000, n_cand_low=40, n_cand_high=40)
            if tleft() > 40:
                limit = time.time() + min(40, tleft() * 0.4)
                h_hires = slsqp_round(h_hires, n_try, target_try,
                                      [1000, 5000, 20000], 800, 1e-15, time_limit=limit)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 9: Final pair CD at best resolution
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(50, tleft() * 0.5)
        h_cd, c5_cd = pair_cd_improved(best_h, n_use, target_use,
                                       time_limit=cd_time, max_sweeps=5000,
                                       n_cand_low=50, n_cand_high=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        limit = time.time() + min(25, tleft() * 0.7)
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000], 1000, 1e-15, time_limit=limit)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
