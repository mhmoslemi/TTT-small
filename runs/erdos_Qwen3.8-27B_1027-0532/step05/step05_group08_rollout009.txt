import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha, active_idx=None):
    corr = compute_corr(h)
    if active_idx is not None:
        c_sub = corr[active_idx]
    else:
        c_sub = corr
    c_max = float(np.max(c_sub))
    if alpha < 1e6:
        shifted = np.clip(alpha * (c_sub - c_max), -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    else:
        return c_max


def lse_grad_full(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-10:
            continue
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wm * (1.0 - h[k:])
            g[k:] -= wm * h[:n - k]
        else:
            ell = -k
            g[ell:] += wm * (1.0 - h[:n - ell])
            g[:n - ell] -= wm * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter, active_idx=None):
    def obj(x):
        val = lse_value(x, alpha, active_idx)
        penalty = mu * (np.sum(x) - target) ** 2

def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    except Exception:
        return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    n_init = len(initial_h_values)
    target_init = n_init / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target_init)
    best_score = compute_c5(h_init)

    n = 800
    target = n / 2.0
    h = interpolate_to(h_init, n, target)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B LogSumExp continuation
    alphas = [10, 30, 80, 200, 500, 1500, 4000, 10000, 30000]
    for alpha in alphas:
        if time.time() > deadline - 250:
            break
        for mu in [10.0, 100.0, 500.0]:
            if time.time() > deadline - 240:
                break
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 300)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: SLSQP with active set
    if time.time() < deadline - 200:
        for alpha in [500, 2000, 8000, 30000]:
            if time.time() > deadline - 180:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 3000)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 3: Multi-lag focused SLSQP
    if time.time() < deadline - 180:
        for m_active in [4, 6, 8, 12]:
            if time.time() > deadline - 150:
                break
            h_ml, sc_ml = multi_lag_slsqp(best_h, n, target, m_active, 5000, 1500, deadline - 140)
            if sc_ml < best_score:
                best_score = sc_ml
                best_h = h_ml

    # Phase 4: Worst lag surgery
    if time.time() < deadline - 120:
        for eps0 in [0.005, 0.003, 0.002, 0.001]:
            if time.time() > deadline - 100:
                break
            h_surg, sc_surg = worst_lag_gradient_step(best_h, n, target, eps0, deadline - 95)
            if sc_surg < best_score:
                best_score = sc_surg
                best_h = h_surg

    # Phase 5: Block perturbations to escape local optima
    if time.time() < deadline - 100:
        block_sizes = [5, 10, 20, 30]
        h_pert, sc_pert = block_perturbation(best_h, n, target, rng, 3, block_sizes, deadline - 80)
        if sc_pert < best_score:
            best_score = sc_pert
            best_h = h_pert

    # Phase 6: Re-optimize after perturbation
    if time.time() < deadline - 80:
        for alpha in [2000, 10000]:
            if time.time() > deadline - 60:
                break
            h_new = lbfgsb_round(best_h, n, target, alpha, 200.0, 200)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        if time.time() < deadline - 50:
            h_new = slsqp_round(best_h, n, target, 20000, 2000)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Try n=1000
    if time.time() < deadline - 200:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [20, 60, 200, 600, 2000, 6000, 20000]:
            if time.time() > deadline - 80:
                break
            for mu in [50.0, 500.0]:
                if time.time() > deadline - 75:
                    break
                h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 200)
                sc2 = compute_c5(h2_new)
                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 60:
            h2_new = slsqp_round(best_h2, n2, target2, 10000, 1500)
            sc2 = compute_c5(h2_new)
            if sc2 < best_score2:
                best_score2 = sc2
                best_h2 = h2_new
        if time.time() < deadline - 40:
            h_surg2, sc_surg2 = worst_lag_gradient_step(best_h2, n2, target2, 0.003, deadline - 35)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2
        if best_score2 < best_score:
            best_score = best_score2
            best_h = best_h2
            n = n2
            target = target2

    # Phase 8: Another round of surgery + SLSQP
    if time.time() < deadline - 60:
        h_surg, sc_surg = worst_lag_gradient_step(best_h, n, target, 0.002, deadline - 50)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        if time.time() < deadline - 40:
            h_new = slsqp_round(best_h, n, target, 50000, 2000)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 9: Gaussian perturbation restarts
    if time.time() < deadline - 50:
        for scale in [0.05, 0.1, 0.03]:
            if time.time() > deadline - 30:
                break
            h_pert = best_h + rng.randn(n) * scale
            h_pert = project_feasible(h_pert, target)
            h_ref = lbfgsb_round(h_pert, n, target, 3000, 500.0, 200)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                if time.time() < deadline - 20:
                    h_ref2 = slsqp_round(best_h, n, target, 20000, 1500)
                    sc2 = compute_c5(h_ref2)
                    if sc2 < best_score:
                        best_score = sc2
                        best_h = h_ref2

    # Phase 10: Final coordinate descent
    if time.time() < deadline - 15:
        h_cd, sc_cd = coordinate_descent_polish(best_h, n, target, deadline - 10, max_rounds=20)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 11: Final SLSQP polish
    if time.time() < deadline - 5:
        h_new = slsqp_round(best_h, n, target, 100000, 1000)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final projection and verification
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    # Also try returning the n=800 version if it's better
    if len(h_final) == 1000 and len(best_h) == 800:
        pass

    return h_final, final_score, len(h_final)
