import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    over = compute_corr(h)[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_val(over, alpha):
    c_max = float(np.max(over))
    shifted = np.clip(alpha * (over - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_obj(h, alpha, n):
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return lse_val(over, alpha)

def lse_obj_grad(h, alpha, n):
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = np.clip(alpha * (over - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        if k == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wk * (1.0 - h[k:n])
            grad[k:] -= wk * h[:n-k]
    return grad * dx

def slsqp_with_lse(h, n, target, alpha_schedule, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)
    
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return ones_vec
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alpha_schedule:
        if time.time() > deadline:
            break
        def obj(x):
            return lse_obj(x, a, n)
        def obj_grad(x):
            return lse_obj_grad(x, a, n)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, jac=obj_grad,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def binary_sa(h, n, target, max_iters, T_init, T_final, deadline, rng):
    """Simulated annealing on the binary search space."""
    h_cur = h.copy()
    # Make it binary
    h_bin = (h_cur > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    h_cur = h_bin
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    
    ones = np.where(h_cur > 0.5)[0].astype(int)
    zeros = np.where(h_cur < 0.5)[0].astype(int)
    
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cooling
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        h_new = h_cur.copy()
        h_new[a] = 0.0
        h_new[b] = 1.0
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        delta = sc_new - cur_score
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h_cur = h_new
            cur_score = sc_new
            # Update ones and zeros
            ones = np.where(h_cur > 0.5)[0].astype(int)
            zeros = np.where(h_cur < 0.5)[0].astype(int)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
    
    return best_h, best_score

def steep_descent(h, n, target, alpha, n_steps, lr, deadline):
    """Steepest descent with the LSE-smoothed objective."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    cur_score = best_score
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        grad = lse_obj_grad(h_cur, alpha, n)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try a few step sizes
        best_step_sc = cur_score
        best_step_h = h_cur
        for lr_try in [lr, lr*0.5, lr*0.25, lr*2.0]:
            h_try = h_cur - lr_try * grad
            h_try = project_feasible(h_try, target)
            sc_try = compute_c5(h_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_step_h = h_try
        
        if best_step_sc < cur_score - 1e-15:
            h_cur = best_step_h
            cur_score = best_step_sc
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
        else:
            lr *= 0.7
    
    return best_h, best_score

def generate_thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        # t[i] = parity of number of 1s in binary representation of i
        t[i] = bin(i).count('1') % 2
    return t

def binary_to_feasible(t, n, target):
    """Convert a binary sequence to a feasible h."""
    h = t.astype(float)
    # Count ones
    n_ones = int(np.sum(h))
    need = int(target) - n_ones
    
    if need > 0:
        # Need to flip 'need' zeros to ones
        zeros = np.where(h == 0)[0]
        to_flip = zeros[:need]
        h[to_flip] = 1.0
    elif need < 0:
        # Need to flip 'need' ones to zeros
        ones = np.where(h == 1)[0]
        to_flip = ones[:(-need)]
        h[to_flip] = 0.0
    
    return project_feasible(h, target)

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)  # length 2n-1
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp  # length 2n-1
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)  # shift
        # d(corr[k])/d(h[j]) = (1-h[j-s]) * [0<=j-s<n] - h[j+s] * [0<=j+s<n]

            if s == 0:

            else:

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_val(over, alpha):
    c_max = float(np.max(over))
    shifted = np.clip(alpha * (over - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def full_lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s == 0:
            grad[:] += wk * (1.0 - 2.0 * h)
        elif s > 0:
            p = s
            if n - p > 0:
                grad[:n-p] -= wk * h[p:]
            grad[p:] += wk * (1.0 - h[:n-p])
        else:
            p = -s
            if n - p > 0:
                grad[:n-p] += wk * (1.0 - h[p:])
                grad[p:] -= wk * h[:n-p]
    return grad * dx


def slsqp_lse(h, n, target, alphas, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n
    ones_vec = np.ones(n)

    def cf(x):
        return np.sum(x) - target

    def cj(x):
        return ones_vec

    cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break

        def obj(x):
            corr = compute_corr(x)
            return lse_val(corr, a)

        def obj_grad(x):
            return full_lse_grad(x, a)

        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=obj_grad,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def proj_grad_descent(h, n, target, alpha, n_steps, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    momentum = 0.7
    v = np.zeros(n)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        grad = full_lse_grad(h_cur, alpha)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-16:
            break
        grad = grad / gnorm * gnorm
        v = momentum * v - lr * grad
        h_try = h_cur + v
        h_try = project_feasible(h_try, target)
        sc_try = compute_c5(h_try)
        if sc_try < best_score:
            best_score = sc_try
            best_h = h_try.copy()
        if sc_try < compute_c5(h_cur) - 1e-16:
            h_cur = h_try
            lr = min(lr * 1.05, 0.02)
        else:
            lr *= 0.7
            v *= 0.5
    return best_h, best_score


def binary_sa(h, n, target, max_iters, T_init, T_final, deadline, rng):
    h_cur = (h > 0.5).astype(float)
    h_cur = project_feasible(h_cur, target)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    ones = np.where(h_cur > 0.5)[0].astype(int)
    zeros = np.where(h_cur < 0.5)[0].astype(int)
    T = T_init
    cooling = (T_final / T_init) ** (1.0 / max(max_iters, 1))
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T *= cooling
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h_new = h_cur.copy()
        h_new[a] = 0.0
        h_new[b] = 1.0
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        delta = sc_new - cur_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            h_cur = h_new
            cur_score = sc_new
            ones = np.where(h_cur > 0.5)[0].astype(int)
            zeros = np.where(h_cur < 0.5)[0].astype(int)
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
    return best_h, best_score


def binary_greedy_search(h, n, target, max_rounds, deadline):
    h_cur = (h > 0.5).astype(float)
    h_cur = project_feasible(h_cur, target)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_indices = np.argsort(corr)[::-1]
        improved = False
        ones = np.where(h_cur > 0.5)[0].astype(int)
        zeros = np.where(h_cur < 0.5)[0].astype(int)
        for k in top_indices[:8]:
            k = int(k)
            s = k - (n - 1)
            if s == 0:
                continue
            if s > 0:
                p = s
                if n - p <= 0:
                    continue
                grad_part = np.zeros(n)
                grad_part[:n-p] -= h_cur[p:]
                grad_part[p:] += (1.0 - h_cur[:n-p])
            else:
                p = -s
                if n - p <= 0:
                    continue
                grad_part = np.zeros(n)
                grad_part[:n-p] += (1.0 - h_cur[p:])
                grad_part[p:] -= h_cur[:n-p]
            if len(ones) < 1 or len(zeros) < 1:
                continue
            gz = grad_part[zeros]
            go = grad_part[ones]
            if len(gz) == 0 or len(go) == 0:
                continue
            a = zeros[np.argmin(gz)]
            b = ones[np.argmax(go)]
            if a == b:
                continue
            h_new = h_cur.copy()
            h_new[a] = 1.0
            h_new[b] = 0.0
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc_new
                improved = True
                ones = np.where(h_cur > 0.5)[0].astype(int)
                zeros = np.where(h_cur < 0.5)[0].astype(int)
                if cur_score < best_score:
                    best_score = cur_score
                    best_h = h_cur.copy()
        if not improved:
            break
    return best_h, best_score


def generate_thue_morse(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = float(bin(i).count('1') & 1)
    return t


def binary_to_feasible(t, n, target):
    h = t.copy().astype(float)
    n_ones = int(np.sum(h))
    need = int(round(target)) - n_ones
    if need > 0:
        zeros = np.where(h < 0.5)[0]
        to_flip = zeros[:need]
        h[to_flip] = 1.0
    elif need < 0:
        ones = np.where(h > 0.5)[0]
        to_flip = ones[:(-need)]
        h[to_flip] = 0.0
    return project_feasible(h, target)

            mx = float(np.max(corr))
            if mx < best_corr or best_pos == -1:
                best_corr = mx
                best_pos = pos
        h[best_pos] = 1.0
    return project_feasible(h, target)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    candidates = [h_init]
    tm = generate_thue_morse(n)
    h_tm = binary_to_feasible(tm, n, target)
    candidates.append(h_tm)
    tm_flip = 1.0 - tm
    h_tmf = binary_to_feasible(tm_flip, n, target)
    candidates.append(h_tmf)
    for trial in range(3):
        h_rand = (rng.rand(n) > 0.5).astype(float)
        h_rand = project_feasible(h_rand, target)
        candidates.append(h_rand)
    for cand in candidates:
        sc = compute_c5(cand)
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    deadline_p1 = start_time + 180
    h_res, sc_res = slsqp_lse(best_h, n, target,
                              [10, 50, 200, 800, 3000], 2000, 1e-15, deadline_p1)
    if sc_res < best_score:
        best_score = sc_res
        best_h = h_res

    if tleft() > 100:
        deadline_p2 = start_time + 350
        h_res, sc_res = proj_grad_descent(best_h, n, target,
                                          alpha=5000, n_steps=800,
                                          lr_init=0.003, deadline=deadline_p2)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 100:
        deadline_p3 = start_time + 450
        h_res, sc_res = slsqp_lse(best_h, n, target,
                                  [3000, 10000, 30000], 3000, 1e-16, deadline_p3)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 150:
        deadline_p4 = start_time + 650
        h_res, sc_res = binary_sa(best_h, n, target, max_iters=30000,
                                  T_init=0.008, T_final=1e-9,
                                  deadline=deadline_p4, rng=rng)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 80:
        deadline_p5 = start_time + 780
        h_res, sc_res = binary_greedy_search(best_h, n, target,
                                             max_rounds=100, deadline=deadline_p5)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res
        deadline_p5b = start_time + 850
        h_res, sc_res = slsqp_lse(best_h, n, target,
                                  [5000, 20000, 100000], 3000, 1e-16, deadline_p5b)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 150:
        deadline_p6 = start_time + 950
        h_res, sc_res = binary_sa(best_h, n, target, max_iters=40000,
                                  T_init=0.003, T_final=1e-10,
                                  deadline=deadline_p6, rng=rng)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 80:
        deadline_p7 = start_time + 1050
        h_res, sc_res = binary_greedy_search(best_h, n, target,
                                             max_rounds=200, deadline=deadline_p7)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res
        deadline_p7b = start_time + 1120
        h_res, sc_res = slsqp_lse(best_h, n, target,
                                  [10000, 50000, 200000], 4000, 1e-17, deadline_p7b)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 60:
        deadline_p8 = start_time + budget_s - 50
        h_res, sc_res = proj_grad_descent(best_h, n, target,
                                          alpha=100000, n_steps=1000,
                                          lr_init=0.001, deadline=deadline_p8)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    if tleft() > 40:
        deadline_p9 = start_time + budget_s - 20
        h_res, sc_res = binary_greedy_search(best_h, n, target,
                                             max_rounds=300, deadline=deadline_p9)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res
        h_res, sc_res = slsqp_lse(best_h, n, target,
                                  [50000, 200000], 5000, 1e-17,
                                  deadline=start_time + budget_s - 5)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final
    return best_h, best_score, n
