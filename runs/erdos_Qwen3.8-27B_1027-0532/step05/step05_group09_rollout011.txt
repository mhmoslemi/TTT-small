    best_score = compute_c5(best_h)

    def tleft():
        return deadline_hard - time.time()

    # Phase 1: FISTA continuation with log-sum-exp
    alphas = [30, 80, 200, 500, 1500, 4000, 10000, 25000, 60000, 150000]
    times = [60, 50, 50, 50, 50, 50, 50, 50, 50, 60]
    for i, (a, td) in enumerate(zip(alphas, times)):
        if tleft() < td + 10:
            break
        dl = start_time + (sum(times[:i+1])) + 5
        dl = min(dl, deadline_hard - 200)
        h_new, sc_new = fista_optimize(best_h, target, a, 200000, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: Active set optimization (direct max with softmax weights)
    if tleft() > 120:
        dl = start_time + min(120, tleft() - 20)
        h_new, sc_new = active_set_optimize(best_h, target, dl, top_m=3, temp=0.002)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: Another FISTA round at very high alpha
    if tleft() > 100:
        dl = start_time + min(100, tleft() - 20)
        h_new, sc_new = fista_optimize(best_h, target, 300000, 200000, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: Active set with different parameters
    if tleft() > 80:
        dl = start_time + min(80, tleft() - 20)
        h_new, sc_new = active_set_optimize(best_h, target, dl, top_m=5, temp=0.005)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        if tleft() > 40:
            dl2 = start_time + min(40, tleft() - 10)
            h_new, sc_new = active_set_optimize(best_h, target, dl2, top_m=2, temp=0.001)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 5: Perturbation restarts
    rng = np.random.RandomState(seed)
    if tleft() > 80:
        for s in [0.02, 0.04, 0.01]:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            dl = start_time + min(25, tleft() - 10)
            h_new, sc_new = fista_optimize(h_pert, target, 50000, 50000, dl)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 6: Final active set polish
    if tleft() > 60:
        dl = start_time + min(60, tleft() - 15)
        h_new, sc_new = active_set_optimize(best_h, target, dl, top_m=3, temp=0.001)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Local search
    if tleft() > 30:
        dl = start_time + min(tleft() - 5, 120)
        h_new, sc_new = local_search(best_h, target, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Final: ensure feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
