def fast_project(h, target):
    """Fast projection maintaining sum = target and 0 <= h <= 1."""
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def subgradient_equalize(h, target, n_active, n_steps, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = float(np.max(corr))
        tol = max(0.005 * c_max, 1e-8)
        active_mask = corr >= (c_max - tol)
        active_idx = np.where(active_mask)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(corr)])
        elif len(active_idx) > n_active:
            top_local = np.argsort(corr[active_idx])[::-1][:n_active]
            active_idx = active_idx[top_local]
        grad = np.zeros(n)
        for ki in active_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])

                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]

                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= len(active_idx)
        grad_norm = np.linalg.norm(grad)

def coordinate_descent_fast(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    def fast_eval(j, v):
        h_cur[j] = v
        diff = target - np.sum(h_cur)
        if abs(diff) > 1e-14:
            if diff > 0:
                mask = h_cur < 1.0
            else:
                mask = h_cur > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h_cur[mask] += diff / cnt
                h_cur = np.clip(h_cur, 0.0, 1.0)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        return float(np.max(corr))

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            orig = h_cur[j]
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = fast_eval(j, c)
            fd = fast_eval(j, d)
            for _ in range(22):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = fast_eval(j, c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = fast_eval(j, d)
            best_v = (a + b) / 2.0
            sc = fast_eval(j, best_v) * (2.0 / n)
            if sc < cur_score - 1e-15:
                cur_score = sc
                improved = True
            else:
                h_cur[j] = orig
                diff = target - np.sum(h_cur)

                        mask = h_cur < 1.0
