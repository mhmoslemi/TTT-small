def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)

    # Phase 5: SLSQP on n=800
    if tleft() > 100:
        h_s2 = slsqp_polish(best_h, n2, target2, [500, 2000, 8000], 1200, 1e-13)
        sc = compute_c5(h_s2)
        if sc < best_score:
            best_score = sc
            best_h = h_s2

    # Phase 6: Another CD round on n=800
    deadline_p6 = start_time + eff * 0.78
    h_cd3, sc_cd3 = paired_coord_desc(best_h, n2, target2, deadline_p6, rng,
                                       recompute_interval=15, stall_thresh=50, perturb_size=0.003)
    if sc_cd3 < best_score:
        best_score = sc_cd3
        best_h = h_cd3

    # Phase 7: Final SLSQP on n=800
    if tleft() > 60:
        h_s3 = slsqp_polish(best_h, n2, target2, [3000, 10000, 30000], 1000, 1e-14)
        sc = compute_c5(h_s3)
        if sc < best_score:
            best_score = sc
            best_h = h_s3

    # Phase 8: Try different perturbation restarts on n=800
    if tleft() > 120:
        noise_scales = [0.02, 0.04, 0.015, 0.06]
        for ns in noise_scales:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n2) * ns
            h_pert = project_feasible(h_pert, target2)
            dl = start_time + min(eff * 0.92, time.time() + tleft() * 0.4)
            h_res, sc_res = paired_coord_desc(h_pert, n2, target2, dl, rng,
                                              recompute_interval=20, stall_thresh=60,
                                              perturb_size=0.003)
            if sc_res < best_score:
                best_score = sc_res
                best_h = h_res

    # Phase 9: Final SLSQP
    if tleft() > 50:
        h_f = slsqp_polish(best_h, n2, target2, [5000, 20000], 800, 1e-15)
        sc = compute_c5(h_f)
        if sc < best_score:
            best_score = sc
            best_h = h_f

    # Phase 10: Final CD
    if tleft() > 20:
        h_fin, sc_fin = paired_coord_desc(best_h, n2, target2,
                                          start_time + budget_s - 10, rng,
                                          recompute_interval=10, stall_thresh=40,
                                          perturb_size=0.002)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Determine best output (could be n=400 or n=800)
    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
