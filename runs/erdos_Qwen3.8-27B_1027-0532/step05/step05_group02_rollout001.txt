def _smooth_max_obj(h, alpha):
    corr = full_corr_fast(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    soft_max = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    dx = 2.0 / len(h)
    return soft_max * dx

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_max_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        extra = min(final_sum - target_sum, len(ones))
        h_r[ones[:extra]] = 0
    elif final_sum < target_sum:
        zeros = np.where(h_r == 0)[0]
        extra = min(target_sum - final_sum, len(zeros))
        h_r[zeros[:extra]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

    best_h = h.copy()
    current_c5 = best_c5

    max_stall = 60000
    positions = np.arange(n)

        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        use_multi = (rng.random() < multi_prob and len(ones) >= multi_size and len(zeros) >= multi_size)

        if use_multi:
            s1 = rng.choice(len(ones), size=multi_size, replace=False)
            s2 = rng.choice(len(zeros), size=multi_size, replace=False)
            ones_to_zero = ones[s1]
            zeros_to_one = zeros[s2]
            h_new = h.copy()
            h_new[ones_to_zero] = 0.0
            h_new[zeros_to_one] = 1.0
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h = h_new
                full = full_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        else:
            if rng.random() < 0.4:
                max_idx = int(np.argmax(full))
                k_star = max_idx - (n - 1)
                if k_star != 0:
                    A = np.zeros(n)
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    A[valid] += h[idx[valid]]
                    B = np.zeros(n)
                    idx = positions - k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    idx = positions + k_star
                    valid = (idx >= 0) & (idx < n)
                    B[valid] += h[idx[valid]]
                    i_cands = ones[A[ones] <= 1.0]
                    j_cands = zeros[B[zeros] >= 1.0]
                    if len(i_cands) > 0 and len(j_cands) > 0:
                        i = i_cands[rng.randint(len(i_cands))]
                        j = j_cands[rng.randint(len(j_cands))]
                    else:
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

            delta = compute_delta(h, i, j)
            new_max = float((full + delta).max())
            c5_new = new_max * dx
            delta_c5 = c5_new - current_c5
            if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
                h[i] = 0.0
                h[j] = 1.0
                full += delta
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1

    return best_h, best_c5

def greedy_lag_targeted(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 800
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        if k_star == 0:
            stall += 1
            continue

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Score for removing a 1 at position i (wants h[i+k_star] and h[i-k_star] to be 0)
        i_scores = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        i_scores[valid] += (1.0 - h[idx[valid]])
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        i_scores[valid] += (1.0 - h[idx[valid]])

        # Score for adding a 1 at position j (wants h[j-k_star] and h[j+k_star] to be 1)
        j_scores = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        j_scores[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        j_scores[valid] += h[idx[valid]]

        top_i = ones[np.argsort(-i_scores[ones])[:min(40, len(ones))]]
        top_j = zeros[np.argsort(-j_scores[zeros])[:min(40, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                if i == j:
                    continue
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            for _ in range(60):
                i = top_i[rng.randint(len(top_i))]
                j = top_j[rng.randint(len(top_j))]
                if i == j:
                    continue
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                for _ in range(40):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    delta = compute_delta(h, i, j)
                    new_max = float((full + delta).max())
                    if new_max < max_val:
                        h[i] = 0.0
                        h[j] = 1.0
                        full += delta
                        stall = 0
                        new_c5 = float(full.max()) * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found = True
                        break
                if not found:
                    stall += 1

    return best_h, best_c5

        # We want to minimize the max corr after placing a 1 here
        best_pos = -1

            # Terms where pos is first factor: +(1-h[pos+k]) for valid pos+k
            # Terms where pos is second factor: -h[pos-k] for valid pos-k (since (1-h[pos]) goes from 1 to 0)
            delta = np.zeros(2 * n - 1)
            k_arr = np.arange(2 * n - 1) - (n - 1)
            
            idx = pos + k_arr
