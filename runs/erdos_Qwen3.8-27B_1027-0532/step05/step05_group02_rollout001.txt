def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += (h[idx[mask]] - 1.0)
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += (1.0 - h[idx[mask]])
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def compute_deltas_for_all_zeros(h, zeros):
    n = len(h)

        mask = (idx >= 0) & (idx < n)

    n = len(h)

    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:

        if need <= len(zeros):

    final_sum = int(h_r.sum())

            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]

def project_feasible(h, target_sum):

    if diff > 0:

    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def projected_gradient_continuous(h_init, n, target, n_iters, lr_start, lr_end, alpha, time_limit, start_time):
    h = h_init.copy().astype(float)
    h = project_feasible(h, target)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        ratio = it / max(n_iters - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** ratio

        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted) / np.sum(np.exp(shifted))

        grad = np.zeros(n)
        k = np.arange(2 * n - 1) - (n - 1)
        for i in range(n):
            g = 0.0
            for ki, w in enumerate(weights):
                kk = k[ki]
                if abs(w) < 1e-12:
                    continue
                idx1 = i + kk
                if 0 <= idx1 < n:
                    g += w * (1.0 - h[idx1])
                idx2 = i - kk
                if 0 <= idx2 < n:
                    g += w * (-h[idx2])
            grad[i] = g

        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        corr = np.correlate(h, 1.0 - h, mode='full')

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    phase1_time = min(100, tleft() * 0.15)
    h_g, c5_g = steepest_descent_v2(best_h, max_iters=200000, time_limit=phase1_time,
                                     start_time=start_time, n_i_candidates=50)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    phase2_time = min(200, tleft() * 0.25)
    n_sa = int(min(3000000, phase2_time * 30000))
    h_sa, c5_sa = sa_adaptive(best_h, n_sa, 0.015, 0.00005, rng,
                              time_limit=phase2_time, start_time=start_time, adaptive=True)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    n_restarts = 4
    restart_time_total = min(250, tleft() * 0.25)
    if tleft() > 60:
        time_per_restart = restart_time_total / n_restarts
        for r in range(n_restarts):
            if tleft() < 40:
                break
            n_pert = rng.randint(5, 12)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0.0
                h_pert[j] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]

            t_allow = min(time_per_restart, tleft() - 30)
            if t_allow < 5:
                break
            n_sa_r = int(min(1000000, t_allow * 25000))
            h_sr, c5_sr = sa_adaptive(h_pert, n_sa_r, 0.012, 0.00003, rng,
                                       time_limit=t_allow, start_time=start_time, adaptive=True)
            if c5_sr < best_score:
                best_score = c5_sr
                best_h = h_sr.copy()

            if tleft() < 30:
                break

            h_g2, c5_g2 = steepest_descent_v2(best_h, max_iters=50000,
                                               time_limit=min(15, tleft() * 0.05),
                                               start_time=start_time, n_i_candidates=40)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()

    if tleft() > 80:
        time_multiswap = min(80, tleft() * 0.12)
        n_ms = int(min(1000000, time_multiswap * 20000))
        h_ms, c5_ms = sa_multiswap(best_h, n_ms, 0.02, 0.001, rng,
                                    time_limit=time_multiswap, start_time=start_time,
                                    swap_sizes=(1, 2, 3))
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()
        h_g3, c5_g3 = steepest_descent_v2(best_h, max_iters=50000,
                                           time_limit=min(20, tleft() * 0.05),
                                           start_time=start_time, n_i_candidates=50)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    if tleft() > 100:
        n_coarse = 200
        target_coarse = n_coarse // 2
        x_new = np.linspace(0, 1, n_coarse)
        x_old = np.linspace(0, 1, n_init)
        h_coarse = np.interp(x_new, x_old, best_h)
        h_coarse_bin = project_binary(h_coarse, target_coarse)

        t_c = min(30, tleft() * 0.1)
        h_cg, c5_cg = steepest_descent_v2(h_coarse_bin, max_iters=100000,
                                           time_limit=t_c * 0.4, start_time=start_time,
                                           n_i_candidates=30)
        t_c2 = min(20, tleft() * 0.08)
        n_sac = int(min(500000, t_c2 * 30000))
        h_cs, c5_cs = sa_adaptive(h_cg, n_sac, 0.01, 0.0001, rng,
                                   time_limit=t_c2, start_time=start_time, adaptive=True)
        if c5_cs < c5_cg:
            h_coarse_best = h_cs
        else:
            h_coarse_best = h_cg

        ratio = n_init // n_coarse
        h_fine = np.zeros(n_init)
        for i in range(n_coarse):
            s = i * ratio
            e = min(s + ratio, n_init)
            h_fine[s:e] = h_coarse_best[i]
        h_fine_bin = project_binary(h_fine, target_init)

        t_f = min(25, tleft() * 0.08)
        h_fg, c5_fg = steepest_descent_v2(h_fine_bin, max_iters=50000,
                                           time_limit=t_f, start_time=start_time,
                                           n_i_candidates=50)
        if c5_fg < best_score:
            best_score = c5_fg
            best_h = h_fg.copy()

    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_cont = best_h.copy().astype(float)
        alphas = [30, 80, 200, 500, 1500]
        t_cont = min(tleft() - 30, 150)
        h_ref, sc = slsqp_refine(h_cont, n_use, target_use, alphas, 3000, 1e-14,
                                  time_limit=t_cont, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        h_g4, c5_g4 = steepest_descent_v2(h_bin_final, max_iters=100000,
                                           time_limit=min(10, tleft() * 0.3),
                                           start_time=start_time, n_i_candidates=60)
        if c5_g4 < best_score:
            best_score = c5_g4
            best_h = h_g4.copy()

    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                  [2000, 5000, 10000], 2000, 1e-15,
                                  time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = project_binary(best_h, int(n_final / 2.0))
        final_score = compute_c5(h_final)

    return h_final, final_score, n_final
