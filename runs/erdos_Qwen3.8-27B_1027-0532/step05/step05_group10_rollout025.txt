import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_g(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def focused_coordinate_descent(h, n, target, max_iter, time_limit, step_sizes=None):
    """Coordinate descent focused on reducing the worst lag."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    
    if step_sizes is None:
        step_sizes = [0.01, 0.05, 0.1, 0.2, 0.5]
    
    idxs = np.arange(n)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g_vals = compute_all_g(h_cur)
        k_star_idx = int(np.argmax(g_vals))
        k_star = k_star_idx - half_n
        
        # Compute gradient of g_{k_star} w.r.t. h
        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            # h[i] appears in g_k as h[i]*(1-h[i+k]) or h[i-k]*(1-h[i])
            # g_k = sum_i h[i]*(1-h[i+k])*dx

            #           - h[i-k]*dx (from the term where h[i] is the second factor in 1-h[i-k])
            if 0 <= i + k_star < n:

            if 0 <= i - k_star < n:

                h_cur = h_try

def compute_c5(h):

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    return h

    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

This is related to "difference sets" or more precisely, we want a set with large self-intersection at all shifts.

For a random set of density 1/2, |S ∩ (S+k)| ≈ n/4 for all k, giving g_k ≈ (n/2 - n/4) * 2/n = 1/2.

To beat 1/2, we need |S ∩ (S+k)| > n/4 for all k, which means the set has "more structure" than random.

The optimal structure is related to arithmetic progressions or other algebraic structures.

- So g_{n/2} = (n/2 - 0) * 2/n = 1 (as we computed)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_g(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def lp_round(h_init, n, target, max_iter, time_limit, n_active=50, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    damping = 0.85

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k_idx in sorted(active_k):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass

        if not solved:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5

        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.95)
        else:
            damping = min(0.97, damping + 0.02)

        if it % 40 == 0 and reg > 1e-8:
            reg = max(1e-8, reg * 0.8)

    return best_h, best_c5


def slsqp_cons_fun(h):
    return np.array([np.sum(h) - _g_target])


def slsqp_cons_jac(h):
    return np.array([np.ones(_g_n)])


def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_cons_fun, 'jac': slsqp_cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(slsqp_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgradient_focused(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    rng = np.random.RandomState(7)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        g_vals = compute_all_g(h_cur)
        k_star_idx = int(np.argmax(g_vals))
        k = k_star_idx - half_n

        grad = np.zeros(n)
        v1 = (idxs >= 0) & (idxs + k < n) if k >= 0 else (idxs < n) & (idxs + k >= 0)
        if k >= 0:
            valid = (idxs < n - k)
            grad[valid] += (1.0 - h_cur[valid + k]) * dx
        else:
            valid = (idxs + k >= 0)
            grad[valid] += (1.0 - h_cur[valid + k]) * dx
        v2 = (idxs - k >= 0) & (idxs - k < n)
        grad[v2] -= h_cur[v2 - k] * dx

        improved = False
        for lr in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved:
            if step % 20 == 0 and step > 20:
                for ns in [0.01, 0.03]:
                    h_try = h_cur + rng.randn(n) * ns
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_c5 - 1e-13:
                        best_c5 = sc
                        best_h = h_try.copy()
                        h_cur = h_try
                        break
            if step > 200:
                break

    return best_h, best_c5


def pair_swap_search(h, n, target, max_rounds, time_limit):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(123)

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        g_vals = compute_all_g(h_cur)
        k_star_idx = int(np.argmax(g_vals))
        k = k_star_idx - (n - 1)

        contributions = np.zeros(n)
        idxs = np.arange(n)
        if k >= 0:
            valid = idxs < n - k
            contributions[valid] = h_cur[valid] * (1.0 - h_cur[valid + k]) * (2.0 / n)
        else:
            valid = idxs + k >= 0
            contributions[valid] = h_cur[valid] * (1.0 - h_cur[valid + k]) * (2.0 / n)

        top_n = 15
        top_contrib = np.argsort(contributions)[-top_n:]
        low_vals = np.argsort(h_cur)[:top_n]

        improved = False
        order = np.random.permutation(len(top_contrib))
        for ii in order:
            i = top_contrib[ii]
            for j in low_vals:
                if i == j:
                    continue
                for frac in [0.1, 0.25, 0.5]:
                    delta = frac * h_cur[i]
                    if delta < 1e-6:
                        continue
                    if h_cur[j] + delta > 1.0:
                        delta = max(0.0, 1.0 - h_cur[j])
                    if delta < 1e-6:
                        continue
                    h_try = h_cur.copy()
                    h_try[i] -= delta
                    h_try[j] += delta
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_c5 - 1e-12:
                        best_c5 = sc
                        best_h = h_try.copy()
                        h_cur = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and rnd > 40:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
