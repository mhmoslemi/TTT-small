    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:

    if diff > 0:

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)

    x_new = np.linspace(0.0, 1.0, n_dst)

    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    if res.status == 0 and res.x is not None:
        return res.x[:n], float(res.x[n])
    return None, None

def lp_refine(h, n, target, max_iters=200, time_limit=60.0, n_active_max=50):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
        spread = c_max - np.min(corr)
        tol = max(0.02 * c_max, 1e-8)
        active_idx = np.where(corr >= c_max - tol)[0]
        if len(active_idx) > n_active_max:
            order = np.argsort(corr[active_idx])
            active_idx = active_idx[order[-n_active_max:]]
        m = len(active_idx)
        if m < 1:
            break
        grads = np.zeros((m, n))
        for i, s_idx in enumerate(active_idx):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grads[i, v1] += (1.0 - h[idxs[v1] - k])
            grads[i, v2] -= h[idxs[v2] + k]
        d, t_lp = lp_step(h, n, target, active_idx, corr, grads)
        if d is None:
            break
        improved = False
        for frac in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = h + frac * d
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved:
            break
    return best_h, best_c5

def active_set_gs(h, n, target, time_limit=60.0, max_steps=5000, n_top=8):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        c5_now = c_max * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
        top_k = np.argsort(corr)[-n_top:]
        alpha_w = 30.0
        w = np.exp(alpha_w * (corr[top_k] - c_max))
        w = w / w.sum()
        grad = np.zeros(n)
        for wi, s_idx in zip(w, top_k):
            k = int(s_idx) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += wi * (1.0 - h[idxs[v1] - k])
            grad[v2] -= wi * h[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr in [0.05, 0.03, 0.02, 0.01, 0.005, 0.002]:
            h_try = project_feasible(h - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h = h_try
                improved = True
                break
        if not improved and step > 500:
            break
    return best_h, best_c5

def pair_cd(h, n, target, time_limit=100.0, max_sweeps=3000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 150:
            break
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        best_trial = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def momentum_gs(h, n, target, time_limit=40.0, max_steps=20000, mu=0.8, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)
        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def local_perturb(h_init, n, target, max_tries=2000, time_limit=30.0):
    start = time.time()
    rng_local = np.random.RandomState(77777)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng_local.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)
        scale = float(rng_local.choice([0.03, 0.08, 0.15, 0.25]))
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    if tleft() > 120:
        h_pol, sc = slsqp_optimize(best_h, n, target, [50, 200, 1000, 5000], 1200)
        if sc < best_score:
            best_score, best_h = sc, h_pol

    if tleft() > 80:
        h_lp, sc = lp_refine(best_h, n, target, max_iters=300, time_limit=min(70, tleft() * 0.4), n_active_max=40)
        if sc < best_score:
            best_score, best_h = sc, h_lp

    if tleft() > 60:
        h_pol, sc = slsqp_optimize(best_h, n, target, [2000, 8000, 30000], 1000)
        if sc < best_score:
            best_score, best_h = sc, h_pol

    if tleft() > 60:
        h_cd, sc = pair_cd(best_h, n, target, time_limit=min(60, tleft() * 0.5), max_sweeps=3000, n_cand=50)
        if sc < best_score:
            best_score, best_h = sc, h_cd

    if tleft() > 50:
        h_lp, sc = lp_refine(best_h, n, target, max_iters=200, time_limit=min(45, tleft() * 0.5), n_active_max=50)
        if sc < best_score:
            best_score, best_h = sc, h_lp

    if tleft() > 120:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_new, sc = slsqp_optimize(h_interp, n_try, target_try, [50, 200, 1000, 5000], 800)
            if tleft() > 40:
                h_lp2, sc2 = lp_refine(h_new, n_try, target_try, max_iters=100, time_limit=min(35, tleft() * 0.4), n_active_max=40)
                if sc2 < sc:
                    h_new, sc = h_lp2, sc2
            if sc < best_score:
                best_score, best_h = sc, h_new

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_as, sc = active_set_gs(best_h, n_use, target_use, time_limit=min(40, tleft() * 0.5), max_steps=20000, n_top=10)
        if sc < best_score:
            best_score, best_h = sc, h_as

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_mom, sc = momentum_gs(best_h, n_use, target_use, time_limit=min(30, tleft() * 0.5), max_steps=20000, mu=0.85, base_lr=0.01)
        if sc < best_score:
            best_score, best_h = sc, h_mom

    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd2, sc = pair_cd(best_h, n_use, target_use, time_limit=min(50, tleft() * 0.5), max_sweeps=2000, n_cand=60)
        if sc < best_score:
            best_score, best_h = sc, h_cd2

    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp3, sc = lp_refine(best_h, n_use, target_use, max_iters=200, time_limit=min(35, tleft() * 0.5), n_active_max=60)
        if sc < best_score:
            best_score, best_h = sc, h_lp3

    if tleft() > 30:
        rng = np.random.RandomState(seed)
        for noise_scale in [0.02, 0.05]:
            if tleft() < 25:
                break
            h_pert = best_h + rng.randn(len(best_h)) * noise_scale
            h_pert = project_feasible(h_pert, len(best_h) / 2.0)
            h_pol2, sc = slsqp_optimize(h_pert, len(best_h), len(best_h) / 2.0, [200, 1000, 5000], 500)
            if sc < best_score:
                best_score, best_h = sc, h_pol2

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert2, sc = local_perturb(best_h, n_use, target_use, max_tries=3000, time_limit=min(25, tleft() * 0.5))
        if sc < best_score:
            best_score, best_h = sc, h_pert2

    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_optimize(best_h, n_use, target_use, [5000, 20000, 80000], 600)
        if sc < best_score:
            best_score, best_h = sc, h_final

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp4, sc = lp_refine(best_h, n_use, target_use, max_iters=100, time_limit=min(15, tleft() * 0.5), n_active_max=80)
        if sc < best_score:
            best_score, best_h = sc, h_lp4

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
