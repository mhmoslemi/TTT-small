        if time.time() - start > time_limit:
            break
        alpha = alpha_init * (alpha_final / alpha_init) ** (it / max(1, max_iter))

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        f_vals = corr * dx
        f_max = np.max(f_vals)
        weights = np.exp(alpha * (f_vals - f_max))
        weights /= weights.sum()

        top_k = np.argsort(weights)[-15:]
        grad = np.zeros(n)
        for ki in top_k:
            k = int(ki) - mid
            w = weights[int(ki)]
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += w * (1.0 - h_cur[idxs[v1] - k])
            grad[v2] -= w * h_cur[idxs[v2] + k]
        grad *= dx
        grad -= np.mean(grad)

        s = np.zeros(n)
        thresh = np.argpartition(grad, int(target))[:int(target)]
        s[thresh] = 1.0

        gamma = 2.0 / (it + 2.0)
        h_try = (1.0 - gamma) * h_cur + gamma * s
        h_try = project_feasible(h_try, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
        h_cur = h_try

    return best_h, best_c5


def local_perturbation_search(h_init, n, target, max_tries, time_limit, seed=12345):
    start = time.time()
    rng = np.random.RandomState(seed)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 4, 6, 8, 12]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.03, 0.05, 0.08, 0.12, 0.18]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
        elif rng.random() < 0.05:
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP to establish baseline
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [200, 1000, 5000], 1000, 1e-14)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Main pair CD (vectorized, efficient)
    if tleft() > 250:
        cd_time = min(300, tleft() * 0.35)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=cd_time,
                                         max_sweeps=6000, n_cand=45)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target, [1000, 5000, 20000, 80000], 1000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Second pair CD round with wider candidate set
    if tleft() > 150:
        cd_time = min(180, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                         time_limit=cd_time,
                                         max_sweeps=4000, n_cand=55)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: Frank-Wolfe for potential escape
    if tleft() > 80:
        h_fw, c5_fw = fw_smooth(best_h, n, target,
                                time_limit=min(50, tleft() * 0.25),
                                alpha_init=30.0, alpha_final=2000.0, max_iter=5000)
        if c5_fw < best_score:
            best_score = c5_fw
            best_h = h_fw
        if tleft() > 40:
            h_pol = slsqp_round(best_h, n, target, [2000, 10000, 50000], 800, 1e-15)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 6: Local perturbation + quick CD
    if tleft() > 100:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                    max_tries=3000,
                                                    time_limit=min(40, tleft() * 0.3),
                                                    seed=seed)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if tleft() > 50:
            cd_time = min(40, tleft() * 0.3)
            h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                             time_limit=cd_time,
                                             max_sweeps=2000, n_cand=40)
            if c5_cd < best_score:
                best_score = c5_cd
                best_h = h_cd

    # Phase 7: Multi-restart with small perturbations + SLSQP
    rng = np.random.RandomState(seed)
    for noise_scale in [0.01, 0.03, 0.06, 0.1]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref = vectorized_pair_cd(h_pert, n, target,
                                   time_limit=min(25, tleft() * 0.15),
                                   max_sweeps=1500, n_cand=35)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        if tleft() > 40:
            h_pol = slsqp_round(h_ref, n, target, [500, 2000, 8000], 600, 1e-14)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 8: Try higher resolution
    if tleft() > 150:
        for n_try in [500, 600]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [100, 500, 2000, 8000], 800, 1e-14)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try
            if tleft() > 60:
                target_use = len(h_hires) / 2.0
                cd_time = min(40, tleft() * 0.25)
                h_cd, c5_cd = vectorized_pair_cd(h_hires, len(h_hires), target_use,
                                                 time_limit=cd_time,
                                                 max_sweeps=2000, n_cand=35)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd

    # Phase 9: Final intensive pair CD at best resolution
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(45, tleft() * 0.5)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n_use, target_use,
                                         time_limit=cd_time,
                                         max_sweeps=3000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [5000, 20000, 100000], 600, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
