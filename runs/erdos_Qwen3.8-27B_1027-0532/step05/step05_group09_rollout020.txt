```python
def swap_delta(h, p, q):
    """Compute the change in corr_k when h[p] goes 1→0 and h[q] goes 0→1."""
    n = len(h)
    delta = np.zeros(n)
    # p loses its "1" contribution
    for k in range(1, min(n - p, n)):
        delta[k] -= (1.0 - h[p + k])
    # p gains "as target" contribution (was 1, now 0, so (1-h_p) goes 0→1)
    for k in range(1, min(p + 1, n)):
        if p - k >= 0 and (n - 1 - k) >= (p - k):  # index p-k is valid source
            delta[k] += h[p - k]
    # q gains its "1" contribution
    for k in range(1, min(n - q, n)):
        delta[k] += (1.0 - h[q + k])
    # q loses "as target" contribution (was 0, now 1, so (1-h_q) goes 1→0)
    for k in range(1, min(q + 1, n)):
        if q - k >= 0 and (n - 1 - k) >= (q - k):
            delta[k] -= h[q - k]
    return delta
```

```python
        new_max = float(np.max(new_corr))
        delta_max = new_max - cur_max
        if delta_max <= 0 or rng.random() < np.exp(-delta_max / max(T, 1e-10)):
            h[p] = 0.0
            h[q] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            corr = new_corr
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    return best_h, best_max * dx

def arc_polish(h, target, max_rounds, deadline, num_active=5, temp=0.01):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        sorted_idx = np.argsort(corr)[::-1]
        active = sorted_idx[:num_active]
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        improved = False
        for eps_try in [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps_try * grad
            h_new = proj(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                improved = True
                break
        if not improved:
            break
    return h_cur, cur_score

def block_cd(h, target, deadline, block_size=10):
    n = len(h)
    dx = 2.0 / n
    h_cur = proj(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(42)
    n_blocks = n // block_size
    remaining = n % block_size
    for sweep in range(5000):
        if time.time() > deadline:
            break
        order = rng.permutation(n_blocks)
        for bi in order:
            if time.time() > deadline:
                break
            start = bi * block_size
            end = min(start + block_size, n)
            block = h_cur[start:end].copy()
            best_block = block
            best_sc = cur_score
            for trial in range(8):
                if time.time() > deadline:
                    break
                pert = rng.randn(block.size) * 0.05
                new_block = np.clip(block + pert, 0.0, 1.0)
                h_trial = h_cur.copy()
                h_trial[start:end] = new_block
                h_trial = proj(h_trial, target)
                sc = compute_c5(h_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_block = h_trial[start:end].copy()
            if best_sc < cur_score:
                h_cur = h_cur.copy()
                h_cur[start:end] = best_block
                h_cur = proj(h_cur, target)
                cur_score = compute_c5(h_cur)
    return h_cur, cur_score

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    start_time = time.time()
    deadline_final = start_time + budget_s - 10.0

    n = len(initial_h_values)
    target = n / 2.0
    h_init = proj(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return deadline_final - time.time()

    # Phase 1: Quick FISTA from initial (30s)
    alpha1 = [100, 500, 2000]
    for a in alpha1:
        if tleft() < 15:
            break
        step = min(1.0 / (np.sqrt(a) * (2.0/n)), 5.0)
        h_new, sc_new = fista_fast(best_h, target, a, step, 30000, start_time + 12)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 2: Binary local search to find diverse starting points (60s)
    if tleft() > 60:
        binary_starts = []
        sa_deadline = start_time + 55
        # Start 1: from current best
        bin1, sc1 = binary_sa(best_h, target, sa_deadline, max_iters=100000)
        binary_starts.append((bin1, sc1))
        # Start 2: random binary
        rng = np.random.RandomState(seed)
        h_rand = np.zeros(n)
        ones_idx = rng.choice(n, n//2, replace=False)
        h_rand[ones_idx] = 1.0
        bin2, sc2 = binary_sa(h_rand, target, sa_deadline, max_iters=50000)
        binary_starts.append((bin2, sc2))
        # Start 3: structured (two blocks)
        h_struct = np.zeros(n)
        half = n // 4
        h_struct[:half] = 1.0
        h_struct[n//2:n//2+half] = 1.0
        if h_struct.sum() > target:
            h_struct = h_struct[:int(target)]
            h_struct = np.concatenate([h_struct, np.zeros(n - len(h_struct))])
        bin3, sc3 = binary_sa(h_struct, target, sa_deadline, max_iters=50000)
        binary_starts.append((bin3, sc3))
        # Refine best binary with FISTA
        for bi, (bh, bs) in enumerate(binary_starts):
            if tleft() < 20:
                break
            # Interpolate binary to continuous
            h_cont = bh.astype(float)
            h_cont = proj(h_cont, target)
            sc_cont = compute_c5(h_cont)
            step = min(1.0 / (np.sqrt(500) * (2.0/n)), 5.0)
            h_new, sc_new = fista_fast(h_cont, target, 500, step, 15000, start_time + 15)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 3: Main FISTA continuation (400s)
    alphas_main = [200, 800, 3000, 10000, 30000, 80000]
    iters_each = [8000, 8000, 10000, 10000, 8000, 8000]
    for i, (a, iters) in enumerate(zip(alphas_main, iters_each)):
        if tleft() < 30:
            break
        step = min(1.0 / (np.sqrt(a) * (2.0/n) * 0.8), 5.0)
        phase_deadline = start_time + min(70, tleft() - 5)
        h_new, sc_new = fista_fast(best_h, target, a, step, iters, phase_deadline)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        # Quick arc polish between FISTA phases
        if tleft() > 40:
            h_ap, sc_ap = arc_polish(best_h, target, 100, start_time + 20, num_active=5, temp=0.01)
            if sc_ap < best_score:
                best_score = sc_ap
                best_h = h_ap

    # Phase 4: Arc polish with various parameters (100s)
    if tleft() > 80:
        configs = [(3, 0.005), (5, 0.01), (8, 0.02), (10, 0.03), (5, 0.005), (3, 0.003)]
        for na, tp in configs:
            if tleft() < 15:
                break
            h_ap, sc_ap = arc_polish(best_h, target, 200, start_time + 15, num_active=na, temp=tp)
            if sc_ap < best_score:
                best_score = sc_ap
                best_h = h_ap

    # Phase 5: Perturbation restarts (200s)
    if tleft() > 150:
        rng = np.random.RandomState(seed + 1)
        pert_configs = [(0.03, 1000), (0.05, 800), (0.08, 600), (0.12, 400), (0.02, 1200)]
        for scale, max_it in pert_configs:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n) * scale
            h_pert = proj(h_pert, target)
            a = 5000
            step = min(1.0 / (np.sqrt(a) * (2.0/n) * 0.8), 3.0)
            h_new, sc_new = fista_fast(h_pert, target, a, step, max_it, start_time + 25)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
            # Also try binary perturbation
            if tleft() > 30:
                h_bin = (best_h > 0.5).astype(float)
                flip_count = int(n * 0.02)
                flip_idx = rng.choice(n, flip_count, replace=False)
                h_bin[flip_idx] = 1.0 - h_bin[flip_idx]
                h_bin = proj(h_bin, target)
                h_new2, sc_new2 = fista_fast(h_bin, target, 2000, 1.5, 5000, start_time + 15)
                if sc_new2 < best_score:
                    best_score = sc_new2
                    best_h = h_new2

    # Phase 6: High-alpha FISTA + polish (100s)
    if tleft() > 80:
        a = 100000
        step = min(1.0 / (np.sqrt(a) * (2.0/n) * 0.5), 2.0)
        h_new, sc_new = fista_fast(best_h, target, a, step, 20000, start_time + 40)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
        h_ap, sc_ap = arc_polish(best_h, target, 300, start_time + 30, num_active=4, temp=0.008)
        if sc_ap < best_score:
            best_score = sc_ap
            best_h = h_ap

    # Phase 7: Block coordinate descent (60s)
    if tleft() > 40:
        h_bd, sc_bd = block_cd(best_h, target, start_time + min(40, tleft()-5), block_size=8)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Final: one more arc polish
    if tleft() > 10:
        h_ap, sc_ap = arc_polish(best_h, target, 200, start_time + min(10, tleft()-2), num_active=5, temp=0.01)
        if sc_ap < best_score:
            best_score = sc_ap
            best_h = h_ap

    h_final = proj(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
