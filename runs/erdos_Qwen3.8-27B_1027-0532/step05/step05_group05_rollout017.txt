import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smooth_max_value(corr, alpha):
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            grad[l:] += w * (1.0 - h[:n-l])
            grad[:n-l] += w * (-h[l:])
        else:
            lp = -l
            grad[:n-lp] += w * (1.0 - h[lp:])
            grad[lp:] += w * (-h[:n-lp])
    
    return grad * (2.0 / n)

def lbfgs_optimize(h_init, target, alphas, maxiter, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if time.time() > deadline:
            break
        
        def obj(x):
            corr = compute_corr(x)
            return smooth_max_value(corr, a)
        
        def grad(x):
            return smooth_max_grad(x, a)
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score

def momentum_subgradient(h_init, target, n_steps, deadline, temperature=0.01, lr_init=0.005):
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n

        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))

            lr = min(lr * 1.005, 0.03)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_max_value(corr, alpha):
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()

    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            grad[l:] += w * (1.0 - h[:n - l])
            grad[:n - l] += w * (-h[l:])
        else:
            lp = -l
            grad[:n - lp] += w * (1.0 - h[lp:])
            grad[lp:] += w * (-h[:n - lp])

    return grad * (2.0 / n)


def lbfgs_optimize(h_init, target, alphas, maxiter, deadline):
    n = len(h_init)
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)

    for a in alphas:
        if time.time() > deadline:
            break

        def obj(x):
            corr = compute_corr(x)
            return smooth_max_value(corr, a)

        def grad(x):
            return smooth_max_grad(x, a)

        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass

    return h_cur, cur_score


def momentum_subgradient(h_init, target, n_steps, deadline, temperature=0.01, lr_init=0.005):
    n = len(h_init)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n

    velocity = np.zeros(n)
    lr = lr_init
    beta = 0.9

    for step in range(n_steps):
        if time.time() > deadline:
            break

        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))

        shifted = (corr - c_max) / temperature
        shifted = np.clip(shifted, -100.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()

        grad = np.zeros(n)
        for k_idx in range(2 * n - 1):
            w = weights[k_idx]
            if w < 1e-8:
                continue
            l = k_idx - (n - 1)
            if l >= 0:
                grad[l:] += w * (1.0 - h_cur[:n - l])
                grad[:n - l] += w * (-h_cur[l:])
            else:
                lp = -l
                grad[:n - lp] += w * (1.0 - h_cur[lp:])
                grad[lp:] += w * (-h_cur[:n - lp])
        grad *= dx

        velocity = beta * velocity - lr * grad
        h_new = h_cur + velocity
        h_new = project_feasible(h_new, target)

        sc_new = compute_c5(h_new)
        if sc_new < c_max * dx:
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
            lr = min(lr * 1.005, 0.03)
        else:
            lr *= 0.95
            velocity *= 0.5
            if lr < 1e-12:
                break

    return best_h, best_score

def top_k_equalize(h_init, target, max_steps, k, deadline):

    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n

        top_idx = np.argsort(corr)[::-1][:k]

        mean_val = float(np.mean(top_vals))
        spread = float(np.max(top_vals) - np.min(top_vals))

    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_parent)
    best_h_parent = h_parent.copy()
    best_score_parent = compute_c5(h_parent)

    n_work = 800
    target_work = n_work / 2.0
    h_work = interpolate_to(h_parent, n_work, target_work)
    best_h_work = h_work.copy()
    best_score_work = compute_c5(h_work)

    # Phase 1: Multi-scale warm start
    for n_scale in [80, 160, 320]:
        if tleft() < 70:
            break
        t = n_scale / 2.0
        h_s = interpolate_to(best_h_work, n_scale, t)
        dl = min(start_time + 20 + (n_scale // 80) * 15, hard_deadline)
        h_s, sc = lbfgs_optimize(h_s, t, [100, 500, 2000], 400, dl)
        h_s, sc = momentum_subgradient(h_s, t, 150, dl, temperature=0.02, lr_init=0.008)
        h_up = interpolate_to(h_s, n_work, target_work)
        sc_up = compute_c5(h_up)
        if sc_up < best_score_work:
            best_score_work = sc_up
            best_h_work = h_up

    # Phase 2: Main L-BFGS-B at n=800
    if tleft() > 250:
        h_work, sc = lbfgs_optimize(best_h_work, target_work,
                                    [200, 800, 3000, 10000], 1500,
                                    min(start_time + 250, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 3: Momentum subgradient
    if tleft() > 200:
        h_work, sc = momentum_subgradient(best_h_work, target_work, 600,
                                          min(start_time + 420, hard_deadline),
                                          temperature=0.008, lr_init=0.004)
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 4: Top-k equalization
    if tleft() > 150:
        h_work, sc = top_k_equalize(best_h_work, target_work, 400, 5,
                                    min(start_time + 520, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work
        h_work, sc = lbfgs_optimize(best_h_work, target_work,
                                    [5000, 20000], 1500,
                                    min(start_time + 580, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 5: Worst lag surgery
    if tleft() > 150:
        h_work, sc = worst_lag_surgery(best_h_work, target_work, 400,
                                       min(start_time + 650, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 6: Another round of refinement
    if tleft() > 200:
        h_work, sc = momentum_subgradient(best_h_work, target_work, 800,
                                          min(start_time + 780, hard_deadline),
                                          temperature=0.004, lr_init=0.003)
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work
        h_work, sc = lbfgs_optimize(best_h_work, target_work,
                                    [10000, 50000, 100000], 2000,
                                    min(start_time + 870, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 7: Top-k equalization round 2
    if tleft() > 120:
        h_work, sc = top_k_equalize(best_h_work, target_work, 300, 8,
                                    min(start_time + 930, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work
        h_work, sc = worst_lag_surgery(best_h_work, target_work, 200,
                                       min(start_time + 970, hard_deadline))
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Phase 8: Random restarts with structured perturbations
    if tleft() > 120:
        for attempt in range(12):
            if tleft() < 50:
                break
            r = rng.rand()
            if r < 0.3:
                shift = rng.randint(1, max(2, n_work // 50))
                h_pert = np.roll(best_h_work, shift)
            elif r < 0.6:
                freq = rng.uniform(2, 12)
                amp = rng.uniform(0.02, 0.08)
                x = np.linspace(0, 2, n_work)
                h_pert = best_h_work + amp * np.sin(freq * np.pi * x + rng.uniform(0, 2 * np.pi))
            else:
                h_pert = best_h_work + rng.randn(n_work) * rng.uniform(0.02, 0.06)

            h_pert = project_feasible(h_pert, target_work)
            dl = min(start_time + 100 + attempt * 45, hard_deadline)
            h_p, sc = lbfgs_optimize(h_pert, target_work, [500, 2000, 8000], 800, dl)
            if sc < best_score_work:
                best_score_work = sc
                best_h_work = h_p

    # Phase 9: Final polish
    if tleft() > 60:
        h_work, sc = momentum_subgradient(best_h_work, target_work, 400,
                                          min(start_time + budget_s - 50, hard_deadline),
                                          temperature=0.003, lr_init=0.002)
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    if tleft() > 30:
        h_work, sc = worst_lag_surgery(best_h_work, target_work, 150, hard_deadline)
        if sc < best_score_work:
            best_score_work = sc
            best_h_work = h_work

    # Compare parent vs work
    if best_score_parent < best_score_work:
        h_final = project_feasible(best_h_parent, target_parent)
        return h_final, compute_c5(h_final), n_parent

    h_final = project_feasible(best_h_work, target_work)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_work
