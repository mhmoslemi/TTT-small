    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    over = compute_over(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    obj = c_max + float(np.log(total)) / alpha
    weights = exp_s / total
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-16:
            continue
        wdx = w * dx
        if k == 0:
            grad[:] += wdx * (1.0 - 2.0 * h)
        else:
            grad[:n-k] += wdx * (1.0 - h[k:])
            grad[k:] -= wdx * h[:n-k]
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

_g_alpha = 100.0
_g_target = 200.0
_g_n = 400
_g_pen_mu = 0.0

def _obj(x):
    return lse_obj_grad(x, _g_alpha)[0]

def _grad(x):
    g = lse_obj_grad(x, _g_alpha)[1]
    if _g_pen_mu > 0:
        g = g + 2.0 * _g_pen_mu * (np.sum(x) - _g_target) * np.ones(_g_n)
    return g

def _obj_pen(x):
    base, _ = lse_obj_grad(x, _g_alpha)
    pen = _g_pen_mu * (np.sum(x) - _g_target)**2
    return base + pen

def _eq_con(x):
    return np.sum(x) - _g_target

def _eq_con_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_alpha, _g_target, _g_n, _g_pen_mu
    _g_n = n
    _g_target = target
    _g_pen_mu = 0.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj, h_cur, jac=_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def lbfgsb_optimize(h, n, target, alphas, maxiter, deadline):
    global _g_alpha, _g_target, _g_n, _g_pen_mu
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        # Adaptive penalty
        mu = 1.0
        for _ in range(20):
            if time.time() > deadline:
                break
            _g_pen_mu = mu
            try:
                res = minimize(_obj_pen, h_cur, jac=_grad, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
                # Check constraint satisfaction
                if abs(np.sum(h_cur) - target) < 1e-6 * n:
                    break
                mu *= 3.0
            except Exception:
                mu *= 3.0
    return h_cur, cur_score

def worst_shift_gradient_step(h, n, target, top_k, step_size, deadline):
    """Take gradient steps on the top-k worst shifts."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(2000):
        if time.time() > deadline:
            break
        over = compute_over(h_cur)
        top_idx = np.argsort(over)[::-1][:top_k]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            wdx = dx / top_k
            if k == 0:
                grad[:] += wdx * (1.0 - 2.0 * h_cur)
            else:
                grad[:n-k] += wdx * (1.0 - h_cur[k:])
                grad[k:] -= wdx * h_cur[:n-k]
        # Line search
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02]:
            h_new = project_feasible(h_cur - step_size * factor * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
                break
        else:
            # No improvement, try smaller step
            step_size *= 0.7
            if step_size < 1e-10:
                break
    return h_cur, cur_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array([project_feasible(p.astype(float), target) for p in init_pop]).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 6
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            elif r == 4:
                # Triangle wave
                tri = 1.0 - np.abs(2.0 * x / 2.0 - 1.0)
                pop[i] = project_feasible(tri + rng.randn(n) * 0.1, target)
            else:
                # Two-peak
                c1 = rng.uniform(0.3, 0.7)
                c2 = 1.0 - c1
                sigma = rng.uniform(0.05, 0.2)
                peak = 0.3 + 0.4 * np.exp(-((x-c1)**2)/sigma**2) + 0.4 * np.exp(-((x-c2)**2)/sigma**2)
                pop[i] = project_feasible(peak + rng.randn(n) * 0.05, target)
    
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            if len(avail) < 3:
                break
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score

def multi_start_slsqp(h_list, n, target, alphas, maxiter, ftol, deadline):

def alternating_shift_min(h, n, target, n_rounds, max_it_per_shift, deadline):

C5/dx = max_k Σ_{i=0}^{n-1-k} h[i](1-h[i+k])

Note that as k increases, the sum has fewer terms. So for large k, even if each term is large, the sum might be small. The maximum is likely achieved at small k (where there are many terms) or at k ≈ n/2 (where the "overlap region" is largest in some sense).

Actually, for k=0: Σ h[i](1-h[i]) = Σ h[i] - Σ h[i]² = n/2 - Σ h[i]²
This is maximized when h is as "spread" as possible (all 0.5).
It's minimized when h is as "concentrated" as possible (0s and 1s).

For k=n/2: Σ_{i=0}^{n/2-1} h[i](1-h[i+n/2])
This measures the "complementarity" between the first half and second half.

- h[i](1-h[i]) to be small (h close to 0 or 1 at each point)
- h[i](1-h[i+k]) to be small for all k (h "shifted by k" should be complementary to h)

- d/dh[j] from h[i] term: (1-h[j+k]) for j ∈ [0, n-1-k]
- d/dh[j] from (1-h[i+k]) term: -h[j-k] for j ∈ [k, n-1]

Combined:
- j ∈ [0, min(k, n-1-k)): only first term: (1-h[j+k])
- j ∈ [k, n-1-k]: both terms: (1-h[j+k]) - h[j-k] (when k ≤ n-1-k, i.e., 2k ≤ n-1)
- j ∈ [max(k, n-k), n-1]: only second term: -h[j-k]
- If k > (n-1)/2: the "both terms" region is empty, and the first region is [0, n-k), second is [k, n-1]

My code handles this correctly with:
```python
grad[:n-k] += wdx * (1.0 - h[k:])  # first term for j = 0, ..., n-k-1
grad[k:] -= wdx * h[:n-k]  # second term for j = k, ..., n-1
```

This is correct! The overlap region [k, n-k-1] gets both contributions, and the non-overlap regions get only one.

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 100.0
_g_target = 200.0
_g_n = 400
_g_pen_mu = 0.0


def compute_over(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:2 * n - 1]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_over(h))) * dx


def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    over = compute_over(h)
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    total = float(np.sum(exp_s))
    obj = c_max + np.log(total) / alpha
    weights = exp_s / total
    grad = np.zeros(n)
    for k in range(n):
        w = weights[k]
        if w < 1e-16:
            continue
        wdx = w * dx
        if k == 0:
            grad[:] += wdx * (1.0 - 2.0 * h)
        else:
            grad[:n - k] += wdx * (1.0 - h[k:])
            grad[k:] -= wdx * h[:n - k]
    return obj, grad


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _obj_func(x):
    return lse_obj_grad(x, _g_alpha)[0]


def _grad_func(x):
    g = lse_obj_grad(x, _g_alpha)[1]
    if _g_pen_mu > 0:
        g = g + 2.0 * _g_pen_mu * (np.sum(x) - _g_target) * np.ones(_g_n)
    return g


def _obj_pen(x):
    base, _ = lse_obj_grad(x, _g_alpha)
    pen = _g_pen_mu * (np.sum(x) - _g_target) ** 2
    return base + pen


def _eq_con(x):
    return np.sum(x) - _g_target


def _eq_con_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_alpha, _g_target, _g_n, _g_pen_mu
    _g_n = n
    _g_target = target
    _g_pen_mu = 0.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_con_jac}]
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_obj_func, h_cur, jac=_grad_func, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def lbfgsb_optimize(h, n, target, alphas, maxiter, deadline):
    global _g_alpha, _g_target, _g_n, _g_pen_mu
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        mu = 0.5
        for _ in range(15):
            if time.time() > deadline:
                break
            _g_pen_mu = mu
            try:
                res = minimize(_obj_pen, h_cur, jac=_grad_func, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    h_cur = h_new
                    cur_score = sc
                if abs(np.sum(h_cur) - target) < 1e-6 * n:
                    break
                mu *= 2.5
            except Exception:
                mu *= 2.5
    return h_cur, cur_score


def worst_shift_optimize(h, n, target, n_focus, max_iter, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for it in range(max_iter):
        if time.time() > deadline:
            break
        over = compute_over(h_cur)
        top_idx = np.argsort(over)[::-1][:n_focus]
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            if k == 0:
                grad[:] += (1.0 - 2.0 * h_cur)
            else:
                grad[:n - k] += (1.0 - h_cur[k:])
                grad[k:] -= h_cur[:n - k]
        grad *= dx / len(top_idx)
        step = 0.5
        improved = False
        for _ in range(12):
            h_new = project_feasible(h_cur - step * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
                improved = True
                break
            step *= 0.5
        if not improved:
            step = 0.01
            h_new = project_feasible(h_cur - step * grad, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
            else:
                break
    return h_cur, cur_score


def alternating_shift_min(h, n, target, n_rounds, n_top, step_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    step = step_init
    for round_ in range(n_rounds):
        if time.time() > deadline:
            break
        over = compute_over(h_cur)
        order = np.argsort(over)[::-1][:n_top]
        for k in order:
            if time.time() > deadline:
                break
            k = int(k)
            grad = np.zeros(n)
            if k == 0:
                grad[:] = (1.0 - 2.0 * h_cur) * dx
            else:
                grad[:n - k] = (1.0 - h_cur[k:]) * dx
                grad[k:] = -h_cur[:n - k] * dx
            for _ in range(30):
                if time.time() > deadline:
                    break
                h_new = project_feasible(h_cur - step * grad, target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
                    step = min(step * 1.05, 0.5)
                else:
                    step *= 0.5
                    if step < 1e-8:
                        break
    return h_cur, cur_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array([project_feasible(p.astype(float), target) for p in init_pop]).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0.0, 2.0, n)
        for i in range(pop_size):
            r = i % 6
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.4)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            elif r == 4:
                tri = 1.0 - np.abs(2.0 * (x / 2.0 - 0.5))
                pop[i] = project_feasible(tri + rng.randn(n) * 0.08, target)
            else:
                c1 = rng.uniform(0.25, 0.75)
                c2 = 1.0 - c1
                sigma = rng.uniform(0.06, 0.18)
                peak = 0.3 + 0.4 * np.exp(-((x - c1) ** 2) / (sigma ** 2)) + 0.4 * np.exp(-((x - c2) ** 2) / (sigma ** 2))
                pop[i] = project_feasible(peak + rng.randn(n) * 0.04, target)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            if len(avail) < 3:
                break
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r_arr = rng.rand(n)
            trial[r_arr > cr] = pop[i][r_arr > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def anneal_low_res(h_parent, n, target, n_steps, t_init, t_final, deadline, rng):
    h_cur = interpolate_to(h_parent, n, target).copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    t = t_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        t = t_init * (t_final / t_init) ** (step / max(n_steps - 1, 1))
        idx = rng.randint(n)
        delta = rng.randn() * 0.15
        h_trial = h_cur.copy()
        h_trial[idx] = np.clip(h_trial[idx] + delta, 0.0, 1.0)
        h_trial = project_feasible(h_trial, target)
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            best_h = h_trial.copy()
        if sc <= compute_c5(h_cur) or rng.rand() < np.exp(-(sc - compute_c5(h_cur)) / (t + 1e-10)):
            h_cur = h_trial
    return best_h, best_score
