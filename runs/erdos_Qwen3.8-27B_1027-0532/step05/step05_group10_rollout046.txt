```python
    # min c^T h + mu * h^T h s.t. sum(h) = target, 0 <= h <= 1
    # KKT: c + 2*mu*h = lambda + nu - wi
    # where nu >= 0 (for h <= 1), wi >= 0 (for h >= 0)
    # h[i] = clip((lambda - c[i]) / (2*mu), 0, 1)
    # Find lambda such that sum(h) = target
    
    lo = c.min() - 2*mu  # h[i] <= 1 => lambda - c[i] <= 2*mu => lambda <= c[i] + 2*mu
    hi = c.max() + 2*mu
    # Actually: h[i] = clip((lambda - c[i])/(2*mu), 0, 1)
    # h[i] = 0 when lambda <= c[i]
    # h[i] = 1 when lambda >= c[i] + 2*mu
    # h[i] = (lambda - c[i])/(2*mu) when c[i] < lambda < c[i] + 2*mu
    
    # Bisection on lambda
    lo = c.min()
    hi = c.max() + 2*mu
    for _ in range(100):
        mid = (lo + hi) / 2
        h_try = np.clip((mid - c) / (2*mu), 0, 1)
        s = h_try.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip((lo - c) / (2*mu), 0, 1)
    # Fix sum exactly
    diff = target - h.sum()
    if diff != 0:
        # Adjust the fractional elements
        frac_idx = np.where((h > 0) & (h < 1))[0]
        if len(frac_idx) > 0:
            h[frac_idx] += diff / len(frac_idx)
            h = np.clip(h, 0, 1)
    return h
```

```python
import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_linear_reg(c, target, mu):

    for _ in range(80):

def smooth_obj_lse(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.25]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 80:
            break

    return best_h, best_c5


def alternating_lp_constraint_gen(h_init, n, target, max_iter, time_limit, damping_init, n_active_init, n_active_max):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active_init:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        t_val = t_var.value if t_var.value is not None else best_c5

        new_violated = []
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k:
                if corr[k_idx] * dx > t_val + dx * 1e-3:
                    new_violated.append(k_idx)
                    if len(new_violated) >= 10:
                        break

        for kv in new_violated:
            active_k.add(kv)

        top_k = set(np.argsort(corr)[-n_active_max:].tolist())
        active_k = top_k | active_k
        if len(active_k) > n_active_max:
            active_k = set(np.argsort(corr)[-n_active_max:].tolist())

        if it % 30 == 0:
            damping = min(0.95, damping + 0.01)
            if damping > 0.9:
                damping = 0.9

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP warmup
    if tleft() > 60:
        h_ref = slsqp_optimize(best_h, n, target, [50, 200], 500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Multiplicative weights + QP (main method, n=400)
    if tleft() > 200:
        mw_time = min(300, tleft() * 0.3)
        h_mw, c5_mw = multiplicative_weights(
            best_h, n, target,
            n_iter=50000, time_limit=mw_time,
            alpha_init=10.0, mu_init=0.05, damping_init=0.7
        )
        if c5_mw < best_score:
            best_score = c5_mw
            best_h = h_mw

    # Phase 3: SLSQP polish
    if tleft() > 80:
        h_ref = slsqp_optimize(best_h, n, target, [200, 1000, 3000], 1500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: LW + QP with different params
    if tleft() > 150:
        mw_time = min(200, tleft() * 0.3)
        h_mw, c5_mw = multiplicative_weights(
            best_h, n, target,
            n_iter=50000, time_limit=mw_time,
            alpha_init=30.0, mu_init=0.02, damping_init=0.8
        )
        if c5_mw < best_score:
            best_score = c5_mw
            best_h = h_mw

    # Phase 5: LP constraint generation
    if tleft() > 150:
        lp_time = min(180, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp_constraint_gen(
            best_h, n, target,
            max_iter=5000, time_limit=lp_time,
            damping_init=0.7, n_active_init=40, n_active_max=80
        )
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 6: SLSQP
    if tleft() > 80:
        h_ref = slsqp_optimize(best_h, n, target, [1000, 5000, 15000], 1500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Try higher resolution n=600
    if tleft() > 250:
        n2 = 600
        target2 = n2 / 2.0
        h_interp = interpolate_to(best_h, n2, target2)
        sc = compute_c5(h_interp)
        if sc < best_score - 0.005:
            h_mw, c5_mw = multiplicative_weights(
                h_interp, n2, target2,
                n_iter=50000, time_limit=min(150, tleft() * 0.4),
                alpha_init=10.0, mu_init=0.05, damping_init=0.7
            )
            sc2 = compute_c5(h_mw)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_mw
                n = n2
                target = target2

            if tleft() > 60:
                h_ref = slsqp_optimize(best_h, n, target, [500, 2000], 1000, 1e-14)
                sc3 = compute_c5(h_ref)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_ref

    # Phase 8: More MW at current best resolution
    if tleft() > 100:
        mw_time = min(120, tleft() * 0.4)
        h_mw, c5_mw = multiplicative_weights(
            best_h, n, target,
            n_iter=50000, time_limit=mw_time,
            alpha_init=50.0, mu_init=0.01, damping_init=0.85
        )
        if c5_mw < best_score:
            best_score = c5_mw
            best_h = h_mw

    # Phase 9: Subgradient refinement
    if tleft() > 40:
        h_sg, c5_sg = subgradient_improve(
            best_h, n, target,
            max_steps=5000, time_limit=min(30, tleft() * 0.5)
        )
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 10: SLSQP final polish
    if tleft() > 60:
        h_ref = slsqp_optimize(best_h, n, target, [3000, 10000, 30000], 1000, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: LP final
    if tleft() > 60:
        lp_time = min(50, tleft() * 0.6)
        h_lp, c5_lp = alternating_lp_constraint_gen(
            best_h, n, target,
            max_iter=2000, time_limit=lp_time,
            damping_init=0.9, n_active_init=60, n_active_max=100
        )
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Perturbed restarts
    for noise in [0.02, 0.05, 0.1]:
        if tleft() < 50:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        if tleft() > 30:
            h_ref = slsqp_optimize(h_pert, n, target, [200, 1000, 5000], 600, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 13: Final MW + SLSQP
    if tleft() > 60:
        h_mw, c5_mw = multiplicative_weights(
            best_h, n, target,
            n_iter=30000, time_limit=min(40, tleft() * 0.5),
            alpha_init=100.0, mu_init=0.005, damping_init=0.9
        )
        if c5_mw < best_score:
            best_score = c5_mw
            best_h = h_mw
        if tleft() > 30:
            h_ref = slsqp_optimize(best_h, n, target, [10000, 50000], 500, 1e-16)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Final check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
```
