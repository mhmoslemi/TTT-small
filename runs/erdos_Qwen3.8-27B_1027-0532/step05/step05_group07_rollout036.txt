import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_full(h, m_star):
    """Compute subgradient of corr[m_star] w.r.t. h"""
    n = len(h)
    shift = m_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:-shift]) if shift > 0 else (1.0 - h)
            g[:n-shift] -= h[shift:] if shift > 0 else h
    else:
        s = -shift
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    return g

def fw_vertex(g, target):
    """Find the FW vertex (vertex of the polytope minimizing <g, v>)"""
    n = len(g)
    idx = np.argsort(g)
    s = np.zeros(n)
    k = int(target)
    s[idx[:k]] = 1.0
    frac = target - k
    if frac > 1e-12 and k < n:
        s[idx[k]] = frac
    return s

def frank_wolfe_optimize(h0, n, target, deadline, rng, max_iters=50000):
    """Frank-Wolfe optimization with line search"""
    h = h0.copy()
    dx = 2.0 / n
    best_score = compute_c5(h)
    best_h = h.copy()
    
    # Track active set for away steps
    active_set = set()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h)
        m_star = int(np.argmax(corr))
        g = subgradient_full(h, m_star) * dx
        
        # FW vertex
        s = fw_vertex(g, target)
        
        # Line search along h + γ(s - h)
        direction = s - h
        best_g = -1.0
        best_score_ls = best_score
        
        # Try several step sizes
        for gamma in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1, 1.0/(it+2)]:
            h_try = h + gamma * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score_ls:
                best_score_ls = sc
                best_g = gamma
        
        if best_g > 0:
            h = h + best_g * direction
            h = project_feasible(h, target)
        
        # Update best
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_vec(h, m_star):
    n = len(h)
    shift = m_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def fw_vertex(g, target):
    n = len(g)
    idx = np.argsort(g)
    s = np.zeros(n)
    k = int(target)
    s[idx[:k]] = 1.0
    frac = target - k
    if frac > 1e-12 and k < n:
        s[idx[k]] = frac
    return s

def min_max_linear(base, coef, lo=0.0, hi=1.0, iters=50):

        mid = (lo + hi) * 0.5
        vals = base + mid * coef
        max_val = np.max(vals)
        active = np.where(np.abs(vals - max_val) < 1e-12)[0]
        if len(active) == 0:

        c_min = np.min(coef[active])
        c_max = np.max(coef[active])
        if c_min <= 1e-14 and c_max >= -1e-14:

        else:

        base = corr - h * 0.0

    return np.sum(x) - _g_target

    return np.ones(_g_n)

def slsqp_polish(h0, n, target, deadline, alphas=None, maxiter=3000):
    global _g_n, _g_target, _g_alpha
    if alphas is None:
        alphas = [200, 1000, 5000, 20000]
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h = h0.copy()
    cur_score = compute_c5(h)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(obj_lse, h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h = h_new
                cur_score = sc
        except Exception:
            pass

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
