import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    omega = h_sorted - np.arange(n)
    # find largest j (1-indexed) such that omega[j-1] > 0
    j = np.searchsorted(-omega, 0.0, side='right')
    if j == 0:
        lo, hi = h.min() - 1.0, h.max() + 1.0
        for _ in range(80):
            mid = (lo + hi) / 2
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        theta = (lo + hi) / 2
    else:
        theta = (h_sorted[:j].sum() - target) / j
    return np.clip(h - theta, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def exact_subgrad(h, top_k=1):
    """Compute subgradient of max of top_k correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    # Get top_k indices
    idx = np.argsort(corr)[::-1][:top_k]
    # Weight by closeness to max
    diffs = corr[idx] - corr[idx[0]]
    w = np.exp(diffs / 0.01)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        ki = int(ki)
        jp = np.arange(0, n - ki)
        grad[jp] += wi * (1.0 - h[jp + ki])
        jn = np.arange(ki, n)
        grad[jn] -= wi * h[jn - ki]
    grad *= dx
    return grad, corr[idx[0]] * dx

def fista_lse(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        
        # Backtracking line search
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step / (1.0 - shrink), lr_init * 2.0)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    omega = h_sorted - np.arange(n).astype(float)
    j = np.searchsorted(-omega, 0.0, side='right')
    if j == 0:
        lo = h.min() - 1.0
        hi = h.max() + 1.0
        for _ in range(80):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        theta = (lo + hi) / 2.0
    else:
        theta = (h_sorted[:j].sum() - target) / j
    return np.clip(h - theta, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_lse(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if lr_init is None:
        lr_init = 1.0 / max(alpha * 0.05, 1.0)
    step = lr_init
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 2.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def dcorr_swap(h, i, j):
    n = len(h)
    dcorr = np.zeros(n)
    fwd_i = (1.0 - h)[i:]
    dcorr[:len(fwd_i)] -= fwd_i
    bwd_i = h[i::-1]
    dcorr[:len(bwd_i)] += bwd_i
    fwd_j = (1.0 - h)[j:]
    dcorr[:len(fwd_j)] += fwd_j
    bwd_j = h[j::-1]
    dcorr[:len(bwd_j)] -= bwd_j
    return dcorr

def sa_binary(h_init, target, deadline, seed=42):

    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:].copy()
    cur_score = float(np.max(corr)) * dx
    best_score = cur_score
    best_h = h.copy()
    start_t = time.time()
    elapsed = start_t
    total_time = deadline - elapsed
    T_start = 0.005 * dx
    T_end = 0.0003 * dx
    it = 0

    ones_list = list(np.where(h == 1)[0])
    zeros_list = list(np.where(h == 0)[0])
    while time.time() < deadline:
        it += 1
        frac = min(it / max_iters, 1.0)
        T = T_start * (T_end / T_start) ** frac
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        dcorr = dcorr_swap(h, i, j)
        new_corr = corr + dcorr
        new_score = float(np.max(new_corr)) * dx
        if new_score <= cur_score:
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            cur_score = new_score
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            accept_prob = np.exp((cur_score - new_score) / max(T, 1e-15))

                h[i] = 0.0
                h[j] = 1.0
                corr = new_corr
                cur_score = new_score
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)

def local_search_binary(h, target, deadline, seed=123):

def coord_descent(h, target, deadline, max_sweeps=300):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(777)
    candidates = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            if abs(old_val) < 1e-9 or abs(old_val - 1.0) < 1e-9:
                cands = [1.0 - old_val]
            else:
                cands = [0.0, 1.0, 0.5, 0.25, 0.75]
            best_cand = None
            best_sc = cur_score
            for c in cands:
                if abs(c - old_val) < 1e-9:
                    continue
                h_cur[idx] = c
                h_p = project_feasible(h_cur, target)
                sc = compute_c5(h_p)
                if sc < best_sc:
                    best_sc = sc
                    best_cand = h_p.copy()
            if best_cand is not None:
                h_cur = best_cand
                cur_score = best_sc
            else:
                h_cur[idx] = old_val
                h_cur = project_feasible(h_cur, target)
    return h_cur, cur_score

def block_sa(h_init, target, deadline, seed=555):

    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()

    cur_score = float(np.max(corr)) * dx
    best_score = cur_score
    best_h = h.copy()
    block_size = 3
    n_blocks = n // block_size
    T_start = 0.003 * dx
    T_end = 0.0002 * dx
    it = 0
    max_iters = 300000
    while time.time() < deadline:
        it += 1
        frac = min(it / max_iters, 1.0)
        T = T_start * (T_end / T_start) ** frac
        b = rng.randint(n_blocks)
        start = b * block_size
        end = start + block_size
        old_block = h[start:end].copy()
        new_block = np.clip(old_block + rng.randn(block_size) * 0.3, 0.0, 1.0)
        h[start:end] = new_block
        h_p = project_feasible(h, target)
        new_corr = np.correlate(h_p, 1.0 - h_p, mode='full')[n-1:]
        new_score = float(np.max(new_corr)) * dx
        if new_score <= cur_score:
            h = h_p
            corr = new_corr
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()

            accept_prob = np.exp((cur_score - new_score) / max(T, 1e-15))

                h = h_p
                corr = new_corr
                cur_score = new_score
