This is still O(n_candidates * n^2) per iteration which is too slow. Let me optimize the swap evaluation.

Key optimization: When we swap h[i] from 1→0 and h[j] from 0→1, only the correlations at shifts affected by i and j change. Specifically:
- h[i] affects corr at shifts where i is the first or second factor
- h[j] similarly

For a shift k (offset from n-1), corr[k+(n-1)] involves h[t] for t where both t and t+k are in range. The change at h[i] affects corr[n-1+k] for all k such that i is in the valid range, i.e., k in [-(n-1-i), n-1-i].

So the number of affected shifts is at most 2n-1 for each of i and j, giving at most 4n-2 affected shifts total. But we can compute the delta for each affected shift in O(1).

Here's the optimized evaluation:

```python
def eval_swap_fast(h, i, j, corr):
    """Evaluate new max after swapping h[i]=1->0 and h[j]=0->1.
    Returns the new maximum correlation value."""
    n = len(h)
    # The shifts affected by position i: k ranges from -(n-1-i) to (n-1-i)
    # The shifts affected by position j: k ranges from -(n-1-j) to (n-1-j)
    
    # For shift k, corr[n-1+k] = sum_t h[t]*(1-h[t+k])
    # Change from h[i]: 1->0:
    #   - As first factor: lose (1-h[i+k]) if 0<=i+k<n
    #   - As second factor: was (1-1)*h[i-k]=0, now (1-0)*h[i-k]=h[i-k] if 0<=i-k<n
    # Change from h[j]: 0->1:
    #   - As first factor: gain (1-h[j+k]) if 0<=j+k<n
    #   - As second factor: was (1-0)*h[j-k]=h[j-k], now (1-1)*h[j-k]=0 if 0<=j-k<n
    
    # Affected shifts: k where any of i+k, i-k, j+k, j-k is in [0, n-1]
    # That's k in [-(n-1-i), n-1-i] union [-(n-1-j), n-1-j]
    
    k_min = max(-(n-1), -max(n-1-i, n-1-j))  # = -max(n-1-i, n-1-j) clamped to -(n-1)
    k_max = min(n-1, max(n-1-i, n-1-j))
    
    # For efficiency, only check a window of shifts
    # The new max can only be at a shift that was already high OR is affected by the swap
    # So we check: all shifts in the affected range, plus the current top shifts
    
    # Actually, the simplest correct approach:
    # new_corr[s] = corr[s] + delta[s] for affected s, else corr[s]
    # new max = max(max(affected new values), max(unaffected old values))
    
    # Unaffected max: we can precompute this
    # But it's complex. Let's just compute the full thing for the affected shifts.
    
    # For a practical implementation, let's compute the delta for all shifts in the affected range
    # and take the max of (corr + delta) over affected shifts, and max(corr) over unaffected shifts.
    
    # The unaffected max is at most the current max (since we're only changing some shifts)
    # Unless all shifts at the current max are affected...
    
    # Simplest: just compute max over ALL shifts, but only actually changing the affected ones
    # For n=606, there are 1211 shifts. Computing delta for each is O(1).
    # So total is O(n) per swap evaluation.
    
    new_max = 0.0
    for s in range(2 * n - 1):
        k = s - (n - 1)
        delta = 0.0
        if 0 <= i + k < n:
            delta -= (1.0 - h[i + k])
        if 0 <= i - k < n:
            delta += h[i - k]
        if 0 <= j + k < n:
            delta += (1.0 - h[j + k])
        if 0 <= j - k < n:
            delta -= h[j - k]
        val = corr[s] + delta
        if val > new_max:
            new_max = val
    return new_max
```

But this is O(n) = O(1211) per swap in Python, which is slow. Let me vectorize it:

```python
def eval_swap_fast_vec(h, i, j, corr):
    n = len(h)
    ks = np.arange(2*n-1) - (n-1)  # all shift values -k to k
    delta = np.zeros(2*n-1)
    
    # Contribution from i (1->0):
    # First factor: -(1-h[i+k]) for valid i+k
    mask = (ks >= -i) & (ks <= n-1-i)
    delta[mask] -= (1.0 - h[i + ks[mask]])
    # Second factor: +h[i-k] for valid i-k
    mask2 = (ks >= i-(n-1)) & (ks <= i)
    delta[mask2] += h[i - ks[mask2]]
    
    # Contribution from j (0->1):
    # First factor: +(1-h[j+k]) for valid j+k
    mask3 = (ks >= -j) & (ks <= n-1-j)
    delta[mask3] += (1.0 - h[j + ks[mask3]])
    # Second factor: -h[j-k] for valid j-k
    mask4 = (ks >= j-(n-1)) & (ks <= j)
    delta[mask4] -= h[j - ks[mask4]]
    
    new_corr = corr + delta
    return new_corr.max()
```

This should be fast in numpy. For n=606, the arrays are length 1211, so this is very fast.

    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:

        if need <= len(ones):

            h_r[ones[order]] = 0
    else:

            final_sum -= 1
        else:

Minimize t

        allowed = min(30, tleft() * 0.05)
        h_g, c5_g = _greedy(best_h, allowed, start_time, rng, n_ci=30, n_cj=60)
        if c5_g < best_score:
            best_score = c5_g
            best_h = h_g.copy()

    # Phase 2: Main SA
    if tleft() > 200:
        allowed = min(250, tleft() * 0.35)
        h_sa, c5_sa = _sa_single(best_h, allowed, start_time, rng, t_init=5.0, t_final=0.03)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Greedy polish after SA
    if tleft() > 40:
        allowed = min(25, tleft() * 0.04)
        h_g2, c5_g2 = _greedy(best_h, allowed, start_time, rng, n_ci=35, n_cj=70)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

    # Phase 4: ILS with multiple restarts
    if tleft() > 100:
        n_restarts = 25
        per_restart = max(3.0, (tleft() * 0.45) / n_restarts)
        for r in range(n_restarts):
            if tleft() < per_restart + 5:
                break
            n_pert = 2 + (r % 5)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                h_pert[ii] = 0
                h_pert[jj] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            allowed_g = min(3.0, per_restart * 0.4)
            h_pg, c5_pg = _greedy(h_pert, allowed_g, start_time, rng, n_ci=25, n_cj=50)
            allowed_sa = min(per_restart - allowed_g - 0.5, tleft() - 5)
            if allowed_sa > 2:
                h_ps, c5_ps = _sa_single(h_pg, allowed_sa, start_time, rng, t_init=3.0, t_final=0.05)
                if c5_ps < c5_pg:
                    h_pg = h_ps
                    c5_pg = c5_ps
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

    # Phase 5: Second main SA from best
    if tleft() > 150:
        allowed = min(180, tleft() * 0.3)
        h_sa2, c5_sa2 = _sa_single(best_h, allowed, start_time, rng, t_init=6.0, t_final=0.02)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 6: Greedy polish
    if tleft() > 30:
        allowed = min(20, tleft() * 0.05)
        h_g3, c5_g3 = _greedy(best_h, allowed, start_time, rng, n_ci=40, n_cj=80)
        if c5_g3 < best_score:
            best_score = c5_g3
            best_h = h_g3.copy()

    # Phase 7: More ILS with different perturbation
    if tleft() > 80:
        n_restarts = 20
        per_restart = max(3.0, (tleft() * 0.3) / n_restarts)
        for r in range(n_restarts):
            if tleft() < per_restart + 5:
                break
            n_pert = 1 + (r % 3)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = int(ones[rng.randint(len(ones))])
                jj = int(zeros[rng.randint(len(zeros))])
                h_pert[ii] = 0
                h_pert[jj] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            allowed_g = min(2.5, per_restart * 0.35)
            h_pg, c5_pg = _greedy(h_pert, allowed_g, start_time, rng, n_ci=20, n_cj=45)
            allowed_sa = min(per_restart - allowed_g - 0.5, tleft() - 5)
            if allowed_sa > 2:
                h_ps, c5_ps = _sa_single(h_pg, allowed_sa, start_time, rng, t_init=2.5, t_final=0.08)
                if c5_ps < c5_pg:
                    h_pg = h_ps
                    c5_pg = c5_ps
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg.copy()

    # Phase 8: SLSQP continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        alphas = [20, 50, 100, 300, 1000, 3000, 10000, 30000]
        h_ref, sc = _slsqp_refine(best_h, alphas, 3000, 1e-15,
                                  min(tleft() - 20, 200), start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 9: Binary project and final polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin = project_binary(best_h, target_use)
        allowed = min(15, tleft() * 0.1)
        h_gf, c5_gf = _greedy(h_bin, allowed, start_time, rng, n_ci=30, n_cj=60)
        if c5_gf < best_score:
            best_score = c5_gf
            best_h = h_gf.copy()

    # Phase 10: Final SA on best binary
    if tleft() > 25:
        n_use = len(best_h)
        allowed = max(5.0, tleft() - 10)
        h_sf, c5_sf = _sa_single(best_h, allowed, start_time, rng, t_init=2.0, t_final=0.1)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf.copy()

    # Phase 11: Final greedy
    if tleft() > 10:
        allowed = max(3.0, tleft() - 5)
        h_gfinal, c5_final = _greedy(best_h, allowed, start_time, rng, n_ci=40, n_cj=80)
        if c5_final < best_score:
            best_score = c5_final
            best_h = h_gfinal.copy()

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
