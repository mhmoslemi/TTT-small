            # d(corr[k])/dx[s] = (1-x[s+k]) for s as first factor + (-x[s-k]) for s as second factor

            # And the total gradient is sum_k weights[k] * d(corr[k])/dx[s]
            
            # Alternative: use the same structure as compute_delta but for the gradient
            # grad[s] = sum_k weights[k] * [(1-x[s+k]) - x[s-k]] (with validity)
            #         = sum_{k: 0<=s+k<n} weights[k] * (1-x[s+k]) - sum_{k: 0<=s-k<n} weights[k] * x[s-k]
            
            # First term: for each s, sum over k in [-s, n-1-s] of weights[n-1+k] * (1-x[s+k])
            # Let's compute this directly
            for s in range(n):

                k_range = np.arange(max(-(n-1), -s), min(n-1, n-1-s)+1)
                valid_k = k_range
                idx1 = s + valid_k
                w1 = weights[n-1+valid_k]
                grad[s] += np.sum(w1 * (1 - x[idx1]))

    result[valid] += h[idx[valid]]
    idx = j + k
    valid = (idx >= 0) & (idx < n)
    result[valid] += 1.0 - h[idx[valid]]
    idx = j - k
    valid = (idx >= 0) & (idx < n)
    result[valid] -= h[idx[valid]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_ls(h, n, rng, max_iters, time_limit, start_time, top_k=40, rand_trials=80):
    h = h.copy().astype(float)
    corr = fast_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    max_stall = 200
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        max_val = float(corr.max())
        max_idx = int(np.argmax(corr))
        k_star = max_idx - (n-1)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] -= h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(top_k, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(top_k, len(zeros))]]
        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, int(i), int(j), n)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[int(i)] = 0.0
                    h[int(j)] = 1.0
                    corr += delta
                    stall = 0
                    new_c5 = new_max * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break
        if not found:
            for _ in range(rand_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, int(i), int(j), n)
                new_max = float((corr + delta).max())
                if new_max < max_val:
                    h[int(i)] = 0.0
                    h[int(j)] = 1.0
                    corr += delta
                    stall = 0
                    new_c5 = float(corr.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    return best_h, best_c5

def sa_focused_tabu(h, n, n_iters, t_init, t_final, rng, time_limit, start_time, tabu_len=100, focus_prob=0.55):
    h = h.copy().astype(float)
    corr = fast_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    positions = np.arange(n)
    tabu = []
    tabu_set = set()
    stall = 0
    max_stall = 60000
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < focus_prob:
            top5_idx = np.argsort(corr)[-5:]
            max_idx = int(top5_idx[rng.randint(5)])
            k_star = max_idx - (n-1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] -= h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                j_cands = zeros[B[zeros] >= -1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        i, j = int(i), int(j)
        if (i, j) in tabu_set or (j, i) in tabu_set:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            i, j = int(i), int(j)
        delta = compute_delta(h, i, j, n)
        new_max = float((corr + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        accept = False
        if delta_c5 < 0:
            accept = True
        elif t > 1e-15:
            if rng.random() < np.exp(-delta_c5 / t):
                accept = True
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr += delta
            current_c5 = c5_new
            tabu.append((i, j))
            tabu_set.add((i, j))
            if len(tabu) > tabu_len:
                old = tabu.pop(0)
                tabu_set.discard(old)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def multi_top_greedy(h, n, rng, max_iters, time_limit, start_time, n_top=3):
    h = h.copy().astype(float)
    corr = fast_corr(h)
    dx = 2.0 / n
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 300:
            break
        top_indices = np.argsort(corr)[-n_top:]
        found_any = False
        for max_idx in top_indices:
            k_star = int(max_idx) - (n-1)
            if k_star == 0:
                continue
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            A = np.zeros(n)
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            B = np.zeros(n)
            idx = positions - k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + k_star
            valid = (idx >= 0) & (idx < n)
            B[valid] -= h[idx[valid]]
            top_i = ones[np.argsort(A[ones])[:min(25, len(ones))]]
            top_j = zeros[np.argsort(-B[zeros])[:min(25, len(zeros))]]
            max_val = float(corr.max())
            for i in top_i:
                for j in top_j:
                    delta = compute_delta(h, int(i), int(j), n)
                    new_max = float((corr + delta).max())
                    if new_max < max_val:
                        h[int(i)] = 0.0
                        h[int(j)] = 1.0
                        corr += delta
                        stall = 0
                        new_c5 = new_max * dx
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        found_any = True
                        break
                if found_any:
                    break
            if found_any:
                break
        if not found_any:
            stall += 1
    return best_h, best_c5

def _smooth_obj(h, alpha, target):
    corr = fast_corr(h)
    c_max = float(corr.max())
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
    penalty = 50.0 * (np.sum(h) - target) ** 2
    return val + penalty

def continuous_refine(h_start, n, target, time_limit, start_time):
    h = h_start.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alphas = [30, 80, 200, 500, 1200, 3000]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h, args=(alpha, float(target)),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, float(target))
            sc = compute_c5(h_new)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coarse_to_fine(h_coarse, n_fine, target_fine):
    n_coarse = len(h_coarse)
    ratio = n_fine // n_coarse
    h_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = start + ratio
        if end > n_fine:
            end = n_fine
        if ratio == 1:
            h_fine[i] = h_coarse[i]
        elif h_coarse[i] > 0.5:
            h_fine[start:end] = 1.0
        else:
            h_fine[start:end] = 0.0
    return project_binary(h_fine, target_fine)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)
    n_best = n_init

    # Phase 1: Greedy polish
    allowed = min(40, tleft() * 0.10)
    h_g, c5_g = greedy_ls(best_h, n_init, rng, 50000, allowed, start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Multi-top greedy
    allowed = min(30, tleft() * 0.08)
    h_mt, c5_mt = multi_top_greedy(best_h, n_init, rng, 30000, allowed, start_time)
    if c5_mt < best_score:
        best_score = c5_mt
        best_h = h_mt.copy()

    # Phase 3: Main ILS loop with focused SA + tabu
    n_ils = 20
    for it in range(n_ils):
        if tleft() < 80:
            break
        # Focused SA with tabu
        allowed = min(60, tleft() * 0.15)
        n_sa = int(min(3000000, allowed * 40000))
        t_init_sa = 0.015
        t_final_sa = 0.00003
        h_sa, c5_sa = sa_focused_tabu(best_h, n_best, n_sa, t_init_sa, t_final_sa,
                                      rng, allowed, start_time, tabu_len=80, focus_prob=0.5)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

        if tleft() < 60:
            break

        # Greedy polish
        allowed = min(20, tleft() * 0.06)
        h_g2, c5_g2 = greedy_ls(best_h, n_best, rng, 20000, allowed, start_time)
        if c5_g2 < best_score:
            best_score = c5_g2
            best_h = h_g2.copy()

        if tleft() < 50:
            break

        # Perturb
        n_pert = 3 + it // 3
        if n_pert > 12:
            n_pert = rng.randint(3, 8)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[int(i)] = 0.0
            h_pert[int(j)] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        # SA from perturbed with slightly higher temperature
        if tleft() < 40:
            break
        allowed = min(45, tleft() * 0.12)
        n_sa2 = int(min(2000000, allowed * 35000))
        h_sa2, c5_sa2 = sa_focused_tabu(h_pert, n_best, n_sa2, 0.02, 0.00005,
                                        rng, allowed, start_time, tabu_len=60, focus_prob=0.4)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    # Phase 4: Coarse-to-fine at n_init
    if tleft() > 120:
        n_coarse = max(100, n_init // 4)
        if n_coarse % 2 != 0:
            n_coarse += 1
        target_coarse = n_coarse // 2
        h_coarse = interpolate_to(best_h, n_coarse, float(target_coarse))
        h_coarse_bin = project_binary(h_coarse, target_coarse)
        allowed = min(25, tleft() * 0.10)
        h_cg, c5_cg = greedy_ls(h_coarse_bin, n_coarse, rng, 10000, allowed * 0.4, start_time)
        allowed2 = min(20, tleft() * 0.08)
        n_sac = int(min(300000, allowed2 * 40000))
        h_cs, c5_cs = sa_focused_tabu(h_cg, n_coarse, n_sac, 0.01, 0.0001,
                                       rng, allowed2, start_time, tabu_len=40)
        if c5_cs < c5_cg:
            h_coarse_best = h_cs
        else:
            h_coarse_best = h_cg
        h_fine = coarse_to_fine(h_coarse_best, n_init, target_init)
        allowed3 = min(30, tleft() * 0.10)
        h_fg, c5_fg = greedy_ls(h_fine, n_init, rng, 15000, allowed3 * 0.5, start_time)
        allowed4 = min(25, tleft() * 0.08)
        n_saf = int(min(400000, allowed4 * 35000))
        h_fs, c5_fs = sa_focused_tabu(h_fg, n_init, n_saf, 0.01, 0.00003,
                                       rng, allowed4, start_time, tabu_len=60)
        if c5_fs < c5_fg:
            h_embed = h_fs
            c5_embed = c5_fs
        else:
            h_embed = h_fg
            c5_embed = c5_fg
        if c5_embed < best_score:
            best_score = c5_embed
            best_h = h_embed.copy()

    # Phase 5: Try different n values
    for n_try in [n_init + 4, n_init + 8, n_init - 4, n_init + 12, n_init - 8]:
        if tleft() < 60:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, float(target_try))
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(20, tleft() * 0.07)
        h_g3, c5_g3 = greedy_ls(h_bin_try, n_try, rng, 8000, allowed, start_time)
        allowed2 = min(15, tleft() * 0.05)
        n_sa3 = int(min(200000, allowed2 * 35000))
        h_s3, c5_s3 = sa_focused_tabu(h_g3, n_try, n_sa3, 0.008, 0.00005,
                                       rng, allowed2, start_time, tabu_len=40)
        if c5_s3 < c5_g3:
            h_try = h_s3
            c5_try = c5_s3
        else:
            h_try = h_g3
            c5_try = c5_g3
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()
            n_best = n_try

    # Phase 6: Second ILS round at best n
    if tleft() > 150:
        for it in range(8):
            if tleft() < 60:
                break
            allowed = min(50, tleft() * 0.15)
            n_sa = int(min(2500000, allowed * 40000))
            h_sa, c5_sa = sa_focused_tabu(best_h, n_best, n_sa, 0.012, 0.00002,
                                          rng, allowed, start_time, tabu_len=120, focus_prob=0.6)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()
            if tleft() < 40:
                break
            allowed = min(15, tleft() * 0.05)
            h_g2, c5_g2 = greedy_ls(best_h, n_best, rng, 15000, allowed, start_time)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()
            n_pert = rng.randint(4, 10)
            h_pert = best_h.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[int(i)] = 0.0
                h_pert[int(j)] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            if tleft() < 30:
                break
            allowed = min(40, tleft() * 0.10)
            n_sa2 = int(min(1500000, allowed * 35000))
            h_sa2, c5_sa2 = sa_focused_tabu(h_pert, n_best, n_sa2, 0.018, 0.00004,
                                            rng, allowed, start_time, tabu_len=50, focus_prob=0.45)
            if c5_sa2 < best_score:
                best_score = c5_sa2
                best_h = h_sa2.copy()

    # Phase 7: Continuous refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(tleft() - 20, 120)
        h_ref, sc = continuous_refine(best_h, n_use, target_use, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final binary SA burst
    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        c5_bin = compute_c5(h_bin_final)
        n_sa_f = int(min(800000, max(20000, (tleft() - 10) * 40000)))
        h_sa_f, c5_f = sa_focused_tabu(h_bin_final, n_use, n_sa_f, 0.006, 0.00001,
                                       rng, max(5.0, tleft() - 10), start_time, tabu_len=60, focus_prob=0.7)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        elif c5_bin < best_score:
            best_score = c5_bin
            best_h = h_bin_final

    # Phase 9: Final continuous refinement if time
    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = max(5.0, tleft() - 8)
        h_ref, sc = continuous_refine(best_h, n_use, target_use, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
