def project_feasible(h, target_sum):
    # Project h onto {0<=h<=1, sum(h)=target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)

            break

            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def subgradient_min_max(h, target, n_iter, top_k=5, eta0=0.01, decay=0.999):

            best_h = h_new.copy()
        h_cur = h_new
        eta *= decay
    return best_h, best_score

            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    s = np.sum(x)
    grad += 2.0 * mu * (s - target)
    return grad

def lbfgs_optimize(h, target, alpha, mu, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return lse_values(x, alpha, target, mu)
    def g(x):
        return lse_grad(x, alpha, target, mu)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h

def subgradient_topk(h, target, n_iter, top_k, eta0, decay, deadline):
    n = len(h)
    h_cur = h.copy()
    eta = eta0
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for ki in idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad /= top_k
        h_new = h_cur - eta * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
        eta *= decay
    return best_h, best_score

def top_k_equalize(h, target, n_top, n_steps, lr, deadline):

                    grad[j2] -= w * h_cur[j2 + lag]

                m = -lag

                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)

            h_cur = h_new
            cur_score = sc
        else:

    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    return np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))

    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    return val

        for ki in top_idx:
            w = float(corr[ki] - top_mean)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return h_cur, cur_score
