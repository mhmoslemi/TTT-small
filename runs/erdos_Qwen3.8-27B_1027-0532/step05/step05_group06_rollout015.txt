def project_feasible(h, target_sum):
    # Project onto {0<=h<=1, sum(h)=target_sum}
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
        else:
            mask = h > 0.0
            cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def get_max_corr_grad(h, top_k=1):
    """Get gradient of max (or sum of top-k) of correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top-k
    top_idx = np.argsort(corr)[::-1][:top_k]
    weights = np.zeros(len(corr))
    for ki in top_idx:
        weights[ki] = 1.0 / top_k
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for ki in top_idx:
        w = 1.0 / top_k
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - h[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - h[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
    return grad, corr

def nesterov_subgradient(h, target, n_iter, deadline, top_k=3, lr_init=0.01):

    h_y = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)[0]

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def top_k_subgradient(h, top_k=3):
    """Compute subgradient of max over top-k correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for ki in top_idx:
        w = 1.0 / top_k
        lag = int(ki) - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                grad[j] += w * (1.0 - h[j - lag])
            for j in range(n - lag):
                grad[j] -= w * h[j + lag]
        else:
            m = -lag
            for j in range(n - m):
                grad[j] += w * (1.0 - h[j + m])
            for j in range(m, n):
                grad[j] -= w * h[j - m]
    return grad, corr

def nesterov_explore(h, target, n_iter, deadline, top_k=3):
    """Fast subgradient exploration with Nesterov acceleration."""
    n = len(h)
    h_x = h.copy()
    h_y = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    lr_base = 0.02
    counter = 0
    
    for i in range(n_iter):
        if time.time() > deadline:
            break
        
        t = i + 1
        # Nesterov extrapolation
        if i == 0:
            h_z = h_y.copy()
        else:
            beta = (1.0 - np.sqrt(1.0 + 4.0 * (i) ** 2)) / 2.0 + i  # ~ i/2 for large i
            # Simpler: use standard Nesterov
            # y_{k+1} = x_{k+1} + (k/(k+2))(x_{k+1} - x_k)
            # Let me use a simpler momentum
            h_z = h_y + (0.9) * (h_y - h_prev)
            h_z = project_feasible(h_z, target)
        
        grad, corr = top_k_subgradient(h_z, top_k)
        lr = lr_base / np.sqrt(t)
        
        h_new = h_z - lr * grad
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            counter = 0
        else:
            counter += 1
            if counter > 1000:
                lr_base *= 0.5
                counter = 0
        
        h_prev = h_y
        h_y = h_new
        h_x = h_new
    
    return best_h, best_score

def exact_cd_1d(h, j, target, n_sweeps_inner=1):
    n = len(h)
    old_val = h[j]
    corr = np.correlate(h, 1.0 - h, mode='full')
    B = np.zeros(2 * n - 1)
    for ki in range(2 * n - 1):
        lag = ki - (n - 1)
        b = 0.0
        jk = j + lag
        if 0 <= jk < n:
            b += (1.0 - h[jk])
        jk = j - lag
        if 0 <= jk < n:
            b -= h[jk]
        B[ki] = b
    A = corr - B * old_val
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = A + B * mid
        max_val = vals.max()
        active = np.where(np.abs(vals - max_val) < 1e-14)[0]
        if len(active) == 0:
            active = np.where(vals == max_val)[0]
        min_B = B[active].min()
        max_B = B[active].max()
        if min_B <= 1e-10 and max_B >= -1e-10:
            return mid
        elif max_B < 0:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def lag_equalization(h, target, top_k, n_steps, deadline):
