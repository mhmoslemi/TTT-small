def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)

                    x_opt = xs2[best_local2]

            h[i] = x_opt

            delta = x_opt - old_val

            sweep_improved = True

            no_improve += 1

def find_min_of_upper_envelope(slopes, intercepts):
    """
    Find x in [0,1] minimizing f(x) = max_j (slopes[j]*x + intercepts[j]).
    slopes and intercepts are 1D arrays of the same length m.
    Returns the optimal x.
    """
    m = len(slopes)
    if m == 0:
        return 0.0
    if m == 1:
        if slopes[0] >= 0:
            return 0.0
        else:
            return 1.0
    
    # Sort by slope
    order = np.argsort(slopes)
    s = slopes[order]
    b = intercepts[order]
    
    # Build upper envelope using a stack
    # The envelope is the upper hull in the (x, f(x)) plane
    # Each line is (s_i, b_i): f_i(x) = s_i * x + b_i
    # The upper envelope from left to right uses lines with increasing slopes
    
    # Stack of (slope, intercept, x_right) where x_right is the right boundary
    stack_s = []
    stack_b = []
    stack_x = []  # left boundary for each line on the envelope
    
    for i in range(m):
        si, bi = s[i], b[i]
        if stack_s:
            # Find the intersection of the new line with the last line on the stack
            # si*x + bi = stack_s[-1]*x + stack_b[-1]
            # x = (stack_b[-1] - bi) / (si - stack_s[-1])
            if si == stack_s[-1]:
                # Same slope: keep the one with higher intercept
                if bi > stack_b[-1]:
                    # Replace the last one
                    # The new line is above, so it takes over from the left boundary of the last
                    stack_b[-1] = bi
                    # Check if it should replace earlier ones too
                    # Actually, since we process in order of increasing slope,
                    # and we're replacing with a higher intercept, the left boundary stays
                    continue
                else:
                    continue  # New line is below, skip
            x_int = (stack_b[-1] - bi) / (si - stack_s[-1])
            # The new line becomes active from x_int onwards
            # But we need to check if x_int is to the right of the previous left boundary
            if stack_x:
                if x_int <= stack_x[-1]:
                    # The new line makes the last one irrelevant, pop it
                    stack_s.pop()
                    stack_b.pop()
                    stack_x.pop()
                    if not stack_s:
                        stack_s.append(si)
                        stack_b.append(bi)
                        stack_x.append(-np.inf)
                    else:
                        # Re-compute intersection with new last
                        x_int2 = (stack_b[-1] - bi) / (si - stack_s[-1])
                        if si == stack_s[-1]:
                            # Shouldn't happen after sorting
                            continue
                        stack_s.append(si)
                        stack_b.append(bi)
                        stack_x.append(x_int2)
                else:
                    stack_s.append(si)
                    stack_b.append(bi)
                    stack_x.append(x_int)
            else:
                stack_s.append(si)
                stack_b.append(bi)
                stack_x.append(x_int)
        else:
            stack_s.append(si)
            stack_b.append(bi)
            stack_x.append(-np.inf)
    
    # Now find the minimum of the envelope on [0,1]
    # The envelope is made of segments, each with a constant slope
    # The minimum is at the point where slope crosses 0, or at an endpoint
    
    # Evaluate at x=0 and x=1, and at all breakpoints in (0,1)
    candidates = [0.0, 1.0]
    for j in range(len(stack_x)):
        if 0.0 < stack_x[j] < 1.0:
            candidates.append(stack_x[j])
    
    # Also, the minimum could be at a breakpoint where slope changes from negative to positive
    best_x = 0.0
    best_val = np.inf
    for xc in candidates:
        val = np.max(s * xc + b)  # max over all lines (not just envelope, same result)
        if val < best_val - 1e-15:
            best_val = val
            best_x = xc
    
    return best_x

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def find_optimal_1d(slopes, intercepts):
    m = len(slopes)
    if m == 0:
        return 0.5
    if m == 1:
        return 0.0 if slopes[0] >= 0 else 1.0
    xs = np.linspace(0.0, 1.0, 400)
    vals = np.max(slopes[:, None] * xs[None, :] + intercepts[:, None], axis=0)
    best_idx = np.argmin(vals)
    lo = max(0.0, xs[max(0, best_idx - 1)])
    hi = min(1.0, xs[min(len(xs) - 1, best_idx + 1)])
    if hi - lo < 1e-12:
        return xs[best_idx]
    xs2 = np.linspace(lo, hi, 400)
    vals2 = np.max(slopes[:, None] * xs2[None, :] + intercepts[:, None], axis=0)
    return float(xs2[np.argmin(vals2)])


def cyclic_coordinate_descent(h_init, n, target, max_sweeps, time_limit,
                              seed=42, order_mode='random'):
    rng = np.random.RandomState(seed)
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    C = compute_c5_vector(h)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if order_mode == 'random':
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for i in order:
            k_min = max(-(n - 1), max(-i, i - n + 1))
            k_max = min(n - 1, min(n - 1 - i, i))
            if k_min > k_max:
                continue
            k_range = np.arange(k_min, k_max + 1)
            slopes = np.zeros(len(k_range), dtype=float)
            mask1 = (k_range >= -i) & (k_range <= n - 1 - i)
            if np.any(mask1):
                ix1 = k_range[mask1] + i
                slopes[mask1] += (1.0 - h[ix1])
            mask2 = (k_range >= i - n + 1) & (k_range <= i)
            if np.any(mask2):
                ix2 = i - k_range[mask2]
                slopes[mask2] += (-h[ix2])
            k_indices = k_range + (n - 1)
            intercepts = C[k_indices] - slopes * h[i]
            x_opt = find_optimal_1d(slopes, intercepts)
            old_val = h[i]
            delta = x_opt - old_val
            if abs(delta) > 1e-15:
                h[i] = x_opt
                C[k_indices] += delta * slopes

        h = project_feasible(h, target)
        C = compute_c5_vector(h)
        c5_val = float(np.max(C))
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
    return best_h, best_c5


def build_lp_constraints(h, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def get_top_lag_indices(h, K):
    overlaps = compute_c5_vector(h)
    n = len(h)
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:K]
    lags = top_idx - n_half
    return lags


def alternating_lp_refined(h_init, n, target, max_iter, time_limit,
                           n_top=180, damping_start=0.95, damping_end=0.80,
                           reg_start=1e-4, reg_end=1e-8):

    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 200 == 0 and stuck > 500:
            break
        g = 1.0 - h
        lags = get_top_lag_indices(h, n_top)
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        h_var, t_var, constraints = build_lp_constraints(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)

def smooth_obj(h, alpha):

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target

    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000, ftol=1e-14):

    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,

            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

            best_score = c5_final
            best_h = h_cd_final

    # Phase 10: Final LP micro-polish
    if tleft() > 40:
        h_lp_final, c5_lp_final = alternating_lp_refined(best_h, n, target,
                                                         max_iter=2000,
                                                         time_limit=min(35, tleft() * 0.8),
                                                         n_top=100, damping_start=0.95,
                                                         damping_end=0.9, reg_start=1e-6,
                                                         reg_end=1e-10)
        if c5_lp_final < best_score:
            best_score = c5_lp_final
            best_h = h_lp_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
