def active_set_direction(h, top_k_indices, K=5):
    n = len(h)
    # For each active lag, compute the gradient
    # grad_k = A_k^T * (h_k_shifted) where A_k is the shift matrix
    # Actually, dC/dh[j] for lag k:
    # If j < n-k: dC/dh[j] = (1 - h[j+k])
    # If j >= k: dC/dh[j] -= h[j-k]
    
    grads = np.zeros((K, n))
    for i, k in enumerate(top_k_indices):
        k = int(k)
        for j in range(n - k):
            grads[i, j] += (1.0 - h[j + k])
        for j in range(k, n):
            grads[i, j] -= h[j - k]
    
    # We want to minimize max_i (grads[i] · d)
    # s.t. ||d||^2 <= 1 (or some bound)
    # This is a convex problem
    # Equivalently: minimize t
    # s.t. grads[i] · d <= t for all i
    #      ||d|| <= 1
    
    # This is an SOCP, but we can solve it as a sequence of QPs or use scipy
    # For simplicity, use scipy.optimize.minimize
    
    # Actually, for K small (5), we can use a simpler approach:
    # The optimal d is the negative of the weighted sum of active gradients
    # where weights come from solving a small LP
    # 
    # LP: minimize c
    # s.t. sum_i w_i * (grad_i · d) <= 0 (for some d)
    #      w_i >= 0, sum w_i = 1
    #
    # Actually, the optimal direction for minimizing max_i f_i is:
    # d* = -argmax_{||d||=1} min_i (-grad_i · d)
    # 
    # This is equivalent to finding the minimum norm point in the convex hull of {grad_i}
    # d* = -(g - p) where p is the projection of 0 onto conv{grad_i}
    
    # For K gradients in R^n, find p = argmin ||p||^2 s.t. p in conv{grad_1, ..., grad_K}
    # This is a QP with K constraints
    
    # Use scipy.optimize.minimize to solve this small QP
    from scipy.optimize import minimize as sp_min
    
    # p = sum_i w_i * grad_i, w_i >= 0, sum w_i = 1
    # minimize ||p||^2 = w^T (G^T G) w where G is (K x n) matrix of gradients
    
    G = grads  # K x n
    M = G @ G.T  # K x K
    
    def qobj(w):
        return w @ M @ w
    
    def qgrad(w):
        return 2 * M @ w
    
    w0 = np.ones(K) / K
    cons = [{'type': 'eq', 'fun': lambda w: w.sum() - 1, 'jac': lambda w: np.ones(K)}]
    bounds = [(0, 1)] * K
    
    res = sp_min(qobj, w0, jac=qgrad, method='SLSQP', bounds=bounds, constraints=cons)
    w_opt = np.clip(res.x, 0, 1)
    w_opt /= w_opt.sum()
    
    d = -grads.T @ w_opt
    return d, w_opt

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    # Binary search for tau such that sum(clip(h - tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    fsize = 2 * n
    Wf = np.fft.rfft(weights[::-1].copy(), fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad, corr

def fista_optimize(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    
    lr_init = 1.0 / (max(alpha * 0.01, 1.0))
    step = lr_init
    no_improve = 0
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y, _ = lse_obj_and_grad(y_clipped, alpha)
        
        # Backtracking line search
        shrink = 0.5
        for _ in range(30):
            x_new = project_feasible(y_clipped - step * grad_y, target)
            obj_new, _, _ = lse_obj_and_grad(x_new, alpha)
            if obj_new <= obj_y - 0.5 * step * np.dot(grad_y, x_new - y_clipped):
                break
            step *= shrink
        else:
            x_new = project_feasible(y_clipped - step * grad_y, target)
        
        step = min(step / shrink, lr_init * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        
        sc = compute_c5(x)
        if sc < best_obj - 1e-12:
            best_obj = sc
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Restart momentum
                x_prev = x.copy()
                t = 1.0
                no_improve = 0
                step = lr_init
    
    return best_x, best_obj

def active_set_step(h, target, K=5, deadline=None):
    """Find optimal direction using active set QP and take a step."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    
    # Find top K lags
    top_k = np.argsort(corr)[::-1][:K]
    corr_top = corr[top_k]
    corr_max = corr_top[0]
    
    # Compute gradients for active lags
    grads = np.zeros((K, n))
    for i, k in enumerate(top_k):
        k = int(k)
        for j in range(n - k):
            grads[i, j] += (1.0 - h[j + k])
        for j in range(k, n):
            grads[i, j] -= h[j - k]
    grads *= dx
    
    # Solve QP: min w^T M w s.t. sum(w)=1, w>=0
    # where M = G G^T
    M = grads @ grads.T
    K_actual = len(M)
    
    from scipy.optimize import minimize as sp_min
    w0 = np.ones(K_actual) / K_actual
    bounds = [(0, 1)] * K_actual
    cons = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1.0, 'jac': lambda w: np.ones(K_actual)}]
    
    def qobj(w):
        return float(w @ M @ w)
    def qgrad(w):
        return 2.0 * (M @ w)
    
    res = sp_min(qobj, w0, jac=qgrad, method='SLSQP', bounds=bounds, constraints=cons,
                 options={'maxiter': 100, 'ftol': 1e-12})
    w_opt = np.clip(res.x, 0, 1)
    w_sum = w_opt.sum()
    if w_sum > 1e-10:
        w_opt /= w_sum
    else:
        w_opt = w0
    
    # Direction: negative weighted sum of gradients
    d = -grads.T @ w_opt
    
    # Project direction to be orthogonal to constraint (sum = n/2)
    d = d - np.mean(d)  # This makes sum(d) = 0
    
    # Line search
    best_h = h.copy()
    best_sc = compute_c5(h)
    for alpha_ls in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        h_new = project_feasible(h + alpha_ls * d, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best_h = h_new
    
    return best_h, best_sc

```python
def solve_equalities_qp(M, K):
    """Solve min w^T M w s.t. sum(w) = 1, w >= 0 using active set method."""
    n_active = list(range(K))
    w = np.ones(K)
    
    for _ in range(K + 1):
        # Solve equality-constrained QP on active set
        Ma = M[np.ix_(n_active, n_active)]
        # KKT: 2 Ma w + lambda * 1 = 0, sum(w) = 1
        # w = -Ma^{-1} * 1 * lambda / 2
        # -lambda/2 * sum(Ma^{-1} * 1) + 1 = 0
        # lambda = -2 / sum(Ma^{-1} * 1)
        ones = np.ones(len(n_active))
        try:
            Ma_inv_ones = np.linalg.solve(Ma, ones)
            s = np.sum(Ma_inv_ones)
            if abs(s) < 1e-15:
                w_active = ones / len(ones)
            else:
                w_active = -Ma_inv_ones / s
        except np.linalg.LinAlgError:
            w_active = ones / len(ones)
        
        w = np.zeros(K)
        for i, idx in enumerate(n_active):
            w[idx] = w_active[i]
        
        # Check feasibility
        if np.min(w) >= -1e-12:
            w = np.clip(w, 0, None)
            w /= w.sum()
            return w
        
        # Find most negative w and fix it to 0
        worst = np.argmin(w)
        n_active = [i for i in n_active if i != worst]
    
    # Fallback
    w = np.zeros(K)
    for idx in n_active:
        w[idx] = 1.0 / len(n_active)
    return w
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    cmax = float(corr.max())
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    exps = np.exp(shifted)

    w = exps / s

    # Gradient via FFT

    V = np.fft.rfft(1.0 - h, fs)
    W = np.fft.rfft(w, fs)
    H = np.fft.rfft(h, fs)
    g1 = np.fft.irfft(Wf * V, fs)[n-1:2*n-1]
    g2 = np.fft.irfft(W * H, fs)[:n]
