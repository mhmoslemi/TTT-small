def coord_min_max(a, b, lo=0.0, hi=1.0, n_iter=50):

    # Evaluate at boundaries
    f_lo = np.max(a * lo + b)
    f_hi = np.max(a * hi + b)
    
    # Check if minimum is at boundary
    # For convex f, if f is decreasing at lo, min is not at lo
    # If f is increasing at hi, min is not at hi
    # Simple approach: just do ternary search on [lo, hi]
    
    l, r = lo, hi
    for _ in range(n_iter):
        m1 = l + (r - l) / 3.0
        m2 = r - (r - l) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 <= f2:
            r = m2
        else:
            l = m1
    x_opt = (l + r) / 2.0
    return x_opt

def coordinate_descent_optimize(h_init, n, target, n_sweeps, time_limit, 
                                 n_iter_1d=50, random_order=True, seed=0):

    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()

    dx = 2.0 / n

    c_active = full_corr * dx  # all lags
    n_lags = len(full_corr)
    lags = np.arange(-(n-1), n)  # actual lag values
    
    for sweep in range(n_sweeps):
        if time.time() - start > time_limit:
            break
        
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for p in order:
            # Compute a_lag for all lags
            # For lag ell: a = (1-h[p+ell]) * I(-p <= ell < n-p) + (-h[p-ell]) * I(p-n < ell <= p)
            a = np.zeros(n_lags)
            
            # First term: valid where -p <= ell < n-p, i.e., index in [p, n+p) in lags array
            # lags array is [-(n-1), ..., -1, 0, 1, ..., n-1]
            # ell >= -p means index >= n-1-p (since lags[n-1-p] = -(n-1)-(-p) = p-(n-1)... 
            # Let me just compute directly
            
            ell_neg = lags  # this is the array of actual lag values
            mask1 = (ell_neg >= -p) & (ell_neg < n - p)
            mask2 = (ell_neg > p - n) & (ell_neg <= p)
            
            if np.any(mask1):
                idx1 = p + ell_neg[mask1]
                a[mask1] = 1.0 - h[idx1]
            if np.any(mask2):
                idx2 = p - ell_neg[mask2]
                a[mask2] = a[mask2] - h[idx2]
            
            b = c_active - a * h[p]
            x_star = coord_min_max(a, b)
            h[p] = x_star
        
        # Project to satisfy sum
        h = project_feasible(h, target)
        
        # Recompute correlation
        full_corr = np.correlate(h, 1.0 - h, mode='full')
        c_active = full_corr * dx
        
        c5_val = float(np.max(full_corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_min_max(a, b, n_iter=50):
    """Minimize max(a*x + b) for x in [0,1]."""
    # Check boundaries first
    f0 = np.max(a * 0.0 + b)
    f1 = np.max(a * 1.0 + b)
    
    # Ternary search on [0, 1]
    l, r = 0.0, 1.0
    for _ in range(n_iter):
        m1 = l + (r - l) / 3.0
        m2 = r - (r - l) / 3.0
        f_m1 = np.max(a * m1 + b)
        f_m2 = np.max(a * m2 + b)
        if f_m1 <= f_m2:
            r = m2
        else:
            l = m1

                       use_random_order=True, proj_every=1):

    best_c5 = compute_c5(h)
    dx = 2.0 / n

    precomp, lags = precompute_masks(n)

    c_vals = full_corr * dx

            if len(i2) > 0:
                a[m2_idx] = a[m2_idx] - h[i2]

            x_star = coord_min_max(a, b, n_iter=n_iter_1d)
            delta = x_star - old_hp

    _global_target = target

            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

```python
import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(40):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def coord_min_max(a, b, n_iter=40):
    l = 0.0
    r = 1.0
    for _ in range(n_iter):
        m1 = l + (r - l) / 3.0
        m2 = r - (r - l) / 3.0
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 <= f2:
            r = m2
        else:
            l = m1
    return (l + r) / 2.0


def precompute_masks(n):
    lags = np.arange(-(n - 1), n)
    precomp = []
    for p in range(n):
        m1 = (lags >= -p) & (lags < n - p)
        m2 = (lags > p - n) & (lags <= p)
        m1_idx = np.where(m1)[0]
        m2_idx = np.where(m2)[0]
        i1 = p + lags[m1] if len(m1_idx) > 0 else np.array([], dtype=np.int64)
        i2 = p - lags[m2] if len(m2_idx) > 0 else np.array([], dtype=np.int64)
        precomp.append((m1_idx, m2_idx, i1, i2))
    return precomp


def coordinate_descent(h_init, n, target, time_limit, n_iter_1d=40, seed=0,
                       proj_every=30, lambda_lr=0.005, random_order=True):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    precomp = precompute_masks(n)
    full_corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = full_corr * dx
    lam = 0.0
    sweep = 0
    stall = 0

    while time.time() - start < time_limit:
        if random_order:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for cnt, p in enumerate(order):
            if time.time() - start > time_limit:
                break
            m1_idx, m2_idx, i1, i2 = precomp[p]
            a_orig = np.zeros(n_lags)
            if len(i1) > 0:
                a_orig[m1_idx] = 1.0 - h[i1]
            if len(i2) > 0:
                a_orig[m2_idx] = a_orig[m2_idx] - h[i2]
            old_hp = h[p]
            b_vec = c_vals - a_orig * old_hp
            a_eff = a_orig + lam
            x_star = coord_min_max(a_eff, b_vec, n_iter=n_iter_1d)
            delta = x_star - old_hp
            if abs(delta) > 1e-16:
                h[p] = x_star
                c_vals += delta * a_orig

        s = np.sum(h) - target
        lam += lambda_lr * s
        lam = np.clip(lam, -0.5, 0.5)

        if sweep % proj_every == 0:
            h = project_feasible(h, target)
            full_corr = np.correlate(h, 1.0 - h, mode='full')
            c_vals = full_corr * dx
            lam = 0.0

            stall = 0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target

    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter=1500, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
