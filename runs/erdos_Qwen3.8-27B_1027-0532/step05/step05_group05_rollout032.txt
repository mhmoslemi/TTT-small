    global _g_n
    _g_n = n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for p in p_list:
        if time.time() > deadline:
            break
        n_it = n_iters_per_p
        t_param = 1.0
        h_prev = h_cur.copy()
        for it in range(n_it):
            if time.time() > deadline:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
            y = h_cur + (t_param - 1.0) / t_new * (h_cur - h_prev)
            grad, _ = pnorm_grad(y, p)
            pen = 50.0 * (np.sum(y) - target)
            grad_total = grad + 2.0 * pen * np.ones(n) / n
            lr = 0.5 * dx * dx * p * p / (max(float(np.max(get_overlap(y))) ** (p - 1), 1e-6) + 1e-6)
            lr = min(lr, 0.05)
            h_new = y - lr * grad_total
            h_new = project_feasible(h_new, target)
            h_prev = h_cur
            h_cur = h_new
            t_param = t_new
            sc = compute_c5(h_cur)
            if sc < best_score:
                best_score = sc
                best_h = h_cur.copy()
    return best_h, best_score

def subgrad_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1: n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            if k > 0:
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 4.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def lag_surgery(h, n, target, max_steps, eps0, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1: n - 1 + n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        if k_star > 0:
            j_neg = np.arange(k_star, n)
            grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.02)
        else:
            eps *= 0.5
            if eps < 1e-10:
                break
    return h_cur, cur_score

def equalize_top(h, n, target, n_top, n_steps, lr, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1: n - 1 + n]
        top_idx = np.argsort(over)[::-1][:n_top]
        mean_top = float(np.mean(over[top_idx]))
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            weight = over[k] - mean_top
            if weight > 0:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += weight * (1.0 - h_cur[j_pos + k])
                if k > 0:
                    j_neg = np.arange(k, n)
                    grad[j_neg] -= weight * h_cur[j_neg - k]
        if np.max(np.abs(grad)) < 1e-15:
            break
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h_cur = h_new
    return best_h, best_score

def de_step(h_pop, n, target, n_gen, f_fac, cr, rng, deadline):
    pop = np.array(h_pop, dtype=float).copy()
    pop_size = pop.shape[0]
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_fac * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_p
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP on parent
    if tleft() > 60:
        h1, sc1 = slsqp_opt(best_h, n_final, target_final, [100, 500, 2000], 1500, 1e-15, min(start_time + 40, hard_deadline))
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: FISTA with increasing p
    if tleft() > 100:
        p_list = [2, 4, 8, 16, 32, 64]
        n_iters = {2: 3000, 4: 3000, 8: 2000, 16: 2000, 32: 1500, 64: 1500}
        h2, sc2 = fista_opt(best_h, n_final, target_final, p_list, 2000, min(start_time + 80, hard_deadline))
        if sc2 < best_score:
            best_score = sc2
            best_h = h2
        # SLSQP polish after FISTA
        h2s, sc2s = slsqp_opt(best_h, n_final, target_final, [500, 2000, 8000], 2000, 1e-16, min(start_time + 130, hard_deadline))
        if sc2s < best_score:
            best_score = sc2s
            best_h = h2s

    # Phase 3: Equalize top lags
    if tleft() > 80:
        for n_top in [2, 3, 5]:
            if tleft() < 30:
                break
            he, sce = equalize_top(best_h, n_final, target_final, n_top, 500, 0.003, min(start_time + 160, hard_deadline))
            if sce < best_score:
                best_score = sce
                best_h = he
            hs, scs = slsqp_opt(best_h, n_final, target_final, [1000, 5000], 1500, 1e-16, min(start_time + 180, hard_deadline))
            if scs < best_score:
                best_score = scs
                best_h = hs

    # Phase 4: Multi-restart with diverse perturbations
    restart_scales = [0.02, 0.04, 0.01, 0.06, 0.03, 0.015, 0.05, 0.025]
    for ri in range(len(restart_scales)):
        if tleft() < 60:
            break
        scale = restart_scales[ri]
        if ri % 3 == 0:
            h_pert = best_h + rng.randn(n_final) * scale
        elif ri % 3 == 1:
            # Lag-targeted perturbation
            corr_r = compute_corr(best_h)
            over_r = corr_r[n_final - 1: n_final - 1 + n_final]
            k_star = int(np.argmax(over_r))
            h_pert = best_h.copy()
            j_pos = np.arange(0, n_final - k_star)
            h_pert[j_pos] += scale * (1.0 - best_h[j_pos + k_star])
            if k_star > 0:
                j_neg = np.arange(k_star, n_final)
                h_pert[j_neg] -= scale * best_h[j_neg - k_star]
        else:
            # Fourier perturbation
            freq = rng.randint(2, 8)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2, n_final)
            h_pert = best_h + scale * np.cos(freq * np.pi * x + phase)
        h_pert = project_feasible(h_pert, target_final)
        # Quick FISTA
        hf, scf = fista_opt(h_pert, n_final, target_final, [8, 32, 64], 500, min(start_time + 250 + ri * 40, hard_deadline))
        # SLSQP polish
        hs, scs = slsqp_opt(hf, n_final, target_final, [500, 2000, 8000], 1500, 1e-16, min(start_time + 300 + ri * 40, hard_deadline))
        if scs < best_score:
            best_score = scs
            best_h = hs

    # Phase 5: More FISTA at very high p on best
    if tleft() > 80:
        h5, sc5 = fista_opt(best_h, n_final, target_final, [64, 128, 256, 512], 1000, min(start_time + 420, hard_deadline))
        if sc5 < best_score:
            best_score = sc5
            best_h = h5
        h5s, sc5s = slsqp_opt(best_h, n_final, target_final, [2000, 8000, 20000], 2000, 1e-16, min(start_time + 480, hard_deadline))
        if sc5s < best_score:
            best_score = sc5s
            best_h = h5s

    # Phase 6: Subgradient polish
    if tleft() > 60:
        h6, sc6 = subgrad_polish(best_h, n_final, target_final, 800, 5, 0.002, 0.997, min(start_time + 540, hard_deadline))
        if sc6 < best_score:
            best_score = sc6
            best_h = h6
        h6s, sc6s = slsqp_opt(best_h, n_final, target_final, [5000, 20000], 2000, 1e-16, min(start_time + 590, hard_deadline))
        if sc6s < best_score:
            best_score = sc6s
            best_h = h6s

    # Phase 7: Lag surgery
    if tleft() > 50:
        h7, sc7 = lag_surgery(best_h, n_final, target_final, 400, 0.002, min(start_time + 630, hard_deadline))
        if sc7 < best_score:
            best_score = sc7
            best_h = h7
        h7s, sc7s = slsqp_opt(best_h, n_final, target_final, [2000, 10000, 30000], 2000, 1e-17, min(start_time + 680, hard_deadline))
        if sc7s < best_score:
            best_score = sc7s
            best_h = h7s

    # Phase 8: More structured restarts
    for ri in range(6):
        if tleft() < 50:
            break
        # Block perturbation: modify a random contiguous block
        block_len = rng.randint(10, 40)
        block_start = rng.randint(0, n_final - block_len)
        h_pert = best_h.copy()
        h_pert[block_start: block_start + block_len] += rng.uniform(-0.04, 0.04)
        h_pert = project_feasible(h_pert, target_final)
        hf, scf = fista_opt(h_pert, n_final, target_final, [16, 64, 128], 400, min(start_time + 720 + ri * 25, hard_deadline))
        hs, scs = slsqp_opt(hf, n_final, target_final, [2000, 8000], 1500, 1e-16, min(start_time + 760 + ri * 25, hard_deadline))
        if scs < best_score:
            best_score = scs
            best_h = hs

    # Phase 9: DE at moderate resolution for global search, then upsample
    if tleft() > 120:
        n_med = 100
        t_med = n_med / 2.0
        x_old = np.linspace(0, 1, n_final)
        x_new = np.linspace(0, 1, n_med)
        h_med = np.interp(x_new, x_old, best_h)
        h_med = project_feasible(h_med, t_med)
        pop = [h_med.copy()]
        for _ in range(14):
            s = rng.uniform(0.03, 0.12)
            p = h_med + rng.randn(n_med) * s
            pop.append(project_feasible(p, t_med))
        hd, scd = de_step(pop, n_med, t_med, 100, 0.7, 0.6, rng, min(start_time + 830, hard_deadline))
        # Upsample back
        x_old2 = np.linspace(0, 1, n_med)
        x_new2 = np.linspace(0, 1, n_final)
        h_up = np.interp(x_new2, x_old2, hd)
        h_up = project_feasible(h_up, target_final)
        h_up = slsqp_opt(h_up, n_final, target_final, [500, 2000, 8000], 2000, 1e-16, min(start_time + 900, hard_deadline))
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 10: Final high-precision polish
    if tleft() > 40:
        h10, sc10 = slsqp_opt(best_h, n_final, target_final, [5000, 20000, 50000], 3000, 1e-17, min(start_time + 960, hard_deadline))
        if sc10 < best_score:
            best_score = sc10
            best_h = h10

    # Phase 11: One more FISTA + SLSQP cycle
    if tleft() > 60:
        h11, sc11 = fista_opt(best_h, n_final, target_final, [128, 256, 512, 1024], 800, min(start_time + 1020, hard_deadline))
        if sc11 < best_score:
            best_score = sc11
            best_h = h11
        h11s, sc11s = slsqp_opt(best_h, n_final, target_final, [10000, 30000, 60000], 2500, 1e-17, min(start_time + 1080, hard_deadline))
        if sc11s < best_score:
            best_score = sc11s
            best_h = h11s

    # Phase 12: Final subgradient + surgery
    if tleft() > 40:
        h12, sc12 = subgrad_polish(best_h, n_final, target_final, 500, 3, 0.001, 0.998, min(start_time + 1120, hard_deadline))
        if sc12 < best_score:
            best_score = sc12
            best_h = h12
        h12s, sc12s = lag_surgery(best_h, n_final, target_final, 200, 0.001, min(start_time + 1160, hard_deadline))
        if sc12s < best_score:
            best_score = sc12s
            best_h = h12s

    # Phase 13: Very final SLSQP
    if tleft() > 20:
        h13, sc13 = slsqp_opt(best_h, n_final, target_final, [20000, 50000, 100000], 2000, 1e-17, hard_deadline)
        if sc13 < best_score:
            best_score = sc13
            best_h = h13

    # Final validation
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    # Make sure we return the best
    if final_score > best_score:
        h_final = best_h.copy()
        final_score = best_score
    return h_final, final_score, len(h_final)
