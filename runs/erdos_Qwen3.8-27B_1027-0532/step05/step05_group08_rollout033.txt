Actually, let me try a completely different approach: **constraint programming / SLP (Successive Linear Programming)**

1. At current h, compute all correlations
2. Add a linear constraint: corr_k(h_new) ≈ corr_k(h) + ∇corr_k(h)^T (h_new - h) ≤ c_k - δ
   for the worst lag k (or top M lags)
3. Solve: min 0 (feasibility) s.t. the above + bounds + sum constraint
   (or: min ||h_new - h||^2 s.t. the constraints)
4. If infeasible, relax δ
5. Repeat

This is essentially a cutting-plane method. Each iteration adds a linear constraint that the new point must satisfy.

The QP in step 3 is:
min ||h_new - h||^2
s.t. ∇corr_k(h)^T h_new ≤ c_k - δ - ∇corr_k(h)^T h for top M k's
     0 ≤ h_new ≤ 1
     ∑ h_new = target

This is a standard QP that can be solved with scipy or cvxpy.

```python
def slp_step(h, n, target, top_m=10, delta_frac=0.01, deadline=None):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    top_idx = np.argsort(corr)[::-1][:top_m]
    
    # Build linearized constraints
    A = np.zeros((top_m, n))
    b = np.zeros(top_m)
    for i, mi in enumerate(top_idx):
        k = mi - (n - 1)
        g = np.zeros(n)
        if k >= 0:
            g[:n-k] += (1 - h[k:])
            g[k:] -= h[:n-k]
        else:
            ell = -k
            g[ell:] += (1 - h[:n-ell])
            g[:n-ell] -= h[ell:]
        g *= 2.0 / n
        A[i] = g
        b[i] = corr[mi] * 2.0/n - delta_frac * c_max * 2.0/n - g @ h
    
    # Solve QP: min ||h_new - h||^2 s.t. A h_new <= b, 0 <= h_new <= 1, sum(h_new) = target
    # This is: min h_new^T h_new - 2 h^T h_new + const
    # Hessian = 2I, linear = -2h
    # Use cvxpy
    import cvxpy as cp
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - h))
    constraints = [A @ x <= b, x >= 0, x <= 1, cp.sum(x) == target]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    
    if x.value is not None:
        h_new = project_feasible(x.value, target)
        return h_new
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h

def sls_qp_round(h, n, target, alpha, maxiter, ftol):
    def obj_slsqp(x):
        return lse_value(x, alpha)
    def grad_slsqp(x):
        return lse_grad(x, alpha)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', jac=grad_slsqp,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slp_descent(h, n, target, max_iter, deadline):
    """Successive Linear Programming (SLP) / cutting-plane descent."""
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    delta_frac = 0.005
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr)) * (2.0/n)
        
        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()
        
        top_m = min(10 + it // 50, 20)
        top_idx = np.argsort(corr)[::-1][:top_m]
        
        # Build linearized constraints
        A = np.zeros((top_m, n))
        b = np.zeros(top_m)
        dx = 2.0 / n
        for i, mi in enumerate(top_idx):
            k = mi - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[:n-k] += (1 - h_cur[k:])
                g[k:] -= h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += (1 - h_cur[:n-ell])
                g[:n-ell] -= h_cur[ell:]
            g *= dx
            A[i] = g
            # Linearization: corr_mi(h_new) ≈ corr[mi]*dx + g @ (h_new - h_cur)
            # We want: corr[mi]*dx + g @ (h_new - h_cur) <= c_max * (1 - delta_frac)
            # => g @ h_new <= c_max * (1-delta_frac) - corr[mi]*dx + g @ h_cur
            b[i] = c_max * (1 - delta_frac) - corr[mi] * dx + g @ h_cur
        
        # Solve QP: min ||h_new - h_cur||^2 s.t. A h_new <= b, 0<=h_new<=1, sum(h_new)=target
        x = cp.Variable(n)
        objective = cp.Minimize(cp.sum_squares(x - h_cur))
        constraints = [A @ x <= b, x >= 0, x <= 1, cp.sum(x) == target]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000)
        except:
            break
        
        if x.value is None:
            # Infeasible - reduce delta and retry
            delta_frac *= 0.5
            if delta_frac < 1e-10:
                break
            continue
        
        h_new = project_feasible(x.value, target)
        c_new = compute_c5(h_new)
        
        if c_new < c_max:
            h_cur = h_new
            delta_frac = min(delta_frac * 1.05, 0.02)
        else:
            # Try a smaller step
            h_step = 0.5 * (h_new - h_cur)
            h_try = project_feasible(h_cur + h_step, target)
            c_try = compute_c5(h_try)
            if c_try < c_max:
                h_cur = h_try
            else:
                delta_frac *= 0.7
                if delta_frac < 1e-12:
                    break
    
    return best_h, best_score

def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):

    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 8
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.02, 0.015)
        else:
            eps *= 0.85
            if eps < 1e-9:
                break
    return h_cur, best_score

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g


def obj_lse(x, alpha, mu, target):
    return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2


def grad_lse(x, alpha, mu, target):
    return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)


def lbfgsb_round(h, n, target, alpha, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_lse, h, method='L-BFGS-B', jac=lambda x: grad_lse(x, alpha, mu, target),
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9, 'factr': 1e7})
        return project_feasible(res.x, target)
    except Exception:
        return h


def obj_slsqp(x, alpha):
    return lse_value(x, alpha)


def grad_slsqp(x, alpha):
    return lse_grad(x, alpha)


def cons_fun(x, target):
    return np.sum(x) - target


def cons_jac(x):
    return np.ones(len(x))


def sls_qp_round(h, n, target, alpha, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: cons_fun(x, target), 'jac': cons_jac}]
    try:
        res = minimize(obj_slsqp, h, method='SLSQP', jac=lambda x: grad_slsqp(x, alpha),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def slp_descent(h, n, target, max_iter, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    delta_frac = 0.004
    dx = 2.0 / n
    infeas_count = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        infeas_count += 1
        if infeas_count > 30:
            break

        corr = compute_corr(h_cur)
        c_max = float(np.max(corr)) * dx

        if c_max < best_score:
            best_score = c_max
            best_h = h_cur.copy()

        top_m = min(8 + it // 40, 18)
        top_idx = np.argsort(corr)[::-1][:top_m]

        A = np.zeros((top_m, n))
        b = np.zeros(top_m)
        for i, mi in enumerate(top_idx):
            k = mi - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[:n - k] += (1.0 - h_cur[k:])
                g[k:] -= h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += (1.0 - h_cur[:n - ell])
                g[:n - ell] -= h_cur[ell:]
            g *= dx
            A[i] = g
            b[i] = c_max * (1.0 - delta_frac) - corr[mi] * dx + g @ h_cur

        x = cp.Variable(n)
        objective = cp.Minimize(cp.sum_squares(x - h_cur))
        constraints = [A @ x <= b, x >= 0, x <= 1, cp.sum(x) == target]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=2000, abs_tol=1e-8, rel_tol=1e-8)
        except Exception:
            break

        if x.value is None or prob.status == 'infeasible':
            delta_frac *= 0.6
            if delta_frac < 1e-10:
                break
            infeas_count += 1
            continue

        infeas_count = 0
        h_new = project_feasible(x.value, target)
        c_new = compute_c5(h_new)

        if c_new < c_max:
            h_cur = h_new
            delta_frac = min(delta_frac * 1.03, 0.015)
        else:
            h_step = 0.5 * (h_new - h_cur)
            h_try = project_feasible(h_cur + h_step, target)
            c_try = compute_c5(h_try)
            if c_try < c_max:
                h_cur = h_try
            else:
                delta_frac *= 0.7
                if delta_frac < 1e-12:
                    break

    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, temp, lr_init, lr_decay, deadline):

    best_h = h_cur.copy()

    dx = 2.0 / n
    lr = lr_init

            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_k = 10
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n - ell])
                g[:n - ell] -= wk * h_cur[ell:]
        g *= dx
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

            best_score = sc_new
            eps = min(eps * 1.015, 0.02)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
```
