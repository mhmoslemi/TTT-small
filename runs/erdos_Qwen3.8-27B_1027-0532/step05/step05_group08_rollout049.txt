def compute_corr(h):

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        h = np.clip(h, 0.0, 1.0)
    return h

def multi_lag_obj(x, active_idx, n):
    corr = compute_corr(x)
    return float(np.sum(corr[active_idx]))

def multi_lag_grad(x, active_idx, n):
    dx = 2.0 / n
    g = np.zeros(n)
    for m in active_idx:
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n-k] += (1.0 - x[k:])
            g[k:] -= x[:n-k]
        else:
            ell = -k
            g[ell:] += (1.0 - x[:n-ell])
            g[:n-ell] -= x[ell:]
    g *= dx
    return g

def slsqp_multilag(h, n, target, active_idx, maxiter=500, ftol=1e-15):
    def obj(x):
        return multi_lag_obj(x, active_idx, n)
    def grad(x):
        return multi_lag_grad(x, active_idx, n)
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', jac=grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

**Alternative idea: Use the structure of the problem**

The problem min_h max_k sum_i h[i+k](1-h[i]) has a special structure. At the optimum, by the "minimax" property, the top lags should be approximately equal. So we can reformulate as:

min t
s.t. sum_i h[i+k](1-h[i]) <= t/dx for K active lags k
     0 <= h <= 1
     sum(h) = n/2

This is a non-convex QP (bilinear constraints). But we can solve it with:
- Sequential Quadratic Programming (which is what SLSQP does internally)
- Or by linearizing the bilinear terms (alternating linearization)

The alternating linearization approach:
- Fix h at its current value h_old
- Linearize: h[i+k]*(1-h[i]) ≈ h_old[i+k] - h_old[i+k]*h[i] + h[i+k]*(1-h_old[i]) - h[i+k]*(1-h_old[i]) ... 

Actually, the linearization of h[i+k]*(1-h[i]) around h_old is:
h_old[i+k]*(1-h_old[i]) + (1-h_old[i])*(h[i+k]-h_old[i+k]) - h_old[i+k]*(h[i]-h_old[i])

= (1-h_old[i])*h[i+k] - h_old[i+k]*h[i] + h_old[i+k]*h_old[i]

This is linear in h! So we can solve a sequence of LPs (or QPs if we add a regularization term to stay close to h_old).

This is essentially the **bilinear programming** approach (Jaumont et al., or the "successive linearization" method).

Algorithm:
1. Initialize h
2. Repeat:
   a. Find the top K lags
   b. Solve LP: min t s.t. linearized constraints, 0<=h<=1, sum(h)=target
      (with a trust region or regularization to h_old for stability)
   c. Update h
3. Until convergence

This could be much faster than SLSQP because each subproblem is an LP (or QP with trust region).

    k = m - (n - 1)
    g = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_single(h, m, n):
    k = m - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        ell = -k
        g[ell:] += (1.0 - h[:n - ell])
        g[:n - ell] -= h[ell:]
    return g


def multi_lag_objective(x, active_idx, n):
    corr = compute_corr(x)
    return float(np.sum(corr[active_idx]))


def multi_lag_gradient(x, active_idx, n):
    g = np.zeros(n)
    for m in active_idx:
        g += corr_grad_single(x, int(m), n)
    dx = 2.0 / n
    return g * dx


def lse_smooth_objective(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def lse_smooth_gradient(x, alpha):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-14:
            continue
        g += w[idx] * corr_grad_single(x, idx, n)
    return g * dx


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_solve_ml(h_init, n, target, active_idx, maxiter, ftol):
    def obj_func(x):
        return multi_lag_objective(x, active_idx, n)

    def grad_func(x):
        return multi_lag_gradient(x, active_idx, n)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj_func, h_init, method='SLSQP', jac=grad_func,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h_init


def lbfgsb_solve_ml(h_init, n, target, active_idx, mu, maxiter):
    def obj_func(x):
        return multi_lag_objective(x, active_idx, n) + mu * (np.sum(x) - target) ** 2

    def grad_func(x):
        return multi_lag_gradient(x, active_idx, n) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_func, h_init, method='L-BFGS-B', jac=grad_func,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h_init


def lbfgsb_solve_lse(h_init, n, target, alpha, mu, maxiter):
    def obj_func(x):
        return lse_smooth_objective(x, alpha) + mu * (np.sum(x) - target) ** 2

        return lse_smooth_gradient(x, alpha) + 2.0 * mu * (np.sum(x) - target)

    bounds = [(0.0, 1.0)] * n

        res = minimize(obj_func, h_init, method='L-BFGS-B', jac=grad_func,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})

    except Exception:
        return h_init

        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_curr = c_max * dx
        if c_curr < best_score:

            best_h = h_cur.copy()

        top_k = 5
        top_idx = np.argsort(corr)[::-1][:top_k]
        diffs = corr[top_idx] - corr[top_idx[0]]
        temp = 0.005 * c_max + 1e-15
        weights = np.exp(diffs / temp)
        weights /= weights.sum()

        g = np.zeros(n)
        for wk, m in zip(weights, top_idx):
            g += wk * corr_grad_single(h_cur, int(m), n)
        g *= dx

        lr = base_lr * (0.8 + 0.4 * rng.rand())
        noise_scale = 0.003 * (T / 0.004)
        h_new = h_cur - lr * g + rng.randn(n) * noise_scale
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        delta = sc_new - c_curr
        if delta < 0 or rng.rand() < np.exp(-max(delta, -1.0) / max(T, 1e-10)):

        else:
            if rng.rand() < 0.03:
