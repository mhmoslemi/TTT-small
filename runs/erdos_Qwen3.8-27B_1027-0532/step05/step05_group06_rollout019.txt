import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))

    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac, 'args': (target,)}]
    try:
        res = minimize(lse_fun, h.copy(), jac=lse_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       args=(alpha, target, mu),
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return h


def upscale(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)


def stochastic_subgrad(h, target, n_iters, top_k, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.008
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        w = (corr[top_idx] - corr[top_idx].min() + 1e-8)
        w = w / w.sum()
        grad = np.zeros(n)
        for ki, wi in zip(top_idx, w):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= h_cur[j2 - m]
        grad += 0.2 * (np.sum(h_cur) - target)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            h_cur = h_new
            cur_score = sc
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-12:
                break
    return h_cur, cur_score
