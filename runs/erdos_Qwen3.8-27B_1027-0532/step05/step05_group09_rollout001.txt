def project_feasible(h, target):

    h = h.astype(float).copy()

        y_clipped = np.clip(y, 0.0, 1.0)

        x_new = project_feasible(y_clipped - step * grad_y, target)

            if obj_new < obj_y:

            step *= 0.7
            x_new = project_feasible(y_clipped - step * grad_y, target)

        x_prev = x
        x = x_new
        t = t_new

            best_x = x.copy()

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1.0
_g_n = 0
_g_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    p = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    f1 = np.correlate(1.0 - h, p, 'full')[n-1:2*n-1]
    f2 = np.convolve(h, p, 'full')[:n]
    grad = dx * (f1 - f2)
    return obj, grad

def _obj(x):
    return lse_obj_and_grad(x, _g_alpha)[0]

def _grad(x):
    return lse_obj_and_grad(x, _g_alpha)[1]

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def trust_constr_optimize(h, alpha, maxiter=400):
    global _g_alpha, _g_n, _g_target
    _g_alpha = alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj, h.copy(), jac=_grad, bounds=bounds,
                   constraints=cons, method='trust-constr',
                   options={'maxiter': maxiter, 'gtol': 1e-12, 'xtol': 1e-14})
    return res.x

def fista_optimize(h0, target, alpha, max_iters, deadline):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = 1.0 / (alpha * 0.5 + 50.0)
    step_cap = 5.0 / (alpha + 10.0)
    for k in range(max_iters):
        if time.time() > deadline:
            break
        frac = min(k / float(max(max_iters, 1)), 1.0)
        mom_scale = 1.0 - 0.4 * frac
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + mom_scale * ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            if obj_new <= obj_y + 1e-15:
                break
            step *= 0.6
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step * 1.2, step_cap)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj

def active_set_polish(h, target, deadline, max_rounds=500):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        sorted_idx = np.argsort(corr)[::-1]
        num_active = min(5, n - 1)
        active = sorted_idx[:num_active]
        c_max = corr[active[0]]
        diffs = corr[active] - c_max
        w = np.exp(diffs / 0.003)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            if ki >= n:
                continue
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[n - 1 - (jn - ki)][::-1]
            grad[jn] = 0.0
            for j in jn:
                grad[j] += -wi * h_cur[j - ki]
        grad *= dx
        found = False
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-16:
                h_cur = h_new
                cur_score = sc
                found = True
                break
        if not found:
            break
    return h_cur, cur_score

def fine_coord_descent(h, target, deadline, max_steps=10000):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(42)
    for it in range(max_steps):
        if time.time() > deadline:
            break
        idx = rng.randint(n)
        old = h_cur[idx]
        if old < 1e-10:
            h_cur[idx] = 0.1
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < cur_score:
                cur_score = sc
            else:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
        elif old > 1.0 - 1e-10:
            h_cur[idx] = 0.9
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < cur_score:
                cur_score = sc
            else:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
        else:
            candidates = [0.0, 1.0, 0.5, old * 0.5, old + (0.5 - old) * 0.5,
                          old - 0.01, old + 0.01]
            for trial in candidates:
                if trial < -0.1 or trial > 1.1:
                    continue
                if abs(trial - old) < 1e-14:
                    continue
                h_cur[idx] = trial
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score:
                    cur_score = sc
                    break
            else:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
    return h_cur, cur_score
