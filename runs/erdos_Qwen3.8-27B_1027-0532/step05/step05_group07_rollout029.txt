3. Repeat

def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    # First term: (1-h[i+m-n+1]) for m where 0 ≤ i+m-n+1 < n
    # i+m-n+1 = j => m = j-i+n-1, j in [0, n-1]
    # m in [n-1-i, n-1-i+n-1] = [n-1-i, 2n-2-i]
    j_range = np.arange(0, n)
    m_range = j_range - i + n - 1
    valid = (m_range >= 0) & (m_range < nl1)
    B[m_range[valid]] += (1.0 - h[j_range[valid]])
    
    # Second term: -h[i-m+n-1] for m where 0 ≤ i-m+n-1 < n
    # i-m+n-1 = j => m = i-j+n-1, j in [0, n-1]
    # m in [i-n+1, i+0+n-1] = [i-n+1, i+n-1]... wait
    # m = i - j + n - 1, j in [0, n-1]
    # When j=0: m = i+n-1
    # When j=n-1: m = i+0 = i
    # So m in [i, i+n-1]
    m_range2 = i + n - 1 - j_range  # m = i - j + n - 1
    valid2 = (m_range2 >= 0) & (m_range2 < nl1)
    B[m_range2[valid2]] -= h[j_range[valid2]]
    
    return B

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def lse_grad(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    weights = weights / wsum
    n = len(x)
    nl1 = 2 * n - 1
    grad = np.zeros(n)
    for m in range(nl1):
        wm = weights[m]
        if wm < 1e-15:
            continue
        i_lo = max(0, n - 1 - m)
        i_hi = min(n - 1, 2 * n - 2 - m)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1) + m - n + 1
            grad[i_lo:i_hi + 1] += wm * (1.0 - x[idx])
        i_lo2 = max(0, m - n + 1)
        i_hi2 = min(n - 1, m)
        if i_lo2 <= i_hi2:
            idx2 = np.arange(i_lo2, i_hi2 + 1) - m + n - 1
            grad[i_lo2:i_hi2 + 1] -= wm * x[idx2]
    return grad


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_optimize(h, n, target, alphas, maxiter=5000, ftol=1e-18):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, jac=lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def compute_B_vec(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    j_range = np.arange(n)
    m_range = j_range - i + n - 1
    valid = (m_range >= 0) & (m_range < nl1)
    B[m_range[valid]] += (1.0 - h[j_range[valid]])
    m_range2 = i + n - 1 - j_range
    valid2 = (m_range2 >= 0) & (m_range2 < nl1)
    B[m_range2[valid2]] -= h[j_range[valid2]]
    return B


def coord_desc(h, n, target, deadline, rng, n_sweeps=100000, recompute_interval=50):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)
            h_cur = project_feasible(h_cur, target)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B_vec(h_cur, i, n)
            Bj = compute_B_vec(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            pos = C > 1e-14
            neg = C < -1e-14
            if np.any(pos):
                hi = float(np.max(A + C * t_max))
            else:
                hi = float(np.max(A))

            lo = 0.0
            for _ in range(45):
                mid = (lo + hi) * 0.5
                zro = ~(pos | neg)
                if np.any(zro) and np.max(A[zro]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid

            v = hi
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 150 and sweep > 50:
            pert = rng.randn(n) * 0.008
            h_cur = project_feasible(h_cur + pert, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def simulated_annealing_binary(h, n, target, deadline, rng, n_iter=50000, T_init=0.5, T_min=0.01):
    dx = 2.0 / n
    h_cur = h.copy()
    h_bin = (h_cur > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target)
    corr = compute_corr(h_bin)
    cur_score = float(np.max(corr)) * dx
    best_h = h_bin.copy()
    best_score = cur_score
    on_indices = np.where(h_bin > 0.5)[0]
    off_indices = np.where(h_bin < 0.5)[0]

    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T_init * (T_min / T_init) ** (it / n_iter)
        if len(on_indices) == 0 or len(off_indices) == 0:
            break

        swap_count = rng.randint(1, 4)
        swaps = []
        for _ in range(swap_count):
            oi = on_indices[rng.randint(len(on_indices))]
            oi_idx = np.where(on_indices == oi)[0][0]
            off_i = off_indices[rng.randint(len(off_indices))]
            off_idx = np.where(off_indices == off_i)[0][0]
            if oi != off_i:
                swaps.append((oi, off_i))

        if not swaps:
            continue

        h_trial = h_bin.copy()
        for oi, off_i in swaps:
            h_trial[oi] = 0.0
            h_trial[off_i] = 1.0

        corr_trial = compute_corr(h_trial)
        score_trial = float(np.max(corr_trial)) * dx
        delta = score_trial - cur_score

        if delta < 0 or rng.rand() < np.exp(-delta / (T * dx * 10)):
            h_bin = h_trial
            cur_score = score_trial
            corr = corr_trial
            on_indices = np.where(h_bin > 0.5)[0]
            off_indices = np.where(h_bin < 0.5)[0]
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_bin.copy()
        else:
            h_bin = best_h.copy()
            cur_score = best_score
            corr = compute_corr(h_bin)
            on_indices = np.where(h_bin > 0.5)[0]
            off_indices = np.where(h_bin < 0.5)[0]

    return best_h, best_score

def oscillation_search(h, n, target, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    phases = np.arange(n)
    rng.shuffle(phases)
    for trial in range(200):
        if time.time() > deadline:
            break
        idx_i = int(phases[trial % n])
        idx_j = int(phases[(trial + n // 3) % n])
        if idx_i == idx_j:
            continue
        c = h_cur[idx_i] + h_cur[idx_j]
        if c < 1e-14 or c > 2.0 - 1e-14:
            continue
        t_min = max(0.0, c - 1.0)
        t_max = min(1.0, c)
        if t_max - t_min < 1e-10:
            continue
        fractions = np.linspace(t_min, t_max, 7)
        best_t = h_cur[idx_i]
        best_s = compute_c5(h_cur)
        for frac in fractions:
            h_trial = h_cur.copy()
            h_trial[idx_i] = frac
            h_trial[idx_j] = c - frac
            s = compute_c5(h_trial)
            if s < best_s:
                best_s = s
                best_t = frac
        h_cur[idx_i] = best_t
        h_cur[idx_j] = c - best_t
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score
