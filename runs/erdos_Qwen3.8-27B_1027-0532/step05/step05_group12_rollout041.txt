import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_rng_local = np.random.RandomState(999)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    total_lags = 2 * n - 1
    if top_m is not None and top_m < total_lags:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(total_lags)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def run_gradient_phase(h_init, n, target, alpha_list, steps_list, topm_list,
                       start_time, deadline, max_lr=2.5, halve=0.7, max_trials=45,
                       beta=0.88, stall_limit=40):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    for ai, alpha in enumerate(alpha_list):
        if time.time() > deadline:
            break
        steps = steps_list[min(ai, len(steps_list) - 1)]
        top_m = topm_list[min(ai, len(topm_list) - 1)]

        momentum_v = np.zeros(n)
        stalled = 0

        for step in range(steps):
            if time.time() > deadline:
                break

            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad

            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-13:
                break

            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, max_lr)

            improved = False
            for trial in range(max_trials):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-15:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= halve

            if not improved:
                stalled += 1
                if stalled >= stall_limit:
                    if stalled % 15 == 0:
                        momentum_v = np.zeros(n)
                    if stalled > stall_limit * 2:
                        break

    return best_h, best_score


def slsqp_polish(h_start, n, target, alphas, maxiter, ftol, deadline):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coordinate_ascent_step(h, n, target, j, j2, deadline):
    """Try to improve by varying h[j] and h[j2] to maintain sum constraint."""
    current_sum = h[j] + h[j2]
    best_local = compute_c5(h)

    for tj in np.linspace(0.0, 1.0, 11):
        for tj2 in np.linspace(0.0, 1.0, 11):
            if abs(tj + tj2 - current_sum) > 1e-6:
                continue
            if tj < -1e-9 or tj > 1.0 + 1e-9 or tj2 < -1e-9 or tj2 > 1.0 + 1e-9:
                continue
            h_test = h.copy()
            h_test[j] = tj
            h_test[j2] = tj2
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_local - 1e-14:
                best_local = sc
                h[:] = h_test
    return h, best_local
