    return h0

def arc_polish_v2(h, target, max_rounds, deadline):
    """Improved arc polish with adaptive temperature."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        sorted_idx = np.argsort(corr)[::-1]
        num_active = max(3, min(10, int(n * 0.02)))
        active = sorted_idx[:num_active]
        
        # Adaptive temperature
        spread = corr[active[0]] - corr[active[-1]]
        if spread < 1e-10:
            temp = 0.01
        else:
            temp = max(0.01, spread / 10.0)
        
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        
        # Line search
        best_eps = 0
        best_sc = cur_score
        for eps_try in np.logspace(-1, -4, 20):
            h_new = h_cur - eps_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_eps = eps_try
        if best_eps > 0:
            h_cur = project_feasible(h_cur - best_eps * grad, target)
            cur_score = compute_c5(h_cur)

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    cm = float(np.max(corr))
    sh = np.clip(alpha * (corr - cm), -700.0, 0.0)
    return (cm + float(np.log(np.exp(sh).sum())) / alpha) * dx

def lse_grad_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    cm = float(np.max(corr))
    sh = np.clip(alpha * (corr - cm), -700.0, 0.0)
    exp_sh = np.exp(sh)
    sum_exp = float(exp_sh.sum())
    w = exp_sh / sum_exp
    wf = w[::-1].copy()
    ws = 2 * n
    Wf = np.fft.rfft(wf, ws)
    W = np.fft.rfft(w, ws)
    V = np.fft.rfft(1.0 - h, ws)
    H = np.fft.rfft(h, ws)
    cp = np.fft.irfft(Wf * V, ws)[n-1:2*n-1]
    cn = np.fft.irfft(W * H, ws)[:n]
    return dx * (cp - cn)

def _penalty_obj(x):
    target = _global['target']
    alpha = _global['alpha']
    count = _global['count']
    deadline = _global['deadline']
    count[0] += 1
    if time.time() > deadline or count[0] > _global['max_evals']:
        raise _Timeout()
    h = project_feasible(x, target)
    return lse_obj_val(h, alpha)

def _penalty_grad(x):
    target = _global['target']
    alpha = _global['alpha']
    h = project_feasible(x, target)
    return lse_grad_val(h, alpha)

def lbfgs_step(h0, target, alpha, deadline, max_evals=300):
    _global['target'] = target
    _global['alpha'] = alpha
    _global['count'] = [0]
    _global['deadline'] = deadline
    _global['max_evals'] = max_evals
    h = project_feasible(h0, target).copy()
    bounds = [(0.0, 1.0)] * len(h)
    try:
        res = minimize(_penalty_obj, h, jac=_penalty_grad, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': 1000, 'maxfun': max_evals, 'gtol': 1e-6})
        return project_feasible(res.x, target)
    except _Timeout:
        return h
    except Exception:
        return h

def fista(h0, target, alpha, max_iters, deadline, lr=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_sc = compute_c5(x)
    if lr is None:
        lr = min(1.0, 2.0 / (alpha * 0.03 + 1.0))
    step = lr
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_c = np.clip(y, 0.0, 1.0)
        g = lse_grad_val(y_c, alpha)
        obj_y = lse_obj_val(y_c, alpha)
        x_new = project_feasible(y_c - step * g, target)
        
        contr = 0.0
        for _ in range(20):
            obj_new = lse_obj_val(x_new, alpha)
            dec = obj_y - obj_new
            quad = 0.5 / step * float(np.dot(x_new - y_c, x_new - y_c))
            if dec >= quad * (1.0 - contr):
                break
            step *= (1.0 - contr)
            x_new = project_feasible(y_c - step * g, target)
        step = min(step / max(1.0 - contr, 0.01), lr * 3.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_sc:
            best_sc = sc
            best_x = x.copy()
    return best_x, best_sc

def directed_descent(h, target, deadline, max_steps=500):
    """Directly target the maximum lag with a specialized step."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_sc = compute_c5(h_cur)
    
    for step_i in range(max_steps):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        k_max = int(np.argmax(corr))
        
        # Gradient of the max lag
        g = np.zeros(n)
        jp = np.arange(0, n - k_max)
        g[jp] += (1.0 - h_cur[jp + k_max])
        jn = np.arange(k_max, n)
        g[jn] -= h_cur[jn - k_max]
        g *= dx
        
        # Try several step sizes
        best_eps = 0
        best_cand_sc = best_sc
        for eps in np.logspace(-3, 0, 15):
            h_new = project_feasible(h_cur - eps * g, target)
            sc = compute_c5(h_new)
            if sc < best_cand_sc:
                best_cand_sc = sc
                best_eps = eps
        if best_eps > 0:
            h_cur = project_feasible(h_cur - best_eps * g, target)
            best_sc = best_cand_sc
        else:
            # Try reducing the second max lag
            sorted_idx = np.argsort(corr)[::-1]
            if len(sorted_idx) > 1:
                k2 = int(sorted_idx[1])
                g2 = np.zeros(n)
                jp2 = np.arange(0, n - k2)
                g2[jp2] += (1.0 - h_cur[jp2 + k2])
                jn2 = np.arange(k2, n)
                g2[jn2] -= h_cur[jn2 - k2]
                g2 *= dx
                for eps in np.logspace(-3, 0, 10):
                    h_new = project_feasible(h_cur - eps * g2, target)
                    sc = compute_c5(h_new)
                    if sc < best_cand_sc:
                        best_cand_sc = sc
                        h_cur = h_new
                        best_eps = eps
                if best_eps > 0:
                    best_sc = best_cand_sc
    
    return h_cur, best_sc

def arc_polish(h, target, max_rounds, deadline):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_sc = compute_c5(h_cur)
    
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = get_corr(h_cur)
        sorted_idx = np.argsort(corr)[::-1]
        num_act = max(2, min(5, int(0.01 * n)))
        active = sorted_idx[:num_act]
        
        spread = float(corr[active[0]] - corr[active[-1]])
        temp = max(0.005, spread * 0.5)
        
        diffs = corr[active] - corr[active[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        
        g = np.zeros(n)
        for wi, ki in zip(w, active):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            g[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            g[jn] -= wi * h_cur[jn - ki]
        g *= dx
        
        best_eps = 0
        best_sc = cur_sc
        for eps in np.logspace(-2, -5, 20):
            h_new = project_feasible(h_cur - eps * g, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_eps = eps
        if best_eps > 0:
            h_cur = project_feasible(h_cur - best_eps * g, target)
            cur_sc = best_sc
        else:
            # Adaptive: try smaller temperature
            pass
    
    return h_cur, cur_sc

def coord_descent_v2(h, target, deadline, max_sweeps=50):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_sc = compute_c5(h_cur)
    rng = np.random.RandomState(77)
    
    stall = 0
    for sweep in range(max_sweeps):

            break

        for idx in rng.permutation(n):
            if time.time() > deadline:
                break
            ov = h_cur[idx]
            if ov < 1e-8 or ov > 1.0 - 1e-8:
                trials = [0.0] if ov > 0.5 else [1.0]
            else:
                trials = [0.0, 1.0, 0.5, ov * 0.5, ov * 1.5]
            best_t = ov
            best_s = cur_sc
            for t in trials:

                improved = True
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return h_cur, cur_sc
