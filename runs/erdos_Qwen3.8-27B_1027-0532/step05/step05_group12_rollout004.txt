def project_feasible(h, target_sum):

            return h

def smooth_obj(h, alpha):

def smooth_grad(h, alpha, dx, top_m=None):

    c_max = np.max(corr)

        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def adaptive_pg(h_init, n, target, deadline, alpha_start=20.0, top_m_start=15,
                max_steps=80000, beta_start=0.8):

    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha = alpha_start
    top_m = top_m_start
    beta = beta_start
    momentum_v = np.zeros(n)
    stalled = 0
    best_streak = 0

    for step in range(max_steps):
        if time.time() > deadline:
            break

        grad = smooth_grad(h, alpha, dx, top_m=top_m)
        momentum_v = beta * momentum_v + (1.0 - beta) * grad

        gnorm = np.max(np.abs(momentum_v))
        if gnorm < 1e-15:
            break

        lr = 2.0 / (alpha * gnorm * dx + 1e-10)
        lr = min(lr, 5.0)

        improved = False
        for _ in range(80):

            h_try = project_feasible(h_try, target)
            score = compute_c5(h_try)

                best_h = h_try.copy()
                h = h_try
                improved = True
                stalled = 0

                break

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, deadline):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                sc_new = compute_c5(h_new)
                sc_cur = compute_c5(h_cur)
                if sc_new < sc_cur - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def alternating_lp(h_init, n, target, deadline, max_iter=10000,
                   damping=0.5, reg=1e-4):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for it in range(max_iter):
        if time.time() > deadline:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()

    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_restart(h, n, target, rng, mode):
    h_p = h.copy()
    if mode == 0:
        h_p += rng.randn(n) * 0.04
    elif mode == 1:
        seg = max(1, n // 8)
        start = rng.randint(0, max(1, n - seg))
        h_p[start:start + seg] = 1.0 - h_p[start:start + seg]
    elif mode == 2:
        shift = rng.randint(1, max(2, n // 20))
        h_p = np.roll(h, shift)
    elif mode == 3:
        seg = max(1, n // 5)
        for i in range(0, n, seg):
            end = min(i + seg, n)
            scale = rng.uniform(0.6, 1.4)
            h_p[i:end] *= scale
    elif mode == 4:
        kernel = np.ones(7) / 7.0
        h_smooth = np.convolve(h, kernel, mode='same')
        thr = np.median(h_smooth)
        h_p = np.where(h_smooth > thr, 0.7, 0.3)
    elif mode == 5:
        h_p = 1.0 - h
    else:
        h_p += rng.randn(n) * 0.06
    return project_feasible(h_p, target)
