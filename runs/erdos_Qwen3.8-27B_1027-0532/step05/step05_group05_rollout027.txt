import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def golden_search(base, c, lo, hi, n_iter=32):
    gr = (1.0 - np.sqrt(5.0)) / 2.0
    a, b = lo, hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = np.max(base + c_pt * c)
    fd = np.max(base + d_pt * c)
    for _ in range(n_iter):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = np.max(base + c_pt * c)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = np.max(base + d_pt * c)
    return (a + b) / 2.0


def coord_descent_loop(h_init, n, target, deadline, rng):
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    n_full = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    gr = (1.0 - np.sqrt(5.0)) / 2.0
    corr = compute_corr(h_cur)
    idx = np.arange(n)
    stall = 0

    while time.time() < deadline:
        rng.shuffle(idx)
        for ii in idx:
            i = int(ii)
            old_val = h_cur[i]
            v1 = (i + lags >= 0) & (i + lags < n)
            v2 = (i - lags >= 0) & (i - lags < n)
            c = np.zeros(n_full)
            if v1.any():
                c[v1] -= h_cur[i + lags[v1]]
            if v2.any():
                c[v2] += (1.0 - h_cur[i - lags[v2]])
            base = corr - old_val * c
            a_g, b_g = 0.0, 1.0
            c_pt = b_g - gr * (b_g - a_g)
            d_pt = a_g + gr * (b_g - a_g)
            fc = np.max(base + c_pt * c)
            fd = np.max(base + d_pt * c)
            for _ in range(30):
                if fc < fd:
                    b_g = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b_g - gr * (b_g - a_g)
                    fc = np.max(base + c_pt * c)
                else:
                    a_g = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a_g + gr * (b_g - a_g)
                    fd = np.max(base + d_pt * c)
            t_new = (a_g + b_g) / 2.0
            delta = t_new - old_val

                h_cur[i] = t_new
                corr += delta * c
        h_cur = project_feasible(h_cur, target)
        corr = compute_corr(h_cur)
        sc = float(np.max(corr)) * (2.0 / n)

            stall = 0

        if stall > 8:

            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            stall = 0
    h_cur = project_feasible(h_cur, target)
    sc = compute_c5(h_cur)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()
    return best_h, best_score


def pair_coord_descent(h_init, n, target, deadline, rng):
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    n_full = 2 * n - 1
    lags = np.arange(-(n - 1), n)
    gr = (1.0 - np.sqrt(5.0)) / 2.0
    corr = compute_corr(h_cur)
    stall = 0

    while time.time() < deadline:
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            a = h_cur[i]
            b = h_cur[j]
            v1_i = (i + lags >= 0) & (i + lags < n)
            v2_i = (i - lags >= 0) & (i - lags < n)
            c_i = np.zeros(n_full)
            if v1_i.any():
                c_i[v1_i] -= h_cur[i + lags[v1_i]]
            if v2_i.any():
                c_i[v2_i] += (1.0 - h_cur[i - lags[v2_i]])
            v1_j = (j + lags >= 0) & (j + lags < n)
            v2_j = (j - lags >= 0) & (j - lags < n)
            c_j = np.zeros(n_full)
            if v1_j.any():
                c_j[v1_j] -= h_cur[j + lags[v1_j]]
            if v2_j.any():
                c_j[v2_j] += (1.0 - h_cur[j - lags[v2_j]])
            c_diff = c_i - c_j
            base = corr - a * c_diff
            lo = max(0.0, a + b - 1.0)
            hi = min(1.0, a + b)
            if lo >= hi:
                continue
            a_g, b_g = lo, hi
            c_pt = b_g - gr * (b_g - a_g)
            d_pt = a_g + gr * (b_g - a_g)
            fc = np.max(base + c_pt * c_diff)
            fd = np.max(base + d_pt * c_diff)
            for _ in range(28):
                if fc < fd:
                    b_g = d_pt
                    d_pt = c_pt
                    fd = fc
                    c_pt = b_g - gr * (b_g - a_g)
                    fc = np.max(base + c_pt * c_diff)
                else:
                    a_g = c_pt
                    c_pt = d_pt
                    fc = fd
                    d_pt = a_g + gr * (b_g - a_g)
                    fd = np.max(base + d_pt * c_diff)
            t_new = (a_g + b_g) / 2.0
            s_new = a + b - t_new
            d_i = t_new - a
            d_j = s_new - b
            if abs(d_i) > 1e-13 or abs(d_j) > 1e-13:
                h_cur[i] = t_new
                h_cur[j] = s_new
                corr += d_i * c_i + d_j * c_j
        h_cur = project_feasible(h_cur, target)
        corr = compute_corr(h_cur)
        sc = float(np.max(corr)) * (2.0 / n)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        if stall > 20:
            h_cur = h_cur + rng.randn(n) * 0.015
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            stall = 0
    h_cur = project_feasible(h_cur, target)
    sc = compute_c5(h_cur)
    if sc < best_score:
        best_score = sc
        best_h = h_cur.copy()
    return best_h, best_score


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, pos in zip(weights, top_idx):
            lag = int(pos) - (n - 1)
            if lag >= 0:
                grad[lag:] += wk * (1.0 - h_cur[:n - lag])
                grad[:n - lag] -= wk * h_cur[lag:]
            else:
                la = -lag
                grad[:n - la] += wk * (1.0 - h_cur[la:])
                grad[la:] -= wk * h_cur[:n - la]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 5.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        n_full = 2 * n - 1
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:] += (1.0 - h_cur[:n - lag])
            grad[:n - lag] -= h_cur[lag:]
        else:
            la = -lag
            grad[:n - la] += (1.0 - h_cur[la:])
            grad[la:] -= h_cur[:n - la]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.03)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def fourier_init(n, rng, n_harmonics=12):
    x = np.linspace(0, 2, n)
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):

        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(np.pi * k * x + phase)
