def equalize_top_lags(h, target, n_top=10, max_iter=100, tol=1e-10):
    n = len(h)
    h_cur = h.copy()
    dx = 2.0/n
    
    for it in range(max_iter):
        corr = np.correlate(h_cur, 1.0-h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        spread = top_vals.max() - top_vals.min()
        
        if spread < tol:
            break
        
        # Compute gradient of each top lag
        grads = []
        for ki in top_idx:
            g = np.zeros(n)
            lag = int(ki) - (n-1)
            if lag >= 0:
                for i in range(n - lag):
                    g[i+lag] += (1.0 - h_cur[i])  # from h[i+lag] term
                    g[i] -= h_cur[i+lag]  # from -h[i]*h[i+lag] term
            else:
                m = -lag
                for i in range(n - m):
                    g[i] += (1.0 - h_cur[i+m])  # from h[i] term
                    g[i+m] -= h_cur[i]  # from -h[i]*h[i+m] term
            grads.append(g)
        grads = np.array(grads)
        
        # We want to find delta such that:
        # f_k(h + delta) ≈ f_k(h) + grad_k · delta is equalized for all k in top
        # Also: sum(delta) = 0 (to maintain constraint)
        # And: h + delta stays in [0,1]
        
        # Simple approach: take gradient in direction that reduces the spread
        # d = weighted average of gradients, weighted to reduce spread
        mean_val = top_vals.mean()
        weights = np.clip(top_vals - mean_val, 0, None)
        if weights.sum() < 1e-15:
            weights = np.ones(n_top)/n_top
        weights = weights / weights.sum()
        
        # Direction that reduces the max
        direction = -np.sum(weights[:, None] * grads, axis=0)
        
        # Project to maintain constraint
        direction -= direction.mean()
        
        # Line search
        alpha = 0.01
        best_h = h_cur
        best_score = compute_c5(h_cur)
        for _ in range(50):
            h_try = h_cur + alpha * direction
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_h = h_try
                best_score = sc
            alpha *= 0.8
            if alpha < 1e-10:
                break
        
        h_cur = best_h
    
    return h_cur, compute_c5(h_cur)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def lbfgs_step(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]
    
    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]
    
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50, 'maxfun': maxiter * 10})
        if res.success or res.fun < 1e-300:
            return project_feasible(res.x, target)
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def active_equalize(h, target, n_top=8, max_iter=200, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        spread = top_vals[-1] - top_vals[0]
        
        if spread < 1e-12:
            break
        
        # Compute weighted gradient to reduce the max lags
        # Weight higher lags more
        w = top_vals - top_vals.min()
        w = w ** 2  # emphasize the highest
        if w.sum() < 1e-30:
            w = np.ones(n_top) / n_top
        else:
            w = w / w.sum()
        
        grad = np.zeros(n)
        for ii in range(n_top):
            ki = top_idx[ii]
            wi = w[ii]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_cur[j2 - m]
        
        # Also subtract gradient of mean (to maintain sum constraint)
        grad -= grad.mean()
        
        # Line search
        lr = 0.02
        for _ in range(30):
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)

                cur_score = sc
                lr = min(lr * 1.1, 0.1)
                break
            lr *= 0.7

                break
        else:

    return h_cur, cur_score

def freq_perturb(h, target, rng, n_modes=5, amp=0.1):
    n = len(h)
    freqs = np.fft.rfftfreq(n)
    # Pick random low-frequency modes
    nonzero = np.where(freqs > 0)[0]
    chosen = rng.choice(nonzero, size=min(n_modes, len(nonzero)), replace=False)
    pert = np.zeros(n)
    for f in chosen:
        pert += np.cos(2 * np.pi * freqs[f] * np.arange(n)) * rng.randn()
    h_new = h + amp * pert
    return project_feasible(h_new, target)

def block_flip(h, target, rng, n_blocks=3, block_size=10):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, n - block_size)
        end = start + block_size
        h_new[start:end] = 1.0 - h_new[start:end]
    return project_feasible(h_new, target)

def coordinate_descent(h, target, n_sweeps=5, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            
            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')

            fc = f1d(c)

                if fc < fd:

                    d = c
                    fd = fc
                    c = b - gr * (b - a)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_step(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50, 'maxfun': maxiter * 15})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def active_equalize(h, target, n_top=10, max_iter=200, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        spread = top_vals[-1] - top_vals[0]
        if spread < 1e-12:
            break

        w = top_vals - top_vals.min()
        w = w ** 2
        if w.sum() < 1e-30:
            w = np.ones(n_top) / n_top
        else:
            w = w / w.sum()

        grad = np.zeros(n)
        for ii in range(n_top):
            ki = top_idx[ii]
            wi = w[ii]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += wi * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= wi * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += wi * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h_cur[j2 - m]

        grad -= grad.mean()
        lr = 0.03
        improved_this_iter = False
        for _ in range(30):
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < cur_score:
                h_cur = h_try
                cur_score = sc
                improved_this_iter = True
                lr = min(lr * 1.15, 0.15)
                break
            lr *= 0.7
            if lr < 1e-12:
                break
        if not improved_this_iter:
            break

    return h_cur, cur_score

def freq_perturb(h, target, rng, n_modes=5, amp=0.1):
    n = len(h)
    freqs = np.fft.rfftfreq(n)
    nonzero = np.where(freqs > 0)[0]
    chosen = rng.choice(nonzero, size=min(n_modes, len(nonzero)), replace=False)
    pert = np.zeros(n)
    for f in chosen:
        pert += np.cos(2.0 * np.pi * freqs[f] * np.arange(n)) * rng.randn()
    h_new = h + amp * pert
    return project_feasible(h_new, target)


def block_flip(h, target, rng, n_blocks=3, block_size=10):
    n = len(h)
    h_new = h.copy()
    for _ in range(n_blocks):
        start = rng.randint(0, max(1, n - block_size))
        end = min(start + block_size, n)
        h_new[start:end] = 1.0 - h_new[start:end]
    return project_feasible(h_new, target)


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur
