def project_box_sum(x, target):
    """Project x onto {h: 0 <= h_i <= 1, sum(h) = target} using water-filling."""
    # We need to find lambda such that sum(clip(x_i + lambda, 0, 1)) = target

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:]
    return float(np.max(over)) * dx

def project_box_sum(x, target):
    """Exact projection onto {h: 0<=h_i<=1, sum(h)=target} via bisection on offset."""
    # Find lambda such that sum(clip(x_i + lambda, 0, 1)) = target
    # clip(x + lambda, 0, 1) is non-decreasing in lambda
    lo = -x.max() - 2.0
    hi = -x.min() + 2.0 + target
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if np.isclose(s, target, atol=1e-15):
            break
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) * 0.5
    return np.clip(x + lam, 0.0, 1.0)

def grad_f_k(h, k, n):
    """Gradient of f_k(h) = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) w.r.t. h."""
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:k] = -h[k:2*k] if 2*k <= n else -h[k:]
        if n - k > k:
            mid = np.arange(k, n - k)
            grad[mid] = 1.0 - h[mid - k] - h[mid + k]
        if n - k < n:
            grad[n-k:] = 1.0 - h[:k]
    return grad

def obj_active_max(h, target, active_lags, smoothing):
    """Objective: smoothed max of f_k over active lags."""
    n = len(h)
    dx = 2.0 / n
    vals = np.zeros(len(active_lags))

            vals[i] = np.sum(h * (1-h)) * dx

            vals[i] = np.sum(h[k:] * (1-h[:-k])) * dx

def bang_bang_with_slack(h, target, n_slack):
    """
    Given h, identify the n_slack coordinates closest to 0.5 (most "slack")
    and optimize only those, keeping others at their current values (clipped to 0 or 1).
    """
    n = len(h)
    # Sort by distance from 0.5 (most intermediate first)
    dist = np.abs(h - 0.5)
    order = np.argsort(dist)
    free_idx = order[:n_slack]
    fixed_idx = order[n_slack:]
    
    # Set fixed coordinates to 0 or 1
    h_bb = h.copy()
    h_bb[fixed_idx] = np.where(h_bb[fixed_idx] >= 0.5, 1.0, 0.0)
    
    # Determine the required sum for free coordinates
    fixed_sum = h_bb[fixed_idx].sum()
    free_target = target - fixed_sum
    
    # Check feasibility
    if free_target < 0 or free_target > n_slack:
        return h, compute_c5(h)
    
    # Optimize free coordinates
    # min max_k f_k(h) s.t. sum(h[free]) = free_target, 0 <= h[free] <= 1
    # This is a small problem (n_slack variables)
    
    def obj_slack(x_free):
        h_full = h_bb.copy()
        h_full[free_idx] = x_free
        return compute_c5(h_full)
    
    x0 = h_bb[free_idx].copy()
    # Use scipy minimize on the small problem
    from scipy.optimize import minimize as sp_min
    res = sp_min(obj_slack, x0, method='Nelder-Mead',

    # Project to ensure sum constraint
    h_opt = project_box_sum(h_opt, target)
    return h_opt, compute_c5(h_opt)
