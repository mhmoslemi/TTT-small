import numpy as np
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = int(np.sum(mask))
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h_init, n, target, max_iter, time_limit, start_time,
                   deadline_offset=30, damping=0.8, n_top=0, solver_list=None):
    if solver_list is None:
        solver_list = ['GLOP', 'CLP', 'ECOS', 'SCS']
    dx = 2.0 / n
    h = project_feasible(h_init.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_full = 2 * n - 1
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        if n_top > 0 and n_top < n_full:
            top_idx = np.argsort(corr)[-n_top:]
        else:
            top_idx = np.arange(n_full)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for idx in top_idx:
            k = int(idx) - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in solver_list:
            try:
                if solver == 'SCS':
                    prob.solve(solver=solver, max_iters=100000, verbose=False)
                elif solver == 'ECOS':
                    prob.solve(solver=solver, max_iter=100000, verbose=False)
                else:
                    prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def smooth_max_gradient(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k in range(len(weights)):
        w = weights[idx_k]
        if w < 1e-12:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            n_rem = n - k
            if n_rem > 0:
                grad[k:] += w * (1.0 - h[:n_rem])
                grad[:n_rem] -= w * h[k:]
        else:
            m = -k
            n_rem = n - m
            if n_rem > 0:
                grad[:n_rem] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n_rem]
    return grad * dx
