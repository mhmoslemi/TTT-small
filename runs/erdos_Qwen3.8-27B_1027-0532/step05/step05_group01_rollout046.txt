def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    """Compute g such that corr_new ≈ corr + delta * g
    when h[i] -= delta and h[j] += delta"""
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    
    # For index p with sign sigma (sigma=+1 means p is in the "increase" group, -1 means "decrease"):
    # Contribution to g from p:
    #   As h[m]: sigma * (1 - h[p+k]) for valid k
    #   As h[m+k]: -sigma * h[p-k] for valid k
    
    # For i (sigma = -1, since h[i] decreases):
    #   As h[m]: -1 * (1 - h[i+k]) = h[i+k] - 1
    #   As h[m+k]: -(-1) * h[i-k] = h[i-k]
    
    # For j (sigma = +1, since h[j] increases):
    #   As h[m]: +1 * (1 - h[j+k])
    #   As h[m+k]: -(+1) * h[j-k] = -h[j-k]
    
    # i as h[m]: valid when 0 <= i+k < n, i.e., -i <= k < n-i
    s_lo = n - 1 - i  # s = k + n - 1, so k = -i gives s = n-1-i
    s_hi = 2*n - 2 - i  # k = n-i-1 gives s = 2n-2-i
    s_lo = max(0, s_lo)
    s_hi = min(n_corr - 1, s_hi)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        k_vals = sv - (n - 1)
        h_idx = i + k_vals  # = i + s - (n-1) = s - n + 1 + i
        g[sv] += h[h_idx] - 1.0  # = -(1 - h[i+k])
    
    # i as h[m+k] (m = i-k): valid when 0 <= i-k < n, i.e., i-n < k <= i
    s_lo = i + 1  # k = i-n+1 gives s = i+1... wait
    # k > i-n means s-(n-1) > i-n means s > i-1, so s >= i
    # k <= i means s-(n-1) <= i means s <= n-1+i
    s_lo = max(0, i)  # Actually s >= i? Let me recheck.
    # i-n < k <= i
    # s = k + n - 1
    # i-n < s-(n-1) <= i
    # i-n+n-1 < s <= i+n-1
    # i-1 < s <= i+n-1
    # So s ranges from i to i+n-1 (since s is integer, i-1 < s means s >= i)
    s_lo = max(0, i)
    s_hi = min(n_corr - 1, i + n - 1)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        k_vals = sv - (n - 1)
        h_idx = i - k_vals  # = i - (s - (n-1)) = i + n - 1 - s
        g[sv] += h[h_idx]  # = +h[i-k]
    
    # j as h[m]: valid when -j <= k < n-j
    s_lo = max(0, n - 1 - j)
    s_hi = min(n_corr - 1, 2*n - 2 - j)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        k_vals = sv - (n - 1)
        h_idx = j + k_vals
        g[sv] += 1.0 - h[h_idx]
    
    # j as h[m+k] (m = j-k): valid when j-n < k <= j
    s_lo = max(0, j)
    s_hi = min(n_corr - 1, j + n - 1)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        k_vals = sv - (n - 1)
        h_idx = j - k_vals
        g[sv] -= h[h_idx]
    
    return g
```

def compute_gain_fast(h, i, j, n):
    n_corr = 2 * n - 1
    g = np.zeros(n_corr)
    
    # Term 1: -(1-h[i-k]) for s in [i, i+n-1]
    s_lo = max(0, i)
    s_hi = min(n_corr - 1, i + n - 1)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        idx = i + n - 1 - sv  # = i-k
        g[sv] -= (1.0 - h[idx])
    
    # Term 2: +h[i+k] for s in [n-1-i, 2n-2-i]
    s_lo = max(0, n - 1 - i)
    s_hi = min(n_corr - 1, 2*n - 2 - i)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        idx = i + sv - n + 1  # = i+k
        g[sv] += h[idx]
    
    # Term 3: +(1-h[j-k]) for s in [j, j+n-1]
    s_lo = max(0, j)
    s_hi = min(n_corr - 1, j + n - 1)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        idx = j + n - 1 - sv  # = j-k
        g[sv] += (1.0 - h[idx])
    
    # Term 4: -h[j+k] for s in [n-1-j, 2n-2-j]
    s_lo = max(0, n - 1 - j)
    s_hi = min(n_corr - 1, 2*n - 2 - j)
    if s_lo <= s_hi:
        sv = np.arange(s_lo, s_hi + 1)
        idx = j + sv - n + 1  # = j+k
        g[sv] -= h[idx]
    
    return g
```
