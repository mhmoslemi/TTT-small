def compute_c5(h):
    n = len(h)

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_now = float(np.max(corr)) * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            break

        top_idx = np.argsort(corr)[-top_m:]
        k_shifts = (top_idx - (n - 1)).astype(int)

        grad = compute_grad_at_shifts(h_cur, k_shifts)
        grad /= float(len(k_shifts))
        grad -= np.mean(grad)

        improved = False
        lr = base_lr
        for _ in range(15):
            h_try = project_feasible(h_cur - lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h_cur = h_try
                improved = True
                break
            lr *= 0.5

    return best_h, best_c5


def momentum_sg(h, n, target, time_limit, max_steps=15000, mu=0.85, base_lr=0.025):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    v = np.zeros(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)

        grad = compute_grad_at_shifts(h_cur, [k_star])
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h_cur - base_lr * v, target)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h_cur = h_new

        if no_improve > 500:
            break

    return best_h, best_c5


def autocorr_gradient_rate(h, n, target, time_limit, max_steps=5000, base_lr=0.03):
    """Maximize min autocorrelation (dual perspective)"""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    idxs = np.arange(n)
    no_improve = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr_orig = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c5_now = float(np.max(corr_orig)) * dx

        ac = np.correlate(h_cur, h_cur, mode='full')
        ac_nonzero = ac[np.abs(idxs - (n - 1)) > 0]
        k_min_ac = int(np.argmin(ac[np.abs(np.arange(len(ac)) - (n-1)) > 0]))
        shift_k = k_min_ac - (n - 1)

        if shift_k == 0:
            shift_k = 1

        grad = np.zeros(n)
        v1 = (idxs - shift_k >= 0) & (idxs - shift_k < n)
        v2 = (idxs + shift_k >= 0) & (idxs + shift_k < n)
        grad[v1] += h_cur[idxs[v1] + shift_k]
        grad[v2] += h_cur[idxs[v2] - shift_k]
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        lr = base_lr
        for _ in range(12):
            h_try = project_feasible(h_cur + lr * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-15:
                h_cur = h_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_cur.copy()
                    no_improve = 0
                improved = True
                break
            lr *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break

    return best_h, best_c5


def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    return np.interp(x_new, x_old, h_src)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    t0 = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Initial SLSQP with adaptive alpha
    if tleft() > 180:
        h1 = slsqp_run(best_h, n, target, [100, 500, 2000, 10000, 50000], 800, 1e-15)
        s1 = compute_c5(h1)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Pair coordinate descent
    if tleft() > 250:
        cd_time = min(250, tleft() * 0.25)
        h2, s2 = pair_cd_fast(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=8000, n_cand=35, top_m=3)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: SLSQP after CD
    if tleft() > 130:
        h3 = slsqp_run(best_h, n, target, [5000, 20000, 80000], 800, 1e-15)
        s3 = compute_c5(h3)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Gradient equalization (multi-shift)
    if tleft() > 100:
        h4, s4 = gradient_descent_eq(best_h, n, target,
                                     time_limit=min(80, tleft() * 0.35),
                                     max_steps=10000, base_lr=0.04, top_m=8)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Autocorrelation-based gradient (dual view)
    if tleft() > 80:
        h5, s5 = autocorr_gradient_rate(best_h, n, target,
                                        time_limit=min(60, tleft() * 0.3),
                                        max_steps=8000, base_lr=0.03)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Pair CD round 2 with more candidates
    if tleft() > 200:
        cd_time = min(180, tleft() * 0.25)
        h6, s6 = pair_cd_fast(best_h, n, target,
                              time_limit=cd_time,
                              max_sweeps=6000, n_cand=45, top_m=5)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: SLSQP after CD2
    if tleft() > 100:
        h7 = slsqp_run(best_h, n, target, [10000, 50000, 150000], 600, 1e-15)
        s7 = compute_c5(h7)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Momentum subgradient for escaping plateaus
    if tleft() > 70:
        h8, s8 = momentum_sg(best_h, n, target,
                             time_limit=min(55, tleft() * 0.4),
                             max_steps=20000, mu=0.85, base_lr=0.02)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Multi-restart with perturbations
    rng = np.random.RandomState(seed)
    for scale in [0.02, 0.05, 0.1]:
        if tleft() < 120:
            break
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        h9 = slsqp_run(h_pert, n, target, [500, 2000, 10000, 50000], 600, 1e-15)
        s9 = compute_c5(h9)
        if s9 < best_score:
            best_score = s9
            best_h = h9
        if tleft() > 80:
            h9b, s9b = pair_cd_fast(h_pert, n, target,
                                    time_limit=min(50, tleft() * 0.3),
                                    max_sweeps=3000, n_cand=35, top_m=3)
            if s9b < best_score:
                best_score = s9b
                best_h = h9b

    # Phase 10: Final pair CD
    if tleft() > 60:
        h10, s10 = pair_cd_fast(best_h, n, target,
                                time_limit=min(50, tleft() * 0.5),
                                max_sweeps=4000, n_cand=45, top_m=5)
        if s10 < best_score:
            best_score = s10
            best_h = h10

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        h11 = slsqp_run(best_h, n, target, [30000, 100000, 300000], 500, 1e-15)
        s11 = compute_c5(h11)
        if s11 < best_score:
            best_score = s11
            best_h = h11

    # Final feasibility check
    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
