       h = np.clip(h, 0.0, 1.0)
       residual = target_sum - np.sum(h)
       if abs(residual) > 1e-10:
           if residual > 0:
               idx2 = np.where(h < 1.0)[0]
           else:
               idx2 = np.where(h > 0.0)[0]
           if len(idx2) > 0:
               h[idx2] += residual / len(idx2)
               h = np.clip(h, 0.0, 1.0)
       return h

   def smooth_corr_obj(h, alpha, penalty):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       c_max = np.max(corr)
       shifted = alpha * (corr - c_max)
       log_sum = c_max + np.log(np.sum(np.exp(shifted))) / alpha
       penalty_term = penalty * (np.sum(h) - n/2.0)**2
       return log_sum * dx + penalty_term

   def smooth_corr_grad(h, alpha, penalty):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       c_max = np.max(corr)
       shifted = alpha * (corr - c_max)
       exp_shifted = np.exp(shifted)
       Z = np.sum(exp_shifted)
       weights = exp_shifted / Z  # sums to 1
       
       grad = np.zeros(2*n-1)  # gradient of corr w.r.t. each... no
       # Actually, I need grad of the objective w.r.t. h
       grad_h = np.zeros(n)
       
       for k in range(-(n-1), n):
           w = weights[k + n - 1] * dx
           if w < 1e-15:
               continue
           # Term 1: w * (1-h[j+k]) for j where 0 <= j+k < n
           js = max(0, -k)
           je = min(n-1, n-1-k)
           if js <= je:
               grad_h[js:je+1] += w * (1.0 - h[js:je+1+k])
           # Term 2: -w * h[j-k] for j where 0 <= j-k < n
           js2 = max(0, k)
           je2 = min(n-1, n-1+k)
           if js2 <= je2:
               grad_h[js2:je2+1] -= w * h[js2-k:je2-k+1]
       
       # Penalty gradient
       grad_h += 2.0 * penalty * (np.sum(h) - n/2.0)
       
       return grad_h

   def lbfgsb_optimize(h_init, n, alpha, maxiter, penalty=1e8):
       bounds = [(0.0, 1.0)] * n
       
       def fun(x):
           return smooth_corr_obj(x, alpha, penalty)
       
       def jac(x):
           return smooth_corr_grad(x, alpha, penalty)
       
       res = minimize(fun, h_init, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-12, 'maxls': 50,
                                               'maxfun': maxiter*5})
       return res.x

   def coordinate_descent(h_init, n, target, time_limit, max_sweeps=10000):

       # and for term 2 is [max(0,k), min(n-1, n-1+k)]
       
       no_improve = 0

               best_h = h.copy()
               no_improve = 0
           else:
               no_improve += 1
               if no_improve > 200:
                   break
           
           # For each coordinate j, find the best delta
           for j in range(n):
               # Compute g_k(j) = (1-h[j+k])*[valid] - h[j-k]*[valid]
               # We need this for all k in [-(n-1), n-1]
               g = np.zeros(n_corr)
               # Term 1: (1-h[j+k]) for k where 0 <= j+k < n => k in [-j, n-1-j]
               k_min1 = max(-(n-1), -j)
               k_max1 = min(n-1, n-1-j)
               if k_min1 <= k_max1:
                   indices = np.arange(j+k_min1, j+k_max1+1)  # j+k values
                   g[k_min1+(n-1):k_max1+(n-1)+1] += (1.0 - h[indices])
               
               # Term 2: -h[j-k] for k where 0 <= j-k < n => k in [j-n+1, j]
               k_min2 = max(-(n-1), j-n+1)
               k_max2 = min(n-1, j)
               if k_min2 <= k_max2:
                   indices = np.arange(j-k_max2, j-k_min2+1)  # j-k values
                   g[k_min2+(n-1):k_max2+(n-1)+1] -= h[indices]
               
               # Find optimal delta that minimizes max(corr + delta * g)
               # subject to 0 <= h[j] + delta <= 1
               delta_lo = -h[j]
               delta_hi = 1.0 - h[j]
               
               # Candidate deltas: 0, boundaries, and crossing points of top lines
               candidates = [0.0, delta_lo, delta_hi]
               
               # Find top m lines and their crossing points
               m = 20
               top_idx = np.argsort(corr)[-m:]

                           if delta_lo < d_cross < delta_hi:
                               candidates.append(d_cross)
               
               # Also add some uniform candidates
               n_uniform = 10
               if delta_hi - delta_lo > 1e-10:
                   candidates.extend(np.linspace(delta_lo, delta_hi, n_uniform).tolist())

               best_max = np.max(corr)

                   h[j] = np.clip(h[j], 0.0, 1.0)
           
           # Project to feasible set after each sweep
           h = project_feasible(h, target)
       
       return best_h, best_c5

   Actually, a better approach: instead of pairwise crossings, use the fact that the optimal δ is where the subgradient of max(corr + δ*g) contains 0. The subgradient is the convex hull of {g_k : c_k + δ*g_k = max(corr + δ*g)}. At the optimum, 0 is in this convex hull.

   This means there exist active k's with some having g_k > 0 and some with g_k < 0 (and the weighted average is 0).

   A simple algorithm:
   1. Start with δ = 0
   2. Find the active set A = {k: c_k + δ*g_k = max}
   3. If all g_k > 0 for k in A: decrease δ (the max is increasing)
   4. If all g_k < 0 for k in A: increase δ
   5. Otherwise: the optimal δ is where two active lines cross

   For step 5: find k1 in A with g_k1 < 0 and k2 in A with g_k2 > 0 (or one = 0). The crossing is at δ* = (c_k2 - c_k1) / (g_k1 - g_k2).
