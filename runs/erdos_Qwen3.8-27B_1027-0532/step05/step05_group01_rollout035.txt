        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _lbfgs_obj(h):
    n = _global_n
    alpha = _global_alpha
    mu = _global_mu
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    s = np.sum(h) - _global_target
    obj += mu * s * s
    return obj


def _lbfgs_grad(h):
    n = _global_n
    alpha = _global_alpha
    mu = _global_mu
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    w = np.exp(shifted - log_sum)
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    grad = np.zeros(n)
    active = np.where(w > 1e-11)[0]
    for s in active:
        k = int(s) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += w[s] * (1.0 - h[idxs[v1] - k])
        grad[v2] += w[s] * (-h[idxs[v2] + k])
    s_diff = np.sum(h) - _global_target
    grad += 2.0 * mu * s_diff
    return grad


def lbfgs_round(h_start, n, target, alphas, maxiter, mu=200.0):
    global _global_n, _global_target, _global_alpha, _global_mu
    _global_n = n
    _global_target = target
    _global_mu = mu
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(_lbfgs_obj, h_cur, jac=_lbfgs_grad,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15,
                                    'gtol': 1e-9, 'maxfun': maxiter * 30,
                                    'maxls': 50, 'm': 20})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            c5_cur = compute_c5(h_cur)
            if c5_new < c5_cur:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def compute_g_matrix(candidate_indices, h, n):
    n_corr = 2 * n - 1
    mid = n - 1
    n_cand = len(candidate_indices)
    g_mat = np.zeros((n_cand, n_corr))
    for c, idx in enumerate(candidate_indices):
        s_lo = max(0, idx + 1 - n)
        s_hi = min(n_corr - 1, idx + mid)
        if s_lo <= s_hi:
            s_range = np.arange(s_lo, s_hi + 1)
            h_indices = idx + mid - s_range
            g_mat[c, s_range] += (1.0 - h[h_indices])
        s_lo2 = max(0, mid - idx)
        s_hi2 = min(n_corr - 1, n - 1 + mid - idx)
        if s_lo2 <= s_hi2:
            s_range2 = np.arange(s_lo2, s_hi2 + 1)
            h_indices2 = idx + s_range2 - mid
            g_mat[c, s_range2] -= h[h_indices2]
    return g_mat


def pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=40):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 300:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        all_cands = np.unique(np.concatenate([low_set, high_set]))
        high_mask = np.zeros(len(all_cands), dtype=bool)
        low_mask = np.zeros(len(all_cands), dtype=bool)
        for ci, ci_val in enumerate(all_cands):
            if ci_val in high_set:
                high_mask[ci] = True
            if ci_val in low_set:
                low_mask[ci] = True

        g_mat = compute_g_matrix(all_cands, h, n)

        best_trial = c5_val
        best_i, best_j, best_delta = -1, -1, 0.0

        high_idx_list = np.where(high_mask)[0]
        low_idx_list = np.where(low_mask)[0]

        for ci in high_idx_list:
            for cj in low_idx_list:
                if ci == cj:
                    continue
                i = int(all_cands[ci])
                j = int(all_cands[cj])
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g = g_mat[ci] - g_mat[cj]
                f_mid_val = np.max(corr)
                active_mask0 = corr >= f_mid_val - 1e-14
                f_prime0 = np.max(g[active_mask0])
                if f_prime0 >= 0:
                    continue

                a_d, b_d = delta_lo, delta_hi
                for _ in range(16):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_v = np.max(vals)
                    a_m = vals >= f_v - 1e-14
                    if a_m.any():
                        f_p = np.max(g[a_m])
                    else:
                        f_p = 0.0
                    if f_p > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial:
                    best_trial = approx_max
                    best_i, best_j, best_delta = i, j, delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def sobol_like_starts(n, target, n_starts=5, seed=42):
    rng = np.random.RandomState(seed)
    starts = []
    for s in range(n_starts):
        if s == 0:
            h = np.full(n, 0.5)
        elif s == 1:
            h = np.where(np.arange(n) < n // 2, 1.0, 0.0).astype(float)
        elif s == 2:
            t = 2.0 * np.pi * (np.arange(n) + 0.5) / (n // 4)
            h = 0.5 + 0.3 * np.sin(t)
        elif s == 3:
            t = np.arange(n) / n
            h = np.where(t < 0.25, 1.0, np.where(t < 0.5, 0.0, np.where(t < 0.75, 1.0, 0.0)))
        else:
            h = rng.rand(n) * 0.5 + 0.25
        h = project_feasible(h, target)
        starts.append(h)
    return starts


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha, _global_mu
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: L-BFGS-B initial polish
    if tleft() > 250:
        h_pol = lbfgs_round(best_h, n, target,
                            [50, 200, 1000, 5000, 20000, 100000],
                            maxiter=300, mu=200.0)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Pair CD
    if tleft() > 250:
        cd_time = min(250, tleft() * 0.2)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time, max_sweeps=5000, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: L-BFGS-B refinement
    if tleft() > 200:
        h_pol = lbfgs_round(best_h, n, target,
                            [2000, 10000, 50000, 200000, 1000000],
                            maxiter=200, mu=500.0)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Another pair CD round
    if tleft() > 200:
        cd_time = min(180, tleft() * 0.25)
        h_cd, c5_cd = pair_cd(best_h, n, target,
                              time_limit=cd_time, max_sweeps=5000, n_cand=50)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 5: L-BFGS-B at high alpha
    if tleft() > 150:
        h_pol = lbfgs_round(best_h, n, target,
                            [50000, 200000, 1000000, 5000000],
                            maxiter=200, mu=1000.0)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 6: Try higher resolution
    if tleft() > 250:
        for n_try in [600, 800]:
            if tleft() < 180:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = lbfgs_round(h_interp, n_try, target_try,
                                  [50, 200, 1000, 5000, 20000, 50000],
                                  maxiter=250, mu=200.0)
            sc = compute_c5(h_hires)
            if sc < best_score - 1e-6:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

            if tleft() > 120:
                cd_t = min(80, tleft() * 0.3)
                h_cd, c5_cd = pair_cd(h_hires, n_try, target_try,
                                      time_limit=cd_t, max_sweeps=2000, n_cand=30)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd
                    n = n_try
                    target = target_try

            if tleft() > 80:
                h_pol2 = lbfgs_round(best_h, n, target,
                                     [20000, 100000, 500000],
                                     maxiter=150, mu=500.0)
                sc2 = compute_c5(h_pol2)
                if sc2 < best_score:
                    best_score = sc2
                    best_h = h_pol2

    # Phase 7: Diverse restarts
    rng = np.random.RandomState(seed)
    n_use = len(best_h)
    target_use = n_use / 2.0
    for noise_scale in [0.01, 0.03, 0.05, 0.08, 0.12]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert = project_feasible(h_pert, target_use)
        h_ref = lbfgs_round(h_pert, n_use, target_use,
                            [100, 1000, 10000, 100000, 500000],
                            maxiter=200, mu=200.0)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Block restarts at structured starts
    if tleft() > 120:
        starts = sobol_like_starts(n_use, target_use, n_starts=3, seed=seed + 1)
        for h_s in starts:
            if tleft() < 60:
                break
            h_ref = lbfgs_round(h_s, n_use, target_use,
                                [50, 500, 5000, 50000, 500000],
                                maxiter=200, mu=200.0)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: Final pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_t = min(50, tleft() * 0.5)
        h_cd, c5_cd = pair_cd(best_h, n_use, target_use,
                              time_limit=cd_t, max_sweeps=3000, n_cand=40)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Final L-BFGS-B polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = lbfgs_round(best_h, n_use, target_use,
                              [50000, 200000, 1000000, 5000000, 20000000],
                              maxiter=300, mu=1000.0)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
