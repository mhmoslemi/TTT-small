    dx = 2.0 / n

    # But we need to be smart about which positions to choose
    # Use a greedy approach: sort by value descending, take top target

import numpy as np
from scipy.optimize import minimize
import time

def compute_f(h):
    """Compute f(k) for all valid shifts k."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def compute_c5(h):
    return float(np.max(compute_f(h)))

def to_bang_bang(h, target):
    """Round h to bang-bang preserving the sum constraint."""
    n = len(h)
    # Sort indices by h value descending
    order = np.argsort(h)[::-1]
    h_bb = np.zeros(n)
    h_bb[order[:int(target)]] = 1.0
    return h_bb

def flip_effect_binary(h, j):
    """Compute delta f when flipping h[j] (assuming h[j] is 0 or 1)."""
    n = len(h)
    dx = 2.0 / n
    delta = np.zeros(n)
    val = h[j]
    n_minus_j = n - j
    if val < 0.5:
        # 0 -> 1
        if n_minus_j > 1:
            delta[0:n_minus_j-1] += (1.0 - h[j+1:n]) * dx
        if j >= 1:
            delta[0:j] -= h[0:j] * dx
    else:
        # 1 -> 0
        if n_minus_j > 1:
            delta[0:n_minus_j-1] -= (1.0 - h[j+1:n]) * dx
        if j >= 1:
            delta[0:j] += h[0:j] * dx
    return delta

def greedy_local_search(h, max_passes=200):
    """Best-improvement local search on binary space."""
    n = len(h)
    h_cur = h.copy()
    f_cur = compute_f(h_cur)
    cur_max = float(np.max(f_cur))
    
    for _ in range(max_passes):
        best_obj = cur_max
        best_j = -1
        best_f = None
        for j in range(n):
            delta = flip_effect_binary(h_cur, j)
            f_new = f_cur + delta
            obj = float(np.max(f_new))
            if obj < best_obj - 1e-15:

                best_j = j
                best_f = f_new
        if best_j >= 0:
            h_cur[best_j] = 1.0 - h_cur[best_j]
            f_cur = best_f
            cur_max = best_obj
        else:

def compute_f(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def compute_c5(h):
    return float(np.max(compute_f(h)))

def flip_delta(h, j):
    n = len(h)
    dx = 2.0 / n
    d = np.zeros(n)

    if h[j] < 0.5:
        if r > 1:
            d[:r-1] += (1.0 - h[j+1:n]) * dx
        if j > 0:
            d[:j] -= h[:j] * dx
    else:
        if r > 1:
            d[:r-1] -= (1.0 - h[j+1:n]) * dx
        if j > 0:
