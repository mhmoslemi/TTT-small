Actually, one more thought on the pairwise CD efficiency. The main bottleneck is the ternary search (40 iterations × O(n) each = 40n operations per pair). Can I do better?

Since f(t) = max_k (c_k + d_k * t) is convex and piecewise linear, I can find the minimum more efficiently:

Observation: The minimum of f(t) over [lo, hi] occurs at:
- A point where the subgradient contains 0, i.e., where there exist k1, k2 with d_{k1} ≤ 0 ≤ d_{k2} and c_{k1} + d_{k1}*t = c_{k2} + d_{k2}*t = f(t), or
- An endpoint lo or hi

The point where two lines cross is: t = (c_{k1} - c_{k2}) / (d_{k2} - d_{k1})

So the algorithm is:
1. Evaluate f at lo and hi.
2. Find the "active" lines at lo (those achieving the max at lo) and at hi.
3. The minimum is at the intersection of the line with the smallest positive slope (active at some point) and the line with the largest negative slope (active at some point), or at an endpoint.

But finding the active set at a point is O(n), and finding the right pair of lines is complex.

A simpler efficient approach: use the fact that the minimum of a convex PL function can be found by binary search on the derivative:
- The "derivative" (slope) of f at t is the set of slopes of active lines
- f is minimized where the slope transitions from negative to positive
- Binary search: at t_mid, if the max active slope < 0, the minimum is to the right; if the min active slope > 0, the minimum is to the left; otherwise we're at the minimum.

Each evaluation is O(n), and binary search takes log2(1/ε) iterations. For ε=1e-10 (which is at floating point precision), that's about 33 iterations. Similar to ternary search.

Actually, for a convex PL function, the bisection on the derivative converges in O(log(1/ε)) steps to the exact minimum (since the minimum is at a breakpoint, and bisection will locate it). This is slightly better than ternary search in theory.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def pair_cd(h_init, n, target, time_limit, start_time, deadline_offset=30, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()

    corr = compute_corr(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    it = 0
    while time.time() - start_time < time_limit - deadline_offset:
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        s_val = h[i] + h[j]
        lo = max(0.0, s_val - 1.0)
        hi = min(1.0, s_val)
        if lo >= hi - 1e-12:
            continue
        sh = shifts
        v1_i = (sh >= i - n + 1) & (sh <= i)
        idx1_i = np.clip(i - sh, 0, n - 1)
        v2_i = (sh >= -i) & (sh <= n - 1 - i)
        idx2_i = np.clip(i + sh, 0, n - 1)
        coeff_i = v1_i * (1.0 - h[idx1_i]) - v2_i * h[idx2_i]
        v1_j = (sh >= j - n + 1) & (sh <= j)
        idx1_j = np.clip(j - sh, 0, n - 1)

            t_opt = lo

                m2 = b2 - (b2 - a2) / 3.0
                f1 = np.max(c_vec + d * m1)
                f2 = np.max(c_vec + d * m2)
                if f1 < f2:
                    b2 = m2
                else:
                    a2 = m1
                if b2 - a2 < 1e-14:

            h[j] = s_val - t_opt
            corr = new_corr
            best_c5 = new_c5
            best_h = h.copy()

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.astype(float).copy()
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})

                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad

            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-12:
                break
            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 2.0)
            improved = False
            for _ in range(30):

                h_try = project_feasible(h_try, target)

                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 80:
                    break
    return best_h, best_score
