        if n_free == 0:
            break
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj(h, alpha, target, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    # constraint penalty
    sv = np.sum(h) - target
    val += 1e6 * sv * sv
    return val

def _smooth_grad(h, alpha, target, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if top_m is not None and top_m < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_m:]
        sub = corr[top_idx]
    else:
        top_idx = np.arange(2 * n - 1)
        sub = corr
    
    shifted = np.clip(alpha * (sub - c_max), -600.0, 0.0)
    exp_s = np.exp(shifted)
    ws = np.sum(exp_s)
    if ws < 1e-300:
        weights = np.ones(len(sub)) / len(sub)
    else:
        weights = exp_s / ws
    
    grad = np.zeros(n)
    nh = n - 1
    
    for idx_k, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k = int(idx_k) - nh
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            m = n - k
            if m > 0:
                grad[k:] += w * (1.0 - h[:m]) * dx
                grad[:m] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n-m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n-m] * dx
    
    # constraint gradient
    sv = np.sum(h) - target
    grad += 2e6 * sv * np.ones(n)
    
    return grad

def lbfgs_optimize(h_init, n, target, alpha, top_m, maxiter, time_limit, start_time, margin=10):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    
    # Custom callback to check time
    state = {'nfev': 0, 'halt': False}
    
    def fun(x):
        state['nfev'] += 1
        if time.time() - start_time > time_limit - margin:
            state['halt'] = True
        return _smooth_obj(x, alpha, target, dx)
    
    def jac(x):
        if state['halt']:
            return np.zeros(n)
        return _smooth_grad(x, alpha, target, dx, top_m=top_m)
    
    try:
        res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': 50000, 'maxls': 50})
        if res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new, compute_c5(h_new)
    except Exception:
        pass
    
    return h, compute_c5(h)
