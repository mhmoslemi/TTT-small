import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def top_lag_gradient_stage(h, n, target, n_steps, n_top, deadline):
    """Aggressive gradient descent targeting the top n_top lags."""
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        # The overlap part is corr[n-1 : 2*n-1] (lags 0 to n-1)
        over = corr[n-1 : 2*n-1]
        # Find top n_top lags
        top_k = np.argsort(over)[::-1][:n_top]
        c_max = over[top_k[0]]
        
        # Compute weighted gradient (weight proportional to how much above median)
        median_c = np.median(over)
        weights = np.zeros(n)
        for k in top_k:
            w = (over[k] - median_c) / (2 * (c_max - median_c + 1e-15)) + 0.3
            # Gradient of c(k) w.r.t. h
            kp = int(k)
            for j in range(n - kp):
                weights[j] += w * (1.0 - h_cur[j + kp])
            for j in range(kp, n):
                weights[j] -= w * h_cur[j - kp]
        weights *= dx
        
        # Try multiple step sizes (line search)
        best_step_h = None
        best_step_score = c_max * dx
        for lr in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:

                best_step_h = h_new

            if sc < best_score:
                best_score = best_step_score
                best_h = h_cur.copy()
        else:

    return best_h, best_score

Actually, I realize I should also try a completely different approach: **alternating minimization**. Here's the idea:

For a fixed set of "active" lags (the top K), the problem of minimizing the maximum of c(k) for k in active set is a subproblem. We can solve this by:
1. Binary search on t: is there a feasible h with c(k) * dx <= t for all active k?
2. The feasibility check is: minimize max_k c(k)*dx subject to constraints, which is still non-convex.

Alternatively, we can use a "penalty" approach:
minimize sum_{k in active} max(0, c(k)*dx - t)² + lambda * (sum(h)*dx - 1)²
subject to 0 <= h <= 1

This is a smooth optimization problem that can be solved with gradient-based methods.

Let me also think about a key insight: the gradient of the MAX of the top correlations is the sum of gradients of those correlations (if they're all at the same maximum). So if we have K lags all at the maximum, the gradient is the sum of their individual gradients. This is exactly what the "top lag gradient" method does.

The issue is that when we take a step in the negative gradient direction, we reduce the top correlations but might increase others. The line search helps with this by finding a step size that doesn't worsen the overall max.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def _lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_optimize(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _g_target,
             'jac': lambda x: np.ones(_g_n)}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def top_lag_gradient(h, n, target, n_steps, n_top, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stall_count = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n-1:2*n-1]
        c_max = float(np.max(over)) * dx
        if c_max < best_score - 1e-15:
            best_score = c_max
            best_h = h_cur.copy()
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                break
        top_k = np.argsort(over)[::-1][:n_top]
        median_c = np.median(over)
        spread = over[top_k[0]] - median_c + 1e-15
        grad = np.zeros(n)
        for k in top_k:
            kp = int(k)
            w = 0.5 + 0.5 * (over[kp] - median_c) / spread
            j_pos = np.arange(0, n - kp)
            grad[j_pos] += w * (1.0 - h_cur[j_pos + kp])
            j_neg = np.arange(kp, n)
            grad[j_neg] -= w * h_cur[j_neg - kp]
        grad *= dx
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-14:
            break
        grad /= grad_norm
        best_step_sc = c_max
        best_step_h = None
        for lr_try in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            h_new = h_cur - lr_try * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_step_sc - 1e-15:
                best_step_sc = sc
                best_step_h = h_new
                best_lr = lr_try
        if best_step_h is not None:
            h_cur = best_step_h
            lr = best_lr * 1.05
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score


def alternating_lag_descent(h, n, target, max_rounds, n_lags, eps_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n-1:2*n-1]
        top_k = np.argsort(over)[::-1][:n_lags]
        improved = False
        for k in top_k:
            if time.time() > deadline:
                break
            kp = int(k)
            grad = np.zeros(n)
            j_pos = np.arange(0, n - kp)
            grad[j_pos] += (1.0 - h_cur[j_pos + kp])
            j_neg = np.arange(kp, n)
            grad[j_neg] -= h_cur[j_neg - kp]
            grad *= dx
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                continue
            grad /= gnorm
            for eps_try in [eps, eps * 0.5, eps * 2.0]:
                h_new = h_cur - eps_try * grad
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_score - 1e-15:
                    h_cur = h_new
                    best_score = sc
                    best_h = h_cur.copy()
                    eps = eps_try * 1.1
                    improved = True
                    break
        if not improved:
            eps *= 0.6
            if eps < 1e-10:
                break
    return best_h, best_score
