import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective_and_grad(x, alpha, pen_coeff, target):
    n = len(x)
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    w = exp_vals / float(np.sum(exp_vals))
    
    lse = c_max + float(np.log(np.sum(exp_vals))) / alpha
    pen = pen_coeff * (np.sum(x) - target) ** 2
    obj = lse + pen
    
    # Gradient
    grad = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-18:
            continue
        k = m - (n - 1)
        if k == 0:
            grad += wm * (1.0 - 2.0 * x)
        elif k > 0:
            # grad[k:] += wm * (1 - x[:n-k])   (first factor)
            # grad[:n-k] -= wm * x[k:]           (second factor)
            grad[k:] += wm * (1.0 - x[:n-k])
            grad[:n-k] -= wm * x[k:]
        else:
            kp = -k
            # grad[:n-kp] += wm * (1 - x[kp:])   (first factor)
            # grad[kp:] -= wm * x[:n-kp]         (second factor)
            grad[:n-kp] += wm * (1.0 - x[kp:])
            grad[kp:] -= wm * x[:n-kp]
    
    grad += 2.0 * pen_coeff * (np.sum(x) - target)
    return obj, grad

    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        pen_coeff = 100.0
        
        def obj_func(x):
            return lse_objective_and_grad(x, alpha, pen_coeff, target)[0]
        
        def grad_func(x):
            return lse_objective_and_grad(x, alpha, pen_coeff, target)[1]
        
        try:
            res_x, res_f, res_info = fmin_l_bfgs_b(
                obj_func, h_cur, fprime=grad_func,
                pgtol=1e-12, maxiter=maxiter,
                # bounds=[(0.0, 1.0)] * n  # L-BFGS-B supports bounds
                bounds=[(0.0, 1.0)] * n,
                iprint=0
            )
            h_new = project_feasible(res_x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new.copy()
            else:
                # Try with lower penalty to see if constraint was the issue
                pass
        except Exception as e:
            pass
    
    return best_h, best_score

def subgradient_polish(h, n, target, n_steps, deadline):
    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    temperature = 0.005
    lr = 0.01
    
    # Running average for adaptive step
    f_hist = [best_score]
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_vals = corr * dx
        
        # Softmax weights
        shifted = (c_vals - c_max) / temperature
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            wm = weights[m]
            if wm < 1e-15:
                continue
            k = m - (n - 1)
            if k == 0:
                grad += wm * (1.0 - 2.0 * h_cur)
            elif k > 0:
                grad[k:] += wm * (1.0 - h_cur[:n-k])
                grad[:n-k] -= wm * h_cur[k:]
            else:
                kp = -k
                grad[:n-kp] += wm * (1.0 - h_cur[kp:])
                grad[kp:] -= wm * h_cur[:n-kp]
        grad *= dx
        
        # Momentum update
        momentum = beta * momentum + (1 - beta) * grad
        momentum_norm = np.linalg.norm(momentum)
        if momentum_norm < 1e-12:
            break
        
        # Adaptive step size
        adaptive_lr = lr / (momentum_norm + 0.1)
        adaptive_lr = min(adaptive_lr, 0.5)
        
        h_new = h_cur - adaptive_lr * momentum
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new.copy()
            h_cur = h_new
            lr = min(lr * 1.05, 0.1)
        else:
            h_cur = project_feasible(0.5 * h_cur + 0.5 * best_h, target)  # move back toward best
            lr *= 0.95
            if lr < 1e-8:
                break
        
        f_hist.append(sc_new)
        if len(f_hist) > 100:
            f_hist.pop(0)
    
    return best_h, best_score

def coordinate_descent(h, n, target, n_passes, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    deltas = [0.001, 0.005, 0.01, 0.02, 0.05]
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in deltas:
                for sign in [1.0, -1.0]:
                    h_try = h_cur.copy()
                    h_try[i] = np.clip(h_cur[i] + sign * d, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-12:
                        h_cur = h_try
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break
    return h_cur, best_score

            if lr_base < 1e-10:
                break

    return best_h, best_score

def worst_lag_descent(h, n, target, n_steps, deadline):
    h_cur = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = 0.01

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        k_star = int(np.argmax(corr)) - (n - 1)

        grad = np.zeros(n)
        if k_star == 0:
            grad = 1.0 - 2.0 * h_cur
        elif k_star > 0:
            grad[k_star:] += (1.0 - h_cur[:n - k_star])
            grad[:n - k_star] -= h_cur[k_star:]
        else:
            kp = -k_star
            grad[:n - kp] += (1.0 - h_cur[kp:])
            grad[kp:] -= h_cur[:n - kp]
        grad *= dx

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        step_size = eps / (gnorm + 0.01)

        h_try = h_cur - step_size * grad
        h_try = project_feasible(h_try, target)
        sc_try = compute_c5(h_try)

        if sc_try < best_score - 1e-15:
            best_score = sc_try
            best_h = h_try.copy()
            h_cur = h_try
            eps = min(eps * 1.05, 0.2)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break

    return best_h, best_score

def coordinate_descent(h, n, target, n_passes, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    deltas = np.array([0.0005, 0.002, 0.005, 0.01, 0.03])

    for p in range(n_passes):
        if time.time() > deadline:
            break
        any_improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for d in deltas:
                improved_here = False
                for sign in [1.0, -1.0]:
                    val = h_cur[i] + sign * d
                    if val < 0.0 or val > 1.0:
                        continue
                    h_try = h_cur.copy()
                    h_try[i] = val
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-14:
                        h_cur = h_try
                        best_score = sc
                        any_improved = True
                        improved_here = True
                        break
                if improved_here:
                    break
        if not any_improved:
            break
    return h_cur, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(h_parent)

    # Phase 1: L-BFGS-B continuation with increasing alpha
    t1 = min(start_time + 180, hard_deadline)
    best_h, best_score = lbfgsb_run(best_h, n_final, target_final,
                                     [200, 1000, 5000], 800, 200.0, t1)

    # Phase 2: Subgradient polish
    t2 = min(start_time + 450, hard_deadline)
    h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 30000, t2, temp=0.005)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg

    # Phase 3: L-BFGS-B with higher alpha
    t3 = min(start_time + 650, hard_deadline)
    best_h, best_score = lbfgsb_run(best_h, n_final, target_final,
                                     [20000, 100000], 1500, 200.0, t3)

    # Phase 4: Worst-lag focused descent
    t4 = min(start_time + 850, hard_deadline)
    h_wl, sc_wl = worst_lag_descent(best_h, n_final, target_final, 10000, t4)
    if sc_wl < best_score:
        best_score = sc_wl
        best_h = h_wl

    # Phase 5: Subgradient with lower temperature
    t5 = min(start_time + 1050, hard_deadline)
    h_sg, sc_sg = subgradient_polish(best_h, n_final, target_final, 30000, t5, temp=0.001)
    if sc_sg < best_score:
        best_score = sc_sg
        best_h = h_sg

    # Phase 6: Coordinate descent
    t6 = min(start_time + 1200, hard_deadline)
    h_cd, sc_cd = coordinate_descent(best_h, n_final, target_final, 8, t6)
    if sc_cd < best_score:
        best_score = sc_cd
        best_h = h_cd

    # Phase 7: Final L-BFGS-B
    t7 = hard_deadline - 10.0
    best_h, best_score = lbfgsb_run(best_h, n_final, target_final,
                                     [500000], 2000, 200.0, t7)

    # Final projection
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
