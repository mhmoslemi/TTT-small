import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_lp(h_init, n, target, time_limit, damping=0.9, reg=1e-6):
    """Solve LP with ALL shift constraints."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    
    for it in range(500):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Add ALL shift constraints (this is the key difference)
        for k_idx in range(1, 2*n - 1):
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=200000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=200000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Decay regularization
        reg = max(1e-8, reg * 0.95)
    
    return best_h, best_c5

from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_lp(h_init, n, target, time_limit, damping_init=0.95, reg_init=1e-5):
    """SLP with ALL constraints."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    reg = reg_init
    half_n = n - 1

    # For each lag offset k in range -(n-1) to (n-1):
    # C[k + half_n] = sum_{i} h[i] * (1-h[i+k]) for valid i
    # In the LP, we linearize (1-h[i+k]) at current h, giving:
    # sum_i (1-h_old[i+k]) * h_new[i] * dx <= t
    
    for it in range(100):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h  # g[i] = 1 - h[i]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in range(-(n-1), n):
            # C index in correlate output: k + half_n
            # C[k+half_n] = sum_i h[i+k] * (1-h[i]) for valid i
            # = sum_i (1-h[i]) * h[i+k]
            # Linearize h[i+k] as variable: we need sum_i (1-h_old[i]) * h_new[i+k]
            # Let j = i+k, so i = j-k: sum_j (1-h_old[j-k]) * h_new[j] for valid j
            # Valid j: 0 <= j < n and 0 <= j-k < n, i.e., max(0,k) <= j < min(n, n+k)
            lo = max(0, k)
            hi = min(n, n + k)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(objective), constraints)

            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=500000)
                solved = h_var.value is not None
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:

            best_h = h.copy()
        
        reg = max(1e-10, reg * 0.9)
        if c5_val > best_c5:

from scipy.sparse import coo_matrix, csr_matrix
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_lp_structure(n):
    """Precompute the sparse structure for all overlap constraints."""
    half_n = n - 1
    rows_list = []
    cols_list = []
    j_minus_s_list = []
    for s in range(-half_n, half_n + 1):
        lo = max(0, s)
        hi = min(n, n + s)
        js = np.arange(lo, hi)
        rows_list.append(np.full(len(js), s + half_n, dtype=np.int32))
        cols_list.append(js.astype(np.int32))
        j_minus_s_list.append((js - s).astype(np.int32))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    j_minus_s = np.concatenate(j_minus_s_list)
    return rows, cols, j_minus_s

def smooth_logsumexp(corr, alpha):
    """Smoothed max via log-sum-exp."""
    c_max = np.max(corr)
    if alpha < 1e10:
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return c_max

def nesterov_gradient(h_init, n, target, time_limit, alpha=1000, step_size=0.001, momentum_init=0.9):
    """Nesterov-accelerated projected gradient on smoothed objective."""
    start = time.time()
    dx = 2.0 / n
    half_n = n - 1
    h = project_feasible(h_init.copy(), target)
    v = h.copy()  # momentum variable
    best_h = h.copy()
    best_score = compute_c5(h)
    momentum = momentum_init
    lr = step_size
    
    for step in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Nesterov lookahead
        y = h + momentum * (h - v)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute gradient of smoothed objective at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        weights = alpha * np.exp(alpha * (corr - np.max(corr)))
        weights = weights / np.sum(weights)
        
        # Gradient is weighted average of subgradients
        grad = np.zeros(n)
        # For each lag k, the subgradient of O(k) = sum_j y[j]*(1-y[j-k])
        # dO(k)/dy[i] = (1-y[i-k]) - y[i+k] (with boundary handling)
        # Multiply by weight_k * dx
        
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-10:
                continue
            k = idx - half_n
            # Contribution from first factor: (1-y[i-k]) for valid i
            lo1 = max(0, k)
            hi1 = min(n, n + k)
            if hi1 > lo1:
                grad[lo1:hi1] += w * (1.0 - y[lo1 - k:hi1 - k])
            # Contribution from second factor: -y[i+k] for valid i
            lo2 = max(0, -k)
            hi2 = min(n, n - k)
            if hi2 > lo2:
                grad[lo2:hi2] -= w * y[lo2 + k:hi2 + k]
        
        grad *= dx
        
        # Update
        v = h.copy()
        h = h - lr * grad
        h = project_feasible(h, target)
        
        score = compute_c5(h)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
        
        # Decay learning rate
        lr = max(1e-5, step_size / (1 + step * 0.01))
        momentum = min(0.999, momentum * (1 + 1e-6))
    
    return best_h, best_score
