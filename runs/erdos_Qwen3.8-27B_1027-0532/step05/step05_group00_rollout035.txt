```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)
```

```python
def fast_swap_delta(h, i, j, n):
    """Compute the delta vector for swapping h[i]=1->0 and h[j]=0->1."""
    delta = np.zeros(2*n - 1)
    # i goes from 1 to 0
    # For shift k (0-indexed in full array, k_full = k + (n-1)):
    # corr[k_full] = sum_valid h[a] * (1-h[a+k]) where k = k_full - (n-1)
    
    # Effect of h[i]: 1->0
    # corr loses h[i]*(1-h[i+k]) at shift k where i+k is valid
    # corr gains h[i-k]*(1-h[i]) at shift -k where i-k is valid
    
    # Shift k means in full array index k + (n-1)
    # h[i]*(1-h[i+k]) is at full index: k + (n-1) where 0 <= i+k < n
    #   k ranges from -i to n-1-i
    #   full index ranges from (n-1-i) to (n-1) + (n-1-i) = 2n-2-i
    
    # Let me just do it directly with loops for clarity, then optimize
    
    # For the "h[i] as first factor" contribution:
    # corr at shift k includes h[i] * (1-h[i+k]) if 0<=i+k<n
    # This is at full index k + (n-1)
    # When h[i] goes 1->0, we subtract (1-h[i+k]) at these positions
    
    k_min = -i  # so that i+k >= 0
    k_max = n - 1 - i  # so that i+k <= n-1
    if k_max >= k_min:
        full_idx_start = k_min + (n-1)
        full_idx_end = k_max + (n-1)
        h_slice = h[i+k_min : i+k_max+1]
        delta[full_idx_start:full_idx_end+1] -= (1.0 - h_slice)
    
    # For the "h[i] as second factor" contribution:
    # corr at shift k includes h[i-k] * (1-h[i]) if 0<=i-k<n
    # When h[i] goes 1->0, (1-h[i]) goes 0->1, so we ADD h[i-k]
    k_min2 = i - (n-1)  # so that i-k >= 0 -> k <= i
    k_max2 = i  # so that i-k >= 0
    # Wait: 0 <= i-k < n means i-n < k <= i, i.e., k in [i-n+1, i]
    # But k also ranges from -(n-1) to n-1
    k_lo = max(i - (n-1), -(n-1))  # = i-n+1 if i-n+1 > -(n-1), else -(n-1)
    k_hi = min(i, n-1)
    if k_hi >= k_lo:
        full_idx_start = k_lo + (n-1)
        full_idx_end = k_hi + (n-1)
        # h[i-k] for k in [k_lo, k_hi]
        # i-k ranges from i-k_lo to i-k_hi (decreasing)
        h_slice2 = h[i-k_hi : i-k_lo+1][::-1]
        delta[full_idx_start:full_idx_end+1] += h_slice2
    
    # Similarly for j going 0->1:
    # corr GAINS h[j]*(1-h[j+k]) at shift k where 0<=j+k<n  (same as above but positive)
    k_min = -j
    k_max = n - 1 - j
    if k_max >= k_min:
        full_idx_start = k_min + (n-1)
        full_idx_end = k_max + (n-1)
        h_slice = h[j+k_min : j+k_max+1]
        delta[full_idx_start:full_idx_end+1] += (1.0 - h_slice)
    
    # corr LOSES h[j-k]*(1-h[j]) at shift k where 0<=j-k<n
    # (1-h[j]) goes 1->0, so we subtract h[j-k]
    k_lo = max(j - (n-1), -(n-1))
    k_hi = min(j, n-1)
    if k_hi >= k_lo:
        full_idx_start = k_lo + (n-1)
        full_idx_end = k_hi + (n-1)
        h_slice2 = h[j-k_hi : j-k_lo+1][::-1]
        delta[full_idx_start:full_idx_end+1] -= h_slice2
    
    return delta
```

```python
def compute_swap_delta(h, i, j, n):
    """Compute delta vector for swapping h[i]:1->0 and h[j]:0->1."""
    delta = np.zeros(2*n - 1)
    half = n - 1
    
    for m in range(2*n - 1):
        k = m - half
        # Relevant n values
        ns = set()
        if 0 <= i < n:
            ns.add(i)
        if 0 <= j < n:
            ns.add(j)
        ni = i - k
        if 0 <= ni < n:
            ns.add(ni)
        nj = j - k
        if 0 <= nj < n:
            ns.add(nj)
        
        d = 0.0
        for nn in ns:
            # old term: h[nn+k]*(1-h[nn])
            nk = nn + k
            if 0 <= nk < n:
                old_t = h[nk] * (1.0 - h[nn])
                # new term
                h_i_new = 0.0 if nk == i else (1.0 if nk == j else h[nk])
                h_n_new = 0.0 if nn == i else (1.0 if nn == j else h[nn])
                new_t = h_i_new * (1.0 - h_n_new)
                d += new_t - old_t
        delta[m] = d
    
    return delta
```

```python
    half = n-1

    # Contribution from n=i (if valid)

    # new_h[i+k] = 0 if i+k==i, 1 if i+k==j, else h[i+k]

    if valid_i.any():

        idx = np.clip(i+k, 0, n-1)

        vals[valid_i & (k==0)] = 0.0

        vals[valid_i & (k==j-i)] = 1.0
```

```python
def project_binary(h, target_sum):
```

```python
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]

            order = np.argsort(h[ones])[:need]

    else:

        d += (1.0 - h[j + k])

            corr = corr_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = hl.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_run(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def continuous_sa(h, target, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    hl = h.astype(float).copy()
    corr = np.correlate(hl, 1.0 - hl, mode='full')
    best_c5 = float(corr.max()) * (2.0 / n)
    best_h = hl.copy()
    cur_c5 = best_c5
    stall = 0
    max_stall = 30000
    dl = 2.0 / n

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        step = (rng.random() * 2.0 - 1.0) * 0.15
        hl_new = hl.copy()
        hl_new[i] += step
        hl_new[j] -= step
        hl_new = np.clip(hl_new, 0.0, 1.0)
        hl_new = project_feasible(hl_new, target)

        corr_new = np.correlate(hl_new, 1.0 - hl_new, mode='full')
        c5_new = float(corr_new.max()) * dl
        d = c5_new - cur_c5

        if d < 0 or (t > 1e-15 and rng.random() < np.exp(-d / max(t, 1e-30))):
            hl = hl_new
            corr = corr_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = hl.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def legendre_init(n, rng):
    p = 1
    while p < n:
        p += 2
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime and p >= n:
            break
    if p < n:
        p = n + 1 if (n + 1) % 2 == 1 else n + 2
    while True:
        is_prime = True
        for d in range(3, int(np.sqrt(p)) + 1, 2):
            if p % d == 0:
                is_prime = False
                break
        if is_prime:
            break
        p += 2
    leg = np.ones(p)
    for a in range(1, p):
        if pow(a, (p - 1) // 2, p) == p - 1:
            leg[a] = -1.0
    seq = (leg >= 0).astype(float)
    if p == n:
        h = seq.copy()
    elif p > n:
        h = seq[:n].copy()
    else:
        h = np.zeros(n)
        h[:p] = seq
    target = n // 2
    h = project_binary(h, target)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    def _time_left():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: Targeted local search
    tl = min(50, _time_left() * 0.1)
    h1, s1 = targeted_ls(best_h, 80000, tl, start_time, scan_width=14)
    if s1 < best_score:
        best_score = s1
        best_h = h1

    # Phase 2: Binary SA (warm)
    tl = min(120, _time_left() * 0.18)
    mi = int(min(200000, tl * 2500))
    h2, s2 = fast_binary_sa(best_h, target_init, mi, 0.015, 0.00005, rng, tl, start_time)
    if s2 < best_score:
        best_score = s2
        best_h = h2

    # Phase 3: SLSQP continuous optimization
    if _time_left() > 100:
        tl = min(250, _time_left() * 0.35)
        alphas = [10, 30, 80, 200, 500, 1200, 3000, 8000, 20000, 50000, 120000]
        h3, s3 = slsqp_run(best_h, n_init, float(target_init), alphas, 8000, 1e-16, tl, start_time)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Binary SA after SLSQP (cool)
    if _time_left() > 80:
        tl = min(100, _time_left() * 0.18)
        h_bin = project_binary(best_h, target_init)
        mi = int(min(150000, tl * 2500))
        h4, s4 = fast_binary_sa(h_bin, target_init, mi, 0.008, 0.00002, rng, tl, start_time)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Continuous SA
    if _time_left() > 80:
        tl = min(100, _time_left() * 0.15)
        h_cs = best_h.astype(float).copy()
        mi = int(min(80000, tl * 800))
        h5, s5 = continuous_sa(h_cs, float(target_init), mi, 0.01, 0.0001, rng, tl, start_time)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Multi-n exploration
    n_options = [n_init + 2, n_init + 4, n_init + 8, n_init + 12, n_init + 20, n_init + 40]
    n_options = [x for x in n_options if x >= 100 and x % 2 == 0 and x != n_init]
    for n_try in n_options:
        if _time_left() < 80:
            break
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        tl = min(20, _time_left() * 0.06)
        h_g, s_g = targeted_ls(h_bin_try, 30000, tl, start_time, scan_width=10)
        if s_g < best_score:
            best_score = s_g
            best_h = h_g

        if _time_left() < 60:
            continue
        tl2 = min(30, _time_left() * 0.08)
        mi2 = int(min(60000, tl2 * 2500))
        h_sa, s_sa = fast_binary_sa(h_g, target_try, mi2, 0.01, 0.00005, rng, tl2, start_time)
        if s_sa < best_score:
            best_score = s_sa
            best_h = h_sa

        if _time_left() > 50:
            tl3 = min(40, _time_left() * 0.08)
            alphas_sm = [50, 200, 800, 3000, 10000, 40000]
            h_ref, s_ref = slsqp_run(h_sa, n_try, float(target_try), alphas_sm, 4000, 1e-16, tl3, start_time)
            if s_ref < best_score:
                best_score = s_ref
                best_h = h_ref

    # Phase 7: Legendre initialization + optimization
    if _time_left() > 100:
        h_leg = legendre_init(n_init, rng)
        tl = min(20, _time_left() * 0.05)
        h_g2, s_g2 = targeted_ls(h_leg, 20000, tl, start_time, scan_width=10)
        if s_g2 < best_score:
            best_score = s_g2
            best_h = h_g2
        if _time_left() > 60:
            tl2 = min(40, _time_left() * 0.08)
            mi3 = int(min(50000, tl2 * 2000))
            h_s3, s_s3 = fast_binary_sa(h_g2, target_init, mi3, 0.01, 0.00005, rng, tl2, start_time)
            if s_s3 < best_score:
                best_score = s_s3
                best_h = h_s3

    # Phase 8: ILS perturbations on best solution
    if _time_left() > 60:
        for pert in range(8):
            if _time_left() < 30:
                break
            h_p = best_h.copy()
            n_pert = 2 + pert
            ones = np.where(h_p == 1)[0]
            zeros = np.where(h_p == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h_p[ii] = 0
                h_p[jj] = 1
                ones = np.where(h_p == 1)[0]
                zeros = np.where(h_p == 0)[0]

            tl = min(15, _time_left() * 0.05)
            h_pg, s_pg = targeted_ls(h_p, 15000, tl, start_time, scan_width=8)
            if s_pg < best_score:
                best_score = s_pg
                best_h = h_pg

            if _time_left() < 25:
                break
            tl2 = min(20, _time_left() * 0.04)
            mi4 = int(min(30000, tl2 * 2000))
            h_ps, s_ps = fast_binary_sa(h_pg, int(len(h_pg) // 2), mi4,
                                        0.005, 0.00002, rng, tl2, start_time)
            if s_ps < best_score:
                best_score = s_ps
                best_h = h_ps

    # Phase 9: Final SLSQP polish (multiple restarts)
    if _time_left() > 30:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_final = [2000, 8000, 30000, 100000]
        tl = min(60, _time_left() - 10)
        h_ref, s_ref = slsqp_run(best_h, n_use, target_use, alphas_final, 5000, 1e-16, tl, start_time)
        if s_ref < best_score:
            best_score = s_ref
            best_h = h_ref

    # Phase 10: Final binary SA on polished solution
    if _time_left() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bf = project_binary(best_h, target_use)
        tl = max(5.0, min(30, _time_left() - 10))
        mi5 = int(min(80000, tl * 3000))
        h_sf, s_sf = fast_binary_sa(h_bf, target_use, mi5, 0.003, 0.00001, rng, tl, start_time)
        if s_sf < best_score:
            best_score = s_sf
            best_h = h_sf

    # Phase 11: Final SLSQP
    if _time_left() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_fin = [10000, 50000, 200000]
        tl = max(3.0, min(20, _time_left() - 5))
        h_rf, s_rf = slsqp_run(best_h, n_use, target_use, alphas_fin, 3000, 1e-17, tl, start_time)
        if s_rf < best_score:
            best_score = s_rf
            best_h = h_rf

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
```
