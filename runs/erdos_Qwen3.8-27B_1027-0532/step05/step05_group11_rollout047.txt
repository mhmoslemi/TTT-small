    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-11, verbose=False)
    except:
        return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def smooth_max_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    if alpha < 1:
        return float(c_max)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights /= wsum
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - (n - 1)
        lo = max(0, k)
        hi = min(n, n + k)
        if hi <= lo:
            continue
        j1 = np.arange(lo, hi)
        grad[j1] += w * (1.0 - h[j1 - k]) * dx
        j2_lo = max(0, lo - k)
        j2_hi = min(n, hi - k)
        if j2_hi > j2_lo:
            j2 = np.arange(j2_lo, j2_hi)
            grad[j2] -= w * h[j2 + k] * dx
    return grad

def slsqp_polish(h, n, target, alphas, maxiter_each=300, time_limit=100):
    start = time.time()
    h_cur = h.copy()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            def obj(x):
                return smooth_max_obj(x, alpha)
            def obj_grad(x):
                return smooth_max_grad(x, alpha)
            def cons_eq(x):
                return x.sum() - target
            def cons_eq_jac(x):
                return np.ones(n)
            res = minimize(
                obj, h_cur, args=(),
                method='SLSQP',
                jac=obj_grad,
                bounds=[(0.0, 1.0)] * n,
                constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}],
                options={'maxiter': maxiter_each, 'ftol': 1e-16}
            )
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except:
            pass
    return h_cur

def lbfgs_with_projection(h, n, target, alpha, maxiter=200, time_limit=60):
    start = time.time()
    h_cur = h.copy()
    try:
        res = minimize(
            smooth_max_obj, h_cur, args=(alpha,),
            method='L-BFGS-B',
            jac=lambda x, a: smooth_max_grad(x, a),
            bounds=[(0.0, 1.0)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        h_new = project_feasible(res.x, target)
        if compute_c5(h_new) <= compute_c5(h_cur):
            return h_new
    except:
        pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def alternating_lp(h_init, n, target, time_limit, use_full=True, n_top=200,
                   damping=1.0, min_improve=1e-14):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    it_count = 0
    while time.time() - start < time_limit:
        it_count += 1
        if use_full:
            h_lp = solve_full_lp(h, n, target)
        else:
            h_lp = solve_top_lp(h, n, target, n_top)
        if h_lp is None:
            break
        h_try = project_feasible(damping * h_lp + (1.0 - damping) * h, target)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - min_improve:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
        else:
            h = h_try
    return best_h, best_c5

def periodic_init(n, period, on_fraction, phase=0):
    h = np.zeros(n)
    on_count = int(period * on_fraction)
    pattern = np.zeros(period)
    pattern[:on_count] = 1.0
    h[:n % period] = np.tile(pattern, (n // period + 1))[:n % period + on_count - (period - on_count)] if n % period > 0 else 0
    h = np.tile(pattern, (n // period + 1))[:n]
    target = n / 2.0
    total_ones = h.sum()
    if total_ones != target:
        h = project_feasible(h, target)
    return h
