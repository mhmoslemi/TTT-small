            best_score = sc_c
            best_h = h_c

    # Phase 4: SLSQP polishing
    if tleft() > 200:
        for alpha in [5000, 30000, 150000]:
            if tleft() < 150:
                break
            best_h = lse_slsqp(best_h, n_final, target_final, alpha, 2500, 1e-16,
                              min(start_time + 850, hard_deadline))

    # Phase 5: Fine cycling with more lags
    if tleft() > 150:
        h_c, sc_c = cycling_optimize(best_h, n_final, target_final,
                                     max_steps=8000, lr_init=0.001,
                                     lr_decay=0.997, deadline=start_time + 1000,
                                     n_worst=5)
        if sc_c < best_score:
            best_score = sc_c
            best_h = h_c

    # Phase 6: Spectral smoothing + re-optimize
    if tleft() > 120:
        h_smooth = spectral_smooth(best_h, n_final, 8.0)
        h_c, sc_c = cycling_optimize(h_smooth, n_final, target_final,
                                     max_steps=3000, lr_init=0.002,
                                     lr_decay=0.996, deadline=start_time + 1100,
                                     n_worst=3)
        if sc_c < best_score:
            best_score = sc_c
            best_h = h_c

    # Phase 7: High-alpha SLSQP
    if tleft() > 100:
        for alpha in [200000, 1000000]:
            if tleft() < 60:
                break
            best_h = lse_slsqp(best_h, n_final, target_final, alpha, 2000, 1e-17,
                              min(start_time + 1250, hard_deadline))

    # Phase 8: Final fine cycling
    if tleft() > 50:
        h_c, sc_c = cycling_optimize(best_h, n_final, target_final,
                                     max_steps=5000, lr_init=0.0004,
                                     lr_decay=0.999, deadline=hard_deadline - 10,
                                     n_worst=2)
        if sc_c < best_score:
            best_score = sc_c
            best_h = h_c

    # Phase 9: Perturbation escape
    if tleft() > 30:
        for pert_scale in [0.03, 0.05]:
            if tleft() < 20:
                break
            h_pert = best_h + rng.randn(n_final) * pert_scale
            h_pert = project_feasible(h_pert, target_final)
            h_c, sc_c = cycling_optimize(h_pert, n_final, target_final,
                                         max_steps=2000, lr_init=0.003,
                                         lr_decay=0.995, deadline=hard_deadline - 5,
                                         n_worst=3)
            if sc_c < best_score:
                best_score = sc_c
                best_h = h_c

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
