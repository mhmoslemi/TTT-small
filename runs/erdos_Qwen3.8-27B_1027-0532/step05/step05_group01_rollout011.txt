def exact_subgradient_step(h, n, target, lr, n_active=None):
    """Take a subgradient step on the exact max objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    mid = n - 1
    idxs = np.arange(n)
    
    # Find active constraints (within tolerance)
    tol = 1e-10 / dx
    active = np.where(corr >= c_max - tol)[0]
    
    if n_active is not None and len(active) > n_active:
        # Keep only top n_active
        sorted_active = active[np.argsort(corr[active])[::-1][:n_active]]
        active = sorted_active
    
    grad = np.zeros(n)
    for j in active:
        k = int(j) - mid
        v1 = (idxs - k >= 0) & (idxs - k < n)
        v2 = (idxs + k >= 0) & (idxs + k < n)
        grad[v1] += (1.0 - h[idxs[v1] - k])
        grad[v2] -= h[idxs[v2] + k]
    
    grad /= len(active)
    grad *= dx
    grad -= np.mean(grad)  # Remove mean to maintain feasibility
    
    h_new = project_feasible(h - lr * grad, target)
    return h_new, compute_c5(h_new)

def smooth_obj(h, alpha):

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n

                          method='SLSQP', bounds=bounds, constraints=cons,

def subgradient_method(h, n, target, time_limit, lr_start=0.05, lr_end=0.001, decay=0.999):

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    lr = lr_start
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        tol = 1e-8
        active = np.where(corr >= c_max - tol)[0]
        
        grad = np.zeros(n)
        for j in active:
            k = int(j) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]
        
        grad /= len(active)
        grad *= dx
        grad -= np.mean(grad)
        
        h_try = project_feasible(h - lr * grad, target)
        c5_try = compute_c5(h_try)
        
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
            h = h_try
        else:
            # Backtrack
            for factor in [0.5, 0.25, 0.1]:

                    h = h_try2
                    if c5_try2 < best_c5 - 1e-14:
                        best_c5 = c5_try2
                        best_h = h.copy()
                    break
        
        lr *= decay
    
    return best_h, best_c5


def binary_push(h, n, target):
    """Push h toward binary and return the best binary relaxation."""
    # Threshold at 0.5
    h_bin = (h >= 0.5).astype(float)
    # Adjust to meet constraint
    n_ones = int(np.sum(h_bin))
    n_target = int(target)
    if n_ones > n_target:
        # Need to flip some 1s to 0s - flip the ones with lowest h values
        ones_idx = np.where(h_bin == 1.0)[0]
        ones_sorted = ones_idx[np.argsort(h[ones_idx])]
        for idx in ones_sorted[:n_ones - n_target]:
            h_bin[idx] = 0.0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        zeros_sorted = zeros_idx[np.argsort(h[zeros_idx])[::-1]]
        for idx in zeros_sorted[:n_target - n_ones]:
            h_bin[idx] = 1.0
    
    return project_feasible(h_bin, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

_g_n = 0
_g_target = 0.0

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _g_n, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def compute_influence(h, m, n):
    n_corr = 2 * n - 1
    mid = n - 1
    inf = np.zeros(n_corr)
    j1s = max(0, n - 1 - m)
    j1e = min(2 * n - 1, 2 * n - 1 - m)
    if j1s < j1e:
        idxs1 = np.arange(j1s, j1e)
        inf[idxs1] += (1.0 - h[m + idxs1 - mid])
    j2s = max(0, m)
    j2e = min(2 * n - 1, n - 1 + m)
    if j2s < j2e:
        idxs2 = np.arange(j2s, j2e)
        inf[idxs2] -= h[m - idxs2 + mid]
    return inf

def pair_cd(h, n, target, time_limit, max_sweeps=10000, n_cand=60, n_bs=15):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    idxs = np.arange(n)
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 400:
            break

        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)

        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]

        candidates = list(set(list(low_set) + list(high_set)))
        influence = {}
        for m in candidates:
            influence[m] = compute_influence(h, m, n)

        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0

        for i in high_set:
            inf_i = influence[i]
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue

                g_move = influence[j] - inf_i
                active_j = int(np.argmax(corr))
                if g_move[active_j] >= -1e-14:
                    continue

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(n_bs):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g_move
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g_move[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g_move) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5

def subgradient_method(h, n, target, time_limit, lr_start=0.04, decay=0.9995):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    idxs = np.arange(n)
    mid = n - 1
    lr = lr_start
    stale = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        tol = c_max * 1e-6 + 1e-10
        active = np.where(corr >= c_max - tol)[0]

        grad = np.zeros(n)
        for j in active:
            k = int(j) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        grad /= len(active)
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h - lr * mult * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                stale = 0
                break
            elif c5_try < compute_c5(h) - 1e-15:
                h = h_try
                improved = True
                break

        if not improved:
            stale += 1
            if stale > 50:
                break

        lr *= decay

    return best_h, best_c5

def binary_push(h, n, target):
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = int(target)
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1.0)[0]
        ones_sorted = ones_idx[np.argsort(h[ones_idx])]
        for idx in ones_sorted[:n_ones - n_target]:
            h_bin[idx] = 0.0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0.0)[0]
        zeros_sorted = zeros_idx[np.argsort(h[zeros_idx])[::-1]]
        for idx in zeros_sorted[:n_target - n_ones]:
            h_bin[idx] = 1.0
    return project_feasible(h_bin, target)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def momentum_subgradient(h, n, target, time_limit, max_steps, mu=0.8, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1
    stale = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        tol = c_max * 1e-5 + 1e-10
        active = np.where(corr >= c_max - tol)[0]

        grad = np.zeros(n)
        for j in active:
            k = int(j) - mid
            v1 = (idxs - k >= 0) & (idxs - k < n)
            v2 = (idxs + k >= 0) & (idxs + k < n)
            grad[v1] += (1.0 - h[idxs[v1] - k])
            grad[v2] -= h[idxs[v2] + k]

        grad /= len(active)
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stale = 0
        else:
            stale += 1
            if stale > 200:
                break

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [200, 1000, 5000], 800, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Aggressive pair CD
    if tleft() > 250:
        cd_time = min(250, tleft() * 0.35)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                              max_sweeps=10000, n_cand=65, n_bs=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP high alpha
    if tleft() > 100:
        h_pol = slsqp_round(best_h, n, target, [5000, 20000, 80000], 1200, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Subgradient
    if tleft() > 80:
        h_sg, c5_sg = subgradient_method(best_h, n, target,
                                         time_limit=min(50, tleft() * 0.3),
                                         lr_start=0.04, decay=0.9995)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 5: Binary push + SLSQP
    if tleft() > 100:
        h_bin = binary_push(best_h, n, target)
        sc_bin = compute_c5(h_bin)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
        h_from_bin = slsqp_round(h_bin, n, target, [100, 500, 2000, 10000], 800, 1e-15)
        sc = compute_c5(h_from_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_from_bin

    # Phase 6: Pair CD round 2
    if tleft() > 150:
        cd_time = min(150, tleft() * 0.35)
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=cd_time,
                              max_sweeps=8000, n_cand=75, n_bs=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 7: Alternating SLSQP + subgradient
    for rnd in range(3):
        if tleft() < 60:
            break
        h_pol = slsqp_round(best_h, n, target, [10000, 50000, 150000], 1000, 1e-16)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
        if tleft() < 40:
            break
        h_sg, c5_sg = subgradient_method(best_h, n, target,
                                         time_limit=min(30, tleft() * 0.25),
                                         lr_start=0.02, decay=0.9997)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Momentum subgradient
    if tleft() > 60:
        h_mom, c5_mom = momentum_subgradient(best_h, n, target,
                                             time_limit=min(40, tleft() * 0.4),
                                             max_steps=20000, mu=0.8, base_lr=0.012)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 9: Final pair CD
    if tleft() > 60:
        h_cd, c5_cd = pair_cd(best_h, n, target, time_limit=min(40, tleft() * 0.4),
                              max_sweeps=4000, n_cand=50, n_bs=15)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 10: Try higher resolution
    if tleft() > 120:
        for n_try in [500, 600]:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires = slsqp_round(h_interp, n_try, target_try,
                                  [100, 500, 2000, 10000], 800, 1e-15)
            sc = compute_c5(h_hires)
            if sc < best_score:
                best_score = sc
                best_h = h_hires
                n = n_try
                target = target_try

    # Phase 11: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [50000, 200000, 500000], 600, 1e-16)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
