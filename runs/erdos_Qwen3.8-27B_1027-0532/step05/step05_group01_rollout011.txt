                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]

                a_d = delta_lo
                b_d = delta_hi
                for _ in range(20):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d

                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue

                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5 - 1e-15:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star

        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def active_set_lp(h_init, n, target, n_active=5, time_limit=60):
    start = time.time()
    dx = 2.0 / n
    mid = n - 1
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for iteration in range(50):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_ids = np.argsort(corr)[-n_active:]

        f_lin = np.zeros(n)
        for s_idx in top_ids:
            k = int(s_idx) - mid
            for j in range(n):
                if j + k >= 0 and j + k < n:
                    f_lin[j] += (1.0 - h[j + k])
                if j - k >= 0 and j - k < n:
                    f_lin[j] -= h[j - k]
        f_lin -= np.mean(f_lin)

        var = cp.Variable(n)
        constraints = [var >= 0, var <= 1, cp.sum(var) == target]

        obj_terms = []
        for s_idx in top_ids:
            k = int(s_idx) - mid
            f_val = float(corr[s_idx])
            f_grad = np.zeros(n)
            for j in range(n):
                if j + k >= 0 and j + k < n:
                    f_grad[j] += (1.0 - h[j + k])
                if j - k >= 0 and j - k < n:
                    f_grad[j] -= h[j - k]
            f_grad -= np.mean(f_grad)
            obj_terms.append(f_val + cp.multiply(f_grad, var - h))

        t_var = cp.Variable()
        constraints.append(t_var >= cp.max(obj_terms))

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=5000, abs_acc=1e-10, rel_acc=1e-10)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=10000)
            except Exception:
                break

        if var.value is None:
            break

        h_try = project_feasible(var.value, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try
        else:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def rounding_search(h_init, n, target, max_tries=200, time_limit=30):
    start = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(12345)

    sorted_vals = np.argsort(h)
    n_ones = int(target)

    for t in range(max_tries):
        if time.time() - start > time_limit:
            break
        h_round = np.zeros(n)
        cutoff = target - sorted_vals[n_ones - 1]
        h_round[sorted_vals[n_ones - 1:]] = 1.0
        h_round[sorted_vals[:n_ones]] = 0.0
        h_round = project_feasible(h_round, target)
        sc = compute_c5(h_round)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = h_round.copy()

        h_mix = h.copy()
        n_pert = int(rng.choice([5, 10, 20]))
        perts = rng.choice(n, size=n_pert, replace=False)
        for p in perts:
            h_mix[p] = 1.0 if rng.rand() > 0.5 else 0.0
        h_mix = project_feasible(h_mix, target)
        sc = compute_c5(h_mix)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = h_mix.copy()

    return best_h, best_c5


def momentum_subgradient(h, n, target, time_limit, max_steps, mu=0.8, base_lr=0.015, n_top=5):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-n_top:]
        grad = multi_shift_grad(h, n, corr, top_k)
        grad *= dx

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with adaptive alpha
    if tleft() > 200:
        alphas = [50, 200, 1000, 5000, 20000, 80000]
        h_pol, sc = slsqp_round(best_h, n, target, alphas, 2000, 1e-16,
                                time_limit=min(180, tleft() * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Enhanced pair CD with multi-shift
    if tleft() > 200:
        cd_time = min(300, tleft() * 0.35)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=8000, n_cand=80, n_active=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: Active-set LP refinement
    if tleft() > 150:
        lp_time = min(120, tleft() * 0.3)
        h_lp, c5_lp = active_set_lp(best_h, n, target, n_active=5, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP after LP
    if tleft() > 100:
        alphas = [500, 2000, 10000, 50000]
        h_pol, sc = slsqp_round(best_h, n, target, alphas, 1500, 1e-16,
                                time_limit=min(80, tleft() * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 5: Another round of enhanced CD
    if tleft() > 150:
        cd_time = min(200, tleft() * 0.35)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n, target,
                                       time_limit=cd_time,
                                       max_sweeps=6000, n_cand=100, n_active=4)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Active-set LP again
    if tleft() > 100:
        lp_time = min(80, tleft() * 0.3)
        h_lp, c5_lp = active_set_lp(best_h, n, target, n_active=7, time_limit=lp_time)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Momentum subgradient
    if tleft() > 80:
        h_mom, c5_mom = momentum_subgradient(best_h, n, target,
                                             time_limit=min(50, tleft() * 0.3),
                                             max_steps=15000, mu=0.8, base_lr=0.012, n_top=5)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 8: Multi-restart with structured perturbations
    rng = np.random.RandomState(seed)
    for noise_scale in [0.03, 0.08, 0.12]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_round(h_pert, n, target,
                                [100, 500, 2000, 10000, 40000],
                                1000, 1e-16,
                                time_limit=min(50, tleft() * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

        h_cd2, c5_cd2 = enhanced_pair_cd(h_ref, n, target,
                                         time_limit=min(30, tleft() * 0.3),
                                         max_sweeps=3000, n_cand=60, n_active=3)
        if c5_cd2 < best_score:
            best_score = c5_cd2
            best_h = h_cd2

    # Phase 9: Rounding search
    if tleft() > 60:
        h_round, c5_round = rounding_search(best_h, n, target,
                                            max_tries=300, time_limit=min(30, tleft() * 0.3))
        if c5_round < best_score:
            best_score = c5_round
            best_h = h_round
            if tleft() > 40:
                h_pol, sc = slsqp_round(best_h, n, target,
                                        [500, 2000, 10000],
                                        800, 1e-16,
                                        time_limit=min(30, tleft() * 0.4), start_time=start_time)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol

    # Phase 10: Try higher resolution
    if tleft() > 200:
        for n_try in [500, 600, 800]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_hires, sc = slsqp_round(h_interp, n_try, target_try,
                                      [100, 500, 2000, 8000, 30000],
                                      1000, 1e-16,
                                      time_limit=min(80, tleft() * 0.4), start_time=start_time)
            if sc < best_score:
                h_cd3, c5_cd3 = enhanced_pair_cd(h_hires, n_try, target_try,
                                                 time_limit=min(40, tleft() * 0.3),
                                                 max_sweeps=3000, n_cand=60, n_active=3)
                if c5_cd3 < sc:
                    sc = c5_cd3
                    h_hires = h_cd3
                if sc < best_score:
                    best_score = sc
                    best_h = project_feasible(h_hires, target_try)

    # Phase 11: Final pair CD
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(40, tleft() * 0.5)
        h_cd, c5_cd = enhanced_pair_cd(best_h, n_use, target_use,
                                       time_limit=cd_time,
                                       max_sweeps=4000, n_cand=80, n_active=3)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 12: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final, sc = slsqp_round(best_h, n_use, target_use,
                                  [2000, 10000, 50000, 200000],
                                  1000, 1e-16,
                                  time_limit=min(25, tleft() * 0.7), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
