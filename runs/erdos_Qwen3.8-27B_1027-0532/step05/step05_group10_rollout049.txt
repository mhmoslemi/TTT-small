        # = [p[N-1], p[N-2], ..., p[0]]
        # The first row is [f(0), f(1), f(2), ..., f(N-1)]

            m_hi = min(n_corr, N_minus_1 + N - i)  # N-1+N-i
            b_vec[i] = dx * np.sum(p[m_lo:m_hi])

        # minimize: h^T A_mat h - b_vec^T h
        # subject to: sum(h) = target, 0 <= h <= 1
        
        h_var = cp.Variable(n)
        obj = cp.quad_form(h_var, A_mat) - b_vec @ h_var
        constraints = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        prob = cp.Problem(cp.Minimize(obj), constraints)
        prob.solve(solver=cp.SCS, eps=1e-10, max_iters=20000)
        
        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_feasible(h_new, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        
        # Update p
        corr = np.correlate(h, 1-h, mode='full') * dx
        k_star = np.argmax(corr)
        p = (1 - eta_p) * p + eta_p * np.eye(n_corr)[k_star]
    
    return best_h, best_c5

Actually, here's another idea: **Iterative Constraint Tightening (ICT)**:
1. Start with t = current C_5
2. Solve: min_h (1/2)||h - h_prev||² s.t. O_k(h) ≤ t for all k, 0≤h≤1, sum(h)=target

Hmm, O_k(h) is quadratic in h, so the constraint O_k(h) ≤ t is a quadratic constraint (not linear). This makes it a QCQP, which is non-convex.

But if we linearize: O_k(h + Δ) ≈ O_k(h) + ∇O_k(h)^T Δ
Then the constraint becomes linear in Δ, and we have a QP in Δ.

This is the **sequential quadratic programming (SQP)** approach!

Algorithm:
1. Start with h_0
2. At each iteration:
   a. Compute O_k(h) for all k, find k* = argmax
   b. Linearize O_k(h + Δ) ≈ O_k(h) + ∇O_k(h)^T Δ for all k
   c. Solve: min ||Δ||² s.t. O_k(h) + ∇O_k(h)^T Δ ≤ t for all k, 
                      0 ≤ h + Δ ≤ 1, sum(h+Δ) = target
   d. Update h ← h + Δ (or h + αΔ with line search)
   e. Gradually decrease t

This is essentially SQP for the maximin problem. The QP at each step is convex (minimizing a convex quadratic subject to linear constraints).

- Each subproblem is a convex QP (solvable exactly)
- We handle all constraints simultaneously (not just a subset)
- The linearization is natural and well-justified

def sqp_maximin(h_init, n, target, max_iter, time_limit, t_init=None, t_decrease=0.999):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    if t_init is None:
        t_init = compute_c5(h) + 0.01
    
    t = t_init
    n_corr = 2 * n - 1
    half_n = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute current overlaps and gradients
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5_cur = np.max(corr)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Compute gradients of O_k w.r.t. h for all k
        # O_k = dx * sum_i h[i](1-h[i+k]) for valid i
        # dO_k/dh[j] = dx * [(1-h[j+k]) if 0<=j+k<n + (-h[j-k]) if 0<=j-k<n]
        
        # For efficiency, compute all gradients at once
        # grad[k, j] = dx * [(1-h[j+k]) * 1[0<=j+k<n] + (-h[j-k]) * 1[0<=j-k<n]]
        
        # Actually, let's think about this differently.
        # O_k = dx * (sum_{i: valid} h[i] - sum_{i: valid} h[i]h[i+k])
        # dO_k/dh[j] = dx * (1[valid for first sum] - (h[j+k]*1[j+k valid] + h[j-k]*1[j-k valid]))
        
        # For the first sum: 1[0<=j<n and 0<=j+k<n] = 1[max(0,-k) <= j < min(n, n-k)]
        # For the second: h[j+k] is valid if 0<=j+k<n, i.e., -k <= j < n-k
        #                  h[j-k] is valid if 0<=j-k<n, i.e., k <= j < n+k
        
        # Let me just compute the gradient for the active constraints (top few k values)
        
        # Get the worst lag
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - half_n
        
        # For the QP, we'll use all constraints but that might be too many.
        # Let's use the top 50 lags.
        top_k_indices = np.argsort(corr)[-50:]
        
        # Build the QP:
        # min ||delta||^2
        # s.t. O_k(h) + grad_k^T delta <= t for k in top_k
        #      0 <= h + delta <= 1
        #      sum(h + delta) = target
        
        delta = cp.Variable(n)
        obj = cp.sum_squares(delta)
        constraints = [cp.sum(h + delta) == target,
                       h + delta >= 0,
                       h + delta <= 1]
        
        for k_idx in top_k_indices:
            k = k_idx - half_n
            # Compute gradient of O_k at h
            grad_k = np.zeros(n)
            lo = max(0, k)
            hi = min(n, n + k)
            # First term: sum_{i=lo}^{hi-1} h[i], grad is 1 for i in [lo, hi)
            grad_k[lo:hi] += 1.0
            # Second term: sum_{i=lo}^{hi-1} h[i]h[i-k]
            # d/dh[j]: h[j-k] for j in [lo+k, hi+k) [from h[i] term where i=j, need i-k=j-k valid]
            #          + h[j] for j in [lo, hi) [from h[i-k] term where i-k=j, so i=j+k]
            # Wait, let me be more careful.
            # term2 = sum_{i=lo}^{hi-1} h[i] * h[i-k]
            # d/dh[j] = h[j-k] * 1[j in [lo,hi) and 0<=j-k<n] + h[j] * 1[j+k in [lo,hi) and 0<=j<n]
            # = h[j-k] * 1[lo<=j<hi, 0<=j-k<n] + h[j] * 1[lo<=j+k<hi, 0<=j<n]

            # lo <= j < hi means max(0,k) <= j < min(n, n+k)
            # 0 <= j-k < n means k <= j < n+k
            # Intersection: max(0,k,k) <= j < min(n, n+k, n+k) = max(0,k) <= j < min(n,n+k)
            # So the first part: h[j-k] for j in [max(0,k), min(n,n+k))
            # But we also need j-k >= 0, i.e., j >= k. Since j >= max(0,k) >= k (when k>0)
            # and j >= 0 (when k<=0), this is automatically satisfied.

            #            Actually: min(n, n+k) - k = min(n-k, n) 
            # If k < 0: max(0,k)+k = k < 0 (since k<0 and max(0,k)=0), so -k+0 <= j... 
            # Let me just do it numerically.

            j_p1_lo = max(lo, k)  # j >= k (for j-k >= 0)
            j_p1_hi = min(hi, n + k)  # j < n+k (for j-k < n)
            j_p1_lo = max(j_p1_lo, 0)
            j_p1_hi = min(j_p1_hi, n)
            if j_p1_hi > j_p1_lo:
                grad_k[j_p1_lo:j_p1_hi] -= h[j_p1_lo - k:j_p1_hi - k]
            
            # Part 2: h[j] for j where lo <= j+k < hi and 0 <= j < n
            # j+k >= lo => j >= lo - k
            # j+k < hi => j < hi - k
            j_p2_lo = max(0, lo - k)
            j_p2_hi = min(n, hi - k)
            if j_p2_hi > j_p2_lo:
                grad_k[j_p2_lo:j_p2_hi] -= h[j_p2_lo:j_p2_hi]
            
            grad_k *= dx
            
            constraints.append(corr[k_idx] + grad_k @ delta <= t)
        
        # Solve QP
        prob = cp.Problem(cp.Minimize(obj), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000)
        except:
            try:
                prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000)
            except:
                break
        
        if delta.value is None:
            break
        
        h_new = project_feasible(h + delta.value, target)
        
        # Line search
        best_alpha = 0
        best_obj = compute_c5(h_new)
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            h_try = project_feasible(h + alpha * delta.value, target)
            obj_try = compute_c5(h_try)
            if obj_try < best_obj:
                best_obj = obj_try
                best_alpha = alpha
        
        h = h + best_alpha * delta.value
        h = project_feasible(h, target)
        
        # Update t
        if it % 10 == 0:
            t = max(compute_c5(h) - 0.001, 0.3)  # decrease t gradually
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:

    else:
        idx = np.where(h > 1e-10)[0]

        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
