def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    # Fix sum exactly
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        idx = (result > 1e-15) & (result < 1.0 - 1e-15)
        cnt = idx.sum()
        if cnt > 0:
            result[idx] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def active_set_gradient(h, active_lags, weights=None):
    """Compute gradient of weighted sum of active correlations."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_lags:
        k = int(k)
        w = weights[k] if weights is not None else 1.0
        if w < 1e-15:
            continue
        jp = np.arange(0, n - k)
        grad[jp] += w * (1.0 - h[jp + k])
        jn = np.arange(k, n)
        grad[jn] -= w * h[jn - k]
    return grad * dx

    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    
    # Adaptive parameters
    alpha = 100.0  # LSE temperature (will decrease)
    step = 0.5  # Initial step size
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        corr = np.correlate(y, 1.0 - y, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        
        # Only use weights above threshold for efficiency
        active_mask = weights > 1e-8
        active_lags = np.where(active_mask)[0]
        
        if len(active_lags) == 0:
            active_lags = np.array([int(np.argmax(corr))])
            w_vec = np.zeros(n-1)
            w_vec[active_lags[0]] = 1.0
        else:
            w_vec = weights

        w_vec = np.where(active_mask, w_vec, 0.0)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    result = np.clip(h - tau, 0.0, 1.0)
    # Fix sum
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def fista_direct(h0, target, deadline, alpha=200.0, use_reject=True):
    """FISTA with direct C5 tracking and optional rejection."""
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_mom = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    step = 1.0 / (alpha * 0.01 + 1.0)
    max_step = 2.0 / (alpha * 0.005 + 1.0)
    
    for it in range(1000000):
        if time.time() > deadline:
            break
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) * 0.5
        y = x + ((t_mom - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute LSE gradient
        corr = np.correlate(y, 1.0 - y, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_vals = np.exp(shifted)
        sum_exp = exp_vals.sum()
        weights = exp_vals / sum_exp
        
        # Compute gradient
        grad = np.zeros(n)
        # Use only significant weights
        thresh = max(1e-6, 1.0 / n * 0.1)
        significant = weights > thresh
        for k in np.where(significant)[0]:
            w = weights[k]
            if w < 1e-15:
                continue
            n_k = int(k)
            grad[:n-n_k] += w * (1.0 - y[n_k:])
            grad[n_k:] -= w * y[:n-n_k]
        grad *= dx
        
        # Take step
        for _ in range(15):
            x_new = project_feasible(y - step * grad, target)
            # Check if this is good
            if use_reject:
                sc_new = compute_c5(x_new)
                if sc_new < best_score + 0.001:  # Allow slight increase
                    break
            else:
                # LSE check
                corr_new = np.correlate(x_new, 1.0 - x_new, mode='full')[n-1:]
                cm = float(np.max(corr_new))
                lse_new = (cm + np.log(max(exp_vals.sum(), 1e-300)) / alpha) * dx
                # Compare with LSE at y
                lse_y = (c_max + np.log(max(sum_exp, 1e-300)) / alpha) * dx
                if lse_new <= lse_y + 1e-10:
                    break
            step *= 0.5
        
        x_prev = x
        x = x_new
        t_mom = t_new
        
        sc = compute_c5(x)
        if sc < best_score:
            best_score = sc
            best_x = x.copy()
        
        step = min(step * 2.0, max_step)
    
    return best_x, best_score

def binary_refine(h, target, deadline):
    """Try to push values towards 0 or 1 while maintaining feasibility."""
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)
    
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                return h_cur, cur_score
            val = h_cur[idx]
            if val < 1e-8 or val > 1.0 - 1e-8:
                continue
            
            # Try 0
            h_test = h_cur.copy()
            h_test[idx] = 0.0
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-10:
                h_cur = h_test
                cur_score = sc
                improved = True
                continue
            
            # Try 1
            h_test = h_cur.copy()
            h_test[idx] = 1.0
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-10:
                h_cur = h_test
                cur_score = sc
                improved = True
        
        if not improved and sweep > 5:
            break
    
    return h_cur, cur_score

def block_refine(h, target, deadline):
    """Try flipping blocks of positions."""
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(444)
    
    for sweep in range(50):
        if time.time() > deadline:
            break
        improved = False
        # Try random blocks
        for _ in range(n):
            if time.time() > deadline:
                return h_cur, cur_score
            length = rng.randint(2, 15)
            start = rng.randint(0, max(1, n - length))
            h_test = h_cur.copy()
            h_test[start:start+length] = 1.0 - h_test[start:start+length]
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-10:
                h_cur = h_test
                cur_score = sc
                improved = True
        if not improved:
            break
    
    return h_cur, cur_score

def multi_start(h_best, target, deadline, n_starts=6):
    """Generate perturbations and optimize each."""
    n = len(h_best)
    best_h = h_best.copy()
    best_score = compute_c5(h_best)
    rng = np.random.RandomState(12345)
    
    time_per_start = max(20, (deadline - time.time()) / max(n_starts, 1) - 5)
    
    for s in range(n_starts):
        if time.time() > deadline - 10:
            break
        scale = 0.02 + 0.03 * s
        h_pert = h_best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        
        deadline_s = time.time() + time_per_start
        h_opt, sc = fista_direct(h_pert, target, deadline_s, alpha=300.0, use_reject=True)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()
    
    return best_h, best_score

  -h[j-k] if j >= k
  Both if k <= j <= n-1-k
  0 otherwise

So:
grad[j] = sum_{k: j+k<n} w[k]*(1-h[j+k]) - sum_{k: k<=j} w[k]*h[j-k]

First term: for fixed j, sum over k from 1 to n-1-j of w[k]*(1-h[j+k])
Second term: for fixed j, sum over k from 1 to j of w[k]*h[j-k]

Wait, let me re-examine. `np.correlate(h, 1-h, mode='full')` gives the full cross-correlation. With `mode='full'`, the result has length 2n-1. The element at index n-1+k corresponds to the correlation at lag k (positive lags).

So corr = np.correlate(h, 1-h, mode='full')[n-1:] gives correlations at lags 0, 1, ..., n-1.
corr[0] = sum_i h[i]*(1-h[i]) (lag 0)
corr[k] = sum_i h[i]*(1-h[i+k]) for i=0,...,n-1-k (lag k)

We want to minimize max_{k=1,...,n-1} corr[k], or maybe including k=0?

Looking at the problem statement: "C₅ = max_k ∫ h(x)(1-h(x+k)) dx"

The integral is over x from 0 to 2-k (or wherever the function is defined). In the discretization, this corresponds to corr[k] for k=1,...,n-1 (positive lags). The k=0 lag would be ∫ h(x)(1-h(x)) dx which is always < 1/4 * 2 = 1/2 and might or might not be included.

Looking at the parent code: `corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]` which includes k=0. And `c_max = float(np.max(corr))`. So k=0 IS included.

For k=0: corr[0] = sum_i h[i]*(1-h[i]) = sum_i h[i] - sum_i h[i]^2

If h is binary, corr[0] = 0 (since h*(1-h) = 0 for binary h).
If h is continuous, corr[0] > 0.

So for a binary solution, the k=0 lag is 0 and doesn't affect the max. For a continuous solution, it might.

This is an important observation! If I push h towards binary, the k=0 lag goes to 0, which means the effective constraint is only on k=1,...,n-1.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    result = np.clip(h - tau, 0.0, 1.0)
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = float(exp_vals.sum())
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    thresh = 1e-7
    for k in range(n - 1):
        wk = weights[k]
        if wk < thresh:
            continue
        n_k = int(k)
        grad[:n - n_k] += wk * (1.0 - h[n_k:])
        grad[n_k:] -= wk * h[:n - n_k]
    grad *= dx
    return grad

def fista_opt(h0, target, deadline, alpha=200.0):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t_mom = 1.0
    best_x = x.copy()
    best_score = compute_c5(x)
    step = 1.0 / (alpha * 0.02 + 1.0)
    max_step = 3.0 / (alpha * 0.01 + 1.0)

    for it in range(5000000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) * 0.5
        y = x + ((t_mom - 1.0) / t_new) * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)

        grad = lse_grad(y, alpha)

        accepted = False
        for _ in range(15):
            x_new = project_feasible(y - step * grad, target)
            sc = compute_c5(x_new)
            if sc < best_score + 0.0005:
                accepted = True
                break
            step *= 0.5

        if accepted:
            x_prev = x
            x = x_new
            t_mom = t_new
            step = min(step * 1.8, max_step)
            sc = compute_c5(x)
            if sc < best_score:
                best_score = sc
                best_x = x.copy()
        else:
            x_prev = x.copy()
            t_mom = 1.0
            step = max(step * 0.5, 1e-4)
            y = x.copy()

    return best_x, best_score

def binary_refine(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(42)

    for sweep in range(500):
        if time.time() > deadline:
            break
        improved = False
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                return h_cur, cur_score
            val = h_cur[idx]
            if val < 5e-8 or val > 1.0 - 5e-8:
                continue
            h_test = h_cur.copy()
            h_test[idx] = 0.0
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-12:
                h_cur = h_test
                cur_score = sc
                improved = True
                continue
            h_test = h_cur.copy()
            h_test[idx] = 1.0
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-12:
                h_cur = h_test
                cur_score = sc
                improved = True
        if not improved and sweep > 3:
            break
    return h_cur, cur_score

def sim_anneal_binary(h_init, target, deadline):
    n = len(h_init)
    sorted_h = np.sort(h_init)[::-1]
    mid_val = sorted_h[n // 2]
    h_bin = np.where(h_init > mid_val, 1.0, 0.0)
    if h_bin.sum() != n // 2:
        idx_sorted = np.argsort(h_init)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx_sorted[:n // 2]] = 1.0

    cur_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_score = cur_score

    T = 0.03
    T_min = 0.0003
    cooling = 0.99995
    rng = np.random.RandomState(31415)
    it = 0
    patience = 0
    max_patience = 80000

    while time.time() < deadline:
        it += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_score = compute_c5(h_bin)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-max(delta, 0.0) / T):
            cur_score = new_score
            patience = 0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            patience += 1

        if it % 2000 == 0:
            T = max(T * cooling, T_min)
        if patience > max_patience:
            T = max(T * 5.0, 0.01)
            patience = 0

    return best_h, best_score

def block_moves(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)

    for sweep in range(100):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(n * 2):
            if time.time() > deadline:
                return h_cur, cur_score
            length = rng.randint(2, 12)
            start = rng.randint(0, max(1, n - length))
            h_test = h_cur.copy()
            h_test[start:start + length] = 1.0 - h_test[start:start + length]
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-10:
                h_cur = h_test
                cur_score = sc
                improved = True
        if not improved:
            break
    return h_cur, cur_score

def multi_start_fista(h_ref, target, deadline, n_starts=8):
    n = len(h_ref)
    best_h = h_ref.copy()
    best_score = compute_c5(h_ref)
    rng = np.random.RandomState(55555)
    time_per = max(15.0, (deadline - time.time()) / max(n_starts, 1) - 3.0)

    for s in range(n_starts):
        if time.time() > deadline - 10.0:
            break
        scale = 0.015 + 0.015 * (s % 4)
        h_pert = h_ref + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        dl = time.time() + time_per
        h_opt, sc = fista_opt(h_pert, target, dl, alpha=300 + 200 * s)
        if sc < best_score:
            best_score = sc
            best_h = h_opt.copy()
    return best_h, best_score
