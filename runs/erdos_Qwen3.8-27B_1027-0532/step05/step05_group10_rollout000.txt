    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def power_mean_obj(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    vals = corr * dx
    vmax = np.max(vals)
    if p < 100:
        return float(np.mean(vals ** p)) ** (1.0 / p)
    shifted = vals / vmax
    log_sum = np.log(np.mean(shifted ** p))
    return float(vmax * np.exp(log_sum / p))


def power_mean_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    vmax = np.max(vals)
    if p < 100:
        phase = np.mean(vals ** (p - 1))
        weights = (vals ** (p - 1)) / phase
    else:
        shifted = vals / vmax
        sp = shifted ** p
        sp_mean = np.mean(sp)
        weights = sp / sp_mean
    grad = np.zeros(n)
    half_n = n - 1
    for k_idx in range(n):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        k = k_idx - half_n
        idx1 = np.arange(max(0, k), min(n, n + k))
        idx2 = np.arange(max(0, -k), min(n, n - k))
        grad[idx1] += w * (1.0 - h[idx1 - k]) * dx
        grad[idx2] -= w * h[idx2 + k] * dx
    return grad


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_power_mean(h_start, n, target, p_list, maxiter_per_p, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    start = time.time()

    for p in p_list:
        if time.time() - start > time_limit:
            break
        _global_p = p
        try:
            def obj_func(x):
                return power_mean_obj(x, p)
            def grad_func(x):
                return power_mean_grad(x, p)
            res = minimize(obj_func, h_cur, jac=grad_func,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_p, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def slsqp_logsumexp(h_start, n, target, alpha_list, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    start = time.time()

    for alpha in alpha_list:
        if time.time() - start > time_limit:
            break
        _global_alpha = alpha
        try:
            def obj_func(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                dx = 2.0 / len(x)
                c_max = np.max(corr) * dx
                shifted = np.clip(alpha * (corr * dx - c_max), -400.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def lp_alternating(h_init, n, target, max_iter, time_limit, n_active, damping_init, reg):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    half_n = n - 1

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=50000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = True
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 1e-4:
                    active_k.add(k_idx)
                    break

        active_k = top_k | active_k
        if len(active_k) > 3 * n_active:
            active_k = set(np.argsort(corr)[-3 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.9)
        else:
            damping = min(0.95, damping + 0.02)

        if it % 50 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.7)

    return best_h, best_c5


def momentum_subgradient(h, n, target, max_steps, time_limit, lr_init=0.01, momentum=0.9):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    velocity = np.zeros(n)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        lr = lr_init / (1.0 + 0.01 * step)
        velocity = momentum * velocity - lr * grad
        h_try = h_cur + velocity
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
        else:
            lr_decayed = lr * 0.5
            velocity = momentum * velocity - lr_decayed * grad
            h_try2 = h_cur + velocity
            h_try2 = project_feasible(h_try2, target)
            sc2 = compute_c5(h_try2)
            if sc2 < best_c5 - 1e-14:
                best_c5 = sc2
                best_h = h_try2.copy()
                h_cur = h_try2

    return best_h, best_c5


def block_coordinate_opt(h, n, target, time_limit, block_size=20, n_blocks=500):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(12345)

    for it in range(n_blocks):
        if time.time() - start > time_limit:
            break

        start_idx = rng.randint(0, n - block_size)
        idx = np.arange(start_idx, start_idx + block_size)
        h_fixed = h_cur.copy()
        h_sub = h_cur[idx].copy()

        def sub_obj(hs):
            h_try = h_fixed.copy()
            h_try[idx] = hs
            corr = np.correlate(h_try, 1.0 - h_try, mode='full')
            return float(np.max(corr)) * dx

        res = minimize(sub_obj, h_sub, bounds=[(0.0, 1.0)] * block_size,
                       method='Nelder-Mead', options={'maxiter': 200, 'xatol': 1e-8, 'fatol': 1e-12})
        h_new_sub = np.clip(res.x, 0.0, 1.0)
        h_cur[idx] = h_new_sub
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)

        if sc < best_c5 - 1e-15:
            best_c5 = sc
            best_h = h_cur.copy()

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multiplicative_weights_lp(h_init, n, target, max_iter, time_limit, n_top=20):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    n_corr = 2 * n - 1

    q = np.ones(n_corr) / n_corr
    eta = 0.1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        top_indices = np.argsort(q)[-n_top:]
        for k_idx in top_indices:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + 1e-4 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-10, max_iters=30000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = 0.7 * h + 0.3 * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        vals = corr * dx
        q *= np.exp(-eta * vals)
        q /= np.sum(q)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha, _global_p
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: Multiplicative weights LP (different from constraint generation)
    if tleft() > 250:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = multiplicative_weights_lp(best_h, n, target,
                                                 max_iter=10000, time_limit=lp_time, n_top=20)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Standard LP alternating
    if tleft() > 200:
        lp_time = min(200, tleft() * 0.25)
        h_lp, c5_lp = lp_alternating(best_h, n, target,
                                      max_iter=15000, time_limit=lp_time,
                                      n_active=30, damping_init=0.8, reg=1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Power-mean SLSQP with increasing p
    if tleft() > 150:
        h_ref, sc = slsqp_power_mean(best_h, n, target,
                                      [100, 500, 2000, 10000], 1500, 1e-14,
                                      time_limit=min(150, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Log-sum-exp SLSQP (different alpha schedule)
    if tleft() > 120:
        h_ref, sc = slsqp_logsumexp(best_h, n, target,
                                     [200, 1000, 5000, 20000], 1500, 1e-15,
                                     time_limit=min(120, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Momentum subgradient
    if tleft() > 80:
        h_sg, c5_sg = momentum_subgradient(best_h, n, target,
                                            max_steps=3000, time_limit=min(60, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: Power-mean SLSQP with very high p
    if tleft() > 100:
        h_ref, sc = slsqp_power_mean(best_h, n, target,
                                      [5000, 20000, 100000], 1200, 1e-15,
                                      time_limit=min(100, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Block coordinate optimization
    if tleft() > 80:
        h_bc, c5_bc = block_coordinate_opt(best_h, n, target,
                                            time_limit=min(60, tleft() * 0.4),
                                            block_size=15, n_blocks=300)
        if c5_bc < best_score:
            best_score = c5_bc
            best_h = h_bc

    # Phase 8: Higher resolution attempt
    if tleft() > 300:
        n_high = 600
        target_high = n_high / 2.0
        h_high = interpolate_to(best_h, n_high, target_high)
        if tleft() > 200:
            h_lp, c5_lp = lp_alternating(h_high, n_high, target_high,
                                          max_iter=8000, time_limit=min(150, tleft() * 0.4),
                                          n_active=40, damping_init=0.85, reg=1e-4)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_high if False else h_lp
                n = n_high
                target = target_high
        if tleft() > 80:
            h_ref, sc = slsqp_power_mean(best_h, len(best_h), len(best_h)/2.0,
                                          [200, 1000, 10000], 1000, 1e-14,
                                          time_limit=min(80, tleft() * 0.4))
            if sc < best_score:
                best_score = sc
                best_h = h_ref
                n = len(best_h)
                target = n / 2.0

    # Phase 9: Perturbed restarts with local refinement
    for noise in [0.02, 0.05, 0.1]:
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        h_ref, sc = slsqp_power_mean(h_pert, n, target,
                                      [500, 5000, 50000], 800, 1e-15,
                                      time_limit=min(50, tleft() * 0.4))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final LP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_lp, c5_lp = lp_alternating(best_h, n_use, target_use,
                                      max_iter=8000, time_limit=min(50, tleft() * 0.6),
                                      n_active=40, damping_init=0.95, reg=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 11: Final SLSQP with very high p
    if tleft() > 40:
        h_ref, sc = slsqp_power_mean(best_h, len(best_h), len(best_h)/2.0,
                                      [10000, 100000, 1000000], 800, 1e-16,
                                      time_limit=min(35, tleft() * 0.7))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 12: Final momentum subgradient polish
    if tleft() > 20:
        h_sg, c5_sg = momentum_subgradient(best_h, len(best_h), len(best_h)/2.0,
                                            max_steps=3000, time_limit=min(18, tleft() * 0.8),
                                            lr_init=0.005, momentum=0.95)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
