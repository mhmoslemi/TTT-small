    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    residual = target_sum - proj.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(proj > 1e-12)[0]
        if len(idx) > 0:
            proj[idx] += residual / len(idx)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_grad_smooth_max(h, alpha, weight_threshold=1e-12):
    """Compute gradient of smooth max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx
    
    max_c = c_vals.max()
    exp_vals = np.exp(alpha * (c_vals - max_c))
    weights = exp_vals / exp_vals.sum()
    
    grad = np.zeros(n)
    half_n = n - 1
    
    for k_idx in range(len(corr)):
        if weights[k_idx] < weight_threshold:
            continue
        d = k_idx - half_n
        w = weights[k_idx] * dx
        
        if d >= 0:
            if d < n:
                grad[:n-d] += w * (1.0 - h[d:])
                grad[d:] += -w * h[:n-d]
        else:
            dp = -d
            grad[dp:] += w * (1.0 - h[:n-dp])
            grad[:n-dp] += -w * h[dp:]
    
    return grad, float(max_c + np.log(exp_vals.sum()) / alpha)

def proximal_gradient(h_init, target, alpha, max_iter, step_init, time_limit, 
                       weight_threshold=1e-12):
    """Proximal gradient for smooth max with projection."""
    n = len(h_init)
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        grad, f_val = compute_grad_smooth_max(h, alpha, weight_threshold)
        h_trial = h - step * grad
        h_trial = project_feasible(h_trial, target)
        
        # Backtracking: if smooth objective increased too much, reduce step
        # (This is a heuristic - we check C5 instead of the smooth obj for efficiency)
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        # Simple backtracking
        c5_current = compute_c5(h)
        if c5_trial > c5_current + step * np.max(np.abs(grad)) * 2 * dx:
            step *= 0.7
        else:
            step = min(step * 1.05, step_init)
        
        h = h_trial
    
    return best_h, best_c5
