def project_feasible(h, target):
    """Project h onto {x: 0<=x<=1, sum(x)=target} using bisection on shift."""
    h = h.astype(float).copy()
    n = len(h)
    # Bisection on tau: find tau such that sum(clip(h - tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

    n = len(y)
    corr = np.correlate(y, 1.0 - y, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = float(exp_s.sum())
    weights = exp_s / sum_exp
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-7:
            continue
        lag = ki - (n - 1)
        grad += w * grad_corr_lag(y, lag, dx)
    return grad, c_max, sum_exp

def fista_lse(h0, target, alpha, max_iters, deadline, step_init=None):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    x_old = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if step_init is None:
        step_init = 0.005 / max(alpha * 0.001, 1.0)
    step = step_init
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_old)
        y = np.clip(y, 0.0, 1.0)
        
        grad, _, _ = lse_gradient(y, alpha, dx)
        
        f_x = compute_c5(x)
        x_new = x
        for _ in range(25):
            x_try = project_feasible(y - step * grad, target)
            f_try = compute_c5(x_try)
            if f_try <= f_x - 1e-13 or step < 1e-12:
                x_new = x_try
                break
            step *= 0.5
        else:
            x_new = project_feasible(y - step * grad, target)
        
        x_old = x
        x = x_new
        step = min(step * 1.3, step_init * 5.0)
        t = t_new
        
        f = compute_c5(x)
        if f < best_obj:
            best_obj = f
            best_x = x.copy()
    
    return best_x, best_obj

def activemax_subgrad(h0, target, deadline, max_active=10, tol_gap=0.005, step_init=0.005):
    n = len(h0)
    dx = 2.0 / n
    x = project_feasible(h0, target).copy()
    best_x = x.copy()
    best_obj = compute_c5(x)
    step = step_init
    
    max_iters = 1000000
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        c_max = float(np.max(corr))
        
        threshold = c_max - tol_gap
        active = np.where(corr >= threshold)[0]
        if len(active) > max_active:
            top_idx = np.argsort(corr)[::-1][:max_active]
            active = top_idx
        elif len(active) == 0:
            active = np.array([int(np.argmax(corr))])
        
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            grad += grad_corr_lag(x, lag, dx)
        grad /= float(len(active))
        
        f_curr = c_max
        x_new = x
        for _ in range(30):
            x_try = project_feasible(x - step * grad, target)
            f_try = compute_c5(x_try)
            if f_try <= f_curr - 1e-13 or step < 1e-12:
                x_new = x_try
                break
            step *= 0.5
        else:
            x_new = project_feasible(x - step * grad, target)
        
        x = x_new
        step = min(step * 1.3, step_init * 5.0)
        
        f = compute_c5(x)
        if f < best_obj:
            best_obj = f
            best_x = x.copy()
    
    return best_x, best_obj

def local_polish(h, target, deadline):
    """Coordinate descent on individual components."""
    n = len(h)
    dx = 2.0 / n
    x = project_feasible(h, target).copy()
    best_obj = compute_c5(x)
    best_x = x.copy()
    
    rng = np.random.RandomState(42)
    
    for sweep in range(50000):
        if time.time() > deadline:
            break
        for idx in range(n):
            if time.time() > deadline:
                break
            old = x[idx]
            # Try a few perturbations
            candidates = []
            if old > 0.01:
                candidates.append(max(0.0, old - 0.02))
                candidates.append(max(0.0, old * 0.7))
                candidates.append(max(0.0, old * 0.5))
            if old < 0.99:
                candidates.append(min(1.0, old + 0.02))
                candidates.append(min(1.0, old * 1.5))
            candidates.append(0.0)
            candidates.append(1.0)
            
            best_trial = None
            best_trial_sc = best_obj
            for c in candidates:
                if abs(c - old) < 1e-10:
                    continue
                x[idx] = c
                x = project_feasible(x, target)
                sc = compute_c5(x)
                if sc < best_trial_sc:
                    best_trial_sc = sc
                    best_trial = x.copy()
            if best_trial is not None:
                x = best_trial
                best_obj = best_trial_sc
                best_x = x.copy()
            else:
                x[idx] = old
                x = project_feasible(x, target)
    
    return best_x, best_obj
