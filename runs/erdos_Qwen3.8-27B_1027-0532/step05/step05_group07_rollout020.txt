def cd_polish(h, n, target, deadline, max_sweeps=100):
    """Coordinate descent polish on the exact max objective."""
    h = h.copy()
    dx = 2.0 / n
    best_score = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Find best h[i] to minimize max_k c_k
            # c_k as function of t = h[i]:
            # c_k(t) = c_k(current) + t * dk where dk = d(c_k)/d(h[i])
            # = c_k_current + t * [(1-h[i-k*])*1_{valid} - h[i+k*]*1_{valid}]
            
            # Precompute the slope for each k
            # dk for shift k (indexed 0 to 2n-2, corresponding to k_shift = idx-(n-1))
            c = compute_corr(h)
            dk = np.zeros(2*n-1)
            for idx in range(2*n-1):
                k_shift = idx - (n-1)
                val = 0.0
                if 0 <= i - k_shift < n:
                    val += (1.0 - h[i - k_shift])
                if 0 <= i + k_shift < n:
                    val -= h[i + k_shift]
                dk[idx] = val
            
            # Now c_k(t) = c[idx] + (t - h[i]) * dk[idx]
            # We want min_t max_k (c[idx] + (t-h[i])*dk[idx]) * dx
            # = dx * (h[i] related stuff)... let's just bisection on t
            
            old_t = h[i]
            # Bisection to find t that minimizes max_k c_k(t)
            lo = 0.0
            hi = 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                c_at_mid = c + (mid - old_t) * dk
                max_at_mid = np.max(c_at_mid)
                
                # Check derivative at lo and hi to determine direction
                # Actually, for min-max of linear functions, the optimal t
                # is where the "active" constraints cross.
                # Simpler: just do ternary search or bisection on the derivative.
                
                # Evaluate at lo and hi
                c_lo = c + (lo - old_t) * dk
                c_hi = c + (hi - old_t) * dk
                
                # The function g(t) = max_k c_k(t) is convex (max of linear)
                # So we can use ternary search
                t1 = lo + (hi - lo) / 3
                t2 = hi - (hi - lo) / 3
                g1 = np.max(c + (t1 - old_t) * dk)
                g2 = np.max(c + (t2 - old_t) * dk)
                if g1 < g2:
                    hi = t2
                else:
                    lo = t1
            
            t_star = (lo + hi) / 2.0
            if abs(t_star - old_t) > 1e-10:
                h[i] = t_star
                improved = True
        
        # Re-project
        h = project_feasible(h, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
        
        if not improved and sweep > 10:
            break
    
    return h, best_score

Wait, ternary search on a convex function works! The function g(t) = max_k (c_k + (t-t0)*d_k) is convex in t (it's the max of linear functions), so ternary search will find the minimum.

But I realize there's an issue: after I change h[i], the c values change for ALL k, not just in the linear way I assumed. Wait, no - c_k is bilinear in h, so if I only change h[i], then c_k changes linearly in h[i]. So the linear approximation IS exact!

c_k = sum_j h[j+k]*(1-h[j])

If I change h[i] from t0 to t:
- Terms where i appears as j+k: h[i]*(1-h[j]) → t*(1-h[j]), change = (t-t0)*(1-h[j])
- Terms where i appears as j: h[j+k]*(1-h[i]) → h[j+k]*(1-t), change = -(t-t0)*h[j+k]

So c_k(t) = c_k(t0) + (t-t0) * [(1-h[i-k*]) - h[i+k*]] (with validity indicators)

This IS linear in t. So the ternary search on the exact function is correct!

The only issue is the projection: after changing h[i], the sum constraint might be violated. But if I only change one coordinate at a time and re-project, the projection might undo my change.

A better approach: do the coordinate descent WITHOUT projection, and only project at the end of each sweep. During a sweep, the sum constraint might be slightly violated, but the final projection fixes it.

Actually, this could cause problems because the optimal t for coordinate i depends on the sum constraint (we want sum(h) = n/2). Without the constraint, the optimal t might push the sum away from n/2.

Alternative: Account for the sum constraint by adding a Lagrange multiplier to the objective:
min_t max_k c_k(t) + λ*(sum(h) - n/2)

The λ term is linear in t: λ*(t - t0) [since only h[i] changes].

So the modified objective is: max_k (c_k + (t-t0)*d_k) + λ*(t-t0)

This is still convex in t (max of linear + linear), so ternary search still works.

I can update λ using a subgradient method: λ ← λ - step*(sum(h) - n/2).

This is getting complex. Let me simplify: just do CD without the sum constraint during the sweep, and project at the end. The CD will still make progress because the objective is being decreased, and the projection only slightly changes the solution.

# Helper functions (top-level)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smoothed_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = compute_corr(h)
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    s = np.sum(exp_vals)
    obj = dx * (c_max + np.log(s) / alpha)
    p = exp_vals / s
    
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    w_idx1 = n - 1 + i_idx - j_idx
    valid1 = (w_idx1 >= 0) & (w_idx1 < 2*n - 1)
    M1 = np.where(valid1, p[np.clip(w_idx1, 0, 2*n-2)], 0.0)
    term1 = M1 @ (1.0 - h)
    
    w_idx2 = n - 1 + j_idx - i_idx
    valid2 = (w_idx2 >= 0) & (w_idx2 < 2*n - 1)
    M2 = np.where(valid2, p[np.clip(w_idx2, 0, 2*n-2)], 0.0)
    term2 = M2 @ h
    
    grad = dx * (term1 - term2)
    return obj, grad

def aug_lag_optimize(h, n, target, alpha, deadline, mu=100.0, max_outer=50, max_inner=300):
    """Augmented Lagrangian with L-BFGS-B inner solver."""
    h_cur = h.copy()
    lam = 0.0
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        def obj(x):
            f, _ = compute_smoothed_obj_grad(x, alpha)
            s = np.sum(x) - target
            return f + 0.5 * mu * s * s + lam * s
        
        def grad(x):
            _, g = compute_smoothed_obj_grad(x, alpha)
            s = np.sum(x) - target
            return g + (mu * s + lam) * np.ones(n)
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = res.x
        except Exception:
            break
        
        s = np.sum(h_new) - target
        if abs(s) < 1e-10:
            h_cur = project_feasible(h_new, target)
            break
        
        lam += mu * s
        h_cur = h_new
    
    return project_feasible(h_cur, target)

def cd_exact(h, n, target, deadline, max_sweeps=50):
    """Exact coordinate descent on the max objective."""
    h = h.copy()
    dx = 2.0 / n
    best_score = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = compute_corr(h)
        changed = False
        
        for i in range(n):

                break
            
            # Compute dk for each correlation index
            dk = np.zeros(2*n-1)
            for idx in range(2*n-1):
                k = idx - (n-1)
                val = 0.0
                if 0 <= i - k < n:
                    val += (1.0 - h[i-k])
                if 0 <= i + k < n:
                    val -= h[i+k]
                dk[idx] = val
            
            # Ternary search for optimal t
            old_t = h[i]
            lo, hi = 0.0, 1.0

                t2 = hi - (hi - lo) / 3.0

                else:
                    lo = t1
            
            t_star = (lo + hi) / 2.0
            if abs(t_star - old_t) > 1e-12:
                h[i] = t_star
                changed = True
        
        h = project_feasible(h, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
        
        if not changed:

    return h, best_score
