import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    n = len(h)
    # Distribute the difference among non-saturated variables
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    # Handle residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            per2 = residual / len(idx2)
            h[idx2] = np.clip(h[idx2] + per2, 0.0, 1.0)
    return h

def compute_gradient_lse(h, alpha):
    """Compute gradient of the log-sum-exp smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Compute p_k = softmax(alpha * corr)
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    p = exp_vals / np.sum(exp_vals)
    # Compute gradient: g[i] = dx * sum_k p[k] * d_corr[k]/d_h[i]
    g = np.zeros(n)
    for k in range(2 * n - 1):
        tau = k - (n - 1)
        w = p[k] * dx
        if abs(w) < 1e-15:
            continue
        # First term: (1-h[i+tau]) for valid i
        if tau >= 0:
            g[0:n-tau] += w * (1.0 - h[tau:n])
        else:
            g[-tau:n] += w * (1.0 - h[0:n+tau])
        # Second term: -h[i-tau] for valid i
        if tau >= 0:
            g[tau:n] -= w * h[0:n-tau]
        else:
            g[0:n+tau] -= w * h[-tau:n]
    return g

def smooth_obj_and_grad(h, alpha):
    """Compute the smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    log_sum = np.log(np.sum(exp_vals))
    obj = max_corr + log_sum / alpha
    grad = compute_gradient_lse(h, alpha)
    return obj, grad

def lbfgsb_optimize(h_init, n, target, alphas, maxiter_per_alpha=500, time_limit=300):
    """Optimize using L-BFGS-B with log-sum-exp smooth approximation."""
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        
        def fun(val):
            val = np.clip(val, 0.0, 1.0)
            obj, _ = smooth_obj_and_grad(val, alpha)
            return obj
        
        def grad(val):
            val = np.clip(val, 0.0, 1.0)
            _, g = smooth_obj_and_grad(val, alpha)
            # Project gradient onto the null space of the constraint
            mean_g = np.mean(g)
            return g - mean_g  # This handles the equality constraint approximately
        
        try:
            res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter_per_alpha,
                                                   'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    
    return best_h, best_score

def subgradient_optimize(h_init, n, target, max_steps, time_limit):
    """Subgradient method with backtracking line search."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        
        grad = np.zeros(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h[idxs[idx_valid2] + k])
        grad *= dx
        
        # Project gradient: remove mean to maintain sum constraint
        grad -= np.mean(grad)
        
        improved = False
        for lr in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        
        if not improved and step > 100:
            # Try random restart from current best
            h = best_h + np.random.randn(n) * 0.01
            h = project_feasible(h, target)
            if compute_c5(h) < best_c5:
                # Keep going
                pass
            else:
                h = best_h.copy()
        
        if not improved and step > 500:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    p = exp_vals / np.sum(exp_vals)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        tau = k - (n - 1)
        w = p[k] * dx
        if abs(w) < 1e-16:
            continue
        if tau >= 0:
            g[0:n - tau] += w * (1.0 - h[tau:n])
            g[tau:n] -= w * h[0:n - tau]
        else:
            g[-tau:n] += w * (1.0 - h[0:n + tau])
            g[0:n + tau] -= w * h[-tau:n]
    return g


def lse_objective(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_corr))
    return max_corr + np.log(np.sum(exp_vals)) / alpha


def lbfgsb_round(h_init, n, target, alphas, maxiter_per_alpha, time_limit):
    start = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break

        def fun(val):
            obj = lse_objective(val, alpha)
            return obj

        def grad(val):
            g = compute_lse_gradient(val, alpha)
            g = g - np.mean(g)
            return g

        try:
            res = minimize(fun, h, jac=grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': maxiter_per_alpha,
                                    'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score - 1e-15:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass

    return best_h, best_score


def alternating_lp(h_init, n, target, max_iter, time_limit, n_active=50, reg=1e-4, damping_init=0.8):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    damping = damping_init

    corr = np.correlate(h, 1.0 - h, mode='full')
    active_k = set(np.argsort(corr)[-n_active:].tolist())

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        sorted_active = sorted(active_k)
        for k_idx in sorted_active:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=1e-12, max_iters=100000, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = h_var.value is not None
            except Exception:
                pass

        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()

        top_k = set(np.argsort(corr)[-n_active:].tolist())

                    break

        active_k = top_k | active_k

        else:
            damping = min(0.98, damping + 0.01)

        if it % 80 == 0 and reg > 1e-7:
            reg = max(1e-7, reg * 0.75)

    return best_h, best_c5


def subgradient_optimize(h_init, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    idxs = np.arange(n)
    stall_count = 0

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star

        grad = np.zeros(n)
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h[idxs[idx_valid2] + k]
        grad *= dx
        grad -= np.mean(grad)

        improved = False
        for lr in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-15:
                best_c5 = sc
                best_h = h_try.copy()
                h = h_try
                improved = True
                stall_count = 0
                break

        if not improved:
            stall_count += 1
            if stall_count > 200:
                break

    return best_h, best_c5


def local_perturbation(h_init, n, target, time_limit, seed=99):
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for _ in range(200000):
        if time.time() - start > time_limit:
            break

        n_pert = rng.randint(1, 4)
        idxs = rng.choice(n, size=n_pert, replace=False)
        h_try = h.copy()
        for idx in idxs:
            delta = rng.uniform(-0.03, 0.03)
            h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)

        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = h_try.copy()
            h = h_try

    return best_h, best_c5


def symmetrize(h, n, target):
    h_sym = (h + h[::-1]) / 2.0
    h_sym = project_feasible(h_sym, target)
    return h_sym


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP with constraint generation (more active constraints)
    if tleft() > 200:
        lp_time = min(300, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                     max_iter=30000, time_limit=lp_time,
                                     n_active=50, reg=1e-4, damping_init=0.8)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: L-BFGS-B with analytical gradient
    if tleft() > 150:
        h_lbf, c5_lbf = lbfgsb_round(best_h, n, target,
                                     alphas=[20, 50, 100, 300, 1000],
                                     maxiter_per_alpha=300,
                                     time_limit=min(200, tleft() * 0.4))
        if c5_lbf < best_score:
            best_score = c5_lbf
            best_h = h_lbf

    # Phase 3: LP with higher active set
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=30000, time_limit=lp_time,
                                     n_active=70, reg=1e-5, damping_init=0.85)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: L-BFGS-B with high alpha
    if tleft() > 120:
        h_lbf, c5_lbf = lbfgsb_round(best_h, n, target,
                                     alphas=[1000, 3000, 10000, 50000],
                                     maxiter_per_alpha=200,
                                     time_limit=min(150, tleft() * 0.4))
        if c5_lbf < best_score:
            best_score = c5_lbf
            best_h = h_lbf

    # Phase 5: Subgradient refinement
    if tleft() > 60:
        h_sg, c5_sg = subgradient_optimize(best_h, n, target,
                                           max_steps=8000,
                                           time_limit=min(80, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: LP polish with tight regularization
    if tleft() > 100:
        lp_time = min(100, tleft() * 0.4)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=30000, time_limit=lp_time,
                                     n_active=80, reg=1e-6, damping_init=0.92)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Symmetrization attempt
    if tleft() > 40:
        h_sym = symmetrize(best_h, n, target)
        sc_sym = compute_c5(h_sym)
        if sc_sym < best_score:
            best_score = sc_sym
            best_h = h_sym
            if tleft() > 40:
                h_lbf, c5_lbf = lbfgsb_round(best_h, n, target,
                                             alphas=[500, 2000, 10000],
                                             maxiter_per_alpha=200,
                                             time_limit=min(40, tleft() * 0.5))
                if c5_lbf < best_score:
                    best_score = c5_lbf
                    best_h = h_lbf

    # Phase 8: Final L-BFGS-B
    if tleft() > 60:
        h_lbf, c5_lbf = lbfgsb_round(best_h, n, target,
                                     alphas=[5000, 20000, 100000, 500000],
                                     maxiter_per_alpha=200,
                                     time_limit=min(60, tleft() * 0.5))
        if c5_lbf < best_score:
            best_score = c5_lbf
            best_h = h_lbf

    # Phase 9: Local perturbation
    if tleft() > 30:
        h_pert, c5_pert = local_perturbation(best_h, n, target,
                                             time_limit=min(30, tleft() * 0.5),
                                             seed=seed + 7)
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert

    # Phase 10: Final subgradient squeeze
    if tleft() > 20:
        h_sg, c5_sg = subgradient_optimize(best_h, n, target,
                                           max_steps=5000,
                                           time_limit=min(20, tleft() * 0.6))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 11: One more LP with very tight settings
    if tleft() > 40:
        lp_time = min(40, tleft() * 0.6)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=20000, time_limit=lp_time,
                                     n_active=100, reg=1e-7, damping_init=0.95)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
