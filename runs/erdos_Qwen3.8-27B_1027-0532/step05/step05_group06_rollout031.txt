        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def weighted_corr_obj_grad(x, p, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    val = float(np.dot(p, corr))
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    grad = np.zeros(n)
    K = len(corr)
    for ki in range(K):
        w = p[ki]
        if w < 1e-15:
            continue
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def multiplicative_weights(h_init, target, n_outer, n_inner, eta_start, eta_end, deadline, mu=50.0):
    n = len(h_init)
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    K = len(corr)
    p = np.ones(K) / K
    best_h = h.copy()
    best_val = float(np.max(corr))
    bounds = [(0.0, 1.0)] * n

    for t in range(n_outer):
        if time.time() > deadline:
            break
        eta = eta_start * (eta_end / eta_start) ** (t / max(1, n_outer - 1))

        def f_inner(x):
            return weighted_corr_obj_grad(x, p, target, mu)[0]

        def g_inner(x):
            return weighted_corr_obj_grad(x, p, target, mu)[1]

        try:
            res = minimize(f_inner, h, jac=g_inner, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': n_inner, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
        except Exception:
            h_new = h

        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        val_new = float(np.max(corr_new))
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        h = h_new

        p_new = np.exp(eta * (corr_new - corr_new.max()))
        p_new = np.clip(p_new, 1e-30, None)
        p = p_new / p_new.sum()

    return best_h, best_val


def top_lag_gradient_descent(h, target, n_top, n_steps, lr_init, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = lr_init

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-13:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
            lr = min(lr * 1.03, 0.05)
        else:
            lr *= 0.75
            if lr < 1e-9:
                break
    return h_cur, cur_score


def coordinate_descent_fast(h, target, n_sweeps, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def slsqp_polish(h, target, alpha, maxiter):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[0]

    def g(x):
        return lse_obj_and_grad(x, alpha, target, 1e10)[1]

    def eq(x):
        return np.sum(x) - target

    def eq_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': eq, 'jac': eq_jac}]
    try:
        res = minimize(f, h_cur, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def upsample_and_optimize(h_base, n_base, target_base, n_fine, target_fine, deadline, alphas, mu):
    x_old = np.linspace(0.0, 1.0, n_base)
    x_new = np.linspace(0.0, 1.0, n_fine)
    h_fine = np.interp(x_new, x_old, h_base)
    h_fine = project_feasible(h_fine, target_fine)
    for a in alphas:
        if time.time() > deadline:
            break
        h_fine = lbfgs_optimize(h_fine, target_fine, a, mu, 300, deadline)
    return h_fine


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    deadline = start_time + budget_s - 30.0
    n = len(initial_h_values)
    target = n / 2.0
    mu = 50.0

    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: LSE L-BFGS-B with geometric alpha schedule
    alphas_p1 = [5, 20, 100, 500, 2000, 10000, 50000, 200000]
    for a in alphas_p1:
        if tleft() < 80:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Top-lag gradient descent
    if tleft() > 100:
        h_eq, sc_eq = top_lag_gradient_descent(best_h, target, n_top=5,
                                               n_steps=400, lr_init=0.003,
                                               deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, target, 50000, mu, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Multiplicative weights primal-dual
    if tleft() > 200:
        h_mw, sc_mw = multiplicative_weights(best_h, target,
                                             n_outer=80, n_inner=80,
                                             eta_start=0.3, eta_end=8.0,
                                             deadline=deadline, mu=mu)
        if sc_mw < best_score:
            best_score = sc_mw
            best_h = h_mw
        h_new = lbfgs_optimize(best_h, target, 100000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Another MW round with different params
    if tleft() > 150:
        h_mw2, sc_mw2 = multiplicative_weights(best_h, target,
                                               n_outer=60, n_inner=100,
                                               eta_start=0.1, eta_end=3.0,
                                               deadline=deadline, mu=mu)
        if sc_mw2 < best_score:
            best_score = sc_mw2
            best_h = h_mw2
        h_new = lbfgs_optimize(best_h, target, 200000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        alphas_fine = [20, 200, 2000, 20000, 100000]
        h_fine = upsample_and_optimize(best_h, n, target, n2, target2,
                                       deadline, alphas_fine, mu)
        sc_fine = compute_c5(h_fine)
        if sc_fine < best_score:
            best_score = sc_fine
            best_h = h_fine
            n = n2
            target = target2

    # Phase 6: Targeted perturbations
    n_cur = len(best_h)
    tgt_cur = n_cur / 2.0
    perturb_strategies = [
        ('gauss', 0.05), ('gauss', 0.1), ('gauss', 0.2),
        ('binary_flip', 0.05), ('binary_flip', 0.1),
        ('block', 8), ('block', 20),
    ]
    for strat, param in perturb_strategies:
        if tleft() < 60:
            break
        h_pert = best_h.copy()
        if strat == 'gauss':
            h_pert += rng.randn(n_cur) * param
        elif strat == 'binary_flip':
            n_flip = max(1, int(n_cur * param))
            idx = rng.choice(n_cur, size=n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif strat == 'block':
            block = param
            start = rng.randint(0, n_cur - block)
            h_pert[start:start + block] = 1.0 - h_pert[start:start + block]
        h_pert = project_feasible(h_pert, tgt_cur)
        h_new = lbfgs_optimize(h_pert, tgt_cur, 2000, mu, 200, deadline)
        h_new = lbfgs_optimize(h_new, tgt_cur, 50000, mu, 200, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Final top-lag equalization + LSE
    if tleft() > 80:
        h_eq, sc_eq = top_lag_gradient_descent(best_h, tgt_cur, n_top=3,
                                               n_steps=300, lr_init=0.002,
                                               deadline=deadline)
        if sc_eq < best_score:
            best_score = sc_eq
            best_h = h_eq
        h_new = lbfgs_optimize(best_h, tgt_cur, 500000, mu, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Coordinate descent polish
    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent_fast(best_h, tgt_cur, n_sweeps=3,
                                              deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: SLSQP final polish
    if tleft() > 50:
        h_slq = slsqp_polish(best_h, tgt_cur, 50000, 3000)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 10: One more MW round if time allows
    if tleft() > 100:
        h_mw3, sc_mw3 = multiplicative_weights(best_h, tgt_cur,
                                               n_outer=50, n_inner=100,
                                               eta_start=0.05, eta_end=2.0,
                                               deadline=deadline, mu=mu)
        if sc_mw3 < best_score:
            best_score = sc_mw3
            best_h = h_mw3
        h_new = lbfgs_optimize(best_h, tgt_cur, 1000000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 11: Final coordinate descent
    if tleft() > 60:
        h_cd, sc_cd = coordinate_descent_fast(best_h, tgt_cur, n_sweeps=2,
                                              deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 12: Final SLSQP
    if tleft() > 30:
        h_final = slsqp_polish(best_h, tgt_cur, 100000, 2000)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_final = project_feasible(best_h, tgt_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
