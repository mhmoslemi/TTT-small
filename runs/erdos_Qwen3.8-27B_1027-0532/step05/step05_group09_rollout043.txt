import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)

    if lr_init is None:
        lr_init = 1.0 / (max(alpha * 0.05, 1.0))
    step = lr_init

    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(30):
            obj_new, _ = lse_obj_and_grad(x_new, alpha)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):

            step *= (1.0 - shrink)

            break
        dl = start_time + min(40, tleft() - 5)
        h_new, sc_new = active_set_balance(best_h, target, m, dl, max_iters=3000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 3: Coordinate descent with line search
    if tleft() > 100:
        dl = start_time + min(100, tleft() - 5)
        h_new, sc_new = coord_descent_v2(best_h, target, dl, max_sweeps=20000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: More FISTA at very high alpha
    if tleft() > 80:
        dl = start_time + min(80, tleft() - 5)
        h_new, sc_new = fista_optimize(best_h, target, 50000, 50000, dl)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 5: Active set balancing again
    if tleft() > 60:
        for m in [3, 5, 7]:
            if tleft() < 20:
                break
            dl = start_time + min(30, tleft() - 5)
            h_new, sc_new = active_set_balance(best_h, target, m, dl, max_iters=3000)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 6: Two-swap local search
    if tleft() > 80:
        dl = start_time + min(80, tleft() - 5)
        h_new, sc_new = two_swap_search(best_h, target, dl, n_iter=50000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 7: Frequency-domain perturbation restarts
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        scales = [0.03, 0.06, 0.02, 0.05]
        for s in scales:
            if tleft() < 25:
                break
            fsize = 2 * n
            H = np.fft.rfft(best_h, fsize)
            noise = (rng.randn(fsize // 2 + 1) + 1j * rng.randn(fsize // 2 + 1)) * s
            H_new = H + noise * np.minimum(np.abs(H), 1.0)
            h_try = np.fft.irfft(H_new, fsize)[:n]
            h_try = project_feasible(h_try, target)
            dl = start_time + min(25, tleft() - 5)
            h_new, sc_new = fista_optimize(h_try, target, 5000, 20000, dl)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 8: Final coordinate descent polish
    if tleft() > 40:
        dl = start_time + min(40, tleft() - 3)
        h_new, sc_new = coord_descent_v2(best_h, target, dl, max_sweeps=20000)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 9: Final active set + coordinate
    if tleft() > 30:
        for m in [3, 5]:
            if tleft() < 10:
                break
            dl = start_time + min(15, tleft() - 3)
            h_new, sc_new = active_set_balance(best_h, target, m, dl, max_iters=2000)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
        if tleft() > 15:
            dl = start_time + min(15, tleft() - 3)
            h_new, sc_new = coord_descent_v2(best_h, target, dl, max_sweeps=10000)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Final projection
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
