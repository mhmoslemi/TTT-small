    # Phase 5: SLSQP high alpha polish (900 - 1100s)
    dl5 = start_time + 1100
    if tleft() > 80:
        h_s, sc_s = slsqp_smooth(best_h, n, target,
                                  [5000, 20000, 50000],
                                  3000, 1e-17, dl5)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 6: Final subgradient micro-polish (1100 - 1250s)
    dl6 = start_time + 1250
    if tleft() > 30:
        h_sg, sc_sg = subgradient_micro(best_h, n, target, 2000, 5, 0.0005, 0.999, dl6)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 7: Final SLSQP (1250 - 1290s)
    if tleft() > 20:
        h_s, sc_s = slsqp_smooth(best_h, n, target, [100000], 2000, 1e-18,
                                  start_time + budget_s - 10)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
