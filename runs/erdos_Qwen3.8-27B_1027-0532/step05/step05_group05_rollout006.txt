For coordinate descent, at each step:
- Pick a coordinate j
- Compute the optimal value of h[j] given all other h[i] fixed
- The objective is max_k f_k(h), and changing h[j] affects f_k for all k
- The optimal h[j] minimizes the maximum of the affected f_k values

- If j <= n-1-k: the term h[j](1-h[j+k]) changes from a(1-h[j+k]) to b(1-h[j+k])
  Change: (b-a)(1-h[j+k])
- If j >= k: the term h[j-k](1-h[j]) changes from h[j-k](1-a) to h[j-k](1-b)
  Change: (a-b)h[j-k] = -(b-a)h[j-k]

So the total change in f_k when changing h[j] from a to b:
Δf_k = (b-a)[(1-h[j+k])*I(j<=n-1-k) - h[j-k]*I(j>=k)]

Let d = b-a. Then Δf_k = d * c_k where:
c_k = (1-h[j+k])*I(j<=n-1-k) - h[j-k]*I(j>=k)

So f_k becomes f_k + d*c_k. And we want to choose d (subject to 0 ≤ a+d ≤ 1 and the sum constraint) to minimize max_k (f_k + d*c_k).

This is a 1D minimax problem! We want:
min_d max_k (f_k + d*c_k)

subject to: -(sum_adjustment_needed) constraints on d (from the sum constraint, changing h[j] by d changes the sum by d, so we'd need to adjust another coordinate... this makes it more complex).

min_d max_k (f_k + d*c_k)

The optimal d is where the two most "extreme" lines f_k + d*c_k cross. This can be found by:
- For each pair (k1, k2), find the d where f_{k1} + d*c_{k1} = f_{k2} + d*c_{k2}
- d = (f_{k2} - f_{k1}) / (c_{k1} - c_{k2})
- Evaluate the max at each such d and find the minimum

            if time.time() > deadline:
                break
            a = h_cur[j]
            if a <= 0.001 or a >= 0.999:
                continue
            over = get_overlap(h_cur)
            c_max = float(np.max(over))
            c_k_vals = np.zeros(n)
            for k in range(n):
                c = 0.0
                if j <= n - 1 - k:
                    c += 1.0 - h_cur[j + k]
                if j >= k:
                    c -= h_cur[j - k]
                c_k_vals[k] = c
            c_k_vals *= (2.0 / n)
            lo = 0.0
            hi = 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(over + m1 * c_k_vals - a * c_k_vals)
                f2 = np.max(over + m2 * c_k_vals - a * c_k_vals)
                if f1 > f2:
                    lo = m1
                else:
                    hi = m2
            new_val = (lo + hi) / 2.0
            h_try = h_cur.copy()
            h_try[j] = new_val
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_score:
                h_cur = h_try
                best_score = sc
    return h_cur, best_score


def multi_start_slsqp(h, n, target, n_starts, scales, deadline, alphas, maxiter, rng):
    best_h = h.copy()
    best_score = compute_c5(h_cur := h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        sc = scales[s % len(scales)]
        h_pert = project_feasible(h + rng.randn(n) * sc, target)
        h_ref = slsqp_optimize(h_pert, n, target, alphas, maxiter, 1e-14, deadline)
        sc_ref = compute_c5(h_ref)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    n = len(initial_h_values)
    target = n / 2.0
    best_h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_score = compute_c5(best_h)

    h_flat = spectral_flatten(best_h, 80, rng)
    sc_flat = compute_c5(h_flat)
    if sc_flat < best_score:
        best_score = sc_flat
        best_h = h_flat

    alphas1 = [5, 10, 25, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    h_ref = slsqp_optimize(best_h, n, target, alphas1, 3000, 1e-14, start_time + 350)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    for m_act in [3, 5, 8, 12, 20]:
        if time.time() > hard_deadline - 250:
            break
        h_ad, sc_ad = active_descent(best_h, n, target, 3000, m_act, time.time() + 40)
        if sc_ad < best_score:
            best_score = sc_ad
            best_h = h_ad

    alphas2 = [2000, 5000, 10000, 20000, 50000, 100000, 200000]
    h_ref = slsqp_optimize(best_h, n, target, alphas2, 5000, 1e-15, start_time + 650)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    for m_act in [3, 5, 8]:
        if time.time() > hard_deadline - 150:
            break
        h_ad, sc_ad = active_descent(best_h, n, target, 5000, m_act, time.time() + 30)
        if sc_ad < best_score:
            best_score = sc_ad
            best_h = h_ad

    h_co, sc_co = coordinate_optimize(best_h, n, target, 2, min(start_time + 850, hard_deadline - 100), rng)
    if sc_co < best_score:
        best_score = sc_co
        best_h = h_co

    if time.time() < hard_deadline - 80:
        h_ref = slsqp_optimize(best_h, n, target, [5000, 20000, 100000, 300000], 5000, 1e-16, hard_deadline - 40)
        sc = compute_c5(h_ref)
        if sc < best_score:

            best_h = h_ref

        if sc_ad < best_score:

            best_h = h_ad

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
