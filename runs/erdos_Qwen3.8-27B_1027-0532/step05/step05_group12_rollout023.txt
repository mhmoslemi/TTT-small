        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                break

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
    else:
        top_indices = np.arange(2 * n - 1)
    sub_corr = corr[top_indices]
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def get_cons_fun(n):
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    return cons_fun


def get_cons_jac(n):
    def cons_jac(h):
        return np.ones(n)
    return cons_jac


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': get_cons_fun(n), 'jac': get_cons_jac(n)}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def projected_gradient(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    top_m_list = [5, 10, 20, 30, 50, 80, 150, 2 * n - 1]
    steps_per_alpha = 2000

    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_list[min(ai, len(top_m_list) - 1)]
        momentum_v = np.zeros(n)
        beta = 0.8
        stalled = 0

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            grad = smooth_grad(h, alpha, dx, top_m=top_m)
            momentum_v = beta * momentum_v + (1.0 - beta) * grad
            gnorm = np.max(np.abs(momentum_v))
            if gnorm < 1e-13:
                break
            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 2.0)
            improved = False
            for _ in range(30):
                h_try = h - lr * momentum_v
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 80:
                    break

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Extended alternating LP at original resolution
    if tleft() > 300:
        h_lp, c5_lp = alternating_lp_fast(
            best_h, n, target, max_iter=50000,
            time_limit=budget_s, start_time=start_time,
            deadline_offset=budget_s - 300, damping_start=0.8, damping_end=0.4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Focused LP (only active constraints)
    if tleft() > 250:
        h_fp, c5_fp = alternating_lp_focused(
            best_h, n, target, max_iter=30000,
            time_limit=budget_s, start_time=start_time,
            deadline_offset=budget_s - 250, damping=0.5)
        if c5_fp < best_score:
            best_score = c5_fp
            best_h = h_fp

    # Phase 3: Projected gradient for fine-tuning
    if tleft() > 200:
        h_pg, c5_pg = projected_gradient(
            best_h, n, target,
            time_limit=budget_s, start_time=start_time,
            deadline_offset=budget_s - 200)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 4: Perturbation restarts at current resolution
    for noise in [0.02, 0.05, 0.1]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_lp, c5_lp = alternating_lp_fast(
            h_pert, n, target, max_iter=10000,
            time_limit=budget_s, start_time=start_time,
            deadline_offset=budget_s - 100, damping_start=0.9, damping_end=0.5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: SLSQP polish
    if tleft() > 80:
        n_use = len(best_h)
        target_use = n_use / 2.0
        _global_n = n_use
        _global_target = target_use
        h_slq = slsqp_optimize(best_h, n_use, target_use,
                               alphas=[100, 500, 2000, 10000, 50000],
                               maxiter=1500, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=40)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 6: Try higher resolution
    for n_try in [600, 800, 1000]:
        if tleft() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        h_lp, c5_lp = alternating_lp_fast(
            h_interp, n_try, target_try, max_iter=30000,
            time_limit=budget_s, start_time=start_time,
            deadline_offset=max(40, budget_s - tleft() + 100),
            damping_start=0.8, damping_end=0.4)

        if tleft() > 80:
            h_fp, c5_fp = alternating_lp_focused(
                h_lp, n_try, target_try, max_iter=15000,
                time_limit=budget_s, start_time=start_time,
                deadline_offset=max(30, budget_s - tleft() + 60),
                damping=0.5)
            if c5_fp < c5_lp:
                h_lp, c5_lp = h_fp, c5_fp

        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
            n = n_try
            target = target_try

    # Phase 7: Final SLSQP at best resolution
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                               alphas=[5000, 20000, 100000],
                               maxiter=1000, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
