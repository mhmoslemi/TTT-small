grad[m] = ∑_{k=0}^{N-1} w_k * [(1-h[m+k])*I(m≤N-1-k) - h[m-k)*I(m≥k))
        + ∑_{j=1}^{N-1} w_{-j} * [(1-h[m-j])*I(m≥j) - h[m+j)*I(m≤N-1-j))

where w_k = exp(α(c_k - c_max)) / ∑_m exp(α(c_m - c_max))

This can be computed efficiently:
- For the first sum: for each k, we add (1-h[k:N-k]) to grad[0:N-k] and subtract h[0:N-k] to grad[k:N]
- For the second sum: for each j, we add (1-h[0:N-j]) to grad[j:N] and subtract h[j:N] to grad[0:N-j]

If we do this naively, it's O(N²). But we can vectorize:

grad_pos = zeros(N)
for k in range(N):
    grad_pos[0:N-k] += w[k] * (1 - h[k:N])
    grad_pos[k:N] -= w[k] * h[0:N-k]

grad_neg = zeros(N)
for j in range(1, N):
    grad_neg[j:N] += w[j+N-1] * (1 - h[0:N-j])  # w for shift -j is at index j+N-1 in the full corr
    grad_neg[0:N-j] -= w[j+N-1] * h[j:N]

total_grad = grad_pos + grad_neg

The loop is O(N²) but with N=400, that's 160,000 operations, which is fast.

Actually, I can do even better with FFT-based convolution:
- The first part of grad_pos is a correlation of w with (1-h): 
  grad_pos1[m] = ∑_{k: k≤N-1-m} w[k] * (1-h[m+k])
  This is like a "truncated" correlation.

Let me just use the loop approach for now since N=400 is small enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def compute_lse_and_weights(h, alpha):
    corr = compute_corr(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    lse = c_max + np.log(weights.sum() * weights.max() / weights.max()) / alpha
    # Actually: LSE = c_max + (1/alpha) * log(sum(exp(alpha*(c-cmax))))
    # = c_max + (1/alpha) * log(sum(weights) * ... )
    # Since weights = exp(alpha*(c-cmax)), sum(weights) = sum(exp(alpha*(c-cmax)))
    # LSE = c_max + (1/alpha) * log(sum(weights))
    lse = float(c_max) + float(np.log(weights.sum())) / alpha
    return lse, weights, corr

def lse_gradient(weights, h, alpha):
    """Compute gradient of LSE w.r.t. h using analytic formula."""
    n = len(h)
    c_len = 2 * n - 1
    grad = np.zeros(n)
    # w[k] corresponds to shift k = k_idx - (n-1) where k_idx is 0..2n-2
    for k_idx in range(c_len):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)  # shift
        if k >= 0:
            # c_k = sum_{i=0}^{n-1-k} h[i](1-h[i+k])
            # dck/dh[j] = (1-h[j+k]) for j <= n-1-k, minus h[j-k] for j >= k
            valid1 = n - 1 - k  # max j for first term
            if valid1 >= 0:
                grad[0:valid1+1] += w * (1.0 - h[k:n])
                grad[k:n] -= w * h[0:valid1+1]
        else:
            j = -k  # positive
            # c_{-j} = sum_{i=j}^{n-1} h[i](1-h[i-j])
            # d/dh[m] = (1-h[m-j]) for m >= j, minus h[m+j] for m <= n-1-j
            valid2 = n - 1 - j  # max m for second term
            grad[j:n] += w * (1.0 - h[0:valid2+1])
            grad[0:valid2+1] -= w * h[j:n]
    return grad / alpha

def slsqp_analytic(h, n, target, alphas, maxiter, ftol, deadline):
    bounds = [(0.0, 1.0)] * n
    
    def _obj(x):
        lse, _, _ = compute_lse_and_weights(x, _alpha)
        return lse
    
    def _obj_grad(x):
        _, weights, _ = compute_lse_and_weights(x, _alpha)
        return lse_gradient(weights, x, _alpha)
    
    def _cons_eq(x):
        return np.sum(x) - target
    
    def _cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        if time.time() > deadline:
            break
        _alpha = float(a)
        try:
            res = minimize(_obj, h_cur, jac=_obj_grad, method='SLSQP', 
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def nesterov_pg(h, n, target, alpha, n_steps, lr_init, momentum, deadline):
    """Nesterov accelerated projected gradient descent for LSE."""
    h_cur = h.copy()
    y = h_cur.copy()  # momentum variable
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dt = lr_init
    mu = momentum
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        # Compute gradient at y (Nesterov lookahead)
        _, weights, _ = compute_lse_and_weights(y, alpha)
        grad = lse_gradient(weights, y, alpha)
        
        # Gradient step
        y_new = y - dt * grad
        
        # Projection with momentum
        h_new = (1 - mu) * h_cur + mu * y_new
        h_new = project_feasible(h_new, target)
        
        # Nesterov update
        y = h_new + (mu / (1 - mu)) * (h_new - h_cur) if mu < 1.0 else h_new
        # Actually standard Nesterov: y_{t+1} = h_{t+1} + (mu_{t+1}/mu_t - 1)*(h_{t+1} - h_t)
        # Simplified: use fixed mu
        y = h_new + mu * (h_new - h_cur)
        
        h_cur = h_new
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
        
        # Adaptive step size
        if step % 50 == 49:
            if sc < best_score + 1e-8:
                dt *= 1.1
            else:
                dt *= 0.7
            dt = min(dt, lr_init * 2)
    
    return best_h, best_score

def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    """DE with current-to-best/1/2 strategy."""
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.6 + 0.2, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 7
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 8)
                amp = rng.uniform(0.2, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x / 2 + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.2, target)
            elif r == 4:
                # Periodic pattern
                period = rng.randint(3, n // 3)
                pattern = rng.rand(period) * 0.6 + 0.2
                base = np.tile(pattern, n // period + 1)[:n]
                pop[i] = project_feasible(base, target)
            elif r == 5:
                # Triangular
                tri = 1.0 - np.abs(x - 1.0) / 1.0
                pop[i] = project_feasible(tri * rng.uniform(0.5, 1.5) + rng.randn(n) * 0.1, target)
            else:
                # Block pattern
                n_blocks = rng.randint(2, 8)
                block_h = rng.uniform(0.1, 0.9, n_blocks)
                base = np.repeat(block_h, n // n_blocks + 1)[:n]
                pop[i] = project_feasible(base + rng.randn(n) * 0.05, target)
    
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]
    
    F_adapt = f_factor
    CR_adapt = cr
    
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        # Adaptive parameters
        F_adapt = f_factor * (0.8 + 0.4 * np.exp(-gen / 50.0))
        CR_adapt = min(0.9, cr * (1.0 + 0.3 * np.exp(-gen / 30.0)))
        
        best_idx = int(np.argmin(scores))
        
        for i in range(pop_size):
            if time.time() > deadline:
                break
            if i == best_idx:
                continue
            # current-to-best/1/2
            mask = np.ones(pop_size, dtype=bool)
            mask[i] = False
            mask[best_idx] = False
            avail = np.where(mask)[0]
            if len(avail) < 3:
                break
            a, b, c = rng.choice(avail, size=3, replace=False)
            
            # Current-to-best mutation
            r1 = rng.rand()
            if r1 < 0.5:
                v = pop[best_idx] + F_adapt * (pop[best_idx] - pop[i]) + F_adapt * (pop[a] - pop[b])
            else:
                v = pop[i] + F_adapt * (pop[best_idx] - pop[i]) + F_adapt * (pop[a] - pop[c])
            
            # Binary crossover
            j_rand = rng.randint(n)
            r = rng.rand(n)
            trial = np.where(r < CR_adapt, v, pop[i])
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    
    return best_h, best_score

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        n_corr = len(corr)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top active shifts
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.005)
        weights /= weights.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wk, k_idx in zip(weights, top_idx):
            k = int(k_idx) - (n - 1)
            if k >= 0:
                valid1 = n - 1 - k
                if valid1 >= 0:
                    grad[0:valid1+1] += wk * (1.0 - h_cur[k:n])
                    grad[k:n] -= wk * h_cur[0:valid1+1]
            else:
                j = -k
                valid2 = n - 1 - j
                grad[j:n] += wk * (1.0 - h_cur[0:valid2+1])
                grad[0:valid2+1] -= wk * h_cur[j:n]
        grad *= dx
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, lr_init * 3.0)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        k_star_idx = int(np.argmax(corr))
        k_star = k_star_idx - (n - 1)
        
        grad = np.zeros(n)
        if k_star >= 0:
            valid1 = n - 1 - k_star
            if valid1 >= 0:
                grad[0:valid1+1] += (1.0 - h_cur[k_star:n])
                grad[k_star:n] -= h_cur[0:valid1+1]
        else:
            j = -k_star
            valid2 = n - 1 - j
            if valid2 >= 0:
                grad[j:n] += (1.0 - h_cur[0:valid2+1])
                grad[0:valid2+1] -= h_cur[j:n]
        grad *= dx
        
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.05, 0.03)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:

        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    return project_feasible(h_out, target_dst)


def compute_lse_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)

    if s < 1e-300:

    else:
        weights = weights / s

    return lse, weights

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):

                j = -k
                nj = n - j
                grad[0:nj] += wk * (1.0 - h_cur[j:n])
                grad[j:n] -= wk * h_cur[0:nj]
        grad *= dx
        h_new = h_cur - lr * grad

        h3_parent = interpolate_to(best_h, n3, t3)
        pop3 = [h3_init.copy(), h3_parent.copy()]
        for _ in range(12):
            if rng.rand() < 0.5:
                pop3.append(project_feasible(h3_init + rng.randn(n3) * 0.06, t3))
            else:
                pop3.append(project_feasible(rng.rand(n3) * 0.6 + 0.2, t3))
        h3, sc3 = de_optimize(n3, t3, 14, 200, 0.6, 0.5, min(start_time + 150, hard_deadline), rng, init_pop=pop3)
        h3 = slsqp_analytic(h3, n3, t3, [100, 500, 2000, 8000], 2000, 1e-14, min(start_time + 180, hard_deadline))
        sc3 = compute_c5(h3)
        h3n, sc3n = nesterov_pg(h3, n3, t3, 500.0, 300, 0.01, min(start_time + 210, hard_deadline))
        if sc3n < sc3:
            h3, sc3 = h3n, sc3n
        h3p = slsqp_analytic(h3_parent, n3, t3, [100, 500, 2000], 1500, 1e-14, min(start_time + 220, hard_deadline))
        if compute_c5(h3p) < sc3:
            h3, sc3 = h3p, compute_c5(h3p)

    h_work = interpolate_to(h3, n_final, target_final)
    h_work_parent = interpolate_to(best_h, n_final, target_final)
    sc_work = compute_c5(h_work)
    sc_parent = compute_c5(h_work_parent)
    if sc_work < sc_parent:
        best_candidate = h_work
        best_candidate_score = sc_work
    else:
        best_candidate = h_work_parent
        best_candidate_score = sc_parent
    if best_candidate_score < best_score:
        best_score = best_candidate_score
        best_h = best_candidate.copy()

    if hard_deadline - time.time() > 150:
        pop5 = [best_candidate.copy(), h_work_parent.copy(), h_work.copy()]
        for _ in range(10):
            scale = rng.uniform(0.03, 0.12)
            pop5.append(project_feasible(best_candidate + rng.randn(n_final) * scale, target_final))
        h5, sc5 = de_optimize(n_final, target_final, len(pop5), 60, 0.5, 0.4, min(start_time + 330, hard_deadline), rng, init_pop=pop5)
        if sc5 < best_score:
            best_score = sc5
            best_h = h5.copy()

    if hard_deadline - time.time() > 150:
        h6 = slsqp_analytic(best_h, n_final, target_final, [100, 500, 2000, 8000, 20000], 3000, 1e-16, min(start_time + 500, hard_deadline))
        sc6 = compute_c5(h6)
        if sc6 < best_score:
            best_score = sc6
            best_h = h6.copy()

    if hard_deadline - time.time() > 100:
        h7, sc7 = nesterov_pg(best_h, n_final, target_final, 1000.0, 500, 0.005, min(start_time + 620, hard_deadline))
        if sc7 < best_score:
            best_score = sc7
            best_h = h7.copy()
        h7s = slsqp_analytic(best_h, n_final, target_final, [2000, 8000, 30000], 2000, 1e-16, min(start_time + 660, hard_deadline))
        if compute_c5(h7s) < best_score:
            best_score = compute_c5(h7s)
            best_h = h7s.copy()

    if hard_deadline - time.time() > 80:
        h8, sc8 = subgradient_polish(best_h, n_final, target_final, 500, 4, 0.003, 0.997, min(start_time + 740, hard_deadline))
        if sc8 < best_score:
            best_score = sc8
            best_h = h8.copy()
        h8s = slsqp_analytic(best_h, n_final, target_final, [5000, 20000, 50000], 2000, 1e-17, min(start_time + 780, hard_deadline))
        if compute_c5(h8s) < best_score:
            best_score = compute_c5(h8s)
            best_h = h8s.copy()

    if hard_deadline - time.time() > 60:
        h9, sc9 = worst_lag_surgery(best_h, n_final, target_final, 300, 0.002, min(start_time + 830, hard_deadline))
        if sc9 < best_score:
            best_score = sc9
            best_h = h9.copy()
        h9s = slsqp_analytic(best_h, n_final, target_final, [10000, 30000, 100000], 1500, 1e-17, min(start_time + 870, hard_deadline))
        if compute_c5(h9s) < best_score:
            best_score = compute_c5(h9s)
            best_h = h9s.copy()

    if hard_deadline - time.time() > 80:
        for attempt in range(6):
            if hard_deadline - time.time() < 40:
                break
            scales = [0.03, 0.06, 0.04, 0.08, 0.05, 0.1]
            h_pert = project_feasible(best_h + rng.randn(n_final) * scales[attempt], target_final)
            h_r = slsqp_analytic(h_pert, n_final, target_final, [500, 2000, 8000], 1000, 1e-15, min(start_time + 950, hard_deadline))
            sc_r = compute_c5(h_r)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r.copy()

    if hard_deadline - time.time() > 50:
        h_f, sc_f = subgradient_polish(best_h, n_final, target_final, 300, 5, 0.002, 0.998, min(start_time + 1000, hard_deadline))
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f.copy()

    if hard_deadline - time.time() > 30:
        h_s, sc_s = worst_lag_surgery(best_h, n_final, target_final, 200, 0.001, min(start_time + 1040, hard_deadline))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s.copy()

    if hard_deadline - time.time() > 20:
        h_fs = slsqp_analytic(best_h, n_final, target_final, [10000, 50000, 100000], 1000, 1e-17, hard_deadline)
        sc_fs = compute_c5(h_fs)
        if sc_fs < best_score:
            best_score = sc_fs
            best_h = h_fs.copy()

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
