from scipy.optimize import minimize
import time

_g_n = 0
_g_alpha = 0.0
_g_target = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = compute_corr(h)
    return float(np.max(c)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(x):
    n = _g_n
    alpha = _g_alpha
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    obj = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    weights = np.exp(shifted)
    wsum = float(np.sum(weights))
    weights /= wsum
    
    # Gradient computation (vectorized)
    i_idx = np.arange(n)[:, None].astype(np.int64)
    m_idx = np.arange(2 * n - 1)[None, :].astype(np.int64)
    
    j1 = i_idx - m_idx + (n - 1)
    v1 = (j1 >= 0) & (j1 < n)
    j2 = i_idx + m_idx - (n - 1)
    v2 = (j2 >= 0) & (j2 < n)
    
    h1 = np.where(v1, x[np.clip(j1, 0, n - 1)], 0.0)
    h2 = np.where(v2, x[np.clip(j2, 0, n - 1)], 0.0)
    
    term1 = np.sum(v1 * (1.0 - h1) * weights[None, :], axis=1)
    term2 = np.sum(v2 * h2 * weights[None, :], axis=1)
    grad = term1 - term2
    
    return obj, grad

def _lse_obj(x):
    return lse_obj_and_grad(x)[0]

def _lse_grad(x):
    return lse_obj_and_grad(x)[1]

def _cons_fun(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def slsqp_lse(h0, target, alpha, maxiter=2000, ftol=1e-18):

    _g_n = len(h0)
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * _g_n
    constraints = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_lse_obj, h0, jac=_lse_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h0.copy()

    h_cur = h.copy()

    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)
    stall_count = 0
