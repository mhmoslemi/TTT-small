def binary_sa(h_init, n_steps, t_start, t_end, seed):
    n = len(h_init)
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    def compute_f(h):
        # f[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k])
        f = np.zeros(n)
        for k in range(n):
            f[k] = np.sum(h[:n-k] * (1-h[k:]))
        return f
    
    f = compute_f(h)
    E = np.max(f)
    best_E = E
    
    one_list = list(np.where(h == 1)[0])
    zero_list = list(np.where(h == 0)[0])
    
    for step in range(n_steps):
        t = t_start * (t_end/t_start) ** (step/n_steps)
        if t < t_end:
            t = t_end
        
        # Choose random swap
        i = one_list[rng.randint(len(one_list))]
        j = zero_list[rng.randint(len(zero_list))]
        
        # Compute Δf for all k
        k_all = np.arange(n)
        df = np.zeros(n)
        
        # Term 1: -(1-h[i+k]) for k < n-i
        mask1 = k_all < (n - i)
        df[mask1] -= (1 - h[i + k_all[mask1]])
        # Special: if i+k = j, i.e., k = j-i
        if j > i:
            ksp = j - i
            if ksp < n:
                # Already handled: h[i+ksp] = h[j] = 0, so -(1-0) = -1 ✓
                pass
        
        # Term 2: (h[i-k] + I(i-k=j)) for k ≤ i
        mask2 = k_all <= i
        idx2 = i - k_all[mask2]
        df[mask2] += h[idx2]
        if i > j:
            ksp = i - j
            df[ksp] += 1  # Because i-k=j, so h'[i-k] = 1 instead of h[j]=0
        
        # Term 3: (1-h[j+k] + I(j+k=i)) for k < n-j
        mask3 = k_all < (n - j)
        df[mask3] += (1 - h[j + k_all[mask3]])
        if i > j:
            ksp = i - j
            if ksp < n:
                df[ksp] += 1  # Because j+k=i, h'[j+k]=0, so 1-0=1 instead of 1-h[i]=0
        
        # Term 4: (-h[j-k] + I(j-k=i)) for k ≤ j
        mask4 = k_all <= j
        idx4 = j - k_all[mask4]
        df[mask4] -= h[idx4]
        if j > i:
            ksp = j - i
            df[ksp] += 1  # Because j-k=i, h'[j-k]=0, so -0=0 instead of -h[i]=-1
        
        new_f = f + df
        new_E = np.max(new_f)
        
        # Accept?
        if new_E <= E or rng.random() < np.exp(-(new_E - E) / t):
            # Apply swap
            h[i] = 0
            h[j] = 1
            one_list.remove(i)
            one_list.append(j)
            zero_list.remove(j)
            zero_list.append(i)
            f = new_f
            E = new_E
            if E < best_E:
                best_E = E
                best_h = h.copy()
    
    return best_h, best_E

import numpy as np
from scipy.optimize import minimize
import time

# Global variables for SLSQP callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_f_vector(h):
    """Compute f[k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) for all k."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n]  # lags 0 to n-1

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    over = corr[_g_n-1:]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -50.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

# Binary SA
def binary_sa(h_bin, n_steps, t_start, t_end, seed, max_restarts=1):
    n = len(h_bin)
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    best_h = h.copy()
    f = compute_f_vector(h)
    E = float(np.max(f))
    best_E = E
    
    one_pos = list(np.where(h == 1)[0])
    zero_pos = list(np.where(h == 0)[0])
    
    for restart in range(max_restarts):
        # Cool down
        for step in range(n_steps):
            frac = step / n_steps
            t = t_start * (t_end / t_start) ** frac
            if t < t_end:
                t = t_end
            
            # Random swap
            i = one_pos[rng.randint(len(one_pos))]
            j = zero_pos[rng.randint(len(zero_pos))]
            if i == j:
                continue
            
            # Compute Δf
            k_all = np.arange(n)
            df = np.zeros(n)
            
            # Term 1: -(1-h[i+k]) for k < n-i
            m1 = k_all < (n - i)
            if m1.any():
                df[m1] -= (1.0 - h[i + k_all[m1]])
            
            # Term 2: h[i-k] for k ≤ i (plus special case)
            m2 = k_all <= i
            if m2.any():
                idx2 = i - k_all[m2]
                df[m2] += h[idx2]
            if i > j:
                ksp = i - j
                df[ksp] += 1.0
            
            # Term 3: (1-h[j+k]) for k < n-j (plus special case)
            m3 = k_all < (n - j)
            if m3.any():
                df[m3] += (1.0 - h[j + k_all[m3]])
            if i > j:
                ksp = i - j
                if ksp < n:
                    df[ksp] += 1.0
            
            # Term 4: -h[j-k] for k ≤ j (plus special case)
            m4 = k_all <= j
            if m4.any():
                idx4 = j - k_all[m4]
                df[m4] -= h[idx4]
            if j > i:
                ksp = j - i
                df[ksp] += 1.0
            
            new_E = float(np.max(f + df))
            
            if new_E <= E or rng.random() < np.exp(-(new_E - E) / max(t, 1e-10)):
                h[i] = 0.0
                h[j] = 1.0
                one_pos.remove(i)
                one_pos.append(j)
                zero_pos.remove(j)
                zero_pos.append(i)
                f = f + df
                E = new_E
                if E < best_E:
                    best_E = E
                    best_h = h.copy()
        
        # Reheat: go back to best and increase temperature
        h = best_h.copy()
        f = compute_f_vector(h)
        E = float(np.max(f))
        one_pos = list(np.where(h == 1)[0])
        zero_pos = list(np.where(h == 0)[0])
        t_start = max(t_end * 10, 0.5)
    
    return best_h, best_E

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= dx * h_cur[idx2 - k]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.015, 0.004)
        else:
            eps *= 0.85
            if eps < 1e-8:
                break
    return h_cur, cur_score

def tabu_search_binary(h_bin, n_iter, tabu_len, rng):
    n = len(h_bin)
    h = h_bin.copy()
    best_h = h.copy()
    f = compute_f_vector(h)
    E = float(np.max(f))
    best_E = E
    one_pos = set(np.where(h == 1)[0])
    zero_pos = set(np.where(h == 0)[0])
    tabu_i = {}
    tabu_j = {}
    k_all = np.arange(n)
    for it in range(n_iter):
        candidates = []
        n_trials = 30
        trial_ones = list(one_pos)[:min(50, len(one_pos))]
        trial_zeros = list(zero_pos)[:min(50, len(zero_pos))]
        for _ in range(n_trials):
            i = trial_ones[rng.randint(len(trial_ones))]
            j = trial_zeros[rng.randint(len(trial_zeros))]
            if i == j:
                continue
            if tabu_i.get(i, -1) >= it - tabu_len:
                continue
            if tabu_j.get(j, -1) >= it - tabu_len:
                continue
            df = np.zeros(n)
            m1 = k_all < (n - i)
            if m1.any():
                df[m1] -= (1.0 - h[i + k_all[m1]])
            m2 = k_all <= i
            if m2.any():
                idx2 = i - k_all[m2]
                df[m2] += h[idx2]
            if i > j:
                ksp = i - j
                if ksp < n:
                    df[ksp] += 1.0
            m3 = k_all < (n - j)
            if m3.any():
                df[m3] += (1.0 - h[j + k_all[m3]])
            if i > j:
                ksp = i - j
                if ksp < n:
                    df[ksp] += 1.0
            m4 = k_all <= j
            if m4.any():
                idx4 = j - k_all[m4]
                df[m4] -= h[idx4]
            if j > i:
                ksp = j - i
                if ksp < n:
                    df[ksp] += 1.0
            new_E = float(np.max(f + df))
            candidates.append((new_E, i, j, df))
        if not candidates:
            continue
        candidates.sort(key=lambda x: x[0])
        best_new_E, bi, bj, bdf = candidates[0]
        if best_new_E < best_E:
            h[bi] = 0.0
            h[bj] = 1.0
            one_pos.remove(bi)
            one_pos.add(bj)
            zero_pos.remove(bj)
            zero_pos.add(bi)
            f = f + bdf
            E = best_new_E
            best_E = E
            best_h = h.copy()
            tabu_i[bi] = it
            tabu_j[bj] = it
        else:
            break
    return best_h, best_E

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Quick SLSQP warm-up
    if tleft() > 150:
        h_ref, sc = slsqp_optimize(best_h, [200, 1000], 2000, 1e-14)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Binary SA - main exploration
    if tleft() > 350:
        h_bin = (best_h >= 0.5).astype(float)
        h_bin = project_feasible(h_bin, target)
        h_sa, E_sa = binary_sa(h_bin, 150000, 4.0, 0.01, seed, max_restarts=2)
        sc_sa = E_sa * (2.0 / n)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    # Phase 3: SLSQP refinement
    if tleft() > 250:
        h_ref, sc = slsqp_optimize(best_h, [2000, 10000], 2500, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Tabu search on binary
    if tleft() > 250:
        h_bin2 = (best_h >= 0.5).astype(float)
        h_bin2 = project_feasible(h_bin2, target)
        rng_tabu = np.random.RandomState(seed + 100)
        h_tab, E_tab = tabu_search_binary(h_bin2, 5000, 50, rng_tabu)
        sc_tab = E_tab * (2.0 / n)
        if sc_tab < best_score:
            best_score = sc_tab
            best_h = h_tab
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Second SA with different params
    if tleft() > 300:
        h_bin3 = (best_h >= 0.45).astype(float)
        diff3 = target - np.sum(h_bin3)
        if abs(diff3) > 1e-6:
            sorted_idx = np.argsort(h_bin3)
            if diff3 > 0:
                h_bin3[sorted_idx[-int(diff3):]] = 1.0
            else:
                h_bin3[sorted_idx[:int(-diff3)]] = 0.0
        h_bin3 = project_feasible(h_bin3, target)
        h_sa3, E_sa3 = binary_sa(h_bin3, 200000, 3.0, 0.005, seed + 200, max_restarts=1)
        sc_sa3 = E_sa3 * (2.0 / n)
        if sc_sa3 < best_score:
            best_score = sc_sa3
            best_h = h_sa3
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000, 100000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Subgradient refinement
    if tleft() > 150:
        h_sg, sc_sg = subgradient_momentum(best_h, 400, 5, 0.002, 0.998, 0.4, 0.003)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref, sc = slsqp_optimize(best_h, [10000, 50000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Surgery passes
    if tleft() > 100:
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=200, initial_epsilon=0.0006, n_lags=3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, max_steps=200, initial_epsilon=0.0008)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2
        h_ref, sc = slsqp_optimize(best_h, [20000, 100000], 1500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 60:
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000], 1000, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final tiny surgery
    if tleft() > 30:
        h_surg, sc_surg = worst_lag_surgery(best_h, max_steps=80, initial_epsilon=0.0004)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = top_lag_surgery(best_h, max_steps=60, initial_epsilon=0.0003, n_lags=2)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    # Phase 10: One last SLSQP
    if tleft() > 15:
        h_ref, sc = slsqp_optimize(best_h, [100000], 500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
