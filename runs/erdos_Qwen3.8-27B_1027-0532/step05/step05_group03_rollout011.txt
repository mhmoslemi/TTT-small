def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:])) * dx

def compute_overlaps(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:]  # lags 0 to n-1

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, alphas, maxiter, ftol):

    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    for a in alphas:
        alpha = float(a)
        
        def obj(x):
            hh = project_feasible(x, target)
            corr = np.correlate(hh, 1.0 - hh, mode='full')
            c_max = float(np.max(corr[n-1:]))
            shifted = alpha * (corr[n-1:] - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:

                cur_score = sc
        except Exception:
            pass
    
    return h_cur, cur_score
