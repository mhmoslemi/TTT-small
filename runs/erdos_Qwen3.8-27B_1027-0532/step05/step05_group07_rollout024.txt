    dx = 2.0 / n

    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    """Compute gradient of soft-max correlation."""
    grad = np.zeros(n)
    one_h = 1.0 - h
    for m_idx in range(len(weights)):
        w_val = weights[m_idx]
        if w_val < 1e-12:
            continue
        m = m_idx - (n - 1)
        # (1-h[i+m]) term: i in [max(0,-m), min(n-1, n-1-m)]
        i_lo = max(0, -m)
        i_hi = min(n - 1, n - 1 - m)
        if i_lo <= i_hi:
            grad[i_lo:i_hi+1] += w_val * one_h[i_lo + m : i_hi + m + 1]
        # h[i-m] term: i in [max(0,m), min(n-1, m+n-1)]
        i_lo2 = max(0, m)
        i_hi2 = min(n - 1, m + n - 1)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2+1] -= w_val * h[i_lo2 - m : i_hi2 - m + 1]

                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=50):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    pair_positions = np.arange(n)

    for sweep in range(50000):
        if time.time() > deadline:
            break

        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        rng.shuffle(pair_positions)

        for p in range(n // 2):
            i = int(pair_positions[2 * p])
            j = int(pair_positions[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            nl1 = 2 * n - 1
            C = np.zeros(nl1)
            for k in range(nl1):
                m = k - (n - 1)
                di = (1.0 - h_cur[i + m]) if 0 <= i + m < n else 0.0
                dj = (1.0 - h_cur[j + m]) if 0 <= j + m < n else 0.0
                di2 = -h_cur[i - m] if 0 <= i - m < n else 0.0
                dj2 = -h_cur[j - m] if 0 <= j - m < n else 0.0
                C[k] = (di + di2) - (dj + dj2)

            B_i = np.zeros(nl1)
            B_j = np.zeros(nl1)
            for k in range(nl1):
                m = k - (n - 1)
                B_i[k] = (1.0 - h_cur[i + m]) if 0 <= i + m < n else 0.0
                B_i[k] += (-h_cur[i - m]) if 0 <= i - m < n else 0.0
                B_j[k] = (1.0 - h_cur[j + m]) if 0 <= j + m < n else 0.0
                B_j[k] += (-h_cur[j - m]) if 0 <= j - m < n else 0.0

            A = corr - h_cur[i] * B_i - h_cur[j] * B_j + B_j * c

            lo = 0.0
            hi = float(np.max(A + C * t_max)) if np.any(C > 0) else float(np.max(A))
            if not np.any(C > 0):
                hi = float(np.max(A + C * t_max))

            for _ in range(40):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                if np.any(~(pos | neg)) and np.max(A[~(pos | neg)]) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    ub = min(ub, float(np.min((mid - A[pos]) / C[pos])))
                lb = t_min
                if np.any(neg):
                    lb = max(lb, float(np.max((mid - A[neg]) / C[neg])))
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            old_i = h_cur[i]
            delta = t_star - old_i
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score >= best_score and sweep > 200:
            if sweep % 500 == 0:
                pert = rng.randn(n) * 0.01
                h_cur = project_feasible(h_cur + pert, target)
                corr = compute_corr(h_cur)

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Adam optimization (main workhorse)
    dl = start_time + (budget_s - 25.0) * 0.25
    h1, s1 = adam_optimize(best_h, n, target, dl, rng,
                           alpha_start=50, alpha_end=5000,
                           lr_start=0.1, lr_end=0.005, max_iters=15000)
    if s1 < best_score:
        best_score = s1
        best_h = h1

    # Phase 2: Paired coordinate descent
    if tleft() > 200:
        dl = start_time + (budget_s - 25.0) * 0.42
        h2, s2 = paired_coord_desc(best_h, n, target, dl, rng, recompute_interval=40)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Multi-resolution coarsening
    if tleft() > 300:
        n_coarse = max(n // 4, 50)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        dl_c = start_time + (budget_s - 25.0) * 0.55
        h_coarse, s_coarse = adam_optimize(h_coarse, n_coarse, target_coarse, dl_c, rng,
                                           alpha_start=30, alpha_end=2000,
                                           lr_start=0.12, lr_end=0.005, max_iters=8000)
        h_up = interpolate_to(h_coarse, n, target)
        h_up, s_up = adam_optimize(h_up, n, target,
                                   start_time + (budget_s - 25.0) * 0.62, rng,
                                   alpha_start=200, alpha_end=8000,
                                   lr_start=0.05, lr_end=0.003, max_iters=8000)
        if s_up < best_score:
            best_score = s_up
            best_h = h_up

    # Phase 4: Top-k focused Adam
    if tleft() > 250:
        dl = start_time + (budget_s - 25.0) * 0.72
        h4, s4 = adam_optimize(best_h, n, target, dl, rng,
                               alpha_start=500, alpha_end=20000,
                               lr_start=0.06, lr_end=0.002,
                               max_iters=10000, top_k=5)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: More paired coordinate descent
    if tleft() > 200:
        dl = start_time + (budget_s - 25.0) * 0.80
        h5, s5 = paired_coord_desc(best_h, n, target, dl, rng, recompute_interval=30)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: SLSQP polish
    if tleft() > 100:
        h6 = slsqp_polish(best_h, n, target, [200, 1000, 5000, 15000], 2000, 1e-16)
        s6 = compute_c5(h6)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Random restarts with Adam
    if tleft() > 150:
        noise_scales = [0.03, 0.05, 0.02, 0.08, 0.04, 0.06, 0.035, 0.045]
        for attempt in range(len(noise_scales)):
            if tleft() < 80:
                break
            h_pert = best_h + rng.randn(n) * noise_scales[attempt]
            h_pert = project_feasible(h_pert, target)
            dl_r = start_time + (budget_s - 25.0) * (0.85 + 0.012 * attempt)
            hr, sr = adam_optimize(h_pert, n, target, dl_r, rng,
                                   alpha_start=100, alpha_end=10000,
                                   lr_start=0.08, lr_end=0.003, max_iters=5000)
            if sr < best_score:
                best_score = sr
                best_h = hr

    # Phase 8: Final paired CD + SLSQP
    if tleft() > 60:
        h8, s8 = paired_coord_desc(best_h, n, target,
                                   start_time + budget_s - 40.0, rng, recompute_interval=20)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    if tleft() > 30:
        h9 = slsqp_polish(best_h, n, target, [5000, 20000], 1500, 1e-17)
        s9 = compute_c5(h9)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Final Adam micro-polish
    if tleft() > 20:
        h10, s10 = adam_optimize(best_h, n, target, start_time + budget_s - 15.0, rng,
                                 alpha_start=5000, alpha_end=50000,
                                 lr_start=0.02, lr_end=0.001, max_iters=3000, top_k=3)
        if s10 < best_score:
            best_score = s10
            best_h = h10

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
