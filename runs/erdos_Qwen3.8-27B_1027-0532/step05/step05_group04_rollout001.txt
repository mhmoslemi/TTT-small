import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def compute_overlaps(h):
    """Return the overlap values for lags 0 to n-1."""
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:n-1+n]

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_gradient(h, weights, dx):
    """Compute gradient of sum_k weights[k] * overlap[k]."""
    n = len(h)
    grad = np.zeros(n)
    for k in range(n):
        wk = weights[k]
        if wk < 1e-12:
            continue
        j_pos = np.arange(0, n - k)
        grad[j_pos] += wk * (1.0 - h[j_pos + k])
        j_neg = np.arange(k, n)
        grad[j_neg] -= wk * h[j_neg - k]
    return grad * dx

def lse_objective(h, alpha, target_sum):

    shifted = alpha * (over - c_max)

def lse_gradient(h, alpha, target_sum):

def smooth_max_objective(h, alpha, dx):

        weights[top_idx] = w_top

        h_new = h_cur + v
        h_new = project_feasible(h_new, target)

            v *= 0.5
            if lr < 1e-8:
                lr = 1e-8
        
        # Update beta
        beta = min(beta * beta_growth, 5000.0)
        
    return best, best_score

    return best, best_sc


def slsqp_sweep(h, n, target, alphas, deadline=None):
    h_cur = h.copy()
    cur_sc = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        h_new = slsqp_call(h_cur, n, target, a, 3000, 1e-17)
        sc = compute_c5(h_new)
        if sc < cur_sc:
            h_cur = h_new
            cur_sc = sc
    return h_cur


def multi_restart(h, n, target, n_trials, scale, deadline=None):
    best = h.copy()
    best_sc = compute_c5(h)
    for i in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        delta = np.random.randn(n) * scale
        h_pert = project_feasible(h + delta, target)
        h_opt = slsqp_call(h_pert, n, target, 1000.0, 2000, 1e-16)
        sc = compute_c5(h_opt)
        if sc < best_sc:
            best_sc = sc
            best = h_opt
    return best, best_sc


def coarse_fine(h_fine, n_fine, target_fine, n_coarse, deadline=None):
    x_old = np.linspace(0.0, 1.0, n_fine)
    x_new = np.linspace(0.0, 1.0, n_coarse)
    h_coarse = np.interp(x_new, x_old, h_fine)
    h_coarse = project_feasible(h_coarse, n_coarse / 2.0)
    sc_coarse = compute_c5(h_coarse)
    h_coarse = slsqp_sweep(h_coarse, n_coarse, n_coarse / 2.0,
                           [100, 500, 2000, 8000], deadline=deadline)
    sc_coarse = compute_c5(h_coarse)
    x_old2 = np.linspace(0.0, 1.0, n_coarse)
    x_new2 = np.linspace(0.0, 1.0, n_fine)
    h_up = np.interp(x_new2, x_old2, h_coarse)
    h_up = project_feasible(h_up, target_fine)
    sc_up = compute_c5(h_up)
    if sc_up < compute_c5(h_fine):
        return h_up
    return h_fine


def equalize_lags(h, n, target, n_steps, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    best = h_cur.copy()
    best_sc = compute_c5(h_cur)
    lr = 0.002
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        over = compute_overlaps(h_cur)
        mx = float(np.max(over))
        active = np.where(over > mx - 0.02 * (over.max() - over.min() + 1e-10))[0]
        if len(active) == 0:
            active = np.argmax(over)
            active = np.array([active])
        w = np.zeros(n)
        ua = np.ones(len(active))
        w[active] = ua / len(active)
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            jp = np.arange(0, n - k)
            grad[jp] += (1.0 - h_cur[jp + k])
            jn = np.arange(k, n)
            grad[jn] -= h_cur[jn - k]
        grad *= dx / len(active)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_sc:
            best_sc = sc
            best = h_new.copy()
        if sc < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.002, 0.01)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return best, best_sc


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_sc = compute_c5(best_h)

    def tleft():
        return budget_s - 40.0 - (time.time() - start_time)

    # Phase 1: Initial SLSQP sweep (90s)
    dl = start_time + 90
    h1 = slsqp_sweep(h_init, n, target, [50, 200, 800, 3000], deadline=dl)
    sc = compute_c5(h1)
    if sc < best_sc:
        best_sc = sc
        best_h = h1

    # Phase 2: Coarse grid refinement (90s)
    if tleft() > 100:
        dl = start_time + 180
        h2 = coarse_fine(best_h, n, target, 100, deadline=dl)
        sc = compute_c5(h2)
        if sc < best_sc:
            best_sc = sc
            best_h = h2

    # Phase 3: Main reweighted gradient (240s)
    if tleft() > 250:
        dl = start_time + 390
        h3, sc3 = rw_pg(best_h, n, target, n_steps=5000, beta_init=30.0,
                        beta_growth=1.015, lr_init=0.0015,
                        m_init=3, m_max=25, deadline=dl)
        if sc3 < best_sc:
            best_sc = sc3
            best_h = h3
        h3p = slsqp_sweep(best_h, n, target, [500, 2000, 8000], deadline=dl)
        sc = compute_c5(h3p)
        if sc < best_sc:
            best_sc = sc
            best_h = h3p

    # Phase 4: Equalize lags (120s)
    if tleft() > 130:
        dl = start_time + 510
        h4, sc4 = equalize_lags(best_h, n, target, n_steps=2000, deadline=dl)
        if sc4 < best_sc:
            best_sc = sc4
            best_h = h4
        h4p = slsqp_sweep(best_h, n, target, [2000, 8000], deadline=dl)
        sc = compute_c5(h4p)
        if sc < best_sc:
            best_sc = sc
            best_h = h4p

    # Phase 5: Multi-restart (150s)
    if tleft() > 160:
        dl = start_time + 660
        h5, sc5 = multi_restart(best_h, n, target, n_trials=8, scale=0.03, deadline=dl)
        if sc5 < best_sc:
            best_sc = sc5
            best_h = h5

    # Phase 6: Second reweighted pass - sharper (200s)
    if tleft() > 210:
        dl = start_time + 860
        h6, sc6 = rw_pg(best_h, n, target, n_steps=4000, beta_init=200.0,
                        beta_growth=1.02, lr_init=0.0008,
                        m_init=2, m_max=15, deadline=dl)
        if sc6 < best_sc:
            best_sc = sc6
            best_h = h6
        h6p = slsqp_sweep(best_h, n, target, [2000, 8000, 20000], deadline=dl)
        sc = compute_c5(h6p)
        if sc < best_sc:
            best_sc = sc
            best_h = h6p

    # Phase 7: Final SLSQP with very high alpha (120s)
    if tleft() > 130:
        dl = start_time + budget_s - 80
        h7 = slsqp_sweep(best_h, n, target, [5000, 15000, 50000], deadline=dl)
        sc = compute_c5(h7)
        if sc < best_sc:
            best_sc = sc
            best_h = h7

    # Phase 8: Final micro-polish (60s)
    if tleft() > 70:
        dl = start_time + budget_s - 30
        h8, sc8 = rw_pg(best_h, n, target, n_steps=1000, beta_init=2000.0,
                        beta_growth=1.01, lr_init=0.0003,
                        m_init=1, m_max=5, deadline=dl)
        if sc8 < best_sc:
            best_sc = sc8
            best_h = h8
        h8p = slsqp_call(best_h, n, target, 50000.0, 5000, 1e-18)
        sc = compute_c5(h8p)
        if sc < best_sc:
            best_sc = sc
            best_h = h8p

    h_final = project_feasible(best_h, target)
    final_sc = compute_c5(h_final)
    if final_sc > best_sc:
        h_final = best_h
        final_sc = best_sc
    return h_final, final_sc, n
