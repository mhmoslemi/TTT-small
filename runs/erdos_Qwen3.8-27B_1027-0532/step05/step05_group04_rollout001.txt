For the Erdős problem, the optimal h is believed to be related to the structure of the "Erdős-Rényi" sequence or similar combinatorial objects. The function h should be roughly 1/2 on average (since integral = 1 over [0,2]), so about half the points should be 1 and half 0.

The goal is to make h(x) and h(x+k) "anti-correlated" for all k simultaneously - when h is 1, h(x+k) should be 0, and vice versa. This is like a "perfect" binary code with no autocorrelation peaks.

For binary sequences, this is related to the "minimum envelope" of the autocorrelation. The theoretical minimum for a binary sequence of length n with n/2 ones is related to n/4 (I think), but let's not get into theory and just optimize.

from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:2*n-1]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_overlaps(h):
    n = len(h)
    corr = compute_corr(h)
    return corr[n-1:2*n-1]

def to_binary(h, n, target):
    """Convert fractional h to binary with exact sum."""
    h_bin = (h >= 0.5).astype(float)
    surplus = target - h_bin.sum()
    if surplus > 0:
        # Need to flip some 0s to 1s
        zeros = np.where(h_bin == 0)[0]
        # Pick the ones with highest h values (closest to 0.5 from above)
        order = np.argsort(-h[zeros])[:int(surplus)]
        h_bin[zeros[order]] = 1.0
    elif surplus < 0:
        # Need to flip some 1s to 0s
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])[:int(-surplus)]
        h_bin[ones[order]] = 0.0
    return h_bin

def build_effects(h, n):
    """Build effect vectors for all zeros and ones."""
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h >= 0.5)[0]
    nz = len(zeros)
    no = len(ones)
    
    eff_plus = np.zeros((nz, n))
    for idx, i in enumerate(zeros):
        i = int(i)
        # (1-h[i+k]) for k=0 to n-1-i
        max_k = n - 1 - i
        if max_k >= 0:
            eff_plus[idx, 0:max_k+1] = 1.0 - h[i:i+max_k+1]
        # -h[i-k] for k=1 to i (k=0 gives -h[i]=0 since h[i]<0.5, approximately 0)
        # Actually for binary: h[i]=0, so -h[i]=0 for k=0
        if i >= 1:
            eff_plus[idx, 1:i+1] = -h[i-1::-1]
    
    eff_minus = np.zeros((no, n))
    for idx, j in enumerate(ones):
        j = int(j)
        # -(1-h[j+k]) for k=0 to n-1-j
        max_k = n - 1 - j
        if max_k >= 0:
            eff_minus[idx, 0:max_k+1] = -(1.0 - h[j:j+max_k+1])
        # +h[j-k] for k=0 to j
        eff_minus[idx, 0:j+1] = h[j::-1]
    
    return zeros, ones, eff_plus, eff_minus

def greedy_binary_search(h_bin, n, target, max_rounds=50, deadline=None):
    """Greedy best-swap local search on binary vectors."""
    h = h_bin.copy()
    overlap = get_overlaps(h)
    cur_max = float(np.max(overlap))
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        
        zeros, ones, eff_plus, eff_minus = build_effects(h, n)
        nz = len(zeros)
        no = len(ones)
        
        if nz == 0 or no == 0:
            break
        
        # Find best swap
        best_score = cur_max
        best_a, best_b = -1, -1
        
        # Process in batches to limit memory
        batch = 30
        for start in range(0, nz, batch):
            end = min(start + batch, nz)
            bp = eff_plus[start:end]  # (batch, n)
            # combined[a, b, k] = overlap[k] + bp[a, k] + eff_minus[b, k]
            # We want min over (a,b) of max over k
            combined = overlap[None, None, :] + bp[:, None, :] + eff_minus[None, :, :]
            scores = np.max(combined, axis=2)  # (batch, no)
            local_best = np.unravel_index(np.argmin(scores), scores.shape)
            local_min = scores[local_best]
            if local_min < best_score:
                best_score = local_min
                best_a = start + local_best[0]
                best_b = local_best[1]
        
        if best_a < 0 or best_score >= cur_max - 1e-14:
            break
        
        i = zeros[best_a]
        j = ones[best_b]
        h[i] = 1.0
        h[j] = 0.0
        overlap = get_overlaps(h)
        cur_max = float(np.max(overlap))
    
    return h, cur_max

def simulated_annealing_binary(h_bin, n, target, n_steps=5000, T_init=0.5, T_final=0.01, deadline=None):
    """Simulated annealing on binary vectors using swaps."""
    h = h_bin.copy()
    overlap = get_overlaps(h)
    cur_max = float(np.max(overlap))
    best_h = h.copy()
    best_max = cur_max
    dx = 2.0 / n
    
    rng = np.random.RandomState(42)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        
        T = T_init * (T_final / T_init) ** (step / max(n_steps, 1))
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # Pick random swap
        a = rng.randint(len(zeros))
        b = rng.randint(len(ones))
        i = zeros[a]
        j = ones[b]
        
        # Compute effect
        # (1-h[i+k]) for k=0 to n-1-i
        max_k_i = n - 1 - i
        max_k_j = n - 1 - j

        if max_k_i >= 0:
            delta[0:max_k_i+1] += (1.0 - h[i:i+max_k_i+1])
        if i >= 1:
            delta[1:i+1] -= h[i-1::-1]
        if max_k_j >= 0:
            delta[0:max_k_j+1] -= (1.0 - h[j:j+max_k_j+1])
        if j >= 0:
            delta[0:j+1] += h[j::-1]
        
        new_overlap = overlap + delta
        new_max = float(np.max(new_overlap))
        
        if new_max < cur_max or (T > 0.001 and rng.rand() < np.exp(-(new_max - cur_max) / T)):
            # Accept
            h[i] = 1.0
            h[j] = 0.0
            overlap = new_overlap
            cur_max = new_max
            # Update zeros and ones
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
            
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
    
    return best_h, best_max

def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[n-1:2*n-1]
        c_max = float(np.max(over))
        alpha = _slsqp_alpha
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    def cons_eq(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    
    global _slsqp_alpha
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def weighted_grad_descent(h, n, target, n_steps=300, beta=200, lr=0.001, deadline=None):

        h_new = h_cur - lr * grad

        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:

def project_feasible(h, target_sum):

def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _slsqp_alpha
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        over = corr[n - 1:2 * n - 1]
        c_max = float(np.max(over))
        alpha = _slsqp_alpha
        shifted = alpha * (over - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_eq(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _slsqp_alpha = a
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def wgd(h, n, target, n_steps, beta, lr_init, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        overlap = get_overlaps(h_cur)
        max_ov = float(np.max(overlap))
        weights = np.exp(beta * (overlap - max_ov))
        weights /= weights.sum()
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            lr = max(lr * 0.92, 1e-6)
    return h_cur, cur_score
