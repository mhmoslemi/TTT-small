Let me think about a **beam search / steepest descent** approach:

At each step:
1. Compute full correlation (FFT)
2. Identify the top M worst shifts
3. For each worst shift, identify the "best" swaps that would reduce it
4. Evaluate all candidate swaps using the delta formula (O(1) per candidate)
5. Pick the best swap
6. Update

The delta formula for a swap (i: 1→0, j: 0→1) at shift k is:
Δc[k] = -(1-h[i+k]) + (1-h[j+k]) + h[i-k] - h[j-k]

This is O(1) per (swap, shift) pair! So we can evaluate many swaps against many shifts very fast.

The key insight: we don't need to recompute the full correlation after each swap. We just need to:
1. Know which shifts are "near the max" (from the previous full computation)
2. For each candidate swap, compute the delta at those near-max shifts
3. If the new max would be lower, accept the swap
4. Periodically (every K moves) recompute the full correlation to update the "near-max" set

This is much faster than recomputing FFT every step!

For a given set of near-max shifts K = {k1, k2, ..., km}, a swap (i, j) has delta:

1. For each i in ones, compute a "cost vector" v_i = [h[i+k1]+h[i-k1], h[i+k2]+h[i-k2], ..., h[i+km]+h[i-km]]
2. For each j in zeros, compute a "benefit vector" w_j = [h[j+k1]+h[j-k1], ..., h[j+km]+h[j-km]]
3. The new max for swap (i,j) is max_k (corr[k] + v_i[k] - w_j[k])
4. We want to minimize this over (i,j)

This is still O(n*m) for computing the vectors, and O(|ones|*|zeros|*m) for the full search. But we can approximate:
- Pick the top 10 i's with smallest v_i (component-wise or by sum)
- Pick the top 10 j's with largest w_j

**Targeted swap strategy**:
1. Compute the full correlation
2. Find the worst shift k*
3. For shift k*, a swap (i:1→0, j:0→1) changes c[k*] by:
   Δc[k*] = h[i+k*] + h[i-k*] - h[j+k*] - h[j-k*]
4. We want Δc[k*] < 0, which means h[i+k*] + h[i-k*] < h[j+k*] + h[j-k*]
5. So: pick i from ones where h[i+k*] + h[i-k*] is small (preferably 0), and j from zeros where h[j+k*] + h[j-k*] is large (preferably 2)

Let me precompute for the current worst shift:
- For each one i: score_i = h[i+k*] + h[i-k*] (0, 1, or 2)
- For each zero j: score_j = h[j+k*] + h[j-k*] (0, 1, or 2)
- Best swap: i with score_i=0, j with score_j=2 → Δc[k*] = -2

If no such pair exists, try score_i=0, score_j=1 → Δc[k*] = -1, etc.

This is a heuristic that biases moves toward the worst shift. Combined with occasional random moves, it should be effective.

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    cur = int(h_r.sum())
    if cur > target_sum:
        excess = cur - target_sum
        oi = np.where(h_r == 1)[0]
        order = np.argsort(h[oi])[:excess]
        h_r[oi[order]] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zi = np.where(h_r == 0)[0]
        order = np.argsort(-h[zi])[:deficit]
        h_r[zi[order]] = 1
    # Fix any remaining
    cur = int(h_r.sum())
    while cur < target_sum:
        z = np.where(h_r == 0)[0]
        h_r[z[0]] = 1
        cur += 1
    while cur > target_sum:
        o = np.where(h_r == 1)[0]
        h_r[o[0]] = 0
        cur -= 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def compute_c5_direct(h):
    n = len(h)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).clip(0, 1).astype(int)
    cur = int(h_r.sum())
    if cur > target_sum:
        excess = cur - target_sum
        oi = np.where(h_r == 1)[0]
        order = np.argsort(h[oi])[:excess]
        h_r[oi[order]] = 0
    elif cur < target_sum:
        deficit = target_sum - cur
        zi = np.where(h_r == 0)[0]
        order = np.argsort(-h[zi])[:deficit]
        h_r[zi[order]] = 1
    cur = int(h_r.sum())
    while cur < target_sum:
        z = np.where(h_r == 0)[0]
        if len(z) == 0: break
        h_r[z[0]] = 1
        cur += 1
    while cur > target_sum:
        o = np.where(h_r == 1)[0]
        if len(o) == 0: break
        h_r[o[0]] = 0
        cur -= 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def sa_fft(h_start, time_limit, start_time, rng, t_init, t_final, 
           iters_per_phase, n_phases, target_prob=0.3):
    """
    Fast SA with FFT evaluation and periodic targeted moves.
    """
    n = len(h_start)
    size = 2 * n - 1
    dx = 2.0 / n
    h = h_start.copy()
    
    corr = np.fft.irfft(np.fft.rfft(h, size) * np.conj(np.fft.rfft(1.0 - h, size)))
    c5 = float(np.max(corr)) * dx
    best_h = h.copy()
    best_c5 = c5
    it = 0
    
    for phase in range(n_phases):
        t_max = iters_per_phase
        phase_start = it
        
        while it - phase_start < t_max:
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            it += 1
            t = t_init * (t_final / t_init) ** ((it - phase_start) / t_max)
            
            # Choose move type
            r = rng.random()
            use_targeted = (it - phase_start) > t_max * 0.5 and r < target_prob
            
            if use_targeted:
                # Target worst shift
                max_idx = int(np.argmax(corr))
                k = max_idx - (n - 1)
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                # Score-based selection
                # For speed, just pick random from ones/zeros but weight toward
                # positions that are "bad" for the worst shift
                # h[i+k]=0 and h[i-k]=0 means i is "free" for this shift
                # We want to remove such i and add j where h[j+k]=1 and h[j-k]=1
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
            else:
                i = rng.randint(n)
                j = rng.randint(n)
            
            if i == j or h[i] == h[j]:
                continue
            
            hi = h[i]
            hj = h[j]
            h[i] = 1.0 - hi
            h[j] = 1.0 - hj
            
            corr = np.fft.irfft(np.fft.rfft(h, size) * np.conj(np.fft.rfft(1.0 - h, size)))
            c5_new = float(np.max(corr)) * dx
            delta = c5_new - c5
            
            if delta <= 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i] = hi
                h[j] = hj

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_global_n, _cons_global_target
    _cons_global_n = n
    _cons_global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5_direct(h_cur)
