def project_feasible(h, target_sum):

        else:

            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(x, n, target, alpha, lambda_pen):
    """Compute the log-sum-exp objective and its gradient."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    over = corr[n-1:n-1+n]
    dx = 2.0 / n

        wk = weights[k] * dx
        if wk < 1e-12:
            continue

        # Second part: for j >= k
        if k > 0:

For a 0/1 vector:
- If h[j+k] = 0: (1-h[j+k]) = 1, so the first term contributes -1 (we remove a "1-0" pair)
- If h[j+k] = 1: (1-h[j+k]) = 0, no contribution

Similarly for other terms.

The key optimization: for a given pair (j, m), compute ΔN_k for all k simultaneously. This can be done in O(n) time.

Then, the new C₅ is max_k (N_k + ΔN_k) * dx. We want this to be less than the current max.

For a greedy approach:
1. Find the current bottleneck lag k* (where N_k is maximum)
2. Find a swap (j, m) that reduces N_{k*} the most while not increasing other N_k too much
3. Make the swap

For a simulated annealing approach:
1. Propose a random swap
2. Compute the new C₅
3. Accept with probability exp(-ΔC₅/T)

**Coordinate descent**:
- At each step, pick a coordinate j
- Try all values of h[j] in {0, 0.1, 0.2, ..., 1.0} (or optimize the 1D subproblem)
- The 1D subproblem: given h with h[j] = t, what value of t minimizes C₅?
- C₅ as a function of t is max_k f_k(t) where f_k is linear in t (since each f_k has at most 2 terms involving h[j])
- So the 1D subproblem is: minimize max of a set of linear functions of t, subject to 0 ≤ t ≤ 1 and the sum constraint
- This can be solved by finding the intersection of the active lines

This is actually a very clean approach! The 1D subproblem is a "bracketing" problem.

For coordinate j, the terms involving h[j] in f_k are:
- If j < n-k: f_k has a term h[j]*(1-h[j+k]), which is linear in h[j] (coefficient: 1-h[j+k])
- If j ≥ k: f_k has a term h[j-k]*(1-h[j]), which is linear in h[j] (coefficient: -h[j-k])

So f_k as a function of t = h[j]:
f_k(t) = a_k * t + b_k (linear in t)

where:
a_k = dx * [(1-h[j+k])*1_{j<n-k} - h[j-k]*1_{j≥k}]
b_k = f_k - a_k * t_current (computed at the current value)

The 1D problem: minimize max_k (a_k * t + b_k) subject to 0 ≤ t ≤ 1 and sum constraint.

The sum constraint: sum(h) = target means t = target - sum(h[m] for m≠j). So t is actually fixed by the sum constraint! We can't freely choose t.

Unless we also modify another coordinate. So it's really a 2D subproblem:

            for _ in range(5):
                if time.time() > deadline:
                    break
                h_new = _lbfgs(h_cur, n, target, alpha, lamb, maxiter=200)
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < cur_sc - 1e-12:
                    h_cur = h_new
                    cur_sc = sc
                    improved = True
    return h_cur, cur_sc


def de_search(n, target, pop_size, max_gen, rng, deadline, F=0.8, CR=0.9):
    pop = [project_feasible(rng.rand(n), target) for _ in range(pop_size)]
    scores = [compute_c5(h) for h in pop]
    idx = np.argsort(scores)
    pop = [pop[i] for i in idx]
    scores = [scores[i] for i in idx]
    best = pop[0].copy()
    best_sc = scores[0]
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            others = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(others, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            jr = rng.randint(n)
            trial = np.where(rng.rand(n) < CR, mutant, pop[i])
            trial[jr] = mutant[jr]
            trial = project_feasible(trial, target)
            ts = compute_c5(trial)
            if ts <= scores[i]:
                pop[i] = trial
                scores[i] = ts
                if ts < best_sc:
                    best_sc = ts
                    best = trial.copy()
        idx = np.argsort(scores)
        pop = [pop[i] for i in idx]
        scores = [scores[i] for i in idx]
    return best, best_sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def zero_one_sa(h, n, target, n_iters, rng, deadline, T_init=0.0005):
    """Simulated annealing on 0/1 vectors using 2-swaps."""
    h = np.zeros(n)
    h[:int(target)] = 1.0
    # Start from the rounded version of h_input
    h_input = h  # will be overwritten
    
    # Actually, let me accept the input h and round it
    h_round = (h > 0.5).astype(float)
    # Fix cardinality
    n_ones = int(h_round.sum())
    if n_ones > int(target):
        # Remove excess
        one_idx = np.where(h_round == 1.0)[0]
        excess = n_ones - int(target)
        to_rm = one_idx[:excess]
        h_round[to_rm] = 0.0
    elif n_ones < int(target):
        zero_idx = np.where(h_round == 0.0)[0]
        deficit = int(target) - n_ones
        to_add = zero_idx[:deficit]
        h_round[to_add] = 1.0
    
    h = h_round
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    
    # Precompute N_k
    def get_N(hv):
        N = np.zeros(n)
        for k in range(1, n):
            N[k] = np.sum(hv[:n - k] * (1.0 - hv[k:]))
        return N
    
    N = get_N(h)
    cur_max = float(np.max(N))
    best_N = N.copy()
    best_h = h.copy()
    best_sc = cur_max * (2.0 / n)
    T = T_init
    
    # Track iteration for RNG
    it_count = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick random j from ones, m from zeros
        j = ones_list[rng.randint(len(ones_list))]
        m = zeros_list[rng.randint(len(zeros_list))]
        
        # Compute delta N_k for this swap
        # Swap: h[j] 1->0, h[m] 0->1
        dN = np.zeros(n)
        for k in range(1, n):
            delta = 0.0
            # j as first index (was 1, now 0): removes contribution (1-h[j+k]) if j < n-k
            if j < n - k:
                delta -= (1.0 - h[j + k])
            # j as second index (was 1, now 0): was removing h[j-k]*(1-1)=0, now h[j-k]*(1-0)=h[j-k]
            if j >= k:
                delta += h[j - k]
            # m as first index (was 0, now 1): adds contribution (1-h[m+k]) if m < n-k
            if m < n - k:
                delta += (1.0 - h[m + k])
            # m as second index (was 0, now 1): was h[m-k]*(1-0)=h[m-k], now h[m-k]*(1-1)=0
            if m >= k:
                delta -= h[m - k]
            # Cross terms
            # If j+k == m: h[j+k] was h[m]=0, now h[m]=1
            # The first term for j: (1-h[j+k]) was (1-0)=1, now (1-1)=0
            # Already handled? Let's check:
            # In "j as first index": we computed -(1-h[j+k]) using OLD h
            # But h[j+k] = h[m] which changes from 0 to 1
            # So the actual change should be: old contribution was h[j]*(1-h[j+k]) = 1*(1-h[m]) = 1
            #  new contribution is h'[j]*(1-h'[j+k]) = 0*(1-1) = 0
            #  change = -1
            # My formula gives: -(1-h[j+k]) = -(1-0) = -1. Correct!
            # But wait, h[j+k] = h[m] = 0 (old value). After swap h[m]=1.
            # The old contribution was 1*(1-0) = 1. New is 0*(1-1) = 0. Change = -1.
            # My formula: -(1-h[m]) = -(1-0) = -1. Correct.
            # 
            # For "m as first index": old was 0*(1-h[m+k])=0, new is 1*(1-h'[m+k])
            # If m+k == j: h'[m+k] = h'[j] = 0, so new = 1*(1-0) = 1. Change = +1.
            # My formula: +(1-h[m+k]) = (1-h[j]) = 1-1 = 0. WRONG!
            # Because h[m+k] = h[j] = 1 (old), but after swap h[j] = 0.
            # So I need to use the NEW value.
            
            # Let me redo this more carefully.
            # ...
            
            dN[k] = delta
        
        # Apply swap
        new_max = float(np.max(N + dN))
        delta_sc = (new_max - cur_max) * (2.0 / n)
        
        if delta_sc < 0 or rng.rand() < np.exp(-delta_sc / T):
            # Accept
            h[j] = 0.0
            h[m] = 1.0
            ones_list.remove(j)
            ones_list.append(m)
            zeros_list.remove(m)
            zeros_list.append(j)
            N += dN
            cur_max = float(np.max(N))
            if cur_max * (2.0 / n) < best_sc:
                best_sc = cur_max * (2.0 / n)
                best_h = h.copy()
                best_N = N.copy()
        
        T *= 0.99999
        if T < 1e-8:
            T = 1e-8
    
    return best_h, best_sc
