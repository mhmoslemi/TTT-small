def subgradient_step(h, n_active, temp, project_back=True):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    overlaps_unscaled = np.correlate(h, 1.0 - h, mode='full')[n-1:n-1+n]
    c_curr = float(np.max(overlaps_unscaled)) * dx
    top_idx = np.argsort(overlaps_unscaled)[::-1][:n_active]
    diffs = overlaps_unscaled[top_idx] - overlaps_unscaled[top_idx[0]]
    weights = np.exp(diffs / (temp / dx))
    weights /= weights.sum()
    grad = np.zeros(n)
    for wk, k in zip(weights, top_idx):
        k = int(k)
        if k == 0:
            grad[:] += wk * dx * (1.0 - 2.0 * h)
        else:
            upper = n - 1 - k
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += wk * dx * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= wk * dx * h[idx2 - k]
    return grad, c_curr

    if tleft(start_time, budget_s) > 200:
        h_sg, sc_sg = subgradient_descent(best_h, 300, 4, 0.003, 0.998, 0.3, 0.005)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
    
    # Phase 3: Binary pair search cycle
    for cycle in range(3):
        if tleft(start_time, budget_s) < 200:
            break
        # Round to binary
        h_bin = make_binary(best_h, target)
        # Binary local search
        h_bin2, sc_bin = binary_pair_search(h_bin, max_swaps=2000, n_cand=25, rng=rng)
        # Convert back to continuous and refine
        h_cont = h_bin2.copy()
        h_cont, sc_cont = slsqp_optimize(h_cont, [500, 2000, 10000], 1500, 1e-15)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont
        # Also check if the binary itself is better (scaled)
        sc_bin_scaled = compute_c5(h_bin2)
        if sc_bin_scaled < best_score:
            best_score = sc_bin_scaled
            best_h = h_bin2.copy()
    
    # Phase 4: Simulated annealing
    if tleft(start_time, budget_s) > 150:
        h_bin = make_binary(best_h, target)
        h_sa, sc_sa = binary_simulated_annealing(h_bin, rng, n_iter=300, t_init=0.008, t_final=0.0005)
        h_sa_cont, sc_sa_cont = slsqp_optimize(h_sa.copy(), [500, 2000, 10000], 1200, 1e-15)
        if sc_sa_cont < best_score:
            best_score = sc_sa_cont
            best_h = h_sa_cont
        sc_sa_bin = compute_c5(h_sa)
        if sc_sa_bin < best_score:
            best_score = sc_sa_bin
            best_h = h_sa.copy()
    
    # Phase 5: More binary pair search with different params
    if tleft(start_time, budget_s) > 150:
        h_bin = make_binary(best_h, target)
        h_bin3, sc_bin3 = binary_pair_search(h_bin, max_swaps=3000, n_cand=40, rng=rng)
        h_ref, sc = slsqp_optimize(h_bin3.copy(), [1000, 5000, 20000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        sc_b = compute_c5(h_bin3)
        if sc_b < best_score:
            best_score = sc_b
            best_h = h_bin3.copy()
    
    # Phase 6: Multi-resolution
    if tleft(start_time, budget_s) > 300:
        for n_fine in [n + 100, n + 200]:
            if tleft(start_time, budget_s) < 150:
                break
            if n_fine % 2 != 0:
                n_fine += 1
            target_fine = n_fine / 2.0
            h_fine = interpolate_to(best_h, n_fine, target_fine)
            h_fine, sc_fine = slsqp_optimize(h_fine, [200, 1000, 5000], 1200, 1e-14)
            h_bin_f = make_binary(h_fine, target_fine)
            h_bin_f2, sc_bf = binary_pair_search(h_bin_f, max_swaps=1500, n_cand=25, rng=rng)
            h_fine2, sc_fine2 = slsqp_optimize(h_bin_f2.copy(), [2000, 10000, 50000], 1200, 1e-15)
            if sc_fine2 < best_score:
                best_score = sc_fine2
                best_h = h_fine2
            sc_bf2 = compute_c5(h_bin_f2)
            if sc_bf2 < best_score:
                best_score = sc_bf2
                best_h = h_bin_f2.copy()
    
    # Phase 7: Random restarts
    if tleft(start_time, budget_s) > 120:
        for attempt in range(5):
            if tleft(start_time, budget_s) < 60:
                break
            n_cur = len(best_h)
            target_cur = n_cur / 2.0
            h_pert = best_h.copy()
            if attempt % 3 == 0:
                h_pert += rng.randn(n_cur) * 0.05
            elif attempt % 3 == 1:
                block = max(8, n_cur // 25)
                for b in range(n_cur // block):
                    if rng.random() < 0.5:
                        h_pert[b*block:min((b+1)*block, n_cur)] += rng.uniform(-0.12, 0.12)
            else:
                h_pert += np.sin(np.arange(n_cur) * 0.05 + rng.rand() * 6.28) * 0.03
            h_pert = project_feasible(h_pert, target_cur)
            h_bin_p = make_binary(h_pert, target_cur)
            h_bin_p2, sc_bp = binary_pair_search(h_bin_p, max_swaps=1500, n_cand=20, rng=rng)
            h_ref2, sc2 = slsqp_optimize(h_bin_p2.copy(), [500, 5000], 1000, 1e-15)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_ref2
            sc_bp2 = compute_c5(h_bin_p2)
            if sc_bp2 < best_score:
                best_score = sc_bp2
                best_h = h_bin_p2.copy()
    
    # Phase 8: Final high-alpha SLSQP
    if tleft(start_time, budget_s) > 80:
        h_ref, sc = slsqp_optimize(best_h, [5000, 20000, 100000], 1500, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 9: Final surgery
    if tleft(start_time, budget_s) > 40:
        h_surg, sc_surg = worst_lag_surgery(best_h, max_steps=100, initial_epsilon=0.0005)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref, sc = slsqp_optimize(best_h, [20000, 100000], 800, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 10: One final binary search + refined SLSQP
    if tleft(start_time, budget_s) > 50:
        h_bin_final = make_binary(best_h, len(best_h) / 2.0)
        h_bin_final2, sc_bf = binary_pair_search(h_bin_final, max_swaps=1000, n_cand=30, rng=rng)
        h_final_cont, sc_final = slsqp_optimize(h_bin_final2.copy(), [10000, 50000, 200000], 800, 1e-17)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cont
        sc_bf2 = compute_c5(h_bin_final2)
        if sc_bf2 < best_score:
            best_score = sc_bf2
            best_h = h_bin_final2.copy()
    
    # Final
    if tleft(start_time, budget_s) > 15:
        h_ref, sc = slsqp_optimize(best_h, [50000], 500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
