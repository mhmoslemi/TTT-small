def sim_annealing_binary(n, n_iters, T0, T1, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    ones_idx = np.arange(0, n, 2)  # even positions are 1
    h[ones_idx] = 1.0
    
    # Compute initial cost
    def cost(h):
        autocorr = np.correlate(h, h, mode='full')
        # autocorr[n-1] = sum h[i]*h[i] = n/2 (for binary)
        # We want overlap_k = n/2 - autocorr[n-1+k] for k=1,...,n-1
        # which is autocorr[n+k-1] for k=1,...,n-1
        # i.e., indices n through 2n-2
        overlaps = n/2.0 - autocorr[n:2*n-1]
        return np.max(overlaps)
    
    best_h = h.copy()
    best_cost = cost(h)
    cur_cost = best_cost
    
    for it in range(n_iters):
        T = T0 * (T1/T0) ** (it / n_iters)
        # Pick random i with h[i]=1 and j with h[j]=0
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        i = ones_pos[rng.randint(len(ones_pos))]
        j = zeros_pos[rng.randint(len(zeros_pos))]
        
        h[i] = 0
        h[j] = 1
        new_cost = cost(h)
        
        if new_cost < cur_cost or rng.random() < np.exp(-(new_cost - cur_cost) / T):
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_cost

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[_g_n-1:_g_n-1+_g_n]
    c_max = float(np.max(over))
    shifted = _g_alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def slsqp_optimize(h, alphas, maxiter, ftol):
    global _g_n, _g_target, _g_alpha
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

        h[idx[:n_ones]] = 1.0

    best_cost = cur_cost
    it = 0

        zeros_mask = (h == 0.0)
        ones_idx = np.where(ones_mask)[0]
        zeros_idx = np.where(zeros_mask)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        if i == j:
            it += 1
            continue
        
        h[i] = 0.0
        h[j] = 1.0
        new_cost = binary_cost(h)
        
        delta = new_cost - cur_cost
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        it += 1
    
    return best_h, best_cost

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.006)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def worst_lag_surgery(h, max_steps=200, initial_epsilon=0.001):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            grad = dx * (1.0 - 2.0 * h_cur)
        else:
            grad = np.zeros(n)
            upper = n - 1 - k_star
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h_cur[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h_cur[idx2 - k_star]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.006)
        else:
            eps *= 0.8
            if eps < 1e-8:
                break
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    from scipy.interpolate import interp1d
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    f = interp1d(x_old, h_src, kind='linear', fill_value='extrapolate')
    h_out = f(x_new)
    return project_feasible(h_out, target_dst)

def try_rounding_improve(h, target):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(h)
    for thresh in [0.25, 0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6]:
        h_r = h.copy()
        h_r[h_r < thresh] = 0.0
        h_r[h_r > 1.0 - thresh] = 1.0
        h_r = project_feasible(h_r, target)
        sc = compute_c5(h_r)
        if sc < best_score:
            best_score = sc
            best = h_r.copy()
    return best, best_score

    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: SLSQP refinement of initial
    if tleft() > 400:
        h_ref, sc = slsqp_optimize(best_h, [100, 500, 2000], 2000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 2: Binary SA (the main new contribution)
    # Try multiple restarts
    n_sa = n  # same resolution
    n_sa_ones = n_sa // 2
    sa_solutions = []
    
    if tleft() > 600:
        # First SA run from random init
        h_sa, sa_cost = sim_annealing_binary(n_sa, 300000, 100.0, 0.5, 
                                              h_init=None, seed=seed, 
                                              timeout=tleft()-300, start_time=start_time)
        sa_solutions.append((h_sa, sa_cost * (2.0/n_sa)))
    
    if tleft() > 500:
        # Second SA run from different random init
        h_sa2, sa_cost2 = sim_annealing_binary(n_sa, 300000, 80.0, 0.3,
                                                h_init=None, seed=seed+1,
                                                timeout=tleft()-250, start_time=start_time)
        sa_solutions.append((h_sa2, sa_cost2 * (2.0/n_sa)))
    
    if tleft() > 400:
        # Third SA run from perturbation of current best
        h_pert = best_h.copy()
        h_pert += rng.randn(n_sa) * 0.1
        h_pert = project_feasible(h_pert, target)
        # Round to binary for SA
        h_binary_pert = (h_pert > 0.5).astype(float)
        # Ensure correct number of ones
        if h_binary_pert.sum() != n_sa_ones:
            idx_sorted = np.argsort(h_pert)[::-1]
            h_binary_pert = np.zeros(n_sa)
            h_binary_pert[idx_sorted[:n_sa_ones]] = 1.0
        h_sa3, sa_cost3 = sim_annealing_binary(n_sa, 200000, 50.0, 0.2,
                                                h_init=h_binary_pert, seed=seed+2,
                                                timeout=tleft()-200, start_time=start_time)
        sa_solutions.append((h_sa3, sa_cost3 * (2.0/n_sa)))
    
    # Check if any SA solution beats current best
    for h_cand, cand_score in sa_solutions:
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_cand.copy()
            # Continuous refinement of the binary solution
            if tleft() > 300:
                h_ref, sc = slsqp_optimize(best_h, [100, 500, 2000], 2000, 1e-15)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref
    
    # Phase 3: Rounding and refinement
    if tleft() > 300:
        h_round, sc_round = try_rounding_improve(best_h, target)
        if sc_round < best_score:
            best_score = sc_round
            best_h = h_round
            h_ref, sc = slsqp_optimize(best_h, [500, 2000, 10000], 3000, 1e-15)
            if sc < best_score:
                best_score = sc
                best_h = h_ref
    
    # Phase 4:
