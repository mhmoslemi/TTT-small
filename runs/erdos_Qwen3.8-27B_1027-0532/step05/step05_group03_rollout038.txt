    # Term 2: +h[a-k] for k where k <= a, i.e., k from 0 to a
    max_k2 = min(a, n - 1)
    if max_k2 >= 0:
        k_range = np.arange(0, max_k2 + 1)
        delta[k_range] += h[a - k_range]
    
    # Term 3: +(1-h[b+k]) for k where b < n-k, i.e., k < n-b
    max_k3 = min(n - b - 1, n - 1)
    if max_k3 >= 0:
        k_range = np.arange(0, max_k3 + 1)
        delta[k_range] += (1.0 - h[b + k_range])
    
    # Term 4: -h[b-k] for k where k <= b, i.e., k from 0 to b
    max_k4 = min(b, n - 1)
    if max_k4 >= 0:
        k_range = np.arange(0, max_k4 + 1)
        delta[k_range] -= h[b - k_range]
    
    return delta

def subgradient_step(h, top_k, lr, momentum, velocity, temp):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    corr = compute_overlaps(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    diffs = corr[top_idx] - corr[top_idx[0]]
    weights = np.exp(diffs / temp)
    weights /= weights.sum()
    grad = np.zeros(n)
    for wk, k in zip(weights, top_idx):
        k = int(k)
        if k == 0:

            grad[idx1] += wk * dx * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= wk * dx * h[idx2 - k]

    h_new = h - lr * velocity
    h_new = project_feasible(h_new, target)
