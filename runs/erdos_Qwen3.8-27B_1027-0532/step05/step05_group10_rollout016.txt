import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0
_global_alpha = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def build_lp_matrix(n, g):
    dx = 2.0 / n
    num_lags = 2 * n - 1
    A = np.zeros((num_lags, n + 1))
    for j in range(num_lags):
        k = j - (n - 1)
        if k >= 0:
            length = n - k
            if length > 0:
                A[j, :length] = g[k:k + length] * dx
        else:
            k2 = -k
            length = n - k2
            if length > 0:
                A[j, k2:] = g[:length] * dx
        A[j, n] = -1.0
    return A


def solve_full_lp(h_current, n, target):
    g = 1.0 - h_current
    A = build_lp_matrix(n, g)
    x = cp.Variable(n + 1)
    i_vec = np.ones(n + 1)
    i_vec[n] = 0.0
    cons = [A @ x <= 0, i_vec @ x == target, x[:n] >= 0, x[:n] <= 1]
    prob = cp.Problem(cp.Minimize(x[n]), cons)
    try:
        prob.solve(solver=cp.SCS, eps=1e-10, max_iters=100000)
    except Exception:
        pass
    if x.value is None:
        try:
            prob.solve(solver=cp.PDLP, max_iter=100000)
        except Exception:
            pass
    if x.value is None:
        return None, None
    return np.clip(x.value[:n], 0.0, 1.0), x.value[n]


def alternating_lp_full(h_init, n, target, max_iter, time_limit, damping=0.9):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_damping = damping
    stall = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        result = solve_full_lp(h, n, target)
        if result[0] is None:
            break
        h_lp = project_feasible(result[0], target)
        h_new = (1.0 - cur_damping) * h + cur_damping * h_lp
        h_new = project_feasible(h_new, target)
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if c5_val > best_c5 + 1e-6:
            cur_damping = max(0.3, cur_damping * 0.75)
        else:
            cur_damping = min(1.0, cur_damping + 0.03)
        h = h_new
        if stall > 300:
            break
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:

                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        idx_v1 = (idxs >= k_star) & (idxs < n + k_star)
        idx_v2 = (idxs >= -k_star) & (idxs - k_star >= 0) & (idxs - k_star < n)
        grad = np.zeros(n)
        mask1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        mask2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[mask1] += (1.0 - h_cur[idxs[mask1] - k_star])
        grad[mask2] -= h_cur[idxs[mask2] + k_star]
        grad *= dx
        improved = False

                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

            break
    return best_h, best_c5
