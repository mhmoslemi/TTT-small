def binary_local_search(h_bin, target, max_iters=20000, n_check=5, rng=None):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    def c5_val(h):
        corr = np.correlate(h, 1.0 - h, mode='full')[n-1:n-1+n]
        return float(np.max(corr)) * dx
    
    def lap_change(h, a, b, k):
        """Compute change in c_k when h_a goes 1->0 and h_b goes 0->1."""
        delta = 0.0
        # Remove a (was 1, now 0)
        if a + k < n:
            delta -= (1.0 - h[a+k])
        if a - k >= 0 and a - k <= n - 1 - k:
            delta += h[a-k]
        # Add b (was 0, now 1)
        if b + k < n:
            delta += (1.0 - h[b+k])
        if b - k >= 0 and b - k <= n - 1 - k:
            delta -= h[b-k]
        return delta
    
    cur_score = c5_val(h)
    best_score = cur_score
    best_h = h.copy()
    stall = 0
    
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')[n-1:n-1+n]
        top_lags = np.argsort(corr)[::-1][:n_check]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_improve = 0
        best_a = -1
        best_b = -1
        
        # Try a subset of swaps for efficiency
        # Focus on elements that are "expensive" to keep in the worst lags
        n_try = min(len(ones) * len(zeros), 2000)
        
        # Sample pairs
        if n_try < len(ones) * len(zeros):
            # Smart sampling: focus on elements near the worst lag pattern
            for _ in range(200):
                a = int(rng.choice(ones))
                b = int(rng.choice(zeros))
                # Quick check on worst lag only
                d = lap_change(h, a, b, top_lags[0])
                if d < -best_improve:
                    # Verify all top lags
                    ok = True
                    for k in top_lags[1:]:
                        if lap_change(h, a, b, int(k)) > 0:
                            ok = False
                            break
                    if ok or d < -best_improve * 2:
                        new_score_est = cur_score + d * dx  # rough
                        if new_score_est < cur_score - best_improve:
                            best_improve = -d
                            best_a = a
                            best_b = b
        else:
            for a in ones:
                for b in zeros:
                    d = lap_change(h, a, b, top_lags[0])
                    if d < -best_improve:
                        best_improve = -d
                        best_a = a
                        best_b = b
        
        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            new_score = c5_val(h)
            if new_score < cur_score:
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = h.copy()
                stall = 0
            else:
                h[best_a] = 1.0
                h[best_b] = 0.0
                stall += 1
        else:
            stall += 1
        
        if stall > 500:
            # Perturb to escape local minimum
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            h[a] = 0.0
            h[b] = 1.0
            if c5_val(h) < best_score:
                best_h = h.copy()
                best_score = c5_val(h)
            stall = 0
    
    return best_h, best_score

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        h[mask] += diff / cnt

    return c_max + float(np.log(exp_s.sum())) / alpha
