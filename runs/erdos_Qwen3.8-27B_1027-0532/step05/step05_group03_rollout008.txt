def identify_blocks(h, thresh=0.5):
    """Identify blocks of 0s and 1s (binarized at thresh)"""
    binary = (h >= thresh).astype(int)
    blocks = []
    i = 0
    while i < len(binary):
        j = i
        while j < len(binary) and binary[j] == binary[i]:
            j += 1
        blocks.append((i, j, binary[i]))
        i = j
    return blocks

def block_flip(h, block_start, block_end, target):
    """Flip a block of h"""
    h_new = h.copy()
    h_new[block_start:block_end] = 1.0 - h_new[block_start:block_end]
    return project_feasible(h_new, target)

    """Shift a block's content"""
    h_new = h.copy()
    seg = h_new[block_start:block_end]
    h_new[block_start:block_end] = np.roll(seg, shift)
    return project_feasible(h_new, target)

def grad_corr_index(h, idx):
    """Compute gradient of corr[idx] with respect to h."""
    n = len(h)
    k = idx - (n - 1)  # the lag
    grad = np.zeros(n)
    if k >= 0:
        # corr = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
        # d/dh[j] = (1-h[j-k]) for j in [k, n-1] + (-h[j+k]) for j in [0, n-1-k]
        upper = n - 1 - k
        # From h[n+k] term: j = n+k, n in [0, upper], so j in [k, n-1]
        grad[k:] += (1.0 - h[:n-k])  # (1-h[j-k]) for j in [k, n-1]
        # From (1-h[n]) term: j = n, n in [0, upper], so j in [0, upper]
        grad[:upper+1] -= h[k:k+n-k]  # -h[j+k] for j in [0, upper]
    else:
        m = -k  # m > 0
        # corr = sum_{n=m}^{n-1} h[n-m] * (1-h[n])
        # d/dh[j] = (1-h[j+m]) for j in [0, n-1-m] + (-h[j-m]) for j in [m, n-1]
        upper = n - 1 - m
        # From h[n-m] term: j = n-m, n in [m, n-1], so j in [0, n-1-m]
        grad[:upper+1] += (1.0 - h[m:m+n-m])  # (1-h[j+m]) for j in [0, upper]
        # From (1-h[n]) term: j = n, n in [m, n-1], so j in [m, n-1]
        grad[m:] -= h[:n-m]  # -h[j-m] for j in [m, n-1]
    return grad
