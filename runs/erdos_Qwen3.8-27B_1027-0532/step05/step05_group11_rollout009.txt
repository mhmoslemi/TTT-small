```python
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:

    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:

        if len(idx2) > 0:

            h = np.clip(h, 0.0, 1.0)
    return h

    """Solve LP with all lag constraints, fixing g=1-h"""
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
    ]

        if k == 0:

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5


def lse_gradient(h, alpha):
    """Compute gradient of log-sum-exp approximation to max overlap"""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    n_half = n - 1
    
    # Softmax weights
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient
    grad = np.zeros(n)
    for k_idx in range(1, 2*n - 1):
        k = k_idx - n_half
        w = weights[k_idx]
        if w < 1e-6:
            continue
        if k >= 0:
            n_valid = n - k
            # d(O_k)/d(h[i]) = dx*(1-h[i+k]) for i in [0, n-1-k]
            # d(O_k)/d(h[i]) = -dx*h[i-k] for i in [k, n-1]
            for i in range(n_valid):
                grad[i] += w * dx * (1 - h[i + k])
            for i in range(k, n):
                grad[i] -= w * dx * h[i - k]
        else:
            k_pos = -k
            n_valid = n - k_pos
            for i in range(n_valid):
                grad[i] += w * dx * (1 - h[i + k_pos])
            for i in range(k_pos, n):
                grad[i] -= w * dx * h[i - k_pos]
    
    # Subtract constraint gradient (for equality constraint sum(h) = target)
    # We'll handle this via projection
    return grad, float(max_ov + np.log(np.sum(exp_shifted)) / alpha)


def lse_gradient_fast(h, alpha):
    """Faster gradient computation using vectorized operations"""
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    
    max_ov = np.max(overlaps)
    shifted = alpha * (overlaps - max_ov)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    
    grad = np.zeros(n)
    n_half = n - 1
    
    for k_idx in range(1, 2*n - 1):
        k = k_idx - n_half
        w = weights[k_idx]
        if w < 1e-5:
            continue
        if k >= 0:
            n_valid = n - k
            grad[:n_valid] += w * dx * (1 - h[k:])
            grad[k:] -= w * dx * h[:n_valid]
        else:
            k_pos = -k
            n_valid = n - k_pos
            grad[:n_valid] += w * dx * (1 - h[k_pos:])
            grad[k_pos:] -= w * dx * h[:n_valid]
    
    obj_val = float(max_ov + np.log(total) / alpha)
    return grad, obj_val


def nesterov_lse_optimize(h_init, n, target, max_steps, time_limit, 
                           alpha_start=10, alpha_end=50000,
                           momentum_init=0.9, momentum_end=0.999):
    """Nesterov-accelerated gradient descent on log-sum-exp objective"""
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        progress = step / max(1, max_steps)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        momentum = momentum_init + (momentum_end - momentum_init) * progress
        
        # Nesterov step
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad, obj_val = lse_gradient_fast(y, alpha)
        
        # Polyak step size
        c5_y = compute_c5(y)
        gap = max(0, c5_y - best_c5)
        norm_grad = np.linalg.norm(grad)
        if norm_grad < 1e-15:
            break
        lr = gap / (norm_grad ** 2 + 1e-20)
        lr = min(max(lr, 1e-4), 0.1)
        
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target)
        
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        h_prev = h.copy()
        h = h_try
    
    return best_h, best_c5


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.success or res.fun < 1:
                h_new = project_feasible(res.x, target)
                if compute_c5(h_new) < compute_c5(h_cur):
                    h_cur = h_new
        except Exception:
            pass
    return h_cur

_global_n = 0
_global_target = 0.0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def generate_golay_pair(n):
    """Generate a Golay complementary pair of length n (power of 2)"""
    # Start with length 1
    a = np.array([1], dtype=float)
    b = np.array([0], dtype=float)
    while len(a) < n:
        a = np.concatenate([a, b])
        b = np.concatenate([a[:-len(b)], -b])
    # a and b are 0/1 sequences
    return a[:n].astype(float), b[:n].astype(float)
```

```python
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def full_lp_solve(h, n, target):
    """Solve LP with all lag constraints. Returns (h_new, t_value) or (None, None)"""
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
    ]
    
    for k in range(1, n):
        # Lag +k
        coeff_pos = np.zeros(n)
        coeff_pos[:n-k] = g[k:] * dx
        constraints.append(coeff_pos @ h_var <= t_var)
        
        # Lag -k (by symmetry of the problem formulation)
        coeff_neg = np.zeros(n)
        coeff_neg[k:] = g[:n-k] * dx
        constraints.append(coeff_neg @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-10, verbose=False)
    except Exception:
        return None, None
    
    if h_var.value is None:
        return None, None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), float(t_var.value)
```

Wait, I realize I'm double-counting. The correlation has 2n-1 lags, from -(n-1) to +(n-1). Lag 0 gives O(0) = 1 - dx*sum(h²) which is always ≤ 1 and typically not the maximum. For lags 1 to n-1, we have both +k and -k.

But actually, O(k) and O(-k) are different in general! So we need both.

For n=400, that's 2*399 = 798 constraints (plus 3 for the basic ones). Total: 801 constraints on 401 variables. This is still very manageable.

Let me reconsider the time. SCS with 801 constraints and 401 variables... the constraint matrix is 801×401 ≈ 320,000 entries. SCS typically converges in 100-1000 iterations for problems this size, with each iteration being O(nnz) ≈ O(320,000). So each solve is about 10-32 seconds?? That might be too slow.

Let me think about this more carefully. SCS uses an ADMM method. Each iteration involves solving a linear system of size ~(n_vars + n_constrs) = ~1202. The cost per iteration is O(1202^1.5) for direct solve or O(1202^2) for iteration. With 1000 iterations, that's O(10^8) to O(10^9) operations. At 1 GFLOP/s, that's 0.1-1 second per solve.

With 50 LP iterations, that's 5-50 seconds. Should fit in the budget.

But wait, I can reduce the number of constraints by noting that O(k) = O(-k) when h is symmetric. But h is not necessarily symmetric, so we can't use this.

Alternative: use a subset of constraints. The key insight is that at the optimum, only O(1) constraints are active (by the LP duality). So we can use a **constraint generation** approach:
1. Start with a small set of constraints (e.g., top 50 lags)
2. Solve the LP
3. Check which constraints are violated
4. Add the most violated ones
5. Repeat

This is the **cutting plane** method, and it's much more efficient than using all constraints.

```python
def cutting_plane_lp(h, n, target, max_rounds=50, n_init=50, n_add=20, time_limit=60):
    """Cutting plane method for the bilinear overlap minimization"""
    start = time.time()
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    # Initialize with top lags
    overlaps = compute_overlaps(h_cur)
    top_idx = np.argsort(overlaps)[::-1][:n_init]
    active_lags = set(top_idx - (n-1))
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        # Solve LP with current active constraints
        g = 1.0 - h_cur
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0,
        ]
        
        for k in sorted(active_lags):
            if k == 0:
                continue
            lo = max(0, k)
            hi = min(n, n + k)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        
        # Check for violated constraints
        g_new = 1.0 - h_new
        overlaps_new = compute_overlaps(h_new)
        t_lp = float(t_var.value)
        
        # Find most violated constraints not in active set
        violated = []
        for k_idx in range(2*n - 1):
            k = k_idx - (n-1)
            if k == 0 or k in active_lags:
                continue
            if overlaps_new[k_idx] > t_lp + 1e-8:
                violated.append((overlaps_new[k_idx] - t_lp, k))
        
        if len(violated) == 0:
            # No violation, we're at a local optimum of the LP relaxation
            c5_val = compute_c5(h_new)
            if c5_val < best_c5 - 1e-12:
                best_c5 = c5_val
                best_h = h_new.copy()
            break
        
        # Add most violated constraints
        violated.sort(reverse=True)
        for _, k in violated[:n_add]:
            active_lags.add(k)
        
        # Update h_cur
        h_cur = h_new
    
    return best_h, best_c5
```

Hmm, but this cutting plane approach has an issue: the LP is a relaxation of the bilinear problem, so even if no LP constraint is violated, the bilinear objective might not be minimized.

The correct approach is the **alternating method**:
1. Fix g = 1-h
2. Solve LP: min t s.t. A_k(h) ≤ t for all k (or active k), sum(h) = target
3. Update h
4. Repeat

The key difference from cutting plane is that we always re-solve with the updated g.

Let me think about what's the most efficient approach:

**Option A: Full LP (all constraints) at each iteration**
- Pros: Exact linearization, no constraint selection needed
- Cons: More constraints in the LP (slower per iteration)
- n=400: 798 constraints, ~1-5s per solve, 20 iterations = 20-100s

**Option B: Top-K LP at each iteration**
- Pros: Fewer constraints (faster per iteration)
- Cons: Might miss important constraints
- n=400, K=100: 200 constraints, ~0.5-2s per solve, 100 iterations = 50-200s

**Option C: Cutting plane (adaptive constraints)**
- Pros: Automatically finds important constraints
- Cons: More complex, might oscillate
- Similar to Option B but with constraint addition

I think Option A is the cleanest and most likely to give the best result. Let me go with it, but limit the number of iterations based on time.

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_lp_full(h, n, target, t_ub=None):
    dx = 2.0 / n
    g = 1.0 - h
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0,
    ]
    for k in range(1, n):
        coeff_pos = np.zeros(n)
        coeff_pos[:n - k] = g[k:] * dx
        constraints.append(coeff_pos @ h_var <= t_var)
        coeff_neg = np.zeros(n)
        coeff_neg[k:] = g[:n - k] * dx
        constraints.append(coeff_neg @ h_var <= t_var)
    if t_ub is not None:
        constraints.append(t_var <= t_ub)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    solvers_to_try = [cp.SCS, cp.CLARABEL, cp.ECOS]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver, verbose=False)
            break
        except Exception:
            continue
    if h_var.value is None:
        return None, None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0), float(t_var.value)


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_start=0.7, damping_end=0.95):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 200:
            break
        progress = min(1.0, it / max(1, max_iter))
        damping = damping_start + (damping_end - damping_start) * progress
        h_new, t_val = solve_lp_full(h, n, target)
        if h_new is None:
            break
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
    return best_h, best_c5


def lse_gradient_fast(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    max_ov = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - max_ov), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    grad = np.zeros(n)
    for k in range(1, n):
        w_pos = weights[k - 1 + (n - 1)]
        w_neg = weights[n - 1 - k]
        w_total = w_pos + w_neg
        if w_total < 1e-6:
            continue
        grad[:n - k] += w_total * dx * (1.0 - h[k:])
        grad[k:] -= w_total * dx * h[:n - k]
    obj_val = float(max_ov + np.log(total) / alpha)
    return grad, obj_val


def nesterov_optimize(h_init, n, target, max_steps, time_limit,
                      alpha_start=50.0, alpha_end=100000.0,
                      mu_start=0.8, mu_end=0.995):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        progress = step / max(1, max_steps)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress
        mu = mu_start + (mu_end - mu_start) * progress
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        grad, obj_val = lse_gradient_fast(y, alpha)
        c5_y = compute_c5(y)
        gap = max(0.0, c5_y - best_c5)
        ng = np.linalg.norm(grad)
        if ng < 1e-15:
            break
        lr = gap / (ng ** 2 + 1e-20)
        lr = min(max(lr, 1e-5), 0.05)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_prev = h.copy()
        h = h_new
    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol=1e-14):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur) - 1e-12:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def binary_local_search(h, n, target, rng, time_limit=60):
    start = time.time()
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = n // 2
    if n_ones > n_target:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - n_target]:
            h_bin[ones_idx[idx]] = 0.0
    elif n_ones < n_target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros_idx])[::-1]
        for idx in order[:n_target - n_ones]:
            h_bin[zeros_idx[idx]] = 1.0
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    for rnd in range(200):
        if time.time() - start > time_limit:
            break
        improved = False
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_tries = 300
        for _ in range(n_tries):
            if time.time() - start > time_limit:
                break
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c5_try = compute_c5(h_bin)
            if c5_try < best_c5 - 1e-10:
                best_c5 = c5_try
                best_h = h_bin.copy()
                improved = True
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            break
    return best_h, best_c5
```
