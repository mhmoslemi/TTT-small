    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def smooth_max_obj_grad(x):
    n = _g_n
    target = _g_target
    alpha = _g_alpha
    mu = _g_mu
    dx = _g_dx
    
    x = np.clip(x, 0.0, 1.0)
    corr = np.correlate(x, 1.0 - x, mode='full')
    over = corr[n-1:n-1+n]
    m = np.max(over)
    exp_term = np.exp(alpha * (over - m))
    val = (m + np.log(np.sum(exp_term)) / alpha) * dx
    
    # Penalty for sum constraint
    sum_dev = np.sum(x) - target
    val += mu * sum_dev * sum_dev
    
    w = exp_term / exp_term.sum()
    v = 1.0 - x
    first = np.correlate(v, w, 'full')[n-1:n-1+n]
    second = fftconvolve(w, x)[:n]
    grad = (first - second) * dx + 2.0 * mu * sum_dev
    return val, grad

                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h

def thin_step(h, n, target, n_snap=None):
    """Push values toward 0 or 1."""
    h_new = h.copy()
    if n_snap is None:
        n_snap = max(10, n // 20)
    
    # Sort by distance to nearest boundary
    dist_to_0 = h_new
    dist_to_1 = 1.0 - h_new
    dist_to_nearest = np.minimum(dist_to_0, dist_to_1)
    
    # Snap the most extreme values
    idx = np.argsort(dist_to_nearest)[:n_snap]
    for i in idx:
        if h_new[i] < 0.5:
            h_new[i] = 0.0
        else:
            h_new[i] = 1.0
    
    h_new = project_feasible(h_new, target)
    return h_new

def de_search(n, target, pop_size, max_gen, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    for gen in range(max_gen):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = list(range(pop_size))
            candidates.remove(i)
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            mask = rng.rand(n) < CR
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial[~mask] = population[i][~mask]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
    best_idx = np.argmin(scores)
    return population[best_idx], scores[best_idx]

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())

            break

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

    target = _g_target
    alpha = _g_alpha
    mu = _g_mu
    dx = _g_dx
