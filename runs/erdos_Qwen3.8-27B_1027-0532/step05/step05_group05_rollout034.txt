   ```python
   def min_max_linear(slopes, intercepts):
       """Find t in [0,1] minimizing max_i (slopes[i]*t + intercepts[i])"""
       # Check boundaries
       f0 = np.max(intercepts)  # value at t=0
       f1 = np.max(intercepts + slopes)  # value at t=1
       
       # Check if the minimum is at a boundary
       # If all slopes >= 0, min is at t=0
       # If all slopes <= 0, min is at t=1
       if np.all(slopes >= 0):
           return 0.0
       if np.all(slopes <= 0):
           return 1.0
       
       # Use the upper hull approach
       # Sort by slope
       order = np.argsort(slopes)
       s_sorted = slopes[order]
       c_sorted = intercepts[order]
       
       # Build upper convex hull
       # The upper hull consists of lines that are "visible" from above
       # Use a stack-based algorithm
       
       hull_slopes = []
       hull_intercepts = []
       hull_x = []  # x-coordinate of the leftmost point of each line on the hull
       
       for k in range(len(s_sorted)):
           s_new = s_sorted[k]
           c_new = c_sorted[k]
           
           while len(hull_slopes) > 0:
               s_prev = hull_slopes[-1]
               c_prev = hull_intercepts[-1]
               if s_new == s_prev:
                   # Keep the one with higher intercept
                   if c_new <= c_prev:
                       break  # skip this line
                   else:
                       hull_slopes.pop()
                       hull_intercepts.pop()
                       if hull_x:
                           hull_x.pop()
                       continue
   ```

   ```python
               term1[valid_plus] = 1.0 - h[is_plus[valid_plus]]
               
               # i - s: which values are in [0, n)?
               is_minus = i - s
               valid_minus = (is_minus >= 0) & (is_minus < n)
               term2 = np.zeros(2*n-1)
               term2[valid_minus] = -h[is_minus[valid_minus]]
               
               coeff = term1 + term2
               
               # Now: new_corr[s] = corr[n-1+s] + (t - hi) * coeff[s]
               # = (corr[n-1+s] - hi*coeff[s]) + t * coeff[s]
               intercept = corr - hi * coeff
               slopes = coeff
               
               # Find t* = argmin_{t in [0,1]} max_s (intercept[s] + t*slopes[s])
               t_star = _min_max_1d(slopes, intercept)
               
               h[i] = t_star
       
       return h
   ```
