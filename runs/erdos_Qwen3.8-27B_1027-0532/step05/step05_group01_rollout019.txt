```python
import numpy as np
from scipy.optimize import linprog, minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def slp_iteration(h0, n, target):
    """One SLP iteration: solve the LP linearization at h0."""
    dx = 2.0 / n
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    mid = n - 1
    
    # Build LP
    # Variables: [h(0), ..., h(n-1), t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each shift: grad_k^T h - t <= rhs_k
    A_rows = []
    b_rows = []
    
    for shift_idx in range(2 * n - 1):
        k = shift_idx - mid
        # Compute gradient of corr at h0 for this shift
        if k >= 0:
            valid_max = n - 1 - k
            if valid_max < 0:
                continue
            p = np.zeros(n)
            p[:valid_max + 1] = 1.0
            Sh = np.zeros(n)
            Sh[k:] += 0.5 * h0[:n-k]
            Sh[:n-k] += 0.5 * h0[k:]
            grad_k = p - 2.0 * Sh
        else:
            kp = -k
            valid_min = kp
            if valid_min >= n:
                continue
            p = np.zeros(n)
            p[valid_min:] = 1.0
            Sh = np.zeros(n)
            Sh[kp:] += 0.5 * h0[:n-kp]
            Sh[:n-kp] += 0.5 * h0[kp:]
            grad_k = p - 2.0 * Sh
        
        row = np.append(grad_k, -1.0)
        rhs = grad_k @ h0 - corr[shift_idx]
        A_rows.append(row)
        b_rows.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Bound constraints: 0 <= h_i <= 1
    # h_i <= 1: row = [e_i, 0], b = 1
    # -h_i <= 0: row = [-e_i, 0], b = 0
    bound_rows = np.zeros((2 * n, n + 1))
    bound_b = np.zeros(2 * n)
    for i in range(n):
        bound_rows[i, i] = 1.0  # h_i <= 1
        bound_b[i] = 1.0
        bound_rows[n + i, i] = -1.0  # -h_i <= 0
        bound_b[n + i] = 0.0
    
    A_ub = np.vstack([A_ub, bound_rows])
    b_ub = np.concatenate([b_ub, bound_b])
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    
    if res.status == 0:
        h_new = res.x[:n]
        t_new = res.x[n]
        return h_new, t_new
    else:
        return h0, None

def slp_solve(h0, n, target, max_iters=100, tol=1e-10):
    """Run SLP until convergence."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iters):
        h_new, t_lp = slp_iteration(h, n, target)
        if h_new is None:
            break
        h = project_feasible(h_new, target)
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
        # Check convergence
        if abs(score - best_score) < tol and it > 5:
            break
        if np.max(np.abs(h_new - h)) < 1e-12:
            break
    
    return best_h, best_score
```
