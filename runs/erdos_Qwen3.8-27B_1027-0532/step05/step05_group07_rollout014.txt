def gradient_step(h, n, target, alpha, step_scale):

    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-15:
            continue
        k = m - (n - 1)
        idx1 = np.arange(n) - k
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[valid1] += w * (1.0 - h[idx1[valid1]])
        idx2 = np.arange(n) + k
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):
            grad[valid2] -= w * h[idx2[valid2]]
    
    grad = grad / alpha
    h_new = h - step_scale * grad
    h_new = project_feasible(h_new, target)

        n_coarse = max(n // 4, 100)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        deadline_coarse = start_time + (budget_s - 25.0) * 0.72
        h_coarse, sc_coarse = paired_coord_desc(h_coarse, n_coarse, target_coarse,
                                                 deadline_coarse, rng,
                                                 recompute_interval=20, sa_temp=0.005,
                                                 sa_cooling=0.99)
        h_up = interpolate_to(h_coarse, n, target)
        h_up = slsqp_polish(h_up, n, target, [100, 500, 2000], 1500, 1e-16)
        deadline_up = start_time + (budget_s - 25.0) * 0.80
        h_up, sc_up = paired_coord_desc(h_up, n, target, deadline_up, rng,
                                         recompute_interval=20, sa_temp=0.0)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 6: Random restarts with varying perturbation
    deadline_p6 = start_time + (budget_s - 25.0) * 0.92
    noise_scales = [0.015, 0.025, 0.04, 0.01, 0.05, 0.02, 0.03, 0.012]
    for attempt in range(len(noise_scales)):
        if tleft() < 60:
            break
        h_pert = best_h + rng.randn(n) * noise_scales[attempt]
        h_pert = project_feasible(h_pert, target)
        sub_deadline = min(deadline_p6, time.time() + 30.0)
        h_res, sc_res = paired_coord_desc(h_pert, n, target, sub_deadline, rng,
                                           recompute_interval=15, sa_temp=0.003,
                                           sa_cooling=0.998)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res

    # Phase 7: Final L-BFGS polish
    if tleft() > 80:
        h_lbfgs, sc_lbfgs = lbfgs_polish(best_h, n, target, [200, 1000, 5000], 1000, 1e-15)
        if sc_lbfgs < best_score:
            best_score = sc_lbfgs
            best_h = h_lbfgs

    # Phase 8: Final SLSQP + CD
    if tleft() > 40:
        h_final = slsqp_polish(best_h, n, target, [500, 2000, 10000], 1000, 1e-17)
        sc = compute_c5(f_final if False else h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if tleft() > 15:
        h_final, sc_final = paired_coord_desc(best_h, n, target,
                                               start_time + budget_s - 10.0, rng,
                                               recompute_interval=15, sa_temp=0.0)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
