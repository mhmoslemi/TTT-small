import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta_single(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def lse_max(corr, alpha):
    """Smooth maximum using log-sum-exp."""
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return lse_max(corr, alpha)

def lbfgs_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    """Refine using L-BFGS-B with log-sum-exp smooth max."""
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)

def topk_greedy(h, max_iters, rng, time_limit, start_time, top_k=3):
    """Greedy that targets the top-k worst shifts simultaneously."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    stall = 0
    max_stall = 500
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        # Get top-k worst shifts
        top_indices = np.argsort(full)[-top_k:][::-1]
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Build combined scoring for each candidate swap
        # For each (i, j) swap, score = weighted sum of delta at top-k shifts
        # Higher score = more reduction
        
        # For efficiency, precompute the contribution vectors
        # A[i] = sum over top-k shifts k of h[i+k] (for i in ones)
        # B[j] = sum over top-k shifts k of h[j-k] (for j in zeros)
        # The delta at shift k for swap (i,j) is h[i+k]-h[j+k]+h[i-k]-h[j-k]
        # We want this to be negative for the worst shifts
        
        # Simpler: just use the primary (worst) shift as before, but with a penalty
        # for increasing other top-k shifts
        max_idx = top_indices[0]
        k_star = int(max_idx - (n - 1))
        
        if k_star == 0:
            stall += 1
            continue

        A = np.zeros(n)
        for ki in top_indices[:min(top_k, 3)]:
            kk = int(ki - (n - 1))
            if kk == 0:
                continue
            idx = positions + kk
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]
            idx = positions - kk
            valid = (idx >= 0) & (idx < n)
            A[valid] += h[idx[valid]]

        B = np.zeros(n)
        for ki in top_indices[:min(top_k, 3)]:
            kk = int(ki - (n - 1))
            if kk == 0:
                continue
            idx = positions - kk
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]
            idx = positions + kk
            valid = (idx >= 0) & (idx < n)
            B[valid] += h[idx[valid]]

        # Score: we want small A[i] (removing i doesn't help much at other shifts)
        # and large B[j] (adding j helps a lot)
        score = A[ones]  # smaller is better for i
        top_i = ones[np.argsort(score)[:min(20, len(ones))]]
        score_j = B[zeros]  # larger is better for j
        top_j = zeros[np.argsort(-score_j)[:min(20, len(zeros))]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta_single(h, i, j)
                new_max = float((full + delta).max())
                max_val = float(full.max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            stall += 1

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2 * n - 1][::-1]


def compute_delta_single(h, i, j):
    n = len(h)
    result = np.zeros(2 * n - 1)
    k = np.arange(2 * n - 1) - (n - 1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def multi_swap_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time, max_swap):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 100000
    k_arr = np.arange(2 * n - 1) - (n - 1)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if ratio < 0.3:
            swap_size = 1 + rng.randint(0, max_swap)
        elif ratio < 0.7:
            swap_size = 1 + rng.randint(0, max(1, max_swap - 1))
        else:
            swap_size = 1
        swap_size = min(swap_size, len(ones), len(zeros))

        idx_ones = rng.choice(len(ones), size=swap_size, replace=False)
        idx_zeros = rng.choice(len(zeros), size=swap_size, replace=False)
        IS = ones[idx_ones]
        JS = zeros[idx_zeros]

        delta = np.zeros(2 * n - 1)
        for ii in IS:
            idx = ii + k_arr
            mask = (idx >= 0) & (idx < n)
            delta[mask] -= (1.0 - h[idx[mask]])
            idx2 = ii - k_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            delta[mask2] += h[idx2[mask2]]
        for jj in JS:
            idx = jj + k_arr
            mask = (idx >= 0) & (idx < n)
            delta[mask] += (1.0 - h[idx[mask]])
            idx2 = jj - k_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            delta[mask2] -= h[idx2[mask2]]

        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            for ii in IS:
                h[ii] = 0.0
            for jj in JS:
                h[jj] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def smooth_obj_lse(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)


def lbfgs_refine(h_start, n, target, alphas, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy().astype(float)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj_lse, h_cur, args=(alpha,),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            best_h = h_ref

    # Phase 7: Final focused SA on binary
    if time_rem(start_time, budget_s) > 40:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        allowed = min(30, time_rem(start_time, budget_s) - 15)
        n_sa_f = int(min(500000, allowed * 35000))
        h_sa_f, c5_f = focused_sa(h_bin_final, n_sa_f, 0.008, 0.00002,
                                   rng, time_limit=allowed, start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    # Phase 8: Final L-BFGS-B
    if time_rem(start_time, budget_s) > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = min(20, time_rem(start_time, budget_s) - 10)
        h_ref, sc = lbfgs_refine(best_h, n_use, target_use,
                                  [2000, 5000, 10000], 400,
                                  time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
