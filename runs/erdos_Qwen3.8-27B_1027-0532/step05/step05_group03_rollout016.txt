    no_improve = 0
    for p in range(max_passes):
        one_idx = np.where(h_cur == 1)[0]
        zero_idx = np.where(h_cur == 0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        n_a = min(batch_size, len(one_idx))
        n_b = min(batch_size, len(zero_idx))
        a_batch = rng.choice(one_idx, n_a, replace=False)
        b_batch = rng.choice(zero_idx, n_b, replace=False)
        best_delta = 1e-12
        best_a = -1
        best_b = -1
        best_new_corr = None
        for ai in a_batch:
            a_int = int(ai)
            fa = f[a_int]
            ga = g[a_int]
            for bi in b_batch:
                b_int = int(bi)
                if a_int == b_int:
                    continue
                fb = f[b_int]
                gb = g[b_int]
                dc = (fa - fb) + (ga - gb)
                new_corr = corr + dc
                new_max = float(np.max(new_corr))
                delta = new_max - cur_score
                if delta < -best_delta:
                    best_delta = -delta
                    best_a = a_int
                    best_b = b_int
                    best_new_corr = new_corr
        if best_a >= 0:
            old_a = h_cur[best_a]
            old_b = h_cur[best_b]
            h_cur[best_a] = 0.0
            h_cur[best_b] = 1.0
            update_influence(f, g, best_a, old_a, 0.0, n)
            update_influence(f, g, best_b, old_b, 1.0, n)
            corr = best_new_corr
            cur_score = cur_score - best_delta
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                break
    return h_cur, cur_score * dx

def simulated_annealing_binary(h, target_sum, n_steps=200000, T_init=0.5, T_final=0.01, seed=42):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    corr_cur = compute_corr(h_cur)
    score_cur = float(np.max(corr_cur)) * dx
    best_h = h_cur.copy()
    best_score = score_cur
    best_corr = corr_cur.copy()
    rng = np.random.RandomState(seed)
    for step in range(n_steps):
        T = T_init * (T_final / T_init) ** (step / n_steps)
        one_idx = np.where(h_cur == 1)[0]
        zero_idx = np.where(h_cur == 0)[0]
        a = int(rng.choice(one_idx))
        b = int(rng.choice(zero_idx))
        dc_a = np.zeros(2 * n - 1)
        dc_b = np.zeros(2 * n - 1)
        for ell_idx in range(2 * n - 1):
            ell = ell_idx - (n - 1)
            ip_a = a + ell
            im_a = a - ell
            if 0 <= ip_a < n:
                dc_a[ell_idx] -= (1.0 - h_cur[ip_a])
            if 0 <= im_a < n:
                dc_a[ell_idx] += h_cur[im_a]
            ip_b = b + ell
            im_b = b - ell
            if 0 <= ip_b < n:
                dc_b[ell_idx] += (1.0 - h_cur[ip_b])
            if 0 <= im_b < n:
                dc_b[ell_idx] -= h_cur[im_b]
        new_corr = corr_cur + dc_a + dc_b
        new_score = float(np.max(new_corr)) * dx
        delta = new_score - score_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h_cur[a] = 0.0
            h_cur[b] = 1.0
            corr_cur = new_corr
            score_cur = new_score
            if score_cur < best_score:
                best_score = score_cur
                best_h = h_cur.copy()
                best_corr = new_corr.copy()
    return best_h, best_score

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def cons_eq(x):
    return np.sum(x) - _g_target

def cons_jac(x):
    return np.ones(_g_n)

def slsqp_optimize(h, alphas, maxiter, ftol, time_limit=120):
    global _g_n, _g_target, _g_alpha
    t0 = time.time()
    _g_n = len(h)
    _g_target = _g_n / 2.0
    bounds = [(0.0, 1.0)] * _g_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() - t0 > time_limit:
            break
        _g_alpha = float(a)
        try:
            res = minimize(obj_soft_max, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def lbfgs_smooth_max(h_init, target, alphas, maxiter=5000, time_limit=120):
    t0 = time.time()
    n = len(h_init)
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        def obj(x):
            h = project_feasible(x, target)
            corr = compute_corr(h)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        def grad(x):
            h = project_feasible(x, target)
            corr = compute_corr(h)
            c_max = float(np.max(corr))
            shifted = alpha * (corr - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            weights = np.exp(shifted)
            weights /= np.sum(weights)
            dx_val = 2.0 / n
            g = np.zeros(n)
            nlags = len(corr)
            for ell_idx in range(nlags):
                ell = ell_idx - (n - 1)
                w = weights[ell_idx]
                if ell == 0:
                    g += w * dx_val * (2.0 * h - 1.0)
                elif ell > 0:
                    upper = n - 1 - ell
                    idx1 = np.arange(0, upper + 1)
                    g[idx1] += w * dx_val * (1.0 - h[idx1 + ell])
                    idx2 = np.arange(ell, n)
                    g[idx2] -= w * dx_val * h[idx2 - ell]
                else:
                    ell_abs = -ell
                    lower = ell_abs
                    idx1 = np.arange(lower, n)
                    g[idx1] += w * dx_val * (1.0 - h[idx1 - ell_abs])
                    idx2 = np.arange(0, n - ell_abs)
                    g[idx2] -= w * dx_val * h[idx2 + ell_abs]
            return g
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj, h_cur, method='L-BFGS-B', jac=grad,
                           bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                                   'gtol': 1e-12, 'maxfun': 100000})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def lag_targeted_surgery(h, target, max_steps=500, n_lags=3, eps_init=0.002):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    eps = eps_init
    for step in range(max_steps):
        corr = compute_corr(h_cur)
        top_lags = np.argsort(corr)[::-1][:n_lags]
        grad = np.zeros(n)
        for ell_idx in top_lags:
            ell = int(ell_idx) - (n - 1)
            if ell == 0:
                grad += dx * (2.0 * h_cur - 1.0)
            elif ell > 0:
                upper = n - 1 - ell
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += dx * (1.0 - h_cur[idx1 + ell])
                idx2 = np.arange(ell, n)
                grad[idx2] -= dx * h_cur[idx2 - ell]
            else:
                ea = -ell
                lower = ea
                idx1 = np.arange(lower, n)
                grad[idx1] += dx * (1.0 - h_cur[idx1 - ea])
                idx2 = np.arange(0, n - ea)
                grad[idx2] -= dx * h_cur[idx2 + ea]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-14:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.008)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, cur_score

def subgradient_momentum(h, n_steps, n_active, lr_init, lr_decay, momentum, temp):
    global _g_n, _g_target
    _g_n = len(h)
    _g_target = _g_n / 2.0
    n = _g_n
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    velocity = np.zeros(n)
    for step in range(n_steps):
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k) - (n - 1)
            if k == 0:
                grad[:] += wk * dx * (2.0 * h_cur - 1.0)
            elif k > 0:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= wk * dx * h_cur[idx2 - k]
            else:
                ea = -k
                lower = ea
                idx1 = np.arange(lower, n)
                grad[idx1] += wk * dx * (1.0 - h_cur[idx1 - ea])
                idx2 = np.arange(0, n - ea)
                grad[idx2] -= wk * dx * h_cur[idx2 + ea]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, _g_target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.003, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def tleft(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_binary = binary_init(h_init, target)
    best_score = min(best_score, compute_c5(h_binary))
    if compute_c5(h_binary) < best_score:
        best_h = h_binary

    if tleft(start_time, budget_s) > 800:
        h_bs, sc_bs = binary_local_search_vectorized(h_binary, target, max_passes=1500, batch_size=80, seed=seed)
        if sc_bs < best_score:
            best_score = sc_bs
            best_h = h_bs

    if tleft(start_time, budget_s) > 600:
        h_sa, sc_sa = simulated_annealing_binary(h_best if 'h_best' in dir() else best_h, target,
                                                  n_steps=150000, T_init=0.3, T_final=0.005, seed=seed+1)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa

    if tleft(start_time, budget_s) > 400:
        h_cont, sc_cont = slsqp_optimize(best_h, [100, 500, 2000, 10000], 3000, 1e-15, time_limit=150)
        if sc_cont < best_score:
            best_score = sc_cont
            best_h = h_cont

    if tleft(start_time, budget_s) > 300:
        h_lb, sc_lb = lbfgs_smooth_max(best_h, target, [200, 1000, 5000, 20000], maxiter=3000, time_limit=120)
        if sc_lb < best_score:
            best_score = sc_lb
            best_h = h_lb

    if tleft(start_time, budget_s) > 250:
        h_surg, sc_surg = lag_targeted_surgery(best_h, target, max_steps=400, n_lags=4, eps_init=0.0015)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_sg, sc_sg = subgradient_momentum(best_h, 400, 5, 0.003, 0.997, 0.3, 0.004)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    if tleft(start_time, budget_s) > 200:
        h_cont2, sc_cont2 = slsqp_optimize(best_h, [5000, 20000, 100000], 2000, 1e-16, time_limit=100)
        if sc_cont2 < best_score:
            best_score = sc_cont2
            best_h = h_cont2

    if tleft(start_time, budget_s) > 150:
        h_lb2, sc_lb2 = lbfgs_smooth_max(best_h, target, [5000, 20000, 100000], maxiter=5000, time_limit=80)
        if sc_lb2 < best_score:
            best_score = sc_lb2
            best_h = h_lb2

    if tleft(start_time, budget_s) > 100:
        deadline_p7 = start_time + (budget_s - 40.0) * 0.82
        for attempt in range(8):
            if time.time() > deadline_p7 or tleft(start_time, budget_s) < 60:
                break
            h_pert = best_h.copy()
            n_cur = len(h_pert)
            target_cur = n_cur / 2.0
            if attempt % 3 == 0:
                h_pert += rng.randn(n_cur) * 0.05
            elif attempt % 3 == 1:
                corr = compute_corr(best_h)
                top_idx = np.argsort(corr)[::-1][:3]
                for ell_idx in top_idx:
                    ell = int(ell_idx) - (n_cur - 1)
                    if ell > 0:
                        shift = rng.uniform(-0.08, 0.08)
                        block = max(5, n_cur // 10)
                        for b in range(n_cur // block):
                            h_pert[b*block:min((b+1)*block, n_cur)] += shift * np.sin(2*np.pi*b/5 + rng.rand())
                    elif ell < 0:
                        shift = rng.uniform(-0.08, 0.08)
                        h_pert += shift * np.cos(2*np.pi*np.arange(n_cur)*(-ell)/n_cur)
                h_pert = project_feasible(h_pert, target_cur)
            else:
                n_flips = n_cur // 8
                flip_idx = rng.choice(n_cur, n_flips, replace=False)
                h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
            h_pert = project_feasible(h_pert, target_cur)
            h_ref, sc = slsqp_optimize(h_pert, [500, 2000, 10000], 1500, 1e-15, time_limit=60)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft(start_time, budget_s) > 80:
        h_sg3, sc_sg3 = subgradient_momentum(best_h, 500, 4, 0.002, 0.998, 0.4, 0.003)
        if sc_sg3 < best_score:
            best_score = sc_sg3
            best_h = h_sg3
        h_surg2, sc_surg2 = lag_targeted_surgery(best_h, target, max_steps=300, n_lags=2, eps_init=0.001)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    if tleft(start_time, budget_s) > 50:
        h_final, sc_final = slsqp_optimize(best_h, [10000, 50000], 1000, 1e-17, time_limit=40)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    if tleft(start_time, budget_s) > 20:
        h_surg3, sc_surg3 = lag_targeted_surgery(best_h, target, max_steps=150, n_lags=2, eps_init=0.0005)
        if sc_surg3 < best_score:
            best_score = sc_surg3
            best_h = h_surg3

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
