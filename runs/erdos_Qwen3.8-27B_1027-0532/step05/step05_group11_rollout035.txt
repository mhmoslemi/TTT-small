import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def fast_correlate(a, v):
    n = len(a)
    m = len(v)
    total = n + m - 1
    if total < 256:
        return np.correlate(a, v, mode='full')
    size = 1
    while size < total:
        size *= 2
    A = np.fft.rfft(a, size)
    Vr = np.fft.rfft(v[::-1], size)
    conv = np.fft.irfft(A * Vr, size)
    return conv[:total][::-1]

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_correlate(h, 1.0 - h)
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, alpha, target_sum, mu):
    """Compute smooth max of overlaps + penalty, and gradient."""
    n = len(h)
    dx = 2.0 / n
    ones = 1.0 - h
    corr = fast_correlate(h, ones)
    overlaps = corr * dx
    
    c_max = np.max(overlaps)
    shifted = alpha * (overlaps - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    smooth_val = c_max + np.log(total) / alpha
    weights = exp_shifted / total
    
    # Penalty term
    sum_dev = np.sum(h) - target_sum
    penalty = mu * sum_dev ** 2
    
    # Gradient
    grad = np.zeros(n)
    n_half = n - 1
    
    # Only process lags with significant weight
    significant = weights > 1e-10
    
    for k_idx in np.where(significant)[0]:
        w = weights[k_idx]
        lag = k_idx - n_half
        abs_lag = abs(lag)
        
        if abs_lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            L = lag
            grad[:n-L] += w * dx * (1.0 - h[L:n])
            grad[L:] -= w * dx * h[:n-L]
        else:
            L = -lag
            grad[L:] += w * dx * (1.0 - h[:n-L])
            grad[:n-L] -= w * dx * h[L:]
    
    # Penalty gradient
    grad += 2.0 * mu * sum_dev * np.ones(n)
    
    return smooth_val + penalty, grad

def lbfgs_smooth_max(h_start, n, target, alphas, mu, maxiter, ftol):
    """Optimize using L-BFGS-B with smooth max objective."""
    bounds = [(0.0, 1.0)] * n
    h_cur = project_feasible(h_start.copy(), target)
    
    for alpha in alphas:
        def obj(x):
            return smooth_max_val_grad(x, alpha, target, mu)[0]
        
        def grad(x):
            return smooth_max_val_grad(x, alpha, target, mu)[1]
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                                   'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

def active_set_lp(h_init, n, target, max_iter, time_limit, n_top, damping, reg, solver='SCS'):
    """LP-based active set refinement."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        top_idx = np.argsort(overlaps)[::-1][:n_top]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def nesterov_smooth(h_init, n, target, alpha, mu, max_steps, lr, momentum):
    """Nesterov accelerated gradient on smooth max objective."""
    h = project_feasible(h_init.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(max_steps):
        # Nesterov lookahead
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        _, grad = smooth_max_val_grad(y, alpha, target, mu)
        
        h_prev = h.copy()
        h = h - lr * grad
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5
