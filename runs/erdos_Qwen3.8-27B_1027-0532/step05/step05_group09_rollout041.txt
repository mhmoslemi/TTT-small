from scipy.optimize import minimize

def compute_c5(h):

    dx = 2.0 / n

def compute_corr_full(h):
    n = len(h)

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:] * dx
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = corr_pos - corr_neg
    return obj, grad

def find_threshold(vals, deficit):
    lo = vals.min() - 1.0
    hi = vals.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(vals - mid, 0.0, 1.0))
        if s > -deficit:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2.0

def diversify_start(h_base, n, target, rng, style):

def coord_descent_fast(h, target, deadline, max_sweeps=100):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    rng = np.random.RandomState(999)

        if time.time() > deadline:
            break
        order = rng.permutation(n)
        for idx in order:
            if time.time() > deadline:
                break
            if (sweep * n + idx) % 3 == 0:
                continue
            old_val = h_cur[idx]
            h_cur[idx] = 1.0 - old_val
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc >= cur_score:
                h_cur[idx] = old_val
                h_cur = project_feasible(h_cur, target)
    return h_cur, cur_score
