import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def topk_subgrad(h, dx, top_indices, n_half):
    n = len(h)
    grad = np.zeros(n)
    m = len(top_indices)
    for idx_k in top_indices:
        k = idx_k - n_half
        if k == 0:
            grad += (1.0 - 2.0 * h) * dx
        elif k > 0:
            n_rem = n - k
            grad[k:] += (1.0 - h[:n_rem]) * dx
            grad[:n_rem] -= h[k:] * dx
        else:
            m_neg = -k
            n_rem = n - m_neg
            grad[:n_rem] += (1.0 - h[m_neg:]) * dx
            grad[m_neg:] -= h[:n_rem] * dx
    return grad / m

            top_indices = np.argpartition(corr, -top_m)[-top_m:]
        else:

        v = mu * v + g

        if gnorm < 1e-14:

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:

        try:

                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
        except Exception:
            pass

    return h_cur, cur_score


def binary_round(h, target):
    n = len(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    return project_feasible(h_bin, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_src = np.linspace(0.0, 1.0, n_src)
    x_dst = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_dst, x_src, h_src)
    return project_feasible(h_out, target_dst)
