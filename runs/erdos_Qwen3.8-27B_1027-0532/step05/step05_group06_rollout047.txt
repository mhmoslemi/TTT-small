import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_slp_matrix(h, n, max_lag):
    """Build the coefficient matrix A and rhs for the SLP linearization."""
    ns_lags = 2 * max_lag + 1
    A = np.zeros((ns_lags, n))
    for i in range(ns_lags):
        lag = i - max_lag
        # coeff[j] = (1-h[j-lag]) if valid - h[j+lag] if valid
        # For j in [0, n):
        #   term1: j-lag in [0, n) -> (1-h[j-lag])
        #   term2: j+lag in [0, n) -> -h[j+lag]
        j = np.arange(n)
        # term1
        mask1 = (j - lag >= 0) & (j - lag < n)
        A[i, mask1] += (1.0 - h[j[mask1] - lag])
        # term2
        mask2 = (j + lag >= 0) & (j + lag < n)
        A[i, mask2] -= h[j[mask2] + lag]
    return A

def solve_slp(h_init, target, max_iter=100, max_lag=None, deadline=None):
    n = len(h_init)
    if max_lag is None:
        max_lag = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        # Build linearization matrix
        A = build_slp_matrix(h, n, max_lag)
        corr_current = compute_corr(h)
        # Extract the relevant lags
        center = n - 1
        lag_start = max(0, center - max_lag)
        lag_end = min(2 * n - 2, center + max_lag)
        corr_sel = corr_current[lag_start:lag_end + 1]
        
        # rhs[k] = A[k,:] · h - corr_current[k]
        rhs = A @ h - corr_sel
        
        # Solve LP
        h_var = cp.Variable(n)
        T_var = cp.Variable()
        ns = A.shape[0]
        constraints = [
            A @ h_var - T_var <= rhs,
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(cp.Minimize(T_var), constraints)
        
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.array(h_var.value).flatten()
        h_new = project_feasible(h_new, target)
        
        # Line search: try h_α = (1-α)h + α*h_new
        best_alpha = 0
        best_sc = best_score
        for alpha in [1.0, 0.75, 0.5, 0.25, 0.125]:
            h_trial = (1 - alpha) * h + alpha * h_new
            h_trial = project_feasible(h_trial, target)
            sc = compute_c5(h_trial)
            if sc < best_sc:
                best_sc = sc
                best_alpha = alpha
                if sc < best_score:
                    best_score = sc
                    best_h = h_trial.copy()
        
        if best_alpha == 0:
            break
        
        h = (1 - best_alpha) * h + best_alpha * h_new
        h = project_feasible(h, target)
        
        if np.max(np.abs(h_new - h)) < 1e-10:
            break
    
    return best_h, best_score


def subgradient_optimize(h_init, target, max_iter=2000, deadline=None, n_active=3):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    
    # Diminishing step size
    eta0 = 0.1
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = compute_corr(h)
        c_max = float(np.max(corr))
        
        # Find active lags
        active = np.where(corr > c_max - 1e-10)[0]
        if len(active) > n_active:
            top_idx = np.argsort(corr)[::-1][:n_active]
            active = top_idx
        
        # Compute subgradient (average over active lags)
        grad = np.zeros(n)
        for ki in active:
            lag = int(ki) - (n - 1)
            j = np.arange(n)
            mask1 = (j - lag >= 0) & (j - lag < n)
            grad[mask1] += (1.0 - h[j[mask1] - lag]) / len(active)
            mask2 = (j + lag >= 0) & (j + lag < n)
            grad[mask2] -= h[j[mask2] + lag] / len(active)
        
        # Step
        eta = eta0 / (1.0 + 0.01 * it)
        h_new = h - eta * grad
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def build_slp_matrix(h, n, max_lag):
    n_lags = 2 * max_lag + 1
    A = np.zeros((n_lags, n))
    j = np.arange(n)
    for i in range(n_lags):
        lag = i - max_lag
        if lag == 0:
            A[i, :] = 1.0 - 2.0 * h
        else:
            mask1 = (j - lag >= 0) & (j - lag < n)
            A[i, mask1] = (1.0 - h[j[mask1] - lag])
            mask2 = (j + lag >= 0) & (j + lag < n)
            A[i, mask2] -= h[j[mask2] + lag]
    return A
