    grad = np.zeros(n)
    for k in range(2*n-1):

        if w < 1e-15:

            # h factor: j in [0, n-1-shift], grad[j] += w*(1-h[j+shift])

def pairwise_cd(h, n, target, n_rounds, top_k, deadline):

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline):

        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        tau = 0.003
        weights = np.exp(diffs / tau)
        weights /= weights.sum()
        w_vec = np.zeros(2 * n - 1)
        for wk, k in zip(weights, top_idx):
            w_vec[k] = wk

```python
def corr_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    # Only process weights above threshold
    threshold = 1e-10
    active = np.where(weights > threshold)[0]
    for k in active:
        w = weights[k]
        shift = k - (n - 1)
        if shift >= 0:
            if n - shift > 0:
                grad[:n - shift] += w * (1.0 - h[shift:])
                grad[shift:] -= w * h[:n - shift]
        else:
            s = -shift
            if n - s > 0:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
    return grad
```
