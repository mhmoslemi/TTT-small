```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def solve_1d_minmax(a_vals, b_vals):
    """Solve min_{t in [0,1]} max_i (a_vals[i]*t + b_vals[i])."""
    K = len(a_vals)
    candidates = np.array([0.0, 1.0])
    # Add pairwise intersection points
    for i in range(K):
        for j in range(i + 1, K):
            da = a_vals[i] - a_vals[j]
            if abs(da) > 1e-15:
                t = (b_vals[j] - b_vals[i]) / da
                if -0.5 <= t <= 1.5:  # slightly relaxed, will clip
                    candidates = np.append(candidates, t)
    
    if len(candidates) == 0:
        return 0.5
    
    candidates = np.clip(candidates, 0.0, 1.0)
    candidates = np.unique(candidates)  # remove duplicates
    
    best_t = candidates[0]
    best_val = np.inf
    for t in candidates:
        val = np.max(a_vals * t + b_vals)
        if val < best_val:
            best_val = val
            best_t = t
    return best_t
```

```python
def coordinate_descent(h_init, target, n_sweeps, K, rng, time_limit, start_time):
    n = len(h_init)
    n_halves = n - 1
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(n_sweeps):
        if time.time() - start_time > time_limit:
            break
        
        # Compute overlaps and determine top-K lags
        overlaps = compute_all_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:K]
        top_lags = top_idx - n_halves
        
        # Randomize coordinate order
        order = rng.permutation(n)
        
        for j in order:
            # Compute a_k for each top lag
            a_vals = np.zeros(K)
            for i in range(K):
                k = top_lags[i]
                jm = j - k
                jp = j + k
                a_k = 0.0
                if 0 <= jm < n:
                    a_k += (1.0 - h[jm])
                if 0 <= jp < n:
                    a_k -= h[jp]
                a_vals[i] = a_k * (2.0 / n)
            
            # b_k = overlap[k] - a_k * h[j]
            # Note: overlaps are stale (computed at start of sweep)
            # but this is OK as an approximation
            b_vals = overlaps[top_idx] - a_vals * h[j]
            
            # Solve 1D subproblem
            t_star = solve_1d_minmax(a_vals, b_vals)
            h[j] = t_star
        
        # Project to sum constraint
        h = project_feasible(h, target)
        
        # Check improvement
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5
```

```python
def solve_1d_minmax(a_vals, b_vals):
    K = len(a_vals)
    # All pairwise intersections
    # t_ij = (b_j - b_i) / (a_i - a_j)
    A = a_vals[:, None] - a_vals[None, :]  # K x K
    B = b_vals[None, :] - b_vals[:, None]  # K x K (transposed)
    # Actually: t = (b_j - b_i) / (a_i - a_j)
    # A[i,j] = a_i - a_j
    # B[i,j] = b_j - b_i
    # t[i,j] = B[i,j] / A[i,j]
    
    with np.errstate(divide='ignore', invalid='ignore'):
        T = B / A
    T = np.where(np.abs(A) > 1e-15, T, np.nan)
    
    candidates = np.concatenate([[0.0, 1.0], T.flatten()])
    candidates = candidates[np.isfinite(candidates)]
    candidates = candidates[(candidates >= 0.0) & (candidates <= 1.0)]
    
    if len(candidates) == 0:
        return 0.5
    
    # Evaluate at each candidate
    # f(t) = max_i (a_i * t + b_i)
    # Vectorized: for each candidate t, compute a_vals * t + b_vals, take max
    # candidates shape: (M,)
    # a_vals * candidates shape: (K, M) if we do a_vals[:,None] * candidates[None,:]
    M = len(candidates)
    # This could be large: K * M values
    # K=30, M is at most K(K-1)/2 + 2 ≈ 452
    # So 30 * 452 = 13560 values. Fine.
    vals = np.max(a_vals[:, None] * candidates[None, :] + b_vals[:, None], axis=0)
    best_idx = np.argmin(vals)
    return candidates[best_idx]
```

Now let me think about the L-BFGS-B approach.

For L-BFGS-B, I need:
- Objective function F(h) = (1/α) * log(∑_{k∈S} exp(α * overlap_k)) + μ * (sum(h) - target)²
- Gradient ∇F(h)

The set S should be the top-K lags. But which K? I can use a fixed K=50 or make it adaptive.

For the gradient:
∇F(h) = ∑_{k∈S} w_k * ∇overlap_k + 2μ*(sum(h)-target)*1

where w_k = exp(α*overlap_k) / ∑_{k'∈S} exp(α*overlap_{k'})

And ∇overlap_k[j] = dx * [(1-h[j-k])*1{0≤j-k≤n-1} - h[j+k]*1{0≤j+k≤n-1}]

I can compute ∑_{k∈S} w_k * ∇overlap_k efficiently:
For each k in S, the contribution to the gradient is a shifted version of h and 1-h.

Specifically:
∑_{k∈S} w_k * dx * (1-h[j-k]) * 1{0≤j-k≤n-1} = dx * ∑_{k∈S} w_k * (1-h[j-k]) * 1{0≤j-k≤n-1}

This is like a weighted sum of shifted (1-h) vectors. I can precompute the shift indices.

For a given set of lags S = {k_1, ..., k_K}:
grad[j] = dx * ∑_{i=1}^{K} w_i * [(1-h[j-k_i])*1{0≤j-k_i≤n-1} - h[j+k_i]*1{0≤j+k_i≤n-1}]

I can compute this in O(n*K) time.

```python
def lbfgs_objective_and_grad(h, lags, alpha, mu, target, dx, n):
    """Compute F(h) and its gradient."""
    overlaps = compute_all_overlaps(h)
    top_idx = np.argsort(overlaps)[::-1][:len(lags)]
    # Actually, I should use fixed lags for stability
    # Let me use the passed lags
    
    lag_values = overlaps[lags + n - 1]
    
    # Log-sum-exp
    max_ov = np.max(lag_values)
    log_sum_exp = max_ov + np.log(np.sum(np.exp(alpha * (lag_values - max_ov))))
    F = log_sum_exp / alpha + mu * (np.sum(h) - target) ** 2
    
    # Weights
    w = np.exp(alpha * (lag_values - max_ov))
    w /= np.sum(w)
    
    # Gradient
    grad = np.zeros(n)
    for i in range(len(lags)):
        k = lags[i]
        wi = w[i]
        # (1-h[j-k]) for valid j-k
        jm = np.arange(n) - k  # j - k for j = 0..n-1
        valid_down = (jm >= 0) & (jm < n)
        grad[valid_down] += wi * dx * (1.0 - h[jm[valid_down]])
        
        # -h[j+k] for valid j+k
        jp = np.arange(n) + k  # j + k for j = 0..n-1
        valid_up = (jp >= 0) & (jp < n)
        grad[valid_up] -= wi * dx * h[jp[valid_up]]
    
    # Sum constraint gradient
    grad += 2.0 * mu * (np.sum(h) - target)
    
    return F, grad
```
