        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lambda x: lse_value(x, alpha, top_m), h, method='SLSQP',
                       jac=lambda x: lse_grad(x, alpha, top_m),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def subgradient_polish(h, n, target, n_steps, top_k, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score - 1e-15:
            best_score = c_curr
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
        top_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for m in top_idx:
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += (1.0 - h_cur[k:])
                g[k:] -= h_cur[:n - k]
            else:
                ell = -k
                g[ell:] += (1.0 - h_cur[:n - ell])
                g[:n - ell] -= h_cur[ell:]
        g *= dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        h_new = h_cur - lr * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, 0.08)
        else:
            lr *= lr_decay
            if lr < 1e-11:
                break
        if stall > 200:
            break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    stall = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = h_cur - eps * g
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-14:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.015, 0.025)
            stall = 0
        else:
            eps *= 0.82
            stall += 1
            if eps < 1e-11 or stall > 100:
                break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 8.0
    rng = np.random.RandomState(seed)

    best_score_native = compute_c5(h_init)

    n = 800
    target = n / 2.0

    h = interpolate_to(h_init, n, target)

    x = np.linspace(0, 2, n)
    h_sine1 = np.clip(0.5 + 0.3 * np.sin(2 * np.pi * 2 * x / 2), 0, 1)
    h_sine1 = project_feasible(h_sine1, target)
    h_sine2 = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * 4 * x / 2 + 0.5), 0, 1)
    h_sine2 = project_feasible(h_sine2, target)
    h_sine3 = np.clip(0.5 + 0.25 * np.cos(2 * np.pi * 3 * x / 2) + 0.15 * np.sin(2 * np.pi * 7 * x / 2), 0, 1)
    h_sine3 = project_feasible(h_sine3, target)

    candidates = [h, h_sine1, h_sine2, h_sine3]
    scores = [compute_c5(c) for c in candidates]
    best_idx = int(np.argmin(scores))
    best_h = candidates[best_idx].copy()
    best_score = scores[best_idx]

    alphas_p1 = [8, 25, 80, 250, 800, 2500, 8000]
    for alpha in alphas_p1:
        if time.time() > deadline - 500:
            break
        for mu in [5.0, 50.0, 300.0]:
            h_new = lbfgsb_round(best_h, n, target, alpha, mu, 250)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    alphas_p2 = [400, 1500, 6000, 25000]
    for alpha in alphas_p2:
        if time.time() > deadline - 400:
            break
        h_new = slsqp_round(best_h, n, target, alpha, 2500, 1e-15)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        for top_m in [2, 4, 6]:
            if time.time() > deadline - 350:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 400, 1e-14, top_m=top_m)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if time.time() < deadline - 250:
        for top_k in [2, 3, 5]:
            if time.time() > deadline - 200:
                break
            h_sg, sc_sg = subgradient_polish(best_h, n, target, 3000, top_k, 0.004, 0.996, deadline - 200)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    if time.time() < deadline - 200:
        h_surg, sc_surg = worst_lag_surgery(best_h, n, target, 600, 0.004, deadline - 200)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    if time.time() < deadline - 120:
        for alpha in [15000, 60000, 200000]:
            if time.time() > deadline - 100:
                break
            h_new = slsqp_round(best_h, n, target, alpha, 1500, 1e-16)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            for top_m in [2, 3]:
                if time.time() > deadline - 90:
                    break
                h_new = slsqp_round(best_h, n, target, alpha, 300, 1e-15, top_m=top_m)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new

    if time.time() < deadline - 350:
        n2 = 1000
        target2 = n2 / 2.0
        h2 = interpolate_to(best_h, n2, target2)
        best_h2 = h2.copy()
        best_score2 = compute_c5(h2)
        for alpha in [10, 50, 200, 800, 3000, 12000]:
            if time.time() > deadline - 120:
                break

                h2_new = lbfgsb_round(best_h2, n2, target2, alpha, mu, 200)

                if sc2 < best_score2:

        if time.time() < deadline - 100:
            for alpha in [3000, 15000, 60000]:
                if time.time() > deadline - 80:

                h2_new = slsqp_round(best_h2, n2, target2, alpha, 1200, 1e-15)

                if sc2 < best_score2:
                    best_score2 = sc2
                    best_h2 = h2_new
        if time.time() < deadline - 80:
            h_sg2, sc_sg2 = subgradient_polish(best_h2, n2, target2, 1500, 3, 0.003, 0.996, deadline - 70)
            if sc_sg2 < best_score2:
                best_score2 = sc_sg2
                best_h2 = h_sg2
        if time.time() < deadline - 60:
            h_surg2, sc_surg2 = worst_lag_surgery(best_h2, n2, target2, 300, 0.003, deadline - 55)
            if sc_surg2 < best_score2:
                best_score2 = sc_surg2
                best_h2 = h_surg2
        if time.time() < deadline - 45:
            h2_new = slsqp_round(best_h2, n2, target2, 100000, 800, 1e-16)

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)

    if best_score_native < final_score:
        return h_init, best_score_native, n_init

    return h_final, final_score, len(h_final)
