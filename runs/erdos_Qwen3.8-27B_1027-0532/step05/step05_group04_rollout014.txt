The number of 1s in positions k to n-1 is a_k + d_k = sum_{i=k}^{n-1} h[i]

And d_k = sum_{i=k}^{n-1} h[i] - a_k

This is a "balance" condition on the sequence.

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def thue_morse(n):
    """Generate Thue-Morse sequence of length n (0s and 1s)."""
    t = np.zeros(n, dtype=int)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    _g_target = target

                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_weights = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def thue_morse_seq(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t


def _obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _obj_weighted(x):
    corr = compute_corr(x)
    over = corr[_g_n - 1:_g_n - 1 + _g_n]
    return float(np.dot(_g_weights, over))


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def slsqp_weighted(h, n, target, n_outer, beta_init, beta_growth,
                   inner_iters, lr, deadline=None):
    global _g_n, _g_target, _g_weights
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    beta = beta_init
    dx = 2.0 / n
    for it in range(n_outer):
        if deadline is not None and time.time() > deadline:
            break

        over = corr[n - 1:n - 1 + n]
        max_over = float(np.max(over))
        _g_weights = np.exp(beta * (over - max_over))
        _g_weights /= _g_weights.sum()

def equalization_search(h, n, target, n_steps, top_k, temp_init, temp_decay,

        top_idx = sorted_idx[:top_k]
        top_vals = over[top_idx]

        diffs = top_vals - top_vals[0]

                h_cur = h_new
                cur_score = f_new

            temp *= temp_decay
            if a < 1e-8:
                break

    return h_cur, cur_score


def momentum_ewgd(h, n, target, n_steps, beta, lr_init, moment, deadline=None):
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    cur_score = compute_c5(h_cur)
    lr = lr_init

    for step in range(n_steps):

            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]

        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()

        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-8:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx

        v = moment * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new
            cur_score = sc_new
        else:
            v *= 0.5
            lr *= 0.9

    return h_cur, cur_score


def coordinate_polish(h, n, target, max_rounds, trials_per_round, rng, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    scale = 0.02

    for rd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(trials_per_round):
            i, j = rng.choice(n, size=2, replace=False)
            delta = rng.uniform(-scale, scale)
            hi = h_cur[i] + delta
            hj = h_cur[j] - delta
            if hi < -1e-10 or hi > 1 + 1e-10 or hj < -1e-10 or hj > 1 + 1e-10:
                continue
            h_new = h_cur.copy()
            h_new[i] = max(0.0, min(1.0, hi))
            h_new[j] = max(0.0, min(1.0, hj))
            h_new = project_feasible(h_new, target)
            f_new = compute_c5(h_new)
            if f_new < cur_score - 1e-13:
                h_cur = h_new
                cur_score = f_new
                improved = True

                break

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def de_search(n, target, pop_size, max_gens, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for _ in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_gens):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score
