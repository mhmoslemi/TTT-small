import time

# Global variables for callbacks
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def power_mean_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    temp = 0.005  # softmax temperature for focusing
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Compute weighted gradient focusing on worst lags
        sorted_idx = np.argsort(over)[::-1]
        top_k = sorted_idx[:n_worst]
        temp = max(0.002, temp * 0.995)  # slow cooling
        diffs = over[top_k] - over[top_k[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)

            k = int(k)

            if k > 0:

        grad *= dx

        else:
            lr *= lr_decay
            vel *= 0.5  # dampen velocity on failure
            if lr < 1e-12:
                break
    return best_h, best_score

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def eval_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    pen = 50.0 * (np.sum(x) - _g_target) ** 2
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha + pen


def slsqp_polish(h, n, target, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if time.time() > deadline:
            break
        _g_alpha = float(a)
        try:
            res = minimize(eval_obj, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score


def subgrad_momentum(h, n, target, n_steps, n_worst, lr_init, lr_decay, momentum, temp_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    vel = np.zeros(n)
    temp = temp_init
    stall = 0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score - 1e-14:
            best_score = c_curr
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                lr = lr_init * 0.5
                vel[:] = 0.0
                stall = 0

        sorted_idx = np.argsort(over)[::-1]
        top_k = sorted_idx[:n_worst]
        temp = max(0.001, temp * 0.997)
        diffs = over[top_k] - over[top_k[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, k in zip(weights, top_k):
            k = int(k)
            if k < n:
                grad[:n - k] += wk * (1.0 - h_cur[k:])
            if k > 0:
                grad[k:] -= wk * h_cur[:n - k]
        grad *= dx

        vel = momentum * vel - lr * grad
        h_new = h_cur + vel
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)

        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.008, lr_init * 5.0)
        else:
            lr *= lr_decay
            vel *= 0.3
            if lr < 1e-13:
                break
    return best_h, best_score


def fine_tune(h, n, target, n_steps, n_worst, lr_base, deadline):

        sorted_idx = np.argsort(over)[::-1]

                grad[k:] -= wk * h_cur[:n - k]

        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr - 1e-15:
            h_cur = h_new
            lr = min(lr * 1.02, lr_base * 3.0)
        else:
            lr *= 0.7

                break
    return best_h, best_score


def push_pairs(h, n, target, max_iters, deadline, rng):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = 0.005
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        top_lags = np.argsort(over)[::-1][:3]
        improved = False
        for _ in range(20):
            k_star = int(top_lags[rng.randint(3)])
            if k_star >= n:
                continue
            i = rng.randint(0, n - k_star)
            j = i + k_star
            if h_cur[i] > h_cur[j] + 1e-4:
                amt = min(eps, h_cur[i] - h_cur[j], 1.0 - h_cur[j])
                if amt < 1e-6:
                    continue
                h_trial = h_cur.copy()
                h_trial[i] -= amt
                h_trial[j] += amt
                sc = compute_c5(h_trial)
                if sc < cur_score - 1e-14:
                    h_cur = h_trial
                    cur_score = sc
                    improved = True
                    break
            elif h_cur[j] > h_cur[i] + 1e-4:
                amt = min(eps, h_cur[j] - h_cur[i], 1.0 - h_cur[i])
                if amt < 1e-6:
                    continue
                h_trial = h_cur.copy()
                h_trial[j] -= amt
                h_trial[i] += amt
                sc = compute_c5(h_trial)
                if sc < cur_score - 1e-14:
                    h_cur = h_trial
                    cur_score = sc
                    improved = True
                    break
        if not improved:
            eps *= 0.85
            if eps < 1e-8:
                break
    return h_cur, cur_score


def structured_perturb(h, n, target, rng, scale):
    x = np.linspace(0.0, 2.0, n)
    h_pert = h.copy()
    n_freqs = rng.randint(2, 5)
    for _ in range(n_freqs):
        freq = rng.uniform(1.0, 12.0)
        amp = scale * rng.uniform(0.4, 1.0)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h_pert += amp * np.sin(freq * np.pi * x + phase)
    return project_feasible(h_pert, target)


def threshold_boost(h, n, target, beta):
    h_t = 1.0 / (1.0 + np.exp(-beta * (2.0 * h - 1.0)))
    return project_feasible(h_t, target)
