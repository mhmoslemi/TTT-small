        corr = compute_corr(h_cur)
        over = corr[n - 1:n - 1 + n]
        k_star = int(np.argmax(over))
        grad = np.zeros(n)
        j_pos = np.arange(0, n - k_star)
        grad[j_pos] += (1.0 - h_cur[j_pos + k_star])
        j_neg = np.arange(k_star, n)
        grad[j_neg] -= h_cur[j_neg - k_star]
        grad *= dx
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-12:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.06, 0.03)
        else:
            eps *= 0.5
            if eps < 1e-9:
                break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 20.0

    def tleft():
        return hard_deadline - time.time()

    n_final = len(initial_h_values)
    target_final = n_final / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float).copy(), target_final)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coarse optimization at n=50
    n1 = 50
    t1 = n1 / 2.0
    h1_init = interpolate_to(best_h, n1, t1)
    pop1 = [h1_init.copy()]
    for _ in range(14):
        pop1.append(project_feasible(rng.rand(n1) * 0.6 + 0.2, t1))
    h1, sc1 = de_mr(n1, t1, 15, 400, 0.8, 0.7, start_time + 20, rng, init_pop=pop1)
    h1, sc1 = slsqp_lse(h1, n1, t1, [50, 200, 800, 3000], 3000, 1e-14, start_time + 30)

    # Phase 2: Medium at n=100
    if tleft() > 200:
        n2 = 100
        t2 = n2 / 2.0
        h2_init = interpolate_to(h1, n2, t2)
        pop2 = [h2_init.copy(), interpolate_to(best_h, n2, t2)]
        for _ in range(12):
            if rng.rand() < 0.4:
                p = h2_init + rng.randn(n2) * 0.06
            else:
                p = project_feasible(rng.rand(n2) * 0.5 + 0.25, t2)
            pop2.append(project_feasible(p, t2))
        h2, sc2 = de_mr(n2, t2, 14, 300, 0.7, 0.6, start_time + 60, rng, init_pop=pop2)
        h2, sc2 = slsqp_lse(h2, n2, t2, [100, 400, 1500, 5000], 3000, 1e-14, start_time + 80)
        h2, sc2 = slsqp_topk(h2, n2, t2, 5, 2000, start_time + 90)

    # Phase 3: n=200
    if tleft() > 180:
        n3 = 200
        t3 = n3 / 2.0
        h3_init = interpolate_to(h2, n3, t3)
        h3_alt = interpolate_to(best_h, n3, t3)
        pop3 = [h3_init.copy(), h3_alt.copy()]
        for _ in range(10):
            if rng.rand() < 0.3:
                p = h3_init + rng.randn(n3) * 0.05
            else:
                p = project_feasible(rng.rand(n3) * 0.5 + 0.25, t3)
            pop3.append(project_feasible(p, t3))
        h3, sc3 = de_mr(n3, t3, 12, 200, 0.6, 0.5, start_time + 130, rng, init_pop=pop3)
        h3, sc3 = slsqp_lse(h3, n3, t3, [200, 800, 3000, 10000], 2500, 1e-14, start_time + 160)
        h3, sc3 = slsqp_topk(h3, n3, t3, 8, 2000, start_time + 175)
        h3_up = interpolate_to(h3, n_final, target_final)
        sc3_up = compute_c5(h3_up)
        if sc3_up < best_score:
            best_score = sc3_up
            best_h = h3_up

    # Phase 4: Full resolution DE at n=400
    if tleft() > 300:
        h4_init = best_h.copy()
        pop4 = [h4_init.copy()]
        h4_alt = interpolate_to(h2, n_final, target_final)
        pop4.append(h4_alt)
        h4_alt2 = interpolate_to(h3, n_final, target_final)
        pop4.append(h4_alt2)
        for _ in range(10):
            ptype = rng.randint(5)
            pop4.append(structured_perturb(h4_init, ptype, rng, target_final))
        h4, sc4 = de_mr(n_final, target_final, 13, 60, 0.5, 0.4, min(start_time + 250, hard_deadline), rng, init_pop=pop4)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Intensive SLSQP with cycling alphas
    if tleft() > 250:
        alphas_cycle1 = [100, 500, 2000, 8000, 30000, 100000]
        h_ref, sc = slsqp_lse(best_h, n_final, target_final, alphas_cycle1, 3000, 1e-15, min(start_time + 400, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_ref, sc = slsqp_topk(best_h, n_final, target_final, 3, 3000, min(start_time + 430, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Subgradient polishing
    if tleft() > 150:
        h_sg, sc_sg = subgradient_polyak(best_h, n_final, target_final, 800, 4, 0.004, 0.996, min(start_time + 520, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref, sc = slsqp_lse(best_h, n_final, target_final, [1000, 5000, 20000, 80000], 2500, 1e-15, min(start_time + 580, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Worst-lag surgery + re-optimize
    if tleft() > 120:
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 400, 0.003, min(start_time + 640, hard_deadline))
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_ref, sc = slsqp_lse(best_h, n_final, target_final, [5000, 20000, 80000, 300000], 2500, 1e-16, min(start_time + 720, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Structured restarts + SLSQP
    if tleft() > 200:
        restart_best_h = best_h.copy()
        restart_best_score = best_score
        for attempt in range(10):
            if tleft() < 40:
                break
            ptype = attempt % 5
            h_pert = structured_perturb(restart_best_h, ptype, rng, target_final)
            h_ref, sc = slsqp_lse(h_pert, n_final, target_final, [500, 3000, 15000, 60000], 2000, 1e-15, min(start_time + 720 + attempt * 25, hard_deadline))
            if sc < restart_best_score:
                restart_best_score = sc
                restart_best_h = h_ref
        if restart_best_score < best_score:
            best_score = restart_best_score
            best_h = restart_best_h

    # Phase 9: Final top-k + high-alpha SLSQP
    if tleft() > 80:
        h_ref, sc = slsqp_topk(best_h, n_final, target_final, 2, 3000, min(start_time + 920, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_ref, sc = slsqp_lse(best_h, n_final, target_final, [50000, 200000, 500000], 2500, 1e-16, min(start_time + 1000, hard_deadline))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final subgradient + surgery
    if tleft() > 50:
        h_sg, sc_sg = subgradient_polyak(best_h, n_final, target_final, 500, 3, 0.002, 0.997, min(start_time + 1060, hard_deadline))
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_surg, sc_surg = worst_lag_surgery(best_h, n_final, target_final, 200, 0.001, min(start_time + 1100, hard_deadline))
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 11: Ultra-high alpha final polish
    if tleft() > 20:
        h_ref, sc = slsqp_lse(best_h, n_final, target_final, [100000, 500000, 1000000], 2000, 1e-16, hard_deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
