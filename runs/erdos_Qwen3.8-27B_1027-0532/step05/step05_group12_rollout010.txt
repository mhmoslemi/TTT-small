        return budget_s - 15.0 - (time.time() - t0)

    hi = np.array(initial_h_values, dtype=float)
    n = len(hi)
    tgt = n / 2.0

    best_h = project_feasible(hi.copy(), tgt)
    best_score = compute_c5(best_h)

    dl1 = t0 + budget_s * 0.28
    if time.time() < dl1:
        h1, s1 = alt_lp(best_h, n, tgt, 10000, dl1, 0.8, 1e-4, 0.95)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    dl2 = t0 + budget_s * 0.48
    if time.time() < dl2:
        h2, s2 = momentum_opt(best_h, n, tgt, dl2,
                              [50, 100, 300, 1000, 3000, 10000, 50000, 200000],
                              2000, [5, 8, 10, 15, 25, 50, 200, 2 * n - 1], 0.9)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    dl3 = t0 + budget_s * 0.58
    if time.time() < dl3:
        h3, s3 = focused_cd(best_h, n, tgt, dl3, 6)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    dl4 = t0 + budget_s * 0.70
    if time.time() < dl4:
        h4, s4 = alt_lp(best_h, n, tgt, 6000, dl4, 0.9, 1e-5, 0.92)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    dl5 = t0 + budget_s * 0.78
    if time.time() < dl5:
        h5, s5 = momentum_opt(best_h, n, tgt, dl5,
                              [50000, 200000, 1000000],
                              3000, [3, 5, 2 * n - 1], 0.95)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    dl6 = t0 + budget_s * 0.86
    if time.time() < dl6:
        h6, s6 = binary_swap_search(best_h, n, tgt, dl6, n_rounds=5)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    dl7 = t0 + budget_s * 0.92
    if time.time() < dl7:
        for noise in [0.01, 0.03, 0.05]:
            if time.time() > dl7:
                break
            hp = project_feasible(best_h + rng.randn(n) * noise, tgt)
            h7, s7 = focused_cd(hp, n, tgt, dl7, 5)
            if s7 < best_score:
                best_score = s7
                best_h = h7
            h7b, s7b = momentum_opt(h7, n, tgt, dl7,
                                    [1000, 10000, 100000],
                                    1000, [5, 10, 2 * n - 1], 0.9)
            if s7b < best_score:
                best_score = s7b
                best_h = h7b

    if tleft() > 100:
        dl8 = t0 + budget_s * 0.96
        for nt in [600, 1000]:
            if time.time() > dl8:
                break
            tt = nt / 2.0
            hi8 = interp_to(best_h, nt, tt)
            h8, s8 = alt_lp(hi8, nt, tt, 3000, dl8, 0.8, 1e-4, 0.95)
            if time.time() < dl8 - 30:
                h8b, s8b = focused_cd(h8, nt, tt, dl8 - 30, 8)
                if s8b < s8:
                    h8, s8 = h8b, s8b
            if s8 < best_score:
                best_score = s8
                best_h = h8
                n = nt
                tgt = tt

    dl9 = t0 + budget_s - 5
    if tleft() > 20:
        n_use = len(best_h)
        tgt_use = n_use / 2.0
        h9, s9 = momentum_opt(best_h, n_use, tgt_use, dl9,
                              [100000, 1000000],
                              2000, [3, 2 * n_use - 1], 0.95)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
