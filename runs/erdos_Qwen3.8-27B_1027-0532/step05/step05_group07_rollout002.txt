    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    f_cur = float(np.max(corr)) * dx
    
    best_h = h_cur.copy()
    best_f = f_cur
    
    T = 0.01  # initial temperature
    T_min = 1e-8
    
    # Use geometric cooling
    n_total_steps = 50000
    cooling_rate = (T_min / T) ** (1.0 / n_total_steps)
    
    for step in range(n_total_steps):
        if time.time() > deadline:
            break
        
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Determine range for delta
        # h[i] + delta in [0,1] => delta in [-h[i], 1-h[i]]
        # h[j] - delta in [0,1] => delta in [h[j]-1, h[j]]
        delta_max = min(1 - h_cur[i], h_cur[j])
        delta_min = max(-h_cur[i], h_cur[j] - 1)
        
        if delta_max - delta_min < 1e-10:
            continue
        
        # Propose delta
        # Scale with temperature
        delta = rng.uniform(delta_min, delta_max)
        
        # Compute new objective
        B_diff = compute_B(h_cur, i, n) - compute_B(h_cur, j, n)
        new_corr = corr + delta * B_diff
        f_new = float(np.max(new_corr)) * dx
        
        delta_f = f_new - f_cur
        
        # Accept or reject
        if delta_f < 0 or (T > T_min and rng.rand() < np.exp(-delta_f / T)):

            h_cur[j] -= delta
            corr = new_corr
            f_cur = f_new
            
            if f_cur < best_f:
                best_f = f_cur
                best_h = h_cur.copy()
        
        # Cool
        T *= cooling_rate
    
    return best_h, best_f

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_B(h, i, n):
    nl1 = 2 * n - 1
    B = np.zeros(nl1)
    ks = np.arange(n)
    mask1 = ks <= i
    mask2 = (i + ks) < n
    b_pos = np.zeros(n)
    if np.any(mask1):
        b_pos[mask1] += (1.0 - h[i - ks[mask1]])
    if np.any(mask2):
        b_pos[mask2] -= h[i + ks[mask2]]
    B[n - 1:n - 1 + n] = b_pos
    ms = np.arange(1, n)
    mask3 = (i + ms) < n
    mask4 = ms <= i
    b_neg = np.zeros(n - 1)
    if np.any(mask3):
        b_neg[mask3] += (1.0 - h[i + ms[mask3]])
    if np.any(mask4):
        b_neg[mask4] -= h[i - ms[mask4]]
    B[0:n - 1] = b_neg
    return B


def paired_coord_desc(h, n, target, deadline, rng, recompute_interval=50, pair_strategy='random'):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    best_score = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    pair_positions = np.arange(n)
    stall_count = 0
    sweep = 0

    while time.time() < deadline:
        sweep += 1
        if sweep % recompute_interval == 0 and sweep > 0:
            corr = compute_corr(h_cur)

        if pair_strategy == 'value_sort':
            order = np.argsort(h_cur)
            pairs = []
            for p in range(n // 2):
                i = order[p]
                j = order[n - 1 - p]
                pairs.append((int(i), int(j)))
        else:
            rng.shuffle(pair_positions)
            pairs = []
            for p in range(n // 2):
                i = int(pair_positions[2 * p])
                j = int(pair_positions[2 * p + 1])
                pairs.append((i, j))

        max_delta = 0.0
        for (i, j) in pairs:
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue

            Bi = compute_B(h_cur, i, n)
            Bj = compute_B(h_cur, j, n)
            C = Bi - Bj
            A = corr - h_cur[i] * Bi - h_cur[j] * Bj + Bj * c

            lo = 0.0
            if np.any(C > 0):
                hi = float(np.max(A + C * t_max))
            else:
                hi = float(np.max(A + C * t_max))

            for _ in range(50):
                mid = (lo + hi) * 0.5
                pos = C > 1e-14
                neg = C < -1e-14
                flat = ~(pos | neg)
                if np.any(flat) and float(np.max(A[flat])) > mid + 1e-13:
                    lo = mid
                    continue
                ub = t_max
                if np.any(pos):
                    candidate = float(np.min((mid - A[pos]) / C[pos]))
                    ub = min(ub, candidate)
                lb = t_min
                if np.any(neg):
                    candidate = float(np.max((mid - A[neg]) / C[neg]))
                    lb = max(lb, candidate)
                if lb <= ub + 1e-11:
                    hi = mid
                else:
                    lo = mid

            v = hi
            pos = C > 1e-14
            neg = C < -1e-14
            ub = t_max
            if np.any(pos):
                ub = min(ub, float(np.min((v - A[pos]) / C[pos])))
            lb = t_min
            if np.any(neg):
                lb = max(lb, float(np.max((v - A[neg]) / C[neg])))
            ub = min(ub, t_max)
            lb = max(lb, t_min)
            if lb > ub:
                t_star = (t_min + t_max) * 0.5
            else:
                t_star = lb + (ub - lb) * 0.5

            delta = t_star - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C
                max_delta = max(max_delta, abs(delta))

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_h = h_cur.copy()
            stall_count = 0
        else:
            stall_count += 1

        if stall_count > 150:
            pert_scale = min(0.05, 0.01 * (1 + stall_count / 200.0))
            h_cur = h_cur + rng.randn(n) * pert_scale
            h_cur = project_feasible(h_cur, target)
            corr = compute_corr(h_cur)
            stall_count = 0

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)


def simulated_annealing(h, n, target, deadline, rng):
    dx = 2.0 / n
    h_cur = h.copy()
    corr = compute_corr(h_cur)
    f_cur = float(np.max(corr)) * dx
    best_h = h_cur.copy()
    best_f = f_cur

    T = f_cur * 0.002
    T_min = 1e-12
    n_steps = 30000
    cool_rate = (T_min / T) ** (1.0 / max(n_steps, 1))

    for step in range(n_steps):
        if time.time() > deadline:
            break

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        dmin = max(-h_cur[i], h_cur[j] - 1.0)
        dmax = min(1.0 - h_cur[i], h_cur[j])
        if dmax - dmin < 1e-10:
            continue

        step_scale = min(1.0, (T / f_cur) * 2.0)
        delta = (dmax - dmin) * step_scale * rng.randn()
        delta = float(np.clip(delta, dmin, dmax))
        if abs(delta) < 1e-13:
            delta = (dmin + dmax) * 0.5 * (1 if rng.rand() < 0.5 else -1)

        Bi = compute_B(h_cur, i, n)
        Bj = compute_B(h_cur, j, n)
        new_corr = corr + delta * (Bi - Bj)
        f_new = float(np.max(new_corr)) * dx

        df = f_new - f_cur
        if df < 0 or (T > T_min and rng.rand() < np.exp(-df / T)):
            h_cur[i] += delta
            h_cur[j] -= delta
            corr = new_corr
            f_cur = f_new
            if f_cur < best_f:
                best_f = f_cur
                best_h = h_cur.copy()

        T *= cool_rate

    h_final = project_feasible(best_h, target)
    return h_final, compute_c5(h_final)


def obj_lse(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def cons_eq(x):
    return np.sum(x) - _g_target


def cons_jac(x):
    return np.ones(_g_n)


def slsqp_polish(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def upsample(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def downsample(h_src, n_dst, target_dst):
    n_src = len(h_src)

    return project_feasible(h_out, target_dst)
