```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, n, target, max_flips, deadline):
    """Try flipping individual bits (0<->1) to reduce max overlap."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    flips = 0
    while flips < max_flips and time.time() < deadline:
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        # Try flipping each position and see if it helps
        best_gain = 0
        best_pos = -1
        best_dir = 0
        for i in range(n):
            # Try setting h[i] to 1 (if currently 0) or to 0 (if currently 1)
            if h_cur[i] < 0.5:
                trial = h_cur.copy()
                trial[i] = 1.0
                trial = project_feasible(trial, target)
                sc = compute_c5(trial)
                gain = cur_score - sc
            else:
                trial = h_cur.copy()
                trial[i] = 0.0
                trial = project_feasible(trial, target)
                sc = compute_c5(trial)
                gain = cur_score - sc
            if gain > best_gain:

                best_pos = i
                best_dir = 1 if h_cur[i] < 0.5 else 0
            flips += 1
            if time.time() > deadline:

        if best_pos >= 0 and best_gain > 1e-12:
```

```python
def momentum_subgradient(h, n, target, n_steps, n_top, lr_init, momentum, deadline):
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top-k worst lags
        top_idx = np.argsort(over)[::-1][:n_top]
        # Soft weights
        diffs = over[top_idx] - over[top_idx[0]]
        temp = 0.005
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (2 * h_cur - 1)  # d/dh of h(1-h) = 1-2h
            else:
                j_pos = np.arange(0, n - k)
                grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
                j_neg = np.arange(k, n)
                grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        # Nesterov momentum step
        v = momentum * v - lr_init * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        
        # Adaptive step size
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            # Roll back
            v = 0.5 * momentum * v
            h_new = h_cur + 0.5 * v
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < c_curr:
                h_cur = h_new
            else:
                v = 0.0
    
    return best_h, best_score
```

        else:
            stall += 1
        if stall > 300:
            break
        top_idx = np.argsort(over)[::-1][:n_top]
        diffs = over[top_idx] - over[top_idx[0]]
        w = np.exp(diffs / temp)
        w /= w.sum()
        grad = np.zeros(n)
        for wk, k in zip(w, top_idx):
            k = int(k)
            if k == 0:
                grad += wk * (2.0 * h_cur - 1.0)
            else:
                grad[:n-k] += wk * (1.0 - h_cur[k:])
                grad[k:] -= wk * h_cur[:n-k]
        grad *= dx
        v = mom * v - lr * grad
        h_try = project_feasible(h_cur + v, target)
        sc = compute_c5(h_try)
        if sc < c_curr - 1e-15:
            h_cur = h_try
            lr = min(lr * 1.03, lr_init * 8.0)
        else:
            v = 0.5 * v
            lr *= 0.93
            if lr < 1e-12:
                break
    return best_h, best_score

def slsqp_polish(h, n, target, alphas, maxiter, ftol, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    dx = 2.0 / n
    for a in alphas:
        if time.time() > deadline:
            break
        alpha = float(a)
        
        def obj(x):
            over = compute_overlaps(x)
            c_max = float(np.max(over))
            shifted = alpha * (over - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        
        def grad(x):
            over = compute_overlaps(x)
            c_max = float(np.max(over))
            shifted = alpha * (over - c_max)
            shifted = np.clip(shifted, -700.0, 0.0)
            w = np.exp(shifted)
            w /= w.sum()
            g = np.zeros(n)
            for k in range(n):
                if w[k] < 1e-12:
                    continue
                if k == 0:
                    g += w[k] * (2.0 * x - 1.0)
                else:
                    g[:n-k] += w[k] * (1.0 - x[k:])
                    g[k:] -= w[k] * x[:n-k]
            return g * dx
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur, cur_score

def de_fast(n, target, pop_size, n_gen, f, cr, deadline, rng, init_pop):
    pop = np.array([project_feasible(p.copy(), target) for p in init_pop[:pop_size]], dtype=float)
    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_h = pop[int(np.argmin(scores))].copy()
    best_score = float(np.min(scores))
    idx = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx != i
            usable = idx[mask]
            if len(usable) < 3:
                continue
            a, b, c = rng.choice(usable, size=3, replace=False)
            v = pop[a] + f * (pop[b] - pop[c])
            jr = rng.randint(n)
            trial = v.copy()
            r = rng.rand(n)
            trial[r > cr] = pop[i][r > cr]
            trial[jr] = v[jr]
            trial = project_feasible(trial, target)
            sc = compute_c5(trial)
            if sc <= scores[i]:
                pop[i] = trial
                scores[i] = sc
                if sc < best_score:
                    best_score = sc
                    best_h = trial.copy()
    return best_h, best_score
