def compute_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = np.correlate(h, 1.0-h, mode='full') * dx
    # Softmax weights
    max_ov = np.max(overlaps)
    exp_ov = np.exp(alpha * (overlaps - max_ov))
    p = exp_ov / np.sum(exp_ov)
    # Gradient: for each i, sum over k of p[k] * dC(k)/dh[i]
    grad = np.zeros(n)
    n_half = n - 1
    for idx in range(len(overlaps)):
        k = idx - n_half  # lag
        w = p[idx]
        if w < 1e-15:
            continue
        # dC(k)/dh[i] for all i
        if k >= 0:
            # i ranges from 0 to n-1-k
            valid_i = range(n - k)
            # contribution from h[i] as h[j]: (1-h[i+k])
            grad[:n-k] += w * (1.0 - h[k:n])
            # contribution from h[i] as (1-h[j+k]) where j=i-k, i ranges from k to n-1
            grad[k:] += w * (-h[:n-k])
        else:
            kk = -k
            # i ranges from kk to n-1
            # contribution from h[i] as h[j]: (1-h[i+k]) = (1-h[i-kk])
            grad[kk:] += w * (1.0 - h[:n-kk])
            # contribution from h[i] as (1-h[j+k]) where j=i-k=i+kk, i ranges from 0 to n-1-kk
            grad[:n-kk] += w * (-h[kk:n])
    grad *= dx
    return grad, np.max(overlaps)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_onto_simplex_box(h, target_sum):
    # Project onto {x : 0 <= x_i <= 1, sum(x) = target_sum}
    # Water-filling / bisection
    def f(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0, 1)

def smooth_max_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = np.max(overlaps)
    exp_ov = np.exp(alpha * (overlaps - max_ov))
    p = exp_ov / np.sum(exp_ov)
    
    # Compute gradient
    n_total = len(overlaps)
    n_half = n - 1
    grad = np.zeros(n)
    
    for idx in range(n_total):
        k = idx - n_half
        w = p[idx]
        if w < 1e-12:
            continue
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] += w * (-h[kk:])
    
    grad *= dx
    smooth_val = (max_ov + np.log(np.sum(exp_ov)) / alpha)
    return smooth_val, grad

    h = h_init.copy()
    y = h.copy()
    t_k = 1.0

        # Let's estimate it numerically
        val, g = smooth_max_and_grad(h, alpha)
        L = np.linalg.norm(g) * 10 + 1.0
        if lr is not None:
            L = 1.0 / lr
    
    best_h = h.copy()
    best_val = compute_c5(h)

        if time.time() - start > time_limit:
            break
        
        val, grad = smooth_max_and_grad(y, alpha)
        
        # FISTA update
        z = y - (1.0 / L) * grad
        h_new = project_onto_simplex_box(z, target)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))

        h = h_new
        t_k = t_new

        c5 = compute_c5(h)
        if c5 < best_val - 1e-15:
            best_val = c5
            best_h = h.copy()

        if it % 100 == 0:

            val_h, _ = smooth_max_and_grad(h, alpha)
            val_z, grad_z = smooth_max_and_grad(h_new, alpha)
            # Not a great check, just keep L

Since C(k) is linear in h[i] (for fixed other coordinates), max_k C(k) as a function of h[i] is the max of linear functions, which is convex. So the optimal h[i] for fixed others is at h[i] = 0 or h[i] = 1 (or possibly any value if the max is flat).

This confirms that coordinate descent on this problem always gives a binary answer at convergence. And the "cycle" is: for each i, try h[i]=0 and h[i]=1 (with sum constraint adjustment), pick the better one.

But "with sum constraint adjustment" is the tricky part. If we set h[i]=0 and h[i] was 1, we need to increase some other h[j] by 1 (or increase multiple by fractions). The simplest is to find a j with h[j]=0 and set h[j]=1 (a pair swap).

So the local search is: try all pair swaps and pick the best improving one. This is what I had before.

For efficiency, instead of trying ALL pair swaps (O(n⁴)), I can:
1. Compute the gradient of the smooth max
2. The gradient gives a "score" for each coordinate
3. Only try the top few candidates

This should work well in practice.

Let me now write the final, clean version of my code. I'll focus on:
1. Efficient FISTA-like optimization (gradient descent with momentum for the nonconvex smooth max)
2. Binary local search with gradient-guided pair swaps
3. Multiple restarts
4. Multi-resolution

One more thought: since the problem is nonconvex, **momentum** (as in Nesterov acceleration) might help, but it's not guaranteed to converge. I'll use it cautiously.

The key advantage of LP is that each subproblem is convex (and solved exactly). My approach using the smooth max gradient is a first-order method that might be less effective per iteration but faster per iteration.

Actually, let me reconsider the LP approach from the previous program. The key idea is:
1. Fix g = 1-h
2. Solve: min t s.t. Σ h[j] * g[j+k] * dx ≤ t for top-k lags, 0≤h≤1, Σh=n/2

This is a block coordinate descent where we alternate between h and the "active constraints" (top lags). It should converge to a local minimum of the relaxed problem.

    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling via bisection
    def f(lam):
        return np.sum(np.clip(h - lam, 0, 1)) - target_sum
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def smooth_max_value_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    max_ov = float(np.max(overlaps))
    exp_ov = np.exp(alpha * (overlaps - max_ov))
    p = exp_ov / np.sum(exp_ov)
    smooth_val = max_ov + np.log(np.sum(exp_ov)) / alpha
    
    n_half = n - 1
    grad = np.zeros(n)
    n_total = 2 * n - 1
    for idx in range(n_total):
        k = idx - n_half
        w = p[idx]
        if w < 1e-10:
            continue
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] += w * (-h[:n-k])
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] += w * (-h[kk:])
    grad *= dx
    return smooth_val, grad

def gradient_descent(h_init, n, target, max_iter, time_limit,
                     alpha=100.0, lr=0.01, momentum=0.9):
    start = time.time()
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        _, grad = smooth_max_value_and_grad(h, alpha)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_feasible(h_new, target)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            h = h_new
        else:
            # Reset velocity on failure
            v *= 0.5
            h = h  # don't update
    
    return best_h, best_c5

def lp_phase(h_init, n, target, max_iter, time_limit, n_top=200,
             damping=0.9, solver='SCS'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        n_half = n - 1
        # Get lags with overlap > 0.85 * max
        threshold = 0.85 * np.max(overlaps)
        active_idx = np.where(overlaps >= threshold)[0]
        lags = active_idx - n_half
        # Limit to top n_top
        if len(lags) > n_top:
            top_orders = np.argsort(overlaps[active_idx])[::-1][:n_top]
            lags = lags[top_orders]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        obj = t_var + 1e-6 * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h, n, target, max_rounds, time_limit, rng,

    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
