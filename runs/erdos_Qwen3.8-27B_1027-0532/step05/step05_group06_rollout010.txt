def lp_step(h, target, top_indices, overlaps):
    """One step of the Linearized Active Set method.
    Returns (h_new, C_lp) where C_lp is the LP's objective value.
    """
    n = len(h)
    K = len(top_indices)
    
    # Build LP
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize C
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i, idx in enumerate(top_indices):
        lag = idx - (n - 1)
        g = compute_overlap_gradient(h, lag)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = float(np.dot(g, h)) - overlaps[idx]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return None, None
    
    h_new = res.x[:n]
    C_new = res.x[n]
    return h_new, C_new
