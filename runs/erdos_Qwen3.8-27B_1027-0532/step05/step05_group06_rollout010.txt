- d^2/(dh[j]dh[j-s]) = -1

So the Hessian is a banded matrix with -1's at positions ±s for each active shift s.

For the 1D subproblem (fixing all h_i for i≠j):
f(h_j) = ∑_{s∈S} [1_{j<n-s}*h_j - h_j*h[j+s]*1_{j<n-s} - h_j*h[j-s]*1_{j>=s}]
= h_j * (∑_{s∈S} 1_{j<n-s} - ∑_{s∈S} h[j+s]*1_{j<n-s} - ∑_{s∈S} h[j-s]*1_{j>=s})

This is LINEAR in h_j (no h_j^2 term because the only quadratic terms are h[j]*h[j±s] which don't involve h_j^2 for s≠0).

If s=0 is in S: f(h_j) includes 1*h_j - 2*h_j^2 (from overlap_0 = ∑h(1-h)).
So if s=0 is active, we have a -2h_j^2 term.

With the constraint (handled by projection), the 1D problem is:
If s=0 in S: minimize a*h_j - 2*h_j^2 (concave!) → minimum at h_j=0 or h_j=1
If s=0 not in S: minimize a*h_j (linear) → minimum at h_j=0 or h_j=1

So in both cases, the optimal h_j is at a boundary! This confirms that coordinate descent will push towards binary solutions.

But we need to maintain the sum constraint. So the practical approach is:
1. For each j, determine if setting h_j to 0 or 1 (whichever is better) while adjusting the sum constraint improves the objective.
2. This is a "swap" operation: change h_j from its current value to 0 or 1, and adjust another coordinate to maintain the sum.

Actually, a simpler approach: just do the coordinate descent with projection. After changing h_j, project back to the feasible set (which adjusts all coordinates slightly). This is what the previous program does, but now we can do it much faster because we only need to evaluate the objective change in O(|S|) time.

Let me also think about whether we should include s=0 in the active set. The overlap at s=0 is ∑h(1-h) = n/2 - ∑h^2. For a binary h, this is n/2. For a smooth h, this is less than n/2. So s=0 is usually NOT the maximum (other lags have higher overlap). We should only include it if it's among the top.

        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:k]
    return top_idx

def active_cd(h, target, active_lags, n_sweeps=50, temperature=0.05, deadline=None, fine=False):
    n = len(h)
    h_cur = h.copy()
    shifts = active_lags.astype(int) - (n - 1)
    best_score = compute_c5(h_cur)
    
    # Compute initial overlaps for weighting
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    top_vals = corr[active_lags]
    w = np.exp((top_vals - top_vals.max()) / temperature)
    w /= w.sum()
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        
        # Update weights periodically
        if sweep % 10 == 0 and sweep > 0:
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            top_vals = corr[active_lags]
            w = np.exp((top_vals - top_vals.max()) / temperature)
            w /= w.sum()
        
        improved = False
        for j in range(n):
            cur_val = h_cur[j]
            
            # Compute gradient contribution g(j) = sum_s w_s * g_s(j)
            g = 0.0
            for idx_s in range(len(shifts)):
                s = shifts[idx_s]
                ws = w[idx_s]
                if s >= 0:
                    if j < n - s:
                        g += ws * (1.0 - h_cur[j + s])
                    if j >= s and s > 0:
                        g += ws * (-h_cur[j - s])
                    elif s == 0:
                        g += ws * (-h_cur[j])
                else:
                    m = -s
                    if j >= m:
                        g += ws * (1.0 - h_cur[j - m])
                    if j < n - m:
                        g += ws * (-h_cur[j + m])
            
            if fine:

                if g < 0:

    h_cur = h.copy()

    def objective(x):

        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        s = float(np.sum(x))
        val += 100.0 * (s - target) ** 2
        return val
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def fast_project(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = int(mask.sum())
    if cnt == 0:
        return h
    h = h.copy()
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def get_top_lags(h, k=30):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.argsort(corr)[::-1][:k]


def compute_lag_gradient(h, j, s):
    n = len(h)
    val = 0.0
    if s >= 0:
        if j < n - s:
            val += (1.0 - h[j + s])
        if j >= s:
            val -= h[j - s]
    else:
        m = -s
        if j >= m:
            val += (1.0 - h[j - m])
        if j < n - m:
            val -= h[j + m]
    return val


def active_cd(h, target, active_lags, n_sweeps=50, temperature=0.1, deadline=None, fine=False, rng=None):

    top_vals = corr[active_lags]
    w = np.exp((top_vals - top_vals.max()) / temperature)

def lbfgs_polish(h, target, alpha, maxiter, deadline=None):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    mu = 200.0

    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        s = float(np.sum(x))
        val += mu * (s - target) ** 2
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum > 1e-300:
            weights = weights / wsum
        else:
            weights = np.ones(len(corr)) / len(corr)
        g = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for ki in active:
            w = weights[ki]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                g[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    g[j2] -= w * x[j2 + lag]
            else:
                ms = -lag
                j1 = np.arange(0, n - ms)
                g[j1] += w * (1.0 - x[j1 + ms])
                if ms > 0:
                    j2 = np.arange(ms, n)
                    g[j2] -= w * x[j2 - ms]
        g += 2.0 * mu * (s - target)
        return val, g

    def f(x):
        return obj_and_grad(x)[0]

    def g_func(x):
        return obj_and_grad(x)[1]

    try:
        res = minimize(f, h_cur, jac=g_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

            best_h = h_cur
        active_lags = get_top_lags(best_h, k=40)
        h_ann, sc_ann = annealing_search(best_h, target, active_lags,
                                         n_iter=300, T_init=0.15, T_final=0.01,
                                         deadline=deadline, rng=rng)
        if sc_ann < best_score:
            best_score = sc_ann
            best_h = h_ann

    # Phase 6: Upsample to 2x resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if time.time() < deadline - 250:
        x_old = np.linspace(0.0, 2.0, n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n2, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, target2)
        active2 = get_top_lags(h_up, k=30)

        for rnd in range(3):
            if time.time() > deadline - 50:
                break
            h_up, sc = active_cd(h_up, target2, active2, n_sweeps=40,
                                 temperature=0.05, deadline=deadline, rng=rng)
            active2 = get_top_lags(h_up, k=30)
            h_up, sc = active_cd(h_up, target2, active2, n_sweeps=30,
                                 temperature=0.03, deadline=deadline, fine=True, rng=rng)
            sc = compute_c5(h_up)
            if sc < best_score:
                best_score = sc
                best_h = h_up
                n = n2
                target = target2
                active_lags = get_top_lags(best_h, k=30)

        if time.time() < deadline - 80:
            h_new = lbfgs_polish(best_h, target, 50000, 200, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Final fine CD
    if time.time() < deadline - 40:
        active_lags = get_top_lags(best_h, k=50)
        h_fin, sc_fin = active_cd(best_h, target, active_lags, n_sweeps=100,
                                  temperature=0.02, deadline=deadline, fine=True, rng=rng)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    # Phase 8: Final L-BFGS
    if time.time() < deadline - 20:
        h_new = lbfgs_polish(best_h, target, 200000, 200, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
