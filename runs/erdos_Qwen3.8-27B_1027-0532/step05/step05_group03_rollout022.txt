    for step in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        k_star = int(np.argmax(over))
        if k_star == 0:
            grad = dx * (1.0 - 2.0 * h_cur)
        else:
            grad = np.zeros(n)
            upper = n - 1 - k_star
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += dx * (1.0 - h_cur[idx1 + k_star])
            idx2 = np.arange(k_star, n)
            grad[idx2] -= dx * h_cur[idx2 - k_star]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.03, 0.006)
        else:
            eps *= 0.8
            if eps < 1e-8:
                break
    return h_cur, cur_score

def top_lag_surgery(h, max_steps=150, initial_epsilon=0.0008, n_lags=2):
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    for step in range(max_steps):
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        top_lags = np.argsort(over)[::-1][:n_lags]
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            if k == 0:
                grad += dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= dx * h_cur[idx2 - k]
        h_new = h_cur - eps * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score - 1e-13:
            h_cur = h_new
            cur_score = sc_new
            eps = min(eps * 1.02, 0.005)
        else:
            eps *= 0.85
            if eps < 1e-8:
                break
    return h_cur, cur_score

def build_single_interval(m, s, n):
    p = 2.0 / m
    w = 1.0 / m
    h = np.zeros(n)
    dx = 2.0 / n
    for k in range(m):
        start = k * p + s
        end = start + w
        i_s = max(0, int(np.round(start / dx)))
        i_e = min(n, int(np.round(end / dx)))
        if i_e > i_s:
            h[i_s:i_e] = 1.0
    target = n / 2.0
    return project_feasible(h, target)

def build_two_intervals(m, w1, s, n):
    p = 2.0 / m
    w2 = 1.0 / m - w1
    if w2 < 0.0:
        w1 = 1.0 / m
        w2 = 0.0
    h = np.zeros(n)
    dx = 2.0 / n
    for k in range(m):
        base = k * p
        i1s = max(0, int(np.round(base / dx)))
        i1e = min(n, int(np.round((base + w1) / dx)))
        if i1e > i1s:
            h[i1s:i1e] = 1.0
        i2s = max(0, int(np.round((base + s) / dx)))
        i2e = min(n, int(np.round((base + s + w2) / dx)))
        if i2e > i2s:
            h[i2s:i2e] = 1.0
    target = n / 2.0
    return project_feasible(h, target)

def build_three_intervals(m, w1, s1, s2, n):
    p = 2.0 / m
    total_w = 1.0 / m
    w2 = max(0.0, total_w - w1 - (total_w - w1) * 0.5)
    w3 = max(0.0, total_w - w1 - w2)
    h = np.zeros(n)
    dx = 2.0 / n
    for k in range(m):
        base = k * p
        for (w_i, s_i) in [(w1, 0.0), (w2, s1), (w3, s2)]:
            if w_i <= 0:
                continue
            i_s = max(0, int(np.round((base + s_i) / dx)))
            i_e = min(n, int(np.round((base + s_i + w_i) / dx)))
            if i_e > i_s:
                h[i_s:i_e] = 1.0
    target = n / 2.0
    return project_feasible(h, target)

def param_search_single(m, n, n_scan=100):
    s_vals = np.linspace(0.0, 1.0 / m, n_scan)
    best_s = 0.0
    best_score = float('inf')
    for s in s_vals:
        h = build_single_interval(m, s, n)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_s = s
    def f_single(s):
        h = build_single_interval(m, s, n)
        return compute_c5(h)
    try:
        res = minimize_scalar(f_single, bounds=(0.0, 1.0 / m), method='bounded',
                              options={'xatol': 1e-9})
        if res.fun < best_score:
            best_score = res.fun
            best_s = res.x
    except Exception:
        pass
    h_best = build_single_interval(m, best_s, n)
    return h_best, best_score

def param_search_two(m, n, n_scan=40):
    best_w1 = 0.5 / m
    best_s = 0.5 / m
    best_score = float('inf')
    w1_vals = np.linspace(0.05 / m, 0.45 / m, n_scan)
    for w1 in w1_vals:
        s_lo = max(0.01 / m, w1)
        s_hi = min(2.0 / m - w1 - 0.01 / m, 1.0 / m + w1)
        if s_hi <= s_lo:
            s_hi = s_lo + 0.001 / m
        s_vals = np.linspace(s_lo, s_hi, n_scan)
        for s in s_vals:
            h = build_two_intervals(m, w1, s, n)
            sc = compute_c5(h)
            if sc < best_score:
                best_score = sc
                best_w1 = w1
                best_s = s
    for step_frac in [0.1, 0.02, 0.005]:
        step_w = step_frac / m
        step_s = step_frac / m
        best_local_w1, best_local_s, best_local_sc = best_w1, best_s, best_score
        dw_vals = np.linspace(-step_w, step_w, 15)
        ds_vals = np.linspace(-step_s, step_s, 15)
        for dw in dw_vals:
            for ds in ds_vals:
                w1_t = best_w1 + dw
                s_t = best_s + ds
                if w1_t <= 0.01 / m or w1_t >= 0.99 / m:
                    continue
                if s_t < 0.005 / m:
                    continue
                h = build_two_intervals(m, w1_t, s_t, n)
                sc = compute_c5(h)
                if sc < best_local_sc:
                    best_local_w1, best_local_s, best_local_sc = w1_t, s_t, sc
        best_w1, best_s, best_score = best_local_w1, best_local_s, best_local_sc
    h_best = build_two_intervals(m, best_w1, best_s, n)
    return h_best, best_score

def param_search_three(m, n, n_scan=20):
    best_params = (0.3 / m, 0.4 / m, 0.8 / m)
    best_score = float('inf')
    for w1 in np.linspace(0.1 / m, 0.4 / m, n_scan):
        for s1 in np.linspace(w1, 1.0 / m, n_scan):
            for s2 in np.linspace(s1 + 0.1 / m, 2.0 / m - 0.1 / m, n_scan):
                h = build_three_intervals(m, w1, s1, s2, n)
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_params = (w1, s1, s2)
    h_best = build_three_intervals(m, *best_params, n=n)
    return h_best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def tleft_calc(start_time, budget_s):
    return budget_s - 30.0 - (time.time() - start_time)

def jitter_periodic(h, m, n, rng, jitter_amt=0.02):
    p = 2.0 / m
    dx = 2.0 / n
    h_j = h.copy()
    for k in range(1, m):
        boundary = k * p
        idx = int(np.round(boundary / dx))
        if 0 < idx < n:
            h_j[idx] += rng.uniform(-jitter_amt, jitter_amt)
    return project_feasible(h_j, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_score = float('inf')

    # Phase 1: Parametric periodic search
    n_param = 400
    for m in [3, 4, 5, 2, 6, 7, 8]:
        if tleft_calc(start_time, budget_s) < 120:
            break
        h_s, sc_s = param_search_single(m, n_param, n_scan=120)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s
        h_t, sc_t = param_search_two(m, n_param, n_scan=30)
        if sc_t < best_score:
            best_score = sc_t
            best_h = h_t

    # Try from initial_h_values
    try:
        n_init = len(initial_h_values)
        h_init = project_feasible(np.array(initial_h_values, dtype=float), n_init / 2.0)
        sc_init = compute_c5(h_init)
        if sc_init < best_score:
            best_score = sc_init
            best_h = h_init
    except Exception:
        pass

    if best_h is None:
        n_init = 400
        best_h = np.ones(n_init) * 0.5
        best_h = project_feasible(best_h, n_init / 2.0)
        best_score = compute_c5(best_h)

    # Phase 2: Jitter the best periodic solution to break symmetry
    if tleft_calc(start_time, budget_s) > 100:
        n_cur = len(best_h)
        for trial in range(5):
            m_try = 3 + trial
            h_jit = jitter_periodic(best_h, m_try, n_cur, rng, jitter_amt=0.03)
            sc_jit = compute_c5(h_jit)
            if sc_jit < best_score:
                best_score = sc_jit
                best_h = h_jit

    # Phase 3: SLSQP refinement
    if tleft_calc(start_time, budget_s) > 300:
        h_ref, sc = slsqp_optimize(best_h, [50, 200, 1000, 5000], 3000, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Subgradient momentum
    if tleft_calc(start_time, budget_s) > 300:
        h_sg, sc_sg = subgradient_momentum(best_h, 500, 4, 0.003, 0.997, 0.3, 0.005)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgradient_momentum(best_h, 400, 5, 0.002, 0.998, 0.4, 0.003)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 5: Surgery
    if tleft_calc(start_time, budget_s) > 150:
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=200, initial_epsilon=0.0008, n_lags=2)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = worst_lag_surgery(best_h, max_steps=200, initial_epsilon=0.001)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    # Phase 6: High-alpha SLSQP
    if tleft_calc(start_time, budget_s) > 200:
        h_ref, sc = slsqp_optimize(best_h, [10000, 50000, 200000], 2000, 1e-16)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Finer resolution
    if tleft_calc(start_time, budget_s) > 350:
        for n_fine in [len(best_h) + 100, len(best_h) + 200]:
            if tleft_calc(start_time, budget_s) < 250:
                break
            target_fine = n_fine / 2.0
            h_fine = interpolate_to(best_h, n_fine, target_fine)
            h_fine, sc_fine = slsqp_optimize(h_fine, [100, 500, 2000], 1500, 1e-14)
            h_fine, sc_fine = slsqp_optimize(h_fine, [5000, 20000], 1500, 1e-15)
            h_sg_f, sc_sg_f = subgradient_momentum(h_fine, 300, 4, 0.002, 0.997, 0.3, 0.004)
            if sc_sg_f < sc_fine:
                h_fine, sc_fine = h_sg_f, sc_sg_f
            h_surg_f, sc_sf = top_lag_surgery(h_fine, max_steps=150, initial_epsilon=0.0008, n_lags=2)
            if sc_sf < sc_fine:
                h_fine, sc_fine = h_surg_f, sc_sf
            h_ref_f, sc_rf = slsqp_optimize(h_fine, [10000, 50000], 1000, 1e-16)
            if sc_rf < sc_fine:
                h_fine, sc_fine = h_ref_f, sc_rf
            if sc_fine < best_score:
                best_score = sc_fine
                best_h = h_fine

    # Phase 8: Random restarts
    deadline_p8 = start_time + (budget_s - 60.0) * 0.72
    for attempt in range(10):
        if time.time() > deadline_p8:
            break
        if tleft_calc(start_time, budget_s) < 80:
            break
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_pert = best_h.copy()
        if attempt % 5 == 0:
            h_pert += rng.randn(n_cur) * 0.05
        elif attempt % 5 == 1:
            block = max(10, n_cur // 15)
            for b in range(n_cur // block):
                if rng.random() < 0.4:
                    shift = rng.uniform(-0.12, 0.12)
                    h_pert[b*block:min((b+1)*block, n_cur)] += shift
        elif attempt % 5 == 2:
            h_pert += np.sin(np.arange(n_cur) * 0.1 + rng.rand() * 6.28) * 0.04
        elif attempt % 5 == 3:
            n_flips = max(1, n_cur // 12)
            flip_idx = rng.choice(n_cur, n_flips, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        else:
            h_pert += rng.rand(n_cur) * 0.06 - 0.03
        h_pert = project_feasible(h_pert, target_cur)
        h_ref, sc = slsqp_optimize(h_pert, [500, 2000, 10000], 1200, 1e-15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final refinement
    if tleft_calc(start_time, budget_s) > 100:
        h_sg, sc_sg = subgradient_momentum(best_h, 400, 4, 0.002, 0.998, 0.3, 0.003)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_surg, sc_surg = top_lag_surgery(best_h, max_steps=150, initial_epsilon=0.0006, n_lags=3)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg

    # Phase 10: Final high-precision SLSQP
    if tleft_calc(start_time, budget_s) > 60:
        h_ref, sc = slsqp_optimize(best_h, [50000, 200000], 1500, 1e-17)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Final surgery
    if tleft_calc(start_time, budget_s) > 30:
        h_surg, sc_surg = worst_lag_surgery(best_h, max_steps=100, initial_epsilon=0.0005)
        if sc_surg < best_score:
            best_score = sc_surg
            best_h = h_surg
        h_surg2, sc_surg2 = top_lag_surgery(best_h, max_steps=80, initial_epsilon=0.0004, n_lags=2)
        if sc_surg2 < best_score:
            best_score = sc_surg2
            best_h = h_surg2

    # Phase 12: One last SLSQP
    if tleft_calc(start_time, budget_s) > 15:
        h_final, sc_final = slsqp_optimize(best_h, [100000, 500000], 500, 1e-17)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
