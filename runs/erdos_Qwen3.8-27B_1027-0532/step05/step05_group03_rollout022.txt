        corr = full_corr(h_cur)
        top_lags = np.argsort(corr)[::-1][:n_lags]
        diffs = corr[top_lags] - corr[top_lags[0]]
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_lags):
            k = int(k)
            if k == 0:
                grad += wk * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                if upper >= 0:
                    idx1 = np.arange(0, upper + 1)
                    grad[idx1] += wk * dx * (1.0 - h_cur[idx1 + k])
                    idx2 = np.arange(k, n)
                    grad[idx2] -= wk * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = project(h_cur - lr_cur * velocity, target)
        sc_new = c5_val(h_new)
        if sc_new < best_sc:
            best_sc = sc_new
            best = h_new.copy()
        if sc_new < c5_val(h_cur):
            h_cur = h_new
            lr_cur = min(lr_cur * 1.005, 0.01)
        else:
            lr_cur *= 0.95
            if lr_cur < 1e-10:
                lr_cur = lr * 0.5
    return best, best_sc

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = c5_val(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: SLSQP from parent
    if tleft() > 200:
        h_ref, sc = slsqp_opt(best_h, [100, 500, 2000], 2000, 1e-14, start_time, 60)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Binarize and local search
    if tleft() > 500:
        h_bin = binarize(best_h, target)
        h_bin, sc_bin = binary_local_search(h_bin, n_iter=2000, n_trials=500, rng=rng)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin

    # Phase 3: SLSQP from binary solution
    if tleft() > 400:
        h_ref, sc = slsqp_opt(best_h, [500, 2000, 10000], 1500, 1e-15, start_time, 80)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Simulated annealing on binary
    if tleft() > 350:
        h_sa, sc_sa = simulated_annealing(best_h, n_iter=6000, T_init=0.8, T_final=0.05,
                                          rng=rng, start_time=start_time, time_limit=100)
        if sc_sa < best_score:
            best_score = sc_sa
            best_h = h_sa
        # Follow-up local search
        h_ls, sc_ls = binary_local_search(h_sa, n_iter=2000, n_trials=500, rng=rng)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    # Phase 5: Paired swap search
    if tleft() > 300:
        h_ps, sc_ps = paired_swap_search(best_h, n_iter=1500, n_trials=200, rng=rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps
        h_ls, sc_ls = binary_local_search(h_ps, n_iter=1500, n_trials=400, rng=rng)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    # Phase 6: SLSQP refinement
    if tleft() > 250:
        h_ref, sc = slsqp_opt(best_h, [2000, 10000, 50000], 1500, 1e-16, start_time, 80)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Lag-equality search
    if tleft() > 200:
        h_le, sc_le = lag_equality_search(best_h, n_iter=2000, n_trials=300, rng=rng,
                                         start_time=start_time, time_limit=60)
        if sc_le < best_score:
            best_score = sc_le
            best_h = h_le
        h_ls, sc_ls = binary_local_search(h_le, n_iter=1000, n_trials=400, rng=rng)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    # Phase 8: Fourier parameterization approach
    if tleft() > 250:
        h_four, sc_four = fourier_optimize(n, n_modes=12, start_time=start_time, time_limit=80)
        if h_four is not None and sc_four < best_score:
            best_score = sc_four
            best_h = h_four
        # Refine fourier solution with local search
        if h_four is not None:
            h_fb, sc_fb = binary_local_search(h_four, n_iter=1500, n_trials=400, rng=rng)
            if sc_fb < best_score:
                best_score = sc_fb
                best_h = h_fb
            h_ref, sc = slsqp_opt(best_h, [5000, 20000], 1000, 1e-15, start_time, 50)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 9: Multi-restart SA with different temperatures
    if tleft() > 150:
        for T_try in [0.3, 1.5, 3.0]:
            if tleft() < 50:
                break
            h_sa, sc_sa = simulated_annealing(best_h, n_iter=4000, T_init=T_try, T_final=T_try*0.05,
                                              rng=rng, start_time=start_time, time_limit=40)
            if sc_sa < best_score:
                best_score = sc_sa
                best_h = h_sa
            h_ls, sc_ls = binary_local_search(h_sa, n_iter=1000, n_trials=300, rng=rng)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls

    # Phase 10: Subgradient with multiple lags
    if tleft() > 120:
        h_sg, sc_sg = subgradient_multi(best_h, n_steps=1500, n_lags=5, lr=0.003,
                                        momentum=0.4, start_time=start_time, time_limit=40)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_sg2, sc_sg2 = subgradient_multi(best_h, n_steps=1000, n_lags=8, lr=0.001,
                                         momentum=0.5, start_time=start_time, time_limit=30)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    # Phase 11: Final SLSQP with very high alpha
    if tleft() > 80:
        h_ref, sc = slsqp_opt(best_h, [5000, 20000, 100000], 1500, 1e-17, start_time, 50)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 12: Final intensive binary local search
    if tleft() > 50:
        h_bin = binarize(best_h, target)
        h_ls, sc_ls = binary_local_search(h_bin, n_iter=5000, n_trials=600, rng=rng)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    # Phase 13: One more SLSQP from best binary
    if tleft() > 25:
        h_ref, sc = slsqp_opt(best_h, [10000, 50000], 800, 1e-17, start_time, 15)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 14: Final paired swap + local search
    if tleft() > 20:
        h_ps, sc_ps = paired_swap_search(best_h, n_iter=500, n_trials=150, rng=rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_h = h_ps

    # Final feasibility check and return
    h_out = project(best_h, len(best_h) / 2.0)
    final_score = c5_val(h_out)
    return h_out, final_score, len(h_out)
