    """Solve min_{x in [0,1]} max_k (slopes[k]*x + intercepts[k]) + lam*x
    
    Returns optimal x.
    """

    # Check endpoints

        return 1.0

                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, eps=scs_eps, max_iters=100000, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved or h_var.value is None:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = set(np.argsort(corr)[-n_active:].tolist())
        t_val = t_var.value if t_var.value is not None else best_c5
        for k_idx in range(1, len(corr) - 1):
            if k_idx not in active_k and k_idx not in top_k:
                if corr[k_idx] * dx > t_val + 5e-5:
                    active_k.add(k_idx)
                    break
        active_k = top_k | active_k
        if len(active_k) > 4 * n_active:
            active_k = set(np.argsort(corr)[-4 * n_active:].tolist())

        if c5_val >= best_c5 - 1e-14:
            damping = max(0.3, damping * 0.93)
        else:
            damping = min(0.98, damping + 0.03)
        if it % 60 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.array([np.sum(h) - _global_target])


def cons_jac(h):
    return np.array([np.ones(_global_n)])


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target, _global_alpha
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        _global_alpha = alpha
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-15:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    half_n = n - 1
    idxs = np.arange(n)
    stall = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - half_n
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad = np.zeros(n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx
        improved = False
        for lr in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1, 0.2, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 80:
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_h = project_feasible(best_h, target)
    best_score = compute_c5(best_h)

    # Phase 1: LP with more active constraints and tighter tolerances
    if tleft() > 150:
        lp_time = min(250, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=20000, time_limit=lp_time,
                                     d0=0.9, reg0=5e-5, n_active=60, scs_eps=1e-11)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Coordinate descent (new approach)
    if tleft() > 100:
        cd_time = min(150, tleft() * 0.15)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=60,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP moderate alpha
    if tleft() > 120:
        h_ref = slsqp_round(best_h, n, target, [100, 500, 2000], 2500, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 4: Another LP round with tighter settings
    if tleft() > 150:
        lp_time = min(200, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                     max_iter=20000, time_limit=lp_time,
                                     d0=0.95, reg0=1e-5, n_active=80, scs_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 5: Coordinate descent again (longer)
    if tleft() > 80:
        cd_time = min(100, tleft() * 0.12)
        h_cd, c5_cd = coord_descent(best_h, target, n_active=80,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: SLSQP high alpha
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [5000, 20000, 80000], 2000, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Try n=500
    if tleft() > 250:
        n_try = 500
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_int = compute_c5(h_interp)
        if tleft() > 200:
            lp_time = min(120, tleft() * 0.3)
            h_lp500, c5_lp500 = alternating_lp(h_interp, n_try, target_try,
                                               max_iter=15000, time_limit=lp_time,
                                               d0=0.9, reg0=5e-5, n_active=60, scs_eps=1e-11)
            if c5_lp500 < best_score:
                best_score = c5_lp500
                best_h = h_lp500
                n = n_try
                target = target_try
            elif sc_int < best_score:
                best_score = sc_int
                best_h = h_interp
                n = n_try
                target = target_try
        if tleft() > 100:
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_ref = slsqp_round(best_h, n_use, target_use,
                                [2000, 10000, 50000], 1500, 1e-15)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 8: Subgradient refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=5000, time_limit=min(40, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 9: Perturbed restarts
    for noise in [0.02, 0.05, 0.1, 0.2]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, target)
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(h_pert, n_use, target_use,
                            [200, 1000, 5000], 1000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Final coordinate descent
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        cd_time = min(40, tleft() * 0.6)
        h_cd, c5_cd = coord_descent(best_h, target_use, n_active=100,
                                    max_passes=20000, time_limit=cd_time, rng=rng)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final LP polish
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        lp_time = min(40, tleft() * 0.7)
        h_lp, c5_lp = alternating_lp(best_h, n_use, target_use,
                                     max_iter=10000, time_limit=lp_time,
                                     d0=0.98, reg0=1e-6, n_active=100, scs_eps=1e-12)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 12: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(best_h, n_use, target_use,
                            [10000, 50000, 200000], 800, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final subgradient micro-refinement
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                          max_steps=3000, time_limit=min(12, tleft() * 0.8))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
