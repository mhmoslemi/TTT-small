def coord_descent_lse(h, n, target, alpha, n_sweeps, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            # Minimize over h[i] in [0,1]
            # LSE objective as function of h[i]
            # The correlation terms involving h[i] are linear in h[i]
            # So LSE is a smooth function of h[i]
            # Use ternary search on [0, 1]
            
            def obj_at(val):
                h_test = h_cur.copy()
                h_test[i] = val
                corr = compute_corr(h_test)
                c_max = float(np.max(corr))
                shifted = alpha * (corr - c_max)
                shifted = np.clip(shifted, -700.0, 0.0)
                return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
            
            # Golden section search on [0, 1]
            a, b = 0.0, 1.0
            gr = (np.sqrt(5) - 1) / 2
            c = b - gr * (b - a)
            d = a + gr * (b - a)
            fc = obj_at(c)
            fd = obj_at(d)
            for _ in range(30):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = obj_at(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = obj_at(d)
            best_val = (a + b) / 2
            h_cur[i] = best_val
        
        # Project back to feasible
        h_cur = project_feasible(h_cur, target)
        sc = compute_c5(h_cur)
        if sc < cur_score:
            cur_score = sc
    return h_cur, cur_score

def subgradient_polish(h, n, target, n_steps, n_top, lr_init, lr_decay, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = get_overlap(corr, n)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Top-k lags with softmax weighting
        top_idx = np.argsort(over)[::-1][:n_top]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / 0.01)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                continue
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        
        # Line search
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1]:
            h_new = h_cur - trial_lr * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < c_curr:
                h_cur = h_new
                lr = min(trial_lr * 1.1, lr_init * 5.0)
                break
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def adaptive_subgradient(h, n, target, n_steps, deadline):

    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = 0.01
    temp = 0.01
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = get_overlap(corr, n)
        c_curr = float(np.max(over)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Adaptive temperature
        temp = max(0.001, temp * 0.999)
        # Adaptive top-k
        n_top = min(10, max(2, int(3 + step * 0.01)))
        
        top_idx = np.argsort(over)[::-1][:n_top]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                continue
            j_pos = np.arange(0, n - k)

            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]

                break
    return best_h, best_score


def worst_lag_surgery(h, n, target, max_steps, initial_epsilon, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    stall_count = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        idx_star = int(np.argmax(corr))
        k_star = idx_star - (n - 1)
        if k_star == 0:
            break
        grad = compute_full_gradient(h_cur, n, k_star)
        grad *= dx

        improved = False
        trial_list = [eps, eps * 0.5, eps * 0.25, eps * 0.1, eps * 2.0, eps * 4.0, eps * 0.05]
        for t_eps in trial_list:
            h_new = h_cur - t_eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc_new
                eps = min(t_eps * 1.05, 0.05)
                improved = True
                stall_count = 0
                break
        if not improved:
            eps *= 0.5
            stall_count += 1
            if eps < 1e-10 or stall_count > 50:
                break
    return h_cur, cur_score


def multi_lag_surgery(h, n, target, max_steps, initial_epsilon, n_active, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = initial_epsilon
    temp = 0.003
    stall_count = 0
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()

        grad = np.zeros(n)
        for wk, idx in zip(weights, top_idx):
            k = int(idx) - (n - 1)
            grad += wk * compute_full_gradient(h_cur, n, k)
        grad *= dx

        improved = False
        trial_list = [eps, eps * 0.5, eps * 0.25, eps * 2.0]
        for t_eps in trial_list:
            h_new = h_cur - t_eps * grad
            h_new = project_feasible(h_new, target)
            sc_new = compute_c5(h_new)
            if sc_new < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc_new
                eps = min(t_eps * 1.04, 0.04)
                improved = True
                stall_count = 0
                break
        if not improved:
            eps *= 0.6
            temp = max(0.0005, temp * 0.995)
            stall_count += 1
            if eps < 1e-11 or stall_count > 80:
                break
    return h_cur, cur_score


def de_optimize(n, target, pop_size, n_gen, f_factor, cr, deadline, rng, init_pop=None):
    if init_pop is not None:
        pop = np.array(init_pop, dtype=float).copy()
        if pop.shape[0] < pop_size:
            n_extra = pop_size - pop.shape[0]
            extra = np.zeros((n_extra, n))
            for i in range(n_extra):
                extra[i] = project_feasible(rng.rand(n) * 0.5 + 0.25, target)
            pop = np.vstack([pop, extra])
        else:
            pop = pop[:pop_size].copy()
    else:
        pop = np.zeros((pop_size, n))
        x = np.linspace(0, 2, n)
        for i in range(pop_size):
            r = i % 5
            if r == 0:
                pop[i] = project_feasible(rng.rand(n), target)
            elif r == 1:
                freq = rng.randint(1, 5)
                amp = rng.uniform(0.15, 0.45)
                phase = rng.uniform(0, 2 * np.pi)
                pop[i] = project_feasible(0.5 + amp * np.cos(freq * np.pi * x + phase), target)
            elif r == 2:
                pop[i] = project_feasible(rng.choice([0.0, 1.0], n).astype(float), target)
            elif r == 3:
                pop[i] = project_feasible(0.5 + rng.randn(n) * 0.25, target)
            else:
                pop[i] = project_feasible(np.clip(0.5 + rng.randn(n) * 0.4, 0.05, 0.95), target)

    scores = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    idx_range = np.arange(pop_size)
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            mask = idx_range != i
            avail = idx_range[mask]
            a, b, c = rng.choice(avail, size=3, replace=False)
            v = pop[a] + f_factor * (pop[b] - pop[c])
            j_rand = rng.randint(n)
            trial = v.copy()
            r_mask = rng.rand(n)
            trial[r_mask > cr] = pop[i][r_mask > cr]
            trial[j_rand] = v[j_rand]
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score <= scores[i]:
                pop[i] = trial
                scores[i] = score
                if score < best_score:
                    best_score = score
                    best_h = trial.copy()
    return best_h, best_score


def coordinate_descent(h, n, target, n_sweeps, deadline):
    """Coordinate descent: for each coordinate, try a few candidate values and pick the best."""
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_val = h_cur[i]
            candidates = [old_val - 0.1, old_val - 0.05, old_val, old_val + 0.05, old_val + 0.1]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]
            best_val = old_val
            best_sc = cur_score
            for cv in candidates:
                if cv == old_val:
                    continue
                h_test = h_cur.copy()
                h_test[i] = cv
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
            if best_val != old_val:
                h_cur[i] = best_val
                h_cur = project_feasible(h_cur, target)
                cur_score = best_sc
                improved = True
        if not improved and sweep > 0:
            break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)
    hard_deadline = start_time + budget_s - 10.0

    def tleft():
        return hard_deadline - time.time()

    n_parent = len(initial_h_values)
    target_parent = n_parent / 2.0
    h_parent = project_feasible(np.array(initial_h_values, dtype=float), target_parent)
    best_h = h_parent.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP improvement at parent resolution
    if tleft() > 120:
        h_s, sc_s = slsqp_optimize(h_parent, n_parent, target_parent,
                                   [200, 1000, 5000], 2000, 1e-16, min(start_time + 50, hard_deadline))
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 2: Work at n=800
    n_main = 800
    target_main = n_main / 2.0
    h_main = interpolate_to(best_h, n_main, target_main)
    sc_main = compute_c5(h_main)

    # Phase 2a: DE at n=800
    if tleft() > 350:
        pop = [h_main.copy()]
        for i in range(14):
            scale = rng.uniform(0.02, 0.10)
            p = h_main + rng.randn(n_main) * scale
            pop.append(project_feasible(p, target_main))
        h_de, sc_de = de_optimize(n_main, target_main, 15, 80, 0.5, 0.4,
                                   min(start_time + 140, hard_deadline), rng, init_pop=pop)
        if sc_de < sc_main:
            sc_main = sc_de
            h_main = h_de

    # Phase 2b: SLSQP at n=800 - multiple alpha values
    if tleft() > 250:
        h_s, sc_s = slsqp_optimize(h_main, n_main, target_main,
                                   [50, 200, 800, 3000, 10000, 30000],
                                   3000, 1e-16, min(start_time + 400, hard_deadline))
        if sc_s < sc_main:
            sc_main = sc_s
            h_main = h_s

    # Phase 2c: Adaptive subgradient
    if tleft() > 150:
        h_sg, sc_sg = adaptive_subgradient(h_main, n_main, target_main,
                                           1500, min(start_time + 600, hard_deadline), lr_init=0.015)
        if sc_sg < sc_main:
            sc_main = sc_sg
            h_main = h_sg

    # Phase 2d: SLSQP after subgradient
    if tleft() > 150:
        h_s, sc_s = slsqp_optimize(h_main, n_main, target_main,
                                   [500, 2000, 10000, 30000, 80000],
                                   3000, 1e-16, min(start_time + 800, hard_deadline))
        if sc_s < sc_main:
            sc_main = sc_s
            h_main = h_s

    # Phase 2e: Multi-lag surgery
    if tleft() > 120:
        h_mls, sc_mls = multi_lag_surgery(h_main, n_main, target_main,
                                           800, 0.008, 5, min(start_time + 950, hard_deadline))
        if sc_mls < sc_main:
            sc_main = sc_mls
            h_main = h_mls

    # Phase 2f: SLSQP after surgery
    if tleft() > 100:
        h_s, sc_s = slsqp_optimize(h_main, n_main, target_main,
                                   [2000, 10000, 50000, 100000],
                                   3000, 1e-17, min(start_time + 1080, hard_deadline))
        if sc_s < sc_main:
            sc_main = sc_s
            h_main = h_s

    # Phase 2g: Final subgradient pass
    if tleft() > 60:
        h_sg, sc_sg = adaptive_subgradient(h_main, n_main, target_main,
                                           800, min(start_time + 1180, hard_deadline), lr_init=0.008)
        if sc_sg < sc_main:
            sc_main = sc_sg
            h_main = h_sg

    # Phase 2h: Worst lag surgery
    if tleft() > 40:
        h_ws, sc_ws = worst_lag_surgery(h_main, n_main, target_main,
                                         400, 0.003, min(start_time + 1240, hard_deadline))
        if sc_ws < sc_main:
            sc_main = sc_ws
            h_main = h_ws

    # Phase 3: Try n=1600 if time allows
    if tleft() > 60:
        n_high = 1600
        target_high = n_high / 2.0
        h_high = interpolate_to(h_main, n_high, target_high)
        sc_high = compute_c5(h_high)
        h_sg_h, sc_sg_h = adaptive_subgradient(h_high, n_high, target_high,
                                               500, min(start_time + 1280, hard_deadline), lr_init=0.008)
        if sc_sg_h < sc_main:
            sc_main = sc_sg_h
            h_main = h_sg_h

    # Track best overall
    if sc_main < best_score:
        best_score = sc_main
        best_h = h_main

    # Final polish with SLSQP if time remains
    if tleft() > 30:
        n_fin = len(best_h)
        tgt_fin = n_fin / 2.0
        h_fin, sc_fin = slsqp_optimize(best_h, n_fin, tgt_fin,
                                        [5000, 20000, 50000], 2000, 1e-17, hard_deadline)
        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_final
