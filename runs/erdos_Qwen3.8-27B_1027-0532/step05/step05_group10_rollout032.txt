    delta = np.zeros(2*n-1)

    valid_plus = (i + k_range >= 0) & (i + k_range < n)

    delta[valid_plus] += (1.0 - h[i + k_range[valid_plus]])

    # We want to minimize max(corr_new(v)) over v in [0,1]

            hi = m2
        else:
            lo = m1
    v_opt = (lo + hi) / 2.0
    return v_opt

def project_feasible(h, target_sum):

        return h
