                score_j[mj2] += h[j_jas[mj2]]
                thresh_i = np.percentile(score_i, 50)
                thresh_j = np.percentile(score_j, 50)
                i_cands = ones[score_i <= thresh_i]
                j_cands = zeros[score_j >= thresh_j]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = int(i_cands[rng.randint(len(i_cands))])
                    j = int(j_cands[rng.randint(len(j_cands))])
                else:
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
            else:
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
        else:
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])

        if i == j:
            continue
        d = compute_delta(h, i, j, n, s_vals)
        new_max = float((full + d).max())
        c5_new = new_max * dx
        dc5 = c5_new - current_c5

        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += d
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def two_swap_search(h, max_iters, rng, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    full = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    s_vals = np.arange(-(n - 1), n)
    stall = 0
    max_stall = 500

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        max_val = float(full.max())
        i1 = int(ones[rng.randint(len(ones))])
        i2 = int(ones[rng.randint(len(ones))])
        j1 = int(zeros[rng.randint(len(zeros))])
        j2 = int(zeros[rng.randint(len(zeros))])
        if i1 == i2 or j1 == j2:
            continue

        d1 = compute_delta(h, i1, j1, n, s_vals)
        h[i1] = 0.0
        h[j1] = 1.0
        d2 = compute_delta(h, i2, j2, n, s_vals)
        total_d = d1 + d2
        new_max = float((full + total_d).max())

        if new_max < max_val:
            h[i2] = 0.0
            h[j2] = 1.0
            full += total_d
            stall = 0
            nc5 = float(full.max()) * dx
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i1] = 1.0
            h[j1] = 0.0
            stall += 1

    return best_h, best_c5

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.logsumexp(shifted)) / alpha

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, alphas, maxiter, time_limit, start_time):
    global _cons_n, _cons_target
    n = len(h_start)
    _cons_n = n
    _cons_target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = np.clip(res.x, 0.0, 1.0)
            h_new = project_binary(h_new, int(n / 2))
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_binary(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_binary(h_out, target_dst)

def multi_start_greedy(h, n_starts, rng, time_limit, start_time):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for s in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        h_pert = h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        n_swap = rng.randint(2, 6)
        for _ in range(n_swap):
            if len(ones) < 1 or len(zeros) < 1:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]

        t_sub = max(1.0, (time_limit - (time.time() - start_time)) / max(n_starts - s, 1))
        t_sub = min(t_sub, 30.0)
        h_g, c5_g = focused_greedy(h_pert, 5000, 3, rng, t_sub, time.time())
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)
    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    phase_times = [
        (0.10, "greedy"),
        (0.25, "sa_focused"),
        (0.15, "two_swap"),
        (0.15, "sa_restart"),
        (0.15, "multistart"),
        (0.10, "slsqp"),
        (0.10, "final"),
    ]

    elapsed_at_phase = 0.0
    for frac, name in phase_times:
        if tleft() < 20:
            break
        allowed = min(frac * budget_s * 0.9, tleft() - 15)
        if allowed < 5:
            break

        if name == "greedy":
            h_g, c5_g = focused_greedy(best_h, 50000, 3, rng, allowed, start_time)
            if c5_g < best_score:
                best_score = c5_g
                best_h = h_g.copy()

        elif name == "sa_focused":
            n_sa = int(min(5000000, allowed * 30000))
            h_sa, c5_sa = sa_search(best_h, n_sa, 0.03, 0.0001, rng,
                                     allowed, start_time, focus=True)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()

        elif name == "two_swap":
            h_ts, c5_ts = two_swap_search(best_h, 200000, rng, allowed, start_time)
            if c5_ts < best_score:
                best_score = c5_ts
                best_h = h_ts.copy()
            h_g2, c5_g2 = focused_greedy(best_h, 20000, 5, rng,
                                          max(1.0, tleft() * 0.05), start_time)
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()

        elif name == "sa_restart":
            n_restarts = 3
            for r in range(n_restarts):
                if tleft() < 20:
                    break
                h_pert = best_h.copy()
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
                n_swaps = rng.randint(3, 8)
                for _ in range(n_swaps):
                    if len(ones) < 1 or len(zeros) < 1:
                        break
                    i = int(ones[rng.randint(len(ones))])
                    j = int(zeros[rng.randint(len(zeros))])
                    h_pert[i] = 0.0
                    h_pert[j] = 1.0
                    ones = np.where(h_pert == 1)[0]
                    zeros = np.where(h_pert == 0)[0]
                t_sub = max(2.0, (tleft() - 15) / (n_restarts - r))
                n_sa_r = int(min(2000000, t_sub * 30000))
                h_sr, c5_sr = sa_search(h_pert, n_sa_r, 0.02, 0.00005, rng,
                                         t_sub, start_time, focus=True)
                if c5_sr < best_score:
                    best_score = c5_sr
                    best_h = h_sr.copy()

        elif name == "multistart":
            h_ms, c5_ms = multi_start_greedy(best_h, 5, rng, allowed, start_time)
            if c5_ms < best_score:
                best_score = c5_ms
                best_h = h_ms.copy()

        elif name == "slsqp":
            n_use = len(best_h)
            alphas = [30, 100, 300, 1000]
            h_ref, sc = slsqp_refine(best_h, alphas, 1500, allowed, start_time)
            if sc < best_score:
                best_score = sc
                best_h = sc and h_ref or best_h
                best_h = h_ref.copy()
            h_bin = project_binary(best_h, int(n_use / 2))
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_score:
                best_score = c5_bin
                best_h = h_bin.copy()

        elif name == "final":
            n_use = len(best_h)
            h_fg, c5_fg = focused_greedy(best_h, 30000, 5, rng,
                                          max(1.0, allowed * 0.5), start_time)
            if c5_fg < best_score:
                best_score = c5_fg
                best_h = h_fg.copy()
            if tleft() > 10:
                n_sf = int(min(1000000, max(10000, tleft() * 20000)))
                h_sf, c5_sf = sa_search(best_h, n_sf, 0.01, 0.00001, rng,
                                         max(3.0, tleft() - 5), start_time, focus=True)
                if c5_sf < best_score:
                    best_score = c5_sf
                    best_h = h_sf.copy()

    n_final = len(best_h)
    target_final = int(n_final / 2.0)
    h_final = project_binary(best_h, target_final)
    final_score = compute_c5(h_final)

    if final_score > best_score:
        h_final = best_h.copy()
        final_score = best_score

    return h_final, final_score, n_final
