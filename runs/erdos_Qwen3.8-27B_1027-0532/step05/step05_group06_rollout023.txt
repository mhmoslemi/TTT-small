            x = x_new
            lr = min(lr * 1.01, 0.05)
        else:
            x = x_new
            lr *= 0.92
            v *= 0.5
            if lr < 1e-9:
                lr = lr_init * 0.1
                v = np.zeros(n)
    return best_h, best_score


def focus_equalize(h, target, n_steps, top_k, deadline):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(best_h)
    lr = 0.003

    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_range = top_vals.max() - top_vals.min()
        if top_range < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_range + 1e-15)
            grad += w * _grad_for_lag(best_h, ki, n)
        x_new = project_feasible(best_h - lr * grad, target)
        sc = compute_c5(x_new)
        if sc < best_score:
            best_score = sc
            best_h = x_new
            lr = min(lr * 1.03, 0.04)
        else:
            lr *= 0.85
            if lr < 1e-10:
                break
    return best_h, best_score


def upsample(h, factor):
    n_old = len(h)
    n_new = n_old * factor
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target_new = n_new / 2.0
    return project_feasible(h_new, target_new)


def downsample(h, factor):
    n_old = len(h)
    n_new = n_old // factor
    h_new = h[:n_new * factor].reshape(n_new, factor).mean(axis=1)
    target_new = n_new / 2.0
    return project_feasible(h_new, target_new)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    deadline = start_time + budget_s - 40

    n = len(initial_h_values)
    target = n / 2.0
    best_h = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_score = compute_c5(best_h)

    # Phase 1: Warmup L-BFGS at low alpha
    alphas_warmup = [20, 100, 500]
    for a in alphas_warmup:
        if tleft() < 80:
            break
        h_new = lbfgs_refine(best_h, target, a, 800, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Subgradient + focus equalization
    if tleft() > 150:
        h_new, sc = subgrad_momentum(best_h, target, 10000, 5, 0.005, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if tleft() > 100:
        h_new, sc = focus_equalize(best_h, target, 300, 5, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_new = lbfgs_refine(best_h, target, 2000, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Progressive alpha increase
    alphas_prog = [2000, 10000, 50000, 200000, 1000000]
    for a in alphas_prog:
        if tleft() < 70:
            break
        h_new = lbfgs_refine(best_h, target, a, 1000, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if tleft() > 40:
            h_new, sc = subgrad_momentum(best_h, target, 5000, 3, 0.003, deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            h_new, sc = focus_equalize(best_h, target, 200, 3, deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Multi-resolution
    n2 = 2 * n
    target2 = n2 / 2.0
    if tleft() > 350:
        h_fine = upsample(best_h, 2)
        alphas_fine = [100, 1000, 10000, 100000]
        for a in alphas_fine:
            if tleft() < 80:
                break
            h_fine = lbfgs_refine(h_fine, target2, a, 500, deadline)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = downsample(h_fine, 2)
        if tleft() > 100:
            h_fine = upsample(best_h, 2)
            h_fine = lbfgs_refine(h_fine, target2, 500000, 800, deadline)
            sc = compute_c5(h_fine)
            if sc < best_score:
                best_score = sc
                best_h = downsample(h_fine, 2)

    # Phase 5: Diverse restarts
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if tleft() > 200:
        for ri in range(8):
            if tleft() < 60:
                break
            noise = 0.03 + 0.04 * rng.rand()
            h_pert = best_h + rng.randn(n_cur) * noise
            h_pert = project_feasible(h_pert, target_cur)
            h_new = lbfgs_refine(h_pert, target_cur, 1000, 400, deadline)
            h_new = lbfgs_refine(h_new, target_cur, 50000, 400, deadline)
            h_new, sc = subgrad_momentum(h_new, target_cur, 3000, 3, 0.003, deadline)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: Intensive final optimization
    if tleft() > 120:
        for a in [500000, 2000000, 5000000]:
            if tleft() < 50:
                break
            h_new = lbfgs_refine(best_h, target_cur, a, 1500, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Subgradient + focus final polish
    if tleft() > 60:
        h_new, sc = subgrad_momentum(best_h, target_cur, 20000, 3, 0.001, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    if tleft() > 30:
        h_new, sc = focus_equalize(best_h, target_cur, 500, 3, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final L-BFGS
    if tleft() > 30:
        h_new = lbfgs_refine(best_h, target_cur, 5000000, 2000, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 9: Ultra-fine subgradient
    if tleft() > 20:
        h_new, sc = subgrad_momentum(best_h, target_cur, 50000, 2, 0.0005, deadline)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
