    h = h.astype(float).copy()
    n = len(h)

    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def pnorm_obj_and_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    corr_pos = np.maximum(corr, 0)  # correlations can be negative
    obj = (np.sum(corr_pos ** p)) ** (1.0/p) * dx
    # Gradient: d/dh_i of (Σ c_k^p)^{1/p}
    # = (1/p) * (Σ c_k^p)^{1/p - 1} * p * Σ c_k^{p-1} * dc_k/dh_i
    # = (Σ c_k^p)^{1/p - 1} * Σ c_k^{p-1} * dc_k/dh_i
    c_pos_p = np.sum(corr_pos ** p)
    factor = c_pos_p ** (1.0/p - 1.0)
    weights = corr_pos ** (p-1) * factor
    # dc_k/dh_i for k > 0:
    # corr[k] = Σ_i h[i]*(1-h[i+k])
    # d(corr[k])/d(h[i]) = (1-h[i+k]) [from the h[i] term] + ... 
    # Actually d/dh_i [Σ_j h[j]*(1-h[j+k])] = (1-h[i+k]) if i ≤ n-1-k, plus h[i-k] if i ≥ k (from the -h[i]*(1-h[i-k]) term with j=i-k... wait)
    
    # Let me be more careful. corr[k] = Σ_{j=0}^{n-1-k} h[j] * (1-h[j+k])
    # d(corr[k])/d(h[i]) = Σ_{j=0}^{n-1-k} [δ_{j,i}*(1-h[j+k]) + h[j]*(-δ_{j+k,i})]
    # = (1-h[i+k]) if i ≤ n-1-k (from j=i term)
    #   + (-h[i-k]) if i ≥ k (from j=i-k term, where j+k=i)
    # So: d(corr[k])/d(h[i]) = (1-h[i+k]) * 1[i≤n-1-k] - h[i-k] * 1[i≥k]
    
    # Total gradient: grad[i] = dx * Σ_k weights[k] * [(1-h[i+k])*1[i≤n-1-k] - h[i-k]*1[i≥k]]
    # = dx * [Σ_{k=0}^{n-1-i} weights[k] * (1-h[i+k]) - Σ_{k=i}^{n-1} weights[k] * h[k-i]]
    # Wait, let me redo this.
    # For a fixed i:
    #   Term 1 (from j=i in corr[k]): valid when i ≤ n-1-k, i.e., k ≤ n-1-i
    #     Contribution: weights[k] * (1-h[i+k]) for k = 0, 1, ..., n-1-i
    #   Term 2 (from j+k=i in corr[k], i.e., j=i-k): valid when i-k ≥ 0, i.e., k ≤ i
    #     Contribution: weights[k] * (-h[i-k]) for k = 0, 1, ..., i
    #     But wait, j must also satisfy j ≤ n-1-k, i.e., i-k ≤ n-1-k, i.e., i ≤ n-1. Always true.
    
    # So: grad[i] = dx * [Σ_{k=0}^{min(n-1-i, n-1)} weights[k] * (1-h[i+k]) - Σ_{k=0}^{min(i, n-1)} weights[k] * h[i-k]]
    
    # Using FFT:
    # First sum: for each i, Σ_k weights[k] * (1-h[i+k]) = (weights_convolved_with_(1-h))[i] at "negative" lags
    # This is the cross-correlation of weights with (1-h), evaluated at position i.
    
    # Let me use the same FFT trick as before.
    # Σ_k weights[k] * (1-h[i+k]) = (weights ⊛ (1-h))[i] where ⊛ is correlation
    # = Σ_k weights[k] * f[i+k] where f = 1-h
    # This is like (reverse(weights) * f)[i] in convolution... 
    # Actually, the standard correlation (a ⊛ b)[i] = Σ_k a[k] * b[i+k]
    # So this is (weights ⊛ (1-h))[i]
    
    # In FFT terms, correlation can be computed as:
    # corr(a, b)[i] = Σ_k a[k] * b[i+k]
    # = Σ_k reverse(a)[n-1-k] * b[i+k]
    # = (reverse(a) * b)[i + n - 1] (convolution)
    
    # Hmm, this is getting complicated. Let me just use the same approach as the LSE gradient.
    
    # Actually, the structure is the same! The gradient is:
    # grad[i] = dx * [Σ_k w[k] * (1-h[i+k]) - Σ_k w[k] * h[i-k]]
    # where the first sum is over k such that i+k < n, and the second over k such that i-k ≥ 0.
    
    # This is exactly the same form as the LSE gradient! So I can use the same FFT code.
    
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    
    return obj, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def periodic_optimize(n, target, period, alpha, max_iters, deadline):
    # Optimize a periodic solution with the given period
    p = period
    reps = n // p
    rem = n % p
    if rem > 0:
        # Handle remainder by adjusting
        # We'll just work with the first p values and tile
        pass
    
    p_target = p / 2.0  # sum of one period should be p/2 (approximately, if n is divisible by p)
    # Actually, if n = reps * p, then total target = reps * p_target, so p_target = p/2
    
    # Initialize
    x = np.random.RandomState(42).rand(p) * 0.5 + 0.25
    x = project_feasible(x, p_target)
    
    # Simple gradient descent on the periodic objective
    # We need to evaluate C5 on the full tiled array
    def periodic_c5(xp):
        h = np.tile(xp, reps)
        if rem > 0:
            h = np.concatenate([h, xp[:rem]])
        return compute_c5(h)
    
    best_x = x.copy()
    best_sc = periodic_c5(x)
    step = 0.01
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        h_full = np.tile(x, reps)
        if rem > 0:
            h_full = np.concatenate([h_full, x[:rem]])
        
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')[n-1:]
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = float(exp_shifted.sum())
        weights = exp_shifted / sum_exp
        
        # Compute gradient for the full array
        w_flipped = weights[::-1].copy()
        v = 1.0 - h_full
        fsize = 2 * n
        Wf = np.fft.rfft(w_flipped, fsize)
        V = np.fft.rfft(v, fsize)
        W = np.fft.rfft(weights, fsize)
        H = np.fft.rfft(h_full, fsize)
        corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
        corr_neg = np.fft.irfft(W * H, fsize)[:n]
        grad_full = (2.0/n) * (corr_pos - corr_neg)
        
        # Average gradient over periods
        grad_p = np.zeros(p)
        for j in range(reps):
            grad_p += grad_full[j*p:(j+1)*p]
        if rem > 0:
            grad_p[:rem] += grad_full[reps*p:]
        grad_p /= reps  # approximate
        
        x_new = x - step * grad_p
        x_new = project_feasible(x_new, p_target)
        x = x_new
        
        if it % 100 == 0:
            sc = periodic_c5(x)
            if sc < best_sc:
                best_sc = sc
                best_x = x.copy()
            # Adaptive step size
            if sc > best_sc * 1.01:
                step *= 0.9
            else:
                step = min(step * 1.01, 0.1)
    
    # Construct full solution
    h_full = np.tile(best_x, reps)
    if rem > 0:
        h_full = np.concatenate([h_full, best_x[:rem]])
    h_full = project_feasible(h_full, target)
    return h_full, compute_c5(h_full)

def simulated_annealing(h0, target, deadline, rng):
    n = len(h0)
    h_cur = h0.copy()
    h_best = h0.copy()
    sc_cur = compute_c5(h_cur)
    sc_best = sc_cur
    
    T_start = 0.02
    T_end = 0.0001
    n_steps = 5000000  # we'll limit by time
    
    step = 0
    total_time = 0.0
    
    while time.time() < deadline:
        step += 1
        # Annealing schedule
        progress = min(step / (n_steps * 0.8), 1.0)  # anneal in first 80%
        T = T_start * (T_end / T_start) ** progress
        
        # Propose a move: modify a random block
        block_size = max(3, int(20 * (1 - progress) + 3))
        start = rng.randint(0, n - block_size)
        end = start + block_size
        noise_scale = 0.05 + 0.1 * (1 - progress)
        
        h_new = h_cur.copy()
        h_new[start:end] += rng.randn(end - start) * noise_scale
        h_new = project_feasible(h_new, target)
        
        sc_new = compute_c5(h_new)
        delta = sc_new - sc_cur
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h_cur = h_new
            sc_cur = sc_new
            if sc_cur < sc_best:
                sc_best = sc_cur
                h_best = h_cur.copy()
    
    return h_best, sc_best

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo = -2.0
    hi = 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)
