            fc = _f1d(c)
            fd = _f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = _f1d(c)
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)
                    fd = _f1d(d)
            best_v = (a + b) / 2.0
            if abs(best_v - h_cur[j]) > 1e-10:
                h_cur[j] = best_v
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    improved = True
        if not improved:
            break
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0

    def tleft():
        return deadline - time.time()

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    mu = 50.0

    # Phase 1: L-BFGS-B annealing
    alphas_p1 = [5, 20, 100, 500, 2000, 10000, 50000, 200000, 500000]
    for a in alphas_p1:
        if tleft() < 40:
            break
        h_new = lbfgs_optimize(best_h, target, a, mu, 600, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Top-lag equalization with varying focus
    if tleft() > 80:
        for n_top in [3, 5, 8, 15]:
            if tleft() < 30:
                break
            h_eq, sc_eq = top_lag_equalize(best_h, target, n_top, 250, deadline)
            if sc_eq < best_score:
                best_score = sc_eq
                best_h = h_eq
            h_new = lbfgs_optimize(best_h, target, 100000, mu, 400, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 3: Direct subgradient on max
    if tleft() > 100:
        h_sg, sc_sg = direct_subgrad(best_h, target, 3000, deadline)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_new = lbfgs_optimize(best_h, target, 200000, mu, 400, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 4: Diverse restarts
    rng = np.random.RandomState(seed)
    n_cur = len(best_h)
    target_cur = n_cur / 2.0

    restart_types = []
    for scale in [0.03, 0.07, 0.12, 0.2, 0.35, 0.5]:
        restart_types.append(('gauss', scale))
    for lo, hi in [(0.0, 0.005), (0.005, 0.02), (0.02, 0.08), (0.08, 0.2), (0.2, 0.5)]:
        restart_types.append(('fourier', (lo, hi)))
    for frac in [0.1, 0.2, 0.35, 0.5]:
        restart_types.append(('binary_flip', frac))
    for frac in [0.1, 0.25, 0.4]:
        restart_types.append(('block_flip', frac))
    for scale in [0.05, 0.15]:
        restart_types.append(('smooth_gauss', scale))

    for strat_spec in restart_types:
        if tleft() < 70:
            break
        h_pert = best_h.copy()
        strat = strat_spec[0]
        if strat == 'gauss':
            h_pert = h_pert + rng.randn(n_cur) * strat_spec[1]
        elif strat == 'fourier':
            lo, hi = strat_spec[1]
            freqs = np.fft.rfftfreq(n_cur)
            mask = (freqs >= lo) & (freqs < hi)
            spec = np.zeros(n_cur // 2 + 1, dtype=complex)
            cnt = int(np.sum(mask))
            if cnt > 0:
                spec[mask] = (rng.randn(cnt) + 1j * rng.randn(cnt)) * 0.4
            h_pert = h_pert + np.fft.irfft(spec, n_cur)
        elif strat == 'binary_flip':
            n_flip = int(n_cur * strat_spec[1])
            idx = rng.choice(n_cur, size=n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif strat == 'block_flip':
            n_blocks = rng.randint(2, 6)
            for _ in range(n_blocks):
                bstart = rng.randint(0, n_cur)
                blen = rng.randint(max(2, int(n_cur * strat_spec[1] * 0.2)),
                                   max(3, int(n_cur * strat_spec[1])))
                bidx = (np.arange(bstart, bstart + blen)) % n_cur
                h_pert[bidx] = 1.0 - h_pert[bidx]
        elif strat == 'smooth_gauss':
            h_pert = h_pert + rng.randn(n_cur) * strat_spec[1]
            h_pert = np.convolve(h_pert, np.ones(9) / 9.0, mode='same')

        h_pert = project_feasible(h_pert, target_cur)
        h_new = lbfgs_optimize(h_pert, target_cur, 200, mu, 150, deadline)
        if tleft() > 40:
            h_new = lbfgs_optimize(h_new, target_cur, 10000, mu, 200, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Binary push and re-optimize
    if tleft() > 80:
        h_bin = binary_push_reopt(best_h, target_cur, deadline)
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin
            h_new = lbfgs_optimize(best_h, target_cur, 200000, mu, 300, deadline)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: Pairwise local search
    if tleft() > 80:
        h_pw, sc_pw = pairwise_improve(best_h, target_cur, 5000, deadline, seed + 1)
        if sc_pw < best_score:
            best_score = sc_pw
            best_h = h_pw
        h_new = lbfgs_optimize(best_h, target_cur, 100000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Final equalization + high-alpha L-BFGS
    if tleft() > 50:
        for n_top in [3, 5, 7]:
            if tleft() < 25:
                break
            h_eq, sc_eq = top_lag_equalize(best_h, target_cur, n_top, 150, deadline)
            if sc_eq < best_score:
                best_score = sc_eq
                best_h = h_eq
        h_new = lbfgs_optimize(best_h, target_cur, 500000, mu, 500, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Coordinate descent polish
    if tleft() > 60:
        h_cd, sc_cd = coordinate_descent_fast(best_h, target_cur, 3, deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 9: SLSQP with exact constraint
    if tleft() > 25:
        h_slq = slsqp_polish(best_h, target_cur, 50000, 10000, deadline)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 10: Final micro-improvements
    if tleft() > 20:
        h_pw2, sc_pw2 = pairwise_improve(best_h, target_cur, 3000, deadline, seed + 99)
        if sc_pw2 < best_score:
            best_score = sc_pw2
            best_h = h_pw2
        h_new = lbfgs_optimize(best_h, target_cur, 500000, mu, 300, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
