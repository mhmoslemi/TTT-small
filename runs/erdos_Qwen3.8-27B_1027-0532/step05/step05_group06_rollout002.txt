import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    # Same as before but maybe faster
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_and_grad(h, top_k=10):
    """Compute correlation and gradient w.r.t. top-k lags"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k lags
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for ki in top_idx:
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += (1.0 - h[j1 - lag])
            j2 = np.arange(0, n - lag)
            grad[j2] -= h[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] -= h[j2 - m]
    return corr, top_idx, grad / len(top_idx)

def lse_objective_and_grad(x, alpha, target, mu):

    n = len(x)

    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    
    grad = np.zeros(n)

        lag = int(ki) - (n - 1)

            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:

            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]

    momentum = 0.7

            break
        
        # Nesterov interpolation
        beta = (t - 1) / (t + 2) if t > 1 else 0
        h_y = h + beta * (h - h_nesterov) if t > 1 else h

        h_new = project_feasible(h_new, target)

            best_sc = sc
            best_h = h_new.copy()

        else:
            lr *= 0.85
            v *= 0.5
            if lr < 1e-10:
                break
        
        h_nesterov = h

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def lse_obj_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)

        return h_cur
