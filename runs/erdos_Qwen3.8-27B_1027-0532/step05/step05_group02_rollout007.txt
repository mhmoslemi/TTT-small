  * max over non-peak s of (c[s] + ε*d[s]) (piecewise linear in ε)

The optimum is where these two curves cross, or at ε=1, or at a kink point.

This can be solved analytically or by bisection.

def soft_swap_optimal_eps(h, i, j, c, n):
    """Find optimal epsilon for moving mass from i to j."""
    # d[s] = h[i-s] + h[i+s] - h[j-s] - h[j+s]
    # s ranges from -(n-1) to n-1
    s = np.arange(1 - n, n)  # shifts
    
    # Compute d safely (out-of-bounds treated as 0)
    di_plus = np.zeros(2*n-1)
    idx = i + s
    valid = (idx >= 0) & (idx < n)
    di_plus[valid] = h[idx[valid]]
    
    di_minus = np.zeros(2*n-1)
    idx = i - s
    valid = (idx >= 0) & (idx < n)
    di_minus[valid] = h[idx[valid]]
    
    dj_plus = np.zeros(2*n-1)
    idx = j + s
    valid = (idx >= 0) & (idx < n)
    dj_plus[valid] = h[idx[valid]]
    
    dj_minus = np.zeros(2*n-1)
    idx = j - s
    valid = (idx >= 0) & (idx < n)
    dj_minus[valid] = h[idx[valid]]
    
    d = di_plus + di_minus - dj_plus - dj_minus
    
    # c_new[s] = c[s] + eps * d[s]
    # Want to minimize max_s c_new[s]
    
    # For eps=0: max is c_max
    # For eps=1: max is max(c + d)
    
    # The optimal is where the decreasing curve (from peaks) meets the increasing curve (from non-peaks)
    c_max = float(c.max())
    
    # Peaks: indices where c[s] == c_max
    peak_idx = np.where(c >= c_max - 1e-10)[0]
    non_peak_idx = np.where(c < c_max - 1e-10)[0]
    
    # For peaks, c_new = c_max + eps*d[peak], which decreases if d[peak] < 0
    # The "peaking" curve is: c_max + eps * min(d[peak_idx])
    d_peak_min = float(d[peak_idx].min()) if len(peak_idx) > 0 else 0
    
    if d_peak_min >= 0:
        # Even moving mass doesn't help the peaks, return eps=0
        return 0.0, c_max
    
    # For non-peaks, the "rival" curve is max over non-peak of (c[s] + eps*d[s])
    # At the optimal point: c_max + eps*d_peak_min = max_non_peak(c + eps*d)
    
    # Use bisection on eps
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        c_new = c + mid * d
        new_max = float(c_new.max())
        
        # Is the peak still the max?
        if c_max + mid * d_peak_min > new_max * 0.999:
            # Peak is still dominant, can increase eps
            lo = mid
        else:
            # A non-peak has caught up or exceeded
            hi = mid
    
    # Actually, let me just do a simple grid search for eps
    eps_vals = np.linspace(0, 1, 1001)
    best_c = c_max
    best_eps = 0.0
    for eps in eps_vals:
        c_new = c + eps * d
        cm = float(c_new.max())
        if cm < best_c:
            best_c = cm
            best_eps = eps
    
    return best_eps, float((c + best_eps * d).max())

For the soft swap, given current h, peak shift k*, and positions i (h[i]=1) and j (h[j]=0):
- Δc[s] = ε * (h[i-s] + h[i+s] - h[j-s] - h[j+s])
- We want to find ε ∈ [0,1] that minimizes max_s (c[s] + ε*Δc[s]/ε) = max_s (c[s] + ε*d[s])
  where d[s] = h[i-s] + h[i+s] - h[j-s] - h[j+s]

The optimal ε can be found by:
- The peak value as function of ε: V(ε) = max_s (c[s] + ε*d[s])
- V(0) = c_max
- V is piecewise linear and convex in ε (max of linear functions)
- We want argmin_{ε∈[0,1]} V(ε)

Since V is convex piecewise linear, the minimum is either at ε=0, ε=1, or at a kink point.

A kink point occurs where two different s values give the same c[s] + ε*d[s] value.

For efficiency, I'll just do a coarse-to-fine search:
1. Check ε = 0, 0.1, 0.2, ..., 1.0
2. Refine around the best

Actually, for a faster approach: since V(ε) is the upper envelope of lines c[s] + ε*d[s], and we're looking for the minimum of this envelope over [0,1], we can note:

V(ε) is decreasing where the active s (the one giving the max) has d[s] < 0, and increasing where the active s has d[s] > 0.

The optimal ε is where the active s switches from one with d < 0 to one with d > 0.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:need]:
            h_r[ones[idx]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in order[:need]:
            h_r[zeros[idx]] = 1
    return h_r.astype(float)

def compute_delta_swap(h, i, j):
    """Compute the change in full correlation when we do h[i]-=1, h[j]+=1."""
    n = len(h)
    s = np.arange(1 - n, n)
    # d[s] = h[i-s] + h[i+s] - h[j-s] - h[j+s]
    di_s = np.zeros(2*n-1)
    idx = i - s
    valid = (idx >= 0) & (idx < n)
    di_s[valid] = h[idx[valid]]
    
    di_ps = np.zeros(2*n-1)
    idx = i + s
    valid = (idx >= 0) & (idx < n)
    di_ps[valid] = h[idx[valid]]
    
    dj_s = np.zeros(2*n-1)
    idx = j - s
    valid = (idx >= 0) & (idx < n)
    dj_s[valid] = h[idx[valid]]
    
    dj_ps = np.zeros(2*n-1)
    idx = j + s
    valid = (idx >= 0) & (idx < n)
    dj_ps[valid] = h[idx[valid]]
    
    return di_s + di_ps - dj_s - dj_ps

def soft_swap_optimal(h, i, j, c):

    n = len(h)
    d = compute_delta_swap(h, i, j)

    eps_vals = np.array([0.0, 0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0])

def slsqp_refine(h_start, alphas, maxiter, time_limit, start_time):

        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break

                         options={'maxiter': maxiter, 'ftol': 1e-14})

            if sc < curr_score:

        except Exception:

            stall += 1

def sa_continuous(h, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h)
    target = n / 2.0
    h = h.copy().astype(float)
    c = full_corr(h)
    dx = 2.0 / n
    best_c5 = float(c.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 80000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        amount = rng.uniform(0.02, 0.15)
        if h[i] < amount or h[j] > 1.0 - amount:
            continue

        h[i] -= amount
        h[j] += amount
        c = full_corr(h)
        new_c5 = float(c.max()) * dx
        delta_c5 = new_c5 - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i] += amount
            h[j] -= amount
            stall += 1

    return best_h, best_c5

def _smooth_obj(h, alpha):
    corr = full_corr(h)
    c_max = float(corr.max())
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

_slq_n = 0
_slq_target = 0.0

def _slq_cons_fun(x):
    return np.sum(x) - _slq_target

def _slq_cons_jac(x):
    return np.ones(_slq_n)

def slsqp_refine(h_start, alphas, maxiter, time_limit, start_time):
    global _slq_n, _slq_target
    n = len(h_start)
    target = n / 2.0
    _slq_n = n
    _slq_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slq_cons_fun, 'jac': _slq_cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _slq_n, _slq_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_binary(h_init, int(target))
    best_score = compute_c5(best_h)

    def remaining():
        return budget_s - 10.0 - (time.time() - start_time)

    # Phase 1: Peak-focused binary local search
    allowed = min(120, remaining() * 0.12)
    h1, c5_1 = peak_binary_search(best_h, max_iters=100000, rng=rng,
                                   time_limit=allowed, start_time=start_time)
    if c5_1 < best_score:
        best_score = c5_1
        best_h = h1.copy()

    # Phase 2: Binary SA
    if remaining() > 80:
        allowed = min(180, remaining() * 0.18)
        n_sa = int(min(3000000, allowed * 40000))
        h2, c5_2 = sa_binary(best_h, n_sa, 0.015, 0.00005, rng,
                              time_limit=allowed, start_time=start_time)
        if c5_2 < best_score:
            best_score = c5_2
            best_h = h2.copy()

    # Phase 3: Continuous soft-swap refinement (single peak)
    if remaining() > 80:
        allowed = min(150, remaining() * 0.15)
        h3, c5_3 = continuous_refine(best_h, max_iters=30000,
                                      time_limit=allowed, start_time=start_time)
        if c5_3 < best_score:
            best_score = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-peak continuous refinement
    if remaining() > 80:
        allowed = min(150, remaining() * 0.15)
        h4, c5_4 = multi_peak_refine(best_h, max_iters=30000,
                                      time_limit=allowed, start_time=start_time)
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4.copy()

    # Phase 5: SLSQP refinement
    if remaining() > 60:
        allowed = min(180, remaining() * 0.2)
        alphas = [50, 100, 300, 1000, 3000, 10000]
        h5, c5_5 = slsqp_refine(best_h, alphas, 2000, time_limit=allowed,
                                 start_time=start_time)
        if c5_5 < best_score:
            best_score = c5_5
            best_h = h5.copy()

    # Phase 6: Iterated perturb + refine
    for it in range(25):
        if remaining() < 45:
            break

        h_pert = best_h.copy()
        n_pert = rng.randint(2, 6)
        idx1 = rng.choice(n, n_pert, replace=False)
        idx2 = rng.choice(n, n_pert, replace=False)
        for a, b in zip(idx1, idx2):
            if a != b:
                amt = min(h_pert[a], 0.08) * rng.uniform(0.5, 1.0)
                if amt > 0.005:
                    h_pert[a] -= amt
                    h_pert[b] += amt
        h_pert = project_feasible(h_pert, target)

        allowed = min(30, remaining() * 0.04)
        if allowed < 5:
            break
        h6, c5_6 = continuous_refine(h_pert, max_iters=10000,
                                      time_limit=allowed, start_time=start_time)
        if c5_6 < best_score:
            best_score = c5_6
            best_h = h6.copy()

        if remaining() < 45:
            break
        allowed2 = min(25, remaining() * 0.03)
        if allowed2 < 5:
            break
        alphas_small = [500, 2000, 8000]
        h6b, c5_6b = slsqp_refine(h6, alphas_small, 1000, time_limit=allowed2,
                                   start_time=start_time)
        if c5_6b < best_score:
            best_score = c5_6b
            best_h = h6b.copy()

    # Phase 7: Try different resolutions
    for dn in [-8, -4, 4, 8, 12]:
        if remaining() < 40:
            break
        n_try = n + dn
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, int(target_try))

        allowed = min(20, remaining() * 0.04)
        h7, c5_7 = peak_binary_search(h_bin_try, max_iters=30000, rng=rng,
                                       time_limit=allowed, start_time=start_time)
        if c5_7 < best_score:
            best_score = c5_7
            best_h = h7.copy()

        if remaining() < 35:
            continue
        allowed2 = min(15, remaining() * 0.03)
        h7b, c5_7b = continuous_refine(h7, max_iters=8000,
                                        time_limit=allowed2, start_time=start_time)
        if c5_7b < best_score:
            best_score = c5_7b
            best_h = h7b.copy()

    # Phase 8: Final SLSQP with high alpha
    if remaining() > 30:
        allowed = min(50, remaining() - 10)
        alphas_final = [5000, 10000, 20000, 50000]
        h8, c5_8 = slsqp_refine(best_h, alphas_final, 1500, time_limit=allowed,
                                 start_time=start_time)
        if c5_8 < best_score:
            best_score = c5_8
            best_h = h8.copy()

    # Phase 9: Final continuous micro-refinement
    if remaining() > 15:
        allowed = min(20, remaining() - 5)
        h9, c5_9 = continuous_refine(best_h, max_iters=10000,
                                      time_limit=allowed, start_time=start_time)
        if c5_9 < best_score:
            best_score = c5_9
            best_h = h9.copy()

    h_final = project_feasible(best_h, float(len(best_h)) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
