    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def greedy_search(h_start, max_iters, rng, time_limit, start_time, top_k=8):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    positions = np.arange(n)
    stall = 0
    max_stall = 50

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        max_val = float(full.max())
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0 or k_star == 0:
            break

        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]

        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]

        tk = min(top_k, len(ones), len(zeros))
        top_i = ones[np.argsort(A[ones])[:tk]]
        top_j = zeros[np.argsort(-B[zeros])[:tk]]

        found = False
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
            if found:
                break

        if not found:
            n_rand = min(10, len(ones), len(zeros))
            for _ in range(n_rand):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                new_max = float((full + delta).max())
                if new_max < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    new_c5 = float(full.max()) * dx
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                    found = True
                    break
        if not found:
            stall += 1

    return best_h, best_c5

def iterated_local_search(h_start, rng, time_limit, start_time, n_kicks=300):
    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)
    kick_sizes = [1, 1, 2, 2, 3, 3, 4, 5, 5, 7, 8, 10, 13, 17, 21]
    kick_idx = 0
    kicks_done = 0
    patience = 0
    max_patience = 30

    while kicks_done < n_kicks:
        if time.time() - start_time > time_limit - 5:
            break

        ks = kick_sizes[kick_idx % len(kick_sizes)]
        kick_idx += 1

        h_kick = best_h.copy()
        ones = np.where(h_kick == 1.0)[0]
        zeros = np.where(h_kick == 0.0)[0]
        for _ in range(ks):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_kick[i] = 0.0
            h_kick[j] = 1.0
            ones = np.where(h_kick == 1.0)[0]
            zeros = np.where(h_kick == 0.0)[0]

        alloc = min(3.0, (time_limit - (time.time() - start_time)) * 0.1)
        h_res, c5_res = greedy_search(h_kick, max_iters=2000, rng=rng,
                                      time_limit=alloc, start_time=start_time)
        kicks_done += 1

        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
            patience = 0
            kick_idx = 0
        else:
            patience += 1

        if patience > max_patience:
            kick_sizes = np.array(kick_sizes) + rng.randint(0, 5)
            kick_sizes = np.clip(kick_sizes, 1, 50).tolist()
            patience = 0

    return best_h, best_c5

def sa_search(h_start, n_iters, t_init, t_final, rng, time_limit, start_time, focused=False):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    positions = np.arange(n)
    accept_count = 0
    total_count = 0

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if focused and it % 3 == 0:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                i_cands = ones[A[ones] <= 1.0]
                j_cands = zeros[B[zeros] >= 1.0]
                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        delta = compute_delta(h, i, j)
        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        total_count += 1

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            accept_count += 1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5

def pair_swap_sa(h_start, n_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 20000

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break

        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break

        i1 = ones[rng.randint(len(ones))]
        j1 = ones[rng.randint(len(ones))]
        if i1 == j1:
            continue
        k1 = zeros[rng.randint(len(zeros))]
        k2 = zeros[rng.randint(len(zeros))]
        if k1 == k2:
            continue

        delta = compute_delta(h, i1, k1)
        delta += compute_delta(np.where(np.arange(n) == j1, 1.0 - h[j1], h), j1, k2)

        new_max = float((full + delta).max())
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5

        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i1] = 0.0
            h[j1] = 0.0
            h[k1] = 1.0
            h[k2] = 1.0
            full += delta
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

    return best_h, best_c5
