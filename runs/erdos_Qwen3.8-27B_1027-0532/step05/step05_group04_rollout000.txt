            grad_iters=40, lr_init=0.0008, deadline=deadline4)
        if sc_f1 < best_score:
            best_score = sc_f1
            h_best = h_f1.copy()
        if remaining() > 60:
            h_s600 = slsqp_smooth(h_best, n1, target1, alpha=5000, maxiter=2000,
                                   ftol=1e-16, deadline=start_time + 730)
            sc = compute_c5(h_s600)
            if sc < best_score:
                best_score = sc
                h_best = h_s600.copy()

    # Phase 5: Interpolate to n=800 and optimize
    n2 = 800
    target2 = n2 / 2.0
    if remaining() > 180:
        h_fine2 = interpolate_h(h_best, n2, target2)
        sc_f2 = compute_c5(h_fine2)
        if sc_f2 < best_score:
            best_score = sc_f2
            h_best = h_fine2.copy()
        deadline5 = start_time + 880
        h_f2, sc_f2o = reweight_gradient_descent(
            h_best, n2, target2,
            n_rounds=6, beta_init=300, beta_growth=1.4,
            grad_iters=30, lr_init=0.0005, deadline=deadline5)
        if sc_f2o < best_score:
            best_score = sc_f2o
            h_best = h_f2.copy()
        if remaining() > 50:
            h_s800 = slsqp_smooth(h_best, n2, target2, alpha=8000, maxiter=1500,
                                   ftol=1e-16, deadline=start_time + 940)
            sc = compute_c5(h_s800)
            if sc < best_score:
                best_score = sc
                h_best = h_s800.copy()

    # Phase 6: Multi-start perturbation at best resolution
    n_final = len(h_best)
    target_final = n_final / 2.0
    if remaining() > 120:
        deadline6 = start_time + budget_s - 120
        h_ms, sc_ms = multi_start(
            h_best, n_final, target_final, rng,
            n_starts=4, per_start_budget=25, deadline=deadline6)
        if sc_ms < best_score:
            best_score = sc_ms
            h_best = h_ms.copy()

    # Phase 7: Subgradient refinement
    if remaining() > 60:
        deadline7 = start_time + budget_s - 50
        h_sg, sc_sg = subgradient_refine(
            h_best, n_final, target_final,
            n_steps=1000, n_active=5, lr_init=0.0008, lr_decay=0.998,
            deadline=deadline7)
        if sc_sg < best_score:
            best_score = sc_sg
            h_best = h_sg.copy()

    # Phase 8: Final SLSQP polish
    if remaining() > 25:
        h_fin = slsqp_smooth(h_best, n_final, target_final,
                             alpha=20000, maxiter=4000,
                             ftol=1e-17, deadline=start_time + budget_s - 10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            h_best = h_fin.copy()

    # Phase 9: Last subgradient micro-steps
    if remaining() > 10:
        h_sg2, sc_sg2 = subgradient_refine(
            h_best, n_final, target_final,
            n_steps=200, n_active=8, lr_init=0.0003, lr_decay=0.999,
            deadline=start_time + budget_s - 3)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            h_best = h_sg2.copy()

    h_final = project_feasible(h_best, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
