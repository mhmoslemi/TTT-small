        h_ref = slsqp_multi_alpha(best_h, n, target, [10000, 50000], 4000, 1e-17, deadline=deadline_p6)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Block coordinate descent
    if tleft() > 80:
        deadline_p7 = start_time + 900
        for bs in [4, 3]:
            if tleft() < 40:
                break
            h_bd = block_cd(best_h, n, target, bs, 4, 5000.0, deadline=deadline_p7)
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    # Phase 8: Projected gradient fine-tuning
    if tleft() > 60:
        deadline_p8 = start_time + 1000
        h_pg2, sc_pg2 = proj_grad_adaptive(best_h, n, target, 5000, 80000, 500, 0.002, deadline=deadline_p8)
        if sc_pg2 < best_score:
            best_score = sc_pg2
            best_h = h_pg2

    # Phase 9: Perturbation restarts
    if tleft() > 60:
        for eps in [0.01, 0.005, 0.02]:
            if tleft() < 30:
                break
            h_pert = perturb_restart(best_h, n, target, eps, rng)
            h_try = slsqp_multi_alpha(h_pert, n, target, [2000, 10000], 3000, 1e-16,
                                      deadline=start_time + budget_s - 80)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try

    # Phase 10: Final SLSQP polish
    if tleft() > 40:
        h_final = slsqp_multi_alpha(best_h, n, target, [30000, 100000], 4000, 1e-17,
                                    deadline=start_time + budget_s - 20)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 11: Final subgradient micro-polish
    if tleft() > 20:
        h_sg2, sc_sg2 = subgradient_fine(best_h, n, target, 200, 8, 0.001, 0.001,
                                         deadline=start_time + budget_s - 10)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2

    h_out = project_feasible(best_h, target)
    final_score = compute_c5(h_out)
    return h_out, final_score, n
