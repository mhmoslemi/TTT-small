    """Project h onto {x: sum x = target_sum, 0 <= x <= 1} using water-filling."""
    n = len(h)
    y = h.astype(float).copy()

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_gradient_step(h, alpha, lr, target_sum):
    """One step of gradient descent on smooth max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    exp_s = np.exp(shifted)

            grad[k:] += w * (1 - h[:n-k]) * dx
        else:
            grad += w * (1 - 2*h) * dx
    
    f_new = h - lr * grad
    f_new = project_feasible(f_new, target_sum)
    return f_new, f_val, c_max

                       lr_init=0.1, cooldown=50):

    start = time.time()

    h = project_feasible(h, target_sum)

    best_c5 = compute_c5(h)

                best_h = h_new.copy()
                h = h_new
                no_improve = 0
                lr = min(lr * 1.05, 0.5)
            else:
                no_improve += 1
                if no_improve > cooldown:
                    lr *= 0.5
                    no_improve = 0
                    h = best_h.copy()
    
    return best_h, best_c5

def project_feasible(h, target_sum):
    n = len(h)
    y = h.astype(float).copy()
    lo = y.min() - 1.0
    hi = y.max() + 1.0

def lp_refine(h_init, target_sum, time_limit, n_top=200, max_iter=5000):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        # Get top lags
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(corr)[::-1][:n_top]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        for idx in top_idx:
            lag = idx - (n - 1)
            lo = max(0, lag)
            hi = min(n, n + lag)
            if hi > lo:
                coeff = g[lo - lag:hi - lag] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        # Regularization
        reg = 1e-5
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps=1e-9)
        except:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h = 0.8 * h + 0.2 * h_new
        h = project_feasible(h, target_sum)
        
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize

def project_feasible(h, target_sum):
    n = len(h)
    y = h.astype(float).copy()
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(y - tau, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = corr.max()
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    w_sum = exp_s.sum()
    weights = exp_s / w_sum
    f_val = c_max + np.log(w_sum) / alpha
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for lag_idx in active:
        w = weights[lag_idx]
        lag = int(lag_idx) - (n - 1)
        if lag > 0:
            grad[:n - lag] += w * (1.0 - h[lag:]) * dx
            grad[lag:] -= w * h[:n - lag] * dx
        elif lag < 0:
            k = -lag
            grad[:n - k] -= w * h[k:] * dx
            grad[k:] += w * (1.0 - h[:n - k]) * dx
        else:
            grad += w * (1.0 - 2.0 * h) * dx
    return f_val, grad, c_max

def adaptive_smooth_max_descent(h_init, target_sum, time_limit, lr_init=0.1):
    start = time.time()
    n = len(h_init)
    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = compute_overlaps(h)
        c_max = corr.max()
        sorted_c = np.sort(corr)[::-1]
        gap5 = sorted_c[0] - sorted_c[min(5, len(sorted_c) - 1)]
        if gap5 < 1e-5:
            alpha = 80.0
        elif gap5 < 5e-5:
            alpha = 200.0
        elif gap5 < 2e-4:
            alpha = 500.0
        elif gap5 < 1e-3:
            alpha = 1500.0
        else:
            alpha = 5000.0

        f_val, grad, c_max2 = smooth_max_gradient(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new

def lp_refine(h_init, target_sum, time_limit, n_top=300, damping=0.6, reg=1e-5, max_iter=5000):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_lags = 2 * n - 1

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        corr = compute_overlaps(h)
        top_idx = np.argsort(corr)[::-1][:n_top]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            coeff = np.zeros(n)
            if lag >= 0:
                coeff[:n - lag] = g[lag:] * dx
            else:
                k = -lag
                coeff[k:] = g[:n - k] * dx
            constraints.append(coeff @ h_var <= t_var)

        progress = min(1.0, it / max(1, max_iter * 0.5))
        reg_eff = reg * (0.01 ** progress)
        objective = t_var + reg_eff * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-10, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target_sum)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h_init, target_sum, time_limit, n_epochs=50):
    start = time.time()
    n = len(h_init)
    dx = 2.0 / n
    h = project_feasible(h_init, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    order = np.arange(n)

    for epoch in range(n_epochs):
        if time.time() - start > time_limit:
            break
        np.random.shuffle(order)
        for i in order:
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            best_local = old_val
            best_local_c5 = compute_c5(h)
            for trial in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(trial - old_val) < 1e-12:
                    continue
                h[i] = trial
                residual = target_sum - np.sum(h)
                if abs(residual) > 1e-10:
                    other = np.where(np.arange(n) != i)[0]
                    mask = other[(h[other] < 1.0) if residual > 0 else (h[other] > 0.0)]
                    if len(mask) > 0:
                        h[mask] += residual / len(mask)
                        h = np.clip(h, 0.0, 1.0)
                c5_try = compute_c5(h)
                if c5_try < best_local_c5 - 1e-15:
                    best_local_c5 = c5_try
                    best_local = trial
            h[i] = best_local
            h = project_feasible(h, target_sum)
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5
