```python
        return budget_s - 10.0 - elapsed()
```

Actually, for the min-max of quadratic functions, a sequential QP approach would be:
1. At h*, identify active lags A = {k : f_k(h*) ≥ max - ε}
2. Solve: min t s.t. f_k(h) ≤ t for k in A, 0≤h≤1, sum(h)=target
3. Since f_k is quadratic, this is a QCP (quadratically constrained program)
4. We can linearize: f_k(h) ≈ f_k(h*) + ∇f_k(h*)^T (h - h*)
5. Then it becomes a QP (or LP if we also linearize the objective)

```python
import cvxopt
from cvxopt import matrix, sparse

def sqp_step(h, n, target, eps=0.01):
    """Solve linearized SQP subproblem."""
    corr = compute_corr(h)
    max_corr = corr.max()
    active = np.where(corr >= max_corr - eps)[0]
    
    # Build the linearized problem
    # min t
    # s.t. grad_f_k(h)^T (h_new - h) + f_k(h) <= t for k in active
    #      0 <= h_new <= 1
    #      sum(h_new) = target
    
    # This is an LP! (linear objective, linear constraints)
    
    # Variables: t, h_new[0], ..., h_new[n-1]
    # Total variables: n + 1
    
    # Objective: minimize t (index 0)
    c = matrix(np.zeros(n+1))
    c[0] = 1.0
    
    # Inequality constraints: G * x <= h_vec
    # For each active lag k: sum_j grad_k[j] * h_new[j] + const_k - t <= 0
    # Rearranged: t - sum_j grad_k[j] * h_new[j] >= const_k
    # i.e., -t + sum_j grad_k[j] * h_new[j] <= -const_k
    
    # Also: 0 <= h_new[j] <= 1, which is:
    # h_new[j] <= 1 (i.e., h_new[j] - 1 <= 0)
    # -h_new[j] <= 0
    
    # Equality: sum(h_new) = target
    
    n_active = len(active)
    # G rows: n_active (for f_k ≤ t) + n (for h_new ≤ 1) + n (for -h_new ≤ 0)
    # h_vec: corresponding RHS
    
    G = np.zeros((2*n + n_active, n+1))
    h_vec = np.zeros(2*n + n_active)
    
    for i, k in enumerate(active):
        # f_k(h_new) ≈ f_k(h) + grad_k^T (h_new - h) <= t
        # grad_k^T h_new - t <= f_k(h) - grad_k^T h
        grad_k = compute_single_grad(k, h, n)  # gradient of f_k at h
        ri = i
        G[ri, 0] = -1.0  # coefficient of t
        G[ri, 1:] = grad_k
        h_vec[ri] = corr[k] - np.dot(grad_k, h)  # f_k(h) - grad_k^T h
    
    # h_new[j] <= 1
    for j in range(n):
        ri = n_active + j
        G[ri, j+1] = 1.0
        h_vec[ri] = 1.0
    
    # -h_new[j] <= 0
    for j in range(n):
        ri = n_active + n + j
        G[ri, j+1] = -1.0
        h_vec[ri] = 0.0
    
    # Equality: sum(h_new) = target
    A = np.zeros((1, n+1))
    A[0, 1:] = 1.0
    b = matrix(np.array([target]))
    
    # Solve LP using CVXOPT
    c_cvx = matrix(c)
    G_cvx = matrix(G)
    h_cvx = matrix(h_vec)
    A_cvx = matrix(A)
    
    result = cvxopt.solvers.lp(c_cvx, G_cvx, h_cvx, A_cvx, b)
    
    if result['status'] == 'optimal':
        h_new = np.array(result['x'])[1:]
        h_new = project_feasible(h_new, target)
        return h_new
    
    return h
```

This is an upper bound on max_k corr_k (since log-sum-exp ≥ max). As α → ∞, F → max.

We minimize F using gradient descent:
h ← h - step * ∇F(h)

∇F(h)[j] = sum_k w_k * ∂corr_k/∂h[j]

where w_k = exp(α*corr_k) / sum_m exp(α*corr_m) = softmax(α * corr)[k]

Now, ∂corr_k/∂h[j]:
corr_k = sum_i h[i] * (1 - h[i+k]) (for the full correlation, with appropriate index handling)

Let me denote the full correlation index as m = 0, ..., 2n-2, corresponding to lag L = m - (n-1).

corr_m = sum_{i: 0≤i<n, 0≤i+L<n} h[i] * (1-h[i+L]) where L = m-(n-1)

∂corr_m/∂h[j] = (1-h[j+L])*[0≤j+L<n] - h[j-L]*[0≤j-L<n]

This is what I had before. Let me code it up efficiently.

For efficiency, I'll use the fact that for each j, the sum over m can be reorganized:

g[j] = sum_m w_m * (1-h[j+L_m]) * [0≤j+L_m<n] - sum_m w_m * h[j-L_m] * [0≤j-L_m<n]

First term: let k = j+L_m, so L_m = k-j, m = k-j+(n-1)
= sum_{k: 0≤k<n, |k-j|≤n-1} w[k-j+(n-1)] * (1-h[k])
= sum_{k=0}^{n-1} w[k-j+(n-1)] * (1-h[k])  (since |k-j| ≤ n-1 always for k,j in [0,n-1])

Second term: let k = j-L_m, so L_m = j-k, m = j-k+(n-1)
= sum_{k: 0≤k<n, |j-k|≤n-1} w[j-k+(n-1)] * h[k]
= sum_{k=0}^{n-1} w[j-k+(n-1)] * h[k]

So: g[j] = sum_{k=0}^{n-1} w[k-j+n-1]*(1-h[k]) - sum_{k=0}^{n-1} w[j-k+n-1]*h[k]
= sum_{k=0}^{n-1} w_ext[k-j]*(1-h[k]) - sum_{k=0}^{n-1} w_ext[j-k]*h[k]

where w_ext[d] = w[d+(n-1)] for d in [-(n-1), n-1] (and 0 otherwise).

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15 * max(1.0, target_sum):
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_gradient(h, w, n):
    """Compute gradient of sum_k w[k] * corr_k w.r.t. h.
    w: array of length 2n-1 (softmax weights for each correlation index)
    """
    g = np.zeros(n)
    omh = 1.0 - h
    for m in range(2 * n - 1):
        wl = w[m]
        if wl < 1e-15:
            continue
        L = m - (n - 1)
        if L >= 0:
            g[:n - L] += wl * omh[L:]
            g[L:] -= wl * h[:n - L]
        else:
            M = -L
            g[M:] += wl * omh[:n - M]
            g[:n - M] -= wl * h[M:]
    return g


def grad_descent(h0, n, target, deadline, rng):
    """Multi-alpha gradient descent with adaptive step size."""
    h = h0.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    alphas = [5, 15, 50, 150, 500, 1500, 5000]
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_h, best_score
    
    # Give more time to later (larger alpha) phases
    weights = [0.5, 0.7, 1.0, 1.3, 1.6, 2.0, 2.5]
    total_w = sum(weights)
    times = [remaining * w / total_w for w in weights]
    
    for i, alpha in enumerate(alphas):
        end = time.time() + times[i]
        if time.time() >= deadline or end > deadline:
            break
        end = min(end, deadline)
        
        step = 0.02
        stall = 0
        momentum = np.zeros(n)
        
        while time.time() < end:
            corr = compute_corr(h)
            cx = corr - corr.max()
            w = np.exp(alpha * cx)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros_like(corr)
                w[np.argmax(corr)] = 1.0
            else:
                w /= ws
            
            g = compute_gradient(h, w, n)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            
            # Momentum update (heavy ball)
            momentum = 0.7 * momentum - step * g / gn
            direction = momentum / (np.linalg.norm(momentum) + 1e-30)
            
            # Line search
            improved = False
            for trial_factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
                s = step * trial_factor
                h_try = project_feasible(h + s * direction, target)
                f_try = compute_c5(h_try)
                if f_try < best_score:
                    best_score = f_try
                    best_h = h_try.copy()
                    h = h_try
                    improved = True
                    break
                elif trial_factor == 0.05:
                    h = h_try  # move anyway to explore
            
            if improved:
                stall = 0
                step = min(step * 1.05, 0.1)
            else:
                stall += 1
                step *= 0.8
                if stall > 100:
                    h = best_h + rng.randn(n) * 0.005
                    h = project_feasible(h, target)
                    momentum = np.zeros(n)
                    stall = 0
                    step = 0.01
        
        h = best_h.copy()
    
    return best_h, best_score


def slsqp_polish(h, n, target, alpha, maxiter=5000, ftol=1e-17):
    """SLSQP with log-sum-exp smoothing."""
    def obj(x):
        corr = compute_corr(x)
        cm = float(np.max(corr))
        sh = alpha * (corr - cm)
        sh = np.clip(sh, -700, 0)
        return cm + np.log(np.sum(np.exp(sh))) / alpha
    
    def obj_grad(x):
        corr = compute_corr(x)
        cm = float(np.max(corr))
        sh = alpha * (corr - cm)
        w = np.exp(np.clip(sh, -700, 0))
        ws = w.sum()
        if ws < 1e-300:
            return np.zeros(n)
        w /= ws
        g = compute_gradient(x, w, n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h
```
