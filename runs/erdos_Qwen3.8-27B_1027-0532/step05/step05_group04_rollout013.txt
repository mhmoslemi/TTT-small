Also, I want to try a key different idea: **using the dual variable (Lagrange multiplier) for the sum constraint more effectively**. Instead of using SLSQP which handles the constraint internally, I can use a penalty method or an explicit Lagrangian:

minimize f(h) + lambda * (sum(h) - n/2)^2 + mu * sum((h[i])^2 indicator[h[i]<0] + (1-h[i])^2 indicator[h[i]>1])

And do gradient descent on this. The sum constraint can be enforced exactly by projecting after each step (which is cheap: just add a constant to all non-saturated coordinates).

    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx


def project_feasible(h, target_sum):

        else:

def grad_overlap(h, k, n):
    """Gradient of overlap at lag k with respect to h."""
    grad = np.zeros(n)
    j_pos = np.arange(0, n - k)
    grad[j_pos] += (1.0 - h[j_pos + k])
    j_neg = np.arange(k, n)
    grad[j_neg] -= h[j_neg - k]
    return grad

    _g_target = target

            if sc < cur_score:

                cur_score = sc
        except Exception:

    return h_cur, cur_score


def nesterov_reweighted(h, n, target, n_steps, beta_init, beta_growth, lr_init,
                         deadline=None, nesterov=True):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    beta = beta_init
    lr = lr_init
    v = np.zeros(n) if nesterov else None
    t = 0
    best_h = h_cur.copy()
    best_score = cur_score
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        over = corr[n-1:n-1+n]
        max_over = float(np.max(over))
        weights = np.exp(beta * (over - max_over))
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k in range(n):
            wk = weights[k]
            if wk < 1e-10:
                continue
            grad += wk * grad_overlap(h_cur, k, n)
        grad *= dx
        
        if nesterov:
            t += 1
            beta_t = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            v_new = ((beta_t - 1.0) / (1.0 + t)) * v + ((1.0 + t) / beta_t) * (h_cur - lr * grad)
            h_new = v_new
            v = v_new
        else:
            h_new = h_cur - lr * grad
        
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < cur_score:
            h_cur = h_new

    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):

        over = corr[n-1:n-1+n]
        c_curr = float(np.max(over)) * dx

            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(over)[::-1][:n_active]
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temperature)
        weights /= weights.sum()

        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    sc_lo = compute_c5(h_lo_new)
    return h_lo_new, sc_lo, bh, bs


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 55.0 - (time.time() - start_time)

    n0 = len(initial_h_values)
    target0 = n0 / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target0)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Initial SLSQP polish at n0
    if tleft() > 80:
        h1, sc1 = slsqp_opt(best_h, n0, target0, [100, 500, 2000], 2000, 1e-16,
                            deadline=start_time + 70)
        if sc1 < best_score:
            best_score, best_h = sc1, h1

    # Phase 2: Multi-resolution at 2x
    n1 = n0 * 2
    target1 = n1 / 2.0
    if tleft() > 150:
        h2, sc2, h2_hi, bs2 = multi_res_cycle(best_h, n0, target0, n1, target1,
                                               deadline=start_time + 160,
                                               n_grad_steps=400,
                                               beta_init=100, beta_growth=1.12,
                                               lr=0.0004)
        if bs2 < best_score:
            best_score = bs2
            best_h = interp_to(h2_hi, n0, target0)
        if tleft() > 60:
            h3, sc3 = slsqp_opt(h2_hi, n1, target1, [500, 2000], 800, 1e-15,
                                deadline=start_time + 240)
            if sc3 < best_score:
                best_score = sc3
                best_h = interp_to(h3, n0, target0)
            h1_cur = h3
        else:
            h1_cur = h2_hi
        if tleft() > 50:
            h4, sc4 = subgrad_polish(h1_cur, n1, target1, 300, 5, 0.001, 0.998,
                                      deadline=start_time + 310, temp=0.002)
            if sc4 < best_score:
                best_score = sc4
                best_h = interp_to(h4, n0, target0)

    # Phase 3: Multi-resolution at 4x
    n2 = n1 * 2
    target2 = n2 / 2.0
    if tleft() > 150:
        h5, _, h5_hi, bs5 = multi_res_cycle(best_h, n0, target0, n2, target2,
                                             deadline=start_time + 420,
                                             n_grad_steps=250,
                                             beta_init=300, beta_growth=1.1,
                                             lr=0.0002)
        if bs5 < best_score:
            best_score = bs5
            best_h = interp_to(h5_hi, n0, target0)
        if tleft() > 50:
            h6, sc6 = slsqp_opt(h5_hi, n2, target2, [1000, 3000], 400, 1e-14,
                                deadline=start_time + 500)
            if sc6 < best_score:
                best_score = sc6
                best_h = interp_to(h6, n0, target0)

    # Phase 4: Transition back to n0, re-polish
    if len(best_h) != n0:
        best_h = interp_to(best_h, n0, target0)
    sc_chk = compute_c5(best_h)
    if sc_chk < best_score:
        best_score = sc_chk

    if tleft() > 120:
        h7, _, bh7, bs7 = momentum_rw(best_h, n0, target0, 500, 500, 1.15,
                                       0.0003, momentum=0.82,
                                       deadline=start_time + 620)
        if bs7 < best_score:
            best_score, best_h = bs7, bh7

    if tleft() > 80:
        h8, sc8 = slsqp_opt(best_h, n0, target0, [5000, 20000], 2500, 1e-17,
                            deadline=start_time + 720)
        if sc8 < best_score:
            best_score, best_h = sc8, h8

    # Phase 5: Alternating multi-res for escape
    if tleft() > 100:
        for cycle in range(3):
            if tleft() < 40:
                break
            dl = start_time + 60 + 30 * (cycle + 2)
            step_n = n0 + cycle * (n0 // 4)
            step_t = step_n / 2.0
            h_m, sc_m, h_m_hi, bs_m = multi_res_cycle(
                best_h, n0, target0, min(step_n + n0 * (cycle + 1) // 4, n0 * 2),
                min(step_n + n0 * (cycle + 1) // 4, n0 * 2) / 2.0,
                deadline=dl, n_grad_steps=100,
                beta_init=1000 + 500 * cycle, beta_growth=1.2, lr=0.0002)
            if bs_m < best_score:
                best_score = bs_m
                best_h = interp_to(h_m_hi, n0, target0)

    # Phase 6: Lag surgery
    if tleft() > 60:
        h9, sc9 = lag_surgery(best_h, n0, target0, 400, 0.003,
                              deadline=start_time + 850)
        if sc9 < best_score:
            best_score, best_h = sc9, h9

    # Phase 7: Subgradient micro-polish
    if tleft() > 50:
        h10, sc10 = subgrad_polish(best_h, n0, target0, 300, 5, 0.0006, 0.999,
                                    deadline=start_time + 950, temp=0.001)
        if sc10 < best_score:
            best_score, best_h = sc10, h10

    # Phase 8: Random perturbation to escape local minima
    if tleft() > 50:
        for mag in [0.04, 0.02, 0.008, 0.003]:
            if tleft() < 15:
                break
            for _ in range(20):
                if tleft() < 10:
                    break
                h_p = best_h.copy()
                idx = rng.choice(n0, size=rng.randint(3, 8), replace=False)
                h_p[idx] += rng.uniform(-mag, mag, size=len(idx))
                h_p = project_feasible(h_p, target0)
                sc = compute_c5(h_p)
                if sc < best_score:
                    best_score = sc
                    best_h = h_p
    # Phase 9: Final SLSQP with very high alpha
    if tleft() > 40:
        h_f, sc_f = slsqp_opt(best_h, n0, target0, [10000, 50000], 2000, 1e-17,
                              deadline=start_time + budget_s - 40)
        if sc_f < best_score:
            best_score, best_h = sc_f, h_f

    # Phase 10: Final subgradient
    if tleft() > 15:
        h_s, sc_s = subgrad_polish(best_h, n0, target0, 150, 3, 0.0003, 0.999,
                                    deadline=start_time + budget_s - 10, temp=0.001)
        if sc_s < best_score:
            best_score, best_h = sc_s, h_s

    best_h = project_feasible(best_h, target0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
