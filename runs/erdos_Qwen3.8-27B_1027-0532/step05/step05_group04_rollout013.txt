def project_feasible(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection on lambda"""
    h = h.astype(float).copy()
    n = len(h)
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining numerical issues
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = int(mask.sum())
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)

            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new.copy()
            sc_cur = compute_c5(h_cur)
            if sc_new <= sc_cur + 1e-15:
                h_cur = h_new
            else:
                v *= 0.5
                lr = max(lr * 0.9, 1e-8)
            lr = max(lr * 0.9995, 1e-10)
    return best_h, best_score

def slsqp_polish(h, n, target, alpha, maxiter, deadline):
    def objective(x):
        ov = compute_overlap_fft(x)
        c_max = float(np.max(ov))
        shifted = alpha * (ov - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

    def cons_eq(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    try:
        res = minimize(objective, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h.copy()

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, deadline):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if time.time() > deadline:
            break
        ov = compute_overlap_fft(h_cur)
        c_curr = float(np.max(ov)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(ov)[::-1][:n_active]
        diffs = ov[top_idx] - ov[top_idx[0]]
        weights = np.exp(diffs / 0.002)
        weights /= weights.sum()
        w_full = np.zeros(n)
        w_full[top_idx] = weights
        g = grad_weighted_fft(h_cur, w_full) * (-dx)
        h_new = project_feasible(h_cur + lr * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= 0.98
            if lr < 1e-10:
                break
    return best_h, best_score

def de_search(n, target, pop_size, max_generations, rng, deadline, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n)
        h = project_feasible(h, target)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_generations):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n)
            j_rand = rng.randint(n)
            for j in range(n):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_feasible(trial, target)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
    return best, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def structured_init(n, target, rng):
    h = np.ones(n) * 0.5
    n_blocks = rng.randint(3, 8)
    block_len = n // n_blocks
    for i in range(n_blocks):
        start = i * block_len
        end = start + block_len if i < n_blocks - 1 else n
        val = rng.choice([0.0, 0.5, 1.0])
        h[start:end] = val + rng.randn(end - start) * 0.1
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick diverse exploration (0-60s)
    deadline_p1 = start_time + 60
    n_c = 100
    h_de, _ = de_search(n_c, n_c / 2.0, 20, 150, rng, deadline=deadline_p1, F=0.7, CR=0.85)
    h_de_fine = interpolate_to(h_de, n, target)
    sc = compute_c5(h_de_fine)
    if sc < best_score:
        best_score = sc
        best_h = h_de_fine.copy()

    # Try structured initializations
    for trial in range(5):
        if time.time() > deadline_p1:
            break
        h_struct = structured_init(n, target, rng)
        sc = compute_c5(h_struct)
        if sc < best_score:
            best_score = sc
            best_h = h_struct.copy()

    # Phase 2: Main gradient optimization (60-350s)
    deadline_p2 = start_time + 350
    h_grad, sc_grad = gradient_round(best_h, n, target,
                                     n_steps=5000, beta_init=100, beta_growth=1.005,
                                     lr_init=0.003, momentum=0.85, lr_decay=0.9998,
                                     deadline=deadline_p2, max_step=0.03)
    if sc_grad < best_score:
        best_score = sc_grad
        best_h = h_grad.copy()

    # Phase 3: SLSQP refinement (350-450s)
    if tleft() > 100:
        deadline_p3 = start_time + 450
        h_ref = slsqp_polish(best_h, n, target, alpha=500, maxiter=3000, deadline=deadline_p3)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 4: Multi-scale gradient (450-600s)
    if tleft() > 150:
        deadline_p4 = start_time + 600
        h_ms, sc_ms = multi_scale_gradient(best_h, n, target,
                                           n_steps=3000, deadline=deadline_p4)
        if sc_ms < best_score:
            best_score = sc_ms
            best_h = h_ms.copy()

    # Phase 5: Second SLSQP (600-700s)
    if tleft() > 100:
        deadline_p5 = start_time + 700
        h_ref = slsqp_polish(best_h, n, target, alpha=2000, maxiter=3000, deadline=deadline_p5)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 6: Aggressive gradient with high beta (700-850s)
    if tleft() > 150:
        deadline_p6 = start_time + 850
        h_grad2, sc_g2 = gradient_round(best_h, n, target,
                                        n_steps=5000, beta_init=1000, beta_growth=1.003,
                                        lr_init=0.001, momentum=0.9, lr_decay=0.9999,
                                        deadline=deadline_p6, max_step=0.015)
        if sc_g2 < best_score:
            best_score = sc_g2
            best_h = h_grad2.copy()

    # Phase 7: Subgradient polish (850-950s)
    if tleft() > 100:
        deadline_p7 = start_time + 950
        h_sg, sc_sg = subgradient_polish(best_h, n, target,
                                         n_steps=2000, n_active=3, lr_init=0.002,
                                         deadline=deadline_p7)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg.copy()

    # Phase 8: SLSQP with very high alpha (950-1050s)
    if tleft() > 100:
        deadline_p8 = start_time + 1050
        h_ref = slsqp_polish(best_h, n, target, alpha=10000, maxiter=3000, deadline=deadline_p8)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 9: Final fine gradient (1050-1180s)
    if tleft() > 130:
        deadline_p9 = start_time + 1180
        h_grad3, sc_g3 = gradient_round(best_h, n, target,
                                        n_steps=8000, beta_init=5000, beta_growth=1.001,
                                        lr_init=0.0005, momentum=0.92, lr_decay=0.99995,
                                        deadline=deadline_p9, max_step=0.008)
        if sc_g3 < best_score:
            best_score = sc_g3
            best_h = h_grad3.copy()

    # Phase 10: Final SLSQP (1180-1270s)
    if tleft() > 90:
        deadline_p10 = start_time + 1270
        h_ref = slsqp_polish(best_h, n, target, alpha=50000, maxiter=5000, deadline=deadline_p10)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    # Phase 11: Ultra-fine subgradient (1270-1290s)
    if tleft() > 20:
        deadline_p11 = start_time + budget_s - 15
        h_sg2, sc_sg2 = subgradient_polish(best_h, n, target,
                                           n_steps=500, n_active=5, lr_init=0.0005,
                                           deadline=deadline_p11)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = h_sg2.copy()

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
