    """Find optimal h[j] given current h and corr. Returns (new_hj, new_corr, new_max)."""

    delta_max = 1.0 - h[j]
    
    # Precompute g vector: g[idx] for idx in range(2n-1), L = idx - (n-1)

        h2 = np.interp(x_new, x_old, best_h)

        h2, sc2 = coord_descent(h2, target2, n_sweeps=10000, rng=rng,

        if sc2 < best_score:
            best_score = sc2
            best_h = h2.copy()
    
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def solve_1d_convex(corr, g, lo, hi, max_iters=60):
    for _ in range(max_iters):
        if hi - lo < 1e-14:
            break
        mid = 0.5 * (lo + hi)
        vals = corr + mid * g
        mx = float(np.max(vals))
        active = vals >= mx - 1e-12
        ga = g[active]
        if np.all(ga > 1e-12):
            hi = mid
        elif np.all(ga < -1e-12):
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)


def is_optimal_at_zero(corr, g):
    mx = float(np.max(corr))
    active = corr >= mx - 1e-12
    ga = g[active]
    if np.all(ga > 1e-12) or np.all(ga < -1e-12):
        return False
    return True


def coordinate_descent(h, target, n_sweeps, rng, deadline=None):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    cur_max = float(np.max(corr))
    lags = np.arange(-(n-1), n)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        max_before = cur_max
        for j in order:
            dlo = -h[j]
            dhi = 1.0 - h[j]
            if dhi - dlo < 1e-15:
                continue
            jp = j + lags
            jm = j - lags
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            g = np.zeros(2 * n - 1)
            if vp.any():
                g[vp] = 1.0 - h[jp[vp]]
            if vm.any():
                g[vm] -= h[jm[vm]]
            
            if is_optimal_at_zero(corr, g):
                continue
            
            best_d = solve_1d_convex(corr, g, dlo, dhi)
            if abs(best_d) > 1e-14:
                h[j] += best_d
                h[j] = np.clip(h[j], 0.0, 1.0)
                corr = compute_corr(h)
                cur_max = float(np.max(corr))
        
        if cur_max > max_before + 1e-10:
            # Something went wrong, recompute
            corr = compute_corr(h)
            cur_max = float(np.max(corr))

        if max_before - cur_max < 1e-14 and sweep > 20:

                    jm = j - lags

                if not any_improve:

def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = float(np.max(corr))
        shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
        val = cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha
        s = float(np.sum(x))
        val += mu * (s - target) ** 2
        return val
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = float(np.max(corr))
        shifted = np.clip(alpha * (corr - cmax), -700.0, 0.0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws > 1e-300:
            w = w / ws
        else:
            w = np.ones(len(corr)) / len(corr)
        grad_v = np.zeros(n)
        active = np.where(w > 1e-10)[0]
        for ki in active:
            wi = w[ki]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad_v[j1] += wi * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad_v[j2] -= wi * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad_v[j1] += wi * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad_v[j2] -= wi * x[j2 - m]
        s = float(np.sum(x))
        grad_v += 2.0 * mu * (s - target)
        return grad_v
    
    bounds = [(0.0, 1.0)] * n
    h0 = h.copy()
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22, 'gtol': 1e-15, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h0
