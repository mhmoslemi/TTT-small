import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, w_lags, n_lags_active):
    """Compute gradient of weighted sum of overlaps.
    w_lags: array of shape (2n-1,) with softmax weights
    Returns gradient of sum_s w_s * O(s) w.r.t. h
    """
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    grad = np.zeros(n)
    
    # Only use the top n_lags_active lags
    if n_lags_active < len(w_lags):
        top_idx = np.argsort(w_lags)[::-1][:n_lags_active]
        w = np.zeros(len(w_lags))
        for i in top_idx:
            w[i] = w_lags[i]
    else:
        w = w_lags
    
    for s_offset in range(2 * n - 1):
        wval = w[s_offset]
        if wval < 1e-15:
            continue
        s = s_offset - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += wval * g[:n-s]
                grad[:n-s] -= wval * h[s:]
        else:
            t = -s
            if t < n:
                grad[:n-t] += wval * g[t:]
                grad[t:] -= wval * h[:n-t]
    
    grad *= dx
    return grad

def softmax_gradient_descent(h_init, n, target, time_limit,
                              beta_start=10.0, beta_end=5000.0,
                              lr_start=0.05, lr_end=0.005,
                              momentum=0.9, n_active=300):
    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    n_lags = 2 * n - 1
    no_improve = 0
    
    max_iter = 100000
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        progress = min(1.0, it / max_iter)
        # Exponential annealing
        beta = beta_start * (beta_end / beta_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress
        
        # Compute overlaps and softmax weights
        overlaps = compute_overlaps(h)
        max_ov = np.max(overlaps)
        log_w = beta * (overlaps - max_ov)
        log_w = np.clip(log_w, -400.0, 0.0)
        w = np.exp(log_w)
        w_sum = np.sum(w)
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(n_lags)
            w[np.argmax(overlaps)] = 1.0
        
        # Compute gradient
        grad = compute_gradient(h, w, n_active)
        
        # Momentum update
        vel = momentum * vel - lr * grad
        h_new = h + vel
        h_new = project_feasible(h_new, target
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
        else:
            # Try plain gradient step
            h_try = h - lr * grad
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                no_improve = 0
            else:
                no_improve += 1
                vel *= 0.5
                if no_improve > 500:
                    # Restart with higher beta
                    vel = np.zeros(n)
                    no_improve = 0
    
    return best_h, best_c5
