    """Compute the change in correlation at lag k when swapping h[p]:1->0 and h[q]:0->1"""
    n = len(h)

    # p was 1, becomes 0
    # As first factor at position p: -(1-h[p+k]) if valid
    pk = p + k
    if 0 <= pk < n:
        delta -= (1.0 - h[pk])
    # As second factor: position i where i+k=p, i=p-k
    pk2 = p - k
    if 0 <= pk2 < n:
        delta += h[pk2]
    # q was 0, becomes 1
    qk = q + k
    if 0 <= qk < n:
        delta += (1.0 - h[qk])
    qk2 = q - k
    if 0 <= qk2 < n:
        delta -= h[qk2]
    return delta

    global_slsqp_target = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    start = time.time()
    for a in alphas:
        if time.time() - start > time_limit:
            break
        def obj(x):
            return soft_max_obj(x, a, global_slsqp_target)
        def cons_eq(x):
            return np.sum(x) - global_slsqp_target
        def cons_jac(x):
            return np.ones(len(x))
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]

            sc = compute_c5(h_new)
            if sc < cur_score:

                cur_score = sc

        if time.time() - start > time_limit:
            break
        corr = compute_corr(h_cur)
        c_curr = float(corr.max()) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / max(temp, 1e-10))
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            k = int(k)
            if k == 0:
                grad[:] += w * dx * (1.0 - 2.0 * h_cur)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += w * dx * (1.0 - h_cur[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= w * dx * h_cur[idx2 - k]
        velocity = momentum * velocity + (1.0 - momentum) * grad
        h_new = h_cur - lr * velocity
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.005, 0.008)
        else:
            lr *= lr_decay
            if lr < 1e-12:
                break
    return best_h, best_score

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), n_dst / 2.0)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, n_dst / 2.0)

def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def make_structured_binary(n, rng):
    pattern_type = rng.randint(5)
    h = np.zeros(n)
    if pattern_type == 0:
        period = rng.randint(3, max(4, n // 10))
        for i in range(n):
            h[i] = 1.0 if (i // period) % 2 == 0 else 0.0
    elif pattern_type == 1:
        freq = rng.randint(1, max(2, n // 10))
        phase = rng.rand() * 2.0 * np.pi
        h = (np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase) > 0).astype(float)
    elif pattern_type == 2:
        block = rng.randint(3, max(4, n // 8))
        for i in range(n):
            h[i] = 1.0 if (i % block) < block // 2 else 0.0
    elif pattern_type == 3:
        half = n // 2
        h[:half] = 1.0
        swap_count = rng.randint(n // 8, n // 4)
        swap_idx = rng.choice(n, swap_count, replace=False)
        h[swap_idx] = 1.0 - h[swap_idx]
    else:
        wave1 = np.sin(2.0 * np.pi * rng.uniform(1, 5) * np.arange(n) / n)
        wave2 = np.sin(2.0 * np.pi * rng.uniform(5, 15) * np.arange(n) / n)
        h = ((wave1 + 0.5 * wave2) > 0).astype(float)
    # Fix to have exactly n//2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n // 2:
        excess = len(ones) - n // 2
        to_zero = rng.choice(ones, excess, replace=False)
        h[to_zero] = 0.0
    elif len(ones) < n // 2:
        deficit = n // 2 - len(ones)
        to_one = rng.choice(zeros, deficit, replace=False)
        h[to_one] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    n_init = len(initial_h_values)
    target_init = n_init / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target_init)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    best_n = n_init

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: Quick SLSQP from parent
    if tleft() > 400:
        h_ref, sc = slsqp_refine(best_h, [100, 500, 2000], 2000, 1e-15, time_limit=120)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 2: Binary local search from parent
    if tleft() > 300:
        h_bin = to_binary(best_h)
        h_ls, sc_ls = binary_local_search(h_bin, max_time=60.0, use_annealing=True, T_init=0.8)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls
            best_n = n_init

    # Phase 3: Multi-start binary search with structured initializations
    if tleft() > 250:
        candidates = []
        for attempt in range(15):
            if tleft() < 100:
                break
            if attempt < 8:
                h_cand = make_structured_binary(n_init, rng)
            else:
                h_cand = make_random_binary(n_init, rng)
            h_ls, sc_ls = binary_local_search(h_cand, max_time=15.0, use_annealing=True, T_init=0.5)
            candidates.append((sc_ls, h_ls))
        candidates.sort(key=lambda x: x[0])
        for sc_c, h_c in candidates[:3]:
            if sc_c < best_score:
                best_score = sc_c
                best_h = h_c.copy()
                best_n = n_init

    # Phase 4: Continuous refinement of best binary
    if tleft() > 200:
        h_cont = best_h.copy()
        h_ref, sc = slsqp_refine(h_cont, [500, 2000, 10000], 1500, 1e-15, time_limit=80)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            best_n = n_init
        h_sg, sc_sg = subgradient_step(best_h, 400, 5, 0.003, 0.997, 0.4, 0.005, time_limit=40)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
            best_n = n_init
        h_ref, sc = slsqp_refine(best_h, [5000, 20000], 1000, 1e-16, time_limit=60)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
            best_n = n_init

    # Phase 5: Try different n values
    if tleft() > 200:
        n_list = []
        for dn in [-100, -50, 50, 100, 200, 300, 500]:
            nc = n_init + dn
            if nc >= 100 and nc <= 2000:
                n_list.append(nc)
        seen = set()
        for n_try in n_list:
            if tleft() < 80:
                break
            if n_try in seen:
                continue
            seen.add(n_try)
            h_try = interpolate_to(best_h, n_try)
            h_bin_try = to_binary(h_try)
            h_ls, sc_ls = binary_local_search(h_bin_try, max_time=20.0, use_annealing=True, T_init=0.6)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls.copy()
                best_n = n_try
        # Also try structured starts at different n
        for n_try in [n_init // 2, n_init * 2]:
            if tleft() < 60:
                break
            if n_try < 50 or n_try > 1500:
                continue
            h_try = make_structured_binary(n_try, rng)
            h_ls, sc_ls = binary_local_search(h_try, max_time=15.0, use_annealing=True, T_init=0.5)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls.copy()
                best_n = n_try

    # Phase 6: Refine best solution at its optimal n
    if tleft() > 100:
        h_cur = best_h.copy()
        n_cur = len(h_cur)
        h_ref, sc = slsqp_refine(h_cur, [1000, 5000, 20000], 1500, 1e-16, time_limit=50)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
        h_sg, sc_sg = subgradient_step(best_h, 500, 6, 0.002, 0.998, 0.5, 0.003, time_limit=30)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref, sc = slsqp_refine(best_h, [10000, 50000], 1000, 1e-17, time_limit=40)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Binary refinement of the best continuous solution
    if tleft() > 60:
        h_bin_final = to_binary(best_h)
        h_ls, sc_ls = binary_local_search(h_bin_final, max_time=30.0, use_annealing=True, T_init=0.3)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls
        h_ref, sc = slsqp_refine(best_h, [5000, 20000, 100000], 1000, 1e-17, time_limit=30)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Final micro-optimization
    if tleft() > 30:
        h_sg, sc_sg = subgradient_step(best_h, 300, 4, 0.001, 0.999, 0.6, 0.002, time_limit=20)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_ref, sc = slsqp_refine(best_h, [50000, 200000], 500, 1e-17, time_limit=20)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final projection and evaluation
    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
