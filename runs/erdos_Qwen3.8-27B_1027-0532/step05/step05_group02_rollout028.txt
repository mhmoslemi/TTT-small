from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:

        # But we need to be careful: delta_single uses the OLD h
        # For the combined delta, we need to account for the fact that
        # after swapping i1→j1, the value at i1 changes, which affects delta_single(i2, j2)
        # However, if all four positions are distinct, the deltas are independent!
        
        # Actually, let me verify:
        # delta_single(i1, j1)[k] = h[i1+k] + h[i1-k] - h[j1+k] - h[j1-k]
        # This uses h at positions i1±k and j1±k

        # This uses h at positions i2±k and j2±k

        # i1±k, i2±k, j1±k, j2±k don't overlap with the swap positions,

        # The issue is: when we compute the TOTAL change from (h with i1=1, i2=1, j1=0, j2=0)
        # to (h with i1=0, i2=0, j1=1, j2=1), is it just the sum of two single-swap deltas?

        # Let's think about it. The correlation is:
        # corr[k] = sum_p h[p] * (1-h)[p+k]
        #
        # After both swaps:
        # corr_new[k] = sum_p h'[p] * (1-h')[p+k]
        #
        # The change is:
        # corr_new[k] - corr[k] = sum_p (h'[p] - h[p]) * (1-h')[p+k] + sum_p h[p] * ((1-h')[p+k] - (1-h)[p+k])
        # 
        # This is NOT simply the sum of two single-swap deltas because of the cross terms.
        #
        # However, for BESSEL-type objectives (like sum of squares), the cross terms might cancel.
        # For our max objective, we just need the exact delta.
        #
        # Let me derive the exact delta for a pair swap (i1,i2 -> j1,j2):
        #
        # h'[i1] = 0 (was 1), h'[i2] = 0 (was 1), h'[j1] = 1 (was 0), h'[j2] = 1 (was 0)
        # (1-h')[i1] = 1, (1-h')[i2] = 1, (1-h')[j1] = 0, (1-h')[j2] = 0
        #

        # delta_pair[k] = sum_p (h'[p] * (1-h')[p+k] - h[p] * (1-h)[p+k])

        # Only p values where h' differs from h contribute, and only p+k values where (1-h') differs.

Wait, there's an issue with the approximation. When we compute delta1 = compute_delta(h, i1, j1) and delta2 = compute_delta(h, i2, j2), both use the OLD h. The sum delta1 + delta2 gives the correct delta ONLY if the swaps don't interact.

Let me think about when they DO interact:
- delta1[k] uses h at positions i1±k and j1±k
- delta2[k] uses h at positions i2±k and j2±k

The interaction occurs when:
- i1+k = j2 (then delta1 uses h[j2]=0, but after swap 2, h[j2]=1)
- i1+k = i2 (then delta1 uses h[i2]=1, but after swap 2, h[i2]=0)
- etc.

For a given k, the probability of such an overlap is about 4/n (since we have 4 "special" positions out of n). For the max over all k, the probability that AT LEAST one k has an overlap is about 4*(2n)/n = 8, which is > 1. So overlaps ARE common!

This means the simple sum approximation can be wrong. Let me think about a better way.

Option 1: Compute the delta exactly.
For a pair swap (i1, i2 → j1, j2), the exact delta at lag k is:

delta[k] = h'[i1]*(1-h')[i1+k] + h'[i2]*(1-h')[i2+k] + h'[j1]*(1-h')[j1+k] + h'[j2]*(1-h')[j2+k]
         - h[i1]*(1-h)[i1+k] - h[i2]*(1-h)[i2+k] - h[j1]*(1-h)[j1+k] - h[j2]*(1-h)[j2+k]
         + (corrections for p where p not in special but p+k is special)

This is getting too complex. Let me use a different approach:

**Option 2: Compute the delta by brute force for the 4 affected positions.**

The correlation at lag k is: corr[k] = ∑_p h[p] * (1-h)[p+k]

After the pair swap, the new correlation is:
corr_new[k] = corr[k] + (contributions from the 4 changed positions as "first factor") + (contributions from the 4 changed positions as "second factor")

As "first factor" (p is one of the 4 special positions):
- p=i1: was h[i1]*(1-h)[i1+k] = 1*(1-h)[i1+k], now 0*(...) = 0. Change: -(1-h)[i1+k]
- p=i2: similar. Change: -(1-h)[i2+k]
- p=j1: was 0, now 1*(1-h')[j1+k]. Change: (1-h')[j1+k]
- p=j2: similar. Change: (1-h')[j2+k]

As "second factor" (p+k is one of the 4 special positions, and p is NOT one of them):
- p+k=i1: p=i1-k. Change: h[i1-k] * ((1-h')[i1] - (1-h)[i1]) = h[i1-k] * (1-0) = h[i1-k]
  (since (1-h)[i1] = 0 and (1-h')[i1] = 1)
- p+k=i2: similar. Change: h[i2-k]
- p+k=j1: p=j1-k. Change: h[j1-k] * ((1-h')[j1] - (1-h)[j1]) = h[j1-k] * (0-1) = -h[j1-k]
- p+k=j2: similar. Change: -h[j2-k]

Now, the tricky part: (1-h')[j1+k] in the first factor.
If j1+k is one of {i1, i2, j1, j2}, then (1-h')[j1+k] is modified.
- If j1+k = i1: (1-h')[i1] = 1
- If j1+k = i2: (1-h')[i2] = 1
- If j1+k = j1: (1-h')[j1] = 0 (i.e., k=0)
- If j1+k = j2: (1-h')[j2] = 0
- Otherwise: (1-h')[j1+k] = (1-h)[j1+k]

Similarly for (1-h')[j2+k].

And for the "second factor" corrections, we need to exclude the case where p is also special.
- p=i1-k: if i1-k ∈ {i1, i2, j1, j2}, we've already counted it in the "first factor" part.
  Specifically, if i1-k = i1 (k=0), or i1-k = i2, etc.

The exact delta at lag k is:

Where Δh[i1] = -1, Δh[i2] = -1, Δh[j1] = +1, Δh[j2] = +1

(with boundary checks, and excluding p that is also in {i1,i2,j1,j2})

Now, (1-h')[j1+k] = 1 - h'[j1+k].

  If i1+k = j1 or j2: Δh = +1, so diff = -1. (1-h')[i1+k] = (1-h)[i1+k] - 1

def compute_delta_pair(h, i1, i2, j1, j2, k_arr, n):
    """Exact delta for pair swap (i1,i2 -> j1,j2) in the full correlation."""
    result = np.zeros(2*n-1)
    
    # Simple sum (ignoring interactions)
    for i, j in [(i1, j1), (i2, j2)]:
        idx = i + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += h[idx[mask]]
        idx = i - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] += h[idx[mask]]
        idx = j - k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= h[idx[mask]]
        idx = j + k_arr
        mask = (idx >= 0) & (idx < n)
        result[mask] -= h[idx[mask]]
    
    # Now add interaction corrections
    # The issue: in compute_delta(h, i1, j1), when computing h[i1+k], if i1+k = j2,
    # we use h[j2] = 0 (old value). But after the swap, h[j2] = 1.
    # Similarly, if i1+k = i2, we use h[i2] = 1 (old value), but after swap h[i2] = 0.
    
    # The interaction terms are:
    # In the "first factor" part:
    # For p=i1: we computed -(1-h)[i1+k] = -1 + h[i1+k]
    #   But if i1+k = j2, the correct (1-h')[j2] = 0, not (1-h)[j2] = 1.
    #   So correction: +1 (change from -1 to 0)
    #   Wait, let me redo.
    #   In the simple sum, the contribution of p=i1 to delta[k] is:
    #   OLD: h[i1]*(1-h)[i1+k] = 1*(1-h)[i1+k]
    #   NEW: h'[i1]*(1-h')[i1+k] = 0
    #   Delta from p=i1: 0 - (1-h)[i1+k] = -(1-h)[i1+k] = -1 + h[i1+k]
    #   
    #   But wait, in the simple sum formula, we have h[i1+k] + h[i1-k] - ... for the "first factor" part from i1.
    #   Actually no, the simple formula is:
    #   delta_single(i,j)[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k]
    #   This represents: -(1-h)[i+k] + h[i-k] + (1-h)[j+k] - h[j-k]
    #                    = -1 + h[i+k] + h[i-k] + 1 - h[j+k] - h[j-k]
    #                    = h[i+k] + h[i-k] - h[j+k] - h[j-k]  ✓
    #
    # So the simple sum for the pair is:
    # h[i1+k] + h[i1-k] - h[j1+k] - h[j1-k] + h[i2+k] + h[i2-k] - h[j2+k] - h[j2-k]
    #
    # The EXACT delta is:
    # -(1-h')[i1+k] + h'[i1-k] + (1-h')[j1+k] - h'[j1-k]  (from swap 1)
    # -(1-h')[i2+k] + h'[i2-k] + (1-h')[j2+k] - h'[j2-k]  (from swap 2)
    #
    # But we need to be careful: the "from swap 1" and "from swap 2" should use the SAME
    # "after" state h'. And we should not double-count if positions overlap.
    #
    # Actually, the cleanest way: the exact delta is:
    # delta[k] = sum_{p special} [h'[p]*(1-h')[p+k] - h[p]*(1-h)[p+k]]
    #            + sum_{p+k special, p not special} h[p]*[(1-h')[p+k] - (1-h)[p+k]]
    #
    # First sum (p ∈ {i1,i2,j1,j2}):

    # p=i2: -(1-h)[i2+k]

    # p=j2: (1-h')[j2+k]

    #   t=i2: +1
    #   t=j1: -1
    #   t=j2: -1

    # = h[i1-k] + h[i2-k] - h[j1-k] - h[j2-k]  (with boundary checks)
    # BUT ONLY for p not in {i1,i2,j1,j2}

    # -(1-h)[i1+k] - (1-h)[i2+k] + (1-h')[j1+k] + (1-h')[j2+k]
    # + h[i1-k] + h[i2-k] - h[j1-k] - h[j2-k]  (with p not special)
