from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent_step(h, i, overlaps, dx, lam):
    """Find optimal h[i] given all other h fixed, using Lagrange multiplier for sum constraint."""
    n = len(h)
    k = np.arange(-(n-1), n)  # 2n-1 values
    
    # Compute c_k = dx * (a_k - b_k)
    idx_a = i + k
    valid_a = (idx_a >= 0) & (idx_a < n)
    a = np.zeros(2*n - 1)
    a[valid_a] = 1.0 - h[idx_a[valid_a]]
    
    idx_b = i - k
    valid_b = (idx_b >= 0) & (idx_b < n)
    b = np.zeros(2*n - 1)
    b[valid_b] = h[idx_b[valid_b]]
    
    c = dx * (a - b)
    
    # f(x) = max_k (overlaps[k] - c[k]*h[i] + (c[k] + lam)*x)
    #      = max_k ((overlaps[k] - c[k]*h[i]) + (c[k]+lam)*x)
    d = overlaps - c * h[i]
    slopes = c + lam
    
    # Minimize f(x) = max_k (d[k] + slopes[k]*x) for x in [0,1]
    # Use grid + refinement
    x_grid = np.linspace(0.0, 1.0, 50)
    f_vals = np.max(d[None, :] + slopes[None, :] * x_grid[:, None], axis=1)
    best_idx = np.argmin(f_vals)
    
    # Refine
    if best_idx == 0:
        return 0.0
    if best_idx == len(x_grid) - 1:
        return 1.0
    
    lo = x_grid[best_idx - 1]
    hi = x_grid[best_idx + 1]
    for _ in range(20):
        mid = (lo + hi) / 2.0
        # Check subgradient at mid
        f_mid = np.max(d + slopes * mid)
        active = np.abs(d + slopes * mid - f_mid) < 1e-12
        if np.any(slopes[active] < -1e-14):
            lo = mid
        elif np.any(slopes[active] > 1e-14):
            hi = mid
        else:
            break
    
    return (lo + hi) / 2.0

def coordinate_descent(h_init, n, target, max_sweeps, time_limit, rng):
    """Coordinate descent with Lagrange multiplier for sum constraint."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lam = 0.0
    
    # Adaptive lam update
    lam_step = 2.0 / (target + 1)
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        # Update order (random for diversity)
        order = rng.permutation(n)
        
        for i in order:
            x_new = coord_descent_step(h, i, overlaps, dx, lam)
            delta = x_new - h[i]
            if abs(delta) > 1e-15:
                # Update overlaps incrementally
                k = np.arange(-(n-1), n)
                idx_a = i + k
                valid_a = (idx_a >= 0) & (idx_a < n)
                a = np.zeros(2*n - 1)
                a[valid_a] = 1.0 - h[idx_a[valid_a]]
                
                idx_b = i - k
                valid_b = (idx_b >= 0) & (idx_b < n)
                b = np.zeros(2*n - 1)
                b[valid_b] = h[idx_b[valid_b]]
                
                c = dx * (a - b)
                overlaps += c * delta
                h[i] = x_new
        
        # Update lambda
        sum_h = np.sum(h)
        lam += lam_step * (sum_h - target) / n
        lam = np.clip(lam, -10.0, 10.0)
        
        # Check improvement
        c5_val = float(np.max(overlaps))
        if c5_val < best_c5 - 1e-12:

            best_h = h.copy()

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def coord_find_opt(x_current, slopes, intercepts, lam, n_grid=50):
    """Find x in [0,1] minimizing max_p(intercept_p + slope_p * x) + lam * x"""
    x_grid = np.linspace(0.0, 1.0, n_grid)
    # f(x) = max_p(intercept_p + slope_p * x) + lam * x
    vals = np.max(intercepts[None, :] + slopes[None, :] * x_grid[:, None], axis=1) + lam * x_grid
    best_idx = np.argmin(vals)
    if best_idx == 0:
        return 0.0
    if best_idx == n_grid - 1:
        return 1.0
    lo = x_grid[best_idx - 1]
    hi = x_grid[best_idx + 1]
    for _ in range(25):
        mid = (lo + hi) / 2.0
        f_mid = np.max(intercepts + slopes * mid) + lam * mid
        active = np.abs(intercepts + slopes * mid - np.max(intercepts + slopes * mid)) < 1e-14
        # Subgradient of max part: slopes[active]
        # Subgradient of f: slopes[active] + lam
        min_sg = np.min(slopes[active]) + lam
        max_sg = np.max(slopes[active]) + lam
        if min_sg > 1e-14:
            # All subgradients positive → f is increasing → go left
            hi = mid
        elif max_sg < -1e-14:
            # All subgradients negative → f is decreasing → go right
            lo = mid
        else:
            break
    return (lo + hi) / 2.0

def coordinate_descent(h_init, n, target, max_sweeps, time_limit, rng):

    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_c5 = float(np.max(overlaps))
    lam = 0.0
    p = np.arange(2 * n - 1)
    
    lam_lr = 1.0
    stuck = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if stuck > 50:
            break
        
        order = rng.permutation(n)
        for i in order:
            # Compute slopes and intercepts
            idx_a = i + p - (n - 1)
            idx_b = i - p + (n - 1)
            valid_a = (idx_a >= 0) & (idx_a < n)
            valid_b = (idx_b >= 0) & (idx_b < n)
            a = np.zeros(2 * n - 1)
            a[valid_a] = 1.0 - h[idx_a[valid_a]]
            b = np.zeros(2 * n - 1)
            b[valid_b] = h[idx_b[valid_b]]
            slopes = dx * (a - b)
            intercepts = overlaps - slopes * h[i]
            
            x_new = coord_find_opt(h[i], slopes, intercepts, lam)
            delta = x_new - h[i]
            if abs(delta) > 1e-15:
                overlaps += slopes * delta
                h[i] = x_new
        
        # Update lambda
        sum_h = np.sum(h)
        lam += lam_lr * (sum_h - target) / n
        lam = np.clip(lam, -5.0, 5.0)
        
        c5_val = float(np.max(overlaps))
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            # Decay lam_lr
            lam_lr *= 0.95
    
    h = project_feasible(h, target)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def coord_find_opt(slopes, intercepts, lam, n_grid=60):
    x_grid = np.linspace(0.0, 1.0, n_grid)
    f_vals = np.max(intercepts[None, :] + slopes[None, :] * x_grid[:, None], axis=1) + lam * x_grid
    best_idx = int(np.argmin(f_vals))
    if best_idx == 0:
        return 0.0
    if best_idx == n_grid - 1:
        return 1.0
    lo = x_grid[best_idx - 1]
    hi = x_grid[best_idx + 1]
    for _ in range(25):
        mid = (lo + hi) / 2.0
        f_at_mid = intercepts + slopes * mid
        f_max = np.max(f_at_mid)
        active = np.abs(f_at_mid - f_max) < 1e-14
        min_sg = np.min(slopes[active]) + lam
        max_sg = np.max(slopes[active]) + lam
        if min_sg > 1e-14:
            hi = mid
        elif max_sg < -1e-14:
            lo = mid
        else:
            break
    return (lo + hi) / 2.0


def coordinate_descent(h_init, n, target, max_sweeps, time_limit, rng):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    overlaps = compute_overlaps(h)
    best_h = h.copy()
    best_c5 = float(np.max(overlaps))
    lam = 0.0
    lam_lr = 0.5
    p = np.arange(2 * n - 1)
    stuck = 0
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 80:
            break

        order = rng.permutation(n)
        for i in order:
            idx_a = i + p - (n - 1)
            idx_b = i - p + (n - 1)
            valid_a = (idx_a >= 0) & (idx_a < n)
            valid_b = (idx_b >= 0) & (idx_b < n)
            a = np.zeros(2 * n - 1)
            a[valid_a] = 1.0 - h[idx_a[valid_a]]
            b = np.zeros(2 * n - 1)
            b[valid_b] = h[idx_b[valid_b]]
            slopes = dx * (a - b)
            intercepts = overlaps - slopes * h[i]

            x_new = coord_find_opt(slopes, intercepts, lam)
            delta = x_new - h[i]
            if abs(delta) > 1e-15:
                overlaps += slopes * delta
                h[i] = x_new

        sum_h = np.sum(h)
        lam += lam_lr * (sum_h - target) / n
        lam = float(np.clip(lam, -5.0, 5.0))

        c5_val = float(np.max(overlaps))
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
            no_improve = 0
        else:
            stuck += 1
            no_improve += 1
            if stuck > 30:
                lam_lr *= 0.9

    h = project_feasible(h, target)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def build_lp(h, g, dx, lags, n, target):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for k in lags:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
    return h_var, t_var, constraints


def get_top_lags(h, K):
    overlaps = compute_overlaps(h)
    n = len(h)
    n_half = n - 1
    top_idx = np.argsort(overlaps)[::-1][:K]
    return top_idx - n_half


def lp_refine(h_init, n, target, max_iter, time_limit, n_top=150,
              damping_start=0.95, damping_end=0.80, reg_start=1e-4, reg_end=1e-9):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 100:
            break

        g = 1.0 - h
        lags = get_top_lags(h, n_top)
        progress = min(1.0, it / max(1, max_iter * 0.3))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress

        h_var, t_var, constraints = build_lp(h, g, dx, lags, n, target)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

    return best_h, best_c5


def soft_max_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lambda x: soft_max_obj(x, alpha, n, target), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def gen_binary_periodic(n):
    h = np.zeros(n)
    h[0::4] = 1.0
    h[1::4] = 1.0
    return project_feasible(h, n / 2.0)


def gen_cosine_pattern(n, freq=1):
    x = np.linspace(0.0, 2.0, n)
    h = 0.5 + 0.35 * np.cos(np.pi * freq * x)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n / 2.0)


def gen_random_binary(n, rng):
    h = (rng.random(n) < 0.5).astype(float)
    return project_feasible(h, n / 2.0)
