```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk < 1e-13:
            continue
        g[:n - k] += wk * (1.0 - h[k:])
        g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk < 1e-13:
            continue
        g[ell:] += wk * (1.0 - h[:n - ell])
        g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g
```

```python
def active_set_descent(h, n, target, M, max_iter, deadline):
    """
    Active set gradient descent with momentum.
    Focus on the top M lags, use Nesterov momentum.
    """
    h_cur = h.copy()
    h_mom = h_cur.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = 0.01
    momentum = 0.9
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * dx
        
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        # Get top M lags
        top_idx = np.argsort(corr)[::-1][:M]
        c_top = corr[top_idx]
        
        # Compute gradient as weighted average of top M lag gradients
        # Weights proportional to how close each lag is to the max
        temp = 0.01
        w = np.exp((c_top - c_top[0]) / temp)
        w /= w.sum()
        
        g = np.zeros(n)
        for wk, m in zip(w, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n-k] += wk * (1.0 - h_cur[k:])
                g[k:] -= wk * h_cur[:n-k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - h_cur[:n-ell])
                g[:n-ell] -= wk * h_cur[ell:]
        g *= dx
        
        # Nesterov momentum update
        h_prev = h_mom.copy()
        h_mom = h_mom - lr * (g + momentum * (g - (h_cur - h_prev)))
        # Actually, standard Nesterov: 
        # h_new = h_mom - lr * grad(h_mom)
        # h_mom = h_new + momentum * (h_new - h_cur)
        
        # Simpler: heavy ball method
        vel = h_mom - h_cur
        h_new = h_cur - lr * g + momentum * vel
        h_new = project_feasible(h_new, target)
        h_mom = h_new
        
        sc_new = compute_c5(h_new)
        if sc_new > c_curr * 1.01:
            lr *= 0.5
            h_cur = h_cur  # don't update
        else:
            h_cur = h_new
    
    return best_h, best_score
```

But the h-optimization is still non-convex (bilinear). However, we can use it as a guide: at the optimum, the λ that minimizes the worst case should spread weight across multiple lags.

This suggests: if the top lag is much worse than the second, we should focus on it. If they're close, we should balance them.

Let me implement this as a "smart" active set update:
- If top lag >> second lag: focus on top lag
- If top lags are close: use equal weights (to balance them)

This is essentially what the soft-max does with the temperature parameter! High alpha → focus on worst lag. Low alpha → balance.

So the previous code's alpha continuation is already doing this. The question is whether the optimization at each alpha is good enough.

**I think the key improvement will be:**
1. Higher resolution (n=800)
2. Better convergence at each alpha (more iterations, tighter tolerances)
3. A final "direct" method that doesn't use soft-max (just the worst lag or top-2 lags)

Let me write the code now. I'll aim for clean, efficient code that makes good use of the time budget.

Let me also consider: **Should I try to enforce symmetry h(x) = h(2-x)?**

If the optimal solution is symmetric, then enforcing it reduces the dimensionality by half and might help convergence. But if the optimal solution is NOT symmetric, enforcing it would hurt.

I won't enforce symmetry, but I'll check if the solution is close to symmetric at the end.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_value(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def lse_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-12:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
    for ell in range(1, n):
        wk = w[n - 1 - ell]
        if wk > 1e-12:
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def topk_grad(h, top_m, temperature):
    n = len(h)
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_m]
    c_top = corr[top_idx]
    w = np.exp((c_top - c_top[0]) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(top_m)
        w[0] = 1.0
    else:
        w /= w_sum
    dx = 2.0 / n
    g = np.zeros(n)
    for wk, m in zip(w, top_idx):
        m = int(m)
        k = m - (n - 1)
        if k >= 0:
            g[:n - k] += wk * (1.0 - h[k:])
            g[k:] -= wk * h[:n - k]
        else:
            ell = -k
            g[ell:] += wk * (1.0 - h[:n - ell])
            g[:n - ell] -= wk * h[ell:]
    g *= dx
    return g

def lbfgsb_optimize(h, n, target, alpha, mu, maxiter=300):
    def obj(x):
        return lse_value(x, alpha) + mu * (np.sum(x) - target) ** 2
    def objg(x):
        return lse_grad(x, alpha) + 2.0 * mu * (np.sum(x) - target)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'factr': 1e8})
        return project_feasible(res.x, target)
    except Exception:
        return h

def slsqp_optimize(h, n, target, alpha, maxiter=2000, ftol=1e-15):
    def cons_fun(x):
        return np.sum(x) - target
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(lse_value, h, method='SLSQP', 
                       jac=lambda x: lse_grad(x, alpha),
                       args=(alpha,),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def active_momentum_descent(h, n, target, M, n_steps, temp, lr_init, deadline):
    h_cur = h.copy()
    vel = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = lr_init
    momentum = 0.85
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_curr = float(np.max(corr)) * 2.0 / n
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        
        g = topk_grad(h_cur, M, temp)
        vel = momentum * vel - lr * g
        h_new = project_feasible(h_cur + vel, target)
        sc_new = compute_c5(h_new)
        
        if sc_new < c_curr:
            h_cur = h_new
        else:
            lr *= 0.7
            vel *= 0.5
            if lr < 1e-12:
                break
    return best_h, best_score

def worst_lag_descent(h, n, target, max_steps, eps_init, deadline):
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    eps = eps_init
    
    for step in range(max_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        m_star = int(np.argmax(corr))
        k_star = m_star - (n - 1)
        g = np.zeros(n)
        if k_star >= 0:
            g[:n - k_star] += (1.0 - h_cur[k_star:])
            g[k_star:] -= h_cur[:n - k_star]
        else:
            ell = -k_star
            g[ell:] += (1.0 - h_cur[:n - ell])
            g[:n - ell] -= h_cur[ell:]
        g *= dx
        h_new = project_feasible(h_cur - eps * g, target)
        sc_new = compute_c5(h_new)
        if sc_new < best_score - 1e-15:
            h_cur = h_new
            best_score = sc_new
            eps = min(eps * 1.01, 0.02)
        else:
            eps *= 0.8
            if eps < 1e-10:
                break
    return h_cur, best_score

def lag_balance_optimize(h, n, target, M, alpha, maxiter=200, deadline=0):
    """Optimize with a penalty for variance of top M lags."""
    h_cur = h.copy()
    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[::-1][:M]
        c_top = corr[top_idx]
        # Soft max of top M
        am = float(np.max(c_top))
        shifted = np.clip(alpha * (c_top - am), -700.0, 0.0)
        lse = am + float(np.log(np.sum(np.exp(shifted)))) / alpha
        # Variance penalty
        var_pen = 0.1 * np.var(c_top)
        return (lse + var_pen) * dx + 100.0 * (np.sum(x) - target) ** 2
    
    def objg(x):
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[::-1][:M]
        c_top = corr[top_idx]
        am = float(np.max(c_top))
        shifted = np.clip(alpha * (c_top - am), -700.0, 0.0)
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(M)
            w[0] = 1.0
        else:
            w /= w_sum
        g = np.zeros(n)
        for wk, m in zip(w, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - x[k:])
                g[k:] -= wk * x[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - x[:n - ell])
                g[:n - ell] -= wk * x[ell:]
        g *= dx
        # Variance penalty gradient (approximate)
        var_m = np.mean(c_top)
        for wk, m in zip((c_top - var_m) / M, top_idx):
            m = int(m)
            k = m - (n - 1)
            if k >= 0:
                g[:n - k] += wk * (1.0 - x[k:])
                g[k:] -= wk * x[:n - k]
            else:
                ell = -k
                g[ell:] += wk * (1.0 - x[:n - ell])
                g[:n - ell] -= wk * x[ell:]
        g *= dx
        g += 200.0 * (np.sum(x) - target)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_cur, method='L-BFGS-B', jac=objg,
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
