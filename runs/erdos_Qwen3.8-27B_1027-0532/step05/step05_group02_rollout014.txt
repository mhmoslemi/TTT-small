def continuous_waterfill(h, n_iters, eta, rng, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    target = n / 2.0
    dx = 2.0 / n
    full = full_corr_fast(h)
    best_c5 = float(full.max()) * dx
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Find the worst lag (max correlation = min overlap)
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            break
        
        # Compute gradient: d(corr[k_star])/dh[i] = (1-h[i+k_star]) - h[i-k_star]
        # We want to DECREASE corr[k_star], so we move opposite to gradient
        grad = np.zeros(n)
        positions = np.arange(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        grad[valid] += (1.0 - h[idx[valid]])
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        grad[valid] -= h[idx[valid]]
        
        # Update: h -= eta * grad (to decrease corr at k_star)
        # But maintain sum constraint: sum(h) = target
        h_new = h - eta * grad
        # Project to maintain sum and bounds
        h_new = project_feasible(h_new, target)
        
        # Check if we improved
        full_new = full_corr_fast(h_new)
        c5_new = float(full_new.max()) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            full = full_new
        else:
            # Try smaller step
            h_new = h - eta * 0.5 * grad
            h_new = project_feasible(h_new, target)
            full_new = full_corr_fast(h_new)
            c5_new = float(full_new.max()) * dx

                full = full_new
            else:
                eta *= 0.95  # Reduce learning rate
