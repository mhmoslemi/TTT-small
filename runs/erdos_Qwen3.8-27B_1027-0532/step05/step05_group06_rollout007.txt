import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def corr_and_grad(h, active_lags):
    """Compute correlation values and gradient w.r.t. h for given active lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    grad = np.zeros(n)
    for ki in active_lags:
        lag = int(ki) - (n - 1)
        if lag >= 0:
            # corr[k] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j]: 
            #   h[j] as first factor: (1-h[j+lag]) for j in [0, n-1-lag]
            #   h[j] as second factor: -h[j-lag] for j in [lag, n-1]
            j1 = np.arange(0, n - lag)
            grad[j1] += (1.0 - h[j1 + lag])
            if lag > 0:
                j2 = np.arange(lag, n)
                grad[j2] -= h[j2 - lag]
        else:
            m = -lag
            # corr[n-1-m] = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
            j1 = np.arange(m, n)
            grad[j1] += (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            grad[j2] -= h[j2 + m]
    return corr, grad

def nesterov_subgrad(h, target, n_iter, deadline, topk=3, lam=0.5):
    """Nesterov-accelerated projected subgradient method on max(corr)."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)  # velocity
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    # Adaptive step size
    step = 0.02
    patience = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cur_max = float(np.max(corr))
        
        # Find active lags (within some tolerance of max)
        tol = 0.05 * cur_max + 1e-6
        active = np.where(corr >= cur_max - tol)[0]
        # Limit to topk most important
        if len(active) > topk:
            top_idx = np.argsort(corr[active])[::-1][:topk]
            active = active[top_idx]
        
        _, grad = corr_and_grad(h_cur, active)
        grad = grad / len(active)
        
        # Nesterov step
        h_trial = h_cur - step * (grad + lam * v)
        h_trial = project_feasible(h_trial, target)
        
        new_corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        new_max = float(np.max(new_corr))
        
        if new_max < cur_max:
            v = lam * v + (h_trial - h_cur)
            h_cur = h_trial
            cur_max = new_max
            if cur_max * dx < best_score:
                best_score = cur_max * dx
                best_h = h_cur.copy()
            patience = 0
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.9
            patience += 1
            if patience > 50:
                break
    
    return best_h, best_score

def lbfgs_smooth(h, target, alpha, mu, maxiter, deadline):
    """L-BFGS-B on smoothed objective."""
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n
    
    def f(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
        s = float(np.sum(x))
        val += mu * (s - target) ** 2
        return val
    
    def g(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum > 1e-300:
            weights = weights / wsum
        else:
            weights = np.ones(len(corr)) / len(corr)
        grad = np.zeros(n)
        active = np.where(weights > 1e-10)[0]
        for ki in active:
            w = weights[ki]
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - x[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * x[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - x[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * x[j2 - m]
        grad += 2.0 * mu * (np.sum(x) - target)
        return grad
    
    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur

def binary_sa(h_init, target, n_iter, deadline, T0=0.1, T1=0.001):
    """Simulated annealing on binary solutions."""
    n = len(h_init)
    n_ones = int(target)
    
    # Start from nearest binary
    h_b = (h_init > 0.5).astype(float)
    # Fix number of ones
    while int(h_b.sum()) > n_ones:
        idx = np.where(h_b == 1)[0]
        h_b[idx[np.random.randint(len(idx))]] = 0
    while int(h_b.sum()) < n_ones:
        idx = np.where(h_b == 0)[0]
        h_b[idx[np.random.randint(len(idx))]] = 1
    
    best_h = h_b.copy()
    best_score = compute_c5(h_b)
    cur_score = best_score
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        
        # Move: flip two random positions (one 1->0, one 0->1) to preserve sum
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        h_b[i] = 0
        h_b[j] = 1
        new_score = compute_c5(h_b)
        
        if new_score < cur_score or np.random.random() < np.exp(-(new_score - cur_score) / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_b.copy()
        else:
            h_b[i] = 1
            h_b[j] = 0
    
    return best_h, best_score

def binary_to_continuous(h_bin, target, n_steps=100, deadline=None):

def top_lag_gradient_descent(h, target, topk=5, n_steps=200, deadline=None):
    """Direct gradient descent on the top-k lags."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.01
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:topk]
        avg_corr = corr[top_idx].mean()
        std_corr = corr[top_idx].std()
        if std_corr < 1e-10:
            # All top lags are equal, use all of them
            weights = np.ones(topk) / topk
        else:
            weights = (corr[top_idx] - avg_corr) / (2 * std_corr)
            weights = np.clip(weights, 0.01, None)
            weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for ki, w in zip(top_idx, weights):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        if sc < compute_c5(h_cur):
            h_cur = h_new
            lr = min(lr * 1.03, 0.05)
        else:

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_and_gradient(h, active_lags):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    grad = np.zeros(n)
    for ki in active_lags:
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(0, n - lag)
            grad[j1] += (1.0 - h[j1 + lag])
            if lag > 0:
                j2 = np.arange(lag, n)
                grad[j2] -= h[j2 - lag]
        else:
            m = -lag
            j1 = np.arange(m, n)
            grad[j1] += (1.0 - h[j1 - m])
            j2 = np.arange(0, n - m)
            grad[j2] -= h[j2 + m]
    return corr, grad


def lse_objective_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_smooth(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    h_cur = h.copy()
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return lse_objective_and_grad(x, alpha, target, mu)[0]

    def g(x):
        return lse_objective_and_grad(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h_cur, jac=g, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20,
                                'gtol': 1e-14, 'maxls': 40})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def nesterov_subgrad(h, target, n_iter, deadline, topk=5, lam=0.7):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step = 0.015
    patience = 0
    prev_corr_max = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full')))

    for it in range(n_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        cur_max = float(np.max(corr))
        tol = 0.08 * cur_max + 1e-6
        active = np.where(corr >= cur_max - tol)[0]
        if len(active) > topk:
            top_idx = np.argsort(corr[active])[::-1][:topk]
            active = active[top_idx]
        _, grad = corr_and_gradient(h_cur, active)
        grad = grad / len(active)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        h_trial = h_cur - step * (grad / gnorm + lam * v / (gnorm + 1e-10))
        h_trial = project_feasible(h_trial, target)
        new_corr = np.correlate(h_trial, 1.0 - h_trial, mode='full')
        new_max = float(np.max(new_corr))
        if new_max < cur_max:
            v = lam * v + (h_trial - h_cur)
            v /= (np.linalg.norm(v) + 1e-10)
            h_cur = h_trial
            cur_max = new_max
            if cur_max * dx < best_score:
                best_score = cur_max * dx
                best_h = h_cur.copy()
            patience = 0
            step = min(step * 1.04, 0.08)
        else:
            step *= 0.92
            patience += 1
            if patience > 80:
                break
    return best_h, best_score


def top_lag_gd(h, target, topk=5, n_steps=300, deadline=None):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lr = 0.008
    stall = 0

    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:topk]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-10:
            weights = np.ones(topk) / topk
        else:
            weights = (top_vals - top_mean) / (2.0 * top_std)
            weights = np.clip(weights, 0.02, None)
            weights = weights / weights.sum()
        grad = np.zeros(n)
        for ki, w in zip(top_idx, weights):
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]
            else:
                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])
                if m > 0:
                    j2 = np.arange(m, n)
                    grad[j2] -= w * h_cur[j2 - m]
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            stall = 0
        if sc < compute_c5(h_cur) - 1e-12:
            h_cur = h_new
            lr = min(lr * 1.03, 0.05)
            stall = 0
        else:
            lr *= 0.9
            stall += 1
            if stall > 100 or lr < 1e-10:
                break
    return best_h, best_score


def binary_sa(h_init, target, n_iter, deadline, T0=0.05, T1=0.0005):
    n = len(h_init)
    n_ones = int(round(target))
    h_b = (h_init > 0.5).astype(float)
    while int(h_b.sum()) > n_ones:
        idx = np.where(h_b == 1)[0]
        h_b[np.random.randint(len(idx))] = 0
    while int(h_b.sum()) < n_ones:
        idx = np.where(h_b == 0)[0]
        h_b[np.random.randint(len(idx))] = 1
    best_h = h_b.copy()
    best_score = compute_c5(h_b)
    cur_score = best_score
    for it in range(n_iter):
        if time.time() > deadline:
            break
        T = T0 * (T1 / T0) ** (it / max(n_iter, 1))
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_b[i] = 0.0
        h_b[j] = 1.0
        new_score = compute_c5(h_b)
        delta = new_score - cur_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_b.copy()
        else:
            h_b[i] = 1.0
            h_b[j] = 0.0
    return best_h, best_score

def diverse_restart(h_base, target, n, strat, param, rng, deadline):
    h_pert = h_base.copy()
    if strat == 'gauss':
        h_pert += rng.randn(n) * param
    elif strat == 'sparse':
        n_flip = max(1, int(n * param))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_pert[idx] = rng.uniform(0, 1, n_flip)
    elif strat == 'smooth':
        freq = np.fft.rfftfreq(n)
        nfreq = n // 2 + 1
        mask = (freq > 0) & (freq < 0.03)
        spec = np.zeros(nfreq)
        spec[mask] = rng.randn(int(mask.sum()))
        pert = np.fft.irfft(spec, n)
        h_pert += pert * param
    elif strat == 'block':
        block_size = max(1, int(n * param))
        n_blocks = n // block_size
        for b in range(n_blocks):
            if rng.random() < 0.3:
                h_pert[b * block_size:(b + 1) * block_size] = 1.0 - h_pert[b * block_size:(b + 1) * block_size]
    h_pert = project_feasible(h_pert, target)
    h_new = lbfgs_smooth(h_pert, target, 500, 50, 250, deadline)
    h_new = lbfgs_smooth(h_new, target, 5000, 50, 250, deadline)

    # Phase 8: Nesterov at current resolution
    if tleft() > 120:
        h_ns2, sc_ns2 = nesterov_subgrad(best_h, target, 300, deadline, topk=3, lam=0.5)
        if sc_ns2 < best_score:
            best_score = sc_ns2
            best_h = h_ns2

    # Phase 9: Top-lag GD
    if tleft() > 100:
        h_tl2, sc_tl2 = top_lag_gd(best_h, target, topk=3, n_steps=200, deadline=deadline)
        if sc_tl2 < best_score:
            best_score = sc_tl2
            best_h = h_tl2

    # Phase 10: L-BFGS polish
    if tleft() > 80:
        h_pol2 = lbfgs_smooth(best_h, target, 20000, 100, 400, deadline)
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 11: Coordinate descent
    if tleft() > 100:
        h_cd, sc_cd = coordinate_descent_fast(best_h, target, n_sweeps=2, deadline=deadline)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 12: Final Nesterov + L-BFGS
    if tleft() > 60:
        h_ns3, sc_ns3 = nesterov_subgrad(best_h, target, 200, deadline, topk=3, lam=0.6)
        if sc_ns3 < best_score:
            best_score = sc_ns3
            best_h = h_ns3
        if tleft() > 30:
            h_final = lbfgs_smooth(best_h, target, 50000, 200, 300, deadline)
            sc = compute_c5(h_final)
            if sc < best_score:
                best_score = sc
                best_h = h_final

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
