def corr_values(h):

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sum = h.sum()
    if abs(h_sum - target) < 1e-10:
        return np.clip(h, 0.0, 1.0)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:

    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp

    v = 1.0 - h

    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)

    corr_neg = np.fft.irfft(W * H, fsize)[:n]

    rng = np.random.RandomState(777)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        any_improve = False
        for idx in order:
            if time.time() > deadline:
                break
            old_val = h_cur[idx]
            
            lo = max(0.0, old_val - 0.15)
            hi = min(1.0, old_val + 0.15)
            
            if hi - lo < 1e-10:
                continue
            
            best_local = cur_score
            best_v = old_val
            ratio = (5.0 - np.sqrt(5.0)) / 2.0
            a = lo
            b = hi
            c = b - ratio * (b - a)
            d = a + ratio * (b - a)
            
            h_cur[idx] = c
            h_c = project_feasible(h_cur.copy(), target)
            sc_c = compute_c5(h_c)
            h_cur[idx] = d
            h_d = project_feasible(h_cur.copy(), target)
            sc_d = compute_c5(h_d)
            
            for _ in range(15):
                if b - a < 1e-8:
                    break
                if sc_c < sc_d:
                    b = d
                    d = c
                    sc_d = sc_c
                    c = b - ratio * (b - a)
                    h_cur[idx] = c
                    h_c = project_feasible(h_cur.copy(), target)
                    sc_c = compute_c5(h_c)
                else:
                    a = c
                    c = d
                    sc_c = sc_d
                    d = a + ratio * (b - a)
                    h_cur[idx] = d
                    h_d = project_feasible(h_cur.copy(), target)
                    sc_d = compute_c5(h_d)
            
            best_v = (a + b) / 2.0
            h_cur[idx] = best_v
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < best_local - 1e-15:
                best_local = sc
                any_improve = True
            else:
                h_cur[idx] = old_val
                h_cur = project_feasible(h_cur, target)
        
        if not any_improve and sweep > 50:
            break
    return h_cur, cur_score if not any_improve else compute_c5(h_cur)

def micro_restart(h, target, rng, deadline, scale):
    n = len(h)
    h_pert = h + rng.randn(n) * scale
    h_pert = project_feasible(h_pert, target)
    h_new, sc = fista_optimize(h_pert, target, 5000, 5000, deadline, objective_type='lse')
    if sc > compute_c5(h):
        return h, compute_c5(h)
    return h_new, sc

def binary_local_search(h_init, target, deadline, max_iters=2000):
    n = len(h_init)
    dx = 2.0 / n
    h_bin = (h_init >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h_bin > 0.5)[0]
        for i in range(n_ones - target_ones):
            h_bin[ones[np.random.RandomState(i).randint(n)]] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h_bin < 0.5)[0]
        for i in range(target_ones - n_ones):
            h_bin[zeros[np.random.RandomState(i + 100).randint(n)]] = 1.0
    h_bin = project_feasible(h_bin, target)
    cur_score = compute_c5(h_bin)
    rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if abs(h_bin[i] - h_bin[j]) < 0.1:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        h_bin = project_feasible(h_bin, target)
        sc = compute_c5(h_bin)
        if sc < cur_score:
            cur_score = sc
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            h_bin = project_feasible(h_bin, target)
    return h_bin, cur_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline_final = start_time + budget_s - 10.0
    
    n = len(initial_h_values)
    target = n / 2.0
    h_init = project_feasible(np.array(initial_h_values, dtype=float), target)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    rng = np.random.RandomState(seed)
    
    def tleft():
        return deadline_final - time.time()
    
    # Phase 1: Binary local search for a good discrete starting point
    if tleft() > 100:
        h_bin, sc_bin = binary_local_search(h_init, target, start_time + 30, max_iters=3000)
        h_bin_continuous = np.clip(h_bin, 0, 1)
        h_bin_continuous = project_feasible(h_bin_continuous, target)
        sc_bin_c = compute_c5(h_bin_continuous)
        if sc_bin_c < best_score:
            best_score = sc_bin_c
            best_h = h_bin_continuous
    
    # Phase 2: FISTA continuation with moderate alpha
    alphas = [100, 500, 2000, 8000, 30000, 100000]
    for a in alphas:
        if tleft() < 30:
            break
        td = 25
        h_new, sc_new = fista_optimize(best_h, target, a, 100000, start_time + td, objective_type='lse')
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 3: Top-k equalization
    if tleft() > 40:
        h_new, sc_new = topk_equalize(best_h, target, 300, start_time + 35, k=5)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 4: High-alpha FISTA
    if tleft() > 60:
        h_new, sc_new = fista_optimize(best_h, target, 100000, 100000, start_time + 50, objective_type='lse')
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 5: Top-k equalization again
    if tleft() > 30:
        h_new, sc_new = topk_equalize(best_h, target, 200, start_time + 25, k=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 6: FISTA with top-k objective
    if tleft() > 50:
        h_new, sc_new = fista_optimize(best_h, target, 1, 50000, start_time + 45, objective_type='topk')
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 7: Golden section coordinate descent
    if tleft() > 120:
        h_new, sc_new = golden_coord_descent(best_h, target, start_time + min(110, tleft() - 10), max_sweeps=500)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 8: Micro-restarts
    if tleft() > 80:
        scales = [0.01, 0.02, 0.015, 0.025, 0.008]
        for s in scales:
            if tleft() < 20:
                break
            h_new, sc_new = micro_restart(best_h, target, rng, start_time + 15, s)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new
    
    # Phase 9: Final top-k polish
    if tleft() > 30:
        h_new, sc_new = topk_equalize(best_h, target, 200, start_time + 25, k=3)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 10: Final short coordinate descent
    if tleft() > 40:
        h_new, sc_new = golden_coord_descent(best_h, target, start_time + min(35, tleft() - 5), max_sweeps=200)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 11: Very fine FISTA
    if tleft() > 30:
        h_new, sc_new = fista_optimize(best_h, target, 200000, 50000, start_time + 25, objective_type='lse')
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    # Phase 12: Final coordinate descent
    if tleft() > 20:
        h_new, sc_new = golden_coord_descent(best_h, target, start_time + min(15, tleft() - 3), max_sweeps=100)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
    
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, len(h_final)
