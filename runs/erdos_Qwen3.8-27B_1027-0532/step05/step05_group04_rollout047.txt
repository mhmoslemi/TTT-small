    n = len(h)

    over = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(over)) * dx

            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        weights = np.exp(diffs / 0.003)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.01, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

A better approach: **minimize the max correlation subject to the sum constraint, using coordinate descent with a Lagrange multiplier for the sum constraint.**

For coordinate i, the sub-problem is:
min_t max_k (A_k * t + B_k)
s.t. t + (target - sum(h_cur) + h_cur[i]) = target  (i.e., just update the sum)

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_lse_cutoff = 700.0
_g_pen_coeff = 50.0
_g_focused_weights = None
_g_n_focus = 0
_g_fourier_m = 15
_g_fourier_n = 0
_g_fourier_target = 0.0
_g_fourier_bound = 0.3


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    over = np.correlate(h, 1.0 - h, mode='full')[n - 1:]
    return float(np.max(over)) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_lse(x):
    n = _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')[n - 1:]
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -_g_lse_cutoff, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def obj_lse_pen(x):
    n = _g_n
    target = _g_target
    a = _g_alpha
    corr = np.correlate(x, 1.0 - x, mode='full')[n - 1:]
    c_max = float(np.max(corr))
    shifted = a * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / a
    pen = _g_pen_coeff * (np.sum(x) - target) ** 2
    return lse + pen


def obj_focus(x):
    n = _g_n
    corr = np.correlate(x, 1.0 - x, mode='full')[n - 1:]
    focused = corr[:_g_n_focus]
    rest = corr[_g_n_focus:]
    return float(np.dot(_g_focused_weights, focused)) + 0.01 * float(np.max(rest))


def cons_eq(x):
    return np.sum(x) - _g_target


def slsqp_pass(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_alpha, _g_target
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq}]
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(obj_lse, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def lbfgsb_pass(h, n, target, alphas, maxiter, deadline=None):
    global _g_n, _g_alpha, _g_target, _g_pen_coeff
    _g_n = n
    _g_target = target
    _g_pen_coeff = 50.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(obj_lse_pen, h_cur, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def fourier_eval(params):
    n = _g_fourier_n
    m = _g_fourier_m
    target = _g_fourier_target
    a0 = params[0]
    x_arr = np.arange(n, dtype=float)
    h = np.full(n, a0)
    for j in range(1, m + 1):
        aj = params[2 * j - 1]
        bj = params[2 * j]
        h += aj * np.cos(2.0 * np.pi * j * x_arr / n)
        h += bj * np.sin(2.0 * np.pi * j * x_arr / n)
    h = np.clip(h, 0.0, 1.0)
    h = project_feasible(h, target)
    return compute_c5(h)


def fourier_de_search(n, target, m, scale, maxiter, popsize, seed, deadline=None):
    global _g_n, _g_target, _g_fourier_m, _g_fourier_n, _g_fourier_target, _g_fourier_bound
    _g_n = n
    _g_target = target
    _g_fourier_m = m
    _g_fourier_n = n
    _g_fourier_target = target
    _g_fourier_bound = scale
    bounds = [(0.0, 1.0)] + [(-scale, scale)] * (2 * m)
    try:
        result = differential_evolution(fourier_eval, bounds, maxiter=maxiter,
                                        popsize=popsize, seed=seed,
                                        tol=1e-12, polish=True, disp=False)
        params = result.x
        a0 = params[0]
        x_arr = np.arange(n, dtype=float)
        h = np.full(n, a0)
        for j in range(1, m + 1):
            aj = params[2 * j - 1]
            bj = params[2 * j]
            h += aj * np.cos(2.0 * np.pi * j * x_arr / n)
            h += bj * np.sin(2.0 * np.pi * j * x_arr / n)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target)
        return h, compute_c5(h)
    except Exception:
        return None, float('inf')


def focused_slq(h, n, target, n_rounds, top_k, maxiter, deadline=None):
    global _g_n, _g_target, _g_focused_weights, _g_n_focus
    _g_n = n
    _g_target = target
    _g_n_focus = top_k
    bounds = [(0.0, 1.0)] * n

    for _ in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n - 1:]
        top_idx = np.argsort(corr)[::-1][:top_k]
        _g_focused_weights = np.ones(top_k) / top_k
        try:
            res = minimize(obj_focus, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})

            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def subgradient_polish(h, n, target, n_steps, n_active, lr_init, lr_decay, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    lr = lr_init
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n - 1:]
        c_curr = float(np.max(corr)) * dx
        if c_curr < best_score:
            best_score = c_curr
            best_h = h_cur.copy()
        top_idx = np.argsort(corr)[::-1][:n_active]
        diffs = corr[top_idx] - corr[top_idx[0]]
        temp = 0.003
        weights = np.exp(diffs / temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for wk, k in zip(weights, top_idx):
            k = int(k)
            j_pos = np.arange(0, n - k)
            grad[j_pos] += wk * (1.0 - h_cur[j_pos + k])
            j_neg = np.arange(k, n)
            grad[j_neg] -= wk * h_cur[j_neg - k]
        grad *= dx
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        if sc_new < c_curr:
            h_cur = h_new
            lr = min(lr * 1.02, 0.005)
        else:
            lr *= lr_decay
            if lr < 1e-10:
                break
    return best_h, best_score

    no_improve = 0

            noise += rng.randn() * np.sin(2.0 * np.pi * j * x_arr / n)
        noise *= cur_sigma / np.sqrt(m_fourier)
        h_pert = best_h + noise
        h_pert = project_feasible(h_pert, target)
        _g_alpha = 3000
        try:

            else:
                no_improve += 1
                if no_improve >= 4:
                    cur_sigma = min(cur_sigma * 1.6, 0.15)
                    no_improve = 0

    return best_h, best_score


def coord_minimax(h, n, target, n_sweeps, deadline=None):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    t_coarse = np.linspace(0.0, 1.0, 41)
    t_fine = np.linspace(0.0, 1.0, 401)
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            old_val = h_cur[i]
            max_k = min(i, n - 1 - i)
            if max_k < 0:
                continue
            A = np.zeros(n)
            for k in range(max_k + 1):
                A[k] = (1.0 - h_cur[i + k]) - h_cur[i - k]
            if i < n - 1 - i:
                for k in range(i + 1, n - i):
                    A[k] = 1.0 - h_cur[i + k]
            elif n - 1 - i < i:
                for k in range(n - i, i + 1):
                    A[k] = -h_cur[i - k]
            active_mask = A != 0.0
            if not np.any(active_mask):
                continue
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n - 1:]
            B = corr / dx - A * old_val
            g_coarse = np.zeros(len(t_coarse))
            for idx in range(len(t_coarse)):
                g_coarse[idx] = np.max(A[active_mask] * t_coarse[idx] + B[active_mask])
            best_idx = np.argmin(g_coarse)
            if 0 < best_idx < len(t_coarse) - 1:
                lo = t_coarse[best_idx - 1]
                hi = t_coarse[best_idx + 1]
                g_fine = np.zeros(len(t_fine))
                t_search = np.linspace(lo, hi, len(t_fine))
                for idx in range(len(t_fine)):
                    g_fine[idx] = np.max(A[active_mask] * t_search[idx] + B[active_mask])
                best_t = t_search[np.argmin(g_fine)]
            else:
                best_t = t_coarse[best_idx]
            h_cur[i] = best_t
        h_cur = project_feasible(h_cur, target)
        new_score = compute_c5(h_cur)
        if new_score < cur_score:
            cur_score = new_score
    return h_cur, cur_score
