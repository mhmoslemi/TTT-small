    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(corr.max()) * dx
    best_h = h.copy()
    
    tabu = {}  # (i,j) -> iteration when it was done
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 2000:
            break
        
        # Find top M shifts
        top_m_idx = np.argsort(corr)[::-1][:top_m]
        top_m_vals = corr[top_m_idx]
        max_val = corr.max()
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For each candidate swap, score based on effect on top M shifts
        # But this is O(|ones| * |zeros| * M) which is too expensive
        # Instead, sample candidates
        
        best_score = max_val
        best_i, best_j = -1, -1
        
        # Sample candidates
        n_sample = min(50, len(ones))
        n_sample_j = min(50, len(zeros))
        
        # Focus on positions that affect the worst shifts
        worst_k = top_m_idx[0] - (n-1)
        
        # For the worst shift, candidates are:
        # i should be a 1 where h[i+worst_k] is 0 (to reduce the worst shift)
        # j should be a 0 where h[j+worst_k] is 1 (to reduce the worst shift)
        
        i_cands = [i for i in ones if 0 <= i + worst_k < n and h[i+worst_k] == 0]
        j_cands = [j for j in zeros if 0 <= j + worst_k < n and h[j+worst_k] == 1]
        
        if not i_cands or not j_cands:
            # Try other top shifts
            found = False
            for k_idx in top_m_idx[1:]:
                k = k_idx - (n-1)
                i_cands = [i for i in ones if 0 <= i + k < n and h[i+k] == 0]
                j_cands = [j for j in zeros if 0 <= j + k < n and h[j+k] == 1]
                if i_cands and j_cands:
                    found = True
                    break
            if not found:
                stall += 10
                continue
        
        # Score top candidates
        i_cands = i_cands[:n_sample]
        j_cands = j_cands[:n_sample_j]
        
        for i in i_cands:
            for j in j_cands:
                if (i,j) in tabu and it - tabu[(i,j)] < tabu_len:

                # Compute max over top M shifts
                new_max = 0

def swap_delta(h, corr, n, i, j):
    """Compute the change in the full correlation vector from swapping h[i]: 1->0, h[j]: 0->1"""
    # Only affects shifts k where i or j is in range
    # For efficiency, compute delta for each k
    # S[k_idx] = S[k_idx] + delta_k
    # where k = k_idx - (n-1)
    
    # For shift k:
    # delta_k = -(1-h[i+k])*[0<=i+k<n] + (1-h[j+k])*[0<=j+k<n] 
    #           + h[i-k]*[0<=i-k<n] - h[j-k]*[0<=j-k<n]
    
    # This is nonzero for k in:
    # [-(n-1-i), n-1-i] ∪ [-(n-1-j), n-1-j] ∪ [i-(n-1), i] ∪ [j-(n-1), j]
    
    # For simplicity, just compute for all k in range
    k_range = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    idx_ik = np.where((i + k_range >= 0) & (i + k_range < n))[0]
    delta[idx_ik] -= (1.0 - h[i + k_range[idx_ik]])
    
    idx_jk = np.where((j + k_range >= 0) & (j + k_range < n))[0]
    delta[idx_jk] += (1.0 - h[j + k_range[idx_jk]])
    
    idx_ik2 = np.where((i - k_range >= 0) & (i - k_range < n))[0]
    delta[idx_ik2] += h[i - k_range[idx_ik2]]
    
    idx_jk2 = np.where((j - k_range >= 0) & (j - k_range < n))[0]
    delta[idx_jk2] -= h[j - k_range[idx_jk2]]
    
    return delta

                i_cands = list(rng.choice(i_cands, mc, replace=False))
            if len(j_cands) > mc:
                j_cands = list(rng.choice(j_cands, mc, replace=False))
            for i in i_cands:
                for j in j_cands:
                    nm = 0.0
                    for ki in range(top_m):
                        kk = int(top_m_idx[ki] - (n - 1))
                        d = 0.0
                        if 0 <= i + kk < n:
                            d -= (1.0 - h[i + kk])
                        if 0 <= i - kk < n:
                            d += h[i - kk]
                        if 0 <= j + kk < n:
                            d += (1.0 - h[j + kk])
                        if 0 <= j - kk < n:
                            d -= h[j - kk]
                        v = float(top_vals[ki]) + d
                        if v > nm:
                            nm = v
                    if nm < best_score:
                        best_score = nm
                        best_i, best_j = i, j
            break

        if not found or best_i < 0:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0
            h[j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[best_i] = 0
            h[best_j] = 1
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')
            sc = float(corr.max()) * dx
            if sc < best_c5:
                best_c5 = sc
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = int(n / 2.0)

    best_h = project_binary(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick greedy initialization (10% budget)
    allowed = min(80, tleft() * 0.10)
    h_g, c5_g = tabu_search(best_h, n, 50000, top_m=6, tabu_len=100,
                            time_limit=allowed, start_time=start_time,
                            rng=rng, kick_interval=5000)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Intensive tabu search (40% budget)
    if tleft() > 50:
        allowed = min(450, tleft() * 0.40)
        h_t, c5_t = tabu_search(best_h, n, 200000, top_m=8, tabu_len=200,
                                time_limit=allowed, start_time=start_time,
                                rng=rng, kick_interval=5000)
        if c5_t < best_score:
            best_score = c5_t
            best_h = h_t.copy()

    # Phase 3: Multi-shift SA (20% budget)
    if tleft() > 50:
        allowed = min(250, tleft() * 0.20)
        max_it = int(min(500000, allowed * 800))
        h_ms, c5_ms = multi_shift_sa(best_h, n, max_it, top_m=8,
                                     time_limit=allowed, start_time=start_time, rng=rng)
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms.copy()

    # Phase 4: Standard SA refinement (15% budget)
    if tleft() > 50:
        allowed = min(180, tleft() * 0.15)
        max_it = int(min(400000, allowed * 1000))
        h_sa, c5_sa = sa_refine(best_h, n, max_it, 0.008, 0.00002,
                                time_limit=allowed, start_time=start_time, rng=rng)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 5: Second tabu round with different params
    if tleft() > 60:
        allowed = min(200, tleft() * 0.15)
        h_t2, c5_t2 = tabu_search(best_h, n, 150000, top_m=10, tabu_len=300,
                                  time_limit=allowed, start_time=start_time,
                                  rng=rng, kick_interval=3000)
        if c5_t2 < best_score:
            best_score = c5_t2
            best_h = h_t2.copy()

    # Phase 6: Try different n values
    n_options = [n + 2, n - 2, n + 4, n + 6]
    for n_try in n_options:
        if tleft() < 50:
            break
        if n_try < 100 or n_try % 2 != 0:
            continue
        target_try = int(n_try / 2)
        x_new = np.linspace(0.0, 1.0, n_try)
        x_old = np.linspace(0.0, 1.0, n)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(40, tleft() * 0.08)
        h_g2, c5_g2 = tabu_search(h_bin, n_try, 30000, top_m=6, tabu_len=100,
                                  time_limit=allowed, start_time=start_time,
                                  rng=rng, kick_interval=3000)
        sc2 = c5_g2
        if sc2 < best_score:
            best_score = sc2
            best_h = h_g2.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 100, 300, 1000, 3000, 10000, 30000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 3000, 1e-15,
                                 time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary SA on refined solution
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(40, tleft() - 10)
        max_it5 = int(min(150000, allowed * 1200))
        h_sf, c5_sf = sa_refine(h_bin_f, n_use, max_it5, 0.004, 0.00001,
                                time_limit=allowed, start_time=start_time, rng=rng)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final quick tabu
    if tleft() > 20:
        n_use = len(best_h)
        allowed = min(25, tleft() - 5)
        h_f, c5_f = tabu_search(best_h, n_use, 50000, top_m=8, tabu_len=150,
                                time_limit=allowed, start_time=start_time,
                                rng=rng, kick_interval=2000)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_f

    # Final SLSQP polish
    if tleft() > 10:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [5000, 20000, 100000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 2000, 1e-16,
                                   time_limit=max(3.0, tleft() - 3), start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
