import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h):
    """Compute overlap values for all positive lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx  # lags 0 to n-1

def compute_c5(h):
    return float(np.max(compute_overlap(h)))

def project_binary(h, target_sum):
    """Project h to binary (0/1) with exact sum."""
    n = len(h)
    # Sort by value, set top 'target_sum' to 1, rest to 0
    # But target_sum must be integer for this to work
    # Actually target_sum = n/2, which is integer if n is even
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:int(target_sum)]] = 1.0
    return h_bin

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_swap_search(h, n_iterations=5000, rng=None):
    """Local search on binary patterns using pair swaps."""
    n = len(h)
    target = n // 2
    h_cur = project_binary(h, target)
    ones = np.where(h_cur == 1)[0]
    zeros = np.where(h_cur == 0)[0]
    cur_score = compute_c5(h_cur)
    best_score = cur_score
    best_h = h_cur.copy()
    
    for it in range(n_iterations):
        if rng is None:
            break
        # Pick a random 1 and a random 0
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        new_score = compute_c5(h_cur)
        
        if new_score <= cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
            # Update sets
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
        else:
            # Revert
            h_cur[i] = 1.0
            h_cur[j] = 0.0
    
    return best_h, best_score

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:n-1+n])) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[n-1:n-1+n] * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

def binary_sa(h_init, n_iters, rng, t0, t1, cooldown):
    """Simulated annealing on binary patterns."""
    n = len(h_init)
    target = n // 2
    h_cur = to_binary(h_init, target)
    best_h = h_cur.copy()
    cur_score = compute_c5(h_cur)
    best_score = cur_score
    ones = np.where(h_cur == 1)[0].tolist()
    zeros = np.where(h_cur == 0)[0].tolist()
    
    for it in range(n_iters):
        T = t0 * (t1/t0) ** (it / max(n_iters-1, 1))
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h_cur[i] = 0.0
        h_cur[j] = 1.0
        new_score = compute_c5(h_cur)
        delta = new_score - cur_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            # Update sets
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_score < best_score:
                best_score = new_score
                best_h = h_cur.copy()
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
    
    return best_h, best_score

def nesterov_pg(h_init, n_steps, lr, target):
    """Nesterov-accelerated projected gradient descent."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for step in range(1, n_steps + 1):
        t_k = (1 + np.sqrt(1 + 4 * (step-1)**2)) / 2
        beta = (step - 1) / t_k
        
        # Compute gradient at y
        overlaps = compute_overlaps(y)
        # Use soft-max to get a smooth gradient
        alpha = 500.0
        c_max = np.max(overlaps)
        weights = np.exp((overlaps - c_max) / (c_max / alpha))
        weights /= weights.sum()
        
        # Gradient of weighted sum of overlaps
        grad = np.zeros(n)
        for k in range(n):
            w = weights[k]
            if w < 1e-10:
                continue
            k = int(k)
            if k == 0:
                grad += w * dx * (1.0 - 2.0 * y)
            else:
                upper = n - 1 - k
                idx1 = np.arange(0, upper + 1)
                grad[idx1] += w * dx * (1.0 - y[idx1 + k])
                idx2 = np.arange(k, n)
                grad[idx2] -= w * dx * y[idx2 - k]
        
        # Nesterov update
        y_new = project_feasible(y - lr * grad, target)
        h_new = y_new + beta * (y_new - y)
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        
        y = y_new
        h = h_new
    
    return best_h, best_score

            h_p = project_feasible(x, target)
            corr = np.correlate(h_p, 1.0 - h_p, mode='full')
            over = corr[n-1:n-1+n] * (2.0/n)

            return grad

            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)

            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new

The product u_i * u_{i+k} can be handled using the "epigraph" trick if we introduce auxiliary variables, but it's still non-convex in general.

So: O_k = ∑ u_i - ∑ u_i u_{i+k} ≤ ∑ u_i - ∑ max(0, u_i + u_{i+k} - 1)

And: O_k ≥ ∑ u_i - ∑ min(u_i, u_{i+k}) = ∑ u_i - ∑ max(0, min(u_i, u_{i+k}))

These bounds are tight when h is binary.

Let me also add a key improvement: **use the "top-p mean" objective** instead of soft-max. The top-p mean is:
f(x) = (1/p) * sum of the p largest components of x

This is a convex function and gives a tighter lower bound on the max than the log-sum-exp (soft-max) for any finite p. As p→1, it approaches the max.

For p=1: f(x) = max(x) (non-differentiable)
For p=2: f(x) = (max(x) + second_max(x)) / 2

The gradient of the top-p mean (when all top-p values are distinct) is:
∇f(x) = (1/p) * 1_{i ∈ top-p}(x)

So it's a uniform weighting over the top-p components. This is actually simpler and potentially better than the soft-max.

    """Compute the top-p mean of the overlaps."""
    over = compute_overlaps(h)

def top_p_mean_grad(h, p=2):
    """Gradient of the top-p mean of overlaps."""
    n = len(h)
    dx = 2.0 / n
    over = compute_overlaps(h)
    top_idx = np.argsort(over)[::-1][:p]
    grad = np.zeros(n)
    for k in top_idx:
        k = int(k)
        if k == 0:
            grad += (1.0/p) * dx * (1.0 - 2.0 * h)
        else:
            upper = n - 1 - k
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += (1.0/p) * dx * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= (1.0/p) * dx * h[idx2 - k]
    return grad

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr[n-1:n-1+n])) * dx
