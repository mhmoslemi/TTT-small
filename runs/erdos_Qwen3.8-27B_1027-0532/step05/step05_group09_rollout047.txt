    n = len(h)

    cur_score = compute_c5(h_cur)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')[n-1:]

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(100):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

**Let me now think about the "equalization" phase more carefully.**

At the optimum (or near it), we have:

            best_h = h_new

    # Phase 6: Coarse-to-fine
    if tleft() > 120:
        n_coarse = n // 4
        target_coarse = n_coarse / 2.0
        x_coarse = np.linspace(0, 1, n_coarse)
        x_fine = np.linspace(0, 1, n)
        h_coarse = np.interp(x_coarse, x_fine, best_h)
        h_coarse = project_feasible(h_coarse, target_coarse)
        h_coarse, _ = fista_adaptive(h_coarse, target_coarse, 500, 1.0, 4, 30000, start_time + 30)
        h_coarse, _ = fista_adaptive(h_coarse, target_coarse, 5000, 2.0, 4, 30000, start_time + 50)
        h_up = np.interp(x_fine, x_coarse, h_coarse)
        h_up = project_feasible(h_up, target)
        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up
            h_new, sc_new = fista_adaptive(best_h, target, 5000, 1.0, 4, 50000, start_time + 40)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 7: Perturbation restarts
    rng = np.random.RandomState(seed)
    if tleft() > 100:
        scales = [0.02, 0.05, 0.08, 0.03, 0.06]
        for s in scales:
            if tleft() < 30:
                break
            h_pert = best_h + rng.randn(n) * s
            h_pert = project_feasible(h_pert, target)
            h_new, sc_new = fista_adaptive(h_pert, target, 2000, 1.0, 4, 30000, start_time + 25)
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_new

    # Phase 8: Structured restarts
    if tleft() > 80:
        for pat in range(5):
            if tleft() < 25:
                break
            h_struct = structured_init(n, pat)
            # Blend with current best
            for frac in [0.5, 0.7]:
                h_blend = (1 - frac) * h_struct + frac * best_h
                h_blend = project_feasible(h_blend, target)
                h_new, sc_new = fista_adaptive(h_blend, target, 1000, 1.0, 5, 20000, start_time + 15)
                if sc_new < best_score:
                    best_score = sc_new
                    best_h = h_new
                if tleft() < 10:
                    break

    # Phase 9: Final high-alpha push
    if tleft() > 60:
        td = min(40, tleft() - 15)
        h_new, sc_new = fista_adaptive(best_h, target, 50000, 5.0, 3, 80000, start_time + td)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 10: Fine polish
    if tleft() > 30:
        h_new, sc_new = fine_polish(best_h, target, start_time + max(tleft() - 5, 10))
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Final: ensure feasibility
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score
    return h_final, final_score, n
