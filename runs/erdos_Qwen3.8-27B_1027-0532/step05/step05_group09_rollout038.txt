def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def find_optimal_v(a, b, tol=1e-10, max_iters=60):
    """Minimize max_k (a[k] + b[k]*v) over v in [0,1]"""
    lo = 0.0
    hi = 1.0
    
    for _ in range(max_iters):
        # Use a 3-point evaluation (ternary search doesn't work for non-smooth, use bisection on subgradient)
        mid = (lo + hi) / 2.0
        vals = a + b * mid
        k_max = np.argmax(vals)
        max_val = vals[k_max]
        
        # Find all active (within tolerance)
        active = np.where(abs(vals - max_val) < 1e-8)[0]
        avg_b = b[active].mean()
        
        if avg_b > 0:
            # Function is increasing, go left
            hi = mid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):

            s = np.sum(np.clip(h - mid, 0.0, 1.0))

                lo = mid
            else:

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    # Gradient via FFT
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

            break

            best_x = x.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = h.astype(float).copy()
    n = len(h)
    h_sorted = np.sort(h)[::-1]
    cumsum = np.concatenate([[0.0], np.cumsum(h_sorted)])
    best_tau = None
    for k in range(1, n + 1):
        tau = (cumsum[k] - target) / k
        lower = h_sorted[k - 1] - 1.0
        upper = h_sorted[k - 1]
        if k < n:
            lower = max(lower, h_sorted[k])
        if lower <= tau + 1e-12 and tau <= upper + 1e-12:
            best_tau = tau
            break
    if best_tau is None:
        lo = h.min() - 1.0
        hi = h.max() + 0.5
        for _ in range(200):
            mid = (lo + hi) / 2.0
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target:
                lo = mid
            else:
                hi = mid
        best_tau = (lo + hi) / 2.0
    return np.clip(h - best_tau, 0.0, 1.0)

def lse_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    weights = exp_shifted / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    w_flipped = weights[::-1].copy()
    v = 1.0 - h
    fsize = 2 * n
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(v, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_opt(h0, target, alpha, deadline, lr_init=None):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    if lr_init is None:
        lr_init = 0.8 / max(alpha * 0.01, 0.5)
    step = lr_init
    last_improve = time.time()
    for k in range(1000000):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = lse_obj_and_grad(y_clipped, alpha)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        sc_new = compute_c5(x_new)
        sc_cur = compute_c5(x)
        if sc_new <= sc_cur + 0.5e-12:
            step = min(step * 1.08, lr_init * 5.0)
            if sc_new < best_obj:
                best_obj = sc_new
                best_x = x_new.copy()
                last_improve = time.time()
        else:
            step *= 0.5
            x_new = project_feasible(y_clipped - step * grad_y, target)
            if compute_c5(x_new) > sc_cur + 0.5e-12:
                step *= 0.5
                x_new = project_feasible(y_clipped - step * grad_y, target)
        x_prev = x
        x = x_new
        t = t_new
        if time.time() - last_improve > 30:
            break
    return best_x, best_obj

def subgradient_active(h, target, max_rounds, deadline, top_k=5):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    stagnant = 0
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
        c_max = float(np.max(corr))
        sorted_corr = np.sort(corr)[::-1]
        spread = sorted_corr[0] - sorted_corr[min(top_k, n-1)]
        temp = max(0.0005, spread * 0.5)
        top_idx = np.argsort(corr)[::-1][:top_k]
        shifted = (corr[top_idx] - c_max) / temp
        shifted = np.clip(shifted, -50.0, 0.0)
        w = np.exp(shifted)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            jp = np.arange(0, n - ki)
            grad[jp] += wi * (1.0 - h_cur[jp + ki])
            jn = np.arange(ki, n)
            grad[jn] -= wi * h_cur[jn - ki]
        grad *= dx
        found = False
        for eps in [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            h_new = h_cur - eps * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < cur_score - 1e-14:
                h_cur = h_new
                cur_score = sc
                found = True
                stagnant = 0
                break
        if not found:
            stagnant += 1
            if stagnant > max_rounds:
                break
    return h_cur, cur_score

def coord_1d_search(h, target, deadline, max_sweeps=200):
    n = len(h)
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    no_improve_sweeps = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        sweep_improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old_val = h_cur[j]
            if old_val < 1e-12 or old_val > 1.0 - 1e-12:
                continue
            h_backup = h_cur.copy()
            h_cur[j] = 0.0
            corr_0 = np.correlate(h_cur, 1.0 - h_cur, mode='full')[n-1:]
            influence = np.zeros(n)
            n_minus_j = n - j
            if n_minus_j > 0:
                if n_minus_j == 1:
                    influence[0] += 1.0
                else:
                    influence[0] += 1.0
                    influence[1:n_minus_j] += 1.0 - h_cur[j+1:j+n_minus_j]
            if j >= 1:
                influence[1:j+1] -= h_cur[j-1::-1]
            lo = 0.0
            hi = 1.0
            for _ in range(35):
                if hi - lo < 1e-12:
                    break
                mid = (lo + hi) / 2.0
                vals = corr_0 + mid * influence
                k_max = int(np.argmax(vals))
                max_v = vals[k_max]
                tol = 1e-12 * max(1.0, abs(max_v))
                active_mask = np.abs(vals - max_v) <= tol
                n_active = int(active_mask.sum())
                if n_active == 0:
                    active_mask[k_max] = True
                    n_active = 1
                avg_b = float(np.sum(influence[active_mask]) / n_active)
                if avg_b > 1e-14:
                    hi = mid
                elif avg_b < -1e-14:
                    lo = mid
                else:
                    break
            best_v = (lo + hi) / 2.0
            if abs(best_v - old_val) < 1e-10:
                h_cur = h_backup
                continue
            h_cur = h_backup
            h_cur[j] = best_v
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < cur_score - 1e-14:
                cur_score = sc
                sweep_improved = True
            else:
                h_cur = h_backup
        if sweep_improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
    return h_cur, cur_score

def coord_boundary(h, target, deadline):
    n = len(h)
    h_cur = project_feasible(h.copy(), target)
    cur_score = compute_c5(h_cur)
    for _ in range(20):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:

            if sc < cur_score - 1e-14:

                improved = True
            else:
