import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(np.int32)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        extra = final_sum - target_sum
        ones = np.where(h_r == 1)[0]
        h_r[ones[:extra]] = 0
    elif final_sum < target_sum:
        extra = target_sum - final_sum
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[:extra]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            c_max = float(np.max(c))
            shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
            return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)
        def cons_fun(x):
            return np.sum(x) - target
        def cons_jac(x):
            return np.ones(n)
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

                              rng=rng, time_limit=min(30, tleft()*0.1),
                              start_time=start_time, top_m=3)
    if c5_cs < c5_cg:
        h_coarse_best = h_cs
    else:
        h_coarse_best = h_cg
    h_fine = np.interp(np.linspace(0, 1, len(best_h)), np.linspace(0, 1, n_coarse), h_coarse_best)
    h_fine_bin = project_binary(h_fine, target_init if len(best_h)==n_init else len(best_h)//2)
    h_fg, c5_fg = greedy_descent(h_fine_bin, max_iters=20000, rng=rng,
                                 time_limit=min(10, tleft()*0.04), start_time=start_time)
    n_saf = int(min(500000, min(20, tleft()*0.08) * 25000))
    h_fs, c5_fs = sa_multilag(h_fg, n_saf, t_init=0.008, t_final=0.00005,
                              rng=rng, time_limit=min(20, tleft()*0.08),
                              start_time=start_time, top_m=3)
    if c5_fs < c5_fg:
        h_embed = h_fs
        c5_embed = c5_fs
    else:
        h_embed = h_fg
        c5_embed = c5_fg
    if c5_embed < best_score:
        best_score = c5_embed
        best_h = h_embed.copy()

    if tleft() > 40:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [30, 80, 200, 600, 2000, 6000]
        allowed = min(tleft() - 20, 150)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1500, 1e-14,
                                 time_limit=allowed, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 20:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_final = project_binary(best_h, target_use)
        n_sa_f = int(min(2000000, max(50000, (tleft() - 10) * 30000)))
        h_sa_f, c5_f = sa_multilag(h_bin_final, n_sa_f, t_init=0.006, t_final=0.00002,
                                   rng=rng, time_limit=max(5.0, tleft() - 8),
                                   start_time=start_time, top_m=3)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    if tleft() > 15:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                 [3000, 8000, 15000], 1000, 1e-15,
                                 time_limit=max(5.0, tleft() - 8), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
