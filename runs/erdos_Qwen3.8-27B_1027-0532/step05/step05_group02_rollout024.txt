```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max_and_grad(h, alpha):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    full = np.correlate(h, 1.0 - h, mode='full')
    
    # Smooth max: (1/alpha) * logsumexp(alpha * full)
    full_max = full.max()
    shifted = alpha * (full - full_max)
    log_sum_exp = full_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    # Weights
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Gradient computation
    j_vals = np.arange(n)[:, None]
    s_vals = np.arange(2*n-1)[None, :] - (n-1)  # shift values
    
    js = j_vals + s_vals
    js_valid = (js >= 0) & (js < n)
    jsm = j_vals - s_vals
    jsm_valid = (jsm >= 0) & (jsm < n)
    
    h_js = np.zeros((n, 2*n-1))
    h_js[js_valid] = h[js[js_valid]]
    h_jsm = np.zeros((n, 2*n-1))
    h_jsm[jsm_valid] = h[jsm[jsm_valid]]
    
    w_exp = weights[None, :]
    T1 = np.sum(w_exp * js_valid * (1.0 - h_js), axis=1)
    T2 = np.sum(w_exp * jsm_valid * h_jsm, axis=1)
    grad = T1 - T2
    
    return log_sum_exp, grad, full_max

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = final_sum - target_sum
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = target_sum - final_sum
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)
```

```python
def lbfgsb_continuous(h_start, alpha, penalty, maxiter):
    n = len(h_start)
    target = n / 2.0
    
    def obj_and_grad(h):
        sm, grad, _ = smooth_max_and_grad(h, alpha)
        sum_diff = np.sum(h) - target
        obj = sm + penalty * sum_diff ** 2
        grad = grad + 2 * penalty * sum_diff
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h_start, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8})
    return res.x, res.fun
```

```python
def compute_c5(h):

def smooth_max_grad(h, alpha):
    n = len(h)
    full = np.correlate(h, 1.0 - h, mode='full')
    full_max = float(full.max())
    shifted = alpha * (full - full_max)
    log_sum = full_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    weights = np.exp(shifted)
    weights /= weights.sum()
    j_vals = np.arange(n)[:, None]
    s_vals = np.arange(2 * n - 1)[None, :] - (n - 1)
    js = j_vals + s_vals
    js_valid = (js >= 0) & (js < n)
    jsm = j_vals - s_vals
    jsm_valid = (jsm >= 0) & (jsm < n)
    h_js = np.zeros((n, 2 * n - 1))
    h_js[js_valid] = h[js[js_valid]]
    h_jsm = np.zeros((n, 2 * n - 1))
    h_jsm[jsm_valid] = h[jsm[jsm_valid]]
    w_exp = weights[None, :]
    T1 = np.sum(w_exp * js_valid * (1.0 - h_js), axis=1)
    T2 = np.sum(w_exp * jsm_valid * h_jsm, axis=1)
    grad = T1 - T2
    return log_sum, grad

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    return h_r.astype(float)

def lbfgsb_step(h_start, alpha, penalty, maxiter, n, time_limit=None, start_time=None):
    target = n / 2.0
    def obj_and_grad(h):
        sm, grad = smooth_max_grad(h, alpha)
        sum_diff = float(np.sum(h)) - target
        obj = sm + penalty * sum_diff * sum_diff
        grad = grad + 2.0 * penalty * sum_diff
        return obj, grad
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_and_grad, h_start, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_start.copy()

            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init // 2

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Multi-scale continuous L-BFGS-B
    resolutions = [50, 100, 200, 400, n_init]
    h_cont = h_init.copy()
    for n_res in resolutions:
        if tleft() < 40:
            break
        target_res = n_res // 2
        h_cur = interp_to_resolution(h_cont, n_res)
        h_cur = project_feasible(h_cur, float(target_res))
        alphas = [5, 20, 80, 300, 1000]
        for alpha in alphas:
            if tleft() < 20:
                break
            maxiter = 150 if alpha < 100 else 80
            h_new = lbfgsb_step(h_cur, alpha, 50.0, maxiter, n_res,
                                time_limit=min(15, tleft() * 0.3),
                                start_time=start_time)
            h_new = project_feasible(h_new, float(target_res))
            h_cur = h_new
        h_cont = h_cur
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()

    # Phase 2: Binary local search at n_init
    h_bin = project_binary(best_h, target_init)
    sc_bin = compute_c5(h_bin)
    if sc_bin < best_score:
        best_score = sc_bin
        best_h = h_bin.copy()
    else:
        h_bin = project_binary(best_h, target_init)

    if tleft() > 150:
        allow = min(200, tleft() * 0.3)
        h_dls, sc_dls = directed_local_search(h_bin, target_init, rng, allow, start_time)
        if sc_dls < best_score:
            best_score = sc_dls
            best_h = h_dls.copy()
        h_bin = h_dls

    # Phase 3: Iterated perturb + SA
    for it in range(8):
        if tleft() < 80:
            break
        h_pert = best_h.copy().astype(float)
        ones = np.where(h_pert == 1.0)[0]
        zeros = np.where(h_pert == 0.0)[0]
        n_pert = rng.randint(4, 10)
        for _ in range(n_pert):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            h_pert[i] = 0.0
            h_pert[j] = 1.0
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]

        allow_dls = min(40, tleft() * 0.15)
        h_d2, sc_d2 = directed_local_search(h_pert, target_init, rng, allow_dls, start_time)
        allow_sa = min(60, tleft() * 0.15)
        n_sa = int(allow_sa * 30000)
        h_s2, sc_s2 = sa_refined(h_d2, n_sa, 0.012, 0.00005, rng, allow_sa, start_time)
        sc_final = min(sc_d2, sc_s2)
        h_final = h_d2 if sc_d2 <= sc_s2 else h_s2
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final.copy()

    # Phase 4: Try higher resolutions
    for n_try in [800, 1000, 1200]:
        if tleft() < 120:
            break
        target_try = n_try // 2
        h_interp = interp_to_resolution(best_h, n_try)
        h_interp = project_feasible(h_interp, float(target_try))
        for alpha in [20, 100, 500]:
            if tleft() < 30:
                break
            h_new = lbfgsb_step(h_interp, alpha, 50.0, 100, n_try,
                                time_limit=min(20, tleft() * 0.3),
                                start_time=start_time)
            h_interp = project_feasible(h_new, float(target_try))
        sc_cont = compute_c5(h_interp)
        h_b2 = project_binary(h_interp, target_try)
        sc_b2 = compute_c5(h_b2)
        if tleft() > 60:
            allow = min(50, tleft() * 0.3)
            h_b2, sc_b2 = directed_local_search(h_b2, target_try, rng, allow, start_time)
        sc_best_try = min(sc_cont, sc_b2)
        h_best_try = h_interp if sc_cont <= sc_b2 else h_b2
        if sc_best_try < best_score:
            best_score = sc_best_try
            best_h = h_best_try.copy()

    # Phase 5: Final SA burst
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_f = project_binary(best_h, target_use)
        allow = max(10.0, tleft() - 15.0)
        n_sa_f = int(min(2000000, allow * 25000))
        h_sa_f, c5_f = sa_refined(h_bin_f, n_sa_f, 0.006, 0.00001, rng, allow, start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f.copy()

    # Phase 6: Final continuous polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        h_c = lbfgsb_step(best_h, 2000, 200.0, 200, n_use,
                          time_limit=max(5.0, tleft() - 12),
                          start_time=start_time)
        h_c = project_feasible(h_c, target_use)
        sc_c = compute_c5(h_c)
        if sc_c < best_score:
            best_score = sc_c
            best_h = h_c.copy()

    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, final_score, n_final
```
