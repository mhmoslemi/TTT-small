    """SA with adaptive temperature and focused moves."""
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(full)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 20000
    positions = np.arange(n)
    
    # Adaptive temperature tracking
    accept_count = 0
    eval_count = 0
    window = 500
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Adapt temperature based on acceptance rate
        if eval_count > window:
            accept_rate = accept_count / eval_count
            if accept_rate < 0.05 and t > t_final:
                t *= 1.5  # increase T if too few acceptances
            elif accept_rate > 0.8 and t < t_init:
                t *= 0.7  # decrease T if too many acceptances
            accept_count = 0
            eval_count = 0
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Choose move: focused or random
        if rng.random() < focus_ratio:
            # Focused: target the worst k
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]

                if len(i_cands) > 0 and len(j_cands) > 0:
                    i = i_cands[rng.randint(len(i_cands))]
                    j = j_cands[rng.randint(len(j_cands))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        new_max = float(np.max(full + delta))
        c5_new = new_max * dx
        delta_c5 = c5_new - current_c5
        eval_count += 1
        
        if delta_c5 < 0 or (t > 1e-15 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5_new
            accept_count += 1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:

        else:
            stall += 1

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted)) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(float)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r > 0.5)[0]
    zeros = np.where(h_r < 0.5)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0.0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1.0
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            o2 = np.where(h_r > 0.5)[0]
            h_r[o2[:final_sum - target_sum]] = 0.0
        else:
            z2 = np.where(h_r < 0.5)[0]
            h_r[z2[:target_sum - final_sum]] = 1.0
    return h_r

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def greedy_polish(h, rng, max_iters, time_limit, start_time):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(full)) * dx
    best_h = h.copy()
    stall = 0
    positions = np.arange(n)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > 200:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        max_val = float(np.max(full))
        max_idx = int(np.argmax(full))
        k_star = max_idx - (n - 1)
        if k_star == 0:
            stall += 1
            continue
        A = np.zeros(n)
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        A[valid] += h[idx[valid]]
        B = np.zeros(n)
        idx = positions - k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        idx = positions + k_star
        valid = (idx >= 0) & (idx < n)
        B[valid] += h[idx[valid]]
        top_i = ones[np.argsort(A[ones])[:min(20, len(ones))]]
        top_j = zeros[np.argsort(-B[zeros])[:min(20, len(zeros))]]
        found = False
        best_nm = max_val
        best_mv = None
        for i in top_i:
            for j in top_j:
                delta = compute_delta(h, i, j)
                nm = float(np.max(full + delta))
                if nm < best_nm:
                    best_nm = nm
                    best_mv = (i, j, delta)
        if best_mv is not None and best_nm < max_val:
            i, j, delta = best_mv
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            stall = 0
            nc5 = float(np.max(full)) * dx
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            for _ in range(20):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                delta = compute_delta(h, i, j)
                nm = float(np.max(full + delta))
                if nm < max_val:
                    h[i] = 0.0
                    h[j] = 1.0
                    full += delta
                    stall = 0
                    nc5 = float(np.max(full)) * dx
                    if nc5 < best_c5:
                        best_c5 = nc5
                        best_h = h.copy()
                    found = True
                    break
            if not found:
                stall += 1
    return best_h, best_c5

def adaptive_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time, focus_ratio=0.5):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(full)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    positions = np.arange(n)
    acc = 0
    evl = 0
    win = 400

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        if evl > win:
            ar = acc / evl
            if ar < 0.03 and t > t_final:
                t = min(t * 2.0, t_init)
            elif ar > 0.7 and t < t_init:
                t = max(t * 0.5, t_final)
            acc = 0
            evl = 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < focus_ratio:
            max_idx = int(np.argmax(full))
            k_star = max_idx - (n - 1)
            if k_star != 0:
                A = np.zeros(n)
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                B = np.zeros(n)
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                ic = ones[A[ones] <= 1.0]
                jc = zeros[B[zeros] >= 1.0]
                if len(ic) > 0 and len(jc) > 0:
                    i = ic[rng.randint(len(ic))]
                    j = jc[rng.randint(len(jc))]
                else:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j)
        nm = float(np.max(full + delta))
        c5n = nm * dx
        dc5 = c5n - current_c5
        evl += 1
        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5n
            acc += 1
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

def multi_k_sa(h, n_iters, t_init, t_final, rng, time_limit, start_time, n_top_k=3):
    n = len(h)
    h = h.copy().astype(float)
    full = full_corr_fast(h)
    dx = 2.0 / n
    best_c5 = float(np.max(full)) * dx
    best_h = h.copy()
    current_c5 = best_c5
    stall = 0
    max_stall = 30000
    positions = np.arange(n)

    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        if stall > max_stall:
            break
        ratio = it / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if it % 3 == 0:
            top_idx = np.argsort(full)[::-1][:n_top_k]
            A = np.zeros(n)
            B = np.zeros(n)
            for ti in top_idx:
                k_star = int(ti) - (n - 1)
                if k_star == 0:
                    continue
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                A[valid] += h[idx[valid]]
                idx = positions - k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
                idx = positions + k_star
                valid = (idx >= 0) & (idx < n)
                B[valid] += h[idx[valid]]
            ic = ones[A[ones] <= n_top_k * 0.7]
            jc = zeros[B[zeros] >= n_top_k * 0.3]
            if len(ic) > 0 and len(jc) > 0:
                i = ic[rng.randint(len(ic))]
                j = jc[rng.randint(len(jc))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        delta = compute_delta(h, i, j)
        nm = float(np.max(full + delta))
        c5n = nm * dx
        dc5 = c5n - current_c5
        if dc5 < 0 or (t > 1e-15 and rng.random() < np.exp(-dc5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            full += delta
            current_c5 = c5n
            if c5n < best_c5:
                best_c5 = c5n
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
    return best_h, best_c5

_cons_n = 0
_cons_target = 0.0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted)) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 12.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = int(n_init / 2.0)

    best_h = project_binary(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Greedy polish at initial resolution
    allowed = min(25, max(5, tleft() * 0.04))
    h_g, c5_g = greedy_polish(best_h, rng, 30000, allowed, start_time)
    if c5_g < best_score:
        best_score = c5_g
        best_h = h_g.copy()

    # Phase 2: Long adaptive SA at n_init
    allowed = min(180, max(20, tleft() * 0.22))
    n_sa = int(allowed * 80000)
    h_sa, c5_sa = adaptive_sa(best_h, n_sa, 0.018, 0.00004, rng,
                              allowed, start_time, focus_ratio=0.5)
    if c5_sa < best_score:
        best_score = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Multi-k focused SA
    if tleft() > 100:
        allowed = min(150, max(20, tleft() * 0.18))
        n_sa2 = int(allowed * 70000)
        h_mk, c5_mk = multi_k_sa(best_h, n_sa2, 0.015, 0.00003, rng,
                                  allowed, start_time, n_top_k=4)
        if c5_mk < best_score:
            best_score = c5_mk
            best_h = h_mk.copy()

    # Phase 4: Iterated local search (SA + perturb)
    if tleft() > 120:
        allowed_total = min(200, max(40, tleft() * 0.22))
        h_cur = best_h.copy()
        sc_cur = best_score
        while time.time() - start_time < start_time + allowed_total:
            allowed = min(25, allowed_total - (time.time() - start_time))
            if allowed < 15:
                break
            n_sa3 = int(allowed * 90000)
            h_i, c5_i = adaptive_sa(h_cur, n_sa3, 0.012, 0.00003, rng,
                                     allowed, start_time, focus_ratio=0.4)
            if c5_i < sc_cur:
                sc_cur = c5_i
                h_cur = h_i.copy()
            if c5_i < best_score:
                best_score = c5_i
                best_h = h_i.copy()
            n_pert = rng.randint(2, 7)
            h_pert = h_cur.copy()
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h_pert[ii] = 0.0
                h_pert[jj] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            if c5_i > best_score * 1.005:
                h_cur = best_h.copy()

    # Phase 5: Coarse-to-fine at higher resolutions
    for n_try in [1000, 1400, 1800, 2400, 3000]:
        if tleft() < 90:
            break
        if n_try % 2 != 0:
            continue
        target_try = n_try // 2
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(15, max(5, tleft() * 0.06))
        h_g2, c5_g2 = greedy_polish(h_bin, rng, 8000, allowed, start_time)

        allowed2 = min(50, max(15, tleft() * 0.12))
        if allowed2 < 15:
            break
        n_sa4 = int(allowed2 * 60000)
        h_s2, c5_s2 = adaptive_sa(h_g2, n_sa4, 0.012, 0.00008, rng,
                                   allowed2, start_time, focus_ratio=0.5)
        c5_try = c5_s2 if c5_s2 < c5_g2 else c5_g2
        h_try = h_s2 if c5_s2 < c5_g2 else h_g2
        if c5_try < best_score:
            best_score = c5_try
            best_h = h_try.copy()

    # Phase 6: SLSQP continuous refinement
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [15, 40, 100, 300, 800, 2000, 5000]
        allowed = min(tleft() - 25, 120)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 2000, 1e-14,
                                  allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final SA burst on binary
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2.0)
        h_bin_f = project_binary(best_h, target_use)
        allowed = max(10, tleft() - 18)
        n_sa_f = int(allowed * 90000)
        h_sf, c5_f = adaptive_sa(h_bin_f, n_sa_f, 0.008, 0.00001, rng,
                                  allowed, start_time, focus_ratio=0.6)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sf

    # Phase 8: Final SLSQP
    if tleft() > 25:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        allowed = max(10, tleft() - 12)
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use,
                                    [2000, 5000, 10000], 2000, 1e-15,
                                    allowed, start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Final feasibility and evaluation
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    h_bin_final = project_binary(best_h, int(target_final))
    score_bin = compute_c5(h_bin_final)

    if score_bin < final_score:
        return h_bin_final, score_bin, n_final
    return h_final, final_score, n_final
