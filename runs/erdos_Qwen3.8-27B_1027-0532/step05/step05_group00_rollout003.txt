def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:
        if final_sum > target_sum:
            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0
            final_sum -= 1
        else:
            idx = np.where(h_r == 0)[0]
            h_r[idx[0]] = 1
            final_sum += 1
    return h_r.astype(float)

# Smooth max for continuous optimization
def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted)) / alpha)

def _cons_fun(h):
    return np.sum(h) - _cons_target

def _cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    while final_sum != target_sum:

            idx = np.where(h_r == 1)[0]
            h_r[idx[0]] = 0

def _smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return c_max + float(np.sum(np.exp(shifted)) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

            best_h = h_sa.copy()

    # Phase 4: Try alternative n values
    n_options = [n_init + 2, n_init + 4, n_init - 2, n_init + 8, n_init - 4, n_init + 6]
    n_options = [x for x in n_options if x >= 400 and x % 2 == 0 and x != n_init]

    for n_try in n_options:
        if tleft() < 40:
            break
        target_try = int(n_try / 2)
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin = project_binary(h_interp, target_try)

        allowed = min(15, tleft() * 0.04)
        h_g2, c5_g2 = greedy_worst_shift(h_bin, 10000, time_limit=allowed, start_time=start_time)

        if tleft() < 25:
            if c5_g2 < best_score:
                best_score = c5_g2
                best_h = h_g2.copy()
            continue

        allowed2 = min(30, tleft() * 0.06)
        max_it2 = int(min(40000, allowed2 * 1500))
        h_sa3, c5_sa3 = sa_shift_aware(h_g2, target_try, max_it2, 0.006, 0.00002, rng,
                                       time_limit=allowed2, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()

    # Phase 5: ILS with diverse perturbations
    if tleft() > 80:
        for pert_iter in range(10):
            if tleft() < 25:
                break
            h_pert = best_h.copy()
            n_pert = 2 + pert_iter // 2
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_pert):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i] = 0
                h_pert[j] = 1
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]

            allowed = min(15, tleft() * 0.03)
            h_pg, c5_pg = greedy_worst_shift(h_pert, 8000, time_limit=allowed,
                                             start_time=start_time)

            if tleft() < 20:
                if c5_pg < best_score:
                    best_score = c5_pg
                    best_h = h_pg.copy()
                continue

            allowed2 = min(25, tleft() * 0.04)
            max_it4 = int(min(30000, allowed2 * 1500))
            h_ps, c5_ps = sa_shift_aware(h_pg, int(len(h_pg) / 2), max_it4,
                                         0.004, 0.00001, rng,
                                         time_limit=allowed2, start_time=start_time)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps.copy()

    # Phase 6: SLSQP continuous refinement with progressive alpha
    if tleft() > 50:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas = [20, 50, 150, 400, 1000, 3000, 8000, 20000, 50000]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 4000, 1e-15,
                                 time_limit=min(tleft() - 20, 250), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Quantized SLSQP (alternating continuous/binary)
    if tleft() > 60:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas_q = [500, 2000, 10000, 40000]
        h_q, sc_q = quantized_slsqp(best_h, n_use, target_use, alphas_q, 3000,
                                    num_cycles=3,
                                    time_limit=min(tleft() - 15, 200),
                                    start_time=start_time, rng=rng)
        if sc_q < best_score:
            best_score = sc_q
            best_h = h_q

    # Phase 8: Final binary SA polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = int(n_use / 2)
        h_bin_f = project_binary(best_h, target_use)
        allowed = min(40, tleft() - 10)
        max_it5 = int(min(80000, allowed * 1500))
        h_sf, c5_sf = sa_shift_aware(h_bin_f, target_use, max_it5, 0.002, 0.000005,
                                     rng, time_limit=allowed, start_time=start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final SLSQP polish
    if tleft() > 20:
        n_use = len(best_h)
        target_use = float(n_use) / 2.0
        alphas2 = [5000, 20000, 80000]
        h_ref2, sc2 = slsqp_refine(best_h, n_use, target_use, alphas2, 3000,
                                   1e-16, time_limit=max(5.0, tleft() - 5),
                                   start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Phase 10: Final greedy on the best
    if tleft() > 10:
        h_fg, c5_fg = greedy_worst_shift(project_binary(best_h, int(len(best_h) / 2)),
                                         5000, time_limit=max(3.0, tleft() - 3),
                                         start_time=start_time)
        if c5_fg < best_score:
            best_score = c5_fg
            best_h = h_fg

    # Final feasibility
    n_final = len(best_h)
    target_final = float(n_final) / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)

    return h_final, final_score, n_final
