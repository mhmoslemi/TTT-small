def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])
        for idx in range(min(need, len(ones))):
            h_r[ones[order[idx]]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])
        for idx in range(min(need, len(zeros))):
            h_r[zeros[order[idx]]] = 1
    # Fix any remaining discrepancy
    final_sum = int(h_r.sum())
    idx_list = list(range(n))
    while final_sum != target_sum:
        if final_sum > target_sum:
            for i in idx_list:
                if h_r[i] == 1:
                    h_r[i] = 0
                    final_sum -= 1
                    break
        else:
            for i in idx_list:
                if h_r[i] == 0:
                    h_r[i] = 1
                    final_sum += 1
                    break
    return h_r.astype(float)

def project_feasible(h, target_sum):

def smooth_obj(h, alpha, n, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    soft_max = c_max + float(np.sum(np.exp(shifted))) / alpha
    return soft_max * (2.0 / n)

_con_target = 0.0

def cons_fun(h):
    return np.sum(h) - _con_target

def cons_jac(h):
    return np.ones(_con_n)

def slsqp_refine(h_start, alphas, maxiter, ftol, time_limit, start_time):
    global _con_n, _con_target
    n = len(h_start)
    _con_n = n
    _con_target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, _con_target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, _con_target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def downsample_upsample(h, factor):
    """Downsample by averaging blocks, then upsample by replication."""
    n = len(h)
    n_small = n // factor
    h_small = np.zeros(n_small)
    for i in range(n_small):
        h_small[i] = np.mean(h[i*factor:(i+1)*factor])
    h_up = np.repeat(h_small, factor)
    # If lengths don't match, trim or pad
    if len(h_up) > n:
        h_up = h_up[:n]
    elif len(h_up) < n:
        h_up = np.pad(h_up, (0, n - len(h_up)), mode='edge')
    return h_small, h_up

        h_ms2, c5_ms2 = sa_restart_strategy(best_h, target, 4, rng, allowed, start_time)
        if c5_ms2 < best_score:
            best_score = c5_ms2
            best_h = h_ms2.copy()

    # Phase 7: SLSQP continuous refinement
    if tleft() > 60:
        alphas = [30, 80, 200, 500, 1200, 3000, 8000, 20000]
        allowed = min(tleft() - 20, 180)
        h_ref, sc = slsqp_refine(best_h, alphas, 2000, 1e-15, allowed, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 8: Binary SA on refined solution
    if tleft() > 40:
        h_bin = project_binary(best_h, target)
        allowed = min(tleft() - 10, 80)
        max_it = int(min(300000, allowed * 2500))
        h_sf, c5_sf = sa_multi_move(h_bin, target, max_it, 0.006, 0.00001,
                                     rng, allowed, start_time)
        if c5_sf < best_score:
            best_score = c5_sf
            best_h = h_sf

    # Phase 9: Final SLSQP polish
    if tleft() > 15:
        alphas2 = [5000, 15000, 40000]
        h_ref2, sc2 = slsqp_refine(best_h, alphas2, 1500, 1e-16,
                                    max(5.0, tleft() - 5), start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    h_final = project_feasible(best_h, float(n) / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, n
