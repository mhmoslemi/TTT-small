            return h
        h = h.copy()

    grad = np.zeros(n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < 2 * n - 1:
        top_idx = np.argsort(corr)[-top_k:]
        sub_corr = corr[top_idx]
    else:
        top_idx = np.arange(2 * n - 1)
        sub_corr = corr
    c_max = np.max(sub_corr)
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum
    grad = np.zeros(n)
    n_half = n - 1
    for idx_k, w in zip(top_idx, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def weighted_grad(h, p, dx):
    n = len(h)
    grad = np.zeros(n)
    n_half = n - 1
    for k_idx in range(len(p)):
        pk = p[k_idx]
        if pk < 1e-13:
            continue
        k = k_idx - n_half
        if k == 0:
            grad += pk * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += pk * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= pk * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += pk * (1.0 - h[m:]) * dx
                grad[m:] -= pk * h[:n - m] * dx
    return grad


def paired_optimize(h_init, n, target, time_limit, start_time, deadline_offset=30,
                    n_outer=500, n_inner=50, eta=1.0):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    n_shifts = 2 * n - 1
    p = np.ones(n_shifts) / n_shifts

    for outer in range(n_outer):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        p = p * np.exp(-eta * corr)
        p_sum = np.sum(p)
        if p_sum < 1e-300:
            p = np.ones(n_shifts) / n_shifts
        else:
            p = p / p_sum
        h_inner = h.copy()
        for inner in range(n_inner):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            grad = weighted_grad(h_inner, p, dx)
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-14:
                break
            lr = min(1.0 / (gnorm + 1e-10), 1.0)
            h_try = h_inner - lr * grad
            h_try = project_feasible(h_try, target)
            h_inner = h_try
        h = project_feasible(h_inner, target)
        sc = compute_c5(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def nesterov_proj_grad(h_init, n, target, time_limit, start_time, deadline_offset=30, top_k=50):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_score = compute_c5(h)
    alpha_list = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    for alpha in alpha_list:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        y = h.copy()
        stalled = 0
        for step in range(5000):
            if time.time() - start_time > time_limit - deadline_offset:
                break
            g = smooth_grad_topk(y, alpha, dx, top_k)
            gnorm = np.max(np.abs(g))
            if gnorm < 1e-13:
                break
            lr = 1.0 / (alpha * gnorm + 1e-10)
            lr = min(lr, 2.0 / (gnorm + 1e-10))
            improved = False
            for _ in range(30):
                h_try = y - lr * g
                h_try = project_feasible(h_try, target)
                score = compute_c5(h_try)
                if score < best_score - 1e-14:
                    best_score = score
                    best_h = h_try.copy()
                obj_old = smooth_obj_topk(y, alpha, top_k)
                obj_new = smooth_obj_topk(h_try, alpha, top_k)
                if obj_new < obj_old - 1e-15:
                    y = h_try
                    improved = True
                    stalled = 0
                    break
                lr *= 0.5
            if not improved:
                stalled += 1
                if stalled > 100:
                    break
        h = project_feasible(y, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def alternating_lp(h_init, n, target, max_iter, time_limit, start_time,
                   deadline_offset=30, damping=0.6, reg=1e-4):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_topk(h_start, n, target, alphas, top_k, maxiter, ftol,
               time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj_topk, h_cur, args=(alpha, top_k),
                           jac=lambda hh: smooth_grad_topk(hh, alpha, dx, top_k),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    cur_score = sc
                    h_cur = h_new
        except Exception:
            pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 10.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP
    if tleft() > 200:
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                     max_iter=200, time_limit=min(150, tleft() * 0.25),
                                     start_time=start_time, deadline_offset=120)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Paired optimization (saddle point)
    if tleft() > 200:
        h_po, c5_po = paired_optimize(best_h, n, target,
                                       time_limit=min(180, tleft() * 0.25),
                                       start_time=start_time, deadline_offset=120,
                                       n_outer=500, n_inner=30, eta=0.8)
        if c5_po < best_score:
            best_score = c5_po
            best_h = h_po

    # Phase 3: Nesterov projected gradient
    if tleft() > 200:
        h_ng, c5_ng = nesterov_proj_grad(best_h, n, target,
                                          time_limit=min(200, tleft() * 0.3),
                                          start_time=start_time, deadline_offset=120,
                                          top_k=40)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    # Phase 4: SLSQP with top-k
    if tleft() > 100:
        h_slq, sc = slsqp_topk(best_h, n, target,
                                [100, 500, 2000, 10000, 50000],
                                top_k=40, maxiter=3000, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=60)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: Another paired optimization round
    if tleft() > 100:
        h_po, c5_po = paired_optimize(best_h, n, target,
                                       time_limit=min(80, tleft() * 0.3),
                                       start_time=start_time, deadline_offset=60,
                                       n_outer=300, n_inner=20, eta=1.5)
        if c5_po < best_score:
            best_score = c5_po
            best_h = h_po

    # Phase 6: SLSQP final polish
    if tleft() > 60:
        h_slq, sc = slsqp_topk(best_h, n, target,
                                [5000, 20000, 100000],
                                top_k=80, maxiter=3000, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=30)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 7: Multi-start perturbations
    for noise in [0.01, 0.02, 0.04, 0.06]:
        if tleft() < 40:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_slq, sc = slsqp_topk(h_pert, n, target,
                                [500, 2000, 10000],
                                top_k=50, maxiter=2000, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=20)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 8: Higher resolution
    if tleft() > 150:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                         max_iter=80, time_limit=min(50, tleft() * 0.3),
                                         start_time=start_time, deadline_offset=70)
            if tleft() > 50:
                h_ng, c5_ng = nesterov_proj_grad(h_lp, n_try, target_try,
                                                  time_limit=min(40, tleft() * 0.3),
                                                  start_time=start_time, deadline_offset=50,
                                                  top_k=50)
                if c5_ng < c5_lp:
                    h_lp, c5_lp = h_ng, c5_ng
            if tleft() > 30:
                h_slq, sc = slsqp_topk(h_lp, n_try, target_try,
                                        [1000, 5000, 20000],
                                        top_k=50, maxiter=1500, ftol=1e-16,
                                        time_limit=budget_s, start_time=start_time,
                                        deadline_offset=20)
                if sc < c5_lp:
                    h_lp, c5_lp = h_slq, sc
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try

    # Phase 9: Final SLSQP
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin, sc = slsqp_topk(best_h, n_use, target_use,
                                [5000, 20000, 100000],
                                top_k=100, maxiter=2000, ftol=1e-16,
                                time_limit=budget_s, start_time=start_time,
                                deadline_offset=10)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
