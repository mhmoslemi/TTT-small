import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_full_lp(h, g, dx, n, target, n_max_lag=None, eps=1e-10):
    """Solve the LP with ALL relevant lags included."""
    if n_max_lag is None:
        n_max_lag = n - 5  # include all lags with overlap region >= 5 samples
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    
    n_half = n - 1
    for k in range(-n_max_lag, n_max_lag + 1):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi - lo < 5:  # skip very short overlaps
            continue
        coeff = np.zeros(n)
        coeff[lo:hi] = g[lo - k:hi - k] * dx
        constraints.append(coeff @ h_var <= t_var)
    
    objective = t_var
    prob = cp.Problem(cp.Minimize(objective), constraints)
    prob.solve(solver=cp.SCS, max_iters=100000, eps=eps, verbose=False)
    
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

def alternating_full_lp(h_init, n, target, max_iter, time_limit, 
                        damping=0.9, eps=1e-10, reg_decay=0.95):
    """Alternating LP with all lags included."""

    h = project_feasible(h, target)

        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n-5), n-5+1):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi - lo < 5:

            coeff = np.zeros(n)

            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        prob.solve(solver=cp.SCS, max_iters=100000, eps=eps, verbose=False)
        
        if h_var.value is None:

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        
        reg *= reg_decay
    
    return best_h, best_c5

    _smooth_n = n
    _smooth_target = target
    bounds = [(0.0, 1.0)] * n

    """Local search on binary solutions."""
    start = time.time()
    # Project to binary
    h_bin = (h_init >= 0.5).astype(float)
    # Fix the sum
    deficit = int(target) - int(h_bin.sum())

        flip = np.argsort(-h_init[zeros])[:deficit]
        h_bin[zeros[flip]] = 1

        flip = np.argsort(h_init[ones])[:(-deficit)]
        h_bin[ones[flip]] = 0

                        break
                if improved2:

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp_v2(h_init, n, target, max_iter, time_limit,
                      n_lags=350, eps=1e-10,
                      damping_start=0.90, damping_end=0.99,
                      reg_start=1e-3, reg_end=1e-10):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    stuck = 0
    n_half = n - 1
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if stuck > 400:
            break
        
        progress = min(1.0, it / max(1, max_iter))
        damping = damping_start + (damping_end - damping_start) * progress
        reg = reg_start * (reg_end / reg_start) ** progress
        
        g = 1.0 - h
        overlaps = compute_overlaps(h)
        top_idx = np.argsort(overlaps)[::-1][:n_lags]
        lags = top_idx - n_half
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in lags:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi - lo < 3:
                continue
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo - k:hi - k] * dx
            constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=eps, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
            except Exception:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
        # If stuck a lot, reduce damping temporarily
        if 100 < stuck < 300:
            damping = max(0.5, damping - 0.2)
    
    return best_h, best_c5

# SLSQP stuff
_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def _smooth_obj(h):
    n = _g_n
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c_vals = corr * dx
    c_max = np.max(c_vals)
    shifted = np.clip(_g_alpha * (c_vals - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _g_alpha)

def _cons_fun(h):
    return np.sum(h) - _g_target

def _cons_jac(h):
    return np.ones(_g_n)

def slsqp_polish(h_start, n, target, alphas, maxiter=2000):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        _g_alpha = alpha
        try:
            res = minimize(_smooth_obj, h_cur, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
