def smooth_objective(h, alpha):
    """Log-sum-exp approximation to max of correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    # For numerical stability
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_objective_grad(h, alpha):
    """Gradient of the log-sum-exp approximation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of c_k w.r.t. h[j]:
    # c_k = dx * sum_i h[i] * (1-h[i+k])
    # dc_k/dh[j] = dx * (1-h[j+k]) - dx * h[j-k] (where indices are valid)
    
    grad = np.zeros(n)
    for lag_offset in range(2*n-1):
        k = lag_offset - (n-1)  # actual lag
        w = weights[lag_offset]
        if w < 1e-10:

        # dc_k/dh[j] for each j
        # Term 1: h[j] appears as h[i] where i=j, so (1-h[j+k])*dx

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def full_corr_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = (1.0 - h)[::-1]
    result = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return result[:2*n-1][::-1]

def compute_delta(h, i, j):
    n = len(h)
    result = np.zeros(2*n-1)
    k = np.arange(2*n-1) - (n-1)
    idx = i + k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = i - k
    mask = (idx >= 0) & (idx < n)
    result[mask] += h[idx[mask]]
    idx = j - k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    idx = j + k
    mask = (idx >= 0) & (idx < n)
    result[mask] -= h[idx[mask]]
    return result

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(np.clip(h, 0, 1))
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        ones = np.where(h_r == 1)[0]
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        zeros = np.where(h_r == 0)[0]
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum != target_sum:
        if final_sum > target_sum:
            diff = final_sum - target_sum
            ones = np.where(h_r == 1)[0]
            h_r[ones[:diff]] = 0
        else:
            diff = target_sum - final_sum
            zeros = np.where(h_r == 0)[0]
            h_r[zeros[:diff]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fast(h) * dx
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fast(h) * dx
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    active = np.where(w > 1e-10)[0]
    grad = np.zeros(n)
    for k_off in active:
        k = int(k_off) - (n - 1)
        wk = w[k_off]
        if k >= 0:
            j_max = n - k
            if j_max > 0:
                grad[:j_max] += wk * (1.0 - h[k:n])
            if k > 0:
                grad[k:] -= wk * h[:j_max]
            else:
                grad[k:] -= wk * h[:j_max]
        else:
            kp = -k
            j_max = n - kp
            if j_max > 0:
                grad[kp:] += wk * (1.0 - h[:j_max])
                grad[:j_max] -= wk * h[kp:]
    grad *= dx
    return grad

def continuous_slsqp(h_start, target, alpha_list, maxiter_list, time_limit, start_time):
    n = len(h_start)
    h = h_start.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alpha_list):
        if time.time() - start_time > time_limit:
            break
        mi = maxiter_list[i] if i < len(maxiter_list) else maxiter_list[-1]
        try:
            res = minimize(smooth_obj, h, args=(alpha,), jac=lambda x: smooth_grad(x, alpha),
                          method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                       'jac': lambda x: np.ones(n)}],
                          options={'maxiter': mi, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score
