def compute_grad_lse(h, alpha, n_active=None):
    """Compute gradient of LSE objective using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    if n_active is not None and n_active < n:
        idx = np.argsort(corr)[::-1][:n_active]
        corr_f = corr[idx]
        c_max = float(np.max(corr_f))
        shifted = np.clip(alpha * (corr_f - c_max), -700.0, 0.0)
        exp_v = np.exp(shifted)
        w_f = exp_v / exp_v.sum()
        weights = np.zeros(n)
        for wi, ki in zip(w_f, idx):
            weights[int(ki)] = wi
    else:
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        exp_v = np.exp(shifted)
        weights = exp_v / exp_v.sum()
    
    # Gradient computation
    fsize = 2 * n
    w_flipped = weights[::-1].copy()
    Wf = np.fft.rfft(w_flipped, fsize)
    V = np.fft.rfft(1.0 - h, fsize)
    W = np.fft.rfft(weights, fsize)
    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)
    corr_neg = np.fft.irfft(W * H, fsize)
    grad = dx * (corr_pos[n-1:2*n-1] - corr_neg[:n])
    return grad

def adaptive_level(h, target, deadline, max_rounds=5000):
    """Adaptively level the top lags with Nesterov momentum."""
    n = len(h)
    h_cur = h.copy()
    h_mom = h.copy()
    best = compute_c5(h_cur)
    h_best = h_cur.copy()
    t = 1.0
    lr = 0.02
    
    for it in range(max_rounds):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Evaluate at momentum point
        h_eval = np.clip(h_mom, 0.0, 1.0)
        h_eval = project_feasible(h_eval, target)
        
        corr = get_corr(h_eval)
        c_max = float(np.max(corr))
        
        # Focus on top lags with soft assignment
        sorted_corr = np.sort(corr)[::-1]
        gap = c_max - sorted_corr[min(10, n-1)]  # gap to 10th lag
        temp = max(gap * 0.5, c_max * 0.001)
        shifted = (corr - c_max) / temp
        weights = np.exp(np.clip(shifted, -20, 0))
        weights /= weights.sum()
        
        grad = weighted_grad(h_eval, weights)
        
        h_new = project_feasible(h_eval - lr * grad, target)
        h_mom = h_new + ((t - 1.0) / t_new) * (h_new - h_cur)
        h_cur = h_new
        t = t_new
        
        sc = compute_c5(h_cur)
        if sc < best:
            best = sc
            h_best = h_cur.copy()
    
    return h_best, best

def coord_polish(h, target, deadline, max_sweeps=200):
    n = len(h)
    h_cur = h.copy()
    best = compute_c5(h_cur)
    rng = np.random.RandomState(42)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        any_improve = False
        for idx in order:
            if time.time() > deadline:
                break
            old = h_cur[idx]
            if old < 1e-9:
                trials = [0.05, 0.1, 0.2]
            elif old > 1.0 - 1e-9:
                trials = [0.8, 0.9, 0.95]
            else:
                trials = [old*0.5, old*1.5, min(old+0.2, 1.0), max(old-0.2, 0.0), 0.5]
            found = False
            for tv in trials:
                tv = float(min(max(tv, 0.0), 1.0))
                if abs(tv - old) < 1e-12:
                    continue
                h_cur[idx] = tv
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best - 1e-14:
                    best = sc
                    any_improve = True
                    found = True
                    break
            if not found:
                h_cur[idx] = old
                h_cur = project_feasible(h_cur, target)
        if not any_improve:
            break
    
    return h_cur, best
