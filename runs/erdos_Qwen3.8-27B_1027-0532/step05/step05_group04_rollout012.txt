    # First term: for each j, sum_{k=0}^{n-1-j} w_k * (1-h[j+k])

    # Actually it's: (w ⊛ (1-h))[j] where ⊛ is cross-correlation, but only the part where both indices are valid
    
    one_minus_h = 1.0 - h
    # Forward: corr_fwd[j] = sum_{k=0}^{n-1-j} w[k] * one_minus_h[j+k]
    # This is like: for each j, dot product of w[:n-j] with one_minus_h[j:n]

    # zero-pad w to size 2n, zero-pad one_minus_h to size 2n
    # circular correlation gives us what we want

    H = np.zeros(2*n)
    W[:n] = w
    H[:n] = one_minus_h
    Wf = np.fft.rfft(W)

    # Hmm, this is getting complicated. Let me just do it directly for n=400.

import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_weights = None


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h


def compute_grad(h, weights, dx):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = dx * (np.dot(weights[:n - j], omh[j:n]) - np.dot(weights[:j + 1], h[j::-1]))
    return grad

    return np.sum(x) - _g_target

    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_one(h, n, target, alpha, maxiter, deadline):

    _g_n = n
    _g_target = target
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_lse, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)

    except Exception:
        return h
