If I swap h[i]: 1→0 and h[j]: 0→1, the change in correlation at lag k is:
- For the i position (going from 1 to 0): 
  - At lag k: change = -(1-h[i+k]) if i+k is valid, and -h[i+k'] if i-k' is valid (for the reverse direction)

Now let me also think about a key insight: for the binary case, C₅ = max_k c_k * dx where c_k = number of i with h[i]=1 and h[i+k]=0 (for k>0) or h[i]=1 and h[i-k]=0 (for k<0, with appropriate range).

Actually, for the 'full' correlate:
- Index 3+k (for n=4) gives ∑_i h[i]*(1-h[i+k]) for valid i
- This counts 1→0 transitions at distance k to the RIGHT

For k<0, say k=-m:
- Index 3-m gives ∑_i h[i]*(1-h[i-m]) for valid i (i≥m)
- This counts 1→0 transitions at distance m to the LEFT

So the max is over both left and right transitions.

For a binary set A = {i : h[i]=1}, the value at lag k>0 is |A \ (A-k)| restricted to valid range = number of elements in A that are k positions from the left of a non-element.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_grad(h, alpha, dx, top_m=None):

    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):

        k = idx_k - n_half

                grad[m:] -= w * h[:n - m] * dx
    return grad

def nesterov_pgd(h_init, n, target, time_limit, start_time, deadline_offset=30):

    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    y = h.copy()  # Nesterov momentum variable
    best_h = h.copy()
    best_score = compute_c5(h)
    
    alpha_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    top_m_schedule = [20, 30, 50, 80, 100, 150, 200, 300, 2*n-1, 2*n-1]
    steps_per_alpha = 2000
    
    for ai, alpha in enumerate(alpha_schedule):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_schedule[min(ai, len(top_m_schedule)-1)]
        
        # Nesterov parameters
        for step in range(steps_per_alpha):
            if time.time() - start_time > time.limit - deadline_offset:
                break
            
            t_k = (1 + np.sqrt(1 + 4*(step+1)**2)) / 2  # Nesterov t
            beta_nest = (t_k - 1) / (t_k + 1)
            
            # Compute gradient at y
            grad_y = smooth_grad(y, alpha, dx, top_m=top_m)
            
            # Step
            lr = 1.0 / (alpha * np.max(np.abs(grad_y)) + 1e-10)
            lr = min(lr, 1.0)

            h_new = project_feasible(h_new, target)

            y = np.clip(y, 0.0, 1.0)

                best_h = h.copy()

def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target

The gradient is: ∇f = ∑_k w_k ∇c_k where w_k = exp(alpha*(c_k - c_max)) / ∑_j exp(alpha*(c_j - c_max))

The Hessian is: ∇²f = alpha * (∑_k w_k ∇c_k ∇c_k^T - (∑_k w_k ∇c_k)(∑_k w_k ∇c_k)^T)

The norm of the Hessian is bounded by alpha * ||∇c_k||_max² ≤ alpha * (2*dx)² = 4*alpha*dx².

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad(h, alpha, dx, top_m=None):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
        sub_corr = corr[top_indices]
    else:
        top_indices = np.arange(2 * n - 1)
        sub_corr = corr
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        return np.zeros(n)
    weights = exp_shifted / wsum

    grad = np.zeros(n)
    n_half = n - 1

    for idx_k, w in zip(top_indices, weights):
        if w < 1e-14:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n - k]) * dx
                grad[:n - k] -= w * h[k:] * dx
        else:
            m = -k
            if n - m > 0:
                grad[:n - m] += w * (1.0 - h[m:]) * dx
                grad[m:] -= w * h[:n - m] * dx
    return grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def nesterov_pgd(h_init, n, target, time_limit, start_time, deadline_offset=30):
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    y = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)

    alphas = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    top_m_sched = [20, 30, 50, 80, 100, 150, 200, 300, 2 * n - 1, 2 * n - 1]
    steps_per_alpha = 3000

    for ai, alpha in enumerate(alphas):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        top_m = top_m_sched[min(ai, len(top_m_sched) - 1)]
        L_bound = 4.0 * alpha * dx * dx + 1e-10

        for step in range(steps_per_alpha):
            if time.time() - start_time > time_limit - deadline_offset:
                break

            t_k = (1.0 + np.sqrt(1.0 + 4.0 * (step + 1) * (step + 1))) / 2.0
            beta_nest = (t_k - 1.0) / (t_k + 1.0)

            grad_y = smooth_grad(y, alpha, dx, top_m=top_m)
            gnorm = np.max(np.abs(grad_y))
            if gnorm < 1e-14:
                break

            lr = 1.0 / (L_bound + 1e-10)
            lr = min(lr, 2.0)

            h_new = y - lr * grad_y
            h_new = project_feasible(h_new, target)
            y = h_new + beta_nest * (h_new - h)
            y = np.clip(y, 0.0, 1.0)
            h = h_new

            score = compute_c5(h)
            if score < best_score - 1e-15:
                best_score = score
                best_h = h.copy()

    return best_h, best_score


def binary_sa(h_start, n, target, time_limit, start_time, deadline_offset=30, seed_val=42):
    rng = np.random.RandomState(seed_val)
    dx = 2.0 / n
    n_target = int(target)

    h_bin = np.zeros(n)
    sorted_idx = np.argsort(-h_start)
    h_bin[sorted_idx[:n_target]] = 1.0

    c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    best_c5 = c5
    cur_c5 = c5

    T = 0.02 * dx
    T_min = 1e-9
    cooling = 0.99998
    max_iter = 20000000

    for it in range(max_iter):
        if time.time() - start_time > time_limit - deadline_offset:
            break
        if T < T_min:
            break

        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0

        T = max(T * cooling, T_min)

    return best_h, best_c5


def slsqp_optimize(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time, deadline_offset=20):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    cur_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit - deadline_offset:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                sc = compute_c5(h_new)
                if sc < cur_score - 1e-15:
                    h_cur = h_new
                    cur_score = sc
        except Exception:
            pass

    return h_cur, compute_c5(h_cur)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def sharpen(h, beta, target):
    h_sharp = 1.0 / (1.0 + np.exp(-beta * (2.0 * h - 1.0)))
    return project_feasible(h_sharp, target)
