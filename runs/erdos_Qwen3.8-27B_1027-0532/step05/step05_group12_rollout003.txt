from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha, dx, top_m=None):
    """Compute smooth max objective and its gradient simultaneously."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    if top_m is not None and top_m < 2 * n - 1:
        top_indices = np.argsort(corr)[-top_m:]
    else:
        top_indices = np.arange(2 * n - 1)
    
    sub_corr = corr[top_indices]
    shifted = np.clip(alpha * (sub_corr - c_max), -600.0, 0.0)
    exp_shifted = np.exp(shifted)
    wsum = np.sum(exp_shifted)
    if wsum < 1e-300:
        obj = c_max * dx
        return obj, np.zeros(n)
    weights = exp_shifted / wsum
    
    obj = (c_max + np.log(wsum) / alpha) * dx
    
    # Compute gradient
    grad = np.zeros(n)
    n_half = n - 1
    
    for idx_k, w in zip(top_indices, weights):
        if w < 1e-15:
            continue
        k = idx_k - n_half
        if k == 0:
            grad += w * (1.0 - 2.0 * h) * dx
        elif k > 0:
            m = n - k
            if m > 0:
                grad[k:] += w * (1.0 - h[:m]) * dx
                grad[:m] -= w * h[k:] * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            return h
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        n_free = np.sum(mask)
        if n_free == 0:
            return h
        h = h.copy()
        h[mask] += diff / n_free
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_val(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -600.0, 0.0)
    wsum = np.sum(np.exp(shifted))
    return float(c_max + np.log(wsum) / alpha)


def smooth_grad_full(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

        if time.time() - start_time > time_limit - deadline_offset:
            break
        n_steps = min(steps_per_alpha[min(ai, len(steps_per_alpha) - 1)], 20000)
        h_prev = h.copy()
        momentum = np.zeros(n)
        beta = 0.9
        stalled = 0
        lr = 1.0 / (alpha * 0.01 + 10.0)

                shifted = np.clip(a * (corr - cm), -600.0, 0.0)
                return float(cm + np.log(np.sum(np.exp(shifted))) / a)

            def obj_grad(x, a=alpha, dx=2.0 / n):
                corr = np.correlate(x, 1.0 - x, mode='full')
                cm = np.max(corr)
                shifted = np.clip(a * (corr - cm), -600.0, 0.0)
                exp_sh = np.exp(shifted)
                ws = np.sum(exp_sh)
                if ws < 1e-300:
                    return np.zeros(len(x))
                weights = exp_sh / ws
                g = np.zeros(len(x))
                nh = len(x) - 1
                for idx_k in range(2 * len(x) - 1):
                    w = weights[idx_k]
                    if w < 1e-14:
                        continue
                    k = idx_k - nh
                    if k == 0:
                        g += w * (1.0 - 2.0 * x) * dx
                    elif k > 0:
                        m = len(x) - k
                        g[k:] += w * (1.0 - x[:m]) * dx
                        g[:m] -= w * x[k:] * dx
                    else:
                        m = -k
                        g[:len(x) - m] += w * (1.0 - x[m:]) * dx
                        g[m:] -= w * x[:len(x) - m] * dx
                return g

            res = minimize(obj_func, h_cur, jac=obj_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(np.asarray(res.x).ravel(), target)
                if compute_c5(h_new) < compute_c5(h_cur) - 1e-15:
                    h_cur = h_new
        except Exception:
            pass

    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Nesterov PG with smooth max - main optimization
    if tleft() > 300:
        pg_limit = min(700, tleft() * 0.6)
        h_pg, c5_pg = nesterov_pg(best_h, n, target,
                                  time_limit=pg_limit,
                                  start_time=start_time,
                                  deadline_offset=budget_s - pg_limit)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 2: SLSQP polish at high alpha
    if tleft() > 150:
        h_slq = slsqp_optimize(best_h, n, target,
                               alphas=[200, 1000, 5000, 20000, 50000],
                               maxiter=2000, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=budget_s * 0.3)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 3: Random restarts from best solution
    for noise in [0.005, 0.01, 0.02, 0.03, 0.05]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        pg_limit = min(120, tleft() * 0.4)
        h_pg, c5_pg = nesterov_pg(h_pert, n, target,
                                  time_limit=pg_limit,
                                  start_time=start_time,
                                  deadline_offset=budget_s - pg_limit)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 4: SLSQP after restarts
    if tleft() > 100:
        h_slq = slsqp_optimize(best_h, n, target,
                               alphas=[1000, 5000, 20000, 50000, 100000],
                               maxiter=1500, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=60)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    # Phase 5: Try higher resolution
    if tleft() > 250:
        for n_try in [500, 600, 800]:
            if tleft() < 150:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            c5_interp = compute_c5(h_interp)
            pg_limit = min(150, tleft() * 0.5)
            h_pg, c5_pg = nesterov_pg(h_interp, n_try, target_try,
                                      time_limit=pg_limit,
                                      start_time=start_time,
                                      deadline_offset=budget_s - pg_limit)
            if tleft() > 80:
                h_slq = slsqp_optimize(h_pg, n_try, target_try,
                                       alphas=[500, 2000, 10000, 50000],
                                       maxiter=1000, ftol=1e-16,
                                       time_limit=budget_s, start_time=start_time,
                                       deadline_offset=40)
                c5_pg = compute_c5(h_slq)
                h_pg = h_slq
            if c5_pg < best_score:
                best_score = c5_pg
                best_h = h_pg
                n = n_try
                target = target_try

    # Phase 6: Final Nesterov PG
    if tleft() > 150:
        n_use = len(best_h)
        target_use = n_use / 2.0
        pg_limit = min(200, tleft() * 0.6)
        h_pg, c5_pg = nesterov_pg(best_h, n_use, target_use,
                                  time_limit=pg_limit,
                                  start_time=start_time,
                                  deadline_offset=budget_s - pg_limit)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 7: Final SLSQP polish
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_optimize(best_h, n_use, target_use,
                               alphas=[5000, 20000, 50000, 100000, 200000],
                               maxiter=2000, ftol=1e-16,
                               time_limit=budget_s, start_time=start_time,
                               deadline_offset=10)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin

    # Final feasibility check
    n_use = len(best_h)
    h_final = project_feasible(best_h, n_use / 2.0)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
