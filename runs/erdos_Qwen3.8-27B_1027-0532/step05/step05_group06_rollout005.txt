import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
    grad += 2.0 * mu * (s - target)
    return val, grad

def gradient_descent_momentum(h, target, alpha, mu, max_iter, deadline, lr=None, momentum=0.9):
    """Nesterov-accelerated gradient descent with projection"""
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    if lr is None:
        lr = 0.01 / (alpha + 1)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = h_cur + momentum * (h_cur - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = lse_obj_and_grad(y, alpha, target, mu)
        h_new = h_cur - lr * grad
        h_new = project_feasible(h_new, target)
        
        # Adaptive step size
        sc_new = compute_c5(h_new)
        sc_cur = compute_c5(h_cur)
        if sc_new < sc_cur:
            h_prev = h_cur
            h_cur = h_new
            if sc_new < best_score:
                best_score = sc_new
                best_h = h_cur.copy()
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7
            if lr < 1e-10:
                break
    return best_h, best_score

def binary_local_search(h, target, n_swaps=5000, deadline=None):
    """Local search on the binary version of h"""
    n = len(h)
    # Round to binary
    h_bin = (h >= 0.5).astype(float)
    # Fix the number of 1s
    n_ones = int(np.sum(h_bin))
    n_target = int(target)
    if n_ones > n_target:
        ones = np.where(h_bin == 1)[0]
        # Remove the ones with lowest h values
        order = np.argsort(h[ones])[::-1]
        for idx in order[n_target:]:
            h_bin[ones[idx]] = 0
    elif n_ones < n_target:
        zeros = np.where(h_bin == 0)[0]
        # Add the zeros with highest h values
        order = np.argsort(h[zeros])[::-1]
        for idx in order[:n_target - n_ones]:
            h_bin[zeros[idx]] = 1
    
    cur_score = compute_c5(h_bin)
    
    for swap in range(n_swaps):
        if deadline and time.time() > deadline:
            break
        # Try a random swap
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h_bin[i] = 0
        h_bin[j] = 1
        new_score = compute_c5(h_bin)
        if new_score > cur_score:
            # Revert
            h_bin[i] = 1
            h_bin[j] = 0
        else:

def top_lag_equalize(h, target, n_top=5, n_steps=200, deadline=None):

    lr = 0.005

        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        top_vals = corr[top_idx]
        top_mean = top_vals.mean()
        top_std = top_vals.std()
        if top_std < 1e-12:
            break
        grad = np.zeros(n)
        for ki in top_idx:
            w = (corr[ki] - top_mean) / (top_std + 1e-15)
            lag = int(ki) - (n - 1)
            if lag >= 0:
                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h_cur[j1 - lag])
                if lag > 0:
                    j2 = np.arange(0, n - lag)
                    grad[j2] -= w * h_cur[j2 + lag]

                m = -lag
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h_cur[j1 + m])

                    j2 = np.arange(m, n)

                    h_cur = project_feasible(h_cur, target)

                    fd = float(np.max(np.correlate(h_cur, 1.0 - h_cur, mode='full')))

def downsample(h, n_new):
    """Downsample h to n_new points"""
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def upsample(h, n_new):

    h_new = np.interp(x_new, x_old, h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def lse_obj_and_grad(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    val = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum > 1e-300:
        weights = weights / wsum
    else:
        weights = np.ones(len(corr)) / len(corr)
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for ki in active:
        w = weights[ki]
        lag = int(ki) - (n - 1)
        if lag >= 0:
            j1 = np.arange(lag, n)
            grad[j1] += w * (1.0 - x[j1 - lag])
            if lag > 0:
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * x[j2 + lag]
        else:
            m = -lag
            j1 = np.arange(0, n - m)
            grad[j1] += w * (1.0 - x[j1 + m])
            if m > 0:
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]

            method='L-BFGS-B',
