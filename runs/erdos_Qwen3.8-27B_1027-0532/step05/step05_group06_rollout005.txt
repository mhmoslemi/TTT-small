def project_feasible(h, target_sum):

        else:

        if cnt == 0:
            break

def coordinate_descent_1d(h, target, n_sweeps=3, deadline=None):

                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')

            fd = f1d(d)
            for _ in range(35):
                if fc < fd:
                    b = d; d = c; fd = fc
                    c = b - gr * (b - a); fc = f1d(c)

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def obj_grad_lse(x, alpha, target, mu):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    log_sum = float(np.log(exp_s.sum()))
    val = c_max + log_sum / alpha
    s = float(np.sum(x))
    val += mu * (s - target) ** 2
    wsum = float(exp_s.sum())
    if wsum > 1e-300:
        w = exp_s / wsum
    else:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        wk = w[ki]
        if wk < 1e-12:
            continue
        lag = ki - (n - 1)
        if lag >= 0:
            if lag > 0:
                i1 = np.arange(0, n - lag)
                grad[i1] += wk * (1.0 - x[i1 + lag])
                i2 = np.arange(lag, n)
                grad[i2] -= wk * x[i2 - lag]
            else:
                grad[:] += wk * (1.0 - x)
        else:
            m = -lag
            i1 = np.arange(m, n)
            grad[i1] += wk * (1.0 - x[i1 - m])
            i2 = np.arange(0, n - m)
            grad[i2] -= wk * x[i2 + m]
    grad += 2.0 * mu * (s - target)
    return val, grad


def lbfgs_optimize(h, target, alpha, mu, maxiter, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def f(x):
        return obj_grad_lse(x, alpha, target, mu)[0]

    def g(x):
        return obj_grad_lse(x, alpha, target, mu)[1]

    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-22,
                                'gtol': 1e-15, 'maxls': 100, 'maxcor': 60})
        return project_feasible(res.x, target)
    except Exception:
        return h.copy()


def top_lag_push(h, target, n_top=8, n_steps=250, deadline=None):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    lr = 0.004
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        sidx = np.argsort(corr)[::-1]
        top_idx = sidx[:n_top]
        top_vals = corr[top_idx]
        max_tv = top_vals.max()
        wv = np.exp(80.0 * (top_vals - max_tv))
        wv /= wv.sum()
        grad = np.zeros(n)
        for i in range(n_top):
            ki = int(top_idx[i])
            wk = wv[i]
            lag = ki - (n - 1)
            if lag >= 0:
                if lag > 0:
                    i1 = np.arange(0, n - lag)
                    grad[i1] += wk * (1.0 - h_cur[i1 + lag])
                    i2 = np.arange(lag, n)
                    grad[i2] -= wk * h_cur[i2 - lag]
                else:
                    grad[:] += wk * (1.0 - h_cur)
            else:
                m = -lag
                i1 = np.arange(m, n)
                grad[i1] += wk * (1.0 - h_cur[i1 - m])
                i2 = np.arange(0, n - m)
                grad[i2] -= wk * h_cur[i2 + m]

            break
        improved = False
        for j in range(n):
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d = a + gr * (b - a)

            def f1d(v):
                h_new = h_cur.copy()
                h_new[j] = v
                h_new = project_feasible(h_new, target)
                corr = np.correlate(h_new, 1.0 - h_new, mode='full')
                return float(np.max(corr))

            fc = f1d(c)
            fd = f1d(d)
            for _ in range(25):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - gr * (b - a)
                    fc = f1d(c)

                    a = c
                    c = d
                    fc = fd
                    d = a + gr * (b - a)

                h_cur = project_feasible(h_cur, target)
