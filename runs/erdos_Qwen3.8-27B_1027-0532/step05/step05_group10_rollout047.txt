def overlap_at_k(h, k_list):
    """Compute overlap integral at specific shifts."""
    n = len(h)
    dx = 2.0 / n
    half_n = n - 1
    results = []
    for k in k_list:
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = np.arange(lo - k, hi - k)
            val = np.sum(h[idx1] * (1.0 - h[idx2])) * dx
        else:
            val = 0.0
        results.append(val)
    return np.array(results)

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h.copy()
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lp_constraint_gen(h, n, target, max_iter, time_limit, n_active=60, 
                      damping=0.85, reg=1e-4, reg_decay=0.95):

    start_t = time.time()
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    current_reg = reg
    
    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0

    """Objective for SLSQP: smooth max over subset of k values."""
    alpha = _global_alpha
    k_list = _global_k_indices
    n = _global_n
    dx = 2.0 / n
    values = np.zeros(len(k_list))
    for j, k in enumerate(k_list):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            values[j] = np.sum(x[idx1] * (1.0 - x[idx2])) * dx
        else:
            values[j] = 0.0
    max_val = np.max(values)
    shifted = np.clip(alpha * (values - max_val), -400.0, 0.0)
    return float(max_val + np.log(np.sum(np.exp(shifted))) / alpha)

def prepare_grad(x):
    """Gradient of smooth max for SLSQP."""
    alpha = _global_alpha
    k_list = _global_k_indices
    n = _global_n
    dx = 2.0 / n
    values = np.zeros(len(k_list))
    for j, k in enumerate(k_list):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            values[j] = np.sum(x[idx1] * (1.0 - x[idx2])) * dx
        else:
            values[j] = 0.0
    max_val = np.max(values)
    shifted = np.clip(alpha * (values - max_val), -400.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for j, k in enumerate(k_list):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            dval = np.zeros(n)
            dval[idx1] += (1.0 - x[idx2]) * dx
            dval[idx2] -= x[idx1] * dx
            grad += weights[j] * dval
    
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        h = h.copy()
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lp_constraint_gen(h, n, target, max_iter, time_limit, n_active=60,
                      damping=0.85, reg=1e-4, reg_decay=0.95, add_violated=True):
    start_t = time.time()
    dx = 2.0 / n
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    half_n = n - 1
    current_reg = reg
    t_prev = best_c5

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        corr = np.correlate(h, g, mode='full')
        active_indices = np.argsort(corr)[-n_active:]

        for k_idx in active_indices:
            k = k_idx - half_n
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        if add_violated:
            t_val = t_prev
            for k_idx in range(1, len(corr) - 1):
                if k_idx not in active_indices and corr[k_idx] * dx > t_val + 1e-5:
                    k = k_idx - half_n
                    lo = max(0, k)
                    hi = min(n, k + n)
                    if hi > lo:
                        coeff = np.zeros(n)
                        coeff[lo:hi] = g[lo - k:hi - k] * dx
                        constraints.append(coeff @ h_var <= t_var)
                        if len(constraints) > n_active + 20:
                            break

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        solved = False
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            solved = h_var.value is not None
        except Exception:
            pass
        if not solved:

                solved = h_var.value is not None

def prepare_obj(x):

    k_list = _global_k_indices
    n = _global_n
    dx = 2.0 / n
    m = len(k_list)
    values = np.zeros(m)

        k = k_list[j]
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            values[j] = np.sum(x[idx1] * (1.0 - x[idx2])) * dx
    max_val = np.max(values)
    shifted = np.clip(alpha * (values - max_val), -400.0, 0.0)
    return float(max_val + np.log(np.sum(np.exp(shifted))) / alpha)


def prepare_grad(x):
    alpha = _global_alpha
    k_list = _global_k_indices
    n = _global_n
    dx = 2.0 / n
    m = len(k_list)
    values = np.zeros(m)
    for j in range(m):
        k = k_list[j]
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            values[j] = np.sum(x[idx1] * (1.0 - x[idx2])) * dx
    max_val = np.max(values)
    shifted = np.clip(alpha * (values - max_val), -400.0, 0.0)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)

    grad = np.zeros(n)
    for j in range(m):
        k = k_list[j]
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            idx1 = np.arange(lo, hi)
            idx2 = idx1 - k
            dval = np.zeros(n)
            dval[idx1] += (1.0 - x[idx2]) * dx
            dval[idx2] -= x[idx1] * dx
            grad += weights[j] * dval

    return grad

            best_h = h_bin
        if tleft() > 50:
            k_list_b = identify_active_k(best_h, n_work, n_active=20)
            h_ref_b = slsqp_active_set(best_h, n_work, target_work, k_list_b,
                                        [500, 2000, 8000], 1000, 1e-15)
            sc_b = compute_c5(h_ref_b)
            if sc_b < best_score:
                best_score = sc_b
                best_h = h_ref_b

    # Phase 7: Subgradient
    if tleft() > 60:
        h_sg, c5_sg = subgradient_improve(best_h, n_work, target_work,
                                           max_steps=3000, time_limit=min(35, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Final LP polish
    if tleft() > 70:
        lp_time = min(50, tleft() * 0.6)
        h_lp, c5_lp = lp_constraint_gen(best_h, n_work, target_work,
                                         max_iter=3000, time_limit=lp_time,
                                         n_active=100, damping=0.95, reg=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: Final active-set SLSQP
    if tleft() > 50:
        k_list = identify_active_k(best_h, n_work, n_active=30)
        h_ref = slsqp_active_set(best_h, n_work, target_work, k_list,
                                  [2000, 8000, 30000], 1200, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 10: Full smooth SLSQP
    if tleft() > 40:
        h_ref = slsqp_full(best_h, n_work, target_work,
                           [2000, 10000], 500, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 11: Try also at n=400
    if n_work != 400 and tleft() > 30:
        h_400 = interpolate_to(best_h, 400, 200.0)
        sc_400 = compute_c5(h_400)
        if sc_400 < best_score:
            best_score = sc_400
            best_h = h_400
            n_work = 400
            target_work = 200.0
            if tleft() > 25:
                k_list_400 = identify_active_k(best_h, 400, n_active=20)
                h_ref_400 = slsqp_active_set(best_h, 400, 200.0, k_list_400,
                                              [1000, 5000], 800, 1e-15)
                sc_400b = compute_c5(h_ref_400)
                if sc_400b < best_score:
                    best_score = sc_400b
                    best_h = h_ref_400

    # Final feasibility
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
