def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        if abs(diff) < 1e-15:
            break

        else:

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

def lse_gradient(x):

    nl1 = 2 * n - 1

    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)

    sw = np.sum(weights)

        return np.zeros(n)
    weights = weights / sw * _g_alpha
    
    grad = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 0.0
_g_target = 0.0
_g_penalty = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(3000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse_val = c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha
    cv = np.sum(x) - _g_target
    return lse_val + _g_penalty * cv * cv


def lse_grad(x):
    n = len(x)
    nl1 = 2 * n - 1
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    base_grad = np.zeros(n)
    if sw > 1e-300:
        weights = _g_alpha * weights / sw
        one_minus_h = 1.0 - x
        for k in range(nl1):
            w_val = weights[k]
            if w_val < 1e-12:
                continue
            m = k - (n - 1)
            if m >= 0:
                base_grad[0:n - m] += w_val * one_minus_h[m:n]
                base_grad[m:n] -= w_val * x[0:n - m]
            else:
                mm = -m
                base_grad[mm:n] += w_val * one_minus_h[0:n - mm]
                base_grad[0:n - mm] -= w_val * x[mm:n]
    cv = np.sum(x) - _g_target
    base_grad += 2.0 * _g_penalty * cv
    return base_grad


def compute_C_pair(h_cur, i, j, n):
    ms = np.arange(-(n - 1), n)
    im = i + ms
    jm = j + ms
    im_neg = i - ms
    jm_neg = j - ms
    valid_im = (im >= 0) & (im < n)
    valid_jm = (jm >= 0) & (jm < n)
    valid_im_neg = (im_neg >= 0) & (im_neg < n)
    valid_jm_neg = (jm_neg >= 0) & (jm_neg < n)
    vi = np.zeros(2 * n - 1)
    vi[valid_im] = h_cur[im[valid_im]]
    vj = np.zeros(2 * n - 1)
    vj[valid_jm] = h_cur[jm[valid_jm]]
    vi_neg = np.zeros(2 * n - 1)
    vi_neg[valid_im_neg] = h_cur[im_neg[valid_im_neg]]
    vj_neg = np.zeros(2 * n - 1)
    vj_neg[valid_jm_neg] = h_cur[jm_neg[valid_jm_neg]]
    return vj - vi - vi_neg + vj_neg


def solve_1d_max(A, C, t_min, t_max):
    pos = C > 1e-14
    neg = C < -1e-14
    zero = ~(pos | neg)
    v_lo = float(np.min(A)) - 1.0
    v_hi = max(float(np.max(A + C * t_min)), float(np.max(A + C * t_max)))
    for _ in range(45):
        v_mid = (v_lo + v_hi) * 0.5
        ub = t_max
        if np.any(pos):
            ub = min(ub, float(np.min((v_mid - A[pos]) / C[pos])))
        lb = t_min
        if np.any(neg):
            lb = max(lb, float(np.max((v_mid - A[neg]) / C[neg])))
        if np.any(zero):
            if float(np.max(A[zero])) > v_mid + 1e-15:
                v_lo = v_mid
                continue
        if lb <= ub + 1e-12:
            v_hi = v_mid
        else:
            v_lo = v_mid
    v_star = v_hi
    ub = t_max
    if np.any(pos):
        ub = min(ub, float(np.min((v_star - A[pos]) / C[pos])))
    lb = t_min
    if np.any(neg):
        lb = max(lb, float(np.max((v_star - A[neg]) / C[neg])))
    if lb > ub:
        t_star = (t_min + t_max) * 0.5
    else:
        t_star = (lb + ub) * 0.5
    return t_star


def coord_desc(h, n, target, deadline, rng, recompute_interval=
25):

            break
        if sweep > 0 and sweep % recompute_interval == 0:
            corr = compute_corr(h_cur)
        rng.shuffle(order)

        for p in range(n // 2):
            i = int(order[2 * p])
            j = int(order[2 * p + 1])
            c = h_cur[i] + h_cur[j]
            if c < 1e-14 or c > 2.0 - 1e-14:
                continue
            t_min = max(0.0, c - 1.0)
            t_max = min(1.0, c)
            if t_max - t_min < 1e-13:
                continue
            C = compute_C_pair(h_cur, i, j, n)
            A = corr - h_cur[i] * C
            t_star = solve_1d_max(A, C, t_min, t_max)
            delta = t_star - h_cur[i]
            if abs(delta) > 1e-14:
                h_cur[i] = t_star
                h_cur[j] = c - t_star
                corr += delta * C

        new_score = float(np.max(corr)) * dx
        if new_score < best_score - 1e-15:
            best_score = new_score
            stall_count = 0
        else:
            stall_count += 1

            corr = compute_corr(h_cur)
            best_score = float(np.max(corr)) * dx

    h_final = project_feasible(h_cur, target)
    return h_final, compute_c5(h_final)
