import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.sum(np.exp(shifted)) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_budget=None):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        if time_budget is not None and time_budget() < 30:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def fast_pair_cd(h, n, target, time_limit, max_sweeps=5000, n_cand=60, sa_temp=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_corr = 2 * n - 1
    mid = n - 1
    shifts = np.arange(-(n - 1), n)
    idxs = np.arange(n)
    consecutive_no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        if consecutive_no_improve > 300:
            break
        
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad -= np.mean(grad)
        
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        
        best_trial_c5 = c5_val
        best_i = -1
        best_j = -1
        best_delta = 0.0
        best_g = None
        
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                g = np.zeros(n_corr)
                v_ip = (i + shifts >= 0) & (i + shifts < n)
                g[v_ip] += h[i + shifts[v_ip]]
                v_im = (i - shifts >= 0) & (i - shifts < n)
                g[v_im] += h[i - shifts[v_im]]
                v_jp = (j + shifts >= 0) & (j + shifts < n)
                g[v_jp] -= h[j + shifts[v_jp]]
                v_jm = (j - shifts >= 0) & (j - shifts < n)
                g[v_jm] -= h[j - shifts[v_jm]]
                
                a_d = delta_lo
                b_d = delta_hi
                for _ in range(25):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * g
                    f_mid = np.max(vals)
                    active_mask = vals >= f_mid - 1e-14
                    if active_mask.any():
                        f_prime = np.max(g[active_mask])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                
                approx_max = np.max(corr + delta_star * g) * dx
                if approx_max < best_trial_c5:
                    best_trial_c5 = approx_max
                    best_i = i
                    best_j = j
                    best_delta = delta_star
                    best_g = g
        
        if best_i >= 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    
    return best_h, best_c5

def block_cd(h, n, target, time_limit, block_sizes=None, max_rounds=50):
    if block_sizes is None:
        block_sizes = [5, 10, 20]
    
    start = time.time()
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for round_idx in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        for B in block_sizes:
            if time.time() - start > time_limit:
                break
            n_blocks = n // B
            for b in range(n_blocks):
                if time.time() - start > time_limit:
                    break
                lo = b * B
                hi = lo + B
                if hi > n:
                    break
                
                # Save the block values
                h_block_orig = h[lo:hi].copy()
                
                # Optimize the block using SLSQP
                def block_obj(x):
                    h_copy = h.copy()
                    h_copy[lo:hi] = x
                    return smooth_obj(h_copy, 5000)
                
                def block_cons(x):
                    return np.sum(x) - np.sum(h_block_orig)
                
                bounds_block = [(0.0, 1.0)] * B
                cons_block = [{'type': 'eq', 'fun': block_cons}]
                
                try:
                    res = minimize(block_obj, h_block_orig, method='SLSQP',
                                   bounds=bounds_block, constraints=cons_block,
                                   options={'maxiter': 50, 'ftol': 1e-12})
                    if res.success or compute_c5(np.concatenate([h[:lo], res.x, h[hi:]])) < best_c5:

                            best_h = h_new.copy()

                    pass

    idxs = np.arange(n)

    grad = np.zeros(n)
    v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
    v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
    grad[v1] += (1.0 - h[idxs[v1] - k_star])
    grad[v2] -= h[idxs[v2] + k_star]
    grad -= np.mean(grad)

    low_set = sorted_idx[:n_cand]
    high_set = sorted_idx[-n_cand:]

    pairs = []
    for i in high_set:
        for j in low_set:
            if i != j:
                delta_lo = max(h[i] - 1.0, -h[j])
                delta_hi = min(h[i], 1.0 - h[j])
                if delta_hi - delta_lo > 1e-12:
                    pairs.append((i, j, delta_lo, delta_hi))
    
    if not pairs:
        return None

    a_d = np.array([p[2] for p in pairs])
    b_d = np.array([p[3] for p in pairs])

        vals = corr[None, :] + mid_d[:, None] * G  # shape: (n_pairs, n_corr)
        f_mid = np.max(vals, axis=1)
        active_mask = vals >= f_mid[:, None] - 1e-14
        # f_prime[p] = max(G[p, k]) over active k
        # Set inactive to -inf, then take max
        G_masked = np.where(active_mask, G, -np.inf)
        f_prime = np.max(G_masked, axis=1)
        mask_pos = f_prime > 1e-14
        b_d[mask_pos] = mid_d[mask_pos]
        a_d[~mask_pos] = mid_d[~mask_pos]
    
    delta_star = 0.5 * (a_d + b_d)

    vals = corr[None, :] + delta_star[:, None] * G
    approx_max = np.max(vals, axis=1) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:

    else:

        approx_max = np.max(vals, axis=1) * dx
        best_local = int(np.argmin(approx_max))

        if approx_max[best_local] < c5_val - 1e-15:
            r = rows[best_local]
            c = cols[best_local]
            i = int(hi_idx[r])
            j = int(lo_idx[c])
            delta = float(delta_star[best_local])
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)

    return best_h, best_c5


def block_cd_fast(h, n, target, time_limit, block_size=8, max_rounds=30):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    n_blocks = n // block_size

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            lo = b * block_size
            hi = lo + block_size
            if hi > n:
                break

            h_orig_block = h[lo:hi].copy()
            block_sum = float(np.sum(h_orig_block))

            def block_obj(x):
                h_c = h.copy()
                h_c[lo:hi] = x
                corr = np.correlate(h_c, 1.0 - h_c, mode='full')
                c_max = np.max(corr)
                alpha_b = 5000.0
                shifted = np.clip(alpha_b * (corr - c_max), -500.0, 0.0)
                return float(c_max + np.sum(np.exp(shifted)) / alpha_b)

            def block_cons(x):
                return np.sum(x) - block_sum

            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_cons}]

            try:
                res = minimize(block_obj, h_orig_block, method='SLSQP',
                               bounds=bounds_b, constraints=cons_b,
                               options={'maxiter': 30, 'ftol': 1e-12})
                h_new = h.copy()
                h_new[lo:hi] = np.clip(res.x, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new
            except Exception:
                pass

    return best_h, best_c5


def momentum_subgradient(h, n, target, time_limit, max_steps, mu=0.9, base_lr=0.015):
    start = time.time()
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)
    idxs = np.arange(n)
    mid = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - mid
        grad = np.zeros(n)
        v1 = (idxs - k_star >= 0) & (idxs - k_star < n)
        v2 = (idxs + k_star >= 0) & (idxs + k_star < n)
        grad[v1] += (1.0 - h[idxs[v1] - k_star])
        grad[v2] -= h[idxs[v2] + k_star]
        grad *= dx
        grad -= np.mean(grad)

        v = mu * v + grad
        h_new = project_feasible(h - base_lr * v, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_cd(h, n, target, time_limit, n_starts=4, noise_scales=None):
    if noise_scales is None:
        noise_scales = [0.01, 0.03, 0.05, 0.08]
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(12345)
    time_per_start = time_limit / n_starts

    for k in range(n_starts):
        if time.time() - start > time_limit:
            break
        scale = noise_scales[k % len(noise_scales)]
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        time_left = time_limit - (time.time() - start)
        if time_left < 30:
            break
        h_cd, c5_cd = vectorized_pair_cd(h_pert, n, target,
                                          time_limit=min(time_per_start, time_left - 20),
                                          max_sweeps=5000, n_cand=50, n_bisect=18)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP polish
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [200, 1000, 5000, 20000, 80000],
                            2000, 1e-15, tleft)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Fast vectorized pair CD (large candidate set)
    if tleft() > 300:
        cd_time = min(350, tleft() * 0.35)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=10000, n_cand=80, n_bisect=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 3: SLSQP after pair CD
    if tleft() > 120:
        h_pol = slsqp_round(best_h, n, target,
                            [2000, 10000, 50000, 200000],
                            2000, 1e-15, tleft)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 4: Block coordinate descent
    if tleft() > 100:
        h_block, c5_block = block_cd_fast(best_h, n, target,
                                           time_limit=min(80, tleft() * 0.25),
                                           block_size=10, max_rounds=20)
        if c5_block < best_score:
            best_score = c5_block
            best_h = h_block
            if tleft() > 60:
                h_pol = slsqp_round(best_h, n, target,
                                    [5000, 20000, 100000],
                                    1500, 1e-15, tleft)
                sc = compute_c5(h_pol)
                if sc < best_score:
                    best_score = sc
                    best_h = h_pol

    # Phase 5: Another round of pair CD with different parameters
    if tleft() > 200:
        cd_time = min(250, tleft() * 0.3)
        h_cd, c5_cd = vectorized_pair_cd(best_h, n, target,
                                          time_limit=cd_time,
                                          max_sweeps=10000, n_cand=100, n_bisect=18)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 6: Multi-start exploration
    if tleft() > 150:
        h_ms, c5_ms = multi_start_cd(best_h, n, target,
                                      time_limit=min(120, tleft() * 0.4),
                                      n_starts=3,
                                      noise_scales=[0.02, 0.04, 0.06])
        if c5_ms < best_score:
            best_score = c5_ms
            best_h = h_ms

    # Phase 7: SLSQP polish
    if tleft() > 80:
        h_pol = slsqp_round(best_h, n, target,
                            [5000, 20000, 100000, 500000],
                            1500, 1e-15, tleft)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 8: Try higher resolution
    if tleft() > 250:
        n_high = 600
        target_high = n_high / 2.0
        h_interp = interpolate_to(best_h, n_high, target_high)
        h_high, c5_high = vectorized_pair_cd(h_interp, n_high, target_high,
                                              time_limit=min(150, tleft() * 0.4),
                                              max_sweeps=8000, n_cand=60, n_bisect=18)
        if c5_high < best_score:
            best_score = c5_high
            best_h = h_high
            n = n_high
            target = target_high
        elif tleft() > 100:
            h_back = interpolate_to(h_interp, n, target)
            h_pol = slsqp_round(h_back, n, target,
                                [5000, 20000],
                                1000, 1e-15, tleft)
            sc = compute_c5(h_pol)
            if sc < best_score:
                best_score = sc
                best_h = h_pol

    # Phase 9: Momentum subgradient for fine-tuning
    if tleft() > 60:
        h_mom, c5_mom = momentum_subgradient(best_h, len(best_h), len(best_h)/2.0,
                                              time_limit=min(40, tleft() * 0.4),
                                              max_steps=50000, mu=0.92, base_lr=0.01)
        if c5_mom < best_score:
            best_score = c5_mom
            best_h = h_mom

    # Phase 10: Final pair CD
    if tleft() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = vectorized_pair_cd(best_h, n_use, target_use,
                                          time_limit=min(50, tleft() * 0.5),
                                          max_sweeps=5000, n_cand=70, n_bisect=20)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 11: Final SLSQP
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_round(best_h, n_use, target_use,
                              [10000, 50000, 200000, 1000000],
                              1000, 1e-15, tleft)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
