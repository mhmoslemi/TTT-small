**Better approach: Use the constraint exactly via reparameterization.**

Let h = P(z) where P is a projection that ensures ∑h = target and 0 ≤ h ≤ 1. Then optimize over z freely.

The projection P: given z, find h = argmin ∑(h-z)² subject to 0≤h≤1, ∑h=target.

This is a quadratic programming problem that can be solved by bisection on the Lagrange multiplier:
h_i = clip(z_i - λ, 0, 1) where λ is chosen so that ∑h_i = target.

This is the "water-filling" solution! It's much better than the uniform distribution.

Let me implement this:

```python
def project_feasible_wf(h, target_sum):
    """Water-filling projection: find h' = argmin ||h'-h||^2 s.t. 0<=h'<=1, sum(h')=target"""
    # Bisection on lambda
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    
    def sum_at_lam(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_lam(mid)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)
```

This is O(n log n) per projection (with the bisection being ~100 iterations of O(n) each). Much better than the iterative approach.

2. Project to feasibility after each step
3. The projection preserves the sum constraint exactly

This is the **projected gradient method** (or its variants like FISTA, Nesterov, Adam+project).

Let me also think about the **dual** of the projection. The water-filling projection is the proximal operator of the indicator function of the constraint set. So projected gradient is equivalent to a proximal gradient method.

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    return float(np.max(over)) * dx

def project_wf(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    lo = float(h.min()) - 1.0
    hi = float(h.max()) + 1.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - float(np.sum(result))
    if abs(diff) > 1e-13:
        mask = (result > 1e-10) & (result < 1.0 - 1e-10)
        cnt = float(mask.sum())
        if cnt > 0:
            result = result.copy()
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def lse_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = alpha * (over - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    s = float(np.sum(exp_s))

        else:
            grad[:n-k] += wk * (1.0 - h[k:]) * dx

    return obj, grad

def weighted_grad(h, n, target, beta):

    corr = compute_corr(h)
    over = corr[n-1:n-1+n]
    c_max = float(np.max(over))
    shifted = beta * (over - c_max)

    active = np.where(w > 1e-10)[0]

            grad[k:] -= wk * h[:n-k] * dx
    return grad

def lbfgs_round(h, n, target, alphas, maxiter, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    M = 1e10
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        cur_alpha = alpha
        def fun(x):
            obj, _ = lse_obj_grad(x, cur_alpha)
            return obj + M * (float(np.sum(x)) - target) ** 2
        def jac(x):
            _, g = lse_obj_grad(x, cur_alpha)
            return g + 2.0 * M * (float(np.sum(x)) - target)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun, h_cur, method='L-BFGS-B', jac=jac, bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': 20000})
            h_new = project_wf(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def fista_optimize(h0, n, target, alpha_lse, max_iter, lr_init, deadline=None):
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    lr = lr_init
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        momentum = (t - 1.0) / t_new
        y = h + momentum * (h - h_prev)
        _, grad = lse_obj_grad(y, alpha_lse)
        h_new = project_wf(y - lr * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        t = t_new
        if it % 200 == 0 and it > 0:
            lr = lr * 0.95
        lr = max(lr, 0.001)
    return best_h, best_score

def nesterov_grad(h0, n, target, beta, max_iter, lr_init, deadline=None):
    h = h0.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):

        v = 0.9 * v - lr * grad

            best_score = sc
            best_h = h_new.copy()
            h = h_new
        else:
            v *= 0.5
            lr = max(lr * 0.95, 1e-6)
            if lr < 1e-8:
                break
    return best_h, best_score

def slsqp_round(h, n, target, alphas, maxiter, deadline=None):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if deadline is not None and time.time() > deadline:
            break
        def obj_fn(x):
            obj, _ = lse_obj_grad(x, alpha)
            return obj
        def eq_fun(x):
            return float(np.sum(x)) - target
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_wf(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_wf(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_wf(h_out, target_dst)

def de_coarse(n_c, target_c, pop_size, max_gen, rng, deadline=None, F=0.8, CR=0.9):
    population = []
    for i in range(pop_size):
        h = rng.rand(n_c)
        h = project_wf(h, target_c)
        population.append(h)
    scores = [compute_c5(h) for h in population]
    order = np.argsort(scores)
    population = [population[i] for i in order]
    scores = [scores[i] for i in order]
    best = population[0].copy()
    best_score = scores[0]
    for gen in range(max_gen):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(pop_size):
            candidates = [j for j in range(pop_size) if j != i]
            r1, r2, r3 = rng.choice(candidates, size=3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])
            trial = np.zeros(n_c)
            j_rand = rng.randint(n_c)
            for j in range(n_c):
                if rng.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            trial = project_wf(trial, target_c)
            trial_score = compute_c5(trial)
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best = trial.copy()
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
    return best, best_score
