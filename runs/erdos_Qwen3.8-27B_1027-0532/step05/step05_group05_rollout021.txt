    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)

    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _cons_eq(x):
    return np.sum(x) - _g_target

def _cons_jac(x):
    return np.ones(_g_n)

def lse_objective(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    _g_target = target

            break

            res = minimize(lse_objective, h_cur, method='SLSQP', bounds=bounds,

            best_h = h_cur.copy()

        temp = 0.002 + 0.008 * (1.0 - frac)
        diffs = over[top_idx] - over[top_idx[0]]
        weights = np.exp(diffs / temp)
        weights /= weights.sum()

            k = int(k)

            j_neg = np.arange(k, n)
