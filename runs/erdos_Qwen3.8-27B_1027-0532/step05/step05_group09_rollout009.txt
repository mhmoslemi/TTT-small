def smooth_obj_and_grad_fft(h, alpha, p=1):
    """Compute smoothed objective and gradient using FFT.
    If p=1, this is log-sum-exp. If p>1, this is power-mean-like."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')[n-1:]
    c_max = float(np.max(corr))

        weights = exp_shifted / sum_exp
        obj = (c_max + np.log(sum_exp) / alpha) * dx
    else:

    H = np.fft.rfft(h, fsize)
    corr_pos = np.fft.irfft(Wf * V, fsize)[n-1:2*n-1]
    corr_neg = np.fft.irfft(W * H, fsize)[:n]
    grad = dx * (corr_pos - corr_neg)
    return obj, grad

def fista_optimize(h0, target, alpha, max_iters, deadline, p=1):
    n = len(h0)
    x = project_feasible(h0, target).copy()
    x_prev = x.copy()
    y = x.copy()
    t = 1.0
    best_x = x.copy()
    best_obj = compute_c5(x)
    lr_init = 1.0 / max(alpha * 0.05, 1.0)
    if p > 1:
        lr_init = 1.0 / max(1.0, n * p * 0.01)
    step = lr_init
    
    for k in range(max_iters):
        if time.time() > deadline:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - x_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = smooth_obj_and_grad_fft(y_clipped, alpha, p)
        x_new = project_feasible(y_clipped - step * grad_y, target)
        shrink = 0.0
        for _ in range(40):
            obj_new, _ = smooth_obj_and_grad_fft(x_new, alpha, p)
            decrease = obj_y - obj_new
            quadratic = 0.5 * (1.0 / step) * np.dot(x_new - y_clipped, x_new - y_clipped)
            if decrease >= quadratic * (1.0 - shrink):
                break
            step *= (1.0 - shrink)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        else:
            step = max(step, 1e-12)
            x_new = project_feasible(y_clipped - step * grad_y, target)
        step = min(step / (1.0 - shrink), lr_init * 2.0)
        x_prev = x
        x = x_new
        t = t_new
        sc = compute_c5(x)
        if sc < best_obj:
            best_obj = sc
            best_x = x.copy()
    return best_x, best_obj
