```python
def get_grad_for_shift(h, k):
    """Gradient of c_k with respect to h"""
    n = len(h)
    idxs = np.arange(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g = np.zeros(n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    mid = n - 1
    no_improve = 0
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step % 500 == 0 and no_improve > 500:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Get top-k active constraints
        top_idx = np.argsort(corr)[-n_top:]
        c_max = np.max(corr)
        
        # Weighted gradient (uniform weights for top-k)
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_for_shift(h, k)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)  # center to maintain sum constraint
        
        # Adam update
        t = step + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        update = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h - update
        h_new = project_feasible(h_new, target)
        h = h_new
    
    return best_h, best_c5
```

```python
        # h[p+k] term: index = p + (idx - (n-1)) = p + idx - n + 1
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    
    return f
```

This is O(n * (2n)) = O(n²) total, which is fine.

Then in the pair CD, for each pair (i, j):
dcorr = f[i] - f[j]

And we need to minimize max(corr + delta * dcorr) over delta.

This is a 1D convex piecewise-linear minimization. The function g(delta) = max_m (corr[m] + delta * dcorr[m]) is convex in delta (max of linear functions). We can find the minimum by binary search on the derivative.

The derivative of g(delta) is dcorr[m*] where m* = argmax_m (corr[m] + delta * dcorr[m]). If this is positive, we should decrease delta; if negative, increase.

Binary search converges in O(log(1/eps)) iterations, each O(n) to compute the max and argmax.

Total per pair: O(n * log(1/eps)) ≈ O(n * 20) for 20 bisection steps.

For all (i,j) pairs, we have dcorr_ij = f[i] - f[j]. We want to find delta_ij that minimizes max(corr + delta_ij * dcorr_ij).

The optimal delta satisfies: the max of (corr + delta * dcorr) is achieved at a point where the slope changes sign. Specifically, if we sort the values corr[m] + delta * dcorr[m], the maximum is at the top, and we want the slope (dcorr[m*]) to be as close to 0 as possible.

This is equivalent to: find delta such that there exist m1, m2 with corr[m1] + delta*dcorr[m1] = corr[m2] + delta*dcorr[m2] and dcorr[m1] > 0 > dcorr[m2] (or dcorr[m1] = 0).

For a fixed pair (i,j), the optimal delta is at the intersection of two lines:
corr[m1] + delta * dcorr[m1] = corr[m2] + delta * dcorr[m2]
delta = (corr[m2] - corr[m1]) / (dcorr[m1] - dcorr[m2])

We want the delta that minimizes the maximum, which is at the "tallest" intersection point.

This is essentially finding the lower envelope of lines... no, the upper envelope. We want to find the delta where the upper envelope is minimized.

For a 1D problem with n lines, the minimum of the upper envelope can be found by checking all O(n²) intersection points and finding which one has the lowest value. But that's O(n³) which is too slow.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_grad_shift(h, k, n):
    idxs = np.arange(n)
    g = np.zeros(n)
    v1 = (idxs - k >= 0) & (idxs - k < n)
    v2 = (idxs + k >= 0) & (idxs + k < n)
    g[v1] += (1.0 - h[idxs[v1] - k])
    g[v2] -= h[idxs[v2] + k]
    return g

def _slsqp_cons_fun(h):
    return np.sum(h) - _G_target

def _slsqp_cons_jac(h):
    return np.ones(_G_n)

_G_n = 0
_G_target = 0.0
_G_alpha = 0.0

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    shifted = np.clip(_G_alpha * (C - max_C), -500.0, 0.0)
    return float(max_C + np.log(np.sum(np.exp(shifted))) / _G_alpha)

def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _G_n, _G_target, _G_alpha
    _G_n = n
    _G_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        _G_alpha = alpha
        try:
            res = minimize(_slsqp_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_c5

def adam_optimize(h_start, n, target, time_limit, max_steps, n_top_start, n_top_end,
                  lr, beta1, beta2, eps):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    m = np.zeros(n)
    v = np.zeros(n)
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        frac = min(step / max(1, max_steps * 0.3), 1.0)
        n_top = int(n_top_start + (n_top_end - n_top_start) * frac)
        n_top = max(n_top, 1)
        top_idx = np.argsort(corr)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(h, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        t = step + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        h = h_new
    return best_h, best_c5

def nesterov_optimize(h_start, n, target, time_limit, max_steps, n_top, lr, mu):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 200 and no_improve > 1000:
            break
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr_y)[-n_top:]
        grad = np.zeros(n)
        for s_idx in top_idx:
            k = int(s_idx) - mid
            grad += get_grad_shift(y, k, n)
        grad /= n_top
        grad *= dx
        grad -= np.mean(grad)
        y_new = y - lr * grad
        h_new = (1.0 - mu) * h + mu * y_new
        h_new = project_feasible(h_new, target)
        y = project_feasible(h_new + mu * (h_new - h), target)
        h = h_new
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if step % 100 == 0 and step > 0:
            mu = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * mu * mu))
    return best_h, best_c5

def precompute_f(h, n):
    f = -np.ones((n, 2 * n - 1))
    idx = np.arange(2 * n - 1)
    for p in range(n):
        idx1 = p + n - 1 - idx
        mask1 = (idx1 >= 0) & (idx1 < n)
        f[p, mask1] += h[idx1[mask1]]
        idx2 = p + idx - n + 1
        mask2 = (idx2 >= 0) & (idx2 < n)
        f[p, mask2] += h[idx2[mask2]]
    return f

def pair_cd(h_start, n, target, time_limit, max_sweeps, n_cand):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    mid = n - 1
    no_improve = 0
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        if no_improve > 250:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        k_star = int(np.argmax(corr)) - mid
        grad = get_grad_shift(h, k_star, n)
        grad -= np.mean(grad)
        sorted_idx = np.argsort(grad)
        low_set = sorted_idx[:n_cand]
        high_set = sorted_idx[-n_cand:]
        f_arr = precompute_f(h, n)
        best_trial = c5_val
        best_move = None
        for i in high_set:
            for j in low_set:
                if i == j:
                    continue
                dcorr = f_arr[i] - f_arr[j]
                a_d = 0.0
                b_d = min(h[i], 1.0 - h[j])
                if b_d <= 1e-14:
                    continue
                for _ in range(18):
                    mid_d = 0.5 * (a_d + b_d)
                    vals = corr + mid_d * dcorr
                    f_mid = np.max(vals)
                    active = vals >= f_mid - 1e-14
                    if active.any():
                        f_prime = np.max(dcorr[active])
                    else:
                        f_prime = 0.0
                    if f_prime > 1e-14:
                        b_d = mid_d
                    else:
                        a_d = mid_d
                delta_star = 0.5 * (a_d + b_d)
                if abs(delta_star) < 1e-12:
                    continue
                trial_c5 = float(np.max(corr + delta_star * dcorr)) * dx
                if trial_c5 < best_trial:
                    best_trial = trial_c5
                    best_move = (i, j, delta_star)
        if best_move is not None:
            i, j, delta = best_move
            h[i] -= delta
            h[j] += delta
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h, target)
    return best_h, best_c5

def soft_weighted_subgrad(h_start, n, target, time_limit, max_steps, alpha_w, base_lr):
    start = time.time()
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid = n - 1
    no_improve = 0
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        if step > 50 and no_improve > 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(np.max(corr)) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        c_max = np.max(corr)
        weights = np.exp(alpha_w * (corr - c_max))
        wsum = weights.sum()
        if wsum < 1e-300:
            break
        weights /= wsum
        sig_idx = np.where(weights > 1e-4 / (2 * n))[0]
        grad = np.zeros(n)
        for s_idx in sig_idx:
            k = int(s_idx) - mid
            grad += weights[s_idx] * get_grad_shift(h, k, n)
        grad *= dx
        grad -= np.mean(grad)
        improved = False
        for lr_mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
            lr_try = base_lr * lr_mult
            h_try = project_feasible(h - lr_try * grad, target)
            c5_try = compute_c5(h_try)
            if c5_try < c5_now - 1e-14:
                h = h_try
                improved = True
                break
        if not improved:
            no_improve += 5
    return best_h, best_c5

def local_perturbation(h_init, n, target, max_tries, time_limit, rng):
    start = time.time()
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    for step in range(max_tries):
        if time.time() - start > time_limit:
            break
        n_pert = int(rng.choice([1, 2, 3, 5, 8]))
        n_pert = min(n_pert, n)
        idxs = rng.choice(n, size=n_pert, replace=False)
        scale = float(rng.choice([0.02, 0.05, 0.1, 0.15, 0.2]))
        h_try = h_cur.copy()
        h_try[idxs] += rng.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
