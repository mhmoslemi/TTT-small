def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_bangbang(h, n_ones):
    h_sorted_idx = np.argsort(h)[::-1]
    h_bb = np.zeros(len(h))
    h_bb[h_sorted_idx[:n_ones]] = 1.0
    return h_bb

def obj_soft_max(x):
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha

    return np.sum(x) - _g_target

    return np.ones(_g_n)

    cur_score = compute_c5(h_cur)

def solve_lp(n):
    """Solve the McCormick LP relaxation at resolution n."""
    target = n / 2.0
    dx = 2.0 / n
    
    h = cp.Variable(n)
    C = cp.Variable()
    
    # McCormick variables for positive lags
    # u_pos[k, i] = h[i] * (1 - h[i+k]) for k=0,...,n-1, i=0,...,n-1-k
    u_pos = {}
    constraints = []
    
    for k in range(n):
        for i in range(n - k):
            u = cp.Variable()
            u_pos[(k, i)] = u
            # McCormick: a = h[i], b = 1-h[i+k]
            constraints.append(u >= 0)
            constraints.append(u >= h[i] + (1 - h[i+k]) - 1)
            constraints.append(u <= h[i])
            constraints.append(u <= 1 - h[i+k])
    
    # McCormick variables for negative lags
    # u_neg[k, i] = h[i+k] * (1 - h[i]) for k=1,...,n-1, i=0,...,n-1-k
    u_neg = {}
    for k in range(1, n):
        for i in range(n - k):
            u = cp.Variable()
            u_neg[(k, i)] = u
            # a = h[i+k], b = 1-h[i]
            constraints.append(u >= 0)
            constraints.append(u >= h[i+k] + (1 - h[i]) - 1)
            constraints.append(u <= h[i+k])
            constraints.append(u <= 1 - h[i])
    
    # C constraints
    for k in range(n):
        total = sum(u_pos[(k, i)] for i in range(n - k))
        constraints.append(total * dx <= C)
    
    for k in range(1, n):
        total = sum(u_neg[(k, i)] for i in range(n - k))
        constraints.append(total * dx <= C)
    
    # h constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    constraints.append(cp.sum(h) == target)
    
    obj = cp.Minimize(C)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    
    return h.value, C.value
