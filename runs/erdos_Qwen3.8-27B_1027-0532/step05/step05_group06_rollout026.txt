def project_feasible(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} using bisection on shift."""
    h = np.asarray(h, dtype=float)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target_sum
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr_and_grad(h, active_indices):
    """Compute correlation and gradient for given active lags."""
    n = len(h)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    grad = np.zeros(n)
    n_half = n - 1
    for ki in active_indices:
        lag = int(ki) - n_half
        if lag >= 0:
            # c_k = sum_{i=lag}^{n-1} h[i]*(1-h[i-lag])
            # grad[i] += (1-h[i-lag]) for i in [lag, n-1]
            # grad[i] -= h[i+lag] for i in [0, n-lag-1]
            grad[lag:] += one_minus_h[:n-lag]
            if lag > 0:
                grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            # c_k = sum_{i=0}^{n-1-m} h[i]*(1-h[i+m])
            # grad[i] += (1-h[i+m]) for i in [0, n-m-1]
            # grad[i] -= h[i-m] for i in [m, n-1]
            grad[:n-m] += one_minus_h[m:]
            if m > 0:
                grad[m:] -= h[:n-m]
    return corr, grad

        h_pert = best_h.copy()
        if i % 3 == 0:
            h_pert += rng.randn(n_cur) * 0.08
        elif i % 3 == 1:
            nf = int(n_cur * 0.05)
            idx = rng.choice(n_cur, size=nf, replace=False)
            h_pert[idx] = rng.uniform(0, 1, nf)
        else:
            freq = np.fft.rfftfreq(n_cur)
            mask = freq < 0.03
            pert = np.fft.irfft(np.where(mask, rng.randn(n_cur // 2 + 1), 0.0), n_cur)
            h_pert += pert * 0.3
        h_pert = project_feasible(h_pert, target_cur)
        h_new = lbfgs_opt(h_pert, target_cur, 500, mu, 300)
        h_new = lbfgs_opt(h_new, target_cur, 10000, mu, 200)
        h_new = project_feasible(h_new, target_cur)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 7: Coordinate descent polish
    if tleft() > 120:
        h_cd, sc_cd = coord_descent(best_h, target_cur, n_sweeps=2)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()

    # Phase 8: Final active set + high alpha
    if tleft() > 80:
        h_new = active_set_iterate(best_h, target_cur, 5, 200000, mu, 3, deadline)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_new = lbfgs_opt(best_h, target_cur, 500000, mu, 400)
        h_new = project_feasible(h_new, target_cur)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    # Phase 9: SLSQP final
    if tleft() > 50:
        h_slq = slsqp_polish(best_h, target_cur, 800)
        h_slq = project_feasible(h_slq, target_cur)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq.copy()

    # Phase 10: One more coord descent + active set
    if tleft() > 50:
        h_cd, sc_cd = coord_descent(best_h, target_cur, n_sweeps=1)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd.copy()
        h_new = lbfgs_opt(best_h, target_cur, 500000, mu, 200)
        h_new = project_feasible(h_new, target_cur)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

    h_final = project_feasible(best_h, target_cur)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    return h_final, best_score, len(h_final)
