def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0

        else:

        if w_sum < 1e-300:

def _proj_grad(h0, target_sum, n_iters=2000, deadline=None):
    n = len(h)
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    
    prev_grad = None
    prev_x = None
    xk_1 = 0.0  # previous BB step
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute gradient (using top-5 active terms)
        corr = np.correlate(x, 1.0 - x, mode='full')
        top5 = np.argsort(corr)[::-1][:5]
        g = np.zeros(n)
        for idx in top5:
            L = idx - (n-1)
            j_all = np.arange(n)
            idx1 = j_all + L
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = j_all - L
            valid2 = (idx2 >= 0) & (idx2 < n)
            gi = np.zeros(n)
            gi[valid1] += (1.0 - x[idx1[valid1]])
            gi[valid2] += -x[idx2[valid2]]
            g += gi * dx / len(top5)
        
        # Barzilai-Borwein step size
        if prev_grad is not None and prev_x is not None:
            s = x - prev_x
            y_bb = g - prev_grad
            ss = s @ s
            sy = s @ y_bb
            yy = y_bb @ y_bb
            if yy > 1e-15:
                if it % 2 == 0:
                    xk_1 = ss / max(sy, 1e-15)
                else:
                    xk_1 = max(sy, 1e-15) / yy
            xk_1 = np.clip(xk_1, 1e-4, 1.0)
        
        if xk_1 == 0.0:
            xk_1 = 0.1
        
        x_new = _project(x - xk_1 * g, target_sum)
        
        cv = _c5_of(x_new)
        if cv < best_val:
            best_val = cv
            best = x_new.copy()
        
        prev_grad = g
        prev_x = x
        x = x_new
    
    return best, best_val

        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:

Let me now think about the BB gradient refinement more carefully.

For the BB gradient, I need to compute the gradient of the objective with respect to h. The objective is:
f(h) = (2/n) * max_ki corr_ki(h)

The gradient (subgradient) is:
∇f(h) = (2/n) * ∇corr_{ki*}(h) where ki* = argmax_ki corr_ki(h)

But the max is not differentiable, so I'll use a smoothing:
f_α(h) = (2/n) * (1/α) * log(sum_ki exp(α * corr_ki(h)))

∇f_α(h) = (2/n) * sum_ki w_ki * ∇corr_ki(h) where w_ki = exp(α*corr_ki) / sum_j exp(α*corr_j)

For the BB step, I'll use the top-K terms to approximate the gradient (to save computation):

    n = len(h0)

    best = x.copy()
    best_val = _c5_of(x)
    
    prev_x = None
    prev_g = None
    bb_step = 0.05
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        g = np.zeros(n)
        for idx in top_idx:
            L = idx - (n-1)
            j_all = np.arange(n)
            idx1 = j_all + L
            valid1 = (idx1 >= 0) & (idx1 < n)
            idx2 = j_all - L
            valid2 = (idx2 >= 0) & (idx2 < n)
            gi = np.zeros(n)
            gi[valid1] += (1.0 - x[idx1[valid1]])
            gi[valid2] += -x[idx2[valid2]]
            g += gi
        
        g *= dx / top_k
        
        # BB step
        if prev_x is not None and prev_g is not None:
            s = x - prev_x
            y_bb = g - prev_g
            ss = s @ s
            sy = s @ y_bb
            yy = y_bb @ y_bb
            if yy > 1e-15 and abs(sy) > 1e-15:

                    bb_step = ss / abs(sy)
                else:
                    bb_step = abs(sy) / yy

        cv = _c5_of(x_new)
        if cv < best_val:
            best_val = cv
            best = x_new.copy()
        
        prev_x = x
        prev_g = g
        x = x_new
    
    return best, best_val
