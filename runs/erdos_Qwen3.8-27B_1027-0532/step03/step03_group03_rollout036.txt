def lse_objective(h, alpha, penalty):
    corr = correlate_vector(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_viol = np.sum(h) - len(h) / 2.0
    return lse + penalty * sum_viol * sum_viol

def lse_gradient(h, alpha, penalty):
    n = len(h)
    corr = correlate_vector(h)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    
    # Compute weighted gradient of correlations
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        w = weights[k]

def project_onto_P(y, target_sum):
    """Project y onto {h : 0 <= h <= 1, sum(h) = target_sum}"""
    n = len(y)
    lo = y.min() - target_sum
    hi = y.max()
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)/2, 0.0, 1.0)
    # Fix any numerical issues
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, weights):
    """Compute sum_k weights[k] * d(c[lag_k])/d(h) where lag_k = k - (n-1)"""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # First term: (1-h[i-lag]) for i in [lag, n-1]
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                # Second term: -h[i+lag] for i in [0, n-1-lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            # First term: (1-h[i+m]) for i in [0, n-1-m]
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                # Second term: -h[i-m] for i in [m, n-1]
                grad[m:] -= w * h[:n - m]
    return grad

        # and i+lag is in [0, n-1]
        if 0 <= i < n and 0 <= i + lag < n:

        coefs[k] = c
    return coefs

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(h, alpha, penalty):
    corr = get_corr(h)
    n = len(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    obj = c_max + np.log(total) / alpha
    weights = exp_s / total
    
    # Gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    
    if penalty > 0:
        sv = np.sum(h) - n / 2.0
        obj += penalty * sv * sv
        grad += 2.0 * penalty * sv
    
    return obj, grad
