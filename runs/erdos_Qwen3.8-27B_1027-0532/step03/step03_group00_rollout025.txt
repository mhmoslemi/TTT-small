import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_indices(n):
    """Precompute index arrays for fast gradient computation."""
    j_idx = np.arange(n)
    k_vals = np.arange(2*n-1) - (n-1)
    J, K = np.meshgrid(j_idx, k_vals, indexing='ij')
    valid1 = (J + K >= 0) & (J + K < n)
    valid2 = (J - K >= 0) & (J - K < n)
    idx1 = np.where(valid1, J + K, 0)
    idx2 = np.where(valid2, J - K, 0)
    return valid1, idx1, valid2, idx2

def compute_smoothed_grad(h, dx, alpha, valid1, idx1, valid2, idx2):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    w = np.exp(alpha * (corr - c_max))
    w /= w.sum()
    
    term1_vals = np.where(valid1, 1.0 - h[idx1], 0.0)
    term2_vals = np.where(valid2, h[idx2], 0.0)
    
    grad = dx * (term1_vals @ w - term2_vals @ w)
    return grad, c_max

def fista_subgradient(h, n, target, dx, deadline, alpha_start, alpha_end, n_iters_max):
    """FISTA-like accelerated projected subgradient method."""
    valid1, idx1, valid2, idx2 = precompute_indices(n)
    
    h_cur = h.copy()
    h_next = h.copy()
    t = 1.0
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    # Adaptive step size
    step = 0.1
    last_improve = 0
    
    for it in range(n_iters_max):
        if time.time() > deadline:
            break
        
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(n_iters_max, 1))
        
        grad, c_max = compute_smoothed_grad(h_cur, dx, alpha, valid1, idx1, valid2, idx2)
        
        # Project gradient onto the affine subspace (remove mean component)
        grad_proj = grad - np.mean(grad)
        
        # FISTA step
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t*t)) / 2.0
        h_y = h_next + (t - 1.0) / t_new * (h_next - h_cur)
        h_y = np.clip(h_y, 0.0, 1.0)
        
        h_new = h_y - step * grad_proj
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h_new.copy()
            last_improve = it
        
        # Backtracking line search
        if sc > best_score * 1.01:
            step *= 0.7
        elif it - last_improve > 100:
            step = min(step * 1.05, 0.5)
        
        h_cur = h_new
        h_next = h_new
        t = t_new
    
    return best_h, best_score

The idea:
1. Find the lag k* that achieves the maximum correlation
2. c_{k*} = ∑_i h[i] * (1-h[i+k*]) * dx
3. To reduce c_{k*}, we want to either:
   - Decrease h[i] where h[i+k*] is large (reduce product)
   - Or increase h[i+k*] where h[i] is large (increase (1-h[i+k*])... wait no, we want to decrease the product)
   
Actually: c_{k*} = ∑_i h[i] * (1-h[i+k*]) * dx
To decrease this:
- Where h[i] and (1-h[i+k*]) are both large, we should either decrease h[i] or increase h[i+k*]
- This suggests: for pairs (i, i+k*) where both h[i] ≈ 1 and h[i+k*] ≈ 0, flip one of them

This is like a "swap" operation: if h[i] is high and h[i+k*] is low, swapping them would reduce c_{k*}.

def swap_local_search(h, n, target, deadline, n_swaps_per_iter=10):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    dx = 2.0 / n
    
    while time.time() < deadline:
        improved = False
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        c_max = np.max(corr)
        
        # Focus on lags near the maximum
        active_lags = np.where(corr > c_max - 0.001)[0]
        
        for _ in range(n_swaps_per_iter):
            if time.time() > deadline:
                break
            # Pick a random active lag
            m = active_lags[np.random.randint(len(active_lags))]
            k = m - (n-1)
            
            # Find the worst position: maximize h[i] * (1-h[i+k])
            if k >= 0:
                valid_i = np.arange(n - k)
            else:
                valid_i = np.arange(-k, n)
            
            products = h_cur[valid_i] * (1.0 - h_cur[valid_i + k])
            j = valid_i[np.argmax(products)]
            
            # Try swapping h[j] and h[j+k]
            h_test = h_cur.copy()
            h_test[j], h_test[j+k] = h_test[j+k], h_test[j]
            # Note: swapping preserves the sum
            
            sc = compute_c5(h_test)
            if sc < cur_score - 1e-10:
                h_cur = h_test
                cur_score = sc
                improved = True
        
        if not improved:
            # Try random swaps to escape
            for _ in range(50):
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j:
                    continue
                h_test = h_cur.copy()
                h_test[i], h_test[j] = h_test[j], h_test[i]
                sc = compute_c5(h_test)
                if sc < cur_score - 1e-12:
                    h_cur = h_test
                    cur_score = sc
                    improved = True
                    break
        
        if not improved:
            break
    
    return h_cur, cur_score

def _interpolate(h_src, n_dst, target_dst):

    for bs in [3, 4, 6]:
        if tleft() < 60:
            break
        h_bd, sc_bd = _block_descent(best_h, n, target, bs, 8, 500.0, deadline_p4, rng)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Phase 5: L-BFGS-B refinement
    if tleft() > 80:
        deadline_p5 = start_time + tleft() * 0.3
        h_ref2, sc_ref2 = _lbbfsb_round(best_h, n, target,
                                         [200, 1000, 5000, 10000],
                                         [1000.0, 10000.0],
                                         3000, deadline_p5)
        if sc_ref2 < best_score:
            best_score = sc_ref2
            best_h = h_ref2

    # Phase 6: Multi-start perturbations
    deadline_p6 = start_time + tleft() * 0.3
    for trial in range(8):
        if tleft() < 80:
            break
        h_pert = best_h.copy()
        n_pert = rng.randint(20, 80)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] += (rng.rand(n_pert) - 0.5) * 0.4
        h_pert = project_feasible(h_pert, target)
        h_ref3, sc_ref3 = _lbbfsb_round(h_pert, n, target,
                                         [100, 500, 2000],
                                         [1000.0, 10000.0],
                                         1500, start_time + 30)
        if sc_ref3 < best_score:
            best_score = sc_ref3
            best_h = h_ref3
        h_swap2, sc_swap2 = _swap_search(h_ref3, n, target, start_time + 20, rng)
        if sc_swap2 < best_score:
            best_score = sc_swap2
            best_h = h_swap2

    # Phase 7: Try higher resolution
    if tleft() > 250:
        n_hi = n * 2
        target_hi = n_hi / 2.0
        h_hi = _interpolate(best_h, n_hi, target_hi)
        _precompute(n_hi)
        _g_target = target_hi
        deadline_p7 = start_time + tleft() * 0.5
        h_hi_ref, sc_hi = _lbbfsb_round(h_hi, n_hi, target_hi,
                                         [50, 200, 1000],
                                         [100.0, 1000.0],
                                         1500, deadline_p7)
        if tleft() > 60:
            h_hi_swap, sc_hi_s = _swap_search(h_hi_ref, n_hi, target_hi, start_time + 60, rng)
            if sc_hi_s < sc_hi:
                sc_hi = sc_hi_s
                h_hi_ref = h_hi_swap
        # Downsample back to n for final answer
        h_back = _interpolate(h_hi_ref, n, target)
        _precompute(n)
        _g_target = target
        h_back_ref, sc_back = _lbbfsb_round(h_back, n, target,
                                             [500, 2000],
                                             [1000.0, 10000.0],
                                             1500, start_time + 30)
        if sc_back < best_score:
            best_score = sc_back
            best_h = h_back_ref

    # Phase 8: Final polish
    if tleft() > 30:
        h_final, sc_final = _lbbfsb_round(best_h, n, target,
                                           [2000, 5000, 10000, 20000],
                                           [10000.0, 100000.0],
                                           3000, start_time + tleft() - 10)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
