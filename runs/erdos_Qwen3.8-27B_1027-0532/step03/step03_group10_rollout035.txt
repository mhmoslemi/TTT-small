    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx

    x = (corr - cmax) / eps

    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s  # n-s elements
            # First term: j in [s, n-1], (1-h[j-s])
            grad[s:] += wk * omh[:m]
            # Second term: j in [0, n-1-s], -h[j+s]
            grad[:m] -= wk * h[s:]

            m = n - r
            # First term: j in [0, n-1-r], (1-h[j+r])
            grad[:m] += wk * omh[r:]
            # Second term: j in [r, n-1], -h[j-r]
            grad[r:] -= wk * h[:m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def lse_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def lse_gradient(h, eps):

    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx

    return grad

def greedy_local_search(h_bin, n_passes=10):
    """Try all single swaps and accept if it improves the max."""
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    
    for p in range(n_passes):
        improved = False
        ones_list = list(np.where(h == 1.0)[0])
        zeros_list = list(np.where(h == 0.0)[0])
        best_delta = 0
        best_swap = None
        for i in ones_list:
            for j in zeros_list:
                if i == j:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                s_range = np.arange(s_min, s_max + 1)
                k_range = s_range + half
                delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
                # Compute new max
                old_vals = corr[k_range].copy()
                new_vals = old_vals + delta
                # The new max is max(new_vals, max of corr outside k_range)
                if len(k_range) < len(corr):
                    outer_max = float(np.max(np.concatenate([corr[:k_range[0]], corr[k_range[-1]+1:]])))
                else:
                    outer_max = -np.inf
                new_max = max(float(np.max(new_vals)), outer_max)
                if new_max < current_max - 1e-12:
                    best_delta = new_max - current_max
                    best_swap = (i, j, s_range, k_range, delta)
                    if best_delta < -0.5:  # Good enough improvement
                        break
            if best_swap is not None and best_delta < -0.5:
                break
        if best_swap is not None:
            i, j, s_range, k_range, delta = best_swap
            corr[k_range] += delta
            h[i] = 0.0
            h[j] = 1.0
            current_max += best_delta
            improved = True
        if not improved:
            break
    return h, current_max
