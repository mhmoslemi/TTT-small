import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(20):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = min(diff / nfree, 1.0 - h[free].max() + 1e-15)
            h[free] += per
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = max(diff / nfree, -h[free].min() + 1e-15)
            h[free] += per
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    """Compute gradient of the correlation at a specific index of the full correlation."""
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def compute_grad_multi_lag(h, lag_indices, weights):
    """Compute weighted sum of gradients at multiple lags."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)

def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):

    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)

    best_score = compute_c5(h)

        else:

        w = np.exp(alpha * (top_values - np.max(top_values)))

            best_h = h_new.copy()
            stagnation = 0

        else:

            lr *= 0.99
        h = h_new

    return best_h, best_score

        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    
    # Additional constraints:
    # sum(h) = target -> handled separately (equality)
    # 0 <= h <= 1 -> bounds
    
    # For linprog: minimize c^T x subject to A_ub x <= b_ub, A_eq x = b_eq, bounds
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [target_sum]
    
    # Bounds: 0 <= h_i <= 1, t >= 0 (or t >= some lower bound)
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds_lp, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        return project_feasible(h_new, target_sum)
    return None

But wait, with n=400 and topk=100, we have 100 inequality constraints + 1 equality + 401 variables. That's manageable for HiGHS.

Let me also think about the step size. The LP gives us the optimal h for the linearized model, but we might want to take a smaller step to avoid overshooting. We can do a line search:
- Solve the LP to get h_lp
- Try h_new = (1-α)h_cur + α*h_lp for α in {1.0, 0.5, 0.25, 0.1}
- Pick the one with lowest C5

Actually, since the LP minimizes the linearized max, taking the full step (α=1) should work well when we're close to the solution. Let me just use the full step and see.

from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)

def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad

def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score

def lp_linearized_step(h_cur, target_sum, topk, step_frac=1.0):
    """Solve a linearized LP to get a step direction."""
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    
    bounds_lp = [(0.0, 1.0)] * n + [(-1e10, 1e10)]
    
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds_lp, method='highs')
    
    if res.success:
        h_lp = res.x[:n]
        h_new = (1.0 - step_frac) * h_cur + step_frac * h_lp
        return project_feasible(h_new, target_sum)
    return None

    lr = lr_init
    stagnation = 0
    
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)

        else:
            top_indices = np.arange(n_corr)

        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)

        else:
            w = w / w_sum

            lr *= 0.99

import numpy as np
from scipy.optimize import minimize, linprog
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(50):
        if diff > 0:
            free = h < 1.0 - 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.minimum(h[free] + per, 1.0)
        else:
            free = h > 1e-14
            nfree = free.sum()
            if nfree == 0:
                break
            per = diff / nfree
            h[free] = np.maximum(h[free] + per, 0.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_single_lag(h, lag_idx):
    n = len(h)
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad


def compute_grad_multi_lag(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n - lag])
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - h[m:])
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def active_set_smooth_obj(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        shifted = np.clip(alpha * (top_vals - c_ref), -500.0, 0.0)
        return c_ref + np.log(np.sum(np.exp(shifted))) / alpha
    else:
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def active_set_smooth_grad(x, alpha, topk):
    corr = overlap_corrs(x)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        c_ref = np.max(top_vals)
        exp_shifted = np.exp(np.clip(alpha * (top_vals - c_ref), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, top_idx, weights)
    else:
        c_max = np.max(corr)
        exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
        total = np.sum(exp_shifted)
        weights = exp_shifted / total
        return compute_grad_multi_lag(x, np.arange(n_corr), weights)


def slsqp_active_set(h_start, target_sum, alpha_topk_pairs, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_score = compute_c5(h_cur)
    for alpha, topk in alpha_topk_pairs:
        try:
            res = minimize(
                active_set_smooth_obj, h_cur, args=(alpha, topk),
                jac=active_set_smooth_grad, method='SLSQP',
                bounds=bounds, constraints=cons,
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target_sum)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def lp_linearized_step(h_cur, target_sum, topk):
    n = len(h_cur)
    corr = overlap_corrs(h_cur)
    n_corr = len(corr)
    if topk < n_corr:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    else:
        top_idx = np.arange(n_corr)
    m = len(top_idx)
    A = np.zeros((m, n + 1))
    b = np.zeros(m)
    for i, idx in enumerate(top_idx):
        g = compute_grad_single_lag(h_cur, idx)
        A[i, :n] = g
        A[i, n] = -1.0
        b[i] = g @ h_cur - corr[idx]
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(-1e6, 1e6)]
    res = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_lp, method='highs')
    if res.success:
        h_lp = res.x[:n]
        return project_feasible(h_lp, target_sum)
    return None


def lp_refinement(h_start, target_sum, max_iterations, topk, time_limit=None):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    stagnation = 0
    for it in range(max_iterations):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        h_lp = lp_linearized_step(h, target_sum, topk)
        if h_lp is None:
            break
        score = compute_c5(h_lp)
        if score < best_score - 1e-15:
            best_score = score
            best_h = h_lp.copy()
            h = h_lp
            stagnation = 0
        else:
            h_half = project_feasible(0.5 * h + 0.5 * h_lp, target_sum)
            sc_half = compute_c5(h_half)
            if sc_half < best_score - 1e-15:
                best_score = sc_half
                best_h = h_half.copy()
                h = h_half
                stagnation = 0
            else:
                stagnation += 1
                if stagnation > 10:
                    break
    return best_h, best_score


def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_multi_lag(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 3.0)
        else:
            stagnation += 1
            lr *= 0.99
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
