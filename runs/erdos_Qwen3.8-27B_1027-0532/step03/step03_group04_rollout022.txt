def get_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        if reg > 0:
            objective = t_var + reg * cp.sum_squares(h_var - 0.5)
        else:
            objective = t_var

        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False, eps=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.PDLP)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        c5_lp = compute_c5(h_new)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_new.copy()

        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 20:
            damping = max(0.02, damping * 0.8)
            stuck_count = 0

    return best_h, best_c5


def coordinate_descent(h_init, n, target, max_iter, time_limit, rng=None, restarts=1):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    if rng is None:
        rng = np.random.RandomState(42)

    for restart in range(restarts):
        local_iter = 0
        while local_iter < max_iter:
            if time.time() - start > time_limit:
                break
            local_iter += 1

            corr = np.correlate(h, 1.0 - h, mode='full')
            c_vals = corr * dx
            k_star = int(np.argmax(c_vals)) - n_half
            current_max = c_vals[k_star]

            i_arr = np.arange(n)
            grad = np.zeros(n)
            valid_pos = ((i_arr + k_star) >= 0) & ((i_arr + k_star) < n)
            if np.any(valid_pos):
                grad[valid_pos] += (1.0 - h[i_arr[valid_pos] + k_star]) * dx
            valid_neg = ((i_arr - k_star) >= 0) & ((i_arr - k_star) < n)
            if np.any(valid_neg):
                grad[valid_neg] -= h[i_arr[valid_neg] - k_star] * dx

            reductions = np.zeros(n)
            directions = np.zeros(n)
            pos_mask = grad > 1e-12
            neg_mask = grad < -1e-12
            reductions[pos_mask] = grad[pos_mask] * h[pos_mask]
            directions[pos_mask] = -1.0
            reductions[neg_mask] = -grad[neg_mask] * (1.0 - h[neg_mask])
            directions[neg_mask] = 1.0

            if np.max(reductions) < 1e-10:
                if restart < restarts - 1:
                    h = np.clip(h + rng.randn(n) * 0.08, 0, 1)
                    h = project_feasible(h, target)
                break

            best_i = int(np.argmax(reductions))
            direction = directions[best_i]

            found = False
            for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:
                if direction < 0:
                    step = h[best_i] * frac
                else:
                    step = (1.0 - h[best_i]) * frac
                if step < 1e-12:
                    continue
                h_try = h.copy()
                h_try[best_i] += direction * step
                h_try = project_feasible(h_try, target)
                c_new = compute_c5(h_try)
                if c_new < current_max - 1e-12:
                    h = h_try
                    if c_new < best_c5:
                        best_c5 = c_new
                        best_h = h.copy()
                    found = True
                    break

            if not found:
                if restart < restarts - 1:
                    h = np.clip(h + rng.randn(n) * 0.04, 0, 1)
                    h = project_feasible(h, target)
                else:
                    break

    return best_h, best_c5


def paired_cd(h_init, n, target, max_iter, time_limit, rng=None):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    if rng is None:
        rng = np.random.RandomState(123)

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        k_star = int(np.argmax(c_vals)) - n_half
        current_max = c_vals[k_star]

        i_arr = np.arange(n)
        grad = np.zeros(n)
        valid_pos = ((i_arr + k_star) >= 0) & ((i_arr + k_star) < n)
        if np.any(valid_pos):
            grad[valid_pos] += (1.0 - h[i_arr[valid_pos] + k_star]) * dx
        valid_neg = ((i_arr - k_star) >= 0) & ((i_arr - k_star) < n)
        if np.any(valid_neg):
            grad[valid_neg] -= h[i_arr[valid_neg] - k_star] * dx

        best_gain = 0
        best_inc = -1
        best_dec = -1
        best_max_step = 0.0

        top_k = 30
        gains_sorted = np.argsort(np.abs(grad))[::-1][:top_k]

        for idx_a in range(len(gains_sorted)):
            i = gains_sorted[idx_a]
            for idx_b in range(len(gains_sorted)):
                if idx_b == idx_a:
                    continue
                j = gains_sorted[idx_b]
                max_step_inc = min(1.0 - h[i], h[j])
                if max_step_inc > 1e-8:
                    gain = (grad[j] - grad[i]) * max_step_inc
                    if gain > best_gain:
                        best_gain = gain
                        best_inc = i
                        best_dec = j
                        best_max_step = max_step_inc
                max_step_dec = min(h[i], 1.0 - h[j])
                if max_step_dec > 1e-8:
                    gain = (grad[i] - grad[j]) * max_step_dec
                    if gain > best_gain:
                        best_gain = gain
                        best_inc = j
                        best_dec = i
                        best_max_step = max_step_dec

        if best_inc < 0 or best_gain < 1e-10:
            break

        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
            d = best_max_step * frac
            h_try = h.copy()
            h_try[best_inc] += d
            h_try[best_dec] -= d
            h_try = np.clip(h_try, 0, 1)
            c_new = compute_c5(h_try)
            if c_new < current_max - 1e-12:
                h = h_try
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                break

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def multi_start_lp(h_base, n, target, num_starts, time_limit, rng):
    start = time.time()
    best_h = h_base.copy()
    best_c5 = compute_c5(h_base)

    for s in range(num_starts):
        if time.time() - start > time_limit:
            break
        noise_level = 0.15 * (s + 1) / num_starts
        h_pert = np.clip(h_base + rng.randn(n) * noise_level, 0, 1)
        h_pert = project_feasible(h_pert, target)
        damping = min(0.9, 0.4 + 0.5 * s / max(1, num_starts - 1))
        single_time = max(20, min(120, (time_limit - (time.time() - start)) / (num_starts - s)))
        h_lp, c5_lp = alternating_lp(h_pert, n, target,
                                      max_iter=5000, time_limit=single_time,
                                      damping_init=damping, reg=0.0)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Primary LP from initial point (no regularization)
    if tleft() > 300:
        lp_time = min(250, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                      max_iter=15000, time_limit=lp_time,
                                      damping_init=0.7, reg=0.0)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Multiple LP restarts with different perturbations
    if tleft() > 300:
        mp_time = min(250, tleft() * 0.3)
        h_mp, c5_mp = multi_start_lp(best_h, n, target, 4, mp_time, rng)
        if c5_mp < best_score:
            best_score = c5_mp
            best_h = h_mp

    # Phase 3: Coordinate descent polish
    if tleft() > 150:
        cd_time = min(120, tleft() * 0.35)
        h_cd, c5_cd = coordinate_descent(best_h, n, target,
                                          max_iter=50000, time_limit=cd_time,
                                          rng=rng, restarts=5)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 4: Paired coordinate descent
    if tleft() > 100:
        pcd_time = min(80, tleft() * 0.25)
        h_pcd, c5_pcd = paired_cd(best_h, n, target,
                                   max_iter=20000, time_limit=pcd_time, rng=rng)
        if c5_pcd < best_score:
            best_score = c5_pcd
            best_h = h_pcd

    # Phase 5: SLSQP with increasing alpha
    if tleft() > 100:
        h_refined = slsqp_round(best_h, n, target,
                                 [50, 200, 500, 1500, 3000, 5000], 2000, 1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 6: LP again from improved point (no reg, low damping)
    if tleft() > 120:
        lp_time = min(100, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=10000, time_limit=lp_time,
                                      damping_init=0.3, reg=0.0)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: More coordinate descent
    if tleft() > 80:
        cd_time = min(60, tleft() * 0.35)
        h_cd, c5_cd = coordinate_descent(best_h, n, target,
                                          max_iter=50000, time_limit=cd_time,
                                          rng=rng, restarts=10)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Phase 8: SLSQP again
    if tleft() > 80:
        h_refined = slsqp_round(best_h, n, target,
                                 [500, 1000, 2000, 5000, 8000], 3000, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 9: Final paired CD
    if tleft() > 50:
        pcd_time = min(40, tleft() * 0.4)
        h_pcd, c5_pcd = paired_cd(best_h, n, target,
                                   max_iter=30000, time_limit=pcd_time, rng=rng)
        if c5_pcd < best_score:
            best_score = c5_pcd
            best_h = h_pcd

    # Phase 10: Try higher resolution
    if tleft() > 250:
        for n_try in [500, 700, 900]:
            if tleft() < 180:
                break
            if n_try % 2 != 0:
                continue
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(120, tleft() * 0.3)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                          max_iter=5000, time_limit=lp_time,
                                          damping_init=0.6, reg=0.0)
            if tleft() > 60 and c5_lp < best_score:
                cd_time = min(40, tleft() * 0.2)
                h_cd, c5_cd = coordinate_descent(h_lp, n_try, target_try,
                                                  max_iter=20000, time_limit=cd_time,
                                                  rng=rng, restarts=3)
                if c5_cd < best_score:
                    best_score = c5_cd
                    best_h = h_cd
                    n = n_try
                    target = target_try
                elif c5_lp < best_score:
                    best_score = c5_lp
                    best_h = h_lp
                    n = n_try
                    target = target_try

    # Phase 11: Final SLSQP polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                   [2000, 5000, 10000], 2000, 1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 12: Final coordinate descent
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cd, c5_cd = coordinate_descent(best_h, n_use, target_use,
                                          max_iter=30000, time_limit=min(15, tleft() * 0.6),
                                          rng=rng, restarts=8)
        if c5_cd < best_score:
            best_score = c5_cd
            best_h = h_cd

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
