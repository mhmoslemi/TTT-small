import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng,
                time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 15)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]

    for i in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
            else:
                break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        move_type = rng.randint(5)
        h_new = h.copy()
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            # Single swap
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            # Double swap
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            h_new[ones_cache[idx1[0]]] = 0
            h_new[ones_cache[idx1[1]]] = 0
            h_new[zeros_cache[idx0[0]]] = 1
            h_new[zeros_cache[idx0[1]]] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) >= 3 and len(zeros_cache) >= 3:
            # Triple swap
            idx1 = rng.choice(len(ones_cache), 3, replace=False)
            idx0 = rng.choice(len(zeros_cache), 3, replace=False)
            for idx in idx1:
                h_new[ones_cache[idx]] = 0
            for idx in idx0:
                h_new[zeros_cache[idx]] = 1
            moved = True
        elif move_type == 3 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            # Proximal swap - prefer nearby positions
            i1 = ones_cache[rng.randint(len(ones_cache))]
            dist = np.abs(zeros_cache - i1).astype(float)
            # Bias towards nearby
            weights = 1.0 / (dist + 1.0)
            weights += rng.random(len(zeros_cache)) * 0.1
            i0 = zeros_cache[np.random.choice(len(zeros_cache), p=weights/weights.sum())]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 4 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            # Block swap - swap adjacent blocks
            block_size = rng.randint(2, 5)
            i1 = ones_cache[rng.randint(len(ones_cache))]
            # Find a nearby zero block
            candidates = zeros_cache[np.abs(zeros_cache - i1) < block_size * 3]
            if len(candidates) > 0:
                i0 = candidates[rng.randint(len(candidates))]
                h_new[i1] = 0
                h_new[i0] = 1
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def focused_local_search(h, target_sum, n_iter, rng, time_limit=None, start_time=None):
    """Local search that focuses on the shifts with highest correlation."""
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(n_iter):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        
        corr = compute_corr(h)
        corr_vals = corr * dx
        max_val = np.max(corr_vals)
        
        # Find shifts near the maximum
        threshold = max_val * 0.98
        critical_shifts = np.where(corr_vals > threshold)[0]
        
        # For the most critical shift, try to improve
        worst_shift = np.argmax(corr_vals)
        
        # Find bad pairs for this shift
        # corr index: for mode='full', index k corresponds to shift (k - (n-1))
        # corr[m] = sum_i h[i] * (1-h[i+m-(n-1)])
        # So shift = worst_shift - (n-1)
        shift = worst_shift - (n - 1)
        
        if shift > 0:
            # h[i] * (1-h[i+shift]) - want to reduce this
            # Bad pairs: h[i]=1 and h[i+shift]=0 for i in [0, n-shift-1]
            bad_ones = np.where(h[:n-shift] == 1)[0]
            bad_zeros_positions = bad_ones + shift
            bad_zeros = np.where(h[bad_zeros_positions] == 0)[0]
            # Actually those should all be 0 by definition
            # So we want to flip some h[i+shift] to 1, or h[i] to 0
        elif shift < 0:
            s = -shift
            # h[i] * (1-h[i-s]) = h[i+s] * (1-h[i]) for i in [0, n-s-1]
            bad_ones = np.where(h[s:] == 1)[0] + s
            bad_zeros = bad_ones - s
            # h[i]=1 and h[i-s]=0, so flip h[i-s] to 1 or h[i] to 0
            
        if shift == 0:
            continue
        
        s = abs(shift)
        if s >= n:
            continue
            
        if shift > 0:
            bad_ones = np.where(h[:n-s] == 1)[0]
            bad_zeros_idx = bad_ones + s
        else:
            bad_ones = np.where(h[s:] == 1)[0] + s
            bad_zeros_idx = bad_ones - s
        
        if len(bad_ones) == 0:
            continue
        
        # Try flipping a random bad pair
        idx = rng.randint(len(bad_ones))
        pos_one = bad_ones[idx]
        pos_zero = bad_zeros_idx[idx]
        
        # We want to flip pos_zero to 1 and pos_one to 0
        # But we need to put the "1" somewhere else
        # Actually, we're doing a swap: h[pos_one] 1->0, h[pos_zero] 0->1
        h[pos_one] = 0
        h[pos_zero] = 1
        
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
        else:
            # Revert
            h[pos_zero] = 0
            h[pos_one] = 1
    
    return h, best_c5

def smooth_obj(h, alpha, corr_cache=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

_cons_n = 0
_cons_target = 0.0

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def multistart_local_search(h_start, target_sum, n_starts, n_iter_each, rng,
                            time_limit=None, start_time=None):

    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    
    for s in range(n_starts):

        zeros = np.where(h_pert == 0)[0]

        flip_zeros = rng.choice(zeros, min(n_flips, len(zeros)), replace=False)
        h_pert[flip_ones] = 0
        h_pert[flip_zeros] = 1

        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

def project_feasible(h, target_sum):

def project_binary(h, target_sum):

    n_use = len(best_h)
    target_use = n_use / 2.0

            best_score = sc
