from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def swap_delta(corr, h, i, j, n):
    """Compute the change in correlation when swapping h[i]=1 with h[j]=0."""
    half = n - 1
    s_min = max(-i, i - half, -j, j - half)
    s_max = min(half - i, i, half - j, j)
    if s_min > s_max:
        return np.array([]), s_min, s_max
    s_range = np.arange(s_min, s_max + 1)
    delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
    return delta, s_min, s_max

def local_search_best_improvement(h, n_evals_per_step, n_max_steps, rng):
    """Local search that evaluates n_evals_per_step random swaps per step and takes the best."""
    n = len(h)
    h = h.astype(np.float64).copy()
    corr = compute_corr(h)
    current_max = float(np.max(corr))
    half = n - 1
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for step in range(n_max_steps):
        # Sample random candidates
        candidates = []
        for _ in range(n_evals_per_step):
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            candidates.append((i, j))
        
        best_gain = 0
        best_pair = None
        best_new_max = current_max

            if i == j:

            delta, s_min, s_max = swap_delta(corr, h, i, j, n)

                continue

            new_vals = corr[k_range] + delta
            max_in = float(np.max(new_vals))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    try:
        h_init_global = np.array(initial_h_values, dtype=float)
    except Exception:
        h_init_global = None
    best_c5 = 1.0
    best_h = None
    best_n = 180
    n_candidates = [184, 180, 186, 182]
    for n_try in n_candidates:
        elapsed = time.time() - start_time
        time_left = budget_s - elapsed
        if time_left < 60:
            break
        target_sum = n_try / 2.0
        if h_init_global is not None and len(h_init_global) == n_try:
            h_start = project_onto_constraint(h_init_global.copy(), target_sum)
        elif h_init_global is not None and len(h_init_global) > 0:
            x_old = np.linspace(0, 2, len(h_init_global), endpoint=False) + 1.0 / len(h_init_global)
            x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
            h_start = np.interp(x_new, x_old, h_init_global)
            h_start = project_onto_constraint(h_start, target_sum)
        else:
            h_start = (rng.random(n_try) < 0.5).astype(float)
            h_start = project_onto_constraint(h_start, target_sum)
        h_bin_start = to_binary(h_start)
        binary_time_budget = min(time_left * 0.45, 250)
        h_cur = h_bin_start.copy()
        n_rounds = 0
        stale_rounds = 0
        prev_best = (2.0 / n_try) * float(np.max(compute_corr(h_cur)))
        while True:
            if time.time() - start_time > budget_s - binary_time_budget - 40:
                break
            if n_rounds >= 60:
                break
            h_cur, cur_max = iterated_ls(h_cur, n_try * 3, 250, rng)
            cur_c5 = (2.0 / n_try) * cur_max
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h_cur.copy()
                best_n = n_try
            if cur_c5 < prev_best - 1e-10:
                prev_best = cur_c5
                stale_rounds = 0
            else:
                stale_rounds += 1
            if stale_rounds > 15:
                h_cur = to_binary(h_start + 0.1 * rng.standard_normal(n_try))
                stale_rounds = 0
            n_rounds += 1
        elapsed = time.time() - start_time
        if budget_s - elapsed > 40 and best_n == n_try:
            h_polish = best_h + 0.015 * rng.standard_normal(n_try)
            h_polish = project_onto_constraint(h_polish, target_sum)
            for eps in [0.01, 0.003, 0.001, 0.0003, 0.0001]:
                if time.time() - start_time > budget_s - 25:
                    break
                h_polish, c5 = slsqp_optimize(h_polish, eps, 250)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_polish.copy()
                    best_n = n_try
    elapsed = time.time() - start_time
    if budget_s - elapsed > 50 and best_h is not None:
        n_best = best_n
        h_bin_final = to_binary(best_h)
        h_final, final_max = full_local_search(h_bin_final, 30)
        final_c5_bin = (2.0 / n_best) * final_max
        if final_c5_bin < best_c5:
            best_c5 = final_c5_bin
            best_h = h_final.copy()
        if budget_s - (time.time() - start_time) > 25:
            h_cp = best_h + 0.005 * rng.standard_normal(n_best)
            h_cp = project_onto_constraint(h_cp, n_best / 2.0)
            h_cp, c5 = slsqp_optimize(h_cp, 0.0005, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cp.copy()
            if budget_s - (time.time() - start_time) > 10:
                h_cp2 = best_h + 0.002 * rng.standard_normal(n_best)
                h_cp2 = project_onto_constraint(h_cp2, n_best / 2.0)
                h_cp2, c5 = slsqp_optimize(h_cp2, 0.0001, 200)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_cp2.copy()
    if best_h is None:
        best_h = (rng.random(180) < 0.5).astype(float)
        best_h = project_onto_constraint(best_h, 90.0)
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
