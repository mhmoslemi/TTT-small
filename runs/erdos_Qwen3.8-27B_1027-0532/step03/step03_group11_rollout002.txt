import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_obj_grad(h, eps):
    """Compute smooth max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    v = corr / eps
    v_shift = v - v.max()
    exp_v = np.exp(v_shift)
    w = exp_v / exp_v.sum()
    obj = eps * (v.max() + np.log(exp_v.sum()))
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        k = s + n - 1
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        if s >= 0:
            m = n - s
            grad[:m] += wk * omh[s:s+m]
            grad[s:s+m] -= wk * h[:m]
        else:
            r = -s
            m = n - r
            grad[r:r+m] += wk * omh[:m]
            grad[:m] -= wk * h[r:r+m]
    return obj, grad

def pgd_optimize(h_init, eps_schedule, max_iter_per_eps, time_limit, start_time):
    """Projected gradient descent with momentum."""
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.copy(), target_sum)
    v = np.zeros(n)  # momentum
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for eps in eps_schedule:
        # Estimate Lipschitz constant (rough)
        # For our problem, L ≈ 4dx * max_w * n (very rough)
        # Use backtracking instead
        lr = 1.0 / (4.0 * (2.0 / n) * n)  # rough initial
        lr = min(lr, 2.0)
        
        for it in range(max_iter_per_eps):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            
            obj, grad = smooth_obj_grad(h, eps)
            
            # Momentum update
            v = beta * v - lr * grad

            h_new = project_onto_constraint(h_new, target_sum)

            alpha = 1.0

                    h_new = h_trial
                    obj_new = obj_trial

                obj_new = obj
                v *= 0.5  # reduce momentum
            
            h = h_new
            
            # Check actual C5

            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
