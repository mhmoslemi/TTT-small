import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0, 1)

def gradient_all_lags(h):
    """Compute the full gradient of each O(k) with respect to h.
    Returns a matrix G of shape (2n-1, n) where G[m, i] = dO(k)/dh[i] for lag k = m-(n-1)."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        k = m - (n-1)
        if k >= 0:
            if k == 0:
                G[m, :] = 1.0 - 2.0*h
            else:
                G[m, :n-k] += -h[k:]       # from (1-h[i]) factor: -h[i+k]
                G[m, k:] += 1.0 - h[:n-k]   # from h[i+k] factor: (1-h[i-k]) wait...
        else:
            m2 = -k
            if m2 == 0:
                G[m, :] = 1.0 - 2.0*h

                G[m, :n-m2] += 1.0 - h[m2:]   # from h[i] factor: (1-h[i+m2])
                G[m, m2:] += -h[:n-m2]         # from (1-h[i]) factor: -h[i+m2-... ]
    return G
