def _nesterov_smooth(h0, n, target_sum, n_iters, alpha, sg_m, sg_alpha, deadline):

def _grad_single(h, corr_idx, n, dx):
    lag = corr_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

        ones = np.where(h_try == 1.0)[0]
        flip_from = zeros[rng.choice(len(zeros), min(n_flip, len(zeros)), replace=False)]
        flip_to = ones[rng.choice(len(ones), min(n_flip, len(ones)), replace=False)]
        h_try[flip_from] = 1.0
        h_try[flip_to] = 0.0
        h_g, c5_g = _binary_greedy(h_try, n, max_rounds=100, deadline=deadline)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
    return best_h, best_c5

def _fractal_start(n, rng):
    h = np.zeros(n)
    for scale in [10, 5, 2, 1]:
        block = n // scale
        for b in range(block):
            if rng.random() < 0.5:
                h[b * scale:(b + 1) * scale] = 1.0
    return _to_binary(h, n // 2)

def _interpolate(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0, 2.0, n_src)
    x_dst = np.linspace(0, 2.0, n_dst)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 150
        h_parent = _fractal_start(n_parent, rng)

    best_h = h_parent.copy()
    best_c5 = _c5_of(h_parent)
    best_n = n_parent
    target = n_parent // 2.0

    h_bin = _to_binary(h_parent, n_parent // 2)
    c5_bin = _c5_of(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if time.time() < deadline - 500:
        h_g, c5_g = _binary_greedy(h_bin, n_parent, max_rounds=400, deadline=deadline - 500)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    if time.time() < deadline - 400:
        h_ms, c5_ms = _multi_start_binary(best_h, n_parent, n_parent // 2,
                                          3, deadline - 400, rng)
        if c5_ms < best_c5:
            best_c5 = c5_ms
            best_h = h_ms.copy()

    if time.time() < deadline - 300:
        h_sa, c5_sa = _sa_binary(best_h, n_parent, n_parent // 2,
                                 5000, 0.5, 0.01, deadline - 250, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    if time.time() < deadline - 200:
        h_g2, c5_g2 = _binary_greedy(best_h, n_parent, max_rounds=300,
                                      deadline=deadline - 200)
        if c5_g2 < best_c5:
            best_c5 = c5_g2
            best_h = h_g2.copy()

    if time.time() < deadline - 200:
        ns = _interpolate(best_h, best_n, n_parent)
        ns = _fix_sum(np.clip(ns, 0.0, 1.0), n_parent / 2.0)
        h_ng, c5_ng = _nesterov_smooth(ns, n_parent, n_parent / 2.0,
                                       200, 0.4, 8, 60.0, deadline - 180)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    if time.time() < deadline - 150:
        h_sl = _slsqp_smooth(best_h, best_n, best_n / 2.0, 200, 500, deadline - 130)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    for n_try in [n_parent - 10, n_parent + 10, n_parent + 20]:
        if n_try < 60 or n_try > 250 or n_try % 2 != 0:
            continue
        if time.time() > deadline - 80:
            break
        h_try = _interpolate(best_h, best_n, n_try)
        h_try = _fix_sum(np.clip(h_try, 0.0, 1.0), n_try / 2.0)
        h_try_bin = _to_binary(h_try, n_try // 2)
        h_g3, c5g3 = _binary_greedy(h_try_bin, n_try, max_rounds=100,
                                     deadline=deadline - 60)
        if c5g3 < best_c5:
            best_c5 = c5g3
            best_h = h_g3.copy()
            best_n = n_try

    if time.time() < deadline - 60:
        h_final_bin = _to_binary(best_h, best_n // 2)
        h_gf, c5_gf = _binary_greedy(h_final_bin, best_n, max_rounds=200,
                                      deadline=deadline - 50)
        if c5_gf < best_c5:
            best_c5 = c5_gf
            best_h = h_gf.copy()

    if time.time() < deadline - 40:
        h_cf = _coord_binary(best_h, best_n, best_n // 2, 30, deadline - 35)
        if h_cf < best_c5 if isinstance(h_cf, float) else False:
            pass
        h_cd, c5_cd = _coord_binary(best_h, best_n, best_n // 2, 30, deadline - 35)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time.time() < deadline - 30:
        h_sl2 = _slsqp_smooth(best_h, best_n, best_n / 2.0, 150, 2000, deadline - 25)
        c5_sl2 = _c5_of(h_sl2)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2.copy()

    if time.time() < deadline - 15:
        h_ng2, c5_ng2 = _nesterov_smooth(best_h, best_n, best_n / 2.0,
                                         300, 0.3, 10, 80.0, deadline - 10)
        if c5_ng2 < best_c5:
            best_c5 = c5_ng2
            best_h = h_ng2.copy()

    if time.time() < deadline - 5:
        h_sl3 = _slsqp_smooth(best_h, best_n, best_n / 2.0, 300, 3000, deadline - 3)
        c5_sl3 = _c5_of(h_sl3)
        if c5_sl3 < best_c5:
            best_c5 = c5_sl3
            best_h = h_sl3.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
