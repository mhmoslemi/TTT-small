    corr = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

def _grad_weighted(h, k_indices, weights, dx):
    n = len(h)
    g = np.zeros(n)
    for idx in range(len(k_indices)):
        k_idx = k_indices[idx]

                g[:m] += w * (-h[lag:lag+m])

            m = n - j
            if m > 0:
                g[:m] += w * (1.0 - h[j:])
                g[j:] += w * (-h[:m])
    return g * dx

    y = h0.copy()

    global_iters = 0

        if time.time() > deadline:
            break
        alpha = 3.0 * (100.0 ** (phase / max(n_phases - 1, 1)))
        alpha = min(alpha, 300.0)
        m_lags = max(3, min(20, 3 + 17 * phase // n_phases))
        
        for it in range(iters_per_phase):
            if time.time() > deadline:

            global_iters += 1
            corr = np.correlate(y, 1.0 - y, 'full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

def _grad_weighted(h, k_indices, weights, dx):

    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    dx = 2.0 / n
    glob_it = 0
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        alpha = 3.0 * (100.0 ** (phase / max(n_phases - 1, 1)))
        alpha = min(alpha, 300.0)
        m_lags = max(3, min(20, 3 + 17 * phase // n_phases))
        
        for it in range(iters_per_phase):
            if time.time() > deadline:
                break
            glob_it += 1
            corr = np.correlate(y, 1.0 - y, 'full')
            max_c = np.max(corr)
            weights = np.exp(alpha * (corr - max_c))
            tw = weights.sum()
            if tw < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights /= tw
            top_idx = np.argsort(weights)[::-1][:m_lags]
            w_top = weights[top_idx].copy()
            w_top /= w_top.sum()

            step = 0.25 / np.sqrt(float(glob_it) + 1.0)

            y = y_new

            y = best.copy()

    corr = np.correlate(x, 1.0 - x, 'full')

def _slsqp_eq(x, target_sum):
    return x.sum() - target_sum

def _slsqp_eq_jac(x, target_sum):
    return np.ones(len(x))

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):

4. Focus optimization on reducing the contributions from the worst contributors

    starts.append(_fix_sum(1.0 - h0, target_sum))
    for s in [0.1, 0.2, 0.4]:
        starts.append(_random_perturb(h0, n, target_sum, s, rng))
    for h_start in starts:
        if time.time() > deadline:
            break
        alloc = max(30, (deadline - time.time()) / max(len(starts), 1))
        h_try, c5_try = _nesterov_opt(h_start, n, target_sum,
                                       min(time.time() + alloc, deadline),
                                       n_phases=12, iters_per_phase=20)
        if c5_try < best_val:
            best_val = c5_try
            best_h = h_try
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)
    rng = np.random.RandomState(seed)
    target_sum = n_init / 2.0
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    if time.time() < deadline - 350:
        h_ns, c5_ns = _multi_start_nesterov(h_init, n_init, target_sum, t_start + 450, rng)
        if c5_ns < best_c5:
            best_c5 = c5_ns
            best_h = h_ns
    if time.time() < deadline - 250:
        h_n = _nesterov_opt(best_h, n_init, target_sum, t_start + 600, n_phases=20, iters_per_phase=25)
        if _c5_of(h_n) < best_c5:
            best_c5 = _c5_of(h_n)
            best_h = h_n
    if time.time() < deadline - 180:
        h_s = _slsqp_opt(best_h, n_init, target_sum, 50, 400)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    if time.time() < deadline - 120:
        h_s = _slsqp_opt(best_h, n_init, target_sum, 120, 800)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    if time.time() < deadline - 90:
        h_cd = _coord_descent_binary(best_h, n_init, target_sum, 8)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    if time.time() < deadline - 60:
        h_cf = _coord_descent_frac(best_h, n_init, target_sum, 5)
        c5_cf = _c5_of(h_cf)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf
    for delta in [20, -20, 40, -40, 60]:
        n_try = n_init + delta
        if n_try < 100 or n_try > 260 or n_try % 2 != 0:
            continue
        if time.time() > deadline - 100:
            break
        t_sum = n_try / 2.0
        h_interp = np.interp(np.linspace(0, n_init - 1, n_try), np.arange(n_init), best_h)
        h_interp = _fix_sum(np.clip(h_interp, 0.0, 1.0), t_sum)
        if time.time() < deadline - 60:
            h_nt, c5_nt = _nesterov_opt(h_interp, n_try, t_sum, t_start + 1000, n_phases=15, iters_per_phase=20)
            if c5_nt < best_c5:
                best_c5 = c5_nt
                best_h = h_nt
                n_init = n_try
                target_sum = t_sum
    n_best = len(best_h)
    t_best = n_best / 2.0
    if time.time() < deadline - 50:
        h_f = _nesterov_opt(best_h, n_best, t_best, t_start + 1100, n_phases=15, iters_per_phase=25)
        c5_f = _c5_of(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f
    if time.time() < deadline - 35:
        h_s = _slsqp_opt(best_h, n_best, t_best, 200, 2000)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    if time.time() < deadline - 20:
        h_cd = _coord_descent_binary(best_h, n_best, t_best, 10)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
    if time.time() < deadline - 10:
        h_cf = _coord_descent_frac(best_h, n_best, t_best, 5)
        c5_cf = _c5_of(h_cf)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf
    if time.time() < deadline - 4:
        h_s = _slsqp_opt(best_h, n_best, t_best, 350, 3000)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    best_h = _fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, len(best_h))
