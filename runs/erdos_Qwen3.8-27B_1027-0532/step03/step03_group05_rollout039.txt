            # OK so for k < 0 (k' = -k > 0):
            #   out[n-1-k'] = sum_{n=k'}^{n-1} h[n-k'] * (1-h[n])
            # Let m = n-k': out[n-1-k'] = sum_{m=0}^{n-1-k'} h[m] * (1-h[m+k'])
            
            # d/dh[j] of sum_{m=0}^{n-1-k'} h[m]*(1-h[m+k']):
            # h[j] appears as h[m] when m=j (if j < n-k'): contribution (1-h[j+k'])
            # h[j] appears as (1-h[m+k']) when m+k'=j, i.e., m=j-k' (if j-k' >= 0): contribution -h[j-k']
            # So grad_h[j] += w * (1-h[j+k']) for j in [0, n-1-k']
            #    grad_h[j] -= w * h[j-k'] for j in [k', n-1]
            
            kp = -k
            grad_h[:n-kp] += w * (1.0 - h[kp:n])
            grad_h[kp:] -= w * h[:n-kp]
    
    # Now apply the gradient step
    gn = np.linalg.norm(grad_h)
    if gn < 1e-12:
        return h, 0.0
    
    # We want to DECREASE the objective, so step in the NEGATIVE gradient direction
    h_new = h - eta * grad_h / gn
    h_new = project_feasible(h_new, target_sum)
    return h_new, gn

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    count = float(np.sum(mask))
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def compute_smoothed_grad(h, alpha):
    """Compute gradient of the softmax-smoothed max overlap."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    shifted = alpha * (corr - max_val)
    shifted = np.clip(shifted, -500.0, 0.0)

    sw = np.sum(weights)

    # Using the formula: grad[j] = sum_{m} W[j-m+n-1] * (1-h[m]) - sum_m W[m-j+n-1] * h[m]

    return grad, np.linalg.norm(grad)

    """FISTA (accelerated proximal gradient) for the smoothed objective."""

    best = h.copy()
    best_score = compute_c5(h)
    
    # FISTA parameters
    L = alpha * n  # Lipschitz constant upper bound (rough)
    step = 1.0 / L
    t = 1.0

        if gn < 1e-14:

        x_new = y - step * grad
        # Proximal step (projection onto feasible set)
        x_new = project_feasible(x_new, target)

        score = compute_c5(x_new)
        if score < best_score - 1e-14:

            best = x_new.copy()

        y = x_new + ((t - 1.0) / t_new) * (x_new - h)
        t = t_new
        h = x_new
        
        # Adaptively adjust step size
        if score < compute_c5(y):
            step = min(step * 1.1, 1.0 / (L * 0.5))
    
    return best, best_score

def project_feasible(h, target_sum):

def compute_smoothed_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights = weights / sw
    
    grad = np.zeros(n)

def slsqp_smooth(h, target, time_limit):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)

    p_values = [8, 32, 128]

    for pi, p in enumerate(p_values):
        remaining = time_limit - (time.time() - start)
        if remaining < 15:
            break
        time_per = remaining / (len(p_values) - pi)
        maxiter = int(max(80, time_per * 2.5))

        def _p_obj(x, p_val=p, _dx=dx, _n=n):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return float(_dx * np.sum(np.abs(corr) ** p_val) ** (1.0 / p_val))

        def _eq_fun(x, _t=target):
            return np.sum(x) - _t

        def _eq_jac(x, _n=n):
            return np.ones(_n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

        try:
            res = minimize(_p_obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def random_restart_search(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    target_sum = target

    attempt = 0
    while time.time() - start < time_limit and attempt < 20:
        perturbation_scale = 0.03 / (1.0 + 0.3 * attempt)
        h_pert = best + rng.randn(n) * perturbation_scale
        h_pert = project_feasible(h_pert, target_sum)
        h_local, s_local = fista_optimize(h_pert, target_sum,
                                          time_limit=min(20, time_limit - (time.time() - start)),
                                          alpha=200)
        if s_local < best_score:
            best_score = s_local
            best = h_local
        attempt += 1

    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    target0 = n0 / 2.0

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: FISTA with multiple alphas at n=400
    if tleft() > 220:
        h1, s1 = multi_alpha_fista(h0, target0, time_limit=min(200, tleft()),
                                   alphas=[50, 100, 200])
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: SLSQP with p-norm continuation
    if tleft() > 150:
        h2, s2 = slsqp_smooth(best_h, target0, time_limit=min(130, tleft()))
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Pairwise swap polish
    if tleft() > 100:
        h3, s3 = pairwise_swap_polish(best_h, target0, time_limit=min(90, tleft()), seed=seed)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Upsample to n=800 and optimize
    if tleft() > 350:
        n_up = 2 * n0
        x_old = np.linspace(0.0, 2.0, n0)
        x_new = np.linspace(0.0, 2.0, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        target_up = n_up / 2.0
        h_up = project_feasible(h_up, target_up)

        h4, s4 = multi_alpha_fista(h_up, target_up, time_limit=min(300, tleft()),
                                   alphas=[50, 100, 200, 400])
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Coordinate descent at current resolution
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h5, s5 = coordinate_polish(best_h, target_cur, time_limit=min(100, tleft()), seed=seed + 1)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: More FISTA with high alphas
    if tleft() > 180:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h6, s6 = multi_alpha_fista(best_h, target_cur, time_limit=min(160, tleft()),
                                   alphas=[200, 400, 800])
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Random restart exploration
    if tleft() > 120:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h7, s7 = random_restart_search(best_h, target_cur, time_limit=min(110, tleft()), seed=seed + 2)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final pairwise swap + coordinate polish
    if tleft() > 60:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h8, s8 = pairwise_swap_polish(best_h, target_cur, time_limit=min(40, max(20, tleft() - 10)), seed=seed + 3)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    if tleft() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h9, s9 = coordinate_polish(best_h, target_cur, time_limit=max(15, min(25, tleft() - 5)), seed=seed + 4)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Final feasibility check
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, n_final
