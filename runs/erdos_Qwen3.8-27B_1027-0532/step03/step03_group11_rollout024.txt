For binary h:
- If h[j-k]=0, h[j+k]=0: change = dx*(1-0-0) = +dx (bad)
- If h[j-k]=1, h[j+k]=0: change = dx*(1-0-1) = 0
- If h[j-k]=0, h[j+k]=1: change = dx*(1-1-0) = 0
- If h[j-k]=1, h[j+k]=1: change = dx*(1-1-1) = -dx (good)

So flipping 0→1 is beneficial for lag k only if both neighbors are 1.

This gives us a way to compute the effect of a flip on ALL lags simultaneously!

For a flip of position j from 1 to 0:
- For each lag k (where the indices are valid):
  - change[k] = dx * (h[j-k] + h[j+k] - 1)

The new max overlap is max(old_max, max over k of (old_corr[k] + change[k]))

This is still O(n) per flip to compute (since there are 2n-1 lags), but we can vectorize it.

Actually, for the targeted search, I want to find the flip that most reduces the current maximum. The current maximum is at some lag k*. If I flip position j, the new value at lag k* is:
old_corr[k*] + dx*(h[j-k*] + h[j+k*] - 1)

So the flip helps at lag k* if h[j-k*]=0 and h[j+k*]=0.

But we also need to make sure other lags don't increase above the old max.

This is getting complex. Let me just implement a version that:
1. Identifies the worst lag k*
2. For each 1-position j where h[j-k*]=0 and h[j+k*]=0, compute the new max after flipping
3. Pick the best such flip

This is O(n) candidates × O(n) to check = O(n²) per round, which is fine for n=180.

```python
import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def get_nft(n):
    N = 2 * n - 1
    Nft = 1
    while Nft < N:
        Nft *= 2
    return Nft

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    Nft = get_nft(n)
    F_h = fft(h, Nft)
    F_b = fft(b, Nft)
    r = np.real(ifft(F_h * np.conj(F_b)))
    full_corr = np.concatenate([r[Nft - (n - 1):], r[:n]]) * dx
    return full_corr

def compute_c5(h):
    return float(np.max(compute_c5_full(h)))

def smooth_objective(h, eps, Nft):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    F_h = fft(h, Nft)
    F_b = fft(b, Nft)
    r = np.real(ifft(F_h * np.conj(F_b)))
    full_corr = np.concatenate([r[Nft - (n - 1):], r[:n]]) * dx
    cmax = np.max(full_corr)
    x = (full_corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps, Nft):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    F_h = fft(h, Nft)
    F_b = fft(b, Nft)
    r = np.real(ifft(F_h * np.conj(F_b)))
    full_corr = np.concatenate([r[Nft - (n - 1):], r[:n]]) * dx
    cmax = np.max(full_corr)
    x = (full_corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-15:
            continue
        k = m - (n - 1)
        # First term: (1-h[j+k]) for valid j
        j_min1 = max(0, -k)
        j_max1 = min(n - 1, n - 1 - k)
        if j_min1 <= j_max1:
            idx = np.arange(j_min1, j_max1 + 1)
            grad[idx] += wm * dx * (1.0 - h[idx + k])
        # Second term: -h[j-k] for valid j
        j_min2 = max(0, k)
        j_max2 = min(n - 1, n - 1 + k)
        if j_min2 <= j_max2:
            idx = np.arange(j_min2, j_max2 + 1)
            grad[idx] -= wm * dx * h[idx - k]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def lbfgs_optimize(h_init, eps, Nft, maxiter=500):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps, Nft),
                          jac=smooth_gradient, method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
```

def flip_delta(h, j, direction, dx, n):
    # direction: -1 for 1→0, +1 for 0→1
    k = np.arange(-(n-1), n)  # all lags
    # delta[m] = direction * dx * (h[j-k] + h[j+k] - 1) for 1→0
    # Actually let me redo this.
    # For 1→0: Δ[m] = dx * (h[j-k] + h[j+k] - 1)
    # For 0→1: Δ[m] = dx * (1 - h[j+k] - h[j-k]) = -Δ_1to0[m]

    jk_minus = j - k  # positions j-k
    jk_plus = j + k   # positions j+k

    # Actually, for the correlation to be valid, we need the indices to be in [0, n-1]
    # If out of range, the term doesn't contribute (h=0 effectively)
    
    v_minus = np.zeros(n * 2 - 1)
    mask_minus = (jk_minus >= 0) & (jk_minus < n)
    v_minus[mask_minus] = h[jk_minus[mask_minus]]
    
    v_plus = np.zeros(n * 2 - 1)
    mask_plus = (jk_plus >= 0) & (jk_plus < n)
    v_plus[mask_plus] = h[jk_plus[mask_plus]]
    
    if direction == -1:  # 1→0
        return dx * (v_minus + v_plus - 1.0)
    else:  # 0→1
        return dx * (1.0 - v_minus - v_plus)

Wait, I need to be more careful about the boundary conditions. The correlation is:
full_corr[m] = dx * Σ_{i: 0≤i<n, 0≤i+k<n} h[i] * (1-h[i+k])

When we flip h[j]:
- If h[j] was 1 (now 0):
  - Term i=j: was 1*(1-h[j+k]), now 0. Valid if 0≤j<n and 0≤j+k<n.
  - Term i=j-k: was h[j-k]*(1-1)=0, now h[j-k]*(1-0)=h[j-k]. Valid if 0≤j-k<n and 0≤j<n.
  
  So: Δ = -dx*(1-h[j+k]) + dx*h[j-k] (with each term only if valid)

- If h[j] was 0 (now 1):
  - Term i=j: was 0, now 1*(1-h[j+k]). Valid if 0≤j<n and 0≤j+k<n.
  - Term i=j-k: was h[j-k]*(1-0)=h[j-k], now h[j-k]*(1-1)=0. Valid if 0≤j-k<n and 0≤j<n.
  
  So: Δ = +dx*(1-h[j+k]) - dx*h[j-k] (with each term only if valid)

For a binary h, this simplifies:
- 1→0: Δ = dx*(h[j-k] - (1-h[j+k])) = dx*(h[j-k] + h[j+k] - 1)
  - Valid terms: first term needs 0≤j-k<n, second needs 0≤j+k<n
- 0→1: Δ = dx*((1-h[j+k]) - h[j-k]) = dx*(1 - h[j+k] - h[j-k])
  - Same validity

```python
def compute_flip_delta(h, j, is_one, n):
    """Compute the change in the full correlation array when flipping h[j]."""
    dx = 2.0 / n
    k = np.arange(-(n-1), n)  # 2n-1 lags
    
    jm = j - k  # j - k for each lag
    jp = j + k  # j + k for each lag
    
    dm = np.zeros(2*n-1)
    dp = np.zeros(2*n-1)
    
    vm = (jm >= 0) & (jm < n)
    vp = (jp >= 0) & (jp < n)
    
    if is_one:  # 1 → 0
        # Δ = dx * (h[j-k] - (1-h[j+k]))
        # = dx * (h[j-k] + h[j+k] - 1)
        # But each term is only present if valid
        # h[j-k] term: valid when 0≤j-k<n, contributes dx*h[j-k]
        # -(1-h[j+k]) term: valid when 0≤j+k<n, contributes -dx*(1-h[j+k])
        dm[vm] = h[jm[vm]]  # dx * h[j-k]
        dp[vp] = -(1.0 - h[jp[vp]])  # -dx * (1-h[j+k])
        return dx * (dm + dp)
    else:  # 0 → 1
        # Δ = dx * ((1-h[j+k]) - h[j-k])
        dm[vm] = -h[jm[vm]]  # -dx * h[j-k]
        dp[vp] = (1.0 - h[jp[vp]])  # +dx * (1-h[j+k])
        return dx * (dm + dp)
```
