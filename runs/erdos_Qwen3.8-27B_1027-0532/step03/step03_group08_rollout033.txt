            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smooth_max_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    if m < -500:
        return m * (2.0 / n)
    sm = np.sum(np.exp(alpha * (corr - m)))
    return (m + np.log(sm) / alpha) * (2.0 / n)

def _smooth_max_val(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    if m < -100:
        best_k = int(np.argmax(corr))
        return m * (2.0 / n), best_k
    sm = np.sum(np.exp(alpha * (corr - m)))
    return (m + np.log(sm) / alpha) * (2.0 / n), int(np.argmax(corr))

def _worst_lag_grad(h, n, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    if m < -100:
        k_star = int(np.argmax(corr))
        weights = np.zeros(len(corr))
        weights[k_star] = 1.0
    else:
        weights = np.exp(alpha * (corr - m))
        sw = weights.sum()
        if sw < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights = weights / sw
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mg = n - lag
            if mg > 0:
                g[lag:lag + mg] += wk * (1.0 - h[:mg])
                g[:mg] += wk * (-h[lag:lag + mg])
        else:
            j = -lag
            mg = n - j
            if mg > 0:
                g[:mg] += wk * (1.0 - h[j:])
                g[j:] += wk * (-h[:mg])
    return g * (2.0 / n)

def _nesterov(h0, n, target_sum, n_iters, alpha, top_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = float(np.max(corr))
        if top_m >= len(corr):
            top_idx = np.arange(len(corr))
        else:
            top_idx = np.argsort(corr)[::-1][:top_m]
        w = np.exp(50.0 * (corr[top_idx] - max_corr))
        tw = w.sum()
        if tw < 1e-300:
            w = np.zeros(len(top_idx))
            w[0] = 1.0
        else:
            w = w / tw
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            k = top_idx[idx]
            if w[idx] < 1e-14:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                mg = n - lag
                if mg > 0:
                    g[lag:lag + mg] += w[idx] * (1.0 - h0[:mg] if False else (1.0 - y[:mg]))
                    g[:mg] += w[idx] * (-y[lag:lag + mg])
            else:
                j = -lag
                mg = n - j
                if mg > 0:
                    g[:mg] += w[idx] * (1.0 - y[j:])
                    g[j:] += w[idx] * (-y[:mg])
        g *= dx
        step = alpha / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_focused(h0, n, target_sum, alpha, maxiter, deadline):
    def obj(x):
        return _smooth_max_obj(x, alpha, n)

    def constr(x):
        return x.sum() - target_sum

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constr, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_out = _fix_sum(res.x, target_sum)
        return h_out
    except Exception:
        return h0

def _coord_descent_fine(h0, n, target_sum, max_rounds, deadline, n_vals=11):
    h = h0.copy()
    vals = np.linspace(0.0, 1.0, n_vals)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-12:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return _fix_sum(h, target_sum)

def _coord_descent_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            co = _c5_of(h)  # h[j] is 1.0, need old value
            h[j] = old
            co = _c5_of(h)
            if c0 < co - 1e-12:
                h[j] = 0.0
                improved = True
            elif c1 < co - 1e-12:
                h[j] = 1.0
                improved = True
        if not improved:
            break
    return _fix_sum(h, target_sum)

def _worst_lag_chase(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        alpha = 300.0 + 200.0 * rnd
        h = _slsqp_focused(h, n, target_sum, alpha, 800, deadline)
        c5 = _c5_of(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def _interpolate_h(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0, n_src - 1, n_dst)
    h_new = np.interp(x_dst, x_src, h_src)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    # Determine n candidates
    n_candidates = [n_init]
    for offset in [20, 40, 50, 80, 100, 150]:
        nc = n_init + offset
        if nc % 2 == 0 and nc <= 350:
            n_candidates.append(nc)
    # Deduplicate
    seen = set()
    n_unique = []
    for nc in n_candidates:
        if nc not in seen:
            seen.add(nc)
            n_unique.append(nc)
    n_candidates = n_unique

    for n in n_candidates:
        if time.time() > deadline - 60:
            break
        target_sum = n / 2.0

        # Interpolate initial
        h_start = _interpolate_h(h_init, n_init, n)
        h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target_sum)
        # Also try from current best
        h_start2 = _interpolate_h(best_h, best_n, n)
        h_start2 = _fix_sum(np.clip(h_start2, 0.0, 1.0), target_sum)
        if _c5_of(h_start2) < _c5_of(h_start):
            h_start = h_start2

        c5_start = _c5_of(h_start)
        if c5_start > best_c5 + 0.003:
            continue

        # Phase 1: Nesterov with multiple settings
        h_curr = h_start.copy()
        nesterov_configs = [
            (150, 0.25, 7, 40.0),
            (150, 0.15, 10, 60.0),
            (100, 0.1, 15, 80.0),
        ]
        for (n_it, lr, tm, sm_a) in nesterov_configs:
            if time.time() > deadline - 30:
                break
            h_n, c5_n = _nesterov(h_curr, n, target_sum, n_it, lr, tm, deadline - 30)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
                best_n = n
                h_curr = h_n.copy()
            elif c5_n < _c5_of(h_curr):
                h_curr = h_n.copy()

        # Phase 2: Worst-lag chasing
        if time.time() > deadline - 40:
            break
        h_chase = h_curr.copy()
        h_ch, c5_ch = _worst_lag_chase(h_chase, n, target_sum, 6, deadline - 40)
        if c5_ch < best_c5:
            best_c5 = c5_ch
            best_h = h_ch.copy()
            best_n = n
            h_curr = h_ch.copy()

        # Phase 3: Coordinate descent (binary)
        if time.time() > deadline - 20:
            break
        h_cd = _coord_descent_binary(h_curr, n, target_sum, 15, deadline - 20)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n
            h_curr = h_cd.copy()

    # Final polish on best solution
    n_best = best_n
    target_best = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 15:
        # High-alpha SLSQP
        h_final = _slsqp_focused(h_final, n_best, target_best, 800, 3000, deadline - 12)
        c5_f = _c5_of(h_final)
        if c5_f > best_c5:
            h_final = best_h.copy()

    if time.time() < deadline - 10:
        # Even higher alpha SLSQP
        h_sl2 = _slsqp_focused(h_final, n_best, target_best, 1500, 2000, deadline - 8)
        c5_s2 = _c5_of(h_sl2)
        if c5_s2 < _c5_of(h_final):
            h_final = h_sl2
        c5_f = _c5_of(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()

    if time.time() < deadline - 8:
        # Fine coordinate descent
        h_fd = _coord_descent_fine(h_final, n_best, target_best, 8, deadline - 6, n_vals=9)
        c5_fd = _c5_of(h_fd)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()
            h_final = h_fd.copy()

    if time.time() < deadline - 5:
        # Another SLSQP round
        h_sl3 = _slsqp_focused(h_final, n_best, target_best, 2000, 5000, deadline - 3)
        c5_s3 = _c5_of(h_sl3)
        if c5_s3 < best_c5:
            best_c5 = c5_s3
            best_h = h_sl3.copy()
            h_final = h_sl3.copy()

    if time.time() < deadline - 3:
        # Final binary coord descent
        h_bd = _coord_descent_binary(h_final, n_best, target_best, 10, deadline - 2)
        c5_bd = _c5_of(h_bd)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
