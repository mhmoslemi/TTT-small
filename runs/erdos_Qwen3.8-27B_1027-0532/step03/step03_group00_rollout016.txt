def project_onto_feasible(h, target):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target}"""
    # Use bisection on lambda
    # x[i] = clip(h[i] - lambda, 0, 1)
    # Need sum(x) = target
    
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # Distribute remaining mass
        mask = (x > 0) & (x < 1)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible_bisect(h, target):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any numerical issues
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_gradient(h, weights):
    """Compute gradient of sum_j weights[j] * corr[j] w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for j in range(len(weights)):
        w = weights[j]
        if w < 1e-12:
            continue
        k = j - (n - 1)
        # First part: d/dh[i] of h[i]*(1-h[i+k]) = (1-h[i+k])
        i1 = max(0, -k)
        i2 = min(n, n - k)
        if i2 > i1:
            grad[i1:i2] += w * (1.0 - h[i1+k:i2+k])
        # Second part: d/dh[i] of h[i-k]*(1-h[i]) = -h[i-k]
        i3 = max(0, k)
        i4 = min(n, n + k)
        if i4 > i3:
            grad[i3:i4] += -w * h[i3-k:i4-k]
    return grad

def slp_optimize(h0, target, n_iter=100, deadline=None):
    """Sequential Linear Programming optimization."""
    n = len(h0)
    n_lags = 2 * n - 1
    h = h0.copy()
    best_score = compute_c5(h)
    
    for it in range(n_iter):
        if deadline and time.time() > deadline:
            break
        
        corr = compute_corr(h)
        cur_max = float(np.max(corr))
        
        # Build LP: minimize t
        # s.t. for each lag j: sum_i grad_j[i] * delta[i] - t <= -corr[j]
        #      sum(delta) = 0
        #      0 <= h[i] + delta[i] <= 1
        
        # Variables: [delta_0, ..., delta_{n-1}, t]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Compute gradients for all lags
        A_ub = np.zeros((n_lags, n + 1))
        b_ub = np.zeros(n_lags)
        
        for j in range(n_lags):
            k = j - (n - 1)
            i1 = max(0, -k)
            i2 = min(n, n - k)
            if i2 > i1:
                A_ub[j, i1:i2] = (1.0 - h[i1+k:i2+k])
            i3 = max(0, k)
            i4 = min(n, n + k)
            if i4 > i3:
                A_ub[j, i3:i4] += -h[i3-k:i4-k]
            A_ub[j, n] = -1.0
            b_ub[j] = -corr[j]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1)
        
        bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
        bounds.append((0.0, None))
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
        
        if not res.success:
            break
        
        delta = res.x[:n]
        if np.allclose(delta, 0):
            break
        
        # Line search
        best_step = 0.0
        best_h = h
        best_sc = best_score
        for step in [0.1, 0.3, 0.5, 0.7, 1.0]:
            h_try = project_feasible_bisect(h + step * delta, target)
            sc = compute_c5(h_try)
            if sc < best_sc:
                best_sc = sc
                best_h = h_try
                best_step = step
        
        if best_step == 0.0:
            break
        
        h = best_h
        best_score = best_sc
    
    return h, best_score

def binary_local_search(h0, n, target, n_swaps=50000, deadline=None):
    """Binary local search via swaps."""
