import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def lag_gradient_vec(h, k_shift):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k_shift == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k_shift > 0:
        grad[0:n - k_shift] += (1.0 - h[k_shift:n])
        grad[k_shift:n] -= h[0:n - k_shift]
    else:
        ks = -k_shift
        grad[ks:n] += (1.0 - h[0:n - ks])
        grad[0:n - ks] -= h[ks:n]
    return grad * dx


def smooth_max_obj_and_grad(h, alpha, w_threshold=1e-8):
    lags = compute_lags(h)
    n = len(h)
    M = np.max(lags)
    shifted = alpha * (lags - M)
    shifted = np.clip(shifted, -400.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    F = M + np.log(sum_exp) / alpha
    w = exp_vals / sum_exp
    active = w > w_threshold
    grad = np.zeros(n)
    active_idx = np.where(active)[0]
    for m in active_idx:
        k_shift = m - (n - 1)
        grad += w[m] * lag_gradient_vec(h, k_shift)
    return F, grad

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    def __init__(self, alpha):
        self.alpha = alpha

    def f(self, x):
        return smooth_max_obj_and_grad(x, self.alpha)[0]

    def g(self, x):
        return smooth_max_obj_and_grad(x, self.alpha)[1]

    prob = SmoothMaxObj(alpha)
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def topk_subgradient_polish(h, target, k, max_iter, tleft_fn):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_val = compute_c5(h_cur)
    step = 0.01
    stall_count = 0
    for it in range(max_iter):
        if tleft_fn() < 20:
            break
        lags = compute_lags(h_cur)
        top_idx = np.argsort(lags)[-k:]
        grad = np.zeros(n)
        for m in top_idx:
            k_shift = m - (n - 1)
            grad += lag_gradient_vec(h_cur, k_shift)
        grad /= float(k)
        h_new = project_feasible(h_cur - step * grad, target)
        val = compute_c5(h_new)

            h_cur = h_new
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 20:
                step *= 0.7
                stall_count = 0
    return best_h, best_val
