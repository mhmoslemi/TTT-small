def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_single_lag(h, m):
    """Gradient of corr[m] with respect to h."""
    n = len(h)
    p = m - (n - 1)
    grad = np.zeros(n)
    # h[j] appears in h[i+p] when i = j-p (contribution: 1-h[j-p])
    # h[j] appears in (1-h[i]) when i = j (contribution: -h[j+p])
    # First term: j-p must be valid i, i.e., 0 <= j-p <= n-1 and 0 <= (j-p)+p <= n-1
    # Second term: j must be valid i, i.e., 0 <= j <= n-1 and 0 <= j+p <= n-1
    for j in range(n):
        g = 0.0
        i1 = j - p  # i such that i+p = j
        if 0 <= i1 <= n-1 and 0 <= i1+p <= n-1:
            g += (1.0 - h[i1])
        # Second: i = j, need 0 <= j+p <= n-1
        if 0 <= j+p <= n-1:
            g -= h[j+p]
        grad[j] = g
    return grad

def gradient_single_lag_fast(h, m):
    """Vectorized gradient of corr[m] with respect to h."""
    n = len(h)
    p = m - (n - 1)
    grad = np.zeros(n)
    # First term: (1-h[j-p]) for j where j-p is valid i
    # j-p valid means 0 <= j-p <= n-1, so j ranges from p to p+n-1 (intersected with [0,n-1])

    j_end1 = min(n-1, p + n - 1)
    if j_start1 <= j_end1:

        grad[j_start1:j_end1+1] += (1.0 - h[i_vals])
    # Second term: -h[j+p] for j where j+p is valid
    j_start2 = max(0, -p)
    j_end2 = min(n-1, n-1-p)
    if j_start2 <= j_end2:

        grad[j_start2:j_end2+1] -= h[idx2]
    return grad

    else:
        corr_sub = corr

        if top_lags is not None:
            actual_m = top_lags[max_idx]
        else:
            actual_m = max_idx
        grad = gradient_single_lag_fast(h, actual_m)
        return F, grad

    sum_exp = np.sum(exp_vals)

        g = gradient_single_lag_fast(h, m_idx)
        grad += w[ii] * g
    return F, grad

    h_work = h.copy()
    corr = np.correlate(h_work, 1.0 - h_work, mode='full')
    n_corr = 2*n - 1
    m0 = n - 1  # the special lag where p=0
    
    for sweep in range(n_sweeps):
        order = list(range(n)) if sweep % 2 == 0 else list(range(n-1, -1, -1))
        for j in order:
            # Compute d for all lags (linear coefficients)
            d = np.zeros(n_corr)
            # For m != m0: d[m] = (1-h[j-p]) - h[j+p] as derived
            # For m == m0: we handle separately
            d[j:n+j] += 1.0 - h_work[n-1::-1]
            d[n-1-j:2*n-1-j] -= h_work

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_single_lag(h, m):
    n = len(h)
    p = m - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, p)
    hi1 = min(n - 1, p + n - 1)
    if lo1 <= hi1:
        grad[lo1:hi1 + 1] += (1.0 - h[np.arange(lo1, hi1 + 1) - p])
    lo2 = max(0, -p)
    hi2 = min(n - 1, n - 1 - p)
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] -= h[np.arange(lo2, hi2 + 1) + p]
    return grad


def smooth_F_grad(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if alpha > 8000:
        mi = np.argmax(corr)
        return m, grad_single_lag(h, mi)
    exp_v = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(len(h))
    for ii in range(len(corr)):
        if w[ii] > 1e-10:
            grad += w[ii] * grad_single_lag(h, ii)
    return F, grad


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgs_polish(h_start, alpha, maxiter):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_F_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_F_grad(x, alpha)
        return g

    try:
        res = minimize(obj_func, h_start.copy(), jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start.copy()


def cd_F_prime(t, c, d, alpha, R_m0, m0):
    n_corr = len(c)
    all_a = alpha * (c + d * t)
    all_a[m0] = alpha * (R_m0 + t - t * t)
    a_max = np.max(all_a)
    exp_shifted = np.exp(all_a - a_max)
    derivs = d.copy()
    derivs[m0] = 1.0 - 2.0 * t
    return float(np.sum(derivs * exp_shifted) / np.sum(exp_shifted))


def coord_desc_smoothed(h, alpha, n_sweeps, target_sum, time_limit):
    n = len(h)
    h_work = h.copy()
    corr = np.correlate(h_work, 1.0 - h_work, mode='full')
    n_corr = 2 * n - 1
    m0 = n - 1

    for sweep in range(n_sweeps):
        if time_limit() < 15:
            break
        if sweep % 2 == 0:
            order = range(n)
        else:
            order = range(n - 1, -1, -1)
        for j in order:
            d = np.zeros(n_corr)
            d[j:n + j] += 1.0 - h_work[n - 1::-1]
            d[n - 1 - j:2 * n - 1 - j] -= h_work

            R_m0 = corr[m0] - h_work[j] * (1.0 - h_work[j])
            c = corr - d * h_work[j]
            c[m0] = R_m0
            d[m0] = 0.0

            fp0 = cd_F_prime(0.0, c, d, alpha, R_m0, m0)
            if fp0 >= -1e-10:
                t_new = 0.0
            else:
                fp1 = cd_F_prime(1.0, c, d, alpha, R_m0, m0)
                if fp1 <= 1e-10:
                    t_new = 1.0
                else:
                    lo = 0.0
                    hi = 1.0
                    for _ in range(40):
                        mid = (lo + hi) / 2.0
                        fp = cd_F_prime(mid, c, d, alpha, R_m0, m0)
                        if fp > 0:
                            hi = mid
                        else:
                            lo = mid
                    t_new = (lo + hi) / 2.0

            old_val = h_work[j]
            if abs(t_new - old_val) > 1e-14:
                h_work[j] = t_new
                delta = t_new - old_val
                corr += d * delta
                corr[m0] += t_new * (1.0 - t_new) - old_val * (1.0 - old_val)

        h_work = project_feasible(h_work, target_sum)
        corr = np.correlate(h_work, 1.0 - h_work, mode='full')

    return h_work


def slsqp_polish(h_start, alpha, maxiter):
    n = len(h_start)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_F_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_F_grad(x, alpha)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj_func, h_start.copy(), jac=jac_func, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h_start.copy()


def subgradient_step(h, n_lags_top, step_size):
    n = len(h)
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(corr)[-n_lags_top:]
    grad = np.zeros(n)
    for m in top_lags:
        grad += grad_single_lag(h, m)
    grad /= n_lags_top
    h_new = h - step_size * grad
    return project_feasible(h_new, target)
