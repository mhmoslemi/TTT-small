def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} via bisection on Lagrange multiplier."""
    lo, hi = -2.0, 2.0

    """Power mean approximation of max(corr)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        c5 = compute_c5(h_opt)
        return h_opt, c5
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def thue_morse_binary(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (bin(i).count('1') % 2 == 0) else 0.0
    # Adjust to have exactly n/2 ones
    s = int(h.sum())
    if s > n // 2:
        excess = s - n // 2
        ones = np.where(h == 1.0)[0]
        h[ones[:excess]] = 0.0
    elif s < n // 2:
        deficit = n // 2 - s
        zeros = np.where(h == 0.0)[0]
        h[zeros[:deficit]] = 1.0
    return h

def periodic_binary(n, period):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i % (2 * period)) < period else 0.0
    s = int(h.sum())
    if s > n // 2:
        excess = s - n // 2
        ones = np.where(h == 1.0)[0]
        h[ones[:excess]] = 0.0
    elif s < n // 2:
        deficit = n // 2 - s
        zeros = np.where(h == 0.0)[0]
        h[zeros[:deficit]] = 1.0
    return h

def binary_sa(h_start, n_steps, t_start, t_end, rng, directed_frac=0.0):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1
    nr = 2 * n - 1

    ones_arr = np.where(h == 1.0)[0].astype(np.int64)
    zeros_arr = np.where(h == 0.0)[0].astype(np.int64)
    n_one = len(ones_arr)
    n_zero = len(zeros_arr)

    step = 0
    no_improve_count = 0
    patience = max(n_steps // 20, 1000)

    while step < n_steps:
        progress = step / max(n_steps - 1, 1)
        t = t_start * (t_end / t_start) ** progress

        if no_improve_count > patience and t > t_end * 10:
            t = t_start * 0.3
            no_improve_count = 0

        use_directed = (directed_frac > 0 and rng.random() < directed_frac)

        if use_directed and current_max > 0:
            worst_k = int(np.argmax(corr))
            worst_s = worst_k - half
            if worst_s != 0:
                valid_mask_i = np.ones(n_one, dtype=bool)
                valid_mask_j = np.ones(n_zero, dtype=bool)
                ones_pos = ones_arr
                zeros_pos = zeros_arr
                valid_mask_i &= (ones_pos + worst_s >= 0) & (ones_pos + worst_s < n)
                valid_mask_i &= (ones_pos - worst_s >= 0) & (ones_pos - worst_s < n)
                valid_mask_j &= (zeros_pos + worst_s >= 0) & (zeros_pos + worst_s < n)
                valid_mask_j &= (zeros_pos - worst_s >= 0) & (zeros_pos - worst_s < n)
                vi = np.where(valid_mask_i)[0]
                vj = np.where(valid_mask_j)[0]
                if len(vi) > 0 and len(vj) > 0:
                    oi = vi[rng.integers(len(vi))]
                    oz = vj[rng.integers(len(vj))]
                else:
                    oi = rng.integers(n_one)
                    oz = rng.integers(n_zero)
            else:
                oi = rng.integers(n_one)
                oz = rng.integers(n_zero)
        else:
            oi = rng.integers(n_one)
            oz = rng.integers(n_zero)

        i = int(ones_arr[oi])
        j = int(zeros_arr[oz])

        if i == j:
            step += 1
            continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)

        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            ones_arr[oi], zeros_arr[oz] = j, i
            step += 1
            continue

        s_range = np.arange(s_min, s_max + 1)
        k0 = s_min + half
        k1 = s_max + half
        delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
        corr[k0:k1+1] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max

        if delta_c5 < -1e-14 or (t > 1e-10 and rng.random() < np.exp(-delta_c5 / t)):
            h[i], h[j] = h[j], h[i]
            ones_arr[oi] = j
            zeros_arr[oz] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            corr[k0:k1+1] -= delta
            no_improve_count += 1

        step += 1

    return best_h, best_c5_val

def exhaustive_local_search(h_bin, time_limit=10.0):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    nr = 2 * n - 1
    start_time = time.time()

    while time.time() - start_time < time_limit:
        best_improvement = 0.0
        best_i = -1
        best_j = -1
        ones_list = np.where(h == 1.0)[0].tolist()
        zeros_list = np.where(h == 0.0)[0].tolist()

        for i in ones_list:
            for j in zeros_list:
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                k0 = s_min + half
                k1 = s_max + half
                delta = h[i + s_min:s_max+1] + h[i - s_max:i - s_min+1][::-1] - h[j + s_min:s_max+1] - h[j - s_max:j - s_min+1][::-1]
                new_vals = corr[k0:k1+1] + delta
                if k0 > 0:
                    out_left = float(np.max(corr[:k0]))
                else:
                    out_left = -1e18
                if k1 < nr - 1:
                    out_right = float(np.max(corr[k1+1:]))
                else:
                    out_right = -1e18
                new_max = max(float(np.max(new_vals)), out_left, out_right)
                if new_max < current_max - 1e-10:
                    improvement = current_max - new_max
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_i = i
                        best_j = j

        if best_i < 0:
            break

        i, j = best_i, best_j
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min <= s_max:
            k0 = s_min + half
            k1 = s_max + half
            delta = h[i + s_min:s_max+1] + h[i - s_max:i - s_min+1][::-1] - h[j + s_min:s_max+1] - h[j - s_max:j - s_min+1][::-1]
            corr[k0:k1+1] += delta
        h[i] = 0.0
        h[j] = 1.0
        current_max -= best_improvement

    return h, current_max

def two_swap_local_search(h_bin, time_limit=5.0):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    start_time = time.time()

    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()

    while time.time() - start_time < time_limit and len(ones_list) >= 2 and len(zeros_list) >= 2:
        np.random.shuffle(ones_list)
        np.random.shuffle(zeros_list)
        improved = False
        for ii in range(min(len(ones_list)-1, 20)):
            i1 = ones_list[ii]
            if h[i1] != 1.0:
                continue
            for jj in range(ii+1, min(len(ones_list), ii+20)):
                i2 = ones_list[jj]
                if h[i2] != 1.0:
                    continue
                # Save state
                h[i1] = 0.0
                h[i2] = 0.0
                ones_list[ii] = -1
                ones_list[jj] = -1

                # Try best pair of zeros
                best_val = float(np.max(corr))
                best_j1 = -1
                best_j2 = -1
                for p in range(min(len(zeros_list)-1, 30)):
                    j1 = zeros_list[p]
                    if h[j1] != 0.0:
                        continue
                    for q in range(p+1, min(len(zeros_list), p+30)):
                        j2 = zeros_list[q]
                        if h[j2] != 0.0:
                            continue
                        h[j1] = 1.0
                        h[j2] = 1.0
                        corr2 = np.correlate(h, 1.0 - h, mode='full')
                        val = float(np.max(corr2))
                        h[j1] = 0.0
                        h[j2] = 0.0
                        if val < best_val - 1e-10:
                            best_val = val
                            best_j1 = j1
                            best_j2 = j2

                if best_j1 >= 0 and best_val < current_max - 1e-10:
                    h[best_j1] = 1.0
                    h[best_j2] = 1.0
                    zeros_list[p] = -1
                    zeros_list[q] = -1
                    ones_list[ii] = best_j1
                    ones_list[jj] = best_j2
                    cors_new = np.correlate(h, 1.0 - h, mode='full')
                    current_max = float(np.max(cors_new))
                    corr = cors_new
                    improved = True
                    break
                else:
                    h[i1] = 1.0
                    h[i2] = 1.0
                    ones_list[ii] = i1
                    ones_list[jj] = i2

            if improved:
                break

        if not improved:
            break

    return h, current_max

def fourier_candidate(n, K, omega, phases, amplitudes, offset):
    x = np.arange(n) * (2.0 / n)
    h = np.full(n, offset)
    for k_idx in range(K):
        h += amplitudes[k_idx] * np.cos((k_idx + 1) * omega * x + phases[k_idx])
    return h

    if remaining() > 15:
        h_final = to_binary(best_h)
        h_ls, ls_max = exhaustive_local_search(h_final, time_limit=min(10.0, remaining() - 5))
        c5_ls = (2.0 / n) * ls_max
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
