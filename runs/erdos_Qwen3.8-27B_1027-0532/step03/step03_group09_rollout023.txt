import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h, n):
    h_sorted = np.argsort(-h)
    result = np.zeros(n)
    result[h_sorted[:n // 2]] = 1.0
    return result

def binary_sa_search(h_binary, time_limit, start_time, rng, t_init=0.05, t_min=0.001, cool_rate=0.99995):
    n = len(h_binary)
    current = h_binary.copy()
    best = h_binary.copy()
    best_c5 = compute_c5(current)
    current_c5 = best_c5
    temperature = t_init

    while time.time() - start_time < time_limit and temperature > t_min:
        corr = compute_corr(current)
        worst_idx = np.argmax(corr)
        worst_s = worst_idx - (n - 1)

        ones_idx = np.where(current == 1.0)[0]
        zeros_idx = np.where(current == 0.0)[0]

        do_2opt = rng.random() < 0.25
        if do_2opt and len(ones_idx) >= 2 and len(zeros_idx) >= 2:
            if worst_s >= 0:
                cand1 = [j for j in ones_idx if j + worst_s < n and current[j + worst_s] == 0]
            else:
                r2 = -worst_s
                cand1 = [j for j in ones_idx if j >= r2 and current[j - r2] == 0]
            if len(cand1) < 2:
                cand1 = list(ones_idx)
            p1 = rng.integers(len(cand1))
            p2 = rng.integers(len(ones_idx))
            a1 = cand1[p1]
            a2 = ones_idx[p2]

def gen_binary_start(n, rng, strategy):
    if strategy == 0:
        idx = rng.permutation(n)
        h = np.zeros(n)
        h[idx[:n // 2]] = 1.0
        return h
    elif strategy == 1:
        h = np.zeros(n)
        for i in range(0, n, 2):
            h[i] = 1.0
        n_ones = h.sum()
        if n_ones > n / 2:
            excess = int(n_ones - n / 2)
            ones = np.where(h == 1.0)[0]
            h[ones[:excess]] = 0.0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for _ in range(max(1, n // 12)):
            i = rng.integers(len(ones))
            j = rng.integers(len(zeros))
            h[ones[i]] = 0.0
            h[zeros[j]] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        return h
    elif strategy == 2:
        h = np.zeros(n)
        p = int(rng.integers(3, max(4, n // 4)))
        for i in range(n):
            if (i % (2 * p)) < p:
                h[i] = 1.0
        total = int(h.sum())
        if total > n // 2:
            excess = total - n // 2
            ones = np.where(h == 1.0)[0]
            h[ones[:excess]] = 0.0
        elif total < n // 2:
            deficit = n // 2 - total
            zeros = np.where(h == 0.0)[0]
            h[zeros[:deficit]] = 1.0
        return h
    else:
        phi = (1.0 + np.sqrt(5.0)) / 2.0
        h = np.zeros(n)

            frac = ((i + 0.5) * phi / n) % 1.0

        total = int(h.sum())

            deficit = n // 2 - total

        return h
