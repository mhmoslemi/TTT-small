def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)[::-1]
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)

            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub

            c5_best = c5_s1
            h_best = h_s1

    if time.time() < deadline - 60:
        h_n, c5_n = _nesterov_subgrad(h_best, n, target_sum, 200, 0.4, 15, 40.0)
        if c5_n < c5_best:
            c5_best = c5_n
            h_best = h_n

    if time.time() < deadline - 50:
        h_s2 = _slsqp(h_best, n, 1500, 4000)
        c5_s2 = _c5_of(h_s2)
        if c5_s2 < c5_best:
            c5_best = c5_s2
            h_best = h_s2

    if time.time() < deadline - 30:
        h_cd = _coord_descent_binary(h_best, n, target_sum, 12)
        c5_cd = _c5_of(h_cd)
        if c5_cd < c5_best:
            c5_best = c5_cd
            h_best = h_cd

    if time.time() < deadline - 20:
        h_cm = _coord_descent_multi(h_best, n, target_sum, 3, 7)
        c5_cm = _c5_of(h_cm)
        if c5_cm < c5_best:
            c5_best = c5_cm
            h_best = h_cm

    if time.time() < deadline - 15:
        h_sh, c5_sh = _shake_refine(h_best, n, target_sum, deadline - 10, rng, 2)
        if c5_sh < c5_best:
            c5_best = c5_sh
            h_best = h_sh

    if time.time() < deadline - 5:
        h_pol = _slsqp(h_best, n, 5000, 8000)
        c5_pol = _c5_of(h_pol)
        if c5_pol < c5_best:
            c5_best = c5_pol
            h_best = h_pol

    if time.time() < deadline - 2:
        h_cd2 = _coord_descent_binary(h_best, n, target_sum, 8)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < c5_best:
            c5_best = c5_cd2
            h_best = h_cd2

    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    # Phase 1: Quick SLSQP on initial
    target_init = n_init / 2.0
    if time.time() < deadline - 400:
        h_s = _slsqp(h_init, n_init, 300, 2000)
        c5_s = _c5_of(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s

    # Phase 2: Subgradient + refine
    if time.time() < deadline - 350:
        h_ng, c5_ng = _nesterov_subgrad(best_h, best_n, best_n / 2.0, 300, 0.5, 12, 30.0)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng
            best_n = best_n

    # Phase 3: Intensive refine at initial resolution
    if time.time() < deadline - 250:
        h_ref, c5_ref = _refine_sequence(best_h, best_n, best_n / 2.0, deadline - 250, rng)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
            best_n = best_n

    # Phase 4: DE at initial resolution
    if time.time() < deadline - 200:
        h_de, c5_de = _de_optimize(best_h, best_n, best_n / 2.0,
                                   45, 90, 0.9, 0.35, 0.55, deadline - 200, rng)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de
            best_n = best_n
        if c5_de < best_c5 + 0.004 and time.time() < deadline - 100:
            h_ref2, c5_ref2 = _refine_sequence(h_de, best_n, best_n / 2.0, deadline - 100, rng)
            if c5_ref2 < best_c5:
                best_c5 = c5_ref2
                best_h = h_ref2

    # Phase 5: Try nearby resolutions
    n_candidates = []
    for d in [2, -2, 4, -4, 6, -6, 8, -8, 10, -10, 12, 16, -16, 20, -20, 24, 28]:
        nt = best_n + d
        if nt % 2 == 0 and 90 <= nt <= 240:
            n_candidates.append(nt)
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)

    for nt in n_unique[:6]:
        if time.time() > deadline - 150:
            break
        h_interp = np.interp(np.linspace(0, best_n - 1, nt), np.arange(best_n), best_h)
        h_interp = _fix_sum(np.clip(h_interp, 0.0, 1.0), nt / 2.0)
        c5_interp = _c5_of(h_interp)
        h_de2, c5_de2 = _de_optimize(h_interp, nt, nt / 2.0,
                                     25, 40, 0.85, 0.4, 0.65, deadline - 80, rng)
        if c5_de2 < c5_interp:
            c5_interp = c5_de2
            h_interp = h_de2
        if c5_interp < best_c5 - 1e-6 and time.time() < deadline - 40:
            h_ref3, c5_ref3 = _refine_sequence(h_interp, nt, nt / 2.0, deadline - 30, rng)
            if c5_ref3 < best_c5:
                best_c5 = c5_ref3
                best_h = h_ref3
                best_n = nt

    # Phase 6: Final intensive refinement
    if time.time() < deadline - 20:
        target_final = best_n / 2.0
        h_fin, c5_fin = _refine_sequence(best_h, best_n, target_final, deadline - 10, rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
            best_n = best_n

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        target = best_n / 2.0
        h_pol = _slsqp(best_h, best_n, 8000, 10000)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol
        best_h = _fix_sum(best_h, target)
        best_c5 = _c5_of(best_h)

    if time.time() < deadline - 2:
        h_cd = _coord_descent_binary(best_h, best_n, best_n / 2.0, 10)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        best_h = _fix_sum(best_h, best_n / 2.0)
        best_c5 = _c5_of(best_h)

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
