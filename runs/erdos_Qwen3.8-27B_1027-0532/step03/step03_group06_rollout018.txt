def bb_momentum_descend(h0, target, K_schedule, n_iter_per_K, max_step):
    h = h0.copy()
    prev_h = h.copy()
    prev_grad = None
    momentum = 0.9
    
    for K in K_schedule:
        for it in range(n_iter_per_K):
            g = get_topk_gradient(h, K)

            if prev_grad is not None:
                s = h - prev_h
                y = g - prev_grad

                    alpha_bb = np.dot(s, s) / sy
                else:
                    alpha_bb = 1.0

            alpha_bb = np.clip(alpha_bb, 1e-8, max_step)
            
            # Nesterov momentum
            if prev_grad is not None:
                grad = g + momentum * (g - prev_grad)
            else:
                grad = g
            momentum = 0.5 * (1 + momentum)  # polyak momentum
            
            h_new = project_feasible(h - alpha_bb * grad, target)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            
            if c5_new < c5_old:
                h = h_new
            # else: keep h, maybe reduce step
            prev_h, prev_grad = h, g
    return h

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_grad(h, K):
    """Gradient of mean of top-K correlation values."""
    n = len(h)
    corr = compute_all_corr(h)
    top_idx = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    
    for m in top_idx:
        s = m - (n - 1)
        if s >= 0:
            js = np.arange(n)
            mask1 = (js >= s) & (js <= n - 1)
            mask2 = (js >= 0) & (js <= n - 1 - s)
            grad[mask1] += 1.0 - h[js[mask1] - s]
            grad[mask2] -= h[js[mask2] + s]
        else:
            p = -s

            mask2 = (js >= p) & (js <= n - 1)
            grad[mask1] += 1.0 - h[js[mask1] + p]
            grad[mask2] -= h[js[mask2] - p]
    
    return grad / K, corr

    # Phase 3: L-BFGS-B with high alpha polish
    if tleft() > 40:
        for alpha in [500, 1500]:
            if tleft() < 30:
                break
            h_new = lbfgs_smooth(best_h, n0, target0, alpha, 300, 1e-15, tleft)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 4: Combinatorial refinement
    if tleft() > 50:
        h_new, sc = combinatorial_refine(best_h, target0, 2000, tleft, rng)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        # Re-optimize after combinatorial
        if tleft() > 30:
            h_new, sc = bb_momentum_descend(best_h, target0, [1, 3, 5, 10, 20], 300, 0.5, tleft)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 5: Multi-start diverse perturbations
    if tleft() > 80:
        h_new, sc = multi_start_refine(best_h, n0, target0, 12, tleft, rng)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Grid refinement to higher resolution
    for n_try in [500, 600, 800]:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        # Quick L-BFGS at new resolution
        h_new = lbfgs_smooth(h_interp, n_try, target_try, 50, 150, 1e-13, tleft)
        sc = compute_c5(h_new)
        # BB-momentum at new resolution
        if tleft() > 40:
            h_new2, sc2 = bb_momentum_descend(h_new, target_try, [1, 3, 5, 10, 20], 300, 0.5, tleft)
            if sc2 < sc:
                h_new = h_new2
                sc = sc2
        # High alpha polish
        if tleft() > 30:
            h_new3 = lbfgs_smooth(h_new, n_try, target_try, 500, 200, 1e-14, tleft)
            sc3 = compute_c5(h_new3)
            if sc3 < sc:
                h_new = h_new3
                sc = sc3
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 7: Final BB-momentum polish at current resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 30:
        h_new, sc = bb_momentum_descend(best_h, target_use, [1, 1, 2, 3, 5], 500, 0.3, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final L-BFGS-B with very high alpha
    if tleft() > 25:
        h_new = lbfgs_smooth(best_h, n_use, target_use, 3000, 200, 1e-15, tleft)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Final projection and evaluation
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
