            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def find_optimal_t(A, B):
    t_min, t_max = 0.0, 1.0
    for _ in range(50):
        t1 = t_min + (t_max - t_min) / 3.0
        t2 = t_max - (t_max - t_min) / 3.0
        if np.max(A * t1 + B) < np.max(A * t2 + B):
            t_max = t2
        else:
            t_min = t1
        if t_max - t_min < 1e-13:
            break
    return (t_min + t_max) / 2.0


def project_sum_exact(h, target_sum):
    lo, hi = -3.0, 3.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)


def coord_descent_one_pass(h, lam, order):
    for m in order:
        h[m] = 0.0
        corr0 = np.correlate(h, 1.0 - h, mode='full')
        h[m] = 1.0
        corr1 = np.correlate(h, 1.0 - h, mode='full')
        A = corr1 - corr0
        B = corr0
        t_opt = find_optimal_t(A + lam, B)
        h[m] = t_opt
    return h


def run_coord_descent(h_init, target_sum, max_passes, seed, lam_init=0.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    lam = lam_init
    eta = 0.08
    best_h = project_sum_exact(h, target_sum).copy()
    best_score = compute_c5(best_h)

    for p in range(max_passes):
        order = rng.permutation(n)
        h = coord_descent_one_pass(h, lam, order)
        h_proj = project_sum_exact(h, target_sum)
        sum_diff = np.sum(h) - target_sum
        lam = np.clip(lam + eta * sum_diff, -5.0, 5.0)
        eta = max(eta * 0.96, 0.005)
        score = compute_c5(h_proj)
        if score < best_score:
            best_score = score
            best_h = h_proj.copy()
    return best_h, best_score


def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    penalty = 50.0 * (np.sum(x) - _SLSQP_TARGET) ** 2
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha + penalty


def _slsqp_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    exp_s = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_s)
    weights = exp_s / total
    n = len(x)
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * (1.0 - x[:n - lag])
                grad[:n - lag] -= w * x[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * (1.0 - x[m:])
                grad[m:] -= w * x[:n - m]
    grad += 100.0 * (np.sum(x) - _SLSQP_TARGET)
    return grad


def _slsqp_cons_fun(x):
    return np.sum(x) - _SLSQP_TARGET


def _slsqp_cons_jac(x):
    return np.ones(_SLSQP_N)


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _SLSQP_N, _SLSQP_TARGET
    _SLSQP_N = len(h_start)
    _SLSQP_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _SLSQP_N
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           jac=_slsqp_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_sum_exact(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_sum_exact(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_sum_exact(h_out, target_dst)


def get_remaining(start_time, budget_s, margin=30.0):
    return budget_s - margin - (time.time() - start_time)


def binary_local_search(h_start, target_sum, max_iter, seed, temperature=0.01):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    diff = target_sum - np.sum(h)
    if diff > 0:
        zeros = np.where(h == 0.0)[0]
        rng.shuffle(zeros)
        h[zeros[:int(diff)]] = 1.0
    elif diff < 0:
        ones = np.where(h == 1.0)[0]
        rng.shuffle(ones)
        h[ones[:int(-diff)]] = 0.0
    h = project_sum_exact(h, target_sum)
    best_h = h.copy()
    best_score = compute_c5(h)
    t = temperature
    cooling = 0.999

    for it in range(max_iter):
        i = rng.randint(n)
        if h[i] > 0.5:
            candidates = np.where(h < 0.5)[0]
        else:
            candidates = np.where(h > 0.5)[0]
        if len(candidates) == 0:
            break
        j = candidates[rng.randint(len(candidates))]
        h_new = h.copy()
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
        h_new = project_sum_exact(h_new, target_sum)
        score_new = compute_c5(h_new)
        delta = score_new - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            h = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
        t *= cooling
        if t < 1e-8:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _SLSQP_N, _SLSQP_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_sum_exact(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent from initial
    if get_remaining(start_time, budget_s) > 80:
        h_cd, sc_cd = run_coord_descent(best_h, target_init, 35, seed)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 2: SLSQP refinement
    if get_remaining(start_time, budget_s) > 120:
        h_sf = slsqp_refine(best_h, target_init, [15, 40, 100, 250], 1500, 1e-14)
        sc_sf = compute_c5(h_sf)
        if sc_sf < best_score:
            best_score = sc_sf
            best_h = h_sf

    # Phase 3: Coordinate descent again (finer)
    if get_remaining(start_time, budget_s) > 80:
        h_cd2, sc_cd2 = run_coord_descent(best_h, target_init, 40, seed + 100, lam_init=0.5)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2

    # Phase 4: SLSQP with higher smoothness
    if get_remaining(start_time, budget_s) > 100:
        h_sf2 = slsqp_refine(best_h, target_init, [50, 150, 400], 2000, 1e-15)
        sc_sf2 = compute_c5(h_sf2)
        if sc_sf2 < best_score:
            best_score = sc_sf2
            best_h = h_sf2

    # Phase 5: Perturbed restarts
    for r in range(4):
        if get_remaining(start_time, budget_s) < 80:
            break
        noise_scale = 0.008 * (r + 1)
        h_pert = best_h + rng.randn(n_init) * noise_scale
        h_pert = project_sum_exact(h_pert, target_init)
        h_cd3, sc_cd3 = run_coord_descent(h_pert, target_init, 30, seed + 500 + r * 100)
        if sc_cd3 < best_score:
            best_score = sc_cd3
            best_h = h_cd3
        if get_remaining(start_time, budget_s) > 60 and sc_cd3 < best_score + 1e-5:
            h_sf3 = slsqp_refine(h_cd3, target_init, [50, 200], 800, 1e-14)
            sc_sf3 = compute_c5(h_sf3)
            if sc_sf3 < best_score:
                best_score = sc_sf3
                best_h = h_sf3

    # Phase 6: Binary local search
    if get_remaining(start_time, budget_s) > 60:
        h_bin, sc_bin = binary_local_search(best_h, target_init, 5000, seed + 2000, 0.005)
        if sc_bin < best_score:
            best_score = sc_bin
            best_h = h_bin
            if get_remaining(start_time, budget_s) > 60:
                h_sf4 = slsqp_refine(best_h, target_init, [30, 100, 300], 1000, 1e-14)
                sc_sf4 = compute_c5(h_sf4)
                if sc_sf4 < best_score:
                    best_score = sc_sf4
                    best_h = h_sf4

    # Phase 7: Higher resolution
    for n_try in [500, 600, 800]:
        if get_remaining(start_time, budget_s) < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_hi, sc_hi = run_coord_descent(h_interp, target_try, 25, seed + 3000 + n_try)
        if sc_hi < best_score:
            best_score = sc_hi
            best_h = h_hi
        if get_remaining(start_time, budget_s) > 60:
            h_sf5 = slsqp_refine(best_h, target_try, [30, 100, 300], 800, 1e-14)
            sc_sf5 = compute_c5(h_sf5)
            if sc_sf5 < best_score:
                best_score = sc_sf5
                best_h = h_sf5

    # Phase 8: Final coordinate descent + SLSQP polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if get_remaining(start_time, budget_s) > 50:
        h_cd_final, sc_cd_final = run_coord_descent(best_h, target_use, 30, seed + 9000, lam_init=0.3)
        if sc_cd_final < best_score:
            best_score = sc_cd_final
            best_h = h_cd_final

    if get_remaining(start_time, budget_s) > 40:
        h_final_sf = slsqp_refine(best_h, target_use, [100, 500, 1000], 800, 1e-15)
        sc_final_sf = compute_c5(h_final_sf)
        if sc_final_sf < best_score:
            best_score = sc_final_sf
            best_h = h_final_sf

    # Phase 9: Tiny final perturbations
    if get_remaining(start_time, budget_s) > 35:
        for _ in range(2):
            if get_remaining(start_time, budget_s) < 35:
                break
            h_tiny = best_h + rng.randn(n_use) * 0.002
            h_tiny = project_sum_exact(h_tiny, target_use)
            h_tiny_cd, sc_tiny = run_coord_descent(h_tiny, target_use, 15, seed + 9500)
            if sc_tiny < best_score:
                best_score = sc_tiny
                best_h = h_tiny_cd

    h_out = project_sum_exact(best_h, target_use)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
