```python
def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        # Distribute diff to positions with most room
        room = 1.0 - h[idx]
        room_sum = np.sum(room)
        if room_sum < diff:
            h[idx] = 1.0
            remaining = diff - room_sum
            idx2 = np.where(h < 1.0)[0]
            if len(idx2) > 0:
                h[idx2] += remaining / len(idx2)
        else:

            h[idx] += diff * room / room_sum

        else:
            h[idx] += diff * room / room_sum
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    h = h.copy()
    if diff > 0:
        mask = h < 1.0
        count = np.sum(mask)
        if count == 0:
            return h
        h[mask] += diff / count
    else:
        mask = h > 0.0
        count = np.sum(mask)
        if count == 0:
            return h
        h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def solve_1d_minimax(A, d, n_iter=40):
    n_active = len(A)
    if n_active == 0:
        return 0.5
    if n_active == 1:
        # Single line: min at boundary
        if d[0] < 0:
            return 1.0  # Line decreasing, min at t=1... wait
            # Actually if d[0] < 0, f(t) = A[0] + d[0]*t is decreasing
            # So min on [0,1] is at t=1
        elif d[0] > 0:
            return 0.0  # Line increasing, min at t=0
        else:
            return 0.5  # Constant, any t works
    
    def f(t):
        vals = A + d * t
        return np.max(vals)
    
    # Golden section search on [0, 1]
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - invphi * (b - a)
    dpt = a + invphi * (b - a)
    fc = f(c)
    fd = f(dpt)
    
    for _ in range(n_iter):
        if fc < fd:
            b = dpt
            dpt = c
            fd = fc
            c = b - invphi * (b - a)
            fc = f(c)
        else:
            a = c
            c = dpt
            fc = fd
            dpt = a + invphi * (b - a)
            fd = f(dpt)
    
    return (a + b) / 2.0

def coordinate_descent(h_init, target_sum, n_passes, topk, time_limit=None, start_time=None):
    n = len(h_init)
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * (2.0 / n)
    dx = 2.0 / n
    
    for p in range(n_passes):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        current_max = np.max(corr)
        if topk < len(corr):
            active = np.argpartition(corr, -topk)[-topk:]
        else:
            active = np.arange(len(corr))
        
        sorted_active = active[np.argsort(corr[active])[::-1]]
        
        for q in range(n):
            offset1 = q + sorted_active - (n - 1)
            offset2 = q - sorted_active + (n - 1)
            valid1 = (offset1 >= 0) & (offset1 < n)
            valid2 = (offset2 >= 0) & (offset2 < n)
            
            d = np.zeros(len(sorted_active))
            if np.any(valid1):
                d[valid1] += (1.0 - h[offset1[valid1]])
            if np.any(valid2):
                d[valid2] -= h[offset2[valid2]]
            
            A = corr[sorted_active] - h[q] * d
            t_star = solve_1d_minimax(A, d, n_iter=35)
            h[q] = t_star
        
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score:
            best_score = score
    
    return h, best_score
```

```python
        prev_score = score
    
    return best_h, best_score

def subgradient_optimize(h_init, target_sum, max_iter, lr_init, topk, alpha, patience, seed, time_limit, start_time):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = project_feasible(h_init.copy(), target_sum)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    best_h = h.copy()
    best_score = float(np.max(corr)) * dx
    lr = lr_init
    v = np.zeros(n)
    momentum = 0.9
    stagnation = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if topk < n_corr:
            active = np.argpartition(corr, -topk)[-topk:]
        else:
            active = np.arange(n_corr)
        
        top_vals = corr[active]
        weights = np.exp(alpha * (top_vals - np.max(top_vals)))
        w_sum = np.sum(weights)
        if w_sum < 1e-300:
            weights = np.ones(len(active)) / len(active)
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for i in range(len(active)):
            w = weights[i]
            if w < 1e-14:
                continue
            k = active[i]
            lag = k - (n - 1)
            if lag >= 0:
                m = lag
                if m < n:
                    grad[m:] += w * (1.0 - h[:n - m])
                    grad[:n - m] -= w * h[m:]
            else:
                m = -lag
                if m < n:
                    grad[:n - m] += w * (1.0 - h[m:])
                    grad[m:] -= w * h[:n - m]
        
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        score = float(np.max(corr_new)) * dx
        
        if score < best_score - 1e-13:
            best_score = score
            best_h = h_new.copy()
            h = h_new
            corr = corr_new
            stagnation = 0
            lr = min(lr * 1.005, 3.0)
        else:
            h = h_new
            corr = corr_new
            stagnation += 1
            lr *= 0.995
            if rng.random() < 0.05:
                h = h + rng.randn(n) * 0.01
                h = project_feasible(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full')
            if stagnation > patience:
                break
    
    return best_h, best_score

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    time_limit = budget_s - 25.0
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)
    
    # Phase 1: Coordinate descent on initial resolution
    if time.time() - start_time < time_limit - 300:
        h_new, score = coord_descent(best_h, target_init,
                                     n_passes=80000, topk=40,
                                     time_limit=time_limit - 500, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 2: Subgradient to escape plateaus
    if time.time() - start_time < time_limit - 250:
        h_try = best_h + rng.randn(n_init) * 0.008
        h_try = project_feasible(h_try, target_init)
        h_new, score = subgradient_optimize(h_try, target_init,
                                            max_iter=15000, lr_init=0.08,
                                            topk=15, alpha=25, patience=400,
                                            seed=seed, time_limit=time_limit - 350, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 3: More coordinate descent for polish
    if time.time() - start_time < time_limit - 200:
        h_new, score = coord_descent(best_h, target_init,
                                     n_passes=80000, topk=55,
                                     time_limit=time_limit - 300, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 4: Try higher resolution n=800
    if time.time() - start_time < time_limit - 200:
        n_try = 800
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_new, score = coord_descent(h_interp, target_try,
                                     n_passes=50000, topk=50,
                                     time_limit=time_limit - 250, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 5: Subgradient on n=800
    if time.time() - start_time < time_limit - 100:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_try = best_h + rng.randn(n_cur) * 0.005
        h_try = project_feasible(h_try, target_cur)
        h_new, score = subgradient_optimize(h_try, target_cur,
                                            max_iter=10000, lr_init=0.05,
                                            topk=20, alpha=40, patience=300,
                                            seed=seed + 200, time_limit=time_limit - 100, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 6: Final coordinate descent polish
    if time.time() - start_time < time_limit - 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_new, score = coord_descent(best_h, target_cur,
                                     n_passes=60000, topk=60,
                                     time_limit=time_limit - 10, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Phase 7: Try n=600 if it might be better
    if time.time() - start_time < time_limit - 80:
        n_try2 = 600
        target_try2 = n_try2 / 2.0
        h_interp2 = interpolate_h(best_h, n_try2, target_try2)
        h_new, score = coord_descent(h_interp2, target_try2,
                                     n_passes=40000, topk=50,
                                     time_limit=time_limit - 30, start_time=start_time)
        if score < best_score:
            best_score = score
            best_h = h_new
    
    # Final projection and evaluation
    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    final_score = compute_c5(best_h)
    
    return best_h, final_score, n_final
```
