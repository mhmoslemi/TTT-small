        c5 = np.max(corr) * (2.0 / n)

            total_w += w

            v = 0.5 * v
            base_lr *= 0.9
            h_new = _project_sum_box(h + v, target_sum)
            h = h_new
    
    return best_h, best_val

            idx2 = rng_choice(len(zeros), min(max_pairs_to_check, len(zeros)))
            candidates = list(zip(ones[idx1], zeros[idx2]))

                change = 0.0

                jm1m = j1 - d
                jm2p = j2 + d
                jm2m = j2 - d
                if 0 <= jm1p < n: change += h_bin[jm1p]
                if 0 <= jm1m < n: change += h_bin[jm1m]
                if 0 <= jm2p < n: change -= h_bin[jm2p]
                if 0 <= jm2m < n: change -= h_bin[jm2m]
                gain -= change  # We want to reduce corr, so negative change is good

                best_gain = gain
                best_j1, best_j2 = j1, j2

        val = np.max(corr) * (2.0 / n)
        if val < best_val:
            best_val = val
            best_h = h_bin.copy()

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_sum_box(y, S):
    n = len(y)
    lo, hi = y.min() - 1.5, y.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
        else:
            order = np.argsort(x)
        for i in order:
            if abs(diff) <= 1e-14:
                break
            if diff > 0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr(h, m_idx, n):
    d = m_idx - (n - 1)
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[d:d+m] += (1.0 - h[:m])
            grad[:m] += -h[d:d+m]
    else:
        j = -d
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad

def _smooth_obj_and_grad(h, n, alpha, top_k=15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:top_k]
    top = corr[idx]
    m = top.max()
    w = np.exp(alpha * (top - m))
    w_sum = w.sum()
    obj = (m + np.log(w_sum) / alpha) * (2.0 / n)
    w = w / w_sum
    g = np.zeros(n)
    for wi, mi in zip(w, idx):
        g += wi * _grad_corr(h, mi, n)
    g *= (2.0 / n)
    return obj, g

def _fista_opt(h0, n, target_sum, alpha, n_iters, lr0, deadline):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best_x = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _, g = _smooth_obj_and_grad(y, n, alpha, 15)
        lr = lr0 / np.sqrt(float(it + 1))
        x_new = _project_sum_box(y - lr * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_sum_box(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best_x = x.copy()
    return best_x, best_val

    def obj(h):
        o, _ = _smooth_obj_and_grad(h, n, alpha, top_k)
        return o
    def obj_grad(h):
        _, g = _smooth_obj_and_grad(h, n, alpha, top_k)
        return g
    def eq_fun(h):
        return h.sum() - target_sum
    def eq_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _binary_greedy(h0, n, target_sum, max_iters, deadline, top_k=3):
    h = np.where(h0 > 0.5, 1.0, 0.0)
    h = _fix_sum(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_val = np.max(corr) * (2.0 / n)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break
        top_idx = np.argsort(corr)[::-1][:top_k]
        best_gain = 1e-10
        best_j1, best_j2 = -1, -1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for j1 in ones:
            for j2 in zeros:
                gain = 0.0
                for m_idx in top_idx:
                    d = m_idx - (n - 1)
                    ch = 0.0
                    if 0 <= j1 + d < n:
                        ch += h[j1 + d]
                    if 0 <= j1 - d < n:
                        ch += h[j1 - d]
                    if 0 <= j2 + d < n:
                        ch -= h[j2 + d]
                    if 0 <= j2 - d < n:
                        ch -= h[j2 - d]
                    gain -= ch
                if gain > best_gain:
                    best_gain = gain
                    best_j1, best_j2 = j1, j2
        if best_j1 < 0:
            break
        h[best_j1] = 0.0
        h[best_j2] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = np.max(corr) * (2.0 / n)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val

def _coord_descent_fast(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old_v = h[j]
            best_delta = 0.0
            best_max = corr.max()
            for v in vals:
                if abs(v - old_v) < 1e-6:
                    continue
                delta = v - old_v
                new_max = 0.0
                for m in range(2 * n - 1):
                    d = m - (n - 1)
                    g = 0.0
                    if 0 <= j + d < n:
                        g += (1.0 - h[j + d])
                    if 0 <= j - d < n:
                        g -= h[j - d]
                    nv = corr[m] + delta * g
                    if nv > new_max:
                        new_max = nv
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_delta = delta
            if best_delta != 0.0:
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                for m in range(2 * n - 1):
                    d = m - (n - 1)
                    g = 0.0
                    if 0 <= j + d < n:
                        g += (1.0 - h[j + d])
                    if 0 <= j - d < n:
                        g -= h[j - d]
                    corr[m] += best_delta * g
                improved = True
        if not improved:
            break
        h = _fix_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
    return h, _c5_of(h)

def _focus_optimize(h0, n, target_sum, deadline, top_k=5, max_rounds=50):
    h = h0.copy()
    best_val = _c5_of(h)
    best_h = h.copy()
    for rd in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_k]
        improved = False
        for j in range(n):
            old_v = h[j]
            best_delta = 0.0
            best_wmax = 0.0
            for m_idx in top_idx:
                best_wmax = max(best_wmax, corr[m_idx])
            for dv in [-0.5, -0.25, -0.1, 0.1, 0.25, 0.5]:
                v_new = old_v + dv
                if v_new < -0.01 or v_new > 1.01:
                    continue
                v_new = np.clip(v_new, 0.0, 1.0)
                if abs(v_new - old_v) < 1e-6:
                    continue
                delta = v_new - old_v
                wmax = 0.0
                for m_idx in top_idx:
                    d = m_idx - (n - 1)
                    g = 0.0
                    if 0 <= j + d < n:
                        g += (1.0 - h[j + d])
                    if 0 <= j - d < n:
                        g -= h[j - d]
                    nv = corr[m_idx] + delta * g
                    if nv > wmax:
                        wmax = nv
                if wmax < best_wmax - 1e-12:
                    best_wmax = wmax
                    best_delta = delta
            if best_delta != 0.0:
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                for m_idx in top_idx:
                    d = m_idx - (n - 1)
                    g = 0.0
                    if 0 <= j + d < n:
                        g += (1.0 - h[j + d])
                    if 0 <= j - d < n:
                        g -= h[j - d]
                    corr[m_idx] += best_delta * g
                improved = True
        if not improved:
            break
        h = _fix_sum(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        val = np.max(corr) * (2.0 / n)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val
