import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def thue_morse(n):
    tm = np.zeros(n, dtype=np.float64)
    length = 1
    while length < n:
        end = min(2 * length, n)
        tm[length:end] = 1.0 - tm[:end - length]
        length = end
    return tm

def golden_seq(n, offset):
    phi = 0.6180339887498949
    seq = np.zeros(n)
    for i in range(n):
        val = ((i + offset) * phi) % 1.0
        seq[i] = 1.0 if val < 0.5 else 0.0
    return seq

def make_binary(h, target_ones):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target_ones]] = 1.0
    return h_bin

def _smooth_max_obj(x, eps):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    cmax = np.max(corr)
    if eps < 1e-12:
        return cmax
    xm = (corr - cmax) / eps
    xm -= np.max(xm)
    return cmax + eps * np.log(np.sum(np.exp(xm)))

def _smooth_max_grad(x, eps):
    n = len(x)
    omh = 1.0 - x
    dx = 2.0 / n
    corr = np.correlate(x, omh, mode='full') * dx
    cmax = np.max(corr)
    if eps < 1e-12:
        # Return gradient of the max (subgradient)
        wk_arr = np.zeros(2 * n - 1)
        wk_arr[np.argmax(corr)] = 1.0
        w = wk_arr
    else:
        xm = (corr - cmax) / eps
        xm -= np.max(xm)
        ex = np.exp(xm)
        w = ex / np.sum(ex)
    grad = np.zeros(n)
    for kk in range(2 * n - 1):
        wk = w[kk]
        if wk < 1e-14:
            continue
        s = kk - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * x[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * x[:m]
    return grad

def _eq_fun(x):
    n = len(x)
    return float(np.sum(x) - n / 2.0)

def _eq_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(_smooth_max_obj, h0, args=(eps,),
                          jac=_smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
