The log-sum-exp function is a smooth approximation of the max:
soft_max(c, α) = (1/α) * log(Σ exp(α * c_k))

As α → ∞, soft_max → max. We can minimize soft_max and increase α over time.

The gradient of soft_max w.r.t. h[j] is:
∂soft_max/∂h[j] = (1/α) * Σ_k exp(α*c_k) * ∂c_k/∂h[j] / Σ_m exp(α*c_m)
= Σ_k p_k * ∂c_k/∂h[j]
where p_k = exp(α*c_k) / Σ_m exp(α*c_m)

This is a weighted average of the gradients of individual c_k's, weighted by the softmax. This is a nice structure that we can exploit for a proximal gradient step.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    num_c = 2 * n - 1
    s_arr = np.arange(num_c) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    sweep = 0
    
    while time.time() - start < max_time:
        sweep += 1
        # Random permutation of coordinates
        order = rng.permutation(n)
        
        for j in order:
            old = h[j]
            # Compute A = d(corr)/d(h[j])
            js_p = j + s_arr
            m1 = (js_p >= 0) & (js_p < n)
            A = np.zeros(num_c)
            A[m1] = 1.0 - h[js_p[m1]]
            js_m = j - s_arr
            m2 = (js_m >= 0) & (js_m < n)
            A[m2] -= h[js_m[m2]]
            
            B = corr - A * old
            # Binary search for optimal v
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = (lo + hi) / 2.0
                k_star = np.argmax(A * mid + B)
                if A[k_star] > 1e-12:
                    hi = mid
                elif A[k_star] < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val
        
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            noise = 0.01 * min(1.0, (no_improve - patience) / patience + 0.1)
            h = best_h + rng.randn(n) * noise
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    
    return best_h, best_score

        # = corr[k] - A_i_contribution
        # where A_i_contribution[k] = (1-h[i+s_k])_{valid} - h[i-s_k]_{valid} (the gradient at current h)
        
        # Similarly, changing h[j] from 0 to 1:
        # corr_new[k] += A_j_contribution[k]
        
        # Let me compute the changes
        js_p_i = i + s_arr
        m1i = (js_p_i >= 0) & (js_p_i < n)
        di = np.zeros(num_c)
        di[m1i] = 1.0 - h[js_p_i[m1i]]
        js_m_i = i - s_arr
        m2i = (js_m_i >= 0) & (js_m_i < n)
        di[m2i] -= h[js_m_i[m2i]]
        
        js_p_j = j + s_arr
        m1j = (js_p_j >= 0) & (js_p_j < n)
        dj = np.zeros(num_c)
        dj[m1j] = 1.0 - h[js_p_j[m1j]]
        js_m_j = j - s_arr
        m2j = (js_m_j >= 0) & (js_m_j < n)
        dj[m2j] -= h[js_m_j[m2j]]
        
        # New corr = corr - di + dj
        corr_new = corr - di + dj
        new_score = float(np.max(corr_new)) * dx
        
        delta = new_score - old_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h[i] = 0
            h[j] = 1
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        
        # Cool down
        T = max(T * 0.999, T_min)
    
    return best_h, best_score

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    num_c = 2 * n - 1
    s_arr = np.arange(num_c) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    
    while time.time() - start < max_time:
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            js_p = j + s_arr
            m1 = (js_p >= 0) & (js_p < n)
            A = np.zeros(num_c)
            A[m1] = 1.0 - h[js_p[m1]]
            js_m = j - s_arr
            m2 = (js_m >= 0) & (js_m < n)
            A[m2] -= h[js_m[m2]]
            B = corr - A * old
            lo, hi = 0.0, 1.0
            for _ in range(22):
                mid = (lo + hi) / 2.0
                k_star = np.argmax(A * mid + B)
                if A[k_star] > 1e-12:
                    hi = mid
                elif A[k_star] < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            noise = 0.01 * min(1.0, 0.1 + (no_improve - patience) / (2.0 * patience))
            h = best_h + rng.randn(n) * noise
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_h, best_score

def binary_sa(h, target_sum, max_time, rng, T0=0.05, T_min=0.001, cool=0.9995):

    dx = 2.0 / n
    num_c = 2 * n - 1
    s_arr = np.arange(num_c) - (n - 1)
    n_target = int(target_sum)
    
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n_target:

        h_bin[to_flip] = 0
    elif n_ones < n_target:
        zeros = np.where(h_bin == 0)[0]
        to_flip = rng.choice(zeros, n_target - n_ones, replace=False)
        h_bin[to_flip] = 1

    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx

    T = T0

        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        # For j: corr changes by: +(1-h[j+s_k])_{valid} - h[j-s_k]_{valid}

        js_mj = j - s_arr
        m2j = (js_mj >= 0) & (js_mj < n)
        delta[m2j] -= h[js_mj[m2j]]
        
        corr_new = corr + delta
        new_score = float(np.max(corr_new)) * dx
        old_score = float(np.max(corr)) * dx
        
        if new_score < old_score or rng.rand() < np.exp(-(new_score - old_score) / T):

            h[j] = 1
            corr = corr_new
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
