def compute_corr_fft(h):

    V = fft(v[::-1], N)

                # Compensate sum
                h_try = project_feasible(h_try, tgt_fin)

        if tl() < 25: break

        sc = evaluate_c5(h_new)
        if sc < best_sc:

    # Phase 6: Micro-search

        h_new = lbfgs(h_p, tgt_fin, 5000, maxit=400)
        sc = evaluate_c5(h_new)
        if sc < best_sc:
            best_sc, best_h = sc, h_new
    
    hf = project_feasible(best_h, len(best_h) / 2.0)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time


def _np2(m):
    return 1 << max(1, (m - 1).bit_length())


def compute_corr_fft(h):
    n = len(h)
    N = _np2(2 * n - 1)
    H = fft(h, N)
    V = fft((1.0 - h)[::-1], N)
    return np.real(ifft(H * V))[:2 * n - 1]


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_fft(h)
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_fg(h, alpha):
    n = len(h)
    corr = compute_corr_fft(h)
    m = np.max(corr)
    s = np.clip(alpha * (corr - m), -500, 0)
    e = np.exp(s)
    se = np.sum(e)
    w = e / se
    F = m + np.log(se) / alpha
    N = _np2(3 * n - 2)
    W = fft(w, N)
    c1 = np.real(ifft(fft(1.0 - h, N) * W))
    c2 = np.real(ifft(fft(h[::-1], N) * W))
    g = c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1][::-1]
    return F, g


def lbfgs_opt(h0, tgt, alpha, maxit=500, ftol=1e-16):
    n = len(h0)
    bds = [(0.0, 1.0)] * n

    def f(x):
        v, _ = smooth_max_fg(x, alpha)
        return v

    def j(x):
        _, g = smooth_max_fg(x, alpha)
        return g

    try:
        r = minimize(f, h0.copy(), jac=j, method='L-BFGS-B', bounds=bds,
                     options={'maxiter': maxit, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(r.x, tgt)
    except Exception:
        return h0


def interp_to(h, nd):
    if len(h) == nd:
        return h.copy()
    return np.interp(np.linspace(0, 1, nd), np.linspace(0, 1, len(h)), h)


def make_binary(rng, n):
    pos = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[pos] = 1.0
    return h


def make_stripe(n, period, duty_start, n_fill=None):
    h = np.zeros(n)
    p = max(2, period)
    d = max(1, int(duty_start * p))
    for i in range(n):
        if (i % p) < d:
            h[i] = 1.0
    if n_fill is not None:
        need = n // 2 - int(h.sum())
        if need > 0:
            zeros = np.where(h == 0)[0]
            if len(zeros) >= need:
                fill = rng.choice(zeros, size=need, replace=False)
                h[fill] = 1.0
        elif need < 0:
            ones = np.where(h == 1)[0]
            if len(ones) >= -need:
                fill = rng.choice(ones, size=-need, replace=False)
                h[fill] = 0.0
    return project_feasible(h, n / 2.0)


def greedy_descent(h, tgt, n_iter=2000, rng=None):
    n = len(h)
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    corr = compute_corr_fft(h).copy()
    current_max = np.max(corr)
    for _ in range(n_iter):
        improved = False
        top_k = np.argsort(corr)[-5:]
        for k in top_k:
            i = rng.randint(n)
            best_delta = 0.0
            best_gain = 0.0
            for delta in [0.02, -0.02, 0.05, -0.05, 0.1, -0.1]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                dj = (1.0 - h[i - k + n - 1]) * (0 <= i - k + n - 1 < n) - \
                     h[i + k - n + 1] * (0 <= i + k - n + 1 < n)
                new_c5 = current_max + delta * dj * dx
                if new_c5 < current_max - 1e-12 and (current_max + delta * dj) < (current_max - best_gain):
                    best_gain = current_max - (current_max + delta * dj)
                    best_delta = delta
            if best_delta != 0.0:
                h[i] += best_delta
                h = project_feasible(h, tgt)
                corr = compute_corr_fft(h)
                new_max = np.max(corr)
                if new_max < current_max:
                    current_max = new_max
                    improved = True
                    break
            j = rng.randint(n)
            if j != i:
                for delta in [0.02, -0.02, 0.05, -0.05]:
                    new_hi = h[i] + delta
                    new_hj = h[j] - delta
                    if new_hi < 0 or new_hi > 1 or new_hj < 0 or new_hj > 1:
                        continue
                    h[i] = new_hi
                    h[j] = new_hj
                    corr_new = compute_corr_fft(h)
                    new_max = np.max(corr_new)
                    if new_max < current_max - 1e-12:
                        current_max = new_max
                        corr = corr_new
                        improved = True
                        break
                    else:
                        h[i] = new_hi - delta
                        h[j] = new_hj + delta
                if improved:
                    break
        if not improved:
            break
    return h, current_max * dx
