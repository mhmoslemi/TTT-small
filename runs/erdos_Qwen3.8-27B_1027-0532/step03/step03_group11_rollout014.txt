def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)

def coordinate_descent(h_init, n_sweeps, lam_iters, time_limit, start_time):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.copy()
    h = project_onto_constraint(h, target_sum)
    
    x_grid = np.linspace(0.0, 1.0, 401)
    
    for sweep in range(n_sweeps):
        if time.time() - start_time > time_limit:
            break
        for p in range(n):
            # Save current value
            hp = h[p]
            
            # Compute base correlation (with h[p] = 0)
            h[p] = 0.0
            base_corr = np.correlate(h, 1.0 - h, mode='full') * dx
            h[p] = hp
            
            # Compute linear coefficients for each k
            s_range = np.arange(-(n-1), n)  # shifts from -(n-1) to n-1
            # For shift s, the linear coefficient of h[p] in c_s:
            # (1-h[p+s]) if 0<=p+s<n else 0, minus h[p-s] if 0<=p-s<n and s!=0 else 0
            # Note: for s=0, we handle the quadratic separately
            
            # p+s valid: 0 <= p+s < n → -p <= s < n-p
            # p-s valid: 0 <= p-s < n → p-n < s <= p, i.e., s >= p-n+1 to s <= p
            
            coeff = np.zeros(2*n - 1)  # indexed by k = s + (n-1)
            for s in range(-(n-1), n):
                k = s + (n - 1)
                c = 0.0
                # First factor: h[p] * (1-h[p+s])
                if 0 <= p + s < n:
                    c += (1.0 - h[p + s])
                # Second factor: h[p-s] * (1-h[p]) → contribution to c_s is -h[p-s] * h[p]
                if s != 0 and 0 <= p - s < n:
                    c -= h[p - s]
                coeff[k] = c * dx
            
            # Quadratic coefficient: only for k=0 (s=0)
            quad_coeff = np.zeros(2*n - 1)
            quad_coeff[n-1] = -dx  # for s=0, the x² term is -dx*x²
            
            # Now solve: minimize max_k (base_corr[k] + coeff[k]*x + quad_coeff[k]*x²) + lam*x
            # We'll do this with a grid search and a simple λ update
            
            best_x = hp
            best_val = np.inf
            for x in x_grid:
                vals = base_corr + coeff * x + quad_coeff * x * x
                v = np.max(vals)
                if v < best_val:
                    best_val = v
                    best_x = x
            
            h[p] = best_x
        
        # Project to constraint
        h = project_onto_constraint(h, target_sum)
    
    return h, compute_c5(h)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
