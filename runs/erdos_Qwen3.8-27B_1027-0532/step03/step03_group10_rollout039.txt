def smooth_objective(h, eps):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]

            grad[:m] -= wk * dx * h[s:]

            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

    return float(np.sum(h) - len(h) / 2.0)

def compute_delta_vec(h, i, j, n):
    half = n - 1
    s_min = min(-i, i-half, -j, j-half)
    s_max = max(half-i, i, half-j, j)
    s_range = np.arange(s_min, s_max + 1)
    delta = np.zeros(len(s_range), dtype=np.float64)
    
    # Case 1: m=i
    mask1 = (i + s_range >= 0) & (i + s_range < n)
    delta[mask1] += h[i + s_range[mask1]] - 1.0
    
    # Case 2: m=j
    mask2 = (j + s_range >= 0) & (j + s_range < n)
    c2 = np.zeros(mask2.sum(), dtype=np.float64)
    js_vals = j + s_range[mask2]
    # default: 1 - h[js]
    c2[:] = 1.0 - h[js_vals]
    # special: js == i → 1
    c2[js_vals == i] = 1.0
    # special: js == j (s=0) → 0
    c2[js_vals == j] = 0.0
    delta[mask2] += c2
    
    # Case 3: m=i-s, skip if s=0 or i-s==j
    mask3 = (i - s_range >= 0) & (i - s_range < n) & (s_range != 0) & (i - s_range != j)
    delta[mask3] += h[i - s_range[mask3]]
    
    # Case 4: m=j-s, skip if s=0 or j-s==i
    mask4 = (j - s_range >= 0) & (j - s_range < n) & (s_range != 0) & (j - s_range != i)
    delta[mask4] -= h[j - s_range[mask4]]
    
    return delta, s_range

       ones_val = val_i[ones]
       zeros_val = val_j[zeros]
