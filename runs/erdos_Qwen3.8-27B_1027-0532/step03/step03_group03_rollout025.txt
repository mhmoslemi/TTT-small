_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.minimum(h[idx] + per, 1.0)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.maximum(h[idx] + per, 0.0)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_max_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_weighted(x, np.arange(2 * len(x) - 1), weights)


def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0

    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 10.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           jac=smooth_max_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            sc_cur = compute_c5(h_cur)
            if sc < sc_cur:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def active_set_polish(h_start, target_sum, n_iter, step_size):

    h = h_start.copy()
    best_h = h.copy()

        corr = overlap_corrs(h)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        threshold = max_val - corr.max() * 0.05
        active = np.where(corr >= threshold)[0]
        
        grad = compute_grad_weighted(h, active, np.ones(len(active)) / len(active))
        
        for trial_lr in [step_size, step_size * 0.5, step_size * 0.25]:

            h_try = project_feasible(h_try, target_sum)
            sc = compute_c5(h_try)
            if sc < best_score:
                best_score = sc
                best_h = h_try.copy()
                h = h_try
                break
    
    return best_h, best_score


def coordinate_polish(h_start, target_sum, n_sweeps, step_size):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            orig_val = h[i]
            for delta in [step_size, -step_size, step_size * 0.5, -step_size * 0.5]:
                new_val = orig_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[i] = new_val
                diff = target_sum - np.sum(h)
                if abs(diff) > 1e-10:
                    partner = (i + 1) % n
                    h[partner] -= diff
                    if h[partner] < 0 or h[partner] > 1:
                        h[i] = orig_val
                        h[partner] += diff
                        continue
                sc = compute_c5(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
                    improved = True
                    break
            if not improved:
                h[i] = orig_val
        if sweep % 10 == 9:
            step_size *= 0.8
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    x_old = np.linspace(0.0, 1.0, n_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
