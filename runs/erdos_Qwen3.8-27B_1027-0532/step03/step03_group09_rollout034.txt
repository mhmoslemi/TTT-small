    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def fast_subgrad(h, topk_idx, w, n):
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def subgrad_momentum(h_init, target_sum, n_iter, top_k, temp, lr_init, time_limit, start_time):
    n = len(h_init)
    h_cur = h_init.astype(float).copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_init
    mom = np.zeros(n)
    beta = 0.85
    no_improve = 0

    for t in range(n_iter):
        if time.time() - start_time > time_limit:
            break

        corr = compute_corr(h_cur)
        cmax = np.max(corr)
        topk_idx = np.argsort(corr)[-top_k:]
        xc = corr[topk_idx] - np.max(corr)
        w = np.exp(xc / temp)
        w = w / np.sum(w)
        grad = fast_subgrad(h_cur, topk_idx, w, n)
        mom = beta * mom - lr * grad
        h_new = project_onto_constraint(h_cur + mom, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.4)
        else:
            no_improve += 1
            if no_improve > 200:
                lr = max(lr * 0.7, 1e-7)
                mom *= 0.3
                no_improve = 0
            else:
                lr = max(lr * 0.97, 1e-8)

        h_cur = h_new

    return best_h, best_c5

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            g[s:] += wk * dx * omh[:m]
            g[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            g[:m] += wk * dx * omh[r:]
            g[r:] -= wk * dx * h[:m]
    return g

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                        'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def gen_perturb(h_best, rng, n, target_sum, strategy):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n, size=max(5, n // 8), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        max_len = max(3, n // 6)
        blen = int(rng.integers(3, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n), target_sum)
    elif strategy == 3:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.4, 0.85))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    elif strategy == 4:
        freq = rng.integers(2, max(3, n // 20))
        ph = rng.uniform(0, 2 * np.pi)
        h_p = h_best + 0.05 * np.sin(2 * np.pi * freq * np.arange(n) / n + ph)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 5:
        h_p = h_best.copy()
        split = n // 2
        h_p[split:] = h_best[:split]
        h_p[:split] = h_best[split:]
        h_p = project_onto_constraint(h_p, target_sum)
        h_p = h_p + 0.01 * rng.standard_normal(n)
        return project_onto_constraint(h_p, target_sum)
    else:
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)

def upscale(h, n_new):
    n_old = len(h)
    old_t = np.linspace(0.0, 1.0, n_old)
    new_t = np.linspace(0.0, 1.0, n_new)
    h_up = np.interp(new_t, old_t, h)
    return project_onto_constraint(h_up, n_new / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time

    def rem():
        return budget_s - elapsed()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(180), 0.0, 1.0)

    n0 = len(h0)

    # Upscale to n=200 for better resolution
    n_target = 200
    if n0 < n_target:
        h0 = upscale(h0, n_target)
    elif n0 > n_target:
        n_target = n0
    h0 = project_onto_constraint(h0, n_target / 2.0)
    n = n_target

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick SLSQP at coarse level
    if rem() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.02, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Subgradient with momentum (top_k=8)
    if rem() > 100:
        tl = min(rem() - 60, 200)
        h_sg, c5_sg = subgrad_momentum(best_h, n / 2.0, 200000, 8, 0.02, 0.06, tl + start_time,
                                      start_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: SLSQP medium
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Subgradient (top_k=3, finer temp)
    if rem() > 80:
        tl = min(rem() - 50, 180)
        h_sg, c5_sg = subgrad_momentum(best_h, n / 2.0, 200000, 3, 0.005, 0.04, tl + start_time,
                                      start_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 5: SLSQP fine
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Restart loop with diverse perturbations
    n_restart = 0
    max_restarts = 30
    eps_cycle = [0.003, 0.002, 0.001, 0.0005, 0.0003]
    topk_cycle = [5, 3, 2, 2, 1]

    while rem() > 120 and n_restart < max_restarts:
        strategy = n_restart % 6
        h_pert = gen_perturb(best_h, rng, n, n / 2.0, strategy)

        eps_use = eps_cycle[min(n_restart // 6, len(eps_cycle) - 1)]
        tk = topk_cycle[min(n_restart // 6, len(topk_cycle) - 1)]

        if rem() > 70:
            h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, 150)
        else:
            h_opt, c5_opt = subgrad_momentum(h_pert, n / 2.0, 50000, tk, 0.005, 0.03,
                                            rem() * 0.4 + start_time, start_time)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if rem() > 50 and (n_restart % 2 == 0):
            tl_pg = min(rem() - 40, 40)
            h_pg, c5_pg = subgrad_momentum(best_h, n / 2.0, 50000, tk, 0.003, 0.03,
                                          tl_pg + start_time, start_time)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        n_restart += 1

    # Phase 7: Resolution increase to 250
    if rem() > 200:
        n1 = 250
        h_up = upscale(best_h, n1)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt1, c5_opt1 = slsqp_optimize(h_up, 0.01, 200)
        if rem() > 100:
            h_sg1, c5_sg1 = subgrad_momentum(h_opt1, n1 / 2.0, 100000, 5, 0.01, 0.05,
                                            rem() * 0.5 + start_time, start_time)
            if c5_sg1 < c5_opt1:
                h_opt1 = h_sg1
                c5_opt1 = c5_sg1
        if rem() > 60:
            h_opt1, c5_opt1 = slsqp_optimize(h_opt1, 0.002, 200)
        if c5_opt1 < best_c5:
            best_c5 = c5_opt1
            best_h = h_opt1.copy()
            n = n1

    # Phase 8: Final SLSQP polish
    if rem() > 40:
        eps_final = 0.0002
        h_opt, c5_opt = slsqp_optimize(best_h, eps_final, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Very fine subgradient
    if rem() > 30:
        tl = min(rem() - 15, 60)
        h_sg, c5_sg = subgrad_momentum(best_h, len(best_h) / 2.0, 100000, 1, 0.001, 0.02,
                                      tl + start_time, start_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Final
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
