def _slsqp_opt(h0, n, target_sum, maxiter, alpha):
    corr = _corr_of(h0)
    k_max = int(np.argmax(corr))
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        c = _corr_of(x)
        m = np.max(c)
        smooth_max = m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha
        return smooth_max * (2.0 / n)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(_obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)

                m = n - lag
                g[lag:lag + m] += (1.0 - y[:m])
                g[:m] += -y[lag:lag + m]
            else:
                j = -lag
                m = n - j
                g[:m] += (1.0 - y[j:])
                g[j:] += -y[:m]

        y_new = _fix_sum(y_new, target_sum)

        y = y_new
        t = t_new

        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref

    if time.time() < deadline - 30:
        h_bin_best = _binarize(best_h, best_n / 2.0)
        sa2_deadline = min(deadline - 20, time.time() + 120)
        h_sa2, c5_sa2 = _binary_sa(h_bin_best, best_n, best_n / 2.0, 60000, 0.01, 0.0003,
                                    sa2_deadline, rng, top_k=5)
        if c5_sa2 < best_c5:
            best_c5 = c5_sa2
            best_h = h_sa2.copy()

    if time.time() < deadline - 15:
        h_ref2, c5_ref2 = _cont_refine(best_h, best_n, best_n / 2.0, deadline - 5)
        if c5_ref2 < best_c5:
            best_c5 = c5_ref2
            best_h = h_ref2

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
