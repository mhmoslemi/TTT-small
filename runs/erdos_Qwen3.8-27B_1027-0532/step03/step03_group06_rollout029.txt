import numpy as np

def compute_c5(h):

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_1d_max_lines(a, c, t_lo=0.0, t_hi=1.0):
    """Solve min_{t in [t_lo, t_hi]} max_k (a[k]*t + c[k])
    where a and c are arrays. Returns the optimal t."""
    n_lines = len(a)
    if n_lines == 0:
        return t_lo
    
    # The function f(t) = max_k (a[k]*t + c[k]) is convex piecewise linear.
    # We find its minimum on [t_lo, t_hi].
    # Strategy: binary search on the subgradient.
    # f'(t) (subgradient) contains 0 at the minimum.
    # The subgradient at t is the convex hull of {a[k] : a[k]*t + c[k] = f(t)}.
    
    # Simple approach: since f is convex, use ternary search or bisection on subgradient.
    # For efficiency, let's use a fine grid + refinement.
    
    # Actually, the cleanest approach for piecewise linear convex:
    # The minimum is at a breakpoint where two lines intersect and the slope changes sign.
    # Or at a boundary.
    
    # Let's evaluate at a set of candidate points:
    # - t_lo, t_hi
    # - Intersection points of pairs of lines (expensive: O(n^2))
    # - Midpoints between sorted a values
    
    # For n=400, O(n^2) intersections is 160K, which is fine.
    # But let's use a simpler approach: bisection on the slope.
    
    lo = t_lo
    hi = t_hi
    
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        # Compute f(mid) and check subgradient
        vals = a * mid + c
        f_val = np.max(vals)
        # Active lines (within tolerance)
        tol = 1e-10 * (1 + abs(f_val))
        active = vals >= f_val - tol
        slopes = a[active]
        if len(slopes) == 0:
            break
        min_slope = np.min(slopes)
        max_slope = np.max(slopes)
        if min_slope >= -1e-12:
            # Function is increasing (or flat) at mid, minimum is to the left
            hi = mid
        elif max_slope <= 1e-12:
            # Function is decreasing at mid, minimum is to the right
            lo = mid
        else:
            # 0 is in the subgradient, we're at the minimum
            lo = hi = mid
            break
    
    return (lo + hi) / 2

def coord_descent(h, n, max_passes=50, tol=1e-12):
    """Coordinate descent for min max_k overlap(k) s.t. 0<=h<=1, sum(h)=n/2."""
    target_sum = n / 2.0
    h = h.copy().astype(float)
    
    for p in range(max_passes):
        max_change = 0.0
        # Compute current overlaps (for all lags)
        # We'll use the formula: overlap(d) = sum_i h[i]*(1-h[i+d]) for valid i, i+d
        # In terms of np.correlate: corr[m] for m=0..2n-2, where lag = m-(n-1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        for i in range(n):
            # Compute a_k and c_k for each lag k
            # For lag d: overlap(d) includes terms:
            #   h[i]*(1-h[i+d]) if i+d in [0,n)
            #   h[i-d]*(1-h[i]) if i-d in [0,n)
            # So: overlap(d) = a_d * h[i] + c_d
            # where a_d = (1-h[i+d]) if i+d<n else 0) + (-h[i-d]) if i-d>=0 else 0)
            #       c_d = overlap(d)_current - a_d * h[i]
            
            # Instead of iterating over all d, let's compute the effect directly.
            # The overlap at lag d (corresponding to correlate index d+n-1) is:
            # corr[d+n-1] = sum_j h[j]*(1-h[j+d]) for valid j
            # 
            # Terms involving h[i]:
            # j=i: h[i]*(1-h[i+d]) [if i+d in [0,n)]
            # j=i-d: h[i-d]*(1-h[i]) [if i-d in [0,n)]
            #
            # So a_d = (1-h[i+d])*[i+d<n] - h[i-d]*[i-d>=0]
            # c_d = corr[d+n-1] - a_d * h[i]
            
            # For the 1D problem, we need to find t = argmin max_d (a_d*t + c_d)
            # But we only need to consider d values where a_d ≠ 0 or where the term is active.
            
            # For efficiency, note that a_d ≠ 0 only for d where i+d or i-d is in range
            # and the corresponding h value is non-zero/non-one.
            
            # Let's just use all d values (there are 2n-1 of them).
            d_vals = np.arange(-(n-1), n)  # all possible lags
            idx = d_vals + (n-1)  # indices into corr
            
            # a_d for each d
            a = np.zeros(2*n-1)
            # i+d in [0,n) means d in [-i, n-1-i]
            valid_plus = (d_vals >= -i) & (d_vals <= n-1-i)
            a[valid_plus] += (1.0 - h[i + d_vals[valid_plus]])
            # i-d in [0,n) means d in [i-(n-1), i]
            valid_minus = (d_vals >= i-(n-1)) & (d_vals <= i)
            a[valid_minus] -= h[i - d_vals[valid_minus]]
            
            # c_d = corr[idx] - a * h[i]
            c = corr[idx] - a * h[i]
            
            # Only consider d values where a != 0 (others don't constrain t)
            mask = np.abs(a) > 1e-15
            if not np.any(mask):

            t_opt = solve_1d_max_lines(a[mask], c[mask])
            new_val = np.clip(t_opt, 0.0, 1.0)
            max_change = max(max_change, abs(new_val - h[i]))
            h[i] = new_val
        
        # Project onto sum constraint
        h = project_feasible(h, target_sum)
        
        if max_change < tol:
            break
    
    return h

Actually, for the binary case, a simpler and effective approach is:
1. Start from a binary solution
2. For each 1-position a, compute the "damage" of removing it (how much the max increases)
3. For each 0-position b, compute the "benefit" of adding it (how much the max decreases)
4. Find the swap (a, b) that maximally reduces the max

The "damage" of removing position a (making h[a]=0):
- For each d: overlap(d) changes by ΔO(d) = -[h[a+d]=0] + [h[a-d]=1] (with validity)
- The max might increase if any O(d) that was below the max now exceeds it

The "benefit" of adding position b (making h[b]=1):
- ΔO(d) = +[h[b+d]=0] - [h[b-d]=1]
- The max might decrease if the max lag's overlap decreases

For a swap (remove a, add b):
- ΔO(d) = -[h[a+d]=0] + [h[a-d]=1] + [h[b+d]=0] - [h[b-d]=1]
- We want to find (a,b) that minimizes max_d (O(d) + ΔO(d))

This is still O(n²) per iteration if done naively, but with n=400 and the operations being simple, it might be OK if I vectorize.

Let me try a different approach: **subgradient method** on the max function.

The objective is F(h) = max_k f_k(h) where f_k(h) = sum_i h[i](1-h[i+k]) * dx.

The subgradient of F at h (with respect to h[i]) is:
∂F/∂h[i] = (1/dx) * sum_{k ∈ K*} ∂f_k/∂h[i]

where K* = {k : f_k(h) = F(h)} (the set of active lags).

∂f_k/∂h[i] = dx * [(1-h[i+k]) if i+k valid] + dx * [-h[i-k]] if i-k valid]
             = dx * [(1-h[i+k])*1_{i+k<n} - h[i-k]*1_{i-k>=0}]

So ∂F/∂h[i] = sum_{k ∈ K*} [(1-h[i+k])*1_{i+k<n} - h[i-k]*1_{i-k>=0}]

This is a valid subgradient, and we can use projected subgradient descent:
h[i] ← clip(h[i] - α * ∂F/∂h[i], 0, 1)
then project onto sum constraint.

This is simple to implement and should work. The step size α can be adaptive (e.g., α_t = c / sqrt(t)).

    target_sum = n / 2.0
    h = h.copy().astype(float)
    best_h = h.copy()
    best_score = compute_c5(h)
    for p in range(max_passes):
        if time_limit_fn is not None and time_limit_fn() < 40:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_change = 0.0
        for i in range(n):
            a = np.zeros(2 * n - 1)
            m_lo = max(0, n - 1 - i)
            m_hi = min(2 * n - 2, 2 * n - 2 - i)
            if m_lo <= m_hi:
                d_vals = np.arange(m_lo, m_hi + 1) - (n - 1)
                a[m_lo:m_hi + 1] += (1.0 - h[i + d_vals])
            m_lo2 = max(0, i)
            m_hi2 = min(2 * n - 2, i + n - 1)
            if m_lo2 <= m_hi2:
                d_vals2 = np.arange(m_lo2, m_hi2 + 1) - (n - 1)
                a[m_lo2:m_hi2 + 1] -= h[i - d_vals2]
            c = corr - a * h[i]
            mask = np.abs(a) > 1e-14
            if not np.any(mask):
                continue
            t_opt = solve_1d_convex_pl(a[mask], c[mask])
            new_val = float(np.clip(t_opt, 0.0, 1.0))
            change = abs(new_val - h[i])
            if change > max_change:
                max_change = change
            h[i] = new_val
        h = project_feasible(h, target_sum)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
        if max_change < tol:
            break
    return best_h, best_score

def subgradient_descent(h, n, max_iter=300, step_init=0.3, step_decay=0.995):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h.copy().astype(float)
    step = step_init
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        f_vals = corr * dx
        max_val = np.max(f_vals)
        tol = 1e-6 * max(1.0, max_val)
        active = f_vals >= max_val - tol
        n_active = float(np.sum(active))
        if n_active == 0:
            break
        grad = np.zeros(n)
        active_idx = np.where(active)[0]
        for k_idx in active_idx:
            d = k_idx - (n - 1)
            if d >= 0:
                if d < n:
                    grad[:n - d] += (1.0 - h[d:n])
                if d > 0:
                    grad[d:] -= h[:n - d]
            else:
                dd = -d
                if dd < n:
                    grad[dd:] += (1.0 - h[:n - dd])
                if dd > 0:
                    grad[:n - dd] -= h[dd:]
        grad /= n_active
        h_new = h - step * grad
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        step *= step_decay
        step = max(step, 0.01)
    return best_h, best_score

def round_to_binary(h, n):
    target = n // 2
    h_sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[h_sorted_idx[:target]] = 1.0
    return h_bin

def binary_local_search(h_bin, n, max_iters=500, seed=42, time_limit_fn=None):
    h = h_bin.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for it in range(max_iters):
        if time_limit_fn is not None and time_limit_fn() < 35:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        max_lag_idx = np.argmax(corr)
        d_max = max_lag_idx - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if d_max >= 0:
            valid_i = np.arange(0, n - d_max)
        else:
            valid_i = np.arange(-d_max, n)
        contributing = [i for i in valid_i if h[i] == 1 and h[i + d_max] == 0]
        improved = False
        for a in contributing[:30]:
            for _ in range(min(30, len(zeros))):
                b = rng.choice(zeros)
                if a == b:
                    continue
                h[a] = 0
                h[b] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_corr)
                h[a] = 1
                h[b] = 0
                if new_max < max_val:
                    h[a] = 0
                    h[b] = 1
                    new_score = float(new_max) * (2.0 / n)
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                    max_val = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            for _ in range(100):
                a = rng.choice(ones)
                b = rng.choice(zeros)
                if a == b:
                    continue
                h[a] = 0
                h[b] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_corr)
                if new_max < max_val:
                    new_score = float(new_max) * (2.0 / n)
                    if new_score < best_score:
                        best_score = new_score
                        best_h = h.copy()
                    max_val = new_max
                    improved = True
                    break
                h[a] = 1
                h[b] = 0
            if not improved:
                break
    return best_h, best_score

def smooth_max_obj(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    return m + np.log(sum_exp) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    grad = np.zeros(n)
    for i in range(n):
        m_lo = max(0, n - 1 - i)
        m_hi = min(2 * n - 2, 2 * n - 2 - i)
        d_vals = np.arange(m_lo, m_hi + 1) - (n - 1)
        grad[i] += np.sum(w[m_lo:m_hi + 1])
        grad[i] -= np.sum(w[m_lo:m_hi + 1] * h[i + d_vals])
        m_lo2 = max(0, i)
        m_hi2 = min(2 * n - 2, i + n - 1)
        d_vals2 = np.arange(m_lo2, m_hi2 + 1) - (n - 1)
        grad[i] -= np.sum(w[m_lo2:m_hi2 + 1] * h[i - d_vals2])
    return grad

def lbfgs_smooth(h_init, n, target, alpha, maxiter, time_limit_fn=None):
    h = h_init.copy()
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(smooth_max_obj, h, jac=smooth_max_grad, args=(alpha,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent (exact 1D subproblems)
    if tleft() > 120:
        h_cd, sc = coord_descent(best_h, n_init, max_passes=300, tol=1e-12, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 2: Subgradient descent for global mixing
    if tleft() > 100:
        h_sg, sc = subgradient_descent(best_h, n_init, max_iter=400, step_init=0.4, step_decay=0.997)
        if sc < best_score:
            best_score = sc
            best_h = h_sg
        # Follow with coordinate descent
        if tleft() > 80:
            h_cd2, sc2 = coord_descent(h_sg, n_init, max_passes=150, tol=1e-11, time_limit_fn=tleft)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_cd2

    # Phase 3: Multi-start coordinate descent with perturbations
    for attempt in range(15):
        if tleft() < 80:
            break
        ptype = attempt % 3
        if ptype == 0:
            amp = 0.02 * (1.0 + 0.3 * attempt)
            h_pert = project_feasible(best_h + rng.randn(n_init) * amp, target_init)
        elif ptype == 1:
            h_pert = best_h.copy()
            seg = rng.randint(5, max(6, n_init // 6))
            a = rng.randint(0, n_init - seg)
            b = rng.randint(0, n_init - seg)
            if abs(a - b) > seg:
                tmp = h_pert[a:a + seg].copy()
                h_pert[a:a + seg] = h_pert[b:b + seg]
                h_pert[b:b + seg] = tmp
            h_pert = project_feasible(h_pert, target_init)
        else:
            shift = rng.randint(1, max(2, n_init // 5))
            h_pert = np.roll(best_h, shift)
            h_pert = project_feasible(h_pert, target_init)
        h_cd, sc = coord_descent(h_pert, n_init, max_passes=100, tol=1e-10, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 4: L-BFGS-B with smooth max (higher alpha for sharper max approximation)
    if tleft() > 120:
        for alpha in [200, 500, 1000, 2000]:
            if tleft() < 60:
                break
            h_new = lbfgs_smooth(best_h, n_init, target_init, alpha, 300, tleft)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
        # Final coordinate descent polish
        if tleft() > 60:
            h_cd, sc = coord_descent(best_h, n_init, max_passes=200, tol=1e-12, time_limit_fn=tleft)
            if sc < best_score:
                best_score = sc
                best_h = h_cd

    # Phase 5: Binary local search
    if tleft() > 60:
        h_bin = round_to_binary(best_h, n_init)
        h_bls, sc = binary_local_search(h_bin, n_init, max_iters=400, seed=seed, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_bls
        # Also try binary search from a perturbed continuous solution
        if tleft() > 40:
            h_pert = project_feasible(best_h + rng.randn(n_init) * 0.05, target_init)
            h_bin2 = round_to_binary(h_pert, n_init)
            h_bls2, sc2 = binary_local_search(h_bin2, n_init, max_iters=300, seed=seed + 1, time_limit_fn=tleft)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_bls2

    # Phase 6: Grid refinement to higher resolution
    for n_try in [500, 600, 800]:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cd, sc = coord_descent(h_interp, n_try, max_passes=200, tol=1e-12, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_cd
        # L-BFGS-B at higher resolution
        if tleft() > 80:
            h_new = lbfgs_smooth(h_cd, n_try, target_try, 500, 200, tleft)
            sc2 = compute_c5(h_new)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_new
            if tleft() > 50:
                h_cd2, sc3 = coord_descent(h_new, n_try, max_passes=150, tol=1e-12, time_limit_fn=tleft)
                if sc3 < best_score:
                    best_score = sc3
                    best_h = h_cd2

    # Phase 7: Final polish - alternating coord descent and subgradient
    n_use = len(best_h)
    target_use = n_use / 2.0
    for _ in range(5):
        if tleft() < 40:
            break
        h_cd, sc = coord_descent(best_h, n_use, max_passes=150, tol=1e-13, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_cd
        if tleft() < 40:
            break
        h_sg, sc2 = subgradient_descent(best_h, n_use, max_iter=100, step_init=0.1, step_decay=0.99)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_sg
        if tleft() < 40:
            break
        h_cd2, sc3 = coord_descent(best_h, n_use, max_passes=100, tol=1e-13, time_limit_fn=tleft)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_cd2

    # Phase 8: Micro-perturbation final search
    for _ in range(8):
        if tleft() < 25:
            break
        amp = rng.uniform(0.001, 0.004)
        h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
        h_new, sc = coord_descent(h_pert, n_use, max_passes=80, tol=1e-13, time_limit_fn=tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
