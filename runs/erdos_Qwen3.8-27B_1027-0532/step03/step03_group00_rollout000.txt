def compute_grad_active(h, k, n):
    """Gradient of corr_k with respect to h"""
    g = np.zeros(n)
    # First term: (1-h[i+k]) for i in [0, n-1-k]
    g[:n-k] += (1.0 - h[k:])
    # Second term: -h[i-k] for i in [k, n-1]
    g[k:] -= h[:n-k]
    return g

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_objective(h, alpha):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def compute_active_gradients(h, k_list, n):
    g = np.zeros(n)
    for k in k_list:
        k = int(k)
        if k < 0 or k >= n:
            continue
        g[:n - k] += (1.0 - h[k:n])
        g[k:] -= h[0:n - k]
    return g


def slsqp_refine(h, target, alphas, maxiter, ftol, deadline):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons_sum = target
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

    for a in alphas:
        if time.time() > deadline:
            break

        def obj(x):
            return lse_objective(x, a)

        def cons_fun(x):
            return np.sum(x) - cons_sum

        def cons_jac(x):
            return np.ones(n)

        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass

    return h_cur, cur_score

def block_slsqp(h, target, block_size, n_rounds, deadline, alpha=200.0):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)

            h_full = h_cur

            def block_obj(x):

                hf[s:e] = np.clip(x, 0.0, 1.0)

def fourier_perturb(h, n, target, rng, n_modes=5, amplitude=0.3):
    h_pert = h.copy()
    x = np.linspace(0, 2, n)
    perturb = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = m
        amp = amplitude / m
        phase = rng.random() * 2 * np.pi
        perturb += amp * np.sin(freq * np.pi * x / 2.0 + phase)
    h_pert = h_pert + perturb
    h_pert = project_feasible(h_pert, target)
    return h_pert


def random_block_perturb(h, n, target, rng, block_len=30, amplitude=0.4):
    h_pert = h.copy()
    start = rng.randint(0, n - block_len)
    shift = (rng.random() - 0.5) * amplitude
    h_pert[start:start + block_len] += shift
    # Also try a triangular bump
    center = rng.randint(0, n)
    width = rng.randint(10, 50)
    bump_h = (rng.random() - 0.5) * 0.5
    idx = np.arange(max(0, center - width), min(n, center + width))
    triangle = 1.0 - np.abs(x - center) / float(width)
    triangle = np.maximum(triangle, 0.0)
    h_pert[idx] += bump_h * triangle
    return project_feasible(h_pert, target)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

        return project_feasible(h_src.copy(), target_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
