# So: k from max(0, n-1-i) to min(2n-2, 2n-2-i)
# Since i ≥ 0: 2n-2-i ≤ 2n-2, so upper bound is 2n-2-i
# And n-1-i ≥ 0 when i ≤ n-1 (always), so lower bound is n-1-i
# Second sum = Σ_{k=n-1-i}^{2n-2-i} w[k] * h[i-n+1+k]
# Let j = i-n+1+k: when k=n-1-i, j=0; when k=2n-2-i, j=n-1
# = Σ_{j=0}^{n-1} w[n-1-i+j-n+1] * h[j] = Σ_{j=0}^{n-1} w[j-i] * h[j]
# Hmm: w[n-1-i+k] where k = j-i+n-1: w[n-1-i+j-i+n-1] = w[j-2i+n-1]... that's not right.

# j = i-n+1+k, so k = j-i+n-1

# = Σ_l h[l-(i-n+1)]*w[l] ... this is a cross-correlation
# = np.correlate(h, w, 'full')[i-n+1+n-1] = np.correlate(h, w, 'full')[i]
# 
# Wait: np.correlate(h, w, 'full')[k] = Σ_n h[n+k-(len(w)-1)] * w[n] = Σ_n h[n+k-n+1] * w[n]
# For k=i: = Σ_n h[n+i-n+1] * w[n] = Σ_n h[n+i+1-n] * w[n] = Σ_n h[i+1] * w[n]... that's not right.
#
# I think I'm overcomplicating this. Let me just use:
# B[i] = Σ_{j=0}^{n-1} w[j-i+n-1] * h[j]
# This is the convolution of w and h (or a correlate) at a specific position.
# 
# np.convolve(w_reversed, h, 'full') where w_reversed = w[::-1]
# conv(w_rev, h, 'full')[m] = Σ_j w_rev[j] * h[m-j] = Σ_j w[n-1-j] * h[m-j]
# Let l = n-1-j: j = n-1-l: = Σ_l w[l] * h[m-n+1+l]
# I want B[i] = Σ_j w[j-i+n-1] * h[j]
# Let m-n+1+l = j: l = j-m+n-1: B[i] = Σ_j w[j-m+n-1] * h[j]
# I want the index in w to be j-i+n-1, so: j-m+n-1 = j-i+n-1 → m = i
# So B[i] = np.convolve(w[::-1], h, 'full')[i]
# 
# Let me verify: conv(w_rev, h, 'full')[i] = Σ_j w_rev[j] * h[i-j] = Σ_j w[n-1-j] * h[i-j]
# Let l = n-1-j: j = n-1-l: = Σ_l w[l] * h[i-n+1+l]
# And I want B[i] = Σ_{j=0}^{n-1} w[j-i+n-1] * h[j]
# Let l = j-i+n-1: j = l+i-n+1: B[i] = Σ_l w[l] * h[l+i-n+1]
# ✓ These match! B[i] = np.convolve(w[::-1], h, 'full')[i]
#
# So the gradient is:
# grad[i] = dx * (A[i] - B[i])
# where A = np.convolve(w, 1-h, 'full')[n-1 : 2n-2]  (indices n-1 to 2n-2, which is n elements)
# and B = np.convolve(w[::-1], h, 'full')[0 : n]  (indices 0 to n-1, which is n elements)

def compute_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = corr - m
    # Clip for numerical stability
    shifted = np.clip(shifted, -50/alpha, 0)
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    A = np.convolve(w, 1.0 - h, mode='full')[n-1 : 2*n-2]
    B = np.convolve(w[::-1], h, mode='full')[0 : n]
    
    grad = dx * (A - B)
    return grad

from scipy.optimize import minimize, differential_evolution
from scipy.signal import fftconvolve
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_overlap(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        if diff > 0:
            space = 1.0 - h[idx]
            total = np.sum(space)
            if total > 1e-15:
                h[idx] += space / total * diff
            else:
                h[idx] += diff / len(idx)
        else:
            space = h[idx]
            total = np.sum(space)
            if total > 1e-15:
                h[idx] -= space / total * (-diff)
            else:
                h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_value_and_weights(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = np.max(corr)
    shifted = np.clip(corr - m, -100.0 / alpha, 0.0)
    exp_vals = np.exp(alpha * shifted)
    w = exp_vals / np.sum(exp_vals)
    F = m + np.log(np.sum(exp_vals)) / alpha
    return F, w, corr


def smooth_max_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    _, w, _ = smooth_max_value_and_weights(h, alpha)
    w_rev = w[::-1]
    A = fftconvolve(w, 1.0 - h, mode='full')[n - 1: 2 * n - 2]
    B = fftconvolve(w_rev, h, mode='full')[0:n]
    grad = dx * (A - B)
    return grad


def smooth_objective(h, alpha):
    F, _, _ = smooth_max_value_and_weights(h, alpha)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 1.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgs_optimize(h_start, n, target, alpha, maxiter, ftol, time_fn=None, min_time_left=40):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

        return smooth_objective(x, alpha)

    def jac_func(x):
        return smooth_max_gradient(x, alpha)

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'gtol': 1e-8, 'maxls': 50})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur
