def slsqp_smooth(h, target_sum, alpha, maxiter=200):
    """SLSQP with log-sum-exp smoothing of the max."""
    n = len(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        # Gradient of max smoothed by log-sum-exp
        # d/dx_j [sum_s w_s * corr_s] where corr_s = sum_j x_j (1-x_{j+s})
        s_arr = np.arange(2 * n - 1) - (n - 1)

            js_minus = np.arange(n) - s_val
            m1 = (js_plus >= 0) & (js_plus < n)
            m2 = (js_minus >= 0) & (js_minus < n)
            contrib = np.zeros(n)
            contrib[m1] += (1.0 - x[js_plus[m1]])
            contrib[m2] -= x[js_minus[m2]]
            g += weights[idx] * contrib
        return g * dx
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                   'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-16})

    threshold = c_max - n_active_frac * c_max

        js_minus = np.arange(n) - s_val
