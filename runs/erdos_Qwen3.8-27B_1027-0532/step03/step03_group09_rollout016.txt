                stall = 0
    return best_h, best_c5

def momentum_topk(h_init, k, temp, lr, mom, n_steps, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    v = np.zeros(n)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    stall = 0
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        h_look = project_onto_constraint(h_cur - mom * v, target_sum)
        grad = topk_subgrad(h_look, k, temp)
        v = mom * v + grad
        h_new = project_onto_constraint(h_cur - lr * v, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 200:
            h_cur = best_h.copy()
            v *= 0.5
            lr = max(lr * 0.7, 1e-5)
            stall = 0
        else:
            h_cur = h_new
    return best_h, best_c5

def gen_perturb(h_best, rng, n0, target_sum, strategy):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.015 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        for _ in range(3):
            f = int(rng.integers(1, 7))
            phase = rng.uniform(0, 2.0 * np.pi)
            h_p += 0.025 * np.cos(2.0 * np.pi * f * np.arange(n0) / n0 + phase)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        h_p = h_best.copy()
        max_len = max(2, n0 // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
        h_bin = (rng.random(n0) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.3, 0.6))
        return project_onto_constraint(alpha * h_best + (1.0 - alpha) * h_bin, target_sum)
    elif strategy == 4:
        h_p = h_best.copy()
        mask = rng.random(n0) < 0.08
        h_p[mask] = 1.0 - h_p[mask]
        return project_onto_constraint(h_p, target_sum)
    else:
        idx = rng.choice(n0, size=max(4, n0 // 12), replace=False)
        h_p = h_best.copy()
        h_p[idx] += 0.1 * rng.standard_normal(len(idx))
        return project_onto_constraint(h_p, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 200
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP warmup
    if rem() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Barrier method (primary optimizer)
    if rem() > 120:
        tl = min(rem() - 60, 300)
        h_bar, c5_bar = barrier_optimize(best_h, 5.0, 500.0, 12, 500, tl, start_time)
        if c5_bar < best_c5:
            best_c5 = c5_bar
            best_h = h_bar.copy()

    # Phase 3: Momentum top-k for fine-tuning
    if rem() > 80:
        tl = min(rem() - 40, 150)
        h_pg, c5_pg = momentum_topk(best_h, 3, 0.008, 0.04, 0.9, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: Fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 350)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: More barrier with higher mu
    if rem() > 100:
        tl = min(rem() - 50, 200)
        h_bar, c5_bar = barrier_optimize(best_h, 100.0, 2000.0, 10, 400, tl, start_time)
        if c5_bar < best_c5:
            best_c5 = c5_bar
            best_h = h_bar.copy()

    # Phase 6: Single worst momentum
    if rem() > 60:
        tl = min(rem() - 30, 100)
        h_pg, c5_pg = momentum_topk(best_h, 1, 0.003, 0.02, 0.85, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 7: Restart loop
    no_improve = 0
    n_restart = 0
    eps_cycle = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
    while rem() > 90 and n_restart < 35 and no_improve < 12:
        strategy = n_restart % 6
        h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)
        eps_use = eps_cycle[min(n_restart // 7, len(eps_cycle) - 1)]

        if rem() > 100:
            h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, 150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                no_improve = 0
            else:
                no_improve += 1

        if rem() > 50:
            tl_pg = min(rem() - 30, 40)
            h_pg, c5_pg = momentum_topk(best_h, 2, 0.005, 0.025, 0.88, 30000, tl_pg, start_time)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                no_improve = 0

        if rem() > 80 and strategy % 2 == 0:
            tl_bar = min(rem() - 40, 50)
            h_bar, c5_bar = barrier_optimize(best_h, 50.0, 500.0, 5, 200, tl_bar, start_time)
            if c5_bar < best_c5:
                best_c5 = c5_bar
                best_h = h_bar.copy()
                no_improve = 0

        n_restart += 1

    # Phase 8: Very fine SLSQP
    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final barrier polish
    if rem() > 40:
        tl = min(rem() - 20, 80)
        h_bar, c5_bar = barrier_optimize(best_h, 200.0, 5000.0, 8, 300, tl, start_time)
        if c5_bar < best_c5:
            best_c5 = c5_bar
            best_h = h_bar.copy()

    # Phase 10: Try higher resolution
    if rem() > 120:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = project_onto_constraint(h_up + 0.005 * rng.standard_normal(n1), n1 / 2.0)

        h_opt_up, c5_up = slsqp_optimize(h_up, 0.005, 200)
        if rem() > 60:
            h_opt_up, c5_up2 = slsqp_optimize(h_opt_up, 0.001, 200)
            if c5_up2 < c5_up:
                c5_up = c5_up2
                h_opt_up = h_opt_up
        if rem() > 40:
            tl_up = min(rem() - 20, 60)
            h_bar_up, c5_bar_up = barrier_optimize(h_opt_up, 20.0, 200.0, 6, 300, tl_up, start_time)
            if c5_bar_up < c5_up:
                c5_up = c5_bar_up
                h_opt_up = h_bar_up
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_opt_up.copy()

    # Final
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
