        if m_lo > m_hi:
            continue
        
        # For each m in [m_lo, m_hi], compute the slope
        # lag k = m - (n-1)
        # If 0 <= j+k < n: slope += (1 - h[j+k])
        # If 0 <= j-k < n: slope -= h[j-k]
        
        ms = np.arange(m_lo, m_hi + 1)
        ks = ms - (n - 1)
        
        # First condition: 0 <= j+k < n
        idx1 = j + ks  # indices into h
        mask1 = (idx1 >= 0) & (idx1 < n)
        
        # Second condition: 0 <= j-k < n
        idx2 = j - ks
        mask2 = (idx2 >= 0) & (idx2 < n)
        
        a_vec = np.zeros(len(ms))
        if np.any(mask1):
            a_vec[mask1] += (1.0 - h[idx1[mask1]])
        if np.any(mask2):
            a_vec[mask2] -= h[idx2[mask2]]
        
        # Add Lagrange multiplier
        a_vec += lam
        
        # Compute b_vec: b = corr[m] - a * old_val
        # But note: the a_vec here includes lam, so:
        # corr[m] + lam * old_val = (a_vec_k) * t + b  where a_vec_k is without lam
        # corr[m] = a_k * old_val + b  →  b = corr[m] - a_k * old_val
        # And we want to minimize max_k (a_k * t + b_k) + lam * t = max_k ((a_k + lam) * t + b_k)
        # So b_k = corr[m] - a_k * old_val (where a_k is WITHOUT lam)
        
        # Let me recompute without lam for b:
        a_vec_nolam = np.zeros(len(ms))
        if np.any(mask1):
            a_vec_nolam[mask1] += (1.0 - h[idx1[mask1]])
        if np.any(mask2):
            a_vec_nolam[mask2] -= h[idx2[mask2]]
        
        b_vec = corr[ms] - a_vec_nolam * old_val
        
        # Find optimal t
        t_new = find_optimal_t_vec(a_vec, b_vec)
        
        delta = t_new - old_val
        if abs(delta) > 1e-12:
            h[j] = t_new
            # Update corr incrementally
            corr[ms] += a_vec_nolam * delta
            changed += 1
    
    return h, changed

    if abs(diff) > 1e-10:

        h = np.clip(h, 0, 1)

    def remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    if remaining() > 600:

        corr = compute_corr(h)
        lam = 0.0
        eta_lam = 1.0 / n
        patience = 0

        phase_start = time.time()
        
        while time.time() - phase_start < phase_limit:
            h, changed = cd_sweep(h, corr, n, lam, rng)
            
            # Update Lagrange multiplier
            sum_h = np.sum(h)
            err = sum_h - target_sum
            if abs(err) > 0.5:
                lam += eta_lam * err

                corr = compute_corr(h)
            elif abs(err) > 0.01:
                lam += 0.5 * eta_lam * err

                if score < best_score - 1e-12:

                    best_h = h.copy()
                    patience = 0
                else:
                    patience += 1

                patience += 1

                h = best_h + 0.01 * np.random.RandomState(seed + int(time.time())).randn(n)
                h = np.clip(h, 0, 1)
                # Project to sum constraint
                diff = target_sum - np.sum(h)
                h += diff / n

                corr = compute_corr(h)
                patience = 0
