**Alternating Linearization (AL) approach:**

At iteration t:
- We have h^(t)
- Define w_i = 1 - h^(t)[i] (fixed weights)
- Solve: min_{h, t} t
  s.t. sum_{i=0}^{n-1-k} h[i+k] * w_i <= t for all k

       0 <= h <= 1

This is a linear program! The constraint for lag k is:
sum_{i=0}^{n-1-k} w_i * h[i+k] <= t

Let j = i+k: sum_{j=k}^{n-1} w_{j-k} * h[j] <= t

So the LP has constraints:
- For each k = 0, ..., n-1: sum_{j=k}^{n-1} w_{j-k} * h[j] <= t
- sum(h) = n/2
- 0 <= h <= 1

This is a feasible LP with O(n) variables and O(n^2) constraints (n lags, each with a linear constraint on h).

Actually, the number of constraints is n (one per lag) plus the sum constraint. So it's O(n) constraints!

For n=400, this is a 401-variable LP with ~402 constraints. Very manageable!

1. Initialize h (from initial_h_values)
2. Repeat:

def grad_lag(h, m):
    """Gradient of c_m = sum_i h[i+m]*(1-h[i]) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if m == 0:
        grad[:] = 1.0 - 2.0 * h
    elif 0 < m < n:

        # Region 2: j in [max(m,0), min(n-m, n)) -> (1-h[j-m]) - h[j+m]

            js = np.arange(r2s, r2e)
            grad[r2s:r2e] = (1.0 - h[js - m]) - h[js + m]

def subgradient_descent(h_init, target, deadline, top_k=5, base_alpha=0.5, momentum=0.7):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)  # momentum
    
    it = 0
    alpha = base_alpha
    while time.time() < deadline:
        it += 1
        corr = compute_corr(h)
        # Top k lags
        top_idx = np.argsort(corr)[-top_k:]
        vals = corr[top_idx]
        # Softmax weights (temperature = 1/max_val for scaling)
        temp = max(vals.max() * 0.1, 1e-6)
        w = np.exp((vals - vals.max()) / temp)
        w /= w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for k_idx in range(top_k):
            # Convert correlation index to lag
            lag = abs(top_idx[k_idx] - (n - 1))
            g = grad_lag(h, lag)
            grad += w[k_idx] * g
        
        grad *= dx  # scale to C5 units
        
        # Momentum
        v = momentum * v - alpha * grad
        h_new = project_feasible(h + v, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
        else:
            # Step size reduction
            alpha *= 0.8
            v = np.zeros(n)
        
        # Periodically increase alpha
        if it % 50 == 0:
            alpha = min(alpha * 1.1, base_alpha * 2)
    
    return best_h, best_score

                h_new = project_feasible(res.x[:n], target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h = h_new
                if len(cuts) < n_cuts:
                    cuts.append(h.copy())
                else:
                    cuts = [h.copy()] + cuts[1:]
        except Exception:
            break
    return best_h, best_score


def focused_subgrad(h_init, target, deadline, mode='single'):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    alpha = 0.3
    stagnant = 0

    while time.time() < deadline:
        corr = compute_corr(h)
        if mode == 'single':
            k_star = int(np.argmax(corr))
            lag = abs(k_star - (n - 1))
            grad = grad_lag(h, lag) * dx
        else:
            top_k = 5
            top_idx = np.argsort(corr)[-top_k:]
            vals = corr[top_idx]
            temp = max(vals.max() * 0.05, 1e-6)
            w = np.exp((vals - vals.max()) / temp)
            w /= w.sum()
            grad = np.zeros(n)
            for ki in range(top_k):
                lag = abs(int(top_idx[ki]) - (n - 1))
                grad += w[ki] * grad_lag(h, lag)
            grad *= dx

        h_new = project_feasible(h - alpha * grad, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            h = h_new
            stagnant = 0
        else:
            h_fail = project_feasible(h - alpha * 0.1 * grad, target)
            sc_fail = compute_c5(h_fail)
            if sc_fail < compute_c5(h):
                h = h_fail
            alpha *= 0.7
            stagnant += 1

        if stagnant > 20:
            alpha = min(alpha * 2.0, 1.0)
            stagnant = 0

        if it_get_count() > 200000:
            break
    return best_h, best_score

    return _g_it_count


def slsqp_round(h, n, target, alphas, maxiter, ftol, deadline=None):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        if deadline is not None and time.time() > deadline:
            break
        _g_alpha = a
        try:
            res = minimize(_lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur


def _lse_obj(x):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def _cons_eq(x):
    return np.sum(x) - _g_target


def _cons_jac(x):
    return np.ones(_g_n)


def _block_obj(x):
    h_full = _g_block_full.copy()
    s = _g_block_start
    e = _g_block_end
    h_full[s:e] = np.clip(x, 0.0, 1.0)
    h_full = project_feasible(h_full, _g_target)
    corr = compute_corr(h_full)
    c_max = float(np.max(corr))
    shifted = _g_alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_target, _g_alpha
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_block_full = h_cur
            _g_block_start = s
            _g_block_end = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                              options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-15})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur


def periodic_perturb(h, n, target, period, amplitude, rng):
    pattern = rng.uniform(-amplitude, amplitude, period)
    full_pattern = np.tile(pattern, n // period + 1)[:n]
    h_new = h + full_pattern
    return project_feasible(h_new, target)


def selective_binary_push(h, n, target, k_push, rng):
    h_new = h.copy()
    indices = rng.choice(n, size=k_push, replace=False)
    for idx in indices:
        if h_new[idx] < 0.5:
            h_new[idx] = 1.0
        else:
            h_new[idx] = 0.0
    return project_feasible(h_new, target)


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_block_full, _g_block_start, _g_block_end
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 45.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating linearization (LP-based) - novel
    deadline_p1 = start_time + (budget_s - 45.0) * 0.20
    h_lp, sc_lp = alternating_linearization(best_h, target, deadline_p1,
                                            max_iters=80, damping=1.0)
    if sc_lp < best_score:
        best_score = sc_lp
        best_h = h_lp

    # Phase 2: Multi-cut LP
    if tleft() > 100:
        deadline_p2 = start_time + (budget_s - 45.0) * 0.35
        h_mc, sc_mc = multi_cut_lp(best_h, target, deadline_p2, n_cuts=4)
        if sc_mc < best_score:
            best_score = sc_mc
            best_h = h_mc

    # Phase 3: Focused subgradient (single worst lag)
    if tleft() > 150:
        deadline_p3 = start_time + (budget_s - 45.0) * 0.50
        h_sg, sc_sg = focused_subgrad(best_h, target, deadline_p3, mode='single')
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg

    # Phase 4: SLSQP polish
    if tleft() > 100:
        h_ref = slsqp_round(best_h, n, target, [50, 200, 800, 3000], 1500, 1e-16)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Block coordinate descent
    if tleft() > 120:
        deadline_p5 = start_time + (budget_s - 45.0) * 0.70
        for bs in [3, 4, 5, 6]:
            if tleft() < 60:
                break
            h_bd = block_descent(best_h, n, target, bs, 6, 300.0, deadline_p5)
            sc = compute_c5(h_bd)
            if sc < best_score:
                best_score = sc
                best_h = h_bd

    # Phase 6: Periodic structured perturbations + SLSQP
    if tleft() > 150:
        deadline_p6 = start_time + (budget_s - 45.0) * 0.85
        for period in [4, 5, 8, 10, 16, 20, 25, 40, 50]:
            if time.time() > deadline_p6:
                break
            for amp in [0.05, 0.1, 0.2, 0.3]:
                if time.time() > deadline_p6:
                    break
                h_pert = periodic_perturb(best_h, n, target, period, amp, rng)
                h_ref = slsqp_round(h_pert, n, target, [100, 500], 800, 1e-14)
                sc = compute_c5(h_ref)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref

    # Phase 7: Binary push perturbations
    if tleft() > 100:
        deadline_p7 = start_time + (budget_s - 45.0) * 0.92
        for k_push in [10, 30, 50, 100]:
            if time.time() > deadline_p7:
                break
            for _ in range(3):
                if time.time() > deadline_p7:
                    break
                h_bp = selective_binary_push(best_h, n, target, k_push, rng)
                h_ref = slsqp_round(h_bp, n, target, [200, 1000], 600, 1e-14)
                sc = compute_c5(h_ref)
                if sc < best_score:
                    best_score = sc
                    best_h = h_ref

    # Phase 8: Final fine subgradient
    if tleft() > 60:
        h_sg2, sc_sg2 = focused_subgrad(best_h, target, start_time + budget_s - 50, mode='single')
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_h = sg_sg2 if False else h_sg2

    # Phase 9: Final SLSQP with very high alpha
    if tleft() > 30:
        h_final = slsqp_round(best_h, n, target, [2000, 8000, 20000], 1500, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
