            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Fix any remaining sum mismatch
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:final_sum - target_sum]] = 0
    elif final_sum < target_sum:
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r.astype(float)

def gradient_descent(h0, n_steps, alpha, lr_init, lr_min, start_time, time_limit):
    h = h0.copy()
    target = len(h) / 2.0
    best_obj = compute_c5(h)
    best_h = h.copy()
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        obj, grad = smooth_obj_grad(h, alpha)
        
        # Adaptive learning rate
        lr = max(lr_init * (0.95 ** step), lr_min)
        
        # Gradient step
        h_new = h - lr * grad
        # Project
        h_new = project_sum(h_new, target)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_obj:
            best_obj = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Halve learning rate
            lr_init = max(lr_init * 0.5, lr_min)
    
    return best_h, best_obj
