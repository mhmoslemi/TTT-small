```python
def coord_descent(h, n_iter=100, time_limit=100, start_time=None):
    n = len(h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h.copy()
    
    for iteration in range(n_iter):
        if time_limit and start_time and (time.time() - start_time) > time_limit:
            break
        
        improved = False
        for j in range(n):
            # Compute the coefficient vector for coordinate j
            # a[k] = change in corr[k] per unit change in h[j]
            a = np.zeros(2*n - 1)
            
            # For lag index k = n-1+s (s >= 0):
            # Δc = (v-h[j]) * [(1-h[j+s]) * [j <= n-1-s] - h[j-s] * [j >= s and j < n-1]]
            for s in range(0, n):
                k = n - 1 + s
                coef = 0.0
                if j <= n - 1 - s:
                    coef += (1.0 - h[j + s])
                if j >= s and j < n - 1:
                    coef -= h[j - s]
                a[k] = coef
            
            # For lag index k = n-1-s (s > 0):
            for s in range(1, n):
                k = n - 1 - s
                coef = 0.0
                if j >= s:
                    coef += (1.0 - h[j - s])
                if j <= n - 1 - s:
                    coef -= h[j + s]
                a[k] = coef
            
            # Now find optimal delta
            # f(delta) = max_k (corr[k] + delta * a[k])
            # We want to minimize this over delta in [-h[j], 1-h[j]]
            
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Binary search for optimal delta
            lo = -h[j]
            hi = 1.0 - h[j]
            
            # Check if current value (delta=0) is already optimal
            # Subgradient at 0: {a[k] : k is argmax of corr}
            cmax = np.max(corr)
            active = np.where(corr >= cmax - 1e-12)[0]
            min_a = np.min(a[active])
            max_a = np.max(a[active])
            
            if min_a <= 0 <= max_a:
                continue  # Already at optimum for this coordinate
            
            if min_a > 0:
                # Need to decrease delta (make h[j] smaller)
                optimal_delta = lo
            elif max_a < 0:
                # Need to increase delta (make h[j] larger)
                optimal_delta = hi
            else:
                # Need to bisect
                for _ in range(30):
                    mid = (lo + hi) / 2
                    new_corr = corr + mid * a
                    cmax_new = np.max(new_corr)
                    active_new = np.where(new_corr >= cmax_new - 1e-12)[0]
                    min_a_new = np.min(a[active_new])
                    max_a_new = np.max(a[active_new])
                    
                    if min_a_new <= 0 <= max_a_new:
                        lo = hi = mid
                        break
                    elif min_a_new > 0:
                        hi = mid
                    else:
                        lo = mid
                
                optimal_delta = (lo + hi) / 2
            
            h[j] += optimal_delta
            improved = True
        
        if not improved:
            break
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_max_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad
```

- At δ_mid, compute the argmax set S = {k : c_k + δ_mid * a_k = max}

```python
                grad[r:] -= wk * dx * h[:m]
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.01, 0.2)
        else:
            stall += 1
            lr = max(lr * 0.95, 1e-8)
        h_cur = h_new
        if stall > stall_limit:
            h_cur = best_h.copy()
            lr = lr_init * 0.6
            stall = 0
            stall_limit = max(150, stall_limit // 2)

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: Coordinate descent (fast local optimization)
    if rem() > 100:
        tl = min(rem() - 60, 150)
        h_cd = coord_descent(best_h, n_passes=300, time_limit=tl, start_time=start_time)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: SLSQP polish
    if rem() > 120:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: More coordinate descent
    if rem() > 80:
        tl = min(rem() - 50, 120)
        h_cd = coord_descent(best_h, n_passes=400, time_limit=tl, start_time=start_time)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Projected gradient for escape
    if rem() > 80:
        tl = min(rem() - 50, 100)
        h_pg, c5_pg = projected_gradient_topk(best_h, 2, 0.01, 0.06, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 5: Random restarts with coordinate descent
    n_restarts = 0
    while rem() > 200 and n_restarts < 8:
        idx = rng.choice(n, size=max(5, n // 15), replace=False)
        h_pert = best_h.copy()
        h_pert[idx] = np.clip(h_pert[idx] + 0.08 * rng.standard_normal(len(idx)), 0, 1)
        h_pert = project_onto_constraint(h_pert, target_sum)

        tl_cd = min(rem() - 100, 40)
        h_cd = coord_descent(h_pert, n_passes=150, time_limit=tl_cd, start_time=start_time)
        c5_cd = compute_c5(h_cd)

        if c5_cd < best_c5 and rem() > 80:
            h_opt, c5_opt = slsqp_optimize(h_cd, 0.001, 200)
            if c5_opt < c5_cd:
                c5_cd = c5_opt
                h_cd = h_opt.copy()

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        n_restarts += 1

    # Phase 6: Resolution increase to 2n
    if rem() > 350:
        h_up = upsample(best_h, factor=2)
        n_new = len(h_up)
        h_up = project_onto_constraint(h_up, n_new / 2.0)

        # Coordinate descent at new resolution
        tl = min(rem() - 150, 200)
        h_cd_up = coord_descent(h_up, n_passes=150, time_limit=tl, start_time=start_time)
        c5_up = compute_c5(h_cd_up)

        if rem() > 120:
            h_opt_up, c5_opt_up = slsqp_optimize(h_cd_up, 0.005, 300)
            if c5_opt_up < c5_up:
                c5_up = c5_opt_up
                h_cd_up = h_opt_up.copy()

        if rem() > 60:
            tl2 = min(rem() - 40, 80)
            h_cd2 = coord_descent(h_cd_up, n_passes=100, time_limit=tl2, start_time=start_time)
            c5_cd2 = compute_c5(h_cd2)
            if c5_cd2 < c5_up:
                c5_up = c5_cd2
                h_cd_up = h_cd2.copy()

        if rem() > 50:
            h_opt2, c5_opt2 = slsqp_optimize(h_cd_up, 0.001, 200)
            if c5_opt2 < c5_up:
                c5_up = c5_opt2
                h_cd_up = h_opt2.copy()

        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_cd_up.copy()

    # Phase 7: Resolution increase to 4n original
    if rem() > 500:
        h_up = upsample(best_h, factor=2)
        n_new = len(h_up)
        h_up = project_onto_constraint(h_up, n_new / 2.0)

        tl = min(rem() - 200, 200)
        h_cd_up = coord_descent(h_up, n_passes=100, time_limit=tl, start_time=start_time)
        c5_up = compute_c5(h_cd_up)

        if rem() > 150:
            h_opt_up, c5_opt_up = slsqp_optimize(h_cd_up, 0.005, 250)
            if c5_opt_up < c5_up:
                c5_up = c5_opt_up
                h_cd_up = h_opt_up.copy()

        if rem() > 80:
            tl2 = min(rem() - 50, 100)
            h_cd2 = coord_descent(h_cd_up, n_passes=80, time_limit=tl2, start_time=start_time)
            c5_cd2 = compute_c5(h_cd2)
            if c5_cd2 < c5_up:
                c5_up = c5_cd2
                h_cd_up = h_cd2.copy()

        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_cd_up.copy()

    # Phase 8: Final coordinate descent
    if rem() > 40:
        tl = min(rem() - 20, 100)
        h_cd = coord_descent(best_h, n_passes=500, time_limit=tl, start_time=start_time)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 9: Final SLSQP
    if rem() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 10: Last coordinate descent
    if rem() > 20:
        tl = min(rem() - 10, 50)
        h_cd = coord_descent(best_h, n_passes=200, time_limit=tl, start_time=start_time)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
```
