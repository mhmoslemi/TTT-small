def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_profile(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def power_obj(h, p):
    """Power-mean objective: (sum corr^p)^(1/p) approximates max for large p."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    c = corr * dx
    # Only consider valid lags (where the overlap is non-trivial)
    # Actually for the problem statement, we use all lags from 'full' mode
    # But the effective lags are from -(n-1) to (n-1)
    # For the constraint integral, only lags where h(x) and h(x+k) both exist matter
    # In the 'full' correlation, the values at extreme lags are small (few terms)
    # Let's just use all of them as the problem states
    return float(np.sum(np.maximum(c, 0)**p) ** (1.0/p))

def slsqp_refine_power(h_start, n, target, ps, maxiter, time_limit=None, start_time=None):
    """SLSQP with power-mean objective."""
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun_eq, 'jac': cons_jac_eq}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for p in ps:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            def obj(x, p=p):
                h = np.clip(x, 0, 1)
                return power_obj(h, p)
            
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def softmax_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)

    c = corr * dx

    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return best_h, best_c5


def best_improve_hill_climb(h, target_sum, max_iter, rng, num_candidates=8):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 200

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0
        best_a = -1
        best_b = -1
        for _c in range(num_candidates):
            a = int(ones[rng.randint(len(ones))])
            b = int(zeros[rng.randint(len(zeros))])
            if a == b:
                continue
            h[a] = 0
            h[b] = 1
            c5_test = compute_c5(h)
            delta = c5_test - best_c5
            h[a] = 1
            h[b] = 0
            if delta < best_delta:
                best_delta = delta
                best_a = a
                best_b = b
        if best_a >= 0:
            h[best_a] = 0
            h[best_b] = 1
            best_c5 += best_delta
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def top3_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 150

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        corr = compute_corr_raw(h)
        top3 = np.argsort(corr)[-3:]
        best_delta = 0
        best_a = -1
        best_b = -1
        num_tries = 12
        for _c in range(num_tries):
            peak = int(top3[rng.randint(3)])
            mid = (2 * n - 1) // 2
            k = peak - mid
            if k != 0:
                good_a = []
                for pos in ones:
                    pos = int(pos)
                    if k > 0 and pos + k < n and h[pos + k] == 0:
                        good_a.append(pos)
                    elif k < 0 and pos + k >= 0 and h[pos + k] == 0:
                        good_a.append(pos)
                if len(good_a) > 0:
                    a = good_a[rng.randint(len(good_a))]
                else:
                    a = int(ones[rng.randint(len(ones))])
            else:
                a = int(ones[rng.randint(len(ones))])
            b = int(zeros[rng.randint(len(zeros))])
            if a == b:
                continue
            h[a] = 0
            h[b] = 1
            c5_test = compute_c5(h)
            delta = c5_test - best_c5
            h[a] = 1
            h[b] = 0
            if delta < best_delta:
                best_delta = delta
                best_a = a
                best_b = b
        if best_a >= 0:
            h[best_a] = 0
            h[best_b] = 1
            best_c5 += best_delta
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def softmax_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def power_obj(h, p):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    dx = 2.0 / n
    c = corr * dx
    c_pos = np.maximum(c, 0.0)
    return float(np.sum(c_pos ** p) ** (1.0 / p))


def cons_fun_eq(h):
    return np.sum(h) - _cons_target


def cons_jac_eq(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, ps, maxiter, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun_eq, 'jac': cons_jac_eq}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(lambda x: softmax_obj(x, alpha), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    for p in ps:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(lambda x: power_obj(x, p), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)
    h_sa_curr = h_bin_start.copy()

    # Phase 1: Guided SA from initial (multiple configs)
    sa_configs = [
        (100000, 0.018, 0.00008, 0.55, 0.12),
        (120000, 0.012, 0.00005, 0.65, 0.15),
        (80000, 0.025, 0.0005, 0.45, 0.10),
        (150000, 0.008, 0.00003, 0.70, 0.18),
        (90000, 0.020, 0.0001, 0.50, 0.20),
    ]

    for ci, (max_it, t_i, t_f, gw, msp) in enumerate(sa_configs):
        if tleft() < 90:
            break
        allowed_time = min(200.0, tleft() * 0.30)
        h_sa, c5_sa = guided_sa(h_sa_curr, target_init, max_it, t_i, t_f,
                                rng, time_limit=allowed_time, start_time=start_time,
                                guide_weight=gw, multi_swap_prob=msp)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        h_sa_curr = h_sa

        if tleft() > 60:
            h_hc, c5_hc = best_improve_hill_climb(h_sa, target_init, 8000, rng, num_candidates=6)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()
                h_sa_curr = h_hc
            if tleft() > 40:
                h_hc2, c5_hc2 = top3_hill_climb(h_sa, target_init, 6000, rng)
                if c5_hc2 < best_score:
                    best_score = c5_hc2
                    best_h = h_hc2.copy()
                    h_sa_curr = h_hc2

    # Phase 2: Random starts
    n_random_starts = 6
    for rs in range(n_random_starts):
        if tleft() < 80:
            break
        h_rand = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_rand[perm[:n_init // 2]] = 1.0
        allowed_time = min(130.0, tleft() * 0.22)
        max_it = int(min(120000, allowed_time * 3500))
        h_sa, c5_sa = guided_sa(h_rand, target_init, max_it, 0.015, 0.00004,
                                rng, time_limit=allowed_time, start_time=start_time,
                                guide_weight=0.55, multi_swap_prob=0.15)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        if tleft() > 50:
            h_hc, c5_hc = best_improve_hill_climb(h_sa, target_init, 6000, rng, num_candidates=6)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()

    # Phase 3: Multi-resolution SA
    n_candidates = [350, 400, 450, 500, 550, 650, 700, 750, 850, 1000]
    n_candidates = [n for n in n_candidates if n != n_init and n >= 100]

    for n_try in n_candidates:
        if tleft() < 70:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_time = min(100.0, tleft() * 0.22)
        max_it = int(min(60000, allowed_time * 2500))
        h_sa, c5_sa = guided_sa(h_bin_try, target_try, max_it, 0.012, 0.00006,
                                rng, time_limit=allowed_time, start_time=start_time,
                                guide_weight=0.55, multi_swap_prob=0.15)
        if tleft() > 35:
            h_hc, c5_hc = best_improve_hill_climb(h_sa, target_try, 4000, rng, num_candidates=5)
            if c5_hc < c5_sa:
                h_sa = h_hc
                c5_sa = c5_hc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Continuous SLSQP refinement
    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        alphas = [30, 100, 300, 800, 2000, 5000]
        ps = [10, 30, 80, 200]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, ps, 600,
                                 time_limit=tleft() * 0.45, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Refine at different resolutions
    n_refine_candidates = [int(n_use * 0.8), int(n_use * 1.1), int(n_use * 1.25), int(n_use * 1.5)]
    n_refine_candidates = [n for n in n_refine_candidates if n >= 100 and n != n_use]

    for n_try in n_refine_candidates:
        if tleft() < 50:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [50, 200, 600, 1500, 4000]
        ps_try = [20, 60, 150]
        h_ref, sc = slsqp_refine(h_interp, n_try, target_try, alphas_try, ps_try, 400,
                                 time_limit=tleft() * 0.4, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Final polish cycles (SA + hill climb + SLSQP)
    for cycle in range(4):
        if tleft() < 30:
            break
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(5000, tleft() * 1500))
        h_sa_f, c5_f = guided_sa(h_bin_final, target_use, max_it, 0.006, 0.00002,
                                  rng, time_limit=max(8.0, tleft() * 0.35), start_time=start_time,
                                  guide_weight=0.7, multi_swap_prob=0.15)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 20:
            h_hc, c5_hc = best_improve_hill_climb(best_h, target_use, 5000, rng, num_candidates=4)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc
        if tleft() > 15:
            h_ref, sc = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                     [200, 800, 3000, 8000], [50, 150, 400], 300,
                                     time_limit=max(5.0, tleft() * 0.3), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Final feasibility check
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
