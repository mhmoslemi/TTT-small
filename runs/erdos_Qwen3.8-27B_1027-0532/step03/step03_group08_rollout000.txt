import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x: 0<=x_i<=1, sum(x)=S}"""
    n = len(y)
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(y - theta, 0.0, 1.0)
    # Fix residual
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:

                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

def _grad_overlap(h, k_idx, dx):
    """Gradient of the overlap at lag k_idx (in full correlation indexing)."""
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            # overlap = sum_{i=0}^{m-1} h[i]*(1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j in [0,m)
            #           -h[j-lag] for j in [lag, n)
            grad[:m] += (1.0 - h[lag:lag+m])
            grad[lag:] += -h[:m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            # overlap = sum_{i=0}^{m-1} h[i+j]*(1-h[i])
            # d/dh[j+i] = (1-h[i]) for i in [0,m)
            # d/dh[i] = -h[i+j] for i in [0,m)
            grad[j:] += (1.0 - h[:m])
            grad[:m] += -h[j:]
    return grad * dx

def _active_set_grad(h, m_active, softmax_temp, dx):
    """Weighted gradient using top m_active lags with softmax weighting."""
    corr = _corr_of(h)
    max_c = np.max(corr)
    # Softmax weights
    w = np.exp((corr - max_c) / softmax_temp)
    w = w / w.sum()
    # Focus on top m
    top_m = np.argsort(w)[::-1][:m_active]
    w_top = w[top_m].copy()
    w_top = w_top / w_top.sum()
    
    grad = np.zeros(len(h))
    for i in range(len(top_m)):
        grad += w_top[i] * _grad_overlap(h, top_m[i], dx)
    return grad, w_top, top_m

def _nesterov_active(h0, n, target_sum, n_iters, alpha0, m_active, softmax_temp, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        grad, _, _ = _active_set_grad(y, m_active, softmax_temp, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(xflat):
        h = np.clip(xflat, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        lse = max_c + np.log(np.sum(np.exp(alpha * (corr - max_c)))) / alpha
        return lse * dx
    
    def grad_obj(xflat):
        h = np.clip(xflat, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        w = np.exp(alpha * (corr - max_c))
        w = w / w.sum()
        grad = np.zeros(n)
        for i in range(len(corr)):
            if w[i] > 1e-15:
                grad += w[i] * _grad_overlap(h, i, dx)
        # Only where h is in (0,1)
        mask = (xflat > 0) & (xflat < 1)
        grad = grad * mask
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=grad_obj,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent_fine(h0, n, target_sum, values, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in values:
                if v == old:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.5
    hi = y.max() + 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    x = np.clip(y - theta, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_overlap(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:lag + m])
            grad[lag:] += -h[:m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[j:] += (1.0 - h[:m])
            grad[:m] += -h[j:]
    return grad * dx

def _lse_grad(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp(alpha * (corr - max_c))
    w = w / w.sum()
    grad = np.zeros(n)
    for i in range(len(corr)):
        if w[i] > 1e-15:
            grad += w[i] * _grad_overlap(h, i, dx)
    return grad

def _active_grad(h, m_active, softmax_temp, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp((corr - max_c) / softmax_temp)
    w = w / w.sum()
    top_m = np.argsort(w)[::-1][:m_active]
    w_top = w[top_m].copy()
    w_top = w_top / w_top.sum()
    grad = np.zeros(n)
    for i in range(len(top_m)):
        grad += w_top[i] * _grad_overlap(h, top_m[i], dx)
    return grad

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha0, alpha_lse, m_active, softmax_temp, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        if alpha_lse > 0:
            grad = _lse_grad(y, alpha_lse, dx)
        else:
            grad = _active_grad(y, m_active, softmax_temp, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_run(h0, n, target_sum, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(xflat):
        h = np.clip(xflat, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        lse = max_c + np.log(np.sum(np.exp(alpha * (corr - max_c)))) / alpha
        return lse * dx

    def grad_obj(xflat):
        h = np.clip(xflat, 0.0, 1.0)
        g = _lse_grad(h, alpha, dx)
        mask = (xflat > 1e-10) & (xflat < 1.0 - 1e-10)
        return g * mask

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h0, jac=grad_obj,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if c0 < co - 1e-14:
                h[j] = 0.0
                improved = True
            elif c1 < co - 1e-14:
                h[j] = 1.0
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def _coord_fine(h0, n, target_sum, values, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in values:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _de_search(h0, n, target_sum, pop_size, max_gen, F, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.25:
            pop[i] = _fix_sum(h0 + 0.15 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.5:
            t = rng.uniform(0.4, 0.6)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif r < 0.75:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        else:
            pop[i] = _fix_sum(1.0 - h0 + 0.08 * (rng.rand(n) - 0.5), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.randint(n)] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0, 1), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
    return best_h, best_val
