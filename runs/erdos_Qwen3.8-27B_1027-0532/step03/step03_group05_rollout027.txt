def efficient_coordinate_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    # Precompute correlation
    corr = np.correlate(best, 1.0 - best, mode='full').copy()
    
    deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.002, -0.002]
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            for d in deltas:
                val = best[j] + d
                if val < 0 or val > 1:
                    continue
                # Compute change in correlation
                # p = j + m - (n-1), q = j + (n-1) - m
                # dr[m] = d * (1 - h[p]) * [0<=p<n] - d * h[q] * [0<=q<n]
                m = np.arange(2*n - 1)
                p = j + m - (n - 1)
                q = j + (n - 1) - m
                mask_p = (p >= 0) & (p < n)
                mask_q = (q >= 0) & (q < n)
                dr = np.zeros(2*n - 1)
                dr[mask_p] = d * (1 - best[p[mask_p]])
                dr[mask_q] -= d * best[q[mask_q]]
                new_corr = corr + dr
                new_score = float(np.max(new_corr)) * (2.0/n)
                if new_score < best_score - 1e-14:
                    best[j] = val
                    corr = new_corr
                    best_score = new_score
                    improved = True
                    break
        if not improved:
            # Shrink deltas
            deltas = [x * 0.5 for x in deltas]
            if max(abs(x) for x in deltas) < 1e-5:
                break
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def efficient_pair_cd(h, target, time_limit, seed=42, use_sa=False, T_init=0.003):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    current = h.copy()
    curr_corr = np.correlate(current, 1.0 - current, mode='full').astype(float)
    best = current.copy()
    best_score = compute_c5(best)
    curr_score = best_score

    m_all = np.arange(2 * n - 1)
    num_lags = 2 * n - 1
    dr = np.zeros(num_lags)

    p_idx = [None] * n
    p_val = [None] * n
    q_idx = [None] * n
    q_val = [None] * n
    for j in range(n):
        p = j + m_all - (n - 1)
        q = j + (n - 1) - m_all
        mp = (p >= 0) & (p < n)
        mq = (q >= 0) & (q < n)
        p_idx[j] = np.where(mp)[0]
        p_val[j] = p[mp]
        q_idx[j] = np.where(mq)[0]
        q_val[j] = q[mq]

    T = T_init
    deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.003, -0.003]
    stall = 0
    sweep_count = 0

    while time.time() - start < time_limit:
        found_improvement = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                vi = current[j] + d
                if vi < 0.0 or vi > 1.0:
                    continue
                for _t in range(5):
                    k = rng.randint(n)
                    if k == j:
                        continue
                    vk = current[k] - d
                    if vk < 0.0 or vk > 1.0:
                        continue
                    dr[p_idx[j]] = d * (1.0 - current[p_val[j]])
                    dr[q_idx[j]] -= d * current[q_val[j]]
                    dr[p_idx[k]] -= d * (1.0 - current[p_val[k]])
                    dr[q_idx[k]] += d * current[q_val[k]]
                    new_score = float(np.max(curr_corr + dr)) * (2.0 / n)
                    accept = new_score < curr_score
                    if not accept and use_sa and T > 1e-7:
                        if rng.rand() < np.exp(-(new_score - curr_score) / T):
                            accept = True
                    if accept:
                        current[j] = vi
                        current[k] = vk
                        curr_corr += dr
                        dr[:] = 0.0
                        curr_score = new_score
                        if new_score < best_score - 1e-15:
                            best_score = new_score
                            best = current.copy()
                            found_improvement = True
                        break
                    else:
                        dr[:] = 0.0
        sweep_count += 1
        if sweep_count % 80 == 0:
            curr_corr = np.correlate(current, 1.0 - current, mode='full').astype(float)
            curr_score = compute_c5(current)
        if use_sa:
            T *= 0.997
        if not found_improvement:
            stall += 1
            if stall >= 4:
                deltas = [x * 0.7 for x in deltas]
                if max(abs(x) for x in deltas) < 1e-5:
                    break
                stall = 0

    return best, best_score


def slsqp_polish(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    alphas = [5, 10, 20, 50, 100, 200, 500]
    na = len(alphas)

    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (na - ai)
        maxiter = max(50, int(time_per * 3))

        def obj(x, a=alpha):
            c = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(c)
            shifted = np.clip(a * (c - cmax), -500.0, 0.0)
            return float(cmax + np.log(np.sum(np.exp(shifted))) / a)

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def smooth_and_refit(h, target, sigma):
    n = len(h)
    kernel = np.exp(-0.5 * (np.arange(-sigma * 4, sigma * 4 + 1) / sigma) ** 2)
    kernel /= kernel.sum()
    h_smooth = np.convolve(h, kernel, mode='same')
    h_smooth = project_feasible(h_smooth, target)
    return h_smooth
