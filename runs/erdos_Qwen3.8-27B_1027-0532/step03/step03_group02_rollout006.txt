def compute_weighted_gradient(h, p, n):
    s_arr = np.arange(2*n-1) - (n-1)  # shifts: -(n-1) to n-1
    g = np.zeros(n)
    for k in range(2*n-1):
        s = s_arr[k]
        w = p[k]
        if w < 1e-15:
            continue
        # First term: h[j] as first factor, valid when 0<=j+s<n
        valid1 = np.arange(n) + s
        mask1 = (valid1 >= 0) & (valid1 < n)
        g[mask1] += w * (1.0 - h[valid1[mask1]])
        # Second term: h[j] as second factor, valid when 0<=j-s<n
        valid2 = np.arange(n) - s
        mask2 = (valid2 >= 0) & (valid2 < n)
        g[mask2] -= w * h[valid2[mask2]]
    return g
```

def compute_weighted_gradient(h, p):
    n = len(h)
    s_arr = np.arange(2*n-1) - (n-1)
    h = h.astype(float)
    
    g = np.zeros(n)
    for k in range(2*n-1):
        s = s_arr[k]
        w = p[k]
        if w < 1e-15:
            continue
        # First term: for each j, contribution is w*(1-h[j+s]) if 0<=j+s<n
        if s >= 0:
            # j ranges from 0 to n-1-s
            g[:n-s] += w * (1.0 - h[s:n])
            # Second term: j ranges from s to n-1
            g[s:] -= w * h[:n-s]
        else:
            s_abs = -s
            # j ranges from s_abs to n-1
            g[s_abs:] += w * (1.0 - h[:n-s_abs])
            # Second term: j ranges from 0 to n-1-s_abs
            g[:n-s_abs] -= w * h[s_abs:]
    return g
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(50):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def compute_weighted_gradient(h, p):
    n = len(h)
    s_arr = np.arange(2*n-1) - (n-1)
    g = np.zeros(n)
    for k in range(2*n-1):
        s = int(s_arr[k])
        w = p[k]
        if w < 1e-15:
            continue
        if s >= 0:
            if n - s > 0:
                g[:n-s] += w * (1.0 - h[s:n])
                g[s:] -= w * h[:n-s]
        else:
            sa = -s
            if n - sa > 0:
                g[sa:] += w * (1.0 - h[:n-sa])
                g[:n-sa] -= w * h[sa:]
    return g

def alternating_maxmin(h_init, target_sum, max_time, rng):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target_sum)
    
    best_score = compute_c5(h)
    best_h = h.copy()
    
    # Initial p
    p = np.ones(2*n-1) / (2*n-1)
    
    beta = 5.0  # temperature for p update
    lr = 0.05
    momentum = 0.9
    v = np.zeros(n)
    
    start = time.time()
    no_improve = 0
    iter_count = 0
    
    while time.time() - start < max_time:
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        
        # Update best
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Update p (focus on worst shifts)
        # Use softmax with temperature
        corr_shifted = corr - c_max
        p_new = np.exp(beta * corr_shifted / max(c_max, 1e-10))
        p_sum = np.sum(p_new)
        if p_sum < 1e-30:
            p_new = np.ones(2*n-1) / (2*n-1)
        else:
            p_new /= p_sum
        p = p_new
        
        # Compute gradient
        g = compute_weighted_gradient(h, p)
        
        # Momentum update
        v = momentum * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, target_sum)
        
        # Check improvement
        score_new = compute_c5(h_new)
        if score_new < score:
            h = h_new
            lr = min(lr * 1.01, 0.5)
        else:
            h = h_new  # still move, but reduce lr
            lr *= 0.95
        
        # Increase beta periodically (focus more on worst shifts)
        if iter_count % 100 == 0:
            beta = min(beta * 1.05, 50.0)
        
        # Add noise if stuck
        if no_improve > 200:
            noise = 0.01 * rng.randn(n)
            h = project_feasible(best_h + noise, target_sum)
            v = np.zeros(n)
            no_improve = 0
            beta = 5.0
            lr = 0.05
        
        iter_count += 1
    
    return best_h, best_score
