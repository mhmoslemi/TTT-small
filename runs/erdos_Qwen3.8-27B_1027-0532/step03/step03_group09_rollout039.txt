import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(dx * np.max(np.correlate(h, 1.0 - h, mode='full')))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def lbfgs_penalty(h_init, eps, lam, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def obj_pen(x):
        return smooth_max_obj(x, eps) + lam * (np.sum(x) - target_sum) ** 2
    
    def grad_pen(x):
        return smooth_max_grad(x, eps) + 2.0 * lam * (np.sum(x) - target_sum) * np.ones(n)
    
    try:
        result = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def differential_evolution(h_init, n_pop, F, CR, start_time, time_limit, rng):

    best_idx = int(np.argmin(fitness))
    if fitness[best_idx] < best_c5:

        return project_onto_constraint(h_base + 0.04 * rng.standard_normal(n), target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(dx * np.max(np.correlate(h, 1.0 - h, mode='full')))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def _cfun(x, target_sum):
    return float(np.sum(x) - target_sum)

def _cjac(x, target_sum):
    return np.ones(len(x))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac, 'args': (target_sum,)}],
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def lbfgs_penalty_optimize(h_init, eps, lam, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def _obj_pen(x):
        return smooth_max_obj(x, eps) + lam * (np.sum(x) - target_sum) ** 2

    def _grad_pen(x):
        g = smooth_max_grad(x, eps)
        g = g + 2.0 * lam * (np.sum(x) - target_sum)
        return g

    try:
        result = minimize(_obj_pen, h0, jac=_grad_pen, method='L-BFGS-B',

            best_h = h_o.copy()

    if rem() > 40:
        h_o, c5_o = slsqp_optimize(best_h, 0.0003, 300, ftol=1e-16)
        if c5_o < best_c5:
            best_c5 = c5_o
            best_h = h_o.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
