```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def differential_evolution(h_init, n_pop, n_gens, time_limit, start_time, rng, F=0.8, CR=0.7):
    n = len(h_init)
    target_sum = n / 2.0
    
    pop = np.zeros((n_pop, n))
    pop[0] = project_onto_constraint(h_init, target_sum)
    for i in range(1, n_pop):
        scale = 0.05 + 0.15 * rng.random()
        pop[i] = project_onto_constraint(h_init + scale * rng.standard_normal(n), target_sum)
    
    pop_c5 = np.array([compute_c5(pop[i]) for i in range(n_pop)])
    best_idx = np.argmin(pop_c5)
    best_h = pop[best_idx].copy()
    best_c5 = pop_c5[best_idx]
    
    for gen in range(n_gens):
        if time.time() - start_time > time_limit:
            break
        
        for i in range(n_pop):
            if time.time() - start_time > time_limit:
                break
            
            # Selection of three distinct indices
            a, b, c = rng.choice([j for j in range(n_pop) if j != i], size=3, replace=False)
            
            # Mutation
            v = pop[a] + F * (pop[b] - pop[c])
            
            # Binary crossover
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            if not mask.any():
                mask[rng.integers(n)] = True
            trial[mask] = v[mask]
            
            # Project
            trial = project_onto_constraint(trial, target_sum)
            c5_trial = compute_c5(trial)
            
            # Selection
            if c5_trial < pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = c5_trial
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = trial.copy()
    
    return best_h, best_c5

def swap_local_search(h, target_sum, n_passes, time_limit, start_time, rng):
    """Greedy + SA swap-based local search on near-binary solutions."""
    n = len(h)
    h_cur = project_onto_constraint(h.copy(), target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    # Create a "binary-like" version: decide each position as 0 or 1
    # based on threshold, then adjust to meet the sum constraint
    h_bin = (h_cur > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = h_bin.sum()
    if n_ones > n / 2:
        # Need to flip some 1s to 0s: pick the smallest ones (closest to 0.5)
        one_indices = np.where(h_bin == 1)[0]
        # Sort by h_cur value (ascending) - flip the smallest first
        order = np.argsort(h_cur[one_indices])
        n_to_flip = int(n_ones - n / 2)
        for idx in order[:n_to_flip]:
            h_bin[one_indices[idx]] = 0.0
    elif n_ones < n / 2:
        zero_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h_cur[zero_indices])
        n_to_flip = int(n / 2 - n_ones)
        for idx in order[:n_to_flip]:
            h_bin[zero_indices[idx]] = 1.0
    
    h_bin = project_onto_constraint(h_bin, target_sum)
    c5_bin = compute_c5(h_bin)
    
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Now do swap-based local search
    h_cur = best_h.copy()
    
    for path_idx in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            if h_cur[i] > 0.5:
                # Try swapping with a 0
                for j in range(n):
                    if j == i or h_cur[j] >= 0.5:
                        continue
                    old_i, old_j = h_cur[i], h_cur[j]
                    h_cur[i], h_cur[j] = 0.0, 1.0
                    c5_try = compute_c5(h_cur)
                    if c5_try < best_c5:
                        best_c5 = c5_try
                        best_h = h_cur.copy()
                        improved = True
                    else:
                        h_cur[i], h_cur[j] = old_i, old_j
                    break  # Only try one j per i for speed
    
    # SA refinement
    temp = 0.005
    h_cur = best_h.copy()
    for sa_iter in range(50000):
        if time.time() - start_time > time_limit:
            break
        i = rng.integers(n)
        if h_cur[i] > 0.5:
            # Find a 0 to swap with
            zeros = np.where(h_cur < 0.5)[0]
            if len(zeros) == 0:
                continue
            j = zeros[rng.integers(len(zeros))]
        else:
            ones = np.where(h_cur > 0.5)[0]
            if len(ones) == 0:
                continue
            j = ones[rng.integers(len(ones))]
        
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
        elif rng.random() < np.exp(-(c5_new - best_c5) / temp):
            pass  # Accept
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]  # Undo
        
        temp *= 0.99999
    
    return best_h, best_c5

def projected_gradient_topk(h, k, temp, lr_init, n_steps, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        grad = topk_subgrad(h_cur, k, temp)
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stall += 1
            lr = max(lr * 0.95, 1e-7)
        
        h_cur = h_new
        
        if stall > 300:
            h_cur = best_h.copy()
            lr = lr_init * 0.7
            stall = 0
    
    return best_h, best_c5

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def gen_structured_init(n, rng, strategy):
    """Generate diverse structured initializations."""
    target_sum = n / 2.0
    if strategy == 0:
        # Random binary
        h = (rng.random(n) < 0.5).astype(float)
    elif strategy == 1:
        # Block pattern: alternating blocks
        block = rng.integers(2, 8)
        h = np.tile([1.0]*block + [0.0]*block, n//(2*block)+1)[:n]
    elif strategy == 2:
        # Sinusoidal
        x = np.linspace(0, 2, n)
        freq = rng.uniform(1, 5)
        h = 0.5 + 0.5 * np.sin(2*np.pi*freq*x + rng.uniform(0, 2*np.pi))
    elif strategy == 3:
        # Smooth noise
        h = 0.5 + 0.3 * rng.standard_normal(n)
        # Smooth with a simple box filter
        kernel = np.ones(3) / 3
        h = np.convolve(h, kernel, mode='same')
    elif strategy == 4:
        # Thue-Morse-like
        h = np.zeros(n)
        h[0] = 1.0
        for i in range(1, n):
            # Count trailing ones in binary representation of (i-1)
            val = i - 1
            cnt = 0
            while val & 1:
                cnt += 1
                val >>= 1
            h[i] = 1.0 if cnt % 2 == 0 else 0.0
    else:
        h = 0.5 + 0.4 * rng.standard_normal(n)
    
    return project_onto_constraint(h, target_sum)
```

Let me also add one more idea: **coordinate descent**. For each coordinate i, find the optimal value of h[i] (given all other coordinates fixed) that minimizes C₅. This is a 1D optimization problem that can be solved by golden section search or similar.

The function to minimize in coordinate i is:
f(h_i) = max_k corr_k(h_i)

where corr_k depends linearly on h_i (for fixed other coordinates). So f is the max of linear functions, which is convex. We can optimize it efficiently.

Actually, let me think about this more carefully. For a fixed set of coordinates except h[i], the correlation at lag k is:
corr_k = sum_j h[j] * (1-h[j-(k-(n-1))])

The terms involving h[i] are:
- h[i] appears as h[j] when j = i: contribution h[i] * (1-h[i-(k-(n-1))])
- h[i] appears as h[j-(k-(n-1))] when j-(k-(n-1)) = i, i.e., j = i + k - (n-1):
  contribution h[j] * (1-h[i])

So corr_k = (terms not involving h[i]) + h[i] * (1-h[i-s_k]) + h[i+s_k] * (1-h[i])
where s_k = k - (n-1) and we need the indices to be valid.

This simplifies to:
corr_k = c_k + h[i] * (1-h[i-s_k]) + h[i+s_k] * (1-h[i])
= c_k + h[i] - h[i]*h[i-s_k] + h[i+s_k] - h[i+s_k]*h[i]
= c_k + h[i] + h[i+s_k] - h[i](h[i-s_k] + h[i+s_k])

So corr_k is quadratic in h[i] (specifically, it's linear in h[i] + h[i+s_k] - h[i]*(h[i-s_k]+h[i+s_k])).

Wait, h[i+s_k] and h[i-s_k] are fixed (they're other coordinates), so corr_k is actually linear in h[i]:
corr_k = c_k + h[i] * (1 - h[i-s_k]) + h[i+s_k] * (1 - h[i])
= c_k + h[i] - h[i]*h[i-s_k] + h[i+s_k] - h[i]*h[i+s_k]
= c_k + h[i+s_k] + h[i] * (1 - h[i-s_k] - h[i+s_k])

So corr_k = A_k + B_k * h[i] where:
A_k = c_k + h[i+s_k] (constant part, depends on which indices are valid)
B_k = 1 - h[i-s_k] - h[i+s_k] (coefficient of h[i])

This is a piecewise linear function (max of linear functions), which is convex. The minimum over h[i] ∈ [0,1] can be found by checking the critical points (where two lines cross) and the boundaries.

This gives an efficient coordinate descent method! Each coordinate update is O(n) (to compute all the A_k and B_k), and a full pass is O(n²). For n=180, that's 32400 operations per pass, which is very fast.

Let me implement this:

```python
def coordinate_descent(h, n_passes, time_limit, start_time, target_sum):
    n = len(h)
    h_cur = project_onto_constraint(h.astype(float).copy(), target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for p in range(n_passes):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start_time > time_limit:
                break
            
            # Compute A_k and B_k for all k
            # For each lag k (0 to 2n-2), s = k - (n-1)
            # corr_k = sum_j h[j] * (1-h[j-s]) for valid j, j-s
            # Terms involving h[i]:
            #   j = i: h[i] * (1-h[i-s])
            #   j-s = i => j = i+s: h[i+s] * (1-h[i])
            
            # For efficiency, compute the "base" correlation (without h[i])
            # and then add the h[i] contributions.
            
            # Actually, let me just compute the correlation at all lags
            # as a function of h_i = t, for t ∈ [0,1].
            
            # corr_k(t) = base_k + t * coeff_k
            # where base_k and coeff_k don't depend on t.
            
            # Let me compute directly:
            h_save = h_cur[i]
            
            # For each possible t, compute corr_k(t) for all k
            # corr_k(t) = sum_{j: j valid, j-s valid, j≠i, j-s≠i} h[j](1-h[j-s])
            #            + [i valid and i-s valid] * t * (1-h[i-s])
            #            + [i+s valid and i valid] * h[i+s] * (1-t)
            
            # Let me compute the sum excluding h[i] contributions:
            # I'll set h_cur[i] = 0, compute corr, then add back.
            
            h_cur[i] = 0.0
            corr_base = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            h_cur[i] = h_save
            
            # Now add back the h[i] contributions
            # h[i] contributes to corr at lag k when:
            #   j = i and j-s = i-s is valid: adds h[i]*(1-h[i-s]) = t*(1-h[i-s])
            #   j-s = i and j = i+s is valid: adds h[i+s]*(1-h[i]) = h[i+s]*(1-t)
            
            # But wait, when I set h_cur[i] = 0, the (1-h_cur) array also changes at position i.
            # So the base correlation already accounts for h[i]=0 in both h and (1-h).
            # I need to add back:
            #   For the h part: h[i] = t instead of 0, so add t * (1-h[j-s]) for j=i
            #   For the (1-h) part: 1-h[i] = 1-t instead of 1, so the terms where (1-h[i]) appears
            #     are: h[j] * (1-h[i]) for j-s = i, i.e., j = i+s
            #     These should be h[i+s] * (1-t) instead of h[i+s] * 1
            #     So the correction is: -h[i+s] * t for each valid s
            
            # This is getting complicated. Let me just do it numerically:
            # Evaluate corr at t=0 and t=1, then linearly interpolate.
            
            h_cur[i] = 0.0
            corr0 = np.correlate(h_cur, 1.0 - h_cur, mode='full') * (2.0/n)
            h_cur[i] = 1.0
            corr1 = np.correlate(h_cur, 1.0 - h_cur, mode='full') * (2.0/n)
            h_cur[i] = h_save
            
            # corr(t) = corr0 + t * (corr1 - corr0)
            # max_k corr_k(t) = max_k (corr0[k] + t * (corr1[k] - corr0[k]))
            # This is max of linear functions of t, hence convex in t.
            # Minimum is at a point where two lines cross, or at a boundary.
            
            diff = corr1 - corr0
            # The max function's minimum: find t* that minimizes max_k (corr0[k] + t*diff[k])
            # This is a 1D convex optimization. Use ternary search or bisection on the derivative.
            
            # Simple approach: evaluate at a few points and pick the best.
            # Since it's convex, the minimum is where the "slope" changes from negative to positive.
            # The subgradient at t is: max over k with corr_k(t) = max of diff[k].
            
            # Let me just do a fine grid search over t ∈ [0, 1]:
            t_vals = np.linspace(0, 1, 21)
            best_t = 0
            best_val = np.inf
            for t in t_vals:
                corr_t = corr0 + t * diff
                val = np.max(corr_t)
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine with a finer grid around best_t
            if 0 < best_t < 1:
                t_fine = np.linspace(max(0, best_t - 0.05), min(1, best_t + 0.05), 21)
                for t in t_fine:
                    corr_t = corr0 + t * diff
                    val = np.max(corr_t)
                    if val < best_val:
                        best_val = val
                        best_t = t
            
            h_cur[i] = best_t
            improved = True
        
        # After a full pass, re-project to maintain the sum constraint
        h_cur = project_onto_constraint(h_cur, target_sum)
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    
    return best_h, best_c5
```

Wait, there's a problem with the coordinate descent approach. When I change h[i], the sum constraint is violated. I need to either:
1. Re-project after each coordinate change (expensive - O(n) per step).
2. Only do the projection after a full pass (might be fine since the changes are small).
3. Use a different parameterization that automatically satisfies the constraint.

Option 2 is the most practical. The sum constraint will be slightly violated during a pass, but the re-projection at the end will fix it.

Actually, there's a subtlety: when I compute corr0 and corr1, I'm computing the correlation with h[i]=0 and h[i]=1 respectively. These don't satisfy the sum constraint (the sum is off by h_save). But that's OK because I'm just using them to determine the optimal value of h[i] - the correlation values are still valid as a function of h[i].

Wait, actually the correlation computation doesn't depend on the sum constraint - it's just a mathematical function. The sum constraint is a separate requirement on the final solution. So the coordinate descent is valid: I find the best h[i] that minimizes C₅ (ignoring the sum constraint), and then project at the end of each pass.

This should work! And it's much faster than SLSQP because each coordinate update is just 2 correlation computations (O(n) each) plus a 41-point grid search (negligible).

Total cost per pass: 2 * n * O(n) = O(n²) = 32400 for n=180. Plus the projection: O(n log n) or O(n) with the bisection.

For 100 passes: 3.2M operations. Very fast! This can run in a fraction of a second.

So I can afford to do many passes, which means the coordinate descent will converge well.

Let me combine coordinate descent with SLSQP:
1. SLSQP for global-ish optimization (with smooth max).
2. Coordinate descent for fine local optimization (exact, no approximation).

The coordinate descent should be more accurate than SLSQP because it doesn't use a smooth approximation of the max - it directly evaluates the max at each step.

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

    bounds = [(0.0, 1.0)] * n

        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
