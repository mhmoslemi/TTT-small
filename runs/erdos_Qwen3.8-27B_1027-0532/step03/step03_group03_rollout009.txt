from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()

    # Phase 10: Final polish
    if _remaining() > 30:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_cd = coord_descent(best_h, target_cur, n_sweeps=20, order_seed=seed + 700)
        sc = compute_c5(h_cd)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, n_use
