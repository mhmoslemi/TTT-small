```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit, damping=0.5, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
        except:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000)
            except:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5
```

```python
                break

            best_h = h.copy()
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
```

**Direct approach: Minimize max_k f_k(h) using bisection + feasibility**

For a target T, check if there exists h with:
- 0 ≤ h ≤ 1
- sum(h) = n/2
- f_k(h) ≤ T for all k

**Key insight**: The alternating LP finds a "saddle point" of the bilinear function (h, g) ↦ max_k h^T A_k g where g = 1-h. At the saddle point, both h and g are optimal responses to each other.

The convergence of the alternating method depends on the "conditioning" of the problem. If the problem is well-conditioned, it converges fast. If not, we might need many iterations.

To improve convergence:
1. Use **extrapolation** (like FISTA) to accelerate
2. Use **averaging** of iterates (take the average of the last K iterates)
3. Use **adaptive step size** based on the observed improvement

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_new, x_old, h_src)
    return project_feasible(h_dst, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    x_new = np.linspace(0.0, 2.0, n_dst)
    h_dst = np.interp(x_new, x_old, h_src)
    return project_feasible(h_dst, target_dst)


def alternating_lp_cg(h_init, n, target, max_iter, time_limit, damping=1.0, reg=1e-4, n_constr=30, avg_window=200):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    avg_h = np.zeros(n)
    count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        top_idx = np.argsort(corr)[-n_constr:]

        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]

        for idx in top_idx:
            k = int(idx) - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        obj = t_var + reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(obj), constraints)
        solved = False
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
            solved = True
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break

        h_lp = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h_new = (1.0 - damping) * h + damping * h_lp
        h_new = project_feasible(h_new, target)

        avg_h += h_new
        count += 1
        if count >= avg_window:
            cand = project_feasible(avg_h / count, target)
            c5_cand = compute_c5(cand)
            if c5_cand < best_c5:

            best_c5 = c5_val
            best_h = h_new.copy()
        h = h_new

    return best_h, best_c5

            h5b, c5_5b = alternating_lp_cg(best_h, n, target, max_iter=4000,
                                            time_limit=t5b, damping=0.4, reg=1e-5,
                                            n_constr=60, avg_window=300)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b

    # Phase 6: Final subgradient polish
    if tleft() > 50:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h6, c5_6 = subgradient_improve(best_h, n_f, target_f,
                                        max_steps=5000, time_limit=min(60, tleft() * 0.5))
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6

    # Phase 7: One more LP round if time permits
    if tleft() > 60:
        n_f = len(best_h)
        target_f = n_f / 2.0
        h7, c5_7 = alternating_lp_cg(best_h, n_f, target_f, max_iter=3000,
                                      time_limit=min(60, tleft() * 0.6), damping=0.3,
                                      reg=1e-6, n_constr=50, avg_window=200)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_c5 = compute_c5(h_final)

    return h_final, final_c5, n_final
```
