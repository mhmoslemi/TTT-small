def subgradient_optimize(h_init, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    # Track average for Polyak step
    h_avg = h_cur.copy()
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        c_star = corr[k_star]
        
        # Compute subgradient of the active constraint
        grad = np.zeros(n)
        k = k_star
        for i in range(n):
            # d/dh[i] of sum_j h[j](1-h[j+k])
            # = (1-h[i+k]) if i+k < n  [from h[i] term]
            #   - h[i-k] if i-k >= 0  [from h[i-k] term where i = j+k]
            if i + k < n and i + k >= 0:
                grad[i] += (1.0 - h_cur[i + k])
            if i - k >= 0 and i - k < n:
                grad[i] -= h_cur[i - k]
        grad *= dx
        
        # Subgradient step with backtracking
        found = False
        for lr in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)

            if sc < best_c5:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try

                break

            sc_avg = compute_c5(h_avg_try)
            if sc_avg < best_c5:
                best_c5 = sc_avg
                best_h = h_avg_try.copy()
    
    return best_h, best_c5

def alternating_lp_cutting(h_init, n, target, max_iter, time_limit, damping_init=0.6, reg=1e-4, max_constraints=50):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg
    
    # Maintain a list of active constraint indices
    active_ks = [0]  # start with k=0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        # Only add constraints for active k values
        for k in active_ks:
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        # Check for violated constraints and add new ones
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr)) - (n - 1)
        if k_max not in active_ks:
            active_ks.append(k_max)
        
        # Also add top few correlations
        top_ks = np.argsort(corr)[-5:]
        for k_idx in top_ks:
            k_val = int(k_idx) - (n - 1)
            if k_val not in active_ks:
                active_ks.append(k_val)
        
        # Limit number of constraints
        if len(active_ks) > max_constraints:
            # Keep the most violated ones
            violations = []
            for k in active_ks:
                lo = max(0, k)
                hi = min(n, k + n)
                if hi > lo:
                    val = np.sum(h[lo:hi] * (1.0 - h[lo-k:hi-k])) * dx
                    violations.append((k, val))
            violations.sort(key=lambda x: -x[1])
            active_ks = [v[0] for v in violations[:max_constraints]]
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
        
        if it % 100 == 0 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.7)
    
    return best_h, best_c5

Let me try a different approach: **projected gradient with the soft-max objective**.

At each step:
1. Compute gradient of smooth_max(corr(h)) w.r.t. h
2. Take a gradient step
3. Project onto the feasible set (0 ≤ h ≤ 1, sum(h) = target)

The gradient computation:
- smooth_max(corr) = max_corr + (1/alpha) * log(∑ exp(alpha * (corr_k - max_corr)))
- d/dh[i] = (1/alpha) * ∑_k exp(alpha*(corr_k - max_corr)) / ∑_j exp(alpha*(corr_j - max_corr)) * d(corr_k)/dh[i]
- d(corr_k)/dh[i] = (1-h[i+k])*dx - h[i-k]*dx (for valid indices)

This is computable in O(n * K) where K is the number of correlations (2n-1). For n=400, that's O(320000) per gradient evaluation, which is fast.

The advantage of projected gradient over SLSQP:
- No need to solve KKT system
- Can use many more iterations
- Easier to implement parallel computation

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target

def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def alternating_lp_cuts(h_init, n, target, max_iter, time_limit,

    stuck = 0

def compute_single_corr(h, k, n):
    dx = 2.0 / n
    lo = max(0, k)
    hi = min(n, k + n)
    if hi <= lo:
        return 0.0
    val = np.sum(h[lo:hi] * (1.0 - h[lo - k:hi - k]))
    return val * dx


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                               max_iter=200000, time_limit=pg_time,
                                               alpha_start=50.0, alpha_end=2000.0, n_top=15)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 4: SLSQP refinement
    if tleft() > 150:
        alphas = [20, 80, 300, 1000, 2000]
        h_refined = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                  alphas, maxiter_per_alpha=1500, ftol=1e-14)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 5: Another round of projected gradient (very high alpha)
    if tleft() > 150:
        pg_time = min(150, tleft() * 0.2)
        h_pg, c5_pg = projected_gradient_smax(best_h, len(best_h), len(best_h) / 2.0,
                                               max_iter=100000, time_limit=pg_time,
                                               alpha_start=200.0, alpha_end=8000.0, n_top=10)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Phase 6: Alternating LP second pass from perturbed start
    for restart in range(3):
        if tleft() < 100:
            break
        noise = [0.02, 0.05, 0.1][restart]
        h_pert = best_h + rng.randn(len(best_h)) * noise
        h_pert = project_feasible(h_pert, len(best_h) / 2.0)
        lp_time = min(80, tleft() * 0.12)
        h_lp, c5_lp = alternating_lp_cuts(h_pert, len(best_h), len(best_h) / 2.0,
                                          max_iter=300, time_limit=lp_time,
                                          damping_init=0.5, reg=1e-4,
                                          max_active=30, momentum=True)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 7: Subgradient polish
    if tleft() > 80:
        h_sg, c5_sg = subgradient_optimize(best_h, len(best_h), len(best_h) / 2.0,
                                            max_steps=5000, time_limit=min(80, tleft() * 0.4))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 60:
        h_final_try = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                    [500, 1500, 4000, 8000], maxiter_per_alpha=1000, ftol=1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 9: Final projected gradient burst
    if tleft() > 40:
        h_pg, c5_pg = projected_gradient_smax(best_h, len(best_h), len(best_h) / 2.0,
                                               max_iter=50000, time_limit=min(40, tleft() * 0.5),
                                               alpha_start=500.0, alpha_end=15000.0, n_top=8)
        if c5_pg < best_score:
            best_score = c5_pg
            best_h = h_pg

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
