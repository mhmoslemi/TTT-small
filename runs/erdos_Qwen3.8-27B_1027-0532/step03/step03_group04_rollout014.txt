def compute_gradient_smooth_max(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    
    max_ov = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    # Build the gradient matrix
    # grad[j] = dx * sum_p weights[p] * ((1-h[j+k_p])*I[0<=j+k_p<n] - h[j-k_p]*I[0<=j-k_p<n])
    # where k_p = p - (n-1)
    
    # Create index matrix
    j_idx = np.arange(n)[:, None]  # (n, 1)
    p_idx = np.arange(2*n-1)[None, :]  # (1, 2n-1)
    k_vals = p_idx - (n-1)  # (1, 2n-1)
    
    # j + k: (n, 2n-1)
    jk = j_idx + k_vals
    jm = j_idx - k_vals
    
    mask1 = (jk >= 0) & (jk < n)  # (n, 2n-1)
    mask2 = (jm >= 0) & (jm < n)  # (n, 2n-1)
    
    # first[j] = sum_p weights[p] * (1-h[j+k_p]) * mask1
    # = sum_p weights[p] * (1-h[jk]) * mask1
    first_terms = (1-h)[jk] * mask1 * weights[None, :]
    second_terms = h[jm] * mask2 * weights[None, :]
    
    first = first_terms.sum(axis=1)
    second = second_terms.sum(axis=1)
    
    return dx * (first - second)

Actually, I want to try one more creative idea. What about using **scipy.optimize.linprog** directly for the LP inner problem? It might be faster than cvxpy for this specific problem.

The inner LP is:
min t
s.t. sum(h) = target
     0 ≤ h ≤ 1
     A_k @ h ≤ t  for k in active_set

This is a standard LP. I can use scipy.optimize.linprog with method='highs' (HiGHS solver), which is very fast and accurate.

Let me reformulate:
- Variables: [h_0, ..., h_{n-1}, t] (total n+1 variables)
- Equality constraint: sum(h) = target → [1,1,...,1,0] @ x = target
- Inequality constraints:
  - -h_i ≤ 0 for each i (i.e., h_i ≥ 0)
  - h_i ≤ 1 for each i
  - A_k @ h - t ≤ 0 for each active k

The objective is to minimize t → c = [0,0,...,0,1]

Let me use this approach with HiGHS:

```python
from scipy.optimize import linprog

def solve_inner_lp(h_cur, n, target, active_shifts, max_shifts=200):
    dx = 2.0 / n
    g = 1.0 - h_cur
    
    overlaps = compute_all_overlaps(h_cur)
    if max_shifts is not None:
        top_idx = np.argsort(overlaps)[::-1][:max_shifts]
    else:
        top_idx = np.arange(2*n-1)
    
    m = len(top_idx) + n + 1  # number of inequality constraints
    
    # Build A_ub (inequality constraints: A_ub @ x <= b_ub)
    # x = [h_0, ..., h_{n-1}, t]
    # Constraints:
    # 1. -h_i <= 0 (n constraints)
    # 2. h_i <= 1 (n constraints)  -- or use bounds
    # 3. A_k @ h - t <= 0 (m constraints)
    
    # Actually, use bounds for h_i: [0, 1]
    # Then A_ub only has the shift constraints
    # A_ub shape: (m, n+1)
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    
    for idx_i, shift_idx in enumerate(top_idx):
        k = shift_idx - (n-1)
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            A_ub[idx_i, lo:hi] = g[lo-k:hi-k] * dx
            A_ub[idx_i, n] = -1.0
            b_ub[idx_i] = 0.0
    
    # Objective: minimize t → c = [0,...,0,1]
    c = np.zeros(n+1)
    c[n] = 1.0
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    # Equality constraint: sum(h) = target
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]
    return None
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_box_sum(h, target):
    n = len(h)
    if n == 0:
        return h.astype(float)
    h = h.astype(float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj_fn(h, alpha):
    overlaps = compute_all_overlaps(h)
    c_max = np.max(overlaps)
    shifted = np.clip(alpha * (overlaps - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_all_overlaps(h)
    c_max = np.max(overlaps)
    exp_vals = np.exp(alpha * (overlaps - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    j_idx = np.arange(n)[:, None]
    k_vals = np.arange(-(n - 1), n)[None, :]
    jk = j_idx + k_vals
    jm = j_idx - k_vals
    mask1 = (jk >= 0) & (jk < n)
    mask2 = (jm >= 0) & (jm < n)
    jk_c = np.clip(jk, 0, n - 1)
    jm_c = np.clip(jm, 0, n - 1)
    
    first = ((1.0 - h)[jk_c] * mask1 * weights[None, :]).sum(axis=1)
    second = (h[jm_c] * mask2 * weights[None, :]).sum(axis=1)
    
    return dx * (first - second)

            return smooth_obj_fn(h, alpha)
        
        def obj_grad(h):
            return smooth_obj_grad(h, alpha)

            return np.sum(h) - target
        
        def con_jac(h):
            return np.ones(n)

            if compute_c5(h_new) <= compute_c5(h_cur) + 1e-12:
