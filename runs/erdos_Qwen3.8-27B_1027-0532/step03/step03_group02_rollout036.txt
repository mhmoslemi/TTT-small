    n = len(h)

    size = 2 * n

    # Use FFT for cross-correlation
    f1 = np.fft.rfft(h, size)
    f2 = np.fft.rfft(g, size)

    corr = np.concatenate([corr[n-1:], corr[:n-1]])
    return corr[:2*n-1]

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_max(corr, alpha):
    """Log-sum-exp approximation of max"""
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def obj_and_grad(h, alpha):
    """Objective and gradient for smooth max of correlation"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    
    # Smooth max value
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    obj = (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx
    
    # Gradient of smooth max
    weights = np.exp(shifted) / np.sum(np.exp(shifted))

    # d(corr_k)/d(h_j) = (1-h[j+k]) - h[j-k] (when these indices are valid)

    # grad_h[j] += weights[k] * ((1-h[j+k]) - h[j-k])

    grad = np.zeros(n)
    s_arr = np.arange(2*n-1) - (n-1)

        if weights[idx] < 1e-10:

        s = s_arr[idx]

        js_plus = np.arange(n) + s

        m1 = (js_plus >= 0) & (js_plus < n)
        m2 = (js_minus >= 0) & (js_minus < n)

        contrib[m1] += (1.0 - h[js_plus[m1]])

def project_onto_simplex(h, target_sum):
    """Project h onto {x : 0<=x<=1, sum(x)=target_sum} using bisection"""
    n = len(h)
    # Binary search for lambda such that sum(clip(h - lambda, 0, 1)) = target_sum
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

One more optimization: in the coordinate descent, instead of doing 20 bisection steps, I can note that the optimal t is at a breakpoint. The breakpoints are where A[idx]*t + B[idx] crosses 0 for the active constraint. But finding them exactly is complex.

Alternative: since the function g(t) = max_k(A[k]*t + B[k]) is convex and piecewise linear, I can find the minimum by:
1. Computing g(0) and g(1)
2. If g(0) ≤ g(1) and g is increasing at 0, the minimum is at 0
3. If g(1) ≤ g(0) and g is decreasing at 1, the minimum is at 1
4. Otherwise, bisection

The "slope" at a point is the A[k*] where k* = argmax (A[k]*t + B[k]). At t=0, k* = argmax(B). At t=1, k* = argmax(A+B).

So:
- slope_at_0 = A[argmax(B)]
- slope_at_1 = A[argmax(A+B)]

If slope_at_0 >= 0: minimum at t=0
If slope_at_1 <= 0: minimum at t=1
Otherwise: bisection (and we'll converge faster because we already know the slope changes sign)

This can reduce the bisection iterations significantly in many cases.

    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    start = time.time()

        improved = False
        for j in range(n):
            old = h[j]
            # Compute A vectorized
            s = s_arr
            jp = j + s
            jm = j - s
            A = np.zeros(2 * n - 1)
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            A[mp] += (1.0 - h[jp[mp]])
            A[mm] -= h[jm[mm]]
            
            B = corr - A * old
            
            # Find minimizer of max_k(A[k]*t + B[k]) on [0,1]
            # Check endpoints
            k0 = np.argmax(B)
            k1 = np.argmax(A + B)
            slope0 = A[k0]
            slope1 = A[k1]
            
            if slope0 >= -1e-14:
                new_val = 0.0
            elif slope1 <= 1e-14:
                new_val = 1.0
            else:
                lo, hi = 0.0, 1.0
                for _ in range(18):

                    vals = A * mid + B
                    k_star = np.argmax(vals)

                    elif A[k_star] < -1e-14:

                corr += delta * A
                h[j] = new_val
                improved = True

        corr = np.correlate(h, 1.0 - h, mode='full')

                h = project_feasible(h, target_sum)

                no_improve = 0
    
    return best_h, best_score

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = np.asarray(h, dtype=np.float64)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-10:
        interior = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if interior.sum() > 0:
            proj[interior] += diff / interior.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_vals = np.exp(shifted)
    Z = exp_vals.sum()
    obj = (c_max + np.log(Z) / alpha) * dx
    weights = exp_vals / Z
    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx]
        if w < 1e-15:
            continue
        s_val = s_arr[idx]
        jp = np.arange(n) + s_val
        jm = np.arange(n) - s_val
        m1 = (jp >= 0) & (jp < n)
        m2 = (jm >= 0) & (jm < n)
        c = np.zeros(n)
        c[m1] += (1.0 - h[jp[m1]])
        c[m2] -= h[jm[m2]]
        grad += w * c
    grad *= dx
    return obj, grad

def fista_optimize(h_init, target_sum, alpha, time_limit, rng=None):
    n = len(h_init)
    h = project_feasible(h_init.copy(), target_sum)
    h_y = h.copy()
    t_k = 1.0
    step = 5.0 / (alpha * n)
    best_score = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stagnant = 0

    while time.time() - start < time_limit:
        obj, grad = smooth_obj_grad(h, alpha)
        h_new = project_feasible(h - step * grad, target_sum)
        f_new = compute_c5(h_new)
        backtracks = 0
        while f_new > obj + 0.5 * step * np.dot(grad, h_new - h) and backtracks < 25:
            step *= 0.5
            h_new = project_feasible(h - step * grad, target_sum)
            f_new = compute_c5(h_new)
            backtracks += 1
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        h_y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        h_y = project_feasible(h_y, target_sum)
        h = h_new
        t_k = t_new
        if f_new < best_score - 1e-14:
            best_score = f_new
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 500:
                break
    return best_h, best_score

def slsqp_polish(h, target_sum, time_limit, alphas=None):
    n = len(h)
    if alphas is None:
        alphas = [200, 500, 1000]
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()

    def obj_func(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(500.0 * (corr - c_max), -500.0, 0.0)
        return (c_max + np.log(np.sum(np.exp(shifted))) / 500.0) * (2.0 / n)

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: x.sum() - target_sum,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: FISTA at initial resolution
    if tleft() > 300:
        h_f, sc_f = fista_optimize(best_h, target_init, 80, min(150, tleft() - 300), rng)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 2: Long coordinate descent at initial resolution
    if tleft() > 200:
        h_cd, sc_cd = coord_descend_fast(best_h, target_init, min(500, tleft() - 200), patience=50, rng=rng)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: SLSQP polish
    if tleft() > 100:
        h_sl, sc_sl = slsqp_polish(best_h, target_init, min(80, tleft() - 100), [300, 800])
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    # Phase 4: Try different resolutions
    n_attempts = [n_init + 100, n_init + 200, n_init + 300, n_init - 100]
    n_attempts = sorted(set(n for n in n_attempts if n >= 200 and n != n_init))
    for n_try in n_attempts:
        if tleft() < 150:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_co, sc_co = coord_descend_fast(h_interp, target_try, min(120, tleft() - 150), patience=30, rng=rng)
        if sc_co < best_score:
            best_score = sc_co
            best_h = h_co

    # Phase 5: Final coordinate descent on best
    n_best = len(best_h)
    target_best = n_best / 2.0
    if tleft() > 100:
        h_final_cd, sc_final = coord_descend_fast(best_h, target_best, min(150, tleft() - 100), patience=60, rng=rng)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final_cd

    # Phase 6: Final SLSQP polish
    if tleft() > 50:
        h_pol, sc_pol = slsqp_polish(best_h, target_best, min(40, tleft() - 50), [500, 1000])
        if sc_pol < best_score:
            best_score = sc_pol
            best_h = h_pol

    # Final feasibility
    n_final = len(best_h)
    h_final = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n_final
