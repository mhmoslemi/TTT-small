            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if c0 < co - 1e-13:
                h[j] = 0.0
                best_c = c0
                improved = True
            elif c1 < co - 1e-13:
                h[j] = 1.0
                best_c = c1
                improved = True
        if not improved:
            break
    return h, best_c

def _pair_cd(h0, n, target_sum, max_rounds=15, deadline=None, rng=None):
    h = h0.copy()
    best_c = _c5_of(h)
    n_try_per_round = min(n * 3, 300)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for _ in range(n_try_per_round):
            i, j = rng.randint(0, n, size=2)
            if i == j:
                continue
            if i > j:
                i, j = j, i
            old_i, old_j = h[i], h[j]
            for delta in [0.02, 0.05, 0.1, 0.15]:
                if h[i] + delta <= 1.0 + 1e-12 and h[j] - delta >= -1e-12:
                    h[i] = min(h[i] + delta, 1.0)
                    h[j] = max(h[j] - delta, 0.0)
                    h = _fix_sum(h, target_sum)
                    c = _c5_of(h)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        break
                    h[i] = old_i
                    h[j] = old_j
                    h = _fix_sum(h, target_sum)
                if h[i] - delta >= -1e-12 and h[j] + delta <= 1.0 + 1e-12:
                    h[i] = max(h[i] - delta, 0.0)
                    h[j] = min(h[j] + delta, 1.0)
                    h = _fix_sum(h, target_sum)
                    c = _c5_of(h)
                    if c < best_c - 1e-14:
                        best_c = c
                        improved = True
                        break
                    h[i] = old_i
                    h[j] = old_j
                    h = _fix_sum(h, target_sum)
            else:
                h[i] = old_i
                h[j] = old_j
                h = _fix_sum(h, target_sum)
                continue
            break
        if not improved:
            break
    return h, best_c

def _gen_binary_start(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _gen_periodic_start(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < period * duty:
            h[i] = 1.0
    return h

def _optimize_one_start(h_start, n, target_sum, deadline, rng, phase_budget_frac):
    h = h_start.copy()
    c5 = _c5_of(h)
    best_h = h.copy()
    best_c5 = c5
    remaining = max(10, (deadline - time.time()) * phase_budget_frac)
    sub_deadline = time.time() + remaining

    # MW phase
    if time.time() < sub_deadline - 30:
        n_mw = max(100, int(remaining * 0.3))
        h_mw, c5_mw = _mw_optimize(h, n, target_sum, n_iters=n_mw,
                                    eta0=15, alpha0=0.25,
                                    deadline=sub_deadline - 30)
        if c5_mw < best_c5:
            best_c5 = c5_mw
            best_h = h_mw.copy()
        h = h_mw

    # Nesterov single
    if time.time() < sub_deadline - 20:
        h_n, c5_n = _nesterov_single(h, n, target_sum, n_iters=200,
                                      alpha0=0.4, deadline=sub_deadline - 20)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()
        h = h_n

    # SLSQP with increasing alpha
    for alpha in [200, 1000, 5000, 15000]:
        if time.time() > sub_deadline - 10:
            break
        h_sl = _slsqp_opt(h, n, target_sum, alpha, maxiter=250)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
        h = h_sl

    # Coordinate descent
    if time.time() < sub_deadline - 5:
        h_cd, c5_cd = _coord_descent(h, n, target_sum, max_rounds=30,
                                      deadline=sub_deadline - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        h = h_cd

    # Pair CD
    if time.time() < sub_deadline - 3:
        h_pc, c5_pc = _pair_cd(h, n, target_sum, max_rounds=10,
                                deadline=sub_deadline - 3, rng=rng)
        if c5_pc < best_c5:
            best_c5 = c5_pc
            best_h = h_pc.copy()
        h = h_pc

    # Final SLSQP polish
    if time.time() < sub_deadline - 2:
        h_pol = _slsqp_opt(h, n, target_sum, 50000, maxiter=300)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5_global = float('inf')
    best_h_global = None
    best_n_global = n_init

    n_candidates = [n_init]
    for delta in [-8, 8, -16, 16, -24, 24, -4, 4, 32, -32]:
        n_try = n_init + delta
        if n_try % 2 == 0 and 100 <= n_try <= 220:
            if n_try not in n_candidates:
                n_candidates.append(n_try)

    n_candidates_sorted = sorted(n_candidates, key=lambda x: abs(x - n_init))

    # Determine time per grid size
    n_grids = len(n_candidates_sorted)
    time_per_grid = (deadline - t_start - 30) / max(n_grids, 1)

    for gi, n in enumerate(n_candidates_sorted):
        if time.time() > deadline - 25:
            break
        target_sum = n / 2.0
        grid_deadline = min(deadline - 20, t_start + 30 + (gi + 1) * time_per_grid)

        # Build starting points
        starts = []
        if n == n_init:
            starts.append(_fix_sum(h_init.copy(), target_sum))
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n),
                                  np.arange(n_init), h_init)
            starts.append(_fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum))

        # Add a binary start
        starts.append(_fix_sum(_gen_binary_start(n, rng), target_sum))

        # Add a periodic start with golden-ratio-like period
        period = max(3, int(n / 5))
        starts.append(_fix_sum(_gen_periodic_start(n, period, 0.5), target_sum))

        # Another periodic with different period
        period2 = max(3, int(n / 7))
        starts.append(_fix_sum(_gen_periodic_start(n, period2, 0.4), target_sum))

        # Interpolate between starts for more diversity
        if len(starts) >= 2:
            starts.append(_fix_sum(0.5 * (starts[0] + starts[1]), target_sum))

        # Evaluate each start and optimize the best ones
        start_scores = []
        for s in starts:
            cs = _c5_of(s)
            start_scores.append((cs, s))
            if cs < best_c5_global:
                best_c5_global = cs
                best_h_global = s.copy()
                best_n_global = n

        start_scores.sort(key=lambda x: x[0])

        # Optimize top starts (spend more time on better ones)
        n_starts_to_optimize = min(3, len(start_scores))
        time_left_for_starts = max(10, grid_deadline - time.time() - 30)
        weights = [1.0 / (i + 1) for i in range(n_starts_to_optimize)]
        w_sum = sum(weights)
        weights = [w / w_sum for w in weights]

        for si in range(n_starts_to_optimize):
            if time.time() > grid_deadline - 15:
                break
            h_s, c5_s = _optimize_one_start(start_scores[si][1], n, target_sum,
                                             grid_deadline, rng, weights[si])
            if c5_s < best_c5_global:
                best_c5_global = c5_s
                best_h_global = h_s.copy()
                best_n_global = n

            # If this is the best for this grid, do extra refinement
            if si == 0 and time.time() < grid_deadline - 10:
                for alpha in [20000, 100000]:
                    if time.time() > grid_deadline - 8:
                        break
                    h_extra = _slsqp_opt(h_s, n, target_sum, alpha, maxiter=400)
                    c5_extra = _c5_of(h_extra)
                    if c5_extra < best_c5_global:
                        best_c5_global = c5_extra
                        best_h_global = h_extra.copy()
                        best_n_global = n
                        h_s = h_extra
                if time.time() < grid_deadline - 5:
                    h_cd_f, c5_cd_f = _coord_descent(h_s, n, target_sum,
                                                      max_rounds=30,
                                                      deadline=grid_deadline - 5)
                    if c5_cd_f < best_c5_global:
                        best_c5_global = c5_cd_f
                        best_h_global = h_cd_f.copy()
                        best_n_global = n

    # Final global polish on the best solution
    if best_h_global is not None and time.time() < deadline - 5:
        n_final = best_n_global
        target_final = n_final / 2.0
        h_fin = best_h_global.copy()
        for alpha in [10000, 50000, 200000]:
            if time.time() > deadline - 4:
                break
            h_fin = _slsqp_opt(h_fin, n_final, target_final, alpha, maxiter=400)
            c5_fin = _c5_of(h_fin)
            if c5_fin < best_c5_global:
                best_c5_global = c5_fin
                best_h_global = h_fin.copy()
        if time.time() < deadline - 3:
            h_cd_fin, c5_cd_fin = _coord_descent(best_h_global, n_final,
                                                  target_final, max_rounds=25,
                                                  deadline=deadline - 3)
            if c5_cd_fin < best_c5_global:
                best_c5_global = c5_cd_fin
                best_h_global = h_cd_fin.copy()
        if time.time() < deadline - 2:
            h_pc_fin, c5_pc_fin = _pair_cd(best_h_global, n_final,
                                            target_final, max_rounds=8,
                                            deadline=deadline - 2, rng=rng)
            if c5_pc_fin < best_c5_global:
                best_c5_global = c5_pc_fin
                best_h_global = h_pc_fin.copy()

    best_h_global = _fix_sum(best_h_global, best_n_global / 2.0)
    best_c5_global = _c5_of(best_h_global)
    return (best_h_global, best_c5_global, best_n_global)
