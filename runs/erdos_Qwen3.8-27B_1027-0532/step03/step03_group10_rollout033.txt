def project_onto_constraint(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum} via water-filling."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def subgradient_at(h, k):
    """Compute subgradient of corr[k] w.r.t. h."""
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        # corr[k] = sum_{i=s}^{n-1} h[i] * (1 - h[i-s])

        #           - h[j+s] for j in [0, n-1-s] (j plays role of i-s)
        m = n - s
        grad[s:] += (1.0 - h[:m])

    else:

        m = n - r
        grad[:m] += (1.0 - h[r:])

    return grad

    n = len(h_init)
    h = h_init.astype(np.float64).copy()

    y = h.copy()  # Nesterov extrapolation

    corr = compute_corr(h)
    
    for t in range(n_iter):

        # Compute smoothed objective value and "soft" weights

from scipy.optimize import minimize
import time

_G_T = 0.0

def proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def c5v(h):
    n = len(h)
    return float(2.0 * np.max(np.correlate(h, 1.0 - h, mode='full')) / n)

def cv(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def tgrad(h, ki, wi):
    n = len(h)
    g = np.zeros(n)
    for kk, ww in zip(ki, wi):
        if ww < 1e-12:
            continue
        s = kk - (n - 1)
        if s >= 0:
            m = n - s
            g[s:] += ww * (1.0 - h[:m])
            g[:m] -= ww * h[s:]
        else:
            r = -s
            m = n - r
            g[:m] += ww * (1.0 - h[r:])
            g[r:] -= ww * h[:m]
    return g

def lb_obj(h, eps, lam, tgt):
    c = cv(h)
    cm = np.max(c)
    x = (c - cm) / eps
    x -= np.max(x)
    return cm + eps * np.log(np.sum(np.exp(x))) + lam * (h.sum() - tgt) ** 2

def lb_grad(h, eps, lam, tgt):
    n = len(h)
    c = cv(h)
    cm = np.max(c)
    x = (c - cm) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    k = min(60, len(c))
    ti = np.argsort(w)[::-1][:k]
    tw = w[ti]
    tw /= tw.sum()
    g = tgrad(h, ti, tw)
    g += 2.0 * lam * (h.sum() - tgt)
    return g

def lbfgs(h0, tgt, niter=300, eps=0.002):

    h = proj(h0, tgt).copy()
    lam = 1.0

        if abs(h.sum() - tgt) < 0.01:

        lam *= 5.0
    h = proj(h, tgt)

    k = min(100, len(c))

    tw = w[ti]
    tw /= tw.sum()
    return tgrad(h, ti, tw)

def sq_cf(h):
    return h.sum() - _G_T

def sq_cj(h):
    return np.ones(len(h))

def slsqp(h0, eps, maxit, tgt):
    global _G_T
    _G_T = tgt
    n = len(h0)
    h = proj(h0, tgt).copy()
    try:
        res = minimize(sq_obj, h, args=(eps,), jac=_sq_grad,
                      method='SLSQP', bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': sq_cf, 'jac': sq_cj}],
                      options={'maxiter': maxit, 'ftol': 1e-13})
        h = proj(res.x, tgt)
    except Exception:
        pass
    return h, c5v(h)

def _sq_grad(h, eps):
    n = len(h)
    c = cv(h)
    cm = np.max(c)
    x = (c - cm) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    k = min(100, len(c))
    ti = np.argsort(w)[::-1][:k]
    tw = w[ti]
    tw /= tw.sum()
    return tgrad(h, ti, tw)

def tbin(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def bsa(h0, nsteps, t0, t1, rng, struct=False):
    n = len(h0)
    h = h0.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    cur = float(np.max(corr))
    best = cur
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    half = n - 1
    no = len(ones)
    nz = len(zeros)
    
    for step in range(nsteps):
        t = t0 * (t1 / t0) ** (step / max(nsteps - 1, 1))
        
        if struct and step > nsteps * 0.2 and rng.random() < 0.35:
            wk = int(np.argmax(corr))
            ws = wk - half
            vi = [x for x in ones if 0 <= x + ws < n and 0 <= x - ws < n]
            vj = [x for x in zeros if 0 <= x + ws < n and 0 <= x - ws < n]
            if vi and vj:
                i = vi[rng.integers(len(vi))]
                j = vj[rng.integers(len(vj))]
            else:
                i = ones[rng.integers(no)]
                j = zeros[rng.integers(nz)]
        else:
            i = ones[rng.integers(no)]
            j = zeros[rng.integers(nz)]
        
        if i == j:
            continue
        
        s1 = max(-i, i - half, -j, j - half)
        s2 = min(half - i, i, half - j, j)
        
        if s1 <= s2:
            sr = np.arange(s1, s2 + 1)
            kr = sr + half
            d = h[i - sr] + h[i + sr] - h[j - sr] - h[j + sr]
            # Fix overlap
            s_ov1 = i - j
            if s1 <= s_ov1 <= s2:
                d[s_ov1 - s1] += 1.0
            s_ov2 = j - i
            if s1 <= s_ov2 <= s2:
                d[s_ov2 - s1] += 1.0
            
            corr[kr] += d
            nm = float(np.max(corr))
            if nm < cur - 1e-12 or (t > 1e-8 and rng.random() < np.exp(-(nm - cur) / t)):
                h[i], h[j] = 0.0, 1.0
                oi = ones.index(i)
                oz = zeros.index(j)
                ones[oi] = j
                zeros[oz] = i
                cur = nm
                if cur < best:
                    best = cur
                    best_h = h.copy()
            else:
                corr[kr] -= d
        else:
            h[i], h[j] = 0.0, 1.0
            oi = ones.index(i)
            oz = zeros.index(j)
            ones[oi] = j
            zeros[oz] = i
    
    return best_h, best
