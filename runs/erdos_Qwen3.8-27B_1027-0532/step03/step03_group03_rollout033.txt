import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(5):
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
            if len(idx) == 0:
                break
            h = h.copy()
            step = diff / len(idx)
            h[idx] = np.minimum(h[idx] + step, 1.0)
        else:
            idx = np.where(h > 1e-14)[0]
            if len(idx) == 0:
                break
            h = h.copy()
            step = diff / len(idx)
            h[idx] = np.maximum(h[idx] + step, 0.0)
        diff = target_sum - np.sum(h)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_obj_and_grad(h, alpha, target_sum):
    n = len(h)
    corr = overlap_corrs(h)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(2 * n - 1):
        w = weights[i]
        if w < 1e-11:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]

    n = len(h_init)
    h_cur = h_init.copy()

        def obj_fn(x):
            val, _ = lse_obj_and_grad(x, alpha, target_sum)
            return val
        def grad_fn(x):
            _, g = lse_obj_and_grad(x, alpha, target_sum)
            return g

    return project_feasible(h_out, target_dst)

def coordinate_descent(h_init, target_sum, n_passes, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_score = compute_c5(h)
    for p in range(n_passes):
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            if old_val <= 1e-14 or old_val >= 1.0 - 1e-14:
                continue
            best_t = old_val
            best_sc = best_score
            for frac in [0.3, 0.5, 0.7, 1.3, 1.5, 1.7]:
                t_new = old_val * frac if frac < 1.0 else 1.0 - (1.0 - old_val) * frac
                if t_new <= 0.0 or t_new >= 1.0:
                    continue
                h_trial = h.copy()
                h_trial[i] = t_new
                h_trial = project_feasible(h_trial, target_sum)
                sc = compute_c5(h_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t_new
                    h_trial_best = h_trial.copy()
                    h = h_trial_best
                    best_score = best_sc
                    break
        if p % 10 == 0:
            h = project_feasible(h, target_sum)
    return h, best_score

def block_local_search(h_init, target_sum, n_trials, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_score = compute_c5(h)
    for _ in range(n_trials):
        block_size = rng.randint(4, min(20, n // 4))
        start = rng.randint(0, n - block_size)
        h_trial = h.copy()
        pert = rng.randn(block_size) * 0.015
        h_trial[start:start + block_size] += pert
        h_trial = project_feasible(h_trial, target_sum)
        sc = compute_c5(h_trial)
        if sc < best_score - 1e-12:
            h = h_trial
            best_score = sc
    return h, best_score
