def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, alpha, mu):
    """Compute smooth max objective with penalty, and its gradient."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = np.log(sum_exp)
    f_smooth = dx * (c_max + log_sum_exp / alpha)
    
    # Penalty term
    s = np.sum(h) - target
    f_penalty = mu * s * s
    f_val = f_smooth + f_penalty
    
    # Gradient
    weights = exp_shifted / sum_exp
    grad = np.zeros(n)
    
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s_shift = k - (n - 1)
        if s_shift >= 0:
            # g[j+s_shift] for j in [0, n-1-s_shift]
            grad[:n - s_shift] += w * g[s_shift:]
            # -h[j-s_shift] for j in [s_shift, n-1]
            grad[s_shift:] -= w * h[:n - s_shift]
        else:
            t = -s_shift
            # g[j-t] for j in [t, n-1]
            grad[t:] += w * g[:n - t]
            # -h[j+t] for j in [0, n-1-t]
            grad[:n - t] -= w * h[t:]
    
    grad *= dx
    # Add penalty gradient
    grad += 2.0 * mu * s * np.ones(n)
    
    return f_val, grad

def lbfgsb_optimize(h_init, alpha_list, mu_list, maxiter_per_alpha):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    
    for alpha in alpha_list:
        for mu in mu_list:
            def obj_and_grad(x):
                return smooth_max_obj_and_grad(x, alpha, mu)
            
            try:
                res = minimize(obj_and_grad, h_cur, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': maxiter_per_alpha,
                                                    'ftol': 1e-16, 'gtol': 1e-8,
                                                    'maxls': 50, 'maxfun': 5000})
                h_new = project_feasible(res.x, target)
                if compute_c5(h_new) < compute_c5(h_cur):
                    h_cur = h_new
            except Exception:
                pass
    
    return h_cur

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 20:
            damping = max(0.05, damping * 0.8)
            stuck_count = 0

        if it % 200 == 0 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.7)

    return best_h, best_c5

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = (n - 1)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        grad = np.zeros(n)
        k = k_star
        idx = np.arange(n)
        valid1 = (idx - k >= 0) & (idx - k < n)
        valid2 = (idx + k >= 0) & (idx + k < n)
        grad[valid1] += (1.0 - h_cur[idx[valid1] - k])
        grad[valid2] -= h_cur[idx[valid2] + k]
        grad *= dx

        for lr in [0.05, 0.1, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                break

    return best_h, best_c5

def local_flip_search(h, n, target, max_rounds, time_limit):
    """Try flipping individual elements to 0 or 1."""
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h_cur[i]
            # Try setting to 0
            if old_val > 0:
                h_cur[i] = 0.0
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_c5 - 1e-12:
                    best_c5 = sc
                    improved = True
                else:
                    h_cur[i] = old_val

                h_cur = project_feasible(h_cur, target)

                if sc < best_c5 - 1e-12:
                    best_c5 = sc
                    improved = True
                else:
                    h_cur[i] = old_val
        
        if not improved:
            break
    
    return h_cur, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_corr_weights(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    f_smooth = c_max + np.log(sum_exp) / alpha
    return f_smooth, weights, c_max

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _smooth_weights(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    f_smooth = c_max + np.log(sum_exp) / alpha
    return f_smooth, weights


def smooth_obj(x, alpha, mu):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    f_s, _ = _smooth_weights(x, alpha)
    f = dx * f_s
    s = np.sum(x) - target
    f += mu * s * s
    return f


def smooth_jac(x, alpha, mu):
    n = len(x)
    dx = 2.0 / n
    target = n / 2.0
    g = 1.0 - x
    _, weights = _smooth_weights(x, alpha)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s_shift = k - (n - 1)
        if s_shift >= 0:
            grad[:n - s_shift] += w * g[s_shift:]
            grad[s_shift:] -= w * x[:n - s_shift]
        else:
            t = -s_shift
            grad[t:] += w * g[:n - t]
            grad[:n - t] -= w * x[t:]
    grad *= dx
    s = np.sum(x) - target
    grad += 2.0 * mu * s
    return grad


def lbfgsb_round(h_init, alpha_list, mu_list, maxiter):
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    h_cur = h_init.copy()
    cur_score = compute_c5(h_cur)
    for alpha in alpha_list:
        for mu in mu_list:
            try:
                res = minimize(smooth_obj, h_cur, args=(alpha, mu),
                               method='L-BFGS-B', jac=smooth_jac,
                               bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-18,
                                         'gtol': 1e-10, 'maxls': 50, 'maxfun': 10000})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, cur_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
