```python
def simulated_annealing_binary(h_bin, n, target, t_start, t_end, n_iters, seed):
    """Simulated annealing on binary solutions with fixed weight."""
    rng = np.random.RandomState(seed)
    h = h_bin.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    best_h = h.copy()
    best_c = compute_c5(h)
    
    corr = compute_corr(h)
    c_max = np.max(corr)
    
    for it in range(n_iters):
        # Temperature schedule
        t = t_start * (t_end/t_start) ** (it / n_iters)
        
        # Pick random 1 and 0 positions
        p = ones[rng.randint(len(ones))]
        q = zeros[rng.randint(len(zeros))]
        
        if p == q:
            continue
        
        # Perform swap
        h[p], h[q] = h[q], h[p]
        
        # Compute new correlation
        new_corr = compute_corr(h)
        new_c = np.max(new_corr)
        
        # Energy change
        delta = new_c - c_max
        
        if delta < 0 or rng.rand() < np.exp(-delta / max(t, 1e-10)):
            # Accept
            c_max = new_c
            corr = new_corr
            # Update tracking
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # Reject
            h[p], h[q] = h[q], h[p]
    
    return best_h, best_c
```

```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, alpha, target, mu):
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    penalty = mu * (float(np.sum(h)) - target) ** 2
    return lse + penalty

def lse_obj_np(x, alpha, target, mu):
    return lse_obj(x, alpha, target, mu)

        h[p], h[q] = h[q], h[p]

        c = compute_c5(h)

        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-15)):

            # Update sets
            if h[p] == 1:  # p was 0, now 1
                ones = np.setdiff1d(ones, [p])
                ones = np.union1d(ones, [q])
                zeros = np.setdiff1d(zeros, [q])
                zeros = np.union1d(zeros, [p])
            else:
                ones = np.setdiff1d(ones, [q])
                ones = np.union1d(ones, [p])
                zeros = np.setdiff1d(zeros, [p])
                zeros = np.union1d(zeros, [q])
            
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            h[p], h[q] = h[q], h[p]
    
    return best_h, best_c
```

  - The penalty method handles the equality constraint approximately
  - With large μ, the constraint is satisfied to high accuracy

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_obj(x, alpha, target, mu):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    penalty = mu * (float(np.sum(x)) - target) ** 2
    return lse + penalty

    """Run L-BFGS-B with penalty method."""
    bounds = [(0.0, 1.0)] * n

        return penalty_obj(x, alpha, target, mu)

                   options={'maxiter': maxiter, 'ftol': 1e-15})
    h_new = project_feasible(res.x, target)
