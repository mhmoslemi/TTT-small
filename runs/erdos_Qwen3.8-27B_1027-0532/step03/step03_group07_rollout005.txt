For the local search, I'll use a "best first" approach:
1. Compute all corr values
2. Generate all possible swaps
3. For each swap, compute the new max corr (using the Δ formula)
4. Accept the best swap
5. Repeat until no improvement or time limit

Let me implement a more efficient version:

For each lag m that is "active" (corr[m] is close to the max), I want to find swaps that reduce corr[m]. From the formula:
Δcorr_m = h[i+m] - h[j+m] + h[i-m] - h[j-m]

To reduce corr_m, I want Δcorr_m < 0, i.e., h[i+m] + h[i-m] < h[j+m] + h[j-m].

For binary h, h[i+m] + h[i-m] can be 0, 1, or 2 (each term is 0 or 1).
Similarly for h[j+m] + h[j-m].

So I want to find i (with h[i]=1) and j (with h[j]=0) such that:
h[i+m] + h[i-m] < h[j+m] + h[j-m]

The best case is h[i+m] + h[i-m] = 0 and h[j+m] + h[j-m] = 2, giving Δcorr_m = -2.

This gives a nice structure for the search:
- Score each position i (with h[i]=1) by S_i(m) = h[i+m] + h[i-m] (lower is better)
- Score each position j (with h[j]=0) by S_j(m) = h[j+m] + h[j-m] (higher is better)
- The best swap for reducing corr_m is to pick i with minimum S_i(m) and j with maximum S_j(m)

For multiple active lags, I can use a weighted combination.

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)

        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add

            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

def _smooth_refine(h, n, target_sum, deadline):
    from scipy.optimize import minimize
    dx = 2.0 / n
    h = h.astype(float).copy()
    for alpha in [50, 100, 300, 1000]:
        if time.time() > deadline:
            break
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            if m < -500:
                return m * dx
            return (m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha) * dx
        def obj_grad(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            m = c.max()
            w = np.exp(alpha * (c - m))
            w = w / w.sum()
            g = np.zeros(n)
            for k_idx in np.where(w > 0.01 / len(w))[0]:
                wi = w[k_idx]
                if wi < 1e-10:
                    continue
                lag = k_idx - (n - 1)
                if lag >= 0:
                    mm = n - lag
                    if mm > 0:
                        g[lag:lag+mm] += wi * (1.0 - x[:mm])
                        g[:mm] += wi * (-x[lag:lag+mm])
                else:
                    jj = -lag
                    mm = n - jj
                    if mm > 0:
                        g[:mm] += wi * (1.0 - x[jj:])
                        g[jj:] += wi * (-x[:mm])
            return g * dx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, h, jac=obj_grad, method='L-BFGS-B',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-15})
            h = _fix_sum(res.x, target_sum)
        except Exception:
            pass
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    elif 'initial_h_values' in globals() and globals()['initial_h_values'] is not None:
        h_parent = np.array(globals()['initial_h_values'], dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    n_parent = len(h_parent)

    n_list = []
    for delta in [0, 10, -10, 20, 28, 30, 40, -20, 50, 60, 80]:
        nt = n_parent + delta
        if nt % 2 == 0 and 80 <= nt <= 360:
            if nt not in n_list:
                n_list.append(nt)
    n_list.sort(key=lambda x: abs(x - n_parent))

    for n_trials in range(min(5, len(n_list))):
        if time.time() > deadline - 80:
            break
        n = n_list[n_trials]
        target = n / 2.0
        target_ones = int(target)

        if n == n_parent:
            h_interp = h_parent.copy()
        else:
            h_interp = np.interp(np.linspace(0, n_parent - 1, n),
                                  np.arange(n_parent), h_parent)

        h_bin = _to_binary(h_interp, target_ones)
        c5_bin = _c5_of(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n

        # Local search on binary
        h_ls, c5_ls = _local_search_binary(h_bin.copy(), n, min(deadline - 60, t_start + 120))
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_n = n

        # DE on binary space
        if time.time() < deadline - 80:
            h_de, c5_de = _de_binary(h_ls, n, target, pop_size=12,
                                      n_gen=60, deadline=min(deadline - 60, time.time() + 120),
                                      rng=rng)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n
            h_ls2, c5_ls2 = _local_search_binary(h_de, n, deadline - 40)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
                best_n = n

        # Simulated annealing
        if time.time() < deadline - 60:
            h_sa, c5_sa = _simulated_anneal(h_ls2.copy(), n, deadline - 30,
                                             t0=3.0, t1=0.05, rng=rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n

        # Smooth refinement
        if time.time() < deadline - 40:
            h_sm = _smooth_refine(h_ls2.astype(float), n, target, deadline - 20)
            c5_sm = _c5_of(h_sm)
            if c5_sm < best_c5:
                best_c5 = c5_sm
                best_h = h_sm.copy()
                best_n = n
            h_bin2 = _to_binary(h_sm, target_ones)
            c5_bin2 = _c5_of(h_bin2)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()
                best_n = n
            h_ls3, c5_ls3 = _local_search_binary(h_bin2, n, deadline - 15)
            if c5_ls3 < best_c5:
                best_c5 = c5_ls3
                best_h = h_ls3.copy()
                best_n = n

    # Also try greedy construction at various n
    for n_g in [140, 160, 180, 200, 220]:
        if time.time() > deadline - 40:
            break
        if n_g % 2 != 0:
            continue
        h_g = _greedy_construct(n_g, seed + n_g)
        c5_g = _c5_of(h_g)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
            best_n = n_g
        h_gl, c5_gl = _local_search_binary(h_g, n_g, min(deadline - 20, time.time() + 60))
        if c5_gl < best_c5:
            best_c5 = c5_gl
            best_h = h_gl.copy()
            best_n = n_g

    # Final intensive local search on best
    if time.time() < deadline - 10 and best_h is not None:
        h_final, c5_final = _local_search_binary(best_h.copy(), best_n, deadline - 5)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    if best_h is None:
        best_h = _to_binary(h_parent, n_parent // 2)
        best_c5 = _c5_of(best_h)
        best_n = n_parent

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
