            m = n - s
            grad[:m] += wk * dx * omh[s:]
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[r:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[r:]
    return obj, grad

            break

            f_y, grad_y = obj_and_grad_circular(y, eps)
        else:

        # Backtracking line search
        alpha = 1.0
        new_x = x.copy()
        for _ in range(30):
            z = y - alpha * grad_y
            new_x = project_onto_constraint(z, target_sum)
            if use_circular:
                f_new, _ = obj_and_grad_circular(new_x, eps)
            else:
                f_new, _ = obj_and_grad_exact(new_x, eps)
            if f_new <= f_y - 0.5 * alpha * np.dot(grad_y, new_x - y):
                break
            alpha *= 0.5
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = new_x + (t - 1.0) / t_new * (new_x - x)

        x = new_x
        t = t_new
        
        if f_new < best_obj:
            best_obj = f_new
            best_x = x.copy()
    
    return best_x, best_obj

import numpy as np

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def circular_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.rfft(h)
    OH = np.fft.rfft(1.0 - h)
    return np.fft.irfft(np.conj(H) * OH, n=n) * dx

def obj_grad_circular(h, eps):
    n = len(h)
    corr = circular_corr_dx(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    obj = cmax + eps * np.log(sexp)
    w = ex / sexp
    H = np.fft.rfft(h)
    W = np.fft.rfft(w)
    A = np.fft.irfft(2.0 * np.real(W) * H, n=n)
    grad = (2.0 / n) * (1.0 - A)
    return obj, grad

def obj_grad_exact(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    obj = cmax + eps * np.log(sexp)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:

        s = k - (n - 1)

    return obj, grad
