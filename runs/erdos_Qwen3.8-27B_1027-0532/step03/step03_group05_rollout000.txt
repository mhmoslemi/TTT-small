from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_softmax_gradient(h, alpha):

    # Softmax weights

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_softmax_gradient_fast(h, alpha):
    n = len(h)
    corr = compute_correlation(h)
    shifted = alpha * (corr - np.max(corr))
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    W = weights
    first = np.convolve(1.0 - h, W, mode='full')[n - 1:n - 1 + n]
    second = np.convolve(h, W[::-1], mode='full')[n - 1:n - 1 + n]
    grad = first - second
    return grad, corr


def momentum_gradient_descent(h, target, time_limit, alpha_start=5.0, alpha_end=150.0,
                               lr_start=0.01, momentum=0.9, seed=42):

    velocity = np.zeros(n)
    patience = 0
    patience_limit = 3000
    while time.time() - start < time_limit:
        frac = min(1.0, (time.time() - start) / time_limit)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (1.0 + 0.3 * np.sin(frac * np.pi))
        grad, corr = compute_softmax_gradient_fast(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        velocity = momentum * velocity + grad / gn
        vn = np.linalg.norm(velocity)
        if vn < 1e-15:
            break
        h_new = h + lr * velocity / vn
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                lr *= 0.8
                patience = 0

def slsqp_refine(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    if alphas is None:
        alphas = [15, 50, 120, 250]
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(80, remaining * 1.5))

        def make_obj(a):
            def _obj(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                c_max = np.max(corr)
                shifted = np.clip(a * (corr - c_max), -400.0, 0.0)
                return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
            def _grad(x):
                g, _ = compute_softmax_gradient_fast(x, a)
                return g
            return _obj, _grad

        obj, obj_grad = make_obj(alpha)
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                       'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best, jac=obj_grad, method='SLSQP', bounds=bounds,
                           constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    return best, best_score

def fine_polish(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    while time.time() - start < time_limit:
        grad, corr = compute_softmax_gradient_fast(best, alpha=200.0)
        order = np.argsort(-np.abs(grad))
        improved = False
        for idx in order[:40]:
            direction = 1.0 if grad[idx] < 0 else -1.0
            for scale in [0.0005, 0.001, 0.002, 0.004, 0.0008]:
                d = direction * scale
                val = best[idx] + d
                if val < 0 or val > 1:
                    continue
                h_trial = best.copy()
                h_trial[idx] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-15:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        if not improved:
            for _ in range(8):
                i = rng.randint(n)
                d = rng.uniform(-0.003, 0.003)
                val = best[i] + d
                if val < 0 or val > 1:
                    continue
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-15:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        if not improved and rng.random() < 0.05:
            i = rng.randint(n)
            d = rng.uniform(-0.01, 0.01)
            val = best[i] + d
            if 0 <= val <= 1:
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score:
                    best = h_trial
                    best_score = s
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    # Phase 1: Momentum gradient descent with annealed alpha
    if tleft() > 280:
        h1, s1 = momentum_gradient_descent(best_h, target, time_limit=min(280, tleft()),
                                           alpha_start=3.0, alpha_end=80.0,
                                           lr_start=0.012, momentum=0.92, seed=seed)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Adaptive coordinate descent guided by gradient
    if tleft() > 250:
        h2, s2 = adaptive_coordinate_descent(best_h, target, time_limit=min(250, tleft()), seed=seed + 1)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: SLSQP with log-sum-exp smoothing
    if tleft() > 200:
        h3, s3 = slsqp_refine(best_h, target, time_limit=min(200, tleft()), alphas=[10, 40, 100, 200])
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Block perturbation to escape local optima
    if tleft() > 200:
        h4, s4 = block_perturbation(best_h, target, time_limit=min(200, tleft()), seed=seed + 2)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Second gradient round with higher alpha
    if tleft() > 180:
        h5, s5 = momentum_gradient_descent(best_h, target, time_limit=min(180, tleft()),
                                           alpha_start=30.0, alpha_end=300.0,
                                           lr_start=0.004, momentum=0.88, seed=seed + 3)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Fine polishing
    if tleft() > 130:
        h6, s6 = fine_polish(best_h, target, time_limit=min(130, tleft()), seed=seed + 4)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final SLSQP pass
    if tleft() > 80:
        h7, s7 = slsqp_refine(best_h, target, time_limit=min(80, tleft()), alphas=[50, 150, 400])
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final micro-adjustments
    if tleft() > 40:
        h8, s8 = fine_polish(best_h, target, time_limit=max(20, tleft() - 5), seed=seed + 5)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
