Wait, but I can do better. Instead of recomputing the full correlation each time, I can compute the change:

When I change h[i] from old to new, the change in corr[m] is:

```python
def coordinate_descent(h, n, target, max_sweeps, rng, deadline):
    """Coordinate descent with exact 1D minimization."""
    h = h.copy()
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            old_val = h[i]
            
            # Compute the coefficient A[m] for each correlation index m
            # corr[m] = sum_j h[j] * (1-h[j+m-n+1])
            # Change in corr[m] when h[i] changes by Δ:
            # Δcorr[m] = Δ * (1-h[i+m-n+1]) [if valid] - Δ * h[i-m+n-1] [if valid]
            
            A = np.zeros(2*n-1)
            
            # Type A: h[i] as first factor, j=i
            # Valid m: 0 <= i+m-n+1 < n → n-1-i <= m <= n-1+i
            m_start_a = max(0, n-1-i)
            m_end_a = min(2*n-2, n-1+i)
            if m_start_a <= m_end_a:
                idx_a = np.arange(m_start_a, m_end_a+1)
                h_idx = i + idx_a - (n-1)  # = i+m-n+1
                A[idx_a] += (1.0 - h[h_idx])
            
            # Type B: h[i] as second factor (inside 1-h[...]), j = i-m+n-1
            # Valid m: 0 <= i-m+n-1 < n → 1+i <= m <= i+n-1... 
            # i-m+n-1 >= 0 → m <= i+n-1
            # i-m+n-1 < n → m > i-1 → m >= i (for integer m)
            m_start_b = max(0, i)
            m_end_b = min(2*n-2, i+n-1)
            if m_start_b <= m_end_b:
                idx_b = np.arange(m_start_b, m_end_b+1)
                h_idx = i - idx_b + (n-1)  # = i-m+n-1
                A[idx_b] -= h[h_idx]
            
            # Now corr[m] + A[m] * (t - old_val) for the new value t
            # We want to minimize max_m (corr[m] + A[m] * (t - old_val)) over t in [0,1]
            # Let corr_current = compute_corr(h) with h[i] = old_val
            # Then new_corr[m] = corr_current[m] + A[m] * (t - old_val)
            
            # For efficiency, compute corr once per sweep (or maintain it)
            # Actually, let me just compute it here for correctness
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # Find t that minimizes max_m (corr[m] + A[m]*(t - old_val))
            # = max_m (corr[m] - A[m]*old_val + A[m]*t)
            # Let f_m(t) = A[m]*t + (corr[m] - A[m]*old_val)
            # Minimize max_m f_m(t) over t in [0,1]
            
            B = corr - A * old_val  # constant part
            # Minimize max_m (A[m]*t + B[m]) over t in [0,1]
            
            # This is a convex piecewise-linear function of t
            # Minimum is at a breakpoint or boundary
            # Quick approach: try t=0, t=1, and a grid
            best_t = old_val
            best_obj = np.max(A * old_val + B)
            
            # Grid search
            grid = np.linspace(0, 1, 31)
            for t in grid:
                obj = np.max(A * t + B)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            h[i] = best_t
            # Don't project yet; do it at the end of the sweep
        
        # Project to maintain sum constraint
        h = project_feasible(h, target)
    
    return h
```

import numpy as np
from scipy.optimize import minimize
import time
import math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_optimize(h, n, target, alpha, maxiter=5000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: lse_obj(x, alpha), h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h

def coordinate_descent(h, n, target, max_sweeps, deadline, grid_size=31):
    """Coordinate descent with incremental correlation update."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        for i in order:
            old_val = h[i]
            
            # Compute A[m]: coefficient of h[i] in corr[m]
            A = np.zeros(2*n-1)
            
            # Type A: j = i (h[i] is the "h" factor)
            # corr[m] includes term h[i] * (1-h[i+m-n+1]) when valid
            m_lo_a = max(0, n-1-i)
            m_hi_a = min(2*n-2, n-1+i)
            if m_lo_a <= m_hi_a:
                idx = np.arange(m_lo_a, m_hi_a+1)
                h_idx = i + idx - (n-1)
                A[idx] += (1.0 - h[h_idx])
            
            # Type B: j+m-n+1 = i, so j = i-m+n-1 (h[i] is inside "1-h[...]" factor)
            # corr[m] includes term h[j] * (1-h[i]) when valid
            # = h[i-m+n-1] * (1-h[i])
            # The change when h[i] → h[i]+Δ: h[i-m+n-1] * (-Δ)
            m_lo_b = max(0, i)  # i-m+n-1 < n → m > i-n → m >= i (since m integer, i-n < 0)
            # Wait: i-m+n-1 >= 0 → m <= i+n-1
            # i-m+n-1 < n → m > i-1 → m >= i (since m is integer)
            m_lo_b = max(0, i)
            m_hi_b = min(2*n-2, i+n-1)
            if m_lo_b <= m_hi_b:
                idx = np.arange(m_lo_b, m_hi_b+1)
                h_idx = i - idx + (n-1)
                A[idx] -= h[h_idx]
            
            # new_corr[m] = corr[m] + A[m] * (t - old_val)
            # = (corr[m] - A[m]*old_val) + A[m]*t
            B = corr - A * old_val
            
            # Minimize max_m (A[m]*t + B[m]) over t in [0,1]
            grid = np.linspace(0.0, 1.0, grid_size)
            objs = A[None, :] * grid[:, None] + B[None, :]
            max_objs = np.max(objs, axis=1)
            best_idx = np.argmin(max_objs)
            best_t = grid[best_idx]
            
            h[i] = best_t
            # Update corr incrementally
            corr = corr + A * (best_t - old_val)
        
        # After each sweep, verify corr and project
        corr_check = np.correlate(h, 1.0 - h, mode='full')
        # Check if our incremental update is accurate
        if np.max(np.abs(corr - corr_check)) > 1e-8:
            corr = corr_check
        h = project_feasible(h, target)
        # Re-compute corr after projection
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h

                A[idx] -= h[i - idx + (n-1)]
            
            # Minimize max_m (corr[m] + A[m]*(t - old_val)) over t in [0,1]
            # = max_m ((corr[m] - A[m]*old_val) + A[m]*t)
            B = corr - A * old_val
            
            # Grid search
            grid = np.linspace(0.0, 1.0, grid_size)
            # Vectorized: compute max_m for each grid point
            # shape: (grid_size, 2n-1)
            vals = grid[:, None] * A[None, :] + B[None, :]
            max_vals = np.max(vals, axis=1)
            best_t = grid[np.argmin(max_vals)]
            
            if best_t != old_val:
                h[i] = best_t
                corr += A * (best_t - old_val)
                improved = True
        
        # Project to maintain constraints
        h = project_feasible(h, target)
        # Recompute corr (projection may have changed things)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        sc = float(np.max(corr)) * (2.0/n)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time
import math


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def _lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _cons_eq(x, target):
    return np.sum(x) - target


def _cons_jac(x, n):
    return np.ones(n)


def slsqp_optimize(h, n, target, alpha, maxiter=3000, ftol=1e-16):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _lse_obj(x, alpha)

    def cfun(x):
        return _cons_eq(x, target)

    def cjac(x):
        return _cons_jac(x, n)

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h


def coordinate_descent(h, n, target, n_sweeps, deadline, grid_size=101, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    no_improve = 0

    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        order = rng.permutation(n)
        improved = False
        for i in order:
            old_val = h[i]
            A = np.zeros(2 * n - 1)
            m_lo = max(0, n - 1 - i)
            m_hi = min(2 * n - 2, n - 1 + i)
            if m_lo <= m_hi:
                idx = np.arange(m_lo, m_hi + 1)
                A[idx] += (1.0 - h[i + idx - (n - 1)])
            m_lo_b = max(0, i)
            m_hi_b = min(2 * n - 2, i + n - 1)
            if m_lo_b <= m_hi_b:
                idx = np.arange(m_lo_b, m_hi_b + 1)
                A[idx] -= h[i - idx + (n - 1)]
            B = corr - A * old_val
            vals = grid[:, None] * A[None, :] + B[None, :]
            max_vals = np.max(vals, axis=1)
            best_t = grid[np.argmin(max_vals)]
            if abs(best_t - old_val) > 1e-10:
                h[i] = best_t
                corr += A * (best_t - old_val)
                improved = True
        h = project_feasible(h, target)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        sc = float(np.max(corr)) * (2.0 / n)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
    return best_h, best_score


def binary_local_search(h, n, target, max_iter, deadline, rng):
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    int_target = int(round(target))
    h_bin[sorted_idx[:int_target]] = 1.0
    best_score = compute_c5(h_bin)
    best_h = h_bin.copy()
    T0 = best_score * 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i] = 0.0
        h_bin[j] = 1.0
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin.copy()
        else:
            T = T0 * max(0.01, 1.0 - it / max_iter)
            if rng.rand() < math.exp(-(sc - best_score) / T):
                pass
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
    return best_h, best_score

    # Phase 3: SLSQP after coordinate descent
    if tleft() > 80:
        for alpha in [500, 2000, 5000]:
            if tleft() < 40:

            h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=3000)

            if sc < best_score:

    if tleft() > 100:

        sc_proj = compute_c5(project_feasible(h_bin, target))

            best_h = project_feasible(h_bin, target)

            if sc_cd2 < best_score:

    # Phase 5: More coordinate descent with perturbations

    sc_cur = best_score

        h_pert[block_start:block_start + block_len] += noise
        h_pert = project_feasible(h_pert, target)

    # Phase 6: Final SLSQP polish

            if tleft() < 30:

            h_new = slsqp_optimize(best_h, n, target, alpha, maxiter=5000, ftol=1e-17)
            sc = compute_c5(h_new)

                best_score = sc
                best_h = h_new

    # Phase 7: Final coordinate descent

            best_score = sc_final
            best_h = h_cd_final

    # Final projection and verification
    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
