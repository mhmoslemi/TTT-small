    exp_vals = np.exp(shifted)
    log_sum = np.log(np.sum(exp_vals))
    val = c_max + log_sum / alpha

    softmax = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    n_half = n - 1
    for k in range(-(n - 1), n):
        w = softmax[k + n_half]
        if w < 1e-15:
            continue
        j_min = max(0, k)
        j_max = min(n, n + k)
        j = np.arange(j_min, j_max)
        n_idx = j - k
        grad[j] += w * g[n_idx] * dx
        grad[n_idx] -= w * h[j] * dx
    return val, grad


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj_and_grad, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol, 'iprint': 0,
                                    'colin': 1e-8, 'acc': 1e-8})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) <= compute_c5(h_cur) + 1e-12:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def lbfgsb_refine(h_start, n, target, alpha, maxiter, pen_lambda):
    dx = 2.0 / n

    def obj_wrapper(h):
        n_g = len(h)
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full') * dx
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_vals = np.exp(shifted)
        val = c_max + np.log(np.sum(exp_vals)) / alpha
        val += pen_lambda * (np.sum(h) - target) ** 2
        return val

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_wrapper, h_start, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                               'gtol': 1e-10, 'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return h_start


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_polish(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        grad = np.zeros(n)
        k = k_star
        j_min = max(0, k)
        j_max = min(n, n + k)
        j = np.arange(j_min, j_max)
        n_idx = j - k
        grad[j] += (1.0 - h_cur[n_idx]) * dx
        grad[n_idx] -= h_cur[j] * dx

        for lr in [0.02, 0.05, 0.1, 0.2, 0.3]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 20.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    # Phase 1: Active-set LP from initial point (n=400)
    if tleft() > 200:
        lp_time = min(200, tleft() * 0.3)
        h_lp, c5_lp = active_set_lp(best_h, n, target, time_limit=lp_time,
                                     K_frac=0.2, damping=0.7, reg=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: More LP with different params
    if tleft() > 200:
        lp_time = min(150, tleft() * 0.2)
        h_lp2, c5_lp2 = active_set_lp(best_h, n, target, time_limit=lp_time,
                                       K_frac=0.1, damping=0.8, reg=1e-6)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    # Phase 3: SLSQP with aggressive alpha schedule
    if tleft() > 250:
        h_slsqp = slsqp_refine(best_h, n, target,
                                [5, 15, 50, 150, 500, 1500, 5000],
                                2000, 1e-14)
        sc = compute_c5(h_slsqp)
        if sc < best_score:
            best_score = sc
            best_h = h_slsqp

    # Phase 4: L-BFGS-B polish
    if tleft() > 150:
        h_lbfgs = lbfgsb_refine(best_h, n, target, alpha=5000, maxiter=500,
                                pen_lambda=50.0)
        sc = compute_c5(h_lbfgs)
        if sc < best_score:
            best_score = sc
            best_h = h_lbfgs

    # Phase 5: Subgradient polish
    if tleft() > 80:
        h_sg, c5_sg = subgradient_polish(best_h, n, target,
                                          max_steps=3000, time_limit=min(60, tleft() * 0.5))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 6: SLSQP from perturbed best
    for noise_scale in [0.05, 0.15]:
        if tleft() < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        h_lp3, c5_lp3 = active_set_lp(h_pert, n, target,
                                       time_limit=min(60, tleft() * 0.2),
                                       K_frac=0.15, damping=0.6, reg=1e-5)
        if c5_lp3 < best_score:
            best_score = c5_lp3
            best_h = h_lp3

    # Final SLSQP from best
    if tleft() > 80:
        h_final_try = slsqp_refine(best_h, n, target,
                                    [200, 1000, 5000, 20000],
                                    3000, 1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 7: Try higher resolution
    if tleft() > 300:
        for n_try in [600, 800]:
            if tleft() < 200:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(180, tleft() * 0.3)
            h_lp_hi, c5_hi = active_set_lp(h_interp, n_try, target_try,
                                            time_limit=lp_time,
                                            K_frac=0.15, damping=0.7, reg=1e-5)
            if tleft() > 100:
                c5_hi_tmp = compute_c5(h_lp_hi)
                h_lp_hi = slsqp_refine(h_lp_hi, n_try, target_try,
                                        [20, 100, 500, 2000, 10000],
                                        2000, 1e-14)
                c5_hi = compute_c5(h_lp_hi)
            if c5_hi < best_score:
                best_score = c5_hi
                best_h = h_lp_hi
                n = n_try
                target = target_try

    # Final polish
    if tleft() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pol = slsqp_refine(best_h, n_use, target_use,
                              [500, 2000, 10000], 2000, 1e-15)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol
        h_sg2, c5_sg2 = subgradient_polish(best_h, n_use, target_use,
                                            max_steps=2000, time_limit=min(30, tleft() * 0.5))
        if c5_sg2 < best_score:
            best_score = c5_sg2
            best_h = h_sg2

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
