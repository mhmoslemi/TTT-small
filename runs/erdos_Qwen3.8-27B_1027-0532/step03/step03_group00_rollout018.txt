        h_bd, c5_bd = block_lbfbsb(best_h, n, target, deadline_p5, block_size=bs, n_rounds=3)
        if c5_bd < best_c5:
            best_c5 = c5_bd
            best_h = h_bd

    # Phase 6: Multi-restart with perturbations
    deadline_p6 = start_time + (budget_s - 30.0) * 0.90
    for _ in range(6):
        if tleft() < 80:
            break
        h_pert = best_h.copy()
        n_pert = rng.randint(n // 4, n // 2)
        idx = rng.choice(n, size=n_pert, replace=False)
        h_pert[idx] += (rng.randn(n_pert) * 0.15)
        h_pert = project_feasible(h_pert, target)
        h_p_rw, c5_p_rw = reweight_optimize(h_pert, n, target,
                                             start_time + 30, n_steps=50,
                                             beta_start=20.0, beta_end=200.0, pen=100.0)
        if c5_p_rw < best_c5:
            best_c5 = c5_p_rw
            best_h = h_p_rw
        h_p_pol, c5_p_pol = lse_polish(h_p_rw, n, target,
                                        start_time + 15, alphas=(100, 500, 1500))
        if c5_p_pol < best_c5:
            best_c5 = c5_p_pol
            best_h = h_p_pol

    # Phase 7: Final aggressive polish
    deadline_p7 = start_time + budget_s - 35.0
    if tleft() > 40:
        h_final, c5_final = lse_polish(best_h, n, target, deadline_p7,
                                        alphas=(200, 800, 3000, 8000))
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Phase 8: Final greedy
    if tleft() > 20:
        h_gr2, c5_gr2 = greedy_coordinate(best_h, n, target,
                                           start_time + budget_s - 32, n_passes=2)
        if c5_gr2 < best_c5:
            best_c5 = c5_gr2
            best_h = h_gr2

    best_h = project_feasible(best_h, target)
    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
