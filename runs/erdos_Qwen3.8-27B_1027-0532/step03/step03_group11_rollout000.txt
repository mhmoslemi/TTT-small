import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_detailed(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def gradient_descent_projected(h_init, eps, max_iter, time_limit, start_time, lr=0.01):
    """Projected gradient descent with decreasing step size."""
    n = len(h_init)
    target_sum = n / 2.0

    h = project_onto_constraint(h, target_sum)
    current = compute_c5(h)
    best_h = h.copy()
    best_val = current

            break
        grad = smooth_gradient(h, eps)
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = project_onto_constraint(h - step * grad, target_sum)
            val_new = smooth_objective(h_new, eps)
            if val_new < current - 0.01 * step * np.sum(grad * (h_new - h)):

            step *= 0.5
        h = h_new
        current = val_new
        true_val = compute_c5(h)
        if true_val < best_val:
            best_val = true_val
            best_h = h.copy()
    
    return best_h, best_val

def binary_local_search(h_bin, max_rounds=1000):
    """Fast binary local search: try all swaps, apply best one."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    current = compute_c5(h)
    
    for _ in range(max_rounds):

def precompute_influence(h, n):
    """Precompute influence_plus[p] for all p.
    influence_plus[p][j] = change in corr[j] when h[p] goes from 0 to 1.
    corr[j] corresponds to shift s = j - (n-1).

    # shift s = j - (n-1), s ranges from -(n-1) to (n-1)
    influence = np.zeros((n, 2*n-1))

            term2 = 0.0
            if 0 <= p - s < n:

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun_bn(h, target):
    return float(np.sum(h) - target)

def constraint_jac_bn(h, target):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, start_time, time_limit):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, t=target_sum: float(np.sum(x) - t),
             'jac': lambda x: np.ones(len(x))}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def precompute_influence(h, n):
    influence = np.zeros((n, 2 * n - 1))
    s_vals = np.arange(-(n - 1), n)
    for p in range(n):
        ps = p + s_vals
        valid1 = (ps >= 0) & (ps < n)
        term1 = np.zeros(2 * n - 1)
        term1[valid1] = 1.0 - h[ps[valid1]]
        pms = p - s_vals
        valid2 = (pms >= 0) & (pms < n)
        term2 = np.zeros(2 * n - 1)
        term2[valid2] = h[pms[valid2]]
        influence[p] = term1 - term2
    return influence

def binary_local_search_influence(h_bin, max_rounds=500, start_time=None, time_limit=None):
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = np.max(corr)
    dx = 2.0 / n
    for _ in range(max_rounds):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        influence = precompute_influence(h, n)
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improvement = 1e-10
        best_oi = -1
        best_zi = -1
        for oi in ones:
            for zi in zeros:
                delta = influence[zi] - influence[oi]
                new_max = np.max(corr + delta)
                if new_max < current_max - 1e-10:
                    improvement = current_max - new_max
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_oi = oi
                        best_zi = zi
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        current_max = np.max(corr)
    return h, float(dx * current_max)

def binary_sa_fast(h_init, n_steps, t_start, t_end, start_time, time_limit):
    n = len(h_init)
    h = h_init.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    dx = 2.0 / n
    rng = np.random.default_rng(12345)
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        new_max = float(np.max(new_corr))
        delta = new_max - current_max
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / (t * n / 2.0))):
            current_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, float(dx * best_max)

def make_binary_from_continuous(h):
    n = len(h)
    order = np.argsort(np.abs(h - 0.5))
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result
