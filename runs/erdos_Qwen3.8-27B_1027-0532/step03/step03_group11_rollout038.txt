2. When we propose a swap, compute delta_c[k] for all k
3. Update c += delta_c
4. Check if max(c) decreased

    """Compute the change in correlation when h[oi] goes 1->0 and h[zi] goes 0->1."""
    # delta_c[k] = h[oi+k] + h[oi-k] - h[zi+k] - h[zi-k] (out of range = 0)
    # k ranges from -(n-1) to n-1, so index in array is k + (n-1)
    delta = np.zeros(2*n-1)
    
    # h[oi+k] for k from -(n-1) to (n-1)
    # oi+k is in [0, n-1] when k is in [-oi, n-1-oi]

    if lo <= hi:
        k_vals = np.arange(lo, hi+1)
        delta[k_vals + (n-1)] += h[oi + k_vals]
    
    # h[oi-k] for k from -(n-1) to (n-1)
    # oi-k is in [0, n-1] when k is in [oi-(n-1), oi]

from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)

                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def compute_delta(h, oi, zi, n):
    """Fast computation of correlation delta for swap (oi: 1->0, zi: 0->1)."""
    offset = n - 1
    size = 2 * n - 1
    delta = np.zeros(size)
    
    # Term 1: h_new[oi+k] for k where 0 <= oi+k <= n-1
    k_min = max(-(n-1), -oi)
    k_max = min(n-1, n-1-oi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max+1)
        idx = oi + ks
        vals = h[idx].astype(float)
        # Correct for new values at swapped positions
        vals[idx == oi] = 0.0   # k = 0
        vals[idx == zi] = 1.0   # k = zi - oi
        delta[ks + offset] += vals
    
    # Term 2: -h_old[zi+k] for k where 0 <= zi+k <= n-1
    k_min = max(-(n-1), -zi)
    k_max = min(n-1, n-1-zi)
    if k_min <= k_max:

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def compute_delta(h, oi, zi, n):
    """Compute correlation delta for swap h[oi]:1->0, h[zi]:0->1."""
    offset = n - 1
    size = 2 * n - 1
    delta = np.zeros(size)
    
    # Term 1: h_new[oi+k]
    k_min = max(-(n-1), -oi)
    k_max = min(n-1, n-1-oi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max+1)
        idx = oi + ks
        vals = h[idx].astype(np.float64)
        mask_oi = (ks == 0)
        if np.any(mask_oi):
            vals[mask_oi] = 0.0
        mask_zi = (idx == zi)
        if np.any(mask_zi):
            vals[mask_zi] = 1.0
        delta[ks + offset] += vals
    
    # Term 2: -h[zi+k]

    k_min = max(-(n-1), oi-(n-1))
    k_max = min(n-1, oi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max+1)
        nvals = oi - ks
        valid = (nvals != oi) & (nvals != zi)
        if np.any(valid):

    k_min = max(-(n-1), zi-(n-1))
    k_max = min(n-1, zi)

        valid = (nvals != oi) & (nvals != zi)
        if np.any(valid):
            delta[ks[valid] + offset] += (1.0 - h[nvals[valid]])
    
    return delta

    Simulated annealing on binary solutions with fast delta computation.

            h_p[idx] = 1.0 - h_p[idx]

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def compute_delta(h, oi, zi, n):
    offset = n - 1
    size = 2 * n - 1
    delta = np.zeros(size)
    k_min = max(-(n - 1), -oi)
    k_max = min(n - 1, n - 1 - oi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max + 1)
        idx = oi + ks
        vals = h[idx].astype(np.float64)
        mask_oi = (ks == 0)
        if np.any(mask_oi):
            vals[mask_oi] = 0.0
        mask_zi = (idx == zi)
        if np.any(mask_zi):
            vals[mask_zi] = 1.0
        delta[ks + offset] += vals
    k_min = max(-(n - 1), -zi)
    k_max = min(n - 1, n - 1 - zi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max + 1)
        delta[ks + offset] -= h[zi + ks]
    k_min = max(-(n - 1), oi - (n - 1))
    k_max = min(n - 1, oi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max + 1)
        nvals = oi - ks
        valid = (nvals != oi) & (nvals != zi)
        if np.any(valid):
            delta[ks[valid] + offset] -= (1.0 - h[nvals[valid]])
    k_min = max(-(n - 1), zi - (n - 1))
    k_max = min(n - 1, zi)
    if k_min <= k_max:
        ks = np.arange(k_min, k_max + 1)
        nvals = zi - ks
        valid = (nvals != oi) & (nvals != zi)
        if np.any(valid):
            delta[ks[valid] + offset] += (1.0 - h[nvals[valid]])
    return delta

def best_improvement_ls(h, corr, max_rounds, time_limit, start_time):
    n = len(h)
    h = h.astype(float).copy()
    corr = corr.astype(float).copy()
    current_c5 = float(np.max(corr))
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        best_new_corr = None
        for oi in ones:
            for zi in zeros:
                d = compute_delta(h, oi, zi, n)
                new_corr = corr + d
                new_max = float(np.max(new_corr))
                delta = new_max - current_c5
                if delta < best_delta - 1e-15:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                    best_new_corr = new_corr
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = best_new_corr
        current_c5 += best_delta
    return h, corr, current_c5
