from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def to_binary(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    cur = h_bin.sum()
    if cur == target_sum:
        return h_bin
    if cur > target_sum:
        excess = cur - target_sum
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])  # least confident ones first
        h_bin[ones[order[:int(excess)]]] = 0
    else:
        deficit = target_sum - cur
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])  # most confident zeros first
        h_bin[zeros[order[:int(deficit)]]] = 1
    return h_bin

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(smooth_max_obj, h0, args=(eps,),
                      jac=smooth_max_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                                    'jac': lambda x: np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})

def thue_morse_binary(n):

        else:
            deficit = target - ones_count
            zeros_idx = np.where(tm == 0)[0]
            tm[zeros_idx[:deficit]] = 1
    return tm

from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def to_binary(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    cur = int(h_bin.sum())
    ts = int(target_sum)
    if cur == ts:
        return h_bin
    if cur > ts:
        excess = cur - ts
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        h_bin[ones[order[:excess]]] = 0
    else:
        deficit = ts - cur
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros])
        h_bin[zeros[order[:deficit]]] = 1
    return h_bin

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          method='SLSQP', jac=smooth_max_grad,
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': sum_constraint,
                                        'jac': sum_jacobian}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)

                            h[i] = 0.0
                            h[b] = 1.0
                            c5_new = compute_c5(h)
                            if c5_new < best_swap_c5:
                                best_swap_c5 = c5_new
                                best_swap = (i, b)
                            h[i] = 1.0
                            h[b] = 0.0
                    op = ones_list[ones_list != i]
                    if len(op) > 0:
                        a = int(op[rng.integers(len(op))])
                        if a != i - r:
                            h[i - r] = 1.0
                            h[a] = 0.0
                            c5_new = compute_c5(h)
                            if c5_new < best_swap_c5:
                                best_swap_c5 = c5_new
                                best_swap = (i - r, a)
                            h[i - r] = 0.0
                            h[a] = 1.0

        for _ in range(n_random):
            a = int(ones_list[rng.integers(len(ones_list))])
            b = int(zeros_list[rng.integers(len(zeros_list))])
            if a == b:
                continue
            h[a] = 0.0
            h[b] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_swap_c5:
                best_swap_c5 = c5_new
                best_swap = (a, b)
            h[a] = 1.0
            h[b] = 0.0

        if best_swap is not None:
            a, b = best_swap
            h[a] = 0.0
            h[b] = 1.0
            best_c5 = best_swap_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_c5

def thue_morse_binary(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = float(bin(i).count('1') % 2)
    target = n // 2
    oc = int(tm.sum())
    if oc != target:
        if oc > target:
            idx = np.where(tm == 1)[0]
            tm[idx[:oc - target]] = 0.0
        else:
            idx = np.where(tm == 0)[0]
            tm[idx[:target - oc]] = 1.0
    return tm

def golden_binary(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = 1.0 if (i * phi) % 1.0 > 0.5 else 0.0
    target = n // 2
    oc = int(seq.sum())
    if oc != target:
        if oc > target:
            idx = np.where(seq == 1)[0]
            seq[idx[:oc - target]] = 0.0
        else:
            idx = np.where(seq == 0)[0]
            seq[idx[:target - oc]] = 1.0
    return seq

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(180), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase A: Binary local search from multiple starts (300s)
    bin_starts = []
    bin_starts.append(to_binary(h0, target_sum))
    bin_starts.append(thue_morse_binary(n0))
    bin_starts.append(golden_binary(n0))
    for _ in range(4):
        rb = (rng.random(n0) < 0.5).astype(float)
        rb = np.sort(rb)[::-1].astype(float)
        rb[:n0 // 2] = 1.0
        rb[n0 // 2:] = 0.0
        rb = rng.permutation(rb)
        bin_starts.append(rb)

    bin_time_each = min(60.0, remaining() / (len(bin_starts) + 1))
    for bs in bin_starts:
        if remaining() < 30:
            break
        h_b, c5_b = binary_hill_climb(bs, min(bin_time_each, remaining() - 20), start_time, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase B: SLSQP refinement from best binary (coarse to fine)
    if remaining() > 80:
        eps_list = [0.01, 0.003, 0.001, 0.0003]
        for eps in eps_list:
            if remaining() < 30:
                break
            h_c, c5_c = slsqp_optimize(best_h, eps, 300)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Phase C: Convert back to binary and do more local search
    if remaining() > 60:
        h_bin_best = to_binary(best_h, len(best_h) / 2.0)
        h_b, c5_b = binary_hill_climb(h_bin_best, min(remaining() - 20, 120), start_time, rng, n_targeted=30, n_random=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase D: Fine SLSQP
    if remaining() > 50:
        eps_fine = [0.0002, 0.0001, 0.00005]
        for eps in eps_fine:
            if remaining() < 25:
                break
            h_c, c5_c = slsqp_optimize(best_h, eps, 400)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c.copy()

    # Phase E: Resolution increase attempt
    if remaining() > 200:
        for n_new in [n0 + 20, n0 + 40]:
            if remaining() < 120:
                break
            old_t = np.linspace(0.0, 1.0, len(best_h))
            new_t = np.linspace(0.0, 1.0, n_new)
            h_up = np.interp(new_t, old_t, best_h)
            h_up = project_onto_constraint(h_up, n_new / 2.0)
            h_up = h_up + 0.005 * rng.standard_normal(n_new)
            h_up = project_onto_constraint(h_up, n_new / 2.0)
            h_u, c5_u = slsqp_optimize(h_up, 0.005, 200)
            if remaining() > 60:
                h_u, c5_u2 = slsqp_optimize(h_u, 0.001, 200)
                if c5_u2 < c5_u:
                    c5_u = c5_u2
                    h_u = h_u  # already updated
            if c5_u < best_c5:
                best_c5 = c5_u
                best_h = h_u.copy()

    # Phase F: Final binary polish + SLSQP
    if remaining() > 40:
        h_bin_final = to_binary(best_h, len(best_h) / 2.0)
        h_b, c5_b = binary_hill_climb(h_bin_final, min(remaining() - 20, 60), start_time, rng, n_targeted=30, n_random=15)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    if remaining() > 30:
        h_c, c5_c = slsqp_optimize(best_h, 0.0001, 300)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()

    # Final projection
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
