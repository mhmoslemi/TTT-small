def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        # Distribute proportionally to available space
        space = 1.0 - h[idx]
        total_space = space.sum()
        if total_space > 0:
            h[idx] += diff * space / total_space
        else:
            h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        space = h[idx]
        total_space = space.sum()
        if total_space > 0:
            h[idx] += diff * space / total_space
        else:
            h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def lse_obj_and_grad(h, alpha, topk=None):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk is not None and topk < len(corr):
        idx = np.argpartition(corr, -topk)[-topk:]
        corr_top = corr[idx]
    else:
        idx = np.arange(len(corr))
        corr_top = corr
    c_max = np.max(corr_top)
    shifted = np.clip(alpha * (corr_top - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = c_max + np.log(total) / alpha
    
    # Gradient
    weights = exp_vals / total
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = idx[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

    except Exception:

        f, _ = lse_obj_grad(x, alpha, topk)
        return f

        _, g = lse_obj_grad(x, alpha, topk)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,

    bounds = [(0.0, 1.0)] * n
    try:

    # Phase 3: Multi-resolution exploration
    res_list = [350, 450, 550, 650]
    for n_try in res_list:
        if time_left() < 120:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_try)
        bounds_try = [(0.0, 1.0)] * n_try

        for alpha in [50, 200, 600]:
            if time_left() < 50:
                break
            h_start = h_interp + rng.randn(n_try) * 0.006
            h_start = project_feasible(h_start, target_try)
            h_new = lbfgs_step(h_start, target_try, alpha, 30, penalty, 250, bounds_try, 1e-15)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new
                if n_try != n:
                    x_old2 = np.linspace(0, 1, n_try)
                    x_new2 = np.linspace(0, 1, n)
                    best_h = np.interp(x_new2, x_old2, h_new)
                    best_h = project_feasible(best_h, target)

    # Phase 4: Structured perturbations + L-BFGS refinement
    if time_left() > 80:
        freqs = [1, 2, 3, 4, 5, 6, 8, 10, 12, 15]
        amps = [0.02, 0.01, 0.005, 0.002]
        for _ in range(3):
            if time_left() < 40:
                break
            best_h, best_score = structured_perturb(best_h, n, rng, freqs, amps)
            if time_left() < 30:
                break
            h_new = lbfgs_step(best_h, target, 500, 20, penalty, 200, bounds, 1e-16)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 5: High-alpha L-BFGS polish
    if time_left() > 50:
        for alpha in [1000, 2000, 5000]:
            if time_left() < 30:
                break
            h_start = best_h + rng.randn(n) * 0.001
            h_start = project_feasible(h_start, target)
            h_new = lbfgs_step(h_start, target, alpha, 15, penalty, 200, bounds, 1e-17)
            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new

    # Phase 6: Coordinate-wise local search
    if time_left() > 30:
        for round_id in range(3):
            if time_left() < 15:
                break
            improved = False
            for coord in range(n):
                if time_left() < 10:
                    break
                orig = best_h[coord]
                for delta in [0.005, -0.005, 0.002, -0.002, 0.001, -0.001]:
                    h_try = best_h.copy()
                    h_try[coord] = np.clip(orig + delta, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_score - 1e-8:
                        best_score = sc
                        best_h = h_try
                        improved = True
                        break

    # Phase 7: Final SLSQP polish
    if time_left() > 40:
        h_new = slsqp_step(best_h, target, 1000, 15, 1000, 1e-15)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 8: Final L-BFGS
    if time_left() > 20:
        h_new = lbfgs_step(best_h, target, 2000, 10, penalty, 150, bounds, 1e-17)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, n
