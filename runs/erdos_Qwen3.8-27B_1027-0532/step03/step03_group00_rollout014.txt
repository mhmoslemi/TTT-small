import numpy as np
from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_mu = 0.0
_g_block_h = None
_g_block_s = 0
_g_block_e = 0

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, alpha):
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -700.0, 0.0)
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for j in range(len(corr)):
        wt = w[j]
        if wt < 1e-12:
            continue
        k = j - (n - 1)
        if k >= 0:
            if k == 0:
                grad += wt * (1.0 - h) - wt * h
            else:
                grad[k:] += wt * (1.0 - h[:n-k])
                grad[:n-k] -= wt * h[k:]
        else:
            s = -k
            grad[:n-s] += wt * (1.0 - h[s:])
            grad[s:] -= wt * h[:n-s]
    return grad

def _pgrad(x):
    return smooth_max_grad(x, _g_alpha) + 2.0 * _g_mu

def _sobj(x):
    return smooth_max_val(x, _g_alpha)

def _sjac(x):
    return smooth_max_grad(x, _g_alpha)

def _ceq(x):
    return np.sum(x) - _g_target

def _cjac(x):
    return np.ones(_g_n)

def _bobj(x):

    h_temp = project_feasible(h_temp, _g_target)

def do_lbfgs(h, alphas, mus, maxiter, deadline):
    global _g_n, _g_target, _g_alpha, _g_mu
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * _g_n
    for a in alphas:
        for mu in mus:
            if time.time() > deadline:

            _g_alpha = a
            _g_mu = mu
            try:

                              bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10})
                h_new = project_feasible(res.x, _g_target)
                sc = compute_c5(h_new)
                if sc < cur:
                    h_cur = h_new
                    cur = sc
            except Exception:
                pass
    return h_cur, cur

def do_slsqp(h, alphas, maxiter, ftol, deadline):
    global _g_n, _g_target, _g_alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)

    for a in alphas:

        _g_alpha = a
        try:

            h_new = project_feasible(res.x, _g_target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur

    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
