import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def smooth_objective_and_grad(h, alpha, target_sum, n):
    """Compute smooth surrogate and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    L = (float(max_corr) + np.log(sum_exp) / alpha) * dx
    
    # Gradient
    w = exp_shifted / sum_exp  # softmax weights, length 2n-1
    
    grad = np.zeros(n)
    for j in range(n):
        m_k = np.arange(2*n-1) - n + 1  # shifts
        # First term: (1-h[j+m]) for valid j+m
        valid1 = (j + m_k >= 0) & (j + m_k < n)
        term1 = np.sum(w[valid1] * (1.0 - h[j + m_k[valid1]]))
        # Second term: h[j-m] for valid j-m
        valid2 = (j - m_k >= 0) & (j - m_k < n)
        term2 = np.sum(w[valid2] * h[j - m_k[valid2]])
        grad[j] = dx * (term1 - term2)
    
    return L, grad

        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def soft_max_obj(h, alpha, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    s = np.exp(alpha * (corr - max_c))
    val = (max_c + np.log(s.sum()) / alpha) * dx
    return val

def soft_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    s = np.exp(alpha * (corr - max_c))
    w = s / s.sum()
    
    h_rev = h[::-1]
    omh = 1.0 - h
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = dx * (np.dot(w[:j+1], omh[n-1-j:n]) - np.dot(w[j:j+n], h_rev))
    return grad

def slsqp_optimize(h_init, n, target, alphas, maxiter, time_limit, start_time):
    h = h_init.copy()
    best = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return soft_max_obj(x, alphas[0], target)  # We'll use fixed alpha in the call
    
    # Actually, let's do it differently: for each alpha, run SLSQP
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def obj_a(x, _alpha=alpha):
            n_local = len(x)
            dx = 2.0 / n_local
            corr = np.correlate(x, 1.0 - x, mode='full')
            max_c = np.max(corr)
            s = np.exp(_alpha * (corr - max_c))
            return (max_c + np.log(s.sum()) / _alpha) * dx
        
        def grad_a(x, _alpha=alpha):
            return soft_max_grad(x, _alpha)
        
        def cons_fun(x, _target=target):
            return np.sum(x) - _target
        
        def cons_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(obj_a, h, jac=grad_a, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h = h_new
        except Exception:
            pass
    
    return h, best

def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 10)
    
    for i in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            break
        
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        move_type = rng.randint(3)
        h_new = h.copy()
        
        if move_type == 0:
            # Single swap
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
        elif move_type == 1 and len(ones) >= 2 and len(zeros) >= 2:
            # Double swap
            idx1 = rng.choice(len(ones), 2, replace=False)
            idx0 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[idx1[0]]] = 0
            h_new[ones[idx1[1]]] = 0
            h_new[zeros[idx0[0]]] = 1
            h_new[zeros[idx0[1]]] = 1
        else:
            # Block move: swap a block of 1s with a block of 0s
            block_size = rng.randint(1, 4)
            # Find a run of 1s of at least block_size
            found = False
            for _ in range(10):
                start1 = rng.randint(0, n - block_size)
                if np.all(h[start1:start1+block_size] == 1):
                    # Find a run of 0s
                    for _2 in range(10):
                        start0 = rng.randint(0, n - block_size)
                        if np.all(h[start0:start0+block_size] == 0):
                            if start0 != start1 and abs(start0 - start1) >= block_size:
                                h_new[start1:start1+block_size] = 0
                                h_new[start0:start0+block_size] = 1
                                found = True
                            break
                    if found:
                        break
            
            if not found:
                # Fallback to single swap
                i1 = ones[rng.randint(len(ones))]
                i0 = zeros[rng.randint(len(zeros))]
                h_new[i1] = 0
                h_new[i0] = 1
        
        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def interpolate_upscale(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgsb_optimize(h_init, n, target, alphas, mu, maxiter, time_limit, start_time):
    h = h_init.copy()
    best = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        
        def obj_pen(x, _alpha=alpha, _mu=mu, _target=target):
            corr = np.correlate(x, 1.0 - x, mode='full')
            max_c = np.max(corr)
            s = np.exp(_alpha * (corr - max_c))
            val = (max_c + np.log(s.sum()) / _alpha) * (2.0 / len(x))
            penalty = _mu * (np.sum(x) - _target) ** 2
            return val + penalty
        
        def grad_pen(x, _alpha=alpha, _mu=mu, _target=target):
            g = soft_max_grad(x, _alpha)
            g += 2 * _mu * (np.sum(x) - _target) * np.ones(n)
            return g
        
        try:
            res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best:
                best = sc
                h = h_new
        except Exception:
            pass
    
    return h, best
