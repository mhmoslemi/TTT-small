def pair_swap_line_search(h, i, j, target):
    """Find optimal delta to swap between h[i] and h[j]."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute current overlaps
    corr = np.correlate(h, 1.0 - h, mode='full')
    base = corr * dx  # O_k values
    
    # Coefficients for h[i] in each O_k
    # O_k = sum_{valid m} h[m]*(1-h[m+k])*dx
    # dO_k/dh[i] = dx * [(1-h[i+k]) if valid - h[i-k] if valid]
    
    k_offsets = np.arange(-(n-1), n)  # all shifts
    
    # For shift s, the contribution of h[i] to O(s):
    # - If i+s is valid (0<=i+s<n): h[i]*(1-h[i+s]) contributes, derivative wrt h[i] is (1-h[i+s])
    # - If i-s is valid (0<=i-s<n): h[i-s]*(1-h[i]) contributes, derivative wrt h[i] is -h[i-s]
    
    a_i = np.zeros(2*n-1)  # coefficient of h[i] in each O_k
    a_j = np.zeros(2*n-1)  # coefficient of h[j] in each O_k
    
    # For h[i]:
    for s_idx, s in enumerate(k_offsets):
        coeff = 0.0
        if 0 <= i + s_idx - (n-1) < n:  # i + s is valid... wait
        # Actually s = s_idx - (n-1)
        s_val = s_idx - (n-1)
        if 0 <= i + s_val < n:
            coeff += (1.0 - h[i + s_val])
        if 0 <= i - s_val < n:
            coeff -= h[i - s_val]
        a_i[s_idx] = coeff * dx
    
    # Similarly for h[j]
    for s_idx, s in enumerate(k_offsets):
        s_val = s_idx - (n-1)
        coeff = 0.0
        if 0 <= j + s_val < n:
            coeff += (1.0 - h[j + s_val])
        if 0 <= j - s_val < n:
            coeff -= h[j - s_val]
        a_j[s_idx] = coeff * dx
    
    # O_k(δ) = base[k] + δ * (a_i[k] - a_j[k])
    # We want to minimize max_k O_k(δ) over δ ∈ [lo, hi]
    # where lo = max(-h[i], -(1-h[j])) and hi = min(1-h[i], h[j])
    
    lo = max(-h[i], -(1.0 - h[j]))
    hi = min(1.0 - h[i], h[j])
    
    if lo >= hi:
        return 0.0, 0.0
    
    # The function f(δ) = max_k (base[k] + δ*(a_i[k] - a_j[k])) is convex piecewise linear
    # Its minimum occurs at a breakpoint or at the boundary
    # Breakpoints: where base[k] + δ*(a_i[k]-a_j[k]) = base[l] + δ*(a_i[l]-a_j[l])
    # => δ = (base[l] - base[k]) / ((a_i[k]-a_j[k]) - (a_i[l]-a_j[l]))
    
    diff = a_i - a_j  # diff[k] = a_i[k] - a_j[k]
    
    # Check boundaries and all breakpoints
    candidates = [lo, hi, 0.0]
    
    # For efficiency, only consider the top few k values
    top_k = np.argsort(base)[-min(50, 2*n-1):]  # top 50 shifts by base value
    
    for idx_k in top_k:
        for idx_l in top_k:
            if idx_k == idx_l:
                continue
            denom = diff[idx_k] - diff[idx_l]
            if abs(denom) < 1e-15:
                continue
            delta = (base[idx_l] - base[idx_k]) / denom
            if lo < delta < hi:
                candidates.append(delta)
    
    # Evaluate at all candidates
    best_delta = 0.0
    best_val = np.max(base)
    for delta in candidates:
        delta = np.clip(delta, lo, hi)
        val = np.max(base + delta * diff)
        if val < best_val:
            best_val = val
            best_delta = delta
    
    return best_delta, best_val

4. Breakpoints are where O_k + δ*d_k = O_l + δ*d_l, i.e., δ = (O_l - O_k)/(d_k - d_l)

def pair_swap_search(h, n, target, max_pairs, time_limit, start_time):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    rng = np.random.RandomState(123)
    improvements = 0
    
    for step in range(max_pairs):
        if time.time() - start_time > time_limit:
            break
        if improvements > 200:
            break
        
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Compute coefficients
        d = np.zeros(2*n-1)
        for k_idx, k in enumerate(range(-(n-1), n)):
            ci = 0.0
            cj = 0.0
            if 0 <= i + k < n:
                ci += (1.0 - h[i+k])
            if 0 <= i - k < n:
                ci -= h[i-k]
            if 0 <= j + k < n:
                cj += (1.0 - h[j+k])
            if 0 <= j - k < n:
                cj -= h[j-k]
            d[k_idx] = (ci - cj) * dx
        
        # Feasible range for delta
        lo = max(-h[i], -(1.0 - h[j]))
        hi = min(1.0 - h[i], h[j])
        
        if lo >= hi - 1e-15:
            continue
        
        # Find optimal delta
        # f(delta) = max_k (corr[k] + delta * d[k])
        # Consider top 30 shifts
        top_indices = np.argsort(corr)[-30:]
        
        candidates = [lo, hi, 0.0]
        for idx_a in top_indices:
            for idx_b in top_indices:
                if idx_a == idx_b:
                    continue
                denom = d[idx_a] - d[idx_b]
                if abs(denom) < 1e-15:
                    continue
                delta = (corr[idx_b] - corr[idx_a]) / denom
                if lo < delta < hi:
                    candidates.append(delta)
        
        best_delta = 0.0
        best_val = np.max(corr)
        for delta in candidates:
            delta = np.clip(delta, lo, hi)
            val = np.max(corr + delta * d)
            if val < best_val - 1e-12:
                best_val = val
                best_delta = delta
        
        if best_delta != 0.0:
            h[i] += best_delta
            h[j] -= best_delta
            h = project_feasible(h, target)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                improvements += 1
            else:
                # Revert
                h[i] -= best_delta
                h[j] += best_delta
                h = project_feasible(h, target)
    
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def build_A(n, g):
    dx = 2.0 / n
    A = np.zeros((2*n-1, n))
    for idx, k in enumerate(range(-(n-1), n)):
        lo = max(0, -k)
        hi = min(n, n-k)
        if hi > lo:
            A[idx, lo:hi] = g[lo+k:hi+k] * dx
    return A

    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0/len(h)))
