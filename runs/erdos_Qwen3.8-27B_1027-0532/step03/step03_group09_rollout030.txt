        h_pg, c5_pg = proj_grad_bt(best_h, 5, 0.02, 0.1, 300000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: Fine SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Basin hopping with diverse strategies
    n_bh = 0
    temp_sa = 0.002
    while rem() > 130 and n_bh < 50:
        strategy = n_bh % 8
        h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)

        h_opt, c5_opt = slsqp_optimize(h_pert, 0.005, 180)

        if rem() > 70:
            tl_pg = min(rem() - 50, 25)
            h_pg, c5_pg = proj_grad_bt(h_opt, 3, 0.01, 0.06, 150000, tl_pg, start_time)
            if c5_pg < c5_opt:
                c5_opt = c5_pg
                h_opt = h_pg

        # Simulated annealing acceptance
        delta = c5_opt - best_c5
        if delta < 0:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            temp_sa = max(0.0005, temp_sa * 0.95)
        elif rng.random() < np.exp(-delta / max(temp_sa, 1e-6)):
            best_h = h_opt.copy()
            temp_sa = min(0.005, temp_sa * 1.2)
        else:
            temp_sa = max(0.0003, temp_sa * 0.9)

        n_bh += 1

    # Phase 6: Very fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Gradient polish (single worst lag)
    if rem() > 50:
        tl = min(rem() - 25, 60)
        h_pg, c5_pg = proj_grad_bt(best_h, 1, 0.003, 0.04, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 8: Another basin hopping round with smaller perturbations
    if rem() > 130:
        n_bh2 = 0
        while rem() > 100 and n_bh2 < 20:
            strategy = 0 if n_bh2 % 2 == 0 else 6
            h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)
            h_opt, c5_opt = slsqp_optimize(h_pert, 0.003, 200)
            if rem() > 60:
                tl_pg = min(rem() - 40, 20)
                h_pg, c5_pg = proj_grad_bt(h_opt, 2, 0.005, 0.05, 100000, tl_pg, start_time)
                if c5_pg < c5_opt:
                    c5_opt = c5_pg
                    h_opt = h_pg
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            n_bh2 += 1

    # Phase 9: Resolution increase
    if rem() > 200:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 2.0, n0)
        new_t = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = project_onto_constraint(h_up + 0.008 * rng.standard_normal(n1), n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.01, 250)
        if rem() > 80:
            h_opt_up2, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.003, 300)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
                h_opt_up = h_opt_up2
        if rem() > 40:
            tl_up = min(rem() - 20, 50)
            h_pg_up, c5_pg_up = proj_grad_bt(h_opt_up, 3, 0.01, 0.05, 100000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n0 = n1

    # Phase 10: Final fine SLSQP
    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 11: Final gradient polish
    if rem() > 20:
        tl = min(rem() - 10, 40)
        h_pg, c5_pg = proj_grad_bt(best_h, 2, 0.002, 0.03, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
