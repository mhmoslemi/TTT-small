    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    dx = 2.0 / n

    k = np.arange(-(n-1), n)  # length 2n-1
    lo = np.maximum(0, k)     # lower bound for valid i
    hi = np.minimum(n-1, n-1+k)  # upper bound for valid i
    
    best_c5 = compute_c5(h)
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        
        c = np.correlate(h, 1.0-h, mode='full')  # will multiply by dx at the end
        improved = False
        
        for j in range(n):
            # Compute slopes
            A = (j >= lo) & (j <= hi)
            B = (j+k >= lo) & (j+k <= hi)
            
            # h[j-k] with out-of-range = 0
            jk_idx = j - k
            hk = np.where((jk_idx >= 0) & (jk_idx < n), h[np.clip(jk_idx, 0, n-1)], 0.0)
            
            # h[j+k] with out-of-range = 0
            jkp_idx = j + k
            hkp = np.where((jkp_idx >= 0) & (jkp_idx < n), h[np.clip(jkp_idx, 0, n-1)], 0.0)
            
            slopes = A * (1.0 - hk) - B * hkp
            
            # c[k] = slopes[k] * h[j] + intercept[k]
            # intercept[k] = c[k] - slopes[k] * h[j]
            # We want to minimize max_k (slopes[k]*t + (c[k] - slopes[k]*h[j])) * dx
            # Since dx is constant, minimize max_k (slopes[k]*t + c[k] - slopes[k]*h[j])
            
            # Let f_k(t) = slopes[k]*t + c[k] - slopes[k]*h[j]
            # = slopes[k]*(t - h[j]) + c[k]
            # So minimize max_k (slopes[k]*(t-h[j]) + c[k]) for t in [0,1]
            # Let u = t - h[j], then u in [-h[j], 1-h[j]]
            # Minimize max_k (slopes[k]*u + c[k]) for u in [-h[j], 1-h[j]]
            
            # Ternary search on u
            u_lo = -h[j]
            u_hi = 1.0 - h[j]
            for _ in range(40):
                u1 = u_lo + (u_hi - u_lo) / 3
                u2 = u_hi - (u_hi - u_lo) / 3
                if np.max(slopes * u1 + c) < np.max(slopes * u2 + c):
                    u_hi = u2
                else:
                    u_lo = u1
            
            t_new = u_lo + (u_hi - u_lo) / 2 + h[j]
            t_new = np.clip(t_new, 0.0, 1.0)
            
            if abs(t_new - h[j]) > 1e-10:
                h[j] = t_new
                improved = True
        
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
        if not improved:
            break
    
    return h, best_c5

_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit):

    _global_n = n
    _global_target = target

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-4):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)

            try:
                prob.solve(solver=cp.ECOS, max_iter=10000)

                break

        h_lp2, c5_lp2 = alternating_lp(best_h, n_use, target_use, max_iter=3000,
                                        time_limit=lp_t, damping_init=0.4, reg=1e-5)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2
        if tleft() > 30:
            h_cd3, c5_cd3 = coord_descent_optimize(best_h, max_sweeps=2000,
                                                     time_limit=min(40, tleft() * 0.4),
                                                     project_every=2)
            if c5_cd3 < best_score:
                best_score = c5_cd3
                best_h = h_cd3

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
