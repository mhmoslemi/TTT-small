import time
import numpy as np
from scipy.optimize import minimize

_g_target = 0.0
_g_n = 0

def _c5(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _binarize(h, S):
    n = len(h)
    k = int(round(S))
    idx = np.argsort(h)[::-1][:k]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def _grad_shift(h, lag, dx):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:lag + m])
            g[lag:lag + m] += (-h[:m])
    else:
        j = -lag
        m = n - j
        if m > 0:
            g[j:] += (1.0 - h[:m])
            g[:m] += (-h[j:])
    return g * dx

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _eq_con_fun(x):
    return x.sum() - _g_target

def _eq_con_jac(x):
    return np.ones(_g_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _g_target, _g_n
    _g_target = target_sum
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov(h0, n, target_sum, n_iters, alpha0, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr))
        g = _grad_shift(y, k_star - (n - 1), dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _compute_G(h, n):
    G = np.zeros((n, 2 * n - 1))
    for p in range(n):
        lo1 = max(0, n - 1 - p)
        hi1 = min(2 * n - 2, 2 * n - 2 - p)
        if lo1 <= hi1:
            lags = np.arange(lo1, hi1 + 1) - (n - 1)
            G[p, lo1:hi1 + 1] += (1.0 - h[p + lags])
        lo2 = max(0, p)
        hi2 = min(2 * n - 2, p + n - 1)
        if lo2 <= hi2:
            lags2 = np.arange(lo2, hi2 + 1) - (n - 1)
            G[p, lo2:hi2 + 1] -= h[p - lags2]
    return G

def _swap_cd(h0, n, target, max_rounds, deadline, rng, sa_temp=0.0):
    h = _binarize(h0, target)
    dx = 2.0 / n
    best_c5 = _c5(h)
    best_h = h.copy()
    stall = 0

    for round_i in range(max_rounds):
        if time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = corr.max() * dx

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        if len(ones) == 0 or len(zeros) == 0:
            break

        G = _compute_G(h, n)
        best_swap_val = c5_curr
        best_j = -1
        best_jp = -1

        for j in ones:
            deltas = G[zeros] - G[j]
            new_corrs = corr[None, :] + deltas
            new_maxes = new_corrs.max(axis=1) * dx
            bl = np.argmin(new_maxes)
            if new_maxes[bl] < best_swap_val - 1e-12:
                best_swap_val = new_maxes[bl]
                best_j = j
                best_jp = zeros[bl]

        if best_j >= 0:
            h[best_j] = 0.0
            h[best_jp] = 1.0
            stall = 0
            if best_swap_val < best_c5:
                best_c5 = best_swap_val
                best_h = h.copy()
        else:
            stall += 1
            if sa_temp > 0.0:
                j = ones[rng.randint(len(ones))]
                jp = zeros[rng.randint(len(zeros))]
                h[best_j if best_j >= 0 else j, best_jp if best_jp >= 0 else jp] = 0.0
                h[best_jp if best_jp >= 0 else jp, best_j if best_j >= 0 else j] = 1.0
                if _c5(h) < best_c5:
                    best_c5 = _c5(h)
                    best_h = h.copy()
            if stall >= 15:
                break

    return best_h, best_c5

def _swap_cd_first(h0, n, target, max_rounds, deadline, rng):
    h = _binarize(h0, target)
    dx = 2.0 / n
    best_c5 = _c5(h)
    best_h = h.copy()
    stall = 0

    for round_i in range(max_rounds):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_curr = corr.max() * dx
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        G = _compute_G(h, n)
        tried = 0
        improved = False
        n_tries = min(len(ones) * len(zeros), 200)
        for _ in range(n_tries):
            tried += 1
            j = ones[rng.randint(len(ones))]
            jp = zeros[rng.randint(len(zeros))]
            new_corr = corr + G[jp] - G[j]
            new_max = new_corr.max() * dx
            if new_max < c5_curr - 1e-12:
                h[j] = 0.0
                h[jp] = 1.0
                c5_curr = new_max
                improved = True
                if c5_curr < best_c5:
                    best_c5 = c5_curr
                    best_h = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                G = _compute_G(h, n)
                corr = new_corr
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 20:
                break
    return best_h, best_c5

def _de_constrained(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.3 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.8 - 0.4 * t_frac
        cr = 0.3 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1, r2, r3 = candidates[r1], candidates[r2], candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.random(n) < cr
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            c_trial = _c5(trial)
            if c_trial <= caches[i]:
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if best_c5 < h0.max() and gen > 50 and caches[best_idx] - best_c5 > 0.01:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = _c5(h_init)
    best_h = h_init.copy()
    best_n = n_init

    n_candidates = [n_init]
    for d in [2, -2, 4, -4, 6]:
        nt = n_init + d
        if nt % 2 == 0 and 60 <= nt <= 200:
            n_candidates.append(nt)
    n_candidates = sorted(set(n_candidates))

    budget_per_n = max(100, int((deadline - time.time()) / len(n_candidates)))

    for n in n_candidates:
        t_n_start = time.time()
        n_deadline = min(deadline - 5, t_n_start + budget_per_n)
        target = n / 2.0

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target)

        c5_start = _c5(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        h_work = h_start.copy()

        # Phase 1: Nesterov subgradient
        if time.time() < n_deadline - 120:
            h_work, c5_n = _nesterov(h_work, n, target, n_iters=300, alpha0=0.6,
                                     deadline=n_deadline - 120)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_work.copy()
                best_n = n

        # Phase 2: Multi-gradient Nesterov
        if time.time() < n_deadline - 100:
            h_work2, c5_n2 = _nesterov_multi(h_work, n, target, n_iters=200,
                                              alpha0=0.4, m_top=5,
                                              deadline=n_deadline - 100)
            if c5_n2 < best_c5:
                best_c5 = c5_n2
                best_h = h_work2.copy()
                best_n = n
                h_work = h_work2

        # Phase 3: Binarize + swap CD (best improvement)
        h_bin = _binarize(h_work, target)
        c5_bin = _c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n

        if time.time() < n_deadline - 150:
            h_bin, c5_swap = _swap_cd(h_bin, n, target, max_rounds=2000,
                                       deadline=n_deadline - 150, rng=rng)
            if c5_swap < best_c5:
                best_c5 = c5_swap
                best_h = h_bin.copy()
                best_n = n

        # Phase 4: First-improvement swap CD for more exploration
        if time.time() < n_deadline - 100:
            h_bin2, c5_swap2 = _swap_cd_first(h_bin, n, target, max_rounds=3000,
                                               deadline=n_deadline - 100, rng=rng)
            if c5_swap2 < best_c5:
                best_c5 = c5_swap2
                best_h = h_bin2.copy()
                best_n = n

        # Phase 5: SLSQP with increasing alpha
        h_smooth = _binarize(h_bin2, target).copy()
        for alpha in [100, 500, 2000, 8000]:
            if time.time() > n_deadline - 30:
                break
            h_sl = _slsqp_opt(h_smooth, n, target, alpha, maxiter=300)
            c5_sl = _c5(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_smooth = h_sl

        # Phase 6: Binarize the SLSQP result and do more swap CD
        h_bin3 = _binarize(h_smooth, target)
        if time.time() < n_deadline - 30:
            h_bin3, c5_sw3 = _swap_cd(h_bin3, n, target, max_rounds=1500,
                                       deadline=n_deadline - 30, rng=rng)
            if c5_sw3 < best_c5:
                best_c5 = c5_sw3
                best_h = h_bin3.copy()
                best_n = n

        # Phase 7: Final SLSQP polish
        if time.time() < n_deadline - 10:
            h_pol = _slsqp_opt(h_smooth, n, target, 5000, maxiter=400)
            c5_pol = _c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                best_n = n

    # Final global polish on the best solution found
    n_best = best_n
    target_best = n_best / 2.0

    if time.time() < deadline - 20:
        h_final = best_h.copy()
        for alpha in [3000, 10000, 30000]:
            if time.time() > deadline - 15:
                break
            h_final = _slsqp_opt(h_final, n_best, target_best, alpha, maxiter=400)
            c5_f = _c5(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    # Final binary refinement
    if time.time() < deadline - 8:
        h_fb = _binarize(best_h, target_best)
        h_fb, c5_fb = _swap_cd(h_fb, n_best, target_best, max_rounds=2000,
                                deadline=deadline - 5, rng=rng)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_fb.copy()

    # One last SLSQP
    if time.time() < deadline - 3:
        h_last = _slsqp_opt(best_h, n_best, target_best, 20000, maxiter=300)
        c5_last = _c5(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5(best_h)
    return (best_h, best_c5, best_n)
