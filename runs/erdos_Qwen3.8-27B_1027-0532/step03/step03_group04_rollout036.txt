                obj_t = corr_t * dx
                c_t = np.max(obj_t)
                if c_t < best_c5 - 1e-12:
                    best_c5 = c_t
                    best_h = h.copy()
                    h = project_feasible(h, target)
                    improved = True
                else:
                    h[i] = orig
                    break

        if not improved:
            break

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target, _global_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return max(0.0, budget_s - 15.0 - (time.time() - start_time))

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_feasible(h_init.copy(), target)
    best_c5 = compute_c5(best_h)

    alphas = [5, 10, 20, 50, 100, 200, 500, 1000]
    for alpha in alphas:
        if tleft() < 80:
            break
        deadline = time.time() + min(150.0, tleft() * 0.45)
        h_new, c5_new = project_gradient_descent(
            best_h, n, target, alpha, max_iters=8000, deadline=deadline, mom=0.9)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    if tleft() > 120:
        for alpha in [100, 500, 2000]:
            if tleft() < 50:
                break
            h_pol = slsqp_polish(best_h, n, target, alpha, maxiter=300)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    for noise_level in [0.03, 0.08, 0.15]:
        if tleft() < 120:
            break
        h_pert = best_h + rng.randn(n) * noise_level
        h_pert = project_feasible(h_pert, target)
        for alpha in [20, 100, 300]:
            if tleft() < 60:
                break
            deadline = time.time() + min(80.0, tleft() * 0.4)
            h_new, c5_new = project_gradient_descent(
                h_pert, n, target, alpha, max_iters=5000, deadline=deadline, mom=0.85)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    if tleft() > 80:
        h_coord, c5_coord = coordinate_refine(best_h, n, target, max_sweeps=50,
                                              deadline=time.time() + min(80.0, tleft() * 0.5))
        if c5_coord < best_c5:
            best_c5 = c5_coord
            best_h = h_coord

    if tleft() > 50:
        for alpha in [500, 2000, 5000]:
            if tleft() < 30:
                break
            h_pol = slsqp_polish(best_h, n, target, alpha, maxiter=500)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol

    if tleft() > 60:
        for n_try in [600, 800]:
            if tleft() < 50:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            for alpha in [20, 100]:
                if tleft() < 40:
                    break
                deadline = time.time() + min(60.0, tleft() * 0.5)
                h_new, c5_new = project_gradient_descent(
                    h_interp, n_try, target_try, alpha, max_iters=4000,
                    deadline=deadline, mom=0.9)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
                    n = n_try
                    target = target_try

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pol = slsqp_polish(best_h, n_use, target_use, 3000, maxiter=800)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
