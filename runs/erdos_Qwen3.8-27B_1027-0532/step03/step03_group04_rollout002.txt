            valid_q = q_candidates[h[q_candidates] < 0.5]

            best_p_idx = np.argmin(cost_p[h[p_candidates] > 0.5])
            best_q_idx = np.argmin(cost_q[h[q_candidates] < 0.5])

def eval_swap_effect(h, p, q, n):
    """Compute the change in overlaps for each shift when swapping h[p] and h[q].
    Assumes h[p]=1 and h[q]=0 (or vice versa, but let's say 1→0 and 0→1)."""
    # h[p]: 1 → 0, h[q]: 0 → 1
    delta = np.zeros(n)
    for k in range(n):
        d = 0.0
        if p >= k:
            d += h[p-k] - 1.0
        if p + k < n:
            d += h[p+k]
        if q >= k:
            d += 1.0 - h[q-k]
        if q + k < n:
            d += -h[q+k]
        delta[k] = d
    return delta

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_shift_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.zeros(n)
    one_mh = 1.0 - h
    for k in range(n):
        overlaps[k] = np.dot(h[k:], one_mh[:n - k])
    return overlaps * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 15:
            damping = max(0.05, damping * 0.8)
            stuck_count = 0
        if it % 100 == 0 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.7)
    return best_h, best_c5


def binary_local_search(h_input, n, target, time_limit, rng, max_rounds=200):
    start = time.time()
    h = h_input.copy().astype(float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = project_feasible(h, target)
    n_ones = np.sum(h > 0.5)
    if n_ones != n // 2:
        idx_ones = np.where(h > 0.5)[0]
        idx_zeros = np.where(h <= 0.5)[0]
        if n_ones > n // 2:
            for j in idx_ones[:int(n_ones - n // 2)]:
                h[j] = 0.0
        else:
            for j in idx_zeros[:int(n // 2 - n_ones)]:
                h[j] = 1.0
    best_h = h.copy()
    best_score = compute_c5(h)
    overlaps = compute_shift_overlaps(h)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        sorted_idx = np.argsort(overlaps)[::-1]
        tried = 0
        improved = False
        for k_star in sorted_idx[:8]:
            if tried > 50:
                break
            k_val = int(k_star)
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
            if len(ones_idx) == 0 or len(zeros_idx) == 0:
                break
            best_gain = 0.0
            best_p = -1
            best_q = -1
            sample_ones = ones_idx if len(ones_idx) <= 300 else rng.choice(ones_idx, 300, replace=False)
            sample_zeros = zeros_idx if len(zeros_idx) <= 300 else rng.choice(zeros_idx, 300, replace=False)
            for p in sample_ones:
                p = int(p)
                for q in sample_zeros:
                    q = int(q)
                    if q == p:
                        continue
                    d = 0.0
                    if p >= k_val:
                        d += h[p - k_val] - 1.0
                    if p + k_val < n:
                        d += h[p + k_val]
                    if q >= k_val:
                        d += 1.0 - h[q - k_val]
                    if q + k_val < n:
                        d += -h[q + k_val]
                    gain = -d
                    if gain > best_gain:
                        best_gain = gain
                        best_p = p
                        best_q = q
            if best_p >= 0 and best_gain > 0:
                h[best_p], h[best_q] = h[best_q], h[best_p]
                new_overlaps = compute_shift_overlaps(h)
                new_max = float(np.max(new_overlaps))
                if new_max < best_score - 1e-12:
                    best_score = new_max
                    best_h = h.copy()
                    overlaps = new_overlaps
                    improved = True
                    tried += 1
                    continue
                else:
                    h[best_p], h[best_q] = h[best_q], h[best_p]
                    tried += 1
        if not improved:
            temperature = max(0.001, 0.01 * (1.0 - rnd / float(max_rounds)))
            found_sa = False
            for _ in range(50):
                p = int(rng.choice(np.where(h == 1.0)[0]))
                q = int(rng.choice(np.where(h == 0.0)[0]))
                if p == q:
                    continue
                h[p], h[q] = h[q], h[p]
                new_overlaps = compute_shift_overlaps(h)
                new_max = float(np.max(new_overlaps))
                delta = new_max - best_score
                if delta < 0 or rng.rand() < np.exp(-delta / temperature):
                    overlaps = new_overlaps
                    if new_max < best_score:
                        best_score = new_max
                        best_h = h.copy()
                    found_sa = True
                    break
                else:
                    h[p], h[q] = h[q], h[p]
            if not found_sa:
                break
    return project_feasible(best_h, target), best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
