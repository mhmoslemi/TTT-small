def power_sum_gradient(h, p):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    fp = np.sum(corr**p) ** (1.0/p)
    # weights: w_k = corr[k]^(p-1) / fp^(p-1)
    w = corr**(p-1) / fp**(p-1)
    # Now compute gradient of sum_k w_k * c[k] w.r.t. h
    # c[k] = sum_i h[i+k] * (1-h[i]) * dx (with appropriate indexing)
    grad = np.zeros(n)

        if wk < 1e-15:

        s = k - (n - 1)
        if s >= 0:
            m = n - s
            # c[k] = sum_{i=0}^{m-1} h[i+s] * (1-h[i]) * dx

            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def make_constraint_fun(target_sum):
    return lambda h: float(np.sum(h) - target_sum)

def make_constraint_jac(n):
    return lambda h: np.ones(n)

def slsqp_optimize(h_init, eps, target_sum, maxiter, use_power_sum=False, p=20):
    n = len(h_init)
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    con_fun = lambda h: float(np.sum(h) - target_sum)
    con_jac = lambda h: np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    if use_power_sum:
        try:
            result = minimize(ps_objective, h0, args=(p, target_sum),
                              jac=ps_gradient, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-12})
        except Exception:
            return h0, compute_c5(h0)
    else:
        try:
            result = minimize(lse_objective, h0, args=(eps,),
                              jac=lse_gradient, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-12})
        except Exception:
            return h0, compute_c5(h0)
    h_opt = project_onto_constraint(result.x, target_sum)
    return h_opt, compute_c5(h_opt)

def lbfgs_sigmoid(h_init, p, target_sum, maxiter=500):
    n = len(h_init)
    z0 = np.zeros(n)
    h0 = h_init.astype(float)
    for i in range(n):
        if h0[i] > 0.99:
            z0[i] = 4.0
        elif h0[i] < 0.01:
            z0[i] = -4.0
        else:
            z0[i] = np.log(h0[i] / (1.0 - h0[i] + 1e-10))
    
    penalty = 500.0
    
    def obj(z):
        hh = 1.0 / (1.0 + np.exp(-np.clip(z, -40, 40)))
        dx = 2.0 / n
        corr = np.correlate(hh, 1.0 - hh, mode='full') * dx
        ps = power_sum_value(corr, p)
        sv = np.sum(hh) - target_sum
        return ps + penalty * sv * sv
    
    def grad(z):
        hh = 1.0 / (1.0 + np.exp(-np.clip(z, -40, 40)))
        sz = hh * (1.0 - hh)
        dx = 2.0 / n
        omhh = 1.0 - hh
        corr = np.correlate(hh, omhh, mode='full') * dx
        ps = power_sum_value(corr, p)
        if ps < 1e-20:
            w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            w = (corr / ps) ** (p - 1) / (2 * n - 1)
        dh = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-16:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                dh[s:] += wk * dx * omhh[:m]
                dh[:m] -= wk * dx * hh[s:]
            else:
                r = -s
                m = n - r
                dh[:m] += wk * dx * omhh[r:]
                dh[r:] -= wk * dx * hh[:m]
        sv = np.sum(hh) - target_sum
        dh += 2.0 * penalty * sv
        return dh * sz
    
    try:
        result = minimize(obj, z0, jac=grad, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-12})
        h_opt = 1.0 / (1.0 + np.exp(-np.clip(result.x, -40, 40)))
        h_opt = project_onto_constraint(h_opt, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_from_continuous(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def fast_binary_local_search(h, max_rounds, time_limit, start_time):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    no_improve = 0
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        improved = False
        for oi in ones:
            if improved:
                break
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                if sc < current - 1e-15:
                    current = sc
                    ones.remove(oi)
                    zeros.remove(zi)
                    improved = True
                    break
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
        else:
            no_improve = 0
    return h, current

def binary_sa(h_init, n_iter, t_start, t_end, time_limit, start_time, rng):
    n = len(h_init)
    h = h_init.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    for step in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(n_iter - 1, 1))
        r = rng.random()
        if r < 0.5:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            h[oi] = 0.0
            h[zi] = 1.0
            new_score = compute_c5(h)
            delta = new_score - score
            if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / t)):
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[zi] = 0.0
        elif r < 0.75:
            blen = int(rng.integers(2, max(3, n // 6) + 1))
            a = int(rng.integers(0, n - blen + 1))
            b = int(rng.integers(0, n - blen + 1))
            if abs(a - b) < blen:
                continue
            block_a = h[a:a + blen].copy()
            block_b = h[b:b + blen].copy()
            h[a:a + blen] = block_b
            h[b:b + blen] = block_a
            new_score = compute_c5(h)
            delta = new_score - score
            if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / t)):
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            else:
                h[a:a + blen] = block_a
                h[b:b + blen] = block_b
        elif r < 0.90:
            nswap = int(rng.integers(2, 6))
            ones_pool = np.where(h == 1.0)[0]
            zeros_pool = np.where(h == 0.0)[0]
            if len(ones_pool) < nswap or len(zeros_pool) < nswap:
                continue
            chosen_ones = rng.choice(ones_pool, size=nswap, replace=False)
            chosen_zeros = rng.choice(zeros_pool, size=nswap, replace=False)
            old_ones = h[chosen_ones].copy()
            old_zeros = h[chosen_zeros].copy()
            h[chosen_ones] = 0.0
            h[chosen_zeros] = 1.0
            new_score = compute_c5(h)
            delta = new_score - score
            if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / t)):
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            else:
                h[chosen_ones] = old_ones
                h[chosen_zeros] = old_zeros
        else:
            blen = int(rng.integers(2, max(3, n // 4) + 1))
            a = int(rng.integers(0, n - blen + 1))
            block_a = h[a:a + blen].copy()
            h[a:a + blen] = 1.0 - h[a:a + blen]
            new_score = compute_c5(h)
            delta = new_score - score
            if delta < 0 or (t > 1e-10 and rng.random() < np.exp(-delta / t)):
                score = new_score
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
            else:
                h[a:a + blen] = block_a
    return best_h, best_score

def continuous_annealing(h_init, n_iter, t_start, t_end, time_limit, start_time, rng, target_sum):
    n = len(h_init)
    h = h_init.copy()
    score = compute_c5(h)
    best_h = h.copy()
    best_score = score
    for step in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(n_iter - 1, 1))
        h_pert = h + rng.normal(0.0, 0.05 * t * 100, n)
        h_pert = project_onto_constraint(h_pert, target_sum)
        new_score = compute_c5(h_pert)
        delta = new_score - score
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / (t * 50))):
            h = h_pert
            score = new_score
            if score < best_score:
                best_score = score
                best_h = h.copy()
    return best_h, best_score

def multi_p_optimize(h_init, target_sum, time_limit, start_time):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    p_values = [4, 8, 16, 32, 64, 128, 256, 512]
    for p in p_values:
        if time.time() - start_time > time_limit:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, target_sum, 150, use_power_sum=True, p=p)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0, 1)
    
    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: Quick LSE optimization from initial
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        h_opt, c5_opt = slsqp_optimize(h0, 0.01, target_sum, 150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 2: Power-sum multi-p optimization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = multi_p_optimize(best_h, target_sum, remaining - 50, start_time)
    
    # Phase 3: L-BFGS-B with sigmoid parameterization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        h_opt, c5_opt = lbfgs_sigmoid(best_h, 32, target_sum, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_opt, c5_opt = lbfgs_sigmoid(best_h, 128, target_sum, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Phase 4: Fine LSE refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        for eps in [0.005, 0.002, 0.001, 0.0005]:
            if time.time() - start_time > budget_s - 100:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, eps, target_sum, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 5: Convert to binary and local search
    bin_h = make_binary_from_continuous(best_h)
    bin_c5 = compute_c5(bin_h)
    best_bin_h = bin_h.copy()
    best_bin_c5 = bin_c5
    
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_bin_h, best_bin_c5 = fast_binary_local_search(best_bin_h, 100, min(remaining - 20, 60), start_time)
    
    # Phase 6: Binary simulated annealing
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        sa_limit = remaining - 120
        best_bin_h, best_bin_c5 = binary_sa(best_bin_h, 500000, 0.005, 0.00005, sa_limit, start_time, rng)
    
    # Post-SA local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        best_bin_h, best_bin_c5 = fast_binary_local_search(best_bin_h, 50, min(remaining - 10, 30), start_time)
    
    # Phase 7: Continuous refinement from best binary
    h_refine = best_bin_h.astype(float)
    h_refine = project_onto_constraint(h_refine, target_sum)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() - start_time > budget_s - 60:
                break
            h_opt, c5_opt = slsqp_optimize(h_refine, eps, target_sum, 250)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_refine = h_opt.copy()
    
    # Phase 8: Multi-restart with diverse strategies
    remaining = budget_s - (time.time() - start_time) - 40
    restart_count = 0
    while remaining > 60 and restart_count < 20:
        strategy = restart_count % 7
        if strategy == 0:
            h_pert = best_h + 0.03 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_bin_h.astype(float) + 0.05 * rng.standard_normal(n0)
        elif strategy == 2:
            h_tmp = best_h + 0.15 * rng.standard_normal(n0)
            h_bin = make_binary_from_continuous(h_tmp)
            h_pert = h_bin.astype(float) + 0.03 * rng.standard_normal(n0)
        elif strategy == 3:
            h_pert = random_binary(n0, rng).astype(float) + 0.08 * rng.standard_normal(n0)
        elif strategy == 4:
            alpha = rng.uniform(0.3, 0.7)
            h_rand = random_binary(n0, rng).astype(float)
            h_pert = alpha * best_h + (1.0 - alpha) * h_rand
        elif strategy == 5:
            h_pert = best_bin_h.astype(float).copy()
            nflip = max(3, n0 // 15)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = h_pert + 0.02 * rng.standard_normal(n0)
        else:
            h_pert = best_h + 0.08 * rng.standard_normal(n0)
        
        h_pert = project_onto_constraint(h_pert, target_sum)
        tl = min(remaining - 20, 45)
        if tl > 30:
            h_opt, c5_opt = slsqp_optimize(h_pert, 0.005, target_sum, 150)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_opt, c5_opt = slsqp_optimize(h_opt, 0.0005, target_sum, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        restart_count += 1
        remaining = budget_s - (time.time() - start_time) - 40
    
    # Phase 9: Resolution upsampling
    remaining = budget_s - (time.time() - start_time) - 20
    if remaining > 150:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            if time.time() - start_time > budget_s - 20:
                break
            h_opt_up, c5_opt_up = slsqp_optimize(h_up, eps, n1 / 2.0, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()
    
    # Phase 10: Final fine-tuning
    remaining = budget_s - (time.time() - start_time) - 10
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    for eps in [0.00005, 0.00002, 0.00001]:
        if time.time() - start_time > budget_s - 5:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, eps, target_cur, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Final validation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5 + 1e-10:
        final_h = best_h
        final_c5 = best_c5
    
    # Ensure valid
    final_h = np.clip(final_h, 0.0, 1.0)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, len(final_h)
