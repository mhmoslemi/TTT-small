        if s > target_sum:
            lo = mid
        else:
            hi = mid

def multi_swap_sa(h_start, n_steps, t_start, t_end, rng):

        nonlocal corr
        h[i], h[j] = h[j], h[i]

        ones_set.add(j)
        zeros_set.discard(j)
        zeros_set.add(i)

        h[i], h[j] = h[j], h[i]

        zeros_set.discard(i)

        s_max = min(half - i, i, half - j, j)

            return None, s_min, s_max
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
        return delta, s_min, s_max

                new_max = float(np.max(corr[k_range] + delta))
                # Also need to check outside k_range
                if k_range[0] > 0:
                    outside_max = float(np.max(corr[:k_range[0]]))
                else:
                    outside_max = -1
                if k_range[-1] < len(corr) - 1:
                    outside_max = max(outside_max, float(np.max(corr[k_range[-1]+1:])))
                else:
                    outside_max = max(outside_max, -1)
                actual_new_max = max(new_max, outside_max)
                delta_max = actual_new_max - current_max
                if delta_max < best_delta:
                    best_delta = delta_max
                    best_pair = (i, j, s_min, s_max)
        
        if best_pair is None or best_delta >= 0:

        # Check if objective improved
        f_old = smooth_objective(h, eps)
        f_new = smooth_objective(h_new, eps)
        
        if f_new > f_old:
            # Reduce learning rate
            lr *= 0.5
            v = 0.5 * v
            h_new = h + v
            h_new = project_onto_constraint(h_new, target_sum)
        
        h = h_new
    
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-15):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def targeted_sa(h_start, n_steps, t_start, t_end, rng, target_frac=0.4):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    n_one = len(ones_list)
    n_zero = len(zeros_list)
    half = (n - 1)
    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        if rng.random() < target_frac:
            worst_k = int(np.argmax(corr))
            worst_s = worst_k - half
            if worst_s != 0:
                valid_i = [x for x in ones_list if 0 <= x + worst_s < n]
                valid_j = [x for x in zeros_list if 0 <= x + worst_s < n]
                if len(valid_i) > 0 and len(valid_j) > 0:
                    i = valid_i[rng.integers(len(valid_i))]
                    j = valid_j[rng.integers(len(valid_j))]
                else:
                    i = ones_list[rng.integers(n_one)]
                    j = zeros_list[rng.integers(n_zero)]
            else:
                i = ones_list[rng.integers(n_one)]
                j = zeros_list[rng.integers(n_zero)]
        else:
            i = ones_list[rng.integers(n_one)]
            j = zeros_list[rng.integers(n_zero)]
        if i == j:
            step += 1
            continue
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            step += 1
            continue
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max
        if delta_c5 < 0 or (t > 1e-8 and rng.random() < np.exp(-delta_c5 / t)):
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1
    return best_h, best_c5_val

def proj_grad_descent(h_init, eps, n_iter, lr_init=0.005, momentum=0.9):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    v = np.zeros(n)
    lr = lr_init
    best_h = h.copy()
    best_obj = smooth_objective(h, eps)
    for it in range(n_iter):
        grad = smooth_gradient(h, eps)
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project_onto_constraint(h_new, target_sum)
        f_new = smooth_objective(h_new, eps)
        if f_new < best_obj:
            best_obj = f_new
            best_h = h_new.copy()
        if f_new > smooth_objective(h, eps):
            lr = max(lr * 0.5, 1e-8)
            v *= 0.5
            h_new = h + v
            h_new = project_onto_constraint(h_new, target_sum)
        h = h_new
    return best_h, compute_c5(best_h)
