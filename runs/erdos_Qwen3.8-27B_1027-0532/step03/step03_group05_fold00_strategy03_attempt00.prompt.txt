def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):

    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_autocorr(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    return np.fft.irfft(H * np.conj(H), N)[:n]


def weighted_gradient(h, weights):
    n = len(h)
    grad = np.zeros(n)
    max_lag = min(n - 1, 200)
    for lag in range(1, max_lag + 1):
        w = weights[lag - 1]
        if w < 1e-8:
            continue
        grad[:n - lag] += w * h[lag:]
        grad[lag:] += w * h[:n - lag]
    return grad
