def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        ones = np.where(h_r == 1)[0]
        # Flip the ones with lowest h value
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == int(target_sum):
        return h_r.astype(float)
    if current_sum > int(target_sum):

        ones = np.where(h_r == 1)[0]

                h_new[i1b] = 0
                h_new[i0a] = 1
                h_new[i0b] = 1

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

def project_binary(h, target_sum):

def smooth_obj(h, alpha, n, target):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)

    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5_fast(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, n, target),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5_fast(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

import numpy as np
from scipy.optimize import minimize
import time

_gn = 0
_gt = 0.0


def compute_c5_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return float(np.max(corr)) * (2.0 / n)


def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    target = int(target_sum)
    if current_sum == target:
        return h_r.astype(float)
    if current_sum > target:
        need = current_sum - target
        ones = np.where(h_r == 1)[0]
        if need >= len(ones):
            h_r[:] = 0
            h_r[:target] = 1
            order = np.argsort(-h[ones])[:target]
            h_r[:] = 0
            h_r[ones[order]] = 1
        else:
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target - current_sum
        zeros = np.where(h_r == 0)[0]
        if need >= len(zeros):
            h_r[:] = 1
            h_r[target:] = 0
        else:
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def thue_morse_binary(n):
    tm = np.zeros(n, dtype=float)
    for i in range(n):
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = 1.0 if (c & 1) else 0.0
    target = n // 2
    current = int(tm.sum())
    if current > target:
        ones = np.where(tm == 1)[0]
        order = rng_global.permutation(len(ones))[:current - target]
        tm[ones[order]] = 0
    elif current < target:
        zeros = np.where(tm == 0)[0]
        order = rng_global.permutation(len(zeros))[:target - current]
        tm[zeros[order]] = 1
    return tm


def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng,
                time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5_fast(h)
    current_c5 = best_c5
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = 10000

    for i in range(max_iters):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        if no_improve > max_no_improve:
            if rng.random() < 0.02:
                no_improve = 0
            else:
                break

        ratio = min(i / max(max_iters - 1, 1), 1.0)
        t = t_init * (t_final / t_init) ** ratio

        r = rng.random()
        h_new = h.copy()
        moved = False

        if r < 0.55 and len(ones) > 0 and len(zeros) > 0:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif r < 0.80 and len(ones) >= 2 and len(zeros) >= 2:
            a1 = ones[rng.randint(len(ones))]
            a2 = ones[rng.randint(len(ones))]
            b1 = zeros[rng.randint(len(zeros))]
            b2 = zeros[rng.randint(len(zeros))]
            if a1 == a2 or b1 == b2:
                a2 = ones[rng.randint(len(ones))]
                b2 = zeros[rng.randint(len(zeros))]
                if a1 == a2 or b1 == b2:
                    continue
            h_new[a1] = 0
            h_new[a2] = 0
            h_new[b1] = 1
            h_new[b2] = 1
            moved = True

            max_dist = max(2, n // 30)

                h_new[i0_cand] = 1
            else:
                i0 = zeros[rng.randint(len(zeros))]

                h_new[i0] = 1
            moved = True

                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

def hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5_fast(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 200

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5_fast(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def _cons_fun(x):
    return np.sum(x) - _gt


def _cons_jac(x):
    return np.ones(_gn)


def smooth_obj(h, alpha):
    n = _gn
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol,
                 time_limit=None, start_time=None):
    global _gn, _gt
    _gn = n
    _gt = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5_fast(h_cur)

    for alpha in alphas:
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5_fast(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
