def compute_d_vec(h, i, j, n):
    # Returns d vector of length 2n-1
    d = np.zeros(2*n-1)
    shifts = np.arange(-(n-1), n)
    
    # h[i + shifts] with out-of-bounds = 0
    idx_i_pos = i + shifts
    mask_i_pos = (idx_i_pos >= 0) & (idx_i_pos < n)
    d -= np.where(mask_i_pos, h[np.where(mask_i_pos, idx_i_pos, 0)], 0.0)
    
    idx_i_neg = i - shifts
    mask_i_neg = (idx_i_neg >= 0) & (idx_i_neg < n)
    d -= np.where(mask_i_neg, h[np.where(mask_i_neg, idx_i_neg, 0)], 0.0)
    
    idx_j_pos = j + shifts
    mask_j_pos = (idx_j_pos >= 0) & (idx_j_pos < n)
    d += np.where(mask_j_pos, h[np.where(mask_j_pos, idx_j_pos, 0)], 0.0)
    
    idx_j_neg = j - shifts
    mask_j_neg = (idx_j_neg >= 0) & (idx_j_neg < n)
    d += np.where(mask_j_neg, h[np.where(mask_j_neg, idx_j_neg, 0)], 0.0)

        t_hi = min(1.0, S)

        t_i = h[i]
        
        # Compute d_s = (h[j+s] + h[j-s]) - (h[i+s] + h[i-s])
        idx_i_pos = i + shifts

        idx_j_pos = j + shifts
        idx_j_neg = j - shifts

        mask_i_neg = (idx_i_neg >= 0) & (idx_i_neg < n)
        mask_j_pos = (idx_j_pos >= 0) & (idx_j_pos < n)
        mask_j_neg = (idx_j_neg >= 0) & (idx_j_neg < n)

        d[mask_i_pos] -= h[idx_i_pos[mask_i_pos]]
        d[mask_i_neg] -= h[idx_i_neg[mask_i_neg]]
        d[mask_j_pos] += h[idx_j_pos[mask_j_pos]]
        d[mask_j_neg] += h[idx_j_neg[mask_j_neg]]
        
        # Find optimal t: minimize max_s (corr[s] + (t - t_i) * d[s])

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)

def binary_sa(h_start, n_steps, t_start, t_end, rng):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_val = current_max
    best_h = h.copy()
    ones_list = sorted(np.where(h == 1.0)[0].tolist())
    zeros_list = sorted(np.where(h == 0.0)[0].tolist())
    half = n - 1
    n_corr = 2 * n - 1
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = ones_list[rng.integers(len(ones_list))]
        j = zeros_list[rng.integers(len(zeros_list))]
        if i == j:
            continue
        
        # Compute affected shift range
        i_lo = max(-i, i - half)
        i_hi = min(half - i, i)
        j_lo = max(-j, j - half)
        j_hi = min(half - j, j)
        s_lo = min(i_lo, j_lo)
        s_hi = max(i_hi, j_hi)

        idx_ip = i + s_range
        idx_in = i - s_range
        idx_jp = j + s_range
        idx_jn = j - s_range
        
        delta = np.zeros(len(s_range))
        m1 = (idx_ip >= 0) & (idx_ip < n)
        m2 = (idx_in >= 0) & (idx_in < n)
        m3 = (idx_jp >= 0) & (idx_jp < n)
        m4 = (idx_jn >= 0) & (idx_jn < n)
        delta[m1] += h[idx_ip[m1]]
        delta[m2] += h[idx_in[m2]]
        delta[m3] -= h[idx_jp[m3]]
        delta[m4] -= h[idx_jn[m4]]

        if s_lo + half == 0 and s_hi + half == n_corr - 1:
            new_max = float(np.max(new_affected))

            k1 = s_hi + half

            elif k0 == 0:

            else:
                un_affected_max = float(np.max(corr[:k0]))
            new_max = max(float(np.max(new_affected)), un_affected_max)

            h[j] = 1.0
            corr[k_range] += delta
            # Update lists

            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            ones_list.sort()
            zeros_list.sort()
            current_max = new_max
            if current_max < best_val:
                best_val = current_max
                best_h = h.copy()
    
    return best_h, best_val
