    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:

            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

    """Nesterov accelerated subgradient method with smooth surrogate."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        
        # Compute gradient using smooth surrogate (top-m weighted)
        max_corr = np.max(corr)
        weights = np.exp(alpha_sg * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            top_m_indices = [int(np.argmax(corr))]
            w_m = np.array([1.0])
        else:
            weights = weights / total_w
            top_m_indices = np.argsort(weights)[::-1][:sg_top_m]
            w_m = weights[top_m_indices].copy()
            w_m = w_m / w_m.sum()
        
        g = np.zeros(n)
        for idx in range(len(top_m_indices)):
            g += w_m[idx] * _grad_at_kstar(y, top_m_indices[idx])
        
        step = base_lr / np.sqrt(float(it + 1))

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_kstar(h, k_star):
    n = len(h)
    dx = 2.0 / n
    offset = k_star - (n - 1)
    grad = np.zeros(n)
    if offset >= 0:
        m = n - offset
        grad[:m] += (1.0 - h[offset:offset + m]) * dx
        grad[offset:offset + m] += (-h[:m]) * dx
    else:
        d = -offset
        m = n - d
        grad[d:d + m] += (1.0 - h[:m]) * dx
        grad[:m] += (-h[d:d + m]) * dx
    return grad

def _nesterov_opt(h0, n, target_sum, n_iters, base_lr, alpha_sg, top_m, deadline):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = np.max(corr)
        if max_c > 1e-30:
            w = np.exp(alpha_sg * (corr - max_c))
            tw = w.sum()
            if tw < 1e-300:
                idx = int(np.argmax(corr))
                g = _grad_kstar(y, idx)
            else:
                w = w / tw
                top = np.argsort(w)[::-1][:top_m]
                wm = w[top].copy()
                wm = wm / wm.sum()
                g = np.zeros(n)
                for ii in range(len(top)):
                    g += wm[ii] * _grad_kstar(y, top[ii])
        else:
            g = np.zeros(n)
        step = base_lr / (np.sqrt(float(it + 1)) + 0.1)
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

        for idx in range(mean(w) > 0.01 and int(np.sum(w > 0.01)) or 1):
            pass
        return w * _grad_kstar(h0, max_k)

    nsamp = len(h0)
    cons = [{'type': 'ineq', 'fun': lambda v: v.sum() - target_sum, 'jac': lambda v: np.ones(nsamp)},
            {'type': 'ineq', 'fun': lambda v: target_sum - v.sum(), 'jac': lambda v: -np.ones(nsamp)}]

def _lse_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _lse_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    e = np.exp(alpha * (corr - m))
    s = e.sum()
    w = alpha * e / s
    top_idx = np.argsort(w)[::-1]
    g = np.zeros(n)
    for ii in range(min(5, len(top_idx))):
        g += w[top_idx[ii]] * _grad_kstar(x, top_idx[ii])
    return g

def _slsqp_refine(h0, n, target_sum, alpha, maxiter):

             'jac': lambda x: np.ones(len(x))}]

        res = minimize(
            _lse_obj, h0.copy(), args=(alpha, n),
            jac=_lse_grad, args=(alpha, n),

            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return _fix_sum(res.x, target_sum)

        return h0

def _de_optimize(h0, n, target_sum, pop_size, max_gen, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.3:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif r < 0.6:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = 0.8 - 0.3 * (gen / max_gen)
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            mask = rng.random(n) < 0.7
            mask[rng.randint(n)] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size * 2:
            worst = np.argsort(pop_vals)[pop_size // 2:]
            for idx in worst:
                t = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t * best_h + (1.0 - t) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

                cur_best = c5_ng
                cur_h = h_ng
            h_ng2, c5_ng2 = _nesterov_opt(cur_h, n, target_sum, 300, 0.15, 100.0, 12, deadline - 60)
            if c5_ng2 < cur_best:
                cur_best = c5_ng2
                cur_h = h_ng2

        # Phase 2: Coordinate descent (fast version)
        if time.time() < deadline - 50:
            h_cd, c5_cd = _coord_descent_fast(cur_h, n, target_sum, 15, deadline - 50)
            if c5_cd < cur_best:
                cur_best = c5_cd
                cur_h = h_cd

        # Phase 3: Nesterov again with finer params
        if time.time() < deadline - 50:
            h_ng3, c5_ng3 = _nesterov_opt(cur_h, n, target_sum, 200, 0.1, 150.0, 15, deadline - 50)
            if c5_ng3 < cur_best:
                cur_best = c5_ng3
                cur_h = h_ng3

        # Phase 4: More coordinate descent
        if time.time() < deadline - 40:
            h_cd2, c5_cd2 = _coord_descent_fast(cur_h, n, target_sum, 20, deadline - 40)
            if c5_cd2 < cur_best:
                cur_best = c5_cd2
                cur_h = h_cd2

        # Phase 5: SLSQP polish
        if time.time() < deadline - 30 and n <= 200:
            h_sl = _slsqp_refine(cur_h, n, target_sum, 200, 500)
            c5_sl = _c5_of(h_sl)
            if c5_sl < cur_best:
                cur_best = c5_sl
                cur_h = h_sl
            h_cd3, c5_cd3 = _coord_descent_fast(cur_h, n, target_sum, 10, deadline - 30)
            if c5_cd3 < cur_best:
                cur_best = c5_cd3
                cur_h = h_cd3

        if cur_best < best_c5:
            best_c5 = cur_best
            best_h = cur_h.copy()
            best_n = n

    # Final polish on best solution
    if best_n is not None and time.time() < deadline - 10:
        target = best_n / 2.0
        h_fin = best_h.copy()

        if time.time() < deadline - 15:
            h_cdf, c5_cdf = _coord_descent_fast(h_fin, best_n, target, 30, deadline - 15)
            if c5_cdf < best_c5:
                best_c5 = c5_cdf
                h_fin = h_cdf

        if time.time() < deadline - 8 and best_n <= 200:
            h_nga, c5_nga = _nesterov_opt(h_fin, best_n, target, 200, 0.08, 200.0, 15, deadline - 8)
            if c5_nga < best_c5:
                best_c5 = c5_nga
                h_fin = h_nga

        if time.time() < deadline - 5:
            h_cdf2, c5_cdf2 = _coord_descent_fast(h_fin, best_n, target, 20, deadline - 5)
            if c5_cdf2 < best_c5:
                best_c5 = c5_cdf2
                h_fin = h_cdf2

        if time.time() < deadline - 3 and best_n <= 180:
            h_slf = _slsqp_refine(h_fin, best_n, target, 300, 2000)
            c5_slf = _c5_of(h_slf)
            if c5_slf < best_c5:
                best_c5 = c5_slf
                h_fin = h_slf

        best_h = h_fin

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
