def swap_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    deltas = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    
    while time.time() - start < time_limit:
        improved = False
        # Random pair of positions
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.choice(deltas) * rng.choice([-1, 1])
        
        # Try: h[i] += d, h[j] -= d (preserves integral)
        if 0 <= best[i] + d <= 1 and 0 <= best[j] - d <= 1:
            h_new = best.copy()
            h_new[i] += d
            h_new[j] -= d
            s = compute_c5(h_new)
            if s < best_score - 1e-14:
                best = h_new
                best_score = s
                improved = True
        
        if not improved and rng.rand() < 0.01:
            # Occasionally try larger moves
            d = rng.uniform(0.01, 0.1) * rng.choice([-1, 1])
            i, j = rng.choice(n, size=2, replace=False)
            if 0 <= best[i] + d <= 1 and 0 <= best[j] - d <= 1:
                h_new = best.copy()
                h_new[i] += d
                h_new[j] -= d
                s = compute_c5(h_new)
                if s < best_score - 1e-14:
                    best = h_new
                    best_score = s
    
    return best, best_score


def coordinate_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    deltas = [0.001, -0.001, 0.002, -0.002, 0.005, -0.005, 0.01, -0.01]
    no_improve = 0
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_new = best.copy()
                h_new[i] = val
                h_new = project_feasible(h_new, target)
                s = compute_c5(h_new)
                if s < best_score - 1e-14:
                    best = h_new
                    best_score = s
                    improved = True
                    break
        no_improve = 0 if improved else no_improve + 1
        if not improved:
            deltas = [d * 0.7 for d in deltas]
            if max(abs(d) for d in deltas) < 1e-6:
                break
    
    return best, best_score


def slsqp_refine(h, target, time_limit, alpha=50):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    remaining = time_limit - (time.time() - start)
    maxiter = int(max(50, remaining * 2))
    
    try:
        res = minimize(obj, best, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new
    except Exception:
        pass
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if not np.any(mask):
            return h
        h[mask] += diff / np.sum(mask)
    else:
        mask = h > 0.0
        if not np.any(mask):
            return h
        h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_weighted_gradient(h, weights_full):
    n = len(h)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = weights_full[m]
        if wm < 1e-14:
            continue
        i_low = max(0, m - (n - 1))
        i_high = min(m, n - 1)
        if i_high < i_low:
            continue
        m_idx = m - np.arange(i_low, i_high + 1)
        grad[i_low:i_high + 1] += wm * (1.0 - 2.0 * h[m_idx])
    return grad


def softmax_over_lags(corr, alpha, top_k=None):
    n = len(corr)
    if top_k is not None and top_k < n:
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(n)
        vals = corr[idx]
        w[idx] = np.exp(alpha * (vals - vals.max()))
        s = w.sum()
        if s > 1e-300:
            w[idx] /= s
        else:
            w[np.argmax(corr)] = 1.0
        return w
    else:
        w = np.exp(alpha * (corr - corr.max()))
        s = w.sum()
        if s > 1e-300:
            w /= s
        else:
            w = np.zeros(n)
            w[np.argmax(corr)] = 1.0
        return w


def nesterov_phase(h, target, time_limit, alpha_start=5.0, lr_start=0.3,
                    momentum=0.85, top_k=100, alpha_growth=1.003):
    n = len(h)
    start = time.time()
    best_h = h.copy()
    best_score = compute_c5(h)
    v = np.zeros(n)
    lr = lr_start
    alpha = alpha_start
    stall = 0
    max_stall = 100
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        weights = softmax_over_lags(corr, alpha, top_k=top_k)
        grad = compute_weighted_gradient(h, weights)
        
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        
        score = compute_c5(h_new)
        if score < best_score - 1e-13:
            best_score = score
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.005, 1.0)
        else:
            stall += 1
            if stall > max_stall:
                lr *= 0.8
                alpha *= 1.5
                v *= 0.5
                stall = 0
                if lr < 1e-4:
                    break
        
        h = h_new
        alpha *= alpha_growth
        alpha = min(alpha, 500.0)
        lr = max(lr * 0.9995, 0.003)
    
    return best_h, best_score


def swap_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    
    while time.time() - start < time_limit:
        i, j = rng.choice(n, size=2, replace=False)
        d = rng.choice([0.001, 0.002, 0.005, 0.01, 0.02, 0.05]) * rng.choice([-1, 1])
        
        if 0 <= best[i] + d <= 1 and 0 <= best[j] - d <= 1:
            h_new = best.copy()
            h_new[i] += d
            h_new[j] -= d
            s = compute_c5(h_new)
            if s < best_score - 1e-14:
                best = h_new
                best_score = s
    
    return best, best_score


def coordinate_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    deltas = [0.002, -0.002, 0.005, -0.005, 0.01, -0.01]
    no_improve_sweeps = 0
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_new = best.copy()
                h_new[i] = val
                h_new = project_feasible(h_new, target)
                s = compute_c5(h_new)
                if s < best_score - 1e-14:
                    best = h_new
                    best_score = s
                    improved = True
                    break
        no_improve_sweeps = 0 if improved else no_improve_sweeps + 1
        if not improved:
            deltas = [d * 0.6 for d in deltas]
            if max(abs(d) for d in deltas) < 1e-6:
                break
    
    return best, best_score


def slsqp_refine(h, target, time_limit, alpha=50):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    bounds = [(0.0, 1.0)] * n
    
    remaining = time_limit - (time.time() - start)
    maxiter = int(max(50, remaining * 3))
    
    try:
        res = minimize(obj, best, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new
    except Exception:
        pass
    
    return best, best_score
