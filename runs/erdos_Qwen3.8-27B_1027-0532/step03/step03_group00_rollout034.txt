```python
def slp_step(h, target, k_active=10, step_size=1.0):
    """One SLP iteration: linearize and solve LP."""
    n = len(h)
    corr = compute_corr(h)
    
    # Get active set (top k_active correlations)
    top_idx = np.argsort(corr)[-k_active:]
    
    # Compute linearization of corr[k] at h
    # corr[k](h + d) ≈ corr[k](h) + sum_j grad_corr[k,j] * d[j]
    # where d = h_new - h
    
    # Build the LP:
    # Variables: x = [d[0], ..., d[n-1], t] (length n+1)
    # Objective: minimize t
    # Constraints:
    #   For each active k: sum_j grad_corr[k,j] * d[j] <= t - corr[k](h)
    #   0 <= h[j] + d[j] <= 1 for all j  =>  -h[j] <= d[j] <= 1-h[j]
    #   sum(h) + sum(d) = target  =>  sum(d) = target - sum(h) = 0 (if h is already feasible)
    
    # Compute gradients for active constraints
    c = n - 1  # center index of full correlation
    A_rows = []
    b_vals = []
    
    for m in top_idx:
        k = m - c
        grad_k = np.zeros(n)
        if k >= 0:
            n_valid = n - k
            grad_k[k:] = (1.0 - h[:n_valid])  # (1-h[j-k]) for j in [k, n-1]
            grad_k[:n_valid] -= h[k:]  # -h[j+k] for j in [0, n-1-k]
        else:
            kp = -k
            n_valid = n - kp
            grad_k[:n_valid] = (1.0 - h[kp:])  # (1-h[j+kp]) for j in [0, n-1-kp]
            grad_k[kp:] -= h[:n_valid]  # -h[j-kp] for j in [kp, n-1]
        
        # Constraint: sum_j grad_k[j] * d[j] - t <= -corr[m]
        # i.e., sum_j grad_k[j] * d[j] - t <= -corr[m]
        row = np.concatenate([grad_k, [-1.0]])
        A_rows.append(row)
        b_vals.append(-corr[m])
    
    # Box constraints: -h[j] <= d[j] <= 1-h[j]
    # sum(d) = 0 (maintain feasibility)
    
    from scipy.optimize import linprog
    
    # Objective: minimize t => c_vec = [0, 0, ..., 0, 1]
    c_vec = np.zeros(n + 1)
    c_vec[-1] = 1.0
    
    # A_ub * x <= b_ub
    A_ub = np.array(A_rows) if A_rows else None
    b_ub = np.array(b_vals) if b_vals else None
    
    # Bounds for d[j]: -h[j] <= d[j] <= 1-h[j]
    # Bounds for t: -inf <= t <= inf (or we can set a reasonable upper bound)
    bounds = [(-h[j], 1.0 - h[j]) for j in range(n)]
    bounds.append((None, None))
    
    # Equality constraint: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    try:
        result = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        if result.success:
            d = result.x[:n]
            t_new = result.x[n]
            h_new = project_feasible(h + step_size * d, target)
            return h_new
    except:
        pass
    
    return h
```

```python
import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def correlation_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        nv = n - k
        if nv > 0:
            grad[k:] += (1.0 - h[:nv])
            grad[:nv] -= h[k:]
    else:
        kp = -k
        nv = n - kp
        if nv > 0:
            grad[:nv] += (1.0 - h[kp:])
            grad[kp:] -= h[:nv]
    return grad

def lse_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    obj = c_max + np.log(total) / alpha
    weights = exp_s / total
    grad = np.zeros(n)
    active = np.where(weights > 1e-6)[0]
    for m in active:
        w = weights[m]
        k = m - (n - 1)
        grad += w * correlation_gradient(h, k)
    return obj, grad

def fista_optimize(h0, target, alpha, n_iters, lr, deadline):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    prev_obj = lse_obj_grad(h, alpha)[0]
    for i in range(n_iters):
        if time.time() > deadline:
            break
        obj, grad = lse_obj_grad(y, alpha)
        y_new = y - lr * grad
        z = project_feasible(y_new, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = z + ((t_k - 1.0) / t_new) * (z - h)
        h = z
        t_k = t_new
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
    return best_h, best_c5

def slp_polish(h, target, k_active, n_iters, deadline, step_frac=1.0):
    n = len(h)
    c = n - 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-k_active:]
        A_rows = []
        b_vals = []
        for m in top_idx:
            k = m - c
            gk = correlation_gradient(best_h, k)
            row = np.concatenate([gk, [-1.0]])
            A_rows.append(row)
            b_vals.append(-corr[m])
        A_ub = np.array(A_rows)
        b_ub = np.array(b_vals)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        bounds = [(-best_h[j], 1.0 - best_h[j]) for j in range(n)]
        bounds.append((-1.0, 1.0))
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                d = res.x[:n] * step_frac
                h_new = project_feasible(best_h + d, target)
                sc = compute_c5(h_new)
                if sc < best_c5 - 1e-12:
                    best_c5 = sc
                    best_h = h_new
        except Exception:
            pass
    return best_h, best_c5

def binary_local_search(h, target, n_swaps, deadline, rng):
    n = len(h)
    h_bin = np.zeros(n)
    sorted_idx = np.argsort(h)[::-1]
    h_bin[sorted_idx[:n // 2]] = 1.0
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(best_h == 1.0)[0]
    zeros = np.where(best_h == 0.0)[0]
    for _ in range(n_swaps):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(200):
            if time.time() > deadline:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            oi = ones[i]
            oz = zeros[j]
            test = best_h.copy()
            test[oi] = 0.0
            test[oz] = 1.0
            sc = compute_c5(test)
            if sc < best_c5 - 1e-10:
                best_c5 = sc
                best_h = test
                ones = np.where(best_h == 1.0)[0]
                zeros = np.where(best_h == 0.0)[0]
                improved = True
                break
    return best_h, best_c5

def topk_smooth_obj(h, k):
    corr = compute_corr(h)
    sorted_corr = np.sort(corr)[::-1]
    return np.sum(sorted_corr[:k])

def slsqp_refine(h, target, maxiter=2000, ftol=1e-16):
    n = len(h)
    def obj(x):
        corr = compute_corr(x)
        alpha = 1000.0
        c_max = float(np.max(corr))
        shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return h
```
