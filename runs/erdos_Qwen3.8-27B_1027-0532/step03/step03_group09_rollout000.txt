```python
def weighted_obj(h, weights):
    corr = compute_corr(h)
    return float(np.sum(weights * corr))

def weighted_grad(h, weights):
    # gradient of sum(w_k * corr_k) w.r.t. h
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n-1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] -= wk * dx * h[:m]
    return grad
```

2. For each iteration:
   a. Solve SLSQP on weighted_obj
   b. Compute new weights based on corr values (e.g., w_k = corr_k^p / sum(corr^p) for some p)
3. Return best found

```python
def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    n = len(h_init)
    target_sum = n / 2.0
    h_cur = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    npoints = 2 * n - 1
    weights = np.ones(npoints) / npoints
    
    p = p_init
    
    for it in range(max_iter):
        def obj(x):
            return weighted_obj(x, weights)
        def grad(x):
            return weighted_grad(x, weights)
        def cfun(x):
            return float(np.sum(x) - target_sum)
        def cjac(x):
            return np.ones(n)

                          jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,

                          options={'maxiter': slsqp_iter, 'ftol': 1e-12})

            h_new = h_cur

def slsqp_smooth(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)
    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def gen_perturb(h_best, rng, n0, target_sum, strategy):

        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n0, size=max(3, n0 // 10), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.15 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        # Block swap
        max_len = max(2, n0 // 8)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def weighted_obj(h, weights):
    corr = compute_corr(h)
    return float(np.dot(weights, corr))

def weighted_grad(h, weights):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def reweighting_optimize(h_init, max_iter, slsqp_iter, p_init, p_growth, start_time, time_limit):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    npoints = 2 * n - 1
    corr_cur = compute_corr(h_cur)
    weights = corr_cur ** p_init
    ws = weights.sum()
    if ws < 1e-30:
        weights = np.ones(npoints) / npoints
    else:
        weights = weights / ws
    p = p_init
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        w_copy = weights.copy()
        def obj(x):
            return weighted_obj(x, w_copy)
        def grad(x):
            return weighted_grad(x, w_copy)
        def cfun(x):
            return float(np.sum(x) - target_sum)
        def cjac(x):
            return np.ones(n)
        try:
            result = minimize(obj, h_cur,
                          jac=grad, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': slsqp_iter, 'ftol': 1e-14})
            h_new = project_onto_constraint(result.x, target_sum)
        except Exception:
            h_new = h_cur
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_cur = h_new
        corr_cur = compute_corr(h_cur)
        weights = corr_cur ** p
        ws = weights.sum()
        if ws < 1e-30:
            weights = np.ones(npoints) / npoints
        else:
            weights = weights / ws
        p = min(p * p_growth, 5000.0)
    return best_h, best_c5

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:

        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def proj_grad_topk(h, k, temp, lr_init, n_steps, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    stall_limit = 400
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        grad = topk_subgrad(h_cur, k, temp)
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.03, 0.25)
        else:
            stall += 1
            lr = max(lr * 0.94, 1e-8)
        h_cur = h_new
        if stall > stall_limit:
            h_cur = best_h.copy()
            lr = max(lr_init * 0.5, 0.005)
            stall = 0
            stall_limit = max(150, stall_limit // 2)
    return best_h, best_c5

def slsqp_smooth(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    def cfun(x):
        return float(np.sum(x) - target_sum)
    def cjac(x):
        return np.ones(n)
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def gen_perturb(h_best, rng, n0, target_sum, strategy):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n0, size=max(3, n0 // 10), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.12 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        max_len = max(2, n0 // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
        h_bin = (rng.random(n0) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.6, 0.9))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    elif strategy == 4:
        n_half = n0 // 2
        h_rev = np.zeros(n0)
        h_rev[:n_half] = h_best[n0 - n_half:]
        h_rev[n0 - n_half:] = h_best[:n_half]
        alpha = float(rng.uniform(0.5, 0.8))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_rev, target_sum)
    else:
        return project_onto_constraint(h_best + 0.005 * rng.standard_normal(n0), target_sum)
```
