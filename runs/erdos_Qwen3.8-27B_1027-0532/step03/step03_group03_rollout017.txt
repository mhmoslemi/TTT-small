def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):

    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):

    _CONS_TARGET = target_sum

    bounds = [(0.0, 1.0)] * _CONS_N

            if compute_c5(h_new) < compute_c5(h_cur):

def pairwise_local_search(h_start, target_sum, max_iter, temporal_steps, seed, deadline=None):
    rng = np.random.RandomState(seed)
    n = len(h_start)

            break
        corr = overlap_corrs(h)
        worst_lag_idx = np.argmax(corr)
        worst_lag = worst_lag_idx - (n - 1)

            valid_i = np.arange(0, n - worst_lag)
        else:
            m = -worst_lag
            valid_i = np.arange(m, n)

        if len(valid_i) == 0:
            break

        if worst_lag >= 0:
            contributions = h[valid_i] * (1.0 - h[valid_i + worst_lag])
        else:
            m = -worst_lag
            contributions = h[valid_i] * (1.0 - h[valid_i - m])

        topk_pairs = min(20, len(valid_i))

    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    n_sg_configs = [
        (3000, 0.12, 3, 30, 0.04, 0.015, 250, 0.92),
        (3000, 0.20, 1, 10, 0.08, 0.03, 200, 0.90),
        (3000, 0.06, 5, 50, 0.02, 0.01, 300, 0.88),
        (2000, 0.15, 2, 20, 0.06, 0.02, 200, 0.93),
    ]

    for i, cfg in enumerate(n_sg_configs):
        if _remaining() < 60:
            break
        h_try = best_h + rng.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
                                        alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
                                        patience=cfg[6], momentum=cfg[7], seed=seed + i * 7)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 150:
        deadline = start_time + budget_s - 35.0
        h_new, score = pairwise_local_search(best_h, target_init, max_iter=20000,
                                              temporal_steps=0.002, seed=seed + 100,
                                              deadline=deadline - 60.0)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 100:
        deadline = start_time + budget_s - 35.0
        h_new, score = multi_pair_search(best_h, target_init, max_iter=30000,
                                          seed=seed + 200, deadline=deadline - 40.0)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 120:
        alphas = [100, 300, 800, 1500, 3000]
        h_ref = slsqp_refine(best_h, target_init, alphas, 2000, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 100:
        deadline = start_time + budget_s - 35.0
        h_new, score = gradient_pair_search(best_h, target_init, max_iter=5000,
                                             seed=seed + 300, deadline=deadline - 30.0)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 80:
        deadline = start_time + budget_s - 35.0
        h_new, score = pairwise_local_search(best_h, target_init, max_iter=15000,
                                              temporal_steps=0.001, seed=seed + 400,
                                              deadline=deadline - 25.0)
        if score < best_score:
            best_score = score
            best_h = h_new

    for n_try in [500, 600, 800]:
        if _remaining() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_new, score = run_subgradient(h_interp, target_try,
                                        max_iter=2000, lr_init=0.08, topk=3,
                                        alpha=30, noise_prob=0.02, noise_scale=0.01,
                                        patience=200, momentum=0.90, seed=seed + 500 + n_try)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 60:
            alphas_hi = [200, 500, 1000]
            h_ref = slsqp_refine(best_h, len(best_h) / 2.0, alphas_hi, 1500, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if _remaining() > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_try = best_h + rng.randn(n_use) * 0.003
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                        max_iter=2000, lr_init=0.03, topk=1,
                                        alpha=80, noise_prob=0.0, noise_scale=0.0,
                                        patience=300, momentum=0.92, seed=seed + 600)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 50:
        alphas_final = [500, 1000, 2000, 5000]
        h_final = slsqp_refine(best_h, len(best_h) / 2.0, alphas_final, 3000, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if _remaining() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        deadline = start_time + budget_s - 35.0
        h_new, score = pairwise_local_search(best_h, target_use, max_iter=10000,
                                              temporal_steps=0.0005, seed=seed + 700,
                                              deadline=deadline)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
