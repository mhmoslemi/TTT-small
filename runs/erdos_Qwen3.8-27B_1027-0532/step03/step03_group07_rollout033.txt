import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    """Project onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx, n):
    """Gradient of corr[k_idx] with respect to h."""
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_max_and_weights(corr, alpha, top_m=None):
    """Compute smooth max and optional top-m weights."""
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    smooth_val = m + np.log(total) / alpha
    weights = exp_vals / total
    if top_m is not None and top_m < len(corr):
        top_idx = np.argsort(weights)[::-1][:top_m]
        w_m = weights[top_idx].copy()
        w_m = w_m / w_m.sum()
        return smooth_val, w_m, top_idx
    return smooth_val, weights, np.arange(len(corr))

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha_schedule, use_top_m=False, top_m=5, deadl=None):
    """Nesterov-accelerated projected gradient with smooth max."""
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if deadl is not None and time.time() > deadl:
            break
        alpha = alpha_schedule(min(it, len(alpha_schedule)-1))
        corr = np.correlate(y, 1.0 - y, mode='full')
        if use_top_m:
            _, w_m, top_idx = _smooth_max_and_weights(corr, alpha, top_m=top_m)
            g = np.zeros(n)
            dx = 2.0 / n
            for idx in range(len(top_idx)):
                g += w_m[idx] * _grad_corr_term(y, top_idx[idx], dx, n)
        else:
            _, w, idx_all = _smooth_max_and_weights(corr, alpha)
            g = np.zeros(n)
            dx = 2.0 / n
            # Subsample if too many terms
            if len(idx_all) > 50:
                top50 = np.argsort(w)[::-1][:50]
                w50 = w[top50] / w[top50].sum()
                for idx in range(50):
                    g += w50[idx] * _grad_corr_term(y, idx_all[top50[idx]], dx, n)
            else:
                for idx in range(len(idx_all)):
                    if w[idx] > 1e-6:
                        g += w[idx] * _grad_corr_term(y, idx_all[idx], dx, n)
        
        lr = 1.0 / (alpha * np.sqrt(it + 1))
        x_new = _project(y - lr * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _cso_explore(h_best, n, target_sum, pop_size, n_iters, deadline, rng):
    """Cat Swarm Optimization inspired exploration."""
    pop = [h_best.copy()]
    # Initialize with diverse starts
    for i in range(pop_size - 1):
        # Mix of random and perturbed
        if i < pop_size // 2:
            h_i = _project(h_best + rng.randn(n) * 0.3, target_sum)
        else:
            h_i = np.clip(rng.rand(n), 0, 1)
            h_i = _fix_sum(h_i, target_sum)
        pop.append(h_i)
    
    scores = [_c5_of(h) for h in pop]
    best_h = pop[0].copy()
    best_score = min(scores)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Sort by score
        order = np.argsort(scores)
        sorted_pop = [pop[i] for i in order]
        sorted_scores = [scores[i] for i in order]
        
        new_pop = [sorted_pop[0]]  # Keep best
        new_scores = [sorted_scores[0]]
        
        for i in range(1, pop_size):
            if time.time() > deadline:
                new_pop.append(sorted_pop[i])
                new_scores.append(sorted_scores[i])
                continue
            
            rng_choice = rng.random()
            if rng_choice < 0.3:
                # Exploration mode - Cauchy perturbation
                cauchy_noise = rng.standard_cauchy(n)
                cauchy_noise = np.clip(cauchy_noise, -2, 2)
                perturbation = 0.5 * cauchy_noise
                candidate = _project(sorted_pop[i] + perturbation, target_sum)
            elif rng_choice < 0.7:
                # Recognition mode - guided by best
                phi = rng.uniform(-np.pi, np.pi)
                if i + 1 < pop_size:
                    follower = sorted_pop[i + 1]
                    direction = sorted_pop[0] - follower
                else:
                    direction = sorted_pop[0] - sorted_pop[rng.randint(1, pop_size)]
                cos_factor = np.cos(phi)
                sin_factor = np.sin(phi)
                candidate = sorted_pop[i] + 0.3 * (cos_factor * direction + sin_factor * rng.randn(n))
                candidate = _project(candidate, target_sum)
            else:
                # Pounce mode - local search
                leader = sorted_pop[rng.choice(min(3, pop_size))]
                n_perturb = max(1, n // 10)
                candidate = leader.copy()
                idx = rng.choice(n, size=n_perturb, replace=False)
                for j in idx:
                    candidate[j] = np.clip(candidate[j] + rng.normal(0, 0.1), 0, 1)
                candidate = _fix_sum(candidate, target_sum)
            
            score = _c5_of(candidate)
            if score <= sorted_scores[i]:
                new_pop.append(candidate)
                new_scores.append(score)
                if score < best_score:
                    best_score = score
                    best_h = candidate.copy()
            else:
                # Keep old if perturbation didn't help, add small noise
                noise = rng.randn(n) * 0.05
                candidate2 = _project(sorted_pop[i] + noise, target_sum)
                new_pop.append(candidate2)
                new_scores.append(_c5_of(candidate2))
        
        pop = new_pop
        scores = new_scores
    
    return best_h, best_score

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):
    """SLSQP with smooth max objective."""
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_vals = np.exp(alpha * (corr - m))
        w = exp_vals / exp_vals.sum()
        dx = 2.0 / n
        g = np.zeros(n)
        # Only use significant weights for efficiency
        thresh = w.max() * 0.01
        active = w > thresh
        if active.sum() == 0:
            active[np.argmax(w)] = True
        w_active = w[active] / w[active].sum()
        for k in np.where(active)[0]:
            g += w_active[np.where(active)[0].tolist().index(k)] * _grad_corr_term(x, k, dx, n)
        return g * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _slsqp_opt_simple(h0, n, target_sum, alpha, maxiter=300):
    """Simpler SLSQP that uses finite differences to avoid gradient issues."""
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum}]
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent_binary(h0, n, target_sum, max_rounds=30):
    """Binary coordinate descent: set each coordinate to 0 or 1."""
    h = h0.copy()
    current_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            if c0 < current_c5 - 1e-14:
                h[j] = 0.0
                current_c5 = c0
                improved = True
            elif c1 < current_c5 - 1e-14:
                h[j] = 1.0
                current_c5 = c1
                improved = True
        if not improved:
            break
    return h

def _coord_descent_values(h0, n, target_sum, max_rounds=10, values=None):
    """Coordinate descent with a set of trial values."""
    if values is None:
        values = [0.0, 0.1, 0.2, 0.25, 1.0/3.0, 0.4, 0.5, 0.6, 2.0/3.0, 0.75, 0.8, 0.9, 1.0]
    h = h0.copy()
    current_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = current_c5
            best_v = old
            for v in values:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                current_c5 = best_c
                improved = True
        if not improved:
            break
    return h

def _anneal_smooth(h0, n, target_sum, alphas, iters_per_stage, deadl):
    """Anneal through a sequence of alpha values."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_of(h)
    
    for alpha in alphas:
        if time.time() > deadl:
            break
        alpha_sched = lambda t: alpha
        h, c5 = _nesterov_smooth(h, n, target_sum, n_iters=iters_per_stage,
                                   alpha_schedule=alpha_sched,
                                   use_top_m=True, top_m=5, deadl=deadl)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def _c5_of(h):

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

def _grad_corr_term(h, k_idx, dx, n):
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

    smooth_val = m + np.log(total) / alpha
    weights = exp_vals / total
    return smooth_val, weights

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):

            g += w_a[kk] * _grad_corr_term(x, active_idx[kk], dx, n)
        return g * dx

        return x.sum() - target_sum

        return np.ones(n)
