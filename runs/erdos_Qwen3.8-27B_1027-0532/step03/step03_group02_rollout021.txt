import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    
    s_arr = np.arange(2*n-1) - (n-1)
    
    while time.time() - start < max_time:
        for j in range(n):
            old = h[j]
            # Compute A vector: how corr changes when h[j] changes
            # corr[s] depends on h[j] through:
            # Term 1: h[j] * (1-h[j+s]) for valid j+s → coefficient (1-h[j+s])
            # Term 2: h[j-s] * (1-h[j]) for valid j-s → coefficient -h[j-s]
            A = np.zeros(2*n-1)
            # j+s valid: 0 <= j+s < n → s >= -j and s < n-j
            s1 = np.arange(max(0, -j), min(n-1-j, n-1) + 1)
            for s in s1:
                idx = s + (n-1)  # index in corr array
                if 0 <= j+s < n:
                    A[idx] += (1.0 - h[j+s])
            # j-s valid: 0 <= j-s < n → s <= j and s > j-n
            s2 = np.arange(max(0-(j-n), j+1))  # s from max(-(j-n), 0-n+1) to j
            for s in s2:
                idx = s + (n-1)
                if 0 <= j-s < n:
                    A[idx] -= h[j-s]
            
            # B = corr - A * old
            B = corr - A * old
            # Binary search for optimal h[j]
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                if A[k_star] > 1e-12:
                    hi = mid
                elif A[k_star] < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val
        
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            break
    
    return best_h, best_score

def pair_optimize(h, i, j, target_sum):
    """Optimize h[i] and h[j] jointly, keeping all else fixed."""
    n = len(h)
    c = target_sum - np.sum(h) + h[i] + h[j]  # required sum of h[i] + h[j]
    if c < 0 or c > 2:
        return h  # infeasible
    
    lo_t = max(0.0, c - 1.0)
    hi_t = min(1.0, c)
    if lo_t >= hi_t:
        t = (lo_t + hi_t) / 2.0
        h[i] = t
        h[j] = c - t
        return h
    
    # Save old values
    old_i = h[i]
    old_j = h[j]
    
    def obj(t):
        h[i] = t
        h[j] = c - t
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo_t, hi_t
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = obj(c_pt)
    fd = obj(d_pt)
    
    for _ in range(80):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = obj(c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = obj(d_pt)
    
    t_opt = (a + b) / 2.0
    h[i] = t_opt
    h[j] = c - t_opt
    return h

        else:
            idx = np.where(h > 0.0)[0]

            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)

        h = np.clip(h, 0.0, 1.0)

        for j in order:
            old = h[j]
            # Vectorized computation of A

**Improved pair selection**: After each successful improvement, compute which shifts are active (close to the max). Then bias the pair selection towards coordinates that are "sensitive" to those shifts.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    while time.time() - start < max_time:
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            A = np.zeros(2 * n - 1)
            s1 = np.arange(-j, n - j)
            v1 = (j + s1 >= 0) & (j + s1 < n)
            A[s1[v1] + n - 1] += (1.0 - h[j + s1[v1]])
            s2 = np.arange(j - n + 1, j + 1)
            v2 = (j - s2 >= 0) & (j - s2 < n)
            A[s2[v2] + n - 1] -= h[j - s2[v2]]
            B = corr - A * old
            lo, hi = 0.0, 1.0
            for _ in range(28):
                mid = 0.5 * (lo + hi)
                k = np.argmax(A * mid + B)
                g = A[k]
                if g > 1e-12:
                    hi = mid
                elif g < -1e-12:
                    lo = mid
                else:
                    break
            nv = 0.5 * (lo + hi)
            d = nv - old
            if abs(d) > 1e-14:
                corr += d * A
                h[j] = nv
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            break
    return best_h, best_score

    n = len(h)
    h = h.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    while time.time() - start < max_time:
        r = rng.random()
        if r < 0.75:
            i = rng.randint(n)
            offset = rng.randint(1, min(30, n))
            j = (i + offset) % n
        else:
            i = rng.randint(n)
            j = rng.randint(n)
        if i == j:
            continue
        if i > j:
            i, j = j, i
        if h[i] >= 0.999 and h[j] >= 0.999:
            continue
        if h[i] <= 0.001 and h[j] <= 0.001:
            continue
        c = target_sum - (h.sum() - h[i] - h[j])
        c = np.clip(c, 0.0, 2.0)
        lo_t = max(0.0, c - 1.0)
        hi_t = min(1.0, c)
        if hi_t - lo_t < 1e-12:
            continue
        a, b = lo_t, hi_t
        c1 = b - gr * (b - a)
        d1 = a + gr * (b - a)
        h[i] = c1

        f_c1 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
        h[i] = d1

                d1 = c1

                c1 = b - gr * (b - a)

                h[j] = c - c1
                f_c1 = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            else:

                c1 = d1

def subgradient_step(h, target_sum, lr, rng):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    active = corr > c_max - 1e-5
    n_act = int(np.sum(active))
    if n_act == 0:
        active = np.zeros(2 * n - 1)
        active[np.argmax(corr)] = 1
        n_act = 1
    g = np.zeros(n)
    for idx in np.where(active)[0]:
        s_val = s_arr[idx]
        contrib = np.zeros(n)
        j_idx = np.arange(n)
        js = j_idx - s_val
        js_p = j_idx + s_val
        m1 = (js >= 0) & (js < n)
        m2 = (js_p >= 0) & (js_p < n)
        contrib[m1] += (1.0 - h[js[m1]])
        contrib[m2] -= h[js_p[m2]]
        g += contrib / n_act
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    return h_new

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_refine(h, target_sum, alphas, maxiter_each, time_limit):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            def obj_func(x, a=alpha):
                return smooth_obj(x, a)
            def cons_fun(x):
                return float(np.sum(x) - target_sum)
            def cons_jac(x):
                return np.ones(len(x))
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score
