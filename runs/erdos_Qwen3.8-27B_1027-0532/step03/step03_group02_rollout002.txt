def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    # Clip and adjust sum
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30, verbose=False):
    """Coordinate descent with exact line search for each coordinate."""
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    
    while time.time() - start < max_time:
        improved = False
        for j in range(n):
            old_val = h[j]
            # Compute A vector: d corr[k] / d h[j]
            js = j - s_arr  # j - (k - (n-1)) = j - k + n - 1
            jp = j + s_arr  # j + (k - (n-1)) = j + k - n + 1
            A = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n)
            m2 = (jp >= 0) & (jp < n)
            A[m1] += (1.0 - h[js[m1]])
            A[m2] -= h[jp[m2]]
            B = corr - A * old_val
            # Binary search for optimal v'
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) * 0.5
                idx = np.argmax(B + A * mid)
                g = A[idx]
                if g > 1e-9:
                    hi = mid
                elif g < -1e-9:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val
                improved = True
        
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            # Perturb and restart
            scale = 0.01 * min(3.0, no_improve / 10.0)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
            if time.time() - start > max_time:
                break
    
    return best_h, best_score

def smooth_obj_and_grad(h, alpha):
    """Compute smooth approximation of max(corr) and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.exp(alpha * (corr - c_max))
    weights = shifted / np.sum(shifted)
    obj = c_max + np.log(np.sum(shifted)) / alpha
    
    # Gradient: sum_k weights[k] * d corr[k] / d h[j]
    # d corr[k] / d h[j] = (1-h[j+s_k]) * I[valid] - h[j-s_k] * I[valid]
    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        s = s_arr[k]
        # (1-h[j+s]) for valid j
        if s >= 0:
            j_range = range(n - s)
        else:
            j_range = range(-s, n)
        # First term: (1-h[j+s])
        js = np.array(list(j_range))
        grad[js] += weights[k] * (1.0 - h[js + s])
        # Second term: -h[j-s]
        if s >= 0:
            j_range2 = range(s, n)  # j-s >= 0 means j >= s
        else:
            j_range2 = range(0, n + s)  # j-s < n means j < n+s
        js2 = np.array(list(j_range2))
        grad[js2] -= weights[k] * h[js2 - s]
    
    return obj, grad
