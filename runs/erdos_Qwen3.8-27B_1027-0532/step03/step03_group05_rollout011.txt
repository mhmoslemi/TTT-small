import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h


def pairwise_cd(h, target, time_limit, seed=42, step_list=None):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(best_corr.max()) * dx

    if step_list is None:
        step_list = [0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    si = 0
    no_improve = 0
    patience = 100

    while time.time() - start < time_limit:
        improved = False
        batch_size = 400
        for _ in range(batch_size):
            if time.time() - start >= time_limit:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            step = step_list[si]
            if best[i] + step <= 1.0 and best[j] - step >= 0.0:
                best[i] += step
                best[j] -= step
                corr = np.correlate(best, 1.0 - best, mode='full')
                s = float(corr.max()) * dx
                if s < best_score:
                    best_score = s
                    best_corr = corr
                    improved = True
                    no_improve = 0
                    continue
                else:
                    best[i] -= step
                    best[j] += step
            if best[i] - step >= 0.0 and best[j] + step <= 1.0:
                best[i] -= step
                best[j] += step
                corr = np.correlate(best, 1.0 - best, mode='full')
                s = float(corr.max()) * dx
                if s < best_score:
                    best_score = s
                    best_corr = corr
                    improved = True
                    no_improve = 0
                    continue
                else:
                    best[i] += step
                    best[j] -= step
        no_improve += 1
        if no_improve >= patience and si < len(step_list) - 1:
            si += 1
            no_improve = 0
            patience = max(50, patience // 2)

    return best, best_score


def block_cd(h, target, time_limit, seed=42, block_size=3):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    steps = [0.05, 0.025, 0.01, 0.005, 0.002, 0.001]
    si = 0
    no_improve = 0

    while time.time() - start < time_limit:
        improved = False
        for _ in range(200):
            if time.time() - start >= time_limit:
                break
            i = rng.randint(0, n - block_size + 1)
            j = rng.randint(0, n - block_size + 1)
            if abs(i - j) < block_size:
                continue
            step = steps[si] * rng.choice([1.0, -1.0])
            h_trial = best.copy()
            feasible = True
            for b in range(block_size):
                if h_trial[i + b] + step < 0.0 or h_trial[i + b] + step > 1.0:

                    break

                    feasible = False
                    break
            if not feasible:

            s = compute_c5(h_trial)

                best = h_trial
                improved = True
                no_improve = 0

            no_improve = 0

def slsqp_refine(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    if alphas is None:
        alphas = [10, 30, 100, 300, 800]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    n_alpha = len(alphas)
    for ai, alpha in enumerate(alphas):
        remaining = time_limit - (time.time() - start)
        if remaining < 15:
            break
        time_per = remaining / (n_alpha - ai)
        maxiter = int(max(50, time_per * 5))

        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

        def obj_grad(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = a * (corr - c_max)
            shifted = np.clip(shifted, -500.0, 0.0)
            weights = np.exp(shifted)
            weights /= weights.sum()
            g = np.zeros(2 * n - 1)
            for lag in range(n):
                ci = n - 1 + lag
                w = weights[ci]
                if w < 1e-10:
                    continue
                g[:n - lag] += w * (1.0 - x[lag:])
                g[lag:] -= w * x[:n - lag]
                ci2 = n - 1 - lag
                if ci2 != ci and ci2 >= 0 and ci2 < 2 * n - 1:
                    w2 = weights[ci2]
                    if w2 < 1e-10:
                        continue
                    idx_lo = max(0, lag)
                    idx_hi = n
                    g[idx_lo:idx_hi] -= w2 * x[idx_lo:idx_hi]
                    g[idx_hi - lag:] += w2 * (1.0 - x[:idx_hi - lag])
            return g

        try:
            res = minimize(obj, best, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def gradient_descent_phase(h, target, time_limit, seed=42):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    alpha = 30.0
    eta = 0.005
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        c_max = float(corr.max())
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-15:
            break
        weights /= wsum

        grad = np.zeros(n)
        for lag in range(n):
            w = weights[n - 1 + lag]
            if w < 1e-10:
                break
            grad[:n - lag] += w * (1.0 - best[lag:])
            grad[lag:] -= w * best[:n - lag]
            if lag > 0:
                w2 = weights[n - 1 - lag]
                if w2 < 1e-10:
                    continue
                grad[lag:] -= w2 * best[lag:]
                grad[:n - lag] += w2 * (1.0 - best[:n - lag])

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            break

        h_trial = best + eta * grad / grad_norm
        h_trial = project_feasible(h_trial, target)
        s = compute_c5(h_trial)
        if s < best_score:
            best_score = s
            best = h_trial
            no_improve = 0
            eta = min(eta * 1.02, 0.03)
        else:
            no_improve += 1
            if no_improve > 40:
                eta *= 0.7
                alpha *= 1.5
                no_improve = 0
                if eta < 1e-5:
                    break

    return best, best_score

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    return np.interp(x_new, x_old, h, period=1)
