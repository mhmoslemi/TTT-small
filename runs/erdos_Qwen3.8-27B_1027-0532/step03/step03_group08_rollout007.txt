```python
    new_overlaps = overlaps.copy()
```

```python
def _swap_delta(h, a, b, n):
    """Compute delta in raw correlation values for swapping h[a]:1->0, h[b]:0->1"""
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    delta = np.zeros(2*n-1)
    
    # a in first position: -(1-h[a+k]) for valid k
    ak = a + lags
    mask = (ak >= 0) & (ak < n)
    delta[mask] -= (1.0 - h[ak[mask]])
    
    # a in second position: +h[a-k] for valid k
    ak2 = a - lags
    mask2 = (ak2 >= 0) & (ak2 < n)
    delta[mask2] += h[ak2[mask2]]
    
    # b in first position: +(1-h[b+k]) for valid k
    bk = b + lags
    mask_b = (bk >= 0) & (bk < n)
    delta[mask_b] += (1.0 - h[bk[mask_b]])
    
    # b in second position: -h[b-k] for valid k
    bk2 = b - lags
    mask_b2 = (bk2 >= 0) & (bk2 < n)
    delta[mask_b2] -= h[bk2[mask_b2]]
    
    # Corrections for shared terms
    # At lag k = b-a: correction +1
    idx_ba = (b - a) + (n - 1)
    delta[idx_ba] += 1.0
    # At lag k = a-b: correction +1
    idx_ab = (a - b) + (n - 1)
    delta[idx_ab] += 1.0
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _all_overlaps_raw(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum_binary(h, n):
    target = n // 2
    h = np.round(h).clip(0, 1).astype(float)
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[:n_ones - target]:
            h[idx] = 0.0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1.0
    return h

def _swap_delta_vec(h, a, b, n, lags):
    delta = np.zeros(2 * n - 1)
    ak = a + lags
    m = (ak >= 0) & (ak < n)
    delta[m] -= (1.0 - h[ak[m]])
    ak2 = a - lags
    m2 = (ak2 >= 0) & (ak2 < n)
    delta[m2] += h[ak2[m2]]
    bk = b + lags

    bk2 = b - lags

    delta[mb2] -= h[bk2[mb2]]
    idx_ba = (b - a) + (n - 1)
    delta[idx_ba] += 1.0
    idx_ab = (a - b) + (n - 1)
    delta[idx_ab] += 1.0
    return delta

    lags = np.arange(-(n - 1), n)

            break

        delta = _swap_delta_vec(h, a, b, n, lags)
        new_raw = raw + delta

    h = h_bin.copy()
    dx = 2.0 / n
    target = n / 2.0
    c5_cur = _c5_of(h)
    c5_best = c5_cur
    h_best = h.copy()

    raw = _all_overlaps_raw(h)

    # If h[i] is 1 and h[i+k] is 0 at the worst lag, reducing h[i] or increasing h[i+k] helps

    n_trials = 2000
    for trial in range(n_trials):
        if time.time() > deadline:

            for i in range(n):

            h = h_new.copy()
```

```python
def _nesterov_refine(h0, n, n_iters, alpha0, deadline, top_m=5, sg_alpha=50.0):
    target = n / 2.0
    dx = 2.0 / n
    def project(y):
        lo, hi = y.min() - 1.0, y.max() + 1.0
        for _ in range(50):
            mid = (lo + hi) / 2.0
            s = np.clip(y - mid, 0, 1).sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        x = np.clip(y - (lo + hi) / 2.0, 0, 1)
        diff = target - x.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                order = np.argsort(1.0 - x)
                for i in order:
                    if diff <= 0:
                        break
                    add = min(diff, 1.0 - x[i])
                    x[i] += add
                    diff -= add
            else:
                order = np.argsort(x)
                for i in order:
                    if diff >= 0:
                        break
                    sub = min(-diff, x[i])
                    x[i] -= sub
                    diff += sub
        return x
    def grad_at(y):
        corr = np.correlate(y, 1.0 - y, mode='full')
        km = float(corr.max())
        sh = sg_alpha * (corr - km)
        sh = np.clip(sh, -700, 700)
        ws = np.exp(sh)
        ws /= ws.sum()
        top = np.argsort(ws)[::-1][:top_m]
        wm = ws[top].copy()
        wm /= wm.sum()
        lags = np.arange(-(n-1), n)
        g = np.zeros(n)
        for idx in range(len(top)):
            k = lags[top[idx]]
            gk = np.zeros(n)
            i1 = np.arange(max(0, -k), min(n, n-k))
            if len(i1) > 0:
                gk[i1] += (1.0 - y[i1 + k])
            i2 = np.arange(max(0, k), min(n, n+k))
            if len(i2) > 0:
                gk[i2] -= y[i2 - k]
            g += wm[idx] * gk
        return g * dx
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = h0.copy()
    best_val = _c5_of(h0)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = grad_at(y)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = project(y - step * g)

        y = y_new
        t = t_new
        cv = _c5_of(x)

            best = x.copy()

    # Phase 4: Final binary refinement at the continuous solution's support
    if time.time() < deadline - 15:
        h_round = _fix_binary(h_cont, best_n)
        c5_round = _c5_of(h_round)
        if c5_round < best_c5:
            best_c5 = c5_round
            best_h = h_round.copy()
        # Quick SA from rounded
        if time.time() < deadline - 5:
            h_sa2, c5_sa2 = _sa_binary_fast(h_round, best_n, 100000, 0.03, 0.001,
                                             min(deadline - 2, time.time() + 10), rng, 0.4)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Final SLSQP polish
    if time.time() < deadline - 3:
        h_final = _slsqp_refine(best_h, best_n, 400, 350, min(deadline - 1, time.time() + 8))
        c5_final = _c5_of(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = _fix_sum_general(best_h, best_n)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
```
