    best_score = compute_c5(h)
    alpha = alpha_init
    eta = eta_init
    no_improve = 0
    t = 0
    while time.time() - start < time_limit:
        t += 1
        corr = compute_corr_array(h_curr)
        max_val = np.max(corr)
        
        # Softmax weighting
        shifted = alpha * (corr - max_val)
        weights = np.exp(np.clip(shifted, -300, 0))
        sw = np.sum(weights)
        if sw < 1e-15:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= sw
        
        # Compute subgradient
        grad = np.zeros(n)
        n_lags = len(corr)
        center = n - 1
        for k_offset in range(n_lags):
            w = weights[k_offset]
            if w < 1e-7:
                continue
            lag = k_offset - center
            if lag >= 0:
                # f = sum_i h[i]*(1-h[i+lag]) for i=0..n-1-lag
                # d/dh[j] = (1-h[j+lag]) for j=0..n-1-lag, minus h[j-lag] for j=lag..n-1
                grad[:n-lag] += w * (1.0 - h_curr[lag:])
                if lag > 0:
                    grad[lag:] -= w * h_curr[:n-lag]
            else:
                lp = -lag
                grad[lp:] += w * (1.0 - h_curr[:n-lp])
                if lp > 0:
                    grad[:n-lp] -= w * h_curr[lp:]
        
        # Momentum update
        v = mu * v + (1 - mu) * grad
        h_new = h_curr + eta * v
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        h_avg = h_avg + (h_new - h_avg) / t
        
        if score < best_score:
            best_score = score
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.8
                alpha = min(alpha * 1.3, alpha_max)
                no_improve = 0
        
        h_curr = h_new
        alpha = min(alpha * 1.001, alpha_max)
    
    # Also check the averaged iterate
    h_avg = project_feasible(h_avg, target)
    score_avg = compute_c5(h_avg)
    if score_avg < best_score:
        best_score = score_avg
        best = h_avg
    
    return best, best_score


def adaptive_slsqp(h, target, time_limit, alphas=(5, 10, 20, 50, 100, 200, 400)):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        
        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        
        def obj_grad(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            sw = np.sum(weights)
            if sw < 1e-15:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights /= sw
            # gradient of the log-sum-exp objective
            # = gradient of c_max/sw * sum(exp(...)) ... 
            # Actually d/dx [c_max + log(sum(exp(a*(corr-c_max))))/a]
            # = (1/a) * sum(exp(a*(corr_k - c_max))) * a * d(corr_k)/dx / sum(exp(a*(corr_k-c_max)))
            # = sum_k weights_k * d(corr_k)/dx
            # where corr_k is the k-th element of the correlation array
            # And c_max also depends on x, but at the optimum, the derivative of c_max 
            # is handled by the fact that we're at the max... 
            # Actually this is more subtle. Let me just use the subgradient approach.
            
            grad = np.zeros(n)
            center = len(x) - 1
            for k_offset in range(len(corr)):
                w = weights[k_offset]
                if w < 1e-7:
                    continue
                lag = k_offset - center
                if lag >= 0:
                    grad[:len(x)-lag] += w * (1.0 - x[lag:])
                    if lag > 0:
                        grad[lag:] -= w * x[:len(x)-lag]
                else:
                    lp = -lag
                    grad[lp:] += w * (1.0 - x[:len(x)-lp])
                    if lp > 0:
                        grad[:len(x)-lp] -= w * x[lp:]
            return grad
        
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        
        maxiter = max(50, int(remaining * 2))
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=obj_grad,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score


def block_coordinate_descent(h, target, time_limit, seed=42):
    """Try block moves of size 2-5 to escape local optima."""
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    block_sizes = [2, 3, 4, 5]
    
    while time.time() - start < time_limit:
        improved = False
        bs = rng.choice(block_sizes)
        n_trials = min(200, max(20, (time_limit - (time.time() - start)) * 5))
        for _ in range(n_trials):
            i = rng.randint(0, n - bs + 1)
            block = best[i:i+bs].copy()
            # Random perturbation of the block
            pert = rng.uniform(-0.02, 0.02, bs)
            new_block = np.clip(block + pert, 0, 1)
            h_trial = best.copy()
            h_trial[i:i+bs] = new_block
            h_trial = project_feasible(h_trial, target)
            s = compute_c5(h_trial)
            if s < best_score - 1e-14:
                best_score = s
                best = h_trial
                improved = True
                break
        if not improved:
            # Try single coordinate with smaller steps
            for _ in range(100):
                i = rng.randint(n)
                d = rng.uniform(-0.005, 0.005)
                val = best[i] + d
                if val < 0 or val > 1:
                    continue
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
            if not improved:
                break
    
    return best, best_score


def smooth_and_reproject(h, target, sigma=2.0):
    """Apply Gaussian smoothing and re-project."""
    n = len(h)
    x = np.arange(n)
    kernel = np.exp(-0.5 * (x - n//2)**2 / sigma**2)
    kernel /= np.sum(kernel)
    h_smooth = np.convolve(h, kernel, mode='same')
    return project_feasible(h_smooth, target)

def optimal_coordinate(h, j, target, n):
    """Find the optimal value of h[j] that minimizes max_k c_k."""
    # For each lag k (indexed by k_offset in [0, 2n-2]):
    # c_k = sum_i h[i]*(1-h[i+lag_k])
    # As a function of t = h[j]:
    # c_k(t) = (A_k - B_k) * t + (B_k + C_k)
    # where A_k = 1 if 0 <= j+lag_k < n (contributes t*(1-h[j+lag_k]))
    #       B_k = h[j-lag_k] if 0 <= j-lag_k < n (contributes h[j-lag_k]*(1-t))
    #       C_k = sum of all other terms
    
    # So slope_k = A_k - B_k
    #    intercept_k = B_k + C_k
    
    # We want to find t in [0,1] that minimizes max_k (slope_k * t + intercept_k)
    
    # The max of lines is a convex piecewise-linear function.
    # The minimum is at a "kink" point, i.e., intersection of two lines.
    
    # To find it: the minimum of max of lines occurs where either:
    # 1. t = 0 or t = 1
    # 2. Two (or more) lines achieve the maximum at the same point
    
    # Efficient approach: binary search on the value, or find the intersection
    # of the two "active" lines at the optimum.
    
    # Actually, the simplest approach for 1D: 
    # Evaluate max_k c_k(t) at t=0, t=1, and at all intersection points of pairs of lines
    # that are in [0,1]. But there are O(n²) pairs, and for n=400 that's 160,000 which is OK.
    
    # Better: use the fact that it's a convex function in 1D, so we can do ternary search
    # or use the subgradient condition.
    
    # Actually, the cleanest way: the minimum of max of affine functions in 1D can be found
    # by: the answer t* satisfies that the subgradient of max_k (a_k t + b_k) contains 0.
    # The subgradient at t is the convex hull of {a_k : c_k(t) = max}.
    # So we need 0 to be in the convex hull of the slopes of the active constraints.
    # This means there exist active constraints with slope > 0 and slope < 0 (or some with slope = 0).
    
    # Simplest implementation: ternary search on [0,1]
    best_t = 0.5

    center = n - 1

    # The "rest" (C_k) = c_k - A_k*t - B_k*(1-t) where t = h[j]
    # But actually, let me just compute slopes and intercepts directly.
    
    t = h[j]
    for k_offset in range(2*n-1):
        lag = k_offset - center
        # c_k = sum of h[i]*(1-h[i+lag]) for valid i
        # Terms involving h[j]:
        # 1. i = j: h[j]*(1-h[j+lag]) = t*(1-h[j+lag]) if j+lag in [0,n-1]
        # 2. i+lag = j: h[j-lag]*(1-h[j]) = h[j-lag]*(1-t) if j-lag in [0,n-1]
        
        a_k = 0.0  # coefficient of t from term 1
        b_k = 0.0  # coefficient of (1-t) from term 2, which is h[j-lag]
        
        if 0 <= j + lag < n:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-14
    else:
        mask = h > 1e-14
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def subgradient_with_momentum(h, target, time_limit, alpha_init=10.0, alpha_max=300.0,
                              eta_init=0.008, mu=0.9):
    n = len(h)
    start = time.time()
    h_curr = h.copy()
    v = np.zeros(n)
    best = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_init
    eta = eta_init
    no_improve = 0
    center = n - 1
    n_corr = 2 * n - 1
    lags = np.arange(-(n - 1), n)

    while time.time() - start < time_limit:
        corr = compute_corr(h_curr) * (2.0 / n)
        max_val = np.max(corr)
        shifted = alpha * (corr - max_val)
        weights = np.exp(np.clip(shifted, -300, 0))
        sw = np.sum(weights)
        if sw < 1e-15:
            weights = np.zeros(n_corr)
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= sw

        grad = np.zeros(n)
        for k_offset in range(n_corr):
            w = weights[k_offset]
            if w < 1e-7:
                continue
            lag = lags[k_offset]
            if lag >= 0:
                grad[:n - lag] += w * (1.0 - h_curr[lag:])
                if lag > 0:
                    grad[lag:] -= w * h_curr[:n - lag]
            else:
                lp = -lag
                grad[lp:] += w * (1.0 - h_curr[:n - lp])
                if lp > 0:
                    grad[:n - lp] -= w * h_curr[lp:]

        v = mu * v + (1.0 - mu) * grad
        h_new = h_curr + eta * v
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                eta *= 0.8
                alpha = min(alpha * 1.3, alpha_max)
                no_improve = 0

        h_curr = h_new
        alpha = min(alpha * 1.002, alpha_max)

    return best, best_score


def exact_coordinate_descent(h, target, time_limit, max_sweeps=50000, bisection_iters=60):

    h_curr = h.copy()

    for sweep in range(max_sweeps):
        if time.time() - start >= time_limit:
            break

        corr = compute_corr(h_curr)
        max_improved = 0.0

        for j in range(n):
            t = h_curr[j]
            j_plus = j + lags
            valid_plus = (j_plus >= 0) & (j_plus < n)
            j_minus = j - lags
            valid_minus = (j_minus >= 0) & (j_minus < n)

            slopes = np.zeros(n_corr)
            slopes[valid_plus] = (1.0 - h_curr[j_plus[valid_plus]])
            slopes[valid_minus] -= h_curr[j_minus[valid_minus]]

                vals = slopes * mid + intercepts

                else:
                    break

            new_t = 0.5 * (lo + hi)

                h_curr[j] = new_t

            break

        h_curr = project_feasible(h_curr, target)
        score = compute_c5(h_curr)
        if score < best_score:

def slsqp_refine(h, target, time_limit, alphas=(20, 50, 100, 300, 800, 2000)):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    ones = np.ones(n)

    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        n_lags_left = len(alphas) - 1 - alphas.index(alpha)
        time_per = remaining / max(1, n_lags_left)
        maxiter = max(50, int(time_per * 3))

        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

        def obj_grad(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            n_x = len(x)
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            weights = np.exp(shifted)
            sw = np.sum(weights)
            if sw < 1e-15:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights /= sw
            grad = np.zeros(n_x)
            ctr = n_x - 1
            for k_offset in range(len(corr)):
                w = weights[k_offset]
                if w < 1e-7:
                    continue
                lag = k_offset - ctr
                if lag >= 0:
                    grad[:n_x - lag] += w * (1.0 - x[lag:])
                    if lag > 0:
                        grad[lag:] -= w * x[:n_x - lag]
                else:
                    lp = -lag
                    grad[lp:] += w * (1.0 - x[:n_x - lp])
                    if lp > 0:
                        grad[:n_x - lp] -= w * x[lp:]
            return grad

        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: ones}]

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, jac=obj_grad,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def randomized_perturbation(h, target, time_limit, seed=42):

        scale = max(scale * 0.98, 0.001)
        h_pert = best + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        s = compute_c5(h_pert)
        if s < best_score:
            best_score = s
            best = h_pert
        scale *= 0.999
        if scale < 0.002:
            break
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    if tleft() > 200:
        h1, s1 = subgradient_with_momentum(best_h, target, time_limit=200,
                                           alpha_init=8.0, alpha_max=400.0,
                                           eta_init=0.006, mu=0.85)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    if tleft() > 350:
        h2, s2 = exact_coordinate_descent(best_h, target, time_limit=350,
                                          max_sweeps=50000, bisection_iters=60)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 150:
        h3, s3 = slsqp_refine(best_h, target, time_limit=150,
                               alphas=(30, 80, 200, 600, 2000))
        if s3 < best_score:
            best_score = s3
            best_h = h3

    if tleft() > 300:
        h4, s4 = exact_coordinate_descent(best_h, target, time_limit=300,
                                          max_sweeps=50000, bisection_iters=60)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    if tleft() > 100:
        h5, s5 = randomized_perturbation(best_h, target, time_limit=80, seed=seed)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    if tleft() > 200:
        h6, s6 = exact_coordinate_descent(best_h, target, time_limit=200,
                                          max_sweeps=50000, bisection_iters=60)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    if tleft() > 80:
        h7, s7 = slsqp_refine(best_h, target, time_limit=70,
                               alphas=(300, 1000, 5000))
        if s7 < best_score:
            best_score = s7
            best_h = h7

    if tleft() > 60:
        h8, s8 = exact_coordinate_descent(best_h, target, time_limit=max(30, tleft() - 10),
                                          max_sweeps=50000, bisection_iters=60)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
