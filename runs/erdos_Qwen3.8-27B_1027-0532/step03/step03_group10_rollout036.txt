def smooth_objective_grad(h, eps):
    """Compute smooth max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    obj = cmax + eps * np.log(np.sum(np.exp(x)))
    ex = np.exp(x)

        wk = w[k]

        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    
    return obj, grad

def smooth_obj_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x_max = float(np.max(x))
    x = x - x_max
    obj = cmax + x_max * eps + eps * np.log(np.sum(np.exp(x)))
    ex = np.exp(x)
    sexp = float(np.sum(ex))
    w = ex / sexp
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    
    return obj, grad

    best_h = h.copy()

    for step in range(n_steps):
        obj, grad = smooth_obj_grad(h, eps)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)

            best_h = h_new.copy()
        # Only accept if it doesn't make things too much worse
        if c5 < obj + 0.01:
            h = h_new
        else:
            h = h.copy()  # revert
        # Decay learning rate
        lr *= 0.999
        
    return best_h, best_c5

    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for step in range(n_steps):
        obj, grad = smooth_obj_grad(h, eps)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        if step % 500 == 499:
            lr *= 0.9
    return best_h, best_c5

def targeted_sa(h_start, n_steps, t_start, t_end, rng, two_swap_prob=0.15):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = compute_correlation(h)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1

    ones_pos = np.where(h == 1.0)[0].copy()
    zeros_pos = np.where(h == 0.0)[0].copy()
    n_one = len(ones_pos)
    n_zero = len(zeros_pos)

    pos_to_oi = np.full(n, -1, dtype=np.int32)
    pos_to_oi[ones_pos] = np.arange(n_one, dtype=np.int32)
    pos_to_zi = np.full(n, -1, dtype=np.int32)
    pos_to_zi[zeros_pos] = np.arange(n_zero, dtype=np.int32)

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        progress = step / max(n_steps - 1, 1)
        target_frac = 0.5 * (1.0 - progress) + 0.1

        if rng.random() < two_swap_prob and n_one >= 2 and n_zero >= 2:
            i1 = int(ones_pos[rng.integers(n_one)])
            i2 = int(ones_pos[rng.integers(n_one)])
            j1 = int(zeros_pos[rng.integers(n_zero)])
            j2 = int(zeros_pos[rng.integers(n_zero)])
            if i1 == i2 or j1 == j2 or i1 == j1 or i1 == j2 or i2 == j1 or i2 == j2:
                step += 1
                continue
            s_min = max(-i1, i1 - half, -i2, i2 - half, -j1, j1 - half, -j2, j2 - half)
            s_max = min(half - i1, i1, half - i2, i2, half - j1, j1, half - j2, j2)
            if s_min > s_max:
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                oi1 = int(pos_to_oi[i1]); oi2 = int(pos_to_oi[i2])
                zi1 = int(pos_to_zi[j1]); zi2 = int(pos_to_zi[j2])
                ones_pos[oi1] = j1; ones_pos[oi2] = j2
                zeros_pos[zi1] = i1; zeros_pos[zi2] = i2
                pos_to_oi[j1] = oi1; pos_to_oi[j2] = oi2
                pos_to_zi[i1] = zi1; pos_to_zi[i2] = zi2
                step += 1
                continue
            s_range = np.arange(s_min, s_max + 1)
            k_range = s_range + half
            delta = (h[i1 + s_range] + h[i2 + s_range] + h[i1 - s_range] + h[i2 - s_range]
                     - h[j1 + s_range] - h[j2 + s_range] - h[j1 - s_range] - h[j2 - s_range])
            new_vals = corr[k_range] + delta
            new_max = float(np.max(new_vals))
            if new_max <= current_max or (t > 1e-10 and rng.random() < np.exp(-(new_max - current_max) / t)):
                corr[k_range] = new_vals
                h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
                oi1 = int(pos_to_oi[i1]); oi2 = int(pos_to_oi[i2])
                zi1 = int(pos_to_zi[j1]); zi2 = int(pos_to_zi[j2])
                ones_pos[oi1] = j1; ones_pos[oi2] = j2
                zeros_pos[zi1] = i1; zeros_pos[zi2] = i2
                pos_to_oi[j1] = oi1; pos_to_oi[j2] = oi2
                pos_to_zi[i1] = zi1; pos_to_zi[i2] = zi2
                current_max = float(np.max(corr))
                if current_max < best_c5_val:
                    best_c5_val = current_max
                    best_h = h.copy()
        else:
            if rng.random() < target_frac:
                worst_k = int(np.argmax(corr))
                worst_s = worst_k - half
                if worst_s >= 0:
                    valid_i = [x for x in range(worst_s, n) if h[x] == 1.0 and h[x - worst_s] == 0.0]
                else:
                    valid_i = [x for x in range(0, n + worst_s) if h[x] == 1.0 and h[x - worst_s] == 0.0]
                if len(valid_i) > 1:
                    i = valid_i[rng.integers(len(valid_i))]
                    j_candidates = [x for x in range(0, n) if h[x] == 0.0 and x != i]
                    j = j_candidates[rng.integers(len(j_candidates))]
                else:
                    i = int(ones_pos[rng.integers(n_one)])
                    j = int(zeros_pos[rng.integers(n_zero)])
            else:
                i = int(ones_pos[rng.integers(n_one)])
                j = int(zeros_pos[rng.integers(n_zero)])
            if i == j or h[i] != 1.0 or h[j] != 0.0:
                step += 1
                continue
            s_min = max(-i, i - half, -j, j - half)
            s_max = min(half - i, i, half - j, j)
            if s_min > s_max:
                h[i] = 0.0; h[j] = 1.0
                oi = int(pos_to_oi[i]); zi = int(pos_to_zi[j])
                ones_pos[oi] = j; zeros_pos[zi] = i
                pos_to_oi[j] = oi; pos_to_zi[i] = zi
                step += 1
                continue
            s_range = np.arange(s_min, s_max + 1)
            k_range = s_range + half
            delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
            new_vals = corr[k_range] + delta
            new_max = float(np.max(new_vals))
            if new_max <= current_max or (t > 1e-10 and rng.random() < np.exp(-(new_max - current_max) / t)):
                corr[k_range] = new_vals
                h[i] = 0.0; h[j] = 1.0
                oi = int(pos_to_oi[i]); zi = int(pos_to_zi[j])
                ones_pos[oi] = j; zeros_pos[zi] = i
                pos_to_oi[j] = oi; pos_to_zi[i] = zi
                current_max = float(np.max(corr))
                if current_max < best_c5_val:
                    best_c5_val = current_max
                    best_h = h.copy()
        step += 1
    return best_h, best_c5_val

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None:
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
        best_n = len(h0)
    else:
        best_n = 180
        best_h = (rng.random(best_n) < 0.5).astype(float)
        best_h = to_binary(best_h)
        best_c5 = compute_c5(best_h)

    target_sum = best_n / 2.0

    # Phase 1: Continuous optimization from initial
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 60
    if remaining > 60:
        h_soft = best_h + 0.03 * rng.standard_normal(best_n)
        h_soft = project_onto_constraint(h_soft, target_sum)
        for eps in [0.01, 0.005, 0.002]:
            if time.time() > start_time + budget_s * 0.1:
                break
            h_soft, c5 = projected_grad_descend(h_soft, eps, 500, 0.03, target_sum)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_soft.copy()

    # Phase 2: Binary SA (main workhorse)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 120
    if remaining > 60:
        h_bin = to_binary(best_h)
        sa_steps = int(remaining * 0.4 * 70000)
        sa_steps = min(sa_steps, 15000000)
        h_bin, bin_max = binary_sa_fast(h_bin, sa_steps, 0.6, 0.01, rng, two_swap_prob=0.25)
        c5_bin = (2.0 / best_n) * bin_max
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Targeted SA
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 100
    if remaining > 60:
        h_bin2 = to_binary(best_h)
        sa_steps = int(remaining * 0.4 * 50000)
        sa_steps = min(sa_steps, 10000000)
        h_bin2, bin_max2 = targeted_sa(h_bin2, sa_steps, 0.4, 0.005, rng, two_swap_prob=0.15)
        c5_bin2 = (2.0 / best_n) * bin_max2
        if c5_bin2 < best_c5:
            best_c5 = c5_bin2
            best_h = h_bin2.copy()

    # Phase 4: Continuous polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 80
    if remaining > 30:
        h_polish = best_h + 0.015 * rng.standard_normal(best_n)
        h_polish = project_onto_constraint(h_polish, target_sum)
        for eps in [0.003, 0.001, 0.0005]:
            if time.time() > start_time + budget_s - 40:
                break
            h_polish, c5 = projected_grad_descend(h_polish, eps, 400, 0.02, target_sum)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    # Phase 5: Multiple SA restarts
    n_restart = 0
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 60
    while remaining > 120 and n_restart < 20:
        strategy = n_restart % 4
        if strategy == 0:
            h_start = to_binary(best_h)
        elif strategy == 1:
            h_pert = best_h + 0.06 * rng.standard_normal(best_n)
            h_start = to_binary(project_onto_constraint(h_pert, target_sum))
        elif strategy == 2:
            h_start = to_binary(best_h)
            n_flip = max(4, best_n // 15)
            idx = rng.choice(best_n, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                h_start[ones[:int(s - target_sum)]] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                h_start[zeros[:int(target_sum - s)]] = 1
            h_start = to_binary(h_start)
        else:
            h_rand = (rng.random(best_n) < 0.5).astype(float)
            alpha = 0.3 + 0.2 * rng.random()
            h_start = to_binary(alpha * best_h + (1 - alpha) * h_rand)

        t_start = max(0.15, 0.5 * (1.0 - n_restart / 25.0))
        t_end = 0.003
        step_budget = int(remaining * 0.35 * 60000)
        step_budget = min(step_budget, 8000000)

        if n_restart % 2 == 0:
            h_r, r_max = binary_sa_fast(h_start, step_budget, t_start, t_end, rng, two_swap_prob=0.2)
        else:
            h_r, r_max = targeted_sa(h_start, step_budget, t_start, t_end, rng, two_swap_prob=0.1)

        c5_r = (2.0 / best_n) * r_max
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            # Quick polish
            if time.time() < start_time + budget_s - 50:
                h_p = h_r + 0.01 * rng.standard_normal(best_n)
                h_p = project_onto_constraint(h_p, target_sum)
                h_p, c5_p = projected_grad_descend(h_p, 0.001, 300, 0.02, target_sum)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

        n_restart += 1
        remaining = budget_s - (time.time() - start_time) - 60

    # Phase 6: Try different n
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 30
    if remaining > 150:
        for alt_n in [200, 220, 240]:
            if time.time() > start_time + budget_s - 30:
                break
            time_for_alt = min(remaining * 0.3, 300)
            h_alt = (rng.random(alt_n) < 0.5).astype(float)
            h_alt = to_binary(h_alt)
            alt_target = alt_n / 2.0
            alt_steps = int(time_for_alt * 0.6 * 60000)
            alt_steps = min(alt_steps, 5000000)
            h_alt, alt_max = binary_sa_fast(h_alt, alt_steps, 0.5, 0.01, rng, two_swap_prob=0.2)
            c5_alt = (2.0 / alt_n) * alt_max
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()
                best_n = alt_n
                # Polish
                if time.time() < start_time + budget_s - 20:
                    h_alt_p = h_alt + 0.01 * rng.standard_normal(alt_n)
                    h_alt_p = project_onto_constraint(h_alt_p, alt_target)
                    h_alt_p, c5_alt_p = projected_grad_descend(h_alt_p, 0.001, 300, 0.02, alt_target)
                    if c5_alt_p < best_c5:
                        best_c5 = c5_alt_p
                        best_h = h_alt_p.copy()
            remaining = budget_s - (time.time() - start_time) - 30
            if remaining < 60:
                break

    # Final polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 5
    if remaining > 20:
        n = len(best_h)
        ts = n / 2.0
        h_final = best_h + 0.005 * rng.standard_normal(n)
        h_final = project_onto_constraint(h_final, ts)
        for eps in [0.0005, 0.0002, 0.0001]:
            if time.time() > budget_s - 3:
                break
            h_final, c5_f = projected_grad_descend(h_final, eps, 200, 0.01, ts)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    # Ensure feasibility
    n = len(best_h)
    best_h = project_onto_constraint(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
