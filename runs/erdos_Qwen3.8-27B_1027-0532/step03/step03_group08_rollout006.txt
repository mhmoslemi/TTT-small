def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)

        return h

            h[i] += add
            diff -= add

    # The max of linear functions is convex, so we can find the minimum
    # by bisection on the subgradient

    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _soft_max(corr, alpha) * (2.0 / n)

def _slsqp_solve(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_v2(h0, n, target_sum, n_iters, base_lr, alpha_start, alpha_end, top_k, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        cos_factor = 0.5 * (1.0 + np.cos(np.pi * float(it) / float(max(n_iters, 1))))
        alpha_cur = alpha_start + (alpha_end - alpha_start) * (1.0 - cos_factor)
        k_cur = max(2, top_k - (top_k - 2) * int(float(it) / float(max(n_iters, 1))))
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        m = np.max(corr)
        exp_w = np.exp(alpha_cur * (corr - m))
        total_w = exp_w.sum()
        if total_w < 1e-300:
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        else:
            w = exp_w / total_w
            top_idx = np.argsort(w)[::-1][:k_cur]
            w_top = w[top_idx].copy()
            w_top = w_top / w_top.sum()
            g = np.zeros(n)
            for idx in range(len(top_idx)):
                g += w_top[idx] * _grad_corr_term(y, top_idx[idx], dx)
        
        step = base_lr / (1.0 + 0.005 * float(it))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _coord_descent_continuous(h0, n, target_sum, max_rounds, deadline, n_candidates=9):
    h = h0.copy()
    best_c = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            # Candidates: 0, 1, and values around old
            cands = np.unique(np.clip([0.0, 1.0, old,
                                        old - 0.25, old - 0.1, old - 0.03,
                                        old + 0.03, old + 0.1, old + 0.25], 0, 1))
            best_c_j = best_c
            best_v_j = old
            for v in cands:
                h[j] = v
                c_try = _c5_of(h)
                if c_try < best_c_j - 1e-13:
                    best_c_j = c_try
                    best_v_j = v
            h[j] = best_v_j
            if best_v_j != old:
                improved = True
                if _c5_of(h) < best_c:
                    best_c = _c5_of(h)
        if not improved:
            break
    return h.copy(), _c5_of(h)

def _anneal_structured(h0, n, target_sum, n_iters, T_start, T_end, deadline, rng):
    h = h0.copy()
    c5_cur = _c5_of(h)
    best_h = h.copy()
    best_c5 = c5_cur
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max(n_iters, 1)))
        h_try = h.copy()
        move_type = rng.randint(6)
        
        if move_type == 0:
            bs = rng.randint(2, max(3, n // 12))
            start = rng.randint(0, n - bs)
            h_try[start:start+bs] = 1.0 - h_try[start:start+bs]
        elif move_type == 1:
            bs = rng.randint(1, max(2, n // 20))
            start = rng.randint(0, n - bs)
            h_try[start:start+bs] = rng.random()
        elif move_type == 2:
            bs = rng.randint(2, max(3, n // 10))
            b1 = rng.randint(0, n - bs)
            b2 = rng.randint(0, n - bs)
            if abs(b1 - b2) >= bs:
                tmp = h_try[b1:b1+bs].copy()
                h_try[b1:b1+bs] = h_try[b2:b2+bs]
                h_try[b2:b2+bs] = tmp
        elif move_type == 3:
            w = rng.randint(5, max(6, n // 5))
            start = rng.randint(0, n - w)
            h_try[start:start+w] += 0.15 * rng.randn(w)
        elif move_type == 4:
            n_pts = rng.randint(1, 4)
            idxs = rng.choice(n, n_pts, replace=False)
            h_try[idxs] = rng.random(n_pts)
        else:
            # Shift a segment
            L = rng.randint(3, max(4, n // 8))
            start = rng.randint(0, n - L)
            shift = rng.randint(1, L)
            seg = h_try[start:start+L].copy()
            h_try[start:start+L] = np.roll(seg, shift)
        
        h_try = _fix_sum(h_try, target_sum)
        c5_try = _c5_of(h_try)
        
        delta = c5_try - c5_cur
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_try
            c5_cur = c5_try
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    
    return best_h, best_c5
