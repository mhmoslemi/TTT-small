    for j in range(n):
        old_val = h[j]
        # Compute A and B for this coordinate
        # A[i] = (1-h[j-l]) * valid1 - h[j+l] * valid2 where l = i-(n-1)
        # B[i] = corr[i] - old_val * A[i]
        
        # For lag l, index i = l + (n-1)
        # valid1: j-l in [0,n-1] i.e. l in [j-n+1, j]
        # valid2: j+l in [0,n-1] i.e. l in [-j, n-1-j]

        A = np.zeros(2*n-1)
        for i in range(2*n-1):
            l = i - (n-1)
            a = 0.0
            if 0 <= j-l <= n-1:
                a += (1.0 - h[j-l])
            if 0 <= j+l <= n-1:
                a -= h[j+l]
            A[i] = a
        B = corr - old_val * A
        
        # Minimize max_i (B[i] + A[i]*t) * dx over t in [0,1]
        # Since dx is constant, minimize max_i (B[i] + A[i]*t)
        
        # Grid search
        t_grid = np.linspace(0, 1, n_grid)
        vals = np.max(B + np.outer(t_grid, A), axis=1)
        best_idx = np.argmin(vals)
        best_t = t_grid[best_idx]

            jpl = j + l

            v2 = (jpl >= 0) & (jpl < n)

            A[v1] = 1.0 - h[jml[v1]]

                changed = True

        if not changed:

    best = h.copy()
    bv = cv

        if time.time() > dl:
            break

        h_new = h.copy()
        ni = rng.randint(1, max(2, n // 10))
        idx = rng.choice(n, size=ni, replace=False)
        for i in idx:
            h_new[i] = np.clip(h_new[i] + rng.normal(0, 0.1), 0, 1)
        h_new = _fix(h_new, T)
        cv_new = _c5(h_new)
        if cv_new < cv or rng.random() < np.exp(-(cv_new - cv) / T_cur):
            h = h_new
            cv = cv_new
            if cv < bv:
                bv = cv
                best = h.copy()
    return best, bv

    """2D local search on random pairs"""
    h = h.copy()
    cv = _c5(h)
    for _ in range(n_pairs):
        j1, j2 = rng.choice(n, size=2, replace=False)
        o1, o2 = h[j1], h[j2]
        l = np.arange(-(n-1), n)
        # For j1
        j1ml = j1 - l; j1pl = j1 + l
        v1_1 = (j1ml >= 0) & (j1ml < n)
        v2_1 = (j1pl >= 0) & (j1pl < n)
        A1 = np.zeros(2*n-1)
        A1[v1_1] = 1.0 - h[j1ml[v1_1]]
        A1[v2_1] -= h[j1pl[v2_1]]
        # For j2
        j2ml = j2 - l; j2pl = j2 + l
        v1_2 = (j2ml >= 0) & (j2ml < n)
        v2_2 = (j2pl >= 0) & (j2pl < n)
        A2 = np.zeros(2*n-1)
        A2[v1_2] = 1.0 - h[j2ml[v1_2]]
        A2[v2_2] -= h[j2pl[v2_2]]
        corr = np.correlate(h, 1.0 - h, mode='full')
        B = corr - o1 * A1 - o2 * A2
        # Minimize max(B + A1*t1 + A2*t2) over t1,t2 in [0,1]
        tg = np.linspace(0, 1, grid)
        # 2D grid
        best_t1, best_t2, best_v = o1, o2, _c5(h)

        if best_t1 != o1 or best_t2 != o2:
            h[j1] = best_t1
            h[j2] = best_t2
