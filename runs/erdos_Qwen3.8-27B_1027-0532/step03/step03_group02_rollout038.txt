                ks = np.arange(k_lo4, k_hi4 + 1)

            delta = new_val - old_val
            if abs(delta) > 1e-12:
                h[j] = new_val
                h[j2] = 1.0 - new_val
                corr += delta * G

        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

        return h

            if k_lo2 <= k_hi2:
                ks = np.arange(k_lo2, k_hi2 + 1)
                G[ks] += -h[j + ks]
            new_val = find_optimal_delta(corr, G, old_val)
            delta = new_val - old_val
            if abs(delta) > 1e-12:
                h[j] = new_val
                corr += delta * G
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            h = best_h + rng.randn(n) * 0.01
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_h, best_score

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_sym_refine(h, maxiter, time_limit):
    n = len(h)
    half = n // 2
    h = enforce_symmetry(h.copy())
    alphas = [100, 300, 800]
    h_cur = h[:half].copy()
    bounds = [(0.0, 1.0)] * half
    start = time.time()
    best_score = compute_c5(h)
    best_h = h.copy()
    for alpha in alphas:
        if time.time() - start > time_limit - 5:
            break
        try:
            def obj_wrapper(x, a=alpha):
                h_full = np.empty(n)
                h_full[:half] = x
                h_full[half:] = 1.0 - x[::-1]
                return smooth_obj(h_full, a)
            res = minimize(obj_wrapper, h_cur, method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = np.empty(n)
            h_new[:half] = res.x
            h_new[half:] = 1.0 - res.x[::-1]
            h_new = np.clip(h_new, 0, 1)
            h_new = enforce_symmetry(h_new)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = best_h[:half].copy()
        except Exception:
            pass
    return best_h, best_score

def coarse_to_fine(h_init, n_target, max_time, rng):
    n_init = len(h_init)
    coarse_n = max(40, n_target // 8)
    if coarse_n % 2 != 0:
        coarse_n += 2
    h_coarse = interpolate_to(h_init, coarse_n)
    h_coarse = enforce_symmetry(h_coarse)
    t_left = max_time - (time.time() - time.time())
    alloc_coarse = min(max_time * 0.2, 60)
    h_c, sc_c = sym_coord_descent(h_coarse, alloc_coarse, rng, patience=15)
    mid_n = max(100, n_target // 3)
    if mid_n % 2 != 0:
        mid_n += 2
    h_mid = interpolate_to(h_c, mid_n)
    h_mid = enforce_symmetry(h_mid)
    elapsed = time.time() - (max_time - alloc_coarse)
    alloc_mid = max(10, min(max_time * 0.3, 120))
    h_m, sc_m = sym_coord_descent(h_mid, alloc_mid, rng, patience=20)
    h_fine = interpolate_to(h_m, n_target)
    h_fine = enforce_symmetry(h_fine)
    return h_fine

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    h_sym = enforce_symmetry(h_init.copy())
    score_sym = compute_c5(h_sym)
    score_asym = compute_c5(h_init)

    best_h = h_sym.copy() if score_sym <= score_asym else h_init.copy()
    best_score = min(score_sym, score_asym)

    if tleft() > 250:
        t_alloc = min(500, tleft() - 250)
        h_cd, sc_cd = sym_coord_descent(best_h, t_alloc, rng, patience=35)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    res_list = [n_init + 1, n_init - 1, n_init + 2, n_init - 2, n_init + 3]
    res_list = [n for n in res_list if n >= 100 and n != n_init]
    for n_try in res_list:
        if tleft() < 120:
            break
        h_interp = interpolate_to(best_h, n_try)
        h_interp = enforce_symmetry(h_interp)
        t_alloc = min(80, tleft() - 120)
        if t_alloc < 15:
            break
        h_cd2, sc_cd2 = sym_coord_descent(h_interp, t_alloc, rng, patience=20)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = interpolate_to(h_cd2, n_init)
            best_h = enforce_symmetry(best_h)

    if tleft() > 150:
        h_c2f = coarse_to_fine(best_h, n_init, min(120, tleft() - 150), rng)
        h_c2f, sc_c2f = sym_coord_descent(h_c2f, min(60, tleft() - 150), rng, patience=20)
        if sc_c2f < best_score:
            best_score = sc_c2f
            best_h = h_c2f

    if tleft() > 100:
        h_final, sc_final = sym_coord_descent(best_h, min(100, tleft() - 100), rng, patience=50)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    n_best = len(best_h)
    if tleft() > 60:
        h_sl, sc_sl = slsqp_sym_refine(best_h, 300, max(20, tleft() - 60))
        if sc_sl < best_score:
            best_score = sc_sl
            best_h = h_sl

    if tleft() > 40:
        h_ns = best_h.copy()
        h_ns[:n_best // 4] += rng.randn(n_best // 4) * 0.003
        h_ns = project_feasible(h_ns, n_best / 2.0)
        h_ns2, sc_ns2 = ns_coord_descent(h_ns, n_best / 2.0, min(30, tleft() - 40), rng, patience=10)
        if sc_ns2 < best_score:
            best_score = sc_ns2
            best_h = h_ns2

    n_final = len(best_h)
    target_final = n_final / 2.0
    best_h = project_feasible(best_h, target_final)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, n_final
