            order = np.argsort(x)
            for i in order:
                if diff >= 0:

                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

        if m > 0:

            grad[:m] += -h[lag:lag+m]

        m = n - j

        if cv < best_val:

            best = x.copy()

import time
import numpy as np
from scipy.optimize import minimize

_gt = 0.0
_gn = 0
_alpha = 100.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_shift(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _eq_fun(x):
    return x.sum() - _gt

def _eq_jac(x):
    return np.ones(_gn)

def _sobj(x):
    n = _gn
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = float(np.max(corr))
    a = _alpha
    shifted = a * (corr - m)
    ms = float(np.max(shifted))
    return m + np.log(np.sum(np.exp(shifted - ms)) + np.exp(-ms)) / a

def _sgrad(x):
    n = _gn
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = float(np.max(corr))
    a = _alpha
    shifted = a * (corr - m)
    ms = float(np.max(shifted))
    exp_s = np.exp(shifted - ms)
    w = exp_s / (np.sum(exp_s) + np.exp(-ms))
    top = np.argsort(w)[::-1][:12]
    wm = w[top].copy()
    wm = wm / wm.sum()
    g = np.zeros(n)
    for idx in range(len(top)):
        g += wm[idx] * _grad_shift(x, top[idx], dx)
    return g

def _nesterov_smooth(h0, target_sum, alpha_start, alpha_end, n_iters, step_scale, deadline):
    global _alpha
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        _alpha = alpha_start * (alpha_end / alpha_start) ** (float(it) / max(1, n_iters))
        g = _sgrad(y)
        step = step_scale / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_polish(h0, target_sum, alpha, maxiter):
    global _gt, _gn, _alpha
    n = len(h0)
    _gt = target_sum
    _gn = n
    _alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h0

    n = len(h0)
    h = h0.copy()
    current_val = _c5_of(h)
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        indices = list(range(n))
        if order == 'rand':
            np.random.shuffle(indices)
        elif order == 'rev':
            indices = indices[::-1]
        for j in indices:
            old = h[j]
            best_c = current_val
            best_v = old
            lo = max(0.0, old - range_lo)
            hi = min(1.0, old + range_hi)

                continue
            candidates = np.linspace(lo, hi, n_evals)

                    best_c = c
                    best_v = v

                current_val = best_c
                improved = True
