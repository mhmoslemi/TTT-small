import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def binary_local_search(h, max_rounds, time_limit, start_time, rng):
    """First-improvement swap local search for binary h."""
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    improved = True
    rounds = 0
    while improved and rounds < max_rounds:
        if time.time() - start_time > time_limit:
            break
        improved = False
        rounds += 1
        # Try random swaps until we find an improvement
        for _ in range(len(ones) * len(zeros)):
            if time.time() - start_time > time_limit:
                break
            oi = rng.integers(len(ones))
            zi = rng.integers(len(zeros))
            a = ones[oi]
            b = zeros[zi]
            h[a] = 0.0
            h[b] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < current:
                current = new_c5
                ones[oi] = b
                zeros[zi] = a
                improved = True
                break
            else:
                h[a] = 1.0
                h[b] = 0.0
    return h, current

def iterated_local_search(best_h, best_c5, time_limit, start_time, rng):
    """Iterated local search with adaptive perturbation."""
    n = len(best_h)
    current_best = best_c5
    current_h = best_h.copy()
    no_improve = 0
    
    while time.time() - start_time < time_limit:
        # Perturb
        perturb_size = min(n // 4, 3 + no_improve // 3)
        perturb_size = max(2, perturb_size)
        h = current_h.copy()
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        swap_ones = rng.choice(ones, size=perturb_size, replace=False)
        swap_zeros = rng.choice(zeros, size=perturb_size, replace=False)
        h[swap_ones] = 0.0
        h[swap_zeros] = 1.0
        
        # Local search
        h_new, c5_new = binary_local_search(h, 50, time_limit, start_time, rng)
        
        if c5_new < current_best:
            current_best = c5_new
            current_h = h_new
            no_improve = 0
        else:
            no_improve += 1
            # Sometimes keep the perturbed solution
            if rng.random() < 0.1:
                current_h = h_new  # or h?
    
    return current_h, current_best

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, target_sum):
    n = len(h)
    h = h.astype(float).copy()
    order = np.argsort(np.abs(h - 0.5))
    result = np.zeros(n)
    result[order[(n - target_sum):]] = 1.0

def local_search_binary(h, start_time, time_limit, rng, max_full_passes=200):

    h = h.copy()

    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    best_c5 = current
    best_h = h.copy()
    
    for _ in range(max_full_passes):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for _ in range(len(ones) * len(zeros) * 2):
            if time.time() - start_time > time_limit:
                break
            oi = rng.integers(len(ones))
            zi = rng.integers(len(zeros))
            a = ones[oi]
            b = zeros[zi]
            h[a] = 0.0
            h[b] = 1.0
            nc = compute_c5(h)
            if nc < current:
                current = nc
                ones[oi] = b
                zeros[zi] = a
                improved = True
                if nc < best_c5:
                    best_c5 = nc
                    best_h = h.copy()
                break
            else:

        if not improved:

    return best_h, best_c5

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-12):

    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)

    try:

                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:

def block_swap_perturb(h, blen, rng):

    h = h.copy()
    if blen >= n:
        return h
    a = rng.integers(0, n - blen + 1)
    b = rng.integers(0, n - blen + 1)

    block_a = h[a:a+blen].copy()
    block_b = h[b:b+blen].copy()
    h[a:a+blen] = block_b
    h[b:b+blen] = block_a
    return h

    n = len(h)
    h = h.copy()
    if blen >= n:
        return h
    a = rng.integers(0, n - blen + 1)

    h[a:a+blen] = 1.0 - h[a:a+blen]

    diff = int(round(sum_before - sum_after))
    if diff != 0:
        if diff > 0:
            block = h[a:a+blen]
            ones_in = np.where(block == 1.0)[0]
            if len(ones_in) >= diff:
                flip = rng.choice(ones_in, size=diff, replace=False)
                h[a+flip] = 0.0
        else:
            block = h[a:a+blen]
            zeros_in = np.where(block == 0.0)[0]
            if len(zeros_in) >= -diff:
                flip = rng.choice(zeros_in, size=-diff, replace=False)
                h[a+flip] = 1.0
    return h
