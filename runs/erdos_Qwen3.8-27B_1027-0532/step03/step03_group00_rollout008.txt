    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    
    if alpha == 0:
        # Subgradient of max
        max_idx = np.argmax(corr)
        # Gradient of corr[max_idx] w.r.t. h
        # c[k] = sum_i h[i] * (1-h[i+k-n+1])
        # dc[k]/dh[j] = (1-h[j+k-n+1]) * 1_{0≤j+k-n+1≤n-1} - h[j-k+n-1] * 1_{0≤j-k+n-1≤n-1}
        k = max_idx
        grad = np.zeros(n)
        for j in range(n):
            idx1 = j + k - n + 1
            idx2 = j - k + n - 1
            if 0 <= idx1 < n:
                grad[j] += (1 - h[idx1])
            if 0 <= idx2 < n:
                grad[j] -= h[idx2]
        return corr, grad, float(np.max(corr))
    else:
        # Soft-max gradient
        c_max = np.max(corr)
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -700.0, 0.0)
        weights = np.exp(shifted)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_grad(h, n, alpha):
    """Compute soft-max objective value and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    obj = c_max + log_sum / alpha
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient computation
    # Only iterate over lags with significant weight
    significant = weights > 1e-8
    k_indices = np.where(significant)[0]
    
    grad = np.zeros(n)
    for k in k_indices:
        w = weights[k]
        # dc[k]/dh[j] = (1-h[j+k-n+1])*1_{0<=j+k-n+1<n} - h[j-k+n-1]*1_{0<=j-k+n-1<n}
        # Vectorize:
        # For the first term: j+k-n+1 in [0, n-1] means j in [n-1-k, 2n-2-k] ∩ [0, n-1]
        #                    = [max(0, n-1-k), min(n-1, 2n-2-k)]
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w * (1 - h[idx + k - n + 1])
        
        # For the second term: j-k+n-1 in [0, n-1] means j in [k-n+1, k] ∩ [0, n-1]
        #                    = [max(0, k-n+1), min(n-1, k)]
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += w * (-h[idx - k + n - 1])
    
    return obj, grad

def projected_grad_descent(h, n, target, alpha, lr, n_steps, deadline, momentum=0.9):
    """Projected gradient descent with momentum."""
    h_cur = h.copy()
    v = np.zeros(n)
    best_score = compute_c5_value(h_cur)
    best_h = h_cur.copy()
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        obj, grad = soft_max_grad(h_cur, n, alpha)
        
        # Gradient step with momentum
        v = momentum * v - lr * grad
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        
        score = compute_c5_value(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        
        # Adaptive step size
        if score > best_score * 1.01:
            lr *= 0.5
            momentum = 0.9
        elif score < best_score:
            lr = min(lr * 1.1, 0.1)
        h_cur = h_new
    
    return best_h, best_score

A good approach: 
1. Compute the "binary version" of h: h_bin[i] = 1 if h[i] > 0.5 else 0
2. Project h_bin to satisfy the sum constraint
3. Compare C_5(h_bin) with C_5(h)
4. If h_bin is better, keep it
5. If not, blend: h_new = (1-t)*h + t*h_bin for some t

Actually, a better binary push is:
- For values clearly above 0.5 (say > 0.6), push them towards 1
- For values clearly below 0.5 (say < 0.4), push them towards 0
- Leave the "transition" values (0.4-0.6) alone

This preserves the structure while pushing the "decided" values to their extremes.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def soft_max_grad(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    obj = c_max + np.log(total) / alpha
    weights = exp_s / total
    
    grad = np.zeros(n)
    significant = np.where(weights > 1e-8)[0]
    for k in significant:
        w = weights[k]
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += w * (1.0 - h[idx + k - n + 1])
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] += w * (-h[idx - k + n - 1])
    
    return obj, grad

        score = compute_c5(h_try)
        
        if score < best_score:
            best_score = score
            best_h = h_try.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            v = np.zeros(n)
            lr *= 0.5
            no_improve = 0
        
        h_cur = h_try
    
    return best_h, best_score

def binary_push(h, target, push_thresh=0.1):
    """Push values away from 0.5 towards 0 or 1."""
    h_new = h.copy()
    for i in range(len(h_new)):
        if h_new[i] < 0.5 - push_thresh:
            h_new[i] = h_new[i] * 0.7  # push towards 0
        elif h_new[i] > 0.5 + push_thresh:
            h_new[i] = 1.0 - (1.0 - h_new[i]) * 0.7  # push towards 1
    return project_feasible(h_new, target)

    cur_score = compute_c5(h_cur)

    t_end = 0.0001

            break
        T = t_start * (t_end / t_start) ** (float(it) / max(1, n_iters - 1))

            cur_score = new_score

                best_score = cur_score
                best_h = h_cur.copy()
    
    return best_h, best_score
