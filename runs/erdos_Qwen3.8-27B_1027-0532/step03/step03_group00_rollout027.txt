        lags = np.argsort(c5_vals)[-n_active:]

        import cvxpy as cp
        h_var = cp.Variable(n)
        t_var = cp.Variable()

            h_var >= 0,
            h_var <= 1

                grad = np.zeros(n)
                grad[valid] = (1 - 2 * h_cur[:n+k]) * dx

            constraints.append(const + np.dot(grad, h_var) <= t_var)

            break

            break

        c_vals = corr * dx
        t_max = float(np.max(c_vals))
        
        # Select active constraints
        top_idx = np.argsort(c_vals)[-n_active:]
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]

            # Compute gradient
            grad = np.zeros(n)
            if k >= 0:
                grad[0:n-k] = (1.0 - h_cur[k:n]) * dx
                grad[k:n] -= h_cur[0:n-k] * dx
            else:
                kk = -k
                grad[kk:n] = (1.0 - h_cur[0:n-kk]) * dx
                grad[0:n-kk] -= h_cur[kk:n] * dx

            const = float(c_vals[idx]) - float(np.dot(grad, h_cur))
            constraints.append(const + grad @ h_var <= t_var)

            prob.solve(solver=solver)

        h_lp = np.clip(h_var.value, 0.0, 1.0)

One more idea: **Second-order information from the Hessian**

The Hessian of c_k is sparse (only non-zero at positions (i, i+k) and (i, i-k)). For a single k, the Hessian has at most 2n non-zero entries. This means we could do a Newton step for the most active constraint:

H_k * d = -∇c_k
d = -H_k^{-1} * ∇c_k

For a single k, H_k has a very special structure (it's like a banded matrix with a specific pattern). The inverse might be computable efficiently.

Actually, for k=0: H_0 has -dx on the superdiagonal and subdiagonal (positions (i, i+1) and (i+1, i) for... wait no.

Let me recompute. c_k = ∑_{i} h[i]*(1-h[i+k])*dx = ∑_{i} h[i]*dx - ∑_{i} h[i]*h[i+k]*dx

∂²c_k/∂h[i]∂h[j]:
- From -∑ h[i]*h[i+k]*dx:
  - ∂/∂h[i]: -h[i+k]*dx (contributes to diagonal? No, h[i+k] ≠ h[i] in general for the quadratic term)
  - Actually: ∂/∂h[j] of (-h[i]*h[i+k]*dx) is:
    - -dx if j=i+k (from the h[i+k] factor)
    - -dx if j-k=i, i.e., j=i+k (same thing)
    - Wait, I need to be more careful.
    
Let me denote f = -∑_{i: i+k<n} h[i]*h[i+k]*dx

∂f/∂h[j] = -dx * (h[j-k] * 1_{j-k≥0} + h[j+k] * 1_{j+k<n})

So ∂²f/∂h[j]∂h[m] = -dx * (1_{m=j-k} + 1_{m=j+k})

This means the Hessian H has H[j,m] = -dx if m = j±k, and 0 otherwise.

For k=1: H is a centrosymmetric matrix with -dx on the first superdiagonal and first subdiagonal. This is a tridiagonal matrix! We can solve H*d = b in O(n) using Thomas algorithm.

For general k: H is a "k-banded" matrix with -dx at distance k. We can still solve it in O(n*k) or O(n) using the fact that it's a Toeplitz-like matrix.

Actually, for the Newton step, we want to minimize c_k (just the one active constraint). The Newton direction is:
d = -H_k^{-1} * ∇c_k

where ∇c_k[j] = (1-h[j+k])*dx * 1_{j+k<n} - h[j-k]*dx * 1_{j-k≥0}

And H_k[j,m] = -dx * (1_{m=j+k} + 1_{m=j-k})

So we need to solve: -dx * (d[j-k] + d[j+k]) = -∇c_k[j]
i.e., d[j-k] + d[j+k] = ∇c_k[j]/dx

This is a linear system with a very specific structure. For k=1, it's:
d[j-1] + d[j+1] = b[j] (tridiagonal, solvable in O(n))

For general k, it's a system where each row has only 2 non-zeros. We can solve it using a specialized algorithm or just use numpy's solver (for n=400, it's fast enough).

Actually, the matrix is singular! The equation d[j-k] + d[j+k] = b[j] has a null space (constant vectors are in the null space when k doesn't divide n-1... actually it's more complex). We need to add a regularization or solve the least-squares problem.

                g[kk:n] = (1.0 - h_cur[0:n-kk]) * dx
                g[0:n-kk] -= h_cur[kk:n] * dx
            grad += g
        grad /= float(k_active)
        vel = momentum * vel - lr * grad
        h_new = h_cur + vel
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h

def lp_cutting_plane(h, n, target, max_iters, n_active, deadline, solver=cp.ECOS):
    global _g_n, _g_target
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    dx = 2.0 / n
    rng = np.random.RandomState(12345)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    stagnation = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_vals = corr * dx
        top_idx = np.argsort(c_vals)[-n_active:]
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0,
            h_var <= 1
        ]
        for idx in top_idx:
            k = int(idx) - (n - 1)
            grad = np.zeros(n)
            if k >= 0:
                grad[0:n-k] = (1.0 - h_cur[k:n]) * dx
                grad[k:n] -= h_cur[0:n-k] * dx
            else:
                kk = -k
                grad[kk:n] = (1.0 - h_cur[0:n-kk]) * dx
                grad[0:n-kk] -= h_cur[kk:n] * dx
            const = float(c_vals[idx]) - float(np.dot(grad, h_cur))
            constraints.append(const + grad.dot(h_var) <= t_var)
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        improved = False
        for alpha_try in [1.0, 0.7, 0.5, 0.3, 0.8, 0.2, 0.15]:
            h_test = (1.0 - alpha_try) * h_cur + alpha_try * h_lp
            h_test = project_feasible(h_test, target)
            sc = compute_c5(h_test)
            if sc < best_score:
                best_score = sc
                best_h = h_test.copy()
                h_cur = h_test.copy()
                improved = True
                stagnation = 0
                break
        if not improved:
            stagnation += 1
            if stagnation > 8:
                h_cur = h_cur + 0.003 * rng.randn(n)
                h_cur = project_feasible(h_cur, target)
                stagnation = 0
            else:
                h_test = h_cur + 0.05 * (h_lp - h_cur)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < best_score:
                    best_score = sc
                    best_h = h_test.copy()
                    h_cur = h_test.copy()
    return best_h

def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_block_target, _g_block_alpha
    _g_block_target = target
    _g_block_alpha = alpha
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = (n + block_size - 1) // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = min(s + block_size, n)
            _g_block_full = h_cur
            _g_block_start = s
            _g_block_end = e
            x0 = h_cur[s:e].copy()
            try:
                res = minimize(_block_obj, x0, method='Nelder-Mead',
                              options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_block_full, _g_block_start, _g_block_end, _g_block_target, _g_block_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP
    h_ref = slsqp_optimize(best_h, n, target, [10, 50, 200, 800], 2000, 1e-16)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 2: Subgradient with momentum (new)
    deadline = start_time + 120.0
    h_sg = subgradient_optimize(best_h, n, target, 50000, 3, 0.02, 0.01, deadline, 0.95)
    sc = compute_c5(h_sg)
    if sc < best_score:
        best_score = sc
        best_h = h_sg

    # Phase 3: LP cutting plane (key new ingredient)
    deadline_lp = start_time + 380.0
    h_lp = lp_cutting_plane(best_h, n, target, 3000, 5, deadline_lp, cp.ECOS)
    sc = compute_c5(h_lp)
    if sc < best_score:
        best_score = sc
        best_h = h_lp

    # Phase 4: SLSQP refinement after LP
    h_ref = slsqp_optimize(best_h, n, target, [200, 800, 3000], 2000, 1e-16)
    sc = compute_c5(h_ref)
    if sc < best_score:
        best_score = sc
        best_h = h_ref

    # Phase 5: Block coordinate descent
    deadline_bd = start_time + 600.0
    for bs in [4, 6]:
        if tleft() < 100:
            break
        h_bd = block_descent(best_h, n, target, bs, 8, 300.0, deadline_bd)
        sc = compute_c5(h_bd)
        if sc < best_score:
            best_score = sc
            best_h = h_bd

    # Phase 6: More subgradient (with different params)
    deadline_sg2 = start_time + 750.0
    h_sg2 = subgradient_optimize(best_h, n, target, 30000, 5, 0.01, 0.005, deadline_sg2, 0.9)
    sc = compute_c5(h_sg2)
    if sc < best_score:
        best_score = sc
        best_h = h_sg2

    # Phase 7: LP cutting plane round 2
    deadline_lp2 = start_time + 950.0
    h_lp2 = lp_cutting_plane(best_h, n, target, 2000, 8, deadline_lp2, cp.ECOS)
    sc = compute_c5(h_lp2)
    if sc < best_score:
        best_score = sc
        best_h = h_lp2

    # Phase 8: Multi-scale restart
    if tleft() > 120:
        n_coarse = max(n // 2, 100)
        target_coarse = n_coarse / 2.0
        h_coarse = interpolate_to(best_h, n_coarse, target_coarse)
        h_coarse = slsqp_optimize(h_coarse, n_coarse, target_coarse, [50, 200, 800], 1500, 1e-15)
        deadline_ls = start_time + budget_s - 40.0
        h_coarse = lp_cutting_plane(h_coarse, n_coarse, target_coarse, 500, 5, deadline_ls, cp.ECOS)
        h_up = interpolate_to(h_coarse, n, target)
        h_up = slsqp_optimize(h_up, n, target, [200, 1000], 1000, 1e-16)
        sc = compute_c5(h_up)
        if sc < best_score:
            best_score = sc
            best_h = h_up

    # Phase 9: Final block descent + SLSQP polish
    if tleft() > 50:
        h_bd = block_descent(best_h, n, target, 4, 5, 800.0, start_time + budget_s - 20.0)
        sc = compute_c5(h_bd)
        if sc < best_score:
            best_score = sc
            best_h = h_bd

    # Phase 10: Final SLSQP
    if tleft() > 20:
        h_final = slsqp_optimize(best_h, n, target, [500, 2000, 8000], 1500, 1e-17)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
