from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smooth_max_objective(h, alpha):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = alpha * corr
    shifted = shifted - np.max(shifted)
    shifted = np.clip(shifted, -500, 0)
    return float((np.max(alpha * corr) + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def compute_gradient_smax(h, alpha):
    """Gradient of the smooth max (log-sum-exp) of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    shifted = a - a_max
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    if total < 1e-15:
        return np.zeros(n)
    p = exp_shifted / total  # probability distribution over lags
    
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = n - 1 + k
        w = p[idx]
        if w < 1e-12:
            continue
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                if n - k > 0:
                    grad[:n-k] += w * (1.0 - h[k:n])
                if k < n:
                    grad[k:] -= w * h[:n-k]
        else:
            m = -k
            if m < n:
                grad[m:] += w * (1.0 - h[:n-m])
            if n - m > 0:
                grad[:n-m] -= w * h[m:n]
    return grad * dx

def adam_optimize(h, target, time_limit, alpha_start=5, alpha_end=200):
    """Adam optimizer with annealing temperature."""
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    # Schedule: linearly interpolate alpha from alpha_start to alpha_end
    # Use a time-based schedule
    total_time = time_limit
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(elapsed / total_time, 1.0)
        # Use exponential schedule for alpha
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        grad = compute_gradient_smax(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        # Adam update
        lr = 0.01 * (1.0 + 0.5 * np.cos(np.pi * frac))  # cosine decay
        t = int(elapsed * 10)  # approximate step count
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
        h = h_new
    
    return best, best_score

def coordinate_descent(h, target, time_limit, seed=42, delta_scale=1.0):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    
    deltas = np.array([0.001, -0.001, 0.003, -0.003, 0.005, -0.005,
                       0.01, -0.01, 0.02, -0.02]) * delta_scale
    no_improve = 0
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        if not improved:
            no_improve += 1
            delta_scale *= 0.5
            deltas = np.array([0.001, -0.001, 0.003, -0.003, 0.005, -0.005,
                               0.01, -0.01, 0.02, -0.02]) * delta_scale
            if max(abs(d) for d in deltas) < 1e-6:

        else:
            no_improve = 0
    
    return best, best_score

def block_local_search(h, target, time_limit, seed=42, block_sizes=None):

def slsqp_refine(h, target, time_limit, use_lse=True):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    alphas = [20, 50, 100, 200, 400] if use_lse else [50, 100, 200, 400]
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(50, remaining * 2))
        
        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            shifted = a * (corr - np.max(corr))
            shifted = np.clip(shifted, -500, 0)
            n_x = len(x)
            dx = 2.0 / n_x
            return float((np.max(a * corr) + np.log(np.sum(np.exp(shifted))) / a) * dx)
        
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_smax_gradient(h, alpha, max_lag=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    shifted = a - a_max
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    if total < 1e-15:
        return np.zeros(n)
    p = exp_s / total

    if max_lag is None:
        max_lag = n - 1

    grad = np.zeros(n)
    for k in range(-max_lag, max_lag + 1):
        idx = n - 1 + k
        w = p[idx]
        if w < 1e-10:
            continue
        if k >= 0:
            if k == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                if n - k > 0:
                    grad[:n - k] += w * (1.0 - h[k:n])
                if k < n:
                    grad[k:] -= w * h[:n - k]
        else:
            m = -k
            if m < n:
                grad[m:] += w * (1.0 - h[:n - m])
            if n - m > 0:
                grad[:n - m] -= w * h[m:n]
    return grad * dx


def smax_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    a = alpha * corr
    a_max = np.max(a)
    shifted = a - a_max
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((a_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def adam_phase(h, target, time_limit, alpha_start=10.0, alpha_end=500.0,
               lr_base=0.02, max_lag=150):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    step_count = 0
    no_improve = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac

        grad = compute_smax_gradient(h, alpha, max_lag=max_lag)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break

        step_count += 1
        decay1 = 1.0 - beta1 ** step_count
        decay2 = 1.0 - beta2 ** step_count
        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad ** 2
        m_hat = m_buf / decay1
        v_hat = v_buf / decay2

        lr = lr_base * (1.0 + np.cos(np.pi * frac)) * 0.5
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target)

        score = compute_c5(h_new)
        if score < best_score - 1e-15:
            best_score = score
            best = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        h = h_new
        if no_improve > 5000:
            break

    return best, best_score


def coordinate_descent_phase(h, target, time_limit, seed=42, base_deltas=None):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)

    if base_deltas is None:
        base_deltas = np.array([0.002, -0.002, 0.005, -0.005,
                                0.01, -0.01, 0.02, -0.02])
    deltas = base_deltas.copy()
    no_improve_sweeps = 0

    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        if not improved:
            no_improve_sweeps += 1
            deltas = deltas * 0.4
            if np.max(np.abs(deltas)) < 1e-6:
                break
        else:
            no_improve_sweeps = 0

    return best, best_score


def slsqp_phase(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)

    if alphas is None:
        alphas = [30, 80, 200, 400, 800]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]

    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(30, remaining * 1.5))

        def obj_func(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            ac = a * corr
            amax = np.max(ac)
            sh = ac - amax
            sh = np.clip(sh, -500.0, 0.0)
            n_x = len(x)
            dx = 2.0 / n_x
            return float((amax + np.log(np.sum(np.exp(sh))) / a) * dx)

        try:
            res = minimize(obj_func, best, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def restart_and_refine(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)

    while time.time() - start < time_limit:
        perturbed = best + rng.normal(0, 0.05, n)
        perturbed = project_feasible(perturbed, target)
        s_pert = compute_c5(perturbed)

        if s_pert < best_score:
            best = perturbed
            best_score = s_pert
            continue

        limit_rem = time_limit - (time.time() - start)
        if limit_rem < 30:
            break
        limit_rem = min(limit_rem * 0.4, 60.0)
        h_adam, s_adam = adam_phase(perturbed, target, limit_rem,
                                    alpha_start=20, alpha_end=300,
                                    lr_base=0.015, max_lag=100)
        if s_adam < best_score:
            best_score = s_adam
            best = h_adam

    return best, best_score
