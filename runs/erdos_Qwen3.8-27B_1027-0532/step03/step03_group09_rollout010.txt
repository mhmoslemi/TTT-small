Actually, the key insight is: for a given h, the function f(delta) = max_k (corr[k] + a_k * delta) is convex in delta (max of linear functions). To minimize it over the range [max(0, -h[i]), min(1, 1-h[i])], we can:

1. Evaluate at the endpoints
2. Find where the subgradient changes sign (bisection)

The subgradient at delta is a_k* where k* is any argmax of (corr + a*delta).

Let me implement this properly:

```python
def find_best_delta(h, i, corr, n_active=5):
    """Find best delta for h[i] to minimize C5."""
    n = len(h)
    dx = 2.0 / n
    h_i = h[i]
    
    # Compute a[k] = effect of delta on corr[k]
    # a[k] = dx * [(1-h[i+s_k]) if valid - h[i-s_k] if valid]
    # where s_k = k - (n-1)
    
    # For efficiency, compute a only for the top lags
    top_k = np.argsort(corr)[-n_active:]
    
    # But we also need to consider that other lags might become the max
    # After changing h[i], the new max could be at a different lag
    # For a safe bound, we need all lags
    
    # Let's compute a for all lags (it's O(n))
    s_all = np.arange(2*n-1) - (n-1)  # s = -(n-1), ..., 0, ..., n-1
    a = np.zeros(2*n-1)

            val += dx * (1.0 - h[i + s])
        if 0 <= i - s < n:
            val -= dx * h[i - s]

    # Find delta* that minimizes f over [lo, hi]
    lo = -h_i
    hi = 1.0 - h_i

        return 0.0

    def subgrad_at(delta):
        vals = corr + a * delta
        return a[np.argmax(vals)]

    sg_hi = subgrad_at(hi)
```

```python
    n = len(h_start)
    h_bin = (h_start > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

        to_flip = rng.choice(ones, excess, replace=False)
        h_bin[to_flip] = 0
    elif len(ones) < n // 2:
        deficit = n // 2 - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h_bin[to_flip] = 1

        T = max(T_min, T_init * (T_min / T_init) ** (it / n_iterations))

        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')

        if delta < 0 or rng.random() < np.exp(-delta / T):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
```

The issue with this approach is that each iteration requires O(n²) work for the correlation, and we might need many iterations (10000+). For n=180, that's 10000 * 32400 = 324M operations. In numpy, each correlate is probably ~1ms, so 10000 iterations would take ~10s. That's feasible.

But I can optimize: instead of recomputing the full correlation, I can update it incrementally. When I swap h[i] and h[j] (i had 1, j had 0):
- For each lag s: corr(s) changes based on whether i or j are in the valid range for that lag

Actually, the correlation is a convolution, so changing two elements changes O(n) values in the output. Specifically:
- Setting h[i] from 1 to 0: subtracts h[i]'s contribution from all lags
- Setting h[j] from 0 to 1: adds h[j]'s contribution to all lags

The contribution of h[i]=1 to corr(s) is:
- As first factor: (1-h[i+s]) for lag s (if i+s valid)
- As second factor: -h[i-s] for lag s (if i-s valid)

So changing h[i] from 1 to 0 and h[j] from 0 to 1:
- For lag s: delta_c(s) = -dx * [(1-h[i+s])*1_{valid} - h[i-s]*1_{valid}] + dx * [(1-h[j+s])*1_{valid} - h[j-s]*1_{valid}]

This is O(n) per swap (compute for all s). So the total is O(n) per iteration instead of O(n²). Much better!

```python
        h[excess] = 0
    elif len(ones) < target:

        h[deficit] = 1

    for it in range(n_iter):

        #   +1*(1-h[j+s]) = (1-h[j+s]) for first factor
```

```python
def compute_swap_delta(h, i, j, n, dx):
    """Compute change in corr when h[i] goes 1→0 and h[j] goes 0→1."""
    s_all = np.arange(2*n-1) - (n-1)
    
    # For position p, the contribution to delta when it changes from old to new:
    # If h[p] goes from 1 to 0: delta_p[k] = -dx * [(1-h[p+s_k]) * 1_{0<=p+s_k<n} - h[p-s_k] * 1_{0<=p-s_k<n}]
    # Wait, I need to be more careful.
    
    # corr[k] = dx * sum_v h[v] * (1 - h[v + s_k]) where s_k = k-(n-1)
    #          = dx * sum_v h[v] * (1 - h[v + s_k]) for valid v
    
    # If h[i] changes by delta_i and h[j] changes by delta_j:
    # delta_c[k] = dx * [delta_i * (1-h[i+s_k]) * 1_{valid} + delta_j * (1-h[j+s_k]) * 1_{valid}
    #           - sum_v' h[v'] * delta_{h[v'+s_k]} * 1_{valid}]
    # where the second term accounts for h[i] or h[j] appearing as the second factor
    
    # More explicitly:
    # delta_c[k] = dx * [delta_i * (1-h[i+s_k]) * 1_{0<=i+s_k<n}
    #           - delta_i * h[i-s_k] * 1_{0<=i-s_k<n}  (h[i] as second factor)
    #           + delta_j * (1-h[j+s_k]) * 1_{0<=j+s_k<n}
    #           - delta_j * h[j-s_k] * 1_{0<=j-s_k<n}]
    
    # For our case: delta_i = -1 (h[i] goes 1→0), delta_j = +1 (h[j] goes 0→1)
    
    delta_c = np.zeros(2*n-1)
    
    # i: delta = -1
    mask1 = (i + s_all >= 0) & (i + s_all < n)
    mask2 = (i - s_all >= 0) & (i - s_all < n)
    delta_c[mask1] += -dx * (1.0 - h[np.clip(i + s_all, 0, n-1)[mask1]])
    delta_c[mask2] += dx * h[np.clip(i - s_all, 0, n-1)[mask2]]
    
    # j: delta = +1
    mask3 = (j + s_all >= 0) & (j + s_all < n)
    mask4 = (j - s_all >= 0) & (j - s_all < n)
    delta_c[mask3] += dx * (1.0 - h[np.clip(j + s_all, 0, n-1)[mask3]])
    delta_c[mask4] += -dx * h[np.clip(j - s_all, 0, n-1)[mask4]]
    
    return delta_c
```

```python
def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:

def binary_anneal(h_start, n_iter, T_init, T_min, rng):
    """Simulated annealing on binary sequences with O(n) per iteration."""
    n = len(h_start)
    dx = 2.0 / n
    s_all = np.arange(2*n-1) - (n-1)
    
    h = (h_start > 0.5).astype(float)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target = n // 2
    if len(ones) > target:
        excess = rng.choice(ones, len(ones) - target, replace=False)
        h[excess] = 0.0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    curr_c5 = float(np.max(corr))
    best_h = h.copy()
    best_c5 = curr_c5
    
    for it in range(n_iter):
        T = T_init * (T_min / T_init) ** (it / max(n_iter - 1, 1))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        
        # Compute delta for swap i(1→0), j(0→1)
        delta_c = np.zeros(2*n-1)
        
        # i was 1, becomes 0 (delta = -1)
        m1 = (i + s_all >= 0) & (i + s_all < n)
        m2 = (i - s_all >= 0) & (i - s_all < n)
        if np.any(m1):
            delta_c[m1] += -dx * (1.0 - h[i + s_all[m1]])
        if np.any(m2):
            delta_c[m2] += dx * h[i - s_all[m2]]
        
        # j was 0, becomes 1 (delta = +1)
        m3 = (j + s_all >= 0) & (j + s_all < n)
        m4 = (j - s_all >= 0) & (j - s_all < n)
        if np.any(m3):
            delta_c[m3] += dx * (1.0 - h[j + s_all[m3]])
        if np.any(m4):
            delta_c[m4] += -dx * h[j - s_all[m4]]
        
        new_corr = corr + delta_c
        new_c5 = float(np.max(new_corr))
        
        if new_c5 < curr_c5 or (T > 0 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    return best_h, best_c5
```
