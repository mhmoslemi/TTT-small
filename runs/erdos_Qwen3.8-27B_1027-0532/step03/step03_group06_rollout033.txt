            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

    """Compute smooth max and its gradient contribution to correlation"""
    shifted = corr * dx - np.max(corr) * dx
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = np.max(corr) * dx + np.log(sum_exp) / alpha
    return F, w

    """
    Gradient of smooth max of correlations with respect to h.
    w are the weights on each correlation lag.
    corr[k] for k in [-(n-1), n-1] is stored at index k+(n-1) in the full output.
    
    d/dh[j] corr[k] = (1-h[j+k]) if 0<=j+k<n, minus h[j-k] if 0<=j-k<n

    grad = np.zeros(n)
    for ki in range(len(w)):
        if w[ki] < 1e-10:
            continue
        k = ki - (n - 1)
        wi = w[ki]
        # Contribution from (1-h[j+k]) term: for j where 0<=j+k<n, i.e., j in [-k, n-1-k]
        j_lo = max(0, -k)
        j_hi = min(n-1, n-1-k)
        if j_lo <= j_hi:

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2 * n - 2, i + n - 1)
        if klo_c <= khi_c:
            j_idx_c = i - np.arange(klo_c, khi_c + 1) + (n - 1)
            term_c = np.sum(w[klo_c:khi_c + 1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad


def lbfgs_stage(h_start, n, target, alpha, maxiter=300, ftol=1e-15):
    h_cur = project_feasible(h_start.copy(), target)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def slsqp_stage(h_start, n, target, alpha, maxiter=200):
    h_cur = project_feasible(h_start.copy(), target)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h_cur


def run_full_refine(h_start, n, target, alpha_schedule, maxiter_per_stage, ftol, time_fn):
    h_cur = project_feasible(h_start.copy(), target)
    current_score = compute_c5(h_cur)
    for alpha in alpha_schedule:
        if time_fn() < 30:
            break
        h_new = lbfgs_stage(h_cur, n, target, alpha, maxiter_per_stage, ftol)
        sc = compute_c5(h_new)
        if sc <= current_score:
            h_cur = h_new
            current_score = sc
    return h_cur, current_score


def solve_lp_for_h(g, n, target):
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))
    for ki in range(n_lags):
        k = ki - (n - 1)
        i_lo = max(0, -k)
        i_hi = min(n - 1, n - 1 - k)
        for i in range(i_lo, i_hi + 1):
            A_ub[ki, i] = g[i + k]
        A_ub[ki, -1] = -1.0
    b_ub = np.zeros(n_lags)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    bounds = [(0, 1)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x[:n]
    except Exception:
        pass
    return None


def alternating_lp(h_init, n, target, max_iters=25, time_fn=None):
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time_fn is not None and time_fn() < 35:
            break
        g = 1.0 - h
        h_new = solve_lp_for_h(g, n, target)
        if h_new is None:
            break
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
    return best_h, best_score


def adam_refine(h_start, n, target, alpha, steps=3000, lr=0.005, b1=0.9, b2=0.999):
    h = project_feasible(h_start.copy(), target)
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    best_h = h.copy()
    best_sc = compute_c5(h)
    for step in range(steps):
        if step % 500 == 0:
            sc = compute_c5(h)
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
        F, grad = smooth_obj_and_grad(h, alpha)
        t += 1
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad ** 2
        m_hat = m_v / (1 - b1 ** t)
        v_hat = v_v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_feasible(h_new, target)
    sc = compute_c5(h)
    if sc < best_sc:
        best_sc = sc
        best_h = h.copy()
    return best_h, best_sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def make_modular_h(n, target, p, phase, rng):
    h = np.zeros(n)
    for i in range(n):
        if ((i + phase) % p) < p // 2:
            h[i] = 1.0
    return project_feasible(h, target)
