def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

        weights[np.argmax(corr)] = 1.0

    obj = (m_max + np.log(total) / alpha) * dx if alpha > 0 else m_max * dx
    
    # Gradient

import time
import numpy as np
from scipy.optimize import minimize

_g_ts = 0.0
_g_n_pts = 0
_g_alpha = 0.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smooth_obj_and_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m_max = np.max(corr)
    shifted = corr - m_max
    scaled = alpha * shifted
    exps = np.exp(scaled)
    total = np.sum(exps)
    if total < 1e-300:
        max_idx = np.argmax(corr)
        weights = np.zeros(len(corr))
        weights[max_idx] = 1.0
        obj = m_max * dx
    else:
        weights = exps / total
        obj = (m_max + np.log(total) / alpha) * dx
    grad = np.zeros(n)
    for m in range(len(corr)):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - (n - 1)
        if k >= 0:
            mm = n - k
            if mm > 0:
                grad[:mm] += w * (1.0 - h[k:k + mm])
                grad[k:] += -w * h[:mm]
        else:
            kp = -k
            mm = n - kp
            if mm > 0:
                grad[kp:] += w * (1.0 - h[:mm])
                grad[:mm] += -w * h[kp:kp + mm]
    grad *= dx
    return obj, grad

def _slsqp_obj(x):
    o, _ = _smooth_obj_and_grad(x, _g_alpha, _g_n_pts)
    return o

def _slsqp_grad(x):
    _, g = _smooth_obj_and_grad(x, _g_alpha, _g_n_pts)
    return g

def _slsqp_con_fun(x):
    return x.sum() - _g_ts

def _slsqp_con_jac(x):
    return np.ones(_g_n_pts)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):
    global _g_ts, _g_n_pts, _g_alpha
    _g_ts = target_sum
    _g_n_pts = n
    _g_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0.copy()

def _coord_descent(h0, n, target_sum, max_rounds=12, ternary_iters=22):
    h = h0.copy()
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old_val = h[j]
            h[j] = 0.0
            f0 = _c5_of(h)
            h[j] = 1.0
            f1 = _c5_of(h)
            h[j] = 0.5
            f_mid = _c5_of(h)
            if f_mid >= min(f0, f1) + 1e-14:
                best_t = 0.0 if f0 <= f1 else 1.0
            else:
                lo_t, hi_t = 0.0, 1.0
                for _ in range(ternary_iters):
                    m1 = lo_t + (hi_t - lo_t) / 3.0
                    m2 = hi_t - (hi_t - lo_t) / 3.0
                    h[j] = m1
                    f_m1 = _c5_of(h)
                    h[j] = m2
                    f_m2 = _c5_of(h)
                    if f_m1 < f_m2:
                        hi_t = m2
                    else:
                        lo_t = m1
                best_t = (lo_t + hi_t) / 2.0
            h[j] = best_t
            if abs(best_t - old_val) > 1e-9:
                improved = True
        h = _project(h, target_sum)
        if not improved:
            break
    return h

def _de_search(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    curves = np.sin(np.linspace(0, 2 * np.pi, n))
    pop = [h0.copy()]
    for i in range(pop_size - 1):
        if i < 3:
            base = h0 + 0.15 * rng.randn(n)
        elif i < 6:
            base = h0 + 0.3 * rng.randn(n)
        else:
            base = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
            if rng.random() < 0.3:
                base = _project(h0 + 0.2 * curves + 0.1 * rng.randn(n), target_sum)
        pop.append(_project(base, target_sum))
    caches = [_c5_of(p) for p in pop]
    best_h = pop[int(np.argmin(caches))].copy()
    best_c5 = min(caches)
    stall = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.85 - 0.4 * t_frac
        cr = 0.5 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            cands = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(cands), size=3, replace=False)
            r1, r2, r3 = cands[r1], cands[r2], cands[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.random(n) < cr
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5 - 1e-12:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved:
            stall += 1
        else:
            stall = 0
        if stall >= 20:
            break
    return best_h, best_c5

def _nesterov_subgrad(h0, n, target_sum, n_iters, alpha0, sg_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        top_m = np.argsort(corr)[::-1][:sg_m]
        weights = np.exp(40.0 * (corr[top_m] - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(sg_m)
            weights[0] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for idx in range(len(top_m)):
            m_idx = top_m[idx]
            k = m_idx - (n - 1)
            if k >= 0:
                mm = n - k
                if mm > 0:
                    g[:mm] += weights[idx] * (1.0 - y[k:k + mm])
                    g[k:] += weights[idx] * (-y[:mm])
            else:
                kp = -k
                mm = n - kp
                if mm > 0:
                    g[kp:] += weights[idx] * (1.0 - y[:mm])
                    g[:mm] += weights[idx] * (-y[kp:kp + mm])
        g *= dx
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _progressive_slsqp(h0, n, target_sum, alphas, maxiter_each, deadline):
    h_curr = h0.copy()
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_new = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=maxiter_each)
        c5_new = _c5_of(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

def _optimize_one_n(h_start, n, target_sum, deadline, rng):
    best_h = h_start.copy()
    best_c5 = _c5_of(h_start)

    # Phase 1: DE global search
    de_deadline = min(deadline, time.time() + 120)
    pop_size = min(14, max(10, n // 10))
    n_gen = min(80, max(40, int((de_deadline - time.time()) / max(0.1, pop_size * 0.03))))
    if n_gen < 5:
        n_gen = 5
    h_de, c5_de = _de_search(h_start, n, target_sum, pop_size, n_gen, de_deadline, rng)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Phase 2: Progressive smooth SLSQP
    if time.time() < deadline - 150:
        alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
        h_sl, c5_sl = _progressive_slsqp(best_h, n, target_sum, alphas, 350, deadline - 150)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()

    # Phase 3: Convex coordinate descent
    if time.time() < deadline - 80:
        h_cd = _coord_descent(best_h, n, target_sum, max_rounds=12, ternary_iters=22)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: More SLSQP at high alpha
    if time.time() < deadline - 60:
        alphas2 = [5000, 10000, 20000]
        h_sl2, c5_sl2 = _progressive_slsqp(best_h, n, target_sum, alphas2, 400, deadline - 60)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2.copy()

    # Phase 5: Final CD
    if time.time() < deadline - 30:
        h_cd2 = _coord_descent(best_h, n, target_sum, max_rounds=8, ternary_iters=25)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    # Phase 6: Nesterov from perturbed starts
    if time.time() < deadline - 20:
        n_start = best_h.copy()
        n_c = _c5_of(n_start)
        for pert in range(5):
            if time.time() > deadline - 15:
                break
            h_pert = _project(n_start + 0.03 * rng.randn(n), target_sum)
            h_nest, c5_nest = _nesterov_subgrad(h_pert, n, target_sum,
                                                 n_iters=150, alpha0=0.4,
                                                 sg_m=5, deadline=deadline - 15)
            if c5_nest < n_c:
                n_c = c5_nest
                n_start = h_nest.copy()
                # Quick SLSQP polish
                h_pol = _slsqp_opt(n_start, n, target_sum, 5000, maxiter=200)
                c5_pol = _c5_of(h_pol)
                if c5_pol < n_c:
                    n_c = c5_pol
                    n_start = h_pol.copy()
        if n_c < best_c5:
            best_c5 = n_c
            best_h = n_start.copy()

    # Phase 7: Final SLSQP + CD
    if time.time() < deadline - 5:
        h_fin = _slsqp_opt(best_h, n, target_sum, 15000, maxiter=400)
        c5_fin = _c5_of(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()
    if time.time() < deadline - 3:
        h_fin_cd = _coord_descent(best_h, n, target_sum, max_rounds=6, ternary_iters=25)
        c5_fin_cd = _c5_of(h_fin_cd)
        if c5_fin_cd < best_c5:
            best_c5 = c5_fin_cd
            best_h = h_fin_cd.copy()

    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):

        h_opt, c5_opt = _optimize_one_n(h_start, n, target_sum, phase_deadline, rng)
