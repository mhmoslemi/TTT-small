def compute_swap_delta_fast(h, a, b, n):
    """Vectorized computation of delta for swap h[a]:1->0, h[b]:0->1"""
    m = np.arange(-n+1, n)  # all possible lags
    delta = np.zeros(2*n-1)
    
    # Case 1: positions where a+m is valid, m≠0
    mask1 = (m != 0) & (a + m >= 0) & (a + m < n)
    if np.any(mask1):
        vals = a[m[mask1]] + m[mask1]  # = a+m for valid m
        delta[m[mask1] + n - 1] -= (1.0 - h[vals])
    
    # Case 2: positions where b+m is valid, m≠0
    mask2 = (m != 0) & (b + m >= 0) & (b + m < n)
    if np.any(mask2):
        vals = b + m[mask2]
        delta[m[mask2] + n - 1] += (1.0 - h[vals])
    
    # Case 3: positions where a-m is valid, m≠0, m≠a-b
    mask3 = (m != 0) & (m != a-b) & (a - m >= 0) & (a - m < n)
    if np.any(mask3):
        vals = a - m[mask3]
        delta[m[mask3] + n - 1] += h[vals]
    
    # Case 4: positions where b-m is valid, m≠0, m≠b-a
    mask4 = (m != 0) & (m != b-a) & (b - m >= 0) & (b - m < n)
    if np.any(mask4):
        vals = b - m[mask4]
        delta[m[mask4] + n - 1] -= h[vals]
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def compute_swap_delta(h, a, b, n):

    m_all = np.arange(-n + 1, n)
    delta = np.zeros(2 * n - 1)

        vals1 = a + m_all[valid1]

    # Wait, h[b] changes 0->1, so the h[n] term changes from 0 to h[b+m]=1*(1-h[b+m])
    # Actually: old = h[b]*(1-h[b+m]) = 0; new = h'[b]*(1-h'[b+m]) = 1*(1-h[b+m])
    # delta[m] += (1-h[b+m]) for valid b+m, m≠0
    valid2 = (m_all != 0) & (b + m_all >= 0) & (b + m_all < n)
    if np.any(valid2):

        vals2 = b + m_all[valid2]

    # Case 3: n_idx + m = a, i.e., n_idx = a-m
    # Old: h[a-m]*(1-h[a]) = h[a-m]*0 = 0
    # New: h[a-m]*(1-h'[a]) = h[a-m]*1 = h[a-m]
