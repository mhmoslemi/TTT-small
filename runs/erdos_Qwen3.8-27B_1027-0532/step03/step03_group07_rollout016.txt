    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])

            diff -= add

def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()

    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    d = target - h.sum()
    if abs(d) < 1e-12:
        return h
    if d > 0:
        for i in np.argsort(1.0 - h):
            if d <= 1e-12:
                break
            a = min(d, 1.0 - h[i])
            h[i] += a
            d -= a
    else:
        for i in np.argsort(h):
            if d >= -1e-12:
                break
            s = min(-d, h[i])
            h[i] -= s
            d += s
    return h
