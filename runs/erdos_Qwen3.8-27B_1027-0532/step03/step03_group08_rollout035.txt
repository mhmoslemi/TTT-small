import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_full(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * (2.0 / len(h))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_eq(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:

def _grad_at_lag(h, lag, n):
    """Gradient of sum_i h[i]*(1-h[i+lag]) with respect to h, for a specific lag offset."""
    # lag is the offset in the correlation: actual_shift = lag - (n-1)
    actual_shift = lag - (n - 1)
    grad = np.zeros(n)
    dx = 2.0 / n
    if actual_shift >= 0:
        m = n - actual_shift
        if m > 0:
            # sum_{i=0}^{m-1} h[i]*(1-h[i+actual_shift])
            # d/dh[j] = (1-h[j+actual_shift]) for j in [0, m-actual_shift) [from first factor]
            #          + (-h[j-actual_shift]) for j in [actual_shift, m-1) [from second factor]
            for j in range(min(m, n - actual_shift)):
                grad[j] += (1.0 - h[j + actual_shift])
            for j in range(actual_shift, m):
                grad[j] += (-h[j - actual_shift])
    else:
        s = -actual_shift
        m = n - s
        if m > 0:
            # sum_{i=s}^{n-1} h[i]*(1-h[i-s])
            # = sum_{i=0}^{m-1} h[i+s]*(1-h[i])
            for j in range(s, n):
                grad[j] += (1.0 - h[j - s])
            for j in range(0, m):
                grad[j] += (-h[j + s])
    return grad * dx

def _smooth_max(corr, alpha):
    """Log-sum-exp approximation of max, for the correlation values."""
    m = np.max(corr)
    if m < -1000:
        return m
    shifted = alpha * (corr - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    return (m + np.log(np.sum(np.exp(shifted))) / alpha)

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha, step0, deadline, top_m=5):
    """Nesterov accelerated gradient with smooth max approximation."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Compute correlation at y
        corr = np.correlate(y, 1.0 - y, mode='full') * dx
        # Find top m lags
        top_idx = np.argsort(corr)[::-1][:top_m]
        # Subgradient: average gradient at top m lags weighted by softmax
        max_c = corr[top_idx[0]]
        weights = np.exp(50.0 * (corr[top_idx] - max_c))
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            g += w * _grad_at_lag(y, idx, n)
        
        # Nesterov step
        step = step0 / np.sqrt(float(it + 1))
        x_new = _project_box_eq(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_eq(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

def _slsqp_polish(h0, n, target_sum, maxiter, alpha, deadline):
    """SLSQP with smooth approximation for polish."""
    corr_full = np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0/n)
        return _smooth_max(c, alpha)
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        if time.time() > deadline:
            return h0
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _fourier_init(n, seed, k_max):

    rng = np.random.RandomState(seed)

    h = np.zeros(n)

        h += a * np.cos(2.0 * np.pi * k * x / 2.0)
        h += b * np.sin(2.0 * np.pi * k * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    h = _fix_sum(h, n / 2.0)
    return h

def _de_optimize(h0, n, target_sum, pop_size, max_gen, deadline, rng):
    """Differential Evolution optimizer."""
    F = 0.8
    CR = 0.7
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        strategy = i % 5
        if strategy == 0:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif strategy == 1:
            pop[i] = _fix_sum(1.0 - h0 + 0.1*(rng.rand(n)-0.5), target_sum)
        elif strategy == 2:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t*h0 + (1-t)*rng.rand(n), target_sum)
        elif strategy == 3:
            perm = rng.permutation(n)
            pop[i] = _fix_sum(h0[perm], target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.2*(rng.rand(n)-0.5), target_sum)
    
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall = 0
    
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F_gen = F - 0.3 * (gen / max_gen)  # Decrease F over time
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1, r2 = rng.choice(pop_size, 2, replace=False)
            while r1 == i or r2 == i or r1 == r2:
                r1, r2 = rng.choice(pop_size, 2, replace=False)
            mutant = pop[best_idx] + F_gen * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            mask[rng.randint(n)] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0, 1), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size * 2:
            # Restart worst half
            half = pop_size // 2
            worst = np.argsort(pop_vals)[half:]
            for idx in worst:
                t_mix = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t_mix*best_h + (1-t_mix)*rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    
    return best_h, best_val

    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)

            sub = min(-diff, h[i])

            diff += sub
    return h

        shifted = sg_alpha * (corr - max_c)
        shifted = np.clip(shifted, -400, 0)
        weights = np.exp(shifted)
        total_w = weights.sum()
        if total_w < 1e-300:
            top_idx = np.array([int(np.argmax(corr))])
            w_val = np.array([1.0])
        else:
            weights = weights / total_w
            top_idx = np.argsort(weights)[::-1][:top_m]
            w_val = weights[top_idx].copy()
            w_val = w_val / w_val.sum()
        g = np.zeros(n)
        for wi, ki in zip(w_val, top_idx):
            shift = int(ki) - (n - 1)
            g += wi * _grad_lag(y, shift, n)
        step = step0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_opt(h0, n, target, pop_size, max_gen, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        mod = i % 7
        if mod == 1:
            pop[i] = _fix_sum(rng.rand(n), target)
        elif mod == 2:
            pop[i] = _fix_sum(1.0 - h0 + 0.08 * (rng.rand(n) - 0.5), target)
        elif mod == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target)
        elif mod == 4:
            pop[i] = _fix_sum(h0[rng.permutation(n)], target)
        elif mod == 5:
            pop[i] = h0.copy()
            bs = max(2, n // 8)
            s1 = rng.randint(0, n - 2 * bs)
            s2 = rng.randint(0, n - 2 * bs)
            pop[i][s1:s1 + bs] = h0[s2:s2 + bs].copy()
            pop[i][s2:s2 + bs] = h0[s1:s1 + bs].copy()
            pop[i] = _fix_sum(pop[i], target)
        elif mod == 6:
            freq = rng.randint(2, 8)
            x_idx = np.arange(n)
            wave = 0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x_idx / n)
            pop[i] = _fix_sum(rng.uniform(0.7, 1.0) * h0 + rng.uniform(0.0, 0.3) * wave, target)
        else:
            pop[i] = _fix_sum(h0 + 0.15 * (rng.rand(n) - 0.5), target)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = 0.9 - 0.4 * (gen / max_gen)
        CR = 0.6 + 0.2 * (gen / max_gen)
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            mask[rng.randint(n)] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size:
            half = pop_size // 2
            worst = np.argsort(pop_vals)[half:]
            for idx in worst:
                t_mix = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t_mix * best_h + (1.0 - t_mix) * rng.rand(n), target)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

def _con_fun(x):
    return x.sum() - _gs_target

def _con_jac(x):
    return np.ones(_gs_n)

def _slsqp(h0, n, target, maxiter, alpha, deadline):
    global _gs_target, _gs_n
    _gs_target = target
    _gs_n = n
    corr_init = np.correlate(h0, 1.0 - n * 0.5 / n, mode='full')

    def _obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        m = np.max(c)
        if m < -500:
            return m
        shifted = alpha * (c - m)
        shifted = np.clip(shifted, -400, 0)
        return float(np.max(c) + np.log(np.sum(np.exp(shifted))) / alpha)

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        if time.time() > deadline:
            return h0
        res = minimize(_obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _coord_descent(h0, n, target, rounds, deadline):
    h = h0.copy()
    current = _c5_of(h)
    vals = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
    for rnd in range(rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            best_v = old
            best_c = current
            for v in vals:
                if abs(v - old) < 1e-7:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
                current = best_c
        if not improved:
            break
    return h, current

def _sa_optimize(h0, n, target, max_iters, T_start, T_end, deadline, rng):
    h = h0.copy()
    h_val = _c5_of(h)
    best_h = h.copy()
    best_val = h_val
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))
        h_new = h.copy()
        n_pert = rng.randint(1, max(2, n // 10))
        indices = rng.choice(n, n_pert, replace=False)
        for idx in indices:
            r = rng.random()
            if r < 0.25:
                h_new[idx] = 0.0
            elif r < 0.5:
                h_new[idx] = 1.0
            elif r < 0.75:
                h_new[idx] = 0.5
            else:
                h_new[idx] = rng.random()
        h_new = _fix_sum(h_new, target)
        new_val = _c5_of(h_new)
        if new_val < h_val or rng.random() < np.exp(-(new_val - h_val) / max(T, 1e-10)):
            h = h_new
            h_val = new_val
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()
    return best_h, best_val

def _interpolate(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.arange(n_src)
    x_new = np.linspace(0, n_src - 1, n_dst)
    return np.interp(x_new, x_old, h_src)

def _fourier_start(n, seed, k_max):
    rng = np.random.RandomState(seed + 7919)
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, k_max + 1):
        a = rng.uniform(-0.08, 0.08)
        b = rng.uniform(-0.08, 0.08)
        h += a * np.cos(np.pi * k * x)
        h += b * np.sin(np.pi * k * x)
    h = np.clip(h, 0.0, 1.0)
    return _fix_sum(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 200
        h_init = _fourier_start(n_init, seed, 8)

    rng = np.random.RandomState(seed)
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    n_list = [n_init]
    for delta in [-10, 10, -20, 20, 30, 50, -30, 70, -50]:
        nn = n_init + delta
        if nn % 2 == 0 and 120 <= nn <= 320:
            n_list.append(nn)
    for nn in [180, 200, 220, 240, 250, 280, 300]:
        if nn % 2 == 0 and nn not in n_list:
            n_list.append(nn)
    n_list = sorted(set(n_list), key=lambda x: abs(x - n_init))

    phase_budget = deadline - t_start

    for n in n_list:
        if time.time() > deadline - 60:
            break
        target = n / 2.0
        remaining = deadline - time.time()
        if remaining < 60:
            break

        h_start = _interpolate(best_h, best_n, n)
        h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target)
        if n != n_init:
            h_alt = _interpolate(h_init, n_init, n)
            h_alt = _fix_sum(np.clip(h_alt, 0.0, 1.0), target)
            if _c5_of(h_alt) < _c5_of(h_start):
                h_start = h_alt

        time_alloc = min(remaining - 30, 300)
        t1 = time.time()

        h1, c5_1 = _nesterov(h_start, n, target, int(200 * time_alloc / 300), 0.4, 7, 60.0, t1 + time_alloc * 0.3)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n

        t2 = time.time()
        if t2 < deadline - 80:
            alloc2 = min(deadline - 80 - t2, 180)
            de_pop = min(35, max(20, int(alloc2 * 2)))
            de_gens = min(80, max(30, int(alloc2 * 0.3)))
            h2, c5_2 = _de_opt(h1, n, target, de_pop, de_gens, deadline - 80, rng)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
                best_n = n
            cur = h2 if c5_2 < c5_1 else h1
        else:
            cur = h1

        t3 = time.time()
        if t3 < deadline - 50:
            h3, c5_3 = _nesterov(cur, n, target, 200, 0.25, 10, 120.0, deadline - 50)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
                best_n = n
            cur = h3 if c5_3 < (c5_2 if c5_2 < c5_1 else c5_1) else cur

        t4 = time.time()
        if t4 < deadline - 30:
            h4 = _slsqp(cur, n, target, 800, 250.0, deadline - 30)
            c5_4 = _c5_of(h4)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()
                best_n = n
            cur = h4 if c5_4 < best_c5 else cur

        t5 = time.time()
        if t5 < deadline - 20:
            h5, c5_5 = _coord_descent(cur, n, target, 10, deadline - 20)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()
                best_n = n
            cur = h5 if c5_5 < best_c5 else cur

    t6 = time.time()
    if t6 < deadline - 20 and best_h is not None:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        h_sa, c5_sa = _sa_optimize(best_h, n_fin, tgt_fin, 2000, 0.03, 0.002, deadline - 15, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_fin

    t7 = time.time()
    if t7 < deadline - 10 and best_h is not None:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        h_ng, c5_ng = _nesterov(best_h, n_fin, tgt_fin, 200, 0.15, 12, 150.0, deadline - 10)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    t8 = time.time()
    if t8 < deadline - 5 and best_h is not None:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        h_fin = _slsqp(best_h, n_fin, tgt_fin, 3000, 350.0, deadline - 5)
        c5_fin = _c5_of(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    t9 = time.time()
    if t9 < deadline - 3 and best_h is not None:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        h_cd, c5_cd = _coord_descent(best_h, n_fin, tgt_fin, 20, deadline - 3)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    t10 = time.time()
    if t10 < deadline - 1 and best_h is not None:
        n_fin = best_n
        tgt_fin = n_fin / 2.0
        h_pol = _slsqp(best_h, n_fin, tgt_fin, 5000, 400.0, deadline - 1)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()
            best_n = n_fin

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
