```python
def slsqp_optimize(h_init, eps, maxiter, n=None):

        n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        dx = 2.0 / n
        x_scaled = (corr * dx - cmax * dx) / eps
        x_scaled -= np.max(x_scaled)
        return cmax * dx + eps * np.log(np.sum(np.exp(x_scaled)))
    
    def obj_grad(x):
        dx = 2.0 / n
        omh = 1.0 - x
        corr = np.correlate(x, 1.0 - x, mode='full')
        cmax = np.max(corr)
        x_scaled = (corr * dx - cmax * dx) / eps
        x_scaled -= np.max(x_scaled)
        ex = np.exp(x_scaled)
        w = ex / np.sum(ex)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * x[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * x[:m]
        return grad
    
    try:
        result = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
```

Actually, here's a key optimization: instead of recomputing the full correlation after each swap, I can compute the delta. For a swap of positions a (1→0) and b (0→1):

The correlation at lag s changes as follows:
- Position a no longer contributes as a "1": for all s where a is a valid position, the term h[a]*(1-h[a-s]) changes from (1-h[a-s]) to 0. So delta_a[s] = -(1-h[a-s]) for valid s.
- Position b now contributes as a "1": delta_b[s] = +(1-h[b-s]) for valid s.

The valid s range for position i is: s from i-(n-1) to i (in terms of the lag), or in corr index k: from i to i+(n-1).

So the total delta for the corr array is:
delta[k] = -(1-h[a-(k-n+1)]) for k in [a, a+n-1] (clamped to [0, 2n-2])
         + (1-h[b-(k-n+1)]) for k in [b, b+n-1] (clamped to [0, 2n-2])

This is O(n) per swap evaluation. For n=180, that's 180 operations. We can vectorize this.

```python
def evaluate_swap(h, corr, a, b, n):
    """
    Compute the new max corr after swapping h[a] (1->0) and h[b] (0->1).
    a is a position with h[a]=1, b is a position with h[b]=0.
    """
    # delta at position k (corr index) from removing a:
    # a contributes to corr[k] for k in [a, a+n-1] (clamped)
    # contribution is h[a]*(1-h[a-(k-(n-1))]) = 1*(1-h[a-k+n-1])
    # After removal: 0. So delta = -(1-h[a-k+n-1])
    
    k_lo_a = max(0, a)
    k_hi_a = min(2*n-2, a + n - 1)
    k_lo_b = max(0, b)
    k_hi_b = min(2*n-2, b + n - 1)
    
    k_lo = min(k_lo_a, k_lo_b)
    k_hi = max(k_hi_a, k_hi_b)
    
    new_corr = corr.copy()
    
    # Apply a's removal
    ka = np.arange(k_lo_a, k_hi_a + 1)
    ja = a - (ka - (n - 1))  # = a - ka + n - 1
    # ja should be in [0, n-1]
    delta_a = -(1.0 - h[ja])
    new_corr[ka] += delta_a
    
    # Apply b's addition
    kb = np.arange(k_lo_b, k_hi_b + 1)
    jb = b - (kb - (n - 1))  # = b - kb + n - 1
    delta_b = (1.0 - h[jb])
    new_corr[kb] += delta_b
    
    return float(np.max(new_corr))
```

- The old contribution of a was h[a]*(1-h[a-s]) = 1*(1-h[a-s])
- The new contribution is 0
- So the change is -(1-h[a-s]), which is what I computed

- The old contribution of b was h[b]*(1-h[b-s]) = 0*(1-h[b-s]) = 0
- The new contribution is 1*(1-h[b-s])
- So the change is +(1-h[b-s]), which is what I computed

But wait: what if a and b are close together? Then h[a] appears in the computation for b's delta, and vice versa. Let me check:

For b's delta: delta_b at position k is (1-h[b-(k-n+1)]). If b-(k-n+1) = a, then we're looking at h[a] which is still 1 in the original. But after the swap, h[a] = 0. So the delta should be (1-0) = 1, but we computed (1-h[a]) = (1-1) = 0. This is wrong!

So when a and b are within n-1 of each other (which they always are since both are in [0,n-1]), there's an interaction.

Let me handle this:
- If b-(k-n+1) = a for some k (i.e., k = b-a+n-1), then the delta_b at that k should be (1-0) = 1 instead of (1-1) = 0.
- Similarly, if a-(k-n+1) = b for some k (i.e., k = a-b+n-1), then the delta_a at that k should be -(1-1) = 0 instead of -(1-0) = -1.

So the corrections are:
- If k = b-a+n-1 is in the valid range for b: delta_b[k] should be 1 instead of 0. So add 1.
- If k = a-b+n-1 is in the valid range for a: delta_a[k] should be 0 instead of -1. So add 1.

Let me verify: if a > b (so a-b > 0), then k = a-b+n-1 is in [n-1, 2n-2]. This is valid if a-b+n-1 <= 2n-2 and a-b+n-1 >= 0, which is always true for a,b in [0,n-1] (given a > b). def evaluate_swap(h, corr, a, b, n):
```python
    """
    Compute new max corr after swapping h[a] (1->0) and h[b] (0->1).
    """
    k_lo_a = max(0, a)
    k_hi_a = min(2*n-2, a + n - 1)
    k_lo_b = max(0, b)
    k_hi_b = min(2*n-2, b + n - 1)
    
    k_lo = min(k_lo_a, k_lo_b)
    k_hi = max(k_hi_a, k_hi_b)
    
    new_corr = corr[k_lo:k_hi+1].copy()
    offset = k_lo
    
    # Apply a's removal
    ka = np.arange(k_lo_a, k_hi_a + 1)
    ja = a - (ka - (n - 1))
    delta_a = -(1.0 - h[ja])
    new_corr[ka - offset] += delta_a
    
    # Apply b's addition
    kb = np.arange(k_lo_b, k_hi_b + 1)
    jb = b - (kb - (n - 1))
    delta_b = (1.0 - h[jb])
    new_corr[kb - offset] += delta_b
    
    # Correct for interaction
    # If a appears in b's range: k = b - a + n - 1
    k_inter1 = b - a + n - 1

        # delta_b at k_inter1 should be 1 (since h[a] will be 0)
        # but we computed (1-h[a]) = (1-1) = 0
        # So add 1
        new_corr[k_inter1 - offset] += 1.0

    return float(np.max(new_corr))
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    n_ones = int(b.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h[one_idx])[:n_ones - target]
        b[one_idx[order]] = 0
    elif n_ones < target:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])[:target - n_ones]
        b[zero_idx[order]] = 1
    return b

def eval_swap(h, corr, a, b, n):
    k_hi_full = 2 * n - 2
    k_lo_a = a
    k_hi_a = min(k_hi_full, a + n - 1)
    k_lo_b = b
    k_hi_b = min(k_hi_full, b + n - 1)
    k_lo = min(k_lo_a, k_lo_b)
    k_hi = max(k_hi_a, k_hi_b)
    new_corr = corr[k_lo:k_hi + 1].astype(float).copy()
    off = k_lo
    ka = np.arange(k_lo_a, k_hi_a + 1)
    ja = a - (ka - (n - 1))
    new_corr[ka - off] -= (1.0 - h[ja])
    kb = np.arange(k_lo_b, k_hi_b + 1)
    jb = b - (kb - (n - 1))
    new_corr[kb - off] += (1.0 - h[jb])
    k_int1 = b - a + n - 1
    if k_lo_b <= k_int1 <= k_hi_b and 0 <= k_int1 <= k_hi_full:
        new_corr[k_int1 - off] += 1.0
    k_int2 = a - b + n - 1
    if k_lo_a <= k_int2 <= k_hi_a and 0 <= k_int2 <= k_hi_full:
        new_corr[k_int2 - off] += 1.0
    return float(np.max(new_corr))

def local_search_binary(h, time_budget, rng, n_candidates=80):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    t0 = time.time()
    corr = corr_full(h).astype(float)
    best_max = float(np.max(corr))
    no_improve = 0

    while time.time() - t0 < time_budget:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        worst_k = int(np.argmax(corr))
        s_worst = worst_k - (n - 1)
        if s_worst >= 0:
            vn = np.arange(0, n - s_worst)
        else:
            vn = np.arange(-s_worst, n)
        cmask = (h[vn] == 0) & (h[vn + s_worst] == 1)
        contrib = vn[cmask]
        best_candidate = None
        best_candidate_val = best_max
        n_try = min(n_candidates, len(ones) * len(zeros))
        for _ in range(n_try):
            if len(contrib) > 0 and rng.random() < 0.7:
                n_flip = contrib[rng.integers(len(contrib))]
                if s_worst >= 0:
                    vm = np.arange(0, n - s_worst)
                else:
                    vm = np.arange(-s_worst, n)
                rmask = (h[vm] == 1) & (h[vm + s_worst] == 0)
                reverses = vm[rmask]
                if len(reverses) > 0:
                    m_flip = reverses[rng.integers(len(reverses))]
                else:
                    m_flip = ones[rng.integers(len(ones))]
                if n_flip == m_flip:
                    m_flip = ones[rng.integers(len(ones))]
                if h[m_flip] != 1 or h[n_flip] != 0:
                    a = ones[rng.integers(len(ones))]
                    b = zeros[rng.integers(len(zeros))]
                    if a == b:
                        continue
                else:
                    a = m_flip
                    b = n_flip
            else:
                a = ones[rng.integers(len(ones))]
                b = zeros[rng.integers(len(zeros))]
                if a == b:
                    continue
            if h[a] != 1 or h[b] != 0:
                continue
            val = eval_swap(h, corr, a, b, n)
            if val < best_candidate_val:
                best_candidate_val = val
                best_candidate = (a, b)
        if best_candidate is not None:
            a, b = best_candidate
            h[a], h[b] = h[b], h[a]
            corr = corr_full(h).astype(float)
            best_max = best_candidate_val
            no_improve = 0
        else:

            if no_improve > 100:

                corr = corr_full(h).astype(float)
                best_max = float(np.max(corr))
                no_improve = 0
    return h, float(dx * best_max)

def smooth_max_obj(x, n, eps):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    xs = (corr - cmax) / eps
    xs -= np.max(xs)
    return cmax + eps * np.log(np.sum(np.exp(xs)))

def smooth_max_grad(x, n, eps):
    dx = 2.0 / n
    omh = 1.0 - x
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    xs = (corr - cmax) / eps
    xs -= np.max(xs)
    ex = np.exp(xs)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * x[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * x[:m]
    return grad

def slsqp_run(h_init, eps, maxiter, n):
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(smooth_max_obj, h0, args=(n, eps), jac=smooth_max_grad,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(res.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_continuation(h_init, eps_list, maxiter_list, n, time_limit, start_time):
    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)
    for i, eps in enumerate(eps_list):
        if time.time() - start_time > time_limit:
            break
        mi = maxiter_list[min(i, len(maxiter_list) - 1)]
        h_new, c5_new = slsqp_run(h_cur, eps, mi, n)
        if c5_new < c5_cur:
            c5_cur = c5_new
            h_cur = h_new.copy()
    return h_cur, c5_cur

def perturb_binary(h, rng, n_flips):

        if a != b:
            h[a], h[b] = h[b], h[a]
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
    return h
```
