import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:

            for i in order:
                if diff >= 0:

                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub

def _grad_corr_at_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_lse(corr, alpha):
    m = np.max(corr)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _softmax_weights(corr, m, temperature):
    max_c = np.max(corr)
    weights = np.exp(temperature * (corr - max_c))
    total = weights.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
        return weights, np.argmax(corr)
    weights = weights / total
    top_idx = np.argsort(weights)[::-1][:m]
    w_m = weights[top_idx].copy()
    w_m = w_m / w_m.sum()
    return w_m, top_idx

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n

    def _objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return _smooth_lse(corr, alpha) * (2.0 / n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})

def _nesterov_subgrad(h0, n, target_sum, n_iters, deadline, m=5, alpha0=0.4,
                       temp=30.0, use_adaptive=True):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    step_scale = alpha0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        w_m, top_idx = _softmax_weights(corr, m, temp)
        g = np.zeros(n)
        for idx in range(len(top_idx)):
            lag = int(top_idx[idx]) - (n - 1)
            g += w_m[idx] * _grad_corr_at_lag(y, lag, n, dx)
        if use_adaptive:
            step = step_scale / (1.0 + 0.01 * it)
        else:
            step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _nesterov_adaptive(h0, n, target_sum, n_iters, deadline, m=5, alpha0=0.5,

                stall = 0
    return best, best_val

def _coord_descent_binary(h0, n, max_rounds=30):
    h = h0.copy()
    best_c = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            if c0 < best_c - 1e-13:
                h[j] = 0.0
                best_c = c0
                improved = True
            elif c1 < best_c - 1e-13:
                h[j] = 1.0
                best_c = c1
                improved = True
        if not improved:
            break
    return h, best_c

def _coord_descent_multi(h0, n, max_rounds=10):
    h = h0.copy()
    vals = np.array([0.0, 0.1, 0.2, 0.25, 1.0/3.0, 0.4, 0.5, 0.6,
                     2.0/3.0, 0.75, 0.8, 0.9, 1.0])
    best_c = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h, best_c

def _simulated_annealing(h0, n, target_sum, t_start, t_end, n_sweeps, deadline, rng):
    h = h0.copy()
    best_h = h0.copy()
    best_c = _c5_of(h)
    current_c = best_c
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        temp = t_start * (t_end / t_start) ** (float(sweep) / max(n_sweeps - 1, 1))
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            new_val = np.clip(old + rng.normal(0, 0.3), 0.0, 1.0)
            h[j] = new_val
            new_c = _c5_of(h)
            delta = new_c - current_c
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                current_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = h.copy()
            else:
                h[j] = old
    return best_h, best_c

def _perturb_and_refine(h0, n, target_sum, deadline, rng, n_perturbs=20, perturb_sigma=0.08):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for _ in range(n_perturbs):
        if time.time() > deadline:
            break
        h_pert = h0.copy()
        n_change = rng.randint(max(1, n // 15), max(2, n // 4) + 1)
        idx = rng.choice(n, size=n_change, replace=False)
        for i in idx:
            h_pert[i] = np.clip(h_pert[i] + rng.normal(0, perturb_sigma), 0.0, 1.0)
        h_pert = _fix_sum(h_pert, target_sum)
        c5_pert = _c5_of(h_pert)
        if c5_pert < best_c5 + 0.002:
            h_n, c5_n = _nesterov_adaptive(h_pert, n, target_sum, n_iters=150,
                                            deadline=min(deadline, time.time() + 30),
                                            m=5, alpha0=0.3, temp=40.0)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
    return best_h, best_c5

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = min(caches)
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.85 - 0.4 * t_frac
        cr = 0.5 + 0.2 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.random(n) < cr
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 12:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start0 = time.time()
    deadline = t_start0 + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    h_init = _fix_sum(h_init, n_init / 2.0)
    c5_init = _c5_of(h_init)
    if c5_init < best_c5:
        best_c5 = c5_init
        best_h = h_init.copy()
        best_n = n_init

    n_candidates = []
    base_ns = [140, 150, 160, 170, 180, 200, 220, 240]
    for nn in base_ns:
        if nn % 2 == 0 and nn != n_init:
            n_candidates.append(nn)
    n_candidates.append(n_init)

    seen = set()
    n_candidates_u = []
    for nn in n_candidates:
        if nn not in seen and 80 <= nn <= 260:
            seen.add(nn)
            n_candidates_u.append(nn)
    n_candidates =sorted(n_candidates_u)

    time_per_n = min(200, max(60, int((deadline - time.time() - 200) / len(n_candidates))))

    for n in n_candidates:
        if time.time() > deadline - 150:
            break
        target_sum = n / 2.0
        n_time_budget = min(time_per_n, int(deadline - time.time() - 100))

        if n == n_init:
            h_start = h_init.copy()
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        phase_deadline = time.time() + n_time_budget * 0.3
        h_de, c5_de = _de_optimize(h_start, n, target_sum,
                                    pop_size=12, n_gen=60,
                                    deadline=min(phase_deadline, deadline - 200),
                                    rng=rng)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n

        phase_deadline = time.time() + n_time_budget * 0.3
        h_n1, c5_n1 = _nesterov_adaptive(h_de, n, target_sum, n_iters=400,
                                          deadline=min(phase_deadline, deadline - 150),
                                          m=5, alpha0=0.45, temp=40.0, patience=40)
        if c5_n1 < best_c5:
            best_c5 = c5_n1
            best_h = h_n1.copy()
            best_n = n

        phase_deadline = time.time() + n_time_budget * 0.2
        h_n2, c5_n2 = _nesterov_subgrad(h_n1, n, target_sum, n_iters=300,
                                         deadline=min(phase_deadline, deadline - 120),
                                         m=3, alpha0=0.25, temp=80.0, use_adaptive=True)
        if c5_n2 < best_c5:
            best_c5 = c5_n2
            best_h = h_n2.copy()
            best_n = n

        h_curr = h_n2
        for alpha in [50, 200, 800, 2000]:
            if time.time() > deadline - 80:
                break
            h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=300)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_curr = h_sl

        if time.time() < deadline - 60:
            h_cd, c5_cd = _coord_descent_binary(h_curr, n, max_rounds=30)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = n

        if time.time() < deadline - 40:
            h_cm, c5_cm = _coord_descent_multi(h_curr, n, max_rounds=8)
            if c5_cm < best_c5:
                best_c5 = c5_cm
                best_h = h_cm.copy()
                best_n = n

    if time.time() < deadline - 100:
        n_best = best_n
        target = n_best / 2.0
        h_pr, c5_pr = _perturb_and_refine(best_h, n_best, target,
                                           deadline - 80, rng,
                                           n_perturbs=25, perturb_sigma=0.06)
        if c5_pr < best_c5:
            best_c5 = c5_pr
            best_h = h_pr.copy()
            best_n = n_best

    if time.time() < deadline - 60:
        n_best = best_n
        target = n_best / 2.0
        h_sa, c5_sa = _simulated_annealing(best_h, n_best, target,
                                            t_start=0.05, t_end=0.001,
                                            n_sweeps=20,
                                            deadline=min(time.time() + 120, deadline - 40),
                                            rng=rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            best_n = n_best

    n_best = best_n
    target = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 30:
        for alpha in [1000, 3000, 8000, 15000]:
            if time.time() > deadline - 20:
                break
            h_final = _slsqp_opt(h_final, n_best, target, alpha, maxiter=400)
            c5_f = _c5_of(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    if time.time() < deadline - 15:
        h_cd_f, c5_cd_f = _coord_descent_binary(best_h, n_best, max_rounds=40)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 10:
        h_cm_f, c5_cm_f = _coord_descent_multi(best_h, n_best, max_rounds=15)
        if c5_cm_f < best_c5:
            best_c5 = c5_cm_f
            best_h = h_cm_f.copy()

    if time.time() < deadline - 5:
        h_pol = _slsqp_opt(best_h, n_best, target, 20000, maxiter=500)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() < deadline - 3:
        h_cd_final, c5_cd_final = _coord_descent_binary(best_h, n_best, max_rounds=20)
        if c5_cd_final < best_c5:
            best_c5 = c5_cd_final
            best_h = h_cd_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
