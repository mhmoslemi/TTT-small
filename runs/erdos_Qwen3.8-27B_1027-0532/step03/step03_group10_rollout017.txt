def coordinate_descent(h, max_passes=100):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    # Compute initial correlations
    # corr[s] for s in range(-(n-1), n) represents C(s)
    # Using numpy correlate: np.correlate(h, 1-h, mode='full')
    # Index k in the output corresponds to shift (k - (n-1))
    
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')  # length 2n-1, index 0 = shift -(n-1)
    
    for p in range(max_passes):
        improved = False
        for i in range(n):
            old_hi = h[i]
            
            # For this i, compute the linear coefficients
            # h[i] appears in C(s) for shifts s where:
            # - As first factor: -i <= s <= n-1-i (i+s in [0,n-1])
            # - As second factor (in 1-h[i+s]): i-s in [0,n-1] means i-n+1 <= s <= i
            
            # h1[s_offset] = coefficient of h[i] in C(s)
            # h2[s_offset] = constant term (not involving h[i]) in C(s)
            
            # Current C(s) = A(s) + old_hi * h1(s)
            # So A(s) = C(s) - old_hi * h1(s)
            
            # Compute h1(s) for all s
            # h1(s) = (1-h[i+s]) if -i <= s <= n-1-i, minus h[i-s] if i-n+1 <= s <= i
            # (with the understanding that both can be true simultaneously)
            
            h1 = np.zeros(2*n-1)  # indexed by s + (n-1)
            offset = n - 1
            
            # As first factor: s in [-i, n-1-i]
            s_start1 = -i
            s_end1 = n - 1 - i
            for s in range(max(s_start1, -(n-1)), min(s_end1, n-1) + 1):
                is = i + s
                h1[s + offset] += (1.0 - h[is])
            
            # As second factor: s in [i-n+1, i], meaning i-s is valid
            s_start2 = i - n + 1
            s_end2 = i
            for s in range(max(s_start2, -(n-1)), min(s_end2, n-1) + 1):
                is = i - s
                h1[s + offset] -= h[is]
            
            # A(s) = C(s) - old_hi * h1(s)
            A = corr - old_hi * h1
            
            # Minimize max_s (A[s] + x * h1[s]) over x in [0, 1]
            # This is a 1D convex piecewise linear minimization
            new_hi = find_1d_min(A, h1)
            
            if abs(new_hi - old_hi) > 1e-12:
                # Update h[i] and correlations
                delta = new_hi - old_hi
                corr += delta * h1
                h[i] = new_hi
                improved = True
        
        if not improved:
            break
    
    return h, dx * np.max(corr)

def project_onto_constraint(h, target_sum):

            lo = mid

def compute_c5(h):

def feasible_1d(A, h1, t, lo, hi):

    n = len(h_init)
    dx = 2.0 / n
    h = h_init.astype(np.float64).copy()

    offset = n - 1
    best_c5 = dx * float(corr.max())
    best_h = h.copy()
    
    for p in range(max_passes):
        total_change = 0.0
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        
        for i in order:
            h1 = np.zeros(2 * n - 1)
            # First factor contribution
            s_lo1 = max(-i, -(n-1))
            s_hi1 = min(n-1-i, n-1)
            if s_lo1 <= s_hi1:
                idx = np.arange(s_lo1, s_hi1+1)
                h1[idx + offset] += (1.0 - h[i + idx])
            # Second factor contribution
            s_lo2 = max(i-(n-1), -(n-1))
            s_hi2 = min(i, n-1)
            if s_lo2 <= s_hi2:
                idx = np.arange(s_lo2, s_hi2+1)
                h1[idx + offset] -= h[i - idx]
            
            old_hi = h[i]
            A = corr - old_hi * h1
            new_hi = find_1d_min(A, h1)
            delta = new_hi - old_hi

                corr += delta * h1
                h[i] = new_hi

            best_c5 = c5
            best_h = h.copy()

delta[mask4] += -h[j - s_range[mask4]]

import numpy as np
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def feasible_1d(A, h1, t, lo, hi):
    pos = h1 > 1e-14
    neg = h1 < -1e-14
    zero = ~(pos | neg)
    if np.any(zero):
        if float(A[zero].max()) > t + 1e-13:
            return False
    lower = lo
    if np.any(neg):
        lower = max(lower, float(((t - A[neg]) / h1[neg]).max()))
    upper = hi
    if np.any(pos):
        upper = min(upper, float(((t - A[pos]) / h1[pos]).min()))
    return lower <= upper + 1e-12

def find_1d_min(A, h1):
    lo, hi = 0.0, 1.0
    f_lo = float(A.max())
    f_hi = float((A + h1).max())
    t_high = min(f_lo, f_hi)
    t_low = float(A.min()) - float(np.abs(h1).max())
    if feasible_1d(A, h1, t_low, lo, hi):
        t_low = t_high - 1e-6
    for _ in range(50):
        t_mid = (t_low + t_high) * 0.5
        if feasible_1d(A, h1, t_mid, lo, hi):
            t_high = t_mid
        else:
            t_low = t_mid
    t_star = t_high
    neg = h1 < -1e-14
    pos = h1 > 1e-14
    lower = 0.0
    if np.any(neg):
        lower = max(lower, float(((t_star - A[neg]) / h1[neg]).max()))
    upper = 1.0
    if np.any(pos):
        upper = min(upper, float(((t_star - A[pos]) / h1[pos]).min()))
    if lower > upper:
        x_star = 0.0 if f_lo <= f_hi else 1.0
    else:
        x_star = (lower + upper) * 0.5
    return x_star

def coordinate_descent(h_init, max_passes=300, recompute_every=5, rng=None):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    offset = n - 1
    best_c5 = dx * float(corr.max())
    best_h = h.copy()
    time_limit = max_passes

    for p in range(time_limit):
        total_change = 0.0
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for i in order:
            h1 = np.zeros(2 * n - 1)
            s_lo1 = max(-i, -(n - 1))
            s_hi1 = min(n - 1 - i, n - 1)
            if s_lo1 <= s_hi1:
                idx = np.arange(s_lo1, s_hi1 + 1)
                h1[idx + offset] += (1.0 - h[i + idx])
            s_lo2 = max(i - (n - 1), -(n - 1))
            s_hi2 = min(i, n - 1)
            if s_lo2 <= s_hi2:
                idx = np.arange(s_lo2, s_hi2 + 1)
                h1[idx + offset] -= h[i - idx]
            old_hi = h[i]
            A = corr - old_hi * h1
            new_hi = find_1d_min(A, h1)
            delta = new_hi - old_hi
            if abs(delta) > 1e-15:
                corr += delta * h1
                h[i] = new_hi
                total_change += abs(delta)
        if (p + 1) % recompute_every == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        c5 = dx * float(corr.max())
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if total_change < 1e-13 * n:
            break
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(corr.max())
    best_c5_val = current_max
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].astype(np.int64).copy()
    zeros = np.where(h == 0.0)[0].astype(np.int64).copy()
    n_one = len(ones)
    half = n - 1
    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        ii = int(rng.integers(n_one))
        jj = int(rng.integers(n_one))
        i = int(ones[ii])
        j = int(zeros[jj])
        if i == j:
            step += 1
            continue
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            ones[ii], zeros[jj] = zeros[jj], ones[ii]
            step += 1
            continue
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = np.zeros(len(s_range))
        mask1 = (i + s_range >= 0) & (i + s_range < n)

        mask4 = (j - s_range >= 0) & (j - s_range < n)
        delta[mask4] += -h[j - s_range[mask4]]
        corr[k_range] += delta
        new_max = float(corr.max())
