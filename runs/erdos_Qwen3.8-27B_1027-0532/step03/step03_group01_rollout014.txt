    corr = np.correlate(h, 1.0 - h, mode='full')

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

    if diff > 0:

    else:

    h = h.copy()

    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

                                          time_limit=min(15.0, tleft() * 0.4), start_time=t0)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls.copy()

    if tleft() > 40:
        n_best = len(best_h)
        target_best = n_best / 2.0
        alphas = [20, 60, 150, 400, 800, 1500]
        h_ref, sc_ref = slsqp_refine(best_h, n_best, target_best, alphas, 500, 1e-13,
                                     time_limit=min(80.0, tleft() * 0.5), start_time=t0)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    if tleft() > 30:
        n_best = len(best_h)
        target_best = n_best / 2.0
        for pert_scale in [0.02, 0.05, 0.1]:
            if tleft() < 15:
                break
            h_pert = project_feasible(best_h + rng.randn(n_best) * pert_scale, target_best)
            h_ref, sc_ref = slsqp_refine(h_pert, n_best, target_best, [200, 600, 1500],
                                         300, 1e-14,
                                         time_limit=min(20.0, tleft() * 0.4), start_time=t0)
            if sc_ref < best_score:
                best_score = sc_ref
                best_h = h_ref

    if tleft() > 20:
        n_best = len(best_h)
        target_best = n_best / 2.0
        h_bin_final = project_binary(best_h, target_best)
        h_ls, sc_ls = greedy_local_search(h_bin_final, target_best, max_passes=30,
                                          time_limit=min(30.0, tleft() * 0.5), start_time=t0)
        if sc_ls < best_score:
            best_score = sc_ls
            best_h = h_ls

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h.copy()
        final_score = best_score

    return h_final, final_score, n_final
