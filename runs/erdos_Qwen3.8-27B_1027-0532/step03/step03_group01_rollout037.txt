- A point where two constraints are tight: C[k1] + δ*(s_i[k1] - s_j[k1]) = C[k2] + δ*(s_i[k2] - s_j[k2])

1. Computing d[k] = s_i[k] - s_j[k] for all k
2. The function f(δ) = max_k (C[k] + δ * d[k]) is convex piecewise-linear
3. Its minimum is where the subgradient contains 0, i.e., where the "upper envelope" has a minimum

For a convex piecewise-linear function, the minimum is at a breakpoint where the slope changes from negative to positive.

The slope of f at a point δ is: sum of d[k] for k in the active set (where C[k] + δ*d[k] = f(δ)).

We can find the minimum by:
- Computing all breakpoints
- Evaluating f at each breakpoint
- Taking the minimum

            h_new, _ = greedy_step(h_new, n)
            moved = True
        elif move_type == 1 and len(ones) > 0 and len(zeros) > 0:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 2 and len(ones) >= 2 and len(zeros) >= 2:
            idx1 = rng.choice(len(ones), 2, replace=False)
            idx0 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[idx1[0]]] = 0
            h_new[ones[idx1[1]]] = 0
            h_new[zeros[idx0[0]]] = 1
            h_new[zeros[idx0[1]]] = 1
            moved = True
        elif move_type == 3 and len(ones) > 0 and len(zeros) > 0:
            h_new, _ = greedy_step(h_new, n, use_multiple=False)
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            if h_new[i1] != h[i1] or h_new[i0] != h[i0]:
                h_new[i1] = 1.0 - h_new[i1]
                h_new[i0] = 1.0 - h_new[i0]
            else:
                h_new[i1] = 1 - h_new[i1]
                h_new[i0] = 1 - h_new[i0]
            moved = True
        elif move_type == 4 and len(ones) >= 3 and len(zeros) >= 3:
            cluster_size = 3
            center = rng.randint(n)
            dist_o = np.abs(ones - center).astype(float)
            dist_z = np.abs(zeros - center).astype(float)
            top_o = ones[np.argsort(dist_o)[:cluster_size]]
            top_z = zeros[np.argsort(dist_z)[:cluster_size]]
            for pos in top_o:
                h_new[pos] = 0
            for pos in top_z:
                h_new[pos] = 1
            moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5


def smooth_logsumexp_value(h, alpha):
    C = compute_corr(h)
    mC = np.max(C)
    return mC + alpha * np.log(np.sum(np.exp((C - mC) / alpha)))


def smooth_logsumexp_grad(h, alpha):
    n = len(h)
    C = compute_corr(h)
    mC = np.max(C)
    exp_C = np.exp((C - mC) / alpha)
    p = exp_C / exp_C.sum()
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    p_idx_A = i_idx - j_idx + (n - 1)
    valid_A = (p_idx_A >= 0) & (p_idx_A < 2 * n - 1)
    A = np.where(valid_A, p[np.clip(p_idx_A, 0, 2 * n - 2)], 0.0) @ h
    p_idx_B = j_idx - i_idx + (n - 1)
    valid_B = (p_idx_B >= 0) & (p_idx_B < 2 * n - 1)
    B = np.where(valid_B, p[np.clip(p_idx_B, 0, 2 * n - 2)], 0.0) @ h
    return 1.0 - A - B


def lbfgs_continuous(h_start, n, target, alphas, time_limit=60, start_time=0):
    h = h_start.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    penalty_weight = 5.0

    def obj_pen(x, alpha):
        x = np.clip(x, 0.0, 1.0)
        base = smooth_logsumexp_value(x, alpha)
        sd = np.sum(x) - target
        return base + penalty_weight * sd * sd

    def grad_pen(x, alpha):
        x = np.clip(x, 0.0, 1.0)
        bg = smooth_logsumexp_grad(x, alpha)
        sd = np.sum(x) - target
        return bg + 2.0 * penalty_weight * sd

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                lambda x: obj_pen(x, alpha),
                h,
                jac=lambda x: grad_pen(x, alpha),
                method='L-BFGS-B',
                bounds=[(0.0, 1.0)] * n,
                options={'maxiter': 1500, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 80}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
    return best_h, best_score


def projected_subgrad(h_start, n, target, alpha, n_iters=5000,
                      lr_init=0.5, lr_decay=0.9995, time_limit=20, start_time=0):
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    for i in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        grad = smooth_logsumexp_grad(h, alpha)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h = h_new
        lr *= lr_decay
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    h_bin, c5_bin = greedy_local_search(h_bin, target_init, max_sweeps=200,
                                         time_limit=min(30, tleft() * 0.1),
                                         start_time=start_time)
    if c5_bin < best_score:
        best_score = c5_bin
        best_h = h_bin.copy()

    sa_configs = [
        (50000, 0.012, 0.00005),
        (70000, 0.008, 0.00003),
        (40000, 0.020, 0.0008),
        (100000, 0.006, 0.00002),
    ]

    h_sa = h_bin.copy()
    for ci, (max_it, t_i, t_f) in enumerate(sa_configs):
        if tleft() < 100:
            break
        allowed = min(180.0, tleft() * 0.3)
        h_sa, c5_sa = sa_optimize_advanced(h_sa, target_init, max_it, t_i, t_f,
                                           rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        if tleft() > 40:
            h_g, c5_g = greedy_local_search(h_sa, target_init, max_sweeps=200,
                                             time_limit=min(15, tleft() * 0.1),
                                             start_time=start_time)
            if c5_g < best_score:
                best_score = c5_g
                best_h = h_g.copy()
                h_sa = h_g

    n_random = 3
    for rs in range(n_random):
        if tleft() < 80:
            break
        h_rand = np.zeros(n_init)
        perm = rng.permutation(n_init)
        h_rand[perm[:n_init // 2]] = 1.0
        h_rand, c5_r = greedy_local_search(h_rand, target_init, max_sweeps=100,
                                            time_limit=min(10, tleft() * 0.05),
                                            start_time=start_time)
        allowed = min(120.0, tleft() * 0.25)
        max_it = int(min(60000, allowed * 500))
        h_sa_r, c5_sa_r = sa_optimize_advanced(h_rand, target_init, max_it,
                                                0.010, 0.00005, rng,
                                                time_limit=allowed, start_time=start_time)
        if c5_sa_r < c5_r:
            h_rand = h_sa_r
            c5_r = c5_sa_r
        h_g, c5_g = greedy_local_search(h_rand, target_init, max_sweeps=150,
                                         time_limit=min(10, tleft() * 0.05),
                                         start_time=start_time)
        if c5_g < c5_r:
            c5_r = c5_g
            h_rand = h_g
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_rand.copy()

    n_multi = [300, 400, 500, 700, 800, 1000]
    n_multi = [n for n in n_multi if n != n_init and n >= 200]
    for n_try in n_multi:
        if tleft() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        h_bin_try, c5_bt = greedy_local_search(h_bin_try, target_try,
                                                max_sweeps=100, time_limit=min(10, tleft()*0.1),
                                                start_time=start_time)
        allowed = min(80.0, tleft() * 0.2)
        max_it = int(min(40000, allowed * 400))
        h_sa_m, c5_sa_m = sa_optimize_advanced(h_bin_try, target_try, max_it,
                                                0.010, 0.00008, rng,
                                                time_limit=allowed, start_time=start_time)
        if c5_sa_m < c5_bt:
            h_bin_try = h_sa_m
            c5_bt = c5_sa_m
        h_g, c5_g = greedy_local_search(h_bin_try, target_try, max_sweeps=100,
                                         time_limit=min(8, tleft()*0.05),
                                         start_time=start_time)
        if c5_g < c5_bt:
            c5_bt = c5_g
            h_bin_try = h_g
        if c5_bt < best_score:
            best_score = c5_bt
            best_h = h_bin_try.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 50:
        h_cont = best_h.copy()
        alphas = [5, 10, 20, 40, 80, 150, 300]
        h_ref, sc = lbfgs_continuous(h_cont, n_use, target_use, alphas,
                                      time_limit=min(80, tleft() * 0.4), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 40:
        alphas_sg = [20, 50, 100, 200]
        h_sg = best_h.copy()
        for a in alphas_sg:
            if tleft() < 20:
                break
            h_sg, sc_sg = projected_subgrad(h_sg, n_use, target_use, a,
                                             n_iters=3000, lr_init=0.3,
                                             lr_decay=0.999,
                                             time_limit=min(15, tleft()*0.2),
                                             start_time=start_time)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg.copy()

    for n_try in [int(n_use * 1.3), int(n_use * 0.7)]:
        if tleft() < 40:
            break
        if n_try < 200 or n_try == n_use:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [10, 30, 80, 200]
        h_ref, sc = lbfgs_continuous(h_interp, n_try, target_try, alphas_try,
                                      time_limit=min(40, tleft() * 0.3), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        h_bin_final, c5_f = greedy_local_search(h_bin_final, target_use,
                                                  max_sweeps=200, time_limit=min(10, tleft()*0.3),
                                                  start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_bin_final.copy()
        if tleft() > 15:
            h_ref, sc = lbfgs_continuous(best_h, len(best_h), len(best_h) / 2.0,
                                          [50, 100, 200], time_limit=max(5, tleft()-8),
                                          start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref.copy()

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
