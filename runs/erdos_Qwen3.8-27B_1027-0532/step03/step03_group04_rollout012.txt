    ]
    for k in range(-(n-1), n):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            coeff = np.zeros(n)
            coeff[lo:hi] = g[lo-k:hi-k] * dx
            constraints.append(coeff @ h_var <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver, max_iters=100000, verbose=False)
            if h_var.value is not None:
                return np.clip(np.asarray(h_var.value).ravel(), 0, 1), t_var.value
        except:
            continue
    return None, None

def alternating_lp(h_init, n, target, time_limit, damping_starts, solvers):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for damping_init in damping_starts:
        if time.time() - start > time_limit:
            break
        damping = damping_init
        stuck = 0
        while time.time() - start < time_limit:
            g = 1.0 - h
            h_new, lp_t = solve_lp_for_h(g, n, target, dx, solvers)
            if h_new is None:
                break
            h_try = (1 - damping) * h + damping * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                stuck = 0
            else:
                h = h_try
                stuck += 1
                if stuck > 5:
                    damping = max(0.1, damping * 0.7)
                if stuck > 30:
                    break
        # After each damping phase, also try full step
        g = 1.0 - best_h
        h_new, _ = solve_lp_for_h(g, n, target, dx, solvers)
        if h_new is not None:
            h_full = project_feasible(h_new, target)
            c5_full = compute_c5(h_full)
            if c5_full < best_c5:
                best_c5 = c5_full
                best_h = h_full
    
    return best_h, best_c5

def projected_gradient_softmax(h, n, target, alphas, time_limit, lr_init=0.01):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    v = np.zeros(n)  # momentum
    
    for alpha in alphas:
        lr = lr_init / alpha * 100  # scale learning rate with alpha
        for it in range(500):
            if time.time() - start > time_limit:
                break
            # Compute gradient
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            scaled = corr * (2.0 / n)
            max_s = np.max(scaled)
            exp_s = np.exp(np.clip(alpha * (scaled - max_s), -500, 0))
            p = exp_s / np.sum(exp_s)
            
            grad = np.zeros(n)
            for k_idx in range(len(p)):
                pk = p[k_idx]
                if pk < 1e-10:
                    continue
                k = k_idx - (n - 1)
                # First term: (1-h[j+k]) for valid j
                j_lo = max(0, -k)
                j_hi = min(n, n - k)
                if j_hi > j_lo:
                    grad[j_lo:j_hi] += pk * (2.0/n) * (1.0 - h_cur[j_lo+k:j_hi+k])
                # Second term: -h[j-k] for valid j
                j2_lo = max(0, k)
                j2_hi = min(n, n + k)
                if j2_hi > j2_lo:
                    grad[j2_lo:j2_hi] -= pk * (2.0/n) * h_cur[j2_lo-k:j2_hi-k]
            
            # Nesterov momentum
            v = 0.9 * v - lr * grad
            h_try = h_cur + v
            h_try = project_feasible(h_try, target)
            h_cur = h_try
            
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5 - 1e-13:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    
    return best_h, best_c5

import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(5):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def precompute_lp_structure(n):
    indices = []
    for k in range(-(n - 1), n):
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            indices.append((lo, hi, lo - k))
    return indices


def solve_lp_step(g, n, target, dx, indices, solver_list):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0.0,
        h_var <= 1.0
    ]
    for lo, hi, gs in indices:
        coeff = np.zeros(n)
        coeff[lo:hi] = g[gs:gs + (hi - lo)] * dx
        constraints.append(coeff @ h_var <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in solver_list:
        try:
            prob.solve(solver=solver, max_iters=100000, verbose=False)
            if h_var.value is not None:
                return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        except Exception:
            continue
    return None

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * (2.0 / len(h))
    c_max = np.max(scaled)
    shifted = np.clip(alpha * (scaled - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    best_c5 = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})

    h_p[flip_idx] = 1.0 - h_p[flip_idx]

    return project_feasible(h_out, target_dst)
