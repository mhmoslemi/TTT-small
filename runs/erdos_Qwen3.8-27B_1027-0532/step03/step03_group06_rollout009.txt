def fft_flatten_step(h, target_sum, strength=0.5):
    """FFT-based spectral flattening step"""
    n = len(h)
    size = 2 * n
    
    H = np.fft.rfft(h, n=size)
    r = np.fft.irfft(H * np.conj(H), n=size)
    
    # r[0] = sum(h^2), r[1..n-1] = positive lag autocorrelation, etc.
    # We want to flatten r[1..n-1] and r[n+1..2n-1]
    
    r_mod = r.copy()
    # Target: make all off-diagonal elements equal to their mean
    off_diag = np.concatenate([r[1:n], r[n+1:2*n]])
    target_val = np.mean(off_diag)
    
    # Interpolate between current and target
    r_mod[1:n] = (1-strength) * r[1:n] + strength * target_val
    r_mod[n+1:2*n] = (1-strength) * r[n+1:2*n] + strength * target_val
    r_mod[n] = 0  # should be ~0 anyway
    
    # Ensure non-negative (power spectrum must be non-negative)
    r_mod = np.maximum(r_mod, 0)
    
    # Compute new power spectrum
    P_new = np.fft.rfft(r_mod, n=size)
    
    # Ensure P_new is non-negative (it should be if r_mod is a valid autocorrelation)
    # Actually, P_new should be real and non-negative if r_mod is a valid autocorrelation
    # But numerical errors might make it slightly negative
    P_new = np.maximum(P_new.real, 0)
    
    # Reconstruct with original phase
    H_orig = np.fft.rfft(h, n=size)
    phase = np.angle(H_orig)
    
    H_new = np.sqrt(P_new) * np.exp(1j * phase)
    h_new = np.fft.irfft(H_new, n=size)[:n]
    
    # Project to feasible set
    h_new = project_feasible(h_new, target_sum)
    return h_new

def _refine(h_start, n, target, tleft, max_iter=200, max_alpha=500):
    """Refine using L-BFGS-B with smooth max objective"""
    h_cur = project_feasible(h_start.copy(), target)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        F, _ = smooth_max_grad(x, max_alpha)
        return F
    
    def jac(x):
        _, g = smooth_max_grad(x, max_alpha)
        return g
    
    try:
        res = minimize(obj, h_cur, jac=jac, method='L-BFGS-B',
                      bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7})
        h_new = project_feasible(res.x, target)
        # Verify improvement
        if compute_c5(h_new) <= compute_c5(h_cur) + 1e-10:
            return h_new
    except Exception:
        pass
    return h_cur

Let me try a different approach: **proximal gradient / forward-backward splitting**

The objective is f(h) = max_k c_k(h) where c_k is the correlation at shift k.

The subgradient of f at h is: (1/|A|) * sum_{k ∈ A} ∇c_k(h) where A = argmax_k c_k(h).

The proximal operator of the indicator of the constraint set {h: 0 ≤ h_i ≤ 1, sum h = n/2} is just the projection (clip + normalize).

So the forward-backward step is:
h_{new} = P(h_old - η * g)

where g is the subgradient and P is the projection.

This is exactly what projected subgradient does. The key is choosing η appropriately.

**Adaptive step size:** Use the "greedy step size" η = (f(h_old) - f(v)) / ||h_old - v||² where v is the minimizer of <g, h> over the constraint set. For our problem, v is the 0/1 vector with n/2 ones placed at the positions where g is most negative.

```python
def frank_wolfe_step(h, target, alpha):
    n = len(h)
    c_vals = compute_corr_values(h)
    m = np.max(c_vals)
    shifted = np.clip(alpha * (c_vals - m), -700, 700)
    exp_vals = np.exp(shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            t1 = np.zeros(n); t1[:n-s] = 1.0 - h[s:]
            t2 = np.zeros(n); t2[s:] = h[:n-s]
        else:
            t1 = np.zeros(n); t1[-s:] = 1.0 - h[:n+s]
            t2 = np.zeros(n); t2[:n+s] = h[-s:]
        grad += wk * (t1 - t2)
    
    # Frank-Wolfe: find v = argmin <grad, v> over {v: 0≤v≤1, sum v = n/2}
    # This means: set v[i] = 1 for the n/2 indices with smallest grad[i], v[i] = 0 otherwise
    idx = np.argsort(grad)
    v = np.zeros(n)
    v[idx[:int(target)]] = 1.0
    
    # Step size
    if np.dot(grad, h - v) > 1e-15:
        gamma = 2.0 / (alpha + 2.0)  # standard FW step size (or use line search)
    else:
        gamma = 0.0
    
    h_new = (1 - gamma) * h + gamma * v
    return project_feasible(h_new, target)
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        adj = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def compute_corr_values(h):

    r_mod[n+1:2*n] = (1-strength)*r[n+1:2*n] + strength*tgt
    r_mod = np.maximum(r_mod, 0)

    H_new = np.sqrt(P) * np.exp(1j * phase)
    h_new = np.fft.irfft(H_new, n=size)[:n]
    return project_feasible(h_new, target)
