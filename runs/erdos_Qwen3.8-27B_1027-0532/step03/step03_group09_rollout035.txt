import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def subgrad_topk(h, k_indices):
    """Compute weighted subgradient for given lag indices."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for kidx in k_indices:
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += dx * omh[:m]
            grad[:m] -= dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += dx * omh[r:]
            grad[r:] -= dx * h[:m]
    return grad

def subgrad_weighted_topk(h, k_indices, weights):
    """Compute weighted subgradient for given lag indices."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for idx, (kidx, w) in enumerate(zip(k_indices, weights)):
        if w < 1e-14:
            continue
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += w * dx * omh[:m]
            grad[:m] -= w * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += w * dx * omh[r:]
            grad[r:] -= w * dx * h[:m]
    return grad

def smooth_max_obj(h, eps):
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]

def sla_slsqp(h_init, eps, maxiter, ftol=1e-14):
    """SLSQP optimization with smooth max."""
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:

def simulated_annealing_topk(h_init, t_start, t_end, n_steps, k_top, time_limit, start_time):

    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

            break
        
        corr = compute_corr_scaled(h_cur)
        # Get top-k worst lags
        topk_idx = np.argsort(corr)[-k_top:]
        vals = corr[topk_idx]
        # Softmax weights
        xc = vals - np.max(vals)
        w = np.exp(xc / (0.01 + t_start / max(1, n_steps) * (1 - step/n_steps)))
        w = w / np.sum(w)

import numpy as np
from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

        return np.ones(n)

            best_h = h_pg.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
