    h = np.clip(h.copy(), 0.0, 1.0)

    return grad * dx

def _nesterov_smooth(h0, n, target_sum, max_iters, alpha0, deadline, use_multi=True, m_top=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        cmax = np.max(corr)
        if use_multi:
            shifted = corr - cmax
            weights = np.exp(80.0 * shifted)
            tw = weights.sum()
            if tw < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / tw
            top_m = np.argsort(weights)[::-1][:m_top]
            w_m = weights[top_m].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m)):
                g = g + w_m[idx] * _grad_corr_at(y, top_m[idx], n)
        else:
            k_star = int(np.argmax(corr))
            g = _grad_corr_at(y, k_star, n)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_run(h0, n, target_sum, alpha, maxiter, deadline):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * (2.0 / n)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    def cons_eq(x):
        return x.sum() - target_sum
    
    try:
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_eq}]
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_out = _fix_sum(res.x, target_sum)
        return h_out, _c5_of(h_out)
    except Exception:
        return h0.copy(), _c5_of(h0)

def _coord_binary(h0, n, target_sum, max_rounds, deadline):
    h = (h0 > 0.5).astype(float).copy()
    h = _fix_sum(h, target_sum)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c_new = _c5_of(h)
                h[i] = 1.0
                h[j] = 0.0
                if c_new < _c5_of(h) - 1e-14:
                    h[i] = 0.0
                    h[j] = 1.0
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
        if not improved:
            break
    return h, _c5_of(h)

def _coord_frac(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h, _c5_of(h)

def _interpolate(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return _fix_sum(np.clip(h_new, 0.0, 1.0), n_new / 2.0)

def _sa_binary(h0, n, target_sum, max_iters, T_start, T_end, deadline, rng):
    h = (h0 > 0.5).astype(float).copy()
    h = _fix_sum(h, target_sum)
    h_val = _c5_of(h)
    best_h = h.copy()
    best_val = h_val
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))
        h_new = h.copy()
        ones = np.where(h_new == 1)[0]
        zeros = np.where(h_new == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_val = _c5_of(h_new)
        acceptance = np.exp(-(new_val - h_val) / max(T, 1e-10))
        if new_val < h_val or rng.random() < acceptance:
            h = h_new
            h_val = new_val
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    best_h = h_init.copy()
    best_c5 = _c5_of(h_init)
    best_n = n_init

    res_list = [n_init]
    for nn in [200, 250, 300, 350]:
        if nn not in res_list and nn != n_init:
            res_list.append(nn)
    if n_init != 150:
        res_list.insert(0, 150)

    for n in res_list:
        if time.time() > deadline - 80:
            break
        target_sum = n / 2.0

        if n == best_n:
            h0 = best_h.copy()
        else:
            h0 = _interpolate(best_h, best_n, n)

        c5_0 = _c5_of(h0)
        if c5_0 < best_c5:
            best_c5 = c5_0
            best_h = h0.copy()
            best_n = n

        # Phase 1: Nesterov with smoothed subgradient
        if time.time() < deadline - 60:
            h_n1, c5_n1 = _nesterov_smooth(h0, n, target_sum, 200, 0.4, deadline - 60, use_multi=True, m_top=5)
            if c5_n1 < best_c5:
                best_c5 = c5_n1
                best_h = h_n1.copy()
                best_n = n
                h0 = h_n1

        # Phase 2: SLSQP with moderate alpha
        if time.time() < deadline - 50:
            h_s1, c5_s1 = _slsqp_run(h0, n, target_sum, 100, 300, deadline - 50)
            if c5_s1 < best_c5:
                best_c5 = c5_s1
                best_h = h_s1.copy()
                best_n = n
                h0 = h_s1

        # Phase 3: Binary coordinate descent
        if time.time() < deadline - 40:
            h_b, c5_b = _coord_binary(h0, n, target_sum, 15, deadline - 40)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
                best_n = n
                h0 = h_b

        # Phase 4: SA on binary
        if time.time() < deadline - 35:
            h_sa, c5_sa = _sa_binary(h0, n, target_sum, 5000, 0.03, 0.001, deadline - 35, rng)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
                best_n = n
                h0 = h_sa

        # Phase 5: Fractional coordinate descent
        if time.time() < deadline - 25:
            h_f, c5_f = _coord_frac(h0, n, target_sum, 6, deadline - 25)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                best_n = n
                h0 = h_f

        # Phase 6: SLSQP with high alpha (polish)
        if time.time() < deadline - 20:
            h_s2, c5_s2 = _slsqp_run(h0, n, target_sum, 300, 500, deadline - 20)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                best_h = h_s2.copy()
                best_n = n
                h0 = h_s2

        # Phase 7: Nesterov with multi-gradient (fine tuning)
        if time.time() < deadline - 15:
            h_n2, c5_n2 = _nesterov_smooth(h0, n, target_sum, 150, 0.2, deadline - 15, use_multi=True, m_top=8)
            if c5_n2 < best_c5:
                best_c5 = c5_n2
                best_h = h_n2.copy()
                best_n = n
                h0 = h_n2

    # Final polishing on the best solution found
    n_final = best_n
    target_final = n_final / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 15:
        h_f1, c5_f1 = _slsqp_run(h_final, n_final, target_final, 500, 2000, deadline - 15)
        if c5_f1 < best_c5:
            best_c5 = c5_f1
            h_final = h_f1

    if time.time() < deadline - 10:
        h_f2, c5_f2 = _coord_binary(h_final, n_final, target_final, 20, deadline - 10)
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            h_final = h_f2

    if time.time() < deadline - 8:
        h_f3, c5_f3 = _coord_frac(h_final, n_final, target_final, 8, deadline - 8)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            h_final = h_f3

    if time.time() < deadline - 5:
        h_f4, c5_f4 = _slsqp_run(h_final, n_final, target_final, 600, 3000, deadline - 5)
        if c5_f4 < best_c5:
            best_c5 = c5_f4
            h_final = h_f4

    if time.time() < deadline - 3:
        h_f5, c5_f5 = _nesterov_smooth(h_final, n_final, target_final, 100, 0.1, deadline - 3, use_multi=True, m_top=10)
        if c5_f5 < best_c5:
            best_c5 = c5_f5
            h_final = h_f5

    if time.time() < deadline - 2:
        h_f6, c5_f6 = _coord_frac(h_final, n_final, target_final, 5, deadline - 2)
        if c5_f6 < best_c5:
            best_c5 = c5_f6
            h_final = h_f6

    h_final = _fix_sum(h_final, n_final / 2.0)
    best_c5 = _c5_of(h_final)

    return (h_final, best_c5, n_final)
