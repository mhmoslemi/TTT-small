from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_dx(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr_dx(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_dx(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

        return float(np.sum(x) - target_sum)

        return np.ones(n)

    h_tm = 0.5 + 0.3 * (2.0 * tm - 1.0)
    pop.append(project_onto_constraint(h_tm, target_sum))
    h_tm2 = 0.5 + 0.2 * (2.0 * tm - 1.0) + 0.1 * np.sin(np.pi * t)
    pop.append(project_onto_constraint(h_tm2, target_sum))

    h_block = np.zeros(n)
    h_block[:n // 4] = 1.0
    h_block[n // 2:3 * n // 4] = 0.8
    h_block[3 * n // 4:] = 1.0
    pop.append(project_onto_constraint(h_block, target_sum))

    period = max(4, n // 12)
    h_comb = np.zeros(n)
    for i in range(n):
        pos_in_period = i % period
        if pos_in_period < period // 3:
            h_comb[i] = 1.0
        elif pos_in_period < 2 * period // 3:
            h_comb[i] = 0.3
    pop.append(project_onto_constraint(h_comb, target_sum))

    for _ in range(6):
        a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.0, 1.0)
        b = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.0, 1.0)
        alpha = rng.uniform(0.3, 0.7)
        h = alpha * a + (1.0 - alpha) * b
        pop.append(project_onto_constraint(h, target_sum))

    for _ in range(8):
        h = h_parent + 0.08 * rng.standard_normal(n)
        pop.append(project_onto_constraint(h, target_sum))

    return pop[:50]

def ga_mutate(h, n, target_sum, rng, strategy):
    if strategy == 0:
        return project_onto_constraint(h + 0.04 * rng.standard_normal(n), target_sum)
    elif strategy == 1:
        blen = int(rng.integers(3, max(4, n // 6) + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            h_new = h.copy()
            blk = h_new[a:a + blen].copy()
            h_new[a:a + blen] = h_new[b:b + blen]
            h_new[b:b + blen] = blk
            return h_new
        return project_onto_constraint(h + 0.03 * rng.standard_normal(n), target_sum)
    elif strategy == 2:
        corr = compute_corr_dx(h)
        worst_k = int(np.argmax(corr))
        grad = lag_grad(h, worst_k)
        h_new = project_onto_constraint(h - 0.08 * grad, target_sum)
        h_new = project_onto_constraint(h_new + 0.02 * rng.standard_normal(n), target_sum)
        return h_new
    elif strategy == 3:
        h_new = h.copy()
        n_flip = int(rng.integers(4, max(8, n // 6) + 1))
        idx = rng.choice(n, size=n_flip, replace=False)
        h_new[idx] = (rng.random(n_flip) < 0.5).astype(float)
        return project_onto_constraint(h_new, target_sum)
    elif strategy == 4:
        return project_onto_constraint(h + 0.15 * rng.standard_normal(n), target_sum)
    elif strategy == 5:
        center = int(rng.integers(0, n))
        width = int(rng.integers(5, max(10, n // 6) + 1))
        h_new = h.copy()
        a = max(0, center - width)
        b = min(n, center + width)
        h_new[a:b] += 0.08 * rng.standard_normal(b - a)
        return project_onto_constraint(h_new, target_sum)
    else:
        h_new = h.copy()
        n_mod = int(rng.integers(10, max(15, n // 4) + 1))
        idx = rng.choice(n, size=n_mod, replace=False)
        h_new[idx] += 0.1 * rng.standard_normal(n_mod)
        return project_onto_constraint(h_new, target_sum)

def ga_crossover(p1, p2, n, target_sum, rng):
    alpha = rng.uniform(0.2, 0.8)
    child = alpha * p1 + (1.0 - alpha) * p2
    child += 0.01 * rng.standard_normal(n)
    return project_onto_constraint(child, target_sum)

def local_polish(h, n, target_sum, time_limit, start_time):
    corr = compute_corr_dx(h)
    worst_k = int(np.argmax(corr))
    for _ in range(5):
        if time.time() - start_time > time_limit:
            break
        grad = lag_grad(h, worst_k)
        h = project_onto_constraint(h - 0.05 * grad, target_sum)
        corr = compute_corr_dx(h)
        worst_k = int(np.argmax(corr))
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.005, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 150:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    ga_time_limit = max(100, int(rem() * 0.45))
    if ga_time_limit > 100:
        pop = init_population(best_h, n, rng, target_sum)
        pop_c5 = [compute_c5(p) for p in pop]
        pop_best = best_h.copy()
        pop_best_c5 = best_c5

        pop_size = len(pop)
        stall_gen = 0
        stall_threshold = 50
        gen = 0

        while rem() > 150 and gen < 20000:
            gen += 1

            sorted_idx = np.argsort(pop_c5)
            new_pop = [pop_best.copy()]
            new_c5 = [pop_best_c5]

            elite_count = 5
            for i in range(min(elite_count, len(pop))):
                idx = sorted_idx[i]
                new_pop.append(pop[idx].copy())
                new_c5.append(pop_c5[idx])

            target_size = min(pop_size, 45)
            attempts = 0
            max_attempts = target_size * 3

            while len(new_pop) < target_size and rem() > 160 and attempts < max_attempts:
                attempts += 1
                i1 = int(rng.integers(0, len(pop)))
                i2 = int(rng.integers(0, len(pop)))
                if pop_c5[i1] <= pop_c5[i2]:
                    p1 = pop[i1]
                else:
                    p1 = pop[i2]

                i3 = int(rng.integers(0, len(pop)))
                i4 = int(rng.integers(0, len(pop)))
                if pop_c5[i3] <= pop_c5[i4]:
                    p2 = pop[i3]
                else:
                    p2 = pop[i4]

                if rng.random() < 0.55:
                    child = ga_crossover(p1, p2, n, target_sum, rng)
                else:
                    strategy = int(rng.integers(0, 7))
                    child = ga_mutate(p1, n, target_sum, rng, strategy)

                if rng.random() < 0.25:
                    child = local_polish(child, n, target_sum, start_time + 1.0, start_time)

                c5_child = compute_c5(child)
                new_pop.append(child)
                new_c5.append(c5_child)

            pop = new_pop
            pop_c5 = new_c5

            min_c5 = min(pop_c5)
            if min_c5 < pop_best_c5 - 1e-9:
                best_idx = int(np.argmin(pop_c5))
                pop_best = pop[best_idx].copy()
                pop_best_c5 = pop_c5[best_idx]
                stall_gen = 0
            else:
                stall_gen += 1

            if stall_gen >= stall_threshold:
                stall_gen = 0
                stall_threshold = max(25, stall_threshold // 2)
                n_replace = max(5, len(pop) // 3)
                for _ in range(n_replace):
                    worst_idx = int(np.argmax(pop_c5))
                    h_rand = np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.0, 1.0)
                    h_mix = 0.4 * pop_best + 0.6 * h_rand
                    pop[worst_idx] = project_onto_constraint(h_mix, target_sum)
                    pop_c5[worst_idx] = compute_c5(pop[worst_idx])

        if pop_best_c5 < best_c5:
            best_c5 = pop_best_c5
            best_h = pop_best.copy()

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 60:
        tl = min(rem() - 30, 120)
        h_pg, c5_pg = proj_grad_descent(best_h, 3, 0.005, 0.06, 300000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 50:
        tl = min(rem() - 25, 80)
        h_pg, c5_pg = proj_grad_descent(best_h, 1, 0.002, 0.03, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 60:
        for trial in range(4):
            if rem() < 40:
                break
            for scale in [0.008, 0.015, 0.03]:
                if rem() < 35:
                    break
                h_pert = project_onto_constraint(best_h + scale * rng.standard_normal(n), target_sum)
                h_opt, c5_opt = slsqp_optimize(h_pert, 0.001, 200)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                if rem() > 30:
                    tl = min(rem() - 20, 25)
                    h_pg, c5_pg = proj_grad_descent(best_h, 2, 0.003, 0.04, 100000, tl, start_time)
                    if c5_pg < best_c5:
                        best_c5 = c5_pg
                        best_h = h_pg.copy()

    if rem() > 250:
        n1 = n + 40
        old_t = np.linspace(0.0, 2.0, n)
        new_t = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = project_onto_constraint(h_up + 0.01 * rng.standard_normal(n1), n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.005, 300)
        if rem() > 100:
            h_opt_up2, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.002, 300)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
                h_opt_up = h_opt_up2
        if rem() > 60:
            h_opt_up3, c5_opt_up3 = slsqp_optimize(h_opt_up, 0.0008, 300)
            if c5_opt_up3 < c5_opt_up:
                c5_opt_up = c5_opt_up3
                h_opt_up = h_opt_up3
        if rem() > 40:
            tl_up = min(rem() - 20, 50)
            h_pg_up, c5_pg_up = proj_grad_descent(h_opt_up, 2, 0.004, 0.04, 200000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()

    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0002, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 25:
        tl = min(rem() - 10, 40)
        h_pg, c5_pg = proj_grad_descent(best_h, 1, 0.001, 0.02, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
