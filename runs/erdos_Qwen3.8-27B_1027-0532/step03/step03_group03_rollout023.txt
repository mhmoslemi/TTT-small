def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Binary search on shift
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(proj < 1.0)[0]
            if len(idx) > 0:
                proj[idx] += residual / len(idx)
        else:
            idx = np.where(proj > 0.0)[0]
            if len(idx) > 0:
                proj[idx] += residual / len(idx)
    return np.clip(proj, 0.0, 1.0)

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(proj > 1e-12)[0]
        if len(idx) > 0:
            proj[idx] += residual / len(idx)
    return np.clip(proj, 0.0, 1.0)

    grid = np.linspace(0.0, 1.0, 80)

        if deadline is not None and time.time() > deadline:

        for i in order:

def slsqp_smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    dx = 2.0 / len(h)
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -500, 0)
    return dx * (cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Compute gradient
    grad = np.zeros(n)
    s_vals = np.arange(-(n-1), n)
    for k in range(len(s_vals)):
        w = weights[k]
        if w < 1e-14:
            continue
        s = s_vals[k]
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            valid_f = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
            valid_b = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
            if np.any(valid_f):
                j = np.where(valid_f)[0]
                grad[j] += w * (1.0 - h[j + s])
            if np.any(valid_b):
                j = np.where(valid_b)[0]
                grad[j] -= w * h[j - s]
    return dx * grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(proj > 1e-12)[0]
        if len(idx) > 0:
            proj[idx] += residual / len(idx)
    return np.clip(proj, 0.0, 1.0)

def coord_desc(h_init, n_sweeps, seed=42, deadline=None):
    n = len(h_init)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    s_vals = np.arange(-(n - 1), n)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    grid = np.linspace(0.0, 1.0, 60)
    n_grid = len(grid)
    idxs = np.arange(n)

    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(n)
        for i in order:
            h_old = h[i]
            valid_f = (idxs + s_vals >= 0) & (idxs + s_vals < n)
            valid_f[i] = False
            valid_b = (idxs - s_vals >= 0) & (idxs - s_vals < n)
            valid_b[i + (n - 1)] = False
            sf = s_vals
            a = np.zeros(2 * n - 1)
            jf = np.where(valid_f)[0]
            if len(jf) > 0:
                ig = i + sf[jf]
                a[jf] = 1.0 - h[ig]
            jb = np.where(valid_b)[0]
            if len(jb) > 0:
                ig = i - sf[jb]
                a[jb] -= h[ig]
            C0 = corr[n - 1] - h_old * (1.0 - h_old)
            b = corr.copy()
            if len(jf) > 0:
                ig = i + sf[jf]
                b[jf] -= h_old * (1.0 - h[ig])
            if len(jb) > 0:
                ig = i - sf[jb]
                b[jb] = b[jb] - h[ig] * (1.0 - h_old) + h[ig]
            par = C0 + grid - grid * grid
            lin = np.max(a[None, :] * grid[:, None] + b[None, :], axis=1)
            fg = dx * np.maximum(par, lin)
            bi = int(np.argmin(fg))
            lo_t = grid[max(0, bi - 1)]
            hi_t = grid[min(n_grid - 1, bi + 1)]
            for _ in range(20):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                f1 = dx * max(C0 + m1 - m1 * m1, float(np.max(a * m1 + b)))
                f2 = dx * max(C0 + m2 - m2 * m2, float(np.max(a * m2 + b)))
                if f1 < f2:
                    hi_t = m2
                else:
                    lo_t = m1
            t_star = (lo_t + hi_t) / 2.0
            delta = t_star - h_old
            if abs(delta) > 1e-15:
                h[i] = t_star
                corr += delta * a
                corr[n - 1] = C0 + t_star - t_star * t_star
        h = project_feasible(h, n / 2.0)
        corr = np.correlate(h, 1.0 - h, mode='full')
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score

def slsqp_obj_wrap(x, alpha, n_pts):
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    dx = 2.0 / n_pts
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return dx * (cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_grad_wrap(x, alpha, n_pts):
    n = n_pts
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    s_vals = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for k in range(len(s_vals)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = s_vals[k]
        if s == 0:
            grad += wk * (1.0 - 2.0 * x)
        else:
            vf = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
            jb = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
            jf = np.where(vf)[0]
            if len(jf) > 0:
                grad[jf] += wk * (1.0 - x[jf + s])
            jb_idx = np.where(jb)[0]
            if len(jb_idx) > 0:
                grad[jb_idx] -= wk * x[jb_idx - s]
    return dx * grad

def slsqp_refine(h_start, target, alphas, maxiter, ftol):
    n = len(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    for alpha in alphas:
        try:
            res = minimize(slsqp_obj_wrap, h_cur, args=(alpha, n),
                           jac=slsqp_grad_wrap, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
