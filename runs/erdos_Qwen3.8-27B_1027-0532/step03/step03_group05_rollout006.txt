def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    n_free = np.sum(mask)
    if n_free == 0:
        return h
    h[mask] += diff / n_free
    return np.clip(h, 0.0, 1.0)

def momentum_subgradient(h, target, time_limit, k_top=10, alpha=30.0, 
                         eta=0.01, momentum=0.9, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    velocity = np.zeros(n)
    no_improve = 0
    eta_local = eta
    
    while time.time() - start < time_limit:
        corr = get_corr(best)
        c_max = np.max(corr)
        
        # Top-k lags with weights
        sorted_idx = np.argsort(corr)[::-1]
        top_k = sorted_idx[:k_top]
        top_vals = corr[top_k]
        
        # Weight: exponential of (value - max)
        w = np.exp(alpha * (top_vals - c_max))
        w_sum = np.sum(w)
        if w_sum < 1e-15:
            w = np.ones(k_top) / k_top
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for wi, pos in zip(w, top_k):
            offset = int(pos) - (n - 1)
            if offset >= 0:
                # corr = sum_{i=0}^{n-1-offset} h[i+offset]*(1-h[i])
                # d/dh[j]: (1-h[j-offset]) for j in [offset, n-1]
                #          -h[j+offset] for j in [0, n-1-offset-1]
                for j in range(offset, n):
                    grad[j] += wi * (1.0 - best[j - offset])
                for j in range(n - offset):
                    grad[j] -= wi * best[j + offset]
            else:
                d = -offset
                # corr = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                # d/dh[j]: (1-h[j+d]) for j in [0, n-1-d]
                #          -h[j-d] for j in [d, n-1]
                for j in range(n - d):
                    grad[j] += wi * (1.0 - best[j + d])
                for j in range(d, n):
                    grad[j] -= wi * best[j - d]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        # Momentum update
        velocity = momentum * velocity - eta_local * grad / gn
        h_new = best + velocity
        h_new = project_feasible(h_new, target)
        s = compute_c5(h_new)
        
        if s < best_score:
            best_score = s
            best = h_new
            no_improve = 0
            eta_local = min(eta_local * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve > 30:
                eta_local *= 0.8
                alpha = min(alpha * 1.2, 200.0)
                velocity *= 0.5
                no_improve = 0
    
    return best, best_score

def slsqp_logsumexp(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    alphas = [10, 30, 100, 300, 1000]
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(50, remaining * 2))
        
        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max * 2.0 / n + np.log(np.sum(np.exp(shifted))) / a * 2.0 / n)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        try:
            res = minimize(obj, best, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best = h_new
                best_score = s
        except Exception:
            pass
    
    return best, best_score

    n = len(h)

When we change h[i] from old_val to new_val (delta = new_val - old_val), how does the correlation change?

corr[p] = sum_k h[k] * (1-h[k+offset_p]) where offset_p = p - (n-1)

The change in corr[p] due to changing h[i]:
- If i is in the "a" position (first array): h[i] changes, so we add delta * (1-h[i+offset_p]) for valid i
- If i is in the "v" position (second array): h[i] changes, so we subtract delta * h[i-offset_p] for valid i (since v = 1-h, changing h[i] by delta changes v[i] by -delta)

More precisely:
- Contribution to corr[p] from h[i] as the "a" element: h[i] appears when k=i, contributing h[i]*(1-h[i+offset_p])
  If i+offset_p is valid (0 ≤ i+offset_p < n), the change is delta*(1-h[i+offset_p])
  
- Contribution to corr[p] from h[i] as the "v" element: v[i] = 1-h[i] appears when k+offset_p = i, i.e., k = i-offset_p
  If i-offset_p is valid (0 ≤ i-offset_p < n), the change is -delta*h[i-offset_p]

So: Δcorr[p] = delta * (1-h[i+offset_p]) * I(0≤i+offset_p<n) - delta*h[i-offset_p] * I(0≤i-offset_p<n)

where offset_p = p - (n-1).

This means for a single coordinate change, we can compute the new correlation at ALL lags in O(n) time (since there are 2n-1 lags, and for each lag we just need to check if i+offset_p or i-offset_p is valid).

Actually, for a given i, which lags are affected?
- i+offset_p is valid when 0 ≤ i+offset_p < n, i.e., -i ≤ offset_p < n-i
  offset_p ranges from -(n-1) to n-1, so this is valid for offset_p in [-i, n-1-i]
  Which corresponds to p in [n-1-i, n-1+n-1-i] = [n-1-i, 2n-2-i]
  
- i-offset_p is valid when 0 ≤ i-offset_p < n, i.e., i-n < offset_p ≤ i
  Which corresponds to p in [n-1+i-n+1, n-1+i] = [i, n-1+i]

So the affected lags for coordinate i are:
p in [n-1-i, 2n-2-i] ∪ [i, n-1+i]

For i=0: [n-1, 2n-2] ∪ [0, n-1] = all lags
For i=n/2: [n/2-1, 3n/2-2] ∪ [n/2, 3n/2-1] ≈ half the lags

In any case, it's O(n) affected lags per coordinate change.

def update_corr_single(corr, h, i, new_val):
    """Update correlation after changing h[i] to new_val. Returns new corr array."""
    n = len(h)
    delta = new_val - h[i]
    if abs(delta) < 1e-15:
        return corr
    
    corr_new = corr.copy()
    offset_base = n - 1
    
    # For each lag position p, offset_p = p - (n-1)
    # Δcorr[p] = delta * (1-h[i+offset_p]) * I(0<=i+offset_p<n) 
    #           - delta*h[i-offset_p] * I(0<=i-offset_p<n)
    
    # First term: valid when 0 <= i+offset_p < n, i.e., -i <= offset_p < n-i
    # offset_p ranges from -(n-1) to n-1
    # So valid offset_p: max(-i, -(n-1)) to min(n-1-i, n-1)
    # = max(-i, -(n-1)) to min(n-i-1, n-1)

    hi1 = min(n - 1 - i, n - 1)

        p = off + offset_base
        corr_new[p] += delta * (1.0 - h[i + off])

    hi2 = min(n - 1, i)
