    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    grad = np.zeros(n)
    one_minus_h = 1.0 - x
    for i in range(len(weights)):
        w = weights[i]
        if w < 1e-14:

            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * one_minus_h[m2:]
                grad[m2:] -= w * x[:n - m2]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0, 1)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0, 1)
    return h

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        m = lags[i]
        lag = m - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * one_minus_h[m2:]
                grad[m2:] -= w * h[:n - m2]
    return grad

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    lags = np.arange(2 * n - 1)
    return compute_grad_weighted(x, lags, weights)

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):

    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new

                if 0 <= i + p < n:
                    slope += (1.0 - h[i + p])
                if 0 <= i - p < n:
                    slope -= h[i - p]
                slopes[idx] = slope
                bases[idx] = corr[m] - old_val * slope
            
            # Minimize max(bases + t*slopes) over t in [0,1]
            candidates = [0.0, 1.0]
            for a in range(k_use):
                for b in range(a+1, k_use):
                    ds = slopes[b] - slopes[a]
                    if abs(ds) > 1e-14:
                        tc = (bases[a] - bases[b]) / ds
                        if 0.0 < tc < 1.0:
                            candidates.append(tc)
            
            best_t = old_val
            best_obj = np.max(bases + old_val * slopes)
            for t in candidates:
                obj = np.max(bases + t * slopes)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            if abs(best_t - old_val) > 1e-12:
                h[i] = best_t
                improved = True
                delta = best_t - old_val
                for idx in range(k_use):
                    corr[top_idx[idx]] += delta * slopes[idx]
            
            if (i + 1) % recompute_every == 0:
                corr = overlap_corrs(h)
        
        h = project_feasible(h, target_sum)
        corr = overlap_corrs(h)
        if not improved and sweep > 3:
            break
    
    return h

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk_start, topk_end,
                    alpha, noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    n_corr = 2 * n - 1
    
    for it in range(max_iter):
        frac = it / max(float(max_iter - 1), 1)
        # Cosine annealing for learning rate
        lr_cos = lr_init * 0.5 * (1 + np.cos(np.pi * frac))
        # Blend with adaptive lr
        current_lr = 0.5 * lr_cos + 0.5 * lr
        
        current_topk = int(topk_start + (topk_end - topk_start) * frac)
        current_topk = max(current_topk, 1)
        
        corr = overlap_corrs(h)
        k_use = min(current_topk, n_corr)
        top_indices = np.argpartition(corr, -k_use)[-k_use:]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(k_use) / k_use
        else:
            w = w / w_sum
        
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - current_lr * v
        h_new = project_feasible(h_new, target_sum)
        
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.03, lr_init * 3)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        
        if stagnation > patience:
            break
    
    return best_h, best_score

def binary_local_search(h_binary, max_swaps=5000, seed=42):
    """Local search on binary solutions using single swaps."""
    rng = np.random.RandomState(seed)
    n = len(h_binary)
    h = h_binary.astype(float).copy()
    corr = overlap_corrs(h)
    current_score = np.max(corr)
    best_score = current_score
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(max_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i] = 0
        h[j] = 1
        new_corr = overlap_corrs(h)
        new_score = np.max(new_corr)
        
        if new_score <= current_score:
            # Accept
            corr = new_corr
            current_score = new_score
            # Update indices
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # Revert
            h[i] = 1
            h[j] = 0
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0, 1)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0, 1)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        m = lags[i]
        lag = m - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += w * one_minus_h[m2:]
                grad[m2:] -= w * h[:n - m2]
    return grad


def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    lags = np.arange(2 * n - 1)
    return compute_grad_weighted(x, lags, weights)


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h


def coordinate_descent_polish(h, n_sweeps=100, recompute_every=5, k_cd=40):
    n = len(h)
    h = h.copy()
    target_sum = n / 2.0
    corr = overlap_corrs(h)
    n_corr = 2 * n - 1
    k_use = min(k_cd, n_corr)

    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            top_idx = np.argpartition(corr, -k_use)[-k_use:]
            p_vals = top_idx - (n - 1)

            ip = i + p_vals
            ip_valid = (ip >= 0) & (ip < n)
            im = i - p_vals
            im_valid = (im >= 0) & (im < n)

            slopes = np.zeros(k_use)
            if np.any(ip_valid):
                slopes[ip_valid] += (1.0 - h[ip[ip_valid]])
            if np.any(im_valid):
                slopes[im_valid] -= h[im[im_valid]]

            bases = corr[top_idx] - old_val * slopes

            candidates = np.array([0.0, 1.0])
            ds_mat = slopes[None, :] - slopes[:, None]
            db_mat = bases[None, :] - bases[:, None]
            with np.errstate(divide='ignore', invalid='ignore'):
                tc_mat = np.where(np.abs(ds_mat) > 1e-14, db_mat / ds_mat, -2.0)
            tc_flat = tc_mat[np.triu_indices(k_use, k=1)]
            tc_valid = tc_flat[(tc_flat > 1e-10) & (tc_flat < 1.0 - 1e-10)]
            if len(tc_valid) > 0:
                candidates = np.concatenate([candidates, tc_valid])

            best_t = old_val
            best_obj = np.max(bases + old_val * slopes)
            for t in candidates:
                obj = np.max(bases + t * slopes)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t

            if abs(best_t - old_val) > 1e-12:
                h[i] = best_t
                improved = True
                delta = best_t - old_val
                corr[top_idx] += delta * slopes

            if (i + 1) % recompute_every == 0:
                corr = overlap_corrs(h)

        h = project_feasible(h, target_sum)
        corr = overlap_corrs(h)
        if not improved and sweep > 5:
            break

    return h

        current_topk = max(current_topk, 1)

        else:
            w = w / w_sum

        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - current_lr * v
        h_new = project_feasible(h_new, target_sum)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.03, lr_init * 3.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new

        if stagnation > patience:
            break

    return best_h, best_score


def binary_local_search(h_binary, max_swaps=20000, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_binary)
    h = h_binary.astype(float).copy()
    corr = overlap_corrs(h)
    current_score = np.max(corr)
    best_score = current_score
    best_h = h.copy()

    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]

    for it in range(max_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        new_corr = overlap_corrs(h)
        new_score = np.max(new_corr)

        if new_score <= current_score + 1e-15:
            corr = new_corr
            current_score = new_score
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):

    start_time = time.time()

        (5000, 0.08, 30, 5, 80, 0.04, 0.012, 400, 0.91),
    ]

    for i, cfg in enumerate(subgrad_configs):
        if _remaining() < 100:
            break
        h_try = best_h + rng.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=cfg[0], lr_init=cfg[1],

                                        momentum=cfg[8], seed=seed + i * 100)

    if _remaining() > 200:
        h_ref = slsqp_refine(best_h, target_init,

        sc = compute_c5(h_cd)

            best_h = h_cd

    if _remaining() > 150:
        h_bin = (best_h >= 0.5).astype(float)
        target_bin = int(n_init / 2)
        current_ones = int(np.sum(h_bin))
        if current_ones > target_bin:
            ones_idx = np.where(h_bin == 1)[0]
            rng.shuffle(ones_idx)
            h_bin[ones_idx[:current_ones - target_bin]] = 0.0
        elif current_ones < target_bin:
            zeros_idx = np.where(h_bin == 0)[0]
            rng.shuffle(zeros_idx)
            h_bin[zeros_idx[:target_bin - current_ones]] = 1.0

        h_bs, bs_score = binary_local_search(h_bin, max_swaps=20000, seed=seed + 5000)
        h_unrounded = h_bs + rng.randn(n_init) * 0.003
        h_unrounded = project_feasible(h_unrounded, target_init)

        h_new, score = run_subgradient(h_unrounded, target_init,
                                        max_iter=3000, lr_init=0.05,
                                        topk_start=10, topk_end=3,
                                        alpha=80, noise_prob=0.01,
                                        noise_scale=0.005, patience=300,
                                        momentum=0.92, seed=seed + 6000)
        if _remaining() > 80:
            h_ref = slsqp_refine(h_new, target_init, [100, 300], 1000, 1e-14)
            sc2 = compute_c5(h_ref)
            if sc2 < score:
                h_new = h_ref
                score = sc2
        h_cd2 = coordinate_descent_polish(h_new, n_sweeps=50, recompute_every=5, k_cd=40)
        sc_cd2 = compute_c5(h_cd2)

    subgrad_configs2 = [

        (3000, 0.02, 5, 2, 200, 0.0, 0.0, 250, 0.94),
    ]

    for i, cfg in enumerate(subgrad_configs2):
        if _remaining() < 60:
            break
        h_try = best_h + rng.randn(n_init) * 0.004
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=cfg[0], lr_init=cfg[1],

        sc = compute_c5(h_cd)

    for n_try in [500, 640, 800]:
        if _remaining() < 150:

        x_new = np.linspace(0.0, 1.0, n_try)

        if _remaining() > 50:
            h_cd3 = coordinate_descent_polish(h_new, n_sweeps=50, recompute_every=5, k_cd=30)
            sc_cd3 = compute_c5(h_cd3)
