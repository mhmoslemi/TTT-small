def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        # Compute all correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        corr_scaled = corr * dx
        max_idx = np.argsort(corr_scaled)[-int(n * top_k_frac):]

    n = len(h)

def nesterov_projected_gradient(h, n, target, max_steps, time_limit):

    start = time.time()

    h_cur = h.copy()

    best_h = h_cur.copy()

        weights /= np.sum(weights)

        indices = np.arange(len(corr))

            idx1 = np.arange(max(0, -k), min(n, n-k))

            best_c5 = c5_val
            best_h = h_new.copy()

    """Project h onto {x: 0 <= x_i <= 1, sum(x) = S}"""

    # Bisection on lambda
    lo = h.min() - S  # or something that makes sum(clip(h-lo, 0, 1)) > S
    hi = h.max() + S  # or something that makes sum(clip(h-hi, 0, 1)) < S
    
    # Actually, the correct bounds:
    # sum(clip(h - λ, 0, 1)) is decreasing in λ
    # At λ → -∞: sum → n
    # At λ → +∞: sum → 0
    # We need sum = S
    
    lo = h.min() - 1 - S  # definitely sum > S (all clipped to 1)
    hi = h.max() + 1 + S  # definitely sum < S (all clipped to 0)
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2, 0, 1)

    h = h.astype(float)
    n = len(h)

def active_set_lp(h_init, n, target, max_iter, time_limit, top_k_frac, damping):
    """LP with only top-k% correlation constraints"""
    start = time.time()
    dx = 2.0 / n
    h = project_sum_box(h_init, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_top = max(5, int(2 * n * top_k_frac))
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_idx = np.argsort(corr)[-n_top:]
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k_pos in top_idx:
            k = int(k_pos) - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=10000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break

        h = (1.0 - damping) * h + damping * h_new
        h = project_sum_box(h, target)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum_box(h, S):
    h = h.astype(float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def nesterov_pg(h_start, n, target, max_steps, time_limit, alpha=200.0):
    start = time.time()
    dx = 2.0 / n
    h_cur = project_sum_box(h_start, target)
    h_mom = h_cur.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    lr_base = 0.05

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        t_k = 1.0 + step
        beta = (t_k - 1.0) / (t_k + 0.5)
        h_mom = (1.0 + beta) * h_mom - beta * h_cur
        h_mom = project_sum_box(h_mom, target)

        corr = np.correlate(h_mom, 1.0 - h_mom, mode='full') * dx
        max_corr = np.max(corr)
        weights = np.exp(np.clip(alpha * (corr - max_corr), -500.0, 0.0))
        sw = weights.sum()
        if sw < 1e-100:
            break
        weights /= sw

        grad = np.zeros(n)
        for k_pos in range(len(weights)):
            w = weights[k_pos]
            if w < 1e-12:
                continue
            k = k_pos - n_half
            if k >= 0:
                idx = np.arange(0, n - k)
                grad[idx] += w * dx * (1.0 - h_mom[idx + k])
                grad[idx + k] -= w * dx * h_mom[idx]
            else:
                ik = -k
                idx = np.arange(ik, n)
                grad[idx] += w * dx * (1.0 - h_mom[idx + k])
                grad[idx + k] -= w * dx * h_mom[idx]

        lr = lr_base / (1.0 + step * 0.01)
        h_new = h_cur - lr * grad
        h_new = project_sum_box(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_new.copy()

        h_cur = h_new

    return best_h, best_c5
