```python
def gradient_step(h, n, target, step_size, top_k=5):
    corr = compute_corr(h)
    # Find top_k lags
    top_indices = np.argsort(corr)[-top_k:]
    # Compute gradient for each top lag and average (weighted by correlation value)
    weights = np.exp(corr[top_indices] * 10)  # softmax-ish
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_indices):
        m = ki - (n - 1)  # lag
        grad += wi * compute_corr_grad(h, m)
    
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, target)
    return h_new
```

```python
        weights = exp_s / sum_exp
        grad = np.zeros(n)
        for wi, ki in zip(weights, range(len(corr))):
            m = ki - (n - 1)
            grad += wi * corr_gradient(x, m)
        return grad
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def make_binary_qr(n, p, offset=0):
    residues = set((a * a) % p for a in range(1, p))
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if ((i + offset) % p) in residues else 0.0
    return h

def make_random_binary(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def binary_local_search(h, n, target, deadline, n_flips_max=20000):
    h_cur = h.copy()
    h_cur[h_cur > 0.5] = 1.0
    h_cur[h_cur <= 0.5] = 0.0
    h_cur = project_feasible(h_cur, target)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score
    
    T = 0.3
    n_flips = 0
    patience = 2000
    no_improve = 0
    
    while time.time() < deadline and n_flips < n_flips_max:
        n_flips += 1
        i = np.random.randint(0, n)
        h_test = h_cur.copy()
        if h_test[i] > 0.5:
            h_test[i] = 0.0
        else:
            h_test[i] = 1.0
        h_test = project_feasible(h_test, target)
        sc_test = compute_c5(h_test)
        
        if sc_test < cur_score or (np.random.random() < np.exp(-(sc_test - cur_score) / T)):
            h_cur = h_test
            cur_score = sc_test
            if cur_score < best_score:
                best_score = cur_score
                best_h = h_cur.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > patience:
            T *= 0.9
            no_improve = 0
            patience = max(500, int(patience * 0.95))
        
        if n_flips % 5000 == 0:
            T *= 0.95
    
    return best_h, best_score

def smooth_binary(h_bin, n, bandwidth):
    kernel = np.zeros(2 * bandwidth + 1)
    for i in range(bandwidth + 1):
        kernel[bandwidth + i] = np.exp(-i ** 2 / (2.0 * (bandwidth / 2.0 + 1) ** 2))
        kernel[bandwidth - i] = kernel[bandwidth + i]
    kernel /= np.sum(kernel)
    h_smooth = np.convolve(h_bin, kernel, mode='same')
    return project_feasible(h_smooth, n / 2.0)

def try_upscale(h, n, target, n_up, deadline):
    dx_old = 2.0 / n
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_up = np.linspace(0, 2, n_up, endpoint=False)
    h_up = np.interp(x_up, x_old, h)
    h_up = project_feasible(h_up, n_up / 2.0)
    h_up, sc_up = gradient_projection(h_up, n_up, n_up / 2.0, deadline, top_k=5)
    return h_up, sc_up, n_up

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_score = compute_c5(best_h)
    
    def elapsed():
        return time.time() - start_time
    
    def time_left():
        return budget_s - 25.0 - elapsed()
    
    # Phase 1: Gradient projection with different parameters
    deadline1 = start_time + budget_s * 0.15
    h_g1, sc_g1 = gradient_projection(best_h, n, target, deadline1, top_k=5, smooth_beta=5.0)
    if sc_g1 < best_score:
        best_score = sc_g1
        best_h = h_g1
    
    # Phase 2: LSE-SLSQP polish
    if time_left() > 60:
        deadline2 = start_time + budget_s * 0.3
        for alpha in [100, 500, 2000]:
            if time_left() < 30:
                break
            h_r, sc_r = lse_slsqp(best_h, n, target, min(deadline2, time.time() + 40), alpha)
            if sc_r < best_score:
                best_score = sc_r
                best_h = h_r
    
    # Phase 3: Gradient projection with different top_k
    if time_left() > 60:
        deadline3 = start_time + budget_s * 0.4
        for tk in [3, 10, 20]:
            if time_left() < 20:
                break
            h_g, sc_g = gradient_projection(best_h, n, target, min(deadline3, time.time() + 25), top_k=tk, smooth_beta=3.0)
            if sc_g < best_score:
                best_score = sc_g
                best_h = h_g
    
    # Phase 4: Binary pattern local search
    if time_left() > 80:
        deadline4 = start_time + budget_s * 0.55
        h_bin = best_h.copy()
        h_bin[h_bin > 0.5] = 1.0
        h_bin[h_bin <= 0.5] = 0.0
        h_bin = project_feasible(h_bin, target)
        h_bl, sc_bl = binary_local_search(h_bin, n, target, min(deadline4, time.time() + 60))
        if sc_bl < best_score:
            best_score = sc_bl
            best_h = h_bl
        # Also smooth the binary result and optimize
        h_sm = smooth_binary(h_bl, n, 3)
        h_sm_opt, sc_sm = gradient_projection(h_sm, n, target, min(deadline4, time.time() + 20), top_k=5)
        if sc_sm < best_score:
            best_score = sc_sm
            best_h = h_sm_opt
    
    # Phase 5: Diverse starting points
    if time_left() > 80:
        deadline5 = start_time + budget_s * 0.78
        candidates = []
        
        for p in [19, 31, 43, 59, 71, 83, 97]:
            if p < n:
                for off in [0, p // 3, 2 * p // 3]:
                    h_qr = make_binary_qr(n, p, off)
                    h_qr = project_feasible(h_qr, target)
                    candidates.append(h_qr)
        
        for _ in range(8):
            h_rb = make_random_binary(n, rng)
            candidates.append(project_feasible(h_rb, target))
        
        # Smoothed versions of current best
        for bw in [2, 4, 8]:
            h_sm = smooth_binary((best_h > 0.5).astype(float), n, bw)
            candidates.append(h_sm)
        
        for cand in candidates:
            if time_left() < 20:
                break
            h_c, sc_c = gradient_projection(cand, n, target, min(deadline5, time.time() + 12), top_k=5)
            if sc_c < best_score:
                best_score = sc_c
                best_h = h_c
    
    # Phase 6: Upscale to higher resolution
    if time_left() > 80:
        for n_up in [600, 800]:
            if time_left() < 40:
                break
            deadline_up = min(start_time + budget_s - 30, time.time() + 50)
            h_u, sc_u, n_u = try_upscale(best_h, n, target, n_up, deadline_up)
            if sc_u < best_score:
                best_score = sc_u
                best_h = h_u
                n = n_u
                target = n / 2.0
    
    # Phase 7: Penalty SLSQP on best
    if time_left() > 40:
        deadline7 = start_time + budget_s - 30
        h_p, sc_p = penalty_slsqp(best_h, n, target, min(deadline7, time.time() + 40),
                                   T_init=best_score, n_outer=12)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p
    
    # Phase 8: Final gradient projection polish
    if time_left() > 15:
        h_f, sc_f = gradient_projection(best_h, n, target, start_time + budget_s - 26, top_k=3, smooth_beta=10.0)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f
    
    # Phase 9: One more LSE-SLSQP
    if time_left() > 15:
        h_l, sc_l = lse_slsqp(best_h, n, target, start_time + budget_s - 26, 5000)
        if sc_l < best_score:
            best_score = sc_l
            best_h = h_l
    
    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
