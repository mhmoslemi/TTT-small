def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * float(np.log(np.sum(np.exp(x))))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = float(np.sum(ex))
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * float(np.log(np.sum(np.exp(x))))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = float(np.sum(ex))
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def binary_sa(h_start, n_steps, t_start, t_end, rng, structured_prob=0.6):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    ones_pos = np.where(h == 1.0)[0].copy()
    zeros_pos = np.where(h == 0.0)[0].copy()
    n_one = len(ones_pos)
    n_zero = len(zeros_pos)
    half = n - 1
    pos2oi = np.full(n, -1, dtype=np.int64)
    pos2zi = np.full(n, -1, dtype=np.int64)
    for idx in range(n_one):
        pos2oi[int(ones_pos[idx])] = idx
    for idx in range(n_zero):
        pos2zi[int(zeros_pos[idx])] = idx

    step = 0
    inv_steps = 1.0 / max(n_steps - 1, 1)
    log_ratio = np.log(max(t_end, 1e-15) / max(t_start, 1e-15))

    while step < n_steps:
        t = t_start * np.exp(log_ratio * step * inv_steps)

        if rng.random() < structured_prob:
            worst_k = int(np.argmax(corr))
            worst_s = worst_k - half
            vi_min = max(0, -worst_s)
            vi_max = min(n - 1, n - 1 - worst_s)
            found = False
            for _ in range(10):
                oi = int(rng.integers(n_one))
                zi = int(rng.integers(n_zero))
                i = int(ones_pos[oi])
                j = int(zeros_pos[zi])
                if vi_min <= i <= vi_max and vi_min <= j <= vi_max:
                    found = True
                    break
            if not found:
                oi = int(rng.integers(n_one))
                zi = int(rng.integers(n_zero))
                i = int(ones_pos[oi])
                j = int(zeros_pos[zi])
        else:
            oi = int(rng.integers(n_one))
            zi = int(rng.integers(n_zero))
            i = int(ones_pos[oi])
            j = int(zeros_pos[zi])

        if i == j:
            step += 1
            continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)

        if s_min > s_max:
            h[i] = 0.0
            h[j] = 1.0
            ones_pos[oi] = j
            zeros_pos[zi] = i
            pos2oi[j] = oi
            pos2zi[i] = zi
            pos2oi[i] = -1
            pos2zi[j] = -1
            current_max = float(np.max(corr))
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
            step += 1
            continue

        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_obj = new_max - current_max
        accept = delta_obj < 0.0
        if not accept and t > 1e-12:
            accept = rng.random() < np.exp(-delta_obj / t)
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            ones_pos[oi] = j
            zeros_pos[zi] = i
            pos2oi[j] = oi
            pos2zi[i] = zi
            pos2oi[i] = -1
            pos2zi[j] = -1
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta

        step += 1

    return best_h, best_c5_val

def deterministic_descent(h_start, n_passes, rng):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1
    ones_arr = np.where(h == 1.0)[0].copy()
    zeros_arr = np.where(h == 0.0)[0].copy()

    for p in range(n_passes):
        improved = False
        n_one = len(ones_arr)
        n_zero = len(zeros_arr)
        if n_one == 0 or n_zero == 0:
            break
        n_sample = min(2000, n_one * n_zero)
        best_delta = 1e10
        best_pair = (-1, -1)
        for _ in range(n_sample):
            i = int(ones_arr[rng.integers(n_one)])
            j = int(zeros_arr[rng.integers(n_zero)])
            if i == j:
                continue
            s_min = max(-i, i - half, -j, j - half)
            s_max = min(half - i, i, half - j, j)
            if s_min > s_max:
                continue
            s_range = np.arange(s_min, s_max + 1)
            k_range = s_range + half
            delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
            new_vals = corr[k_range] + delta
            max_in = float(np.max(new_vals))
            if k_range[0] > 0:
                max_out = float(np.max(corr[:k_range[0]]))
                if k_range[-1] < len(corr) - 1:
                    max_out = max(max_out, float(np.max(corr[k_range[-1]+1:])))
            else:
                max_out = 0.0
                if k_range[-1] < len(corr) - 1:
                    max_out = float(np.max(corr[k_range[-1]+1:]))
            candidate = max(max_in, max_out)
            d = candidate - current_max
            if d < best_delta:
                best_delta = d
                best_pair = (i, j)

        if best_pair[0] >= 0 and best_delta < -1e-10:
            i, j = best_pair
            s_min = max(-i, i - half, -j, j - half)
            s_max = min(half - i, i, half - j, j)
            if s_min <= s_max:
                s_range = np.arange(s_min, s_max + 1)
                k_range = s_range + half
                delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
                corr[k_range] += delta
            h[i] = 0.0
            h[j] = 1.0
            ones_arr = ones_arr[ones_arr != i]
            ones_arr = np.append(ones_arr, j)
            zeros_arr = zeros_arr[zeros_arr != j]
            zeros_arr = np.append(zeros_arr, i)
            current_max = float(np.max(corr))
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
            improved = True

        if not improved:
            break

    return best_h, best_c5_val

                best_h = h_pol.copy()

    # Phase 7: One more SA round if time permits
    if time_left() > 80:
        h_bin_final = to_binary(best_h)
        n_final = len(h_bin_final)
        n_steps = min(int((time_left() - 30) * 3000), 500000)
        if n_steps > 3000:
            h_bin_final, bin_max = binary_sa(h_bin_final, n_steps, 0.2, 0.0005, rng, structured_prob=0.8)
            c5_final_bin = (2.0 / n_final) * bin_max
            if c5_final_bin < best_c5:
                best_c5 = c5_final_bin
                best_h = h_bin_final.copy()
                if time_left() > 20:
                    h_pol, c5 = slsqp_optimize(best_h, 0.0001, 100)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_pol.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
