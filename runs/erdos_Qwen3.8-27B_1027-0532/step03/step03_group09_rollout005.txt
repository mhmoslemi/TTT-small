            if time.time() - start_time > time_limit:
                raise ValueError("timeout")
        return smooth_max_grad(x, eps)

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(obj, h0, args=(), jac=grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient_bb(h_init, topk, temp, max_iter, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = project_onto_constraint(h_init.copy(), target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    g_prev = None
    h_prev = None
    lr = 0.1
    obj_cur = smooth_max_obj(h_cur, temp)
    no_improve = 0
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h_cur)
        topk_idx = np.argsort(corr)[-topk:]
        xc = corr[topk_idx]
        xc = xc - np.max(xc)
        w = np.exp(xc / temp)
        w = w / np.sum(w)
        dx = 2.0 / n
        omh = 1.0 - h_cur
        grad = np.zeros(n)
        for i in range(len(topk_idx)):
            wk = w[i]
            if wk < 1e-12:
                continue
            kidx = topk_idx[i]
            s = kidx - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:m]
        if h_prev is not None and g_prev is not None:
            s_vec = h_cur - h_prev
            y_vec = grad - g_prev
            ys = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if ss > 1e-15:
                if abs(ys) > 1e-12:
                    lr = np.clip(ss / abs(ys), 0.003, 0.5)
                else:
                    lr *= 0.9
        alpha = lr
        for ls in range(12):
            h_new = project_onto_constraint(h_cur - alpha * grad, target_sum)
            obj_new = smooth_max_obj(h_new, temp)
            gsq = np.dot(grad, grad)
            if obj_new < obj_cur - 1e-10 * alpha * gsq or ls >= 11:
                break
            alpha *= 0.5
        h_prev = h_cur
        g_prev = grad
        h_cur = h_new
        obj_cur = obj_new
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 3000:
            break
    return best_h, best_c5

def gen_init_type(h_base, rng, n, target_sum, typ):
    if typ == 0:
        return project_onto_constraint(h_base + 0.03 * rng.standard_normal(n), target_sum)
    elif typ == 1:
        idx = rng.choice(n, size=max(5, n // 8), replace=False)
        hp = h_base.copy()
        hp[idx] = np.clip(hp[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(hp, target_sum)
    elif typ == 2:
        freq = float(rng.integers(1, 5))
        x = np.linspace(0, 2, n)
        amp = float(rng.uniform(0.2, 0.45))
        h_sin = np.clip(0.5 + amp * np.cos(2 * np.pi * freq * x / 2.0), 0, 1)
        alpha = float(rng.uniform(0.4, 0.8))
        return project_onto_constraint(alpha * h_base + (1 - alpha) * h_sin, target_sum)
    elif typ == 3:
        h_bin = (rng.random(n) < 0.5).astype(float)
        alpha = float(rng.uniform(0.5, 0.9))
        return project_onto_constraint(alpha * h_base + (1 - alpha) * h_bin, target_sum)
    elif typ == 4:
        max_len = max(2, n // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            hp = h_base.copy()
            blk = hp[a:a + blen].copy()
            hp[a:a + blen] = hp[b:b + blen]
            hp[b:b + blen] = blk
            return project_onto_constraint(hp, target_sum)
        return project_onto_constraint(h_base + 0.01 * rng.standard_normal(n), target_sum)
    elif typ == 5:
        n_blocks = int(rng.integers(3, 8))
        h_blk = np.zeros(n)
        edges = np.sort(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
        h_blk[edges] = 1.0
        h_blk[(np.roll(h_blk, 1) != h_blk).astype(int) == 1] = 0.5
        h_blk = np.cumsum(h_blk) % 2
        return project_onto_constraint(h_blk, target_sum)
    else:
        return project_onto_constraint(h_base + 0.01 * rng.standard_normal(n), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(180), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: Quick projected gradient to enter a good basin
    if rem() > 150:
        h_pg, c5_pg = projected_gradient_bb(best_h, 5, 0.05, 30000, rem() - 130, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 2: SLSQP moderate eps
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 300, rem() - 60, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Projected gradient with smaller topk and temp
    if rem() > 200:
        h_pg, c5_pg = projected_gradient_bb(best_h, 3, 0.02, 50000, rem() - 180, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: SLSQP smaller eps
    if rem() > 70:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300, rem() - 50, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Projected gradient single worst
    if rem() > 150:
        h_pg, c5_pg = projected_gradient_bb(best_h, 1, 0.005, 80000, rem() - 130, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 6: SLSQP very small eps
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 400, rem() - 40, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Multi-restart with diverse perturbations
    restart_count = 0
    while rem() > 100 and restart_count < 30:
        strat = restart_count % 6
        h_pert = gen_init_type(best_h, rng, n0, target_sum, strat)
        # Quick PG
        tl1 = min(rem() - 80, 40)
        if tl1 > 10:
            h_pg, c5_pg = projected_gradient_bb(h_pert, 3, 0.02, 20000, tl1, time.time())
            if c5_pg < compute_c5(h_pert):
                h_pert = h_pg
        # SLSQP polish
        if rem() > 60:
            h_opt, c5_opt = slsqp_optimize(h_pert, 0.001, 150, rem() - 40, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        # Additional PG on best
        if rem() > 50:
            tl3 = min(rem() - 35, 30)
            if tl3 > 10:
                h_pg2, c5_pg2 = projected_gradient_bb(best_h, 2, 0.005, 30000, tl3, time.time())
                if c5_pg2 < best_c5:
                    best_c5 = c5_pg2
                    best_h = h_pg2.copy()
        restart_count += 1

    # Phase 8: Resolution increase
    if rem() > 250:
        n1 = n0 + 60
        old_t = (np.arange(n0) + 0.5) / n0
        new_t = (np.arange(n1) + 0.5) / n1
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = project_onto_constraint(h_up + 0.003 * rng.standard_normal(n1), n1 / 2.0)

        tl_up1 = min(rem() - 200, 100)
        if tl_up1 > 20:
            h_up, c5_up = projected_gradient_bb(h_up, 5, 0.05, 20000, tl_up1, time.time())
        if rem() > 180:
            h_up, c5_up = slsqp_optimize(h_up, 0.008, 200, min(rem() - 160, 80), start_time)
        if rem() > 120:
            h_up, c5_up = projected_gradient_bb(h_up, 3, 0.01, 40000, min(rem() - 100, 80), time.time())
        if rem() > 60:
            h_up, c5_up = slsqp_optimize(h_up, 0.002, 200, min(rem() - 40, 60), start_time)

        c5_up_final = compute_c5(h_up)
        if c5_up_final < best_c5:
            best_c5 = c5_up_final
            best_h = h_up.copy()
            n0 = n1

    # Phase 9: Final fine polish
    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300, rem() - 20, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 25:
        h_pg, c5_pg = projected_gradient_bb(best_h, 1, 0.001, 50000, rem() - 10, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
