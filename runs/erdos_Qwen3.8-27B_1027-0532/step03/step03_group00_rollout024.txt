    n = len(h)

    C = np.max(corr)
    k_star = np.argmax(corr)

    shifted = _g_alpha * (corr - C)
    shifted = np.clip(shifted, -700.0, 0.0)
    S = np.sum(np.exp(shifted))

            g[ellp:] -= h[:n-ellp]
        grad_h += wc * g
    return grad_h

In practice, for L-BFGS-B, this approximation (ignoring the projection in the gradient) works well when the solution is already close to feasible (which it should be after a few iterations, since we keep projecting).

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
