def power_sum_objective(h, p, target_sum):
    """Smooth approximation using power sum."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Power sum approximation of max
    c_max = np.max(corr)
    # Shift to avoid overflow
    shifted = corr - c_max
    # Use exp scaling
    alpha = p
    val = c_max + np.log(np.sum(np.exp(alpha * shifted))) / alpha
    # Add constraint penalty
    constraint_violation = np.sum(h) - target_sum
    val += 1000.0 * constraint_violation ** 2
    return val

def nesterov_subgradient(h, target_sum, max_time, rng):
    """Nesterov-accelerated subgradient method."""
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, target_sum).copy()
    
    best_score = compute_c5(h)
    best_h = h.copy()
    
    # Nesterov parameters
    beta = 0.9  # momentum
    lr = 0.001
    v = np.zeros(n)  # velocity
    
    elapsed_start = time.time()
    stagnant = 0
    
    while time.time() - elapsed_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 3000:
                break
        
        # Subgradient: active set
        tol = (score - best_score) * 0.1
        if tol < 0:
            tol = 1e-6
        active = corr >= c_max - max(tol, 1e-6)
        n_active = max(1, int(np.sum(active)))
        
        s_arr = np.arange(2 * n - 1) - (n - 1)
        g = np.zeros(n)
        for idx in np.where(active)[0]:
            s_val = s_arr[idx]
            # Gradient of corr[idx] w.r.t. h
            # corr[idx] = sum_i h[i]*(1-h[i+s]) for valid i
            # d/corr[idx] / dh[j] = (1-h[j+s]) * [0<=j+s<n] - h[j-s] * [0<=j-s<n]
            A = np.zeros(n)
            s_lo = 0
            s_hi = n - 1
            # (1-h[j+s]) for j+s in [0,n): j in [-s, n-1-s] intersect [0,n-1]
            j_lo = max(0, -s_val)
            j_hi = min(n-1, n-1-s_val)
            if j_lo <= j_hi:
                A[j_lo:j_hi+1] += (1.0 - h[j_lo+s_val : j_hi+s_val+1])
            
            # -h[j-s] for j-s in [0,n): j in [s, n-1+s] intersect [0,n-1]
            j_lo2 = max(0, s_val)
            j_hi2 = min(n-1, n-1+s_val)
            if j_lo2 <= j_hi2:
                A[j_lo2:j_hi2+1] -= h[j_lo2-s_val : j_hi2-s_val+1]
            
            g += A / n_active
        
        # Nesterov update
        v = beta * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        
        # Line search
        score_new = compute_c5(h_new)
        if score_new < score:
            h = h_new
            lr = min(lr * 1.1, 0.005)
        else:
            lr *= 0.7
            v *= 0.5
            if lr < 1e-7:
                break
    
    return best_h, best_score

def block_swap_search(h, target_sum, max_time, rng):

        h_new = project_feasible(h_new, target_sum)
        sc = compute_c5(h_new)
        
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h_new.copy()
            h = h_new.copy()
    
    return best_h, best_score

    return h

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coord_descend_incr(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, target_sum).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()

    while time.time() - elapsed_start < max_time:
        order = rng.permutation(n)
        for j in order:
            if time.time() - elapsed_start > max_time:
                break
            old_val = h[j]
            A = np.zeros(2 * n - 1)
            s_lo = max(-(n - 1), -j)
            s_hi = min(n - 1, n - 1 - j)
            if s_lo <= s_hi:
                sv = np.arange(s_lo, s_hi + 1)
                A[sv + n - 1] += (1.0 - h[j + sv])
            s_lo2 = max(-(n - 1), j - (n - 1))
            s_hi2 = min(n - 1, j)
            if s_lo2 <= s_hi2:
                sv = np.arange(s_lo2, s_hi2 + 1)
                A[sv + n - 1] -= h[j - sv]

            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = int(np.argmax(vals))
                if A[k_star] > 1e-12:
                    hi = mid
                elif A[k_star] < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val

        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            noise_scale = 0.003 + 0.008 * min(1.0, no_improve / (2.0 * patience))
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_h, best_score

def smooth_obj_grad(x, p_val, n_pts):
    dx_val = 2.0 / n_pts
    corr_raw = np.correlate(x, 1.0 - x, mode='full')
    c = corr_raw * dx_val
    c_max = float(np.max(c))
    shifted = np.clip(p_val * (c - c_max), -200.0, 0.0)
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    w_norm = weights / w_sum
    val = c_max + np.log(w_sum) / p_val

    s_arr = np.arange(2 * n_pts - 1) - (n_pts - 1)
    grad = np.zeros(n_pts)
    for idx in range(2 * n_pts - 1):
        if w_norm[idx] < 1e-12:
            continue
        s_val = s_arr[idx]
        wt = w_norm[idx] * dx_val
        j_lo = max(0, -s_val)
        j_hi = min(n_pts - 1, n_pts - 1 - s_val)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += wt * (1.0 - x[j_lo + s_val:j_hi + s_val + 1])
        j_lo2 = max(0, s_val)
        j_hi2 = min(n_pts - 1, n_pts - 1 + s_val)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= wt * x[j_lo2 - s_val:j_hi2 - s_val + 1]
    return val, grad

def slsqp_with_grad(h, target_sum, p_values, maxiter_each, time_limit):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()

    for p in p_values:
        if time.time() - start > time_limit:
            break
        try:
            def objective(x, p_use=p, n_use=n):
                return smooth_obj_grad(x, p_use, n_use)[0]

            def gradient(x, p_use=p, n_use=n):
                return smooth_obj_grad(x, p_use, n_use)[1]

            def constraint_fun(x, t=target_sum):
                return np.sum(x) - t

            def constraint_jac(x, n_use=n):
                return np.ones(n_use)

            res = minimize(objective, h_cur, jac=gradient, method='SLSQP',
                           bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': constraint_fun,
                                        'jac': constraint_jac}],
                           options={'maxiter': maxiter_each, 'ftol': 1e-12})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

def nesterov_subgrad(h, target_sum, max_time, rng):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h, target_sum).copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    lr = 0.0008
    beta = 0.85
    v = np.zeros(n)
    elapsed_start = time.time()
    stagnant = 0
    s_arr = np.arange(2 * n - 1) - (n - 1)

    while time.time() - elapsed_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 2000:
                break

        tol = max((score - best_score) * 0.2, 1e-7)
        active = (corr - c_max) >= -tol
        n_active = max(1, int(np.sum(active)))
        g = np.zeros(n)
        active_idx = np.where(active)[0]
        for idx in active_idx:
            s_val = s_arr[idx]
            j_lo = max(0, -s_val)
            j_hi = min(n - 1, n - 1 - s_val)
            if j_lo <= j_hi:
                g[j_lo:j_hi + 1] += (1.0 - h[j_lo + s_val:j_hi + s_val + 1])
            j_lo2 = max(0, s_val)
            j_hi2 = min(n - 1, n - 1 + s_val)
            if j_lo2 <= j_hi2:
                g[j_lo2:j_hi2 + 1] -= h[j_lo2 - s_val:j_hi2 - s_val + 1]
        g /= n_active

        v = beta * v - lr * g
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, target_sum)
        score_new = compute_c5(h_new)
        if score_new < score - 1e-15:
            h = h_new
            lr = min(lr * 1.08, 0.004)
        else:
            lr *= 0.75
            v *= 0.4
            if lr < 1e-7:
                break
    return best_h, best_score

def pair_flip_search(h, target_sum, max_time, rng):
    n = len(h)
    h = project_feasible(h, target_sum).copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    elapsed_start = time.time()
    while time.time() - elapsed_start < max_time:
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] == h[j]:
            continue
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        h_new = project_feasible(h_new, target_sum)
        sc = compute_c5(h_new)
        if sc < best_score - 1e-10:
            best_score = sc
            best_h = h_new.copy()
            h = h_new.copy()
    return best_h, best_score

def binary_projection(h, target_sum):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    n_target = int(round(target_sum))
    if n_ones > n_target:
        order = np.argsort(h)
        for idx in order[:n_ones - n_target]:
            h_bin[idx] = 0.0
    elif n_ones < n_target:
        order = np.argsort(-h)
        for idx in order[:n_target - n_ones]:
            h_bin[idx] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent (main workhorse)
    if tleft() > 180:
        h1, s1 = coord_descend_incr(best_h, target_init, min(280, tleft() - 180), rng, patience=25)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: SLSQP with analytical gradient
    if tleft() > 120:
        h2, s2 = slsqp_with_grad(best_h, target_init, [300, 800, 1500], 200, max(30, tleft() - 120))
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Nesterov subgradient
    if tleft() > 80:
        h3, s3 = nesterov_subgrad(best_h, target_init, min(70, tleft() - 80), rng)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Pair flip local search
    if tleft() > 50:
        h4, s4 = pair_flip_search(best_h, target_init, min(40, tleft() - 50), rng)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Multi-resolution exploration
    n_opts = [n_init + 80, n_init - 60, n_init + 160, n_init + 50]
    for n_try in n_opts:
        if tleft() < 100:
            break
        if n_try < 100 or n_try == n_init:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h5, s5 = coord_descend_incr(h_interp, target_try, min(55, tleft() - 100), rng, patience=15)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Binary projection + refine
    if tleft() > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_bin = binary_projection(best_h, target_cur)
        h_bin_smooth = h_bin + rng.randn(n_cur) * 0.02
        h_bin_smooth = project_feasible(h_bin_smooth, target_cur)
        h6, s6 = coord_descend_incr(h_bin_smooth, target_cur, min(50, tleft() - 80), rng, patience=15)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final CD polish
    n_best = len(best_h)
    target_best = n_best / 2.0
    if tleft() > 50:
        h7, s7 = coord_descend_incr(best_h, target_best, min(40, tleft() - 50), rng, patience=20)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final SLSQP
    if tleft() > 40:
        h8, s8 = slsqp_with_grad(best_h, target_best, [1000, 2000], 150, max(20, tleft() - 40))
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Small random perturbations
    for _ in range(3):
        if tleft() < 30:
            break
        h_pert = best_h + rng.randn(n_best) * 0.003
        h_pert = project_feasible(h_pert, target_best)
        h9, s9 = coord_descend_incr(h_pert, target_best, min(25, tleft() - 30), rng, patience=10)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        h_final = h_final
    else:
        h_final = project_feasible(best_h, target_final)

    return h_final, best_score, n_final
