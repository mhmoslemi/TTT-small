    """Compute the change in correlation when h[a] goes 1->0 and h[b] goes 0->1.
    Returns array of length 2n-1 (the full correlation change)."""
    delta = np.zeros(2*n-1)
    
    # h[a]: 1 -> 0
    # For shifts s >= 0 (index k = (n-1) + s, s >= 0):
    #   -(1-h[a+s]) if a+s < n
    #   +h[a-s] if 0 <= a-s <= n-1-s, i.e., s <= a and a-s >= 0
    # For shifts s < 0 (index k = (n-1) + s, s < 0, let t = -s > 0):
    #   -(1-h[a-t]) if a >= t, i.e., t <= a
    #   +h[a+t] if a+t < n, i.e., t < n-a
    
    # h[b]: 0 -> 1
    # For shifts s >= 0:
    #   +(1-h[b+s]) if b+s < n
    #   -h[b-s] if s <= b
    # For shifts s < 0 (t = -s):
    #   +(1-h[b-t]) if b >= t
    #   -h[b+t] if b+t < n
    
    # Let me index by shift s from -(n-1) to n-1
    # corr index k = s + (n-1)
    
    # Position a (1->0):
    # Positive shifts s = 0, 1, ..., n-1-a:
    #   delta at shift s: -(1-h[a+s])
    for s in range(n - a):
        delta[s + (n-1)] += -(1.0 - h[a+s])
    #   Also +h[a-s] for s = min(1,a) to a (or s=0 if a=0):
    #   Actually for s >= 0: +h[a-s] if s <= a, i.e., s = 0, 1, ..., min(a, n-1)
    #   But also need a-s >= 0, which is s <= a. And a-s <= n-1-s is always true.
    for s in range(0, min(a, n-1) + 1):
        delta[s + (n-1)] += h[a-s]
    
    # Negative shifts: s = -t where t = 1, 2, ..., min(a, n-1)
    #   -(1-h[a-t]) if t <= a: shifts s = -1, -2, ..., -min(a, n-1)
    for t in range(1, min(a, n-1) + 1):
        delta[(n-1) - t] += -(1.0 - h[a-t])
    #   +h[a+t] if a+t < n: t = 1, 2, ..., n-1-a
    for t in range(1, n - a):
        delta[(n-1) - t] += h[a+t]
    
    # Position b (0->1):
    # Positive shifts s = 0, ..., n-1-b:
    for s in range(n - b):
        delta[s + (n-1)] += (1.0 - h[b+s])
    #   -h[b-s] for s = 0, ..., min(b, n-1)
    for s in range(0, min(b, n-1) + 1):
        delta[s + (n-1)] += -h[b-s]
    
    # Negative shifts: t = 1, ..., min(b, n-1)
    for t in range(1, min(b, n-1) + 1):
        delta[(n-1) - t] += (1.0 - h[b-t])
    #   -h[b+t] for t = 1, ..., n-1-b
    for t in range(1, n - b):
        delta[(n-1) - t] += -h[b+t]
    
    return delta

Actually, I can vectorize the O(n) delta computation across all swaps simultaneously!

For a fixed position a (1→0), the delta at shift s is:
- For s ≥ 0: δ_a(s) = -(1-h[a+s])*[a+s<n] + h[a-s]*[s≤a≤n-1]
- For s < 0 (t=-s): δ_a(-t) = -(1-h[a-t])*[t≤a] + h[a+t]*[a+t<n]

This can be precomputed for all a in O(n²) total. Similarly for all b positions.

Then for each pair (a,b), the total delta is δ_a + δ_b, and we check if max(C + δ_a + δ_b) < max(C).

But checking max(C + δ_a + δ_b) for all pairs is still O(n² * n) = O(n³).

Alternatively, we can be smarter:
- The max of (C + δ_a + δ_b) is at most max(C + δ_a + δ_b) over all k.
- For the current worst shift k*, the new value is C[k*] + δ_a(k*) + δ_b(k*).
- If this is ≥ max(C), the swap might still be valid if other shifts don't exceed it.
- But if C[k*] + δ_a(k*) + δ_b(k*) < max(C) - margin, it's likely safe.

import numpy as np

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_and_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * (2.0 / len(h))), corr, idx

def _to_binary(h):
    n = len(h)
    target = n // 2
    # Sort indices by value descending, take top target
    idx = np.argsort(h)[::-1][:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def _guided_local_search(h, n, deadline, rng, use_best_improvement=False):
    """Local search on binary vectors with swap moves."""
    cur = h.copy()
    cur_val = _c5_of(cur)
    best_val = cur_val
    best_h = cur.copy()
    
    stall = 0
    max_stall = 50000
    
    while time.time() < deadline:
        if stall > max_stall:
            break
        
        corr = np.correlate(cur, 1.0 - cur, mode='full')
        worst_idx = int(np.argmax(corr))
        s_star = worst_idx - (n - 1)
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        
        if s_star >= 0:
            i_arr = np.arange(0, n - s_star)
            j_arr = i_arr + s_star
        else:
            i_arr = np.arange(-s_star, n)
            j_arr = i_arr + s_star
        
        # Conflicting: cur[i]=1 and cur[j]=0
        conflict_mask = (cur[i_arr] == 1.0) & (cur[j_arr] == 0.0)
        conflict_ones = i_arr[conflict_mask]
        
        if use_best_improvement:
            best_swap_val = cur_val
            best_a = -1
            best_b = -1
            # Try all conflicting ones with all zeros (or a subset)

def _sa_binary(h, n, deadline, rng, T_start=0.003, T_end=0.0001):
    """Simulated annealing for binary vectors."""
    cur = h.copy()
    cur_val = _c5_of(cur)
    best_val = cur_val
    best_h = cur.copy()
    
    n_iters = 200000
    it = 0
    while time.time() < deadline and it < n_iters:
        it += 1
        T = T_start * (T_end / T_start) ** (float(it) / n_iters)
        
        ones = np.where(cur == 1.0)[0]
        zeros = np.where(cur == 0.0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        cur[a] = 0.0
        cur[b] = 1.0
        new_val = _c5_of(cur)
        
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur.copy()
        else:
            cur[a] = 1.0
            cur[b] = 0.0
    
    return best_h, best_val
