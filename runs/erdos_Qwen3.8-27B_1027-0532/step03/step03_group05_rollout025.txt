def lse_gradient(h, t, n):
    """Compute gradient of log-sum-exp approximation of max of full correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    shifted = t * corr
    shifted -= np.max(shifted)  # for numerical stability
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient: sum_m weights[m] * dx * (1-2*h[i+m-(n-1)]) for valid i+m-(n-1)
    # For each m, the contribution to grad[i] is weights[m] * dx * (1-2*h[i+m-(n-1)])
    # Let j = i + m - (n-1), so i = j - m + (n-1)
    # grad[i] = dx * sum_m weights[m] * (1-2*h[i+m-(n-1)]) for valid i+m-(n-1)
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if weights[m] < 1e-15:
            continue
        # m corresponds to lag k = m - (n-1)
        k = m - (n - 1)
        # grad[i] gets contribution weights[m] * dx * (1-2*h[i+k]) for valid i+k
        # i+k valid means 0 ≤ i+k < n, i.e., -k ≤ i < n-k
        if k >= 0:
            idx = np.arange(0, n-k)
            grad[idx] += weights[m] * dx * (1.0 - 2.0 * h[idx + k])
        else:
            idx = np.arange(-k, n)
            grad[idx] += weights[m] * dx * (1.0 - 2.0 * h[idx + k])

def nesterov_phase(h, target, time_limit, t_start=10, t_end=500, n_iters=2000):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    
    best = h.copy()
    best_score = compute_c5(best)
    x = h.copy()
    y = h.copy()  # Nesterov momentum
    t = t_start
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute LSE gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        shifted = t * corr - t * np.max(corr)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        weights = exp_vals / total
        
        # Gradient
        grad = np.zeros(n)
        for m in range(2 * n - 1):
            w = weights[m]
            if w < 1e-12:
                continue
            k = m - (n - 1)
            if k >= 0:
                idx = np.arange(0, n - k)
                grad[idx] += w * dx * (1.0 - 2.0 * y[idx + k])
            else:
                idx = np.arange(-k, n)
                grad[idx] += w * dx * (1.0 - 2.0 * y[idx + k])
        
        # Nesterov step
        gamma = 2.0 / (it + 2.0)
        step = 0.01 * gamma  # learning rate
        
        x_new = project_feasible(x - step * grad, target)
        y = (1 + gamma) * x_new - gamma * x
        y = project_feasible(y, target)
        x = x_new
        
        # Check improvement
        score = compute_c5(x)
        if score < best_score:
            best_score = score
            best = x.copy()
        
        # Increase temperature
        t = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
    
    return best, best_score

    best = project_feasible(best, target)
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(np.max(corr)) * dx
    return best, best_score


def pair_cd(h, target, time_limit, seed=42, n_pairs_per_sweep=200, base_stop=0.01):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)

    best = h.copy()
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(np.max(corr)) * dx

    m_starts, m_ends, j_starts, j_ends = _prepare_indices(n)
    n_full = 2 * n - 1
    no_improve = 0
    stop = base_stop

    while time.time() - start < time_limit and stop > 1e-6:
        improved = False
        for _ in range(n_pairs_per_sweep):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            ms_i = m_starts[i]
            me_i = m_ends[i]
            js_i = j_starts[i]
            je_i = j_ends[i]
            ms_j = m_starts[j]
            me_j = m_ends[j]
            js_j = j_starts[j]
            je_j = j_ends[j]

            m_union_start = min(ms_i, ms_j)
            m_union_end = max(me_i, me_j)
            m_union = np.arange(m_union_start, m_union_end + 1)

            for sign in [1, -1]:
                d = sign * stop
                new_vi = best[i] + d
                new_vj = best[j] - d
                if new_vi < 0 or new_vi > 1 or new_vj < 0 or new_vj > 1:
                    continue

                delta_i = d * (1.0 - 2.0 * best[js_i:je_i + 1])
                delta_j = (-d) * (1.0 - 2.0 * best[js_j:je_j + 1])

                corr_i = np.zeros(n_full)
                corr_i[ms_i:me_i + 1] = delta_i
                corr_j = np.zeros(n_full)
                corr_j[ms_j:me_j + 1] = delta_j

                corr[m_union] += corr_i[m_union] + corr_j[m_union]
                ns = np.max(corr) * dx
                if ns < best_score - 1e-13:
                    best[i] = new_vi
                    best[j] = new_vj
                    best_score = ns
                    improved = True
                else:
                    corr[m_union] -= corr_i[m_union] + corr_j[m_union]
        if not improved:
            stop *= 0.7
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    best = project_feasible(best, target)
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(np.max(corr)) * dx
    return best, best_score


def nesterov_lse(h, target, time_limit, t_start=5.0, t_end=800.0, lr_init=0.05):
    n = len(h)
    dx = 2.0 / n
    start = time.time()

    best = h.copy()
    best_score = compute_c5(best)
    x = h.copy()
    y = h.copy()
    n_iters = 5000

    for it in range(n_iters):
        if time.time() - start > time_limit:
            break

        frac = it / max(n_iters - 1, 1)
        t = t_start * (t_end / t_start) ** frac

        corr = np.correlate(y, 1.0 - y, mode='full')
        cmax = np.max(corr)
        shifted = t * (corr - cmax)
        exp_vals = np.exp(shifted)
        total = np.sum(exp_vals)
        if total < 1e-300:
            break
        weights = exp_vals / total

        grad = np.zeros(n)
        for m in range(2 * n - 1):
            w = weights[m]
            if w < 1e-10:
                continue
            k = m - (n - 1)
            if k >= 0:
                idx = np.arange(0, n - k)
                grad[idx] += w * dx * (1.0 - 2.0 * y[idx + k])
            else:
                idx = np.arange(-k, n)
                grad[idx] += w * dx * (1.0 - 2.0 * y[idx + k])

        gamma = 2.0 / (it + 2.0)
        lr = lr_init * gamma / (1.0 + 0.01 * it)

        y_new = project_feasible(x - lr * grad, target)
        y_next = (1.0 + gamma) * y_new - gamma * y
        y_next = project_feasible(y_next, target)
        x = y_new
        y = y_next

        score = compute_c5(x)
        if score < best_score:
            best_score = score
            best = x.copy()

    return best, best_score


def smooth_refine(h, target, time_limit, use_pmeans=True):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    n_full = 2 * n - 1

    p_values = [10, 30, 100, 300]
    for p in p_values:
        if time.time() - start > time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = max(50, int(remaining * 2))

        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            shifted = p * (corr - cmax)
            return float(cmax + np.log(np.sum(np.exp(shifted))) / p) * dx

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def bisection_cd(h, target, time_limit, seed=42, base_stop=0.001):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    rng = np.random.RandomState(seed)

    best = h.copy()
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(np.max(corr)) * dx

    m_starts, m_ends, j_starts, j_ends = _prepare_indices(n)
    stop = base_stop
    no_improve = 0

    while time.time() - start < time_limit and stop > 1e-7:
        improved = False
        order = rng.permutation(n)
        for i in order:
            h_val = best[i]
            ms = m_starts[i]
            me = m_ends[i]
            js = j_starts[i]
            je = j_ends[i]
            m_idx = np.arange(ms, me + 1)
            for d in [stop, -stop, stop * 0.3, -stop * 0.3]:
                new_val = h_val + d
                if new_val < 0.0 or new_val > 1.0:
                    continue
                delta = d * (1.0 - 2.0 * best[js:je + 1])
                corr[m_idx] += delta
                ns = np.max(corr) * dx
                if ns < best_score - 1e-13:
                    best[i] = new_val
                    h_val = new_val
                    best_score = ns
                    improved = True
                else:
                    corr[m_idx] -= delta
        if not improved:
            stop *= 0.6
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    best = project_feasible(best, target)
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_score = float(np.max(corr)) * dx
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = project_feasible(h, target)
    best_score = compute_c5(best_h)

    # Phase 1: Fast coordinate descent with larger steps
    if tleft() > 300:
        h1, s1 = fast_cd(best_h, target, time_limit=min(280, tleft()), seed=seed,
                         steps=[0.03, 0.015, 0.008, 0.004, -0.03, -0.015, -0.008, -0.004])
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Pair coordinate descent
    if tleft() > 200:
        h2, s2 = pair_cd(best_h, target, time_limit=min(180, tleft()), seed=seed + 10,
                         n_pairs_per_sweep=500, base_stop=0.008)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Nesterov accelerated gradient on LSE
    if tleft() > 250:
        h3, s3 = nesterov_lse(best_h, target, time_limit=min(230, tleft()),
                              t_start=5.0, t_end=1000.0, lr_init=0.04)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Fast CD with medium steps
    if tleft() > 250:
        h4, s4 = fast_cd(best_h, target, time_limit=min(230, tleft()), seed=seed + 20,
                         steps=[0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001])
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Smooth SLSQP refinement
    if tleft() > 150:
        h5, s5 = smooth_refine(best_h, target, time_limit=min(130, tleft()))
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Fine fast CD
    if tleft() > 200:
        h6, s6 = fast_cd(best_h, target, time_limit=min(180, tleft()), seed=seed + 30,
                         steps=[0.004, 0.002, 0.001, 0.0005, -0.004, -0.002, -0.001, -0.0005])
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Nesterov with higher temperature range
    if tleft() > 200:
        h7, s7 = nesterov_lse(best_h, target, time_limit=min(180, tleft()),
                              t_start=50.0, t_end=3000.0, lr_init=0.02)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Ultra-fine CD
    if tleft() > 150:
        h8, s8 = bisection_cd(best_h, target, time_limit=min(130, tleft()), seed=seed + 40,
                              base_stop=0.0005)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Final pair CD
    if tleft() > 100:
        h9, s9 = pair_cd(best_h, target, time_limit=min(80, tleft()), seed=seed + 50,
                         n_pairs_per_sweep=1000, base_stop=0.003)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Phase 10: Final fine CD
    if tleft() > 60:
        h10, s10 = fast_cd(best_h, target, time_limit=max(30, tleft() - 5), seed=seed + 60,
                           steps=[0.002, 0.001, 0.0005, -0.002, -0.001, -0.0005])
        if s10 < best_score:
            best_score = s10
            best_h = h10

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
    else:
        h_final = best_h

    return h_final, best_score, n
