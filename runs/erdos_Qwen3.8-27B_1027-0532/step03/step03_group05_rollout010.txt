    return float(np.max(corr)) * dx

def project_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0

def subgradient_step(h, target, alpha, eta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    # Active set: lags where corr is close to max
    active = corr > max_corr - 0.01 * max_corr
    # Subgradient for each active lag
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        lag = k_idx - (n - 1)
        if lag >= 0:
            # f_k = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
            # df_k/dh[i] = (1-h[i+lag]) - h[i] (for i in 0..n-1-lag)
            # df_k/dh[i+lag] = -h[i] (for i in 0..n-1-lag)
            grad[:n-lag] += (1.0 - h[lag:]) - h[:n-lag]
            grad[lag:] -= h[:n-lag]
        else:
            lag_neg = -lag
            grad[lag_neg:] += (1.0 - h[:n-lag_neg]) - h[lag_neg:]
            grad[:n-lag_neg] -= h[lag_neg:]
    
    # Average over active lags
    if np.sum(active) > 0:
        grad /= np.sum(active)
    
    # Project gradient ascent (we want to minimize, so go against gradient)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h, None
    h_new = h - eta * grad / gn
    h_new = project_feasible(h_new, target)
    return h_new, compute_c5(h_new)

def subgradient_phase(h, target, time_limit, alpha_init=20.0, eta_init=0.003):
    n = len(h)
    start = time.time()
    eta = eta_init

    no_improve = 0

            no_improve = 0
            eta = min(eta * 1.02, 0.01)
        else:

            if no_improve > 50:

        h = h_new if s is not None else h
        if s is not None and s < best_score + 0.001:
            h = h_new  # accept small increases occasionally
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def soft_subgradient(h, target, temperature):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = (corr - max_corr) / temperature
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights = weights / sw
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w * (1.0 - h[lag:])
            grad[lag:] -= w * h[:n - lag]
        else:
            kl = -lag
            grad[kl:] += w * (1.0 - h[:n - kl])
            grad[:n - kl] -= w * h[kl:]
    return grad


def subgradient_phase(h, target, time_limit, temp_init=0.01, eta_init=0.004):
    n = len(h)
    start = time.time()
    eta = eta_init
    temp = temp_init
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    while time.time() - start < time_limit:
        grad = soft_subgradient(h, target, temp)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = h - eta * grad / gn
        h_new = project_feasible(h_new, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.015)
        else:
            no_improve += 1
            if no_improve > 40:
                eta *= 0.7
                temp *= 0.8
                no_improve = 0
                if eta < 1e-6:
                    break
        h = h_new
    return best, best_score


def fast_pair_cd(h, target, time_limit, seed=42, n_pairs=4, initial_step=0.01):
    n = len(h)
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    step = initial_step
    steps = [step, step * 0.5, step * 0.25]
    no_improve = 0
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if best[i] < 1e-8 or best[i] > 1.0 - 1e-8:
                js = rng.choice(n, size=n_pairs, replace=False)
            else:
                js = rng.choice(n, size=n_pairs, replace=False)
            for j in js:
                if i == j:
                    continue
                for d in steps:
                    if best[i] + d <= 1.0 and best[j] - d >= 0.0:
                        best[i] += d
                        best[j] -= d
                        s = compute_c5(best)
                        if s < best_score - 1e-15:
                            best_score = s
                            improved = True
                            break
                        best[i] -= d
                        best[j] += d
                    if best[i] - d >= 0.0 and best[j] + d <= 1.0:
                        best[i] -= d
                        best[j] += d
                        s = compute_c5(best)
                        if s < best_score - 1e-15:
                            best_score = s
                            improved = True
                            break
                        best[i] += d
                        best[j] -= d
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                steps = [s * 0.5 for s in steps]

def smoothing_trick(h, target):
    n = len(h)
    results = []
    for kw in [3, 5, 7, 9]:
        kernel = np.ones(kw) / kw
        hs = np.convolve(h, kernel, mode='same')
        hs = project_feasible(hs, target)
        s = compute_c5(hs)
        results.append((s, hs))
    best_s, best_h = min(results, key=lambda x: x[0])
    return best_h, best_s


def logsumexp_slsqp(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    if alphas is None:
        alphas = [10, 30, 100, 300]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = max(100, int(remaining * 4))

        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            cmax = np.max(corr)
            shifted = np.clip(a * (corr - cmax), -500.0, 0.0)
            return float(cmax + np.log(np.sum(np.exp(shifted))) / a)

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    return best, best_score


def block_cd(h, target, time_limit, seed=42, block_size=4):
    n = len(h)
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    step = 0.005
    no_improve = 0
    start = time.time()
    while time.time() - start < time_limit:
        improved = False
        n_blocks = n // block_size
        for _ in range(min(n_blocks, 200)):
            start_idx = rng.randint(0, n - block_size)
            end_idx = start_idx + block_size
            block = best[start_idx:end_idx].copy()
            perturbation = rng.uniform(-step, step, size=block_size)
            new_block = np.clip(block + perturbation, 0.0, 1.0)
            if np.any(perturbation != 0):
                diff = np.sum(new_block) - np.sum(block)
                if abs(diff) > 1e-10:
                    mask_rest = np.ones(n, dtype=bool)
                    mask_rest[start_idx:end_idx] = False
                    rest_mask = mask_rest & ((best > 1e-6) | (best < 1.0 - 1e-6))
                    if np.any(rest_mask):
                        best[rest_mask] += diff / np.sum(rest_mask)
                        best = np.clip(best, 0.0, 1.0)
                best[start_idx:end_idx] = new_block
                s = compute_c5(best)
                if s < best_score - 1e-15:
                    best_score = s
                    improved = True
                else:
                    best[start_idx:end_idx] = block
                    if abs(diff) > 1e-10:
                        if np.any(rest_mask):
                            best[rest_mask] -= diff / np.sum(rest_mask)
                            best = np.clip(best, 0.0, 1.0)
        if not improved:
            no_improve += 1
            if no_improve >= 4:
                step *= 0.6
                if step < 5e-6:
                    break
                no_improve = 0
    return best, best_score


def multi_resolution(h, target, time_limit, seed=42):
    n = len(h)
    n_large = n * 2
    target_large = n_large / 2.0
    h_large = np.interp(
        np.linspace(0, 2, n_large, endpoint=False),
        np.linspace(0, 2, n, endpoint=False),
        h
    )
    h_large = project_feasible(h_large, target_large)
    s_large = compute_c5(h_large)
    h_opt, s_opt = fast_pair_cd(h_large, target_large,
                                time_limit=min(60, time_limit // 2),
                                seed=seed, n_pairs=3, initial_step=0.005)
    h_back = np.interp(
        np.linspace(0, 2, n, endpoint=False),

        h_opt
    )
    h_back = project_feasible(h_back, target)
    s_back = compute_c5(h_back)
    if s_back < s_large and s_back < compute_c5(h):
        h_back, s_back = fast_pair_cd(h_back, target,
                                      time_limit=time_limit // 3,
                                      seed=seed + 1, n_pairs=3, initial_step=0.003)
    return h_back, s_back
