    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2n-1,)
    
    # Softmax
    shifted = alpha * (corr - np.max(corr))
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    obj = np.max(corr) + np.log(np.sum(exp_shifted)) / alpha
    
    # Gradient
    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-10:
            continue
        s = s_arr[k]
        # ∂g_k/∂h[j] = dx * [(1-h[j+s]) - h[j-s]] (with bounds)
        js = np.arange(n) + s
        js_m = np.arange(n) - s
        term1 = np.zeros(n)
        m1 = (js >= 0) & (js < n)
        term1[m1] = (1.0 - h[js[m1]]) * dx
        m2 = (js_m >= 0) & (js_m < n)
        term1[m2] -= h[js_m[m2]] * dx
        grad += p[k] * term1
    
    # Penalty gradient
    grad += mu * (np.sum(h) - target_sum)
    
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = h.astype(float).copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        for j in range(n):
            old = h[j]
            js = j - s_arr
            js_p = j + s_arr
            A = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n)
            A[m1] += (1.0 - h[js[m1]])
            m2 = (js_p >= 0) & (js_p < n)
            A[m2] -= h[js_p[m2]]
            B = corr - A * old
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) * 0.5
                vals = B + mid * A
                kstar = np.argmax(vals)
                grad_k = A[kstar]
                if grad_k > 1e-12:
                    hi = mid
                elif grad_k < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            scale = 0.01 * min(1.0, no_improve / patience)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            no_improve = 0
    
    return best_h, best_score

def smooth_obj_and_grad(h, alpha, mu, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = alpha * (corr - cmax)
    exp_sh = np.exp(np.clip(shifted, -500, 0))
    p = exp_sh / np.sum(exp_sh)
    obj = cmax + np.log(np.sum(exp_sh)) / alpha
    
    s_arr = np.arange(2 * n - 1) - (n - 1)
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = s_arr[k]
        if s >= 0:
            T1[:n-s] += pk * (1.0 - h[s:])
            T2[s:] += pk * h[:n-s]
        else:
            T1[-s:] += pk * (1.0 - h[:n+s])
            T2[:n+s] += pk * h[-s:]
    
    grad = dx * (T1 - T2) + mu * (np.sum(h) - target_sum)
    return obj, grad

def lbfgsb_optimize(h, target_sum, max_time, alphas, mu):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    t0 = time.time()
    
    for alpha in alphas:
        if time.time() - t0 > max_time - 10:
            break
        def obj(x):
            return smooth_obj_and_grad(x, alpha, mu, target_sum)[0]
        def grad(x):
            return smooth_obj_and_grad(x, alpha, mu, target_sum)[1]
        
        try:
            res = minimize(obj, h_cur, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    
    return best_h, best_score

def make_gaussian_bump(n, sigma=0.3):
    x = np.linspace(0, 2, n)
    h = np.exp(-0.5 * ((x - 1.0) / sigma) ** 2)
    h = h / np.sum(h) * (n / 2.0)
    return project_feasible(h, n / 2.0)

def make_piecewise_constant(n_pieces, n_grid):
    # Create a piecewise constant function with n_pieces pieces
    # Return on n_grid grid
    x = np.linspace(0, 2, n_grid)
    edges = np.linspace(0, 2, n_pieces + 1)
    # Initialize with a bump shape
    centers = (edges[:-1] + edges[1:]) / 2
    base = np.exp(-2.0 * (centers - 1.0) ** 2)
    base = base / np.sum(base) * (n_pieces / 2.0)
    h_pieces = np.clip(base, 0, 1)
    # Map to grid
    h = np.zeros(n_grid)
    for i in range(n_pieces):
        mask = (x >= edges[i]) & (x < edges[i+1])
        h[mask] = h_pieces[i]
    return project_feasible(h, n_grid / 2.0)

def smooth_obj_grad(h, alpha, mu, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(corr))
    shifted = alpha * (corr - cmax)
    exp_sh = np.exp(np.clip(shifted, -500.0, 0.0))
    p = exp_sh / np.sum(exp_sh)
    obj = cmax + np.log(np.sum(exp_sh)) / alpha
    
    s_arr = np.arange(2 * n - 1) - (n - 1)
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = s_arr[k]
        if s >= 0:
            T1[:n-s] += pk * (1.0 - h[s:])
            T2[s:] += pk * h[:n-s]
        else:
            T1[-s:] += pk * (1.0 - h[:n+s])
            T2[:n+s] += pk * h[-s:]
    
    grad = dx * (T1 - T2) + 2.0 * mu * (np.sum(h) - target_sum)
    return obj, grad

def _smooth_obj_grad(h, alpha, mu, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = float(np.max(corr))
    shifted = alpha * (corr - cmax)
    exp_sh = np.exp(np.clip(shifted, -500.0, 0.0))
    p = exp_sh / np.sum(exp_sh)
    obj = cmax + np.log(np.sum(exp_sh)) / alpha
    s_arr = np.arange(2 * n - 1) - (n - 1)
    T1 = np.zeros(n)
    T2 = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = int(s_arr[k])
        if s >= 0:
            T1[:n - s] += pk * (1.0 - h[s:])
            T2[s:] += pk * h[:n - s]
        else:
            T1[-s:] += pk * (1.0 - h[:n + s])
            T2[:n + s] += pk * h[-s:]
    grad = dx * (T1 - T2) + 2.0 * mu * (np.sum(h) - target_sum)
    return obj, grad
