    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    per = diff / len(idx)
    h[idx] += per
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def compute_softmax_c5_and_p(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max_val = float(np.max(corr))
    exponents = alpha * (corr - c_max_val)
    exponents = np.clip(exponents, -700, 0)
    p_raw = np.exp(exponents)
    p_sum = float(np.sum(p_raw))
    if p_sum < 1e-300:
        p = np.zeros_like(corr)
        p[int(np.argmax(corr))] = 1.0
        p_sum = 1.0
    else:
        p = p_raw / p_sum
    soft_c5 = (c_max_val + np.log(p_sum) / alpha) * dx
    return soft_c5, p
