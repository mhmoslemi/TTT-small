def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h

    best_score = compute_c5(h_cur)

            # We need to handle the constraint sum(h) = target and 0 <= h <= 1
            # Use a projected gradient: h_new = project(h_cur - lr * grad)
            
            grad_norm = np.linalg.norm(grad)

                break

                obj_try = smooth_obj(h_try, alpha)
                if obj_try <= obj - 1e-12 * lr * grad_norm:

                lr *= 0.5
            
            h_cur = h_try
            c5_try = compute_c5(h_cur)
            if c5_try < best_score:
                best_score = c5_try
                best_h = h_cur.copy()

        h_sa = sa_block(best_h, n, target, rng, time_limit=min(300, tleft()*0.4))
        sc = compute_c5(h_sa)
        if sc < best_score:

            best_h = h_sa
    
    # Phase 3: Another round of gradient from SA result
    if tleft() > 100:
        alphas2 = [100, 500, 2000]
        h_cur2 = best_h.copy()
        for alpha in alphas2:
            if tleft() < 50:

            h_cur2 = pgd_step(h_cur2, n, target, alpha, min(60, tleft()*0.3))
            sc = compute_c5(h_cur2)
            if sc < best_score:
                best_score = sc
                best_h = h_cur2.copy()

        n2 = 2 * n
        target2 = n2 / 2.0
        h_interp = interpolate_up(best_h, n2)

        for alpha in [50, 200, 1000]:
            if tleft() < 100:

            h_interp = pgd_step(h_interp, n2, target2, alpha, min(80, tleft()*0.3))
            sc = compute_c5(h_interp)
            if sc < best_score:
                best_score = sc
                best_h = h_interp.copy()

            best_h = h_poly

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    tot = exp_vals.sum()
    if tot < 1e-300:
        return c_max, np.zeros(n)
    obj = c_max + np.log(tot) / alpha
    w = exp_vals / tot
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if w[m] < 1e-12:
            continue
        shift = m - (n - 1)
        wm = w[m]
        if shift >= 0:
            if shift < n:
                i1 = np.arange(shift, n)
                grad[i1] += wm * (1.0 - h[i1 - shift])
                i2 = np.arange(0, n - shift)
                grad[i2] -= wm * h[i2 + shift]
        else:
            sp = -shift
            if sp < n:
                i1 = np.arange(0, n - sp)
                grad[i1] += wm * (1.0 - h[i1 + sp])
                i2 = np.arange(sp, n)
                grad[i2] -= wm * h[i2 - sp]
    return obj, grad * dx


def pgd_optimize(h, n, target, alpha, time_limit):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    best_obj, _ = smooth_obj_and_grad(h_cur, alpha)
    
    step = 0.5 / alpha
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_and_grad(h_cur, alpha)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h_cur.copy()
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        for trial in range(30):
            h_try = project_feasible(h_cur - step * grad, target)
            obj_try, _ = smooth_obj_and_grad(h_try, alpha)
            if obj_try <= obj - 1e-4 * step * gnorm:
                h_cur = h_try
                step = min(step * 1.1, 2.0 / alpha)
                break
            step *= 0.5
        else:
            if step < 1e-10:
                break
            h_try = project_feasible(h_cur - step * grad, target)
            h_cur = h_try
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    return best_h, best_score


def sa_block_optimize(h, n, target, rng, time_limit):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    current_score = best_score
    T = 0.005
    T_min = 1e-6
    cool = 0.9997
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        if T < T_min:
            break
        block_size = rng.choice([5, 10, 15, 25, 50])
        if block_size >= n:
            block_size = max(5, n // 10)
        i0 = rng.randint(0, max(1, n - block_size))
        h_try = h_cur.copy()
        noise_std = 0.1 + 0.3 * T / 0.005
        h_try[i0:i0 + block_size] += rng.randn(block_size) * noise_std
        h_try = project_feasible(h_try, target)
        sc_try = compute_c5(h_try)
        delta = sc_try - current_score
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            h_cur = h_try
            current_score = sc_try
            if sc_try < best_score:
                best_score = sc_try
                best_h = h_cur.copy()
        T *= cool
    return best_h, best_score


def interpolate_up(h_src, n_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return h_out


def slsqp_polish(h, n, target, alphas, max_iter):
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        def obj_func(x):
            o, _ = smooth_obj_and_grad(x, alpha)
            return o
        def obj_grad(x):
            _, g = smooth_obj_and_grad(x, alpha)
            return g
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                 'jac': lambda x: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(obj_func, h_cur, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': max_iter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score


def binary_snap(h, target, threshold=0.02):
    h_snap = h.copy()
    h_snap[h_snap < threshold] = 0.0
    h_snap[h_snap > 1.0 - threshold] = 1.0
    h_snap = project_feasible(h_snap, target)
    return h_snap


def run(seed=42, budget_s=1300, **kwargs):
