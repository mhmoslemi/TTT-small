    n = 2 * len(p)

    m = len(p)
    dx = 2.0 / n
    h = unpack_sym(p, n)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)

def coord_descend_sym(h, n, max_time, rng, patience=25):
    m = n // 2
    p = h[:m].copy()
    best_score = compute_c5(h)
    best_p = p.copy()
    no_improve = 0
    start = time.time()
    s_arr = np.arange(2 * n - 1) - (n - 1)

    while time.time() - start < max_time:
        h_cur = unpack_sym(p, n)
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full').copy()

        for j in range(m):
            old_val = p[j]
            nj = n - 1 - j
            A = np.zeros(2 * n - 1)

            mask1 = (s_arr >= -j) & (s_arr <= n - 1 - j)
            A[mask1] += (1.0 - h_cur[j + s_arr[mask1]])

            mask2 = (s_arr >= j - n + 1) & (s_arr <= j)
            A[mask2] -= h_cur[j - s_arr[mask2]]

            mask3 = (s_arr >= -nj) & (s_arr <= n - 1 - nj)
            A[mask3] -= (1.0 - h_cur[nj + s_arr[mask3]])

            mask4 = (s_arr >= nj - n + 1) & (s_arr <= nj)
            A[mask4] += h_cur[nj - s_arr[mask4]]

            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                if A[k_star] > 1e-12:
                    hi = mid
                elif A[k_star] < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val

                corr += delta * A
                p[j] = new_val

        h_cur = unpack_sym(p, n)
        score = compute_c5(h_cur)

            best_score = score
            best_p = p.copy()
            no_improve = 0

            no_improve += 1

            p = best_p + rng.randn(m) * noise
