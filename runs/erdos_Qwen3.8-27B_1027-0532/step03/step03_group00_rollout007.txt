2. **Primal-dual approach**:

   - Dual: max_λ min_h ∑_k λ_k f_k(h) s.t. λ ≥ 0, ∑λ = 1
   - The inner problem (for fixed λ) is: min_h ∑_k λ_k f_k(h) s.t. 0≤h≤1, ∑h=n/2
   - This is a QP (quadratic program) since f_k are quadratic in h!

   ∑_k λ_k ∑_i h_i - ∑_k λ_k ∑_i h_i h_{i+k}
   = (∑_k λ_k) ∑_i h_i - ∑_i ∑_k λ_k h_i h_{i+k}
   = ∑_i h_i - ∑_i h_i (∑_k λ_k h_{i+k})
   
   This is a quadratic function of h (specifically, it's h'Ah + b'h form where A is related to λ).
   
   Is it convex or concave? The quadratic form involves -h_i h_{i+k} terms with positive coefficients λ_k. So the Hessian has negative off-diagonal entries (for the pairs (i, i+k)). This means the quadratic form is NOT convex in general - it's indefinite.
   
   So the inner problem is a non-convex QP. This makes the primal-dual approach harder.

3. **Alternative: Semidefinite relaxation?**
   If we could relax h_i h_{i+k} → z_{i,i+k} and impose PSD constraints on the matrix Z = hh^T, we'd get an SDP. But with n=400, this would be a 400x400 PSD constraint which is too large for practical solvers in the time budget.

   - Given λ, minimize ∑_k λ_k f_k(h) using SLSQP (non-convex QP with bounds and linear constraint)
   - Update λ based on which lags have the highest f_k values
   - Repeat

    n = len(h)
    corr = compute_corr(h)
    n_corr = len(corr)
    lam = np.ones(n_corr) / n_corr  # initial uniform
    
    best_h = h.copy()
    best_score = compute_c5(best_h)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        # Update lam: exponential weights based on current f_k values
        corr = compute_corr(best_h)
        # Normalize for numerical stability
        corr_shifted = corr - np.max(corr)
        lam = np.exp(corr_shifted * 5.0)  # temperature 1/5
        lam /= lam.sum()
        
        # Given lam, minimize sum_k lam_k * f_k(h)
        def obj(x):
            x = project_feasible(x, target)
            c = compute_corr(x)
            return float(np.dot(lam, c))
        
        res = minimize(obj, best_h, method='SLSQP',
                       bounds=[(0,1)]*n,
                       options={'maxiter': 1000, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        
        if sc < best_score:
            best_score = sc
            best_h = h_new
    
    return best_h, best_score

    G = np.zeros(n)
    # For each correlation index idx = 0..2n-2:
    # lag k = idx - (n-1)
    # If k >= 0:
    #   G[j] += lam[idx] * dx * (1-h[j-k]) for j in [k, n-1]
    #   G[j] -= lam[idx] * dx * h[j+k] for j in [0, n-1-k]

    #   G[j] += lam[idx] * dx * (1-h[j+k']) for j in [0, n-1-k']
    #   G[j] -= lam[idx] * dx * h[j-k'] for j in [k', n-1]

        if w < 1e-15:
            continue
        if k >= 0:

            G[k:] += w * (1.0 - h[:n-k])
            # G[j] -= w * h[j+k] for j in [0, n-1-k]

            # G[j] -= w * h[j-kp] for j in [kp, n-1]

from scipy.optimize import minimize
import time

_g_n = 0
_g_target = 0.0
_g_alpha = 0.0
_g_block_full = None
_g_block_start = 0
_g_block_end = 0
_g_lam = None
_g_dx = 0.0


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_max(corr, alpha):
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def _weighted_obj(x):
    x = project_feasible(x, _g_target)
    c = compute_corr(x)
    return float(np.dot(_g_lam, c))


def _weighted_grad(x):
    x = project_feasible(x, _g_target)
    n = _g_n
    dx = _g_dx
    G = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        w = _g_lam[idx] * dx
        if w < 1e-18:

        else:

            G[:n - kp] += w * (1.0 - x[kp:])
            G[kp:] -= w * x[:n - kp]
    return G

    e = _g_block_end

def _block_descent(h, block_size, n_rounds, alpha, deadline):
    global _g_block_full, _g_block_start, _g_block_end, _g_target, _g_alpha
    _g_target = len(h) / 2.0
    _g_alpha = alpha
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_block_full = h_cur
            _g_block_start = s

def _multiplicative_weights(h, target, n_iter, deadline, temp0=3.0, temp1=15.0):

    n_corr = 2 * n - 1

        lam = np.exp(corr_shifted * temp)
        lam_sum = lam.sum()

    best_h = h.copy()
    best_score = compute_c5(best_h)
    v = np.zeros(n)
    lr = lr0

    for it in range(n_steps):
        if time.time() > deadline:
            break
        h_curr = project_feasible(best_h + v, target)
        corr = compute_corr(h_curr)
        c_max = float(np.max(corr))
        active = np.where(corr >= c_max * 0.995)[0]

        G = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                G[k:] += (1.0 - h_curr[:n - k])
                G[:n - k] -= h_curr[k:]
            else:
                kp = -k
                G[:n - kp] += (1.0 - h_curr[kp:])
                G[kp:] -= h_curr[:n - kp]
        G *= dx / len(active)

        lr = lr0 / (1.0 + 0.01 * it)
        v = momentum * v - lr * G
        h_new = project_feasible(best_h + v, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        else:
            v *= 0.5

    return best_h, best_score


def _pair_block_descent(h, target, deadline, alpha=500.0):
    """Optimize pairs of adjacent blocks simultaneously for more flexibility"""
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    pair_size = 4
    n_pairs = n // (2 * pair_size)
    for rd in range(4):
        if time.time() > deadline:
            break
        for bi in range(n_pairs):
            if time.time() > deadline:
                break
            s1 = bi * 2 * pair_size
            e1 = s1 + pair_size
            s2 = e1
            e2 = e1 + pair_size

            def _pair_obj(x):
                hf = _g_block_full.copy()
                hf[s1:e1] = np.clip(x[:pair_size], 0.0, 1.0)
                hf[s2:e2] = np.clip(x[pair_size:], 0.0, 1.0)
                hf = project_feasible(hf, _g_target)
                return lse_max(compute_corr(hf), alpha)

            global _g_block_full, _g_target
            _g_block_full = h_cur
            _g_target = target
            x0 = np.concatenate([h_cur[s1:e1], h_cur[s2:e2]])
            try:
                res = minimize(_pair_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s1:e1] = np.clip(res.x[:pair_size], 0.0, 1.0)
                h_test[s2:e2] = np.clip(res.x[pair_size:], 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    cur_score = sc
                    h_cur = h_test
            except Exception:
                pass
    return h_cur, cur_score


def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target, _g_alpha, _g_block_full, _g_block_start, _g_block_end, _g_lam, _g_dx
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Quick SLSQP cleanup with moderate alphas
    for alpha in [100, 500, 2000]:
        if tleft() < 60:
            break
        _g_n = n
        _g_target = target
        _g_alpha = alpha
        _g_dx = dx
        bounds = [(0.0, 1.0)] * n
        res = minimize(_lse_obj, best_h, method='SLSQP', bounds=bounds,
                       options={'maxiter': 1500, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 2: Multiplicative weights (key innovation)
    deadline_p2 = start_time + budget_s * 0.35
    h_mw, sc_mw = _multiplicative_weights(best_h, target, 25, deadline_p2,
                                           temp0=3.0, temp1=12.0)
    if sc_mw < best_score:
        best_score = sc_mw
        best_h = h_mw

    # Phase 3: Nesterov subgradient
    if tleft() > 60:
        deadline_p3 = start_time + budget_s * 0.45
        h_ns, sc_ns = _nesterov_subgrad(best_h, target, 0.02, 0.85, 150, deadline_p3)
        if sc_ns < best_score:
            best_score = sc_ns
            best_h = h_ns

    # Phase 4: Block coordinate descent with varying block sizes
    deadline_p4 = start_time + budget_s * 0.65
    for bs in [4, 3, 5]:
        if tleft() < 80:
            break
        h_bd, sc_bd = _block_descent(best_h, bs, 6, 1000.0, deadline_p4)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Phase 5: Pair block descent
    if tleft() > 80:
        deadline_p5 = start_time + budget_s * 0.75
        h_pb, sc_pb = _pair_block_descent(best_h, target, deadline_p5, alpha=2000.0)
        if sc_pb < best_score:
            best_score = sc_pb
            best_h = h_pb

    # Phase 6: Another round of multiplicative weights (finer)
    if tleft() > 100:
        deadline_p6 = start_time + budget_s * 0.82
        h_mw2, sc_mw2 = _multiplicative_weights(best_h, target, 15, deadline_p6,
                                                 temp0=8.0, temp1=25.0)
        if sc_mw2 < best_score:
            best_score = sc_pb
            best_h = h_mw2

    # Phase 7: Ultra-fine block descent (blocks of size 2-3)
    if tleft() > 60:
        deadline_p7 = start_time + budget_s * 0.92
        for bs in [2, 3]:
            if tleft() < 30:
                break
            h_bd, sc_bd = _block_descent(best_h, bs, 8, 5000.0, deadline_p7)
            if sc_bd < best_score:
                best_score = sc_bd
                best_h = h_bd

    # Phase 8: Final SLSQP with very high alpha
    if tleft() > 25:
        for alpha in [10000, 30000]:
            if tleft() < 12:
                break
            _g_n = n
            _g_target = target
            _g_alpha = alpha
            bounds = [(0.0, 1.0)] * n
            res = minimize(_lse_obj, best_h, method='SLSQP', bounds=bounds,
                           options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 9: Final single-element polish
    if tleft() > 15:
        n = len(best_h)
        h_fin = best_h.copy()
        sc_fin = best_score
        # Sweep through elements, try small perturbations
        for i in range(n):
            if tleft() < 3:
                break
            for delta in [0.001, -0.001, 0.005, -0.005]:
                h_try = h_fin.copy()
                h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
                h_try = project_feasible(h_try, target)
                sc_try = compute_c5(h_try)
                if sc_try < sc_fin:
                    sc_fin = sc_try
                    h_fin = h_try

        if sc_fin < best_score:
            best_score = sc_fin
            best_h = h_fin

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, n
