    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient(h, top_k=1):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top_k indices
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in top_idx:
        lag = m - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[:n-lag] += (1 - h[lag:n])
            if lag < n:
                grad[lag:] -= h[:n-lag]
        else:
            l = -lag
            if l < n:
                grad[l:] += (1 - h[:n-l])
            if n - 1 - l >= 0:
                grad[:n-l] -= h[l:]
    grad /= top_k
    return grad, corr

def subgradient_descent(h, target, max_iter, step_init, time_fn):
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)  # momentum
    mu = 0.9  # momentum coefficient
    step = step_init
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    top_k = 5
    
    for it in range(max_iter):
        if time_fn() < 30:
            break
        g, corr = subgradient(h_cur, top_k)
        # Nesterov momentum
        v = mu * v - step * g
        h_new = h_cur + v
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            h_cur = h_new
        else:
            # Reduce step size
            step *= 0.7
            v *= 0.5
            # Revert
            h_new = h_cur
            # Reapply
            h_new = h_cur - step * g

            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                h_cur = h_new
            else:
                step *= 0.5
        # Adapt top_k
        if it % 100 == 0 and it > 0:

            idx = np.where(h > 0.0)[0]

        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)

def subgradient_vectorized(h, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        if lag >= 0:
            if n - 1 - lag >= 0:
                grad[:n - lag] += (1.0 - h[lag:n])
            if lag < n:
                grad[lag:] -= h[:n - lag]
        else:
            l = -lag
            if l < n:
                grad[l:] += (1.0 - h[:n - l])
            if n - 1 - l >= 0:
                grad[:n - l] -= h[l:]
    grad /= float(top_k)
    return grad, corr


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        lag_lo = -(n - 1)
        lag_hi = n - 1
        m_lo = max(0, n - 1 - (n - 1 - i))
        m_hi = min(2 * n - 2, n - 1 + i)
        valid_m = np.arange(m_lo, m_hi + 1)
        lags = valid_m - (n - 1)
        first_idx = i + lags
        first_valid = (first_idx >= 0) & (first_idx < n)
        second_idx = i - lags
        second_valid = (second_idx >= 0) & (second_idx < n)
        term1 = np.sum(w[valid_m[first_valid]] * (1.0 - h[i + lags[first_valid]]))
        term2 = np.sum(w[valid_m[second_valid]] * h[i - lags[second_valid]])
        grad[i] = term1 - term2
    return F, grad


def interpolate_to(h_src, n_dst, target_dst):

    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def lbfgs_stage(h_start, n, target, alpha, maxiter, ftol):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):

        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

            h_pert = best_h.copy()
            mask = rng.random(n_init) < 0.1
            h_pert[mask] = np.clip(h_pert[mask] + rng.randn(np.sum(mask)) * 0.02, 0, 1)
            h_pert = project_feasible(h_pert, target_init)

        h_new, sc = run_full_refine(h_pert, n_init, target_init,
                                    [50, 200, 600], 150, 1e-13, tleft)
        if tleft() > 30:
            h_sg, sc2 = subgradient_momentum(h_new, target_init, 2000, tleft,
                                             step_init=0.005, mu=0.88)
            if sc2 < sc:
                sc = sc2
                h_new = h_sg
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Grid refinement to higher resolution
    for n_try in [600, 800, 1000]:
        if tleft() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_refined, sc = run_full_refine(h_interp, n_try, target_try,
                                        [10, 40, 150, 500], 200, 1e-14, tleft)
        if tleft() > 50:
            h_sg, sc2 = subgradient_momentum(h_refined, target_try, 3000, tleft,
                                             step_init=0.005, mu=0.88)
            if sc2 < sc:
                sc = sc2
                h_refined = h_sg
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 7: Coordinate descent polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 80:
        h_cd, sc = coordinate_descent(best_h, target_use, 10, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 8: Final L-BFGS-B with high alpha
    if tleft() > 50:
        h_final2, sc = run_full_refine(best_h, n_use, target_use,
                                       [1000, 3000, 5000], 300, 1e-15, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_final2

    # Phase 9: Final subgradient
    if tleft() > 40:
        h_sg_final, sc = subgradient_momentum(best_h, target_use, 2000, tleft,
                                              step_init=0.002, mu=0.92)
        if sc < best_score:
            best_score = sc
            best_h = h_sg_final

    # Phase 10: Tiny perturbations
    for _ in range(8):
        if tleft() < 30:
            break
        amp = rng.uniform(0.001, 0.003)
        h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
        h_new, sc = run_full_refine(h_pert, n_use, target_use,
                                    [2000, 5000], 150, 1e-15, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
