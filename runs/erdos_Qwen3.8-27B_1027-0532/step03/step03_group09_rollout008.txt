def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def proj_grad_topk(h, k, temp, lr_init, n_steps, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        grad = topk_subgrad(h_cur, k, temp)
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.02, 0.3)
        else:
            stall += 1
            lr = max(lr * 0.95, 1e-7)
        
        h_cur = h_new
        if stall > 500:
            h_cur = best_h.copy()
            lr = lr_init * 0.7
            stall = 0
    
    return best_h, best_c5

import numpy as np

import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def binary_c5(h_bin):
    n = len(h_bin)
    dx = 2.0 / n
    hf = h_bin.astype(np.float64)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    return float(dx * np.max(corr))

def repair_binary_count(h, n_target_ones, seed=0):
    n = len(h)
    h = (h >= 0.5).astype(np.int8)
    c = int(h.sum())
    if c == n_target_ones:
        return h
    if c > n_target_ones:
        idx = np.where(h == 1)[0]
        r = np.random.default_rng(seed)
        r.shuffle(idx)
        h[idx[:c - n_target_ones]] = 0
    else:
        idx = np.where(h == 0)[0]
        r = np.random.default_rng(seed + 7)
        r.shuffle(idx)
        h[idx[:n_target_ones - c]] = 1
    return h

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_obj(x, eps):
    return smooth_max_obj(x, eps)

def slsqp_jac(x, eps):
    return smooth_max_grad(x, eps)

def cfun_sum(x, target_sum):
    return float(np.sum(x) - target_sum)

def cjac_sum(x, target_sum):
    return np.ones(len(x))

                          constraints=[{'type': 'eq', 'fun': cfun_sum, 'args': (target_sum,), 'jac': cjac_sum}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def proj_grad_topk(h, k, temp, lr_init, n_steps, start_time, time_limit_s):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    deadline = start_time + time_limit_s

    for step in range(n_steps):
        if time.time() > deadline:
            break
        grad = topk_subgrad(h_cur, k, temp)
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.015, 0.25)
        else:
            stall += 1
            lr = max(lr * 0.97, 1e-7)
        h_cur = h_new
        if stall > 400:
            h_cur = best_h.copy()
            lr = lr_init * 0.6
            stall = 0
    return best_h, best_c5

def dequantize(h_bin, noise_scale, rng, target_sum):
    h = h_bin.astype(float)
    h = h + noise_scale * rng.standard_normal(len(h))
    h = project_onto_constraint(h, target_sum)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    n_target_ones = n0 // 2
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Binary SA from multiple starts
    starts = []
    starts.append(repair_binary_count(h0, n_target_ones, seed=1))
    starts.append(repair_binary_count(0.3 + 0.4 * h0, n_target_ones, seed=2))
    starts.append(repair_binary_count(0.6 + 0.3 * h0, n_target_ones, seed=3))
    h_rand = rng.integers(0, 2, n0).astype(np.int8)
    starts.append(repair_binary_count(h_rand, n_target_ones, seed=4))

    best_bin_h = None
    best_bin_c5 = float('inf')
    sa_budget_each = max(40, int((budget_s * 0.30) / len(starts)))

    for i, sb in enumerate(starts):
        if rem() < 80:
            break
        tl = min(sa_budget_each, rem() - 40)
        h_b, c5_b = sa_binary(sb, 0.4, 0.0005, tl, start_time, seed=seed + 200 * i, patience=200000)
        if c5_b < best_bin_c5:
            best_bin_c5 = c5_b
            best_bin_h = h_b.copy()

    # Binary polish
    if best_bin_h is not None and rem() > 50:
        tl = min(25, rem() - 30)
        best_bin_h, best_bin_c5 = binary_polish(best_bin_h, 30, rng, start_time, tl)

    # Phase 2: Dequantize + continuous optimization from binary
    if best_bin_h is not None:
        h_deq = dequantize(best_bin_h, 0.04, rng, target_sum)
        if rem() > 80:
            h_opt, c5_opt = slsqp_optimize(h_deq, 0.01, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        if rem() > 50:
            h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 3: Also optimize the original continuous start
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(h0, 0.01, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if rem() > 40:
            h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Projected gradient
    if rem() > 50:
        tl = min(rem() - 30, 180)
        h_pg, c5_pg = proj_grad_topk(best_h, 3, 0.01, 0.08, 500000, start_time, tl)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 5: Binary SA from continuous result
    if rem() > 150:
        h_bin2 = repair_binary_count(best_h, n_target_ones, seed=10)
        tl = min(rem() - 80, 180)
        h_b, c5_b = sa_binary(h_bin2, 0.25, 0.0003, tl, start_time, seed=seed + 500, patience=150000)
        if c5_b < best_bin_c5:
            best_bin_c5 = c5_b
            best_bin_h = h_b.copy()

        if best_bin_h is not None and rem() > 40:
            h_deq2 = dequantize(best_bin_h, 0.03, rng, target_sum)
            h_opt, c5_opt = slsqp_optimize(h_deq2, 0.005, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Fine SLSQP
    if rem() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Final gradient polish
    if rem() > 25:
        tl = min(rem() - 10, 80)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.03, 300000, start_time, tl)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 8: Resolution increase
    if rem() > 160:
        n1 = n0 + 60
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_bin_up = repair_binary_count(h_up, n1 // 2, seed=20)
        tl_sa = min(50, rem() - 100)
        h_b_up, c5_b_up = sa_binary(h_bin_up, 0.2, 0.001, tl_sa, start_time, seed=seed + 800, patience=100000)

        h_up2 = dequantize(h_b_up, 0.03, rng, n1 / 2.0)
        if rem() > 60:
            h_opt_up, c5_opt_up = slsqp_optimize(h_up2, 0.008, 200)
            if rem() > 30:
                h_opt_up, c5_opt_up = slsqp_optimize(h_opt_up, 0.002, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
                n0 = n1
                target_sum = n0 / 2.0

    # Phase 9: Very fine final SLSQP
    if rem() > 25:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
