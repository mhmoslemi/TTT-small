    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5_of(h)
    s_vals = np.arange(-(n - 1), n)
    for _round in range(n_rounds):
        improved = False
        for j in range(n):
            old_j = h[j]
            j_p = j + s_vals
            j_m = j - s_vals
            valid_p = (j_p >= 0) & (j_p < n)
            valid_m = (j_m >= 0) & (j_m < n)
            A = np.zeros(2 * n - 1)
            A[valid_p] = 1.0 - h[j_p[valid_p]]
            A[valid_m] += -h[j_m[valid_m]]
            corr_old = np.correlate(h, 1.0 - h, mode='full')
            grid = np.linspace(0.0, 1.0, n_evals)
            grid = np.unique(np.concatenate([grid, [old_j]]))
            best_c = best_c5
            best_x = old_j
            for x in grid:
                if abs(x - old_j) < 1e-15:
                    continue
                corr_new = corr_old + (x - old_j) * A
                c = float(np.max(corr_new) * dx)
                if c < best_c:
                    best_c = c
                    best_x = x
            h[j] = best_x
            if best_x != old_j:
                improved = True
                if best_c < best_c5:
                    best_c5 = best_c
        h = _fix_sum(h, target_sum)
        if not improved:
            break
    return h, _c5_of(h)

def _coord_fine(h, target_sum, n_rounds=20):
    n = len(h)
    h = h.copy()
    best_c5 = _c5_of(h)
    for _round in range(n_rounds):
        improved = False
        for j in range(n):
            old_j = h[j]
            candidates = np.unique(np.concatenate([
                [0.0, 1.0, old_j],
                old_j + np.array([-0.15, -0.1, -0.05, 0.05, 0.1, 0.15])
            ]))
            candidates = candidates[(candidates >= 0.0) & (candidates <= 1.0)]
            if len(candidates) < 2:
                continue
            best_c = best_c5
            best_x = old_j
            for x in candidates:
                if abs(x - old_j) < 1e-14:
                    continue
                h[j] = x
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_x = x
            h[j] = best_x
            if best_x != old_j:
                improved = True
                if best_c < best_c5:
                    best_c5 = best_c
        h = _fix_sum(h, target_sum)
        if not improved:
            break
    return h, _c5_of(h)

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_subgrad(h0, n, target_sum, n_iters, alpha0, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            m = n - s
            if m > 0:
                g[s:s+m] += (1.0 - y[:m])
                g[:m] += -y[s:s+m]
        else:
            j = -s
            m = n - j
            if m > 0:
                g[:m] += (1.0 - y[j:])
                g[j:] += -y[:m]
        g *= dx
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.4 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.9 - 0.5 * t_frac
        cr = 0.4 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 12:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start0 = time.time()
    deadline = t_start0 + budget_s - 20

    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 200
        h_init = np.full(n_init, 0.5)

    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_candidates = []
    base_n = n_init
    for delta in [0, 20, -20, 40, -40, 60, -60, 80, -80, 100, 140, 180]:
        n_try = base_n + delta
        if 100 <= n_try <= 500 and n_try % 2 == 0:
            n_candidates.append(n_try)
    seen = set()
    n_candidates_u = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_candidates_u.append(nn)
    n_candidates = n_candidates_u

    for n in n_candidates:
        if time.time() > deadline - 120:
            break
        target_sum = n / 2.0

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        # Quick DE for diversity
        pop_size = min(12, max(8, n // 10))
        n_gen = min(60, max(20, int((deadline - time.time() - 150) / max(1, pop_size * 0.03))))
        h_de, c5_de = _de_optimize(h_start, n, target_sum, pop_size, n_gen,
                                    deadline - 150, rng)
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n

        # Analytic coordinate descent (key innovation)
        if time.time() < deadline - 80:
            h_ac, c5_ac = _coord_analytic(h_de, target_sum, n_rounds=25, n_evals=20)
            if c5_ac < best_c5:
                best_c5 = c5_ac
                best_h = h_ac.copy()
                best_n = n

        # Nesterov subgradient
        if time.time() < deadline - 60:
            h_ns, c5_ns = _nesterov_subgrad(h_ac, n, target_sum, n_iters=200, alpha0=0.4,
                                             deadline=deadline - 60)
            if c5_ns < best_c5:
                best_c5 = c5_ns
                best_h = h_ns.copy()
                best_n = n

        # SLSQP with moderate alpha
        if time.time() < deadline - 40:
            h_sl = _slsqp_opt(best_h if best_n == n else h_ns, n, target_sum, 500, maxiter=300)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n

        # More analytic CD
        if time.time() < deadline - 30:
            h_ac2, c5_ac2 = _coord_analytic(h_sl, target_sum, n_rounds=20, n_evals=25)
            if c5_ac2 < best_c5:
                best_c5 = c5_ac2
                best_h = h_ac2.copy()
                best_n = n

    # Intensive refinement on the best solution found
    n_best = best_n
    target = n_best / 2.0
    h_curr = best_h.copy()

    # Phase: iterative CD + SLSQP cycles
    for cycle in range(6):
        if time.time() > deadline - 30:
            break

        h_cd, c5_cd = _coord_analytic(h_curr, target, n_rounds=30, n_evals=20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n_best
            h_curr = h_cd.copy()

        alpha = 200 * (2 ** cycle)
        if time.time() > deadline - 25:
            break
        h_sl = _slsqp_opt(h_curr, n_best, target, min(alpha, 8000), maxiter=400)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n_best
            h_curr = h_sl.copy()

        h_cf, c5_cf = _coord_fine(h_curr, target, n_rounds=15)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf.copy()
            best_n = n_best
            h_curr = h_cf.copy()

    # Final aggressive polishing
    if time.time() < deadline - 15:
        h_cd_f, c5_cd_f = _coord_analytic(best_h, target, n_rounds=40, n_evals=30)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 10:
        h_sl_f = _slsqp_opt(best_h, best_n, target, 15000, maxiter=500)
        c5_sl_f = _c5_of(h_sl_f)
        if c5_sl_f < best_c5:
            best_c5 = c5_sl_f
            best_h = h_sl_f.copy()

    if time.time() < deadline - 5:
        h_cd_final, c5_cd_final = _coord_analytic(best_h, target, n_rounds=30, n_evals=30)
        if c5_cd_final < best_c5:
            best_c5 = c5_cd_final
            best_h = h_cd_final.copy()

    if time.time() < deadline - 3:
        h_cf_final, c5_cf_final = _coord_fine(best_h, target, n_rounds=20)
        if c5_cf_final < best_c5:
            best_c5 = c5_cf_final
            best_h = h_cf_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
