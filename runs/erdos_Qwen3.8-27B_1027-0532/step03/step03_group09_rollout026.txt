    old_t = np.linspace(0.0, 1.0, n_old)
    new_t = np.linspace(0.0, 1.0, n_new)
    return np.interp(new_t, old_t, h)

def run(seed=42, budget_s=1300, **kwargs):
    global _target_sum_global
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    def rem():
        return budget_s - (time.time() - start_time)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Quick SLSQP with moderate smoothing
    h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 350)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt.copy()

    # Phase 2: Momentum projected gradient (top-3 lags)
    if rem() > 180:
        tl = min(rem() - 100, 180)
        h_pg, c5_pg = proj_grad_momentum(best_h, 3, 0.01, 0.06, 200000, tl, start_time, momentum=0.9)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Fine SLSQP
    if rem() > 120:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Very fine SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Single-worst-lag momentum gradient (cooler)
    if rem() > 60:
        tl = min(rem() - 30, 120)
        h_pg, c5_pg = proj_grad_momentum(best_h, 1, 0.001, 0.04, 200000, tl, start_time, momentum=0.85)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 6: Restart loop with diverse perturbation strategies
    eps_cycle = [0.002, 0.001, 0.0005, 0.0003, 0.0001]
    n_restart = 0
    max_restarts = 25
    while rem() > 140 and n_restart < max_restarts:
        strategy = n_restart % 6
        h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)
        eps_use = eps_cycle[min(n_restart // 5, len(eps_cycle) - 1)]

        h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if rem() > 80:
            tl_pg = min(rem() - 50, 35)
            h_pg, c5_pg = proj_grad_momentum(best_h, 2, 0.005, 0.03, 60000, tl_pg, start_time, momentum=0.9)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        n_restart += 1

    # Phase 7: Ultra-fine SLSQP on best solution
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final momentum gradient polish (very cold)
    if rem() > 40:
        tl = min(rem() - 20, 60)
        h_pg, c5_pg = proj_grad_momentum(best_h, 1, 0.0005, 0.02, 150000, tl, start_time, momentum=0.92)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 9: Resolution increase to n=240
    if rem() > 200:
        n1 = n0 + 60
        h_up = upsample_h(best_h, n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.004 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        tl_up = min(rem() - 120, 70)
        h_pg_up, c5_up = proj_grad_momentum(h_up, 3, 0.01, 0.05, 80000, tl_up, start_time, momentum=0.9)

        if rem() > 100:
            tl_up2 = min(rem() - 60, 50)
            h_pg_up2, c5_up2 = proj_grad_momentum(h_pg_up, 1, 0.003, 0.03, 80000, tl_up2, start_time, momentum=0.9)
            if c5_up2 < c5_up:
                c5_up = c5_up2
                h_pg_up = h_pg_up2

        if rem() > 50:
            tl_up3 = min(rem() - 30, 30)
            h_pg_up3, c5_up3 = proj_grad_momentum(h_pg_up, 1, 0.001, 0.02, 60000, tl_up3, start_time, momentum=0.95)
            if c5_up3 < c5_up:
                c5_up = c5_up3
                h_pg_up = h_pg_up3

        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_pg_up.copy()

    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
