def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)

            m = n - s

            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):

                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)

                        max_after = -1e9

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def build_F(h):
    n = len(h)
    half = n - 1
    F = np.zeros((n, 2 * n - 1))
    for p in range(n):
        s_lo = max(-p, p - half)
        s_hi = min(half - p, p)
        for s in range(s_lo, s_hi + 1):
            F[p, s + half] = h[p + s] + h[p - s]
    return F

def update_F_row(F, h, p):
    n = len(h)
    half = n - 1
    F[p, :] = 0.0
    s_lo = max(-p, p - half)
    s_hi = min(half - p, p)
    for s in range(s_lo, s_hi + 1):
        F[p, s + half] = h[p + s] + h[p - s]

def fast_exhaustive_ls(h_bin, max_passes=2000, rng=None):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    F = build_F(h)
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    
    for p in range(max_passes):
        best_gain = 0.0
        best_move = None
        for i in ones_list:
            for j in zeros_list:
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                k_lo = s_min + half
                k_hi = s_max + half
                delta = F[i, k_lo:k_hi+1] - F[j, k_lo:k_hi+1]
                new_vals = corr[k_lo:k_hi+1] + delta
                max_changed = float(np.max(new_vals))
                if k_lo > 0:
                    max_before = float(np.max(corr[:k_lo]))
                else:
                    max_before = -1.0
                if k_hi < 2 * n - 2:
                    max_after = float(np.max(corr[k_hi+1:]))
                else:
                    max_after = -1.0
                max_unchanged = max(max_before, max_after)
                new_max = max(max_changed, max_unchanged)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j, k_lo, k_hi, delta.copy())
        
        if best_move is None:
            break
        i, j, k_lo, k_hi, delta = best_move
        corr[k_lo:k_hi+1] += delta
        h[i] = 0.0
        h[j] = 1.0
        current_max -= best_gain
        update_F_row(F, h, i)
        update_F_row(F, h, j)
        ones_list.remove(i)
        zeros_list.remove(j)
        ones_list.append(j)
        zeros_list.append(i)
    
    return h, current_max

def fast_binary_sa_v2(h_start, n_steps, t_start, t_end, rng, directed_frac=0.4):

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def build_F(h):
    n = len(h)
    half = n - 1
    F = np.zeros((n, 2 * n - 1))
    for p in range(n):
        s_lo = max(-p, p - half)
        s_hi = min(half - p, p)
        for s in range(s_lo, s_hi + 1):
            F[p, s + half] = h[p + s] + h[p - s]
    return F

    if ls_time > 20:
        h_ls, ls_max = fast_exhaustive_ls(to_binary(best_h), time_limit=ls_time)
        c5_ls = (2.0 / n0) * ls_max
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    n_rounds = 0
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 80:
            break
        n_rounds += 1

        strategy = n_rounds % 6
        if strategy == 0:
            h_start = to_binary(best_h)
        elif strategy == 1:
            h_pert = best_h + 0.04 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, target_sum)
            h_start = to_binary(h_pert)
        elif strategy == 2:
            h_start = to_binary(best_h).copy()
            n_flip = max(3, n0 // 12)
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                extra = int(s - target_sum)
                h_start[ones[:extra]] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(target_sum - s)
                h_start[zeros[:need]] = 1
        elif strategy == 3:
            alpha = rng.uniform(0.5, 0.85)
            h_rand = (rng.random(n0) < 0.5).astype(float)
            h_start = to_binary(alpha * best_h + (1.0 - alpha) * h_rand)
        elif strategy == 4:
            freq = rng.integers(3, n0 // 3 + 1)
            x = np.arange(n0) * (2.0 / n0)
            phase = rng.uniform(0, 2 * np.pi)
            h_struct = 0.5 + 0.4 * np.sin(np.pi * freq * x + phase)
            h_struct += 0.1 * np.sin(2 * np.pi * x + phase * 0.7)
            h_start = to_binary(h_struct)
        else:
            h_start = to_binary(best_h).copy()
            n_flip = max(5, n0 // 8)
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                extra = int(s - target_sum)
                h_start[ones[:extra]] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(target_sum - s)
                h_start[zeros[:need]] = 1

        t_start = max(0.08, 0.25 - 0.008 * n_rounds)
        t_end = 0.002
        n_steps = min(int(remaining * 0.45 * 5000), 1200000)
        dfrac = 0.3 + 0.2 * rng.random()

        h_bin_tmp, bin_max = fast_binary_sa_v3(h_start, n_steps, t_start, t_end, rng, directed_frac=dfrac)
        c5_bin = (2.0 / n0) * bin_max

        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_tmp.copy()
            elapsed = time.time() - start_time
            ls_remaining = min(budget_s - elapsed - 60, 50)
            if ls_remaining > 15:
                h_ls, ls_max = fast_exhaustive_ls(best_h, time_limit=ls_remaining)
                c5_ls = (2.0 / n0) * ls_max
                if c5_ls < best_c5:
                    best_c5 = c5_ls
                    best_h = h_ls.copy()

        elapsed = time.time() - start_time
        if budget_s - elapsed > 60 and (c5_bin < best_c5 + 0.0005):
            h_polish = best_h + 0.008 * rng.standard_normal(n0)
            h_polish = project_onto_constraint(h_polish, target_sum)
            h_polish, c5 = slsqp_optimize(h_polish, 0.0003, 150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()
                h_bin2 = to_binary(best_h)
                c5_b2 = compute_c5(h_bin2)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_bin2.copy()

    elapsed = time.time() - start_time
    ls_final_time = min(budget_s - elapsed - 20, 40)
    if ls_final_time > 10:
        h_final_ls, ls_max = fast_exhaustive_ls(to_binary(best_h), time_limit=ls_final_time)
        c5_final = (2.0 / n0) * ls_max
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_ls.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        h_polish = best_h + 0.002 * rng.standard_normal(n0)
        h_polish = project_onto_constraint(h_polish, target_sum)
        for eps in [0.00005, 0.00002]:
            elapsed = time.time() - start_time
            if budget_s - eligible < 10 if False else budget_s - elapsed < 10:
                break
            h_polish, c5 = slsqp_optimize(h_polish, eps, 150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()
        hb = to_binary(best_h)
        c5_b = compute_c5(hb)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = hb.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h.copy()
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
