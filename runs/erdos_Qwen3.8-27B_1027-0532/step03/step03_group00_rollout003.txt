        sc_up = compute_c5(h_up)
        if sc_up < best_score:
            best_score = sc_up
            best_h = h_up

    # Phase 8: Final binarization + polish
    if tleft() > 80:
        h_binq = binarize_push(best_h, target, strength=0.5)
        h_ref = slsqp_round(h_binq, n, target, [200, 1000, 3000], 2000, 1e-16,
                            start_time + budget_s - 40)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 9: Final SLSQP polish
    if tleft() > 30:
        h_final = slsqp_round(best_h, n, target, [1000, 5000, 10000], 2000, 1e-17,
                              start_time + budget_s - 10)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 10: Final subgradient micro-steps
    if tleft() > 20:
        h_sg = subgradient_descent(best_h, target, start_time + budget_s - 5,
                                    top_m=3, lr_init=0.005, patience=30)
        sc = compute_c5(h_sg)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    best_h = project_feasible(best_h, target)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
