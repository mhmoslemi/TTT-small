def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def kronecker_binary(n, target_sum, alpha):
    """Generate binary sequence using Kronecker sequence for 1-positions."""
    h = np.zeros(n)
    positions = [int((i * alpha) % 1.0 * n) for i in range(target_sum)]
    # Handle duplicates by finding nearest empty spot
    used = set()
    for pos in positions:
        if pos not in used:
            h[pos] = 1.0
            used.add(pos)
    # Fill remaining if needed
    count = int(h.sum())
    if count < target_sum:
        empty = np.where(h == 0)[0]
        for pos in empty[:target_sum - count]:
            h[pos] = 1.0
    elif count > target_sum:
        full = np.where(h == 1)[0]
        for pos in full[:count - target_sum]:
            h[pos] = 0.0
    return h

def sturmian_binary(n, target_sum):
    """Generate a Sturmian word as binary sequence."""
    alpha = target_sum / n
    h = np.zeros(n)
    for i in range(n):
        if int((i + 1) * alpha) > int(i * alpha):
            h[i] = 1.0
    # Fix if sum isn't exactly target_sum
    total = int(h.sum())
    if total != target_sum:
        if total > target_sum:
            ones = np.where(h == 1)[0]
            h[ones[:total - target_sum]] = 0
        else:
            zeros = np.where(h == 0)[0]
            h[zeros[:target_sum - total]] = 1
    return h

def van_der_corput_binary(n, target_sum, base=2):
    """Generate binary sequence using Van der Corput positions."""
    h = np.zeros(n)
    positions = set()
    i = 0
    while len(positions) < target_sum:
        # Van der Corput number
        vdc = 0.0
        f = 1.0 / base
        m = i
        while m > 0:
            vdc += (m % base) * f
            m //= base
            f /= base
        pos = int(vdc * n) % n
        positions.add(pos)
        i += 1
    for pos in positions:
        h[pos] = 1.0
    return h

def smooth_max_obj(h, alpha):
    """Smooth approximation of max of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def flat_obj(h, alpha, beta):
    """Objective that penalizes both the max and the spread of correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    # Penalize values above the mean
    mean_c = np.mean(corr)
    spread_penalty = np.mean(np.maximum(corr - mean_c, 0.0) ** 2)
    # Smooth max
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return float(smooth_max + beta * spread_penalty)

def slsqp_refine(h_start, n, target, maxiter, ftol, time_limit=None, start_time=None,
                 use_flat=False, beta=0.1):
    global _conf_n, _conf_target
    _conf_n = n
    _conf_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': conf_fun, 'jac': conf_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    alphas = [20, 50, 100, 200, 500, 1000, 2000, 5000]
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if use_flat:
            obj_func = lambda x: flat_obj(x, alpha, beta)
        else:
            obj_func = lambda x: smooth_max_obj(x, alpha)
        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def hill_climb_targeted(h, target_sum, max_iter, rng):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                        break
                if found:
                    break
            if not found:
                i1 = ones_cache[rng.randint(len(ones_cache))]
                i0 = zeros_cache[rng.randint(len(zeros_cache))]
                h_new[i1] = 0
                h_new[i0] = 1
                moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5


def hill_climb_targeted(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    stall = 0
    max_stall = 400

    for _ in range(max_iter):
        if stall > max_stall:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_len = min(5, len(corr))
        worst_indices = np.argsort(corr)[-top_len:]
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        found = False
        for wi in worst_indices:
            lag = wi - (n - 1)
            for i1 in ones:
                j = i1 - lag
                if 0 <= j < n and h[j] == 0:
                    h[i1] = 0
                    h[j] = 1
                    c5_new = compute_c5(h)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        stall = 0
                        found = True
                        break
                    else:
                        h[i1] = 1
                        h[j] = 0
            if found:
                break

        if not found:
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0
            h[i0] = 1
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                stall = 0
            else:
                h[i1] = 1
                h[i0] = 0
                stall += 1

    return h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def obj_smooth_max(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = np.clip(_al_val * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / _al_val)


def obj_flat(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    mean_c = np.mean(corr)
    spread = float(np.mean(np.maximum(corr - mean_c, 0.0) ** 2))
    shifted = np.clip(_al_val * (corr - c_max), -500.0, 0.0)
    smooth_max = float(c_max + np.log(np.sum(np.exp(shifted))) / _al_val)
    return smooth_max + _flat_beta * spread


def conf_fun(h):
    return np.sum(h) - _conf_target


def conf_jac(h):
    return np.ones(_conf_n)


def slsqp_refine(h_start, n, target, time_limit=None, start_time=None, use_flat=False):
    global _conf_n, _conf_target, _al_val, _flat_beta
    _conf_n = n
    _conf_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': conf_fun, 'jac': conf_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    alphas = [20, 50, 100, 200, 500, 1000, 2000, 4000]
    _flat_beta = 0.05 if use_flat else 0.0

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        _al_val = alpha
        if use_flat:
            obj_func = obj_flat
        else:
            obj_func = obj_smooth_max
        try:
            res = minimize(obj_func, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def iterated_local_search(h_start, target_sum, n_restarts, t_init, t_final, rng,
                          time_limit=None, start_time=None):
    n = len(h_start)
    h_cur = project_binary(h_start, target_sum)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for r in range(n_restarts):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        sa_iters = max(10000, min(50000, int((time_limit - (r * 0.1 * time_limit)) * 2000) if time_limit else 30000))
        time_limit_r = time_limit * (1.0 - (r + 1) / (n_restarts + 2.0)) if time_limit else None
        h_sa, c5_sa = targeted_sa(h_cur, target_sum, sa_iters, t_init, t_final,
                                   rng, time_limit=time_limit_r, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
        h_hc, c5_hc = hill_climb_targeted(h_sa, target_sum, 3000, rng)
        if c5_hc < best_c5:
            best_c5 = c5_hc
            best_h = h_hc.copy()
        perturb_strength = 0.05 + 0.1 * r / max(1, n_restarts - 1)
        n_perturb = int(perturb_strength * target_sum)
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(n_perturb):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            h_pert[i1] = 0
            h_pert[i0] = 1
        h_cur = h_pert

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _conf_n, _conf_target, _al_val, _flat_beta
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n_init = len(h_init)
    else:
        n_init = 600
        h_init = np.zeros(n_init)
        h_init[::2] = 1.0

    best_h = None
    best_score = 1.0
    n = n_init
    target_sum = n // 2

    candidates = []

    h_bin_init = project_binary(h_init, target_sum)
    c5_init = compute_c5(h_bin_init)
    candidates.append((h_bin_init, c5_init))
    if c5_init < best_score:
        best_score = c5_init
        best_h = h_bin_init.copy()

    golden = (1.0 + np.sqrt(5.0)) / 2.0
    alpha_kron = 1.0 / golden
    h_kron = kronecker_binary(n, target_sum, alpha_kron)
    c5_kron = compute_c5(h_kron)
    candidates.append((h_kron, c5_kron))
    if c5_kron < best_score:
        best_score = c5_kron
        best_h = h_kron.copy()

    h_stur = sturmian_binary(n, target_sum)
    c5_stur = compute_c5(h_stur)
    candidates.append((h_stur, c5_stur))
    if c5_stur < best_score:
        best_score = c5_stur
        best_h = h_stur.copy()

    h_vdc = vdc_binary(n, target_sum, base=3)
    c5_vdc = compute_c5(h_vdc)
    candidates.append((h_vdc, c5_vdc))
    if c5_vdc < best_score:
        best_score = c5_vdc
        best_h = h_vdc.copy()

    h_vdc2 = vdc_binary(n, target_sum, base=5)
    c5_vdc2 = compute_c5(h_vdc2)
    candidates.append((h_vdc2, c5_vdc2))
    if c5_vdc2 < best_score:
        best_score = c5_vdc2
        best_h = h_vdc2.copy()

    for rs in range(3):
        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:target_sum]] = 1.0
        c5_r = compute_c5(h_rand)
        candidates.append((h_rand, c5_r))
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_rand.copy()

    candidates.sort(key=lambda x: x[1])

    for ci in range(min(4, len(candidates))):
        if tleft() < 120:
            break
        h_cand, score_cand = candidates[ci]
        allowed = min(200.0, tleft() * 0.3)
        h_sa, c5_sa = targeted_sa(h_cand, target_sum, 80000, 0.015, 0.00004,
                                   rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        if tleft() > 40:
            h_hc, c5_hc = hill_climb_targeted(h_sa, target_sum, 6000, rng)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()
                h_sa = h_hc

    if tleft() > 200:
        allowed = min(250.0, tleft() * 0.4)
        h_ils, c5_ils = iterated_local_search(best_h, target_sum, 3,
                                               0.012, 0.00005, rng,
                                               time_limit=allowed, start_time=start_time)
        if c5_ils < best_score:
            best_score = c5_ils
            best_h = h_ils.copy()

    n_high = n + 200
    target_high = n_high // 2
    if tleft() > 100:
        h_interp = interpolate_to(best_h, n_high, target_high)
        h_bin_high = project_binary(h_interp, target_high)
        allowed = min(150.0, tleft() * 0.3)
        h_sa_h, c5_h = targeted_sa(h_bin_high, target_high, 60000, 0.012, 0.00005,
                                    rng, time_limit=allowed, start_time=start_time)
        if tleft() > 30:
            h_hc_h, c5_hc_h = hill_climb_targeted(h_sa_h, target_high, 4000, rng)
            if c5_hc_h < c5_h:
                h_sa_h = h_hc_h
                c5_h = c5_hc_h
        if c5_h < best_score:
            best_score = c5_h
            best_h = h_sa_h.copy()
            n = n_high
            target_sum = target_high

    n_high2 = n + 200
    target_high2 = n_high2 // 2
    if tleft() > 80:
        h_interp2 = interpolate_to(best_h, n_high2, target_high2)
        h_bin_h2 = project_binary(h_interp2, target_high2)
        allowed = min(120.0, tleft() * 0.3)
        h_sa_h2, c5_h2 = targeted_sa(h_bin_h2, target_high2, 50000, 0.010, 0.00004,
                                      rng, time_limit=allowed, start_time=start_time)
        if tleft() > 25:
            h_hc_h2, c5_hc_h2 = hill_climb_targeted(h_sa_h2, target_high2, 3000, rng)
            if c5_hc_h2 < c5_h2:
                h_sa_h2 = h_hc_h2
                c5_h2 = c5_hc_h2
        if c5_h2 < best_score:
            best_score = c5_h2
            best_h = h_sa_h2.copy()
            n = n_high2
            target_sum = target_high2

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        h_ref, sc = slsqp_refine(best_h, n_use, target_use,
                                  time_limit=tleft() - 50, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 60:
        h_flat, sc_f = slsqp_refine(best_h, n_use, target_use,
                                     time_limit=tleft() - 40, start_time=start_time,
                                     use_flat=True)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_flat

    n_use = len(best_h)
    target_use = n_use / 2.0
    fine_attempts = min(6, int(max(2, tleft() / 30)))

    for fa in range(fine_attempts):
        if tleft() < 25:
            break
        noise_scale = 0.01 + 0.02 * fa
        h_pert = best_h + rng.randn(n_use) * noise_scale
        h_pert = project_feasible(h_pert, target_use)
        h_ref, sc = slsqp_refine(h_pert, n_use, target_use,
                                  time_limit=min(25.0, tleft() - 12), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use // 2
        h_bin_f = project_binary(best_h, target_use)
        max_it = int(max(3000, min(30000, tleft() * 1000)))
        h_sa_f, c5_f = targeted_sa(h_bin_f, target_use, max_it, 0.003, 0.00002,
                                    rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 10:
            h_hc_f, c5_hc_f = hill_climb_targeted(best_h, len(best_h) // 2, 3000, rng)
            if c5_hc_f < best_score:
                best_score = c5_hc_f
                best_h = h_hc_f

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-6:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
