def subgradient_step(h, target_sum, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)  # shift
    
    # Gradient of corr[k_star] with respect to h
    m_range = np.arange(2 * n - 1)
    # For the active constraint k_star:
    # g[j] = (1 - h[j + s]) - h[j - s] (for valid indices)
    g = np.zeros(n)
    # Term 1: (1 - h[j+s]) for j where 0 <= j+s < n
    mask1 = (np.arange(n) + s >= 0) & (np.arange(n) + s < n)
    g[mask1] += (1.0 - h[np.arange(n)[mask1] + s])
    # Term 2: -h[j-s] for j where 0 <= j-s < n
    mask2 = (np.arange(n) - s >= 0) & (np.arange(n) - s < n)
    g[mask2] += -h[np.arange(n)[mask2] - s]
    
    # Move in negative gradient direction, projected
    h_new = h - lr * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_feasible(h_new, target_sum)
    return h_new, compute_c5(h_new)
