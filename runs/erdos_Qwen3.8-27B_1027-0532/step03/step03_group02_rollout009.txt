    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return h

def coord_descend(h, target_sum, max_time, rng, patience=30, bisect_iters=45, noise_base=0.002):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()
    
    while time.time() - elapsed_start < max_time:
        for j in range(n):
            old_val = h[j]
            js_fwd = j + s_arr
            js_bwd = j - s_arr
            mask_fwd = (js_fwd >= 0) & (js_fwd < n)
            mask_bwd = (js_bwd >= 0) & (js_bwd < n)
            
            A = np.zeros(2 * n - 1)
            A[mask_fwd] += 1.0 - h[js_fwd[mask_fwd]]
            A[mask_bwd] -= h[js_bwd[mask_bwd]]
            
            C = corr - A * old_val
            
            lo, hi = 0.0, 1.0
            for _ in range(bisect_iters):
                mid = (lo + hi) * 0.5
                vals = A * mid + C
                k_star = int(np.argmax(vals))
                grad = A[k_star]
                if grad > 1e-12:
                    hi = mid
                elif grad < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
        
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            noise_scale = noise_base * (1.0 + 0.5 * min(no_improve / patience, 3.0))
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    
    return best_h, best_score

def weighted_gradient_step(h, target_sum, corr, n, s_arr, w, lr):
    """Take a gradient step on the weighted objective sum_k w[k]*corr[k]."""
    n_shifts = 2 * n - 1
    grad = np.zeros(n)
    
    for k in range(n_shifts):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = s_arr[k]
        js_fwd = np.arange(n) + s
        js_bwd = np.arange(n) - s
        mask_fwd = (js_fwd >= 0) & (js_fwd < n)
        mask_bwd = (js_bwd >= 0) & (js_bwd < n)
        
        contrib = np.zeros(n)
        contrib[mask_fwd] = (1.0 - h[js_fwd[mask_fwd]])
        contrib[mask_bwd] -= h[js_bwd[mask_bwd]]
        grad += wk * contrib
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new

def minimax_descent(h, target_sum, max_time, rng, patience=100):
    """Minimax-style optimization to escape local minima."""
    n = len(h)
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    s_arr = np.arange(n_shifts) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    
    beta = 5.0
    lr = 0.0005
    no_improve = 0
    elapsed_start = time.time()
    
    while time.time() - elapsed_start < max_time:
        # Update weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        beta_eff = min(beta, 50.0)
        w = np.exp(beta_eff * (corr - c_max))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(n_shifts)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= w_sum
        
        # Gradient step
        h_new = weighted_gradient_step(h, target_sum, corr, n, s_arr, w, lr)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        score_new = float(np.max(corr_new)) * dx
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_h = h_new.copy()
            h = h_new
            no_improve = 0
            lr = min(lr * 1.02, 0.002)
            beta = min(beta * 1.005, 100.0)
        else:
            no_improve += 1
            lr *= 0.8

                # Restart

                h = best_h + rng.randn(n) * noise_scale
                h = project_feasible(h, target_sum)
                no_improve = 0
                lr = 0.0005
                beta = 5.0
            if lr < 1e-7:
                break
    
    # Final coordinate descent polish
    h_cd, sc_cd = coord_descend(h, target_sum, min(30, max(5, max_time - (time.time() - elapsed_start))), rng, patience=10)
    if sc_cd < best_score:
        best_h = h_cd
        best_score = sc_cd
    
    return best_h, best_score

def interpolate_resample(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

One idea: instead of going through coordinates in order (0, 1, 2, ..., n-1), use a random order or a priority-based order (e.g., optimize coordinates that contribute most to the worst shift first).

Another idea: "block" coordinate descent - optimize groups of coordinates together.

For the block approach: given a block of m coordinates, the objective is still a max of linear functions (each being linear in the m block variables). This is a Linear Program! We can solve it efficiently.

Specifically, for a block S of coordinates, we want:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return h

def get_worst_shift_gradient(corr, h, n, s_arr):
    """Compute gradient of the worst shift's correlation."""
    k_star = int(np.argmax(corr))
    s = s_arr[k_star]
    grad = np.zeros(n)
    js_fwd = np.arange(n) + s
    js_bwd = np.arange(n) - s
    mask_fwd = (js_fwd >= 0) & (js_fwd < n)
    mask_bwd = (js_bwd >= 0) & (js_bwd < n)
    grad[mask_fwd] = (1.0 - h[js_fwd[mask_fwd]])
    grad[mask_bwd] -= h[js_bwd[mask_bwd]]
    return grad, k_star

def smart_restart(h_best, corr, n, s_arr, target, rng, strength=0.003):
    """Restart with targeted perturbation towards reducing the worst shift."""
    grad, k_star = get_worst_shift_gradient(corr, h_best, n, s_arr)
    # Negative gradient direction (to reduce the worst shift's correlation)
    neg_grad = -grad
    # Project to feasible direction (orthogonal to all-ones vector for the sum constraint)
    mean_ng = neg_grad.mean()
    neg_grad_centered = neg_grad - mean_ng
    # Scale to desired magnitude
    ng_norm = np.linalg.norm(neg_grad_centered)
    if ng_norm > 1e-15:
        direction = neg_grad_centered / ng_norm
        perturbation = strength * direction + strength * 0.5 * rng.randn(n)
    else:
        perturbation = strength * rng.randn(n)
    h_new = h_best + perturbation
    h_new = project_feasible(h_new, target)
    return h_new

def coord_descend(h, target_sum, max_time, rng, patience=30, bisect_iters=50, 
                  noise_base=0.002, use_smart_restart=True):
    n = len(h)
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    s_arr = np.arange(n_shifts) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()
    
    while time.time() - elapsed_start < max_time:
        # One full sweep
        for j in range(n):
            old_val = h[j]
            js_fwd = j + s_arr
            js_bwd = j - s_arr
            mask_fwd = (js_fwd >= 0) & (js_fwd < n)
            mask_bwd = (js_bwd >= 0) & (js_bwd < n)
            
            A = np.zeros(n_shifts)
            if np.any(mask_fwd):
                A[mask_fwd] += 1.0 - h[js_fwd[mask_fwd]]
            if np.any(mask_bwd):
                A[mask_bwd] -= h[js_bwd[mask_bwd]]
            
            C = corr - A * old_val
            
            lo, hi = 0.0, 1.0
            for _ in range(bisect_iters):
                mid = (lo + hi) * 0.5
                k_star = int(np.argmax(A * mid + C))
                grad = A[k_star]
                if grad > 1e-12:
                    hi = mid
                elif grad < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
        
        # Project and evaluate
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            if use_smart_restart:

                                  strength=noise_base * (1.0 + 0.3 * min((no_improve - patience) / 20.0, 2.0)))
            else:
                h = best_h + rng.randn(n) * noise_base
                h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    
    return best_h, best_score

    n = len(h)

    n_shifts = 2 * n - 1

    h = project_feasible(h.copy(), target_sum)

    best_h = h.copy()

    lr = 0.0003
    no_improve = 0

    while time.time() - elapsed_start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        
        if score < best_score - 1e-15:
            best_score = score
            best_h = h.copy()
            no_improve = 0

        beta_eff = min(beta, 30.0)
        w = np.exp(beta_eff * (corr - c_max))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(n_shifts)
            w[int(np.argmax(corr))] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for k in range(n_shifts):
            wk = w[k]
            if wk < 1e-12:
                continue
            s = s_arr[k]
            js_fwd = np.arange(n) + s
            js_bwd = np.arange(n) - s
            mask_fwd = (js_fwd >= 0) & (js_fwd < n)
            mask_bwd = (js_bwd >= 0) & (js_bwd < n)
