    scaled = alpha * corr * dx
    max_s = np.max(scaled)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h2 = h.copy()
    h2[idx] += diff / len(idx)
    return np.clip(h2, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    if current_sum != target_sum:
        if current_sum > target_sum:
            need = current_sum - target_sum
            ones = np.where(h_r == 1)[0]
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
        else:
            need = target_sum - current_sum
            zeros = np.where(h_r == 0)[0]
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def generate_random_binary(n, rng):
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    return h

def smooth_obj(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    return (max_s + np.log(np.sum(np.exp(scaled - max_s)))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    softmax = exps / np.sum(exps)
    w_full = softmax
    a = 1.0 - h
    first_sum = np.correlate(a, w_full, 'full')[n - 1: 3 * n - 2]
    second_sum = np.convolve(h, w_full, 'full')[n - 1: 3 * n - 2]
    return dx * (first_sum - second_sum)

def _slsqp_wrapper(h, alpha, n, target):
    return smooth_obj(h, alpha) + 50.0 * (np.sum(h) - target) ** 2

def _slsqp_grad_wrapper(h, alpha, n, target):
    return smooth_grad(h, alpha) + 100.0 * (np.sum(h) - target) * np.ones(n)

def slsqp_refine(h_start, n, target, alpha, time_limit=None, start_time=None, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    try:
        if time_limit is not None and start_time is not None:
            remaining = time_limit - (time.time() - start_time)
            if remaining < 5:
                return best_h, best_val
            maxiter = int(min(maxiter, max(20, remaining * 40)))
        res = minimize(_slsqp_wrapper, h_start, args=(alpha, n, target),
                       jac=lambda x: _slsqp_grad_wrapper(x, alpha, n, target),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7, ' iprint': -1})
        h_new = project_feasible(res.x, target)
        val = compute_c5(h_new)
        if val < best_val:
            best_h = h_new
            best_val = val
    except Exception:
        pass
    return best_h, best_val

def slsqp_continuation(h_start, n, target, alphas, time_limit=None, start_time=None):
    best_h = h_start.copy()
    best_val = compute_c5(best_h)
    h_cur = h_start.copy()

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

        h_sa_o, c5_o = sa_optimize(h_bin_cur, target_use, 40000, 0.015, 0.00005,
                                   rng, time_limit=min(60.0, tleft() - 40),
                                   start_time=start_time)
        if c5_o < best_score:
            best_score = c5_o
            best_h = h_sa_o
        if tleft() > 40:
            h_smooth_o = project_feasible(0.5 * best_h + 0.25, target_use)
            h_ref_o, sc_o = slsqp_continuation(h_smooth_o, n_use, target_use,
                                               [50, 200, 800, 3000],
                                               time_limit=min(40.0, tleft() - 25),
                                               start_time=start_time)
            if sc_o < best_score:
                best_score = sc_o
                best_h = h_ref_o

    # Phase 5: Final binary polish
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_f = project_binary(best_h, target_use)
        h_sa_f, c5_f = sa_optimize(h_bin_f, target_use, 50000, 0.008, 0.00003,
                                   rng, time_limit=max(5.0, tleft() - 12),
                                   start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 15:
            h_smooth_f = project_feasible(best_h + 0.1 * (rng.randn(n_use) - 0.5), target_use)
            h_ref_f, sc_f = slsqp_continuation(h_smooth_f, n_use, target_use,
                                               [500, 2000],
                                               time_limit=max(5.0, tleft() - 8),
                                               start_time=start_time)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_ref_f

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
