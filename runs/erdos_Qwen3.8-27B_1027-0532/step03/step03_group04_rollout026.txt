def compute_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    corr = np.correlate(h, 1.0-h, mode='full') * dx
    # Softmax weights
    max_corr = np.max(corr)
    log_weights = alpha * (corr - max_corr)
    log_weights = np.clip(log_weights, -500, 0)
    weights = np.exp(log_weights)
    weights /= np.sum(weights)
    # weights has length 2n-1, indexed 0 to 2n-2

    diff = target_sum - np.sum(h)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def compute_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-300:
        return np.zeros(n)
    weights /= total_w
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-15:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += w * one_minus_h[k:]
            if n - k > 0:
                grad[k:] -= w * h[:n-k]
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:] += w * one_minus_h[:n-kk]
                grad[:n-kk] -= w * h[kk:]
    
    grad *= dx
    return grad

def gradient_descent(h_init, n, target, alpha_schedule, time_limit, max_iters_per_alpha=5000):
    start_time = time.time()
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for alpha in alpha_schedule:
        if time.time() - start_time > time_limit:
            break
        
        step = 1.0 / sqrt_alpha if (sqrt_alpha := np.sqrt(max(alpha, 1.0))) > 0 else 1.0
        no_improve = 0
        
        for it in range(max_iters_per_alpha):
            if time.time() - start_time > time_limit:
                return best_h, best_c5
            
            grad = compute_smooth_gradient(h, alpha)
            
            # Simple backtracking
            step_try = step
            obj_cur = smooth_obj(h, alpha)
            
            for bt in range(30):
                h_try = project_feasible(h - step_try * grad, target)
                obj_try = smooth_obj(h_try, alpha)
                if obj_try <= obj_cur:

                step_try *= 0.5
            
            h = h_try
            step = min(step * 1.3, 10.0 / max(alpha, 1.0))
            
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-14:
                best_c5 = c5_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 100:
                break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-300:
        return np.zeros(n)
    weights /= total_w
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - (n - 1)
        if k >= 0:
            nk = n - k
            if nk > 0:
                grad[:nk] += w * one_minus_h[k:]
                grad[k:] -= w * h[:nk]
        else:
            kk = -k
            nk = n - kk
            if nk > 0:
                grad[kk:] += w * one_minus_h[:nk]
                grad[:nk] -= w * h[kk:]
    
    grad *= dx
    return grad
