import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return h_new

    h_pad = np.zeros(3 * n)
    h_pad[n:2 * n] = h_work
    C = np.correlate(h_work, 1.0 - h_work, mode='full')
    current_c5 = float(np.max(C))
    best_c5 = current_c5
    best_h = h_work.copy()

    n_iters = 0
    last_recompute = 0
    stall = 0

    while time.time() - start < time_limit:
        if stall > 20000:
            break
        if n_iters - last_recompute > 20000:
            C = np.correlate(h_work, 1.0 - h_work, mode='full')
            current_c5 = float(np.max(C))
            h_pad[n:2 * n] = h_work
            last_recompute = n_iters

        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            n_iters += 1
            continue

        idx_a = n + i - lags
        idx_b = n + j - lags
        idx_c = n + i + lags
        idx_d = n + j + lags
        G = h_pad[idx_a] - h_pad[idx_b] + h_pad[idx_c] - h_pad[idx_d]

        d_min = max(-h_work[j], h_work[i] - 1.0)
        d_max = min(1.0 - h_work[j], h_work[i])
        if d_min >= d_max:
            n_iters += 1
            continue

        top_idx = np.argmax(C)
        g_top = G[top_idx]
        if abs(g_top) > 1e-10:
            d_opt = -C[top_idx] / g_top
        else:
            d_opt = (d_min + d_max) / 2.0
        d_opt = float(np.clip(d_opt, d_min, d_max))

        candidates = [d_opt, d_min, d_max, (d_min + d_max) / 2.0]
        best_d = d_opt
        best_val = float(np.max(C + d_opt * G))
        for dc in candidates:
            if dc < d_min or dc > d_max:
                continue
            vc = float(np.max(C + dc * G))
            if vc < best_val:
                best_val = vc
                best_d = dc

        if best_val < current_c5 - 1e-14:
            h_work[i] += best_d
            h_work[j] -= best_d
            h_pad[n + i] = h_work[i]
            h_pad[n + j] = h_work[j]
            C += best_d * G
            current_c5 = best_val
            stall = 0
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_work.copy()
        else:
            stall += 1
        n_iters += 1

    return best_h, best_c5


def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    n_harm = 4
    for k in range(1, n_harm + 1):
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    h = 0.5 + h
    h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target = n / 2.0

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    if tleft() > 280:
        h1, s1 = sa_pair_swaps(h0, target, time_limit=280, seed=seed, T_init=0.01)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    if tleft() > 200:
        h2, s2 = directed_hill_climb(best_h, target, time_limit=200, seed=seed + 100)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 150:
        h3, s3 = random_hill_climb(best_h, target, time_limit=150, seed=seed + 200)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    if tleft() > 250:
        h_sa2, s_sa2 = sa_pair_swaps(best_h, target, time_limit=200, seed=seed + 300, T_init=0.005)
        if s_sa2 < best_score:
            best_score = s_sa2
            best_h = h_sa2
        if tleft() > 40:
            h_dc2, s_dc2 = directed_hill_climb(best_h, target, time_limit=max(30, tleft() - 5), seed=seed + 350)
            if s_dc2 < best_score:
                best_score = s_dc2
                best_h = h_dc2

    if tleft() > 350:
        n2 = 2 * n
        h2n = upsample(best_h, n2)
        h2n = project_feasible(h2n, n2 / 2.0)
        s2n_init = compute_c5(h2n)
        h4, s4 = sa_pair_swaps(h2n, n2 / 2.0, time_limit=250, seed=seed + 400, T_init=0.008)
        if s4 < best_score:
            best_score = s4
            best_h = h4
        if tleft() > 80:
            h5, s5 = directed_hill_climb(best_h, n2 / 2.0, time_limit=60, seed=seed + 500)
            if s5 < best_score:
                best_score = s5
                best_h = h5
        if tleft() > 50:
            h6, s6 = random_hill_climb(best_h, n2 / 2.0, time_limit=40, seed=seed + 550)
            if s6 < best_score:
                best_score = s6
                best_h = h6

    if tleft() > 120:
        n_best = len(best_h)
        h7, s7 = sa_pair_swaps(best_h, n_best / 2.0, time_limit=80, seed=seed + 600, T_init=0.003)
        if s7 < best_score:
            best_score = s7
            best_h = h7
        if tleft() > 30:
            h8, s8 = random_hill_climb(best_h, n_best / 2.0, time_limit=max(20, tleft() - 5), seed=seed + 700)
            if s8 < best_score:
                best_score = s8
                best_h = h8

    if tleft() > 40:
        n_best = len(best_h)
        h9, s9 = directed_hill_climb(best_h, n_best / 2.0, time_limit=max(20, tleft() - 5), seed=seed + 800)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    n_best = len(best_h)
    final_h = project_feasible(best_h, n_best / 2.0)
    final_score = compute_c5(final_h)
    if final_score < best_score:
        best_score = final_score
        best_h = final_h

    return best_h, best_score, len(best_h)
