    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per_elem = diff / len(idx)
        h[idx] = np.minimum(h[idx] + per_elem, 1.0)
        # If some hit 1.0, redistribute
        remaining = target_sum - np.sum(h)
        if remaining > 1e-12:
            idx2 = np.where(h < 1.0)[0]
            if len(idx2) > 0:
                h[idx2] += remaining / len(idx2)
                h = np.clip(h, 0.0, 1.0)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per_elem = diff / len(idx)
        h[idx] = np.maximum(h[idx] + per_elem, 0.0)
        remaining = target_sum - np.sum(h)
        if remaining < -1e-12:
            idx2 = np.where(h > 0.0)[0]
            if len(idx2) > 0:
                h[idx2] += remaining / len(idx2)
                h = np.clip(h, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def smooth_obj_and_grad(h, p, target_sum):
    corr = get_corrs(h)
    # Log-sum-exp for numerical stability
    max_c = np.max(corr)
    log_sum_exp = max_c + np.log(np.sum(np.exp(p * (corr - max_c))))
    obj = log_sum_exp / p
    
    # Gradient
    weights = np.exp(p * (corr - max_c))
    weights = weights / np.sum(weights)
    
    n = len(h)
    grad = np.zeros(n)
    # corr[k] for k = 0, ..., 2n-2
    # lag = k - (n-1)
    one_minus_h = 1.0 - h
    for k in range(len(corr)):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return obj, grad

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    corr_all = get_corrs(h_start)
    max_c = np.max(corr_all)
    obj_val = max_c

    def obj_fn(x, alpha):
        c = get_corrs(x)
        mc = np.max(c)
        sh = np.clip(alpha * (c - mc), -500.0, 0.0)
        return mc + np.log(np.sum(np.exp(sh))) / alpha

    def obj_grad(x, alpha):
        c = get_corrs(x)
        mc = np.max(c)
        es = np.exp(np.clip(alpha * (c - mc), -500.0, 0.0))
        tw = es / np.sum(es)
        return corr_grad_weighted(x, np.arange(len(c)), tw)

    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(obj_fn, h_cur, args=(alpha,),
                           jac=lambda x: obj_grad(x, alpha),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def structured_init(n, seed, kind):
    rng_local = np.random.RandomState(seed)
    if kind == 0:
        h = np.ones(n) / 1.0
        mask = rng_local.permutation(n)[:n // 2]
        h = np.zeros(n)
        h[mask] = 1.0
    elif kind == 1:
        h = np.zeros(n)
        for i in range(n):
            h[i] = 1.0 if (i % 4) < 2 else 0.0
    elif kind == 2:
        t = np.linspace(0, 4 * np.pi, n)
        h = 0.5 + 0.5 * np.sin(t)
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: quick optimization at initial resolution
    if remaining() > 100:
        h1, s1 = heavy_ball(best_h, target_init, p=40, max_iter=2000,
                            lr_init=0.1, momentum=0.85, patience=150)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    if remaining() > 80:
        h1, s1 = peak_shave(best_h, target_init, max_iter=2000,
                            lr_init=0.08, topk=5, alpha=30)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: scale to 600 points
    if remaining() > 250:
        n600 = 600
        h600 = interpolate_h(best_h, n600, n600 / 2.0)
        h600, s600 = heavy_ball(h600, n600 / 2.0, p=30, max_iter=3000,
                                lr_init=0.08, momentum=0.88, patience=200)
        if remaining() > 100:
            h600, s600 = peak_shave(h600, n600 / 2.0, max_iter=2000,
                                    lr_init=0.06, topk=5, alpha=40)
        if s600 < best_score:
            best_score = s600
            best_h = h600

    # Phase 3: scale to 800 points
    if remaining() > 300:
        n800 = 800
        h800 = interpolate_h(best_h, n800, n800 / 2.0)
        h800, s800 = heavy_ball(h800, n800 / 2.0, p=25, max_iter=4000,
                                lr_init=0.06, momentum=0.90, patience=250)
        if remaining() > 120:
            h800, s800 = peak_shave(h800, n800 / 2.0, max_iter=3000,
                                    lr_init=0.05, topk=4, alpha=50)
        if s800 < best_score:
            best_score = s800
            best_h = h800

    # Phase 4: scale to 1000 points
    if remaining() > 300:
        n1000 = 1000
        h1000 = interpolate_h(best_h, n1000, n1000 / 2.0)
        h1000, s1000 = heavy_ball(h1000, n1000 / 2.0, p=20, max_iter=4000,
                                  lr_init=0.05, momentum=0.90, patience=250)
        if remaining() > 120:
            h1000, s1000 = peak_shave(h1000, n1000 / 2.0, max_iter=2500,
                                      lr_init=0.04, topk=5, alpha=40)
        if s1000 < best_score:
            best_score = s1000
            best_h = h1000

    # Phase 5: SLSQP polish at current resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if remaining() > 200:
        h_ref = slsqp_refine(best_h, target_cur, [40, 100, 250], 800, 1e-13)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: additional heavy ball + peak shave cycles
    for cycle in range(3):
        if remaining() < 80:
            break
        h_try = best_h + rng.randn(n_cur) * 0.008
        h_try = project_feasible(h_try, target_cur)
        h_new, s_new = peak_shave(h_try, target_cur, max_iter=2000,
                                  lr_init=0.04, topk=3, alpha=60)
        if s_new < best_score:
            best_score = s_new
            best_h = h_new

    # Phase 7: final SLSQP
    if remaining() > 60:
        h_final = slsqp_refine(best_h, target_cur, [200, 500], 600, 1e-14)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 8: try structured inits at high resolution if time permits
    if remaining() > 150:
        n_struct = len(best_h)
        for kind in range(3):
            if remaining() < 80:
                break
            h_str = structured_init(n_struct, seed + 500 + kind, kind)
            h_str, s_str = heavy_ball(h_str, n_struct / 2.0, p=20,
                                      max_iter=2000, lr_init=0.05,
                                      momentum=0.88, patience=150)
            if s_str < best_score:
                best_score = s_str
                best_h = h_str

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
