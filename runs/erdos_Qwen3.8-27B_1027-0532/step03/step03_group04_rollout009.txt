            current_reg = max(1e-6, current_reg * 0.3)
        if it % 200 == 0 and current_reg > 1e-7:
            current_reg = max(1e-7, current_reg * 0.2)
    return best_h, best_c5

def swap_descent(h, n, target, max_rounds=80, time_limit=60, seed=0):
    start = time.time()
    h = project_feasible(h.copy(), target)
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        one_idx = np.where(h > 0.7)[0]
        zero_idx = np.where(h < 0.3)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        rng.shuffle(one_idx)
        rng.shuffle(zero_idx)
        improved = False
        max_tries = min(len(one_idx) * len(zero_idx), 3000)
        tries = 0
        for i in one_idx[:60]:
            if time.time() - start > time_limit or tries >= max_tries:
                break
            for j in zero_idx[:60]:
                tries += 1
                if tries >= max_tries:
                    break
                h_try = h.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5 - 1e-12:
                    h = h_try
                    best_c5 = sc
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def make_structured_init(n, seed, pattern_type):
    rng = np.random.RandomState(seed)
    if pattern_type == 0:
        block = max(1, n // 8)
        h = np.zeros(n)
        for i in range(0, n, block * 2):
            h[i:i + block] = 1.0
        return project_feasible(h, n / 2.0)
    elif pattern_type == 1:
        h = (rng.rand(n) < 0.5).astype(float)
        return project_feasible(h, n / 2.0)
    elif pattern_type == 2:
        x = np.linspace(0, 4 * np.pi, n)
        h = (np.sin(x) > 0).astype(float)
        return project_feasible(h, n / 2.0)
    elif pattern_type == 3:
        x = np.linspace(0, 2 * np.pi, n)
        h = (np.sin(x) + 0.5 * np.sin(3 * x) > 0).astype(float)
        return project_feasible(h, n / 2.0)
    else:
        x = np.linspace(0, 2 * np.pi * 5, n)
        h = (np.sin(x) + 0.3 * np.cos(2 * x) > 0).astype(float)
        return project_feasible(h, n / 2.0)

def hard_threshold(h, n, target):
    n_ones = int(target)
    idx = np.argsort(h)[::-1][:n_ones]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 280:
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                      max_iter=10000, time_limit=min(260, tleft() * 0.28),
                                      damping_init=0.5, reg=1e-4, solver_name='PDLP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 200:
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=6000, time_limit=min(160, tleft() * 0.25),
                                      damping_init=0.4, reg=1e-5, solver_name='SCS')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 350:
        for pt in range(5):
            if tleft() < 150:
                break
            h_struct = make_structured_init(n, seed + pt * 137, pt)
            lp_time = min(70, tleft() * 0.1)
            h_lp, c5_lp = alternating_lp(h_struct, n, target,
                                          max_iter=4000, time_limit=lp_time,
                                          damping_init=0.5, reg=1e-4, solver_name='PDLP')
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp

    for restart in range(6):
        if tleft() < 120:
            break
        noise = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4][restart]
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        damping = [0.3, 0.5, 0.4, 0.6, 0.35, 0.55][restart]
        solver = 'PDLP' if restart % 2 == 0 else 'SCS'
        lp_time = min(70, tleft() * 0.12)
        h_lp, c5_lp = alternating_lp(h_pert, n, target,
                                      max_iter=4000, time_limit=lp_time,
                                      damping_init=damping, reg=1e-5, solver_name=solver)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 80:
        h_swap, c5_swap = swap_descent(best_h, n, target,
                                        max_rounds=100, time_limit=min(60, tleft() * 0.35),
                                        seed=seed)
        if c5_swap < best_score:
            best_score = c5_swap
            best_h = h_swap

    if tleft() > 100:
        h_hard = hard_threshold(best_h, n, target)
        sc_hard = compute_c5(h_hard)
        if sc_hard < best_score:
            best_score = sc_hard
            best_h = h_hard
        h_lp, c5_lp = alternating_lp(best_h, n, target,
                                      max_iter=6000, time_limit=min(80, tleft() * 0.4),
                                      damping_init=0.4, reg=1e-6, solver_name='PDLP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 150:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(100, tleft() * 0.3)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                          max_iter=5000, time_limit=lp_time,
                                          damping_init=0.5, reg=1e-4, solver_name='PDLP')
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try
            if tleft() > 50:
                n_use = len(best_h)
                h_swap, c5_swap = swap_descent(best_h, n_use, n_use / 2.0,
                                                max_rounds=50, time_limit=min(35, tleft() * 0.3),
                                                seed=seed + 7)
                if c5_swap < best_score:
                    best_score = c5_swap
                    best_h = h_swap
            if tleft() < 60:
                break

    if tleft() > 50:
        n_use = len(best_h)
        h_lp, c5_lp = alternating_lp(best_h, n_use, n_use / 2.0,
                                      max_iter=5000, time_limit=min(40, tleft() * 0.6),
                                      damping_init=0.3, reg=1e-7, solver_name='PDLP')
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 30:
        n_use = len(best_h)
        h_swap, c5_swap = swap_descent(best_h, n_use, n_use / 2.0,
                                        max_rounds=50, time_limit=min(20, tleft() * 0.5),
                                        seed=seed + 13)
        if c5_swap < best_score:
            best_score = c5_swap
            best_h = h_swap

        h_hard = hard_threshold(best_h, n_use, n_use / 2.0)
        sc_hard = compute_c5(h_hard)
        if sc_hard < best_score:
            best_score = sc_hard
            best_h = h_hard

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
