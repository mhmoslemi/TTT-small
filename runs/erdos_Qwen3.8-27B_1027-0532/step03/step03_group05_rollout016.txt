def logsumexp_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute all I_l for l = 0, ..., n-1
    I = np.zeros(n)
    for l in range(n):
        I[l] = dx * np.dot(h[:n-l], 1.0 - h[l:])
    
    # Log-sum-exp
    I_max = np.max(I)
    shifted = alpha * (I - I_max)
    exp_vals = np.exp(shifted)
    log_sum_exp = I_max + np.log(np.sum(exp_vals)) / alpha
    
    # Weights
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient
    grad = np.zeros(n)
    for j in range(n):
        # sum_{l < n-j} w_l * (1 - h[j+l])
        s1 = 0.0
        for l in range(min(n - j, n)):
            s1 += w[l] * (1.0 - h[j + l])
        # sum_{l <= j} w_l * h[j-l]
        s2 = 0.0
        for l in range(min(j + 1, n)):
            s2 += w[l] * h[j - l]
        grad[j] = dx * (s1 - s2)
    
    return log_sum_exp, grad

    from scipy.optimize import minimize
    n = len(h)
    start = time.time()
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 
             'jac': lambda x: np.ones(n)}]
    
    def obj_and_grad(x):
        obj, grad = logsumexp_obj_grad(x, alpha)
        return obj, grad
    
    remaining = time_limit - (time.time() - start)
    maxiter = int(max(50, remaining * 5))
    
    try:
        res = minimize(obj_and_grad, h, method='L-BFGS-B', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})

    # Compute I_l for l = 0, ..., n-1
    I = np.zeros(n)

        grad *= dx
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        momentum = beta_mom * momentum + (1.0 - beta_mom) * (grad / gn)
        h_new = h_curr - eta * momentum
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve > 60:
                eta *= 0.8
                alpha *= 1.1
                no_improve = 0
        h_curr = h_new
    return best, best_score


def pairwise_cd(h, target, time_limit, seed):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    base_deltas = [0.002, 0.005, 0.01, 0.02, 0.05]
    scale = 1.0
    no_improve = 0
    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = base_deltas[rng.randint(len(base_deltas))] * scale
        if rng.rand() < 0.5:
            d = -d
        val_i = best[i] + d
        val_j = best[j] - d
        if val_i < 0.0 or val_i > 1.0 or val_j < 0.0 or val_j > 1.0:
            continue
        trial = best.copy()
        trial[i] = val_i
        trial[j] = val_j
        s = compute_c5(trial)
        if s < best_score:
            best_score = s
            best = trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 1500:
                scale *= 0.6
                if scale < 0.005:
                    break
                no_improve = 0
    return best, best_score


def fine_cd(h, target, time_limit, seed):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    base_deltas = [0.0005, 0.001, 0.002, 0.005]
    scale = 1.0
    no_improve = 0
    while time.time() - start < time_limit:
        i = rng.randint(n)
        d = base_deltas[rng.randint(len(base_deltas))] * scale
        if rng.rand() < 0.5:
            d = -d
        val = best[i] + d
        if val < 0.0 or val > 1.0:
            continue
        trial = best.copy()
        trial[i] = val
        trial = project_feasible(trial, target)
        s = compute_c5(trial)
        if s < best_score:
            best_score = s
            best = trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 800:
                scale *= 0.65
                if scale < 0.01:
                    break
                no_improve = 0
    return best, best_score


def structured_perturb_and_optimize(h, target, time_limit, seed):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    for _ in range(5):
        if time.time() - start >= time_limit:
            break
        freq = rng.randint(2, 8)
        amp = rng.uniform(0.01, 0.04)
        phase = rng.uniform(0, 2 * np.pi)
        perturbation = amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
        h_pert = best + perturbation
        h_pert = project_feasible(h_pert, target)
        h_opt, score = lbfgs_phase(h_pert, target, min(60, time_limit - (time.time() - start)), 100)
        if score < best_score:
            best_score = score
            best = h_opt
    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    alphas_first = [10, 30, 80, 200]
    for alpha in alphas_first:
        if tleft() < 80:
            break
        time_alloc = min(120, tleft() * 0.1)
        h_new, score = lbfgs_phase(best_h, target, time_alloc, alpha)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 100:
        time_alloc = min(120, tleft() * 0.12)
        h_new, score = subgrad_phase(best_h, target, time_alloc, alpha_init=80, eta_init=0.008)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 150:
        time_alloc = min(250, tleft() * 0.2)
        h_new, score = pairwise_cd(best_h, target, time_alloc, seed)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 100:
        time_alloc = min(150, tleft() * 0.15)
        h_new, score = fine_cd(best_h, target, time_alloc, seed + 1)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 80:
        time_alloc = min(120, tleft() * 0.12)
        h_new, score = lbfgs_phase(best_h, target, time_alloc, 500)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 100:
        time_alloc = min(150, tleft() * 0.15)
        h_new, score = structured_perturb_and_optimize(best_h, target, time_alloc, seed + 2)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 80:
        time_alloc = min(120, tleft() * 0.15)
        h_new, score = lbfgs_phase(best_h, target, time_alloc, 800)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 100:
        time_alloc = min(150, tleft() * 0.2)
        h_new, score = pairwise_cd(best_h, target, time_alloc, seed + 3)
        if score < best_score:
            best_score = score
            best_h = h_new

    if tleft() > 80:
        time_alloc = max(30, tleft() * 0.7)
        h_new, score = fine_cd(best_h, target, time_alloc, seed + 4)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
