from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def topk_objective(x, target, K, alpha):
    """Minimize sum of top K correlation values with log-sum-exp tail."""
    h = project_feasible(x, target)
    corr = compute_corr(h)
    # Sort in descending order
    sorted_corr = np.sort(corr)[::-1]
    top_k = sorted_corr[:K]
    # Objective: weighted sum emphasizing the max
    # Use: sum of top K + alpha * log-sum-exp of the rest
    if len(corr) > K:
        rest = sorted_corr[K:]
        shifted = alpha * (rest - sorted_corr[K-1])
        shifted = np.clip(shifted, -700.0, 0.0)
        tail = float(np.max(shifted)) + float(np.log(np.sum(np.exp(shifted - np.max(shifted))))) / alpha
    else:
        tail = 0.0
    return float(np.sum(top_k)) / K + tail

    """Objective using top-K + log-sum-exp tail for smooth max approximation."""
    h = project_feasible(x, _g_target)
    corr = compute_corr(h)
    n_corr = len(corr)
    K = _g_topk
    alpha = _g_alpha
    
    if K >= n_corr:
        return float(np.max(corr))
    
    # Get top K values
    idx = np.argpartition(corr, -K)[-K:]
    top_vals = np.sort(corr[idx])[::-1]
    
    # Log-sum-exp of all, but weighted to emphasize top
    # Use: alpha * (log-sum-exp(corr - top_vals[0]) ) / alpha + top_vals[0]
    # This approximates max but with contribution from all
    shifted = alpha * (corr - top_vals[0])
    shifted = np.clip(shifted, -700.0, 0.0)
    lse = top_vals[0] + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    return lse

    _g_target = target
    _g_alpha = alpha

        grad_mean = grad.mean()
        grad = grad - grad_mean  # Ensure update preserves sum

1. Add a trust region: ||h - h_t||_∞ ≤ Δ (or ||h - h_t||_2 ≤ Δ)
2. Or solve a sequence of LPs with decreasing Δ

Actually, the standard SCP approach is:
1. Start with h_0 (the initial solution)
2. At iteration t:
   a. Linearize at h_t
   b. Solve the resulting convex problem (LP in our case) to get h_{t+1}
   c. Check convergence: if ||h_{t+1} - h_t|| < ε, stop
   d. Optionally: take a step h_{t+1} = (1-α)*h_t + α*h_{t+1} with α ∈ (0,1] for stability
3. Return h_final

The advantage is that each subproblem is a simple LP that can be solved very accurately.
