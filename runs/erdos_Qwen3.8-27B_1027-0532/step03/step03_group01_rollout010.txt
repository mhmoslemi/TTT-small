import time

def fast_c5(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    g = 1.0 - h
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return float(np.max(corr)) * (2.0 / n)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = h_r.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h_r
    if current_sum > target_sum:
        ones_idx = np.where(h_r == 1)[0]
        need = int(current_sum - target_sum)
        order = np.argsort(h[ones_idx])[:need]
        h_r[ones_idx[order]] = 0.0
    else:
        zeros_idx = np.where(h_r == 0)[0]
        need = int(target_sum - current_sum)
        order = np.argsort(-h[zeros_idx])[:need]
        h_r[zeros_idx[order]] = 1.0
    return h_r

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time


def fast_c5(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    g = 1.0 - h
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return float(np.max(corr)) * (2.0 / n)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    current_sum = h_r.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h_r
    if current_sum > target_sum:
        ones_idx = np.where(h_r == 1)[0]
        need = int(current_sum - target_sum)
        order = np.argsort(h[ones_idx])[:need]
        h_r[ones_idx[order]] = 0.0
    else:
        zeros_idx = np.where(h_r == 0)[0]
        need = int(target_sum - current_sum)
        order = np.argsort(-h[zeros_idx])[:need]
        h_r[zeros_idx[order]] = 1.0
    return h_r


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def smooth_objective(x, alpha, target_sum, n_pts):
    n = n_pts
    N = 1
    while N < 2 * n:
        N *= 2
    g = 1.0 - x
    H = np.fft.rfft(x, N)
    G = np.fft.rfft(g, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    penalty = 50.0 * (np.sum(x) - target_sum) ** 2
    return (float(smooth_max) + float(penalty)) * (2.0 / n)

def smooth_refine(h_start, n, target, alphas, maxiter, time_limit=None, start_time=None):

            h_new = project_feasible(res.x, target)
            sc = fast_c5(h_new)
            if sc < curr_score - 1e-12:

                h_cur = h_new
