```python
def _sop_obj(h, dx, p=4):
    corr = np.correlate(h, 1-h, mode='full')
    return (np.sum(corr**p) ** (1.0/p)) * dx

def _sop_grad(h, dx, p=4):
    corr = np.correlate(h, 1-h, mode='full')
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = p * (corr[k] ** (p-1)) / (np.sum(corr**p)) ** ((p-1)/p) * dx
        # grad of corr[k] w.r.t. h
        lag = k - (n-1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[lag:lag+m] += w * (1 - h[:m])
                grad[:m] += w * (-h[lag:lag+m])
        else:
            j = -lag
            m = n - j
            if m > 0:
                grad[:m] += w * (1 - h[j:])
                grad[j:] += w * (-h[:m])
    return grad
```

```python
def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-13:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-13:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

    lo = y.min() - 1.0
    hi = y.max() + 1.0

        if top_m > 0:
            max_corr = np.max(corr)
            weights = np.exp(80.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_indices = np.argsort(weights)[::-1][:top_m]
            w_m = weights[top_indices].copy()
            w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_indices)):
                g += w_m[idx] * _grad_corr(y, top_indices[idx], dx)
        else:
            m = np.max(corr)
            w = np.exp(alpha_ls * (corr - m))
            w = w / w.sum()
            g = np.zeros(n)
            for k in range(2*n - 1):
                if w[k] < 1e-8:
                    continue
                g += w[k] * _grad_corr(y, k, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _block_cd(h0, n, target_sum, max_rounds=15, deadline=None):
    h = h0.copy()
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(0, n - 1, 2):
            s_pair = h[p] + h[p+1]
            lo = max(0.0, s_pair - 1.0)
            hi = min(1.0, s_pair)
            if hi - lo < 1e-12:
                continue
            old_c = _c5_of(h)
            best_t = h[p]
            best_c = old_c
            for t_val in np.linspace(lo, hi, 15):
                h[p] = t_val
                h[p+1] = s_pair - t_val
                c = _c5_of(h)
                if c < best_c:
                    best_c = c
                    best_t = t_val
            h[p] = best_t
            h[p+1] = s_pair - best_t
            if best_t != h[p] or best_c < old_c - 1e-14:
                improved = True
        if not improved:
            break
    return h

def _block_cd_overlap(h0, n, target_sum, max_rounds=15, deadline=None):
    h = h0.copy()
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for p in range(1, n - 1, 2):
            s_pair = h[p] + h[p+1]
            lo = max(0.0, s_pair - 1.0)
            hi = min(1.0, s_pair)
            if hi - lo < 1e-12:
                continue
            old_c = _c5_of(h)
            best_t = h[p]
            best_c = old_c
            for t_val in np.linspace(lo, hi, 15):
                h[p] = t_val
                h[p+1] = s_pair - t_val
                c = _c5_of(h)
                if c < best_c:
                    best_c = c
                    best_t = t_val
            h[p] = best_t
            h[p+1] = s_pair - best_t
            if best_c < old_c - 1e-14:
                improved = True
        if not improved:
            break
    return h

def _bin_cd(h0, n, target_sum, max_rounds=15, deadline=None):
    h = h0.copy()
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _swap_refine(h0, n, target_sum, max_rounds=5, deadline=None, rng=None):
    h = h0.copy()
    if rng is None:
        rng = np.random.RandomState(0)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        n_swaps = max(n // 4, 10)
        for _ in range(n_swaps):
            i, j = rng.choice(n, 2, replace=False)
            h[i], h[j] = h[j], h[i]
            c_new = _c5_of(h)
            c_old = _c5_of(h0) if not improved else _c5_of(h)
            if c_new > _c5_of(h) - 1e-14:
                h[i], h[j] = h[j], h[i]
            else:
                improved = True
        if not improved:
            break
    return h

def _perturb_and_refine(h0, n, target_sum, rng, deadline, n_trials=10):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for trial in range(n_trials):
        if time.time() > deadline:
            break
        scale = 0.15 + 0.1 * (n_trials - trial) / n_trials
        h_pert = h0 + scale * (rng.rand(n) * 2 - 1)
        h_pert = _fix_sum(np.clip(h_pert, 0, 1), target_sum)
        h_n, c5_n = _nesterov(h_pert, n, target_sum, n_iters=80, alpha0=0.25,
                              alpha_ls=100, top_m=3, deadline=deadline)
        if c5_n < best_c5:
            h_sl = _ls_slsqp(h_n, n, target_sum, 500, maxiter=150)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_list = []
    for d in [0, 10, -10, 20, -20, 30, -30, 40, -40, 60, 80]:
        nn = n_init + d
        if 60 <= nn <= 400 and nn % 2 == 0:
            n_list.append(nn)
    for nn in [100, 120, 160, 200, 240, 280, 320]:
        n_list.append(nn)
    n_list = sorted(set(n_list))

    phase1_budget = deadline - 120

    for n in n_list:
        if time.time() > phase1_budget:
            break
        target_sum = n / 2.0

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.arange(n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0, 1), target_sum)

        c5_s = _c5_of(h_start)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_start.copy()
            best_n = n

        h_curr = h_start.copy()

        # Nesterov with smooth log-sum-exp gradient
        if time.time() < phase1_budget - 10:
            h_curr, c5_1 = _nesterov(h_curr, n, target_sum, n_iters=120,
                                      alpha0=0.35, alpha_ls=60, deadline=phase1_budget)
        # Nesterov with top-m gradient
        if time.time() < phase1_budget - 5:
            h_curr, c5_2 = _nesterov(h_curr, n, target_sum, n_iters=100,
                                      alpha0=0.25, alpha_ls=100, top_m=3,
                                      deadline=phase1_budget)

        # SoP SLSQP p=4
        if time.time() < phase1_budget - 5:
            h_sop = _sop_slsqp(h_curr, n, target_sum, p=4, maxiter=150,
                               deadline=phase1_budget)
            c5_sop = _c5_of(h_sop)
            if c5_sop < _c5_of(h_curr):
                h_curr = h_sop

        # SoP SLSQP p=8
        if time.time() < phase1_budget - 3:
            h_sop2 = _sop_slsqp(h_curr, n, target_sum, p=8, maxiter=100,
                                deadline=phase1_budget)
            c5_sop2 = _c5_of(h_sop2)
            if c5_sop2 < _c5_of(h_curr):
                h_curr = h_sop2

        # Log-sum-exp SLSQP escalating
        for alpha in [150, 400, 1000, 2500]:
            if time.time() > phase1_budget - 2:
                break
            h_new = _ls_slsqp(h_curr, n, target_sum, alpha, maxiter=200)
            c5_new = _c5_of(h_new)
            if c5_new <= _c5_of(h_curr) + 1e-12:
                h_curr = h_new
            else:
                break

        # Block coordinate descent (even pairs)
        if time.time() < phase1_budget - 2:
            h_curr = _block_cd(h_curr, n, target_sum, max_rounds=10, deadline=phase1_budget)
        # Block coordinate descent (odd pairs)
        if time.time() < phase1_budget - 1:
            h_curr = _block_cd_overlap(h_curr, n, target_sum, max_rounds=10, deadline=phase1_budget)

        # Binary coordinate descent
        if time.time() < phase1_budget:
            h_curr = _bin_cd(h_curr, n, target_sum, max_rounds=12, deadline=phase1_budget)

        c5_final = _c5_of(h_curr)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_curr.copy()
            best_n = n

    # Phase 2: Focus on best and nearby grid sizes with more intensive optimization
    n_best = best_n
    target_best = n_best / 2.0
    h_focus = best_h.copy()

    # More perturbations on best grid
    if time.time() < deadline - 50:
        h_p, c5_p = _perturb_and_refine(h_focus, n_best, target_best, rng,
                                         deadline - 50, n_trials=8)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p.copy()
            h_focus = h_p.copy()

    # Try n_best +/- small deltas with intensive optimization
    for ndelta in [-4, 4, -8, 8]:
        nn = n_best + ndelta
        if nn < 60 or nn % 2 != 0 or time.time() > deadline - 40:
            continue
        ts = nn / 2.0
        h_interp = np.interp(np.arange(nn), np.arange(n_best), best_h)
        h_try = _fix_sum(np.clip(h_interp, 0, 1), ts)
        c5_try = _c5_of(h_try)
        if c5_try < best_c5 + 0.005:
            h_try, c5_t1 = _nesterov(h_try, nn, ts, n_iters=100, alpha0=0.3,
                                      alpha_ls=80, top_m=3, deadline=deadline - 40)
            if time.time() < deadline - 35:
                h_try = _ls_slsqp(h_try, nn, ts, 800, maxiter=200)
            if time.time() < deadline - 30:
                h_try = _block_cd(h_try, nn, ts, max_rounds=10, deadline=deadline - 30)
            if time.time() < deadline - 25:
                h_try = _bin_cd(h_try, nn, ts, max_rounds=10, deadline=deadline - 25)
            c5_try = _c5_of(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                best_n = nn
                n_best = nn

    # Phase 3: Final intensive polishing
    n_best = best_n
    target = n_best / 2.0
    h = best_h.copy()

    # Escalating log-sum-exp SLSQP
    for alpha in [1000, 3000, 8000, 20000, 50000]:
        if time.time() > deadline - 12:
            break
        h_new = _ls_slsqp(h, n_best, target, alpha, maxiter=300)
        c5_new = _c5_of(h_new)
        if c5_new <= _c5_of(h) + 1e-12:
            h = h_new

    # SoP final
    if time.time() < deadline - 10:
        h_sop = _sop_slsqp(h, n_best, target, p=4, maxiter=200, deadline=deadline - 8)
        if _c5_of(h_sop) < _c5_of(h):
            h = h_sop
        if time.time() < deadline - 8:
            h_sop2 = _sop_slsqp(h, n_best, target, p=8, maxiter=150, deadline=deadline - 6)
            if _c5_of(h_sop2) < _c5_of(h):
                h = h_sop2

    # Block CD
    if time.time() < deadline - 8:
        h = _block_cd(h, n_best, target, max_rounds=15, deadline=deadline - 6)
    if time.time() < deadline - 6:
        h = _block_cd_overlap(h, n_best, target, max_rounds=15, deadline=deadline - 4)

    # Binary CD
    if time.time() < deadline - 5:
        h = _bin_cd(h, n_best, target, max_rounds=20, deadline=deadline - 3)

    # One more SLSQP at very high alpha
    if time.time() < deadline - 4:
        h_new = _ls_slsqp(h, n_best, target, 100000, maxiter=200)
        if _c5_of(h_new) < _c5_of(h):
            h = h_new

    # Final block + bin CD
    if time.time() < deadline - 3:
        h = _block_cd(h, n_best, target, max_rounds=10, deadline=deadline - 2)
    if time.time() < deadline - 2:
        h = _bin_cd(h, n_best, target, max_rounds=15, deadline=deadline - 1)

    h = _fix_sum(h, target)
    final_c5 = _c5_of(h)

    return (h, final_c5, n_best)
```
