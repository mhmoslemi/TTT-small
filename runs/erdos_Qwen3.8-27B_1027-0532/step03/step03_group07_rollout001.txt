def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

    return grad * dx

import time
import numpy as np
from scipy.optimize import minimize

_g_ts = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_vals(h):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / len(h))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)

                sub = min(-diff, x[i])
                x[i] -= sub

    return x

def _grad_corr(h, k_idx, dx):
    n = len(h)
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _weighted_grad(h, alpha, top_m=15):
    corr = _corr_vals(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400, 400)
    exps = np.exp(shifted)
    total = exps.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / total
    top_idx = np.argsort(weights)[::-1][:top_m]
    w_n = weights[top_idx].copy()
    w_n = w_n / w_n.sum()
    dx = 2.0 / len(h)
    g = np.zeros(len(h))
    for idx in range(len(top_idx)):
        g += w_n[idx] * _grad_corr(h, top_idx[idx], dx)
    return g

def _smooth_val(h, alpha):
    corr = _corr_vals(h)
    m = np.max(corr)
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -400, 400)
    return m + np.log(np.exp(shifted).sum()) / alpha

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha, step_scale, deadline=None):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        g = _weighted_grad(y, alpha, top_m=15)
        step = step_scale / (1.0 + 0.1 * np.sqrt(float(it + 1)))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0

        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _paired_cd(h0, n, target_sum, max_rounds=30, n_trials_per_round=300, rng=None):
    h = h0.copy()
    current = _c5_of(h)
    if rng is None:
        rng = np.random.RandomState(0)
    deltas = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    for _ in range(max_rounds):
        improved = False
        for _ in range(n_trials_per_round):
            i = rng.randint(0, n)
            j = rng.randint(0, n)
            if i == j:
                continue
            d = deltas[rng.randint(0, len(deltas))]
            for sign in [1, -1]:
                ni = h[i] + sign * d
                nj = h[j] - sign * d
                if ni < -1e-9 or ni > 1.0 + 1e-9 or nj < -1e-9 or nj > 1.0 + 1e-9:
                    continue
                h[i] = ni
                h[j] = nj
                c = _c5_of(h)
                if c < current - 1e-13:
                    current = c
                    improved = True
                else:
                    h[i] -= sign * d
                    h[j] += sign * d
        if not improved:
            break
    return h, current
