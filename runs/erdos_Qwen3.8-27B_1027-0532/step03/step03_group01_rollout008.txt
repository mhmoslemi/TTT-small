def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def find_worst_lags(corr, n, top_k=5, exclude_center=True):
    center = n - 1
    c = corr.copy()
    if exclude_center:
        c[center] = 0.0
    threshold = c.max() * 0.9
    candidates = np.where(c >= threshold)[0]
    if len(candidates) == 0:
        candidates = np.array([center + 1])
    order = c[candidates][::-1].argsort()
    selected = candidates[order[:top_k]]
    lags = selected - center
    lags = lags[lags != 0]
    if len(lags) == 0:
        lags = np.array([1])
    return lags

def subgradient_continuous(h_start, target_sum, max_iters, step_init, step_final, rng, time_limit, start_time):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        ratio = i / max(max_iters - 1, 1)
        step = step_init * (step_final / step_init) ** ratio
        
        corr = get_corr(h)
        center = n - 1
        top_lags = find_worst_lags(corr, n, top_k=5)
        
        grad = np.zeros(n)
        for k in top_lags:
            k = int(k)
            g = np.zeros(n)
            for i_idx in range(n):
                val = 0.0
                if 0 <= i_idx - k < n:
                    val += (1.0 - h[i_idx - k])
                if 0 <= i_idx + k < n:
                    val -= h[i_idx + k]
                g[i_idx] = val
            grad += g
        grad /= len(top_lags)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    cons_fun_target = target
    cons_fun_n = n
    
    def obj_smooth(h_in, alpha):
        corr = np.correlate(h_in, 1.0 - h_in, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_f(h_in):
        return np.sum(h_in) - cons_fun_target
    
    def cons_j(h_in):
        return np.ones(cons_fun_n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(obj_smooth, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def fast_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 100
    
    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a_idx = ones[rng.randint(len(ones))]
        b_idx = zeros[rng.randint(len(zeros))]
        h[a_idx] = 0.0
        h[b_idx] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[a_idx] = 1.0
            h[b_idx] = 0.0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5
