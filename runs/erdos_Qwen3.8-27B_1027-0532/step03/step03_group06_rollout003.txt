            h_pert = project_feasible(h_pert, target_init)
        elif ptype == 3:
            h_pert = partial_binarize(best_h, target_init, rng.uniform(0.1, 0.4))
        else:
            seg_len = rng.randint(10, max(11, n_init // 5))
            a = rng.randint(0, n_init - seg_len)
            b = rng.randint(0, n_init - seg_len)
            if abs(a - b) > seg_len:
                h_pert = best_h.copy()
                tmp = h_pert[a:a + seg_len].copy()
                h_pert[a:a + seg_len] = h_pert[b:b + seg_len]
                h_pert[b:b + seg_len] = tmp
                h_pert = project_feasible(h_pert, target_init)
            else:
                h_pert = project_feasible(best_h + rng.randn(n_init) * 0.01, target_init)

        alpha_seq = [20, 80, 250, 600]
        h_cur = h_pert
        for alpha in alpha_seq:
            if tleft() < 20:
                break
            h_new = lbfgs_stage(h_cur, n_init, target_init, alpha, 200, 1e-14, tleft)
            sc = compute_c5(h_new)
            if sc <= compute_c5(h_cur):
                h_cur = h_new

        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur

        if tleft() > 50:
            h_ds, sc_ds = direct_subgrad(best_h, n_init, target_init, 50, 0.002, tleft)
            if sc_ds < best_score:
                best_score = sc_ds
                best_h = h_ds

    for n_try in [500, 600, 800]:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        sc_interp = compute_c5(h_interp)
        if sc_interp < best_score:
            best_score = sc_interp
            best_h = h_interp

        for alpha in [15, 50, 150, 400, 1000]:
            if tleft() < 30:
                break
            h_new = lbfgs_stage(best_h, n_try, target_try, alpha, 250, 1e-14, tleft)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

        h_ds, sc_ds = direct_subgrad(best_h, n_try, target_try, 40, 0.001, tleft)
        if sc_ds < best_score:
            best_score = sc_ds
            best_h = h_ds

    n_use = len(best_h)
    target_use = n_use / 2.0

    for alpha in [2000, 5000]:
        if tleft() < 25:
            break
        h_new = lbfgs_stage(best_h, n_use, target_use, alpha, 500, 1e-16, tleft)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_ds, sc_ds = direct_subgrad(best_h, n_use, target_use, 80, 0.0005, tleft)
    if sc_ds < best_score:
        best_score = sc_ds
        best_h = h_ds

    if tleft() > 30:
        h_mom, sc_mom = momentum_optimize(best_h, n_use, target_use, 3000, 200, 0.0003, tleft)
        if sc_mom < best_score:
            best_score = sc_mom
            best_h = h_mom

    if tleft() > 25:
        for _ in range(5):
            if tleft() < 20:
                break
            amp = rng.uniform(0.0005, 0.003)
            h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
            h_new = lbfgs_stage(h_pert, n_use, target_use, 5000, 300, 1e-16, tleft)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
            h_ds2, sc_ds2 = direct_subgrad(h_new, n_use, target_use, 30, 0.0003, tleft)
            if sc_ds2 < best_score:
                best_score = sc_ds2
                best_h = h_ds2

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
