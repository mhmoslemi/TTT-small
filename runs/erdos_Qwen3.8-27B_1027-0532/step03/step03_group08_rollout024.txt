import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_overlap(h, k_idx):
    """Gradient of overlap at correlation index k_idx w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _weighted_grad(h, top_k_idx, weights):
    """Weighted gradient from multiple lags."""
    n = len(h)
    grad = np.zeros(n)
    for w, idx in zip(weights, top_k_idx):
        grad += w * _grad_overlap(h, idx)
    return grad

def _nesterov_topm(h0, n, target_sum, n_iters, alpha0, M, beta, deadline):
    """Nesterov-accelerated method with top-M subgradients."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    stall = 0
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        # Compute all overlaps
        overlaps = _all_overlaps(y)
        
        # Get top M lags
        top_idx = np.argsort(overlaps)[::-1][:M]
        top_vals = overlaps[top_idx]
        
        # Weights: softmax-like
        max_v = top_vals[0]
        weights = np.exp(beta * (top_vals - max_v))
        weights /= weights.sum()
        
        # Compute weighted gradient
        g = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            g += w * _grad_overlap(y, idx)
        
        # Nesterov step
        step = alpha0 / (1.0 + 0.01 * it)
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val - 1e-10:
            best_val = cv
            best = x.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                # Reset momentum and perturb
                y = x + 0.01 * (np.random.rand(n) - 0.5)
                y = _project(y, target_sum)
                t = 1.0
                stall = 0
    
    return best, best_val

def _coord_descent_binary(h0, n, target_sum, max_rounds):
    """Coordinate descent trying 0 and 1 for each position."""
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if c0 < co - 1e-12:
                h[j] = 0.0
                improved = True
            elif c1 < co - 1e-12:
                h[j] = 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds):

    h = h0.copy()

                    best_v = v

                improved = True
        if not improved:
            break
    return h

    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * (2.0 / n)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        ws = np.exp(alpha * (corr - m))
        ws /= ws.sum()
        g = np.zeros(n)
        # Only compute gradient for top few terms
        top_m = np.argsort(ws)[::-1][:min(10, len(ws))]
        for idx in top_m:
            g += ws[idx] * _grad_overlap(h, idx)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, args=(),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except Exception:
        return h0

def _bottleneck_swap(h0, n, target_sum, deadline):
    """Try to reduce the worst lag by swapping values."""
    h = h0.copy()
    best_c = _c5_of(h)
    
    for _ in range(20):
        if time.time() > deadline:
            break

        # or increase h[i+lag] where h[i] is large.

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_overlap(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _multi_grad(h, top_idx, weights):
    n = len(h)
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        grad += w * _grad_overlap(h, idx)
    return grad

def _nesterov_topm(h0, n, target_sum, n_iters, alpha0, M, beta, deadline, alpha_decay=0.999):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    stall = 0
    alpha = alpha0

    for it in range(n_iters):
        if time.time() > deadline:
            break
        overlaps = _all_overlaps(y)
        top_idx = np.argsort(overlaps)[::-1][:M]
        top_vals = overlaps[top_idx]
        max_v = top_vals[0]
        weights = np.exp(beta * (top_vals - max_v))
        sw = weights.sum()
        if sw > 1e-300:
            weights /= sw
        else:
            weights = np.zeros(M)
            weights[0] = 1.0
        g = _multi_grad(y, top_idx, weights)
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        alpha *= alpha_decay
        cv = _c5_of(x)

            best_val = cv
            best = x.copy()

def _coord_descent_binary(h0, n, target_sum, max_rounds):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if c0 < co - 1e-13:
                h[j] = 0.0
                improved = True
            elif c1 < co - 1e-13:
                h[j] = 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds, nvals=7):
    h = h0.copy()
    if nvals <= 4:
        vals = [0.0, 0.5, 1.0]
    elif nvals <= 6:
        vals = [0.0, 0.25, 0.5, 0.75, 1.0]
    else:
        vals = np.linspace(0.0, 1.0, nvals).tolist()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _smooth_corr_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / n)

def _smooth_corr_grad(h, alpha, n, top_M=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    ws = np.exp(alpha * (corr - m))
    total = ws.sum()
    if total < 1e-300:
        ws = np.zeros(len(corr))
        ws[np.argmax(corr)] = 1.0
    else:
        ws /= total
    top_idx = np.argsort(ws)[::-1][:top_M]
    g = np.zeros(n)
    for idx in top_idx:
        g += ws[idx] * _grad_overlap(h, idx)
    return g * (2.0 / n) * alpha / (alpha * (2.0 / n))

def _slsqp_opt(h0, n, target_sum, alpha, maxiter, top_M=10):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
        return val * dx

    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        ws = np.exp(alpha * (corr - m))
        total = ws.sum()
        if total < 1e-300:
            ws = np.zeros(len(corr))
            ws[np.argmax(corr)] = 1.0
        else:
            ws /= total
        top_idx = np.argsort(ws)[::-1][:top_M]
        g = np.zeros(n)
        for idx in top_idx:
            g += ws[idx] * _grad_overlap(h, idx)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

        h_lt, c5_lt = _lag_targeted_opt(best_h, best_n, target_best, deadline - 100, 4)
        if c5_lt < best_c5:
            best_c5 = c5_lt
            best_h = h_lt.copy()

    if time.time() < deadline - 80:
        h_sls2 = _slsqp_opt(best_h, best_n, target_best, 300, 800, 12)
        c5_sls2 = _c5_of(h_sls2)
        if c5_sls2 < best_c5:
            best_c5 = c5_sls2
            best_h = h_sls2.copy()

    if time.time() < deadline - 50:
        h_nest3 = _nesterov_topm(best_h, best_n, target_best, 500, 0.2, 8, 80.0, deadline - 40)
        c5_nest3 = _c5_of(h_nest3)
        if c5_nest3 < best_c5:
            best_c5 = c5_nest3
            best_h = h_nest3.copy()

    if time.time() < deadline - 30:
        h_cd2 = _coord_descent_binary(best_h, best_n, target_best, 15)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()

    if time.time() < deadline - 15:
        h_cd3 = _coord_descent_frac(best_h, best_n, target_best, 10, 9)
        c5_cd3 = _c5_of(h_cd3)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    if time.time() < deadline - 5:
        h_sls3 = _slsqp_opt(best_h, best_n, target_best, 350, 3000, 15)
        c5_sls3 = _c5_of(h_sls3)
        if c5_sls3 < best_c5:
            best_c5 = c5_sls3
            best_h = h_sls3.copy()

    if time.time() < deadline - 2:
        h_cd4 = _coord_descent_frac(best_h, best_n, target_best, 8, 11)
        c5_cd4 = _c5_of(h_cd4)
        if c5_cd4 < best_c5:
            best_c5 = c5_cd4
            best_h = h_cd4.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
