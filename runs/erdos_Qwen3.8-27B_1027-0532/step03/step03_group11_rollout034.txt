            grad[r:] -= wk * dx * h[:m]
    return grad

def replica_exchange_search(h_init, time_limit, start_time, rng, n_replicas=5):
    n = len(h_init)
    
    # Initialize replicas
    replicas = np.zeros((n_replicas, n))
    replicas[0] = h_init
    for r in range(1, n_replicas):
        # Perturb the initial solution
        h_pert = h_init.copy()
        nflip = int(n * 0.1 * (r + 1))
        flip_idx = rng.choice(n, size=min(nflip, n//2), replace=False)
        h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        replicas[r] = h_pert
    
    scores = np.array([compute_c5(replicas[r]) for r in range(n_replicas)])
    best_h = replicas[int(np.argmin(scores))].copy()
    best_score = float(scores.min())
    
    # Temperatures (thresholds for great deluge at each level)
    # Level 0: strict (best), Level K-1: very loose
    thresholds = np.zeros(n_replicas)
    thresholds[0] = scores[0] + 0.001
    for r in range(1, n_replicas):
        thresholds[r] = scores[r] + 0.01 * (r + 1)
    
    step = 0
    swap_period = 500
    last_swap = 0
    
    while time.time() - start_time < time_limit:
        # Pick a replica
        r = int(rng.integers(n_replicas))
        
        # Make a move on replica r
        move_result = make_move(replicas[r], rng, n)
        if move_result is None:
            step += 1
            continue
        h_new, undo_info = move_result
        replicas[r] = h_new
        new_score = compute_c5(h_new)
        
        # Accept if below threshold
        if new_score <= thresholds[r]:
            scores[r] = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # Undo
            replicas[r] = undo_info['old_h']
        
        # Periodic swap
        if step - last_swap >= swap_period and step > 0:
            last_swap = step
            # Try to swap adjacent replicas
            r1 = int(rng.integers(n_replicas - 1))
            r2 = r1 + 1
            # Metropolis swap criterion
            delta = (thresholds[r1] - scores[r2]) - (thresholds[r2] - scores[r1])
            # Actually for replica exchange, the criterion is:
            # Accept swap with prob min(1, exp(-(E_new - E_old) / T))
            # where E = score / (1/T) = score * T
            # Let me use a simpler criterion:
            # Accept if the swap reduces the "free energy" or with some probability
            # Simple: accept if scores[r1] > scores[r2] (put better solution in higher temp)
            # Or use: accept with prob exp(-(scores[r2]-scores[r1]) * 10)
            
            # Actually, standard replica exchange:
            # We have systems at temperatures T_1 > T_2 > ... > T_K
            # Energies E_1, E_2, ..., E_K
            # Try to swap systems i and i+1
            # Accept with prob min(1, exp(-T_i * (E_{i+1} - E_i) + T_{i+1} * (E_i - E_{i+1})))
            # = min(1, exp((T_{i+1} - T_i) * (E_i - E_{i+1})))
            
            # In our case, "temperature" is the threshold (higher = more accepting)
            # "Energy" is the score
            # We want the best solutions at the lowest temperatures (strict thresholds)
            
            # Let me just use: swap if scores improve (put better at stricter level)
            if scores[r1] > scores[r2]:
                # Better solution is at looser temp - swap them
                replicas[r1], replicas[r2] = replicas[r2], replicas[r1]
                scores[r1], scores[r2] = scores[r2], scores[r1]
            elif rng.random() < np.exp(-(scores[r1] - scores[r2]) * 50):
                replicas[r1], replicas[r2] = replicas[r2], replicas[r1]
                scores[r1], scores[r2] = scores[r2], scores[r1]
            
            # Occasionally reset a poor replica
            worst = int(np.argmax(scores))
            if scores[worst] > best_score + 0.02:
                h_reset = best_h.copy()
                nflip = int(n * 0.2)
                flip_idx = rng.choice(n, size=min(nflip, n//2), replace=False)
                h_reset[flip_idx] = 1.0 - h_reset[flip_idx]
                replicas[worst] = h_reset
                scores[worst] = compute_c5(h_reset)
        
        step += 1
        if step % 2000 == 0 and time.time() - start_time >= time_limit:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_from_continuous(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def steepest_binary(h, max_rounds, time_limit, start_time):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        improved = False
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                if sc < current:
                    current = sc
                    improved = True
                    ones.remove(oi)
                    zeros.remove(zi)
                    break
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, current

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def steepest_binary(h, max_rounds, time_limit, start_time):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        improved = False
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                if sc < current:
                    current = sc
                    improved = True
                    ones.remove(oi)
                    zeros.remove(zi)
                    break
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, current
