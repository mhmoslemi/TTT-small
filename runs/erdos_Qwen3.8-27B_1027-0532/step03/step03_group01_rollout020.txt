```python
import numpy as np
from scipy.optimize import minimize
from scipy.optimize import differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def smooth_obj(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    scaled_max = np.max(scaled)
    obj = (scaled_max + np.log(np.sum(np.exp(scaled - scaled_max)))) / alpha
    sum_h = np.sum(h)
    obj += mu * (sum_h - n * 0.5) ** 2
    return obj

def smooth_grad(h, alpha, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * dx * corr
    scaled_max = np.max(scaled)
    exp_vals = np.exp(scaled - scaled_max)
    p = exp_vals / np.sum(exp_vals)
    
    g = np.zeros(n)
    for j in range(n):
        l_min = max(-(n-1), -j)
        l_max = min(n-1, n-1-j)
        if l_min <= l_max:
            lr = np.arange(l_min, l_max+1)
            g[j] += np.dot(p[lr + (n-1)], 1.0 - h[j + lr])
        l_min_b = max(-(n-1), j-(n-1))
        l_max_b = min(n-1, j)
        if l_min_b <= l_max_b:
            lr_b = np.arange(l_min_b, l_max_b+1)
            g[j] -= np.dot(p[lr_b + (n-1)], h[j - lr_b])
    g *= dx
    g += 2.0 * mu * (np.sum(h) - n * 0.5)
    return g

def smooth_obj_grad(h, alpha, mu):
    return smooth_obj(h, alpha, mu), smooth_grad(h, alpha, mu)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - int(current_sum))
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:int(final_sum - target_sum)]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:int(target_sum - final_sum)]] = 1
    return h_r.astype(float)

def sa_optimize(h_start, target_sum, max_iters, t_init, t_final, rng,
                time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(5000, max_iters // 10)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]

    for i in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        move_type = rng.randint(3)
        h_new = h.copy()
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            h_new[ones_cache[idx1[0]]] = 0
            h_new[ones_cache[idx1[1]]] = 0
            h_new[zeros_cache[idx0[0]]] = 1
            h_new[zeros_cache[idx0[1]]] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            # Proximity-based move
            dist = np.abs(zeros_cache.astype(float) - i1) + rng.random(len(zeros_cache)) * 3.0
            i0 = zeros_cache[np.argmin(dist)]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True

        if not moved:
            continue

        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5

def lbfgsb_refine(h_start, n, target, alphas, mu, maxiter, time_limit=None, start_time=None):
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj_grad, h_cur, args=(alpha, mu),
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                                    'maxfun': maxiter * 3})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    
    # Phase 1: L-BFGS-B refinement with smooth max objective
    if tleft() > 30:
        alphas = [5, 15, 50, 150, 500, 1500, 5000, 15000]
        h_ref, sc = lbfgsb_refine(best_h, n_init, target_init, alphas, 
                                   mu=50.0, maxiter=300, 
                                   time_limit=tleft()-30, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 2: Binary SA from the refined solution
    if tleft() > 50:
        h_bin = project_binary(best_h, target_init)
        h_sa, c5_sa = sa_optimize(h_bin, target_init, 100000, 0.015, 0.00005,
                                   rng, time_limit=min(200.0, tleft()-30), start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa
    
    # Phase 3: L-BFGS-B refinement of SA result
    if tleft() > 30:
        h_ref, sc = lbfgsb_refine(best_h, n_init, target_init, 
                                   [100, 500, 2000, 10000], mu=50.0, maxiter=200,
                                   time_limit=tleft()-20, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref
    
    # Phase 4: Multi-resolution search
    n_candidates = [400, 500, 700, 800, 1000]
    n_candidates = [n for n in n_candidates if n != n_init]
    
    for n_try in n_candidates:
        if tleft() < 40:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        
        # L-BFGS-B at this resolution
        h_ref, sc = lbfgsb_refine(h_interp, n_try, target_try,
                                   [20, 100, 500, 2000], mu=30.0, maxiter=200,
                                   time_limit=min(80.0, tleft()-30), start_time=start_time)
        
        if sc < best_score:
            best_score = sc
            best_h = h_ref.copy()
        
        # Try SA at this resolution too
        if tleft() > 30:
            h_bin_try = project_binary(h_ref, target_try)
            h_sa_try, c5_sa_try = sa_optimize(h_bin_try, target_try, 50000, 0.012, 0.0001,
                                               rng, time_limit=min(60.0, tleft()-20), start_time=start_time)
            if c5_sa_try < best_score:
```
