import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)))

def project_feasible(h, target_sum):
    """Project h onto {x: 0≤x≤1, sum(x)=target_sum}"""
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_lags(h, n):
    """Compute f_d(h) for d = 0, 1, ..., n-1"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # The full correlation has length 2n-1
    # Index n-1 corresponds to lag 0
    # Index n-1+d corresponds to lag d (positive)
    # Index n-1-d corresponds to lag -d (negative)
    # We want f_d for d=0..n-1, which are at indices n-1, n, ..., 2n-2
    # And by symmetry f_d = f_{-d}, so lags n-1-d (d=1..n-1) are same as lags d
    # So f_d for d=0..n-1 are at indices n-1, n, ..., 2n-2
    return corr[n-1:]  # This gives f_0, f_1, ..., f_{n-1}

def subgradient_active(h, n, active_dags, target):
    """Compute subgradient of (1/M) * sum_{d in active} f_d(h)"""
    grad = np.zeros(n)
    for d in active_dags:
        if d == 0:
            # f_0(h) = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # grad = 1 - 2*h
            grad += (1.0 - 2.0 * h)
        else:
            grad[0:n-d] += 1.0
            grad[0:n-d] -= h[d:n]
            grad[d:n] -= h[0:n-d]
    grad /= len(active_dags)
    return grad

def lse_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha

def slsqp_refine(h, n, target, alphas, maxiter, ftol):
    """SLSQP refinement with log-sum-exp smoothing"""
    bounds = [(0.0, 1.0)] * n
    eq_const = n / 2.0
    
    def cons_eq(x):
        return np.sum(x) - eq_const
    def cons_jac(x):
        return np.ones(n)
    
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    for a in alphas:
        def obj(x, a=a):
            return lse_obj(x, a)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur
