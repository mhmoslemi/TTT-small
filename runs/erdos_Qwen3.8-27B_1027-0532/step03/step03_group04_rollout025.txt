            h = np.clip(h, 0.0, 1.0)
    return h


def active_set_lp(h, n, target, time_limit, top_m_init=30, top_m_max=200, damping_init=0.6):
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = project_feasible(h.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    top_m = top_m_init
    damping = damping_init
    iter_count = 0

    while time.time() - start < time_limit:
        iter_count += 1
        if iter_count > 80000:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_m >= 2 * n - 1:
            top_indices = np.arange(2 * n - 1)
        else:
            top_indices = np.argpartition(corr, -top_m)[-top_m:]

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for idx in top_indices:
            k = int(idx) - n_half
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except Exception:
            break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            if damping < 0.85:
                damping = min(0.85, damping + 0.005)
        else:
            if iter_count % 100 == 0:
                damping = max(0.3, damping * 0.9)

        if iter_count % 50 == 0 and top_m < top_m_max:
            top_m = min(top_m_max, int(top_m * 1.2) + 1)

    return best_h, best_c5


def nesterov_subgradient(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = project_feasible(h.copy(), target)
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = 1e-6

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        beta = (1.0 + np.sqrt(1.0 + 4.0 * t0 * t0)) / 2.0
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t0 * t0)) / 2.0
        theta = (1.0 + beta) / (1.0 + t_new)

        y = h + (t0 / t_new) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        k = k_star

        grad = np.zeros(n)
        ar = np.arange(n)
        idx1 = (ar - k >= 0) & (ar - k < n)
        idx2 = (ar + k >= 0) & (ar + k < n)
        grad[idx1] += (1.0 - y[ar[idx1] - k])
        grad[idx2] -= y[ar[idx2] + k]
        grad *= dx

        step_size = 2.0 * t0 / (t0 + 1.0) * 0.1
        h_new = h - step_size * grad
        h_new = project_feasible(h_new, target)

        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h_new.copy()
        h_prev = h
        h = h_new
        t0 = t_new

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def subgradient_simple(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h_cur = project_feasible(h.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    li = []
    gs = []

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        k = k_star

        grad = np.zeros(n)
        ar = np.arange(n)
        idx1 = (ar - k >= 0) & (ar - k < n)
        idx2 = (ar + k >= 0) & (ar + k < n)
        grad[idx1] += (1.0 - h_cur[ar[idx1] - k])
        grad[idx2] -= h_cur[ar[idx2] + k]
        grad *= dx

        li.append(grad)
        gs.append(best_c5 - compute_c5(h_cur))

        L = len(li)
        stack_g = np.stack(li)
        stack_l = np.array(gs)
        A = stack_g @ stack_g.T + 1e-10 * np.eye(L)
        b = -stack_l
        try:
            lam_vec = np.linalg.solve(A, b)
            lam_vec = np.maximum(lam_vec, 0)
            lam_vec = lam_vec / (np.sum(lam_vec) + 1e-30)
            step_size = np.dot(lam_vec, stack_l) / (np.linalg.norm(np.dot(lam_vec, stack_g))**2 + 1e-30)
        except Exception:
            step_size = 0.01
        
        step_size = min(step_size, 0.5)
        agg_grad = np.dot(lam_vec, stack_g) if L > 1 else grad
        h_try = h_cur - step_size * agg_grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

        if L > 200:
            li = li[-100:]
            gs = gs[-100:]

    return best_h, best_c5


def try_binary_round(h, n, target):
    threshold = 0.5
    h_bin = (h >= threshold).astype(float)
    h_bin = project_feasible(h_bin, target)
    return h_bin


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    if tleft() > 400:
        t_alloc = min(500, tleft() * 0.45)
        h_lp, c5_lp = active_set_lp(h_init, n, target, t_alloc,
                                     top_m_init=30, top_m_max=200, damping_init=0.6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    if tleft() > 200:
        t_alloc = min(200, tleft() * 0.3)
        h_bin = try_binary_round(best_h, n, target)
        h_lp2, c5_lp2 = active_set_lp(h_bin, n, target, t_alloc,
                                       top_m_init=50, top_m_max=300, damping_init=0.5)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    if tleft() > 150:
        h_slq = slsqp_refine(best_h, n, target,
                              [20, 80, 300, 800, 2000], 2000, 1e-14)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    if tleft() > 120:
        t_alloc = min(120, tleft() * 0.3)
        h_ng, c5_ng = nesterov_subgradient(best_h, n, target,
                                            max_steps=20000, time_limit=t_alloc)
        if c5_ng < best_score:
            best_score = c5_ng
            best_h = h_ng

    for noise in [0.03, 0.08, 0.15]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_slq = slsqp_refine(h_pert, n, target,
                              [50, 200, 500, 1500], 1000, 1e-14)
        sc = compute_c5(h_slq)
        if sc < best_score:
            best_score = sc
            best_h = h_slq

    if tleft() > 150:
        for n_try in [600, 800]:
            if tleft() < 100:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            t_alloc = min(100, tleft() * 0.3)
            h_lp3, c5_lp3 = active_set_lp(h_interp, n_try, target_try, t_alloc,
                                            top_m_init=50, top_m_max=n_try, damping_init=0.5)
            if c5_lp3 < best_score:
                best_score = c5_lp3
                best_h = h_lp3
                n = n_try
                target = target_try

    if tleft() > 80:
        t_alloc = min(60, tleft() * 0.3)
        h_sg, c5_sg = subgradient_simple(best_h, n, target,
                                          max_steps=30000, time_limit=t_alloc)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    if tleft() > 40:
        h_final_try = slsqp_refine(best_h, n, target,
                                    [200, 600, 1500, 3000, 5000], 1500, 1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
