def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]

            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:

    return h
