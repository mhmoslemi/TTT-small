    sw = np.sum(exp_s)

import numpy as np
from scipy.optimize import minimize
import time

_algo_state = {'alpha': 20.0, 'top_m': 200}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def get_corr_and_weights(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_s = np.exp(shifted)
    sw = np.sum(exp_s)
    obj = c_max + np.log(sw) / alpha
    weights = exp_s / sw
    return obj, corr, weights

def smooth_grad(h, alpha, top_m=100):
    n = len(h)
    dx = 2.0 / n
    obj, corr, weights = get_corr_and_weights(h, alpha)
    top_idx = np.argsort(-weights)[:top_m]
    top_w = weights[top_idx]
    top_sw = np.sum(top_w)
    if top_sw < 1e-15:
        return np.zeros(n)
    top_w = top_w / top_sw
    top_lags = top_idx.astype(int) - (n - 1)
    grad = np.zeros(n)
    for j in range(len(top_lags)):
        w = top_w[j]
        if w < 1e-12:
            continue
        lag = int(top_lags[j])
        k = abs(lag)
        if k >= n or k == 0:
            continue
        if lag > 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] -= w * h[k:]
    grad *= dx
    return grad

def _slsqp_obj(x):
    obj, _, _ = get_corr_and_weights(x, _algo_state['alpha'])
    return obj

def _slsqp_jac(x):
    return smooth_grad(x, _algo_state['alpha'], top_m=_algo_state['top_m'])

def subgradient_with_ls(h, target, time_limit, alpha_init=20.0, top_m=80):
    n = len(h)
    start = time.time()
    alpha = alpha_init
    eta = 0.02
    best_local = h.copy()
    best_local_score = compute_c5(h)
    no_improve = 0
    it = 0
    while time.time() - start < time_limit and it < 20000:
        it += 1
        grad = smooth_grad(h, alpha, top_m)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        best_h_trial = h.copy()
        best_score_trial = compute_c5(h)
        step = eta
        for ls in range(8):
            h_trial = h - step * grad / gn
            h_trial = project_feasible(h_trial, target)
            s = compute_c5(h_trial)
            if s < best_score_trial - 1e-15:
                best_h_trial = h_trial
                best_score_trial = s
                break
            step *= 0.5
        if best_score_trial < best_local_score - 1e-15:
            best_local_score = best_score_trial
            best_local = best_h_trial.copy()
            no_improve = 0
            eta = min(eta * 1.08, 0.15)
        else:
            no_improve += 1
            if no_improve > 25:
                eta *= 0.7
                alpha = min(alpha * 1.2, 5000.0)
                no_improve = 0
        h = best_h_trial
    return best_local, best_local_score

def slsqp_phase(h, target, time_limit, alphas, top_m=200):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_cons_fun, 'jac': _ones_jac}]
    n_alphas = len(alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (n_alphas - ai)
        maxiter = max(50, int(time_per * 1.5))
        _algo_state['alpha'] = alpha
        _algo_state['top_m'] = top_m
        try:
            res = minimize(_slsqp_obj, best, jac=_slsqp_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score - 1e-14:
                best_score = s
                best = h_new
        except Exception:
            pass
    return best, best_score

def _sum_cons_fun(x):
    return np.sum(x) - _cons_target[0]

def _ones_jac(x):
    return np.ones(_cons_target[1])

def block_cd(h, target, time_limit, block_size=3, n_trials=8, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    no_improve_rounds = 0
    scale = 0.03
    while time.time() - start < time_limit:
        improved = False
        for _ in range(max(1, n // block_size)):
            si = rng.randint(0, n - block_size + 1)
            ei = si + block_size
            block = best[si:ei].copy()
            for _t in range(n_trials):
                delta = rng.uniform(-scale, scale, block_size)
                tb = np.clip(block + delta, 0.0, 1.0)
                ht = best.copy()
                ht[si:ei] = tb
                ht = project_feasible(ht, target)
                s = compute_c5(ht)
                if s < best_score - 1e-14:
                    best_score = s
                    best = ht
                    block = best[si:ei].copy()
                    improved = True
                    break
        if not improved:
            no_improve_rounds += 1
            scale *= 0.7
            if no_improve_rounds > 8 or scale < 1e-5:
                break
        else:
            no_improve_rounds = 0
    return best, best_score
