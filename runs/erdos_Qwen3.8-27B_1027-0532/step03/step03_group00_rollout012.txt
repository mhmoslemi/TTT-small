```python
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection on the threshold
    lo, hi = -max(h), 1 - min(h)  # range of possible thresholds
    for _ in range(60):  # enough for double precision
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Final correction for any numerical error
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-15:

        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        if mask.sum() > 0:
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_g(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_top_m(h, m_top=5):
    """Compute approximate gradient of max_k g_k using top-m lags"""
    n = len(h)
    all_g = compute_all_g(h)
    # Find top m lags
    top_idx = np.argsort(all_g)[-m_top:]
    # Compute gradient as weighted average
    weights = np.exp(all_g[top_idx] * 100)  # sharp weights
    weights /= weights.sum()
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)  # convert to lag
        grad += weights[list(top_idx).index(idx)] * _grad_gk(h, k, n)
    return grad

def _grad_gk(h, k, n):
    """Gradient of g_k"""
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] += (-h[k:])
    else:
        p = -k
        if p == 0:
            grad[:] = 1.0 - 2.0 * h
        else:
            grad[:n-p] += (1.0 - h[p:])
            grad[p:] += (-h[:n-p])
    return grad

def project_box_sum(h, target_sum):
    """Project onto {x: 0<=x<=1, sum(x)=target_sum}"""

    if abs(diff) > 1e-12:
        if diff > 0:

            h_p[mask] += diff / cnt
```

For the BLP approach, I need to solve an LP. Let me use scipy.optimize.linprog.

The LP formulation:
min t
s.t. For each active lag k: ∇g_k(h_cur)ᵀ h - t ≤ g_k(h_cur) - ∇g_k(h_cur)ᵀ h_cur
     0 ≤ h_i ≤ 1 for all i
     sum(h) = N/2

Variables: x = [h_0, ..., h_{N-1}, t] (N+1 variables)
Objective: c = [0, ..., 0, 1] (minimize t, the last variable)

Inequality constraints (A_ub @ x ≤ b_ub):
For active lag k (let's say M of them):
  Row: [∇g_k[0], ∇g_k[1], ..., ∇g_k[N-1], -1]
  RHS: g_k(h_cur) - ∇g_k(h_cur)·h_cur

For h_i ≤ 1:
  Row: [0, ..., 1(at i), ..., 0, 0]
  RHS: 1

For -h_i ≤ 0:
  Row: [0, ..., -1(at i), ..., 0, 0]
  RHS: 0

Equality constraint (A_eq @ x = b_eq):
  Row: [1, 1, ..., 1, 0]
  RHS: N/2

Let me code the BLP:

```python
def blp_solve(h_cur, n, target, m_top=20, max_iter=500, deadline=None):
    """Branch and Linear Programming for the min-max problem"""
    best_h = h_cur.copy()
    best_t = compute_c5(h_cur)
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute all g_k values and gradients for top m lags
        all_g = compute_all_g(h_cur)
        top_idx = np.argsort(all_g)[-m_top:]
        
        # Build LP
        n_vars = n + 1
        c = np.zeros(n_vars)
        c[-1] = 1.0  # minimize t
        
        A_ub_rows = []
        b_ub_vals = []
        
        for idx in top_idx:
            k = idx - (n - 1)
            grad_k = _grad_gk(h_cur, k, n)
            gk_val = all_g[idx]
            # Constraint: grad_k · h - t ≤ gk_val - grad_k · h_cur
            row = np.zeros(n_vars)
            row[:n] = grad_k
            row[-1] = -1.0
            rhs = gk_val - grad_k.dot(h_cur)
            A_ub_rows.append(row)
            b_ub_vals.append(rhs)
        
        # Box constraints: h_i ≤ 1
        for i in range(n):
            row = np.zeros(n_vars)
            row[i] = 1.0
            A_ub_rows.append(row)
            b_ub_vals.append(1.0)
        
        # Box constraints: -h_i ≤ 0
        for i in range(n):
            row = np.zeros(n_vars)
            row[i] = -1.0
            A_ub_rows.append(row)
            b_ub_vals.append(0.0)
        
        A_ub = np.array(A_ub_rows)
        b_ub = np.array(b_ub_vals)
        
        # Equality: sum(h) = target
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Solve LP
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         method='highs', options={'presolve': True})
            if res.success:
                h_lp = res.x[:n]
                h_lp = project_box_sum(h_lp, target)
                t_lp = compute_c5(h_lp)  # actual C5, not the LP value
                if t_lp < best_t:
                    best_t = t_lp
                    best_h = h_lp
                    h_cur = h_lp
                else:
                    # Check if we're at a local optimum (LP solution doesn't improve)
                    # Try a small perturbation
                    h_cur = h_lp  # still move to the LP solution
        except Exception:
            break
    
    return best_h, best_t
```

Actually, g_k(h) is bilinear, not concave. So the linearization at h_cur is:
g_k(h) ≈ g_k(h_cur) + ∇g_k(h_cur)ᵀ(h - h_cur)

This approximation can be either above or below the true value, depending on the direction. So the LP doesn't give a valid lower or upper bound in general. The BLP approach might not converge in the standard sense.

However, in practice, it often works well as a heuristic: the linearized constraints guide the optimization in a useful direction.

Let me also consider: since g_k is bilinear (and specifically, it's the sum of a linear term and a negative semidefinite quadratic term... wait, is it?), maybe the linearization is an overestimate in some directions and underestimate in others.

g_k(h) = sum_n h[n+k]*(1-h[n]) = sum_n h[n+k] - sum_n h[n+k]*h[n]

The Hessian of g_k has eigenvalues in [-2, 2] (as I computed earlier). So g_k is neither convex nor concave in general. The linearization can overestimate or underestimate.
