    trial_vals = np.linspace(0.0, 1.0, n_trials)

    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        improved = False
        for i in order:
            best_local = h_cur[i]
            best_local_score = best_score
            for tv in trial_vals:
                h_try = h_cur.copy()
                h_try[i] = tv
                adj = h_cur[i] - tv
                others = np.where(np.arange(n) != i)[0]
                room_idx = others[np.argmax(np.where(h_try[others] > 0.5,
                                                     h_try[others],
                                                     1.0 - h_try[others]))]
                h_try[room_idx] = np.clip(h_try[room_idx] - adj, 0.0, 1.0)
                h_try = project_feasible(h_try, target_sum)
                sc = compute_c5(h_try)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = h_try.copy()
            if best_local_score < best_score:
                h_cur = best_local
                best_score = best_local_score
                improved = True
        if not improved:
            break
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 35 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: SLSQP with gradual alpha increase
    if time_left() > 120:
        alphas1 = [8, 15, 30, 60, 120, 250, 500]
        best_h, best_score = run_slsqp_sequence(
            best_h, target_init, alphas1, 4000, 1e-15, start_time, budget_s)

    # Phase 2: Diversification via subgradient from perturbations
    subgrad_configs = [
        (3000, 0.12, 5, 25, 0.04, 0.015, 250, 0.90),
        (3000, 0.06, 8, 50, 0.02, 0.008, 250, 0.85),
        (2500, 0.20, 3, 12, 0.08, 0.03, 200, 0.92),
        (2500, 0.04, 10, 80, 0.0, 0.0, 180, 0.80),
        (3000, 0.09, 6, 35, 0.03, 0.01, 250, 0.88),
    ]

    for i, cfg in enumerate(subgrad_configs):
        if time_left() < 100:
            break
        noise_scale_init = 0.012 if i % 2 == 0 else 0.008
        h_try = best_h + rng.randn(n_init) * noise_scale_init
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                       max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
                                       alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
                                       patience=cfg[6], momentum=cfg[7], seed=seed + i * 131)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 3: SLSQP refinement with higher alphas
    if time_left() > 200:
        alphas2 = [40, 100, 250, 500, 1000, 2000]
        best_h, best_score = run_slsqp_sequence(
            best_h, target_init, alphas2, 5000, 1e-16, start_time, budget_s)

    # Phase 4: Coordinate descent polishing
    if time_left() > 60:
        best_h, best_score = coord_descent_polish(
            best_h, target_init, n_sweeps=3, n_trials=11, seed=seed + 100)

    # Phase 5: Try higher resolutions
    for n_try in [500, 650, 800]:
        if time_left() < 250:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)

        h_new, score = run_subgradient(h_interp, target_try,
                                       max_iter=2500, lr_init=0.08, topk=5,
                                       alpha=30, noise_prob=0.03, noise_scale=0.01,
                                       patience=200, momentum=0.90,
                                       seed=seed + 3000 + n_try)
        if score < best_score:
            best_score = score
            best_h = h_new

        if time_left() > 180:
            alphas3 = [15, 40, 100, 250, 500]
            best_h, best_score = run_slsqp_sequence(
                best_h, target_try, alphas3, 3000, 1e-15, start_time, budget_s)

        if time_left() > 80:
            best_h, best_score = coord_descent_polish(
                best_h, target_try, n_sweeps=2, n_trials=9, seed=seed + 2000 + n_try)

    # Phase 6: Final SLSQP polish at best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if time_left() > 80:
        alphas_final = [300, 800, 1500, 3000]
        best_h, best_score = run_slsqp_sequence(
            best_h, target_use, alphas_final, 3000, 1e-16, start_time, budget_s)

    # Phase 7: One more subgradient burst
    if time_left() > 50:
        h_try = best_h + rng.randn(n_use) * 0.005
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                       max_iter=2000, lr_init=0.04, topk=5,
                                       alpha=60, noise_prob=0.0, noise_scale=0.0,
                                       patience=150, momentum=0.91, seed=seed + 9999)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Final projection and evaluation
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
