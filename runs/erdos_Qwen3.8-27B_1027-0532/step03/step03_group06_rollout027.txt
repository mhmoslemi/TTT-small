def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:

        if diff > 0:

        else:

        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)

def compute_slopes(h, i, n):
    k = np.arange(2 * n - 1)
    m = k - (n - 1)
    a = np.zeros(2 * n - 1)
    valid1 = (i + m >= 0) & (i + m < n)
    a[valid1] += 1.0 - h[i + m[valid1]]
    valid2 = (i - m >= 0) & (i - m < n)
    a[valid2] -= h[i - m[valid2]]
    return a

        return c.max(), 0.0

        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2 * n - 2, i + n - 1)
        if klo_c <= khi_c:
            j_idx_c = i - np.arange(klo_c, khi_c + 1) + (n - 1)
            term_c = np.sum(w[klo_c:khi_c + 1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad


def lbfgs_stage(h_start, n, target, alpha, maxiter, ftol):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coord_descent_single(h, n, target, n_passes, rng, time_check):
    best_h = h.copy()
    best_score = compute_c5(h)
    for p in range(n_passes):
        if time_check() < 30:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        for i in order:
            a = compute_slopes(h, i, n)
            lo = -h[i]
            hi = 1.0 - h[i]
            if hi - lo < 1e-14:
                continue
            _, delta = solve_1d_minimax(c, a, lo, hi)
            if abs(delta) < 1e-15:
                continue
            h[i] += delta
            c += delta * a
        h = project_feasible(h, target)
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def coord_descent_pair(h, n, target, n_passes, rng, time_check):
    best_h = h.copy()
    best_score = compute_c5(h)
    for p in range(n_passes):
        if time_check() < 30:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        for _ in range(n):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            a_i = compute_slopes(h, i, n)
            a_j = compute_slopes(h, j, n)
            d = a_i - a_j
            if np.max(np.abs(d)) < 1e-14:
                continue
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi - lo < 1e-14:
                continue
            _, delta = solve_1d_minimax(c, d, lo, hi)
            if abs(delta) < 1e-15:
                continue
            h[i] += delta
            h[j] -= delta
            c += delta * d
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def coord_descent_greedy(h, n, target, n_passes, rng, time_check):
    best_h = h.copy()
    best_score = compute_c5(h)
    for p in range(n_passes):
        if time_check() < 30:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(c))
        m_star = k_star - (n - 1)
        candidates = []
        for i in range(n):
            if 0 <= i + m_star < n:
                val = h[i] * (1.0 - h[i + m_star])
                candidates.append((val, i, 'first'))
            if 0 <= i - m_star < n:
                val = h[i - m_star] * (1.0 - h[i])
                candidates.append((val, i, 'second'))
        candidates.sort(key=lambda x: -x[0])
        tried = set()
        for _, i, _ in candidates[:min(20, len(candidates))]:
            if time_check() < 15:
                break
            if i in tried:
                continue
            tried.add(i)
            a_i = compute_slopes(h, i, n)
            for j in range(n):
                if j == i or j in tried:
                    continue
                a_j = compute_slopes(h, j, n)
                d = a_i - a_j
                if abs(d[k_star]) < 1e-10:
                    continue
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if hi - lo < 1e-14:
                    continue
                _, delta = solve_1d_minimax(c, d, lo, hi)
                if abs(delta) < 1e-12:
                    continue
                h[i] += delta
                h[j] -= delta
                c += delta * d
                tried.add(j)
                break
        sc = compute_c5(h)
        if sc < best_score:
            best_score = sc
            best_h = h.copy()
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B warm start
    if tleft() > 60:
        for alpha in [30, 100, 300]:
            if tleft() < 40:
                break
            h_new = lbfgs_stage(best_h, n_init, target_init, alpha, 200, 1e-14)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 2: Single-coordinate descent (no constraint, project per pass)
    if tleft() > 40:
        best_h, best_score = coord_descent_single(best_h, n_init, target_init,
                                                  30, rng, tleft)

    # Phase 3: 2-coordinate descent (constraint maintained)
    if tleft() > 40:
        best_h, best_score = coord_descent_pair(best_h, n_init, target_init,
                                                50, rng, tleft)

    # Phase 4: Greedy 2-coordinate descent targeting the max
    if tleft() > 40:
        best_h, best_score = coord_descent_greedy(best_h, n_init, target_init,
                                                  20, rng, tleft)

    # Phase 5: L-BFGS-B polish with high alpha
    if tleft() > 50:
        for alpha in [1000, 3000]:
            if tleft() < 35:
                break
            h_new = lbfgs_stage(best_h, n_init, target_init, alpha, 300, 1e-15)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: More single-coordinate descent (cyclic order)
    if tleft() > 40:
        for _ in range(20):
            if tleft() < 25:
                break
            c = np.correlate(best_h, 1.0 - best_h, mode='full')
            h_cur = best_h.copy()
            for i in range(n_init):
                a = compute_slopes(h_cur, i, n_init)
                lo = -h_cur[i]
                hi = 1.0 - h_cur[i]
                if hi - lo < 1e-14:
                    continue
                _, delta = solve_1d_minimax(c, a, lo, hi)
                if abs(delta) < 1e-15:
                    continue
                h_cur[i] += delta
                c += delta * a
            h_cur = project_feasible(h_cur, target_init)
            sc = compute_c5(h_cur)
            if sc < best_score:
                best_score = sc
                best_h = h_cur

    # Phase 7: More 2-coordinate descent
    if tleft() > 40:
        best_h, best_score = coord_descent_pair(best_h, n_init, target_init,
                                                80, rng, tleft)

    # Phase 8: Greedy refinement
    if tleft() > 30:
        best_h, best_score = coord_descent_greedy(best_h, n_init, target_init,
                                                  15, rng, tleft)

    # Phase 9: Final 2-coordinate with random restart perturbations
    for attempt in range(5):
        if tleft() < 30:
            break
        h_pert = best_h.copy()
        idx = rng.choice(n_init, size=min(10, n_init // 10), replace=False)
        h_pert[idx] += rng.uniform(-0.02, 0.02, size=len(idx))
        h_pert = project_feasible(h_pert, target_init)
        h_pert, sc = coord_descent_pair(h_pert, n_init, target_init,
                                        30, rng, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_pert

    # Phase 10: Final single-coordinate polish
    if tleft() > 30:
        best_h, best_score = coord_descent_single(best_h, n_init, target_init,
                                                  15, rng, tleft)

    # Phase 11: Final pair polish
    if tleft() > 25:
        best_h, best_score = coord_descent_pair(best_h, n_init, target_init,
                                                30, rng, tleft)

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
