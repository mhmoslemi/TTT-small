import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def cross_corr_grad(h, corr_weights):
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w = corr_weights[d + n - 1]
        if abs(w) < 1e-15:
            continue
        if d == 0:
            # C[n-1] = sum h[i]*(1-h[i]) = sum h - sum h^2
            # grad = 1 - 2*h
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            j_lo, j_hi = 0, n - 1 - d
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo+d:j_hi+1+d])
            j_lo2, j_hi2 = d, n - 1
            grad[j_lo2:j_hi2+1] -= w * h[j_lo2-d:j_hi2+1-d]
        else:
            dd = -d
            j_lo, j_hi = 0, n - 1 - dd
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo-dd:j_hi+1-dd])
            j_lo2, j_hi2 = dd, n - 1
            grad[j_lo2:j_hi2+1] -= w * h[j_lo2+dd:j_hi2+1+dd]
    return grad

def adam_soft_gradient(h, target, time_limit, alpha_init=50.0, lr=0.005, seed=42):

    start = time.time()
    rng = np.random.RandomState(seed)
    alpha = alpha_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    while time.time() - start < time_limit:
        C = np.correlate(h, 1.0 - h, mode='full')
        C_max = np.max(C)
        shifted = alpha * (C - C_max)
        shifted = np.clip(shifted, -500, 0)

            v = np.zeros(n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def cross_corr_grad(h, corr_weights):
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w = corr_weights[d + n - 1]
        if abs(w) < 1e-15:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            j_lo, j_hi = 0, n - 1 - d
            grad[j_lo:j_hi+1] += w * (1.0 - h[j_lo+d:j_hi+1+d])
            j_lo2, j_hi2 = d, n - 1
            grad[j_lo2:j_hi2+1] -= w * h[j_lo2-d:j_hi2+1-d]
        else:
            dd = -d
            j_lo1, j_hi1 = dd, n - 1
            grad[j_lo1:j_hi1+1] += w * (1.0 - h[j_lo1-dd:j_hi1+1-dd])
            j_lo2, j_hi2 = 0, n - 1 - dd
            grad[j_lo2:j_hi2+1] -= w * h[j_lo2+dd:j_hi2+1+dd]
    return grad

def get_softmax_weights(C, alpha):
    C_max = np.max(C)
    shifted = alpha * (C - C_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_w = np.exp(shifted)
    sw = np.sum(exp_w)
    if sw < 1e-15:
        return np.full(len(C), 1.0/len(C))
    return exp_w / sw

def adam_soft_gradient(h, target, time_limit, alpha_init=30.0, lr_init=0.002, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    alpha = alpha_init
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    best_alpha = alpha
    
    while time.time() - start < time_limit:

        weights = get_softmax_weights(C, alpha)
        grad = cross_corr_grad(h, weights)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = h - step
        h_new = project_feasible(h_new, target)
        
        score = compute_c5(h_new)
        if score < best_score - 1e-13:
            best_score = score
            best = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                no_improve = 0

def coordinate_descent_fine(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)

        for i in order:

def slsqp_refine(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    alphas = [30, 80, 200, 500]
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        
        def obj(x, a=alpha):
            C = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(C)
            shifted = np.clip(a * (C - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        
        def obj_grad(x, a=alpha):
            C = np.correlate(x, 1.0 - x, mode='full')
            weights = get_softmax_weights(C, a)
            return cross_corr_grad(x, weights)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        maxiter = int(max(50, remaining * 1.5))

            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new.copy()
        except Exception:
            pass
    return best, best_score

This is the standard ADAM update. But I need to make sure the step size is appropriate. The gradient magnitude |g| for this problem is roughly O(n) (since each gradient component is a sum of up to n terms, each of magnitude ≤ 1, weighted by softmax weights that sum to 1). So |g| ≈ O(1) to O(n/2n) = O(1) roughly. Actually, the softmax weights sum to 1, and each weight is at most 1, and the sum of absolute values of the unweighted gradient for a single lag is at most n (each coordinate contributes at most 1). So the weighted gradient has magnitude at most n, but typically much less.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def get_softmax_weights(C, alpha):
    C_max = np.max(C)
    shifted = alpha * (C - C_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_w = np.exp(shifted)
    sw = np.sum(exp_w)
    if sw < 1e-15:
        return np.full(len(C), 1.0 / len(C))
    return exp_w / sw

def cross_corr_grad(h, corr_weights):
    n = len(h)
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        w = corr_weights[d + n - 1]
        if abs(w) < 1e-15:
            continue
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            j_lo, j_hi = 0, n - 1 - d
            grad[j_lo:j_hi + 1] += w * (1.0 - h[j_lo + d:j_hi + 1 + d])
            j_lo2, j_hi2 = d, n - 1
            grad[j_lo2:j_hi2 + 1] -= w * h[j_lo2 - d:j_hi2 + 1 - d]
        else:
            dd = -d
            j_lo1, j_hi1 = dd, n - 1
            grad[j_lo1:j_hi1 + 1] += w * (1.0 - h[j_lo1 - dd:j_hi1 + 1 - dd])
            j_lo2, j_hi2 = 0, n - 1 - dd
            grad[j_lo2:j_hi2 + 1] -= w * h[j_lo2 + dd:j_hi2 + 1 + dd]
    return grad
