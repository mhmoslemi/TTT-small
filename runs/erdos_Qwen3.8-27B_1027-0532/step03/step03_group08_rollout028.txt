    h = np.clip(h.copy(), 0.0, 1.0)

import time

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = float(y.min()) - 1.5
    hi = float(y.max()) + 1.5
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = S - float(x.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_shift(h, k_idx, dx, n):
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[lag:lag + m] += (1.0 - h[:m])
            g[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            g[:m] += (1.0 - h[j:])
            g[j:] += -h[:m]
    return g * dx

def _nesterov(h0, n, tsum, iters, a0, top_m, sg_a, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    bv = _c5_of(x)
    for it in range(iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = float(np.max(corr))
        ev = np.exp(sg_a * (corr - mc))
        tw = float(ev.sum())
        if tw < 1e-300:
            w = np.zeros(len(corr))
            w[int(np.argmax(corr))] = 1.0
        else:
            w = ev / tw
        ti = np.argsort(w)[::-1][:top_m]
        wm = w[ti].copy()
        wm = wm / wm.sum()
        g = np.zeros(n)
        for idx in range(len(ti)):
            g += wm[idx] * _grad_shift(y, int(ti[idx]), dx, n)
        step = a0 / np.sqrt(float(it + 1))
        xn = _project(y - step * g, tsum)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        yn = xn + ((t - 1.0) / tn) * (xn - x)
        yn = _project(yn, tsum)
        x = xn
        y = yn
        t = tn
        cv = _c5_of(x)
        if cv < bv:
            bv = cv
            best = x.copy()
    return best, bv

def _nesterov_fixed(h0, n, tsum, iters, a0, top_m, sg_a, deadline):

                x = xn
    return best, bv

def _de(h0, n, tsum, ps, mg, Fs, Fe, CR, dl, rng):
    pop = np.zeros((ps, n))
    pop[0] = h0.copy()
    for i in range(1, ps):
        r = i % 6
        if r == 0:
            pop[i] = _fix_sum(rng.rand(n), tsum)
        elif r == 1:
            pop[i] = _fix_sum(1.0 - h0 + 0.1 * (rng.rand(n) - 0.5), tsum)
        elif r == 2:
            tt = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(tt * h0 + (1 - tt) * rng.rand(n), tsum)
        elif r == 3:
            pop[i] = _fix_sum(h0 + 0.5 * (rng.rand(n) - 0.5), tsum)
        elif r == 4:
            pop[i] = _fix_sum(h0[rng.permutation(n)], tsum)
        else:
            pop[i] = h0.copy()
            bs = max(2, n // 5)
            s1 = rng.randint(0, n - 2 * bs)
            s2 = rng.randint(0, n - 2 * bs)
            tmp = pop[i][s1:s1 + bs].copy()
            pop[i][s1:s1 + bs] = h0[s2:s2 + bs]
            pop[i][s2:s2 + bs] = tmp
            pop[i] = _fix_sum(pop[i], tsum)
    pv = np.array([_c5_of(pop[i]) for i in range(ps)])
    bi = int(np.argmin(pv))
    bv = float(pv[bi])
    bh = pop[bi].copy()
    stall = 0
    for gen in range(mg):
        if time.time() > dl:
            break
        F = Fs + (Fe - Fs) * float(gen) / float(max(1, mg))
        for i in range(ps):
            if time.time() > dl:
                break
            r1 = rng.randint(ps)
            r2 = rng.randint(ps)
            if r1 == i:
                r1 = (r1 + 1) % ps
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % ps
            mu = pop[bi] + F * (pop[r1] - pop[r2])
            tr = pop[i].copy()
            cm = rng.random(n) < CR
            cm[rng.randint(n)] = True
            tr[cm] = mu[cm]
            tr = _fix_sum(np.clip(tr, 0, 1), tsum)
            tv = _c5_of(tr)
            if tv < pv[i]:
                pop[i] = tr
                pv[i] = tv
                if tv < bv:
                    bv = tv
                    bh = tr.copy()
                    bi = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > ps * 2:
            whalf = np.argsort(pv)[ps // 2:]
            for idx in whalf:
                tm = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(tm * bh + (1 - tm) * rng.rand(n), tsum)
                pv[idx] = _c5_of(pop[idx])
                if pv[idx] < bv:
                    bv = pv[idx]
                    bh = pop[idx].copy()
                    bi = int(idx)
            stall = 0
    return bh, bv

def _sa(h0, n, tsum, mi, Ts, Te, dl, rng):
    h = h0.copy()
    hv = _c5_of(h)
    bh = h.copy()
    bv = hv
    for it in range(mi):
        if time.time() > dl:
            break
        T = Ts * (Te / Ts) ** (float(it) / float(max(1, mi)))
        hn = h.copy()
        npert = rng.randint(1, max(2, n // 10))
        idxs = rng.choice(n, npert, replace=False)
        for idx in idxs:
            r = rng.random()
            if r < 0.3:
                hn[idx] = 0.0
            elif r < 0.6:
                hn[idx] = 1.0
            else:
                hn[idx] = rng.random()
        hn = _fix_sum(hn, tsum)
        nv = _c5_of(hn)
        if nv < hv or rng.random() < np.exp(-(nv - hv) / max(T, 1e-10)):
            h = hn
            hv = nv
            if hv < bv:
                bv = hv
                bh = h.copy()
    return bh, bv

def _eqf(x):
    return x.sum() - _gs_ts

def _eqj(x):
    return np.ones(_gs_n)

def _so(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = float(np.max(corr))
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _slsqp(h0, n, tsum, alpha, mi, dl):
    global _gs_ts, _gs_n
    _gs_ts = tsum
    _gs_n = n
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _eqf, 'jac': _eqj}]
    try:
        if dl is not None and time.time() > dl:
            return h0
        res = minimize(_so, h0, args=(alpha, n), method='SLSQP',
                       bounds=bd, constraints=cs,
                       options={'maxiter': mi, 'ftol': 1e-14})
        return _fix_sum(res.x, tsum)
    except Exception:
        return h0

def _cd_binary(h0, n, tsum, rounds, dl):
    h = h0.copy()
    for _ in range(rounds):
        if dl is not None and time.time() > dl:
            break
        imp = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            hf = _fix_sum(h, tsum)
            c0 = _c5_of(hf)
            h[j] = 1.0
            hf = _fix_sum(h, tsum)
            c1 = _c5_of(hf)
            h[j] = 0.0 if c0 <= c1 else 1.0
            h = _fix_sum(h, tsum)
            if h[j] != old:
                imp = True
        if not imp:
            break
    return h

def _cd_frac(h0, n, tsum, rounds, dl):

def _focal_cd(h0, n, tsum, rounds, dl, focus_m=15):
    h = h0.copy()
    for _ in range(rounds):
        if dl is not None and time.time() > dl:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_shifts = np.argsort(corr)[::-1][:focus_m]
        active = set()
        for k_idx in top_shifts:
            lag = k_idx - (n - 1)
            if lag >= 0:
                active.update(range(lag, n))
                active.update(range(0, n - lag))
            else:
                j = -lag
                active.update(range(0, n - j))
                active.update(range(j, n))
        active_list = sorted(active)
        imp = False
        for j in active_list:
            if dl is not None and time.time() > dl:
                break
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in [0.0, 1.0]:
                h[j] = v
                hf = _fix_sum(h, tsum)
                c = _c5_of(hf)
                if c < best_c:
                    best_c = c
                    best_v = v
            h[j] = best_v
            h = _fix_sum(h, tsum)
            if best_v != old:

        if not imp:

def _pair_swap_search(h0, n, tsum, n_swaps, dl, rng):
    h = h0.copy()
    hv = _c5_of(h)
    bh = h.copy()
    bv = hv
    for _ in range(n_swaps):
        if time.time() > dl:
            break
        i, j = rng.choice(n, 2, replace=False)
        if i == j:
            continue
        hi_old, hj_old = h[i], h[j]
        h[i], h[j] = hj_old, hi_old
        h = _fix_sum(h, tsum)
        c = _c5_of(h)
        if c < hv - 1e-12:
            hv = c
            if c < bv:
                bv = c
                bh = h.copy()
        else:
            h[i], h[j] = hi_old, hj_old
            h = _fix_sum(h, tsum)
    return bh, bv
