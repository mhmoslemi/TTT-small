        else:
            stall += 1
            lr = max(lr * 0.98, 1e-8)

        h_cur = h_new
        if stall > stall_limit:
            h_cur = best_h.copy()
            lr = 0.08
            stall = 0
            stall_limit = max(100, stall_limit * 3 // 4)

    return best_h, best_c5

def gen_perturbaries(h_best, rng, n, target_sum, strategy):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n, size=max(5, n // 8), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        max_len = max(3, n // 6)
        blen = int(rng.integers(3, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)
    elif strategy == 3:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.6, 0.9))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    elif strategy == 4:
        freq_pert = np.zeros(n)
        n_freqs = rng.integers(2, 5)
        for _ in range(n_freqs):
            freq = int(rng.integers(1, n // 2 + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            freq_pert += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
        return project_onto_constraint(h_best + freq_pert, target_sum)
    elif strategy == 5:
        h_p = h_best.copy()
        n_swaps = int(rng.integers(3, 8))
        for _ in range(n_swaps):
            i, j = rng.integers(0, n, size=2)
            if i != j:
                h_p[i], h_p[j] = h_p[j], h_p[i]
        return project_onto_constraint(h_p, target_sum)
    else:
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)

def upsample_h(h, new_n):
    old_n = len(h)
    old_t = np.linspace(0, 1, old_n)
    new_t = np.linspace(0, 1, new_n)
    h_up = np.interp(new_t, old_t, h)
    return h_up

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = np.clip(0.5 + 0.3 * np.random.default_rng(0).standard_normal(180), 0.0, 1.0)

    n_init = len(h0)

    def elapsed():
        return time.time() - start_time

    def rem():
        return budget_s - elapsed()

    n_work = 360
    h_work = upsample_h(h0, n_work)
    target_sum = n_work / 2.0
    h_work = project_onto_constraint(h_work, target_sum)

    best_h = h_work.copy()
    best_c5 = compute_c5(h_work)

    # Phase 1: Coarse SLSQP at working resolution
    if rem() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 300, target_sum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Medium SLSQP
    if rem() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300, target_sum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Projected gradient with adaptive active set
    if rem() > 60:
        tl = min(rem() - 30, 120)
        h_pg, c5_pg = adaptive_proj_grad(best_h, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 4: Fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 400, target_sum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Multi-start with diverse perturbations
    eps_cycle = [0.002, 0.001, 0.0005, 0.0003, 0.0002]
    n_restart = 0
    max_restarts = 12
    while rem() > 120 and n_restart < max_restarts:
        strategy = n_restart % 6
        h_pert = gen_perturbaries(best_h, rng, n_work, target_sum, strategy)
        eps_use = eps_cycle[min(n_restart // 3, len(eps_cycle) - 1)]
        maxiter = 200 if rem() > 200 else 120
        h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, maxiter, target_sum)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if rem() > 60 and c5_opt < best_c5 + 0.001:
            tl_pg = min(rem() - 30, 30)
            h_pg, c5_pg = proj_grad_topk(best_h, 2, 0.003, 0.05, 50000, tl_pg, start_time)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

        n_restart += 1

    # Phase 6: Very fine SLSQP
    if rem() > 50:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 300, target_sum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Final projected gradient polish
    if rem() > 40:
        tl = min(rem() - 20, 80)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.03, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 8: Resolution increase to 480
    if rem() > 200:
        n_up = 480
        target_sum_up = n_up / 2.0
        h_up = upsample_h(best_h, n_up)
        h_up = project_onto_constraint(h_up, target_sum_up)
        h_up = h_up + 0.005 * rng.standard_normal(n_up)
        h_up = project_onto_constraint(h_up, target_sum_up)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.005, 250, target_sum_up)
        if rem() > 100:
            h_opt_up, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.001, 250, target_sum_up)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
                h_opt_up = h_opt_up.copy()
        if rem() > 60:
            tl_up = min(rem() - 30, 60)
            h_pg_up, c5_pg_up = proj_grad_topk(h_opt_up, 2, 0.003, 0.04, 100000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up.copy()
        if rem() > 40:
            h_opt_up, c5_opt_up3 = slsqp_optimize(h_opt_up, 0.0003, 200, target_sum_up)
            if c5_opt_up3 < c5_opt_up:
                c5_opt_up = c5_opt_up3
                h_opt_up = h_opt_up.copy()

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n_work = n_up

    # Phase 9: Final polish at current resolution
    if rem() > 40:
        csum = len(best_h) / 2.0
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 200, csum)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 10: Another resolution bump if time allows
    if rem() > 250 and len(best_h) < 600:
        n_up2 = len(best_h) + 120
        target_sum_up2 = n_up2 / 2.0
        h_up2 = upsample_h(best_h, n_up2)
        h_up2 = project_onto_constraint(h_up2, target_sum_up2)
        h_up2 = h_up2 + 0.004 * rng.standard_normal(n_up2)
        h_up2 = project_onto_constraint(h_up2, target_sum_up2)

        h_opt2, c5_opt2 = slsqp_optimize(h_up2, 0.004, 200, target_sum_up2)
        if rem() > 100:
            h_opt2, c5_opt2b = slsqp_optimize(h_opt2, 0.001, 200, target_sum_up2)
            if c5_opt2b < c5_opt2:
                c5_opt2 = c5_opt2b
                h_opt2 = h_opt2.copy()
        if rem() > 50:
            tl2 = min(rem() - 25, 50)
            h_pg2, c5_pg2 = proj_grad_topk(h_opt2, 1, 0.002, 0.03, 80000, tl2, start_time)
            if c5_pg2 < c5_opt2:
                c5_opt2 = c5_pg2
                h_opt2 = h_pg2.copy()

        if c5_opt2 < best_c5:
            best_c5 = c5_opt2
            best_h = h_opt2.copy()

    # Final: ensure feasibility and return
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)

    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, len(final_h)
