                h_start[ones[:extra]] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(target_sum - s)
                h_start[zeros[:need]] = 1
        elif strategy == 3:
            # Blend with fresh random
            h_rand = random_binary(n0, rng)
            alpha = rng.uniform(0.6, 0.85)
            h_blend = alpha * best_h + (1 - alpha) * h_rand
            h_start = to_binary(h_blend)
        else:
            # Small perturbation of binary
            h_start = to_binary(best_h).copy()
            n_swap = rng.integers(1, 4)
            oi = np.where(h_start == 1)[0]
            zi = np.where(h_start == 0)[0]
            for _ in range(n_swap):
                i = oi[rng.integers(len(oi))]
                j = zi[rng.integers(len(zi))]
                h_start[i], h_start[j] = h_start[j], h_start[i]
        
        # Choose SA method
        if n_rounds % 3 == 0:
            base_E = float(np.max(np.correlate(h_start, 1.0 - h_start, mode='full')))
            t_cold = max(0.3, base_E * 0.015)
            t_hot = base_E * 0.6
            n_reps = 6
            temps_r = np.geomspace(t_cold, t_hot, n_reps).tolist()
            n_steps_r = min(int(remaining() * 0.3 * 8000), 800000)
            h_sa, sa_E = pt_sa_binary(h_start, n_steps_r, temps_r, rng,
                                      targeted_frac=0.55 + 0.1 * (n_rounds % 5) / 5.0,
                                      swap_rate=0.25)
        elif n_rounds % 3 == 1:
            n_steps_r = min(int(remaining() * 0.3 * 10000), 1000000)
            t_s = 0.15 + 0.2 * (1.0 - n_rounds / 30.0)
            h_sa, sa_E = standard_sa_binary(h_start, n_steps_r, t_s, 0.003, rng,
                                            targeted_frac=0.6)
        else:
            # Multi-start: run SA from h_start then from a perturbed version
            n_steps_r = min(int(remaining() * 0.15 * 10000), 500000)
            h_sa1, e1 = standard_sa_binary(h_start, n_steps_r, 0.2, 0.005, rng, targeted_frac=0.5)
            h_start2 = h_start + 0.04 * rng.standard_normal(n0)
            h_start2 = to_binary(project_onto_constraint(h_start2, target_sum))
            h_sa2, e2 = standard_sa_binary(h_start2, n_steps_r, 0.15, 0.004, rng, targeted_frac=0.55)
            if e1 <= e2:
                h_sa, sa_E = h_sa1, e1
            else:
                h_sa, sa_E = h_sa2, e2
        
        c5_sa = (2.0 / n0) * sa_E
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
            # Quick greedy
            h_g, g_E = greedy_local_improve(h_sa, max_passes=30)
            c5_g = (2.0 / n0) * g_E
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
        
        n_rounds += 1
    
    # Phase 4: Try different n values
    if remaining() > 100:
        for n_try in [n0 - 4, n0 + 4, n0 - 8, n0 + 8]:
            if n_try < 100 or remaining() < 60:
                break
            h_try = to_binary(random_binary(n_try, rng))
            h_try, try_E = greedy_local_improve(h_try, max_passes=20)
            n_steps_t = min(int(remaining() * 0.15 * 8000), 400000)
            base_E_t = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')))
            t_c = max(0.3, base_E_t * 0.02)
            t_h = base_E_t * 0.5
            temps_t = np.geomspace(t_c, t_h, 5).tolist()
            h_try, try_E = pt_sa_binary(h_try, n_steps_t, temps_t, rng,
                                        targeted_frac=0.5, swap_rate=0.2)
            h_try, try_E = greedy_local_improve(h_try, max_passes=30)
            c5_try = (2.0 / n_try) * try_E
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 5: Final continuous polish
    remaining_now = remaining()
    if remaining_now > 40:
        h_polish = best_h + 0.008 * rng.standard_normal(len(best_h))
        h_polish = project_onto_constraint(h_polish, len(best_h) / 2.0)
        h_final, c5_final = polish_continuous(h_polish, len(best_h),
                                              min(remaining_now - 15, 120),
                                              eps_list=[0.005, 0.001, 0.0003, 0.0001, 0.00003])
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        
        # Also try polishing the binary directly
        if remaining() > 30:
            h_polish2 = best_h + 0.003 * rng.standard_normal(len(best_h))
            h_polish2 = project_onto_constraint(h_polish2, len(best_h) / 2.0)
            h_final2, c5_final2 = polish_continuous(h_polish2, len(best_h),
                                                    min(remaining() - 10, 60),
                                                    eps_list=[0.001, 0.0003, 0.0001])
            if c5_final2 < best_c5:
                best_c5 = c5_final2
                best_h = h_final2.copy()
    
    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, len(final_h)
