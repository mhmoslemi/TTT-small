import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:

                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1
    
    return best_h, best_c5_val

            j = int(zeros[rng.integers(n_zero)])
            if i == j:
                step += 1
                continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = int(pos2oi[i])
            oz = int(pos2zj[j])
            ones[oi] = j
            zeros[oz] = i
            pos2oi[j] = oi
            pos2zj[i] = oz
            step += 1
            continue

        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max
        if delta_c5 < 0 or (t > 1e-8 and rng.random() < np.exp(-delta_c5 / t)):
            h[i], h[j] = h[j], h[i]
            oi = int(pos2oi[i])
            oz = int(pos2zj[j])
            ones[oi] = j
            zeros[oz] = i
            pos2oi[j] = oi
            pos2zj[i] = oz
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1

    return best_h, best_c5_val

def sa_binary_greedy(h_start, n_steps, t_start, t_end, rng, target_frac=0.7):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1

    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    n_one = len(ones)
    n_zero = len(zeros)

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))

        if rng.random() < target_frac:
            top3_k = np.argsort(corr)[-3:]
            worst_k = int(top3_k[rng.integers(3)])
            worst_s = worst_k - half
            s_val = worst_s
            found = False
            for _ in range(min(50, n_one)):
                i = int(ones[rng.integers(n_one)])
                js = i + s_val
                if 0 <= js < n and h[js] == 0.0:
                    j = js
                    found = True
                    break
                if s_val != 0:
                    jm = i - s_val
                    if 0 <= jm < n and h[jm] == 0.0:
                        j = jm
                        found = True
                        break
            if not found:
                i = int(ones[rng.integers(n_one)])
                j = int(zeros[rng.integers(n_zero)])
                if i == j:
                    step += 1
                    continue
        else:
            i = int(ones[rng.integers(n_one)])
            j = int(zeros[rng.integers(n_zero)])
            if i == j:
                step += 1
                continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = ones.index(i)
            ones[oi] = j
            oz = zeros.index(j)
            zeros[oz] = i
            step += 1
            continue

        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max
        if delta_c5 < 0 or (t > 1e-8 and rng.random() < np.exp(-delta_c5 / t)):
            h[i], h[j] = h[j], h[i]
            oi = ones.index(i)
            ones[oi] = j
            oz = zeros.index(j)
            zeros[oz] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1

    return best_h, best_c5_val

def continuous_sa(h_start, n_steps, t_start, t_end, rng, step_size=0.1):
    n = len(h_start)
    target_sum = n / 2.0
    h = h_start.astype(np.float64).copy()
    h = project_onto_constraint(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            step += 1
            continue
        delta_i = step_size * (2.0 * rng.random() - 1.0)
        h_new = h.copy()
        h_new[i] = np.clip(h_new[i] + delta_i, 0.0, 1.0)
        adjust = target_sum - h_new.sum()
        h_new[j] = np.clip(h_new[j] + adjust, 0.0, 1.0)
        if abs(h_new.sum() - target_sum) > 1e-6:
            h_new = project_onto_constraint(h_new, target_sum)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full') * (2.0 / n)
        new_max = float(np.max(corr_new))
        delta_obj = new_max - current_max
        if delta_obj < 0 or (t > 1e-8 and rng.random() < np.exp(-delta_obj / t)):
            h = h_new
            corr = corr_new
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        step += 1

    return best_h, best_c5_val

def optimize_at_n(h_init, n, budget_s, rng, start_time):
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return h_init, compute_c5(h_init), n

    if len(h_init) != n:
        h_resized = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(h_init)), h_init)
        h_resized = project_onto_constraint(h_resized, n / 2.0)
    else:
        h_resized = h_init.copy()

    best_h = h_resized.copy()
    best_c5 = compute_c5(h_resized)

    for eps in [0.01, 0.003, 0.001, 0.0003]:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        h_new, c5 = slsqp_optimize(best_h, eps, 300)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.7:
        h_bin = to_binary(best_h)
        n_steps = min(int((budget_s - (time.time() - start_time)) * 3000), 500000)
        if n_steps > 50000:
            h_sa, sa_max = sa_binary_fast(h_bin, n_steps, 0.4, 0.01, rng, 0.6)
            c5_sa = (2.0 / n) * sa_max
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.85:
        h_polish = best_h + 0.01 * rng.standard_normal(n)
        h_polish = project_onto_constraint(h_polish, n / 2.0)
        for eps in [0.0001, 0.00003]:
            elapsed = time.time() - start_time
            if elapsed > budget_s:
                break
            h_new, c5 = slsqp_optimize(h_polish, eps, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_polish = h_new

    return best_h, best_c5, n

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.12:
        for eps in [0.005, 0.002, 0.0008, 0.0003, 0.0001]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.12:
                break
            h_new, c5 = slsqp_optimize(best_h, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 60
    n_sa_rounds = 0
    max_sa_rounds = 25
    while remaining > 120 and n_sa_rounds < max_sa_rounds:
        strategy = n_sa_rounds % 5
        if strategy == 0:
            h_start = to_binary(best_h)
        elif strategy == 1:
            h_pert = best_h + 0.06 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, target_sum)
            h_start = to_binary(h_pert)
        elif strategy == 2:
            h_start = to_binary(best_h)
            n_flip = max(3, n0 // 15)
            idx = rng.choice(n0, size=n_flip, replace=False)
            for idx_i in idx:
                h_start[idx_i] = 1.0 - h_start[idx_i]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                extra = int(np.ceil(s - target_sum))
                h_start[np.random.choice(ones, size=min(extra, len(ones)), replace=False)] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(np.ceil(target_sum - s))
                h_start[np.random.choice(zeros, size=min(need, len(zeros)), replace=False)] = 1
        elif strategy == 3:
            h_rand = (rng.random(n0) < 0.5).astype(float)
            alpha = rng.uniform(0.5, 0.8)
            h_mix = alpha * best_h + (1 - alpha) * h_rand
            h_start = to_binary(h_mix)
        else:
            h_start = to_binary(best_h)
            n_flip = max(5, n0 // 8)
            idx = rng.choice(n0, size=n_flip, replace=False)
            for idx_i in idx:
                h_start[idx_i] = 1.0 - h_start[idx_i]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                extra = int(np.ceil(s - target_sum))
                h_start[np.random.choice(ones, size=min(extra, len(ones)), replace=False)] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(np.ceil(target_sum - s))
                h_start[np.random.choice(zeros, size=min(need, len(zeros)), replace=False)] = 1

        t_start = 0.35 * (1.0 - 0.5 * n_sa_rounds / max_sa_rounds)
        t_end = 0.003
        time_per_round = remaining / max(1, max_sa_rounds - n_sa_rounds)
        n_steps = int(min(time_per_round * 4000, 1500000))
        n_steps = max(n_steps, 20000)

        elapsed = time.time() - start_time
        if n_sa_rounds % 3 == 0:
            h_bin, bin_max = sa_binary_fast(h_start, n_steps, t_start, t_end, rng, 0.6)
        elif n_sa_rounds % 3 == 1:
            h_bin, bin_max = sa_binary_fast(h_start, n_steps, t_start, t_end, rng, 0.5)
        else:
            h_bin, bin_max = sa_binary_greedy(h_start, n_steps, t_start, t_end, rng, 0.7)

        c5_bin = (2.0 / n0) * bin_max
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            elapsed = time.time() - start_time
            if elapsed < budget_s * 0.7:
                h_polish = h_bin + 0.012 * rng.standard_normal(n0)
                h_polish = project_onto_constraint(h_polish, target_sum)
                for eps in [0.0003, 0.0001, 0.00003]:
                    elapsed = time.time() - start_time
                    if elapsed > budget_s * 0.7:
                        break
                    h_p, c5_p = slsqp_optimize(h_polish, eps, 250)
                    if c5_p < best_c5:
                        best_c5 = c5_p
                        best_h = h_p.copy()
                    h_polish = h_p

        n_sa_rounds += 1
        remaining = budget_s - (time.time() - start_time) - 60

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.85:
        h_final = best_h + 0.003 * rng.standard_normal(n0)
        h_final = project_onto_constraint(h_final, target_sum)
        for eps in [0.00003, 0.00001, 0.000003]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.85:
                break
            h_new, c5 = slsqp_optimize(h_final, eps, 500)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h_final = h_new

    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.92:
        for n_try in [160, 200]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.92:
                break
            sub_budget = (budget_s * 0.92 - elapsed) / max(1, 200 - n_try + n_try - 160)
            h_try, c5_try, _ = optimize_at_n(best_h, n_try, sub_budget, rng, start_time)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
