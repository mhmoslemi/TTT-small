import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

        if i == j:
            continue
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]

    return best_h, best_c5

def gen_perturb(h_best, rng, target_sum, strategy):
    n = len(h_best)
    if strategy == 0:
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n), target_sum)
    elif strategy == 1:
        max_len = max(3, n // 5)
        blen = int(rng.integers(3, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n), target_sum)
    elif strategy == 2:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.6, 0.85))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    elif strategy == 3:
        freq = int(rng.integers(2, max(3, n // 15)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.05, 0.15))
        pert = amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
        return project_onto_constraint(h_best + pert, target_sum)
    elif strategy == 4:
        idx = rng.choice(n, size=max(5, n // 8), replace=False)
        h_p = h_best.copy()
        h_p[idx] = np.clip(h_p[idx] + 0.2 * rng.standard_normal(len(idx)), 0.0, 1.0)
        return project_onto_constraint(h_p, target_sum)
    else:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: Quick SLSQP on initial
    if rem() > 400:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Projected gradient
    if rem() > 350:
        tl = min(rem() - 300, 60)
        h_pg, c5_pg = proj_grad(best_h, 3, 0.01, 0.06, 150000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 3: Multi-start exploration
    inits = []

    # Binary randoms
    for _ in range(4):
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        inits.append(h_bin)

    # Sine-based
    for freq in [3, 5, 7, 11, 13]:
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.15, 0.35))
        hs = 0.5 + amp * np.sin(2.0 * np.pi * freq * np.arange(n) / n + phase)
        hs = project_onto_constraint(hs, target_sum)
        inits.append(hs)

    # Mixes with best
    for alpha in [0.3, 0.5, 0.7]:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        h_mix = project_onto_constraint(alpha * best_h + (1.0 - alpha) * h_bin, target_sum)
        inits.append(h_mix)

    # Perturbations of best
    for s in range(4):
        inits.append(gen_perturb(best_h, rng, target_sum, s))

    for h_init in inits:
        if rem() < 100:
            break
        h_opt, c5_opt = slsqp_optimize(h_init, 0.005, 120)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Fine SLSQP on best
    if rem() > 200:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 150:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Projected gradient + swap search
    if rem() > 120:
        tl = min(rem() - 80, 50)
        h_pg, c5_pg = proj_grad(best_h, 2, 0.005, 0.04, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 80:
        tl = min(rem() - 50, 40)
        h_sw, c5_sw = swap_local_search(best_h, 500000, tl, start_time, rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 6: Resolution increase
    if rem() > 350:
        n1 = 2 * n
        old_t = np.linspace(0.0, 1.0, n)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = project_onto_constraint(h_up + 0.008 * rng.standard_normal(n1), n1 / 2.0)

        h_opt_u, c5_opt_u = slsqp_optimize(h_up, 0.005, 200)
        if rem() > 150:
            h_opt_u, c5_2 = slsqp_optimize(h_opt_u, 0.002, 200)
            if c5_2 < c5_opt_u:
                c5_opt_u = c5_2
                h_opt_u = h_opt_u
        if rem() > 80:
            h_opt_u, c5_3 = slsqp_optimize(h_opt_u, 0.001, 150)
            if c5_3 < c5_opt_u:
                c5_opt_u = c5_3
                h_opt_u = h_opt_u

        if c5_opt_u < best_c5:
            best_c5 = c5_opt_u
            best_h = h_opt_u.copy()
            n = n1
            target_sum = n / 2.0

    # Phase 7: More restarts at current resolution
    n_restart = 0
    while rem() > 150 and n_restart < 20:
        strategy = n_restart % 6
        h_pert = gen_perturb(best_h, rng, target_sum, strategy)
        eps_use = 0.002 if n_restart < 8 else 0.001
        h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, 150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        n_restart += 1

    # Phase 8: Final fine SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final polish
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0002, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 40:
        tl = min(rem() - 20, 30)
        h_pg, c5_pg = proj_grad(best_h, 1, 0.001, 0.02, 100000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 30:
        tl = min(rem() - 15, 20)
        h_sw, c5_sw = swap_local_search(best_h, 300000, tl, start_time, rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
