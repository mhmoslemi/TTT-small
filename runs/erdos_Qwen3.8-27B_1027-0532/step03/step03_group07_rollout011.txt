def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid

    if abs(diff) > 1e-10:

from scipy.optimize import minimize

_global_n_pts = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _top_m_subgrad(h, m, alpha_w, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:m]
    w = np.exp(alpha_w * (corr[top_idx] - corr[top_idx[0]]))
    w = w / w.sum()
    g = np.zeros(len(h))
    for i in range(m):
        g += w[i] * _grad_corr_term(h, top_idx[i], dx)
    return g, corr

def _nesterov_opt(h0, n, target_sum, n_iters, deadline, alpha0=0.4, m_start=3, m_end=8, w_alpha=30.0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        m_cur = m_start + (m_end - m_start) * it // max(1, n_iters)
        g, _ = _top_m_subgrad(y, m_cur, w_alpha, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _equality_con(x):
    return x.sum() - _global_target_sum

def _equality_jac(x):
    return np.ones(_global_n_pts)

def _lse_objective(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500.0:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _slsqp_solve(h0, n, target_sum, alpha, maxiter, deadline=None):
    global _global_target_sum, _global_n_pts
    _global_target_sum = target_sum
    _global_n_pts = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _equality_con, 'jac': _equality_jac}]
    try:
        res = minimize(_lse_objective, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        if time.time() > (deadline or 0):
            return h0
        return h0

def _coord_descent_continuous(h0, n, target_sum, max_rounds, deadline, n_trials=7):
    """Coordinate descent with continuous line search on each coordinate."""
    h = h0.copy()
    best_c = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            # Golden section search for optimal value of h[j]
            lo, hi = 0.0, 1.0
            # Try a few points first for a quick assessment
            candidates = np.linspace(0.0, 1.0, n_trials)
            best_cand_c = best_c
            best_cand_v = old

                h[j] = v
                c = _c5_of(h)
                if c < best_cand_c - 1e-14:
                    best_cand_c = c
                    best_cand_v = v
            if best_cand_v != old:
                h[j] = best_cand_v
                best_c = best_cand_c
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h

def _coord_descent_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h

            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _block_cd(h0, n, target_sum, block_size, n_rounds, deadline, rng):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    h = h0.copy()
    best_c = _c5_of(h)
    n_blocks = n // block_size
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        # Randomize block order
        block_order = rng.permutation(n_blocks)
        for bi in block_order:
            if time.time() > deadline:
                break
            start = bi * block_size
            end = start + block_size
            old_block = h[start:end].copy()
            # Try perturbations of this block
            best_block_c = best_c
            best_block = old_block.copy()
            for trial in range(5):
                candidate = old_block + rng.normal(0, 0.15, block_size)
                candidate = np.clip(candidate, 0.0, 1.0)
                h[start:end] = candidate
                # Fix sum approximately by adjusting this block
                diff = target_sum - h.sum()
                if abs(diff) > 1e-10:
                    adj = diff / block_size
                    h[start:end] = np.clip(old_block + adj, 0.0, 1.0)
                    # If still off, do a proper fix
                    if abs(h.sum() - target_sum) > 1e-8:
                        h = _fix_sum(h, target_sum)
                        h[start:end] = h[start:end]  # re-read after fix
                c = _c5_of(h)
                if c < best_block_c - 1e-14:
                    best_block_c = c
                    best_block = h[start:end].copy()
            h[start:end] = best_block
            h = _fix_sum(h, target_sum)
            c = _c5_of(h)
            if c < best_c:
                best_c = c
                improved = True
        if not improved:
            break
    return h, best_c

def _variance_balance(h0, n, target_sum, n_iters, deadline):
    """Try to equalize the top-M correlations by moving in the variance-reducing direction."""
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    M = 5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_M = np.argsort(corr)[::-1][:M]
        c_top = corr[top_M]
        c_mean = c_top.mean()
        # Variance-reducing direction: sum (c_i - mean) * grad_i
        g = np.zeros(n)
        for i in range(M):
            g += (c_top[i] - c_mean) * _grad_corr_term(x, top_M[i], dx)
        # Normalize
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        # Line search
        alpha = 0.3 / (1.0 + 0.1 * it)
        x_new = _project(x - alpha * g, target_sum)
        cv = _c5_of(x_new)
        if cv >= best_val:
            # Try smaller step
            for shr in [0.5, 0.25, 0.1]:
                x_try = _project(x - alpha * shr * g, target_sum)
                cv_try = _c5_of(x_try)
                if cv_try < best_val:
                    x_new = x_try
                    cv = cv_try
                    break
            else:
                break
        x = x_new
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _lse_grad(h, alpha, n):
    """Gradient of the log-sum-exp objective."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights = weights / weights.sum()
    g = np.zeros(n)
    dx = 2.0 / n
    # Only compute gradients for significant weights
    threshold = 1e-6
    significant = np.where(weights > threshold)[0]
    for idx in significant:
        g += weights[idx] * _grad_corr_term(h, idx, dx)
    return g * (2.0 / n)  # scale by the 2/n factor in the objective

def _lse_line_search(h, g, alpha_lse, n, target_sum, max_step):
    """Backtracking line search on the LSE objective."""
    f0 = _lse_objective(h, alpha_lse, n)
    step = max_step
    for _ in range(20):
        h_new = _project(h - step * g, target_sum)
        f_new = _lse_objective(h_new, alpha_lse, n)
        if f_new < f0 - 1e-4 * step * np.dot(g, h_new - h):
            return h_new
        step *= 0.5
    return _project(h - step * g, target_sum)

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n_pts = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _lse_value(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500.0:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _lse_grad(h, alpha, n, w_thresh=1e-6):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    total = weights.sum()
    if total < 1e-300:
        return np.zeros(n)
    weights = weights / total
    g = np.zeros(n)
    dx = 2.0 / n
    significant = np.where(weights > w_thresh)[0]
    for idx in significant:
        g += weights[idx] * _grad_corr_term(h, idx, dx)
    return g

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n_pts)

def _slsqp_solve(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n_pts
    _global_target_sum = target_sum
    _global_n_pts = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_lse_value, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_lse(h0, n, target_sum, n_iters, deadline, alpha_lse=100.0,
                  init_step=1.0, shrink=0.7):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    step = init_step
    stall = 0
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = _lse_grad(y, alpha_lse, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val - 1e-10:
            best_val = cv
            best = x.copy()
            stall = 0
            step = min(step * 1.1, 2.0)
        else:
            stall += 1
            step *= shrink
            if stall > 80:
                t = 1.0
                y = x.copy()
                stall = 0
                step = min(step * 2.0, 1.5)
    return best, best_val

def _nesterov_subgrad(h0, n, target_sum, n_iters, deadline, alpha0=0.4,
                      m_start=3, m_end=12, w_alpha=50.0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        m_cur = m_start + (m_end - m_start) * min(it, n_iters) // max(1, n_iters)
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:m_cur]
        w = np.exp(w_alpha * (corr[top_idx] - corr[top_idx[0]]))
        w = w / w.sum()
        g = np.zeros(n)
        for i in range(m_cur):
            g += w[i] * _grad_corr_term(y, top_idx[i], dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        g = g / g_norm
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _block_cd(h0, n, target_sum, block_size, n_rounds, deadline, rng):
    h = h0.copy()
    best_c = _c5_of(h)
    n_blocks = n // block_size
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        block_order = rng.permutation(n_blocks)
        for bi in block_order:
            if time.time() > deadline:
                break
            start = bi * block_size
            end = start + block_size
            old_block = h[start:end].copy()
            best_block = old_block.copy()
            best_block_c = best_c
            for trial in range(6):
                scale = 0.2 * (1.0 - 0.5 * trial / 6.0)
                candidate = old_block + rng.normal(0, scale, block_size)
                candidate = np.clip(candidate, 0.0, 1.0)
                h[start:end] = candidate
                h = _fix_sum(h, target_sum)
                h[start:end] = np.clip(h[start:end], 0.0, 1.0)
                c = _c5_of(h)
                if c < best_block_c - 1e-14:
                    best_block_c = c
                    best_block = h[start:end].copy()
                    h_saved = h.copy()
            h[start:end] = best_block
            h = _fix_sum(h, target_sum)
            c = _c5_of(h)
            if c < best_c:
                best_c = c
                improved = True
        if not improved:
            break
    return h, best_c

def _coord_descent_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return h
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_fractional(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    n_trials = 9
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            best_cv = _c5_of(h)
            best_v = old
            candidates = np.linspace(0.0, 1.0, n_trials)
            for v in candidates:
                if v == old:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_cv - 1e-14:
                    best_cv = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _perturb_refine(h0, n, target_sum, deadline, rng, n_perturbs=25):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for _ in range(n_perturbs):
        if time.time() > deadline:
            break
        h_p = h0.copy()
        n_chg = rng.randint(max(1, n // 30), max(2, n // 10) + 1)
        idx = rng.choice(n, size=n_chg, replace=False)
        for i in idx:
            h_p[i] = np.clip(h_p[i] + rng.normal(0, 0.1), 0.0, 1.0)
        h_p = _fix_sum(h_p, target_sum)
        h_n, c5_n = _nesterov_lse(h_p, n, target_sum,
                                   n_iters=60, deadline=deadline,
                                   alpha_lse=100.0, init_step=0.5)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()
    return best_h, best_c5

def _variance_balance(h0, n, target_sum, n_iters, deadline):
    dx = 2.0 / n
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    M = 5
    alpha = 0.3
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_M = np.argsort(corr)[::-1][:M]
        c_top = corr[top_M]
        c_mean = c_top.mean()
        spread = c_top.max() - c_top.min()
        if spread < 1e-6:
            break
        g = np.zeros(n)
        for i in range(M):
            g += (c_top[i] - c_mean) * _grad_corr_term(x, top_M[i], dx)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        step = alpha / (1.0 + 0.2 * it)
        x_new = _project(x - step * g, target_sum)
        cv = _c5_of(x_new)
        if cv >= best_val:
            for shr in [0.5, 0.25, 0.1]:
                x_try = _project(x - step * shr * g, target_sum)
                cv_try = _c5_of(x_try)
                if cv_try < best_val:
                    x_new = x_try
                    cv = cv_try
                    break
            else:
                break
        x = x_new
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

            best_h = h_cf.copy()

    if time.time() < deadline - 3:
        h_pol = _nesterov_lse(best_h, n_best, target,
                               n_iters=200, deadline=deadline - 1,
                               alpha_lse=500.0, init_step=0.3)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
