    """Compute smooth max and softmax weights of corr with parameter alpha."""

    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)

def softmax_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    return exp_vals / np.sum(exp_vals)


def compute_grad_weights(h, weights):
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    m_max = n - 1
    for m in range(-m_max, m_max + 1):
        w = weights[m + m_max]
        if w < 1e-14:
            continue
        if m >= 0:
            grad[:n-m] += w * g[m:]
        else:
            grad[-m:] += w * g[:n+m]
        if m <= 0:
            grad[:n+m] -= w * h[-m:]
        else:
            grad[m:] -= w * h[:n-m]
    return grad


def lbfgs_phase(h, target, time_limit, alphas=(10, 30, 80, 150, 300)):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    n_alphas = len(alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        
        def obj(x, a=alpha):
            x_clip = np.clip(x, 0.0, 1.0)
            corr = compute_full_corr(x_clip)
            val = smooth_max_value(corr, a)
            w = softmax_weights(corr, a)
            grad = compute_grad_weights(x_clip, w)
            return val, grad
        
        x0 = project_feasible(best, target)
        maxiter = int(max(50, remaining / (n_alphas - ai) * 2))
        
        try:
            res = minimize(obj, x0, method='L-BFGS-B', jac=True,
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-20,
                                   'gtol': 1e-8, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score


def coord_descent_fast(h, target, time_limit, seed=42):

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def compute_D_matrix(h):
    n = len(h)
    m = 2 * n - 1
    g = 1.0 - h
    D = np.zeros((n, m))
    
    # Compute R[k] and S[k]
    R = np.zeros(m)
    S = np.zeros(m)
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(n, k + 1)
        corr_idx = np.arange(j_start, j_end) + k - (n - 1)
        R[k] = g[corr_idx].sum() / n
        S[k] = h[corr_idx].sum() / n
    
    for k in range(m):
        s = k - n + 1
        P = np.zeros(n)
        if s >= 0:
            P[:n - s] = g[s:]
        else:
            P[-s:] = g[:n + s]
        Q = np.zeros(n)
        if s > 0:
            Q[s:] = h[:n - s]
        elif s < 0:
            Q[:n + s] = h[-s:]
        else:
            Q[:] = h
        D[:, k] = P - Q - R[k] + S[k]
    
    return D

            break
        
        def obj(x, a=alpha):
            corr = compute_full_corr(x)
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            exp_vals = np.exp(shifted)
            val = c_max + np.log(np.sum(exp_vals)) / a
            w = exp_vals / np.sum(exp_vals)
            grad = compute_grad_weights(x, w)
            return val, grad
        
        x0 = project_feasible(best, target)
        try:
            res = minimize(obj, x0, method='L-BFGS-B', jac=True,
                          bounds=bounds,
                          options={'maxiter': 300, 'ftol': 1e-20,
                                   'gtol': 1e-10, 'maxls': 30})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score


def compute_grad_weights(h, weights):
    n = len(h)
    g = 1.0 - h
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        w = weights[m + n - 1]
        if w < 1e-14:
            continue
        if m >= 0:
            grad[:n-m] += w * g[m:]
        else:
            grad[-m:] += w * g[:n+m]
        if m <= 0:
            grad[:n+m] -= w * h[-m:]
        else:
            grad[m:] -= w * h[:n-m]
    return grad

Hmm wait, I have a bug. In the fast path, I'm checking `new_max + (2.0/n) < best_score - 1e-13`. The `(2.0/n)` is an upper bound on how much the true C5 could differ from the fast estimate (due to the O(d^2) term and any bound violations). But this might be too conservative.

Actually, the fast formula is exact when the projection is clean. The only error source is:
1. The O(d^2) term in the correlation (from the -correlate(delta_h, delta_h) term)
2. The projection being different when bounds are hit

For (1): the error is at most d^2 * n (very small for d < 0.01)
For (2): if no element hits bounds, the formula is exact.

So the correct check is: if the fast formula says new_max < best_score, verify with the full computation. This is what I should do.

Let me simplify: just use the fast formula to identify promising (i, d) pairs, then verify with the full computation.

Actually, for the verification, I only need to do the full computation when the fast formula suggests an improvement. The fast formula is O(n) (vector add + max), while the full computation is also O(n^2). But the fast formula will reject most candidates (those that don't improve), so the full computation is only done for a few candidates per sweep.
