    return np.correlate(h, 1.0 - h, mode='full') * dx

    corr = compute_corr_vec(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_vec(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):

        if wk < 1e-14:

            grad[:m] += wk * dx * omh[r:]

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr_vec(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_vec(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:

        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r

                        if idx + r < n and h_bin[idx + r] >= 0.5:
                            bad_shift_indices.append((idx + r, idx))
            if len(bad_shift_indices) > 0:
                i, j = bad_shift_indices[rng.integers(len(bad_shift_indices))]
            else:
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
        else:
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]

        h_bin[i] = 0.0
        h_bin[j] = 1.0
        c5_new = compute_c5(h_bin)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_bin.copy()
            consec_fail = 0
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
        else:
            h_bin[i] = 1.0
            h_bin[j] = 0.0
            consec_fail += 1

        if consec_fail > max_consec_fail:
            h_bin = best_h.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            consec_fail = 0
            max_consec_fail = max(5000, max_consec_fail // 2)

    return best_h, best_c5

def nesterov_proj_grad(h_init, lr, n_steps, time_limit, start_time, k=1, temp=0.01):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    beta = 0.9

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        blend = beta * (1.0 + 0.5 / (step + 1))
        h_blend = blend * h_cur + (1.0 - blend) * h_prev

        corr = compute_corr_vec(h_blend)
        cmax = np.max(corr)
        x = (corr - cmax) / temp
        x = x - np.max(x)
        ex = np.exp(x)
        w = ex / np.sum(ex)

        grad = np.zeros(n)
        omh = 1.0 - h_blend
        for kk in range(2 * n - 1):
            wk = w[kk]
            if wk < 1e-14:
                continue
            s = kk - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * omh[:m]
                grad[:m] -= wk * h_blend[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * omh[r:]
                grad[r:] -= wk * h_blend[:m]

        dx = 2.0 / n
        v = beta * v - lr * grad * dx
        h_new = project_onto_constraint(h_cur + v, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

        h_prev = h_cur
        h_cur = h_new

    return best_h, best_c5

def multi_start_random_binary(n, rng, n_starts, time_limit, start_time):
    best_h = None
    best_c5 = 1.0
    per_start = max(10, (time_limit - (time.time() - start_time)) / n_starts)

    for _ in range(n_starts):
        if time.time() - start_time > time_limit:
            break
        h_rand = (rng.random(n) < 0.5).astype(float)
        h_rand = to_binary(h_rand, rng)
        h_ls, c5_ls = binary_local_search(h_rand, rng, per_start, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP from initial (up to 120s)
    if rem() > 400:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Binary local search (up to 400s)
    h_bin = to_binary(best_h, rng)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    if rem() > 300:
        tl = min(rem() - 200, 400)
        h_ls, c5_ls = binary_local_search(h_bin, rng, tl, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 3: Targeted binary search (up to 150s)
    if rem() > 200:
        tl = min(rem() - 150, 150)
        h_ls, c5_ls = targeted_binary_search(best_h, rng, tl, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: SLSQP from best binary (up to 120s)
    if rem() > 180:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Simulated annealing in binary space (up to 120s)
    if rem() > 150:
        h_bin_sa = to_binary(best_h, rng)
        tl = min(rem() - 100, 120)
        h_sa, c5_sa = binary_local_search(h_bin_sa, rng, tl, start_time, use_sa=True, sa_temp=0.008)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Nesterov projected gradient (up to 100s)
    if rem() > 120:
        tl = min(rem() - 80, 100)
        h_ng, c5_ng = nesterov_proj_grad(best_h, 0.1, 100000, tl, start_time, k=1, temp=0.005)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    # Phase 7: More binary local search (up to 100s)
    h_bin2 = to_binary(best_h, rng)
    if rem() > 100:
        tl = min(rem() - 60, 100)
        h_ls, c5_ls = binary_local_search(h_bin2, rng, tl, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 8: Fine SLSQP (up to 80s)
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Multi-restart with random binaries
    if rem() > 80:
        tl = min(rem() - 40, 60)
        h_mr, c5_mr = multi_start_random_binary(n0, rng, 5, tl + start_time - start_time, start_time)
        if h_mr is not None and c5_mr < best_c5:
            best_c5 = c5_mr
            best_h = h_mr.copy()

    # Phase 10: Final binary polish (up to 60s)
    h_bin3 = to_binary(best_h, rng)
    if rem() > 40:
        tl = min(rem() - 20, 60)
        h_ls, c5_ls = binary_local_search(h_bin3, rng, tl, start_time)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 11: Final SLSQP with very small eps
    if rem() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0003, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final: compare binary vs continuous
    h_bin_final = to_binary(best_h, rng)
    c5_bin_final = compute_c5(h_bin_final)
    h_cont = project_onto_constraint(best_h, len(best_h) / 2.0)
    c5_cont = compute_c5(h_cont)

    if c5_bin_final <= c5_cont:
        final_h = h_bin_final
        final_c5 = c5_bin_final
    else:
        final_h = h_cont
        final_c5 = c5_cont

    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, len(final_h)
