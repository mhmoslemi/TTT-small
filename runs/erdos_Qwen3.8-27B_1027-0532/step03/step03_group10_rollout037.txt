```python
def compute_delta_vectorized(h, i, j, N):
    """Returns (k_range, delta) for all k where the correlation changes."""
    # Full range of affected k
    k_lo = min(i - N + 1, -i, j - N + 1, -j)
    k_hi = max(i, N - 1 - i, j, N - 1 - j)
    k_range = np.arange(k_lo, k_hi + 1)
    
    # Initialize delta to 0
    delta = np.zeros(len(k_range))
    
    # Term 1: n = i-k valid when i-N+1 <= k <= i
    mask1 = (k_range >= i - N + 1) & (k_range <= i)
    if np.any(mask1):
        idx = i - k_range[mask1]  # all in [0, N)
        delta[mask1] -= (1 - h[idx])
    
    # Term 2: n = i valid when -i <= k <= N-1-i
    mask2 = (k_range >= -i) & (k_range <= N - 1 - i)
    if np.any(mask2):
        idx = i + k_range[mask2]  # all in [0, N)
        delta[mask2] += h[idx]
    
    # Term 3: n = j-k valid when j-N+1 <= k <= j
    mask3 = (k_range >= j - N + 1) & (k_range <= j)
    if np.any(mask3):
        idx = j - k_range[mask3]  # all in [0, N)
        delta[mask3] += (1 - h[idx])
    
    # Term 4: n = j valid when -j <= k <= N-1-j
    mask4 = (k_range >= -j) & (k_range <= N - 1 - j)
    if np.any(mask4):
        idx = j + k_range[mask4]  # all in [0, N)
        delta[mask4] -= h[idx]
    
    # Overlap corrections
    if (i - j) >= k_lo and (i - j) <= k_hi:
        pos = (i - j) - k_lo
        delta[pos] += 1
    if (j - i) >= k_lo and (j - i) <= k_hi:
        pos = (j - i) - k_lo
        delta[pos] += 1
    
    return k_range, delta
```
