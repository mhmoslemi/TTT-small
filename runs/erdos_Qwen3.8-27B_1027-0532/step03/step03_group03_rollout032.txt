def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def compute_grad_full_softmax(h, alpha):
    """Compute gradient of logsumexp(alpha * corr) / alpha using softmax weights over all lags."""
    n = len(h)
    corr = overlap_corrs(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    lag_indices = np.arange(2 * n - 1)
    return compute_grad_weighted(h, lag_indices, weights), c_max

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def adam_subgrad(h_init, target_sum, max_iter, alpha_start, alpha_end, 
                 lr_init, patience, seed):
    """Adam-style subgradient with annealed alpha."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    n_all = 2 * n - 1
    
    for it in range(max_iter):
        # Anneal alpha
        frac = it / max(max_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        corr = overlap_corrs(h)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_s = np.exp(shifted)
        total = np.sum(exp_s)
        weights = exp_s / total
        
        grad = compute_grad_weighted(h, np.arange(n_all), weights)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr *= 1.01
        else:
            stagnation += 1
            lr *= 0.995
        
        h = h_new
        if stagnation > patience:
            break
    
    return best_h, best_score

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def multiplicative_weights(h_init, target_sum, max_iter, gamma, lr, patience, seed):
    """Multiplicative weights for the saddle point formulation."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    n_lags = 2 * n - 1
    h = h_init.copy()
    log_p = np.zeros(n_lags)  # log of probability over lags
    best_h = h.copy()
    best_score = compute_c5(h)
    stagnation = 0
    
    for it in range(max_iter):
        p = np.exp(log_p - np.max(log_p))
        p = p / p.sum()
        
        # Compute weighted gradient
        grad = compute_grad_weighted(h, np.arange(n_lags), p)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Update p using multiplicative weights
        corr = overlap_corrs(h_new)
        log_p = log_p - gamma * (corr - np.max(corr))
        # Normalize to prevent overflow
        log_p = log_p - np.max(log_p)
        
        h = h_new
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        if stagnation > patience:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(lag_indices)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lag_indices[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * omh[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * omh[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)

def adam_optimize(h_init, target_sum, max_iter, alpha_start, alpha_end,
                  lr_init, topk, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    n_all = 2 * n - 1
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    
    for it in range(max_iter):
        frac = it / max(max_iter - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        corr = overlap_corrs(h)
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        w = np.exp(alpha * (top_vals - np.max(top_vals)))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        
        grad = compute_grad_weighted(h, top_idx, w)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 3.0)
        else:
            stagnation += 1
            lr *= 0.995
        h = h_new
        if stagnation > patience:
            break
    
    return best_h, best_score

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def binary_local_search(h_init, target_sum, max_rounds, seed):
    """Local search that tries to improve C5 by adjusting individual values."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        indices = rng.permutation(n)
        for i in indices:
            # Try setting h[i] to 0 or 1 (adjusting another element to maintain sum)
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-6:
                    continue
                h_new = h.copy()
                h_new[i] = val
                # Adjust sum: find another element to adjust
                diff = target_sum - h_new.sum()
                if abs(diff) < 1e-10:
                    pass
                else:
                    # Find j ≠ i to adjust
                    if diff > 0:
                        candidates = np.where((h_new < 1.0) & (np.arange(n) != i))[0]
                    else:
                        candidates = np.where((h_new > 0.0) & (np.arange(n) != i))[0]
                    if len(candidates) == 0:
                        continue
                    j = rng.choice(candidates)
                    h_new[j] += diff
                    h_new = np.clip(h_new, 0.0, 1.0)
                
                if abs(h_new.sum() - target_sum) > 1e-6:
                    continue
                    
                score = compute_c5(h_new)
                if score < best_score - 1e-10:
                    h = h_new
                    best_score = score
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    
    return best_h, best_score
