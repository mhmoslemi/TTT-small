def smooth_max_obj_and_grad(h, alpha):
    """Compute smoothed max of correlations and its gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    
    # Compute gradient
    grad = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i]
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n-lag]
                grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n-m]
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            h[idx] -= (-diff) / len(idx)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def build_full_from_half(h_half):
    n_half = len(h_half)
    h = np.empty(2 * n_half)
    h[:n_half] = h_half
    h[n_half:] = 1.0 - h_half
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n-lag]
                grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n-m]
    return grad

def antiperiodic_obj_grad(x, alpha):
    h = build_full_from_half(x)
    obj = smooth_max_obj(h, alpha)
    grad_full = smooth_max_grad(h, alpha)
    n_half = len(x)
    grad = grad_full[:n_half] - grad_full[n_half:]
    return obj, grad

    bounds = [(0.0, 1.0)] * n_half
    best_h = h_half_init.copy()
    best_score = compute_c5(build_full_from_half(best_h))
    
    for alpha in alphas:
        if budget_check is not None and budget_check():
            break
        def fun(x):
            return antiperiodic_obj_grad(x, alpha)[0]
        def jac(x):
            return antiperiodic_obj_grad(x, alpha)[1]
        try:
            res = minimize(antiperiodic_obj_grad, h_half_init, args=(alpha,),
                          jac=True, method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
            h_new = res.x
            score = compute_c5(build_full_from_half(h_new))
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
            h_half_init = h_new
        except Exception:

    return best_h, best_score

def slsqp_refine_full(h_start, target_sum, alphas, maxiter, ftol):

            score = compute_c5(h_new)
            if score < best_score:
                best_score = score
                h_cur = h_new
        except Exception:
            pass

    """Minimize smooth max w.r.t. h[j], keeping other h fixed."""
    n = len(h)
    v = h[j]
    
    # Compute base correlation (h[j] = 0) and sensitivity
    h_base = h.copy()
    h_base[j] = 0.0
    corr_base = np.correlate(h_base, 1.0 - h_base, mode='full')
    
    # Sensitivity: ∂corr_k/∂h[j]
    # For lag l >= 0:
    #   If l <= j <= n-1: contrib = (1-h[j-l]) from h[j] as first factor
    #   If j <= n-1-l: contrib += -h[j+l] from h[j] in (1-h[j]) as second factor
    # For lag l < 0 (m = -l):
    #   If j >= m: contrib = (1-h[j-m]) from h[j] as first factor (h[j] = h[i+m] where i=j-m)
    #   If j-m < n and j < n: contrib += -h[j-m] ... 
    
    # Actually, let me just compute it numerically:
    # corr with h[j] = 1 minus corr with h[j] = 0
    h_one = h.copy()
    h_one[j] = 1.0
    corr_one = np.correlate(h_one, 1.0 - h_one, mode='full')
    sensitivity = corr_one - corr_base
    
    # Now corr with h[j] = v' is: corr_base + sensitivity * v'
    # (since it's linear in h[j])
    
    # Minimize smooth max of (corr_base + sensitivity * v') over v' in [0, 1]
    def obj_v(vp):
        corr_v = corr_base + sensitivity * vp
        c_max = np.max(corr_v)
        shifted = np.clip(alpha * (corr_v - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    # Use golden section or scipy
    from scipy.optimize import minimize_scalar
    res = minimize_scalar(obj_v, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-10})
    h_new = h.copy()
    h_new[j] = res.x
    return h_new, res.fun

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) == 0:
                break
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 1e-15)[0]
            if len(idx) == 0:
                break
            h[idx] -= (-diff) / len(idx)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    omh = 1.0 - h
    grad = np.zeros(n)
    for i in range(len(corr)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = i - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * omh[:n-lag]
                grad[:n-lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n-m] += w * omh[m:]
                grad[m:] -= w * h[:n-m]
    return grad

def _slsqp_jac(x, alpha):
    return smooth_max_grad(x, alpha)

    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol, time_limit=60):
    global _GLOBAL_N, _GLOBAL_TARGET
    _GLOBAL_N = len(h_start)
    _GLOBAL_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _GLOBAL_N
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    best_score = compute_c5(h_cur)
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        try:
            res = minimize(_slsqp_fun, h_cur, jac=_slsqp_jac, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score
