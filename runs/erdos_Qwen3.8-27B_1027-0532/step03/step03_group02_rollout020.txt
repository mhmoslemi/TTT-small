def interpolate_to(h_src, n_dst):
    """Interpolate h to new resolution, maintaining symmetry"""
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_new = np.interp(x_new, x_old, h_src)
    # Enforce symmetry
    for i in range(n_dst // 2):
        h_new[n_dst - 1 - i] = 1.0 - h_new[i]
    return h_new

            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def build_h_symmetric(x):
    n_half = len(x)
    n = 2 * n_half
    h = np.empty(n)
    h[:n_half] = x
    h[n_half:] = 1.0 - x[::-1]
    return h

def extract_x(h):
    return h[:len(h) // 2].copy()

def coord_descend_sym(x, max_time, rng, patience=50, noise_base=0.008):
    n_half = len(x)
    n = 2 * n_half
    dx = 2.0 / n
    M = 2 * n - 1
    s_arr = np.arange(M) - (n - 1)
    x = np.clip(x.astype(float), 0.0, 1.0).copy()
    h = build_h_symmetric(x)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_x = x.copy()
    no_improve = 0
    t0 = time.time()
    sweep = 0
    while time.time() - t0 < max_time:
        sweep += 1
        for j in range(n_half):
            old_val = x[j]
            j_sym = n - 1 - j
            A = np.zeros(M)
            idx1 = j + s_arr
            m1 = (idx1 >= 0) & (idx1 < n)
            A[m1] += (1.0 - h[idx1[m1]])
            idx2 = j - s_arr
            m2 = (idx2 >= 0) & (idx2 < n)
            A[m2] -= h[idx2[m2]]
            idx3 = j_sym + s_arr
            m3 = (idx3 >= 0) & (idx3 < n)
            A[m3] -= (1.0 - h[idx3[m3]])
            idx4 = j_sym - s_arr
            m4 = (idx4 >= 0) & (idx4 < n)
            A[m4] += h[idx4[m4]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(32):
                mid = (lo + hi) * 0.5
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-14:
                    hi = mid
                elif grad < -1e-14:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                x[j] = new_val
                h[j] = new_val
                h[j_sym] = 1.0 - new_val
                corr += delta * A
        h = build_h_symmetric(x)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:
            best_score = score
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            progress = min(1.0, no_improve / (patience * 3))
            scale = noise_base * (1.0 - 0.5 * progress)
            n_mutate = max(1, int(n_half * 0.1 * (1.0 + 0.5 * progress)))
            x = best_x.copy()
            mu = rng.randint(n_half, size=n_mutate)
            x[mu] += rng.randn(n_mutate) * scale * 3.0
            x = np.clip(x, 0.0, 1.0)
            h = build_h_symmetric(x)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_x, best_score, build_h_symmetric(best_x)

def smooth_max(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_grad(h, alpha):
    n = len(h)
    M = 2 * n - 1
    s_arr = np.arange(M) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    w = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    w /= np.sum(w)
    dx = 2.0 / n
    n_half = n // 2
    grad = np.zeros(n_half)
    for j in range(n_half):
        j_sym = n - 1 - j
        A = np.zeros(M)
        idx1 = j + s_arr
        m1 = (idx1 >= 0) & (idx1 < n)
        A[m1] += (1.0 - h[idx1[m1]])
        idx2 = j - s_arr
        m2 = (idx2 >= 0) & (idx2 < n)
        A[m2] -= h[idx2[m2]]
        idx3 = j_sym + s_arr
        m3 = (idx3 >= 0) & (idx3 < n)
        A[m3] -= (1.0 - h[idx3[m3]])
        idx4 = j_sym - s_arr
        m4 = (idx4 >= 0) & (idx4 < n)
        A[m4] += h[idx4[m4]]
        grad[j] = np.dot(w, A) * dx
    return grad

def lbfgs_sym(x, max_time, alphas=None):
    if alphas is None:
        alphas = [200, 600, 1500, 4000]
    n_half = len(x)
    n = 2 * n_half
    dx = 2.0 / n
    x_cur = np.clip(x.astype(float), 0.0, 1.0).copy()
    best_score = compute_c5(build_h_symmetric(x_cur))
    best_x = x_cur.copy()
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > max_time:
            break
        def obj(xv):
            h = build_h_symmetric(xv)
            corr = np.correlate(h, 1.0 - h, mode='full')
            return smooth_max(corr, alpha) * dx
        def grad(xv):
            h = build_h_symmetric(xv)
            return smooth_grad(h, alpha)
        try:
            res = minimize(obj, x_cur, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n_half,
                          options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-14, 'maxfun': 2000})
            h_new = build_h_symmetric(res.x)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_x = res.x.copy()
                x_cur = res.x.copy()
        except Exception:
            pass
    return best_x, best_score, build_h_symmetric(best_x)

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_new = np.interp(x_new, x_old, h_src)
    for i in range(n_dst // 2):
        h_new[n_dst - 1 - i] = 1.0 - h_new[i]
    return h_new

def coord_descend_asym(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    M = 2 * n - 1
    s_arr = np.arange(M) - (n - 1)
    h = h.copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    t0 = time.time()
    while time.time() - t0 < max_time:
        for j in range(n):
            old_val = h[j]
            A = np.zeros(M)
            js = j - s_arr
            jp = j + s_arr
            mask1 = (jp >= 0) & (jp < n)
            mask2 = (js >= 0) & (js < n)
            A[mask1] += (1.0 - h[jp[mask1]])
            A[mask2] -= h[js[mask2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) * 0.5
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-14:
                    hi = mid
                elif grad < -1e-14:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                h[j] = new_val
                corr += delta * A
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            scale = 0.006 * min(1.0, no_improve / patience)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_score = compute_c5(best_h)
    if n_init % 2 != 0:
        n_init = n_init - 1
        h_init = h_init[:n_init]
        best_h = h_init.copy()
        best_score = compute_c5(best_h)
    for i in range(n_init // 2):
        h_init[n_init - 1 - i] = 1.0 - h_init[i]
    x_init = extract_x(h_init)
    h_sym = build_h_symmetric(x_init)
    sc_sym = compute_c5(h_sym)
    if sc_sym < best_score:
        best_score = sc_sym
        best_h = h_sym
    if tleft() > 400:
        x_cd, sc_cd, h_cd = coord_descend_sym(x_init, min(450, tleft() - 200), rng, patience=60)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            x_init = x_cd
    if tleft() > 150:
        x_lb, sc_lb, h_lb = lbfgs_sym(x_init, min(120, tleft() - 100), alphas=[200, 600, 2000])
        if sc_lb < best_score:
            best_score = sc_lb
            best_h = h_lb
            x_init = x_lb
    if tleft() > 100:
        x_cd2, sc_cd2, h_cd2 = coord_descend_sym(x_init, min(100, tleft() - 80), rng, patience=80, noise_base=0.004)
        if sc_cd2 < best_score:
            best_score = sc_cd2
            best_h = h_cd2
            x_init = x_cd2
    resolutions = [500, 600, 350, 700, 450]
    for n_try in resolutions:
        if tleft() < 180:
            break
        if n_try % 2 != 0 or n_try < 200:
            continue
        h_interp = interpolate_h(best_h, n_try)
        x_try = h_interp[:n_try // 2].copy()
        x_try = np.clip(x_try, 0.0, 1.0)
        x_res, sc_res, h_res = coord_descend_sym(x_try, min(70, tleft() - 150), rng, patience=25)
        if sc_res < best_score:
            best_score = sc_res
            best_h = h_res
            x_init = x_res
    n_best = len(best_h)
    if n_best % 2 == 0:
        x_best = best_h[:n_best // 2]
        if tleft() > 60:
            x_fin, sc_fin, h_fin = coord_descend_sym(x_best, min(50, tleft() - 40), rng, patience=100, noise_base=0.003)
            if sc_fin < best_score:
                best_score = sc_fin
                best_h = h_fin
                x_best = x_fin
        if tleft() > 30:
            x_pol, sc_pol, h_pol = lbfgs_sym(x_best, min(25, tleft() - 25), alphas=[500, 2000, 5000])
            if sc_pol < best_score:
                best_score = sc_pol
                best_h = h_pol
    else:
        target_best = n_best / 2.0
        if tleft() > 60:
            h_fin, sc_fin = coord_descend_asym(best_h, target_best, min(50, tleft() - 40), rng, patience=50)
            if sc_fin < best_score:
                best_score = sc_fin
                best_h = h_fin
    n_final = len(best_h)
    if n_final % 2 == 0:
        for i in range(n_final // 2):
            best_h[n_final - 1 - i] = 1.0 - best_h[i]
    else:
        best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score
    return best_h, best_score, n_final
