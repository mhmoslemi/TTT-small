```python
    k = np.arange(2*n-1)

    j2 = m + (n-1) - k
```

Since f(δ) = max_k (corr[k] + slopes[k]*δ) is convex piecewise linear, the minimum is at a breakpoint. The breakpoints are at:
δ_ij = (corr[j] - corr[i]) / (slopes[i] - slopes[j]) for all pairs where slopes[i] ≠ slopes[j]

There are O((2n)²) = O(n²) such breakpoints. For n=400, that's 160K breakpoints. Evaluating f at each is O(n), giving O(n³) total. Too slow.

But we can do better! The key observation is that we only need to consider breakpoints that are "relevant" - i.e., where both lines i and j are on the upper envelope. The upper envelope has at most 2n-1 = 799 segments (one per line), so at most 798 breakpoints.

Algorithm to find the minimum of the upper envelope:
1. Find the line with the minimum value at δ = 0 (or at the boundary)
2. Follow the upper envelope from that point, tracking which lines are active
3. At each breakpoint, check if the next segment is going up or down
4. The minimum is where the envelope changes from decreasing to increasing

This is essentially the "rotating calipers" approach for the upper envelope. It's O(n log n) to build the envelope and O(1) to find the minimum.

But implementing this correctly is complex. For practical purposes with n=400, the grid search is fine and much simpler.

Actually, let me use a smarter 1D optimization: since f is convex, I can use the subgradient method:
1. Start at δ = 0
2. Compute the subgradient: all slopes[k] where corr[k] + slopes[k]*δ is within ε of the max
3. If all subgradients > 0, decrease δ
4. If all subgradients < 0, increase δ
5. Otherwise (0 is in the subgradient), we're at the optimum
6. Use bisection to find the optimal δ

This converges in O(log(1/ε)) iterations, each taking O(n). Total: O(n log(1/ε)) per coordinate.

```python
import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    cur_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < cur_score:
                h_cur = h_new
                cur_score = sc
        except Exception:
            pass
    return h_cur

def coord_descent_phase(h_init, target_sum, deadline, rng, relaxation=1.0):
    """Coordinate descent with exact per-coordinate line search."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    k_arr = np.arange(2 * n - 1)
    
    no_improve_sweeps = 0
    max_no_improve = 50
    
    while time.time() < deadline:
        order = rng.permutation(n)
        sweep_start_corr = np.max(corr)
        
        for m in order:
            j1 = m + k_arr - (n - 1)
            j2 = m + (n - 1) - k_arr
            mask1 = (j1 >= 0) & (j1 < n)
            mask2 = (j2 >= 0) & (j2 < n)
            
            slopes = np.zeros(2 * n - 1)
            if np.any(mask1):
                slopes[mask1] = 1.0 - h[j1[mask1]]
            if np.any(mask2):
                slopes[mask2] -= h[j2[mask2]]
            
            d_lo = -h[m]
            d_hi = 1.0 - h[m]
            
            if d_hi - d_lo < 1e-12:
                continue
            
            # Coarse grid search
            grid = np.linspace(d_lo, d_hi, 17)
            vals = np.max(corr + np.outer(grid, slopes), axis=1)
            bi = np.argmin(vals)
            
            # Fine grid search
            ilo = max(0, bi - 1)
            ihi = min(16, bi + 1)
            grid2 = np.linspace(grid[ilo], grid[ihi], 17)
            vals2 = np.max(corr + np.outer(grid2, slopes), axis=1)
            bi2 = np.argmin(vals2)
            delta = grid2[bi2] * relaxation
            
            # Clamp delta to valid range
            delta = max(d_lo, min(d_hi, delta))
            
            if abs(delta) > 1e-12:
                h[m] += delta
                corr += slopes * delta
        
        # End of sweep
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            
        if no_improve_sweeps > max_no_improve:
            break

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    h = h.copy()

    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    per = diff / len(idx)
    h[idx] = np.clip(h[idx] + per, 0.0, 1.0)
    return h

def overlap_corrs(h):

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def smooth_lse_grad(x, alpha):

    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))

        if t_left > 80:
            h_ref = slsqp_refine(best_h, target_try, [100, 300, 800], 1000, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = score
                best_h = h_ref

    t_left = budget_s - 30.0 - (time.time() - start_time)
    if t_left > 60:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final = slsqp_refine(best_h, target_use, [500, 1000, 2000], 1000, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    t_left = budget_s - 30.0 - (time.time() - start_time)
    if t_left > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        t_cd2 = time.time() + t_left * 0.6
        h_new, score = coord_descent_phase(best_h, target_use, t_cd2, rng, relaxation=1.0, max_stall=40)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
