def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

import time
import numpy as np
from scipy.optimize import minimize

_global_target = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _obj_slsqp(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _con_fun(x):
    return x.sum() - _global_target

def _con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target, _global_n
    _global_target = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_slsqp, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_subgrad(h0, n, target_sum, iters, alpha0, top_m, sg_alpha, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        if sg_alpha < 10:
            weights = np.ones(len(corr)) / len(corr)
        else:
            weights = np.exp(sg_alpha * (corr - max_corr))
            total = weights.sum()
            if total < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights /= total
        top_indices = np.argsort(weights)[::-1][:top_m]
        w_top = weights[top_indices].copy()
        w_top /= w_top.sum()
        g = np.zeros(n)
        for idx in range(len(top_indices)):
            g += w_top[idx] * _grad_corr(y, top_indices[idx], dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_descent_fine(h0, n, target_sum, max_rounds, deadline, n_cand=30):
    h = h0.copy()
    best_val = _c5_of(h)
    candidates = np.linspace(0.0, 1.0, n_cand)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 2:
                break
            old = h[j]
            best_c = best_val
            best_v = old
            for v in candidates:
                if abs(v - old) < 0.005:
                    continue
                h[j] = v
                h_fixed = _fix_sum(h, target_sum)
                c = _c5_of(h_fixed)
                if c < best_c - 1e-13:

                    best_v = v
            if best_v != old:
                h[j] = best_v
                h = _fix_sum(h, target_sum)
                best_val = _c5_of(h)
                improved = True
        if not improved:
            break
    return h, best_val

def _perturb_reoptimize(h0, n, target_sum, deadline, rng):

def _multi_nesterov(h0, n, target_sum, deadline):
    best = h0.copy()
    best_val = _c5_of(h0)

def _interp_res(h_src, n_src, n_dst):

    x_src = np.linspace(0, n_src - 1, n_src)
