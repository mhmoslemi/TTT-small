    h = h.copy()
    for round in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            # Compute active lags' coefficients
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            # Find active lags (within some tolerance)
            active = np.where(corr >= max_corr - 0.01 * max_corr)[0]
            # For each active lag, compute a_d and b_d for coordinate j
            candidates = [old, 0.0, 1.0]
            a_vals = []
            b_vals = []
            c_vals = []  # constant term (corr[d] - a_d*h[j] - b_d)
            for d_idx in active:
                lag = d_idx - (n - 1)
                # Compute a_j and b_j for this lag
                a = 1.0
                b = 0.0
                if 0 <= j + lag < n:
                    a -= h[j + lag]
                if 0 <= j - lag < n:
                    a -= h[j - lag]
                    b += h[j - lag]
                # c_d = corr[d_idx] - a * old - b
                c = corr[d_idx] - a * old - b
                a_vals.append(a)
                b_vals.append(b + c)
                # Find intersections
                for prev_idx in range(len(a_vals) - 1):
                    # a_vals[prev_idx]*x + b_vals[prev_idx] = a_vals[-1]*x + b_vals[-1]
                    denom = a_vals[-1] - a_vals[prev_idx]
                    if abs(denom) > 1e-10:
                        x_int = (b_vals[prev_idx] - b_vals[-1]) / denom
                        if 0.0 < x_int < 1.0:
                            candidates.append(x_int)
            # Evaluate max at all candidates
            best_x = old
            best_val = np.max([a * old + b for a, b in zip(a_vals, b_vals)])
            for x_cand in candidates:
                if x_cand < 0 or x_cand > 1:
                    continue
                val = np.max([a * x_cand + b for a, b in zip(a_vals, b_vals)])
                if val < best_val - 1e-14:
                    best_val = val
                    best_x = x_cand
            h[j] = best_x
            if abs(best_x - old) > 1e-10:
                improved = True
        if not improved:
            break
        # Fix sum
        h = _fix_sum(h, target_sum)
    return h

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _exact_coord_descent(h0, n, target_sum, max_rounds=80, deadline=None, n_active_frac=0.15):
    h = h0.copy()
    dx = 2.0 / n
    n_corr = 2 * n - 1
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        n_active = max(5, int(n_active_frac * n_corr))
        top_indices = np.argsort(corr)[::-1][:n_active]
        improved_any = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                break
            old = h[j]
            a_vals = np.zeros(n_active)
            b_vals = np.zeros(n_active)
            for idx, d_idx in enumerate(top_indices):
                lag = d_idx - (n - 1)
                a = 1.0
                b = 0.0
                jp = j + lag
                jm = j - lag
                if 0 <= jp < n:
                    a -= h[jp]
                if 0 <= jm < n:
                    a -= h[jm]
                    b += h[jm]
                a_vals[idx] = a
                b_vals[idx] = corr[d_idx] - a * old + b
            f0 = np.max(b_vals)
            f1 = np.max(a_vals + b_vals)
            best_x = old
            best_val = np.max(a_vals * old + b_vals)
            if f0 <= best_val + 1e-15:
                candidate = 0.0
                val = f0
                if val < best_val - 1e-14:
                    best_val = val
                    best_x = candidate
            if f1 <= best_val + 1e-15:
                candidate = 1.0
                val = f1
                if val < best_val - 1e-14:
                    best_val = val
                    best_x = candidate
            m = n_active
            if m >= 2:
                a_i = a_vals[:, None]
                a_j = a_vals[None, :]
                b_i = b_vals[:, None]
                b_j = b_vals[None, :]
                denom = a_j - a_i
                numer = b_i - b_j
                abs_denom = np.abs(denom)
                safe = abs_denom > 1e-12
                x_int = np.where(safe, numer / np.where(safe, denom, 1.0), -1.0)
                valid = (x_int > 1e-10) & (x_int < 1.0 - 1e-10) & safe
                if np.any(valid):
                    valid_xs = x_int[valid]
                    valid_xs = np.unique(valid_xs)
                    for xv in valid_xs:
                        fv = np.max(a_vals * xv + b_vals)
                        if fv < best_val - 1e-14:
                            best_val = fv
                            best_x = xv
                    if len(valid_xs) > 20:
                        best_on_valid = np.min([np.max(a_vals * xv + b_vals) for xv in valid_xs])
                        for xv in valid_xs:
                            fv = np.max(a_vals * xv + b_vals)
                            if abs(fv - best_on_valid) < 1e-12 and fv < best_val - 1e-14:
                                best_val = fv
                                best_x = xv
            if abs(best_x - old) > 1e-10:
                h[j] = best_x
                improved_any = True
                for idx, d_idx in enumerate(top_indices):
                    corr[d_idx] += a_vals[idx] * (best_x - old)
        h = _fix_sum(h, target_sum)
        if not improved_any:
            break
    return h

def _nesterov_active(h0, n, target_sum, n_iters, alpha0, deadline=None, top_m=8):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    momentum = 0.9
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        threshold = max_corr - 0.02 * max_corr
        active = np.where(corr >= threshold)[0]
        if len(active) > top_m:
            top_idx = np.argsort(corr[active])[::-1][:top_m]
            active = active[top_idx]
        weights = corr[active] - max_corr + 0.02 * max_corr
        weights = weights ** 3
        if weights.sum() > 0:
            weights = weights / weights.sum()
        else:
            weights = np.ones(len(active)) / len(active)
        g = np.zeros(n)
        for wi, idx in zip(weights, active):
            g += wi * _grad_corr_term(y, idx, dx)
        momentum = max(0.7, momentum * 0.999)
        alpha = alpha0 / (1.0 + 0.05 * it)
        x_new = _project(y - alpha * g + momentum * (y - x), target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.4 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.8 - 0.4 * t_frac
        cr = 0.3 + 0.4 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 12:
            break
    return best_h, best_c5

def _pairwise_swap_descent(h0, n, target_sum, max_rounds=20, deadline=None):
    h = h0.copy()
    best_c5 = _c5_of(h)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        rng = np.random.RandomState(42 + rnd)
        n_pairs = min(50, n * (n - 1) // 2)
        for _ in range(n_pairs):
            i, j = rng.choice(n, size=2, replace=False)
            if i == j:
                continue
            di = h[i]
            dj = h[j]
            for delta in [0.05, 0.1, 0.15, 0.2]:
                ni = np.clip(di + delta, 0, 1)
                nj = np.clip(dj - delta, 0, 1)
                h[i], h[j] = ni, nj
                c = _c5_of(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    improved = True
                    break
                h[i], h[j] = di, dj
        if not improved:

    return h, best_c5

        c5_f = _c5_of(h_final)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_final.copy()

    if time.time() < deadline - 10:
        h_sl2 = _slsqp_opt(best_h, n_best, target, 8000, maxiter=400)
        c5_sl2 = _c5_of(h_sl2)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = h_sl2.copy()

    if time.time() < deadline - 8:
        h_cd_final = _exact_coord_descent(best_h, n_best, target, max_rounds=20,
                                           deadline=deadline - 6, n_active_frac=0.04)
        c5_cd_final = _c5_of(h_cd_final)
        if c5_cd_final < best_c5:
            best_c5 = c5_cd_final
            best_h = h_cd_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
