import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def lse_value(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def lse_gradient(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    exps = np.exp(shifted)
    w = exps / exps.sum()
    
    P = np.zeros(n)
    Q = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            P[k:] += wm * (1.0 - h[:n - k])
            Q[:n - k] += wm * h[k:]
        else:
            j = -k
            P[:n - j] += wm * (1.0 - h[j:])
            Q[j:] += wm * h[:n - j]
    
    return dx * (P - Q)


def grad_descent_phase(h, target, dx, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    h = h.copy()
    best = h.copy()
    best_score = compute_c5(h)
    
    alpha = 30.0
    lr = 0.05
    no_improve = 0
    best_no_improve_lr = lr
    
    while time.time() - start < time_limit:
        grad = lse_gradient(h, alpha, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Try several step sizes
        improved = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05]:
            step = lr * mult
            h_new = h - step * grad / gnorm
            h_new = project_feasible(h_new, target)
            score = compute_c5(h_new)
            if score < best_score - 1e-15:
                best = h_new.copy()
                best_score = score
                h = h_new
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.85
                alpha = min(alpha * 1.3, 500.0)
            if no_improve > 200:
                # Add small random perturbation
                h = h + rng.normal(0, 0.001, n)
                h = project_feasible(h, target)
                no_improve = 0
                alpha = 30.0
                lr = 0.05
    
    return best, best_score


def slsqp_analytic(h, target, dx, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    
    alphas = [20, 80, 250, 700]
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(50, remaining * 5))
        
        def fun(x, a=alpha):
            return lse_value(x, a, dx)
        
        def grad(x, a=alpha):
            return lse_gradient(x, a, dx)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        try:
            res = minimize(fun, best, jac=grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score


def pair_swap_search(h, target, dx, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    # Precompute which indices are "high" and "low"
    # We'll do random pair swaps
    delta_values = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
    
    no_improve = 0
    max_no_improve = 5000
    
    while time.time() - start < time_limit and no_improve < max_no_improve:
        # Pick a random high and low position
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if best[i] <= best[j]:
            i, j = j, i
        if best[i] - best[j] < 0.01:
            continue
        
        for d in delta_values:
            if d > best[i] - best[j] + 1e-9:
                continue
            if d > best[i] or d > 1.0 - best[j]:
                continue
            h_trial = best.copy()
            h_trial[i] -= d
            h_trial[j] += d
            s = compute_c5(h_trial)
            if s < best_score - 1e-14:
                best_score = s
                best = h_trial
                no_improve = 0
                break
        else:
            no_improve += 1
    
    return best, best_score


def fine_coordinate_descent(h, target, dx, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    deltas = [0.001, -0.001, 0.003, -0.003, 0.005, -0.005, 0.01, -0.01]
    no_improve_sweeps = 0
    max_sweeps_no_improve = 3
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                val = best[i] + d
                if val < -1e-9 or val > 1.0 + 1e-9:
                    continue
                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        no_improve_sweeps = 0 if improved else no_improve_sweeps + 1
        if no_improve_sweeps >= max_sweeps_no_improve:
            deltas = [d * 0.5 for d in deltas]
            if max(abs(d) for d in deltas) < 5e-5:
                break
            no_improve_sweeps = 0
    
    return best, best_score

Actually, I just realized something important. Let me reconsider the problem from scratch.

The objective is C5 = max_k ∫ h(x)(1-h(x+k)) dx.

For a binary h (values 0 or 1), this equals max_k |{x: h(x)=1, h(x+k)=0}| (the measure).

The total measure where h=1 is 1 (since ∫h = 1 and h is binary on [0,2]).
The total measure where h=0 is also 1.

For a given lag k, the overlap = measure of {x ∈ [0,2] : h(x)=1, h(x+k)=0}.

If S = {x: h(x)=1}, then overlap at lag k = |S \ (S-k)| where S-k = {x: x+k ∈ S} = {x: h(x+k)=1}.

We want to minimize max_k |S \ (S-k)|.

Equivalently: maximize min_k |S ∩ (S-k)| since |S \ (S-k)| = |S| - |S ∩ (S-k)| = 1 - |S ∩ (S-k)|.

So we want to maximize min_k |S ∩ (S-k)|.

|S ∩ (S-k)| is the autocorrelation of the characteristic function of S at lag k.

So the problem is: find a set S ⊂ [0,2] of measure 1 that maximizes min_k |S ∩ (S-k)|.

This is a combinatorial/geometric problem! For a discrete version: find a binary string of length n with exactly n/2 ones that maximizes the minimum autocorrelation over all shifts.

For such problems, there are known constructions:
- If n is composite, choose S as a "lattice" or arithmetic progression
- Residue classes: S = {i : i mod p ∈ R} for some prime p and subset R

For example, if n = 400 and we choose p = 20, then S = {i : i mod 20 ∈ {0,1,2,3,4,5,6,7}} (8 out of 20 residues). This gives a periodic pattern.

The autocorrelation of such a periodic pattern at lag k would be:
- If k is a multiple of p: autocorr = (8/20)^2 * n = 0.16 * 400 = 64 (times dx = 0.128)
- Otherwise: it depends on the specific residues

Actually, for a periodic pattern with period p, the autocorrelation at lag k is:
autocorr(k) = ∑_{i=0}^{n-1} s[i]*s[i+k] (approximately, for large n)
For a truly periodic pattern (repeating every p), this is:
= (n/p) * ∑_{j=0}^{p-1} s[j]*s[(j+k) mod p]
= (n/p) * (autocorrelation of one period at lag k mod p)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def lse_value(h, alpha, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def lse_gradient(h, alpha, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = c.max()
    shifted = np.clip(alpha * (c - c_max), -400.0, 0.0)
    exps = np.exp(shifted)
    total = exps.sum()
    if total < 1e-300:
        return np.zeros(n)
    w = exps / total
    
    P = np.zeros(n)
    Q = np.zeros(n)
    
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            P[k:] += wm * (1.0 - h[:n - k])
            Q[:n - k] += wm * h[k:]
        else:
            j = -k
            P[:n - j] += wm * (1.0 - h[j:])
            Q[j:] += wm * h[:n - j]
    
    return dx * (P - Q)


class _SL:
    def __init__(self, alpha, dx, target, n):
        self.alpha = alpha
        self.dx = dx
        self.target = target
        self.n = n
    
    def fun(self, x):
        return lse_value(x, self.alpha, self.dx)
    
    def grad(self, x):
        return lse_gradient(x, self.alpha, self.dx)
    
    def cfun(self, x):
        return np.sum(x) - self.target
    
    def cjac(self, x):
        return np.ones(self.n)


def slsqp_phase(h, target, dx, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [15, 40, 100, 300, 800]
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(100, remaining * 8))
        
        prob = _SL(alpha, dx, target, n)
        try:
            res = minimize(prob.fun, best, jac=prob.grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': prob.cfun, 'jac': prob.cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score

def pair_swap_phase(h, target, dx, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    no_improve = 0
    max_no_improve = 20000
    
    while time.time() - start < time_limit and no_improve < max_no_improve:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if best[i] < best[j]:
            i, j = j, i
        
        diff = best[i] - best[j]
        if diff < 0.02:
            no_improve += 1
            continue
        
        base_d = min(diff * 0.5, 0.15)
        deltas = [base_d, base_d * 0.5, base_d * 0.25, base_d * 2.0]
        
        improved = False
        for d in deltas:
            if d > min(best[i], 1.0 - best[j]) - 1e-10:
                continue
            h_trial = best.copy()
            h_trial[i] -= d
            h_trial[j] += d
            s = compute_c5(h_trial)
            if s < best_score - 1e-14:
                best_score = s
                best = h_trial
                improved = True
                no_improve = 0
                break
        
        if not improved:
            no_improve += 1
    
    return best, best_score
