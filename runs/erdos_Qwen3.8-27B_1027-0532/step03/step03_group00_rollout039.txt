    """SLSQP with analytical gradient."""
    N = len(h0)
    
    def obj(x):
        val, _ = lse_grad(x, alpha)
        return val
    
    def grad(x):
        _, g = lse_grad(x, alpha)
        return g
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * N
    cons = {'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}
