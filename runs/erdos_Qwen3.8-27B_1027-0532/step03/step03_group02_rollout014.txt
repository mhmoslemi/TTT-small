    corr = np.correlate(h, 1.0 - h, mode='full')

    m = len(x)
    n = 2 * m
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def compute_overlaps(h):
    """Compute overlap(s) for all s using np.correlate."""
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_and_grad(x, alpha, dx):
    """Compute smooth objective and gradient w.r.t. x."""
    h = build_h_from_x(x)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    weights = np.exp(alpha * (corr - c_max))
    w_sum = np.sum(weights)
    obj = (c_max + np.log(w_sum) / alpha) * dx
    
    # Gradient computation
    # grad_x[k] = sum_s weights[s] * ((1-h[k+s]) - h[k-s]) / w_sum
    # where k is index into x (0 to m-1)
    # and h[K] = x[K] for K < m, h[K] = 1-x[n-1-K] for K >= m
    
    # Full gradient with respect to h:
    # grad_h[j] = sum_s weights[s] * ((1-h)[j+s] - h[j-s])
    
    # Using the correlation structure:
    # sum_s weights[s_indexed] * (1-h)[j+s] where s_indexed = s + (n-1)
    # This is cross-correlation of weights with (1-h)
    
    b = 1.0 - h
    # cross_corr[j] = sum_{valid s} corr[s] * b[j+s]
    # conv[j] = sum_{valid s} corr[s] * h[j-s]
    
    # Method: direct computation for n=400 is fast enough
    m = len(x)
    grad = np.zeros(m)
    
    # For each k in 0..m-1:
    # grad_x[k] = grad_h[k] - grad_h[n-1-k] (since dh[n-1-k]/dx[k] = -1)
    # grad_h[j] = sum_s w[s_idx] * ((1-h)[j+s] - h[j-s])
    
    # Let's compute grad_h first (length n), then extract grad_x
    grad_h = np.zeros(n)
    s_offset = n - 1  # corr is indexed from 0 to 2n-2, corresponding to s = -(n-1) to n-1

        s_idx = s_val + s_offset
        w_val = weights[s_idx]
        if w_val < 1e-15:
            continue

        j_start = max(0, -s_val)
        j_end = min(n-1, n-1-s_val)
        if j_start <= j_end:
            grad_h[j_start:j_end+1] += w_val * b[j_start+s_val:j_end+s_val+1]
        # -h[j-s] for valid j: j ranges from max(0, s) to min(n-1, n-1+s)
        j_start2 = max(0, s_val)
        j_end2 = min(n-1, n-1+s_val)
        if j_start2 <= j_end2:

    grad_h /= w_sum
    # Convert to grad_x
    for k in range(m):
        grad[k] = grad_h[k] - grad_h[n-1-k]
    grad *= dx
    
    return obj, grad

    return float(np.max(corr)) * dx

def build_h_from_x(x):
    m = len(x)
    n = 2 * m
    h = np.empty(n)
    h[:m] = x
    h[m:] = 1.0 - x[::-1]
    return h

def smooth_max_obj(x, alpha, dx):
    h = build_h_from_x(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (c_max + log_sum / alpha) * dx
