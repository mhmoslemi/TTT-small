            order = np.argsort(x)
            for i in order:
                if diff >= -1e-14:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_lag(h, out_idx, n):
    lag = out_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad

def _smoothed_gradient(h, alpha, m_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_vals = np.exp(alpha * shifted)
    total = exp_vals.sum()
    if total < 1e-300:
        k_star = int(np.argmax(corr))
        return _grad_lag(h, k_star, n)
    weights = exp_vals / total
    top_m = np.argsort(weights)[::-1][:m_top]
    w_m = weights[top_m].copy()
    w_sum = w_m.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w_m /= w_sum
    g = np.zeros(n)
    for idx in range(len(top_m)):
        g += w_m[idx] * _grad_lag(h, top_m[idx], n)
    return g

def _nesterov_bt(h0, n, target_sum, n_iters, alpha0, m_top, alpha_smooth):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    f_prev = best_val
    for it in range(n_iters):
        g = _smoothed_gradient(y, alpha_smooth, m_top)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        step = alpha0 / np.sqrt(float(it + 1))
        step = min(step, 1.0 / (g_norm + 1e-10))
        x_new = _project(y - step * g, target_sum)
        f_new = _c5_of(x_new)
        bt_count = 0
        while f_new > f_prev + 1e-4 * step * g_norm and bt_count < 8:
            step *= 0.5
            x_new = _project(y - step * g, target_sum)
            f_new = _c5_of(x_new)
            bt_count += 1
        x = x_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = x + ((t - 1.0) / t_new) * (x - y)
        y = _project(y, target_sum)
        t = t_new
        f_prev = f_new
        if f_new < best_val:
            best_val = f_new
            best = x.copy()
    return best, best_val

def _slsqp_smooth(h0, n, target_sum, alpha, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.sum(np.exp(alpha * (corr - m)))
        return (m + np.log(s) / alpha) * dx
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _penalized_lbfgs(h0, n, target_sum, penalty, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        s = np.sum(np.exp(100.0 * (corr - m)))
        smooth_max = (m + np.log(s) / 100.0) * dx
        penalty_term = penalty * (x.sum() - target_sum) ** 2
        return smooth_max + penalty_term
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        exp_vals = np.exp(100.0 * (corr - m))
        total = exp_vals.sum()
        if total < 1e-300:
            return np.zeros(n)
        weights = exp_vals / total
        g = np.zeros(n)
        top = np.argsort(weights)[::-1][:8]
        for idx in top:
            g += weights[idx] * _grad_lag(x, idx, n)
        g *= dx
        g += 2.0 * penalty * (x.sum() - target_sum) * np.ones(n)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = fmin_l_bfgs_b(obj, h0, fprime=grad, pgtol=1e-14, maxiter=maxiter,
                            approx_grad=False, bounds=bounds)
        return _fix_sum(res, target_sum)
    except Exception:
        return h0

def _coord_descent_fine(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    current_val = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                h = _fix_sum(h, target_sum)
                return h, _c5_of(h)
            old = h[j]
            if old < 1e-8 or old > 1.0 - 1e-8:
                continue
            best_v = old
            best_val = current_val
            for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
                v_minus = max(0.0, old - frac)
                v_plus = min(1.0, old + frac)
                if abs(v_minus - old) > 1e-8:
                    h[j] = v_minus
                    val = _c5_of(h)
                    if val < best_val - 1e-13:
                        best_val = val
                        best_v = v_minus
                if abs(v_plus - old) > 1e-8:
                    h[j] = v_plus
                    val = _c5_of(h)
                    if val < best_val - 1e-13:
                        best_val = val
                        best_v = v_plus
            h[j] = best_v
            if best_v != old:
                current_val = best_val
                improved = True
        if not improved:
            break
    h = _fix_sum(h, target_sum)
    return h, _c5_of(h)

def _de_explore(h0, n, target_sum, pop_size, max_gen, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.25:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif r < 0.45:
            pop[i] = _fix_sum(1.0 - h0 + 0.05 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.65:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif r < 0.80:
            pop[i] = _fix_sum(h0 + 0.15 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.90:
            block = max(5, n // 10)
            p = rng.permutation(n)
            pop[i] = _fix_sum(h0[p], target_sum)
        else:
            s1 = rng.randint(0, n - 2 * 10)
            s2 = rng.randint(0, n - 2 * 10)
            tmp = h0.copy()
            tmp[s1:s1+10] = h0[s2:s2+10]
            tmp[s2:s2+10] = h0[s1:s1+10]
            pop[i] = _fix_sum(tmp, target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = 0.6 + 0.3 * (1.0 - gen / float(max_gen))
        CR = 0.6
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross = (rng.random(n) < CR)
            cross[j_rand] = True
            trial[cross] = mutant[cross]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size * 2:
            worst = np.argsort(pop_vals)[pop_size // 2:]
            for idx in worst:
                t = rng.uniform(0.2, 0.8)
                pop[idx] = _fix_sum(t * best_h + (1.0 - t) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

def _refine_pipeline(h0, n, target_sum, deadline):
    h_best = h0.copy()
    c5_best = _c5_of(h0)
    if time.time() > deadline - 80:
        return h_best, c5_best
    h1, c1 = _nesterov_bt(h_best, n, target_sum, 4000, 0.4, 8, 100.0)
    if c1 < c5_best:
        c5_best, h_best = c1, h1
    if time.time() > deadline - 60:
        return h_best, c5_best
    h2 = _slsqp_smooth(h_best, n, target_sum, 80, 500)
    c2 = _c5_of(h2)
    if c2 < c5_best:
        c5_best, h_best = c2, h2
    if time.time() > deadline - 40:
        return h_best, c5_best
    h3, c3 = _nesterov_bt(h_best, n, target_sum, 3000, 0.25, 5, 200.0)
    if c3 < c5_best:
        c5_best, h_best = c3, h3
    if time.time() > deadline - 25:
        return h_best, c5_best
    h4 = _penalized_lbfgs(h_best, n, target_sum, 1e6, 500)
    c4 = _c5_of(h4)
    if c4 < c5_best:
        c5_best, h_best = c4, h4
    if time.time() > deadline - 15:
        return h_best, c5_best
    h5 = _slsqp_smooth(h_best, n, target_sum, 200, 1000)
    c5 = _c5_of(h5)
    if c5 < c5_best:
        c5_best, h_best = c5, h5
    if time.time() > deadline - 5:
        return h_best, c5_best
    h6, c6 = _coord_descent_fine(h_best, n, target_sum, 15, deadline - 2)
    if c6 < c5_best:
        c5_best, h_best = c6, h6
    return h_best, c5_best

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    n_candidates = [n_init]
    for d in [10, -10, 5, -5, 14, -14, 20]:
        nt = n_init + d
        if 100 <= nt <= 220 and nt % 2 == 0 and nt not in n_candidates:
            n_candidates.append(nt)

    for n in n_candidates:
        if time.time() > deadline - 80:
            break
        target_sum = float(n) / 2.0
        if n == n_init:
            h_start = h_init.copy()
        else:
            h_start = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target_sum)
            h_alt = np.interp(np.linspace(0, best_n - 1, n), np.arange(best_n), best_h)
            h_alt = _fix_sum(np.clip(h_alt, 0.0, 1.0), target_sum)
            if _c5_of(h_alt) < _c5_of(h_start):
                h_start = h_alt

        if time.time() < deadline - 250:
            h_de, c5_de = _de_explore(h_start, n, target_sum, 35, 100, deadline - 250, rng)
            if c5_de < best_c5:
                best_c5, best_h, best_n = c5_de, h_de.copy(), n
            if c5_de < _c5_of(h_start):
                h_start = h_de

        h_ref, c5_ref = _refine_pipeline(h_start, n, target_sum, deadline - 30)
        if c5_ref < best_c5:
            best_c5, best_h, best_n = c5_ref, h_ref.copy(), n

    if time.time() < deadline - 40:
        h_final, c5_final = _refine_pipeline(best_h, best_n, best_n / 2.0, deadline - 5)
        if c5_final < best_c5:
            best_c5, best_h = c5_final, h_final

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
