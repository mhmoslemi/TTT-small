       else:
           deficit = int(target_sum - current)
           zeros = np.where(h_r == 0)[0]
           if deficit <= len(zeros):
               order = np.argsort(-h[zeros])[:deficit]
               h_r[zeros[order]] = 1
       # Verify
       final = h_r.sum()
       if final != target_sum:
           if final > target_sum:
               ones = np.where(h_r == 1)[0]
               h_r[ones[:int(final - target_sum)]] = 0
           else:
               zeros = np.where(h_r == 0)[0]
               h_r[zeros[:int(target_sum - final)]] = 1
       return h_r.astype(float)

   def sa_fast(h_start, target_sum, max_time, t_init, t_final, rng, start_time):
       n = len(h_start)
       h = h_start.copy()
       best_h = h.copy()
       best_c5 = compute_c5(h)
       cur_c5 = best_c5
       
       ones = np.where(h == 1)[0]
       zeros = np.where(h == 0)[0]
       
       total_iters = 0
       no_improve = 0
       max_no_improve = max(5000, int(max_time * 2000))
       
       while (time.time() - start_time) < max_time:
           total_iters += 1
           
           if no_improve > max_no_improve:
               break
           
           ratio = min(total_iters / max(total_iters, 1), 1.0)
           # Use a time-based temperature schedule
           elapsed = time.time() - start_time
           ratio_t = min(elapsed / max_time, 1.0)
           t = t_init * (t_final / t_init) ** ratio_t
           
           # Choose move type
           mt = rng.randint(5)
           h_new = h.copy()
           moved = False
           
           if mt == 0 and len(ones) > 0 and len(zeros) > 0:
               # Single swap
               i1 = int(ones[rng.randint(len(ones))])
               i0 = int(zeros[rng.randint(len(zeros))])
               h_new[i1] = 0
               h_new[i0] = 1
               moved = True
           elif mt == 1 and len(ones) >= 2 and len(zeros) >= 2:
               # Double swap
               perm1 = rng.permutation(len(ones))[:2]
               perm2 = rng.permutation(len(zeros))[:2]
               h_new[ones[perm1[0]]] = 0
               h_new[ones[perm1[1]]] = 0
               h_new[zeros[perm2[0]]] = 1
               h_new[zeros[perm2[1]]] = 1
               moved = True
           elif mt == 2 and len(ones) > 0 and n > 10:
               # Slide: move a 1 by a small amount
               i1 = int(ones[rng.randint(len(ones))])
               dist = int(rng.randint(1, 6)) * (1 if rng.random() < 0.5 else -1)
               i0 = i1 + dist
               if 0 <= i0 < n and h[i0] == 0:
                   h_new[i1] = 0
                   h_new[i0] = 1
                   moved = True
               else:
                   # Fallback to random swap
                   i0 = int(zeros[rng.randint(len(zeros))])
                   h_new[i1] = 0
                   h_new[i0] = 1
                   moved = True
           elif mt == 3 and len(ones) > 0 and len(zeros) > 0:
               # Long jump swap
               i1 = int(ones[rng.randint(len(ones))])
               i0 = int(zeros[rng.randint(len(zeros))])
               # Bias towards nearby zeros
               dist = np.abs(zeros - i1).astype(float)
               dist += rng.random(len(zeros)) * 20.0
               i0 = int(zeros[np.argmin(dist)])
               h_new[i1] = 0
               h_new[i0] = 1
               moved = True
           elif mt == 4 and len(ones) >= 3 and len(zeros) >= 3:
               # Triple swap
               perm1 = rng.permutation(len(ones))[:3]
               perm2 = rng.permutation(len(zeros))[:3]
               for p in perm1:
                   h_new[ones[p]] = 0
               for p in perm2:
                   h_new[zeros[p]] = 1
               moved = True
           
           if not moved:
               continue
           
           c5_new = compute_c5(h_new)
           delta = c5_new - cur_c5
           
           if delta < 0 or (t > 1e-15 and rng.random() < np.exp(-delta / t)):
               h = h_new
               cur_c5 = c5_new
               ones = np.where(h == 1)[0]
               zeros = np.where(h == 0)[0]
               if c5_new < best_c5:
                   best_c5 = c5_new
                   best_h = h.copy()
                   no_improve = 0
               else:
                   no_improve += 1
           else:
               no_improve += 1
       
       return best_h, best_c5

   def softmax_obj(h, alpha):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full') * dx
       c_max = np.max(corr)
       shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
       return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

   def softmax_grad(h, alpha):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full') * dx
       c_max = np.max(corr)
       shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
       exp_s = np.exp(shifted)
       w = exp_s / np.sum(exp_s)
       
       lags = np.arange(len(w)) - (n - 1)
       grad = np.zeros(n)
       for j in range(len(w)):
           if w[j] < 1e-12:
               continue
           k = lags[j]
           d = np.zeros(n)
           if k >= 0:
               d[:n-k] += (1.0 - h[k:n])
               d[k:] -= h[:n-k]
           else:
               ak = -k
               d[ak:] += (1.0 - h[:n-ak])
               d[:n-ak] -= h[ak:n]
           grad += w[j] * dx * d
       return grad

   def lmbfgs_refine(h_start, target_sum, alphas, time_limit, start_time):
       n = len(h_start)
       h_cur = h_start.copy()
       best_score = compute_c5(h_cur)
       bounds = [(0.0, 1.0)] * n
       
       for alpha in alphas:
           if time.time() - start_time > time_limit:
               break
           try:
               def obj(x):
                   hx = project_feasible(x, target_sum)
                   return softmax_obj(hx, alpha)
               
               def grad_f(x):
                   hx = project_feasible(x, target_sum)
                   return softmax_grad(hx, alpha)
               
               res = minimize(obj, h_cur, jac=grad_f, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10})
               h_new = project_feasible(res.x, target_sum)
               sc = compute_c5(h_new)
               if sc < best_score:
                   best_score = sc
                   h_cur = h_new
           except Exception:
               pass
       
       return h_cur, best_score
