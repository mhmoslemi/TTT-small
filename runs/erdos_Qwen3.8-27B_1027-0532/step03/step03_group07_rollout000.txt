import time
import numpy as np
from scipy.optimize import minimize

_global_n = 0
_global_target_sum = 0.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

def _nesterov_topk(h0, n, target_sum, n_iters, alpha0, top_k, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    avg = x.copy()
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        top_m = np.argsort(corr)[::-1][:top_k]
        weights = np.exp(100.0 * (corr[top_m] - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(top_k)
            weights[0] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for idx in range(top_k):
            g = g + weights[idx] * _grad_corr_term(y, top_m[idx], dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        avg = avg * (it / (it + 1)) + x * (1.0 / (it + 1))
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
        cv_avg = _c5_of(avg)
        if cv_avg < best_val:
            best_val = cv_avg
            best = avg.copy()
    return best, best_val

def _cat_swarm(h0, n, target_sum, deadline, rng, pop_size=12, max_iters=80):
    cats = [h0.copy()]
    scores = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        cats.append(h_rand)
        scores.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = scores[0]
    exploration_rate = 0.7
    Si = 4
    for it in range(max_iters):
        if time.time() > deadline:
            break
        exploration_rate = max(0.2, 0.7 - 0.005 * it)
        new_cats = []
        new_scores = []
        for i in range(pop_size):
            if rng.random() < exploration_rate:
                h_new = cats[i].copy()
                n_pert = rng.randint(n // 10, n // 3 + 1)
                idx = rng.choice(n, size=n_pert, replace=False)
                scale = 0.15 + 0.2 * (it / max_iters)
                for j in idx:

                h_new = _project(h_new, target_sum)

                n_muts = rng.randint(2, Si)

                best_idx = int(np.argmin(scores))

                        h_new[j] += 0.1 * (cats[best_idx][j] - h_new[j])
                    else:
                        h_new[j] += 0.08 * (rng.rand() * 2 - 1)
                h_new = _project(h_new, target_sum)
            s_new = _c5_of(h_new)
            if s_new < scores[i]:
                new_cats.append(h_new)
                new_scores.append(s_new)
                if s_new < best_c5:
                    best_c5 = s_new
                    best_h = h_new.copy()
            else:
                new_cats.append(cats[i].copy())
                new_scores.append(scores[i])
        cats = new_cats
        scores = new_scores
        if best_c5 <= 0.381:

    return h

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_candidates = []
    for dn in range(0, 41, 4):
        n_try = n_init + dn
        if n_try % 2 == 0 and 50 <= n_try <= 220:
            n_candidates.append(n_try)
    for dn in range(-4, -40, -4):
        n_try = n_init + dn
        if n_try % 2 == 0 and 50 <= n_try <= 220:
            n_candidates.append(n_try)
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)
    n_candidates = sorted(n_unique)

    time_per_n = (deadline - t_start - 60) / len(n_candidates)
    time_per_n = max(30.0, min(120.0, time_per_n))

    n_processed = 0
    for n in n_candidates:
        if time.time() > deadline - 50:
            break
        if n_processed >= 8:
            break
        n_processed += 1
        target_sum = n / 2.0
        n_deadline = min(deadline - 30, t_start + time_per_n * (n_processed + 1))

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n),
                                  np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        h_curr = h_start.copy()
        c5_curr = c5_start

        # CSO for global exploration
        h_cso, c5_cso = _cat_swarm(h_curr, n, target_sum,
                                    n_deadline - 40, rng,
                                    pop_size=10, max_iters=60)
        if c5_cso < c5_curr:
            c5_curr = c5_cso
            h_curr = h_cso.copy()
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

        # Multi Nesterov with different top-k
        h_nest, c5_nest = _multi_nesterov(h_curr, n, target_sum, n_deadline - 30)
        if c5_nest < c5_curr:
            c5_curr = c5_nest
            h_curr = h_nest.copy()
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

        # SLSQP with increasing alpha
        for alpha in [100, 400, 1200, 3000]:
            if time.time() > n_deadline - 20:
                break
            h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=300)
            c5_sl = _c5_of(h_sl)
            if c5_sl < c5_curr:
                c5_curr = c5_sl
                h_curr = h_sl.copy()
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Enhanced coordinate descent
        if time.time() < n_deadline - 15:
            h_cd = _coord_descent_enhanced(h_curr, n, target_sum,
                                            max_rounds=15, deadline=n_deadline - 10)
            c5_cd = _c5_of(h_cd)
            if c5_cd < c5_curr:
                c5_curr = c5_cd
                h_curr = h_cd.copy()
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Binary coordinate descent
        if time.time() < n_deadline - 10:
            h_bd = _binary_cd(h_curr, n, target_sum, max_rounds=20,
                               deadline=n_deadline - 8)
            c5_bd = _c5_of(h_bd)
            if c5_bd < c5_curr:
                c5_curr = c5_bd
                h_curr = h_bd.copy()
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Pounce refinement
        if time.time() < n_deadline - 10:
            h_pounce, c5_pounce = _pounce_refine(h_curr, n, target_sum,
                                                   n_deadline - 5, rng,
                                                   n_pounce=4)
            if c5_pounce < c5_curr:
                c5_curr = c5_pounce
                h_curr = h_pounce.copy()
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

        # Penalty method targeting below current best
        if time.time() < n_deadline - 10:
            target_pen = best_c5 - 0.0005
            h_pen = _penalty_slsqp(h_curr, n, target_pen, pen_weight=500,
                                    maxiter=200)
            c5_pen = _c5_of(h_pen)
            if c5_pen < c5_curr:
                c5_curr = c5_pen
                h_curr = h_pen.copy()
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

    # Final aggressive refinement on best_n
    n_best = best_n
    target = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 20:
        # Very high alpha SLSQP
        for alpha in [5000, 10000, 20000]:
            if time.time() > deadline - 15:
                break
            h_f = _slsqp_opt(h_final, n_best, target, alpha, maxiter=500)
            c5_f = _c5_of(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
            h_final = h_f

    if time.time() < deadline - 15:
        h_cd_f = _coord_descent_enhanced(best_h, n_best, target,
                                          max_rounds=20, deadline=deadline - 10)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 12:
        h_bd_f = _binary_cd(best_h, n_best, target, max_rounds=25,
                             deadline=deadline - 10)
        c5_bd_f = _c5_of(h_bd_f)
        if c5_bd_f < best_c5:
            best_c5 = c5_bd_f
            best_h = h_bd_f.copy()

    if time.time() < deadline - 10:
        h_pounce_f, c5_pounce_f = _pounce_refine(best_h, n_best, target,
                                                   deadline - 5, rng, n_pounce=6)
        if c5_pounce_f < best_c5:
            best_c5 = c5_pounce_f
            best_h = h_pounce_f.copy()

    if time.time() < deadline - 8:
        h_pen_f = _penalty_slsqp(best_h, n_best, best_c5 - 0.0003,
                                  pen_weight=1000, maxiter=300)
        c5_pen_f = _c5_of(h_pen_f)
        if c5_pen_f < best_c5:
            best_c5 = c5_pen_f
            best_h = h_pen_f.copy()

    # Final SLSQP + CD cycle
    if time.time() < deadline - 5:
        h_pol = _slsqp_opt(best_h, n_best, target, 15000, maxiter=400)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() < deadline - 3:
        h_cd_final = _coord_descent_enhanced(best_h, n_best, target,
                                              max_rounds=10, deadline=deadline - 1)
        c5_cd_final = _c5_of(h_cd_final)
        if c5_cd_final < best_c5:
            best_c5 = c5_cd_final
            best_h = h_cd_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
