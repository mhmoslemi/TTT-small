def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

                corr += delta * g

        h = project_feasible(h, target_sum)
        corr = compute_corr(h)
        score = float(np.max(corr)) * dx
        if score < best_score:
            best_score = score
    
    return h, best_score

    return project_feasible(h_out, target_dst)


def coordinate_descent(h_start, target_sum, n_sweeps, rng, time_limit=None,
                       start_time=None, golden_iters=22, kick_interval=80):
    n = len(h_start)
    dx = 2.0 / n
    h = project_feasible(h_start.copy(), target_sum)
    corr = compute_corr(h)
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    no_improve = 0

    for sweep in range(n_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        if no_improve > kick_interval and rng.random() < 0.3:
            n_kick = max(3, n // 20)
            idx = rng.choice(n, n_kick, replace=False)
            h[idx] += rng.randn(n_kick) * 0.05
            h = project_feasible(h, target_sum)
            corr = compute_corr(h)
            no_improve = 0

        order = rng.permutation(n)
        sweep_best = float(np.max(corr))

        for j in order:
            g = np.zeros(2 * n - 1)
            g[n - 1:2 * n - 1 - j] += 1.0 - h[j:]
            g[n - 1 - j:n] += -h[:j + 1][::-1]

            old_val = h[j]
            a_lo = 0.0
            a_hi = 1.0
            for _ in range(golden_iters):
                m1 = a_hi - gr * (a_hi - a_lo)
                m2 = a_lo + gr * (a_hi - a_lo)
                f1 = np.max(corr + (m1 - old_val) * g)
                f2 = np.max(corr + (m2 - old_val) * g)
                if f1 < f2:
                    a_hi = m2
                else:
                    a_lo = m1
            t = (a_lo + a_hi) * 0.5
            delta = t - old_val
            if abs(delta) > 1e-13:
                h[j] = t
                corr = corr + delta * g

        h = project_feasible(h, target_sum)
        corr = compute_corr(h)
        score = float(np.max(corr)) * dx

        if score < best_score - 1e-10:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_h, best_score


def continuous_sa(h_start, target_sum, n_iters, t_init, t_final, rng,
                  time_limit=None, start_time=None):
    n = len(h_start)
    h = project_feasible(h_start.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5

    for i in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break

        progress = i / max(n_iters - 1, 1)
        t = t_init * (t_final / t_init) ** progress

        move_type = rng.randint(4)
        h_new = h.copy()

        if move_type == 0:
            n_pert = rng.randint(1, max(2, n // 15))
            idx = rng.choice(n, n_pert, replace=False)
            h_new[idx] += rng.randn(n_pert) * 0.08
        elif move_type == 1:
            n_pert = rng.randint(5, max(6, n // 8))
            start = rng.randint(0, max(1, n - n_pert))
            h_new[start:start + n_pert] = 1.0 - h_new[start:start + n_pert]
        elif move_type == 2:
            i1, i2 = rng.choice(n, 2, replace=False)
            tmp = h_new[i1]
            h_new[i1] = h_new[i2]
            h_new[i2] = tmp
        else:
            n_pert = rng.randint(2, max(3, n // 25))
            idx = rng.choice(n, n_pert, replace=False)
            h_new[idx] = rng.uniform(0.0, 1.0, n_pert)

        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        delta = c5_new - current_c5

        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()

    return best_h, best_c5


def _smooth_obj(x, alpha):
    corr = compute_corr(x)
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def _cons_fun(x):
    return np.sum(x) - _cons_target


def _cons_jac(x):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = project_feasible(h_start.copy(), target)
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(_smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init.copy(), target_init)
    best_score = compute_c5(best_h)

    def _elapsed():
        return time.time() - start_time

    def _remaining():
        return budget_s - 10.0 - _elapsed()

    # Phase 1: Coordinate descent on initial solution
    phase1_time = min(400.0, _remaining() * 0.35)
    n_sweeps1 = int(phase1_time * 25)
    h_cd, score_cd = coordinate_descent(best_h, target_init, n_sweeps1, rng,
                                        time_limit=phase1_time, start_time=start_time)
    if score_cd < best_score:
        best_score = score_cd
        best_h = h_cd.copy()

    # Phase 2: SA + coordinate descent loop
    sa_rounds = 4
    for sr in range(sa_rounds):
        if _remaining() < 80:
            break
        sa_time = min(60.0, _remaining() * 0.2)
        n_sa_iters = int(sa_time * 80)
        t_init_sa = 0.008 * (1.0 + 0.3 * sr)
        h_sa, score_sa = continuous_sa(best_h, target_init, n_sa_iters,
                                       t_init_sa, 0.0001, rng,
                                       time_limit=sa_time, start_time=start_time)

        h_work = h_sa if score_sa < best_score else best_h
        cd_time = min(80.0, _remaining() * 0.3)
        n_sweeps2 = int(cd_time * 25)
        h_cd2, score_cd2 = coordinate_descent(h_work, target_init, n_sweeps2, rng,
                                              time_limit=cd_time, start_time=start_time)
        if score_cd2 < best_score:
            best_score = score_cd2
            best_h = h_cd2.copy()

    # Phase 3: Multi-resolution coordinate descent
    n_candidates = [400, 500, 650, 750, 900]
    n_candidates = [n for n in n_candidates if n != n_init and n >= 150]

    for n_try in n_candidates:
        if _remaining() < 60:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        cd_time = min(80.0, _remaining() * 0.3)
        n_sweeps3 = int(cd_time * 25)
        h_cd3, score_cd3 = coordinate_descent(h_interp, target_try, n_sweeps3, rng,
                                              time_limit=cd_time, start_time=start_time)
        if score_cd3 < best_score:
            best_score = score_cd3
            best_h = h_cd3.copy()
            n_init = n_try
            target_init = target_try

    # Phase 4: SLSQP refinement on the best
    if _remaining() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas = [50, 200, 800, 2000]
        slsqp_time = min(80.0, _remaining() * 0.4)
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 500,
                                 time_limit=slsqp_time, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Final coordinate descent polish
    n_final = len(best_h)
    target_final = n_final / 2.0
    if _remaining() > 10:
        cd_time_final = _remaining() - 5.0
        n_sweeps_final = int(cd_time_final * 30)
        h_cd_f, score_cd_f = coordinate_descent(best_h, target_final, n_sweeps_final, rng,
                                                time_limit=cd_time_final, start_time=start_time)
        if score_cd_f < best_score:
            best_score = score_cd_f
            best_h = h_cd_f.copy()

    # Phase 6: Tiny SLSQP polish if time remains
    if _remaining() > 8:
        n_final = len(best_h)
        target_final = n_final / 2.0
        h_ref, sc = slsqp_refine(best_h, n_final, target_final, [2000, 5000], 200,
                                 time_limit=min(10.0, _remaining() - 3), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility check
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = compute_c5(best_h)

    return best_h, final_score, n_out
