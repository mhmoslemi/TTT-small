    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    m = np.max(corr)

    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp  # weights, length 2n-1
    F = m + np.log(sum_exp) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0  # sum of w_k * (1-h[i-k]) for valid i-k
        s2 = 0.0  # sum of w_k * h[i+k] for valid i+k

Now let me think about the augmented Lagrangian approach.

The augmented Lagrangian for the constraint ∑h_i = n/2 is:
L_ρ(h, μ) = F(h) + μ(∑h_i - n/2) + (ρ/2)(∑h_i - n/2)²

∂L/∂h_i = ∇F(h)_i + μ + ρ(∑h_i - n/2)

So the gradient for L-BFGS-B is:
grad_aug[i] = grad_F[i] + μ + ρ * (sum(h) - n/2)

And we update μ after each L-BFGS-B call:
μ_new = μ + ρ * (sum(h_new) - n/2)

We start with μ = 0 and increase ρ over time (e.g., ρ = 1, 10, 100, 1000).

This should give much better constraint satisfaction than the parent's approach.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_objective(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if alpha == 0:
        return m, np.zeros_like(corr)
    shifted = alpha * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -700, 700)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    return F, w

def compute_grad_from_weights(h, w):
    n = len(h)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    nm1 = n - 1
    for i in range(n):
        k_lo1 = max(-nm1, i - n + 1)
        k_hi1 = min(nm1, i)
        if k_lo1 <= k_hi1:
            ks = np.arange(k_lo1, k_hi1 + 1)
            s1 = np.dot(w[ks + nm1], one_minus_h[i - ks])
        else:
            s1 = 0.0
        k_lo2 = max(-nm1, -i)
        k_hi2 = min(nm1, n - 1 - i)
        if k_lo2 <= k_hi2:
            ks2 = np.arange(k_lo2, k_hi2 + 1)
            s2 = np.dot(w[ks2 + nm1], h[i + ks2])
        else:
            s2 = 0.0
        grad[i] = s1 - s2
    return grad

def objective_and_grad(h, alpha, mu, rho, target_sum):
    F, w = smooth_objective(h, alpha)
    grad_F = compute_grad_from_weights(h, w)
    # Augmented Lagrangian
    constraint_violation = np.sum(h) - target_sum
    grad = grad_F + mu + rho * constraint_violation
    F_aug = F + mu * constraint_violation + 0.5 * rho * constraint_violation**2
    return F_aug, grad

def optimize_stage(h_start, alpha, maxiter, mu_init=0.0, rho=1.0, n_rho_updates=5):
    n = len(h_start)
    target = n / 2.0
    h_cur = project_feasible(h_start.copy(), target)
    mu = mu_init
    bounds = [(0.0, 1.0)] * n
    
    for rho_step in range(n_rho_updates):
        def obj_func(x):
            return objective_and_grad(x, alpha, mu, rho, target)[0]
        
        def jac_func(x):
            return objective_and_grad(x, alpha, mu, rho, target)[1]
        
        try:
            res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            h_new = res.x
            # Update Lagrange multiplier
            constraint_violation = np.sum(h_new) - target
            mu = mu + rho * constraint_violation
            h_cur = h_new
            rho = rho * 2.0
        except Exception:
            break
    
    h_cur = project_feasible(h_cur, target)
    return h_cur

def downsample(h, n_dst):
    n_src = len(h)
    if n_src == n_dst:
        return h.copy()
    x_new = np.linspace(0, 1, n_dst)
    x_old = np.linspace(0, 1, n_src)
    return np.interp(x_new, x_old, h)

def interpolate_to(h_src, n_dst, target_dst):
    h_dst = downsample(h_src, n_dst)
    return project_feasible(h_dst, target_dst)

def bang_bang_local_search(h_continuous, n_all_swaps=False, max_sweeps=20):
    """Local search on bang-bang solutions."""
    n = len(h_continuous)
    target = int(n / 2)
    # Threshold at 0.5, then adjust
    h_bb = (h_continuous > 0.5).astype(float)
    # Ensure exactly target ones
    n_ones = int(np.sum(h_bb))
    if n_ones > target:
        # Turn off some ones (those closest to 0.5)
        ones_idx = np.where(h_bb == 1)[0]
        order = np.argsort(h_continuous[ones_idx])  # ones with lowest h_continuous first
        for idx in order[:n_ones - target]:
            h_bb[ones_idx[idx]] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bb == 0)[0]
        order = np.argsort(-h_continuous[zeros_idx])  # zeros with highest h_continuous first
        for idx in order[:target - n_ones]:
            h_bb[zeros_idx[idx]] = 1
    
    best_h = h_bb.copy()
    best_c5 = compute_c5(best_h)
    
    for sweep in range(max_sweeps):
        improved = False
        ones_idx = np.where(best_h == 1)[0]
        zeros_idx = np.where(best_h == 0)[0]
        
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        best_swap = None
        best_new_c5 = best_c5
        
        # Try all single swaps (or a random subset for speed)
        n_try = len(ones_idx) * len(zeros_idx)
        if n_try > 5000:
            # Random subset
            rng = np.random.RandomState(sweep)
            indices = rng.choice(n_try, size=5000, replace=False)
        else:
            indices = range(n_try)
        
        idx = 0
        for i_idx in ones_idx:
            for j_idx in zeros_idx:
                if idx not in indices if n_try > 5000 else True:
                    idx += 1
                    continue
                # Try swap
                trial = best_h.copy()
                trial[i_idx] = 0
                trial[j_idx] = 1
                c5_trial = compute_c5(trial)
                if c5_trial < best_new_c5 - 1e-12:
                    best_new_c5 = c5_trial
                    best_swap = (i_idx, j_idx)
                idx += 1
        
        if best_swap is not None:
            best_h[best_swap[0]] = 0
            best_h[best_swap[1]] = 1
            best_c5 = best_new_c5
            improved = True
        
        if not improved:
            break
    
    return best_h, best_c5
