```python
import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, lag_idx, n):
    """Compute gradient of corr[lag_idx] w.r.t. h.
    lag_idx is index in the full correlation output (0 to 2n-2).
    lag = lag_idx - (n-1)"""
    lag = lag_idx - (n - 1)
    grad = np.zeros(n)
    # S_lag = {i : 0 <= i <= n-1 and 0 <= i-lag <= n-1}
    # i-lag >= 0 => i >= lag
    # i-lag <= n-1 => i <= n-1+lag
    # i >= 0 and i <= n-1
    i_lo = max(0, lag)
    i_hi = min(n-1, n-1+lag)
    if i_lo > i_hi:
        return grad
    
    # First term: (1 - h[m-lag]) for m in S_lag
    m_vals = np.arange(i_lo, i_hi + 1)
    h_shifted = h[m_vals - lag]
    grad[m_vals] += (1.0 - h_shifted)
    
    # Second term: -h[m+lag] for m+lag in S_lag
    # m+lag in S_lag => 0 <= m+lag <= n-1 and 0 <= m <= n-1
    # m >= -lag and m <= n-1-lag, and 0 <= m <= n-1
    m_lo2 = max(0, -lag)
    m_hi2 = min(n-1, n-1-lag)
    if m_lo2 <= m_hi2:
        m_vals2 = np.arange(m_lo2, m_hi2 + 1)
        grad[m_vals2] -= h[m_vals2 + lag]
    
    return grad

def active_set_gradient(h, n, top_fraction=0.1):
    """Compute gradient as average over top correlation lags."""
    corr = compute_correlations(h)
    n_corr = len(corr)
    n_active = max(1, int(n_fraction * n_corr))
    
    # Find the lags with highest correlation
    # We want the max, so find top values
    threshold = np.partition(corr, n_corr - n_active)[n_corr - n_active]
    active = np.where(corr >= threshold)[0]
    
    grad = np.zeros(n)
    for lag_idx in active:
        grad += grad_for_lag(h, lag_idx, n)
    grad /= len(active)
    
    return grad, corr, active

def line_search(h, grad, target_sum, max_step=0.1):
    """Backtracking line search for the projected gradient step."""
    current_f = compute_c5(h)
    step = max_step
    for _ in range(30):
        h_trial = project_feasible(h - step * grad, target_sum)
        f_trial = compute_c5(h_trial)
        if f_trial < current_f:
            return h_trial, f_trial
        step *= 0.7
    return h, current_f
```

```python
def grad_min_autocorr(h, active_lags, n):
    """Gradient of min autocorrelation (average over active lags).
    active_lags is an array of lag values (not indices), can be negative."""
    n = len(h)
    grad = np.zeros(n)
    for k in active_lags:
        # ∂R_k/∂h[m] = h[m-k]*1_{valid} + h[m+k]*1_{valid}
        m = np.arange(n)
        idx1 = (m - k >= 0) & (m - k < n)
        idx2 = (m + k >= 0) & (m + k < n)
        grad[idx1] += h[m[idx1] - k]
        grad[idx2] += h[m[idx2] + k]
    grad /= len(active_lags)
    return grad
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_corr_lag(h, k, n):
    """Gradient of corr at lag k (actual lag value, can be negative) w.r.t. h.
    corr_k = sum_{i in S_k} h[i]*(1-h[i-k])
    dcorr_k/dh[m] = (1-h[m-k])*1_{valid(m-k)} - h[m+k]*1_{valid(m+k)}
    """
    m = np.arange(n)
    # First term: (1-h[m-k]) where m-k is valid
    mk = m - k
    valid1 = (mk >= 0) & (mk < n)
    # Second term: -h[m+k] where m+k is valid
    mkp = m + k
    valid2 = (mkp >= 0) & (mkp < n)
    
    grad = np.zeros(n)
    grad[valid1] = (1.0 - h[mk[valid1]])
    grad[valid2] -= h[mkp[valid2]]
    return grad

def active_set_subgradient(h, n, active_fraction=0.1, min_active=5):
    """Compute subgradient of max corr, averaging over active lags."""
    corr = compute_corr_array(h)
    n_corr = len(corr)
    n_active = max(min_active, int(active_fraction * n_corr))
    
    # Get the top n_active lags by correlation value
    # Use partition for efficiency
    if n_active < n_corr:

        active_indices = np.arange(n_corr)
    
    # Convert to lag values
    active_lags = active_indices - (n - 1)
    
    # Compute average gradient
    grad = np.zeros(n)
    for k in active_lags:
        grad += grad_corr_lag(h, int(k), n)
    grad /= len(active_lags)
    
    return grad, corr, float(np.max(corr))

def projected_steepest_descent(h, n, target_sum, max_iters=5000, 
                                  active_fraction=0.1, initial_step=0.05):
    """Active-set projected gradient descent."""
    best_h = h.copy()
    best_f = compute_c5(h)
    
    step = initial_step
    stuck_count = 0
    
    for it in range(max_iters):
        grad, corr, current_max = active_set_subgradient(h, n, active_fraction)
        
        # Backtracking line search
        current_f = compute_c5(h)
        improved = False
        for _ in range(30):
            h_trial = project_feasible(h - step * grad, target_sum)
            f_trial = compute_c5(h_trial)
            if f_trial < current_f * (1 - 1e-12):
                h = h_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = h_trial.copy()
                stuck_count = 0
                improved = True
                # Slightly increase step if we improved a lot
                if f_trial < current_f * 0.99:
                    step = min(0.1, step * 1.1)
                break
            step *= 0.6
        if not improved:
            if step < 1e-10:
                break
            stuck_count += 1
            if stuck_count > 50:
                # Escape: random perturbation
                h = project_feasible(h + 0.005 * np.random.randn(n), target_sum)
                step = initial_step
                stuck_count = 0
    
    return best_h, best_f
```

```python
        h_ref, sc = asgd_plain(best_h, n_final, target_final, 10000, 0.02, 0.002, tleft, rng)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
```
