```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_array(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')
    
def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def c_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def c_jac(h):
    return np.ones(len(h))

def slsqp(h_init, eps, maxiter, ftol=1e-16):
    n = len(h_init)
    h0 = project_onto_constraint(h_init.astype(float), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c_fun, 'jac': c_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(eps,),
                       jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(res.x, n / 2.0)
        return h_opt, compute_c5(h_opt)
    except:
        return h0, compute_c5(h0)

def round_to_binary(h):
    n = len(h)
    # Sort by distance from 0.5, assign 1 to the top n/2
    order = np.argsort(-np.abs(h - 0.5))
    result = np.zeros(n)
    result[order[:n // 2]] = 1.0
    return result

def binary_greedy(h, max_rounds=100, time_budget=None, start_time=None):
    """Greedy binary local search: at each step, find the best improving flip."""
    n = len(h)
    h = h.astype(float).copy()
    current = compute_c5(h)
    
    for _ in range(max_rounds):
        if time_budget is not None and time.time() - start_time > time_budget:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 1e-10
        best_pair = None
        
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                h[oi] = 1.0
                h[zi] = 0.0
                delta = sc - current
                if delta < -best_delta:
                    best_delta = -delta
                    best_pair = (oi, zi)
        
        if best_pair is None:
            break
        h[best_pair[0]] = 0.0
        h[best_pair[1]] = 1.0
        current -= best_delta
    
    return h, current
```

The cross-correlation at lag k is:
corr[k] = Σᵢ h[i](1-h[i+k])

For binary h, this counts the number of positions i where h[i]=1 and h[i+k]=0.

If we think of h as a binary sequence, corr[k] is the number of "10" patterns at distance k.

We want to minimize the maximum number of "10" patterns over all distances k.

This is related to the **Lempel-Ziv complexity** and **balance** of the sequence.

A perfectly balanced sequence would have corr[k] ≈ (n-k)/4 for all k (since on average, 1/4 of positions have a "10" pattern at any given distance). The maximum would be around (n-1)/4 * 2/n = 1/2.

  At lag 1: pairs are (1,0),(0,1),(1,0),... → corr = (n-1)/2
  At lag 2: pairs are (1,1),(0,0),(1,1),... → corr = 0
  At lag 3: pairs are (1,0),(0,1),... → corr = (n-3)/2

Let me also think about the smooth objective more carefully. The log-sum-exp approximation:
f_smooth(h) = cmax + ε * log(Σ exp((corr_k - cmax)/ε))

As ε → 0, this approaches max_k corr_k. The gradient of this smooth function is:
∇f_smooth = Σ_k w_k * ∂corr_k/∂h

where w_k = exp((corr_k - cmax)/ε) / Σ exp((corr_j - cmax)/ε)

For very small ε, w_k is concentrated on the active set (lags achieving the max). The gradient is then a weighted average of the gradients of the active correlations.

The issue with very small ε is numerical stability. Let me use ε = 0.001 as the "fine" value and ε = 0.01 as the "coarse" value.

Another thought: **what if I use a different smooth approximation?** For example, the entropic regularization:
f_ε(h) = ε * log(Σ exp(corr_k/ε))

This is smoother and has better numerical properties. The gradient is:
∇f_ε = (1/ε) * Σ_k (exp(corr_k/ε) / Σ exp(corr_j/ε)) * ∂corr_k/∂h
= Σ_k w_k * ∂corr_k/∂h

where w_k = exp(corr_k/ε) / Σ exp(corr_j/ε)

Same form, but the weights are computed differently (without subtracting cmax). This can be more stable for small ε.

Actually, the two formulations are equivalent up to a constant:
ε * log(Σ exp(corr_k/ε)) = cmax + ε * log(Σ exp((corr_k - cmax)/ε))

So they're the same. The subtraction of cmax is just for numerical stability.

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def _c_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def _c_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    h0 = project_onto_constraint(h_init.astype(float).copy(), n / 2.0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(smooth_obj, h0, args=(eps,),
                       jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        h_opt = project_onto_constraint(res.x, n / 2.0)
        return h_opt, compute_c5(h_opt)
    except:
        return h0, compute_c5(h0)

def round_binary(h):
    n = len(h)
    sorted_idx = np.argsort(-h)  # sort by value descending
    # Actually, we want to make h[sorted_idx[i]] = 1 for the top n/2
    # But we need to be careful: if h[i] > 0.5, we want h[i] = 1
    # Simple approach: threshold at 0.5, then adjust
    result = (h >= 0.5).astype(float)
    # If we don't have exactly n/2 ones, adjust
    while result.sum() > n / 2:
        # Flip the one closest to 0.5
        ones = np.where(result == 1.0)[0]
        flip = ones[np.argmin(h[ones])]
        result[flip] = 0.0
    while result.sum() < n / 2:
        zeros = np.where(result == 0.0)[0]
        flip = zeros[np.argmax(h[zeros])]
        result[flip] = 1.0
    return result
