def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)

            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:

                break
            sub = min(-diff, h[i])

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_vector(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):

        else:
            hi = mid

def _solve_1d_minimax(A, B, t_lo, t_hi, grid=200):

        return t_lo
    t_vals = np.linspace(t_lo, t_hi, grid)
    max_vals = (A + np.outer(t_vals, B)).max(axis=1)
    idx = np.argmin(max_vals)

    t_fine = np.linspace(lo, hi, grid)
    max_fine = (A + np.outer(t_fine, B)).max(axis=1)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n

    def _smooth_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * (2.0 / n)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

    def _eq_con_fun(x):
        return x.sum() - _global_target_sum

    def _eq_con_jac(x):
        return np.ones(_global_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha0, sg_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = _corr_vector(y)
        max_corr = np.max(corr)
        weights = np.exp(40.0 * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= total_w
        top_m = np.argsort(weights)[::-1][:sg_m]
        w_m = weights[top_m].copy()
        w_m /= w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

            if n_try not in n_candidates:
                n_candidates.append(n_try)
    n_candidates.sort(key=lambda x: abs(x - n_init))

    for n in n_candidates:
        if time.time() > deadline - 80:
            break
        target_sum = n / 2.0

        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)

        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        # Phase 1: Nesterov with smooth gradient
        if time.time() < deadline - 40:
            h_n, c5_n = _nesterov_smooth(h_start, n, target_sum,
                                          n_iters=200, alpha0=0.4, sg_m=5,
                                          deadline=deadline - 40)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
                best_n = n

        # Phase 2: Active-set projected gradient
        if time.time() < deadline - 35:
            h_as, c5_as = _active_set_peg(h_n, n, target_sum,
                                           n_iters=150, alpha0=0.25,
                                           n_active=5, deadline=deadline - 35)
            if c5_as < best_c5:
                best_c5 = c5_as
                best_h = h_as.copy()
                best_n = n

        # Phase 3: SLSQP with increasing alpha
        h_curr = h_as
        for alpha in [50, 200, 600, 1500, 3000]:
            if time.time() > deadline - 30:
                break
            h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=300)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_curr = h_sl

        # Phase 4: Exact single-coordinate descent
        if time.time() < deadline - 25:
            h_ecd = _exact_single_cd(h_curr, target_sum, max_rounds=25,
                                      grid=200, deadline=deadline - 25)
            c5_ecd = _c5_of(h_ecd)
            if c5_ecd < best_c5:
                best_c5 = c5_ecd
                best_h = h_ecd.copy()
                best_n = n

        # Phase 5: Exact pair-coordinate descent
        if time.time() < deadline - 20:
            h_pcd = _exact_pair_cd(best_h, best_n / 2.0, max_rounds=20,
                                    grid=200, deadline=deadline - 20)
            c5_pcd = _c5_of(h_pcd)
            if c5_pcd < best_c5:
                best_c5 = c5_pcd
                best_h = h_pcd.copy()
                best_n = best_n

        # Phase 6: SLSQP polish after coordinate descent
        if time.time() < deadline - 15:
            for alpha in [2000, 5000, 10000]:
                if time.time() > deadline - 12:
                    break
                h_pol = _slsqp_opt(best_h, best_n, best_n / 2.0, alpha, maxiter=400)
                c5_pol = _c5_of(h_pol)
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = h_pol.copy()

    # Final refinement on the best solution found
    n_best = best_n
    target = n_best / 2.0
    h_final = best_h.copy()

    # Alternating: pair CD + SLSQP
    for cycle in range(4):
        if time.time() > deadline - 10:
            break
        h_final = _exact_pair_cd(h_final, target, max_rounds=15,
                                  grid=200, deadline=deadline - 10)
        c5_tmp = _c5_of(h_final)
        if c5_tmp < best_c5:
            best_c5 = c5_tmp
            best_h = h_final.copy()
        if time.time() > deadline - 8:
            break
        h_sl = _slsqp_opt(h_final, n_best, target, 5000, maxiter=500)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            h_final = h_sl
        else:
            h_final = h_final

    # Final exact CD
    if time.time() < deadline - 6:
        h_ecd_f = _exact_single_cd(best_h, target, max_rounds=30,
                                    grid=300, deadline=deadline - 6)
        c5_ecd_f = _c5_of(h_ecd_f)
        if c5_ecd_f < best_c5:
            best_c5 = c5_ecd_f
            best_h = h_ecd_f.copy()

    # Final pair CD
    if time.time() < deadline - 4:
        h_pcd_f = _exact_pair_cd(best_h, target, max_rounds=25,
                                  grid=300, deadline=deadline - 4)
        c5_pcd_f = _c5_of(h_pcd_f)
        if c5_pcd_f < best_c5:
            best_c5 = c5_pcd_f
            best_h = h_pcd_f.copy()

    # One last SLSQP
    if time.time() < deadline - 2:
        h_last = _slsqp_opt(best_h, n_best, target, 15000, maxiter=500)
        c5_last = _c5_of(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
