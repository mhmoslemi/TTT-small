def compute_corr(h):
    """Compute the full correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

    """Compute C5 value"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr)) * dx

    """Project h onto the symmetry constraint h[i] + h[n-1-i] = 1"""
    n = len(h)
    return 0.5 * (h + (1.0 - h[::-1].copy()))

    """Extract free variables (first half)"""
    return h[:len(h)//2].copy()

    m = n // 2

    h[:m] = v
    h[n-m:] = 1.0 - v[::-1].copy()
    if n % 2 == 1:

    return h
