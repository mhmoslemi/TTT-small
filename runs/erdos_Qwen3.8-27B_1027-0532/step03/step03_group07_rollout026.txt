```
def _sa_pair_moves(h, n, target_sum, n_iter, t0, t_end, deadline, rng):
    """Simulated annealing with pair moves that preserve the sum."""
    h_curr = h.copy()
    c5_curr = _c5_of(h_curr)
    best_h = h_curr.copy()
    best_c5 = c5_curr
    t = t0
    cooling = (t_end / t0) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t *= cooling
        # Random pair move: h[i] += delta, h[j] -= delta
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(-0.1, 0.1)
        # Ensure bounds
        new_i = h_curr[i] + delta
        new_j = h_curr[j] - delta
        if (new_i < 0 or new_i > 1 or new_j < 0 or new_j > 1):
            continue
        h_curr[i] = new_i
        h_curr[j] = new_j
        c5_new = _c5_of(h_curr)
        if c5_new < c5_curr or rng.random() < np.exp((c5_curr - c5_new) / max(t, 1e-10)):
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_curr.copy()
        else:
            h_curr[i] -= delta
            h_curr[j] += delta
    return best_h, best_c5
```

This is O(n) per evaluation (for the correlation), so each iteration is fast. We can do many iterations.

The key advantage of pair moves is that they **exactly preserve the sum constraint**, so no projection is needed.

Actually, for binary sequences, the "overlap" at lag k is:
Σ_i 1[h[i]=1] * 1[h[i+k]=0] = (number of 1s) - (number of positions where both h[i]=1 and h[i+k]=1)
= n/2 - (autocorrelation at lag k)

So minimizing the max overlap is equivalent to maximizing the min autocorrelation.

This is related to the "Golay sequence" problem. Known constructions give autocorrelation values that are O(√n), which translates to C₅ ≈ 1/2 - O(1/√n) for the overlap... wait that doesn't sound right.

Let me recalculate. For binary h with n/2 ones:
Overlap at lag k = Σ_{i=0}^{n-1-k} h[i] * (1-h[i+k]) * dx
= (number of i where h[i]=1 and h[i+k]=0) * 2/n

Total number of i where h[i]=1 is n/2.
Number of i where h[i]=1 and h[i+k]=0 = n/2 - (number of i where h[i]=1 and h[i+k]=1)

Let A(k) = number of i where h[i]=1 and h[i+k]=1 (autocorrelation count).
Then overlap at lag k = (n/2 - A(k)) * 2/n = 1 - 2A(k)/n.

To minimize the max overlap, we want to maximize min_k A(k).

For a random binary sequence with n/2 ones, A(k) ≈ n/4 (by linearity of expectation), giving overlap ≈ 1/2.

Good constructions (like Golay or Barker sequences) can make A(k) more uniform, but the max overlap is still around 0.3-0.4 for practical n.

So the theoretical best for binary is probably around 0.3-0.35, and the continuous relaxation might improve this slightly.

```python
import time
import numpy as np
from scipy.optimize import minimize

_g_target = 0.0
_g_n = 0
_g_alpha = 100.0
_g_topm = 15

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo, hi = y.min() - 1.0, y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _smooth_val(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = corr.max()
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _smooth_grad(h, alpha, n, top_m=15):
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    m = corr.max()
    exps = np.exp(alpha * (corr - m))
    total = exps.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / total
    if top_m is not None and top_m < len(corr):
        top_idx = np.argsort(weights)[::-1][:top_m]
        w_sum = weights[top_idx].sum()
        if w_sum > 1e-15:
            mask = np.zeros(len(corr), dtype=bool)
            mask[top_idx] = True
            weights *= mask
            weights[top_idx] /= w_sum
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-12:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            kk = lag
            m_ = n - kk
            if m_ <= 0:
                continue
            g = np.zeros(n)
            g[kk:kk + m_] += (1.0 - h[:m_])
            g[:m_] -= h[kk:kk + m_]
        else:
            kk = -lag
            m_ = n - kk
            if m_ <= 0:
                continue
            g = np.zeros(n)
            g[:m_] += (1.0 - h[kk:kk + m_])
            g[kk:kk + m_] -= h[:m_]
        grad += w * dx * g
    return grad

def _slsqp_obj(h):
    return _smooth_val(h, _g_alpha, _g_n)

def _slsqp_jac(h):
    return _smooth_grad(h, _g_alpha, _g_n, top_m=_g_topm)

def _slsqp_eq(h):
    return h.sum() - _g_target

def _slsqp_eq_jac(h):
    return np.ones(_g_n)

def _slsqp_opt(h0, n, target, alpha, maxiter, topm=15):
    global _g_target, _g_n, _g_alpha, _g_topm
    _g_target = target
    _g_n = n
    _g_alpha = alpha
    _g_topm = topm
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_obj, h0, jac=_slsqp_jac,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _nesterov_smooth(h0, n, target, alpha_start, alpha_end, lr_base, n_iters, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr_base / np.sqrt(float(it + 1))
        g = _smooth_grad(y, alpha, n, top_m=10)
        x_new = _project(y - lr * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

        elif move_type == 1:

def _coord_bin(h0, n, target, max_rounds):
    h = h0.copy()
    cur_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            bc = c0 if c0 < c1 else c1
            if bc < cur_c5 - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                cur_c5 = bc
                improved = True
        if not improved:
            break
    return h

def _coord_frac(h0, n, target, max_rounds):
    h = h0.copy()
    cur_c5 = _c5_of(h)
    vals = [0.0, 0.1, 0.2, 0.25, 1.0 / 3.0, 0.4, 0.5, 0.6, 2.0 / 3.0, 0.75, 0.8, 0.9, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            bc = cur_c5
            bv = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < bc - 1e-13:
                    bc = c
                    bv = v
            h[j] = bv
            if bv != old:
                cur_c5 = bc
                improved = True
        if not improved:
            break
    return h

                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if improved:
            stall = 0
        else:
            stall += 1
        if stall >= 20:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = h_init.copy()
    best_n = n_init

    n_list = []
    for nn in [n_init, n_init + 10, n_init - 10, n_init + 20, n_init + 40, n_init + 60]:
        if 80 <= nn <= 300 and nn % 2 == 0 and nn not in n_list:
            n_list.append(nn)
    n_list = sorted(n_list)

    for n in n_list:
        if time.time() > deadline - 120:
            break
        target = n / 2.0
        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target)
        c5_s = _c5_of(h_start)
        if c5_s < best_c5:
            best_c5, best_h, best_n = c5_s, h_start.copy(), n

        h_de, c5_de = _multi_start_de(h_start, n, target, 12, 60, deadline - 200, rng)
        if c5_de < best_c5:
            best_c5, best_h, best_n = c5_de, h_de.copy(), n

        if time.time() < deadline - 150:
            h_n1, c5_n1 = _nesterov_smooth(h_de, n, target, 50, 400, 0.8, 400, deadline - 150)
            if c5_n1 < best_c5:
                best_c5, best_h, best_n = c5_n1, h_n1.copy(), n

        if time.time() < deadline - 120:
            h_sa, c5_sa = _sa_simple(h_n1, n, target, 3000, 0.2, 0.005, deadline - 120, rng)
            if c5_sa < best_c5:
                best_c5, best_h, best_n = c5_sa, h_sa.copy(), n

        if time.time() < deadline - 80:
            h_n2, c5_n2 = _nesterov_smooth(h_sa, n, target, 100, 2000, 0.4, 300, deadline - 80)
            if c5_n2 < best_c5:
                best_c5, best_h, best_n = c5_n2, h_n2.copy(), n

        h_curr = h_n2
        for alpha in [200, 500, 1500, 4000]:
            if time.time() > deadline - 50:
                break
            h_sl = _slsqp_opt(h_curr, n, target, alpha, 250, topm=12)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5, best_h, best_n = c5_sl, h_sl.copy(), n
            h_curr = h_sl

        if time.time() < deadline - 30:
            h_cd = _coord_bin(h_curr, n, target, 20)
            c5_cd = _c5_of(h_cd)
            if c5_cd < best_c5:
                best_c5, best_h, best_n = c5_cd, h_cd.copy(), n
            if time.time() < deadline - 20:
                h_cf = _coord_frac(h_cd, n, target, 8)
                c5_cf = _c5_of(h_cf)
                if c5_cf < best_c5:
                    best_c5, best_h, best_n = c5_cf, h_cf.copy(), n

    target_f = best_n / 2.0
    if time.time() < deadline - 15:
        hp = _slsqp_opt(best_h, best_n, target_f, 3000, 400, topm=10)
        c5p = _c5_of(hp)
        if c5p < best_c5:
            best_c5, best_h = c5p, hp.copy()

    if time.time() < deadline - 10:
        h_cd2 = _coord_bin(best_h, best_n, target_f, 30)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2.copy()

    if time.time() < deadline - 6:
        h_cf2 = _coord_frac(best_h, best_n, target_f, 10)
        c5_cf2 = _c5_of(h_cf2)
        if c5_cf2 < best_c5:
            best_c5, best_h = c5_cf2, h_cf2.copy()

    if time.time() < deadline - 3:
        hp2 = _slsqp_opt(best_h, best_n, target_f, 8000, 350, topm=8)
        c5p2 = _c5_of(hp2)
        if c5p2 < best_c5:
            best_c5, best_h = c5p2, hp2.copy()

    if time.time() < deadline - 1:
        h_cd3 = _coord_bin(best_h, best_n, target_f, 15)
        c5_cd3 = _c5_of(h_cd3)
        if c5_cd3 < best_c5:
            best_c5, best_h = c5_cd3, h_cd3.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
```
