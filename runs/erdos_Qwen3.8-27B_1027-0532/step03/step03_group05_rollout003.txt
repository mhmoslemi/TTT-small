import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights /= sw
    n_corr = len(corr)
    shifts = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for ki in range(n_corr):
        w = weights[ki]
        if w < 1e-10:
            continue
        s = shifts[ki]
        p1s = max(0, s)
        p1e = min(n, n + s)
        if p1s < p1e:
            grad[p1s:p1e] += w * (1.0 - h[p1s - s:p1e - s])
        p2s = max(0, -s)
        p2e = min(n, n - s)
        if p2s < p2e:
            grad[p2s:p2e] -= w * h[p2s + s:p2e + s]
    return grad

def slsqp_single(h, target, alpha, maxiter, ftol=1e-15):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_max_obj(x, alpha)
    
    def obj_grad(x):
        return smooth_max_grad(x, alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def subgradient_descent(h, target, time_limit, alpha=50, eta=0.01):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    while time.time() - start < time_limit:
        grad = smooth_max_grad(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        h_new = h - eta * grad / gn
        h_new = project_feasible(h_new, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new.copy()
            eta = min(eta * 1.02, 0.05)
        else:
            eta *= 0.9
        h = h_new
    return best, best_score

def pairwise_descent(h, target, time_limit, seed=42):
    """Try moving mass from one coordinate to another."""
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    dx = 2.0 / n
    delta = 0.01
    
    improved_any = True
    while time.time() - start < time_limit and improved_any:
        improved_any = False
        n_pairs = min(5000, n * 5)
        for _ in range(n_pairs):
            if time.time() - start > time_limit:
                break
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            for d in [delta, -delta, delta*0.5, -delta*0.5]:
                val_i = best[i] + d
                val_j = best[j] - d
                if val_i < 0 or val_i > 1 or val_j < 0 or val_j > 1:

                trial = best.copy()
                trial[i] = val_i
                trial[j] = val_j

                s = np.max(corr) * dx
                if s < best_score - 1e-14:
                    best_score = s
                    best = trial
                    improved_any = True
                    break
        if not improved_any:
            delta *= 0.7
            if delta < 0.001:
                break
    
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_max_grad(x, alpha):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights = weights / sw
    n_corr = len(corr)
    shifts = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for ki in range(n_corr):
        w = weights[ki]
        if w < 1e-12:
            continue
        s = shifts[ki]
        p1s = max(0, s)
        p1e = min(n, n + s)
        if p1s < p1e:
            grad[p1s:p1e] += w * (1.0 - x[p1s - s:p1e - s])
        p2s = max(0, -s)
        p2e = min(n, n - s)
        if p2s < p2e:
            grad[p2s:p2e] -= w * x[p2s + s:p2e + s]
    return grad


def slsqp_optimize(h, target, alpha, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return smooth_max_obj(x, alpha)

    def obj_jac(x):
        return smooth_max_grad(x, alpha)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, jac=obj_jac, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 200})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h.copy(), compute_c5(h)

    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
