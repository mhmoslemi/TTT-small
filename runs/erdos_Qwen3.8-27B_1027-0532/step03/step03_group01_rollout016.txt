    zeros = np.where(h == 0)[0]

    best_i = -1
    best_j = -1
    current_max = float(np.max(corr))
    
    if len(ones) == 0 or len(zeros) == 0:
        return h, corr, current_max, 0
    
    # Precompute add_matrix[j_idx, k] = -h[j+k] + (1-h[j-k]) for j in zeros
    # k ranges from -(n-1) to (n-1), stored at index k + (n-1)
    # For a position j, add_j[k] = -h[j+k] + (1-h[j-k])
    # j+k is valid when 0 <= j+k < n, i.e., -j <= k < n-j
    # j-k is valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
    
    n_corr = 2 * n - 1
    # add_matrix shape: (len(zeros), n_corr)
    # For zero at position j, and lag k (stored at index k + n - 1):
    # add = -h[j+k] (if 0<=j+k<n) + (1-h[j-k]) (if 0<=j-k<n)
    
    # Vectorized computation
    # j values: zeros array
    # k values: -(n-1), ..., 0, ..., (n-1)
    k_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    j_vals = zeros  # shape (n/2,)

    jk = j_vals[:, None] + k_vals[None, :]  # (n/2, 2n-1)
    jmk = j_vals[:, None] - k_vals[None, :]  # (n/2, 2n-1)
    
    # h[j+k] with bounds
    valid_jk = (jk >= 0) & (jk < n)
    valid_jmk = (jmk >= 0) & (jmk < n)
    
    h_jk = np.where(valid_jk, h[np.clip(jk, 0, n-1)], 0.0)
    h_jmk = np.where(valid_jmk, h[np.clip(jmk, 0, n-1)], 0.0)
    
    add_matrix = -h_jk + (1.0 - h_jmk)  # (n/2, 2n-1)
    
    # Now for each i in ones, compute remove_i and find best j
    for i in ones:
        # remove_i[k] = h[i+k] - (1-h[i-k])
        ik = i + k_vals  # (2n-1,)
        imk = i - k_vals  # (2n-1,)
        
        valid_ik = (ik >= 0) & (ik < n)
        valid_imk = (imk >= 0) & (imk < n)
        
        h_ik = np.where(valid_ik, h[np.clip(ik, 0, n-1)], 0.0)
        h_imk = np.where(valid_imk, h[np.clip(imk, 0, n-1)], 0.0)

        # new_corr[k] = corr[k] + remove_i[k] + add_matrix[j, k]
        # = (corr + remove_i)[k] + add_matrix[j, k]
        base = corr + remove_i  # (2n-1,)
        
        # For each j: new_max_j = max_k (base[k] + add_matrix[j, k])
        # This is (n/2, 2n-1) -> max along axis 1
        new_maxs = (add_matrix + base[None, :]).max(axis=1)
        
        # Find best j (minimum new_max)
        best_j_idx = np.argmin(new_maxs)

            best_i = i
            best_j = zeros[best_j_idx]
    
    if best_i < 0:

    # Apply the best swap
    h = h.copy()
    h[best_i] = 0
    h[best_j] = 1
    corr = get_corr(h)
    new_max = float(np.max(corr))
    
    return h, corr, new_max, best_improve

_cons_n = 0
_cons_target = 0.0

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def golden_ratio_h(n, seed=0):
    """Create a binary h using golden ratio spacing."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    positions = np.array([int(i * phi * n / (n // 2 + 1)) % n for i in range(n // 2)])
    # Ensure no duplicates: if there are, shift
    unique_positions = []
    used = set()
    for p in positions:
        while p in used:
            p = (p + 1) % n
        used.add(p)
        unique_positions.append(p)
    h[unique_positions[:n//2]] = 1.0
    return h
