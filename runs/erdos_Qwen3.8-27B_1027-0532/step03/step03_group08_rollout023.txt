    s = h.sum()
    if s == target:

    h = h.copy()
    if s > target:

    """Compute delta in correlation array for swapping h[i]=1 and h[j]=0.

    # k ranges from -(n-1) to n-1

    # delta at such k: -h[j+k]

        if 0 <= idx < n:

    # delta at such k: 1 - h[j-k]
    k_lo3 = max(j-(n-1), -(n-1))

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_corr(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    return float(np.max(corr)) * dx, corr * dx

def _fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _round_to_binary(h, target):
    h = h.copy()
    s = h.sum()
    # Sort by distance to nearest integer
    dist = np.minimum(h, 1-h)
    order = np.argsort(dist)
    h_binary = np.zeros_like(h)
    # Greedily assign 0 or 1
    for i in order:
        if h[i] > 0.5:
            h_binary[i] = 1
        else:
            h_binary[i] = 0
    # Fix sum
    diff = target - h_binary.sum()
    if diff > 0:
        # Need more 1s
        zeros = np.where(h_binary == 0)[0]
        # Prefer those with highest original h value
        order2 = np.argsort(h[zeros])[::-1]
        for k in range(int(diff)):
            h_binary[zeros[order2[k]]] = 1
    elif diff < 0:
        ones = np.where(h_binary == 1)[0]
        order2 = np.argsort(h[ones])
        for k in range(int(-diff)):
            h_binary[ones[order2[k]]] = 0
    return h_binary

def _local_search(h, max_rounds=1000, deadline=None, rng=None):
    n = len(h)
    best_c5 = _c5_of(h)
    h = h.copy()
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_swap = None
        best_swap_c5 = best_c5
        # Try all swaps
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c = _c5_of(h)
                h[i] = 1
                h[j] = 0
                if c < best_swap_c5 - 1e-12:
                    best_swap_c5 = c
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            best_c5 = best_swap_c5
        else:
            break
    return h, best_c5

def _sa_search(h, max_iters=50000, T_start=0.01, T_end=0.0001, deadline=None, rng=None):
    n = len(h)
    h = h.copy()
    cur_val = _c5_of(h)
    best_h = h.copy()
    best_val = cur_val
    
    for it in range(max_iters):
        if deadline and time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0
        h[j] = 1
        new_val = _c5_of(h)
        
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-10)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_val

def _fractional_refine(h0, n, target_sum, deadline):
    # Use SLSQP with smooth objective

        corr = np.correlate(x, 1.0 - x, mode='full')
        # Smooth max using log-sum-exp
        alpha = 100.0
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)
    
    def obj_grad(x):

        alpha = 100.0
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        w /= w.sum()

        #                   -x[j-k_off] if j is the "second" index
        # This is complex, let me just use numerical gradient or skip it.
        # For SLSQP, we can provide the gradient or let it estimate.
        # Let's just return None and let SLSQP estimate.
        return None
    
    def eq_fun(x):
        return x.sum() - target_sum
    
    def eq_jac(x):
        return np.ones(n)

import time

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_with_lags(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)
    scaled = corr * dx
    max_val = float(np.max(scaled))
    max_lags = np.where(scaled > max_val - 0.001)[0]
    return max_val, max_lags, scaled

def _fix_sum(h, target):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_sum(y, S):
    """Project y onto {x: 0≤x≤1, sum(x)=S} using bisection on threshold."""
    lo, hi = y.min(), y.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any residual
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:

def _grad_c_at_lag(h, lag, n):
    """Gradient of c_lag with respect to h, where lag is the full-mode index (0 to 2n-2).
    c_lag = sum_t h[t] * (1-h[t+k]) where k = lag - (n-1)"""
    k = lag - (n - 1)
    g = np.zeros(n)
    # c_lag = sum_{t: 0≤t<n, 0≤t+k<n} h[t] * (1-h[t+k])
    # d/dh[j]: 
    #   If j is "first index" (t=j): contributes (1-h[j+k]) if 0≤j+k<n
    #   If j is "second index" (t+k=j, t=j-k): contributes -h[j-k] if 0≤j-k<n
    if k >= 0:
        m = n - k
        for t in range(m):
            g[t] += (1.0 - h[t + k])
            g[t + k] += (-h[t])
    else:
        j = -k
        m = n - j
        for t in range(m):
            g[t + j] += (1.0 - h[t])
            g[t] += (-h[t + j])
    return g

def _grad_c_at_lag_vec(h, lag, n):
    """Vectorized gradient of c_lag."""
    k = lag - (n - 1)
    g = np.zeros(n)
    if k == 0:
        g[:] = h - 1.0  # d/dh[j] of h[j](1-h[j]) = 1-2h[j]... wait
        # c_0 = sum h[t](1-h[t]) = sum h[t] - sum h[t]^2
        # d/dh[j] = 1 - 2h[j]
        # But for our purpose, h[j](1-h[j]) for binary is 0, and the derivative
        # at the boundary would be 1 (from below) or -1 (from above)
        # For the smoothed version, gradient is 1-2h[j]
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        m = n - k
        g[:m] += (1.0 - h[k:k+m])
        g[k:k+m] += -h[:m]
    else:
        j = -k
        m = n - j
        g[j:j+m] += (1.0 - h[:m])
        g[:m] += -h[j:j+m]
    return g

def _projected_grad_descent(h0, n, target_sum, n_iters=5000, alpha0=0.1, 
                            alpha_smooth=50.0, deadline=None):
    """Projected gradient descent with log-sum-exp smoothing."""
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_of(h)
    
    for it in range(n_iters):
        if deadline and time.time() > deadline:
            break
        
        # Compute current correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        scaled = corr * dx
        
        # Softmax weights
        max_s = np.max(scaled)
        exp_s = np.exp(alpha_smooth * (scaled - max_s))
        p = exp_s / exp_s.sum()
        
        # Find top few lags
        top_lags = np.argsort(p)[::-1][:10]
        
        # Compute gradient as weighted sum
        g = np.zeros(n)
        for lag in top_lags:
            if p[lag] > 0.01:
                g += p[lag] * _grad_c_at_lag_vec(h, lag, n)
        
        # Step
        alpha = alpha0 / np.sqrt(1.0 + 0.01 * it)
        h_new = _project_sum(h - alpha * g, target_sum)
        
        # Evaluate
        val = _c5_of(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_val
