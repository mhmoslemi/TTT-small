```python
def compute_delta(h, i, j, n):
    """Compute the change in correlation when swapping h[i]=1->0 and h[j]=0->1."""
    delta = np.zeros(2*n-1)
    half = n-1
    for s in range(-(n-1), n):
        L = max(0, -s)
        R = min(n-1, n-1-s)
        d = 0
        # Check each affected position
        for t in [i, j, i-s, j-s]:
            if L <= t <= R:
                # Compute contribution change
                # Before: h[t] * (1-h[t+s])
                # After: h'[t] * (1-h'[t+s])
                # where h' differs from h only at i and j
                ht = 0 if t == i else (1 if t == j else h[t])
                ht_s = 0 if (t+s) == i else (1 if (t+s) == j else h[t+s])
                ht_old = h[t]
                ht_s_old = h[t+s]
                d += ht * (1 - ht_s) - ht_old * (1 - ht_s_old)
        if d != 0:
            delta[s + half] = d
    return delta
```

```python
    n = len(h)
    half = n - 1
    # For each of the 4 affected positions, compute valid s range and contribution
    # Position i: valid for s in [-i, n-1-i], contribution = h[i+s] - 1
    # Position j: valid for s in [-j, n-1-j], contribution = 1 - h'[j+s]
    # Position i-s: valid for s in [i-(n-1), i], contribution = h'[i-s]
    # Position j-s: valid for s in [j-(n-1), j], contribution = -h[j-s]
    
    s_min = min(-i, -j, i-half, j-half)
    s_max = max(n-1-i, n-1-j, i, j)
    
    delta = np.zeros(s_max - s_min + 1)
    for idx, s in enumerate(range(s_min, s_max + 1)):

        if -i <= s <= n-1-i:
            d += h[i+s] - 1
        # t = j
        if -j <= s <= n-1-j:
            hs = j + s
            if 0 <= hs < n:
                h_new_hs = 0 if hs == i else (1 if hs == j else h[hs])
                d += 1 - h_new_hs

        if i-half <= s <= i:
            t = i - s
            if 0 <= t < n:
                h_new_t = 1 if t == j else (0 if t == i else h[t])
                d += h_new_t
        # t = j-s
        if j-half <= s <= j:
            t = j - s
            if 0 <= t < n:
                d += -h[t]
        delta[idx] = d
    return delta, s_min
```

```python
def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad
```

```python
            r = -s
            m = n - r
            grad[r:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[r:]
    return obj, grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target)
    bounds = [(0.0, 1.0)] * n
    def obj_fun(x):
        return smooth_obj_and_grad(x, eps)[0]
    def grad_fun(x):
        return smooth_obj_and_grad(x, eps)[1]
    def con_fun(x):
        return float(np.sum(x) - target)
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        result = minimize(obj_fun, h0, jac=grad_fun, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def fast_binary_sa(h_start, n_steps, t_start, t_end, rng):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1

    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    n_one = n // 2
    n_zero = n // 2

    pos_to_idx_one = np.full(n, -1, dtype=np.int32)
    pos_to_idx_zero = np.full(n, -1, dtype=np.int32)
    for idx, pos in enumerate(ones_list):
        pos_to_idx_one[pos] = idx
    for idx, pos in enumerate(zeros_list):
        pos_to_idx_zero[pos] = idx

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        i = ones_list[rng.integers(n_one)]
        j = zeros_list[rng.integers(n_zero)]

        s_lo = min(-i, -j, i - half, j - half)
        s_hi = max(half - i, half - j, i, j)
        if s_lo > s_hi:
            h[i] = 0.0
            h[j] = 1.0
            oi = pos_to_idx_one[i]
            pos_to_idx_one[i] = -1
            pos_to_idx_one[j] = oi
            ones_list[oi] = j
            oz = pos_to_idx_zero[j]
            pos_to_idx_zero[j] = -1
            pos_to_idx_zero[i] = oz
            zeros_list[oz] = i
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            current_max = float(np.max(corr))
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
            step += 1
            continue

        s_range = np.arange(s_lo, s_hi + 1)
        k_range = s_range + half
        delta = np.zeros(len(s_range), dtype=np.float64)

        mask_i = (s_range >= -i) & (s_range <= half - i)
        if np.any(mask_i):
            delta[mask_i] += h[i + s_range[mask_i]] - 1.0

        mask_j = (s_range >= -j) & (s_range <= half - j)
        if np.any(mask_j):
            js = j + s_range[mask_j]
            contrib = 1.0 - h[js].astype(np.float64)
            c_i = (js == i)
            c_j = (js == j)
            contrib[c_i] = 1.0
            contrib[c_j] = 0.0
            delta[mask_j] += contrib

        mask_is = (s_range >= i - half) & (s_range <= i)
        if np.any(mask_is):
            vals = i - s_range[mask_is]
            contrib = h[vals].astype(np.float64)
            c_j2 = (vals == j)
            c_i2 = (vals == i)
            contrib[c_j2] = 1.0
            contrib[c_i2] = 0.0
            delta[mask_is] += contrib

        mask_js = (s_range >= j - half) & (s_range <= j)
        if np.any(mask_js):
            vals = j - s_range[mask_js]
            delta[mask_js] += -h[vals].astype(np.float64)

        s_coinc1 = j - i
        if s_lo <= s_coinc1 <= s_hi:
            delta[s_coinc1 - s_lo] += 1.0

        s_coinc2 = i - j
        if s_lo <= s_coinc2 <= s_hi:
            delta[s_coinc2 - s_lo] -= 1.0

        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max

        if delta_c5 < -1e-14 or (t > 1e-12 and rng.random() < np.exp(-delta_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            oi = pos_to_idx_one[i]
            pos_to_idx_one[i] = -1
            pos_to_idx_one[j] = oi
            ones_list[oi] = j
            oz = pos_to_idx_zero[j]
            pos_to_idx_zero[j] = -1
            pos_to_idx_zero[i] = oz
            zeros_list[oz] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta

        step += 1

    return best_h, best_c5_val

def greedy_improve(h_bin, n_passes, rng):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1

    for p in range(n_passes):
        improved = False
        ones_list = np.where(h == 1.0)[0].tolist()
        zeros_list = np.where(h == 0.0)[0].tolist()
        rng.shuffle(ones_list)
        rng.shuffle(zeros_list)
        for i in ones_list:
            if h[i] != 1.0:
                continue
            best_j = -1
            best_new_max = current_max
            for j in zeros_list:
                if h[j] != 0.0:
                    continue
                s_lo = min(-i, -j, i - half, j - half)
                s_hi = max(half - i, half - j, i, j)
                if s_lo > s_hi:
                    continue
                s_range = np.arange(s_lo, s_hi + 1)
                k_range = s_range + half
                delta = np.zeros(len(s_range), dtype=np.float64)

                mask_i = (s_range >= -i) & (s_range <= half - i)
                if np.any(mask_i):
                    delta[mask_i] += h[i + s_range[mask_i]] - 1.0

                mask_j = (s_range >= -j) & (s_range <= half - j)
                if np.any(mask_j):
                    js = j + s_range[mask_j]
                    contrib = 1.0 - h[js].astype(np.float64)
                    contrib[js == i] = 1.0
                    contrib[js == j] = 0.0
                    delta[mask_j] += contrib

                mask_is = (s_range >= i - half) & (s_range <= i)
                if np.any(mask_is):
                    vals = i - s_range[mask_is]
                    contrib = h[vals].astype(np.float64)
                    contrib[vals == j] = 1.0
                    contrib[vals == i] = 0.0
                    delta[mask_is] += contrib

                mask_js = (s_range >= j - half) & (s_range <= j)
                if np.any(mask_js):
                    vals = j - s_range[mask_js]
                    delta[mask_js] += -h[vals].astype(np.float64)

                s_coinc1 = j - i
                if s_lo <= s_coinc1 <= s_hi:
                    delta[s_coinc1 - s_lo] += 1.0

                s_coinc2 = i - j
                if s_lo <= s_coinc2 <= s_hi:
                    delta[s_coinc2 - s_lo] -= 1.0

                corr_trial = corr.copy()
                corr_trial[k_range] += delta
                new_max = float(np.max(corr_trial))
                if new_max < best_new_max - 1e-14:
                    best_new_max = new_max
                    best_j = j
            if best_j >= 0:
                j = best_j
                s_lo = min(-i, -j, i - half, j - half)
                s_hi = max(half - i, half - j, i, j)
                s_range = np.arange(s_lo, s_hi + 1)
                k_range = s_range + half
                delta = np.zeros(len(s_range), dtype=np.float64)

                mask_i = (s_range >= -i) & (s_range <= half - i)
                if np.any(mask_i):
                    delta[mask_i] += h[i + s_range[mask_i]] - 1.0

                mask_j = (s_range >= -j) & (s_range <= half - j)
                if np.any(mask_j):
                    js = j + s_range[mask_j]
                    contrib = 1.0 - h[js].astype(np.float64)
                    contrib[js == i] = 1.0
                    contrib[js == j] = 0.0
                    delta[mask_j] += contrib

                mask_is = (s_range >= i - half) & (s_range <= i)
                if np.any(mask_is):
                    vals = i - s_range[mask_is]
                    contrib = h[vals].astype(np.float64)
                    contrib[vals == j] = 1.0
                    contrib[vals == i] = 0.0
                    delta[mask_is] += contrib

                mask_js = (s_range >= j - half) & (s_range <= j)
                if np.any(mask_js):
                    vals = j - s_range[mask_js]
                    delta[mask_js] += -h[vals].astype(np.float64)

                s_coinc1 = j - i
                if s_lo <= s_coinc1 <= s_hi:
                    delta[s_coinc1 - s_lo] += 1.0

                s_coinc2 = i - j
                if s_lo <= s_coinc2 <= s_hi:
                    delta[s_coinc2 - s_lo] -= 1.0

                corr[k_range] += delta
                h[i] = 0.0
                h[j] = 1.0
                current_max = best_new_max
                improved = True
                break
        if not improved:
            break
    return h, current_max

def upsample(h, factor):
    n = len(h)
    return np.repeat(h, factor)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target0 = n0 / 2.0
    h0 = project_onto_constraint(h0, target0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n0

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        h_soft = best_h.copy()
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005]:
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.12:
                break
            h_soft, c5 = slsqp_optimize(h_soft, eps, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_soft.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        h_bin = to_binary(best_h)
        remaining = budget_s - elapsed - 150
        n_sa = max(50000, int(remaining * 3000))
        h_bin, bin_max = fast_binary_sa(h_bin, n_sa, 0.4, 0.01, rng)
        c5_bin = (2.0 / n0) * bin_max
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        elapsed = time.time() - start_time
        if budget_s - elapsed > 80:
            h_polish = h_bin + 0.02 * rng.standard_normal(n0)
            h_polish = project_onto_constraint(h_polish, target0)
            h_polish, c5 = slsqp_optimize(h_polish, 0.001, 200)
            h_polish, c5 = slsqp_optimize(h_polish, 0.0003, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    n_sa_rounds = 0
    elapsed = time.time() - start_time
    while budget_s - elapsed > 250 and n_sa_rounds < 40:
        strategy = n_sa_rounds % 5
        if strategy == 0:
            h_start = to_binary(best_h)
        elif strategy == 1:
            h_pert = best_h + 0.1 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, target0)
            h_start = to_binary(h_pert)
        elif strategy == 2:
            h_start = to_binary(best_h)
            n_flip = max(3, n0 // 15)
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            s = h_start.sum()
            if s > target0:
                ones = np.where(h_start == 1)[0]
                extra = int(s - target0)
                h_start[ones[:extra]] = 0
            elif s < target0:
                zeros = np.where(h_start == 0)[0]
                need = int(target0 - s)
                h_start[zeros[:need]] = 1
        elif strategy == 3:
            alpha = rng.uniform(0.3, 0.7)
            h_rand = (rng.random(n0) < 0.5).astype(float)
            h_start = to_binary(alpha * best_h + (1 - alpha) * h_rand)
        else:
            h_start = to_binary(best_h)
            n_swap = max(2, n0 // 20)
            for _ in range(n_swap):
                ones = np.where(h_start == 1)[0]
                zeros = np.where(h_start == 0)[0]
                ii = rng.integers(len(ones))
                jj = rng.integers(len(zeros))
                h_start[ones[ii]], h_start[zeros[jj]] = h_start[zeros[jj]], h_start[ones[ii]]

        t_start = 0.3 + 0.3 * max(0, 1.0 - n_sa_rounds / 30.0)
        t_end = 0.003
        remaining = budget_s - (time.time() - start_time) - 200
        n_steps = min(int(remaining * 4000), 1500000)

        h_bin, bin_max = fast_binary_sa(h_start, n_steps, t_start, t_end, rng)
        c5_bin = (2.0 / n0) * bin_max
        if c5_bin < best_c5 - 1e-6:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            elapsed = time.time() - start_time
            if budget_s - elapsed > 80:
                h_polish = h_bin + 0.01 * rng.standard_normal(n0)
                h_polish = project_onto_constraint(h_polish, target0)
                h_polish, c5 = slsqp_optimize(h_polish, 0.0005, 150)
                h_polish, c5 = slsqp_optimize(h_polish, 0.0001, 150)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_polish.copy()
        n_sa_rounds += 1
        elapsed = time.time() - start_time

    elapsed = time.time() - start_time
    if budget_s - elapsed > 120:
        n2 = 360
        try:
            h2 = upsample(to_binary(best_h), 2)
            h2 = h2 + 0.3 * rng.standard_normal(n2)
            h2 = project_onto_constraint(h2, n2 / 2.0)
            h2, c5_2 = slsqp_optimize(h2, 0.005, 100)
            h2, c5_2 = slsqp_optimize(h2, 0.001, 150)
            if c5_2 < best_c5:
                h_bin2 = to_binary(h2)
                remaining2 = budget_s - (time.time() - start_time) - 60
                n_sa2 = max(20000, int(remaining2 * 2000))
                h_bin2, bin_max2 = fast_binary_sa(h_bin2, n_sa2, 0.3, 0.005, rng)
                c5_bin2 = (2.0 / n2) * bin_max2
                if c5_bin2 < best_c5:
                    best_c5 = c5_bin2
                    best_h = h_bin2.copy()
                    best_n = n2
        except Exception:
            pass

    elapsed = time.time() - start_time
    if budget_s - elapsed > 50:
        h_final = best_h + 0.003 * rng.standard_normal(best_n)
        h_final = project_onto_constraint(h_final, best_n / 2.0)
        for eps in [0.0001, 0.00003]:
            if time.time() - start_time > budget_s - 20:
                break
            h_final, c5 = slsqp_optimize(h_final, eps, 200)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
```
