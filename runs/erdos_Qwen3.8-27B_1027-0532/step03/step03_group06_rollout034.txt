def compute_flip_delta(h, i, j):

    # delta(s) for s in range(-(n-1), n)
    # -(1-h[i-s]) for valid i-s
    # + h[i+s] for valid i+s
    # + (1-h[j-s]) for valid j-s
    # - h[j+s] for valid j+s

    delta = np.zeros(2*n-1)

    valid1 = (i - s >= 0) & (i - s < n)
    delta[valid1] -= (1 - h[i - s[valid1]])
    
    # Term 2: +h[i+s] where 0 <= i+s < n
    valid2 = (i + s >= 0) & (i + s < n)
    delta[valid2] += h[i + s[valid2]]
    
    # Term 3: +(1-h[j-s]) where 0 <= j-s < n
    valid3 = (j - s >= 0) & (j - s < n)
    delta[valid3] += (1 - h[j - s[valid3]])
    
    # Term 4: -h[j+s] where 0 <= j+s < n
    valid4 = (j + s >= 0) & (j + s < n)
    delta[valid4] -= h[j + s[valid4]]
    
    return delta

Let me write the gradient computation more carefully:

The smooth objective: F(h) = max_k corr[k] ≈ (1/α) log(∑_k exp(α * corr[k]))

Gradient: ∂F/∂h[j] = (1/∑_k exp(α*corr[k])) * ∑_k exp(α*corr[k]) * ∂corr[k]/∂h[j]

And ∂corr[k]/∂h[j] = (1-h[j-s_k]) * [j-s_k valid] - h[j+s_k] * [j+s_k valid]

where s_k = k - (n-1) is the shift corresponding to correlation index k.

Let me index this properly. Let k range from 0 to 2n-2. Shift s = k - (n-1) ranges from -(n-1) to n-1.

∂corr[k]/∂h[j] = (1-h[j-s]) * I(0 ≤ j-s < n) - h[j+s] * I(0 ≤ j+s < n)

For the gradient at position j:
∂F/∂h[j] = (1/Σ) * Σ_k w_k * [(1-h[j-s_k]) * I(0≤j-s_k<n) - h[j+s_k] * I(0≤j+s_k<n)]

where w_k = exp(α*corr[k]) / Σ (normalized weights).

I can precompute:
- term1[j] = Σ_k w_k * (1-h[j-s_k]) * I(0≤j-s_k<n)
- term2[j] = Σ_k w_k * h[j+s_k] * I(0≤j+s_k<n)
- grad[j] = term1[j] - term2[j]

For term1: for each j, sum over k where j-s_k is valid:
j-s_k = j - (k-(n-1)) = j - k + n - 1
Valid when 0 ≤ j - k + n - 1 < n, i.e., 1 ≤ k ≤ j + n

Let me define m = j - s_k = j - k + n - 1 (the index of the "other" h value)
Then term1[j] = Σ_{k: m valid} w_k * (1-h[m])

This is a "weighted sum" of (1-h) over a sliding window, weighted by w_k.

Similarly, term2[j] = Σ_{k: j+s_k valid} w_k * h[j+s_k]
= Σ_{k: l=j+s_k=j+k-(n-1) valid} w_k * h[l]
= Σ_{k: 0 ≤ j+k-(n-1) < n} w_k * h[j+k-(n-1)]
= Σ_{k: n-1-j ≤ k ≤ 2n-2-j} w_k * h[j+k-(n-1)]

These are both "weighted convolution" type operations. I can compute them efficiently using FFT or direct convolution if the weights are simple, but since w_k is arbitrary (depends on the current corr values), I need to do it directly or use some trick.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
    return h


def project_to_binary(h, n):
    target = n // 2
    sorted_idx = np.argsort(h)[::-1]

def compute_C_fast(h):
    n = len(h)
    p = np.arange(n)[:, None]
    t = np.arange(-(n - 1), n)[None, :]
    idx_a = p - t
    valid_a = (idx_a >= 0) & (idx_a < n)
    A = np.where(valid_a, 1.0 - h[np.clip(idx_a, 0, n - 1)], 0.0)
    idx_b = p + t
    valid_b = (idx_b >= 0) & (idx_b < n)
    B = np.where(valid_b, h[np.clip(idx_b, 0, n - 1)], 0.0)
    return A - B

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = exp_vals.sum()
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2 * n - 1)[None, :]
    s = k_idx - (n - 1)
    idx1 = j_idx - s
    valid1 = (idx1 >= 0) & (idx1 < n)
    term1 = np.where(valid1, w[None, :] * (1.0 - h[np.clip(idx1, 0, n - 1)]), 0.0).sum(axis=1)
    idx2 = j_idx + s
    valid2 = (idx2 >= 0) & (idx2 < n)
    term2 = np.where(valid2, w[None, :] * h[np.clip(idx2, 0, n - 1)], 0.0).sum(axis=1)
    grad = term1 - term2
    return F, grad

    F, _ = smooth_obj_grad(x, alpha)
    return F


def _jac_alpha(x, alpha):
    _, g = smooth_obj_grad(x, alpha)

    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s_star = k_star - (n - 1)
    grad = np.zeros(n)

        if 0 <= j - s_star < n:
            val += (1.0 - h[j - s_star])
        if 0 <= j + s_star < n:
            val -= h[j + s_star]
        grad[j] = val
    return grad, s_star, k_star
