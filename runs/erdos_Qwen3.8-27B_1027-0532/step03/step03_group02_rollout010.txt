def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def project_with_bisection(h, target_sum):

        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid

    weights = np.exp(shifted)
    weights /= np.sum(weights)

    sum1 = conv_w_rev[(n-1):(n-1)+n]
    
    grad = 1.0 - sum1 - sum2

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    sh = alpha * (corr - cmax)
    w = np.exp(sh)
    w /= np.sum(w)

    Wr = np.fft.rfft(w_rev, size)

    cwr = np.fft.irfft(Wr * H, size)

    s1 = cwr[(n - 1):(n - 1) + n]
    return 1.0 - s1 - s2
