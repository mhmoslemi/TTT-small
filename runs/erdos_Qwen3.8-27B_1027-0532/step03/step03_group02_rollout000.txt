        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    result = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        if len(idx) > 0:
            result = result.copy()
            adj = diff / len(idx)
            result[idx] += adj
            result = np.clip(result, 0.0, 1.0)
    return result

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_exact(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()

    while time.time() - elapsed_start < max_time:
        for j in range(n):
            old_val = h[j]
            js = j - s_arr
            js_plus = j + s_arr
            A = np.zeros(2 * n - 1)
            mask1 = (js >= 0) & (js < n)
            mask2 = (js_plus >= 0) & (js_plus < n)
            if np.any(mask1):
                A[mask1] += (1.0 - h[js[mask1]])
            if np.any(mask2):
                A[mask2] -= h[js_plus[mask2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad_val = A[k_star]
                if grad_val > 1e-13:
                    hi = mid
                elif grad_val < -1e-13:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = new_val

        h = project_exact(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            freq1 = rng.randint(1, 5)
            freq2 = rng.randint(5, 15)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            noise = (np.sin(2.0 * np.pi * freq1 * x) * rng.randn() +
                     np.sin(2.0 * np.pi * freq2 * x) * rng.randn() * 0.5 +
                     rng.randn(n) * 0.4)
            scale = 0.015 * min(1.0, no_improve / patience)
            h = best_h + noise * scale
            h = project_exact(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0

    return best_h, best_score

def smooth_gradient_step(h, target_sum, alpha, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum <= 0:
        return h
    w /= w_sum

    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        s = s_arr
        m1 = (j + s >= 0) & (j + s < n)
        m2 = (j - s >= 0) & (j - s < n)
        val = 0.0
        if np.any(m1):
            val += np.dot(w[m1], 1.0 - h[j + s[m1]])
        if np.any(m2):
            val -= np.dot(w[m2], h[j - s[m2]])
        grad[j] = val
    grad *= dx

    h_new = project_exact(h - lr * grad, target_sum)
    return h_new

def gradient_descent(h, target_sum, max_time, rng):
    n = len(h)
    h = project_exact(h.copy(), target_sum)
    best_score = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    alphas = [15, 40, 100, 250, 600, 1200]
    for alpha in alphas:
        if time.time() - start > max_time:
            break
        lr = 0.03
        stagnant = 0
        prev_score = best_score
        n_iter = 0

        while time.time() - start < max_time:
            h_new = smooth_gradient_step(h, target_sum, alpha, lr)
            score = compute_c5(h_new)

            if score < prev_score - 1e-14:
                h = h_new
                prev_score = score
                lr = min(lr * 1.08, 0.15)
                stagnant = 0
                if score < best_score - 1e-12:
                    best_score = score
                    best_h = h.copy()
            else:
                lr *= 0.7
                stagnant += 1
                if stagnant > 80:
                    break
            n_iter += 1
            if n_iter > 5000:
                break

        h = best_h + rng.randn(n) * 0.003
        h = project_exact(h, target_sum)

    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_exact(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_exact(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Coordinate descent at initial resolution
    if tleft() > 80:
        h_cd, sc = coord_descend(best_h, target_init, min(150, tleft() - 80), rng, patience=35)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 2: Try higher resolutions
    higher_ns = [n_init + 200, n_init + 400, n_init + 600]
    for n_try in higher_ns:
        if tleft() < 150:
            break
        target_try = float(n_try) / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_cd, sc = coord_descend(h_interp, target_try, min(120, tleft() - 150), rng, patience=25)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 3: Gradient descent at best resolution
    n_best = len(best_h)
    target_best = float(n_best) / 2.0
    if tleft() > 120:
        h_gd, sc = gradient_descent(best_h, target_best, min(120, tleft() - 120), rng)
        if sc < best_score:
            best_score = sc
            best_h = h_gd

    # Phase 4: More coordinate descent
    n_best = len(best_h)
    target_best = float(n_best) / 2.0
    if tleft() > 80:
        h_cd, sc = coord_descend(best_h, target_best, min(120, tleft() - 80), rng, patience=40)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 5: Multiple restarts with different perturbation scales
    n_best = len(best_h)
    target_best = float(n_best) / 2.0
    perturb_scales = [0.05, 0.08, 0.12, 0.04, 0.15, 0.06]
    for scale in perturb_scales:
        if tleft() < 60:
            break
        h_restart = best_h + rng.randn(n_best) * scale
        h_restart = project_exact(h_restart, target_best)
        h_cd, sc = coord_descend(h_restart, target_best, min(50, tleft() - 60), rng, patience=15)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 6: Try a completely different starting point (sinusoidal)
    if tleft() > 80:
        n_try = max(n_init, 400)
        target_try = float(n_try) / 2.0
        x = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_sine = 0.5 + 0.35 * np.sin(np.pi * x) + 0.15 * np.sin(3.0 * np.pi * x)
        h_sine = project_exact(h_sine, target_try)
        h_cd, sc = coord_descend(h_sine, target_try, min(80, tleft() - 80), rng, patience=30)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Phase 7: Final gradient descent polish
    n_best = len(best_h)
    target_best = float(n_best) / 2.0
    if tleft() > 60:
        h_gd, sc = gradient_descent(best_h, target_best, min(60, tleft() - 60), rng)
        if sc < best_score:
            best_score = sc
            best_h = h_gd

    # Phase 8: Final coordinate descent polish
    n_best = len(best_h)
    target_best = float(n_best) / 2.0
    if tleft() > 30:
        h_cd, sc = coord_descend(best_h, target_best, min(50, tleft() - 30), rng, patience=50)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Final feasibility check
    n_out = len(best_h)
    target_out = float(n_out) / 2.0
    h_out = project_exact(best_h, target_out)
    final_score = compute_c5(h_out)
    if final_score < best_score:
        best_score = final_score

    return h_out, best_score, n_out
