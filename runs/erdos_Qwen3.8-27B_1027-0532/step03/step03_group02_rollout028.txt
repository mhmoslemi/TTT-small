```python
def lsmax_obj(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    lsmax = m + np.log(np.sum(np.exp(shifted))) / alpha
    sum_pen = lam * (np.sum(h) - n / 2.0) ** 2
    return lsmax + sum_pen

def lsmax_grad(h, alpha, lam):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = float(np.max(corr))
    shifted = alpha * (corr - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of logsumexp(α * corr) with respect to h
    # d/dh[j] logsumexp(α * corr) = α * sum_k weights[k] * d(corr[k])/d(h[j])
    # d(corr[k])/d(h[j]) = (1-h[j+s[k]]) - h[j-s[k]] (with bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-15:
            continue
        s = s_arr[k]
        w = alpha * weights[k]
        # d(corr[k])/d(h[j]) for all j:
        # corr[k] = sum_i h[i] * (1-h[i+s]) where i+s in [0,n) and i in [0,n)
        # = sum_i h[i] - sum_i h[i]*h[i+s]
        # d/dh[j]: 1 - h[j+s] (if j+s in [0,n)) - h[j-s] (if j-s in [0,n))
        js = np.arange(n) + s
        js2 = np.arange(n) - s
        m1 = (js >= 0) & (js < n)
        m2 = (js2 >= 0) & (js2 < n)
        contrib = np.zeros(n)
        contrib[m1] += (1.0 - h[js[m1]])
        contrib[m2] -= h[js2[m2]]
        grad += w * contrib
    
    # Plus gradient of penalty
    grad += 2 * lam * (np.sum(h) - n / 2.0)
    
    return grad
```

```python
def project_feasible(h, target_sum):

def project_feasible_fast(h, target_sum):
```

```python
            for _ in range(30):
                mid = (lo + hi) * 0.5
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-13:
                    hi = mid
                elif grad < -1e-13:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val

        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-12:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            strategy = rng.randint(4)
            if strategy == 0:
                noise = noise_base * min(4.0, 1.0 + 0.4 * no_improve / patience)
                h = best_h + rng.randn(n) * noise
            elif strategy == 1:
                h = best_h.copy()
                bs = rng.randint(max(5, n // 10), max(10, n // 3))
                st = rng.randint(0, n - bs)
                h[st:st + bs] = rng.rand(bs)
            elif strategy == 2:
                h = best_h.copy()
                ns = rng.randint(5, max(10, n // 15))
                for _ in range(ns):
                    i, j = rng.choice(n, size=2, replace=False)
                    h[i], h[j] = h[j], h[i]
            else:
                h = symmetrize(best_h)
                h = h + rng.randn(n) * noise_base * 0.5
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            no_improve = 0

    return best_h, best_score

def lsmax_smooth(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c = float(np.max(corr))
    shifted = alpha * (corr - c)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_polish(h, target_sum, max_time):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()
    alphas = [20, 50, 100, 200, 400, 800, 1500]
    for alpha in alphas:
        if time.time() - start > max_time - 10:
            break
        try:
            res = minimize(
                lambda x, a=alpha: lsmax_smooth(x, a),
                h_cur,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                            'jac': lambda x: np.ones(n)}],
                options={'maxiter': 250, 'ftol': 1e-15}
            )
            if res.x is not None:
                h_new = project_feasible(res.x, target_sum)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
        except Exception:
            pass
    return best_h, best_score

def lift_resolution(h_src, n_dst):
    n_src = len(h_src)
    target_dst = n_dst / 2.0
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Symmetrize initial
    h_sym_init = symmetrize(h_init)
    sc_sym = compute_c5(h_sym_init)
    if sc_sym < best_score:
        best_score = sc_sym
        best_h = h_sym_init

    # Phase 1: Coordinate descent at initial resolution
    if time_left() > 300:
        t1 = min(250, time_left() - 50)
        h_cd, sc = coord_descent(best_h, target_init, t1, rng, patience=25, noise_base=0.012)
        if sc < best_score:
            best_score = sc
            best_h = h_cd

    # Symmetrize after coordinate descent
    h_sy = symmetrize(best_h)
    sc_sy = compute_c5(h_sy)
    if sc_sy < best_score:
        best_score = sc_sy
        best_h = h_sy

    # Phase 2: SLSQP polish at initial resolution
    if time_left() > 180:
        t2 = min(130, time_left() - 40)
        h_sl, sc = slsqp_polish(best_h, target_init, t2)
        if sc < best_score:
            best_score = sc
            best_h = h_sl

    # Phase 3: Try higher resolution
    for n_try in [n_init + 120, n_init + 240]:
        if time_left() < 180:
            break
        target_try = n_try / 2.0
        h_interp = lift_resolution(best_h, n_try)
        t3 = min(150, time_left() - 60)
        h_cd2, sc2 = coord_descent(h_interp, target_try, t3, rng, patience=20, noise_base=0.008)
        sc2_sym = compute_c5(symmetrize(h_cd2))
        if sc2_sym < sc2:
            sc2 = sc2_sym
            h_cd2 = symmetrize(h_cd2)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_cd2

    # Phase 4: SLSQP at current best resolution
    n_cur = len(best_h)
    target_cur = n_cur / 2.0
    if time_left() > 80:
        t4 = min(60, time_left() - 30)
        h_sl2, sc = slsqp_polish(best_h, target_cur, t4)
        if sc < best_score:
            best_score = sc
            best_h = h_sl2

    # Phase 5: Final coordinate descent with small noise
    if time_left() > 50:
        t5 = min(35, time_left() - 15)
        h_f, sc_f = coord_descent(best_h, target_cur, t5, rng, patience=30, noise_base=0.004)
        sc_f_sym = compute_c5(symmetrize(h_f))
        if sc_f_sym < sc_f:
            sc_f = sc_f_sym
            h_f = symmetrize(h_f)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 6: Final SLSQP
    if time_left() > 25:
        t6 = max(10, time_left() - 5)
        h_p, sc_p = slsqp_polish(best_h, target_cur, t6)
        if sc_p < best_score:
            best_score = sc_p
            best_h = h_p

    # Final symmetrization and feasibility
    n_final = len(best_h)
    h_final = symmetrize(best_h)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final
    else:
        h_final = project_feasible(best_h, n_final / 2.0)
        final_score = compute_c5(h_final)
        if final_score < best_score:
            best_score = final_score

    return h_final, best_score, n_final
```
