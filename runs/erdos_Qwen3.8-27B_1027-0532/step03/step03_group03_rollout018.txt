import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

_N_G = 0
_T_G = 0.0

def smooth_obj(x, alpha):
    corr = overlap_corrs(x)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / len(x))

def smooth_grad(x, alpha):
    corr = overlap_corrs(x)
    cmax = np.max(corr)
    exp_s = np.exp(np.clip(alpha * (corr - cmax), -500.0, 0.0))
    total = np.sum(exp_s)
    w = exp_s / total
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    omh = 1.0 - x
    for p in range(2 * n - 1):
        wi = w[p]
        if wi < 1e-13:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * omh[:n - lag]
                grad[:n - lag] -= wi * x[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += wi * omh[m:]
                grad[m:] -= wi * x[:n - m]
    return grad * dx

    return np.sum(x) - _T_G

def eq_jac(x):
    return np.ones(_N_G)

def slsqp_refine(h0, target, alphas, maxiter, ftol):
    global _N_G, _T_G
    _N_G = len(h0)
    _T_G = target
    h = h0.copy()
    bounds = [(0.0, 1.0)] * _N_G
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    cur_best = compute_c5(h)
    for a in alphas:
        try:
            res = minimize(smooth_obj, h, args=(a,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_best:
                h = h_new
                cur_best = sc
        except Exception:
            pass
    return h

                noise_p, noise_s, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best = compute_c5(h)
    lr_cur = lr_init
    stag = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        w = np.exp(alpha_ns * (top_vals - top_vals.max()))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / ws
        grad = np.zeros(n)
        omh = 1.0 - h
        for i in range(topk):
            wi = w[i]
            if wi < 1e-14:
                continue
            p = top_idx[i]
            lag = p - (n - 1)
            if lag >= 0:
                if lag < n:

                    grad[:n - lag] -= wi * h[lag:]
            else:

                if m < n:
                    grad[:n - m] += wi * omh[m:]
                    grad[m:] -= wi * h[:n - m]
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr_cur * v

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

_N_G = 0
_T_G = 0.0

def smooth_obj(x, alpha):
    corr = overlap_corrs(x)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / len(x))

def smooth_grad(x, alpha):
    corr = overlap_corrs(x)
    cmax = np.max(corr)
    exp_s = np.exp(np.clip(alpha * (corr - cmax), -500.0, 0.0))
    total = np.sum(exp_s)
    w = exp_s / total
    n = len(x)
    dx = 2.0 / n
    grad = np.zeros(n)
    omh = 1.0 - x
    for p in range(2 * n - 1):
        wi = w[p]
        if wi < 1e-13:
            continue
        lag = p - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * omh[:n - lag]
                grad[:n - lag] -= wi * x[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += wi * omh[m:]
                grad[m:] -= wi * x[:n - m]
    return grad * dx

def eq_fun(x):
    return np.sum(x) - _T_G

def eq_jac(x):
    return np.ones(_N_G)

def slsqp_refine(h0, target, alphas, maxiter, ftol):
    global _N_G, _T_G
    _N_G = len(h0)
    _T_G = target
    h = h0.copy()
    bounds = [(0.0, 1.0)] * _N_G
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    cur_best = compute_c5(h)
    for a in alphas:
        try:
            res = minimize(smooth_obj, h, args=(a,), jac=smooth_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur_best:
                h = h_new
                cur_best = sc
        except Exception:
            pass
    return h

def subgrad_opt(h_init, target, max_iter, lr_init, topk, alpha_ns,
                noise_p, noise_s, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best = compute_c5(h)
    lr_cur = lr_init
    stag = 0
    for it in range(max_iter):
        corr = overlap_corrs(h)
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        w = np.exp(alpha_ns * (top_vals - top_vals.max()))
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / ws
        grad = np.zeros(n)
        omh = 1.0 - h
        for i in range(topk):

            if wi < 1e-14:

            p = top_idx[i]
            lag = p - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[lag:] += wi * omh[:n - lag]
                    grad[:n - lag] -= wi * h[lag:]
            else:
                m = -lag
                if m < n:
                    grad[:n - m] += wi * omh[m:]
                    grad[m:] -= wi * h[:n - m]
        v = momentum * v + (1.0 - momentum) * grad

def interp_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_interp = interp_to(best_h, n_try, tt)
        h_new, score = subgrad_opt(h_interp, tt, 2500, 0.06, 5, 30,
                                    0.03, 0.008, 250, 0.90, seed + 5000 + n_try)
        if score < best_c5:
            best_c5 = score
            best_h = h_new
        if remaining() > 60:
            h_ref = slsqp_refine(h_new, tt, [50, 150, 400, 800], 600, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_ref
        if remaining() > 40:
            h_bp, sc = binary_push(best_h, tt, seed=seed + 6000 + n_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_bp
        if remaining() > 40:
            h_ls, sc = local_search(best_h, tt, max_sweeps=100, init_step=0.008, seed=seed + 7000 + n_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_ls

    nu = len(best_h)
    tu = nu / 2.0
    if remaining() > 30:
        h_ls, sc = local_search(best_h, tu, max_sweeps=200, init_step=0.004, seed=seed + 9000)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ls

    if remaining() > 30:
        h_ref = slsqp_refine(best_h, tu, [500, 1500, 3000], 500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ref

    if remaining() > 25:
        h_ls, sc = local_search(best_h, tu, max_sweeps=150, init_step=0.003, seed=seed + 9500)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_ls

    h_final = project_feasible(best_h, tu)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
