    n = len(h)

    best_h = h.copy()

        t = step / max(1, n_steps - 1)

        h_y = h - momentum * v
        
        # Project to feasible region
        h_y = project_feasible(h_y, target_sum)
        
        # Compute correlations
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')
        
        # Find top-k lags
        top_indices = np.argsort(corr)[-top_k:]
        
        # Compute gradient of average top-k correlations
        grad = np.zeros(n)
        for k in top_indices:
            lag = k - (n - 1)
            # For each j, the contribution to grad[j] from this lag:
            # If j+lag is valid: (1 - h_y[j+lag])
            # If j-lag is valid: -h_y[j-lag]
            if lag == 0:
                # corr = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
                # d/dh[j] = 1 - 2*h[j]
                grad[j] += (1.0 - 2.0 * h_y[j])

                if lag > 0:

                    valid2 = slice(0, n - lag_abs)
                    grad[valid1] += (1.0 - h_y[0:n-lag_abs])
                    grad[valid2] += (-h_y[lag_abs:n])

        h_new = h_y - v
        h_new = project_feasible(h_new, target_sum)

            best_score = score
            best_h = h_new.copy()

import time

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):

            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def topk_obj_and_grad(h, alpha, k_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-k_top:]
    top_vals = corr[top_idx]
    m = float(np.max(top_vals))
    exp_vals = np.exp(alpha * (top_vals - m))
    sum_exp = float(np.sum(exp_vals))
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for ii in range(k_top):
        k_val = int(top_idx[ii])
        lag = k_val - (n - 1)
        wt = w[ii]
        if lag == 0:
            grad += wt * (1.0 - 2.0 * h)

            grad[:n - lag] += wt * (1.0 - h[lag:])

        else:
            la = -lag
            grad[la:] += wt * (1.0 - h[:-la])
            grad[:n - la] += wt * (-h[la:])
    return F, grad


def lbfgs_optimize(h_start, target, alpha, k_top, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        f, _ = topk_obj_and_grad(x, alpha, k_top)
        return f

    def jac_func(x):
        _, g = topk_obj_and_grad(x, alpha, k_top)
        return g

    try:
        res = minimize(obj_func, h_start, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol,
                                               'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def nesterov_subgrad(h_start, target, n_steps, top_k, lr_init, lr_end, momentum):

    h = project_feasible(h_start.copy(), target)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    for step in range(n_steps):
        t = step / max(1, n_steps - 1)
        lr = lr_init * (lr_end / lr_init) ** t
        h_y = h - momentum * v
        h_y = project_feasible(h_y, target)
        corr = np.correlate(h_y, 1.0 - h_y, mode='full')

            lo = max(0, edge - width)
            hi = min(n_init, edge + width + 1)
            h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
            h_pert = project_feasible(h_pert, target_init)

        h_new, sc_new = full_refine_pipeline(h_pert, target_init, tleft)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    # Phase 4: Grid refinement to higher resolution
    for n_try in [600, 800, 1000]:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        x_old = np.linspace(0.0, 1.0, n_init)
        x_new = np.linspace(0.0, 1.0, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_feasible(h_interp, target_try)

        # Quick refinement on higher grid
        if tleft() > 40:
            h_ns, sc_ns = nesterov_subgrad(h_interp, target_try, 2000, 15, 0.003, 0.0003, 0.85)
            if sc_ns < compute_c5(h_interp):
                h_interp = h_ns
        h_new, sc_new = full_refine_pipeline(h_interp, target_try, tleft)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new
            n_init = n_try
            target_init = target_try

    # Phase 5: Final aggressive polish
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 30:
        h_new = lbfgs_optimize(best_h, target_use, 5000, 3, 600, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if tleft() > 20:
        h_new = lbfgs_optimize(best_h, target_use, 10000, 3, 400, 1e-16)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 6: Micro perturbations at the end
    for _ in range(5):
        if tleft() < 15:
            break
        amp = rng.uniform(0.0005, 0.002)
        h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
        h_new, sc_new = full_refine_pipeline(h_pert, target_use, tleft)
        if sc_new < best_score:
            best_score = sc_new
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
