        threshold = c_max * 0.999
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:
            active = np.argpartition(corr, -5)[-5:]

        w = np.ones(len(active)) / len(active)
        grad = compute_grad_weighted(best_h, active, w)
        h_new = best_h - lr * grad
        h_new = project_feasible(h_new, target_sum)

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
            lr = min(lr * 1.01, 1.0)
        else:
            lr *= 0.9
    return best_h, best_score

def compute_c5(h):

    dx = 2.0 / n

    return float(np.max(corr)) * dx

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad
