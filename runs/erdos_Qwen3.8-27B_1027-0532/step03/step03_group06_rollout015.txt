import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(60):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
    return h


def exact_subgradient(h, p_star):
    n = len(h)
    k = p_star - (n - 1)
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += (1.0 - h[:n - k])
        grad[:n - k] -= h[k:]
    else:
        k2 = -k
        grad[:n - k2] += (1.0 - h[k2:])
        grad[k2:] -= h[:n - k2]
    return grad


def subgradient_descent(h_start, target_sum, n_iters, step_init, step_decay, rng):
    n = len(h_start)
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    step = step_init
    for it in range(n_iters):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        p_star = int(np.argmax(corr))
        grad = exact_subgradient(h_cur, p_star)
        h_try = project_feasible(h_cur - step * grad, target_sum)
        sc = compute_c5(h_try)
        if sc < best_score:
            best_score = sc
            best_h = h_try.copy()
            h_cur = h_try
        else:
            h_cur = project_feasible(0.5 * h_cur + 0.5 * h_try, target_sum)
            sc2 = compute_c5(h_cur)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_cur.copy()
        step *= step_decay
    return best_h, best_score


def softmax_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2 * n - 2, i + n - 1)
        if klo_c <= khi_c:
            j_idx_c = i - np.arange(klo_c, khi_c + 1) + (n - 1)
            term_c = np.sum(w[klo_c:khi_c + 1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad


def lbfgs_refine(h_start, target_sum, alpha, maxiter, ftol):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_func(x):
        F, _ = softmax_obj_and_grad(x, alpha)
        return F

    def jac_func(x):
        _, g = softmax_obj_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj_func, h_start, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target_sum)
        return h_new
    except Exception:
        return h_start


def try_binary_thresholds(h, n, target_sum):
    flat = h.ravel()
    order = np.argsort(flat)[::-1]
    best_score = float('inf')
    best_h = None
    n_ones = int(target_sum)
    if n_ones <= 0 or n_ones >= n:
        return None, float('inf')
    for margin in range(0, min(5, n // 4)):
        idx = order[:n_ones + margin]
        h_bin = np.zeros(n)
        h_bin[idx[:n_ones]] = 1.0
        sc = compute_c5(h_bin)
        if sc < best_score:
            best_score = sc
            best_h = h_bin.copy()
        if margin > 0:
            idx2 = order[n_ones - margin:n_ones]

            h_bin2[order[:n_ones - margin]] = 1.0
            h_bin2[idx2] = 1.0
            sc2 = compute_c5(h_bin2)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_bin2.copy()
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coordinate_local_search(h, target_sum, n_trials, rng):
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    for _ in range(n_trials):
        idx = rng.randint(n)
        delta = rng.uniform(0.01, 0.15) * rng.choice([-1, 1])
        h_try = best_h.copy()
        h_try[idx] = np.clip(h_try[idx] + delta, 0.0, 1.0)
        h_try = project_feasible(h_try, target_sum)
        sc = compute_c5(h_try)
        if sc < best_score - 1e-12:
            best_score = sc
            best_h = h_try
    return best_h, best_score


def multi_start_refine(h_start, target_sum, alpha_schedule, maxiter, ftol, time_fn, min_time_left):
    n = len(h_start)
    h_cur = project_feasible(h_start.copy(), target_sum)
    current_score = compute_c5(h_cur)
    for alpha in alpha_schedule:
        if time_fn() < min_time_left:
            break
        h_new = lbfgs_refine(h_cur, target_sum, alpha, maxiter, ftol)
        sc = compute_c5(h_new)
        if sc + 1e-12 < current_score:
            h_cur = h_new
            current_score = sc
    return h_cur, current_score
