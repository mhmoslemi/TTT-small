def coord_descent_step(h, j, target_sum, h_excluded):
    # h_excluded: sum of all h[i] for i != j (should equal target_sum - h[j])
    # Compute c[k] as function of t = h[j]
    # c[k] = base[k] + t * coef[k]
    
    n = len(h)
    # Set h[j] = 0
    h0 = h.copy()
    h0[j] = 0.0
    c0 = np.correlate(h0, 1.0 - h0, mode='full')
    
    # Set h[j] = 1
    h1 = h.copy()
    h1[j] = 1.0
    c1 = np.correlate(h1, 1.0 - h1, mode='full')
    
    base = c0
    coef = c1 - c0
    
    # Now find t in [0,1] that minimizes max_k (base[k] + t*coef[k])
    # Walk along the upper envelope starting from t=0
    best_t = 0.0
    best_val = np.max(base)
    
    # The upper envelope at t=0 is argmax of base
    # Walk forward: at each step, find which line becomes the new max
    active = np.argmax(base)
    t_curr = 0.0
    
    # We want to find where the envelope stops decreasing
    # i.e., where coef of the active line becomes >= 0
    
    for _ in range(2 * n):  # iterate at most 2n times
        active_val = base[active] + t_curr * coef[active]
        if active_val < best_val:
            best_val = active_val
            best_t = t_curr
        
        if coef[active] >= -1e-15:
            break
        
        # Find the next line that catches up
        # base[i] + t*coef[i] = base[active] + t_curr*coef[active] + (t - t_curr)*coef[active]
        # base[i] + t*coef[i] = base[active] + t*coef[active]
        # t*(coef[i] - coef[active]) = base[active] - base[i]
        # t = (base[active] - base[i]) / (coef[i] - coef[active])
        
        diff_coef = coef - coef[active]
        diff_base = base[active] - base
        # We need t > t_curr and diff_coef[i] > 0 (line i is steeper or same)
        valid = (diff_coef > 1e-15) & (base + t_curr * coef > base[active] + t_curr * coef[active] - 1e-10)
        # Actually, we need lines that are below the active line at t_curr but will cross it
        # At t_curr: base[i] + t_curr*coef[i] < base[active] + t_curr*coef[active]
        # Crossing: t = (base[active] - base[i]) / (coef[i] - coef[active])
        # We need this t > t_curr
        
        below = (base + t_curr * coef) < (base[active] + t_curr * coef[active])
        steeper = (coef > coef[active])
        valid = below & steeper & (diff_coef > 1e-15)
        
        if not np.any(valid):
            # No line will catch up, go to t=1
            t_next = 1.0
        else:
            t_cross = diff_base[valid] / diff_coef[valid]
            t_cross = t_cross[t_cross > t_curr]
            if len(t_cross) == 0:
                t_next = 1.0
            else:
                t_next = np.min(t_cross)
        
        if t_next >= 1.0:
            t_curr = 1.0
            # Update best
            val_at_1 = np.max(base + coef)
            if val_at_1 < best_val:
                best_val = val_at_1
                best_t = 1.0
            break
        
        t_curr = t_next
        # Find new active (the line that achieved the max at t_curr)
        active = np.argmax(base + t_curr * coef)
    
    # Also check t=1
    val_at_1 = np.max(base + coef)
    if val_at_1 < best_val:
        best_val = val_at_1
        best_t = 1.0
    
    return best_t, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute evenly among coordinates that can absorb the change
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h = h.copy()
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def coord_descent(h_init, target_sum, max_sweeps=200, lambda_init=0.0,
                  lambda_lr=0.1, seed=42):
    """Coordinate descent with Lagrange multiplier for sum constraint."""
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    lam = lambda_init
    
    for sweep in range(max_sweeps):
        # Shuffle coordinates for stochastic ordering
        order = rng.permutation(n)
        
        for j in order:
            # Compute c[k] as linear function of t = h[j]
            h0 = h.copy()
            h0[j] = 0.0
            c0 = np.correlate(h0, 1.0 - h0, mode='full')
            
            h1 = h.copy()
            h1[j] = 1.0
            c1 = np.correlate(h1, 1.0 - h1, mode='full')
            
            base = c0
            coef = c1 - c0 + lam  # Lagrange multiplier adds to coef
            
            # Find t in [0,1] minimizing max_k (base[k] + t*coef[k])
            t_opt = _minimize_max_linear(base, coef)
            
            h[j] = t_opt
        
        # Update Lagrange multiplier
        sum_err = np.sum(h) - target_sum
        lam = lam - lambda_lr * sum_err
        lam = np.clip(lam, -10.0, 10.0)
        
        # Periodic projection to maintain feasibility
        if sweep % 10 == 0:
            h = project_feasible(h, target_sum)
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    # Final projection
    h = project_feasible(h, target_sum)
    final_score = compute_c5(h)
    if final_score < best_score:
        best_score = final_score
        best_h = h
    
    return best_h, best_score

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    n = len(h)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 0.999999
    else:
        mask = h > 0.000001
    if not np.any(mask):
        return h
    h = h.copy()
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def min_max_linear(base, coef):
    """Find t in [0,1] minimizing max_k (base[k] + t * coef[k])."""
    n_lines = len(base)
    t = 0.0
    vals = base.copy()
    active = int(np.argmax(vals))
    best_t = 0.0
    best_val = float(vals[active])
    
    for _ in range(n_lines + 1):
        if coef[active] >= -1e-12:
            break
        diff_c = coef - coef[active]
        diff_b = base[active] - base

            t_cross = t_cross[(t_cross > t + 1e-10) & (t_cross <= 1.0 + 1e-10)]
            if len(t_cross) > 0:
                t = float(np.min(t_cross))
            else:
                t = 1.0
        else:
            t = 1.0

        active = int(np.argmax(vals))

            best_t = t
    
    vals_1 = base + coef
    v1 = float(np.max(vals_1))

        best_t = 1.0

def coord_descent(h_init, target_sum, max_sweeps=200, lambda_lr=0.5, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    n = len(h)

        if sweep < 5:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        
        for j in order:
            h[j] = 0.0
            c0 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            h[j] = 0.0  # reset

        best_h = h

    hj_orig = h[j]
    hm_orig = h[m]
    
    # Set h[j]=0, h[m]=S
    h[j] = 0.0
    h[m] = S
    c0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Set h[j]=1, h[m]=S-1
    h[j] = 1.0
    h[m] = S - 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    
    # Restore
    h[j] = hj_orig
    h[m] = hm_orig
    
    base = c0
    coef = c1 - c0
    
    # Range of t: max(0, S-1) <= t <= min(1, S)
    t_min = max(0.0, S - 1.0)
    t_max = min(1.0, S)
    
    t_opt = min_max_linear_range(base, coef, t_min, t_max)
    
    h[j] = t_opt
    h[m] = S - t_opt

```python
def min_max_linear_range(base, coef, t_min, t_max):
    """Find t in [t_min, t_max] minimizing max_k (base[k] + t * coef[k])."""
    n_lines = len(base)
    t = t_min
    vals = base + t * coef
    active = int(np.argmax(vals))
    best_t = t
    best_val = float(vals[active])
    
    for _ in range(n_lines + 1):
        if coef[active] >= -1e-12 or t >= t_max:
            break
        diff_c = coef - coef[active]
        diff_b = base[active] - base
        valid = (diff_c > 1e-12)
        if np.any(valid):
            t_cross = diff_b[valid] / diff_c[valid]
            t_cross = t_cross[(t_cross > t + 1e-10) & (t_cross <= t_max + 1e-10)]
            if len(t_cross) > 0:
                t = float(np.min(t_cross))
            else:
                t = t_max
        else:
            t = t_max
        
        if t > t_max:
            t = t_max
        
        vals = base + t * coef
        active = int(np.argmax(vals))
        if vals[active] < best_val:
            best_val = float(vals[active])
            best_t = t
    
    # Check t_max
    vals_end = base + t_max * coef
    v_end = float(np.max(vals_end))
    if v_end < best_val:
        best_val = v_end
        best_t = t_max
    
    return best_t
```

def paired_coord_descent(h_init, target_sum, max_sweeps=200, seed=42):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_score = compute_c5(h)
    
    # Create pairs
    pairs = [(i, i + 1) for i in range(0, n - 1, 2)]
    # If n is odd, the last element can be paired with the first
    if n % 2 == 1:
        pairs.append((n - 1, 0))
    
    for sweep in range(max_sweeps):
        if sweep < 5:
            rng.shuffle(pairs)
        
        for (j, m) in pairs:
            S = h[j] + h[m]
            hj_orig = h[j]
            hm_orig = h[m]
            
            h[j] = 0.0
            h[m] = S
            c0 = np.correlate(h, 1.0 - h, mode='full')
            
            h[j] = 1.0
            h[m] = S - 1.0
            c1 = np.correlate(h, 1.0 - h, mode='full')
            
            h[j] = hj_orig
            h[m] = hm_orig
            
            base = c0
            coef = c1 - c0
            
            t_min = max(0.0, S - 1.0)
            t_max = min(1.0, S)
            
            t_opt = min_max_linear_range(base, coef, t_min, t_max)
            h[j] = t_opt
            h[m] = S - t_opt
        
        score = compute_c5(h)
        if score < best_score:
            best_score = score
            best_h = h.copy()
    
    return best_h, best_score

Actually, I realize I can speed this up significantly by noting that:
c1[k] - c0[k] only depends on the change at positions j and m.

When I change h[j] from 0 to 1 and h[m] from S to S-1:
- The change in c[k] is:
  Δc[k] = (contribution of h[j]=1 instead of 0) + (contribution of h[m]=S-1 instead of S)

The contribution of h[j] = t to c[k] is:
- In Σ h[i+k]: contributes t when i+k = j, i.e., i = j-k
- In -Σ h[i+k]*h[i]: contributes -t*h[j-k] when i+k=j, and -h[j+k]*t when i=j

So the coefficient of t in c[k] (holding other h fixed) is:
B_j[k] = 1_{j-k valid} - h[j-k]*1_{j-k valid} - h[j+k]*1_{j+k valid}
= (1 - h[j-k])*1_{j-k valid} - h[j+k]*1_{j+k valid}

This can be computed in O(1) per k (just look up h[j-k] and h[j+k]).

So I can compute base and coef in O(n) time (for the 2n-1 correlation values) instead of O(n²) (for the full correlate).

Wait, but I still need to compute c0 (the baseline correlation) which requires the full h array. Unless I maintain a running correlation and update it incrementally.

Here's the key optimization: **maintain the correlation array incrementally**.

When I change h[j] from old_val to new_val (and h[m] from old_val_m to new_val_m), the change in c[k] is:
Δc[k] = (new_val - old_val) * B_j[k] + (new_val_m - old_val_m) * B_m[k]

where B_j[k] is the "sensitivity" of c[k] to h[j] (holding other h fixed).

If I maintain c (the correlation array) and B_j[k] for the current j and m, I can update in O(n) time.

But computing B_j[k] also requires knowing h, which changes as I go. So I'd need to recompute B_j[k] each time... which is O(n) per coordinate.

Let me also double-check the formula for B_j[k]:

c[k] = Σ_i h[i+k] * (1-h[i])

∂c[k]/∂h[j] = Σ_i [∂h[i+k]/∂h[j]] * (1-h[i]) + h[i+k] * ∂(1-h[i])/∂h[j]
= (1-h[j-k]) * 1_{j-k valid} + h[j+k] * (-1) * 1_{j+k valid}
= (1-h[j-k]) * 1_{j-k valid} - h[j+k] * 1_{j+k valid}

Yes, that's correct.

But wait, this is the derivative. Since c[k] is bilinear in h (product of two h's), the change when I increase h[j] by δ is NOT exactly δ * ∂c[k]/∂h[j] unless I'm at the correct point.

Actually, c[k] = Σ_i h[i+k] * (1-h[i]) is bilinear in h (linear in each argument when the other is fixed). So:
c[k](h + δe_j) - c[k](h) = δ * [(1-h[j-k]) * 1_{j-k valid} - h[j+k] * 1_{j+k valid}]

This is EXACT (not just a derivative) because the function is linear in h[j] (when other components are fixed).

So if I have the current c and I want to change h[j] by δ:
c_new[k] = c[k] + δ * B_j[k]

where B_j[k] = (1-h[j-k])*1_{j-k valid} - h[j+k]*1_{j+k valid} (computed using the CURRENT h, before the update).

This is exact! No approximation needed.

So the algorithm is:
1. Maintain c (the correlation array)
2. For each pair (j, m):
   a. Compute B_j[k] and B_m[k] using current h
   b. The change in c when we set h[j]=t, h[m]=S-t:
      Δc[k] = (t - h[j]) * B_j[k] + ((S-t) - h[m]) * B_m[k]
             = (t - h[j]) * B_j[k] + (h[j] - t) * B_m[k]
             = (t - h[j]) * (B_j[k] - B_m[k])
      
      Wait: h[m] goes from h[m] to S-t = h[j]+h[m]-t. The change is (S-t) - h[m] = h[j]-t.
      And h[j] goes from h[j] to t. The change is t - h[j].
      
      Δc[k] = (t - h[j]) * B_j[k] + (h[j] - t) * B_m[k]
             = (t - h[j]) * (B_j[k] - B_m[k])
   
   c. So c_new[k] = c[k] + (t - h[j]) * (B_j[k] - B_m[k])
      = (c[k] - h[j]*(B_j[k] - B_m[k])) + t * (B_j[k] - B_m[k])
      
      Let base[k] = c[k] - h[j]*(B_j[k] - B_m[k])
      Let coef[k] = B_j[k] - B_m[k]
      
      Then c_new[k] = base[k] + t * coef[k]
   
   d. Find t minimizing max_k c_new[k] subject to t_min ≤ t ≤ t_max
   e. Update: h[j] = t, h[m] = S-t, c[k] = base[k] + t*coef[k]

This is O(n) per pair, and very clean!

def compute_B(h, j):

def compute_B_fast(h, j):
    n = len(h)
    B = np.zeros(2 * n - 1)
    
    # First term: (1-h[j-k]) for valid j-k
    k_start1 = max(0, j)
    k_end1 = min(2 * n - 1, j + n)
    if k_start1 < k_end1:
        i_min = j - (k_end1 - 1) + (n - 1)
        i_max = j - k_start1 + (n - 1)
        B[k_start1:k_end1] += (1.0 - h[i_min:i_max + 1])[::-1]
    
    # Second term: -h[j+k] for valid j+k
    k_start2 = max(0, n - 1 - j)
    k_end2 = min(2 * n - 1, 2 * n - 1 - j)
    if k_start2 < k_end2:
        i2_start = j + k_start2 - (n - 1)
        i2_end = j + (k_end2 - 1) - (n - 1)
        B[k_start2:k_end2] -= h[i2_start:i2_end + 1]
    
    return B
