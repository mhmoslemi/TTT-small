import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, alpha, target):
    """Log-sum-exp objective and gradient for the Erdős problem."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (max_c + np.log(sum_exp) / alpha) * dx
    
    if alpha < 1e-6:
        return obj, np.zeros(n)
    
    w = exp_shifted / sum_exp
    S1 = np.zeros(n)
    S2 = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-15:
            continue
        lag = m - n + 1
        if lag >= 0:
            S1[lag:] += wm * x[:n-lag]
            S2[:n-lag] += wm * x[lag:]
        else:
            S1[:n+lag] += wm * x[-lag:]
            S2[-lag:] += wm * x[:n+lag]
    grad = dx * (1.0 - S1 - S2)
    return obj, grad

def lbfgs_phase(h, target, time_limit, alpha_start=10.0, alpha_end=200.0, n_alphas=8):
    """Run L-BFGS-B with increasing alpha (annealing)."""
    n = len(h)
    start_time = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    alphas = np.linspace(alpha_start, alpha_end, n_alphas)
    bounds = [(0.0, 1.0)] * n
    
    for ai, alpha in enumerate(alphas):
        if time.time() - start_time >= time_limit:
            break
        elapsed = time.time() - start_time
        remaining = time_limit - elapsed
        time_per_phase = remaining / (n_alphas - ai)
        maxiter = int(max(50, time_per_phase * 5))
        
        def obj_and_grad(x, a=alpha):
            o, g = lse_obj_grad(x, a, target)
            return o, g
        
        try:
            res = minimize(obj_and_grad, best, method='L-BFGS-B',
                          bounds=bounds, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                                   'maxls': 50})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score

def pairwise_swap(h, target, time_limit, seed=42):
    """Try pairwise swaps to improve the solution."""
    n = len(h)
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    # Precompute which pairs to try
    # Focus on pairs that are far apart (different lags)
    no_improve = 0
    step_size = 0.05
    
    while time.time() - start_time < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Try moving mass from i to j or vice versa
        # h[i] += d, h[j] -= d (or vice versa)
        for d in [step_size, -step_size, step_size*0.3, -step_size*0.3]:
            val_i = best[i] + d
            val_j = best[j] - d
            if val_i < 0 or val_i > 1 or val_j < 0 or val_j > 1:
                continue
            trial = best.copy()
            trial[i] = val_i
            trial[j] = val_j
            s = compute_c5(trial)
            if s < best_score - 1e-14:
                best_score = s
                best = trial
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 200:
                step_size *= 0.7
                no_improve = 0
                if step_size < 1e-5:
                    break
    
    return best, best_score

def lse_obj_grad_fast(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (max_c + np.log(sum_exp) / alpha) * dx
    
    w = exp_shifted / sum_exp
    # S1 = conv(x, w)[n-1:2n-1]
    # S2 = conv(x, w[::-1])[n-1:2n-1]
    # Use FFT for speed
    N = 2 * (2 * n - 1)  # FFT size
    X = np.fft.rfft(x, N)
    W = np.fft.rfft(w, N)
    WR = np.fft.rfft(w[::-1], N)
    S1 = np.fft.irfft(X * W, N)[:2*n-1]
    S2 = np.fft.irfft(X * WR, N)[:2*n-1]
    S1 = S1[n-1:2*n-1]
    S2 = S2[n-1:2*n-1]
    grad = dx * (1.0 - S1 - S2)
    return obj, grad

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def lse_obj_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    obj = (max_c + np.log(sum_exp) / alpha) * dx
    
    w = exp_shifted / sum_exp
    # FFT-based gradient
    N = 1
    while N < 3 * n - 2:
        N *= 2
    X = np.fft.rfft(x, N)
    W = np.fft.rfft(w, N)
    WR = np.fft.rfft(w[::-1], N)
    conv1 = np.fft.irfft(X * W, N)
    conv2 = np.fft.irfft(X * WR, N)
    S1 = conv1[n-1 : 2*n-1]
    S2 = conv2[n-1 : 2*n-1]
    grad = dx * (1.0 - S1 - S2)
    return obj, grad

def lbfgs_phase(h, target, time_limit, alpha_start=5.0, alpha_end=200.0, n_alphas=10, maxiter_per=200):
    n = len(h)
    start_time = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    bounds = [(0.0, 1.0)] * n
    
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start_time >= time_limit:
            break
        remaining = time_limit - (time.time() - start_time)
        time_per = remaining / (n_alphas - ai)
        maxiter = int(max(50, min(maxiter_per, time_per * 10)))
        
        def obj_grad(x, a=alpha):
            return lse_obj_grad(x, a)
        
        try:
            res = minimize(obj_grad, best, method='L-BFGS-B',
                          bounds=bounds, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-12,
                                   'maxls': 30, 'maxfun': maxiter * 5})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score

def frank_wolfe_phase(h, target, time_limit, alpha=50.0):
    """Frank-Wolfe (conditional gradient) with backtracking."""
    n = len(h)
    start_time = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    k = int(target)
    iter_count = 0
    
    while time.time() - start_time < time_limit:
        o, g = lse_obj_grad(best, alpha)
        # Linear minimization oracle
        threshold_idx = np.argpartition(g, k)[:k]
        v = np.zeros(n)
        v[threshold_idx] = 1.0
        
        # Backtracking line search
        best_t = 2.0 / (2.0 * iter_count + 2.0)
        best_obj = o
        for _ in range(20):
            h_trial = (1.0 - best_t) * best + best_t * v
            h_trial = np.clip(h_trial, 0.0, 1.0)
            # Check sum constraint
            s_sum = np.sum(h_trial)
            if abs(s_sum - target) > 1e-6:
                h_trial = project_feasible(h_trial, target)
            obj_trial = compute_c5(h_trial)
            if obj_trial <= o - 0.01 * best_t * np.dot(g, best - v) + 1e-15:
                break
            best_t *= 0.5
        
        best = (1.0 - best_t) * best + best_t * v
        best = project_feasible(best, target)
        s = compute_c5(best)
        if s < best_score:
            best_score = s
        iter_count += 1
    
    return best, best_score

def pairwise_swap(h, target, time_limit, seed=42):
    n = len(h)
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    no_improve = 0
    step = 0.05
    
    while time.time() - start_time < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in [step, -step, step*0.3, -step*0.3, step*0.1, -step*0.1]:
            vi = best[i] + d
            vj = best[j] - d
            if vi < 0 or vi > 1 or vj < 0 or vj > 1:
                continue
            trial = best.copy()
            trial[i] = vi
            trial[j] = vj
            s = compute_c5(trial)
            if s < best_score - 1e-14:
                best_score = s
                best = trial
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.7
                no_improve = 0
                if step < 1e-6:
                    break
    
    return best, best_score

def fine_coord_descent(h, target, time_limit, seed=42):
    n = len(h)
    start_time = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    deltas = [0.001, -0.001, 0.003, -0.003, 0.005, -0.005]
    
    while time.time() - start_time < time_limit:
        improved = False
        for i in range(n):
            if time.time() - start_time >= time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0 or val > 1:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = project_feasible(trial, target)
                s = compute_c5(trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = trial
                    improved = True
                    break
        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if max(abs(d) for d in deltas) < 1e-6:
                break
    
    return best, best_score

def perturb_and_refine(h, target, time_limit, seed=42, n_perturb=3):
    """Add random perturbations and re-optimize."""
    n = len(h)
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    
    for p in range(n_perturb):
        if time.time() - start_time >= time_limit:
            break
        # Add perturbation
        noise = rng.randn(n) * 0.02
        h_pert = project_feasible(best + noise, target)
        
        # Quick L-BFGS refinement
        time_per = (time_limit - (time.time() - start_time)) / (n_perturb - p)
        h_ref, s_ref = lbfgs_phase(h_pert, target, time_limit=time_per,
                                    alpha_start=50.0, alpha_end=300.0, n_alphas=5,
                                    maxiter_per=100)
        if s_ref < best_score:
            best_score = s_ref
            best = h_ref
    
    return best, best_score
