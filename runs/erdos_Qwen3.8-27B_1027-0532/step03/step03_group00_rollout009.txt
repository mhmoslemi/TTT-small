def compute_gradient(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    # softmax
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # For each position m, gradient = sum_s p[s_idx] * [(1-h[m-s]) - h[m+s]]
    # where s = s_idx - (n-1)
    shifts = np.arange(-(n-1), n)
    
    grad = np.zeros(n)
    for s_idx, s in enumerate(shifts):
        if p[s_idx] < 1e-15:
            continue
        # First term: (1-h[m-s]) for valid m
        # m-s in [0, n-1] => m in [s, s+n-1] intersect [0, n-1]
        # Second term: -h[m+s] for valid m  
        # m+s in [0, n-1] => m in [-s, n-1-s] intersect [0, n-1]
        
        # First term
        m_start1 = max(0, s)
        m_end1 = min(n, s + n)  # exclusive
        if m_start1 < m_end1:
            # m ranges from m_start1 to m_end1-1
            # h[m-s] ranges from h[0] to h[m_end1-1-s]
            idx = np.arange(m_start1, m_end1) - s
            grad[m_start1:m_end1] += p[s_idx] * (1.0 - h[idx])
        
        # Second term
        m_start2 = max(0, -s)
        m_end2 = min(n, n - s)  # exclusive: m < n-s means m+s < n
        if m_start2 < m_end2:
            idx = np.arange(m_start2, m_end2) + s
            grad[m_start2:m_end2] -= p[s_idx] * h[idx]
    
    return grad

def project_feasible(h, target_sum):

        h[mask] += diff / cnt

def lse_grad(h, alpha, shifts, n):
