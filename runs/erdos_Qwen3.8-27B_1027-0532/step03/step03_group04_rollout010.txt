        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r
            n = n3
            target = target3

    # Phase 6: Momentum subgradient at n=800
    if tleft() > 60:
        h_sg, c5_sg = momentum_subgradient(best_h, n, target, 5000, min(50, tleft() * 0.5), 0.01, 0.92, 5)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 7: SLSQP at n=800 with very high alpha
    if tleft() > 80:
        h_r, c5_r = slsqp_optimize(best_h, n, target, [500, 2000, 8000], 1200, 1e-15)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r

    # Phase 8: Alternating LP
    if tleft() > 150:
        h_lp, c5_lp = alternating_lp(best_h, n, target, 3000, min(120, tleft() * 0.5), 0.7, 1e-4)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 9: SLSQP after LP
    if tleft() > 60:
        h_r, c5_r = slsqp_optimize(best_h, n, target, [300, 1000, 4000, 10000], 1000, 1e-15)
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_r

    # Phase 10: Final subgradient
    if tleft() > 30:
        h_sg, c5_sg = momentum_subgradient(best_h, n, target, 3000, min(30, tleft() * 0.6), 0.005, 0.95, 7)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
