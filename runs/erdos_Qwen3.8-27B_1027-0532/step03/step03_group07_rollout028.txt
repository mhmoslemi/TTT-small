def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project(y, S):
    # Euclidean projection onto {x: sum(x) = S, 0 <= x <= 1}
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix rounding
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12: break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12: break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_at_k(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

def _topk_grad(h, k, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_at_k(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] -= h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] -= h[:m]
    return grad * dx

def _topk_grad(h, k, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    weights = np.exp(100.0 * (corr - max_corr))
    total_w = weights.sum()
    if total_w < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total_w
    top_idx = np.argsort(weights)[::-1][:k]
    w = weights[top_idx].copy()
    w = w / w.sum()
    g = np.zeros(len(h))
    for idx in range(len(top_idx)):
        g += w[idx] * _grad_at_k(h, top_idx[idx], dx)
    return g

def _nesterov_refined(h0, n, target_sum, n_iters, deadline, k_start=5, k_end=1):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        t_frac = it / max(n_iters - 1, 1)
        k = int(round(k_start + (k_end - k_start) * t_frac))
        k = max(1, min(k, 8))
        g = _topk_grad(y, k, dx)
        alpha = 1.0 / (0.3 * np.sqrt(float(it + 1)))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_refine(h0, n, target_sum, alpha, maxiter):
    dx = 2.0 / n

    def _obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * dx
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    def _obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        weights = np.exp(alpha * (corr - m))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if weights[k_idx] > 1e-8:
                g += weights[k_idx] * _grad_at_k(h, k_idx, dx)
        return g

    def _con_fun(h):
        return h.sum() - target_sum

    def _con_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h0, jac=_obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _project(res.x, target_sum)
    except Exception:
        return h0

    for rnd in range(n_rounds):

            # Fine grid around current value plus key values

                candidates.append(min(1.0, old + dv))
            for v in [0.0, 0.1, 0.2, 0.25, 1.0/3.0, 0.4, 0.5, 0.6, 2.0/3.0, 0.75, 0.8, 0.9, 1.0]:
                candidates.append(v)
            best_v = old
            best_c_local = best_c
            seen_vals = set()
            for v in candidates:
                vq = round(v, 6)
                if vq in seen_vals:
                    continue
                seen_vals.add(vq)
                if not (0.0 <= vq <= 1.0):
                    continue
                h[j] = vq
                c = _c5_of(h)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_v = vq
            h[j] = best_v
            if best_v != old:
                improved = True
                best_c = best_c_local
        h = _project(h, target_sum)

        if c5_r < best_c5:

            best_h = h_ref.copy()
