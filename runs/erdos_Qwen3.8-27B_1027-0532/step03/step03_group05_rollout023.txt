```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    # Center lag (m=0)
    w0 = weights[n-1]
    if abs(w0) > 1e-18:
        grad += w0 * (1.0 - 2.0 * h)
    # Off-center lags
    for m in range(1, n):
        k_pos = n - 1 + m
        k_neg = n - 1 - m
        w_pos = weights[k_pos]
        w_neg = weights[k_neg]
        if abs(w_pos) > 1e-18:
            grad[:n-m] += w_pos * (1.0 - h[m:])
            grad[m:] += -w_pos * h[:n-m]
        if abs(w_neg) > 1e-18:
            grad[m:] += w_neg * (1.0 - h[:n-m])
            grad[:n-m] += -w_neg * h[m:]
    return grad

def softmax_weights(corr, alpha):
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    if s < 1e-300:
        return np.full(len(corr), 1.0/len(corr))
    return exps / s

def lbfgsb_phase(h, target, alpha, time_limit):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    def obj(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def grad(x):
        corr = compute_corr(x)
        y = softmax_weights(corr, alpha)
        return dx * exact_grad_weighted(x, y)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    
    elapsed = time.time() - start
    maxiter = max(50, int((time_limit - elapsed) * 2))
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxls': 50})
        h_new = project_feasible(res.x, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new
    except Exception:
        pass
    
    return best, best_score

def momentum_phase(h, target, time_limit, alpha=100.0, lr=0.01, momentum=0.9):
    n = len(h)
    dx = 2.0 / n
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    v = np.zeros(n)
    
    # Compute initial weights
    corr = compute_corr(h)
    y = softmax_weights(corr, alpha)
    
    step = 0
    while time.time() - start < time_limit:
        grad = dx * exact_grad_weighted(h, y)
        v = momentum * v - lr * grad
        h_new = project_feasible(h + v, target)
        score = compute_c5(h_new)
        
        if score < best_score:
            best_score = score
            best = h_new.copy()
        
        h = h_new
        step += 1
        
        # Update weights periodically
        if step % 20 == 0:
            corr = compute_corr(h)
            y = softmax_weights(corr, alpha)
            # Decay learning rate
            lr *= 0.998
            if lr < 1e-5:
                break
    
    return best, best_score

def coordinate_descent(h, target, time_limit, deltas=None, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    if deltas is None:
        deltas = [0.005, -0.005, 0.01, -0.01, 0.002, -0.002, 
                  0.02, -0.02, 0.001, -0.001]
    best = h.copy()
    best_score = compute_c5(best)
    
    no_improve = 0
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in deltas:
                val = best[i] + d
                if val < -1e-10 or val > 1.0 + 1e-10:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = project_feasible(trial, target)
                s = compute_c5(trial)
                if s < best_score - 1e-15:
                    best_score = s
                    best = trial
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
            deltas = [d * 0.5 for d in deltas]
        else:
            no_improve = 0
    
    return best, best_score

def round_to_binary(h, target):
    """Round h to {0,1} while maintaining sum = target."""
    n = len(h)
    # Sort by value, set top target/1 ones
    # (target should be n/2 for our problem)
    num_ones = int(round(target))
    thresh = np.sort(h)[::-1][num_ones - 1] if 0 < num_ones < n else 0.5
    h_bin = (h >= thresh).astype(float)
    # Fix if sum is off
    while np.sum(h_bin) < num_ones:
        # Find the 0 closest to 1
        zeros = h_bin == 0
        if not np.any(zeros):
            break
        vals = h[zeros]
        idx = np.where(zeros)[0][np.argmax(vals)]
        h_bin[idx] = 1
    while np.sum(h_bin) > num_ones:
        ones = h_bin == 1
        if not np.any(ones):
            break
        vals = h[ones]
        idx = np.where(ones)[0][np.argmin(vals)]
        h_bin[idx] = 0
    return h_bin
```

One more optimization: instead of calling np.correlate each time in coordinate descent (which is expensive), I can precompute the correlation and update it incrementally when a single coordinate changes.

When h[i] changes by delta:
- corr[k] changes for all k where i is in the valid range
- For m = k - (n-1):
  - If m >= 0: i < n-m, so m < n-i, i.e., m in [0, n-i-1]
  - If m < 0: i >= -m, so -m <= i, i.e., m in [-(i), -(i)-1]... 
    
Actually, for m >= 0: corr[k] = sum_{j=0}^{n-m-1} h[j]*(1-h[j+m])
If h[i] changes by delta:
- h[i] appears as h[j] when j=i (if i < n-m): change = delta*(1-h[i+m])
- h[i] appears as h[j+m] when j=i-m (if i-m >= 0 and i-m < n-m, i.e., m <= i < n): change = -h[i-m]*delta

So for m in [0, min(i, n-1-i)]:
  corr_change = delta * [(1-h[i+m]) - h[i-m]]
For m in [min(i, n-1-i)+1, n-1-i] (only if i < n-1-i):
  corr_change = delta * (1-h[i+m])
For m in [min(i, n-1-i)+1, i] (only if i >= n-1-i):
  corr_change = -delta * h[i-m]

This is O(n) to compute the change for all lags, which is the same as recomputing the full correlation. So it doesn't save much.

But we can use it to quickly check if the max changes without recomputing the full correlation. If the max is at lag k*, and the change at k* doesn't make it exceed the second max, then we know the new max without a full recomputation.

This is a useful optimization for coordinate descent! Let me implement it.

```python
    n = len(h)
    xv = h[i] + delta
    dh = delta
    
    # Compute changes to all lags at once
    # For lag m (m from -(n-1) to n-1):
    # change[m] = ...
    
    m_vals = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For each m, the change is:
    # If m == 0: dh * (1 - xv - h[i])
    # If m > 0: dh * (1-h[i+m]) * (i < n-m) + (-dh) * h[i-m] * (i >= m)
    # If m < 0 (p=-m): dh * (1-h[i-p]) * (i >= p) + (-dh) * h[i+p] * (i < n-p)
    
    changes = np.zeros(2*n-1)
    
    # m = 0
    changes[n-1] = dh * (1.0 - xv - h[i])
    
    # m > 0
    pos_m = np.arange(1, n)  # m = 1, 2, ..., n-1
    # i+m < n → m < n-i
    valid_plus = pos_m < (n - i)
    # i-m >= 0 → m <= i
    valid_minus = pos_m <= i
    
    c_plus = np.zeros(n-1)
    c_plus[valid_plus] = dh * (1.0 - h[i + pos_m[valid_plus]])
    c_minus = np.zeros(n-1)
    c_minus[valid_minus] = -dh * h[i - pos_m[valid_minus]]
    changes[n:n-1+n-1] = c_plus + c_minus  # indices n to 2n-2
    
    # m < 0 (p = -m, p = 1, ..., n-1)
    # i-p >= 0 → p <= i
    valid_plus2 = pos_m <= i
    # i+p < n → p < n-i
    valid_minus2 = pos_m < (n - i)
    
    c_plus2 = np.zeros(n-1)
    c_plus2[valid_plus2] = dh * (1.0 - h[i - pos_m[valid_plus2]])
    c_minus2 = np.zeros(n-1)
    c_minus2[valid_minus2] = -dh * h[i + pos_m[valid_minus2]]
    changes[0:n-1] = c_plus2 + c_minus2  # indices 0 to n-2
    
    new_max = np.max(corr + changes)
    return new_max * (2.0 / n)
```
