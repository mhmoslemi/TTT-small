import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)

    n = 2 * len(f)
    h = np.empty(n)
    h[:len(f)] = f
    h[len(f):] = 1.0 - f[::-1]
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def build_h(f):
    m = len(f)
    h = np.empty(2 * m)
    h[:m] = f
    h[m:] = 1.0 - f[::-1]
    return h

def project_anti(h_in):
    n = len(h_in)
    half = n // 2
    f = np.zeros(half)
    for i in range(half):
        f[i] = (h_in[i] - h_in[n - 1 - i] + 1.0) / 2.0
    return np.clip(f, 0.0, 1.0)

def reduced_cd(f_init, max_time, rng, patience=20, noise_base=0.003):
    m = len(f_init)
    n = 2 * m
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    
    f = np.clip(f_init.astype(float).copy(), 0.0, 1.0)
    h = build_h(f)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_f = f.copy()
    no_improve = 0
    t0 = time.time()
    
    while time.time() - t0 < max_time:
        # One full sweep
        for j in range(m):
            k = s_arr
            js = j - k
            jp = j + k
            j2 = n - 1 - j
            js2 = j2 - k
            jp2 = j2 + k
            
            m1 = (js >= 0) & (js < n)
            m2 = (jp >= 0) & (jp < n)
            m1b = (js2 >= 0) & (js2 < n)
            m2b = (jp2 >= 0) & (jp2 < n)
            
            A = np.zeros(2 * n - 1)
            A[m1] += (1.0 - h[js[m1]])
            A[m2] -= h[jp[m2]]
            A[m1b] -= (1.0 - h[js2[m1b]])
            A[m2b] += h[jp2[m2b]]
            
            B = corr - A * f[j]
            
            lo, hi = 0.0, 1.0
            for _ in range(25):
                mid = (lo + hi) / 2.0
                idx = np.argmax(A * mid + B)
                g = A[idx]
                if g > 1e-12:
                    hi = mid
                elif g < -1e-12:
                    lo = mid
                else:
                    break
            
            nv = (lo + hi) * 0.5
            d = nv - f[j]
            if abs(d) > 1e-15:
                corr += d * A
                f[j] = nv
                h[j] = nv
                h[n - 1 - j] = 1.0 - nv
        
        # Recompute to avoid drift
        f = np.clip(f, 0.0, 1.0)
        h = build_h(f)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
        score = float(np.max(corr)) * dx
        
        if score < best_score - 1e-14:
            best_score = score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            noise = noise_base * min(1.0, (no_improve - patience) / max(1, patience))
            f = best_f + rng.randn(m) * noise
            f = np.clip(f, 0.0, 1.0)
            h = build_h(f)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
            no_improve = 0
    
    return best_f, best_score

def smooth_obj(f, alpha):
    h = build_h(f)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_grad(f, alpha):
    m = len(f)
    n = 2 * m
    h = build_h(f)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    exp_v = np.exp(shifted)
    denom = float(np.sum(exp_v))
    
    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(m)
    
    for j in range(m):
        k = s_arr
        js = j - k
        jp = j + k
        j2 = n - 1 - j
        js2 = j2 - k
        jp2 = j2 + k
        
        m1 = (js >= 0) & (js < n)
        m2 = (jp >= 0) & (jp < n)
        m1b = (js2 >= 0) & (js2 < n)
        m2b = (jp2 >= 0) & (jp2 < n)
        
        dC = np.zeros(2 * n - 1)
        dC[m1] += (1.0 - h[js[m1]])
        dC[m2] -= h[jp[m2]]
        dC[m1b] -= (1.0 - h[js2[m1b]])
        dC[m2b] += h[jp2[m2b]]
        
        grad[j] = float(np.dot(dC, exp_v)) / denom
    
    return grad

def lbfgsb_refine(f_init, max_time, alphas):
    m = len(f_init)
    bounds = [(0.0, 1.0)] * m
    best_f = np.clip(f_init.copy(), 0.0, 1.0)
    best_score = compute_c5(build_h(best_f))
    t0 = time.time()
    
    for alpha in alphas:
        if time.time() - t0 > max_time:
            break
        try:
            res = minimize(
                lambda x: smooth_obj(x, alpha),
                best_f,
                jac=lambda x: smooth_grad(x, alpha),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-10, 'maxfun': 5000}
            )
            f_new = np.clip(res.x, 0.0, 1.0)
            sc = compute_c5(build_h(f_new))
            if sc < best_score:
                best_score = sc
                best_f = f_new
        except Exception:
            pass
    
    return best_f, best_score

def interpolate_f(f_src, n_dst):
    """Interpolate anti-symmetric free variables to new resolution"""
    m_src = len(f_src)
    n_src = 2 * m_src
    n_dst_total = n_dst
    m_dst = n_dst_total // 2
    
    if m_src == m_dst:
        return f_src.copy()
    
    h_src = build_h(f_src)
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, n_dst_total)
    h_dst = np.interp(x_dst, x_src, h_src)
    
    # Project to anti-symmetric
    f_dst = np.zeros(m_dst)
    for i in range(m_dst):
        f_dst[i] = (h_dst[i] - h_dst[n_dst_total - 1 - i] + 1.0) / 2.0
    
    return np.clip(f_dst, 0.0, 1.0)

import numpy as np
from scipy.optimize import minimize
import time

_g_alpha = 1000.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def build_h(f):
    m = len(f)
    h = np.empty(2 * m)
    h[:m] = f
    h[m:] = 1.0 - f[::-1]
    return h

def project_anti(h_in):
    n = len(h_in)
    half = n // 2
    f = np.zeros(half)
    for i in range(half):
        f[i] = (h_in[i] - h_in[n - 1 - i] + 1.0) / 2.0
    return np.clip(f, 0.0, 1.0)

def reduced_cd(f_init, max_time, rng, patience=25, noise_base=0.003):
    m = len(f_init)
    n = 2 * m
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    f = np.clip(f_init.astype(float).copy(), 0.0, 1.0)
    h = build_h(f)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_score = float(np.max(corr)) * dx
    best_f = f.copy()
    no_improve = 0
    t0 = time.time()
    while time.time() - t0 < max_time:
        for j in range(m):
            k = s_arr
            js = j - k
            jp = j + k
            j2 = n - 1 - j
            js2 = j2 - k
            jp2 = j2 + k
            m1 = (js >= 0) & (js < n)
            m2 = (jp >= 0) & (jp < n)
            m1b = (js2 >= 0) & (js2 < n)
            m2b = (jp2 >= 0) & (jp2 < n)
            A = np.zeros(2 * n - 1)
            A[m1] += (1.0 - h[js[m1]])
            A[m2] -= h[jp[m2]]
            A[m1b] -= (1.0 - h[js2[m1b]])
            A[m2b] += h[jp2[m2b]]
            B = corr - A * f[j]
            lo = 0.0
            hi = 1.0

                    lo = mid
                else:

                corr += d * A
                f[j] = nv
                h[j] = nv
                h[n - 1 - j] = 1.0 - nv

            h = build_h(f)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

def smooth_obj(x, alpha):
    h = build_h(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha

def smooth_grad(x, alpha):
    m = len(x)
    n = 2 * m
    h = build_h(x)
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = float(np.max(corr))
    shifted = np.clip(alpha * (corr - cmax), -500.0, 500.0)
    exp_v = np.exp(shifted)
    denom = float(np.sum(exp_v))
    s_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(m)
    for j in range(m):
        k = s_arr
        js = j - k
        jp = j + k
        j2 = n - 1 - j
        js2 = j2 - k
        jp2 = j2 + k
        m1 = (js >= 0) & (js < n)
        m2 = (jp >= 0) & (jp < n)
        m1b = (js2 >= 0) & (js2 < n)
        m2b = (jp2 >= 0) & (jp2 < n)
        dC = np.zeros(2 * n - 1)
        dC[m1] += (1.0 - h[js[m1]])
        dC[m2] -= h[jp[m2]]
        dC[m1b] -= (1.0 - h[js2[m1b]])
        dC[m2b] += h[jp2[m2b]]
        grad[j] = float(np.dot(dC, exp_v)) / denom
    return grad

    return smooth_obj(x, _g_alpha)

def _smo_grad(x):
    return smooth_grad(x, _g_alpha)

    best_f = np.clip(f_init.copy(), 0.0, 1.0)
