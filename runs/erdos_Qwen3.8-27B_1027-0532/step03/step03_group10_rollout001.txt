from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

                current_max = new_max

                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
        if not improved:
            break
    return best_h, (2.0/n) * best_max

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def projected_gradient_min(h_init, eps, target_sum, n_iters, lr0):
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    f_prev = smooth_objective(h, eps)
    for it in range(n_iters):
        g = smooth_gradient(h, eps)
        h_new = h - lr * g
        h_new = project_onto_constraint(h_new, target_sum)
        f_new = smooth_objective(h_new, eps)
        if f_new < f_prev:
            h = h_new
            f_prev = f_new
            lr = min(lr * 1.05, 0.5)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def fast_binary_sa(h_start, n_steps, t_start, t_end, rng, target_worst=True):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    n_one = len(ones_list)
    n_zero = len(zeros_list)
    half = n - 1

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        use_structured = target_worst and rng.random() < 0.4
        if use_structured:
            worst_k = int(np.argmax(corr))
            worst_s = worst_k - half
            valid_i = []
            valid_j = []
            for x in ones_list:
                if worst_s != 0 and 0 <= x + worst_s < n and 0 <= x - worst_s < n:
                    valid_i.append(x)
            for x in zeros_list:
                if worst_s != 0 and 0 <= x + worst_s < n and 0 <= x - worst_s < n:
                    valid_j.append(x)
            if len(valid_i) > 0 and len(valid_j) > 0:
                i = valid_i[rng.integers(len(valid_i))]
                j = valid_j[rng.integers(len(valid_j))]
            else:
                i = ones_list[rng.integers(n_one)]
                j = zeros_list[rng.integers(n_zero)]
        else:
            i = ones_list[rng.integers(n_one)]
            j = zeros_list[rng.integers(n_zero)]
        if i == j:
            step += 1
            continue
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            step += 1
            continue
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
        corr[k_range] += delta
        new_max = float(np.max(corr))
        delta_c5 = new_max - current_max
        if delta_c5 < -1e-12 or (t > 1e-10 and rng.random() < np.exp(-delta_c5 / max(t, 1e-10))):
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1
    return best_h, best_c5_val

def greedy_local_search(h_bin, max_rounds, rng):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()

    for r in range(max_rounds):
        improved = False
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if h[i] != 1.0:
                continue
            best_j = -1
            best_new_max = current_max
            best_delta = None
            best_kr = None
            for j in zeros:
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                sr = np.arange(s_min, s_max + 1)
                kr = sr + half
                d = h[i + sr] + h[i - sr] - h[j + sr] - h[j - sr]
                new_affected = corr[kr] + d
                nm_affected = float(np.max(new_affected))
                if kr[0] > 0:
                    nm_before = float(np.max(corr[:kr[0]]))
                else:
                    nm_before = -1e9
                if kr[-1] < len(corr) - 1:
                    nm_after = float(np.max(corr[kr[-1]+1:]))
                else:
                    nm_after = -1e9
                new_max = max(nm_affected, nm_before, nm_after)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_j = j
                    best_delta = d
                    best_kr = kr
            if best_j >= 0 and best_new_max < current_max - 1e-12:
                corr[best_kr] += best_delta
                h[i] = 0.0
                h[best_j] = 1.0
                ones.remove(i)
                ones.append(best_j)
                zeros.remove(best_j)
                zeros.append(i)
                current_max = best_new_max
                improved = True
                break
        if not improved:
            break
    return h, current_max
