    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_onto_feasible(h, target):
    h = h.astype(float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(result < 1.0 - 1e-15)[0]
        else:
            idx = np.where(result > 1e-15)[0]
        if len(idx) > 0:
            result[idx] += diff / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result


def subgradient_max(h, top_k=12):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_val = np.max(corr)
    threshold = max_val - 1e-8
    active = np.where(corr >= threshold)[0]
    if len(active) > top_k:
        sorted_idx = np.argsort(-corr[active])
        active = active[sorted_idx[:top_k]]
    grad = np.zeros(n)
    for k in active:
        d_k = k - (n - 1)
        i_lo = max(0, -d_k)
        i_hi = min(n - 1, n - 1 - d_k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += g[i_lo + d_k:i_hi + d_k + 1]
        i_lo2 = max(0, d_k)
        i_hi2 = min(n - 1, n - 1 + d_k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= h[i_lo2 - d_k:i_hi2 - d_k + 1]
    grad /= float(len(active))
    return grad, max_val


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_val = np.max(corr)
    exp_vals = np.exp(alpha * (corr - max_val))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = max_val + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    threshold = 1e-12
    significant = np.where(w > threshold)[0]
    for k in significant:
        d_k = k - (n - 1)
        w_k = w[k]
        i_lo = max(0, -d_k)
        i_hi = min(n - 1, n - 1 - d_k)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += w_k * g[i_lo + d_k:i_hi + d_k + 1]
        i_lo2 = max(0, d_k)
        i_hi2 = min(n - 1, n - 1 + d_k)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= w_k * h[i_lo2 - d_k:i_hi2 - d_k + 1]
    return F, grad


def subgradient_phase(h_start, target, n_it, tleft_fn, best_score_tracker):
    n = len(h_start)
    h_cur = h_start.copy()
    v = np.zeros(n)
    momentum = 0.9
    step = 0.4
    current_score = compute_c5(h_cur)
    for it in range(n_it):
        if tleft_fn() < 30:
            break
        if it % 200 == 0 and tleft_fn() < 15:
            break
        grad, val = subgradient_max(h_cur, top_k=12)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            step_polyak = max(0.005, min(1.0, (best_score_tracker[0] - val + 0.001) / (grad_norm ** 2 + 1e-10)))
            step = 0.6 * step + 0.4 * step_polyak
        v = momentum * v - step * grad
        h_new = project_onto_feasible(h_cur + v, target)
        score_new = compute_c5(h_new)
        if score_new > val + 0.02:
            h_new = project_onto_feasible(h_cur - 0.3 * step * grad, target)
            score_new = compute_c5(h_new)
        if score_new < best_score_tracker[0]:
            best_score_tracker[0] = score_new
            best_score_tracker[1] = h_new.copy()
        h_cur = h_new
        step *= 0.998
        momentum = 0.9 + 0.05 * (it / max(1, n_it))
    return h_cur


def lbfgs_polish(h_start, target, alpha, maxiter, tleft_fn):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj_f(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F

    def jac_f(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj_f, h_start, jac=jac_f, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_new = project_onto_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def simulated_annealing(h_start, target, n_iter, t0, t1, tleft_fn, best_score_tracker):
    n = len(h_start)
    h_cur = h_start.copy()
    current_score = compute_c5(h_cur)
    for it in range(n_iter):
        if tleft_fn() < 25:
            break
        if it % 100 == 0 and tleft_fn() < 10:
            break
        T = t0 * (t1 / t0) ** (it / max(1, n_iter - 1))
        amp = 0.02 + 0.03 * T
        h_prop = project_onto_feasible(h_cur + rng_sa.randn(n) * amp, target)
        score_prop = compute_c5(h_prop)
        delta = score_prop - current_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-8)):
            h_cur = h_prop
            current_score = score_prop
        if current_score < best_score_tracker[0]:
            best_score_tracker[0] = current_score
            best_score_tracker[1] = h_cur.copy()
    return h_cur


rng_sa = np.random.RandomState(123)


def run(seed=42, budget_s=1300, **kwargs):
    global rng_sa
    rng_sa = np.random.RandomState(seed + 7)
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = project_onto_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)
    best_tracker = [best_score, best_h.copy()]

    # Phase 1: Subgradient descent from initial
    h_cur = subgradient_phase(best_h, target, 1500, tleft, best_tracker)
    best_h = best_tracker[1].copy()
    best_score = best_tracker[0]

    # Phase 2: L-BFGS-B with increasing alpha
    for alpha in [15, 40, 100, 300, 800, 2000]:
        if tleft() < 40:
            break
        h_new = lbfgs_polish(best_h, target, alpha, 250, tleft)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Simulated annealing explorations
    for trial in range(5):
        if tleft() < 50:
            break
        h_pert = project_onto_feasible(best_h + rng.randn(n) * 0.03, target)
        h_sa = simulated_annealing(h_pert, target, 800, 0.5, 0.01, tleft, best_tracker)
        sc = compute_c5(h_sa)
        if sc < best_score:
            best_score = sc
            best_h = h_sa

    # Phase 4: More L-BFGS-B refinement
    for alpha in [500, 1500, 4000]:
        if tleft() < 40:
            break
        h_new = lbfgs_polish(best_h, target, alpha, 300, tleft)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Multi-start with diverse perturbations + subgradient
    for attempt in range(20):
        if tleft() < 40:
            break
        ptype = attempt % 5
        if ptype == 0:
            amp = 0.005 * (1.0 + 0.2 * attempt)
            h_pert = project_onto_feasible(best_h + rng.randn(n) * amp, target)
        elif ptype == 1:
            h_pert = best_h.copy()
            seg = rng.randint(5, max(6, n // 7))
            a = rng.randint(0, n - seg)
            b = rng.randint(0, n - seg)
            if abs(a - b) > seg:
                tmp = h_pert[a:a + seg].copy()
                h_pert[a:a + seg] = h_pert[b:b + seg]
                h_pert[b:b + seg] = tmp
            h_pert = project_onto_feasible(h_pert, target)
        elif ptype == 2:
            h_pert = best_h.copy()
            edge = rng.randint(0, n)
            w = rng.randint(4, max(5, n // 10))
            lo = max(0, edge - w)
            hi = min(n, edge + w + 1)
            h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
            h_pert = project_onto_feasible(h_pert, target)
        elif ptype == 3:
            s = rng.randint(1, max(2, n // 6))
            h_pert = np.roll(best_h, s)
            h_pert = project_onto_feasible(h_pert, target)
        else:
            h_smooth = gaussian_filter1d(best_h, sigma=2.0)
            h_pert = project_onto_feasible(h_smooth + rng.randn(n) * 0.015, target)

        h_trial = subgradient_phase(h_pert, target, 400, tleft, best_tracker)
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            best_h = h_trial

    # Phase 6: Refine to higher resolution
    for n_hires in [600, 800]:
        if tleft() < 100:
            break
        target_hires = n_hires / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, n_hires)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_onto_feasible(h_interp, target_hires)

        h_high = subgradient_phase(h_interp, target_hires, 800, tleft, best_tracker)
        sc = compute_c5(h_high)
        if sc < best_score:
            best_score = sc
            best_h = h_high

        for alpha in [200, 1000, 3000]:
            if tleft() < 50:
                break
            h_new = lbfgs_polish(best_h, target_hires, alpha, 200, tleft)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Final micro-refinement
    n_final = len(best_h)
    target_final = n_final / 2.0
    for alpha in [3000, 8000, 15000]:
        if tleft() < 30:
            break
        h_new = lbfgs_polish(best_h, target_final, alpha, 400, tleft)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 8: Final micro-perturbation search
    for _ in range(12):
        if tleft() < 20:
            break
        amp = rng.uniform(0.001, 0.004)
        h_pert = project_onto_feasible(best_h + rng.randn(n_final) * amp, target_final)
        h_trial = subgradient_phase(h_pert, target_final, 200, tleft, best_tracker)
        sc = compute_c5(h_trial)
        if sc < best_score:
            best_score = sc
            best_h = h_trial

    h_final = project_onto_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
