from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_value(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def soft_max_weights(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    return ex / np.sum(ex)

def compute_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    w = soft_max_weights(corr, eps)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)

        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] -= wk * dx * h[:m]
    return grad, corr

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):

def dc_lp_solve(h_current, dx, target_sum):
    """Solve the LP for one DC iteration."""
    n = len(h_current)
    g = 1.0 - h_current
    n_shfits = 2 * n - 1
    # Build constraint matrix
    # For shift s (0 to 2n-2, corresponding to actual shift s - (n-1)):
    # corr_s = dx * sum_j h[j] * g[j-s]
    
    A_rows = []
    for m in range(2 * n - 1):
        s = m - (n - 1)
        # corr_m = dx * sum_j h[j] * g[j-s] where 0 <= j-s < n and 0 <= j < n
        row = np.zeros(n)
        if s >= 0:
            # j >= s and j < n, and j-s >= 0 → j in [s, n)
            for j in range(s, n):
                row[j] = g[j - s]
        else:
            # j < n and j-s >= 0 → j in [0, n+s) where n+s = n-|s|
            r = -s
            for j in range(0, n - r):
                row[j] = g[j + r]
        A_rows.append(row)
    
    A = np.array(A_rows) * dx
    # Add -1 column for t
    n_constraints = len(A_rows)
    A_full = np.column_stack([A, -np.ones(n_constraints)])
    b_full = np.zeros(n_constraints)
    
    # Equality: sum(h) = target_sum
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Bounds: 0 <= h_i <= 1, t >= 0 (or t >= 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_full, b_ub=b_full, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n]
    else:
        return h_current

import numpy as np
from scipy.optimize import minimize, linprog
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def soft_max_value(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def soft_max_weights(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    return ex / np.sum(ex)

def dc_iteration(h_current):
    """One DC iteration: fix g=1-h, solve LP."""
    n = len(h_current)
    dx = 2.0 / n
    g = 1.0 - h_current
    n_shifts = 2 * n - 1
    target_sum = n / 2.0
    
    # Build constraint rows: for each shift m (0 to 2n-2)
    # corr_m = dx * sum_j h[j] * g[j-s] where s = m-(n-1)
    A = np.zeros((n_shifts, n))
    for m in range(n_shifts):
        s = m - (n - 1)
        if s >= 0:
            # j in [s, n): A[m, j] = g[j-s]
            A[m, s:] = g[:n-s]
        else:
            # j in [0, n+s): A[m, j] = g[j+r] where r=-s
            r = -s
            A[m, :n-r] = g[r:]
    
    A = A * dx
    # Full A_ub: [A | -1]
    A_full = np.zeros((n_shifts, n + 1))
    A_full[:, :n] = A
    A_full[:, n] = -1.0
    b_full = np.zeros(n_shifts)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_full, b_ub=b_full, 
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return h_current, compute_c5(h_current)

def dc_algorithm(h_init, max_iter=50, tol=1e-10):
    """Run DC algorithm."""
    h = h_init.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(max_iter):
        h_new, t_val = dc_iteration(h)
        if not np.all(np.isfinite(h_new)):
            break
        diff = np.max(np.abs(h_new - h))
        h = h_new
        c5 = compute_c5(h)
        if c5 < best_val:
            best_val = c5
            best_h = h.copy()
        if diff < tol:
            break
    return best_h, best_val

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def obj_fn(h):
        corr = compute_corr(h)
        return soft_max_value(corr, eps)
    
    def grad_fn(h):
        corr = compute_corr(h)
        w = soft_max_weights(corr, eps)
        n_h = len(h)
        grad = np.zeros(n_h)
        dx = 2.0 / n_h
        for k in range(2 * n_h - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n_h - 1)
            if s >= 0:
                m = n_h - s
                grad[s:] += wk * dx * (1.0 - h[:m])
                grad[:m] -= wk * dx * h[s:]
            else:
                r = -s
                m = n_h - r
                grad[:m] += wk * dx * (1.0 - h[r:])
                grad[r:] -= wk * dx * h[:m]
        return grad
    
    def constr_fun(h):
        return float(h.sum() - target_sum)
    
    def constr_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constr_fun, 'jac': constr_jac}]
    try:
        result = minimize(obj_fn, h0, jac=grad_fn, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    n = len(h)
    order = np.argsort(np.abs(h - 0.5))
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def binary_local_search(h_init, max_rounds, time_limit, start_time):
    n = len(h_init)
    h = h_init.copy()
    current = compute_c5(h)
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        improved = False
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                if sc < current - 1e-15:
                    current = sc
                    improved = True
                    break
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
            if improved:
                break
        if not improved:
            break
    return h, current

def sa_binary(h_init, n_steps, t_start, t_end, time_limit, start_time, rng):
    n = len(h_init)
    h = h_init.copy()
    current = compute_c5(h)
    best_h = h.copy()
    best_c5 = current
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        r = rng.random()
        if r < 0.7:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
        else:
            # Block swap
            blen = int(rng.integers(2, max(3, n // 8) + 1))
            a = int(rng.integers(0, n - blen + 1))
            b = int(rng.integers(0, n - blen + 1))
            if abs(a - b) < blen:
                step += 1
                continue
            # Swap blocks
            block_a = h[a:a+blen].copy()
            block_b = h[b:b+blen].copy()
            h[a:a+blen] = block_b
            h[b:b+blen] = block_a
            new_score = compute_c5(h)
            delta = new_score - current
            if delta < 0 or (t > 1e-9 and rng.random() < np.exp(-delta / t)):
                current = new_score
                if new_score < best_c5:
                    best_c5 = new_score
                    best_h = h.copy()
            else:
                h[a:a+blen] = block_a
                h[b:b+blen] = block_b
            step += 1
            continue
        h[oi] = 0.0
        h[zi] = 1.0
        new_score = compute_c5(h)
        delta = new_score - current
        if delta < 0 or (t > 1e-9 and rng.random() < np.exp(-delta / t)):
            current = new_score
            if new_score < best_c5:
                best_c5 = new_score
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
import time


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def soft_max_value(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))


def soft_max_weights(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    return ex / np.sum(ex)


def soft_obj(h, eps):
    corr = compute_corr(h)
    return soft_max_value(corr, eps)


def soft_grad(h, eps):
    n_h = len(h)
    dx = 2.0 / n_h
    corr = compute_corr(h)
    w = soft_max_weights(corr, eps)
    grad = np.zeros(n_h)
    for k in range(2 * n_h - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n_h - 1)
        if s >= 0:
            m = n_h - s
            grad[s:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n_h - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] -= wk * dx * h[:m]
    return grad


def cons_fun(h):
    return float(h.sum() - len(h) / 2.0)


def cons_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        result = minimize(soft_obj, h0, args=(eps,), jac=soft_grad,
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def dc_iteration(h_current):
    n = len(h_current)
    dx = 2.0 / n
    g = 1.0 - h_current
    n_shifts = 2 * n - 1
    target_sum = n / 2.0
    A = np.zeros((n_shifts, n))
    for m in range(n_shifts):
        s = m - (n - 1)
        if s >= 0:
            A[m, s:] = g[:n - s]
        else:
            r = -s
            A[m, :n - r] = g[r:]
    A = A * dx
    A_full = np.zeros((n_shifts, n + 1))
    A_full[:, :n] = A
    A_full[:, n] = -1.0
    b_full = np.zeros(n_shifts)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]
    result = linprog(c_obj, A_ub=A_full, b_ub=b_full,
                     A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    if result.success:
        return result.x[:n], float(result.x[n])
    return h_current, compute_c5(h_current)


def dc_algorithm(h_init, max_iter=50, tol=1e-10, time_limit=None, start_time=None):
    h = h_init.copy()

    best_val = compute_c5(h)
    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        h_new, t_val = dc_iteration(h)
        if not np.all(np.isfinite(h_new)):
            break
        diff = float(np.max(np.abs(h_new - h)))
        h = h_new
        c5_this = compute_c5(h)

            best_h = h.copy()
        if diff < tol:

        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    n_final = len(best_h)
    final_h = project_onto_constraint(best_h, n_final / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, n_final
