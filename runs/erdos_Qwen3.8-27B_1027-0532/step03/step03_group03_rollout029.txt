    h = np.clip(h.astype(float), 0.0, 1.0)

    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]

        if lag >= 0:
            if lag < n:

        else:

            if m < n:
                grad[:n - m] += w * one_minus_h[m:]

    thresh = 1e-10

    if np.sum(mask) < 1:
        mask = np.ones(len(corr), dtype=bool)
    return compute_grad_weighted(x, np.where(mask)[0], weights[mask])


def slsqp_phase(h_start, target_sum, alphas, maxiter, ftol, deadline):
    n = len(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    for alpha in alphas:
        if time.time() > deadline:
            break
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc_new = compute_c5(h_new)
            sc_old = compute_c5(h_cur)
            if sc_new < sc_old:
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def subgradient_refine(h_start, target_sum, max_iter, lr_init, topk, alpha,
                       seed, deadline, momentum=0.90):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_indices = np.argpartition(corr, -topk)[-topk:]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.015, 3.0)
        else:
            stagnation += 1
            lr *= 0.985
        h = h_new
        if stagnation > 300:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_swap_search(h_start, target_sum, max_iters, deadline, seed=123):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n - 1)
        j = i + 1
        delta = h[i] - h[j]
        if abs(delta) < 1e-12:
            continue
        eps = delta * 0.5
        if h[i] + eps > 1.0 or h[j] - eps < 0.0:
            continue
        h_test = h.copy()
        h_test[i] += eps
        h_test[j] -= eps
        sc = compute_c5(h_test)
        if sc < best_score:
            best_score = sc
            best_h = h_test.copy()
            h = h_test
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    hard_deadline = start_time + budget_s - 15.0
    rng = np.random.RandomState(seed)

    def _deadline():
        return min(hard_deadline, start_time + budget_s - 30.0 - (time.time() - start_time))

    def _remaining():
        return hard_deadline - time.time()

    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    best_score = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Polish at initial resolution
    if _remaining() > 150:
        dl = start_time + 60.0
        h_pol = slsqp_phase(h, target, [20, 50, 150], 600, 1e-13, dl)
        sc = compute_c5(h_pol)
        if sc < best_score:
            best_score = sc
            best_h = h_pol

    # Phase 2: Progressive resolution increase with SLSQP
    resolutions = [550, 700, 850, 1000, 1200]
    for n_next in resolutions:
        if _remaining() < 100:
            break
        target_next = n_next / 2.0
        h_cur = interpolate_h(best_h, n_next, target_next)

        # Subgradient warm-up
        if _remaining() > 80:
            dl_sg = time.time() + min(40.0, _remaining() * 0.25)
            h_sub, sc_sub = subgradient_refine(h_cur, target_next,
                                                max_iter=800, lr_init=0.06,
                                                topk=5, alpha=25, seed=seed,
                                                deadline=dl_sg)
            if sc_sub < compute_c5(h_cur):
                h_cur = h_sub

        # SLSQP at this resolution
        alphas = [30, 80, 200, 500]
        if _remaining() > 250:
            alphas = [30, 80, 200, 500, 1000, 2000]
        dl_slsqp = time.time() + min(_remaining() * 0.7, 300.0)
        h_ref = slsqp_phase(h_cur, target_next, alphas, 1500, 1e-14, dl_slsqp)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Additional refinement at best resolution
    n_final = len(best_h)
    target_final = n_final / 2.0

    if _remaining() > 80:
        h_pol2 = slsqp_phase(best_h, target_final, [200, 500, 1000, 2000],
                             800, 1e-15, time.time() + min(60.0, _remaining() * 0.6))
        sc = compute_c5(h_pol2)
        if sc < best_score:
            best_score = sc
            best_h = h_pol2

    # Phase 4: Subgradient micro-refinement with multiple configs
    if _remaining() > 50:
        for cfg_idx in range(3):
            if _remaining() < 20:
                break
            h_try = best_h + rng.randn(n_final) * 0.004
            h_try = project_feasible(h_try, target_final)
            dl_sg2 = time.time() + min(25.0, _remaining() * 0.4)
            h_sg, sc_sg = subgradient_refine(h_try, target_final,
                                              max_iter=1500, lr_init=0.02,
                                              topk=5, alpha=40,
                                              seed=seed + 5000 + cfg_idx * 100,
                                              deadline=dl_sg2)
            if sc_sg < best_score:
                best_score = sc_sg
                best_h = h_sg

    # Phase 5: Local swap search
    if _remaining() > 30:
        h_swap, sc_swap = local_swap_search(best_h, target_final,
                                            max_iters=3000,
                                            deadline=time.time() + min(20.0, _remaining() * 0.5),
                                            seed=seed + 9999)
        if sc_swap < best_score:
            best_score = sc_swap
            best_h = h_swap

    # Final projection
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
