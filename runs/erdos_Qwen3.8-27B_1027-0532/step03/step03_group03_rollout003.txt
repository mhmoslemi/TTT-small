   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       return float(np.max(corr)) * dx

   def project_feasible(h, target_sum):
       h = h.astype(float)
       lo = -1.0
       hi = 2.0
       for _ in range(80):
           mid = (lo + hi) / 2.0
           s = np.sum(np.clip(h - mid, 0.0, 1.0))
           if s > target_sum:
               lo = mid
           else:
               hi = mid
       return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

   def compute_smooth_obj(x, alpha, mu, target_sum):
       corr = np.correlate(x, 1.0 - x, mode='full')
       c_max = np.max(corr)
       shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
       obj = c_max + np.log(np.sum(np.exp(shifted))) / alpha
       obj += mu * (np.sum(x) - target_sum) ** 2
       return float(obj)

   def compute_smooth_grad(x, alpha, mu, target_sum):
       corr = np.correlate(x, 1.0 - x, mode='full')
       n = len(x)
       c_max = np.max(corr)

       weights = exp_shifted / total

                   grad[:m] += w * one_minus_x[lag:]
                   grad[lag:] -= w * x[:m]
           else:
               m = -lag
               if m < n:
                   grad[:n-m] += w * one_minus_x[m:]
                   grad[m:] -= w * x[:n-m]
       grad += 2.0 * mu * (np.sum(x) - target_sum)
