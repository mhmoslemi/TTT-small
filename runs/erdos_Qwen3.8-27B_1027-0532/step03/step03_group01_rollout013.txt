def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_with_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, np.argmax(corr)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def soft_max_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    max_val = np.max(scaled)
    soft_max = (max_val + np.log(np.sum(np.exp(scaled - max_val))) / n) / alpha * (2.0/n)
    return soft_max

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h_new = h.copy()
        moved = False

            i0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 1:
            # Contiguous block swap
            block_size = rng.choice([2, 3, 4, 5, 6, 8, 10])

            runs = []

                    if j - i >= block_size:

                    i = j

            if runs:

                actual_size = min(block_size, end - start + 1)
                # Find a contiguous run of 0s of at least actual_size
                zero_runs = []
                i = 0
                while i < n:
                    if h[i] == 0:
                        j = i

                        i = j

                    zactual = min(actual_size, zend - zstart + 1)
                    # Do the swap
                    h_new[start:start+zactual] = 0
                    h_new[zstart:zstart+zactual] = 1
                    moved = True

                h_new[ones[idx1]] = 0
                h_new[zeros[idx0]] = 1
                moved = True
        else:

            # Pick k random positions, try to improve them
            k = rng.choice([3, 5, 8])
            positions = rng.choice(n, k, replace=False)
            # Try all 2^k assignments (only feasible ones)
            current_sum = h[positions].sum()
            for mask in range(2**k):

from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    return h_r.astype(float)

def softmax_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_val = np.max(scaled)
    obj = float((max_val + np.log(np.sum(np.exp(scaled - max_val)))) / alpha)
    weights = np.exp(scaled - max_val)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if s == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n-s])
        else:
            p = -s
            grad[p:] += w * (1.0 - h[:n-p])
            grad[:n-p] += w * (-h[p:])
    
    grad *= dx
    return obj, grad

def cons_fun(h):
    return np.sum(h) - _cons_target

def cons_jac(h):
    return np.ones(_cons_n)

def slsqp_with_grad(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            def obj_func(x, a=alpha):
                return softmax_obj_grad(x, a)[0]
            def grad_func(x, a=alpha):
                return softmax_obj_grad(x, a)[1]
            
            res = minimize(obj_func, h_cur, jac=grad_func,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def project_grad_descent(h, target, alphas, lr_init, max_iter_total, rng,
                          time_limit=None, start_time=None):

    h = h.copy()
    best_h = h.copy()
    best_score = compute_c5(h)
    iter_per_alpha = max_iter_total // len(alphas)
    
    for alpha in alphas:

            if (time.time() - start_time) > time_limit:
                break

        for i in range(iter_per_alpha):

                if (time.time() - start_time) > time_limit:

def compute_c5(h):

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]

    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1

def softmax_obj_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr * dx
    max_val = np.max(scaled)
    obj = float((max_val + np.log(np.sum(np.exp(scaled - max_val)))) / alpha)
    weights = np.exp(scaled - max_val)
    weights /= weights.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        w = weights[k]
        if w < 1e-14:
            continue
        if s == 0:
            grad[:] += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n - s])
        else:
            p = -s
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] += w * (-h[p:])
    grad *= dx
    return obj, grad

def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target, _current_alpha
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 15.0 - (time.time() - start_time)

    h_bin_start = project_binary(h_init, target_init)
    bin_score = compute_c5(h_bin_start)
    if bin_score < best_score:
        best_score = bin_score
        best_h = h_bin_start.copy()

    if tleft() > 150:
        allowed_time = min(250.0, tleft() * 0.25)
        h_sa, c5_sa = targeted_sa(h_bin_start, target_init, 100000, 0.015, 0.00005,
                                   rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 100:
        alphas = [20, 50, 150, 400, 1000, 2000, 4000, 8000]
        h_ref, sc = slsqp_with_grad(best_h, n_init, target_init, alphas, 500, 1e-14,
                                     time_limit=tleft() - 70, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 80:
        alphas_pg = [50, 200, 800, 2000, 5000]
        h_pg, sc_pg = project_grad_descent(best_h, target_init, alphas_pg,
                                            0.15, 6000, rng,
                                            time_limit=tleft() - 50, start_time=start_time)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg

    n_candidates = [800, 1000, 400, 500]
    n_candidates = [n for n in n_candidates if n != n_init and n >= 200 and n % 2 == 0]
    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_time = min(80.0, tleft() * 0.2)
        max_it = int(min(40000, allowed_time * 3000))
        h_sa, c5_sa = targeted_sa(h_bin_try, target_try, max_it, 0.012, 0.00008,
                                   rng, time_limit=allowed_time, start_time=start_time)
        if tleft() > 40:
            alphas_r = [50, 200, 800, 2000, 5000]
            h_ref, sc = slsqp_with_grad(h_sa, n_try, target_try, alphas_r, 300, 1e-13,
                                         time_limit=tleft() - 30, start_time=start_time)
            if sc < c5_sa:
                h_sa = h_ref
                c5_sa = sc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    for rs in range(3):
        if tleft() < 80:
            break
        n_try = n_init
        h_rand = np.zeros(n_try)
        perm = rng.permutation(n_try)
        h_rand[perm[:n_try // 2]] = 1.0
        allowed_time = min(80.0, tleft() * 0.2)
        max_it = int(min(40000, allowed_time * 3000))
        h_sa, c5_sa = targeted_sa(h_rand, n_try / 2.0, max_it, 0.015, 0.0001,
                                   rng, time_limit=allowed_time, start_time=start_time)
        if tleft() > 30:
            alphas_r = [50, 200, 800, 2000]
            h_ref, sc = slsqp_with_grad(h_sa, n_try, n_try / 2.0, alphas_r, 200, 1e-13,
                                         time_limit=tleft() - 20, start_time=start_time)
            if sc < c5_sa:
                h_sa = h_ref
                c5_sa = sc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(5000, tleft() * 2000))
        h_sa_f, c5_f = targeted_sa(h_bin_final, target_use, max_it, 0.005, 0.00003,
                                    rng, time_limit=max(5.0, tleft() - 12), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

    if tleft() > 10:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas_f = [1000, 3000, 8000, 15000]
        h_ref, sc = slsqp_with_grad(best_h, n_use, target_use, alphas_f, 200, 1e-15,
                                     time_limit=max(3.0, tleft() - 5), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
