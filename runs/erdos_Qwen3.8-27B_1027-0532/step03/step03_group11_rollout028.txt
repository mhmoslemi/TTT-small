    swaps_made = 0

    while swaps_made < max_swaps:
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                delta = sc - current
                if delta < best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                h[oi] = 1.0
                h[zi] = 0.0
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        current += best_delta
        swaps_made += 1
    return h, current

def simulated_annealing_binary(h_init, n_pop, time_limit, start_time, rng):
    n = len(h_init)
    pop = np.zeros((n_pop, n))
    scores = np.zeros(n_pop)
    pop[0] = h_init.copy()
    scores[0] = compute_c5(pop[0])

    for p in range(1, n_pop):
        if p == 1:
            h_pert = h_init + 0.2 * rng.standard_normal(n)
            pop[p] = make_binary_from_continuous(h_pert)
        elif p == 2:
            h_pert = h_init.copy()
            nflip = max(3, n // 10)
            idx = rng.choice(n, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            pop[p] = make_binary_from_continuous(h_pert)
        else:
            pop[p] = random_binary(n, rng)
        scores[p] = compute_c5(pop[p])

    best_h = pop[np.argmin(scores)].copy()
    best_score = float(scores.min())

    total_steps = 500000
    t_start = 0.005
    t_end = 0.00001
    steps_since_improve = 0

    for step in range(total_steps):
        if time.time() - start_time > time_limit:
            break
        t = t_start * (t_end / t_start) ** (step / max(total_steps - 1, 1))
        p = step % n_pop
        r = rng.random()

        if r < 0.6:
            ones = np.where(pop[p] == 1.0)[0]
            zeros = np.where(pop[p] == 0.0)[0]
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            pop[p][oi] = 0.0
            pop[p][zi] = 1.0
            new_score = compute_c5(pop[p])
            delta = new_score - scores[p]
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
                scores[p] = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = pop[p].copy()
                    steps_since_improve = 0
                else:
                    steps_since_improve += 1
            else:
                pop[p][oi] = 1.0
                pop[p][zi] = 0.0
                steps_since_improve += 1
        elif r < 0.8:
            blen = int(rng.integers(2, max(3, n // 8) + 1))
            a = int(rng.integers(0, n - blen + 1))
            b = int(rng.integers(0, n - blen + 1))
            if abs(a - b) < blen:
                continue
            block_a = pop[p][a:a + blen].copy()
            block_b = pop[p][b:b + blen].copy()
            pop[p][a:a + blen] = block_b
            pop[p][b:b + blen] = block_a
            new_score = compute_c5(pop[p])
            delta = new_score - scores[p]
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
                scores[p] = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = pop[p].copy()
                    steps_since_improve = 0
                else:
                    steps_since_improve += 1
            else:
                pop[p][a:a + blen] = block_a
                pop[p][b:b + blen] = block_b
                steps_since_improve += 1
        else:
            blen = int(rng.integers(2, max(3, n // 6) + 1))
            a = int(rng.integers(0, n - blen + 1))
            b = int(rng.integers(0, n - blen + 1))
            if abs(a - b) < blen:
                continue
            block_a = pop[p][a:a + blen].copy()
            block_b = pop[p][b:b + blen].copy()
            pop[p][a:a + blen] = 1.0 - block_b
            pop[p][b:b + blen] = 1.0 - block_a
            new_score = compute_c5(pop[p])
            delta = new_score - scores[p]
            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
                scores[p] = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_h = pop[p].copy()
                    steps_since_improve = 0
                else:
                    steps_since_improve += 1
            else:
                pop[p][a:a + blen] = block_a
                pop[p][b:b + blen] = block_b
                steps_since_improve += 1

        if steps_since_improve > 20000:
            for pp in range(n_pop):
                if scores[pp] > best_score + 0.03:
                    pop[pp] = random_binary(n, rng)
                    scores[pp] = compute_c5(pop[pp])
            steps_since_improve = 0

    return best_h, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = project_onto_constraint(
            np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0, 1), n0 / 2.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 120:
        h_opt, c5_opt = alg_optimize(h0, 0.01, 3, 300, budget_s - 60, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_opt, c5_opt = fista_optimize(best_h, 0.005, 2000, budget_s - 60, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_opt, c5_opt = alg_optimize(best_h, 0.003, 5, 200, budget_s - 40, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        h_opt, c5_opt = fista_optimize(best_h, 0.002, 3000, budget_s - 30, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    h_bin = make_binary_from_continuous(best_h)
    bin_c5 = compute_c5(h_bin)

    sa_time_limit = budget_s - 250
    h_bin_best, bin_c5_best = simulated_annealing_binary(h_bin, 8, sa_time_limit, start_time, rng)

    h_bin_best, bin_c5_best = fast_greedy_binary(h_bin_best, 100, budget_s - 200, start_time)

    h_refine = h_bin_best.astype(float) + 0.02 * rng.standard_normal(n0)
    h_refine = project_onto_constraint(h_refine, target_sum)

    eps_list = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001]
    for eps in eps_list:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 80:
            break
        limit = min(budget_s - 60, elapsed + 60)
        h_opt, c5_opt = alg_optimize(h_refine, eps, 4, 300, limit, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_refine = h_opt.copy()

    remaining = budget_s - (time.time() - start_time) - 100
    restart_count = 0
    while remaining > 60 and restart_count < 20:
        strategy = restart_count % 5
        if strategy == 0:
            h_pert = best_h + 0.03 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = h_bin_best.astype(float) + 0.05 * rng.standard_normal(n0)
        elif strategy == 2:
            h_tmp = best_h + 0.15 * rng.standard_normal(n0)
            h_pert = make_binary_from_continuous(h_tmp).astype(float)
            h_pert += 0.03 * rng.standard_normal(n0)
        elif strategy == 3:
            h_pert = random_binary(n0, rng).astype(float) + 0.08 * rng.standard_normal(n0)
        else:
            alpha = rng.uniform(0.3, 0.7)
            h_rand = random_binary(n0, rng).astype(float)
            h_pert = alpha * best_h + (1.0 - alpha) * h_rand

        h_pert = project_onto_constraint(h_pert, target_sum)
        tl = min(remaining - 30, 45)
        if tl > 20:
            h_opt, c5_opt = alg_optimize(h_pert, 0.005, 3, 200, start_time + tl, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_opt2, c5_opt2 = alg_optimize(h_opt, 0.001, 3, 200, start_time + tl + 30, start_time)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()

        restart_count += 1
        remaining = budget_s - (time.time() - start_time) - 100

    remaining = budget_s - (time.time() - start_time) - 40
    if remaining > 180:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.02 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        eps_up = [0.01, 0.005, 0.003, 0.001, 0.0005]
        for eps in eps_up:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 40:
                break
            limit = min(budget_s - 30, elapsed + 50)
            h_opt_up, c5_opt_up = alg_optimize(h_up, eps, 3, 200, limit, start_time)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()

        h_bin_up = make_binary_from_continuous(h_up)
        h_bin_up, bin_c5_up = fast_greedy_binary(h_bin_up, 50, budget_s - 60, start_time)
        if budget_s - (time.time() - start_time) > 40:
            h_ref_up = h_bin_up.astype(float) + 0.01 * rng.standard_normal(n1)
            h_ref_up = project_onto_constraint(h_ref_up, n1 / 2.0)
            for eps in [0.003, 0.001, 0.0005]:
                elapsed = time.time() - start_time
                if budget_s - elapsed < 30:
                    break
                h_opt_up2, c5_opt_up2 = alg_optimize(h_ref_up, eps, 3, 200, budget_s - 20, start_time)
                if c5_opt_up2 < best_c5:
                    best_c5 = c5_opt_up2
                    best_h = h_opt_up2.copy()
                h_ref_up = h_opt_up2.copy()

    remaining = budget_s - (time.time() - start_time) - 10
    if remaining > 20:
        n_cur = len(best_h)
        for eps in [0.0003, 0.0001, 0.00005]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 10:
                break
            h_opt, c5_opt = fista_optimize(best_h, eps, 5000, budget_s - 5, start_time)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5

    return final_h, final_c5, len(final_h)
