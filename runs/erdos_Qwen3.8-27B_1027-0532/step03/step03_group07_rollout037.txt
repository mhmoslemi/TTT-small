    """Simulated annealing with binary swap moves"""
    h = h.copy()
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(result) * dx
    best_h = h.copy()
    sig = _precompute_sig(h, n)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = t_start * (t_end / t_start) ** (it / max(max_iters-1, 1))
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

        i = int(rng.choice(ones))
        # Pick j within window of i, from zeros
        j_candidates = zeros[np.abs(zeros - i) <= window]
        if len(j_candidates) == 0:
            continue

        new_max = np.max(new_result) * dx
        delta = new_max - best_c  # Wait, should compare to current, not best

            result = new_result
            h[i] = 0.0
            h[j] = 1.0

            cur_c = np.max(result) * dx
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()

def _grad_corr(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_obj(h, alpha, n, active_lags, weights):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = np.array([corr[k] for k in active_lags])
    max_v = np.max(vals)
    if max_v < -500:
        return max_v
    lse = max_v + np.log(np.sum(np.exp(alpha * (vals - max_v)))) / alpha
    return lse * (2.0 / n)

def _slsqp_refine(h0, n, target_sum, alpha, maxiter, deadline):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    top_m = np.argsort(corr)[::-1][:8]
    top_m = sorted(top_m.tolist())
    w = np.exp(alpha * 0.3 * (corr[top_m] - corr[top_m[0]]))
    w = w / w.sum()
    try:
        res = minimize(lambda x: _smooth_obj(x, alpha, n, top_m, w), h0,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

        return _fix_sum(res.x, target_sum)

        return h0

def _nesterov(h0, n, target_sum, n_iters, alpha0, sg_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(40.0 * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        top = np.argsort(weights)[::-1][:sg_m]
        w_m = weights[top].copy()
        w_m = w_m / w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top)):
            g = g + w_m[idx] * _grad_corr(y, top[idx], dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start0 = time.time()
    deadline = t_start0 + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_list = [n_init]
    for d in [2, -2, 4, -4, 6, -6]:
        nt = n_init + d
        if nt % 2 == 0 and 50 <= nt <= 250:
            if nt not in n_list:
                n_list.append(nt)

    for n in n_list:
        if time.time() > deadline - 80:
            break
        target = n / 2.0

        if n == n_init:
            h_curr = _fix_sum(h_init.copy(), target)
        else:
            h_interp = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_curr = _fix_sum(np.clip(h_interp, 0.0, 1.0), target)

        c5_curr = _c5_of(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
            best_n = n

        # Phase 1: Binary local search (novel)
        h_bin = _snap_to_binary(h_curr, target)
        h_bin, c5_bin = _binary_greedy(h_bin, n, max_rounds=300, window=40)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            best_n = n

        # Phase 2: ILS with simulated annealing
        if time.time() < deadline - 60:
            h_ils, c5_ils = _ils_binary(h_bin, n, deadline - 40, rng, n_cycles=8)
            if c5_ils < best_c5:
                best_c5 = c5_ils
                best_h = h_ils.copy()
                best_n = n

        # Phase 3: Nesterov on fractional
        if time.time() < deadline - 50:
            h_nest, c5_nest = _nesterov(h_ils, n, target, n_iters=300, alpha0=0.4,
                                         sg_m=6, deadline=deadline - 40)
            if c5_nest < best_c5:
                best_c5 = c5_nest
                best_h = h_nest.copy()
                best_n = n

        # Snap Nesterov output to binary and refine
        if time.time() < deadline - 40:
            h_snap = _snap_to_binary(h_nest, target)
            h_snap2, c5_snap = _binary_greedy(h_snap, n, max_rounds=200, window=40)
            if c5_snap < best_c5:
                best_c5 = c5_snap
                best_h = h_snap2.copy()
                best_n = n

        # Phase 4: SLSQP with increasing alpha
        h_work = h_nest.copy()
        for al in [50, 200, 800, 3000, 10000]:
            if time.time() > deadline - 15:
                break
            h_sl = _slsqp_refine(h_work, n, target, al, maxiter=400, deadline=deadline)
            c5_sl = _c5_of(h_sl)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()
                best_n = n
            h_work = h_sl

            # Also try snapping SLSQP output
            if time.time() < deadline - 15:
                h_s2 = _snap_to_binary(h_sl, target)
                h_s3, c5_s3 = _binary_greedy(h_s2, n, max_rounds=100, window=40)
                if c5_s3 < best_c5:
                    best_c5 = c5_s3
                    best_h = h_s3.copy()
                    best_n = n
                    h_work = _fix_sum(h_s3.astype(float), target)

    # Final polish on best solution
    n_best = best_n
    target_final = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 10:
        h_final_snapped = _snap_to_binary(h_final, target_final)
        h_final_bp, c5_fp = _binary_greedy(h_final_snapped, n_best, max_rounds=200, window=50)
        if c5_fp < best_c5:
            best_c5 = c5_fp
            best_h = h_final_bp.copy()

    if time.time() < deadline - 8:
        h_final = _nesterov(best_h, n_best, target_final, n_iters=200, alpha0=0.3,
                            sg_m=8, deadline=deadline - 5)
        c5_fn = _c5_of(h_final)
        if c5_fn < best_c5:
            best_c5 = c5_fn
            best_h = h_final.copy()

    if time.time() < deadline - 5:
        h_final2 = _snap_to_binary(best_h, target_final)
        h_final3, c5_f3 = _binary_greedy(h_final2, n_best, max_rounds=150, window=50)
        if c5_f3 < best_c5:
            best_c5 = c5_f3
            best_h = h_final3.copy()

    if time.time() < deadline - 3:
        h_sl_final = _slsqp_refine(_fix_sum(best_h.astype(float), target_final),
                                    n_best, target_final, 5000, maxiter=500,
                                    deadline=deadline - 1)
        c5_sf = _c5_of(h_sl_final)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sl_final.copy()

    # Final snap and binary polish
    if time.time() < deadline - 2:
        h_fs = _snap_to_binary(best_h, target_final)
        h_fs2, c5_fs = _binary_greedy(h_fs, n_best, max_rounds=100, window=50)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_fs2.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
