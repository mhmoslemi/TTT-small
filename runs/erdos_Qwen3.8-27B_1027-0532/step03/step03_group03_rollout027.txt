def compute_grad_c(h, lag_indices):
    """
    Compute gradient of correlation values at given indices w.r.t. h.
    lag_indices: array of indices into the full correlation (0 to 2n-2)
    Returns: array of shape (len(lag_indices), n)
    """

    K = len(lag_indices)
    grad = np.zeros((K, n))
    for ki, li in enumerate(lag_indices):
        m = li - (n - 1)
        # c[li] = sum over valid i of h[i] * (1 - h[i+m])

- Its gradient
- Box constraints (which L-BFGS-B handles natively)

For the equality constraint, I'll use a penalty approach:

And I'll increase the penalty over time.

For the smooth max, I'll use:
L_α(corr) = (1/α) * log(Σ exp(α * corr_k))

The gradient of this is:
∇L_α = (1/α) * Σ_k [exp(α*corr_k) / Σ_j exp(α*corr_j)] * ∇corr_k

where p_k = exp(α*corr_k) / Σ_j exp(α*corr_j) is a softmax over the correlations.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        step = diff / len(idx)
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, alpha, penalty, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    # Smooth max
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    smooth = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    # Penalty for sum constraint
    sum_pen = penalty * (np.sum(x) - target) ** 2
    return smooth + sum_pen

def smooth_max_grad(x, alpha, penalty, target):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500, 0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total  # softmax weights, sum to 1
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-14:
            continue
        m = ki - (n - 1)
        if m >= 0:
            if m < n:
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - x[j1 + m])
                j2 = np.arange(m, n)
                grad[j2] -= w * x[j2 - m]
        else:

                grad[j1] += w * (1.0 - x[j1 + m])
                j2 = np.arange(0, n - am)
                grad[j2] -= w * x[j2 + am]
    
    # Add gradient of penalty
    grad += 2.0 * penalty * (np.sum(x) - target)
    return grad

def lbfgsb_optimize(h_start, target, alphas, penalties, maxiter):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for i, alpha in enumerate(alphas):
        pen = penalties[min(i, len(penalties)-1)]
        res = minimize(
            smooth_max_obj, h_cur,
            args=(alpha, pen, target),
            jac=smooth_max_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14}
        )
        h_new = project_feasible(res.x, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score

        # Get top-k indices

        else:

        # Softmax weights over top-k
        max_top = np.max(corr[top_idx])
        w = np.exp(alpha * (corr[top_idx] - max_top))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        for ki in range(len(top_idx)):
            wi = w[ki]
            if wi < 1e-14:
                continue
            m = top_idx[ki] - (n - 1)
            if m >= 0:
                if m < n:
                    j1 = np.arange(0, n - m)
                    grad[j1] += wi * (1.0 - h[j1 + m])
                    j2 = np.arange(m, n)
                    grad[j2] -= wi * h[j2 - m]
            else:

                    j2 = np.arange(0, n - am)

        # Momentum update

        h_new = h - lr_cur * v

        if noise_prob > 0 and rng.random() < noise_prob:

            stagnation += 1
            lr_cur *= 0.95
        h = h_new

            break

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:
            idx = np.where(h > 1e-15)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad(h, lag_indices, weights):
    n = len(h)
    grad = np.zeros(n)
    for ki in range(len(lag_indices)):
        w = weights[ki]
        if w < 1e-15:
            continue
        m = lag_indices[ki] - (n - 1)
        if m >= 0:
            if m < n:
                j1 = np.arange(0, n - m)
                grad[j1] += w * (1.0 - h[j1 + m])
                j2 = np.arange(m, n)
                grad[j2] -= w * h[j2 - m]
        else:
            am = -m
            if am < n:
                j1 = np.arange(am, n)
                grad[j1] += w * (1.0 - h[j1 + m])
                j2 = np.arange(0, n - am)
                grad[j2] -= w * h[j2 + am]
    return grad

def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    exp_s = np.exp(shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    smooth_val = max_c + np.log(total) / alpha
    grad = weighted_grad(h, np.arange(len(corr)), weights)
    return smooth_val, grad

def _lbfgs_obj(x, alpha, penalty, target):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    smooth = max_c + np.log(np.sum(np.exp(shifted))) / alpha
    pen = penalty * (np.sum(x) - target) ** 2
    return smooth + pen

def _lbfgs_grad(x, alpha, penalty, target):
    _, g = smooth_max_and_grad(x, alpha)
    g = g + 2.0 * penalty * (np.sum(x) - target) * np.ones_like(x)
    return g

def _slsqp_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = np.clip(alpha * (corr - max_c), -500.0, 0.0)
    return max_c + np.log(np.sum(np.exp(shifted))) / alpha

def _slsqp_grad(x, alpha):
    _, g = smooth_max_and_grad(x, alpha)
    return g

def _slope_fun(x):
    return np.sum(x) - _G_TARGET

def _slope_jac(x):
    return np.ones(_G_N)

_G_N = 0
_G_TARGET = 0.0

def slsqp_refine(h_start, target, alphas, maxiter, ftol):
    global _G_N, _G_TARGET
    _G_N = len(h_start)
    _G_TARGET = target
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _slope_fun, 'jac': _slope_jac}]
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,), jac=_slsqp_grad,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_sc

def lbfgsb_refine(h_start, target, alphas, penalties, maxiter):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    for i in range(len(alphas)):
        alpha = alphas[i]
        pen = penalties[min(i, len(penalties) - 1)]
        try:
            res = minimize(_lbfgs_obj, h_cur, args=(alpha, pen, target),
                           jac=_lbfgs_grad, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_sc:
                best_sc = sc
                best_h = h_new
            h_cur = h_new
        except Exception:
            pass
    return best_h, best_sc

def subgrad_optimize(h_start, target, max_iter, topk, alpha, lr,
                     noise_prob, noise_scale, momentum, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr_cur = lr
    stag = 0
    n_all = 2 * n - 1

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topk < n_all:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_all)
        max_top = np.max(corr[top_idx])
        w = np.exp(alpha * (corr[top_idx] - max_top))
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / ws

        grad = weighted_grad(h, top_idx, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr_cur * v
        h_new = project_feasible(h_new, target)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stag = 0
            lr_cur = min(lr_cur * 1.03, 20.0)
        else:
            stag += 1
            lr_cur *= 0.96
        h = h_new
        if stag > patience:
            break
    return best_h, best_score

def balanced_subgrad(h_start, target, max_iter, topk, alpha, lr,
                     momentum, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr_cur = lr
    stag = 0
    n_all = 2 * n - 1
    beta = 0.5

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topk < n_all:
            top_idx = np.argpartition(corr, -topk)[-topk:]
        else:
            top_idx = np.arange(n_all)

        max_top = np.max(corr[top_idx])
        w = np.exp(alpha * (corr[top_idx] - max_top))
        ws = np.sum(w)
        if ws < 1e-300:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w = w / ws

        grad_main = weighted_grad(h, top_idx, w)

        all_w = np.exp(alpha * (corr - np.max(corr)))
        all_ws = np.sum(all_w)
        if all_ws < 1e-300:
            all_w = np.ones(n_all) / n_all
        else:
            all_w = all_w / all_ws
        grad_all = weighted_grad(h, np.arange(n_all), all_w)

        grad = (1.0 - beta) * grad_main + beta * grad_all

        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr_cur * v
        h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stag = 0
            lr_cur = min(lr_cur * 1.02, 15.0)
        else:
            stag += 1
            lr_cur *= 0.96
        h = h_new
        if stag > patience:
            break
    return best_h, best_score
