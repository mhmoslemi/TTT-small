```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def smooth_max_and_grad(h, alpha, beta=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / np.sum(exp_shifted)
    obj = c_max + np.log(np.sum(exp_shifted)) / alpha
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        m = k - (n - 1)
        w = p[k] * dx
        i_start1 = max(0, m)
        i_end1 = min(n, n + m)
        if i_start1 < i_end1:
            idx = np.arange(i_start1, i_end1)
            grad[idx] += w * (1.0 - h[idx - m])
        i_start2 = max(0, -m)
        i_end2 = min(n, n - m)
        if i_start2 < i_end2:
            idx = np.arange(i_start2, i_end2)
            grad[idx] += w * (-h[idx + m])
    
    if beta > 0:
        grad += 2.0 * beta * (np.sum(h) - n / 2.0)
        obj += beta * (np.sum(h) - n / 2.0) ** 2
    
    return obj, grad

def l_bfgs_refine(h, target, time_limit, alphas=None, betas=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    if alphas is None:
        alphas = [5, 10, 20, 50, 100, 200]
    if betas is None:
        betas = [1.0, 10.0, 100.0]
    
    for alpha in alphas:
        for beta in betas:
            if time.time() - start >= time_limit:
                break
            if time.time() - start < time_limit - 5:
                obj, grad = smooth_max_and_grad(best, alpha, beta)
                def fun(x):
                    return smooth_max_and_grad(x, alpha, beta)[0]
                def gfun(x):
                    return smooth_max_and_grad(x, alpha, beta)[1]
                maxiter = int(max(50, (time_limit - (time.time() - start)) * 2))
                try:
                    res = minimize(fun, best, jac=gfun, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
                    h_new = project_feasible(res.x, target)
                    s = compute_c5(h_new)
                    if s < best_score:
                        best_score = s
                        best = h_new
                except Exception:
                    pass
    return best, best_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def smooth_max_obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        m = k - (n - 1)
        w = pk * dx
        # First term: (1-h[i-m]) for valid i
        i_s = max(0, m)
        i_e = min(n, n + m)
        if i_s < i_e:
            idx = np.arange(i_s, i_e)
            grad[idx] += w * (1.0 - h[idx - m])
        # Second term: -h[i+m] for valid i
        i_s2 = max(0, -m)
        i_e2 = min(n, n - m)
        if i_s2 < i_e2:
            idx2 = np.arange(i_s2, i_e2)
            grad[idx2] += w * (-h[idx2 + m])
    
    if beta > 0:
        s = np.sum(h) - n / 2.0
        obj += beta * s * s
        grad += 2.0 * beta * s
    
    return obj, grad


def lbfgs_optimize(h, target, time_limit, alpha, beta, maxiter=None):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        return smooth_max_obj_grad(x, alpha, beta)[0]
    
    def gfun(x):
        return smooth_max_obj_grad(x, alpha, beta)[1]
    
    if maxiter is None:
        maxiter = 300
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def coordinate_descent(h, target, time_limit, deltas, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = project_feasible(trial, target)
                s = compute_c5(trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = trial
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                deltas = [d * 0.5 for d in deltas]
                if max(abs(d) for d in deltas) < 1e-6:
                    break
                no_improve = 0
    
    return best, best_score


def subgradient_phase(h, target, time_limit, alpha=30.0, eta=0.003):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    while time.time() - start < time_limit:
        _, grad = smooth_max_obj_grad(h, alpha, 0.0)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_new = h - eta * grad / gn
        h_new = project_feasible(h_new, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new
            no_improve = 0
            eta = min(eta * 1.02, 0.01)
        else:
            no_improve += 1
            if no_improve > 50:
                alpha *= 1.3
                eta *= 0.8
                no_improve = 0
        h = h_new
    
    return best, best_score


def make_spread_start(n, freq=1.0, amp=0.3):
    x = np.arange(n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = project_feasible(h, target)
    return h


def make_random_start(n, seed=42):
    rng = np.random.RandomState(seed)
    h = rng.uniform(0.0, 1.0, n)
    target = n / 2.0
    h = project_feasible(h, target)
    return h

    for alpha, beta in alpha_beta_seq:

                                  deltas=[0.01, -0.01, 0.005, -0.005, 0.002, -0.002],

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-15
    else:
        mask = h > 1e-15
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def get_fft_size(n):
    # Return next power of 2 >= 3n-2
    size = 3 * n - 2
    p2 = 1
    while p2 < size:
        p2 *= 2
    return p2


_smooth_cache = {}


def smooth_max_obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 500.0)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    
    n_fft = get_fft_size(n)
    # First term: g1[i] = (W * (1-h))[i + n - 1]
    W = np.zeros(n_fft)
    W[:2*n-1] = p * dx
    F = np.zeros(n_fft)
    F[:n] = 1.0 - h
    C1 = np.fft.irfft(np.fft.rfft(W) * np.fft.rfft(F), n_fft)
    g1 = C1[n-1:2*n-1]
    
    # Second term: g2[i] = -(W_rev * h)[i + n - 1]
    W_rev = np.zeros(n_fft)
    W_rev[:2*n-1] = p[::-1] * dx
    G = np.zeros(n_fft)
    G[:n] = h
    C2 = np.fft.irfft(np.fft.rfft(W_rev) * np.fft.rfft(G), n_fft)
    g2 = -C2[n-1:2*n-1]
    
    grad = g1 + g2
    
    if beta > 0:
        s = np.sum(h) - n / 2.0
        obj += beta * s * s
        grad = grad + 2.0 * beta * s
    
    return obj, grad


def lbfgs_optimize(h, target, time_limit, alpha, beta, maxiter=500):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    
    def fun(x):
        return smooth_max_obj_grad(x, alpha, beta)[0]
    
    def gfun(x):
        return smooth_max_obj_grad(x, alpha, beta)[1]
    
    try:
        res = minimize(fun, h, jac=gfun, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10,
                                'maxfun': 5000})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def coordinate_descent(h, target, time_limit, deltas, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    no_improve_sweeps = 0
    
    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start >= time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                trial = best.copy()
                trial[i] = val
                trial = project_feasible(trial, target)
                s = compute_c5(trial)
                if s < best_score - 1e-14:
                    best_score = s
                    best = trial
                    improved = True
                    break
        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                deltas = [d * 0.5 for d in deltas]
                if max(abs(d) for d in deltas) < 1e-7:
                    break
                no_improve_sweeps = 0
    
    return best, best_score


def subgradient_step(h, alpha, eta, target):
    _, grad = smooth_max_obj_grad(h, alpha, 0.0)
    gn = np.linalg.norm(grad)
    if gn < 1e-15:
        return h
    h_new = h - eta * grad / gn
    return project_feasible(h_new, target)


def subgradient_phase(h, target, time_limit, alpha_init=50.0, eta_init=0.002):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_init
    eta = eta_init
    no_improve = 0
    h_curr = h.copy()
    
    while time.time() - start < time_limit:
        h_new = subgradient_step(h_curr, alpha, eta, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.008)
        else:
            no_improve += 1
            if no_improve > 40:
                alpha *= 1.2
                eta *= 0.85
                no_improve = 0
        h_curr = h_new
    
    return best, best_score
