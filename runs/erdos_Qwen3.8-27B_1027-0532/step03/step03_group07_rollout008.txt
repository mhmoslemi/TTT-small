        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_ls_mean_max, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_flat(h0, n, target_sum, n_iters, alpha0, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = _corr_of(y)
        max_corr = np.max(corr)
        progress = it / max(n_iters, 1)
        temp = 20.0 * (1.0 - progress) + 500.0 * progress
        shifted = corr - max_corr
        weights = np.exp(temp * shifted)
        total_w = weights.sum()
        if total_w < 1e-300:
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        else:
            weights = weights / total_w
            n_active = min(n + 1, max(3, int(40.0 / temp + 3)))
            top_idx = np.argsort(weights)[::-1][:n_active]
            w_top = weights[top_idx]
            w_top = w_top / w_top.sum()
            g = np.zeros(n)
            for idx in range(len(top_idx)):
                g += w_top[idx] * _grad_corr_term(y, top_idx[idx], dx)
        step = alpha0 / (1.0 + 0.5 * np.sqrt(float(it)))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _adam_opt(h0, n, target_sum, n_iters, lr0, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.95
    eps = 1e-8
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = _corr_of(x)
        max_corr = np.max(corr)
        temp = 80.0 + 600.0 * (it / max(n_iters - 1, 1))
        shifted = corr - max_corr
        weights = np.exp(temp * shifted)
        total_w = weights.sum()
        if total_w < 1e-300:
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(x, k_star, dx)
        else:
            weights = weights / total_w
            n_active = min(n + 1, max(3, int(50.0 / temp + 3)))
            top_idx = np.argsort(weights)[::-1][:n_active]
            w_top = weights[top_idx]
            w_top = w_top / w_top.sum()
            g = np.zeros(n)
            for idx in range(len(top_idx)):
                g += w_top[idx] * _grad_corr_term(x, top_idx[idx], dx)
        m_arr = beta1 * m_arr + (1.0 - beta1) * g
        v_arr = beta2 * v_arr + (1.0 - beta2) * (g * g)
        m_hat = m_arr / (1.0 - beta1 ** (it + 1))
        v_hat = v_arr / (1.0 - beta2 ** (it + 1))
        lr = lr0 / (1.0 + 0.05 * it)
        x_new = _project(x - lr * m_hat / (np.sqrt(v_hat) + eps), target_sum)
        x = x_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.8 - 0.4 * t_frac
        cr = 0.5 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 12:
            break
    return best_h, best_c5

def _basin_hopping(h0, n, target_sum, deadline, rng, n_hops=60):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for hop in range(n_hops):
        if time.time() > deadline:
            break
        h_pert = best_h.copy()
        mode = rng.randint(5)
        if mode == 0:
            block_len = rng.randint(2, max(3, n // 15))
            start = rng.randint(0, n - block_len)
            h_pert[start:start+block_len] = 1.0 - h_pert[start:start+block_len]
        elif mode == 1:
            n_change = rng.randint(max(3, n // 20), max(5, n // 6))
            idx = rng.choice(n, size=n_change, replace=False)
            h_pert[idx] += rng.normal(0, 0.15, n_change)
        elif mode == 2:
            n_change = rng.randint(max(3, n // 10), max(5, n // 4) + 1)
            idx = rng.choice(n, size=n_change, replace=False)
            h_pert[idx] = rng.uniform(0, 1, n_change)
        elif mode == 3:
            block_len = rng.randint(4, max(5, n // 8))
            start1 = rng.randint(0, n - block_len)
            start2 = rng.randint(0, n - block_len)
            h_pert[start1:start1+block_len], h_pert[start2:start2+block_len] = \
                h_pert[start2:start2+block_len].copy(), h_pert[start1:start1+block_len].copy()
        else:
            h_pert += rng.normal(0, 0.08, n)
        h_pert = _fix_sum(h_pert, target_sum)
        h_opt, c5_opt = _nesterov_flat(h_pert, n, target_sum, n_iters=120,
                                         alpha0=0.4, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        if time.time() > deadline:
            break
    return best_h, best_c5

def _coord_descent(h0, n, target_sum, max_rounds=30):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-13:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_frac(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    vals = [0.0, 0.2, 0.25, 1.0/3.0, 0.5, 2.0/3.0, 0.75, 0.8, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _simulated_anneal(h0, n, target_sum, n_iters, t0, t1, deadline, rng):
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        T = t0 * (t1 / t0) ** (it / max(n_iters, 1))
        n_change = rng.randint(1, max(2, n // 15))
        idx = rng.choice(n, size=n_change, replace=False)
        x_trial = x.copy()
        x_trial[idx] += rng.normal(0, max(0.05, 0.3 * T), n_change)
        x_trial = _fix_sum(x_trial, target_sum)
        c_trial = _c5_of(x_trial)
        c_cur = _c5_of(x)
        if c_trial < c_cur or rng.random() < np.exp(-(c_trial - c_cur) / max(T, 1e-10)):
            x = x_trial
            if c_trial < best_val:
                best_val = c_trial
                best = x_trial.copy()
    return best, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start0 = time.time()
    deadline = t_start0 + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = h_init.copy()
    best_n = n_init

    n = n_init
    target_sum = n / 2.0

    c5_init = _c5_of(h_init)
    if c5_init < best_c5:
        best_c5 = c5_init
        best_h = h_init.copy()

    # Phase 1: DE for global exploration
    pop_size = min(18, max(10, n // 7))
    n_gen = min(150, max(60, int((deadline - time.time() - 300) / max(1, pop_size * 0.025))))
    h_de, c5_de = _de_optimize(h_init, n, target_sum, pop_size, n_gen,
                                 deadline - 300, rng)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Phase 2: Nesterov with adaptive flattening from best
    if time.time() < deadline - 250:
        h_n1, c5_n1 = _nesterov_flat(best_h, n, target_sum, n_iters=400,
                                       alpha0=0.5, deadline=deadline - 250)
        if c5_n1 < best_c5:
            best_c5 = c5_n1
            best_h = h_n1.copy()

    # Phase 3: Adam optimization
    if time.time() < deadline - 200:
        h_ad, c5_ad = _adam_opt(best_h, n, target_sum, n_iters=300,
                                  lr0=0.06, deadline=deadline - 200)
        if c5_ad < best_c5:
            best_c5 = c5_ad
            best_h = h_ad.copy()

    # Phase 4: Basin hopping with structured perturbations
    if time.time() < deadline - 150:
        h_bh, c5_bh = _basin_hopping(best_h, n, target_sum,
                                       deadline - 150, rng, n_hops=50)
        if c5_bh < best_c5:
            best_c5 = c5_bh
            best_h = h_bh.copy()

    # Phase 5: Second Nesterov pass (finer)
    if time.time() < deadline - 100:
        h_n2, c5_n2 = _nesterov_flat(best_h, n, target_sum, n_iters=500,
                                       alpha0=0.3, deadline=deadline - 100)
        if c5_n2 < best_c5:
            best_c5 = c5_n2
            best_h = h_n2.copy()

    # Phase 6: SLSQP with increasing alpha
    h_curr = best_h.copy()
    for alpha in [50, 200, 500, 1500, 4000, 10000]:
        if time.time() > deadline - 25:
            break
        h_sl = _slsqp_opt(h_curr, n, target_sum, alpha, maxiter=500)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
        h_curr = h_sl

    # Phase 7: Coordinate descent
    if time.time() < deadline - 15:
        h_cd = _coord_descent(h_curr, n, target_sum, max_rounds=40)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    if time.time() < deadline - 10:
        h_cf = _coord_descent_frac(h_cd, n, target_sum, max_rounds=15)
        c5_cf = _c5_of(h_cf)
        if c5_cf < best_c5:
            best_c5 = c5_cf
            best_h = h_cf.copy()

    # Phase 8: Another basin hop + SLSQP if time allows
    if time.time() < deadline - 30:
        h_bh2, c5_bh2 = _basin_hopping(best_h, n, target_sum,
                                         deadline - 30, rng, n_hops=20)
        if c5_bh2 < best_c5:
            best_c5 = c5_bh2
            best_h = h_bh2.copy()

    if time.time() < deadline - 15:
        h_n3, c5_n3 = _nesterov_flat(best_h, n, target_sum, n_iters=300,
                                       alpha0=0.25, deadline=deadline - 15)
        if c5_n3 < best_c5:
            best_c5 = c5_n3
            best_h = h_n3.copy()

    # Final aggressive SLSQP
    if time.time() < deadline - 8:
        h_pol = _slsqp_opt(best_h, n, target_sum, 20000, maxiter=800)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() < deadline - 5:
        h_cd_f = _coord_descent(best_h, n, target_sum, max_rounds=30)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 3:
        h_cf_f = _coord_descent_frac(best_h, n, target_sum, max_rounds=15)
        c5_cf_f = _c5_of(h_cf_f)
        if c5_cf_f < best_c5:
            best_c5 = c5_cf_f
            best_h = h_cf_f.copy()

    if time.time() < deadline - 2:
        h_final_sl = _slsqp_opt(best_h, n, target_sum, 30000, maxiter=600)
        c5_final = _c5_of(h_final_sl)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_sl.copy()

    best_h = _fix_sum(best_h, n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n)
