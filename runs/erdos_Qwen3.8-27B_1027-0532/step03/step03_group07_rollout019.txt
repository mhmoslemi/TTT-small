import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _subgrad_momentum(h0, n, target_sum, n_iters, alpha0, beta0, beta1, m_lags, deadline):
    """Subgradient method with heavy-ball momentum and top-m smoothing."""
    dx = 2.0 / n
    x = h0.copy()
    v = np.zeros(n)
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        
        # Get top-m lags for smoothed subgradient
        m_use = min(m_lags, len(corr))
        top_m = np.argsort(corr)[::-1][:m_use]
        
        # Exponential weighting based on how close to max
        max_corr = corr.max()
        weights = np.exp(20.0 * (corr[top_m] - max_corr))
        weights = weights / weights.sum()
        
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += weights[idx] * _grad_corr_term(x, top_m[idx], dx)
        
        alpha = alpha0 / (1.0 + 0.01 * it)
        v = beta0 * v + alpha * g
        x_new = _project(x - v, target_sum)
        
        # Heavy ball momentum
        beta = beta0 + (beta1 - beta0) * it / max(n_iters - 1, 1)
        x = x_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

        if not improved:
            break
    
    # Re-fix sum (the CD might have broken it slightly)
    h = _fix_sum(h, target_sum)
    return h

def _compute_slopes(h, j, n):
    """Compute slope of each correlation term with respect to h[j]."""
    n_corr = 2*n - 1
    slopes = np.zeros(n_corr)
    for k in range(n_corr):
        lag = k - (n - 1)
        if lag == 0:
            slopes[k] = 1 - 2*h[j]
        elif lag > 0:
            m = n - lag
            s = 0.0
            if j < m:
                s += (1 - h[j + lag])
            if j >= lag:
                s += (-h[j - lag])
            slopes[k] = s
        else:
            jp = -lag
            m = n - jp
            s = 0.0
            if j >= jp:
                s += (1 - h[j - jp])
            if j + jp < n:
                s += (-h[j + jp])
            slopes[k] = s
    return slopes

import time
import numpy as np
from scipy.optimize import minimize


_global_target_sum = 0.0
_global_n = 0


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h


def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag + m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag + m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx


def _compute_slopes(h, j, n):
    n_corr = 2 * n - 1
    slopes = np.zeros(n_corr)
    for k in range(n_corr):
        lag = k - (n - 1)
        if lag == 0:
            slopes[k] = 1.0 - 2.0 * h[j]
        elif lag > 0:
            m = n - lag
            s = 0.0
            if j < m:
                s += (1.0 - h[j + lag])
            if j >= lag:
                s += (-h[j - lag])
            slopes[k] = s
        else:
            jp = -lag
            m = n - jp
            s = 0.0
            if j >= jp:
                s += (1.0 - h[j - jp])
            if j + jp < n:
                s += (-h[j + jp])
            slopes[k] = s
    return slopes


def _exact_coord_descent(h0, n, target_sum, max_rounds=50, deadline=None):
    h = h0.copy()
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            old = h[j]
            slopes = _compute_slopes(h, j, n)
            a_lo, a_hi = 0.0, 1.0
            for _it in range(60):
                m1 = a_lo + (a_hi - a_lo) / 3.0
                m2 = a_hi - (a_hi - a_lo) / 3.0
                v1 = np.max(corr + slopes * (m1 - old))
                v2 = np.max(corr + slopes * (m2 - old))
                if v1 < v2:
                    a_hi = m2
                else:
                    a_lo = m1
            t_star = (a_lo + a_hi) / 2.0
            v_star = np.max(corr + slopes * (t_star - old))
            v_0 = np.max(corr + slopes * (0.0 - old))
            v_1 = np.max(corr + slopes * (1.0 - old))
            if v_0 < v_star:
                t_star = 0.0
            if v_1 < v_star:
                t_star = 1.0
            if abs(t_star - old) > 1e-10:
                h[j] = t_star
                corr += (t_star - old) * slopes
                improved = True
        if not improved:
            break
    h = _fix_sum(h, target_sum)
    return h


def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)


def _eq_con_fun(x):
    return x.sum() - _global_target_sum


def _eq_con_jac(x):
    return np.ones(_global_n)


def _slsqp_refine(h0, n, target_sum, alpha, maxiter=300):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum), _c5_of(_fix_sum(res.x, target_sum))
    except Exception:
        return h0, _c5_of(h0)


def _nesterov_refine(h0, n, target_sum, n_iters, alpha0, m_lags, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        m_use = min(m_lags, len(corr))
        top_m = np.argsort(corr)[::-1][:m_use]
        weights = np.exp(30.0 * (corr[top_m] - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(top_m))
            weights[0] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += weights[idx] * _grad_corr_term(y, top_m[idx], dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val


def _subgrad_momentum(h0, n, target_sum, n_iters, alpha0, m_lags, deadline, rng):
    dx = 2.0 / n
    x = h0.copy()
    v = np.zeros(n)
    best = x.copy()
    best_val = _c5_of(x)
    beta = 0.9
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        max_corr = np.max(corr)
        m_use = min(m_lags, len(corr))
        top_m = np.argsort(corr)[::-1][:m_use]
        weights = np.exp(25.0 * (corr[top_m] - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(top_m))
            weights[0] = 1.0
        else:
            weights = weights / total_w
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += weights[idx] * _grad_corr_term(x, top_m[idx], dx)
        alpha = alpha0 / (1.0 + 0.005 * it)
        v = beta * v + alpha * g
        x = _project(x - v, target_sum)
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val


def _simulated_annealing(h0, n, target_sum, n_iters, T_start, T_end, deadline, rng):
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    curr_val = best_val
    for it in range(n_iters):
        if time.time() > deadline:
            break
        frac = it / max(n_iters - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        n_change = rng.randint(1, 4)
        x_new = x.copy()
        idx = rng.choice(n, size=n_change, replace=False)
        scale = 0.1 * (1.0 + frac)
        for i in idx:
            x_new[i] = np.clip(x_new[i] + rng.normal(0, scale), 0.0, 1.0)
        x_new = _project(x_new, target_sum)
        val_new = _c5_of(x_new)
        delta = val_new - curr_val
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            x = x_new
            curr_val = val_new
            if val_new < best_val:
                best_val = val_new
                best = x.copy()
    return best, best_val

                best_c5 = c5_curr
                best_h = h_curr.copy()
                best_n = n

    if best_h is not None and time.time() < deadline - 15:
        n_best = best_n
        target_best = n_best / 2.0
        h_final = best_h.copy()

        for alpha in [5000, 20000, 80000, 200000]:
            if time.time() > deadline - 10:
                break
            h_final, c5_f = _slsqp_refine(h_final, n_best, target_best, alpha, maxiter=400)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

        if time.time() < deadline - 8:
            h_final = _exact_coord_descent(best_h, best_n, target_best, max_rounds=50,
                                            deadline=deadline - 5)
            c5_f = _c5_of(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

        if time.time() < deadline - 3:
            h_final, c5_f = _nesterov_refine(best_h, best_n, target_best, n_iters=300,
                                              alpha0=0.05, m_lags=20,
                                              deadline=deadline - 2)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()
            if time.time() < deadline - 1:
                h_final = _exact_coord_descent(best_h, best_n, target_best, max_rounds=30,
                                                deadline=deadline - 0.5)
                c5_f = _c5_of(h_final)
                if c5_f < best_c5:
                    best_c5 = c5_f
                    best_h = h_final.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
