def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

            jm = j - shifts  # j - shift for each k

            h_m = np.where(valid_m, h[np.clip(jm, 0, n-1)], 0.0)

One more thought: the ternary search for the 1D problem can be replaced with a more direct method. Since f(t) = max_k (A_k + B_k * t) is convex and piecewise linear, the minimum is at a point where the subgradient contains 0. The subgradient at t is conv{B_k : k ∈ argmax(A_k + B_k * t)}. So I need to find t where min{B_k : k ∈ argmax} ≤ 0 ≤ max{B_k : k ∈ argmax}.

A direct method:
1. Sort the lines by their value at t=0 (i.e., by A_k)
2. Start with the upper envelope at t=0
3. Move t to the right, updating the envelope at each intersection
4. Stop when the subgradient contains 0 or t reaches 1

This is O(n log n) for sorting + O(n) for the sweep, giving O(n log n) per coordinate. But the constant might be larger than ternary search in practice.

For simplicity, I'll stick with ternary search (40 iterations is more than enough).

Actually, there's an even simpler approach for the 1D problem. Since f(t) is convex:
- f(0) = max(A)
- f(1) = max(A + B)
- If the minimum is in the interior, it's where the derivative changes sign

The derivative (subgradient) at t is any value in [min{B_k : k ∈ argmax}, max{B_k : k ∈ argmax}].

A bisection on the subgradient:
- Evaluate the active set at t_mid
- If all B_k > 0 for active k, the minimum is to the left
- If all B_k < 0 for active k, the minimum is to the right
- Otherwise, we're at the minimum (or very close)

This converges in O(log(1/ε)) iterations, same as ternary search.

            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        top_m = np.argsort(weights)[::-1][:sg_m]
        w_m = weights[top_m].copy()
        w_m = w_m / w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g = g + w_m[idx] * _grad_corr(y, top_m[idx], dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y_new = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t_k = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    a = _global_alpha
    if a < 1.0:
        return m * _global_dx
    shifted = a * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    return (m + np.log(np.sum(np.exp(shifted))) / a) * _global_dx

def _slsqp_eq(x):
    return x.sum() - _global_target_sum

def _slsqp_eq_jac(x):
    return np.ones(_global_n)

def _slsqp_run(h0, n, target_sum, maxiter, alpha):
    global _global_target_sum, _global_n, _global_alpha, _global_dx
    _global_target_sum = target_sum
    _global_n = n
    _global_alpha = alpha
    _global_dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_out = _fix_sum(res.x, target_sum)
        return h_out, _c5_of(h_out)
    except Exception:
        return h0.copy(), _c5_of(h0)

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        mod = i % 5
        if mod == 1:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif mod == 2:
            pop[i] = _fix_sum(rng.uniform(0.0, 1.0, n), target_sum)
        elif mod == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif mod == 4:
            pop[i] = _fix_sum(h0[rng.permutation(n)], target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = float(pop_vals[best_idx])
    best_h = pop[best_idx].copy()
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (gen / float(max_gen))
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(150, 0.5)

    n_init = len(h_init)
    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c5 = _c5_of(best_h)
    best_n = n_init
    target = best_n / 2.0

    h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, target, 300)
    if c5_ccd < best_c5:
        best_c5 = c5_ccd
        best_h = h_ccd

    if time.time() < deadline - 120:
        h_sl, c5_sl = _slsqp_run(best_h, best_n, target, 300, 200.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    if time.time() < deadline - 100:
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, target, 300)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    if time.time() < deadline - 80:
        h_sl, c5_sl = _slsqp_run(best_h, best_n, target, 500, 500.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    if time.time() < deadline - 60:
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, target, 500)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    if time.time() < deadline - 120:
        h_ncd, c5_ncd = _ccds_perturb(best_h, best_n, target, 80, 5, 0.08, rng, deadline - 60)
        if c5_ncd < best_c5:
            best_c5 = c5_ncd
            best_h = h_ncd

    if time.time() < deadline - 100:
        h_nd, c5_nd = _nesterov_opt(best_h, best_n, target, 200, 0.3, 10, 100.0, deadline - 50)
        if c5_nd < best_c5:
            best_c5 = c5_nd
            best_h = h_nd
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, target, 300)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    n_tries = []
    for d in (-20, 20, -30, 30, -40, 40, -50, 50, -10, 10, 70, -70):
        nt = n_init + d
        if nt % 2 == 0 and 80 <= nt <= 260:
            n_tries.append(nt)
    seen = set()
    n_unique = []
    for nn in n_tries:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)
    n_tries = n_unique

    for n_try in n_tries:
        if time.time() > deadline - 80:
            break
        target_t = n_try / 2.0
        h_try = np.interp(np.linspace(0, best_n - 1, n_try), np.arange(best_n), best_h)
        h_try = _fix_sum(np.clip(h_try, 0.0, 1.0), target_t)
        c5_try = _c5_of(h_try)
        if c5_try > best_c5 + 0.01:
            continue
        h_ccd_t, c5_ccd_t = _cont_coord_descent(h_try, n_try, target_t, 200)
        if c5_ccd_t < best_c5:
            best_c5 = c5_ccd_t
            best_h = h_ccd_t
            best_n = n_try
        if c5_ccd_t < best_c5 + 0.005 and time.time() < deadline - 50:
            h_sl_t, c5_sl_t = _slsqp_run(h_ccd_t, n_try, target_t, 300, 300.0)
            if c5_sl_t < best_c5:
                best_c5 = c5_sl_t
                best_h = h_sl_t
                best_n = n_try
            h_ccd_t2, c5_ccd_t2 = _cont_coord_descent(best_h, best_n, target_t, 300)
            if c5_ccd_t2 < best_c5:
                best_c5 = c5_ccd_t2
                best_h = h_ccd_t2

    if time.time() < deadline - 40:
        target_f = best_n / 2.0
        h_sl, c5_sl = _slsqp_run(best_h, best_n, target_f, 200, 150.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, target_f, 400)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    if time.time() < deadline - 20:
        h_sl, c5_sl = _slsqp_run(best_h, best_n, best_n / 2.0, 500, 400.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, best_n / 2.0, 500)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    if time.time() < deadline - 10:
        h_sl, c5_sl = _slsqp_run(best_h, best_n, best_n / 2.0, 1000, 400.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl

    if time.time() < deadline - 5:
        h_ccd, c5_ccd = _cont_coord_descent(best_h, best_n, best_n / 2.0, 500)
        if c5_ccd < best_c5:
            best_c5 = c5_ccd
            best_h = h_ccd

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
