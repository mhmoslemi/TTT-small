def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()

            if diff <= 1e-12:

        pass
    return h0

def _de_global(h0, n, target_sum, pop_size, max_gen, deadline, rng):
    dx = 2.0 / n
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    pop[1] = _fix_sum(1.0 - h0 + 0.03 * (rng.rand(n) - 0.5), target_sum)
    pop[2] = _fix_sum(rng.uniform(0.2, 0.8, n), target_sum)
    for i in range(3, pop_size):
        t = rng.uniform(0.3, 0.7)
        pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    F_start, F_end = 0.9, 0.3
    CR = 0.7
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline - 30:
            break
        F = F_start + (F_end - F_start) * (gen / float(max_gen))
        for i in range(pop_size):
            if time.time() > deadline - 30:
                break
            r1 = int(rng.randint(pop_size))
            r2 = int(rng.randint(pop_size))
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            mask = rng.random(n) < CR
            jr = int(rng.randint(n))
            mask[jr] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size * 2:
            for idx in np.argsort(pop_vals)[pop_size//2:]:
                t = 0.3 + 0.4 * rng.random()
                pop[idx] = _fix_sum(t * best_h + (1.0-t) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

def _local_refine(h0, n, deadline):
    h_best = h0.copy()
    c5_best = _c5_of(h0)
    if time.time() > deadline - 60:
        return h_best, c5_best
    h1, c1 = _pair_cd(h_best, n, max_sweeps=40, grid_size=30, deadline=deadline)
    if c1 < c5_best:
        c5_best, h_best = c1, h1
    if time.time() > deadline - 50:
        return h_best, c5_best
    h2, c2 = _nesterov_accel(h_best, n, n/2.0, 300, 0.15, M_top=8, alpha_sm=120.0, deadline=deadline)
    if c2 < c5_best:
        c5_best, h_best = c2, h2
    if time.time() > deadline - 40:
        return h_best, c5_best
    h3, c3 = _slp_cvxpy(h_best, n, max_iters=30, deadline=deadline)
    if c3 < c5_best:
        c5_best, h_best = c3, h3
    if time.time() > deadline - 30:
        return h_best, c5_best
    h4 = _slsqp_polish(h_best, n, n/2.0, maxiter=2000, alpha=400, deadline=deadline)
    c4 = _c5_of(h4)
    if c4 < c5_best:
        c5_best, h_best = c4, h4
    if time.time() > deadline - 20:
        return h_best, c5_best
    h5, c5v = _pair_cd(h_best, n, max_sweeps=30, grid_size=40, deadline=deadline)
    if c5v < c5_best:
        c5_best, h_best = c5v, h5
    if time.time() > deadline - 10:
        return h_best, c5_best
    h6, c6 = _slp_cvxpy(h_best, n, max_iters=30, deadline=deadline)
    if c6 < c5_best:
        c5_best, h_best = c6, h6
    return h_best, c5_best

def _resample(h_from, n_from, n_to):
    if n_from == n_to:
        return h_from.copy()
    x_new = np.linspace(0, n_from - 1, n_to)
    h_new = np.interp(x_new, np.arange(n_from), h_from)
    h_new = np.clip(h_new, 0.0, 1.0)
    return _fix_sum(h_new, n_to / 2.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_c5 = _c5_of(h_init)
    best_n = n_init

    # Phase 1: Quick local refinement at initial n
    h_ref, c5_ref = _local_refine(best_h, best_n, deadline)
    if c5_ref < best_c5:
        best_c5, best_h = c5_ref, h_ref

    # Phase 2: Try nearby resolutions with SLP+pair CD
    n_candidates = [best_n]
    for d in [5, -5, 10, -10, 15, -15, 20, -20, 25, 30]:
        nt = best_n + d
        if nt % 2 == 0 and 80 <= nt <= 350:
            n_candidates.append(nt)
    # Deduplicate and sort by proximity to best_n
    seen = set()
    n_unique = []
    for nn in n_candidates:
        if nn not in seen:
            seen.add(nn)
            n_unique.append(nn)
    n_unique.sort(key=lambda x: abs(x - best_n))

    for n_try in n_unique[1:6]:
        if time.time() > deadline - 100:
            break
        target = n_try / 2.0
        h_try = _resample(best_h, best_n, n_try)
        # SLP
        h_slp, c5_slp = _slp_cvxpy(h_try, n_try, max_iters=40, deadline=deadline)
        if c5_slp < best_c5:
            best_c5, best_h, best_n = c5_slp, h_slp, n_try
        # Pair CD
        h_cd = _pair_cd(h_try if c5_slp >= best_c5 else h_slp, n_try, max_sweeps=40, grid_size=30, deadline=deadline)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h, best_n = c5_cd, h_cd, n_try
        # Nesterov
        h_ne, c5_ne = _nesterov_accel(best_h, best_n, best_n/2.0, 400, 0.12, M_top=8, alpha_sm=150, deadline=deadline)
        if c5_ne < best_c5:
            best_c5, best_h, best_n = c5_ne, h_ne, n_try
        # Final SLP
        h_slp2, c5_slp2 = _slp_cvxpy(best_h, best_n, max_iters=40, deadline=deadline)
        if c5_slp2 < best_c5:
            best_c5, best_h, best_n = c5_slp2, h_slp2, n_try

    # Phase 3: DE global search at best_n
    if time.time() < deadline - 200:
        n_de = best_n
        h_de, c5_de = _de_global(best_h, n_de, n_de/2.0, pop_size=40, max_gen=80, deadline=deadline, rng=rng)
        if c5_de < best_c5:
            best_c5, best_h, best_n = c5_de, h_de, n_de
            # Refine the DE result
            h_ref2, c5_ref2 = _local_refine(best_h, best_n, deadline)
            if c5_ref2 < best_c5:
                best_c5, best_h = c5_ref2, h_ref2

    # Phase 4: Additional DE with different params
    if time.time() < deadline - 150:
        n_de2 = best_n
        h_de2, c5_de2 = _de_global(best_h + 0.1*(rng.rand(n_de2)-0.5), n_de2, n_de2/2.0,
                                     pop_size=30, max_gen=60, deadline=deadline, rng=rng)
        h_de2 = _fix_sum(h_de2, n_de2/2.0)
        c5_de2 = _c5_of(h_de2)
        if c5_de2 < best_c5:
            best_c5, best_h, best_n = c5_de2, h_de2, n_de2
            h_ref3, c5_ref3 = _local_refine(best_h, best_n, deadline)
            if c5_ref3 < best_c5:
                best_c5, best_h = c5_ref3, h_ref3

    # Phase 5: Final polishing
    if time.time() < deadline - 10:
        h_final = _pair_cd(best_h, best_n, max_sweeps=50, grid_size=50, deadline=deadline)
        c5_f = _c5_of(h_final)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_final
    if time.time() < deadline - 8:
        h_final2, c5_f2 = _slp_cvxpy(best_h, best_n, max_iters=40, deadline=deadline)
        if c5_f2 < best_c5:
            best_c5, best_h = c5_f2, h_final2
    if time.time() < deadline - 5:
        h_final3 = _slsqp_polish(best_h, best_n, best_n/2.0, maxiter=5000, alpha=500, deadline=deadline)
        c5_f3 = _c5_of(h_final3)
        if c5_f3 < best_c5:
            best_c5, best_h = c5_f3, h_final3
    if time.time() < deadline - 3:
        h_final4, c5_f4 = _pair_cd(best_h, best_n, max_sweeps=40, grid_size=50, deadline=deadline)
        if c5_f4 < best_c5:
            best_c5, best_h = c5_f4, h_final4

    # Ensure feasibility
    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
