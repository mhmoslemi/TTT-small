import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def generate_periodic_init(n, rng):
    h = np.zeros(n)
    period = rng.integers(7, 15)
    pattern_len = period
    n_ones_in_pattern = period // 2
    if rng.random() < 0.5:
        n_ones_in_pattern = (period + 1) // 2
    pattern = np.zeros(pattern_len)
    ones_positions = rng.choice(pattern_len, size=n_ones_in_pattern, replace=False)
    pattern[ones_positions] = 1.0
    h = np.tile(pattern, n // pattern_len + 1)[:n]
    current_sum = h.sum()
    target = n / 2
    if current_sum > target:
        ones_idx = np.where(h == 1.0)[0]
        extra = int(current_sum - target)
        to_flip = rng.choice(ones_idx, size=min(extra, len(ones_idx)), replace=False)
        h[to_flip] = 0.0
    elif current_sum < target:
        zeros_idx = np.where(h == 0.0)[0]
        need = int(target - current_sum)
        to_flip = rng.choice(zeros_idx, size=min(need, len(zeros_idx)), replace=False)
        h[to_flip] = 1.0
    return h

        n_steps = int(min(max(time_per_round * 30000, 200000), 3000000))
        n_steps = max(n_steps, 100000)

        if sa_rounds % 3 == 0:
            h_bin, bin_max = fast_sa(h_start, n_steps, t_start, t_end, rng)
        elif sa_rounds % 3 == 1:
            h_bin, bin_max = targeted_sa(h_start, n_steps, t_start, t_end, rng)
        else:
            h_bin, bin_max = fast_sa(h_start, n_steps, 0.5, 0.001, rng)

        c5_bin = dx * bin_max
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            elapsed = time.time() - start_time
            if elapsed < budget_s * 0.85:
                h_polish = h_bin + 0.02 * rng.standard_normal(n)
                h_polish = project_onto_constraint(h_polish, target_sum)
                h_polish, c5 = slsqp_optimize(h_polish, 0.001, 250)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_polish.copy()
                h_polish2 = h_bin + 0.01 * rng.standard_normal(n)
                h_polish2 = project_onto_constraint(h_polish2, target_sum)
                h_polish2, c5 = slsqp_optimize(h_polish2, 0.0005, 250)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_polish2.copy()

        sa_rounds += 1
        remaining = budget_s - (time.time() - start_time) - 60

    remaining = budget_s - (time.time() - start_time) - 40
    if remaining > 60:
        h_ls = to_binary(best_h)
        h_ls, ls_max = local_search_first(h_ls, 200)
        c5_ls = dx * ls_max
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining2 = budget_s - (time.time() - start_time) - 30
        if remaining2 > 40:
            h_ls2, ls2_max = local_search_best(h_ls, 30)
            c5_ls2 = dx * ls2_max
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()

    remaining = budget_s - (time.time() - start_time) - 20
    if remaining > 30:
        h_final = best_h + 0.005 * rng.standard_normal(n)
        h_final = project_onto_constraint(h_final, target_sum)
        eps_schedule = [0.001, 0.0005, 0.0002, 0.0001, 0.00003]
        for eps_val in eps_schedule:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 15:
                break
            h_final, c5 = slsqp_optimize(h_final, eps_val, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
        h_final2 = best_h + 0.002 * rng.standard_normal(n)
        h_final2 = project_onto_constraint(h_final2, target_sum)
        h_final2, c5 = slsqp_optimize(h_final2, 0.00005, 500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final2.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
