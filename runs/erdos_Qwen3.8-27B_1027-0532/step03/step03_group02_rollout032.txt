C_s = sum_{i: valid} h[i+s]*(1-h[i])

For s >= 0: valid i are 0 to n-1-s
C_s = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])

∂C_s/∂h[j]:
- h[j] appears in h[i+s] when i+s=j, i.e., i=j-s. Valid if 0≤j-s≤n-1-s, i.e., s≤j≤n-1.
  Contribution: (1-h[j-s])
- h[j] appears in (1-h[i]) when i=j. Valid if 0≤j≤n-1-s, i.e., j≤n-1-s.
  Contribution: -h[j+s] (since h[i+s] = h[j+s])

So for s ≥ 0:
∂C_s/∂h[j] = (1-h[j-s]) * 1[s≤j≤n-1] + (-h[j+s]) * 1[0≤j≤n-1-s]

For s < 0, let s' = -s > 0:
C_{s'} (with negative sign convention) = sum_{i: valid} h[i-s']*(1-h[i])
Valid i: i-s' ≥ 0 and i < n, so i ≥ s' and i < n.
C_{-s'} = sum_{i=s'}^{n-1} h[i-s']*(1-h[i])

∂C_{-s'}/∂h[j]:
- h[j] appears in h[i-s'] when i-s'=j, i.e., i=j+s'. Valid if s'≤j+s'≤n-1, i.e., 0≤j≤n-1-s'.

  Contribution: -h[j-s']

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -700.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    obj = (c_max + np.log(sum_exp) / alpha) * dx
    s_arr = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        s = int(s_arr[idx])
        w_s = w[idx]
        if s >= 0:
            if s < n:
                j1 = np.arange(s, n)
                grad[j1] += w_s * (1.0 - h[j1 - s])
                if n - s > 0:
                    j2 = np.arange(0, n - s)
                    grad[j2] -= w_s * h[j2 + s]
        else:
            sp = -s
            if n - sp > 0:
                j1 = np.arange(0, n - sp)
                grad[j1] += w_s * (1.0 - h[j1 + sp])
            if sp < n:
                j2 = np.arange(sp, n)
                grad[j2] -= w_s * h[j2 - sp]
    grad *= dx
    return obj, grad

def nesterov_optimize(h, target_sum, alpha, max_time, lr_init=0.005):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    h_prev = h.copy()
    t_k = 1.0
    lr = lr_init
    best_score = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    while time.time() - start < max_time:
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_next
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj_y, grad_y = smooth_obj_and_grad(y, alpha)
        lr_try = lr
        h_new = h
        score_new = best_score
        for _ in range(60):
            candidate = y - lr_try * grad_y
            candidate = project_feasible(candidate, target_sum)
            sc = compute_c5(candidate)
            if sc < obj_y or sc < best_score:
                h_new = candidate
                score_new = sc
                break
            lr_try *= 0.5
        if lr_try < 1e-12:
            break
        h_prev = h
        h = h_new
        t_k = t_next
        sc = compute_c5(h)
        if sc < best_score - 1e-14:
            best_score = sc
            best_h = h.copy()
            lr = min(lr * 1.05, 0.05)
        else:
            lr *= 0.85
        if lr < 1e-9:
            break
    return best_h, best_score

def active_set_cycling(h, target_sum, max_time, n_active=5, lr_init=0.01):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    best_score = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    stall = 0
    while time.time() - start < max_time:
        corr = compute_corr(h)
        c_max = np.max(corr)
        score = c_max * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
        top_idx = np.argsort(corr)[-n_active:]
        s_arr = np.arange(-(n - 1), n)
        grad_avg = np.zeros(n)
        for idx in top_idx:
            s = int(s_arr[idx])
            if s >= 0:
                if s < n:
                    j1 = np.arange(s, n)
                    grad_avg[j1] += (1.0 - h[j1 - s])
                if n - s > 0:
                    j2 = np.arange(0, n - s)
                    grad_avg[j2] -= h[j2 + s]
            else:
                sp = -s
                if n - sp > 0:
                    j1 = np.arange(0, n - sp)
                    grad_avg[j1] += (1.0 - h[j1 + sp])
                if sp < n:
                    j2 = np.arange(sp, n)
                    grad_avg[j2] -= h[j2 - sp]
        grad_avg /= n_active
        lr_try = lr
        for _ in range(60):
            h_new = h - lr_try * grad_avg
            h_new = project_feasible(h_new, target_sum)
            sc_new = compute_c5(h_new)
            if sc_new < score:
                break
            lr_try *= 0.5
        if lr_try < 1e-12:
            break
        h = h_new
        sc = compute_c5(h)
        if sc < score * 0.999:
            lr = min(lr * 1.1, 0.05)
        else:
            lr *= 0.9
    return best_h, best_score

def coord_descend_fast(h, target_sum, max_time, patience=25):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    while time.time() - start < max_time:
        for j in range(n):
            old_val = h[j]
            js = j - s_arr
            js_plus = j + s_arr
            A = np.zeros(2 * n - 1)
            mask1 = (js >= 0) & (js < n)
            mask2 = (js_plus >= 0) & (js_plus < n)
            A[mask1] += (1.0 - h[js[mask1]])
            A[mask2] -= h[js_plus[mask2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-12:
                    hi = mid
                elif grad < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            break
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgs_refine(h, target_sum, alpha, maxiter):
    n = len(h)

    def obj_fun(x):
        x_proj = project_feasible(x, target_sum)
        o, _ = smooth_obj_and_grad(x_proj, alpha)
        return o

    def grad_fun(x):
        x_proj = project_feasible(x, target_sum)
        _, g = smooth_obj_and_grad(x_proj, alpha)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_fun, h, jac=grad_fun, method='L-BFGS-B',
                       bounds=bounds, maxiter=maxiter,
                       options={'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': maxiter * 2})
        h_new = project_feasible(res.x, target_sum)
    except Exception:
        h_new = h
    return h_new, compute_c5(h_new)
