import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):

    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)

    if abs(residual) > 1e-10:

            idx2 = np.where(h < 1.0)[0]
        else:

            h = np.clip(h, 0.0, 1.0)

def alternating_lp_tophk(h_init, n, target, max_iter, time_limit, top_k=5, reg_init=1e-3, damping_init=0.5):
    """Alternating LP that penalizes top-k correlations."""
    start = time.time()

    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg_init
    stuck_count = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_simulated_annealing(h0, n, target, time_limit, rng, T_init=1.0, T_min=0.01):
    """Simulated annealing in binary space."""
    start = time.time()
    h = (h0 >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones_count = int(np.sum(h))
    if ones_count > target:
        ones_idx = np.where(h == 1)[0]
        excess = ones_count - target
        to_zero = rng.choice(ones_idx, excess, replace=False)
        h[to_zero] = 0
    elif ones_count < target:
        zeros_idx = np.where(h == 0)[0]
        needed = target - ones_count
        to_one = rng.choice(zeros_idx, needed, replace=False)
        h[to_one] = 1
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = T_init
    
    total_iters = 0
    while time.time() - start < time_limit:
        total_iters += 1
        # Simulated annealing schedule
        T = max(T_min, T_init * np.exp(-0.001 * total_iters))
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        j = rng.choice(ones)
        m = rng.choice(zeros)
        
        h[j] = 0
        h[m] = 1
        c5_new = compute_c5(h)
        delta = c5_new - current_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[j] = 1
            h[m] = 0
    
    return best_h, best_c5

def alternating_lp_v2(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg_init=0.0):
    """Alternating LP with CLARABEL and adaptive regularization."""
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg_init
    stuck_count = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        if current_reg > 1e-10:
            objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        else:
            objective = t_var
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        solved = False
        for solver in [cp.CLARABEL, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        if stuck_count > 15:
            damping = max(0.1, damping * 0.8)
            current_reg = max(0.0, current_reg * 0.5)
            stuck_count = 0
        elif it % 30 == 0:
            current_reg = max(0.0, current_reg * 0.9)
    
    return best_h, best_c5

def nesterov_smooth(h0, n, target, time_limit, alpha_init=50, alpha_end=500):
    """Nesterov-accelerated gradient on log-sum-exp."""
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    h = h0.copy()
    h_mom = h.copy()
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    j_arr = np.arange(n)
    
    total_steps = 0
    while time.time() - start < time_limit:
        total_steps += 1
        # Anneal alpha
        progress = min(1.0, total_steps / 5000)
        alpha = alpha_init * (alpha_end / alpha_init) ** progress
        
        # Nesterov lookahead
        h_look = (1 + beta) * h - beta * h_mom
        h_look = np.clip(h_look, 0.0, 1.0)
        # Fix sum constraint
        h_look = project_feasible(h_look, target)
        
        # Compute smooth objective and gradient
        corr = np.correlate(h_look, 1.0 - h_look, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        weights = exp_shifted / sum_exp
        
        # Gradient
        grad = np.zeros(n)
        for idx_k in range(2 * n - 1):
            w = weights[idx_k]
            if w < 1e-10:
                continue
            k = idx_k - n_half
            valid1 = (j_arr + k >= 0) & (j_arr + k < n)
            valid2 = (j_arr - k >= 0) & (j_arr - k < n)
            grad[valid1] += w * dx * (1.0 - h_look[j_arr[valid1] + k])
            grad[valid2] -= w * dx * h_look[j_arr[valid2] - k]
        
        # Step size: use fixed decreasing step
        step = 2.0 / (np.sqrt(total_steps + 1))
        step = min(step, 0.5)
        
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)
        
        c5_val = compute_c5(h_new)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h_new.copy()
        
        h_mom = h.copy()
        h = h_new
        beta = (1 + np.sqrt(1 + 4 * beta**2)) / 2.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h0, n, target, time_limit, rng):
    start = time.time()
    h = (h0 + 0.5).astype(int).astype(float)
    h = np.clip(h, 0, 1)
    ones_count = int(np.sum(h))
    tc = int(target)
    if ones_count > tc:
        ones_idx = np.where(h == 1)[0]
        excess = ones_count - tc
        to_zero = rng.choice(ones_idx, min(excess, len(ones_idx)), replace=False)
        h[to_zero] = 0
    elif ones_count < tc:
        zeros_idx = np.where(h == 0)[0]
        needed = tc - ones_count
        to_one = rng.choice(zeros_idx, min(needed, len(zeros_idx)), replace=False)
        h[to_one] = 1
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    T = 0.5
    total = 0
    while time.time() - start < time_limit:
        total += 1
        T = max(0.001, 0.5 * np.exp(-0.0005 * total))
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = rng.choice(ones)
        m = rng.choice(zeros)
        h[j] = 0
        h[m] = 1
        c5_new = compute_c5(h)
        delta = c5_new - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = c5_new
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[j] = 1
            h[m] = 0
    return best_h, best_c5

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg_init=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg_init
    stuck_count = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        if current_reg > 1e-12:
            objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        else:
            objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
                solved = True
            except Exception:
                pass
        if not solved:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
                solved = True
            except Exception:
                break
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 10:
            damping = max(0.1, damping * 0.85)
            current_reg = max(0.0, current_reg * 0.5)
            stuck_count = 0
        elif it % 40 == 0:
            current_reg = max(0.0, current_reg * 0.92)
    return best_h, best_c5

def momentum_subgrad(h0, n, target, time_limit, alpha_seq):
    start = time.time()
    dx = 2.0 / n
    n_half = n - 1
    j_arr = np.arange(n)
    h = h0.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.7
    for alpha in alpha_seq:
        steps_this_alpha = 0
        while time.time() - start < time_limit:
            steps_this_alpha += 1
            step = 0.05 / np.sqrt(1.0 + 0.01 * steps_this_alpha)
            h_look = h + momentum * (h - h_prev)
            h_look = project_feasible(h_look, target)
            corr = np.correlate(h_look, 1.0 - h_look, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
            exp_s = np.exp(shifted)
            sum_exp = np.sum(exp_s)
            if sum_exp < 1e-300:
                break
            weights = exp_s / sum_exp
            grad = np.zeros(n)
            for idx_k in range(2 * n - 1):
                w = weights[idx_k]
                if w < 1e-8:
                    continue
                k = idx_k - n_half
                valid1 = (j_arr + k >= 0) & (j_arr + k < n)
                valid2 = (j_arr - k >= 0) & (j_arr - k < n)
                grad[valid1] += w * dx * (1.0 - h_look[j_arr[valid1] + k])
                grad[valid2] -= w * dx * h_look[j_arr[valid2] - k]
            h_prev = h.copy()
            h = h - step * grad
            h = project_feasible(h, target)
            c5_val = compute_c5(h)
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
            if steps_this_alpha > 3000:
                break
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def topk_lp(h_init, n, target, max_iter, time_limit, k_top=3, damping_init=0.6, reg_init=0.0):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    current_reg = reg_init
    stuck_count = 0
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        corr_vals = np.zeros(2 * n - 1)
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            corr_vals[k + n - 1] = np.sum(g[lo - k:hi - k] * h[lo:hi]) * dx
        top_k_idx = np.argsort(corr_vals)[::-1][:k_top]
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for idx_k in top_k_idx:
            k = idx_k - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        if current_reg > 1e-12:
            objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        else:
            objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        try:
            prob.solve(solver=cp.CLARABEL, verbose=False)
            solved = True
        except Exception:
            pass
        if not solved:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
                solved = True
            except Exception:
                pass
        if not solved or h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 8:
            damping = max(0.15, damping * 0.8)
            current_reg = max(0.0, current_reg * 0.4)
            stuck_count = 0
    return best_h, best_c5
