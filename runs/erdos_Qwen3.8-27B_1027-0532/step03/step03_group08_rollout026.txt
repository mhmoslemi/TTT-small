```python
def _corr_delta_all_lags(h, j, delta):
    n = len(h)
    # m ranges from -(n-1) to (n-1)
    # For each m, delta_corr[m] = delta*(1-h[j+m]) [if j+m valid] - delta*h[j-m] [if j-m valid]
    m_vals = np.arange(-(n-1), n)
    delta_corr = np.zeros(2*n-1)
    # j+m valid: 0 <= j+m <= n-1, i.e., -j <= m <= n-1-j
    valid1 = (m_vals >= -j) & (m_vals <= n-1-j)
    if np.any(valid1):
        idx1 = j + m_vals[valid1]
        delta_corr[valid1] += delta * (1.0 - h[idx1])
    # j-m valid: 0 <= j-m <= n-1, i.e., j-(n-1) <= m <= j
    valid2 = (m_vals >= j-(n-1)) & (m_vals <= j)
    if np.any(valid2):
        idx2 = j - m_vals[valid2]
        delta_corr[valid2] -= delta * h[idx2]
    return delta_corr
```

```python
import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in range(len(order)):
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in range(len(order)):
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _corr_delta(h, j, delta):
    """Compute the change in all correlation terms when h[j] changes by delta."""
    n = len(h)
    m_vals = np.arange(-(n-1), n)
    delta_corr = np.zeros(2*n-1)
    valid1 = (m_vals >= -j) & (m_vals <= n-1-j)
    if np.any(valid1):
        idx1 = j + m_vals[valid1]
        delta_corr[valid1] += delta * (1.0 - h[idx1])
    valid2 = (m_vals >= j-(n-1)) & (m_vals <= j)
    if np.any(valid2):
        idx2 = j - m_vals[valid2]
        delta_corr[valid2] -= delta * h[idx2]
    return delta_corr

def _fast_coord_descent(h0, n, target_sum, max_rounds=50, values=None):
    """Fast coordinate descent using incremental correlation updates."""
    if values is None:
        values = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    current_max = np.max(corr)
    
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = current_max
            best_v = old
            for v in values:
                if abs(v - old) < 1e-12:
                    continue
                delta = v - old
                dc = _corr_delta(h, j, delta) * dx
                new_corr = corr + dc
                new_max = np.max(new_corr)
                if new_max < best_c - 1e-12:
                    best_c = new_max
                    best_v = v
            if best_v != old:
                delta = best_v - old
                corr += _corr_delta(h, j, delta) * dx
                h[j] = best_v
                current_max = np.max(corr)
                improved = True
        if not improved:
            break
    return h, current_max

def _subgradient(h, n, dx):
    """Compute subgradient of max_k corr[k] * dx at h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Active set: lags where corr is within tolerance of max
    tol = 1e-6
    active = np.where(corr >= max_val - tol)[0]
    # Gradient of corr[k] w.r.t. h
    # corr[k] = sum_i h[i] * (1 - h[i+m]) where m = k - (n-1)
    # d(corr[k])/d(h[j]) = (1-h[j+m]) if j+m valid, -h[j-m] if j-m valid
    grad = np.zeros(n)
    for k_idx in active:
        m = k_idx - (n - 1)
        # h[j] as first factor: j+m valid means 0<=j<=n-1 and 0<=j+m<=n-1
        #   => j in [max(0,-m), min(n-1, n-1-m)]
        j_lo = max(0, -m)
        j_hi = min(n-1, n-1-m)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += (1.0 - h[j_lo+m:j_hi+1+m])
        # h[j] as second factor (inside 1-h): j-m valid means 0<=j-m<=n-1 and 0<=j<=n-1
        #   => j in [max(0,m), min(n-1, n-1+m)]
        j_lo2 = max(0, m)
        j_hi2 = min(n-1, n-1+m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= h[j_lo2-m:j_hi2+1-m]
    # Average over active set (subgradient)
    grad /= len(active)
    return grad * dx

def _subgrad_descent(h0, n, target_sum, n_iters=2000, alpha0=0.01, use_nesterov=True, deadline=None):
    """Subgradient / Nesterov method with exact subgradient."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy() if use_nesterov else h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        g = _subgradient(y, n, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        if use_nesterov:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
            y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
            y_new = _project(y_new, target_sum)
        else:
            y_new = x_new
        x = x_new
        y = y_new
        if use_nesterov:
            t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter, deadline=None):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

                else:

                stall_count += 1
        if stall_count > pop_size:
```

```python
def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in range(len(order)):
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in range(len(order)):
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _corr_delta(h, j, delta):
    n = len(h)
    m_vals = np.arange(-(n-1), n)
    dc = np.zeros(2*n-1)
    v1 = (m_vals >= -j) & (m_vals <= n-1-j)
    if np.any(v1):
        dc[v1] += delta * (1.0 - h[j + m_vals[v1]])
    v2 = (m_vals >= j-(n-1)) & (m_vals <= j)
    if np.any(v2):
        dc[v2] -= delta * h[j - m_vals[v2]]
    return dc

def _subgrad(h, n, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    tol = max(1e-6, 0.01 * (np.max(corr) - np.min(corr)))
    active = np.where(corr >= max_val - tol)[0]
    grad = np.zeros(n)
    for k_idx in active:
        m = k_idx - (n - 1)
        j_lo = max(0, -m)
        j_hi = min(n-1, n-1-m)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += (1.0 - h[j_lo+m:j_hi+1+m])
        j_lo2 = max(0, m)
        j_hi2 = min(n-1, n-1+m)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= h[j_lo2-m:j_hi2+1-m]
    grad /= max(len(active), 1)
    return grad * dx

def _nesterov_sg(h0, n, target_sum, n_iters, alpha0, deadline=None):

    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):

        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()

    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    # Phase 1: Refine initial
    if time.time() < deadline - 200:
        hr, cr = _refine(h_init, n_init, n_init/2.0, deadline - 150)
        if cr < best_c5:
            best_c5 = cr; best_h = hr.copy()

    # Phase 2: DE at multiple n
    n_candidates = []
    for d in [0, -2, 2, -4, 4, -6, 6, -8, 8]:
        nc = n_init + d
        if nc % 2 == 0 and 120 <= nc <= 190:
            if nc not in n_candidates:
                n_candidates.append(nc)

    for n in n_candidates:
        if time.time() > deadline - 80:
            break
        ts = n / 2.0
        if n == best_n:
            h_start = best_h.copy()
        else:
            h_start = np.interp(np.linspace(0, best_n-1, n), np.arange(best_n), best_h)
            h_start = _fix_sum(np.clip(h_start, 0, 1), ts)
        h_de, c5_de = _de_search(h_start, n, ts, 35, 80, 0.8, 0.6, deadline - 60, rng)
        if c5_de < best_c5:
            best_c5 = c5_de; best_h = h_de.copy(); best_n = n
        if time.time() < deadline - 40:
            hr, cr = _refine(h_de, n, ts, deadline - 30)
            if cr < best_c5:
                best_c5 = cr; best_h = hr.copy(); best_n = n

    # Phase 3: SA escape
    if time.time() < deadline - 40:
        h_sa, c5_sa = _sa_fast(best_h, best_n, best_n/2.0, 2000, 0.03, 0.001, deadline - 30, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa; best_h = h_sa.copy()
        if time.time() < deadline - 20:
            hr, cr = _refine(best_h, best_n, best_n/2.0, deadline - 10)
            if cr < best_c5:
                best_c5 = cr; best_h = hr.copy()

    # Phase 4: Final intensive polish
    if time.time() < deadline - 3:
        nb = best_n
        tb = nb / 2.0
        hf = best_h.copy()
        for _ in range(4):
            if time.time() > deadline - 2:
                break
            hs, cs = _nesterov_sg(hf, nb, tb, 800, 0.015, deadline - 2)
            if cs < best_c5:
                best_c5 = cs; hf = hs
            hc, cc = _fast_bin_cd(hf, nb, tb, 40)
            if cc < best_c5:
                best_c5 = cc; hf = hc
        if time.time() < deadline - 2:
            hsl = _slsqp_opt(hf, nb, tb, 500, 3000)
            csl = _c5_of(hsl)
            if csl < best_c5:
                best_c5 = csl; hf = hsl
            hcd,ccd = _fast_frac_cd(hf, nb, tb, 30, 9)
            if ccd < best_c5:
                best_c5 = ccd; hf = hcd
        best_h = hf
        best_n = nb

    best_h = _fix_sum(best_h, best_n/2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
```
