            n_blocks = rng.randint(2, 5)
            for _ in range(n_blocks):
                pos = rng.randint(0, n_best)
                bw = rng.randint(3, max(4, n_best // 20))
                lo = max(0, pos - bw)
                hi = min(n_best, pos + bw)
                h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
            h_pert = project_feasible(h_pert, target_best)

        h_new = lbfgs_refine(h_pert, n_best, target_best, 500, maxiter=300)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 5: Final aggressive polish
    if tleft() > 30:
        for alpha in [8000, 20000, 50000]:
            if tleft() < 20:
                break
            h_new = lbfgs_refine(best_h, n_best, target_best, alpha, maxiter=1000)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 6: Focused final polish
    if tleft() > 25:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_k = max(20, len(corr) // 6)
        focus = np.argsort(corr)[-top_k:]
        for alpha in [10000, 30000]:
            if tleft() < 15:
                break
            h_new = lbfgs_focused(best_h, n_best, target_best, alpha, focus, maxiter=800)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    # Phase 7: Ultra-fine micro-perturbations
    for _ in range(20):
        if tleft() < 15:
            break
        amp = rng.uniform(0.0005, 0.008)
        h_pert = project_feasible(best_h + rng.randn(n_best) * amp, target_best)
        h_new = lbfgs_refine(h_pert, n_best, target_best, 20000, maxiter=800)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
