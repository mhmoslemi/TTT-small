Actually wait - I just realized that using SLSQP on each block requires setting up constraints for each block call. But the sum constraint involves ALL variables, not just the block. So if I'm optimizing a block of 8 variables, the constraint is:
sum(block) = target - sum(other variables)

This is a simple linear constraint on the block variables. Let me implement this.

Actually, for the block optimization, I can just:
1. Fix all variables outside the block
2. The constraint becomes: sum(block) = target - sum(fixed)
3. Optimize the block with SLSQP (bounds + one equality constraint)
4. After optimization, the full h is automatically feasible

This is cleaner than the projection approach.

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_g_n = 400
_g_target = 200
_g_alpha = 100.0
_g_h_base = None
_g_s = 0
_g_e = 0
_g_block_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def con_eq(x):
    return np.sum(x) - _g_target


def con_jac(x):
    return np.ones(_g_n)


def block_con_eq(x):
    return np.sum(x) - _g_block_target


def block_con_jac(x):
    return np.ones(_g_e - _g_s)


def block_obj(x):
    hf = _g_h_base.copy()
    hf[_g_s:_g_e] = np.clip(x, 0.0, 1.0)
    corr = np.correlate(hf, 1.0 - hf, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(_g_alpha * (corr - c_max), -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / _g_alpha


def slsqp_round(h, n, target, alphas, maxiter=3000, ftol=1e-16):
    global _g_n, _g_target, _g_alpha
    _g_n = n
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    for a in alphas:
        _g_alpha = a
        try:
            res = minimize(lse_obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < cur:
                h_cur = h_new
                cur = sc
        except Exception:
            pass
    return h_cur, cur


def block_slsqp_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    global _g_h_base, _g_s, _g_e, _g_target, _g_alpha, _g_block_target
    _g_target = target
    _g_alpha = alpha
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        step = n_blocks // n_rounds + 1
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            _g_h_base = h_cur
            _g_s = s
            _g_e = e
            _g_block_target = target - np.sum(h_cur[:s]) - np.sum(h_cur[e:])
            x0 = h_cur[s:e].copy()
            bounds_b = [(0.0, 1.0)] * block_size
            cons_b = [{'type': 'eq', 'fun': block_con_eq, 'jac': block_con_jac}]
            try:
                res = minimize(block_obj, x0, method='SLSQP', bounds=bounds_b,
                              constraints=cons_b, options={'maxiter': 500, 'ftol': 1e-14})
                ht = h_cur.copy()
                ht[s:e] = np.clip(res.x, 0.0, 1.0)
                ht = project_feasible(ht, target)
                sc = compute_c5(ht)
                if sc < cur:
                    h_cur = ht
                    cur = sc
            except Exception:
                pass
    return h_cur, cur


def two_interval_h(n, L1):
    L2 = n // 2 - L1
    h = np.zeros(n)
    h[:L1] = 1.0
    h[n - L2:] = 1.0
    return h


def periodic_h(n, period, duty):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < duty:
            h[i] = 1.0
    total = int(np.sum(h))
    if total > n // 2:
        h[:total - n // 2] = 0.0
    elif total < n // 2:
        deficit = n // 2 - total
        mask = h == 0
        indices = np.where(mask)[0]
        for j in range(deficit):
            h[indices[j]] = 1.0
    return h


def refined_two_interval(n, L1, edge_frac):
    h = np.zeros(n)
    edge = int(L1 * edge_frac)
    h[:edge] = 1.0
    h[edge:L1] = edge_frac
    L2 = n // 2 - L1
    edge2 = int(L2 * edge_frac)
    h[n - L2:] = 1.0
    h[n - L2:n - L2 + edge2] = 1.0 - edge_frac
    total = np.sum(h)
    h = project_feasible(h, n // 2.0)
    return h


def annealing_descent(h, n, target, n_iter, t_start, t_end, deadline):
    h_cur = h.copy()
    cur = compute_c5(h_cur)
    best_local = h_cur.copy()
    best_local_sc = cur
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t_start * (t_end / t_start) ** (it / max(n_iter, 1))
        i = np.random.randint(0, n)
        j = np.random.randint(0, n)
        if i == j:
            j = (j + 1) % n
        delta1 = (np.random.rand() - 0.5) * 0.3
        delta2 = -delta1
        h_new = h_cur.copy()
        h_new[i] = np.clip(h_new[i] + delta1, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] + delta2, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        sc_new = compute_c5(h_new)
        delta_sc = sc_new - cur
        if delta_sc < 0 or np.random.rand() < np.exp(-delta_sc / max(t, 1e-10)):
            h_cur = h_new
            cur = sc_new
            if cur < best_local_sc:
                best_local_sc = cur
                best_local = h_cur.copy()
    return best_local, best_local_sc
