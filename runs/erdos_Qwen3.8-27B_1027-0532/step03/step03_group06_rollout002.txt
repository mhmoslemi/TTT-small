    # Phase 1: Warm-up with increasing alpha
    for alpha in [10, 30, 100, 300, 800]:
        if tleft() < 80:
            break
        h = lbfgs_round(h, n, target, alpha, maxiter=300)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 2: Combined anti-peak + balance + L-BFGS iterations
    for it in range(40):
        if tleft() < 80:
            break
        h = anti_peak_step(h, n, target, 8, 0.004)
        h = balance_step(h, n, target, 15, 0.008)
        h = lbfgs_round(h, n, target, 500, maxiter=100)
        h = anti_peak_step(h, n, target, 5, 0.002)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase 3: Multi-start with diverse perturbations
    for attempt in range(35):
        if tleft() < 50:
            break
        ptype = attempt % 6
        if ptype == 0:
            amp = 0.015 + 0.005 * (attempt % 4)
            h_pert = project_feasible(best_h + rng.randn(n) * amp, target)
        elif ptype == 1:
            h_pert = best_h.copy()
            seg = rng.randint(8, max(9, n // 8))
            a = rng.randint(0, n - seg)
            b = rng.randint(0, n - seg)
            if abs(a - b) > seg:
                tmp = h_pert[a:a + seg].copy()
                h_pert[a:a + seg] = h_pert[b:b + seg]
                h_pert[b:b + seg] = tmp
            h_pert = project_feasible(h_pert, target)
        elif ptype == 2:
            edge = rng.randint(0, n)
            width = rng.randint(5, max(6, n // 12))
            h_pert = best_h.copy()
            lo = max(0, edge - width)
            hi = min(n, edge + width + 1)
            h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
            h_pert = project_feasible(h_pert, target)
        elif ptype == 3:
            shift = rng.randint(1, max(2, n // 10))
            h_pert = project_feasible(np.roll(best_h, shift), target)
        elif ptype == 4:
            h_pert = project_feasible(1.0 - best_h + rng.randn(n) * 0.01, target)
        else:
            # Fourier perturbation
            freq = rng.randint(2, max(3, n // 10))
            h_pert = best_h + 0.01 * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 2 * np.pi))
            h_pert = project_feasible(h_pert, target)

        h_pert, sc = multi_stage_refine(h_pert, n, target, [100, 400, 1000], 150, tleft)
        if sc < best_c5:
            best_c5, best_h = sc, h_pert.copy()

    # Phase 4: Resolution upgrade
    for n_try in [500, 600, 750]:
        if tleft() < 90:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_interp, sc = multi_stage_refine(h_interp, n_try, target_try,
                                          [50, 200, 600, 1500], 200, tleft)
        # Extra anti-peak at higher resolution
        for _ in range(10):
            if tleft() < 70:
                break
            h_interp = anti_peak_step(h_interp, n_try, target_try, 5, 0.002)
            sc = compute_c5(h_interp)
            if sc < best_c5:
                best_c5, best_h = sc, h_interp.copy()
                n, target = n_try, target_try
        if sc < best_c5:
            best_c5, best_h = sc, h_interp.copy()
            n, target = n_try, target_try

    # Phase 5: Final high-alpha L-BFGS polish
    for alpha in [2000, 5000]:
        if tleft() < 30:
            break
        best_h = lbfgs_round(best_h, n, target, alpha, maxiter=400)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 6: Final micro anti-peak steps
    for _ in range(30):
        if tleft() < 20:
            break
        best_h = anti_peak_step(best_h, n, target, 3, 0.001)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    # Phase 7: Tiny perturbation search
    for _ in range(15):
        if tleft() < 15:
            break
        amp = rng.uniform(0.001, 0.004)
        h_try = project_feasible(best_h + rng.randn(n) * amp, target)
        h_try = lbfgs_round(h_try, n, target, 3000, maxiter=200)
        h_try = anti_peak_step(h_try, n, target, 3, 0.001)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5, best_h = c5, h_try.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
