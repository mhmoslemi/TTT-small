1. At the current point h*, compute the gradient of each active correlation
2. Linearize the max function: approximate max_k corr[k] ≈ corr[k*] + ∇corr[k*]·(h-h*) for the active k*
3. Solve the resulting LP
4. Update h* and repeat

        h = project_sum(h, target)
    
    return best_h, best_score

def coord_descend_refined(h, max_time, rng, target, patience=50, grid_size=31):
    n = len(h)
    start = time.time()
    h = project_sum(h.copy(), target)
    best_score = compute_c5_h(h)
    best_h = h.copy()
    no_improve = 0
    
    while time.time() - start < max_time:
        any_improved = False
        for j in range(n):
            old = h[j]
            # Adaptive grid centered on current value
            if 0 < old < 1:
                span = 0.15
                local_grid = np.linspace(max(0, old - span), min(1, old + span), grid_size)
                global_grid = np.linspace(0, 1, 7)
                candidates = np.unique(np.concatenate([local_grid, global_grid, [old]]))
            else:
                candidates = np.linspace(0, 1, grid_size)
            
            best_v = old
            best_sc = np.inf
            for v in candidates:
                h[j] = v
                sc = compute_c5_h(h)
                if sc < best_sc:
                    best_sc = sc
                    best_v = v
            
            # Golden section refinement
            lo = max(0.0, best_v - 0.02)
            hi = min(1.0, best_v + 0.02)
            if hi - lo > 1e-5:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                a, b = lo, hi
                c = b - gr * (b - a)
                d = a + gr * (b - a)
                h[j] = c
                fc = compute_c5_h(h)
                h[j] = d
                fd = compute_c5_h(h)
                for _ in range(35):
                    if fc < fd:
                        b, d = d, c
                        fd = fc
                        c = b - gr * (b - a)
                        h[j] = c
                        fc = compute_c5_h(h)
                    else:
                        a, c = c, d
                        fc = fd
                        d = a + gr * (b - a)
                        h[j] = d
                        fd = compute_c5_h(h)
                best_v = (a + b) / 2.0
                h[j] = best_v
                sc = compute_c5_h(h)
                if sc < best_sc:
                    best_sc = sc
            
            h[j] = best_v
            h = project_sum(h, target)
            sc_final = compute_c5_h(h)
            if sc_final < best_score - 1e-13:
                best_score = sc_final
                best_h = h.copy()
                any_improved = True
        
        if not any_improved:
            no_improve += 1
        else:
            no_improve = 0
        
        if no_improve >= patience:
            scale = min(0.03, 0.005 * (no_improve - patience + 1))
            h = project_sum(np.clip(best_h + rng.randn(n) * scale, 0, 1), target)
            no_improve = 0
    
    return best_h, best_score

    sc_sym = compute_c5_h(h_sym)

    if tleft() > 200:

        if sc_cd < best_score:
