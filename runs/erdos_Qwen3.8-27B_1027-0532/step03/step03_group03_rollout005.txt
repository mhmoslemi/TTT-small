def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # More careful projection
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        # Distribute remaining mass to elements with most room
        room = 1.0 - h[idx]
        total_room = np.sum(room)
        if total_room > 0:
            frac = min(1.0, diff / total_room)
            h[idx] += room * frac
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        mass = h[idx]
        total_mass = np.sum(mass)
        if total_mass > 0:
            frac = min(1.0, -diff / total_mass)
            h[idx] -= mass * frac
    # Final clip and correction
    h = np.clip(h, 0.0, 1.0)
    # If still not exact, do one more pass

            h[idx] += diff2 / len(idx)

def grad_at_lags(h, lags, weights):
    """Compute weighted sum of gradients at given lags."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    offset = n - 1

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        room = 1.0 - h[idx]
        total_room = np.sum(room)
        if total_room > 0:
            frac = min(1.0, diff / total_room)
            h = h.copy()
            h[idx] += room * frac
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        mass = h[idx]
        total_mass = np.sum(mass)
        if total_mass > 0:
            frac = min(1.0, -diff / total_mass)
            h = h.copy()
            h[idx] -= mass * frac
    h = np.clip(h, 0.0, 1.0)
    diff2 = target_sum - np.sum(h)
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx] += diff2 / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_at_lags(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    offset = n - 1
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lags[i] - offset
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad


def smooth_max_obj(x, alpha):
    corr = corr_full(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -400.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha):
    corr = corr_full(x)
    c_max = np.max(corr)
    exp_arr = np.exp(np.clip(alpha * (corr - c_max), -400.0, 0.0))
    total = np.sum(exp_arr)
    if total < 1e-300:
        return np.zeros(len(x))
    weights = exp_arr / total
    n = len(x)
    return grad_at_lags(x, np.arange(2 * n - 1), weights)


def cons_eq(x):
    return np.sum(x) - _global_target


def cons_eq_jac(x):
    return np.ones(_global_n)


def slsqp_polish(h_start, target, alphas, maxiter, ftol):
    global _global_n, _global_target
    _global_n = len(h_start)
    _global_target = target
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _global_n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_eq_jac}]
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for alpha in alphas:
        try:
            res = minimize(smooth_max_obj, h_cur, args=(alpha,),
                           jac=smooth_max_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            if res.x is not None:
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                h_cur = h_new
        except Exception:
            pass
    return best_h


def subgradient_opt(h_init, target, max_iter, lr_init, topk, alpha,
                    patience, momentum, noise_prob, noise_scale, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0

    for it in range(max_iter):
        corr = corr_full(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = grad_at_lags(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 1.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def structured_perturb(h, rng, n_modes, scale):
    n = len(h)
    x = np.linspace(0.0, 2.0, n)
    h_pert = h.copy()
    for k in range(1, n_modes + 1):
        amp = rng.uniform(-scale, scale)
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h_pert += amp * np.sin(k * np.pi * x + phase)
    return project_feasible(h_pert, n / 2.0)


def targeted_perturb(h, scale):
    """Perturb in direction that reduces top bottleneck lags."""
    n = len(h)
    corr = corr_full(h)
    c_max = np.max(corr)
    top_lags = np.where(corr > c_max - 0.05)[0]
    if len(top_lags) == 0:
        top_lags = np.array([np.argmax(corr)])
    offset = n - 1
    grad = np.zeros(n)
    for lag_idx in top_lags:
        lag = lag_idx - offset
        one_minus_h = 1.0 - h
        if lag >= 0 and lag < n:
            grad[lag:] += one_minus_h[:n - lag]
            grad[:n - lag] -= h[lag:]
        elif -lag < n:
            m = -lag
            grad[:n - m] += one_minus_h[m:]
            grad[m:] -= h[:n - m]
    h_pert = h - scale * grad / (np.linalg.norm(grad) + 1e-10)
    return project_feasible(h_pert, n / 2.0)
