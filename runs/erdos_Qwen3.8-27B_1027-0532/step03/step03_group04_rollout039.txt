import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def cutting_plane_lp(h_init, n, target, max_iter, time_limit, num_top=40, damping=0.5):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        corr_vals = compute_all_corr(h)
        
        # Get top num_top shifts
        top_idx = np.argsort(corr_vals)[::-1][:num_top]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            0.0 <= h_var,
            h_var <= 1.0
        ]
        
        for idx in top_idx:
            k = int(idx) - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = g[lo-k:hi-k] * dx
                # Need to map to the right indices in h_var
                c = np.zeros(n)
                c[lo:hi] = coeff
                constraints.append(c @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=100000, verbose=False)
            except Exception:

        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_all_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def inp_str(v):
    return str(v)


def fast_lp_solve(h, n, target, g, top_idx, reg, num_constraints):
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        0.0 <= h_var,
        h_var <= 1.0
    ]
    for idx in top_idx[:num_constraints]:
        k = int(idx) - (n - 1)
        lo = max(0, k)
        hi = min(n, k + n)
        if hi > lo:
            c = np.zeros(n)
            c[lo:hi] = g[lo - k:hi - k] * (2.0 / n)
            constraints.append(c @ h_var <= t_var)
    obj = t_var + reg * cp.sum_squares(h_var - 0.5)
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-7, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iter=50000, verbose=False)
        except Exception:
            return None
    if h_var.value is None:
        return None
    return np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)


def cutting_plane_lp(h_init, n, target, max_iter, time_limit, num_top, damping, reg):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improve = 0
    current_damping = damping
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        corr_vals = compute_all_corr(h)
        top_idx = np.argsort(corr_vals)[::-1][:num_top]
        h_new = fast_lp_solve(h, n, target, g, top_idx, current_reg, num_top)
        if h_new is None:
            break
        h = (1.0 - current_damping) * h + current_damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            current_damping = max(0.1, current_damping * 0.7)
            no_improve = 0
        if it % 200 == 199 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.5)
    return best_h, best_c5


def smooth_obj_fn(h, alpha, target):
    corr = compute_all_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def smooth_grad_fn(h, alpha, target):
    n = len(h)
    dx = 2.0 / n
    corr = compute_all_corr(h)
    c_max = np.max(corr)
    log_w = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    w = np.exp(log_w)
    w_sum = np.sum(w)
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum
    g = np.zeros(n)
    significant = w > 1e-10
    sig_idx = np.where(significant)[0]
    for idx in sig_idx:
        k = int(idx) - (n - 1)
        wi = w[idx] * dx
        if k >= 0:
            if n - k > 0:
                g[0:n - k] += wi * (1.0 - h[k:n])
                g[k:n] += wi * (-h[0:n - k])
        else:
            kk = -k
            if n - kk > 0:
                g[kk:n] += wi * (1.0 - h[0:n - kk])
                g[0:n - kk] += wi * (-h[kk:n])
    return g


def slsqp_solve(h_init, n, target, alphas, maxiter_per, ftol, time_limit):
    start = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    bounds = [(0.0, 1.0)] * n

    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(
                smooth_obj_fn, h, args=(alpha, target),
                jac=lambda v, a, t: smooth_grad_fn(v, a, t),
                method='SLSQP', bounds=bounds,
                options={'maxiter': maxiter_per, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5


def binary_local_search(h_init, n, target, time_limit, max_rounds=50):
    start = time.time()
    h = h_init.copy()
    binary = (h > 0.5).astype(float)
    ones_count = int(np.sum(binary))
    if ones_count > n // 2:
        extra = ones_count - n // 2
        one_positions = np.where(binary == 1.0)[0]
        order = np.argsort(h[one_positions])[::-1]
        for i in range(min(extra, len(order))):
            binary[one_positions[order[i]]] = 0.0
    elif ones_count < n // 2:
        deficit = n // 2 - ones_count
        zero_positions = np.where(binary == 0.0)[0]
        order = np.argsort(h[zero_positions])[::-1]
        for i in range(min(deficit, len(order))):
            binary[zero_positions[order[i]]] = 1.0
    best_binary = binary.copy()
    best_c5 = compute_c5(best_binary)
    dx = 2.0 / n

    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        corr_vals = np.correlate(best_binary, 1.0 - best_binary, mode='full')
        max_corr = np.max(corr_vals)
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = best_binary[i]
            new_val = 1.0 - old_val
            best_binary[i] = new_val
            new_corr = np.correlate(best_binary, 1.0 - best_binary, mode='full')
            new_max = np.max(new_corr)
            if new_max < max_corr - 1e-14:
                max_corr = new_max
                best_c5 = new_max * dx
                improved = True
            else:
                best_binary[i] = old_val
        if not improved:
            break
    return best_binary, best_c5


def pair_flip_search(h_init, n, target, time_limit, max_rounds=30):

    binary = (h_init > 0.5).astype(float)
    ones_count = int(np.sum(binary))
    if ones_count != n // 2:
        if ones_count > n // 2:
            extra = ones_count - n // 2
            one_pos = np.where(binary == 1.0)[0]
            for i in range(extra):
                binary[one_pos[i]] = 0.0
        else:
            deficit = n // 2 - ones_count
            zero_pos = np.where(binary == 0.0)[0]
            for i in range(deficit):
                binary[zero_pos[i]] = 1.0
    best_binary = binary.copy()
    best_c5 = compute_c5(best_binary)
    dx = 2.0 / n
    one_indices = np.where(best_binary == 1.0)[0]
    zero_indices = np.where(best_binary == 0.0)[0]

        if time.time() - start > time_limit:
            break

        max_corr = np.max(np.correlate(best_binary, 1.0 - best_binary, mode='full'))
        for oi in one_indices:
            if time.time() - start > time_limit:
                break
            for zi in zero_indices:
                if oi == zi:
                    continue
                best_binary[oi] = 0.0
                best_binary[zi] = 1.0
                new_max = np.max(np.correlate(best_binary, 1.0 - best_binary, mode='full'))
                if new_max < max_corr - 1e-14:
                    max_corr = new_max
                    best_c5 = new_max * dx

                else:

                    best_binary[zi] = 0.0

def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half
        grad = np.zeros(n)
        k = k_star
        if k >= 0:
            if n - k > 0:
                grad[0:n - k] += (1.0 - h_cur[k:n])
                grad[k:n] += (-h_cur[0:n - k])
        else:
            kk = -k
            if n - kk > 0:
                grad[kk:n] += (1.0 - h_cur[0:n - kk])
                grad[0:n - kk] += (-h_cur[kk:n])
        grad *= dx

        for lr in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                break
    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)

    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
