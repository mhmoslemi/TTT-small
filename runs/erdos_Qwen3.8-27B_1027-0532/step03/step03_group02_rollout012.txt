```python
import time

def construct_full(h_sym):
    m = len(h_sym)
    h_full = np.empty(2 * m)
    h_full[:m] = h_sym
    h_full[m:] = 1.0 - h_sym[::-1]
    return h_full

def c5_score(h_full):
    n = len(h_full)
    dx = 2.0 / n
    corr = np.correlate(h_full, 1.0 - h_full, mode='full')
    return float(np.max(corr)) * dx, corr.astype(float)

def coord_descent_sym(h_sym, max_time, seed=42, noise_base=0.01):

    h_full = construct_full(h_sym)
    corr = np.correlate(h_full, 1.0 - h_full, mode='full').astype(float)
    dx = 2.0 / n
    score = float(np.max(corr)) * dx
    best_score = score
    best_h = h_sym.copy()
    stagnant = 0
    start = time.time()
    
    shifts = np.arange(2 * n - 1) - (n - 1)

        for j in range(m):
            old_val = h_sym[j]
            b = n - 1 - j
            
            js = j + shifts
            jm = j - shifts
            bs = b + shifts
            bm = b - shifts
            
            a_vec = np.zeros(2 * n - 1)
            
            mask = (js >= 0) & (js < n)
            if np.any(mask):
                a_vec[mask] += (1.0 - h_full[js[mask]])
            
            mask = (jm >= 0) & (jm < n)
            if np.any(mask):
                a_vec[mask] -= h_full[jm[mask]]
            
            mask = (bs >= 0) & (bs < n)
            if np.any(mask):
                a_vec[mask] -= (1.0 - h_full[bs[mask]])
            
            mask = (bm >= 0) & (bm < n)
            if np.any(mask):
                a_vec[mask] += h_full[bm[mask]]

                elif slope < -1e-12:
                    lo = mid

                corr += a_vec * (new_val - old_val)
                h_sym[j] = new_val
```

```python
def constrained_metropolis(h_full, max_time, seed=42):
    rng = np.random.RandomState(seed)
    n = len(h_full)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    h_full = h_full.copy()
    corr = np.correlate(h_full, 1.0 - h_full, mode='full').astype(float)
    score = float(np.max(corr)) * dx
    best_score = score
    best_h = h_full.copy()
    stagnant = 0
    start = time.time()
    
    shifts = np.arange(2 * n - 1) - (n - 1)
    eps = 0.01
    
    while time.time() - start < max_time and stagnant < 10000:
        # Pick two indices
        j, k = rng.randint(0, n, size=2)
        if j == k:
            continue
        
        # Transfer eps from j to k
        if h_full[j] < eps:
            continue
        
        # Compute effect on corr
        # Changing h[j] by -eps and h[k] by +eps
        # Effect on corr[idx]: -eps * a_j[idx] + eps * a_k[idx]
        
        js = j + shifts
        jm = j - shifts
        ks = k + shifts
        km = k - shifts
        
        a_j = np.zeros(2 * n - 1)
        mask = (js >= 0) & (js < n)
        if np.any(mask):
            a_j[mask] += (1.0 - h_full[js[mask]])
        mask = (jm >= 0) & (jm < n)
        if np.any(mask):
            a_j[mask] -= h_full[jm[mask]]
        
        a_k = np.zeros(2 * n - 1)
        mask = (ks >= 0) & (ks < n)
        if np.any(mask):
            a_k[mask] += (1.0 - h_full[ks[mask]])
        mask = (km >= 0) & (km < n)
        if np.any(mask):
            a_k[mask] -= h_full[km[mask]]
        
        delta_corr = eps * (a_k - a_j)
        new_corr = corr + delta_corr
        new_score = float(np.max(new_corr)) * dx
        
        if new_score < score:
            corr = new_corr
            h_full[j] -= eps
            h_full[k] += eps
            if new_score < best_score:
                best_score = new_score
                best_h = h_full.copy()
                stagnant = 0
        else:
            stagnant += 1
        
        # Occasionally add noise
        if stagnant % 500 == 0:
```

def paired_coord_descent(h_full, max_time, seed=42):
    """Coordinate descent paired moves to maintain sum constraint."""
    rng = np.random.RandomState(seed)
    n = len(h_full)
    target_sum = n / 2.0
    dx = 2.0 / n
    
    h_full = h_full.copy()
    corr = np.correlate(h_full, 1.0 - h_full, mode='full').astype(float)
    score = float(np.max(corr)) * dx
    best_score = score
    best_h = h_full.copy()
    stagnant = 0
    start = time.time()
    
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    def compute_a(idx):
        """Compute gradient vector for a single index."""
        s_plus = idx + shifts
        s_minus = idx - shifts
        a = np.zeros(2 * n - 1)
        mask = (s_plus >= 0) & (s_plus < n)
        if np.any(mask):
            a[mask] += (1.0 - h_full[s_plus[mask]])
        mask = (s_minus >= 0) & (s_minus < n)
        if np.any(mask):
            a[mask] -= h_full[s_minus[mask]]
        return a

        improved = False
        for j in range(n):
            a_j = compute_a(j)

            # Change: h[j] += δ, h[k] -= δ (or vice versa)
            # Effect on corr: δ * (a_j - a_k) or δ * (a_k - a_j)
            # We want to minimize max(corr + δ*(a_j - a_k))
            
            # For efficiency, try a few candidate k values
            # The best k is the one where a_k is most "opposite" to a_j
            # at the bottleneck shift

            # Or try a few random k's
            candidates = [(j + n // 2) % n]
            for _ in range(5):
                candidates.append(rng.randint(0, n))
            
            best_delta = 0.0

                if k == j:
                    continue
                a_k = compute_a(k)
                # Try both directions: h[j]+δ, h[k]-δ and h[j]-δ, h[k]+δ
                for direction in [1, -1]:

                    # This is convex in δ, find by binary search

                    if direction == 1:
                        delta_max = min(1.0 - h_full[j], h_full[k])
                    else:
                        delta_max = min(1.0 - h_full[k], h_full[j])
                    
                    if delta_max < 1e-12:
                        continue
                    
                    lo, hi = 0.0, delta_max
                    for _ in range(25):

                        vals = corr + mid * d_vec
