def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)

    if current_sum == target_sum:
        return h_r.astype(float)

import numpy as np
from scipy.optimize import minimize
import time

_cons_n = 0
_cons_target = 0.0

def fast_correlate(a, b):
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    size = np.fft.next_fast_len(out_len)
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    result = np.fft.irfft(fa * np.conj(fb), size)
    return np.concatenate([result[size - m + 1:], result[:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_correlate(h, 1.0 - h)
    return float(np.max(corr)) * dx

def compute_corr(h):
    return fast_correlate(h, 1.0 - h)

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = int(h_r.sum())
    target_int = int(round(target_sum))
    if current_sum == target_int:
        return h_r.astype(float)
    if current_sum > target_int:
        need = current_sum - target_int
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_int - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    return h_r.astype(float)


def greedy_peak_attack(h, target_sum, max_iter=100000, time_limit=None, start_time=None, rng=None):
    """Local search that specifically targets the highest correlation peak."""
    n = len(h)
    h = project_binary(h, target_sum).astype(int)
    corr = compute_corr(h)
    curr_c5 = float(np.max(corr)) * (2.0 / n)
    best_c5 = curr_c5
    best_h = h.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > 100000:
            break
            
        # Find the peak
        max_pos = np.argmax(corr)
        s_star = max_pos - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Find i: h[i]=1 and (h[i+s_star]=0 if in range)
        i_pos = ones + s_star
        valid_i = (i_pos >= 0) & (i_pos < n)
        i_cand_all = ones[valid_i]
        # Prefer those where h[i+s_star]=0 (reduces the peak)
        i_good = i_cand_all[h[i_pos[valid_i]] == 0]
        if len(i_good) == 0:
            i_good = i_cand_all if len(i_cand_all) > 0 else ones
        
        # Find j: h[j]=0 and (h[j+s_star]=1 if in range)
        j_pos = zeros + s_star
        valid_j = (j_pos >= 0) & (j_pos < n)
        j_cand_all = zeros[valid_j]
        j_good = j_cand_all[h[j_pos[valid_j]] == 1]
        if len(j_good) == 0:
            j_good = j_cand_all if len(j_cand_all) > 0 else zeros
        
        # Try multiple candidates
        n_tries = min(30, max(5, len(i_good) + len(j_good)))
        found = False
        for _ in range(n_tries):
            i = i_good[rng.randint(len(i_good))]
            j = j_good[rng.randint(len(j_good))]
            if i == j:
                continue
            h[i] = 0
            h[j] = 1
            corr = compute_corr(h)
            new_c5 = float(np.max(corr)) * (2.0 / n)
            if new_c5 < curr_c5:
                curr_c5 = new_c5
                found = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            else:
                h[i] = 1
                h[j] = 0
                no_improve += 1
        
        if not found:
            # Escalate: try random moves
            n_esc = 5
            for _ in range(n_esc):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0
                h[j] = 1
                corr = compute_corr(h)
                new_c5 = float(np.max(corr)) * (2.0 / n)
                if new_c5 < curr_c5:
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                    found = True
                    break
                else:
                    h[i] = 1
                    h[j] = 0
                    no_improve += 1
    
    return best_h.astype(float), best_c5

              time_limit=None, start_time=None):
    """SA with a mix of targeted and random moves."""
    n = len(h_start)
    h = project_binary(h_start, target_sum).astype(int)
    corr = compute_corr(h)
    current_c5 = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    best_c5 = current_c5
    no_improve = 0
    max_no_improve = max(5000, max_iters // 5)

    for i in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve and rng.random() > 0.03:
            break

        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        if t < t_init * 0.3:
            # Cool phase: targeted moves
            max_pos = np.argmax(corr)

                j = j_good[rng.randint(len(j_good))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]

                j = zeros[rng.randint(len(zeros))]

                        no_improve = 0
