Now, dh[i]/dv[j]:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def build_from_half(v):
    m = len(v)
    n = 2 * m
    h = np.empty(n)
    h[:m] = v
    h[m:] = 1.0 - v[::-1]
    return h

def smooth_obj_grad_sym(v, alpha):
    m = len(v)
    n = 2 * m
    h = build_from_half(v)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_sh = np.exp(shifted)
    s = np.sum(exp_sh)
    w = exp_sh / s
    obj = (c_max + np.log(s) / alpha) * dx
    grad = np.zeros(m)
    n_corr = 2 * n - 1
    k_lags = np.arange(n_corr) - (n - 1)
    for j in range(m):
        idx1 = j + k_lags
        m1 = (idx1 >= 0) & (idx1 < n)
        t = np.zeros(n_corr)
        t[m1] = 1.0 - h[idx1[m1]]
        idx2 = n - 1 - j + k_lags
        m2 = (idx2 >= 0) & (idx2 < n)
        t[m2] -= (1.0 - h[idx2[m2]])
        idx3 = j - k_lags
        m3 = (idx3 >= 0) & (idx3 < n)
        t[m3] -= h[idx3[m3]]
        idx4 = n - 1 - j - k_lags
        m4 = (idx4 >= 0) & (idx4 < n)
        t[m4] += h[idx4[m4]]
        grad[j] = np.dot(w, t)
    grad *= dx
    return obj, grad

def coord_descend(h, target_sum, max_time, rng, patience=30):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = h.copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()
    while time.time() - elapsed_start < max_time:
        for j in range(n):
            old_val = h[j]
            js = j - s_arr
            js_plus = j + s_arr
            A = np.zeros(2 * n - 1)
            mask1 = (js >= 0) & (js < n)
            mask2 = (js_plus >= 0) & (js_plus < n)
            A[mask1] += (1.0 - h[js[mask1]])
            A[mask2] -= h[js_plus[mask2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-10:
                    hi = mid
                elif grad < -1e-10:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            noise_scale = 0.003 * min(1.0, no_improve / patience)
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    return best_h, best_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slb_obj(x, a_val):
    return smooth_obj_grad_sym(x, a_val)[0]

def slb_grad(x, a_val):
    return smooth_obj_grad_sym(x, a_val)[1]

def optimize_sym_lbfgs(v_init, alphas, maxiter_each, start_time, time_budget):
    m = len(v_init)
    bounds = [(0.0, 1.0)] * m
    v_cur = v_init.copy()
    best_c5 = compute_c5_full(build_from_half(v_init))
    best_v = v_init.copy()
    for alpha in alphas:
        if time.time() - start_time > time_budget:
            break
        try:
            res = minimize(slb_obj, v_cur, args=(alpha,), jac=slb_grad,
                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': maxiter_each, 'ftol': 1e-18, 'gtol': 1e-12})
            v_new = np.clip(res.x, 0.0, 1.0)
            h_new = build_from_half(v_new)
            c5_new = compute_c5_full(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_v = v_new.copy()
            v_cur = v_new
        except Exception:
            pass
    return best_v, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5_full(best_h)

    # Phase 1: Symmetric L-BFGS-B with increasing alpha
    if tleft() > 300 and n_init % 2 == 0:
        m = n_init // 2
        v_init = np.clip(h_init[:m].copy(), 0.0, 1.0)
        alphas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000]
        v_best, c5_sym = optimize_sym_lbfgs(v_init, alphas, 300, start_time, min(250, tleft() - 50))
        h_sym = build_from_half(v_best)
        if c5_sym < best_score:
            best_score = c5_sym
            best_h = h_sym.copy()

    # Phase 2: Asymmetric coordinate descent refinement
    if tleft() > 200:

        tgt_cur = n_cur / 2.0
        h_cd, sc_cd = coord_descend(best_h, tgt_cur, min(300, tleft() - 200), rng, patience=25)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    # Phase 3: Multi-resolution exploration
    if tleft() > 250:
        n_cur = len(best_h)
        offsets = [-100, -50, 50, 100, 200, 300, 400]
        n_attempts = [n_cur + d for d in offsets if n_cur + d >= 100 and n_cur + d != n_cur]
        for n_try in n_attempts:
            if tleft() < 150:
                break
            tgt_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, tgt_try)

            if sc_cd2 < best_score:

    # Phase 4: Second symmetric L-BFGS pass on best solution
    n_cur = len(best_h)
    if n_cur % 2 == 0 and tleft() > 100:
        m_cur = n_cur // 2
        v_cur = np.clip(best_h[:m_cur].copy(), 0.0, 1.0)

        v_best2, c5_sym2 = optimize_sym_lbfgs(v_cur, alphas2, 400, start_time, min(120, tleft() - 60))

        if c5_sym2 < best_score:

            best_h = h_sym2.copy()

    # Phase 5: Final asymmetric coordinate descent polish

    tgt_cur = n_cur / 2.0

    # Final feasibility check

    target_final = n_final / 2.0
    h_projeted = project_feasible(best_h, target_final)
    final_score = compute_c5_full(h_projeted)
