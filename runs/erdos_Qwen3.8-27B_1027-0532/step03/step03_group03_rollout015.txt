def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:

def es_optimize(h_init, n, target, budget, rng, sigma_init=0.1, mu=10, lam=30):
    """Evolutionary strategy"""
    pop = np.zeros((lam, n))
    pop[0] = h_init.copy()
    for i in range(1, lam):
        pop[i] = project_feasible(h_init + sigma_init * rng.randn(n), target)
    
    fitness = np.array([compute_c5(pop[i]) for i in range(lam)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]
    
    sigma = sigma_init
    start = time.time()
    gen = 0
    
    while time.time() - start < budget:
        gen += 1
        # Select top mu
        order = np.argsort(fitness)
        parents = pop[order[:mu]]
        
        # Generate children
        children = np.zeros((lam, n))
        for i in range(lam):
            p = parents[i % mu]
            mutation = sigma * rng.randn(n)
            children[i] = project_feasible(p + mutation, target)
        
        # Evaluate
        for i in range(lam):
            fitness[i] = compute_c5(children[i])
            if fitness[i] < best_score:
                best_score = fitness[i]
                best_h = children[i].copy()
        
        pop = children
        
        # Adaptive sigma
        if gen % 10 == 0:
            top5 = np.sort(fitness)[:5]
            bottom5 = np.sort(fitness)[-5:]
            if top5[-1] > bottom5[0]:
                # No improvement, reduce sigma
                sigma *= 0.9
            else:
                sigma *= 1.05
            sigma = np.clip(sigma, 0.001, 0.5)
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

_SLSQP_N = 0
_SLSQP_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def compute_O(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def slsqp_eq_fun(x):
    return np.sum(x) - _SLSQP_TARGET


def slsqp_eq_jac(x):
    return np.ones(_SLSQP_N)


def topk_smooth_obj(x, k, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    O = corr * dx
    topk = np.partition(O, -k)[-k:]
    m = np.max(topk)
    shifted = np.clip(alpha * (topk - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def topk_smooth_grad(x, k, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    O = corr * dx
    topk_idx = np.argpartition(O, -k)[-k:]
    topk_vals = O[topk_idx]
    m = np.max(topk_vals)
    exp_s = np.exp(np.clip(alpha * (topk_vals - m), -500.0, 0.0))
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for idx in range(k):
        wi = w[idx]
        lag = topk_idx[idx] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * (1.0 - x[:n - lag])
                grad[:n - lag] -= wi * x[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += wi * (1.0 - x[m2:])
                grad[m2:] -= wi * x[:n - m2]
    return grad * dx


def full_smooth_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    O = corr * dx
    m = np.max(O)
    shifted = np.clip(alpha * (O - m), -500.0, 0.0)
    return m + np.log(np.sum(np.exp(shifted))) / alpha


def full_smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    O = corr * dx
    m = np.max(O)
    exp_s = np.exp(np.clip(alpha * (O - m), -500.0, 0.0))
    w = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for idx in range(len(O)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * (1.0 - x[:n - lag])
                grad[:n - lag] -= wi * x[lag:]
        else:
            m2 = -lag
            if m2 < n:
                grad[:n - m2] += wi * (1.0 - x[m2:])
                grad[m2:] -= wi * x[:n - m2]
    return grad * dx


def find_opt_delta(O, B, dx, h_i):
    lo = -h_i
    hi = 1.0 - h_i
    if hi - lo < 1e-15:
        return 0.0
    for _ in range(30):
        mid = (lo + hi) * 0.5
        vals = O + dx * mid * B
        max_val = np.max(vals)
        tol = 1e-12 * max(1.0, abs(max_val))
        active = np.where(vals >= max_val - tol)[0]
        b_max = np.max(B[active])
        b_min = np.min(B[active])
        if b_min <= 0.0 <= b_max:
            return mid
        elif b_min > 0.0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5


def compute_B(h, i, n):
    B = np.zeros(2 * n - 1)
    B[n - 1] = 1.0 - 2.0 * h[i]
    lpos = np.arange(1, n)
    valid_left = lpos <= i
    valid_right = lpos <= (n - 1 - i)
    if np.any(valid_left):
        vl = lpos[valid_left]
        B[n - 1 + vl] += (1.0 - h[i - vl])
        B[n - 1 - vl] -= h[i - vl]
    if np.any(valid_right):
        vr = lpos[valid_right]
        B[n - 1 + vr] -= h[i + vr]
        B[n - 1 - vr] += (1.0 - h[i + vr])
    return B


def coord_desc_optimize(h, n, rng, max_sweeps, t0, budget):
    dx = 2.0 / n
    target = n / 2.0
    O = compute_O(h)
    best_c5 = float(np.max(O))
    sweep = 0
    while sweep < max_sweeps:
        if time.time() - t0 > budget - 30.0:
            break
        order = rng.permutation(n)
        for i in order:
            B = compute_B(h, i, n)
            delta = find_opt_delta(O, B, dx, h[i])
            new_hi = h[i] + delta
            new_hi = np.clip(new_hi, 0.0, 1.0)
            actual_delta = new_hi - h[i]
            if abs(actual_delta) > 1e-14:
                h[i] = new_hi
                O += dx * actual_delta * B
        h = project_feasible(h, target)
        O = compute_O(h)
        c5 = float(np.max(O))
        if c5 < best_c5:
            best_c5 = c5
        sweep += 1
    return h, best_c5


def es_optimize(h_init, n, target, t0, budget, rng, sigma_init=0.05, mu=8, lam=20):
    pop = np.zeros((lam, n))
    pop[0] = h_init.copy()
    for i in range(1, lam):
        pop[i] = project_feasible(h_init + sigma_init * rng.randn(n), target)
    fitness = np.array([compute_c5(pop[i]) for i in range(lam)])
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]
    sigma = sigma_init
    gen = 0
    while time.time() - t0 < budget - 30.0:
        gen += 1
        order = np.argsort(fitness)
        parents = pop[order[:mu]]
        for i in range(lam):
            p = parents[i % mu]
            pop[i] = project_feasible(p + sigma * rng.randn(n), target)
        for i in range(lam):
            fitness[i] = compute_c5(pop[i])
            if fitness[i] < best_score:
                best_score = fitness[i]
                best_h = pop[i].copy()
        if gen % 5 == 0:
            sorted_f = np.sort(fitness)
            if sorted_f[mu - 1] < fitness[best_idx] + 1e-6:
                sigma = max(sigma * 0.85, 0.001)
            else:
                sigma = min(sigma * 1.1, 0.3)
    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_run(h, n, target, k, alpha, maxiter, ftol):
    global _SLSQP_N, _SLSQP_TARGET
    _SLSQP_N = n
    _SLSQP_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    try:
        res = minimize(topk_smooth_obj, h, args=(k, alpha),
                       jac=topk_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def slsqp_full_run(h, n, target, alpha, maxiter, ftol):
    global _SLSQP_N, _SLSQP_TARGET
    _SLSQP_N = n
    _SLSQP_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_eq_fun, 'jac': slsqp_eq_jac}]
    try:
        res = minimize(full_smooth_obj, h, args=(alpha,),
                       jac=full_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h


def run(seed=42, budget_s=1300, **kwargs):
    global _SLSQP_N, _SLSQP_TARGET
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    h = project_feasible(h_init, target)
    best_h = h.copy()
    best_score = compute_c5(h)

    def rem():
        return budget_s - 25.0 - (time.time() - t0)

    # Phase 1: SLSQP with top-k, increasing alpha
    slsqp_configs = [
        (3, 50, 3000, 1e-16),
        (5, 100, 3000, 1e-16),
        (3, 300, 3000, 1e-16),
        (5, 500, 3000, 1e-16),
        (3, 1000, 3000, 1e-16),
        (5, 2000, 3000, 1e-16),
    ]
    for cfg in slsqp_configs:
        if rem() < 80:
            break
        h_new = slsqp_run(h, n, target, cfg[0], cfg[1], cfg[2], cfg[3])
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
            h = h_new

    # Phase 2: Coordinate descent
    if rem() > 100:
        h_cd = best_h.copy()
        h_cd, sc_cd = coord_desc_optimize(h_cd, n, rng, 5000, t0, budget_s)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            h = h_cd

    # Phase 3: SLSQP after CD
    if rem() > 80:
        for cfg in [(3, 500, 2000, 1e-16), (5, 1000, 2000, 1e-16)]:
            if rem() < 50:
                break
            h_new = slsqp_run(h, n, target, cfg[0], cfg[1], cfg[2], cfg[3])
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h = h_new

    # Phase 4: More coordinate descent
    if rem() > 100:
        h_cd = best_h.copy()
        h_cd, sc_cd = coord_desc_optimize(h_cd, n, rng, 5000, t0, budget_s)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd
            h = h_cd

    # Phase 5: ES for diversity
    if rem() > 150:
        h_es = best_h + rng.randn(n) * 0.02
        h_es = project_feasible(h_es, target)
        h_es, sc_es = es_optimize(h_es, n, target, t0, budget_s, rng,
                                   sigma_init=0.03, mu=8, lam=20)
        if sc_es < best_score:
            best_score = sc_es
            best_h = h_es
            h = h_es
        if rem() > 80:
            h_cd = best_h.copy()
            h_cd, sc_cd = coord_desc_optimize(h_cd, n, rng, 2000, t0, budget_s)
            if sc_cd < best_score:
                best_score = sc_cd
                best_h = h_cd

    # Phase 6: Higher resolution
    for n_try in [500, 600, 800]:
        if rem() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        # SLSQP at higher res
        if rem() > 80:
            h_interp = slsqp_run(h_interp, n_try, target_try, 3, 200, 2000, 1e-16)
            h_interp = slsqp_run(h_interp, n_try, target_try, 5, 500, 2000, 1e-16)
        # CD at higher res
        if rem() > 60:
            h_interp, sc_hi = coord_desc_optimize(h_interp, n_try, rng,
                                                   max(500, int(rem() / 0.012)),
                                                   t0, budget_s)
            if sc_hi < best_score:
                best_score = sc_hi
                best_h = h_interp

    # Phase 7: Final refinement on best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    if rem() > 60:
        h_fin = slsqp_run(best_h, n_use, target_use, 3, 3000, 2000, 1e-17)
        sc = compute_c5(h_fin)
        if sc < best_score:
            best_score = sc
            best_h = h_fin
    if rem() > 40:
        h_cd = best_h.copy()
        h_cd, sc_cd = coord_desc_optimize(h_cd, n_use, rng, 3000, t0, budget_s)
        if sc_cd < best_score:
            best_score = sc_cd
            best_h = h_cd

    best_h = project_feasible(best_h, target_use)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
