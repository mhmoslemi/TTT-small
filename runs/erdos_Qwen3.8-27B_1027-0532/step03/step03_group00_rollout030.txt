    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k < n:
            grad[0:n-k] -= h[k:]
            grad[k:n] += (1.0 - h[0:n-k])
    else:

        if kk < n:

            grad[0:n-kk] += (1.0 - h[kk:n])
    return grad

def active_subgradient(h, active_frac=0.05):
    """Compute subgradient of max(corr) using active set."""
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    
    # Active set: lags within active_frac of the range
    corr_range = c_max - float(np.min(corr))
    if corr_range < 1e-10:
        # All lags are at max
        threshold = c_max - 1e-10
    else:
        threshold = c_max - active_frac * corr_range
    
    active_indices = np.where(corr >= threshold)[0]
    
    # Convert indices to actual lags
    lags = active_indices - (n - 1)
    
    # Average gradient over active lags
    grad = np.zeros(n)
    for lag in lags:
        grad += corr_grad_single(h, lag)
    grad /= len(lags)
    
    return grad, c_max, corr

def lse_objective_and_grad(h, alpha):
    """LSE smoothed objective and its gradient."""
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_shifted = np.exp(shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    obj = c_max + np.log(np.sum(exp_shifted)) / alpha

    grad = np.zeros(n)
    lags = np.arange(-(n-1), n)
    for i, lag in enumerate(lags):

            grad += weights[i] * corr_grad_single(h, lag)
    
    return obj, grad

    h_new = h - step_size * grad
    # Project onto {0 <= h <= 1, sum(h) = target}
    h_new = project_feasible(h_new, target)
    return h_new

    best_h = h_cur.copy()

        h_new = projected_subgradient_step(h_cur, grad, target, step)

            best_h = h_new.copy()

        return lse_obj_and_grad(x, alpha)[1]

def active_subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    """Polish using active-set subgradient."""
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n-k] -= h[k:]
                    grad[k:] += (1.0 - h[:n-k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h[:n-kk]
                    grad[:n-kk] += (1.0 - h[kk:])
        grad /= len(active)

        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()

- d(c[1])/d(b) = (1-a) - c = (1-h[0]) - h[2] ✓ (j=1: (1-h[1-1]) - h[1+1] = (1-h[0]) - h[2])
- d(c[1])/d(c) = (1-b) = (1-h[1]) ✓ (j=2: (1-h[2-1]) - 0 = (1-h[1]))

import numpy as np
from scipy.optimize import minimize
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def lse_obj_and_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_v = np.exp(shifted)
    sum_exp = np.sum(exp_v)
    w = exp_v / sum_exp
    obj = c_max + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            if k < n:
                grad[:n - k] -= wi * h[k:]
                grad[k:] += wi * (1.0 - h[:n - k])
        else:
            kk = -k
            if kk < n:
                grad[kk:] -= wi * h[:n - kk]
                grad[:n - kk] += wi * (1.0 - h[kk:])
    return obj, grad


def _slsqp_one_alpha(h, target, alpha, n, bounds, maxiter, ftol):
    cur_score = compute_c5(h)

    def fun(x):
        return lse_obj_and_grad(x, alpha)[0]

    def jac(x):
        return lse_obj_and_grad(x, alpha)[1]

    def cfun(x):
        return np.sum(x) - target

    def cjac(x):
        return np.ones(n)

    try:
        res = minimize(fun, h, jac=jac, method='SLSQP',
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < cur_score:
            return h_new, sc
    except Exception:
        pass
    return h, cur_score


def slsqp_refine(h, target, alpha_list, maxiter, ftol, deadline):
    n = len(h)
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_list:
        if time.time() > deadline:
            break
        h_cur, cur_score = _slsqp_one_alpha(h_cur, target, alpha, n, bounds, maxiter, ftol)
    return h_cur, cur_score


def active_subgrad_polish(h, target, n_steps, step_size, active_frac, deadline):
    n = len(h)
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    for it in range(n_steps):
        if time.time() > deadline:
            break
        corr = compute_corr(h_cur)
        c_max = float(np.max(corr))
        c_min = float(np.min(corr))
        threshold = c_max - active_frac * (c_max - c_min)
        active = np.where(corr >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            if k >= 0:
                if k < n:
                    grad[:n - k] -= h_cur[k:]
                    grad[k:] += (1.0 - h_cur[:n - k])
            else:
                kk = -k
                if kk < n:
                    grad[kk:] -= h_cur[:n - kk]
                    grad[:n - kk] += (1.0 - h_cur[kk:])
        grad /= len(active)
        step = step_size / (1.0 + 0.02 * it)
        h_new = h_cur - step * grad
        h_new = project_feasible(h_new, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_score


def block_descent(h, n, target, block_size, n_rounds, alpha, deadline):
    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    n_blocks = n // block_size
    for rd in range(n_rounds):
        if time.time() > deadline:
            break
        for bi in range(n_blocks):
            if time.time() > deadline:
                break
            s = bi * block_size
            e = s + block_size
            h_fixed = h_cur.copy()
            x0 = h_cur[s:e].copy()

            def block_fun(x):
                h_temp = h_fixed.copy()
                h_temp[s:e] = np.clip(x, 0.0, 1.0)
                h_temp = project_feasible(h_temp, target)
                return lse_obj_and_grad(h_temp, alpha)[0]

            try:
                res = minimize(block_fun, x0, method='Nelder-Mead',
                               options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-14})
                h_test = h_cur.copy()
                h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                h_test = project_feasible(h_test, target)
                sc = compute_c5(h_test)
                if sc < cur_score:
                    h_cur = h_test
                    cur_score = sc
            except Exception:
                pass
    return h_cur, cur_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
