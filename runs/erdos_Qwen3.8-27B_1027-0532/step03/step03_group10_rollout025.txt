def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def multi_start_binary(n, rng, n_starts=10):
    """Generate diverse binary initializations"""
    starts = []
    for r in range(n_starts):
        if r == 0:
            # Alternating
            h = np.zeros(n)
            h[::2] = 1.0
            if h.sum() != n / 2:
                h[n//2] = 0  # adjust
                h[0] = 1
            starts.append(h)
        elif r == 1:
            # Anti-alternating (shifted)
            h = np.zeros(n)
            h[1::2] = 1.0
            if h.sum() != n / 2:
                h[n//2] = 1
                h[0] = 0
            starts.append(h)
        elif r == 2:
            # Block pattern: 11001100...
            pattern = np.array([1, 1, 0, 0], dtype=float)
            h = np.tile(pattern, n // 4 + 1)[:n]
            if h.sum() != n / 2:
                # Adjust
                if h.sum() > n / 2:
                    idx = np.where(h == 1)[0]
                    h[idx[0]] = 0
                else:
                    idx = np.where(h == 0)[0]
                    h[idx[0]] = 1
            starts.append(h)
        elif r == 3:
            # Sinusoidal threshold
            x = np.linspace(0, 1, n, endpoint=False)
            h = (x < 0.5).astype(float)
            # Perturb
            h += 0.1 * rng.standard_normal(n)
            h = to_binary(h)
            starts.append(h)
        else:
            # Random with bias
            alpha = 0.3 + 0.4 * r / max(n_starts - 4, 1)
            h_rand = (rng.random(n) < 0.5).astype(float)
            h = to_binary(alpha * h_rand + (1-alpha) * rng.random(n))
            starts.append(h)
    return starts

    cmax = np.max(corr)

    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):

            n_steps_quick = min(n_steps_quick, 300000)
            h_try_bin, bin_max = binary_sa(h_try_bin, n_steps_quick, 0.15, 0.01, rng)
            c5_try = (2.0 / n_try) * bin_max
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try_bin.copy()
                n0 = n_try
                target_sum = n0 / 2.0
                if elapsed() < budget_s * 0.9:
                    h_pol = h_try_bin + 0.008 * rng.standard_normal(n0)
                    h_pol = project_onto_constraint(h_pol, target_sum)
                    h_pol, c5 = slsqp_optimize(h_pol, 0.0003, 200)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_pol.copy()
    
    # Phase 4: Final coordinate descent on binary
    if elapsed() < budget_s * 0.9:
        h_bin = to_binary(best_h)
        h_bin, cd_max = coordinate_descent(h_bin, n_passes=3)
        c5_cd = (2.0 / len(h_bin)) * cd_max
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_bin.copy()
    
    # Phase 5: Final SLSQP polishing
    if elapsed() < budget_s * 0.97:
        n0 = len(best_h)
        target_sum = n0 / 2.0
        h_fin = best_h + 0.002 * rng.standard_normal(n0)
        h_fin = project_onto_constraint(h_fin, target_sum)
        for eps in [0.0002, 0.00005, 0.00001]:
            if elapsed() > budget_s * 0.98:
                break
            h_fin, c5 = slsqp_optimize(h_fin, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_fin.copy()
    
    # Ensure exact constraint satisfaction
    n0 = len(best_h)
    final_h = project_onto_constraint(best_h, n0 / 2.0)
    final_c5 = compute_c5(final_h)
    
    return final_h, final_c5, len(final_h)
