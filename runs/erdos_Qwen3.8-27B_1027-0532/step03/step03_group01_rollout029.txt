def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

import numpy as np
from scipy.optimize import minimize

_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def softmax_subgradient_step(h, target_sum, lr, alpha_smooth, rng):
    n = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)

    shifted = alpha_smooth * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)

    grad = np.zeros(n)
    n1 = n - 1

    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        m = k - n1
        idx = np.arange(n)
        partner1 = idx + m
        valid1 = (partner1 >= 0) & (partner1 < n)
        grad[valid1] += weights[k] * (1.0 - h[partner1[valid1]])
        partner2 = idx - m
        valid2 = (partner2 >= 0) & (partner2 < n)
        grad[valid2] -= weights[k] * h[partner2[valid2]]

    grad -= np.mean(grad)

    h_new = h - lr * grad
    h_new = project_feasible(h_new, target_sum)
    return h_new


def subgradient_optimize(h_start, target_sum, max_iter, lr_init, alpha_smooth, rng):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init

    for i in range(max_iter):
        h = softmax_subgradient_step(h, target_sum, lr, alpha_smooth, rng)
        c5_cur = compute_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        lr *= 0.998
        if i % 50 == 49:
            h = project_feasible(h, target_sum)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    def smooth_obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 35.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_score:
        best_score = c5_bin
        best_h = h_bin.copy()

    # Phase 1: Intensive SA from initial
    h_sa = h_bin.copy()
    sa_configs = [
        (120000, 0.020, 0.00005),
        (100000, 0.012, 0.00008),
        (80000, 0.025, 0.00015),
        (150000, 0.008, 0.00002),
        (100000, 0.015, 0.00006),
    ]

    for ci, (max_it, t_i, t_f) in enumerate(sa_configs):
        if tleft() < 80:
            break
        allowed_time = min(200.0, tleft() * 0.3)
        h_sa, c5_sa = sa_optimize(h_sa, target_init, max_it, t_i, t_f,
                                  rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

        if tleft() > 40:
            h_hc, c5_hc = fast_hill_climb(h_sa, target_init, 10000, rng)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()
                h_sa = h_hc

    # Phase 2: Diverse starting points
    # Thue-Morse based
    if tleft() > 60:
        h_tm = thue_morse_binary(n_init, target_init)
        allowed_time = min(120.0, tleft() * 0.2)
        max_it = int(min(80000, allowed_time * 2500))
        h_sa2, c5_sa2 = sa_optimize(h_tm, target_init, max_it, 0.015, 0.0001,
                                    rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()
        if tleft() > 30:
            h_hc2, c5_hc2 = fast_hill_climb(h_sa2, target_init, 5000, rng)
            if c5_hc2 < best_score:
                best_score = c5_hc2
                best_h = h_hc2.copy()

    # Block-pattern based
    for rs in range(2):
        if tleft() < 50:
            break
        block_size = rng.randint(4, 12)
        h_pat = np.zeros(n_init)
        phase = rng.randint(block_size)
        for i in range(n_init):
            if ((i + phase) // block_size) % 2 == 0:
                h_pat[i] = 1.0
        h_pat = project_binary(h_pat, target_init)
        allowed_time = min(100.0, tleft() * 0.2)
        max_it = int(min(60000, allowed_time * 2000))
        h_sa3, c5_sa3 = sa_optimize(h_pat, target_init, max_it, 0.015, 0.0001,
                                    rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa3 < best_score:
            best_score = c5_sa3
            best_h = h_sa3.copy()
        if tleft() > 25:
            h_hc3, c5_hc3 = fast_hill_climb(h_sa3, target_init, 4000, rng)
            if c5_hc3 < best_score:
                best_score = c5_hc3
                best_h = h_hc3.copy()

    # Random perturbation of best
    for rp in range(2):
        if tleft() < 50:
            break
        noise_frac = rng.choice([0.05, 0.1, 0.15])
        h_pert = best_h.copy()
        n_flip = int(n_init * noise_frac)
        flip_idx = rng.choice(n_init, n_flip, replace=False)
        h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_binary(h_pert, target_init)
        allowed_time = min(100.0, tleft() * 0.2)
        max_it = int(min(60000, allowed_time * 2000))
        h_sa4, c5_sa4 = sa_optimize(h_pert, target_init, max_it, 0.010, 0.00005,
                                    rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa4 < best_score:
            best_score = c5_sa4
            best_h = h_sa4.copy()
        if tleft() > 25:
            h_hc4, c5_hc4 = fast_hill_climb(h_sa4, target_init, 4000, rng)
            if c5_hc4 < best_score:
                best_score = c5_hc4
                best_h = h_hc4.copy()

    # Phase 3: Multi-resolution exploration
    n_candidates = [800, 1000, 700, 1200]
    for n_try in n_candidates:
        if tleft() < 60:
            break
        if n_try == n_init or n_try < 200:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)

        allowed_time = min(100.0, tleft() * 0.25)
        max_it = int(min(50000, allowed_time * 1500))
        h_sa_r, c5_r = sa_optimize(h_bin_try, target_try, max_it, 0.012, 0.0001,
                                   rng, time_limit=allowed_time, start_time=start_time)

        if tleft() > 25:
            h_hc_r, c5_hc_r = fast_hill_climb(h_sa_r, target_try, 4000, rng)
            if c5_hc_r < c5_r:
                h_sa_r = h_hc_r
                c5_r = c5_hc_r

        if c5_r < best_score:
            best_score = c5_r
            best_h = h_sa_r.copy()

    # Phase 4: Subgradient descent on continuous relaxation
    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 60:
        h_cont = best_h.copy()
        for lr_start in [0.03, 0.015, 0.008]:
            if tleft() < 30:
                break
            h_sg, c5_sg = subgradient_optimize(h_cont, target_use, 3000,
                                               lr_start, 10.0, rng)
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg.copy()
            h_cont = h_sg

        # Also try with higher smoothing
        if tleft() > 30:
            h_sg2, c5_sg2 = subgradient_optimize(h_cont, target_use, 2000,
                                                 0.02, 30.0, rng)
            if c5_sg2 < best_score:
                best_score = c5_sg2
                best_h = h_sg2.copy()

    # Phase 5: SLSQP refinement
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60:
        alphas = [15, 50, 150, 400, 1000, 2500]
        h_ref, sc = slsqp_refine(best_h, n_use, target_use, alphas, 1200, 1e-14,
                                 time_limit=tleft() - 40, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 6: Try SLSQP at different resolutions
    for n_try in [int(n_use * 1.3), int(n_use * 0.75)]:
        if tleft() < 50:
            break
        if n_try < 100 or n_try == n_use:
            continue
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alphas_try = [30, 100, 300, 800, 2000]
        h_ref2, sc2 = slsqp_refine(h_interp, n_try, target_try, alphas_try, 800, 1e-13,
                                   time_limit=tleft() - 35, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    # Phase 7: Final SA polish on the best solution
    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(3000, tleft() * 1500))
        h_sa_f, c5_f = sa_optimize(h_bin_final, target_use, max_it, 0.004, 0.00003,
                                   rng, time_limit=max(5.0, tleft() - 10),
                                   start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f

        if tleft() > 8:
            h_ref_f, sc_f = slsqp_refine(best_h, len(best_h), len(best_h) / 2.0,
                                         [500, 1500, 3000], 300, 1e-15,
                                         time_limit=max(3.0, tleft() - 5),
                                         start_time=start_time)
            if sc_f < best_score:
                best_score = sc_f
                best_h = h_ref_f

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
