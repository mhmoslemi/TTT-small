    corr_sub = corr[critical_indices]
    m = np.max(corr_sub)
    exp_vals = np.exp(alpha * (corr_sub - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = (m + np.log(sum_exp) / alpha) * dx
    
    grad = np.zeros(n)
    for pos, idx in enumerate(critical_indices):
        if w[pos] < 1e-12:
            continue
        lag = idx - (n - 1)
        abs_lag = abs(lag)
        grad += compute_gradient_for_lag(h, abs_lag, w[pos])
    
    grad *= dx
    return F, grad
```

This is O(n * |critical_lags|) per evaluation, which is much faster than O(n^2) when |critical_lags| << n.

grad[1:] += -h[:3] → positions 1,2,3 get -h[0], -h[1], -h[2] = -0.5, -0.3, -0.7
Total: [0.7, 0.3-0.5, 0.5-0.3, -0.7] = [0.7, -0.2, 0.2, -0.7] ✓ (matches!)

So the correct general formula is:
If lag ≥ 0:
  grad[lag:] += (1 - h[:n-lag])
  grad[:n-lag] += -h[lag:]
If lag < 0 (L = -lag):
  grad[:n-L] += (1 - h[L:])
  grad[L:] += -h[:n-L]

These are NOT the same! The first is "shifted right" and the second is "shifted left".

Interestingly, for the symmetric case (which is when h is symmetric), they would give the same result. But in general they differ.

Now, an important observation: the correlation vector is NOT symmetric in general. correlate(h, 1-h, k) ≠ correlate(h, 1-h, -k) in general (unless h has special symmetry).

Wait, actually: correlate(h, 1-h, k) = ∑_j h[j+k]*(1-h[j])
And correlate(h, 1-h, -k) = ∑_j h[j-k]*(1-h[j]) = ∑_i h[i]*(1-h[i+k]) (substituting i=j-k)

These are different! The first has h on the "left" and (1-h) on the "right", while the second has (1-h) on the "left" and h on the "right".

So I need to handle positive and negative lags differently in my gradient computation.

def gradient_for_output_index(h, k):
    """Grad of np.correlate(h, 1-h, 'full')[k] with respect to h."""
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] += -h[lag:]
    else:
        L = -lag
        if L < n:
            grad[:n-L] += (1.0 - h[L:])
            grad[L:] += -h[:n-L]
    return grad

def smooth_obj_grad(h, alpha, critical_indices=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if critical_indices is None:
        critical_indices = np.arange(len(corr))
    
    c_sub = corr[critical_indices]
    m = np.max(c_sub)
    exp_vals = np.exp(alpha * (c_sub - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = (m + np.log(sum_exp) / alpha) * dx
    
    grad = np.zeros(n)
    for pos, k in enumerate(critical_indices):
        if w[pos] < 1e-14:
            continue
        grad += w[pos] * gradient_for_output_index(h, k)
    
    grad *= dx
    return F, grad
```

import numpy as np
from scipy.optimize import minimize, differential_evolution
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def gradient_for_output_index(h, k):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] += -h[lag:]
    else:
        L = -lag
        if L < n:
            grad[:n - L] += (1.0 - h[L:])
            grad[L:] += -h[:n - L]
    return grad


def smooth_obj_grad_critical(h, alpha, critical_indices):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_sub = corr[critical_indices]
    m = float(np.max(c_sub))
    exp_vals = np.exp(alpha * (c_sub - m))
    sum_exp = float(np.sum(exp_vals))
    w = exp_vals / sum_exp
    F = (m + np.log(sum_exp) / alpha) * dx
    grad = np.zeros(n)
    for pos in range(len(critical_indices)):
        if w[pos] < 1e-13:
            continue
        k = critical_indices[pos]
        grad += w[pos] * gradient_for_output_index(h, k)
    grad *= dx
    return F, grad


def identify_critical(h, n_top):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[-n_top:]
    return np.sort(idx)


def lbfgs_refine(h_start, target, critical_indices, alpha, maxiter):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    obj_fn = lambda x: smooth_obj_grad_critical(x, alpha, critical_indices)[0]
    jac_fn = lambda x: smooth_obj_grad_critical(x, alpha, critical_indices)[1]

    try:
        res = minimize(obj_fn, h_start, jac=jac_fn, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-9, 'maxfun': 5000})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def binary_local_search(h_binary, max_time, rng, budget_fn):
    n = len(h_binary)
    h = h_binary.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    current = compute_c5(h)
    t_start = time.time()
    accepted = 0
    stalls = 0
    max_stalls = 50000
    while time.time() - t_start < max_time and budget_fn() > 20:
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_val = compute_c5(h)
        if new_val < current - 1e-12:
            current = new_val
            accepted += 1
            stalls = 0
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
        else:
            h[j] = 0.0
            h[i] = 1.0
            stalls += 1
        if stalls > max_stalls:
            temp = max(0.05, 5.0 * (1.0 - (time.time() - t_start) / max_time))
            if rng.random() > temp:
                h[j] = 0.0
                h[i] = 1.0
                stalls += 1
            else:
                h[i] = 0.0
                h[j] = 1.0
                current = compute_c5(h)
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h <= 0.5)[0]
                stalls = 0
    return h, current


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def refine_at_resolution(h_start, n, target, alpha_schedule, maxiter_base, budget_fn):
    h_cur = project_feasible(h_start.copy(), target)
    best_score = compute_c5(h_cur)
    for alpha, n_crit in alpha_schedule:
        if budget_fn() < 25:
            break
        critical = identify_critical(h_cur, n_crit)
        maxiter = maxiter_base
        h_new = lbfgs_refine(h_cur, target, critical, alpha, maxiter)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            h_cur = h_new
    return h_cur, best_score
