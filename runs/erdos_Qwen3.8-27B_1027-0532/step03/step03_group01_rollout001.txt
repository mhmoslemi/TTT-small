        t[i] = np.binary_repr(i).count('1') % 2
    return t.astype(float)


def generate_starts(n, rng, num_starts):
    starts = []
    h_init = np.array(initial_h_values, dtype=float)
    h_bin = project_binary(h_init, n / 2.0)
    starts.append(h_bin)
    for _ in range(3):
        h_rand = np.zeros(n)
        perm = rng.permutation(n)
        h_rand[perm[:n // 2]] = 1.0
        starts.append(h_rand)
    h_tm = thue_morse_pattern(n)
    h_tm = project_binary(h_tm, n / 2.0)
    starts.append(h_tm)
    if n >= 4:
        h_block = np.zeros(n)
        block_size = max(2, n // 200)
        for i in range(0, n, block_size):
            if (i // block_size) % 2 == 0:
                h_block[i:min(i + block_size, n)] = 1.0
        h_block = project_binary(h_block, n / 2.0)
        starts.append(h_block)
    return starts[:num_starts]


def informed_sa(h_start, target_sum, max_iters, t_init, t_final, rng,
                time_limit=None, start_time=None, top_m=5):
    n = len(h_start)
    h = h_start.copy()
    if not np.allclose(h, np.round(h)):
        h = project_binary(h, target_sum)
    best_h = h.copy()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr))
    current_c5 = best_c5
    no_improve = 0
    max_no_improve = max(3000, max_iters // 15)
    ones_cache = np.where(h == 1)[0]
    zeros_cache = np.where(h == 0)[0]
    n_corr = len(corr)

    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
            else:
                break

        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio

        move_type = rng.randint(3)
        h_new = h.copy()
        moved = False

        if move_type == 0 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            i1 = ones_cache[rng.randint(len(ones_cache))]
            if rng.random() < 0.6:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - (n - 1)
                target_pos1 = (i1 + k_star) % n
                target_pos2 = (i1 - k_star) % n
                dist1 = np.abs(zeros_cache.astype(int) - target_pos1).astype(float)
                dist2 = np.abs(zeros_cache.astype(int) - target_pos2).astype(float)
                dist = np.minimum(dist1, dist2) + rng.random(len(zeros_cache)) * 8.0
                i0 = zeros_cache[np.argmin(dist)]
            else:
                i0 = zeros_cache[rng.randint(len(zeros_cache))]
            h_new[i1] = 0
            h_new[i0] = 1
            moved = True
        elif move_type == 1 and len(ones_cache) >= 2 and len(zeros_cache) >= 2:
            idx1 = rng.choice(len(ones_cache), 2, replace=False)
            idx0 = rng.choice(len(zeros_cache), 2, replace=False)
            i1a = ones_cache[idx1[0]]
            i1b = ones_cache[idx1[1]]
            if rng.random() < 0.5:
                worst_idx = np.argmax(corr)
                k_star = worst_idx - (n - 1)
                t1 = (i1a + k_star) % n
                t2 = (i1b - k_star) % n
                d1 = np.abs(zeros_cache.astype(int) - t1).astype(float) + rng.random(len(zeros_cache)) * 6.0
                d2 = np.abs(zeros_cache.astype(int) - t2).astype(float) + rng.random(len(zeros_cache)) * 6.0
                i0a = zeros_cache[np.argmin(d1)]
                i0b = zeros_cache[np.argmin(d2)]
                if i0a == i0b:
                    i0b = zeros_cache[rng.randint(len(zeros_cache))]
            else:
                i0a = zeros_cache[idx0[0]]
                i0b = zeros_cache[idx0[1]]
            h_new[i1a] = 0
            h_new[i1b] = 0
            h_new[i0a] = 1
            h_new[i0b] = 1
            moved = True
        elif move_type == 2 and len(ones_cache) > 0 and len(zeros_cache) > 0:
            worst_indices = np.argsort(-corr)[:top_m]
            i1 = ones_cache[rng.randint(len(ones_cache))]
            best_score_val = 1e9
            best_i0 = zeros_cache[0]
            num_candidates = min(20, len(zeros_cache))
            cand_idx = rng.choice(len(zeros_cache), num_candidates, replace=False)
            for ci in cand_idx:
                i0 = zeros_cache[ci]
                score = 0.0
                for wi in worst_indices[:top_m]:
                    k = wi - (n - 1)
                    j1 = i1 + k
                    j2 = i0 + k
                    val1 = 0.0
                    val2 = 0.0
                    if 0 <= j1 < n:
                        val1 = h[j1]
                    if 0 <= j2 < n:
                        val2 = h[j2]
                    delta = -val1 + (1.0 - val2) + (1.0 - val1) - (1.0 - val2)
                    j1b = i1 - k
                    j2b = i0 - k
                    val1b = 0.0
                    val2b = 0.0
                    if 0 <= j1b < n:
                        val1b = h[j1b]
                    if 0 <= j2b < n:
                        val2b = h[j2b]
                    delta += -val1b + (1.0 - val2b) + (1.0 - val1b) - (1.0 - val2b)
                    score += max(0.0, corr[wi] + delta * (2.0 / n) - best_c5)
                if score < best_score_val:
                    best_score_val = score
                    best_i0 = i0
            h_new[i1] = 0
            h_new[best_i0] = 1
            moved = True

        if not moved:
            continue

        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new))
        delta = c5_new - current_c5

        if delta < -1e-12 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            corr = corr_new
            current_c5 = c5_new
            ones_cache = np.where(h == 1)[0]
            zeros_cache = np.where(h == 0)[0]
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_h, best_c5


def top_p_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top_vals = np.sort(corr)[-p:]
    return float(np.sum(top_vals)) / p


def top_p_smooth(h, alpha, p):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_corr = np.sort(corr)
    top = sorted_corr[-p:]
    max_top = np.max(top)
    shifted = np.clip(alpha * (top - max_top), -500.0, 0.0)
    return float(max_top + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _cons_target


def cons_jac(h):
    return np.ones(_cons_n)


def slsqp_top_p(h_start, n, target, p_vals, alphas, maxiter, ftol,
                time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for p in p_vals:
        for alpha in alphas:
            if time_limit is not None and start_time is not None:
                if (time.time() - start_time) > time_limit:
                    return h_cur, curr_score
            try:
                res = minimize(lambda x: top_p_smooth(x, alpha, p), h_cur,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': ftol})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < curr_score:
                    curr_score = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, curr_score


def slsqp_softmax(h_start, n, target, alphas, maxiter, ftol,
                  time_limit=None, start_time=None):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            res = minimize(lambda x: top_p_smooth(x, alpha, len(x) - 1), h_cur,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def quick_hill_climb(h, target_sum, max_iter, rng):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stall = 0
    max_stall = 100
    corr = compute_corr(h)

    for _ in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        h[i1] = 0
        h[i0] = 1
        corr_new = compute_corr(h)
        c5_new = float(np.max(corr_new))
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            corr = corr_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1
            if stall > max_stall:
                break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _cons_n, _cons_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = np.array(h_init, dtype=float)
    best_score = compute_c5(best_h)

    h_bin_start = project_binary(h_init, target_init)

    starts = generate_starts(n_init, rng, 6)

    for si in range(len(starts)):
        if tleft() < 100:
            break
        h_s = starts[si]
        allowed_time = min(180.0, tleft() * 0.4)
        max_it = int(min(100000, allowed_time * 2500))
        h_sa, c5_sa = informed_sa(h_s, target_init, max_it, 0.015, 0.00005,
                                  rng, time_limit=allowed_time, start_time=start_time)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()
        if tleft() > 60:
            h_hc, c5_hc = quick_hill_climb(h_sa, target_init, 5000, rng)
            if c5_hc < best_score:
                best_score = c5_hc
                best_h = h_hc.copy()

    n_candidates = []
    for nf in [0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
        n_try = int(n_init * nf)
        if n_try % 2 == 0 and n_try != n_init and n_try >= 200:
            n_candidates.append(n_try)
    n_candidates = sorted(set(n_candidates))

    for n_try in n_candidates:
        if tleft() < 80:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed_time = min(120.0, tleft() * 0.35)
        max_it = int(min(60000, allowed_time * 2000))
        h_sa, c5_sa = informed_sa(h_bin_try, target_try, max_it, 0.012, 0.00008,
                                  rng, time_limit=allowed_time, start_time=start_time)
        if tleft() > 40:
            h_hc, c5_hc = quick_hill_climb(h_sa, target_try, 4000, rng)
            if c5_hc < c5_sa:
                h_sa = h_hc
                c5_sa = c5_hc
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0

    if tleft() > 80:
        p_vals = [max(3, n_use // 50), max(5, n_use // 30), max(10, n_use // 20)]
        alphas = [50, 150, 400, 1000]
        h_ref, sc = slsqp_top_p(best_h, n_use, target_use, p_vals, alphas,
                                600, 1e-14, time_limit=tleft() - 60, start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 60:
        alphas_sm = [200, 500, 1500, 3000]
        h_ref2, sc2 = slsqp_softmax(best_h, n_use, target_use, alphas_sm,
                                    500, 1e-14, time_limit=tleft() - 50, start_time=start_time)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_ref2

    n_refine_list = []
    for nf in [0.9, 1.1, 1.25, 1.5]:
        n_r = int(n_use * nf)
        if n_r % 2 == 0 and n_r != n_use and n_r >= 200:
            n_refine_list.append(n_r)

    for n_r in n_refine_list:
        if tleft() < 50:
            break
        target_r = n_r / 2.0
        h_interp = interpolate_to(best_h, n_r, target_r)
        h_bin_r = project_binary(h_interp, target_r)
        allowed = min(60.0, tleft() * 0.3)
        max_it = int(min(30000, allowed * 2000))
        h_sa_r, c5_r = informed_sa(h_bin_r, target_r, max_it, 0.010, 0.00005,
                                   rng, time_limit=allowed, start_time=start_time)
        if tleft() > 30:
            p_vals_r = [max(3, n_r // 50), max(8, n_r // 25)]
            alphas_r = [200, 600, 1500]
            h_ref_r, sc_r = slsqp_top_p(h_sa_r, n_r, target_r, p_vals_r, alphas_r,
                                        400, 1e-14, time_limit=tleft() - 30, start_time=start_time)
            if sc_r < c5_r:
                h_sa_r = h_ref_r
                c5_r = sc_r
        if c5_r < best_score:
            best_score = c5_r
            best_h = h_sa_r.copy()

    n_use = len(best_h)
    target_use = n_use / 2.0
    fine_attempts = min(8, int(max(2, tleft() / 35)))

    for fa in range(fine_attempts):
        if tleft() < 25:
            break
        noise = rng.choice([0.005, 0.01, 0.02, 0.03])
        h_pert = best_h + rng.randn(n_use) * noise
        h_pert = project_feasible(h_pert, target_use)
        alphas_fine = [500, 1500, 3000]
        p_fine = [max(3, n_use // 40)]
        h_ref, sc = slsqp_top_p(h_pert, n_use, target_use, p_fine, alphas_fine,
                                300, 1e-15, time_limit=min(25.0, tleft() - 10), start_time=start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(5000, tleft() * 1500))
        h_sa_f, c5_f = informed_sa(h_bin_final, target_use, max_it, 0.005, 0.00003,
                                   rng, time_limit=max(5.0, tleft() - 10), start_time=start_time)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 10:
            h_ref, sc = slsqp_top_p(best_h, len(best_h), len(best_h) / 2.0,
                                    [max(3, len(best_h) // 40)], [1000, 3000],
                                    200, 1e-15, time_limit=max(3.0, tleft() - 5), start_time=start_time)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-14:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
