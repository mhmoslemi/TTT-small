    dx = 2.0 / n

    best = h.copy()
    best_score = compute_c5(best)
    
    alpha_lse = 50.0  # Temperature for log-sum-exp
    eta = 1e-3  # Step size for gradient
    max_iter = 200
    
    for it in range(max_iter):
        if time.time() - start > max_time:
            break
        # Compute gradient of log-sum-exp
        corr = np.correlate(best, 1.0 - best, mode='full')
        shifted = alpha_lse * (corr - np.max(corr))

        obj = float(np.max(corr) + np.log(np.sum(np.exp(shifted))) / alpha_lse) * dx
        
        # Numerical gradient (sparse)
        grad = np.zeros(n)
        eps = 1e-7
        for j in range(n):  # This is O(n) evaluations, might be slow

            h_plus[j] += eps

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def lse_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

def lse_gradient(h, alpha):
    """Compute gradient of log-sum-exp objective wrt h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-30:
        return np.zeros(n)
    weights = weights / sw  # Now sum to 1
    
    # weights[m] corresponds to lag = m - n + 1
    # grad[j] = sum_m weights[m] * dC[m]/dh[j]
    # dC[m]/dh[j] = (1 - h[j-lag]) * 1[0 <= j-lag <= n-1] - h[j] * 1[0 <= j+lag <= n-1]
    # where lag = m - n + 1
    
    grad = np.zeros(n)
    # For efficiency, loop over lags with significant weight
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        lag = m - n + 1
        # First term: (1-h[j-lag]) for j where 0 <= j-lag <= n-1
        # j ranges from max(0, lag) to min(n-1, n-1+lag)
        if lag >= 0:
            j_start = lag
            j_end = n  # exclusive, so j goes from lag to n-1
        else:
            j_start = 0
            j_end = n + lag  # exclusive, so j goes from 0 to n+lag-1 = n-1+lag
        
        if j_end > j_start:
            grad[j_start:j_end] += w * (1.0 - h[j_start - lag:j_end - lag])
        
        # Second term: -h[j] for j where 0 <= j+lag <= n-1
        # j ranges from max(0, -lag) to min(n-1, n-1-lag)
        if lag >= 0:
            j_start2 = 0
            j_end2 = n - lag  # exclusive
        else:
            j_start2 = -lag
            j_end2 = n
        
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w * h[j_start2:j_end2]
    
    return grad * dx

def projected_gradient_descent(h, target, time_limit, alpha_start=20.0, alpha_end=500.0, eta_start=0.01):

    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    alpha = alpha_start
    eta = eta_start
    
    while time.time() - start < time_limit:
        # Animate alpha
        # Compute gradient
        grad = lse_gradient(best, alpha)
        # Project out mean (to maintain sum constraint)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        
        # Line search
        best_trial = None
        best_trial_score = best_score
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = eta * factor * gnorm  # Scale by gradient norm
            trial = best + step * grad / gnorm
            trial = project_feasible(trial, target)
            score = compute_c5(trial)
            if score < best_trial_score:
                best_trial_score = score
                best_trial = trial

            # Also try the lse value for comparison

            if lse_val < lse_objective(best, alpha):
                best_trial = trial
                best_trial_score = compute_c5(trial)
                break
        
        if best_trial is not None and best_trial_score < best_score:
            best = best_trial
            best_score = best_trial_score
            eta = min(eta * 1.05, 0.1)
            alpha *= 1.3
        else:
            eta *= 0.8
            if eta < 1e-6:
                alpha *= 1.5
                eta = eta_start * 0.1
            if alpha > alpha_end * 5:
                break
    
    return best, best_score

def targeted_cd(h, target, time_limit, seed=42, delta_init=1e-3, delta_min=1e-6):

            if improved:

    """Compute np.correlate(a, b, 'full') using FFT."""
    n = len(a)
    m = len(b)

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def lse_objective(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx


def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-30:
        return np.zeros(n)
    weights = weights / sw
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-12:
            continue
        lag = m - n + 1
        if lag >= 0:
            j_start = lag
            j_end = n
        else:
            j_start = 0
            j_end = n + lag
        if j_end > j_start:
            grad[j_start:j_end] += w * (1.0 - h[j_start - lag:j_end - lag])
        if lag >= 0:
            j_start2 = 0
            j_end2 = n - lag
        else:
            j_start2 = -lag
            j_end2 = n
        if j_end2 > j_start2:
            grad[j_start2:j_end2] -= w * h[j_start2:j_end2]
    return grad * dx


def gradient_phase(h, target, time_limit, alpha_start=20.0, alpha_end=500.0, eta_start=0.02):

    alpha = alpha_start
    eta = eta_start
    no_improve = 0
    while time.time() - start < time_limit - 5:
        grad = lse_gradient(best, alpha)
        grad -= np.mean(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:

        found = False
