    delta = np.zeros(2*n-1)

    # As first index: delta = h[i+k] - 1 for p in [n-1-i, 2n-2-i]
    lo = max(0, n-1-i)
    hi = min(2*n-2, 2*n-2-i)

    idx_hi = i + hi - (n-1)
    delta[lo:hi+1] += h[idx_lo:idx_hi+1].astype(float) - 1.0
    
    # As second index: delta = h[i-k] for p in [i, i+n-1]
    lo2 = max(0, i)
