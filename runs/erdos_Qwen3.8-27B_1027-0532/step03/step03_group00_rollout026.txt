    n = len(h)
    corr = compute_corr(h)
    # corr has 2n-1 elements
    # LP: min t s.t. for each k: g_k^T h <= t - (corr[k] - g_k^T h)
    #     which is: g_k^T h + (corr[k] - g_k^T h) <= t
    #     i.e., the linearized value at h must be <= t
    #     linearized value at h = corr[k] + g_k^T (h - h_current)
    # So: corr[k] + g_k^T (h - h_current) <= t for all k
    #     g_k^T h <= t - corr[k] + g_k^T h_current
    
    # Compute gradients for all k

    # This is an approximation but should work

    # Compute gradient for each top k
    grads = np.zeros((m, n))
    for idx, k_idx in enumerate(top_m):
        k = k_idx - (n - 1)
        grads[idx] = compute_grad_k(h, k)
    
    # Solve LP: min t s.t. grads[i] @ h <= t + (grads[i] @ h - corr[top_m[i]]) for all i
    #            0 <= h <= 1, sum(h) = target
    # Rearranged: grads[i] @ h - t <= grads[i] @ h - corr[top_m[i]]
    #            i.e., linearized_k(h) <= t where linearized_k(h) = corr[k] + grads[i] @ (h - h_current)
    
    # Actually: linearized_value_k(h) = corr[k_idx] + grads[idx] @ (h - h)
    #                                        = corr[k_idx]  (at h = h_current)
    # At the new h: linearized_value_k(h_new) = corr[k_idx] + grads[idx] @ (h_new - h_current)
    
    # So the LP is:
    # min t
    # s.t. corr[idx] + grads[idx] @ (h_new - h_current) <= t  for all idx
    #      0 <= h_new <= 1
    #      sum(h_new) = target
    
    # Let's use cvxpy
    import cvxpy as cp
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

        constraints.append(lin_val <= t_var)
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.CLARABEL)  # or GLOP, or ECOS

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(mask.sum())

            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_shift(h, k):
    """Compute gradient of f(k) = sum_i h[i+k]*(1-h[i]) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # i ranges 0 to n-1-k
        i_vals = np.arange(0, n - k)
        # Term 1: h[i+k] is the first factor, d/dh[j] where j=i+k
        # grad[i+k] += (1 - h[i])

        # Term 2: (1-h[i]) is the second factor, d/dh[j] where j=i
        # grad[i] += -h[i+k]
        grad[0:n-k] += (-h[k:n])
    else:
        kp = -k
        i_vals = np.arange(0, n - kp)
        # f(-kp) = sum_{i=0}^{n-1-kp} h[i] * (1-h[i+kp])
        # grad[i] += (1-h[i+kp])
        grad[0:n-kp] += (1.0 - h[kp:n])

        grad += compute_grad_shift(h, k)
    grad /= m_top

    h_new = project_feasible(h_new, target)
    return h_new

    """L-BFGS-B with penalty for sum constraint"""
    best_h = h.copy()
    best_score = compute_c5(best_h)

        def obj(x):
            h_clip = np.clip(x, 0, 1)
            sh = soft_max_val(h_clip, alpha)
            pen = penalty * (np.sum(h_clip) - target)**2
            return sh + pen
        
        res = minimize(obj, best_h, method='L-BFGS-B', 
                      bounds=[(0,1)]*n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    return best_h, best_score

def al_iteration(h, target, m_top):

    top_m = np.argsort(corr)[-m_top:]

        except:
            return h, None

        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_feasible(h_new, target)
        return h_new, t_var.value
    return h, None

                    cur_score = sc
                    improved = True
            except:
                pass
        if not improved:
            break
    return h_cur, cur_score

        h1, sc1 = lbfgsb_round(best_h, n, target, alphas_p1, 3000, 200.0)
        if sc1 < best_score:
            best_score = sc1
            best_h = h1

    # Phase 2: AL (Alternating Linearization) with CVXPY LP
    if tleft() > 250:
        al_deadline = start_time + (budget_s - 45.0) * 0.35
        h_al = best_h.copy()
        prev_al_score = compute_c5(h_al)
        for it in range(500):
            if time.time() > al_deadline or tleft() < 120:
                break
            h_al, _ = _al_iteration(h_al, target, m_top=25)
            sc = compute_c5(h_al)
            if sc < best_score:
                best_score = sc
                best_h = h_al
            if it > 5 and abs(sc - prev_al_score) < 1e-8:
                if it > 30:
                    break
            prev_al_score = sc

    # Phase 3: Momentum subgradient descent
    if tleft() > 200:
        deadline_p3 = start_time + (budget_s - 45.0) * 0.45
        h_sg = best_h.copy()
        h_sg = momentum_subgradient(h_sg, target, m_top=15, n_iters=3000,
                                    step_init=0.3, beta=0.7, deadline=deadline_p3)
        sc = compute_c5(h_sg)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 4: L-BFGS-B with high alpha (sharp soft-max)
    if tleft() > 120:
        alphas_p4 = [2000, 5000, 10000]
        h4, sc4 = lbfgsb_round(best_h, n, target, alphas_p4, 2000, 500.0)
        if sc4 < best_score:
            best_score = sc4
            best_h = h4

    # Phase 5: Block coordinate descent
    deadline_p5 = start_time + (budget_s - 45.0) * 0.65
    for bs in [4, 6]:
        if tleft() < 80:
            break
        h_bd, sc_bd = block_descent(best_h, n, target, bs, 6, 800.0, deadline_p5)
        if sc_bd < best_score:
            best_score = sc_bd
            best_h = h_bd

    # Phase 6: More AL iterations with larger m_top
    if tleft() > 150:
        al_deadline = start_time + (budget_s - 45.0) * 0.80
        h_al = best_h.copy()
        for it in range(400):
            if time.time() > al_deadline or tleft() < 60:
                break
            h_al, _ = _al_iteration(h_al, target, m_top=50)
            sc = compute_c5(h_al)
            if sc < best_score:
                best_score = sc
                best_h = h_al

    # Phase 7: Randomized restarts + quick L-BFGS-B
    if tleft() > 150:
        deadline_p7 = start_time + (budget_s - 45.0) * 0.90
        for _ in range(8):
            if tleft() < 60 or time.time() > deadline_p7:
                break
            h_pert = randomized_perturb(best_h, rng, n, target, 0.4)
            h_rp, sc_rp = lbfgsb_round(h_pert, n, target, [100, 500, 2000], 1000, 300.0)
            sc = compute_c5(h_rp)
            if sc < best_score:
                best_score = sc
                best_h = h_rp

    # Phase 8: Final momentum subgradient with small steps
    if tleft() > 60:
        deadline_p8 = start_time + budget_s - 50.0
        h_sg = best_h.copy()
        h_sg = momentum_subgradient(h_sg, target, m_top=10, n_iters=5000,
                                    step_init=0.1, beta=0.8, deadline=deadline_p8)
        sc = compute_c5(h_sg)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    # Phase 9: Final L-BFGS-B polish
    if tleft() > 30:
        alphas_final = [5000, 15000]
        h_f, sc_f = lbfgsb_round(best_h, n, target, alphas_final, 1500, 1000.0)
        if sc_f < best_score:
            best_score = sc_f
            best_h = h_f

    # Phase 10: Final subgradient micro-steps
    if tleft() > 10:
        h_sg = best_h.copy()
        deadline_p10 = start_time + budget_s - 5.0
        h_sg = momentum_subgradient(h_sg, target, m_top=5, n_iters=3000,
                                    step_init=0.05, beta=0.9, deadline=deadline_p10)
        sc = compute_c5(h_sg)
        if sc < best_score:
            best_score = sc
            best_h = h_sg

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
