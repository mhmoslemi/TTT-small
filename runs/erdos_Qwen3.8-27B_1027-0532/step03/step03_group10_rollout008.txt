def project_onto_constraint(h, target_sum):
    # Binary search for the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def soft_max(x, eps):
    m = np.max(x)
    if m < -1e10:
        return m
    x_shifted = (x - m) / eps
    x_shifted -= np.max(x_shifted)  # for numerical stability
    return m + eps * np.log(np.sum(np.exp(x_shifted)))

def soft_max_grad(x, eps):
    m = np.max(x)
    x_shifted = (x - m) / eps
    x_shifted -= np.max(x_shifted)
    e = np.exp(x_shifted)
    return e / np.sum(e)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return soft_max(corr, eps)

    dx = 2.0 / n

            j = int(zeros_idx[rng.integers(len(zeros_idx))])

        if i == j:
            step += 1
            continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            current_max = float(np.max(corr))
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            s_range = np.arange(s_min, s_max + 1)
            k_range = s_range + half
            delta_corr = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
            corr[k_range] += delta_corr
            new_max = float(np.max(corr))
            delta = new_max - current_max
            if delta < 0 or rng.random() < np.exp(-delta / t):
                h[i], h[j] = h[j], h[i]
                current_max = new_max
                if current_max < best_c5_val:
                    best_c5_val = current_max
                    best_h = h.copy()
            else:
                corr[k_range] -= delta_corr
        step += 1
    return best_h, best_c5_val

def greedy_one_pass(h_bin):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    improved = True
    passes = 0
    while improved and passes < 50:
        improved = False
        passes += 1
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        for i in ones_idx:
            if h[i] != 1.0:
                continue
            for j in zeros_idx:
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                s_range = np.arange(s_min, s_max + 1)
                k_range = s_range + half
                delta_corr = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
                new_vals = corr[k_range] + delta_corr
                new_max_region = float(np.max(new_vals))
                new_max_other = float(np.max(np.concatenate([corr[:k_range[0]], corr[k_range[-1]+1:]]))) if k_range[0] > 0 or k_range[-1] < len(corr)-1 else 0.0
                new_max = max(new_max_region, new_max_other)
                if new_max < current_max - 1e-12:
                    corr[k_range] += delta_corr
                    h[i] = 0.0
                    h[j] = 1.0
                    current_max = new_max
                    improved = True
                    break
            if improved:
                break
    return h, current_max

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        for eps in [0.01, 0.003, 0.001, 0.0003]:
            if time.time() - start_time > budget_s * 0.08:
                break
            h_new, c5 = slsqp_optimize(best_h, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        h_bin = to_binary(best_h)
        h_polish = h_bin + 0.01 * rng.standard_normal(n0)
        h_polish = project_onto_constraint(h_polish, target_sum)
        for eps in [0.001, 0.0003, 0.0001]:
            if time.time() - start_time > budget_s * 0.12:
                break
            h_polish, c5 = slsqp_optimize(h_polish, eps, 300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    n_sa_total = 0
    while time.time() - start_time < budget_s - 120:
        strategy = n_sa_total % 6
        if strategy == 0:
            h_start = to_binary(best_h)
        elif strategy == 1:
            h_pert = best_h + 0.05 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, target_sum)
            h_start = to_binary(h_pert)
        elif strategy == 2:
            h_start = to_binary(best_h)
            n_flip = max(3, n0 // 12)
            idx = rng.choice(n0, size=n_flip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
            s = h_start.sum()
            if s > target_sum:
                ones = np.where(h_start == 1)[0]
                extra = int(s - target_sum)
                h_start[ones[:extra]] = 0
            elif s < target_sum:
                zeros = np.where(h_start == 0)[0]
                need = int(target_sum - s)
                h_start[zeros[:need]] = 1
        elif strategy == 3:
            h_r = (rng.random(n0) < 0.5).astype(float)
            alpha = rng.uniform(0.5, 0.8)
            h_mix = alpha * best_h + (1 - alpha) * h_r
            h_start = to_binary(h_mix)
        elif strategy == 4:
            h_start = to_binary(best_h)
            n_seg = int(rng.integers(2, 6))
            for _ in range(n_seg):
                a = int(rng.integers(0, n0 - 5))
                b = min(a + int(rng.integers(3, 15)), n0)
                h_start[a:b] = h_start[a:b][::-1]
                s = h_start.sum()
                if int(s) != n0 // 2:
                    if s > target_sum:
                        ones = np.where(h_start == 1)[0]
                        extra = int(s - target_sum)
                        h_start[ones[:extra]] = 0
                    elif s < target_sum:
                        zeros = np.where(h_start == 0)[0]
                        need = int(target_sum - s)
                        h_start[zeros[:need]] = 1
        else:
            h_start = to_binary(best_h)
            n2 = min(10, n0 // 8)
            for _ in range(n2):
                i = int(rng.integers(n0))
                j = int(rng.integers(n0))
                if h_start[i] != h_start[j]:
                    h_start[i], h_start[j] = h_start[j], h_start[i]

        t_start = 0.4 * (1.0 + 0.5 * rng.random())
        t_end = 0.003 + 0.003 * rng.random()
        n_steps = int(min(800000, max(50000, (budget_s - 150 - (time.time() - start_time)) * 800)))

        if n_sa_total % 3 == 0:
            h_bin, bin_max = binary_sa_fast(h_start, n_steps, t_start, t_end, rng, rev_prob=0.15)
        elif n_sa_total % 3 == 1:
            h_bin, bin_max = binary_sa_focused(h_start, n_steps, t_start, t_end, rng)
        else:
            h_bin, bin_max = binary_sa_fast(h_start, n_steps, t_start, t_end, rng, rev_prob=0.3)

        c5_bin = (2.0 / n0) * bin_max

        h_polish_start = h_bin + 0.015 * rng.standard_normal(n0)
        h_polish_start = project_onto_constraint(h_polish_start, target_sum)

        if time.time() - start_time < budget_s - 80:
            h_pol, c5_pol = slsqp_optimize(h_polish_start, 0.001, 300)
            h_pol, c5_pol = slsqp_optimize(h_pol, 0.0003, 300)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
            if c5_pol < c5_bin:
                c5_bin = c5_pol
                h_bin = h_pol.copy()

        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
            if time.time() - start_time < budget_s - 60:
                h_deep = h_bin + 0.005 * rng.standard_normal(n0)
                h_deep = project_onto_constraint(h_deep, target_sum)
                h_deep, c5_deep = slsqp_optimize(h_deep, 0.0001, 500)
                h_deep, c5_deep = slsqp_optimize(h_deep, 0.00003, 500)
                if c5_deep < best_c5:
                    best_c5 = c5_deep
                    best_h = h_deep.copy()

        n_sa_total += 1

    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        h_final = best_h + 0.003 * rng.standard_normal(n0)
        h_final = project_onto_constraint(h_final, target_sum)
        for eps in [0.0001, 0.00003, 0.00001]:
            if time.time() - start_time > budget_s - 15:
                break
            h_final, c5 = slsqp_optimize(h_final, eps, 600)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()

        h_flat = best_h.copy()
        for _ in range(3000):
            g = c5_smooth_grad(h_flat, 0.00005)
            g = g - np.mean(g)
            h_flat = h_flat - 0.001 * g
            h_flat = project_onto_constraint(h_flat, target_sum)
        c5_flat = compute_c5(h_flat)
        if c5_flat < best_c5:
            best_c5 = c5_flat
            best_h = h_flat.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
