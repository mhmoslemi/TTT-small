def active_gradient(h, k_list):
    """Compute gradient of sum of corr_full[n-1+k] for k in k_list."""
    n = len(h)
    grad = np.zeros(n)
    for k in k_list:
        k = int(k)
        if k == 0:
            # corr_full[n-1] = sum(h * (1-h))
            # grad = (1-2h)
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n-kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n-kk]
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def active_gradient(h, k_list):
    n = len(h)
    grad = np.zeros(n)
    for k in k_list:
        k = int(k)
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            if k < n:
                grad[k:] += (1.0 - h[:n - k])
                grad[:n - k] -= h[k:]
        else:
            kk = -k
            if kk < n:
                grad[:n - kk] += (1.0 - h[kk:])
                grad[kk:] -= h[:n - kk]
    return grad


def peak_shaving(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    step = 0.05
    M = 10
    patience = 0
    max_patience = 50

    while time.time() - start < time_limit:
        corr = np.correlate(best, 1.0 - best, mode='full')
        idx = np.argsort(corr)[::-1][:M]
        k_vals = (idx - (n - 1)).astype(int)

        grad = active_gradient(best, k_vals)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break

        h_new = best - step * grad / gn
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)

        if new_score < best_score - 1e-15:
            best = h_new
            best_score = new_score
            patience = 0
            step = min(step * 1.03, 0.15)
            M = min(M + 1, 80)
        else:
            patience += 1
            if patience > max_patience:
                step *= 0.7
                M = max(5, M // 2)
                if step < 0.001:
                    perturbation = 0.002 * rng.randn(n)
                    h_pert = best + perturbation
                    h_pert = project_feasible(h_pert, target)
                    s_pert = compute_c5(h_pert)
                    if s_pert < best_score:
                        best = h_pert
                        best_score = s_pert
                    step = 0.05
                    M = 10
                    patience = 0
                    max_patience = int(max_patience * 0.8)

    return best, best_score


def coordinate_descent(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    deltas = np.array([0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 0.001, -0.001])
    no_improve = 0

    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            for d in deltas:
                val = best[i] + d
                if val < 0.0 or val > 1.0:
                    continue
                old = best[i]
                best[i] = val
                best = project_feasible(best, target)
                s = compute_c5(best)
                if s < best_score - 1e-14:
                    best_score = s
                    improved = True
                    break
                else:
                    best[i] = old
                    best = project_feasible(best, target)
        no_improve = 0 if improved else no_improve + 1
        if no_improve >= 3:
            deltas = deltas * 0.5
            if np.max(np.abs(deltas)) < 1e-6:
                break
            no_improve = 0

    return best, best_score


def slsqp_smooth_max(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    alphas = [20, 80, 300]
    n_alpha = len(alphas)
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (n_alpha - ai)
        maxiter = int(max(50, time_per * 2))

        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -400.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)

        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score

def spectral_leveling(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    step = 0.02
    patience = 0

    while time.time() - start < time_limit:
        H = np.fft.rfft(best, 2 * n)
        power = np.abs(H) ** 2
        total_power = np.sum(power)
        if total_power < 1e-15:
            break
        norm_power = power / total_power
        excess = norm_power - 1.0 / len(norm_power)

        grad_spec = np.zeros(2 * n)
        grad_spec[:len(excess) + 1] = 2.0 * excess
        grad_spec[len(excess) + 1:] = 2.0 * excess[::-1][:max(0, 2 * n - len(excess))]

        grad_h = np.fft.irfft(np.real(np.fft.fftshift(grad_spec)))[:n]
        gn = np.linalg.norm(grad_h)
        if gn < 1e-15:
            break

        h_new = best - step * grad_h / gn
        h_new = project_feasible(h_new, target)
        s_new = compute_c5(h_new)

        if s_new < best_score - 1e-15:
            best = h_new
            best_score = s_new
            patience = 0
            step = min(step * 1.02, 0.1)
        else:
            patience += 1
            if patience > 40:
                step *= 0.7
                patience = 0
                if step < 0.002:
                    break

    return best, best_score

        h5, s5 = slsqp_smooth_max(best_h, target, min(150, tleft()))
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Another peak shaving round (now with refined starting point)
    if tleft() > 150:
        h6, s6 = peak_shaving(best_h, target, min(150, tleft()), seed=seed + 40)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Fine coordinate descent
    if tleft() > 100:
        h7, s7 = coordinate_descent(best_h, target, min(100, tleft()), seed=seed + 50)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final SLSQP polish
    if tleft() > 80:
        h8, s8 = slsqp_smooth_max(best_h, target, min(80, tleft()))
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Final projection and evaluation
    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return h_final, best_score, n
