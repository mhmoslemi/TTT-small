```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_from_continuous(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def sa_fast(h, t_start, t_end, total_steps, rng, time_limit, start_time):
    """Fast simulated annealing for binary h."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full').astype(np.float64)
    current = float(np.max(corr)) * dx
    best = current
    best_h = h.copy()
    
    s = np.arange(-(n-1), n)
    
    step = 0
    while step < total_steps:
        if time.time() - start_time > time_limit:
            break
        
        t = t_start * (t_end / t_start) ** (step / max(total_steps - 1, 1))
        
        # Single swap move
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        
        # Compute delta
        h[oi] = 0.0
        h[zi] = 1.0
        omh_new = 1.0 - h
        corr_new = np.correlate(h, omh_new, mode='full').astype(np.float64)
        new_score = float(np.max(corr_new)) * dx
        delta = new_score - current
        
        if delta < 0 or (t > 1e-9 and rng.random() < np.exp(-delta / t)):
            current = new_score
            corr = corr_new
            omh = omh_new
            if new_score < best:
                best = new_score
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
            # Revert omh
            omh[oi] = 0.0
            omh[zi] = 1.0
        
        step += 1
    
    return best_h, best
```

Let me implement parallel tempering! It's a well-known technique that can significantly improve the exploration of the solution space.

In parallel tempering:
- Maintain K chains at temperatures T_1 > T_2 > ... > T_K
- At each step, evolve one chain (or all chains) by a few SA steps
- Every L steps, try to swap states between adjacent chains (accept with probability min(1, exp((1/T_low - 1/T_high) * (E_low - E_high))))

This is very effective because the hot chains explore the landscape while the cold chains refine.

Let me implement this:

```python
def parallel_tempering(h_init, K, t_high, t_low, steps_per_chain, swap_interval, 
                       total_steps, rng, time_limit, start_time):
    n = len(h_init)
    # Initialize K chains
    chains = np.zeros((K, n))
    energies = np.zeros(K)
    for i in range(K):
        if i == 0:
            chains[i] = h_init
        else:
            # Perturb: flip some random positions
            h_pert = h_init.copy()
            nflip = max(1, n // (10 + i * 5))
            idx = rng.choice(n, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            # Rebalance
            target = n // 2
            current_sum = h_pert.sum()
            if current_sum > target:
                ones_to_flip = np.where(h_pert == 1.0)[0]
                extra_ones = int(current_sum - target)
                to_flip = rng.choice(ones_to_flip, size=extra_ones, replace=False)
                h_pert[to_flip] = 0.0
            elif current_sum < target:
                zeros_to_flip = np.where(h_pert == 0.0)[0]
                extra_zeros = int(target - current_sum)
                to_flip = rng.choice(zeros_to_flip, size=extra_zeros, replace=False)
                h_pert[to_flip] = 1.0
            chains[i] = h_pert
            # Rebalance exactly
            h_pert = make_binary_from_continuous(h_pert)
            chains[i] = h_pert
    
    for i in range(K):
        energies[i] = compute_c5(chains[i])
    
    # Temperatures: geometric from t_low to t_high
    temps = t_low * (t_high / t_low) ** (np.arange(K) / (K - 1))
    # temps[0] = t_low (coldest), temps[K-1] = t_high (hottest)
    
    best = float(np.min(energies))
    best_h = chains[np.argmin(energies)].copy()
    
    step = 0
    while step < total_steps:
        if time.time() - start_time > time_limit:
            break
        
        # Evolve one chain (round-robin)
        c = step % K
        ones = np.where(chains[c] == 1.0)[0]
        zeros = np.where(chains[c] == 0.0)[0]
        oi = ones[rng.integers(len(ones))]
        zi = zeros[rng.integers(len(zeros))]
        
        chains[c][oi] = 0.0
        chains[c][zi] = 1.0
        new_e = compute_c5(chains[c])
        delta = new_e - energies[c]
        
        if delta < 0 or rng.random() < np.exp(-delta / temps[c]):
            energies[c] = new_e
            if new_e < best:
                best = new_e
                best_h = chains[c].copy()
        else:
            chains[c][oi] = 1.0
            chains[c][zi] = 0.0
        
        # Try swap with adjacent chain
        if step % swap_interval == 0 and c > 0:
            # Try swap chains[c] and chains[c-1]
            # Accept with prob min(1, exp((1/T_{c-1} - 1/T_c) * (E_c - E_{c-1})))
            # T_{c-1} < T_c (c-1 is colder)
            # 1/T_{c-1} > 1/T_c
            # If E_c < E_{c-1} (hottest has lower energy), swap is always accepted
            # If E_c > E_{c-1}, swap with prob exp((1/T_{c-1} - 1/T_c) * (E_c - E_{c-1}))
            # Wait, the standard formula:
            # P(accept) = min(1, exp(-(1/T_high - 1/T_low) * (E_low - E_high)))
            # = min(1, exp((1/T_high - 1/T_low) * (E_high - E_low)))
            # Where T_high > T_low, E_low is energy of cold chain, E_high of hot chain
            # Let me use the standard Metropolis criterion for the swap:
            # Delta = (1/T_{c-1} - 1/T_c) * (E_{c-1} - E_c)
            # Accept with prob min(1, exp(-Delta))
            # Since T_{c-1} < T_c, 1/T_{c-1} > 1/T_c
            # So (1/T_{c-1} - 1/T_c) > 0
            # If E_{c-1} < E_c (cold has lower energy): Delta > 0, P < 1
            # If E_{c-1} > E_c (hot has lower energy): Delta < 0, P = 1
            delta_swap = (1.0/temps[c-1] - 1.0/temps[c]) * (energies[c-1] - energies[c])
            if delta_swap < 0 or rng.random() < np.exp(-delta_swap):
                # Swap
                chains[c-1], chains[c] = chains[c].copy(), chains[c-1].copy()
                energies[c-1], energies[c] = energies[c], energies[c-1]
        
        step += 1
    
    return best_h, best
```

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s_val = k - (n - 1)
        if s_val >= 0:
            m = n - s_val
            grad[s_val:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s_val:]
        else:
            r = -s_val
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_from_continuous(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):

def parallel_tempering(h_init, K, t_low, t_high, swap_interval, total_steps, rng, time_limit, start_time):
