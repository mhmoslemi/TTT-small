import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_carr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adjust = diff / len(idx)
        h[idx] = np.clip(h[idx] + adjust, 0.0, 1.0)
    # Final exact fix
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_shift(h, k):
    """Compute gradient of c[k] w.r.t. h, where k is index in full correlate output."""
    n = len(h)
    m = n - 1
    grad = np.zeros(n)
    
    # S_k = {i : max(0, m-k) <= i <= min(n-1, 2m-k)}
    lo = max(0, m - k)
    hi = min(n - 1, 2 * m - k)
    
    if lo > hi:
        return grad
    
    idx = np.arange(lo, hi + 1)
    # First term: (1 - h[idx + k - m]) for i in S_k, j = i
    shift1 = idx + k - m
    # These are valid by construction of S_k
    grad[idx] += (1.0 - h[shift1])
    
    # Second term: -h[idx + m - k] where the index in h is i + k - m = j
    # So for gradient w.r.t. h[j], we need i = j - k + m in S_k
    # j = i + k - m, so j ranges from lo + k - m to hi + k - m
    lo2 = lo + k - m

def subgrad_descent(h, target_sum, n_iter, lr0, momentum, time_check):
    h_cur = h.copy()
    v = np.zeros_like(h_cur)
    best_h = h_cur.copy()
    best_c = compute_c5(h_cur)
    
    for it in range(n_iter):
        if it % 500 == 0 and time_check() < 30:
            break
        
        lr = lr0 / (1.0 + 0.05 * it)
        carr = compute_corr_full(h_cur)

        top_idx = np.argsort(carr)[-top_k:]
        
        grad = np.zeros_like(h_cur)
        for k in top_idx:
            grad += grad_at_shift(h_cur, k)
        grad /= top_k
        
        v = momentum * v - lr * grad
        h_trial = project_feasible(h_cur + v, target_sum)
        
        carr_new = compute_corr_full(h_trial)
        c_new = float(np.max(carr_new))

            best_h = h_trial.copy()
