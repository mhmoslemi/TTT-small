Actually, I realize there's an even simpler and potentially more effective approach:

**Subgradient method with Polyak step size:**
1. Compute the subgradient of C₅ at current h
2. Take a step in the negative subgradient direction
3. Project to the feasible set
4. Use Polyak step size if we have an upper bound on the optimum

The subgradient of C₅ = max_k f_k(h) at h is:
- If k* is the unique maximizing k: subgradient = ∇f_{k*}(h)
- If multiple k's tie: subgradient is any convex combination of their gradients

∇f_k(h) with respect to h_i:
f_k(h) = Σ_valid h[i] * (1-h[i+k-(n-1)]) * dx

For a fixed shift s = k - (n-1) (the actual lag):
f_s(h) = Σ_{i: 0≤i<n, 0≤i+s<n} h[i] * (1-h[i+s]) * dx

∂f_s/∂h[i] = dx * (1-h[i+s]) - dx * h[i-s]  (with boundary conditions: terms are 0 if index is out of range)

Wait, more carefully:
f_s(h) = dx * Σ_i h[i] * (1 - h[i+s]) [for valid i and i+s]

∂f_s/∂h[i] = dx * (1 - h[i+s]) [from the term where h[i] is the first factor]
             - dx * h[i-s] [from the term where h[i] appears as h[(i-s)+s] in the second factor, if i-s is valid]

So ∂f_s/∂h[i] = dx * (1 - h[i+s]) - dx * h[i-s] (with out-of-range terms = 0)

For the max function, the subgradient is just the gradient of the active (maximizing) function.

This is straightforward to compute and use in a subgradient method.

            h = np.clip(h, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    s = k_star - (n - 1)
    grad = np.zeros(n)
    for i in range(n):

        j2 = i - s
        if 0 <= j2 < n:
            grad[i] -= h[j2] * dx

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-5):

    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL]
    solver_idx = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0

        for k in range(-(n - 1), n):

            hi = min(n, k + n)
            if hi > lo:

                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        solved = False
        for attempt in range(3):
            try:
                solver = solvers[(solver_idx + attempt) % len(solvers)]

                break

                continue
        if not solved:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)

        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:

            stuck_count = 0

    _global_n = n
    _global_target = target

                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass

        return project_feasible(h_src.copy(), target_dst)

_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgradient(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    arange_n = np.arange(n)
    for k_star in top_idx:
        s = k_star - (n - 1)
        idx1 = arange_n + s
        mask1 = (idx1 >= 0) & (idx1 < n)
        safe_idx1 = np.where(mask1, idx1, 0)
        term1 = np.where(mask1, 1.0 - h[safe_idx1], 0.0)
        idx2 = arange_n - s
        mask2 = (idx2 >= 0) & (idx2 < n)
        safe_idx2 = np.where(mask2, idx2, 0)
        term2 = np.where(mask2, h[safe_idx2], 0.0)
        grad += dx * (term1 - term2)
    grad /= top_k
    return grad, float(np.max(corr)) * dx

def subgradient_method(h, target, max_steps, time_limit, lr_init=0.1):

def pair_swap_descent(h, target, max_sweeps, time_limit, num_partners=4):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    arange_n = np.arange(n)
    rng = np.random.RandomState(6789)
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:

        improved = False
        order = rng.permutation(n)
        for i in order:

            partners.add((i + half_n) % n)

            room = np.minimum(h_cur, 1.0 - h_cur).copy()
            room[i] = -2.0

            for j in partners:

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg
    solvers = [cp.ECOS, cp.SCS, cp.CLARABEL]
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        solved = False
        for attempt in range(3):
            try:
                solver = solvers[attempt % len(solvers)]
                prob.solve(solver=solver, verbose=False)
                solved = True
                break
            except Exception:
                continue
        if not solved:
            break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        if stuck_count > 25:
            damping = max(0.05, damping * 0.85)
            stuck_count = 0
        if it % 200 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.6)
    return best_h, best_c5

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def cons_fun(h):
    return np.sum(h) - _global_target

def cons_jac(h):
    return np.ones(_global_n)

def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def block_coordinate_descent(h, target, max_sweeps, time_limit, n_blocks=4):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    block_size = n // n_blocks
    boundaries = [b * block_size for b in range(n_blocks)] + [n]
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        improved = False
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            lo_b = boundaries[b]
            hi_b = boundaries[b + 1]
            block_len = hi_b - lo_b
            block_fixed_sum = target - np.sum(h_cur[lo_b:hi_b])
            if abs(block_fixed_sum) > n * 0.6:
                continue
            h_var = cp.Variable(block_len)
            t_var = cp.Variable()
            g = 1.0 - h_cur
            constraints = [
                cp.sum(h_var) == target - np.sum(h_cur[np.arange(n) < lo_b]) - np.sum(h_cur[np.arange(n) >= hi_b]),
                h_var >= 0.0,
                h_var <= 1.0
            ]
            for k in range(-(n - 1), n):
                coeff_full = np.zeros(n)
                lo_k = max(0, k)
                hi_k = min(n, k + n)
                if hi_k > lo_k:
                    coeff_full[lo_k:hi_k] = g[lo_k - k:hi_k - k] * dx
                coeff_block = coeff_full[lo_b:hi_b]
                if np.all(np.abs(coeff_block) < 1e-15):
                    continue
                const = dx * np.sum(g[np.arange(n) * 0 + np.where((np.arange(n) >= lo_b) & (np.arange(n) < hi_b), 0, 1)])
                other_sum = np.sum(coeff_full) - np.sum(coeff_block)
                constraints.append(coeff_block @ h_var <= t_var - other_sum * 0.0)
            try:
                prob = cp.Problem(cp.Minimize(t_var), constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                if h_var.value is not None:
                    h_new_block = np.clip(h_var.value.ravel(), 0.0, 1.0)
                    h_cur[lo_b:hi_b] = h_new_block
                    h_cur = project_feasible(h_cur, target)
                    sc = compute_c5(h_cur)
                    if sc < best_c5 - 1e-14:
                        best_c5 = sc
                        best_h = h_cur.copy()
                        improved = True
            except Exception:
                pass
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP from initial
    if tleft() > 200:
        lp_time = min(280, tleft() * 0.3)
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                      max_iter=10000, time_limit=lp_time,
                                      damping_init=0.6, reg=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Pair-swap coordinate descent
    if tleft() > 120:
        ps_time = min(180, tleft() * 0.3)
        h_ps, c5_ps = pair_swap_descent(best_h, target,
                                         max_sweeps=8000, time_limit=ps_time,
                                         num_partners=4)
        if c5_ps < best_score:
            best_score = c5_ps
            best_h = h_ps

    # Phase 3: Subgradient with multi-witness
    if tleft() > 80:
        sg_time = min(120, tleft() * 0.25)
        h_sg, c5_sg = subgradient_method(best_h, target,
                                          max_steps=80000, time_limit=sg_time,
                                          lr_init=0.08)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: More alternating LP with perturbations
    for restart in range(3):
        if tleft() < 100:
            break
        noise = [0.04, 0.1, 0.2][restart]
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        damping = [0.4, 0.7, 0.55][restart]
        lp_time = min(80, tleft() * 0.2)
        h_lp, c5_lp = alternating_lp(h_pert, n, target,
                                      max_iter=5000, time_limit=lp_time,
                                      damping_init=damping, reg=1e-5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp
        if tleft() > 50:
            h_ps, c5_ps = pair_swap_descent(best_h, target,
                                             max_sweeps=3000, time_limit=25,
                                             num_partners=4)
            if c5_ps < best_score:
                best_score = c5_ps
                best_h = h_ps

    # Phase 5: Higher resolution
    if tleft() > 120:
        for n_try in [600, 800]:
            if tleft() < 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(70, tleft() * 0.3)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                          max_iter=3000, time_limit=lp_time,
                                          damping_init=0.6, reg=1e-5)
            candidate_h = h_lp
            candidate_score = c5_lp
            if tleft() > 30:
                h_ps2, c5_ps2 = pair_swap_descent(candidate_h, target_try,
                                                   max_sweeps=2000, time_limit=20,
                                                   num_partners=3)
                if c5_ps2 < candidate_score:
                    candidate_h = h_ps2
                    candidate_score = c5_ps2
            if candidate_score < best_score:
                best_score = candidate_score
                best_h = candidate_h
                n = n_try
                target = target_try

    # Phase 6: SLSQP refinement
    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ref = slsqp_round(best_h, n_use, target_use,
                             [50, 200, 500, 1200], 800, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final pair-swap and subgradient
    if tleft() > 25:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_ps, c5_ps = pair_swap_descent(best_h, target_use,
                                         max_sweeps=4000, time_limit=min(20, tleft() * 0.5),
                                         num_partners=4)
        if c5_ps < best_score:
            best_score = c5_ps
            best_h = h_ps
        if tleft() > 10:
            h_sg, c5_sg = subgradient_method(best_h, target_use,
                                              max_steps=30000, time_limit=min(10, tleft() * 0.5),
                                              lr_init=0.04)
            if c5_sg < best_score:
                best_score = c5_sg
                best_h = h_sg

    # Final feasibility and return
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
