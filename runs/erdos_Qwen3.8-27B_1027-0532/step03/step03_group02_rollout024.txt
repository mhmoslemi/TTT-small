def smooth_logsumexp(h, alpha):
    """Smooth approximation of max(corr) using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def enforce_symmetry(h):
    n = len(h)
    h_sym = np.zeros(n)
    half = (n + 1) // 2
    for i in range(half):
        val = (h[i] + 1.0 - h[n - 1 - i]) * 0.5
        if val < 0.0:
            val = 0.0
        elif val > 1.0:
            val = 1.0
        h_sym[i] = val
        h_sym[n - 1 - i] = 1.0 - val
    return h_sym

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def sym_cd(h, max_time, rng, patience=40, n_bisect=40):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = enforce_symmetry(h.copy())
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()
    half = (n + 1) // 2
    order = np.arange(half)
    rng.shuffle(order)
    
    while time.time() - elapsed_start < max_time:
        for j in order:
            a = n - 1 - j
            ia_p = a + s_arr
            ia_m = a - s_arr
            ij_p = j + s_arr
            ij_m = j - s_arr
            m1 = (ia_p >= 0) & (ia_p < n)
            m2 = (ia_m >= 0) & (ia_m < n)
            m3 = (ij_p >= 0) & (ij_p < n)
            m4 = (ij_m >= 0) & (ij_m < n)
            D = np.zeros(2 * n - 1)
            D[m1] += h[ia_p[m1]]
            D[m2] += h[ia_m[m2]]
            D[m3] -= h[ij_p[m3]]
            D[m4] -= h[ij_m[m4]]
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-15:
                continue
            for _ in range(n_bisect):
                mid = (lo + hi) * 0.5
                idx = int(np.argmax(corr + mid * D))
                sl = D[idx]
                if sl > 1e-10:
                    hi = mid
                elif sl < -1e-10:
                    lo = mid
                else:
                    break
            nv = h[j] + (lo + hi) * 0.5
            if nv < 0.0:
                nv = 0.0
            elif nv > 1.0:
                nv = 1.0
            delta = nv - h[j]
            if abs(delta) > 1e-14:
                h[j] = nv
                h[a] = 1.0 - nv
                corr += delta * D
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            scale = 0.003 * (1.0 + no_improve * 0.1)
            h = best_h + rng.randn(n) * scale
            h = enforce_symmetry(h)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
            rng.shuffle(order)
    
    return best_h, best_score

def reg_cd(h, target_sum, max_time, rng, patience=30, n_bisect=35):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()
    
    while time.time() - elapsed_start < max_time:
        for j in range(n):
            ov = h[j]
            jp = j + s_arr
            jm = j - s_arr
            m1 = (jp >= 0) & (jp < n)
            m2 = (jm >= 0) & (jm < n)
            A = np.zeros(2 * n - 1)
            A[m1] += 1.0 - h[jp[m1]]
            A[m2] -= h[jm[m2]]
            B = corr - A * ov
            lo = 0.0
            hi = 1.0
            for _ in range(n_bisect):
                mid = (lo + hi) * 0.5
                idx = int(np.argmax(A * mid + B))
                sl = A[idx]
                if sl > 1e-10:
                    hi = mid
                elif sl < -1e-10:
                    lo = mid
                else:
                    break
            nv = (lo + hi) * 0.5
            delta = nv - ov
            if abs(delta) > 1e-14:
                corr += delta * A
                h[j] = nv
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= patience:
            scale = 0.003 * (1.0 + no_improve * 0.1)
            h = best_h + rng.randn(n) * scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0
    
    return best_h, best_score

def interp_sym(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return enforce_symmetry(h_src.copy())
    xo = np.linspace(0.0, 2.0, n_src)
    xn = np.linspace(0.0, 2.0, n_dst)
    return enforce_symmetry(np.interp(xn, xo, h_src))

def interp_reg(h_src, n_dst, target):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target)
    xo = np.linspace(0.0, 2.0, n_src)
    xn = np.linspace(0.0, 2.0, n_dst)
    return project_feasible(np.interp(xn, xo, h_src), target)

def slsqp_refine(h, max_time, alphas=None):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    if alphas is None:
        alphas = [200, 500, 1000]
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start > max_time:
            break
        try:
            def smooth_obj(x, a=alpha):
                c = np.correlate(x, 1.0 - x, mode='full')
                cm = np.max(c)
                sh = np.clip(a * (c - cm), -500.0, 0.0)
                return (cm + np.log(np.sum(np.exp(sh))) / a) * dx
            
            res = minimize(smooth_obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq',
                                         'fun': lambda x: np.sum(x) - target,
                                         'jac': lambda x: np.ones(n)}],
                           options={'maxiter': 300, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_cur = h_new
        except Exception:
            pass
    return best_h, best_score
