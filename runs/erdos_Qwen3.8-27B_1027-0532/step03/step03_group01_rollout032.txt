import numpy as np
from scipy.optimize import minimize
import time

def fast_correlate(a, b):
    """Compute np.correlate(a, b, 'full') using FFT"""
    na, nb = len(a), len(b)
    n_full = na + nb - 1
    size = 1
    while size < n_full:
        size *= 2
    fa = np.fft.rfft(a, size=size)
    fb = np.fft.rfft(b[::-1], size=size)
    result = np.fft.irfft(fa * fb, size=size)
    return result[:n_full]

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fast_correlate(h, one_minus_h)
    return float(np.max(corr)) * dx

def compute_c5_and_lag(h):
    """Returns (c5, lag_of_max)"""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fast_correlate(h, one_minus_h)
    max_idx = np.argmax(corr)
    return float(corr[max_idx]) * dx, max_idx - (n - 1), corr

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def soft_max_subgradient_direct(h, alpha):
    n = len(h)
    dx = 2.0 / n
    n_full = 2 * n - 1
    
    # Compute correlation
    size = 1 << (n_full - 1).bit_length()
    fa = np.fft.rfft(h, n=size)
    fb = np.fft.rfft((1.0 - h)[::-1], n=size)
    corr = np.fft.irfft(fa * fb, n=size)[:n_full]
    
    # Soft-max weights
    max_c = np.max(corr)
    exp_vals = np.clip(alpha * (corr - max_c), -50, 0)
    weights = np.exp(exp_vals)
    weights /= np.sum(weights)
    
    # lags: k - (n-1) for k = 0..n_full-1
    lags = np.arange(n_full) - (n - 1)  # from -(n-1) to (n-1)
    
    # For each j, grad[j] = dx * [1 - sum_k w[k]*h[j-lak] - sum_k w[k]*h[j+lag_k]]
    # Vectorized: create matrix M where M[j, k] = h[j - lag_k] (with 0 for out of bounds)
    # Then sum_k w[k] * M[j, k] gives the first term
    
    # This is O(n * n_full) memory which is 600 * 1199 ≈ 720K - fine
    j_idx = np.arange(n)[:, None]  # n x 1
    lag_vals = lags[None, :]  # 1 x n_full
    
    # First term: h[j - lag_k]
    idx1 = j_idx - lag_vals  # n x n_full
    mask1 = (idx1 >= 0) & (idx1 < n)
    idx1_clipped = np.clip(idx1, 0, n-1)
    term1 = np.sum(weights[None, :] * np.where(mask1, h[idx1_clipped], 0.0), axis=1)
    
    # Second term: h[j + lag_k]
    idx2 = j_idx + lag_vals
    mask2 = (idx2 >= 0) & (idx2 < n)
    idx2_clipped = np.clip(idx2, 0, n-1)
    term2 = np.sum(weights[None, :] * np.where(mask2, h[idx2_clipped], 0.0), axis=1)
    
    grad = dx * (1.0 - term1 - term2)
    return grad, weights

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
