import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

2. Identify the top-k lags (e.g., k=5)

6. Iterate

- smooth_max approximates the max of ALL lags (diluted by the soft max)
- Active set optimization focuses on the ACTUAL max (sharp focus)

```python
def active_set_optimize(h_init, k_active, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    
    corr = compute_corr(h)
    active_idx = np.argsort(corr)[-k_active:]
    
    # Now minimize the sum of the active correlations
    # Each active correlation is a smooth function of h
    # The gradient of corr[k] with respect to h is:
    # For lag s = k - (n-1):
    #   d(corr[s])/d(h[i]) = dx * (1-h[i+s]) for i in valid range (positive lag)
    #                        or -dx * h[i+s] for the other part
    
    def active_obj(h):
        c = compute_corr(h)
        return float(np.sum(c[active_idx]))
    
    def active_grad(h):
        n = len(h)
        dx = 2.0 / n
        omh = 1.0 - h
        grad = np.zeros(n)
        for kidx in active_idx:
            s = kidx - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += dx * omh[:m]
                grad[:m] -= dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += dx * omh[r:]
                grad[r:] -= dx * h[:m]
        return grad
    
    # Use SLSQP on the active set objective
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    result = minimize(active_obj, h, jac=active_grad, method='SLSQP',
                      bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    
    h_opt = project_onto_constraint(result.x, target_sum)
    return h_opt, compute_c5(h_opt)
```

import numpy as np
from scipy.optimize import minimize
import time

_CONSTRAINT_TARGET = [0.0]

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def constraint_fun(x):
    return float(np.sum(x) - _CONSTRAINT_TARGET[0])

def constraint_jac(x):
    return np.ones(len(x))

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def active_set_obj(h, active_idx):
    c = compute_corr(h)
    return float(np.sum(c[active_idx]))

def active_set_grad(h, active_idx):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    for kidx in active_idx:
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += dx * omh[:m]
            grad[:m] -= dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += dx * omh[r:]
            grad[r:] -= dx * h[:m]
    return grad

def slsqp_smooth(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    _CONSTRAINT_TARGET[0] = target_sum
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_active(h_init, k_active, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    _CONSTRAINT_TARGET[0] = target_sum
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    corr = compute_corr(h0)
    active_idx = np.argsort(corr)[-k_active:]
    active_idx = np.sort(active_idx)
    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(active_set_obj, h0, args=(active_idx,),
                          jac=active_set_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def sa_search(h_init, rng, time_limit, start_time, T0=0.003, T_factor=0.9997):
    n = len(h_init)
    h = h_init.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T0
    stall_counter = 0
    stall_limit = 2000

    while time.time() - start_time < time_limit:
        h_old = h.copy()
        mt = rng.random()

        if mt < 0.30:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
        elif mt < 0.55:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            hi = i if h[i] > h[j] else j
            lo = j if h[i] > h[j] else i
            delta = rng.uniform(0.03, 0.25)
            transfer = min(delta, h[hi] - h[lo])
            if transfer <= 0.001:
                continue
            h[hi] -= transfer
            h[lo] += transfer
        elif mt < 0.80:
            blen = int(rng.integers(2, max(3, n // 8)))
            a = int(rng.integers(0, n - blen))
            b = int(rng.integers(0, n - blen))
            if abs(a - b) < blen:
                continue
            tmp = h[a:a+blen].copy()
            h[a:a+blen] = h[b:b+blen]
            h[b:b+blen] = tmp
        else:
            high_mask = h > 0.55
            low_mask = h < 0.45
            high_idx = np.where(high_mask)[0]
            low_idx = np.where(low_mask)[0]
            if len(high_idx) == 0 or len(low_idx) == 0:
                continue
            i = high_idx[rng.integers(len(high_idx))]
            j = low_idx[rng.integers(len(low_idx))]
            h[i] = 0.0
            h[j] = 1.0

        c5_new = compute_c5(h)

        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            stall_counter = 0
        elif rng.random() < np.exp(-(c5_new - best_c5) / T):
            h = h.copy()
            stall_counter += 1
        else:
            h = h_old
            stall_counter += 1

        if stall_counter > stall_limit:
            h = best_h.copy()
            T = T0 * 0.5
            stall_counter = 0
            stall_limit = max(1000, stall_limit // 2)

        T *= T_factor
        if T < 1e-9:
            T = 1e-9

    return best_h, best_c5

def gen_perturb(h_best, rng, n0, strategy):
    target_sum = n0 / 2.0
    if strategy == 0:
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n0, size=max(5, n0 // 8), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        max_len = max(2, n0 // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            tmp = h_p[a:a+blen].copy()
            h_p[a:a+blen] = h_p[b:b+blen]
            h_p[b:b+blen] = tmp
            return h_p
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n0), target_sum)
    else:
        h_bin = (rng.random(n0) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.4, 0.7))
        return project_onto_constraint(alpha * h_best + (1.0 - alpha) * h_bin, target_sum)

def active_set_iterate(h_init, k_active, iters_per_round, n_rounds, time_limit, start_time):
    h = h_init.copy()
    best_c5 = compute_c5(h)
    for r in range(n_rounds):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h)
        active_idx = np.argsort(corr)[-k_active:]
        active_idx = np.sort(active_idx)
        n = len(h)
        target_sum = n / 2.0
        _CONSTRAINT_TARGET[0] = target_sum
        bounds = [(0.0, 1.0)] * n
        try:
            result = minimize(active_set_obj, h, args=(active_idx,),
                              jac=active_set_grad, method='SLSQP',
                              bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}],
                              options={'maxiter': iters_per_round, 'ftol': 1e-14})
            h_new = project_onto_constraint(result.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
        except Exception:
            break
    return h, best_c5
