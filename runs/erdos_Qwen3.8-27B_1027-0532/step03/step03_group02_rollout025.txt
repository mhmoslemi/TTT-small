import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def sym_coord_descend(h_init, n, max_time, rng, patience=25):
    half = n // 2
    h_half = np.clip(h_init[:half].copy(), 0.0, 1.0)
    h_sym = np.zeros(n)
    h_sym[:half] = h_half
    h_sym[half:] = 1.0 - h_half[::-1]
    s_arr = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h_sym, 1.0 - h_sym, mode='full').copy()
    best_score = float(np.max(corr)) * (2.0 / n)
    best_h_half = h_half.copy()
    no_improve = 0
    start = time.time()

    while time.time() - start < max_time:
        for j in range(half):
            old_val = h_half[j]
            m = n - 1 - j
            idx1 = j + s_arr
            idx2 = j - s_arr
            idx3 = m + s_arr
            idx4 = m - s_arr
            A = np.zeros(2 * n - 1)
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            v3 = (idx3 >= 0) & (idx3 < n)
            v4 = (idx4 >= 0) & (idx4 < n)
            A[v1] += (1.0 - h_sym[idx1[v1]])
            A[v2] -= h_sym[idx2[v2]]
            A[v3] -= (1.0 - h_sym[idx3[v3]])
            A[v4] += h_sym[idx4[v4]]

            lo = max(-old_val, h_sym[m] - 1.0)
            hi = min(1.0 - old_val, h_sym[m])
            if hi - lo < 1e-15:
                continue
            left, right = lo, hi
            for _ in range(45):
                mid = (left + right) / 2.0
                vals = corr + mid * A
                f_mid = np.max(vals)
                active = vals >= f_mid - 1e-13
                a_min = np.min(A[active])
                a_max = np.max(A[active])
                if a_min > 1e-11:
                    right = mid
                elif a_max < -1e-11:
                    left = mid
                else:
                    break
            delta = (left + right) / 2.0
            if abs(delta) > 1e-15:
                corr += delta * A
                h_half[j] = old_val + delta
                h_sym[j] = old_val + delta
                h_sym[m] = 1.0 - old_val - delta

        score = float(np.max(corr)) * (2.0 / n)

            best_h_half = h_half.copy()
            no_improve = 0
        else:

def interpolate_to(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_new = np.linspace(0.0, 2.0, n_dst)
    x_old = np.linspace(0.0, 2.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, n_dst / 2.0)

def smooth_max(corr, alpha):
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def slsqp_refine(h_init, max_iter=500, time_limit=60.0, alpha=500.0):
    n = len(h_init)
    dx = 2.0 / n
    h0 = h_init.copy()
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    start = time.time()
    best_score = compute_c5(h0)
    best_h = h0.copy()

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(c, alpha) * dx

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, target)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
    except Exception:
        pass

    return best_h, best_score
