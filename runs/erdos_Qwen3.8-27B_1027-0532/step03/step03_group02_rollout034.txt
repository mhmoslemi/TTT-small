import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_symmetric(h):
    n = len(h)
    h = h.copy().astype(float)
    half = n // 2
    for i in range(half):
        a = (h[i] + (1.0 - h[n-1-i])) / 2.0
        a = np.clip(a, 0.0, 1.0)
        h[i] = a
        h[n-1-i] = 1.0 - a
    return h

def vec_project_symmetric(h):
    n = len(h)
    h = np.array(h, dtype=float)
    half = n // 2
    a = (h[:half] + (1.0 - h[n-1-half:-1:-1])) / 2.0
    a = np.clip(a, 0.0, 1.0)
    h[:half] = a
    h[n-1-half:-1:-1] = 1.0 - a
    return h

def coord_descend_symmetric(h, max_time, rng, patience=20):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    half = n // 2
    h = vec_project_symmetric(h)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()

    while time.time() - elapsed_start < max_time:
        for j in range(half):
            old_val = h[j]
            j2 = n - 1 - j
            # Compute gradient G(s) for this coordinate
            js = j - s_arr      # index for h[j-s]
            js_plus = j + s_arr  # index for h[j+s]
            j2s = j2 - s_arr     # index for h[j2-s]
            j2s_plus = j2 + s_arr  # index for h[j2+s]
            
            G = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n)
            m2 = (js_plus >= 0) & (js_plus < n)
            m3 = (j2s >= 0) & (j2s < n)
            m4 = (j2s_plus >= 0) & (j2s_plus < n)
            
            G[m1] += (1.0 - h[js[m1]])
            G[m2] -= h[js_plus[m2]]
            G[m3] -= (1.0 - h[j2s[m3]])
            G[m4] += h[j2s_plus[m4]]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def vec_project_symmetric(h):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    half = n // 2
    h2 = h[n-1 : n-1-half : -1]
    a = (h[:half] + (1.0 - h2)) / 2.0
    a = np.clip(a, 0.0, 1.0)
    h[:half] = a
    h[n-1 : n-1-half : -1] = 1.0 - a
    return h

def coord_descend_sym(h, max_time, rng, patience=15):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    half = n // 2
    h = vec_project_symmetric(h)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    t0 = time.time()

    while time.time() - t0 < max_time:
        for j in range(half):
            j2 = n - 1 - j
            old = h[j]
            js = j - s_arr
            jp = j + s_arr
            j2s = j2 - s_arr
            j2p = j2 + s_arr
            
            G = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n)
            m2 = (jp >= 0) & (jp < n)
            m3 = (j2s >= 0) & (j2s < n)
            m4 = (j2p >= 0) & (j2p < n)
            G[m1] += (1.0 - h[js[m1]])
            G[m2] -= h[jp[m2]]
            G[m3] -= (1.0 - h[j2s[m3]])
            G[m4] += h[j2p[m4]]
            
            lo = -old
            hi = 1.0 - old
            if hi - lo < 1e-14:
                continue
            for _ in range(30):
                mid = (lo + hi) / 2.0
                nv = corr + mid * G
                ks = int(np.argmax(nv))
                g = G[ks]
                if g > 1e-12:
                    hi = mid
                elif g < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            d = new_val - old
            if abs(d) > 1e-14:
                corr += d * G
                h[j] = new_val
                h[j2] = 1.0 - new_val
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                ns = 0.005 * min(1.0, no_improve / patience)
                h = best_h + rng.randn(n) * ns
                h = vec_project_symmetric(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                no_improve = 0

    return best_h, best_score

def subgrad_sym(h, max_time, rng):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    half = n // 2
    h = vec_project_symmetric(h)
    best_score = compute_c5(h)
    best_h = h.copy()
    lr = 0.01
    t0 = time.time()
    stag = 0
    
    while time.time() - t0 < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        cmax = float(np.max(corr))
        score = cmax * dx
        
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
            if stag > 5000:
                break
        
        # Find active set
        active = corr >= cmax - 1e-7
        na = int(np.sum(active))
        if na == 0:
            active = np.zeros(2*n-1)
            active[np.argmax(corr)] = 1
            na = 1
        
        # Compute subgradient (only for first half)
        g_half = np.zeros(half)
        for idx in np.where(active)[0]:
            s = s_arr[idx]
            for j in range(half):
                j2 = n - 1 - j
                val = 0.0
                js = j - s
                jp = j + s
                j2s = j2 - s
                j2p = j2 + s
                if 0 <= js < n:
                    val += (1.0 - h[js])
                if 0 <= jp < n:
                    val -= h[jp]
                if 0 <= j2s < n:
                    val -= (1.0 - h[j2s])
                if 0 <= j2p < n:
                    val += h[j2p]
                g_half[j] += val / na
        
        h_new = h.copy()
        h_new[:half] -= lr * g_half
        h_new[n-1:n-1-half:-1] = 1.0 - h_new[:half]
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = vec_project_symmetric(h_new)
        
        sc_new = compute_c5(h_new)
        if sc_new < score:
            h = h_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    
    return best_h, best_score

def smooth_c5(h, alpha):
    """Smoothed version of max(corr) using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # log-sum-exp for numerical stability
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -50, 0)
    smooth_max = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return smooth_max * dx

def smooth_grad(h, alpha):
    """Gradient of smooth_c5 with respect to h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient of corr(s) with respect to h[j]
    # corr(s) = sum_i h[i+s] * (1-h[i])
    # d/dh[j] corr(s) = (1-h[j-s])*I(j-s valid) - h[j+s]*I(j+s valid)
    
    s_arr = np.arange(2*n-1) - (n-1)
    g = np.zeros(n)
    for idx in range(2*n-1):
        if weights[idx] < 1e-10:
            continue
        s = s_arr[idx]
        js = np.arange(n) - s
        jp = np.arange(n) + s
        m1 = (js >= 0) & (js < n)
        m2 = (jp >= 0) & (jp < n)
        g[m1] += weights[idx] * (1.0 - h[js[m1]])
        g[m2] -= weights[idx] * h[jp[m2]]
    return g * dx
