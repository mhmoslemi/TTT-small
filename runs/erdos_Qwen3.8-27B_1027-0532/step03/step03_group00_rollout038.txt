    deadline4 = start_time + budget_s - 25.0
    h4, sc4 = coord_descent(best_h, n, target, deadline4, rng, max_sweeps=100000, stall_limit=30)
    if sc4 < best_score:
        best_score = sc4
        best_h = h4

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if not np.isfinite(final_score):
        h_final = project_feasible(h_init, target)
        final_score = compute_c5(h_final)
    return h_final, final_score, n
