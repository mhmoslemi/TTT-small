import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_active_gradient(h, k):
    """Compute gradient of c[k] with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # c[k] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
        # d/dh[i] = (1-h[i-k]) for k <= i <= n-1
        #           -h[i+k] for 0 <= i <= n-1-k
        if k < n:
            grad[k:] += (1.0 - h[:n-k])  # (1-h[i-k]) for i in [k, n-1]
            grad[:n-k] -= h[k:]  # -h[i+k] for i in [0, n-1-k]
    else:
        k2 = -k
        if k2 < n:
            # c[-k2] = sum_{n=k2}^{n-1} h[n-k2] * (1-h[n])
            # d/dh[i] = (1-h[i+k2]) for 0 <= i <= n-1-k2
            #           -h[i-k2] for k2 <= i <= n-1
            grad[:n-k2] += (1.0 - h[k2:])  # (1-h[i+k2]) for i in [0, n-k2-1]
            grad[k2:] -= h[:n-k2]  # -h[i-k2] for i in [k2, n-1]
    return grad

def directed_descent(h, target, time_limit, max_iter=None):
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_score = compute_c5(best)
    start = time.time()
    iter_count = 0
    
    while time.time() - start < time_limit:
        if max_iter and iter_count >= max_iter:
            break
        iter_count += 1
        
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_idx = int(np.argmax(corr))
        k = max_idx - (n - 1)
        
        grad = compute_active_gradient(best, k)
        
        # Find best direction: increase where grad most negative, decrease where most positive
        inc_idx = int(np.argmin(grad))
        dec_idx = int(np.argmax(grad))
        
        if grad[inc_idx] >= 0 or grad[dec_idx] <= 0:
            break
        
        # Line search
        best_step = 0
        best_trial = best_score
        steps = [0.05, 0.1, 0.2, 0.3, 0.5, 0.15, 0.01, 0.005]
        for step in steps:
            if best[inc_idx] + step > 1.0 or best[dec_idx] - step < 0.0:
                continue
            h_trial = best.copy()
            h_trial[inc_idx] += step
            h_trial[dec_idx] -= step
            s = compute_c5(h_trial)
            if s < best_trial:
                best_trial = s
                best_step = step
        
        if best_step == 0:
            break
        
        best[inc_idx] += best_step
        best[dec_idx] -= best_step
        best_score = best_trial
    
    return best, best_score

def multi_lag_directed_descent(h, target, time_limit, top_k=3):

    start = time.time()

        max_val = np.max(corr)

            if weights[idx] < 1e-6:

            k = idx - (n - 1)
            g = compute_active_gradient(best, k)
            grad += weights[idx] * g

        dec_idx = int(np.argmax(grad))

            break

    return best, best_score

        # Update
        best[inc_idx] += step
        best[dec_idx] -= step
        
        # Incrementally update correlation
        # Δc[k'] = step * (∂c[k']/∂h[inc_idx] - ∂c[k']/∂h[dec_idx])
        grad_inc = compute_all_lags_gradient(best, inc_idx)  # O(n)
        grad_dec = compute_all_lags_gradient(best, dec_idx)  # O(n)
        corr += step * (grad_inc - grad_dec)
        
        steps_since_recompute += 1
        if steps_since_recompute % recompute_every == 0:
            corr = compute_full_corr(best)
        
        # Check actual C5 (periodically or based on corr)
        actual_c5 = np.max(corr) * dx
        if actual_c5 < best_score:
            best_score = actual_c5
        else:
            # Revert and try smaller step
            best[inc_idx] -= step
            best[dec_idx] += step
            corr -= step * (grad_inc - grad_dec)
            if step < 0.001:
                break
            step *= 0.5
    
    # Final verification
    final_corr = compute_full_corr(best)
    final_score = float(np.max(final_corr)) * dx
    if final_score < best_score:
        best_score = final_score
    return best, best_score
```

Wait, there's an issue with the incremental update. When I update h[inc_idx] and h[dec_idx], the gradient formulas I used (compute_all_lags_gradient) were computed BEFORE the update. The actual change in c[k] is:

Δc[k] = step * (∂c[k]/∂h[i] - ∂c[k]/∂h[j])

where the derivatives are evaluated at the OLD h. This is a first-order approximation. For small steps, it's accurate. For larger steps, there's a second-order correction.

The second-order term comes from the fact that c[k] is bilinear in h (each term is h[a]*(1-h[b]) which is linear in h[a] and linear in h[b], so the product is bilinear). Wait, actually h[a]*(1-h[b]) = h[a] - h[a]*h[b]. The h[a]*h[b] term is second-order (product of two h values). So the increment is:

Δc[k] = step * (∂c[k]/∂h[i] - ∂c[k]/∂h[j]) + step² * (second order terms)

The second order terms are non-zero only when both h[i] and h[j] appear in the same product h[a]*h[b] within c[k]. This happens when {i,j} = {a, a+k} for some a, k.

For accuracy, I should either:
1. Use small steps (step < 0.01) so the second-order term is negligible
2. Include the second-order correction
3. Periodically recompute the full correlation

I'll go with option 3: recompute every 50-100 steps. This should keep the error small.

Actually, let me reconsider. The issue is that if the incremental correlation drifts too far from the true correlation, the argmax might be wrong, leading to poor search directions. With a step of 0.1 and recompute every 100 steps, the accumulated error could be significant.

A safer approach: use the incremental update for the search direction, but verify with a full computation before accepting a step. This gives us the speed of O(n) per step for the direction, plus O(n²) for verification (every few steps).

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_active_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0 and k < n:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    elif k < 0 and -k < n:
        k2 = -k
        grad[:n-k2] += (1.0 - h[k2:])
