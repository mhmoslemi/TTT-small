import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            if m > 0:
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            if m > 0:
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def greedy_improve(h_bin):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    for _ in range(10):
        improved = False
        ones_set = set(np.where(h == 1.0)[0].tolist())
        zeros_set = set(np.where(h == 0.0)[0].tolist())
        for i in list(ones_set):
            if h[i] != 1.0:
                continue
            for j in list(zeros_set):
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                s_range = np.arange(s_min, s_max + 1)
                k_range = s_range + half
                delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
                new_vals = corr[k_range] + delta
                unchanged_max = np.max(np.concatenate([corr[:k_range[0]], corr[k_range[-1]+1:]])) if k_range[0] > 0 or k_range[-1] < len(corr)-1 else 0.0
                new_max = max(float(np.max(new_vals)), unchanged_max)
                if new_max < current_max - 1e-12:
                    corr[k_range] += delta
                    h[i] = 0.0
                    h[j] = 1.0
                    current_max = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, current_max

def make_structured_start(n, rng, style):
    h = np.zeros(n)
    if style == 0:
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
    elif style == 1:
        block = n // 12
        for b in range(12):
            if b % 2 == 0:
                h[b*block:(b+1)*block] = 1.0
        excess = h.sum() - n//2
        if excess > 0:
            ones = np.where(h == 1.0)[0]
            h[ones[:excess]] = 0
        elif excess < 0:
            zeros = np.where(h == 0.0)[0]
            h[zeros[:abs(excess)]] = 1
    elif style == 2:
        p = rng.integers(7, 14)
        q = int(p * 0.5)
        pattern = np.zeros(p)
        pattern[:q] = 1.0
        tiled = np.tile(pattern, n // p + 1)[:n]
        h = tiled
        excess = h.sum() - n//2
        if excess > 0:
            ones = np.where(h == 1.0)[0]
            rng.shuffle(ones)
            h[ones[:excess]] = 0
        elif excess < 0:
            zeros = np.where(h == 0.0)[0]
            rng.shuffle(zeros)
            h[zeros[:abs(excess)]] = 1
    else:
        freq = rng.uniform(0.3, 0.7)
        h = (rng.random(n) < freq).astype(float)
        excess = h.sum() - n//2
        if excess > 0:
            ones = np.where(h == 1.0)[0]
            rng.shuffle(ones)
            h[ones[:excess]] = 0
        elif excess < 0:
            zeros = np.where(h == 0.0)[0]
            rng.shuffle(zeros)
            h[zeros[:abs(excess)]] = 1
    return h

        time_per_round = remaining() * 0.35
        n_steps = int(time_per_round * 30000)
        n_steps = max(n_steps, 50000)
        n_steps = min(n_steps, 5000000)

        if n_rounds % 3 == 0:
            h_result, res_max = fast_sa(h_start, n_steps, t_start, t_end, rng, focus_prob=0.6)
        elif n_rounds % 3 == 1:
            h_result, res_max = fast_sa(h_start, n_steps, t_start, t_end, rng, focus_prob=0.3)
        else:
            h_result, res_max = fast_sa_two_point(h_start, n_steps, t_start, t_end, rng)

        c5_res = (2.0 / n0) * res_max
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_result.copy()
            # Quick greedy polish
            h_g, g_max = greedy_improve(h_result)
            c5_g = (2.0 / n0) * g_max
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
            # Quick continuous polish
            if remaining() > 60:
                h_polish = h_best = best_h + 0.01 * rng.standard_normal(n0)
                h_polish = project_onto_constraint(h_polish, target_sum)
                h_polish, c5_p = slsqp_optimize(h_polish, 0.001, 200)
                h_polish, c5_p = slsqp_optimize(h_polish, 0.0002, 200)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_polish.copy()
        n_rounds += 1

    # Phase 3: Final intensive SA on best solution
    if remaining() > 60:
        h_bin_final = to_binary(best_h)
        n_steps_final = int(remaining() * 0.5 * 30000)
        h_final, final_max = fast_sa(h_bin_final, n_steps_final, 0.2, 0.001, rng, focus_prob=0.7)
        c5_final = (2.0 / n0) * final_max
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 4: Final continuous polish
    if remaining() > 30:
        h_polish = best_h + 0.005 * rng.standard_normal(n0)
        h_polish = project_onto_constraint(h_polish, target_sum)
        for eps in [0.0003, 0.00005, 0.00001]:
            if remaining() < 10:
                break
            h_polish, c5 = slsqp_optimize(h_polish, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)

    # Also try binary version
    final_bin = to_binary(final_h)
    final_bin_c5 = compute_c5(final_bin)
    if final_bin_c5 < final_c5:
        final_h = final_bin
        final_c5 = final_bin_c5

    return final_h, final_c5, len(final_h)
