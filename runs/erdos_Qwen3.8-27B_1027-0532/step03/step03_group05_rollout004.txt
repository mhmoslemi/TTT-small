    exp_shifted = np.exp(shifted)
    logsumexp = c_max + np.log(np.sum(exp_shifted)) / alpha
    weights = exp_shifted / np.sum(exp_shifted)
    grad = compute_weighted_grad(h, weights)
    return logsumexp, grad

def compute_weighted_grad(h, weights):
    """
    Compute sum_k weights[k] * d(corr[k])/dh
    where weights has length 2n-1 (indexed 0 to 2n-2, corresponding to lag -(n-1) to (n-1))
    """
    n = len(h)
    grad = np.zeros(n)
    offset = n - 1
    for k in range(len(weights)):
        L = k - offset
        w = weights[k]
        if w < 1e-15:
            continue
        if L >= 0:
            # First term: (1-h[j-L]) for j >= L
            rem = n - L
            if rem > 0:
                grad[L:] += w * (1.0 - h[:rem])
                grad[:rem] -= w * h[L:]
        else:
            M = -L
            rem = n - M
            if rem > 0:
                grad[:rem] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:rem]
    return grad

def lbfgsb_optimize(h, target, alpha, maxiter=500):
    """Optimize using L-BFGS-B with log-sum-exp smooth objective."""
    n = len(h)
    dx = 2.0 / n
    
    def obj(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return c_max + np.log(np.sum(np.exp(shifted))) / alpha
    
    def jac(x):
        corr = compute_corr(x)
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        exp_shifted = np.exp(shifted)
        weights = exp_shifted / np.sum(exp_shifted)
        return compute_weighted_grad(x, weights)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
    h_new = project_feasible(res.x, target)
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def weighted_grad(h, weights):
    n = len(h)
    grad = np.zeros(n)
    offset = n - 1
    for k in range(len(weights)):
        L = k - offset
        w = weights[k]
        if w < 1e-14:
            continue
        if L >= 0:
            rem = n - L
            if rem > 0:
                grad[L:] += w * (1.0 - h[:rem])
                grad[:rem] -= w * h[L:]
        else:
            M = -L
            rem = n - M
            if rem > 0:
                grad[:rem] += w * (1.0 - h[M:])
                grad[M:] -= w * h[:rem]
    return grad

def lse_obj_jac(x, alpha, target, lam, top_k=None):
    corr = compute_corr(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    ss = np.sum(exp_s)
    lse_val = c_max + np.log(ss) / alpha
    w = exp_s / ss
    if top_k is not None and top_k < len(w):
        idx = np.argsort(w)[-top_k:]
        ws = np.zeros_like(w)
        ws[idx] = w[idx]
        w = ws
    g = weighted_grad(x, w)
    pen = lam * (np.sum(x) - target)
    g = g + 2.0 * pen * np.ones(len(x))
    val = lse_val + pen * (np.sum(x) - target)
    return val, g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
