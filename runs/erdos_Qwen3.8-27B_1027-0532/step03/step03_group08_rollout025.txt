import time
import numpy as np
from scipy.optimize import minimize

_global_n = 0
_global_target = 0.0


def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))


def _corr_vals(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx


def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 1e-12:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= -1e-12:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x


def _grad_lag(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] = (1.0 - h[:m])
            grad[:m] = grad[:m] - h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] = (1.0 - h[j:])
            grad[j:] = grad[j:] - h[:m]
    return grad * dx


def _nesterov_focus(h0, target_sum, n_iters, alpha0, top_k, sg_alpha, deadline=None):
    n = len(h0)
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = _corr_vals(y)
        idx = np.argsort(corr)[::-1][:top_k]
        vals = corr[idx]
        m_val = vals.max()
        w = np.exp(sg_alpha * (vals - m_val))
        total = w.sum()
        if total < 1e-300:
            w = np.zeros(len(idx))
            w[0] = 1.0
        else:
            w = w / total
        g = np.zeros(n)
        for i in range(len(idx)):
            if w[i] > 1e-10:
                g += w[i] * _grad_lag(y, idx[i], dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val


def _slsqp_wrap(h0, target_sum, alpha, maxiter, deadline=None):
    n = len(h0)
    global _global_n, _global_target
    _global_n = n
    _global_target = target_sum
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        w = np.exp(alpha * (corr - m))
        total = w.sum()
        if total < 1e-300:
            idx = np.argmax(corr)
            return _grad_lag(x, idx, dx)
        w = w / total
        g = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-6:
                g += w[k_idx] * _grad_lag(x, k_idx, dx)
        return g

    def eq_fun(x):
        return x.sum() - target_sum

    def eq_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _project_box_sum(res.x, target_sum)
    except Exception:
        return h0


def _coord_descent_cd(h0, target_sum, n_rounds, deadline=None):
    n = len(h0)
    h = h0.copy()
    best_c = _c5_of(h)
    for r in range(n_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(0, n, 1):
            if deadline is not None and time.time() > deadline:
                break
            old = h[j]
            if old <= 1e-6 or old >= 1.0 - 1e-6:
                continue
            deltas = np.array([-0.15, -0.08, -0.03, 0.03, 0.08, 0.15])
            for d in deltas:
                h[j] = np.clip(old + d, 0.0, 1.0)
                h = _project_box_sum(h, target_sum)
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    improved = True
                else:
                    h[j] = old
            if old > 0.1 and old < 0.9:
                h[j] = (old + (0.5 if old < 0.5 else 0.5)) / 2.0
                h = _project_box_sum(h, target_sum)
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    improved = True
                else:
                    h[j] = old
        if not improved:
            break
    return h, best_c

def _downsample(h, n_new):

    x_old = np.linspace(0, n_old, n_old + 1)
    x_new = np.linspace(0, n_old, n_new + 1)
    h_new = np.interp(x_new[:-1], x_old[:-1], h, period=n_old)
    return _project_box_sum(np.clip(h_new, 0.0, 1.0), n_new / 2.0)

    t_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(t_new, t_old, h)

def _gen_diverse_starts(h_base, n, target_sum, rng, count):

        if mod == 1:
            s = _project_box_sum(h_base + 0.2 * (rng.rand(n) - 0.5), target_sum)
        elif mod == 2:
            t = rng.uniform(0.3, 0.7)
            s = _project_box_sum(t * h_base + (1.0 - t) * rng.rand(n), target_sum)
        elif mod == 3:
            shift = rng.randint(1, n // 4)
            s = _project_box_sum(np.roll(h_base, shift), target_sum)
        elif mod == 4:
            s = _project_box_sum(1.0 - h_base + 0.05 * (rng.rand(n) - 0.5), target_sum)
        elif mod == 5:
            bs = max(2, n // 8)
            s = h_base.copy()
            s1 = rng.randint(0, n - 2 * bs)
            s2 = rng.randint(0, n - 2 * bs)
            s[s1:s1+bs] = h_base[s2:s2+bs].copy()
            s[s2:s2+bs] = h_base[s1:s1+bs].copy()
            s = _project_box_sum(s, target_sum)
        elif mod == 6:
            s = _project_box_sum(h_base + 0.5 * rng.standard_normal(n), target_sum)
        else:
            freq = rng.randint(2, 10)
            phase = rng.uniform(0, 2 * np.pi)

            s = _project_box_sum(np.clip(h_base + pert, 0, 1), target_sum)

    for n in n_candidates:

    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
