def softmax_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    # Numerical stability
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    
    # Gradient computation (vectorized)
    grad = np.zeros(n)
    for i in range(n):
        # h[i] appears in corr[k] for k in range:
        # corr[k] = sum_j h[j] * (1-h)[j + k - n + 1]
        # h[i] as first factor: j=i, need 0<=i+k-n+1<n → n-1-i <= k <= 2n-2-i
        klo = max(0, n - 1 - i)
        khi = min(2*n - 2, 2*n - 2 - i)
        term_a = np.sum(w[klo:khi+1])
        # For each such k, the (1-h) factor is at index i + k - n + 1
        h_idx = i + np.arange(klo, khi+1) - n + 1
        term_b = np.sum(w[klo:khi+1] * h[h_idx])
        
        # h[i] as second factor: (1-h)[i] means j + k - n + 1 = i → j = i - k + n - 1
        # Need 0<=j<n → 0 <= i-k+n-1 < n → i+n-1 >= k and i < k
        # So k > i and k <= i + n - 1
        klo_c = max(i + 1, 0)  # k > i means k >= i+1
        khi_c = min(i + n - 1, 2*n - 2)
        if klo_c <= khi_c:
            j_idx = i - np.arange(klo_c, khi_c+1) + n - 1
            term_c = np.sum(w[klo_c:khi_c+1] * h[j_idx])
        else:
            term_c = 0.0
        
        grad[i] = term_a - term_b - term_c
    
    return F, grad

    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x):
        F, _ = softmax_obj_and_grad(x, alpha)
        return F
    
    def jac_func(x):
        _, g = softmax_obj_and_grad(x, alpha)
        return g

        res = minimize(obj_func, project_feasible(h_start.copy(), target),
                      jac=jac_func, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_start.copy(), target)

def interpolate(h_src, n_dst, target_dst):
    if len(h_src) == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

                                              [10, 30, 100, 300, 1000], 300, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 3: Binarization + re-optimization
    n_use = len(best_h)
    target_use = n_use / 2.0
    for lo, hi in [(0.03, 0.97), (0.08, 0.92), (0.12, 0.88)]:
        if tleft() < 50:
            break
        h_bin = snip_binary(best_h, lo, hi, target_use)
        h_refined, sc = refine_alpha_schedule(h_bin, target_use,
                                              [50, 200, 500, 1500], 300, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 4: Subgradient + L-BFGS-B hybrid
    if tleft() > 80:
        h_sg, sc_sg = subgrad_optimize(best_h, target_use, maxiter=150, frac=0.06)
        if sc_sg < best_score:
            best_score = sc_sg
            best_h = h_sg
        h_refined, sc = refine_alpha_schedule(h_sg, target_use, [100, 400, 1000], 300, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 5: Perturbation search
    for attempt in range(18):
        if tleft() < 40:
            break
        ptype = attempt % 4
        if ptype == 0:
            amp = 0.005 + 0.005 * (attempt // 4)
            h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
        elif ptype == 1:
            seg_len = rng.randint(4, max(5, n_use // 12))
            a = rng.randint(0, n_use - seg_len)
            h_pert = best_h.copy()
            h_pert[a:a + seg_len] = 1.0 - h_pert[a:a + seg_len]
            h_pert = project_feasible(h_pert, target_use)
        elif ptype == 2:
            shift = rng.randint(1, max(2, n_use // 10))
            h_shifted = np.roll(best_h, shift)
            w_blend = 0.5 + 0.3 * rng.random()
            h_pert = project_feasible(w_blend * best_h + (1 - w_blend) * h_shifted, target_use)
        else:
            # Swap two segments
            seg_len = rng.randint(3, max(4, n_use // 15))
            a = rng.randint(0, n_use - seg_len)
            b = rng.randint(0, n_use - seg_len)
            if abs(a - b) > seg_len + 1:
                h_pert = best_h.copy()
                tmp = h_pert[a:a + seg_len].copy()
                h_pert[a:a + seg_len] = h_pert[b:b + seg_len]
                h_pert[b:b + seg_len] = tmp
                h_pert = project_feasible(h_pert, target_use)
            else:
                h_pert = project_feasible(best_h + rng.randn(n_use) * 0.01, target_use)

        h_refined, sc = refine_alpha_schedule(h_pert, target_use,
                                              [30, 100, 400, 1000], 200, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 6: Final high-resolution push
    if tleft() > 120:
        n_final = max(1000, n_use)
        if n_final > 1200:
            n_final = 1200
        target_final = n_final / 2.0
        h_interp = interpolate(best_h, n_final, target_final)
        h_refined, sc = refine_alpha_schedule(h_interp, target_final,
                                              [30, 100, 300, 1000, 3000], 400, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 7: Final polish with very high alpha
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 40:
        h_refined, sc = refine_alpha_schedule(best_h, target_use,
                                              [3000, 8000, 15000], 500, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Final micro-perturbation
    for _ in range(5):
        if tleft() < 20:
            break
        amp = rng.uniform(0.001, 0.004)
        h_pert = project_feasible(best_h + rng.randn(n_use) * amp, target_use)
        h_new, sc = refine_alpha_schedule(h_pert, target_use, [5000, 12000], 300, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    h_final = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
