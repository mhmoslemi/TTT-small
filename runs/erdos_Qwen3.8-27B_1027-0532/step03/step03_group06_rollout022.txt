= ∑_i δh[i](1-h[i+k]) - ∑_i h[i-k]δh[i]
= ∑_i δh[i]((1-h[i+k]) - h[i-k])

The new max is approximately max_k (corr[k] + δ * (s_k[i] - s_k[j]))

**Mirror Descent / Frank-Wolfe approach:**

The Frank-Wolfe (conditional gradient) method for minimizing f(h) over a convex polytope P:
1. Compute subgradient g at current point h
2. Find v = argmin_{v ∈ P} g · v (the "linear oracle")
3. Update: h ← (1-t)h + t*v where t is a step size

For our problem, P = {h : 0 ≤ h ≤ 1, ∑h = n/2}.

The linear oracle: min g·h s.t. 0≤h≤1, ∑h=n/2.
Solution: sort g, set h[i]=1 for the n/2 smallest g[i], h[i]=0 for the rest.
(With ties broken arbitrarily.)

This is O(n log n) per iteration!

The Frank-Wolfe method is guaranteed to converge for convex problems, but our problem is non-convex. However, it might still work well in practice because:

    dx = 2.0 / n
    v = np.zeros(n)  # momentum
    beta = 0.9
    step = 0.01
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for it in range(max_iters):
        if time_limit_fn() < 30:
            break
        
        # Compute all overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        f_max = np.max(corr)
        k_star = np.argmax(corr)
        
        # Subgradient of f_max
        k = k_star - (n - 1)
        s = np.zeros(n)
        if k >= 0:
            idx = np.arange(n - k)
            s[idx] += (1.0 - h[idx + k])
            s[idx + k] += (-h[idx])

            kp = -k
            idx = np.arange(n - kp)
            s[idx + kp] += (1.0 - h[idx])
            s[idx] += (-h[idx + kp])
        
        # Momentum update

        # Part 2: h[i] appears as the second factor (1-h[i])
        # h[i] = h[j+shift] where j+shift=i, so j=i-shift
        # Need 0<=j<n => 0<=i-shift<n => i-n+1 <= shift <= i

        hi2 = min(2 * n - 2, n - 1 + i)

            j_vals = i - shifts2
            valid2 = (j_vals >= 0) & (j_vals < n)
            wv2 = w[lo2:hi2+1]
            hv2 = h[j_vals[valid2]]
            grad[i] += np.sum(-wv2[valid2] * hv2)
    
    return F, grad


def subgradient_descent(h_start, target, max_iters, time_limit_fn, seed=42):
    """Projected subgradient with momentum"""
    n = len(h_start)
    rng = np.random.RandomState(seed)
    h = h_start.copy()
    v = np.zeros(n)
    beta = 0.85
    best_h = h.copy()
    best_f = compute_c5(h)
    step0 = 0.005
    no_improve = 0
    patience = 300
    
    for it in range(max_iters):
        if time_limit_fn() < 30:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        s = get_subgradient(h, k)
        
        v = beta * v + (1.0 - beta) * s
        step = step0 * (1.0 + it / 2000.0) ** (-0.5)
        
        h_new = h - step * v
        h_new = project_feasible(h_new, target)
        
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h = h_new
        
        # Restart if stuck
        if no_improve >= patience:
            h = project_feasible(best_h + rng.randn(n) * 0.02, target)
            v = np.zeros(n)
            no_improve = 0
    
    return best_h, best_f


def lbfgs_smooth(h_start, n, target, alpha, maxiter=200):
    """L-BFGS-B with smooth objective"""
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return smooth_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return smooth_obj_and_grad(x, alpha)[1]
    
    try:
        res = minimize(obj, h_start, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h_start

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def get_subgradient(h, shift):
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        m = n - shift
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + shift])
            grad[idx + shift] -= h[idx]
    else:
        kp = -shift
        m = n - kp
        if m > 0:
            idx = np.arange(m)
            grad[idx + kp] += (1.0 - h[idx])
            grad[idx] -= h[idx + kp]
    return grad


def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    cs = corr - m
    ev = np.exp(alpha * cs)
    se = np.sum(ev)
    w = ev / se
    F = m + np.log(se) / alpha
    
    grad = np.zeros(n)
    for j in range(n):
        lo1 = n - 1 - j
        hi1 = 2 * n - 2 - j
        s1 = np.arange(lo1, hi1 + 1) - (n - 1)
        grad[j] += np.sum(w[lo1:hi1+1] * (1.0 - h[j + s1]))
        lo2 = j
        hi2 = j + n - 1
        s2 = np.arange(lo2, hi2 + 1) - (n - 1)
        grad[j] += np.sum(-w[lo2:hi2+1] * h[j - s2])
    
    return F, grad


def subgrad_momentum(h0, target, n_iter, tleft_fn, seed):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_f = compute_c5(h)
    no_impr = 0
    
    for it in range(n_iter):
        if tleft_fn() < 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = np.argmax(corr)
        shift = ks - (n - 1)
        s = get_subgradient(h, shift)
        v = beta * v + (1.0 - beta) * s
        step = 0.003 / (1.0 + it / 3000.0)
        h = project_feasible(h - step * v, target)
        f = compute_c5(h)
        if f < best_f:
            best_f = f
            best_h = h.copy()
            no_impr = 0
        else:
            no_impr += 1
        if no_impr > 500:
            h = project_feasible(best_h + rng.randn(n) * 0.03, target)
            v = np.zeros(n)
            no_impr = 0
    return best_h, best_f


def lbfgs_run(h0, target, alpha, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lambda x: smooth_obj_grad(x, alpha)[0],
                       h0.copy(),
                       jac=lambda x: smooth_obj_grad(x, alpha)[1],
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return h0


def local_refine(h0, target, tleft_fn, seed, n_iter=3000):
    n = len(h0)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for it in range(n_iter):
        if tleft_fn() < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        ks = np.argmax(corr)
        shift = ks - (n - 1)
        s = get_subgradient(h, shift)
        
        # Greedy: move opposite to subgradient at the worst coordinate
        # Try a few random steps guided by subgradient
        for _ in range(3):
            if rng.random() < 0.5:
                # Move in subgradient direction
                idx = rng.choice(n)
                delta = -np.sign(s[idx]) * rng.uniform(0.002, 0.01)
            else:
                # Random
                idx = rng.choice(n)
                delta = rng.uniform(-0.01, 0.01)
            
            h_new = h.copy()
            h_new[idx] += delta
            h_new = project_feasible(h_new, target)
            f_new = compute_c5(h_new)
            if f_new < best_f:
                best_f = f_new
                best_h = h_new.copy()
                h = h_new
                break
    
    return best_h, best_f
