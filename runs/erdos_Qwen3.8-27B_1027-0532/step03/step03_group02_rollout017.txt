```python
def build_and_solve_lp(h_curr, active_indices, n):
    s_arr = np.arange(2*n - 1) - (n-1)
    m = len(active_indices)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    corr_curr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
    for i, k in enumerate(active_indices):
        s = int(s_arr[k])
        grad = np.zeros(n)
        if s == 0:
            grad[:] = 1.0 - 2.0 * h_curr
        else:
            if s >= 0:
                idx1 = np.arange(n - s)
                grad[idx1] += (1.0 - h_curr[idx1 + s])
            else:
                idx1 = np.arange(-s, n)
                grad[idx1] += (1.0 - h_curr[idx1 + s])
            if s > 0:
                idx2 = np.arange(s, n)
                grad[idx2] -= h_curr[idx2 - s]
            else:
                idx2 = np.arange(0, n + s)
                grad[idx2] -= h_curr[idx2 - s]
        A_ub[i, :n] = grad
        A_ub[i, n] = -1.0
        b_ub[i] = -(corr_curr[k] - np.dot(grad, h_curr))
    A_eq = np.ones((1, n + 1))
    A_eq[0, n] = 0.0
    b_eq = np.array([n / 2.0])
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n].copy()
    except Exception:
        pass
    return None

def slp_refine(h, target_sum, time_limit, start_time):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_score = compute_c5(h)
    best_h = h.copy()
    stagnant = 0
    it = 0
    while time.time() - start_time < time_limit and it < 500:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = float(np.max(corr))
        score = c_max * dx
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                break
        # Active set: top 20 or threshold-based
        threshold = c_max - 0.5 * n * dx  # relative to dx scale
        active_mask = corr >= threshold
        if np.sum(active_mask) < 5:
            idx_top = np.argsort(corr)[::-1][:10]
            active_indices = idx_top
        elif np.sum(active_mask) > 40:
            idx_top = np.argsort(corr)[::-1][:30]
            active_indices = idx_top
        else:
            active_indices = np.where(active_mask)[0]
        h_new = build_and_solve_lp(h, active_indices, n)
        if h_new is None:
            break
        h_new = project_feasible(h_new, target_sum)
        score_new = compute_c5(h_new)
        if score_new < score - 1e-14:
            h = h_new
        else:
            h_mid = 0.5 * (h + h_new)
            h_mid = project_feasible(h_mid, target_sum)
            score_mid = compute_c5(h_mid)
            if score_mid < score - 1e-14:
                h = h_mid
            else:
                # Try smaller step
                h_q = 0.25 * h + 0.75 * h_new
                h_q = project_feasible(h_q, target_sum)
                score_q = compute_c5(h_q)
                if score_q < score - 1e-14:
                    h = h_q
                else:
                    break
    
    return best_h, best_score
```

A[k] = (1-h[j+k-(n-1)]) * [n-1-j ≤ k ≤ n-1+j] - h[j-k+(n-1)] * [j-(n-1) ≤ k ≤ j]

This can be computed with numpy slicing:

```python
k_start1 = max(0, n-1-j)
k_end1 = min(2*n-1, n-1+j+1)  # +1 for numpy slice end
idx1 = np.arange(k_start1, k_end1)
A[idx1] += (1.0 - h[j + idx1 - (n-1)])

# Second term: k from max(0, j-(n-1)) to min(2n-2, j)
k_start2 = max(0, j-(n-1))
k_end2 = min(2*n-1, j+1)
idx2 = np.arange(k_start2, k_end2)
A[idx2] -= h[j - idx2 + (n-1)]
```
