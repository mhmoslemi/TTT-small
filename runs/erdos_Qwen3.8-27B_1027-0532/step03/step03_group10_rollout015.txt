```python
        if s > target_sum:

        else:

def smooth_max_and_grad(vals, eps):
    """Smooth max with gradient"""
    cmax = np.max(vals)
    x = (vals - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
```

```python
    n = len(h_init)
    h = h_init.astype(np.float64).copy()
    target_sum = n / 2.0
    h = project_onto_constraint(h, target_sum)
    
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for it in range(n_iterations):
        # Compute gradient
        val, grad = gradient_c5_smooth(h, eps)
        
        # Adaptive step size
        step = lr / (1.0 + it * 0.001)
        
        # Gradient step
        h_new = h - step * grad
        
        # Project
        h_new = project_onto_constraint(h_new, target_sum)
        
        # Check improvement
        val_new = compute_c5(h_new)
        if val_new < val:
            h = h_new
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # Backtrack
            step *= 0.5
            h_new = h - step * grad
            h_new = project_onto_constraint(h_new, target_sum)
            val_new = compute_c5(h_new)
            if val_new < val:
                h = h_new
                val = val_new
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
    
    return best_h, best_val
```

```python
            if local_max > current_max:
                current_max = local_max
                max_pos = k_min + int(np.argmax(corr[k_min:k_max+1]))
        
        delta_c5 = current_max - old_max

            h[i] = 0; h[j] = 1
            ones_pos[i_idx] = j; zeros_pos[j_idx] = i
            if current_max < best:
                best = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
    
    return best_h, best
```

This should be faster because we avoid the full `np.max(corr)` in most cases. The `np.max(corr[k_min:k_max+1])` is only over the affected range, which is at most n values.

Actually, `np.max(corr[k_min:k_max+1])` is still O(n) in the worst case (when s_min=0 and s_max=n-1, the range is the full array). But in practice, for most i,j pairs, the range is smaller.

The expected range size: s_min = max(-i, i-half, -j, j-half) and s_max = min(half-i, i, half-j, j). For random i,j in [0,n-1], the range is roughly n/2 on average. So we're not saving much.

Let me just use the simple version with full `np.max(corr)` and focus on getting more steps done.

For the speed, the key bottlenecks are:
1. `rng.integers(n_one)` - fast, ~100ns
2. `np.arange(s_min, s_max+1)` - creates an array, ~1μs
3. Array indexing `h[i + s_range]` - ~1μs each, 4 of them
4. `corr[k_range] += delta` - ~1μs
5. `np.max(corr)` - ~1μs
6. Python loop overhead - ~1μs

Total per step: ~10-15 μs. For 10M steps: 100-150 seconds. That's feasible!

Let me now write the complete solution. I'll structure it as:

1. Helper functions (projection, c5 computation, etc.)
2. Fast binary SA
3. Soft optimization (projected gradient)
4. Main `run` function with time management

Here's my final code plan:

import numpy as np
from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def gradient_c5_smooth(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    val = cmax + eps * np.log(sexp)
    
    grad = np.zeros(n)
    half = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - half
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return val, grad

def to_binary(h):
    n = len(h)
    k = n // 2
    threshold = np.partition(h, n - k)[n - k]
    h_bin = (h >= threshold).astype(float)
    # Ensure exactly k ones
    n_ones = int(h_bin.sum())
    if n_ones > k:
        # Set some 1s to 0
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by h value (lowest h first among the ones)
        order = np.argsort(h[ones_idx])
        h_bin[ones_idx[order[:n_ones - k]]] = 0
    elif n_ones < k:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        h_bin[zeros_idx[order[:k - n_ones]]] = 1
    return h_bin

def binary_sa(h_start, n_steps, t_start, t_end, rng, check_interval=10000):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    half = n - 1
    current_max = float(np.max(corr))
    best = current_max
    best_h = h.copy()
    
    ones_pos = list(np.where(h == 1.0)[0])
    zeros_pos = list(np.where(h == 0.0)[0])
    n_half = len(ones_pos)
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        i_idx = int(rng.integers(n_half))
        j_idx = int(rng.integers(n_half))
        i = ones_pos[i_idx]
        j = zeros_pos[j_idx]
        
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        
        if s_min > s_max:
            h[i] = 0.0; h[j] = 1.0
            ones_pos[i_idx] = j; zeros_pos[j_idx] = i
            continue
        
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
        
        old_max = current_max
        corr[k_range] += delta
        new_max = float(np.max(corr))
        
        if new_max < old_max - 1e-15 or (t > 1e-10 and rng.random() < np.exp(-(new_max - old_max) / t)):
            h[i] = 0.0; h[j] = 1.0
            ones_pos[i_idx] = j; zeros_pos[j_idx] = i
            current_max = new_max
            if current_max < best:
                best = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
    
    return best_h, best

def soft_optimize(h_init, n_iters, eps_start, eps_end, lr, rng):
    n = len(h_init)
    target = n / 2.0
    h = project_onto_constraint(h_init.copy(), target)
    best_val = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        eps = eps_start * (eps_end / eps_start) ** frac
        step = lr * (1.0 - frac * 0.5)
        
        val, grad = gradient_c5_smooth(h, eps)
        h_new = h - step * grad
        h_new = project_onto_constraint(h_new, target)
        
        val_new = compute_c5(h_new)
        if val_new < val:
            h = h_new
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h.copy()

            h_new2 = h - 0.5 * step * grad
            h_new2 = project_onto_constraint(h_new2, target)
            val_new2 = compute_c5(h_new2)
            if val_new2 < val:
                h = h_new2
                val = val_new2
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
    
    return best_h, best_val
