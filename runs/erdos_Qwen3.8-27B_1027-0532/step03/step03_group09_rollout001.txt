import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(x, eps):
    corr = compute_corr_scaled(x)
    cmax = np.max(corr)
    z = (corr - cmax) / eps
    z -= np.max(z)
    return cmax + eps * np.log(np.sum(np.exp(z)))

def smooth_max_grad(x, eps):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr_scaled(x)
    cmax = np.max(corr)
    z = (corr - cmax) / eps
    z -= np.max(z)
    ex = np.exp(z)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    omh = 1.0 - x
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * x[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * x[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def proj_grad_refine(h, k_top, temp, lr_init, n_steps, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    stall_limit = 400

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr_scaled(h_cur)
        topk_idx = np.argsort(corr)[-k_top:]
        xc = corr[topk_idx]
        xc = xc - np.max(xc)
        w = np.exp(xc / temp)
        w = w / np.sum(w)
        grad = np.zeros(n)
        omh = 1.0 - h_cur
        for i in range(len(topk_idx)):
            wk = w[i]
            if wk < 1e-15:
                continue
            kidx = topk_idx[i]
            s = kidx - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * omh[:m]
                grad[:m] -= wk * h_cur[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * omh[r:]
                grad[r:] -= wk * h_cur[:m]
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.005, 0.15)
        else:

            lr = max(lr * 0.98, 1e-9)
        h_cur = h_new

            h_cur = best_h.copy()
            lr = max(lr_init * 0.6, 1e-6)
            stall = 0
            stall_limit = max(100, stall_limit // 2)
    return best_h, best_c5

def local_search_swap(h, n_trials, target_sum, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_trials):
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = 0.005 * rng.standard_normal()
        h_try = best_h.copy()
        h_try[i] = np.clip(h_try[i] + delta, 0.0, 1.0)
        h_try[j] = np.clip(h_try[j] - delta * (best_h[i] - h_try[i]) / max(abs(delta), 1e-12), 0.0, 1.0)
        h_try = project_onto_constraint(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    return best_h, best_c5

def gen_perturb(h_best, rng, n0, target_sum, strategy, scale=1.0):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.008 * scale * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        n_swap = max(3, n0 // 12)
        idx = rng.choice(n0, size=n_swap, replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.08 * scale * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        max_len = max(2, n0 // 8)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.004 * scale * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
        h_bin = (rng.random(n0) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.75, 0.92))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    else:
        return project_onto_constraint(h_best + 0.003 * scale * rng.standard_normal(n0), target_sum)
