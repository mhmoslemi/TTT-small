def _corr_of(h):

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:

        else:
            hi = mid

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _min_norm_subgrad(h, n):
    dx = 2.0 / n
    corr = _corr_of(h)
    max_corr = np.max(corr)
    threshold = max_corr * 0.99
    active = np.where(corr >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    if len(active) == 1:
        return _grad_corr_term(h, active[0], dx).copy()
    grads = np.array([_grad_corr_term(h, k, dx) for k in active])
    m = len(active)
    A = grads @ grads.T
    try:
        A_inv = np.linalg.inv(A + 1e-12 * np.eye(m))
        ones_vec = np.ones(m)
        lam = A_inv @ ones_vec
        denom = ones_vec @ lam
        if denom > 1e-15:
            lam = lam / denom
        if np.all(lam >= -1e-10):
            lam = np.clip(lam, 0.0, 1.0)
            s = lam.sum()
            if s > 1e-15:
                lam = lam / s
                return grads.T @ lam
    except Exception:
        pass
    try:
        res = minimize(lambda v: float(v @ A @ v), np.ones(m) / m,
                       method='SLSQP', bounds=[(0, 1)] * m,
                       constraints=[{'type': 'eq', 'fun': _qobj_eq, 'jac': _qobj_jac}],
                       options={'maxiter': 50, 'ftol': 1e-14})
        lam = np.clip(res.x, 0.0, 1.0)
        s = lam.sum()
        if s > 1e-15:
            lam = lam / s
        else:
            lam = np.ones(m) / m
        return grads.T @ lam
    except Exception:
        return grads.T @ (np.ones(m) / m)

def _qobj_eq(v):
    return v.sum() - 1.0

def _qobj_jac(v):
    return np.ones(len(v))

def _subgrad_method(h0, n, target_sum, n_iters, alpha0, deadline):
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        g = _min_norm_subgrad(x, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        alpha = alpha0 / np.sqrt(float(it + 1)) / g_norm
        best_alpha = alpha
        best_x = _project(x - alpha * g, target_sum)
        best_x_val = _c5_of(best_x)
        for _bt in range(25):
            if best_x_val < best_val - 1e-15:
                break
            alpha *= 0.6
            cand = _project(x - alpha * g, target_sum)
            cand_val = _c5_of(cand)
            if cand_val < best_x_val:
                best_x_val = cand_val
                best_x = cand
                best_alpha = alpha
        x = best_x
        if best_x_val < best_val:
            best_val = best_x_val
            best = x.copy()
    return best, best_val

def _nesterov_mnsg(h0, n, target_sum, n_iters, alpha0, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        g = _min_norm_subgrad(y, n)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-16:
            break
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g / g_norm, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _sa_explore(h0, n, target_sum, T0, T_end, n_steps, deadline, rng):
    x = h0.copy()
    best = x.copy()
    best_val = _c5_of(x)
    cur_val = best_val
    for it in range(n_steps):
        if time.time() > deadline:
            break
        frac = float(it) / max(n_steps - 1, 1)
        T = T0 * (T_end / T0) ** frac
        n_change = rng.randint(1, max(2, n // 15) + 1)
        idx = rng.choice(n, size=n_change, replace=False)
        x_new = x.copy()
        for i in idx:
            x_new[i] = np.clip(x_new[i] + rng.normal(0, 0.15), 0.0, 1.0)
        x_new = _fix_sum(x_new, target_sum)
        new_val = _c5_of(x_new)
        delta = new_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / (T + 1e-12)):
            x = x_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = x.copy()
    return best, best_val

def _pairwise_descent(h0, n, target_sum, max_rounds, deadline):

    for rnd in range(max_rounds):

                    h[i] += delta

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    dx = 2.0 / n
    global _g_ts, _g_n
    _g_ts = target_sum
    _g_n = n

    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        if m < -500:
            return m * dx
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * dx

    def _eq_fun(x):
        return x.sum() - _g_ts

    def _eq_jac(x):
        return np.ones(_g_n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_obj, h0.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0.copy()
