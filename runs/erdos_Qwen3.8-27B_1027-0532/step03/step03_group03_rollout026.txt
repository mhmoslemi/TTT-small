def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.minimum(1.0, h[idx] + per)
        # Re-clip and redistribute if needed
        h = np.clip(h, 0.0, 1.0)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.maximum(0.0, h[idx] + per)
        h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_pg(h_init, target_sum, n_iter, alpha_lse, tol=1e-14):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    
    t = 1.0
    t0 = 10.0  # Nesterov offset
    
    for it in range(n_iter):
        # Compute log-sum-exp objective and gradient at y
        corr = compute_corr(y)
        f_vals = corr * dx  # These are the individual "costs"
        
        # Log-sum-exp
        m = np.max(f_vals)
        lse = m + np.log(np.sum(np.exp(f_vals - m))) / alpha_lse
        g = np.exp(alpha_lse * (f_vals - m))
        g = g / np.sum(g)
        
        # Gradient of log-sum-exp w.r.t. h
        # ∂/∂h[j] of corr[k] * dx
        # corr[k] = sum_i h[i] * (1-h[i+k-(n-1)])  (with valid indices)
        # ∂corr[k]/∂h[j] = (1-h[j+k-(n-1)]) - h[j-k-(n-1)]  (with valid indices)
        grad = np.zeros(n)
        for k in range(2*n-1):
            if g[k] < 1e-15:
                continue
            lag = k - (n-1)
            if lag >= 0:
                # corr[k] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
                # ∂/∂h[j] = (1-h[j+lag]) for j in [0, n-1-lag] (first factor)
                #           - h[j-lag] for j in [lag, n-1] (second factor)
                if lag < n:
                    grad[:n-lag] += g[k] * (1.0 - y[lag:])
                    grad[lag:] -= g[k] * y[:n-lag]
            else:
                m2 = -lag
                # corr[k] = sum_{i=m2}^{n-1} h[i] * (1-h[i-m2])
                if m2 < n:
                    grad[m2:] += g[k] * (1.0 - y[:n-m2])
                    grad[:n-m2] -= g[k] * y[m2:]
        
        grad *= dx
        
        # Nesterov step
        step = 2.0 / (t + t0)
        h_new = y - step * grad
        h_new = project_feasible(h_new, target_sum)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t*t)) / 2.0
        y_new = h_new + (t / t_next) * (h_new - h)
        y_new = project_feasible(y_new, target_sum)
        
        h = h_new
        y = y_new
        t = t_next
        
        f_val = compute_c5(h)
        if f_val < best_f:
            best_f = f_val
            best_h = h.copy()
    
    return best_h, best_f

def coordinate_search(h, target_sum, n_sweeps=100, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(n_sweeps):
        max_change = 0.0
        for j in range(n):
            # Compute c_k for this coordinate
            # c_k = (1-h[j+s_k]) - h[j-s_k] (with 0 for invalid)
            corr = compute_corr(h)
            c = np.zeros(2*n-1)
            for k in range(2*n-1):
                s = k - (n-1)
                val = 0.0
                if 0 <= j+s <= n-1:
                    val += (1.0 - h[j+s])
                if 0 <= j-s <= n-1:
                    val -= h[j-s]
                c[k] = val
            
            # Now minimize max_k (corr[k] + delta * c[k]) * dx + mu * delta
            # subject to: -h[j] <= delta <= 1-h[j] and sum constraint
            
            # For the sum constraint, we'll handle it by adjusting another coordinate
            # after the 1D search. For now, ignore it and fix later.
            
            # 1D convex minimization: f(delta) = max_k (corr[k] + delta * c[k]) * dx
            # Find minimum over delta in [-h[j], 1-h[j]]
            
            # The minimum of max of linear functions occurs where two lines cross
            # at the boundary, or at a breakpoint
            # Simple approach: ternary search
            a = -h[j]
            b = 1.0 - h[j]
            
            for _ in range(50):  # 50 iterations of ternary search gives ~1e-10 precision
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * c) * dx
                f2 = np.max(corr + m2 * c) * dx
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            
            delta = (a + b) / 2.0
            new_val = h[j] + delta
            new_val = np.clip(new_val, 0.0, 1.0)
            max_change = max(max_change, abs(new_val - h[j]))
            h[j] = new_val
        
        # Project to satisfy sum constraint
        h = project_feasible(h, target_sum)
        
        if max_change < tol:
            break
    
    return h

- In each 1D subproblem, minimize max_k corr_k * dx + μ * δ
- After each sweep, update μ based on the constraint violation

```python
def coordinate_descent_alaug(h_init, target_sum, n_sweeps=50, n_search=40, 
                              mu_init=0.0, mu_lr=0.1):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    mu = mu_init
    n_corr = 2 * n - 1
    
    for sweep in range(n_sweeps):
        for j in range(n):
            corr = compute_corr(h)
            # Compute sensitivity c[k] = ∂corr[k]/∂h[j]
            c = np.zeros(n_corr)
            for k in range(n_corr):
                s = k - (n-1)
                val = 0.0
                if 0 <= j+s <= n-1:
                    val += (1.0 - h[j+s])
                if 0 <= j-s <= n-1:
                    val -= h[j-s]
                c[k] = val
            
            # 1D minimization: min over δ of max_k (corr[k] + δ*c[k])*dx + μ*δ
            # subject to -h[j] ≤ δ ≤ 1-h[j]
            a = -h[j]
            b = 1.0 - h[j]
            
            # Ternary search (convex function)
            for _ in range(n_search):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * c) * dx + mu * m1
                f2 = np.max(corr + m2 * c) * dx + mu * m2
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            
            delta = (a + b) / 2.0
            h[j] = np.clip(h[j] + delta, 0.0, 1.0)
        
        # Update multiplier
        violation = np.sum(h) - target_sum
        mu += mu_lr * violation / dx
        # Clip h to [0,1]
        h = np.clip(h, 0.0, 1.0)
    
    return project_feasible(h, target_sum)
```

Hmm, but this is still O(n * n_corr * n_search) in Python loops, which is slow. Let me vectorize the c[k] computation.

For a given j:
c[k] = (1-h[j+s_k]) * 1_{valid} - h[j-s_k] * 1_{valid}
where s_k = k - (n-1)

Let me compute this vectorized:
```python
s = np.arange(n_corr) - (n - 1)  # shape: (2n-1,)
# For first term: j+s must be in [0, n-1]
mask1 = (j + s >= 0) & (j + s < n)
# For second term: j-s must be in [0, n-1]
mask2 = (j - s >= 0) & (j - s < n)

c = np.zeros(n_corr)
c[mask1] += (1.0 - h[j + s[mask1]])
c[mask2] -= h[j - s[mask2]]
```

This is O(n) vectorized operations per coordinate. Total: O(n^2) per sweep. Fast!

And the ternary search: each evaluation is O(n_corr) = O(n). With n_search evaluations per coordinate: O(n * n_search) per coordinate. Total: O(n^2 * n_search) per sweep.

For n=400, n_search=40: 400^2 * 40 = 6.4 million numpy operations per sweep. At ~1μs per operation (array of size 800), that's ~6.4 seconds per sweep. With 50 sweeps, that's 320 seconds. That's reasonable!

Actually, I can speed up the ternary search by pre-computing the breakpoints. The function f(δ) = max_k (a_k + b_k * δ) where a_k = corr[k]*dx and b_k = c[k]*dx. The breakpoints are at δ_k = (a_i - a_j) / (b_j - b_i) for all pairs (i,j) where b_i ≠ b_j. There are O(n^2) breakpoints, but the minimum of the max always occurs at either a breakpoint or a boundary.

So I can:
1. Find all breakpoints: O(n^2)
2. Sort them: O(n^2 log n)
3. Evaluate f at each breakpoint and the boundaries: O(n^3) in the worst case

But O(n^3) = 400^3 = 64 million, which is too slow.

Let me stick with ternary search but reduce the number of iterations. 30 iterations of ternary search gives precision (b-a) * (2/3)^30 ≈ 1 * 10^(-6), which is plenty.

Actually, I realize there's an even better approach for the 1D subproblem. Since f(δ) = max_k (a_k + b_k * δ) is convex and piecewise linear, I can find its minimum by:
1. Compute the subgradient at δ=0: it's the set of b_k for all k that achieve the max
2. If 0 is in the subgradient, δ=0 is optimal
3. Otherwise, the minimum is at a breakpoint

A practical algorithm:
1. Start with δ = 0
2. Compute which constraint(s) are active: k* = argmax_k (a_k + b_k * δ)
3. If the active constraint has b_k* > 0, decrease δ (move left)
   If b_k* < 0, increase δ (move right)
   If b_k* = 0, we're at the minimum
4. Move to the next breakpoint in the appropriate direction
5. Repeat

Each step involves computing the max and finding the next breakpoint, which is O(n). Total: O(n * number_of_steps) = O(n^2) in the worst case.

But in practice, the function is "smooth" (many constraints are inactive), so the number of steps should be small.

Let me just use a simpler approach: a 1D bisection on the derivative (subgradient).

At a point δ, the subgradient of f(δ) = max_k (a_k + b_k * δ) is:
∂f(δ) = {Σ_λ b_k : λ_k ≥ 0, Σ λ_k = 1, a_k + b_k * δ = f(δ) for all k with λ_k > 0}

The minimum is where 0 ∈ ∂f(δ), i.e., there exist active constraints with b_k values that "straddle" 0.

Practical algorithm:
1. At δ, find max_val = max_k (a_k + b_k * δ)
2. Find all k with a_k + b_k * δ = max_val (active set)
3. Let b_min = min{b_k : k active}, b_max = max{b_k : k active}
4. If b_min ≤ 0 ≤ b_max: δ is optimal
5. If b_min > 0: all active constraints have positive slope, so f is decreasing. Move to the right.
   New δ = min over active k of the next breakpoint to the right.
6. If b_max < 0: move to the left.

This is essentially a "recycling" algorithm for 1D piecewise linear minimization. It should converge in a small number of steps.

But implementing this correctly is tricky. Let me just use the ternary search approach with 30 iterations. It's simple and reliable.

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TARGET = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-13)[0]
        else:
            idx = np.where(h > 1e-13)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

def _cons_fun(x):
    return np.sum(x) - _G_TARGET

def _cons_jac(x):
    return np.ones(_G_N)

def _lse_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    f = corr * dx
    m = np.max(f)
    return float(m + np.log(np.sum(np.exp(f - m))) / alpha)

def _lse_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = compute_corr(x)
    f = corr * dx
    m = np.max(f)
    raw = alpha * (f - m)
    exp_raw = np.exp(np.clip(raw, -500.0, 0.0))
    weights = exp_raw / np.sum(exp_raw)
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[:n - lag] += w * (1.0 - x[lag:])
                grad[lag:] -= w * x[:n - lag]
        else:
            m2 = -lag
            if m2 < n:
                grad[m2:] += w * (1.0 - x[:n - m2])
                grad[:n - m2] -= w * x[m2:]
    return grad * dx

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _G_N, _G_TARGET
    _G_N = len(h_start)
    _G_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _G_N
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    for alpha in alphas:
        try:
            res = minimize(_lse_obj, h_cur, args=(alpha,),
                           jac=_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def coord_descent(h_init, target_sum, n_sweeps, n_search, time_limit):
    """Coordinate descent with augmented Lagrangian for sum constraint."""
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    mu = 0.0
    mu_lr = 0.5
    n_corr = 2 * n - 1
    s = np.arange(n_corr) - (n - 1)  # pre-compute shifts
    start_t = time.time()
    
    for sweep in range(n_sweeps):
        for j in range(n):
            if time.time() - start_t > time_limit:
                return project_feasible(h, target_sum)
            
            corr = compute_corr(h)
            # Compute c[k] = partial corr[k] / partial h[j]
            c = np.zeros(n_corr)
            js = j + s
            jm = j - s
            mask1 = (js >= 0) & (js < n)
            mask2 = (jm >= 0) & (jm < n)
            if np.any(mask1):
                c[mask1] += (1.0 - h[js[mask1]])
            if np.any(mask2):
                c[mask2] -= h[jm[mask2]]
            
            # 1D minimization: min_{delta} max_k (corr[k] + delta*c[k]) * dx + mu * delta
            # subject to -h[j] <= delta <= 1 - h[j]
            a = -h[j]
            b = 1.0 - h[j]
            
            # Ternary search
            for _ in range(n_search):
                m1 = a + (b - a) / 3.0
                m2 = b - (b - a) / 3.0
                f1 = np.max(corr + m1 * c) * dx + mu * m1
                f2 = np.max(corr + m2 * c) * dx + mu * m2
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            
            delta = (a + b) / 2.0
            h[j] = np.clip(h[j] + delta, 0.0, 1.0)
        
        # Update multiplier
        violation = np.sum(h) - target_sum
        mu += mu_lr * violation
        h = np.clip(h, 0.0, 1.0)
    
    return project_feasible(h, target_sum)

def subgradient_optimize(h_init, target_sum, max_iter, lr_init, topk, temp,
                         noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    n_corr = 2 * n - 1
    
    for it in range(max_iter):
        corr = compute_corr(h)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(temp * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        
        grad = np.zeros(n)
        for i in range(len(top_indices)):
            wi = w[i]
            if wi < 1e-14:
                continue
            lag = top_indices[i] - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += wi * (1.0 - h[lag:])
                    grad[lag:] -= wi * h[:n - lag]
            else:
                m2 = -lag
                if m2 < n:
                    grad[m2:] += wi * (1.0 - h[:n - m2])
                    grad[:n - m2] -= wi * h[m2:]
        grad *= dx
        
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        
        score = compute_c5(h_new)
        if score
