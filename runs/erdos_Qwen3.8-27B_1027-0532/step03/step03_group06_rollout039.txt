from scipy.optimize import minimize, differential_evolution

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        # First term: h[i] as first factor
        p = i + np.arange(klo, khi + 1) - (n - 1)
        term_a = np.sum(w[klo:khi + 1] * (1.0 - h[p]))
        # Second term: h[i] as second factor
        q = i + (n - 1) - np.arange(klo, khi + 1)
        term_b = np.sum(w[klo:khi + 1] * h[q])
        grad[i] = term_a - term_b
    
    return F, grad

def lbfgs_optimize(h_start, bounds, alpha, maxiter=300):
    n = len(h_start)
    target = n / 2.0
    
    def obj(x):
        return smooth_obj_and_grad(x, alpha)[0]
    
    def jac(x):
        return smooth_obj_and_grad(x, alpha)[1]
    
    try:
        res = minimize(obj, h_start, jac=jac, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except:
        return h_start

def to_01(h, n):
    """Convert to 0-1 vector with exactly n/2 ones."""
    h_sorted = np.sort(h)[::-1]
    # Take top n/2 as 1, rest as 0
    threshold = h_sorted[n // 2 - 1]
    h01 = (h >= threshold).astype(float)
    # Fix if we don't have exactly n/2 ones
    n_ones = int(np.sum(h01))
    if n_ones > n // 2:

        # Remove the ones with lowest h values
        order = np.argsort(h[one_idx])
        for idx in order[:n_ones - n // 2]:
            h01[one_idx[idx]] = 0
    elif n_ones < n // 2:
        # Add some
        zero_idx = np.where(h01 == 0)[0]
        order = np.argsort(-h[zero_idx])
        for idx in order[:n // 2 - n_ones]:
            h01[zero_idx[idx]] = 1
    return h01

            corr = new_corr

            one_set.add(j)
            n_accepted += 1
        else:
            h[i] = 1
            h[j] = 0
    
    return h, curr * dx, n_accepted

        else:
            h[i] = 1
            h[j] = 0

    n = len(h)

    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)

    F = m + np.log(sum_exp) / alpha

        klo = max(0, n - 1 - i)

        grad[i] = np.sum(w[klo:khi+1] * (1.0 - h[p])) - np.sum(w[klo:khi+1] * h[q])
