import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.minimum(h[idx] + per, 1.0)
        # Handle edge case where some hit the cap
        remaining = target_sum - np.sum(h)
        if remaining > 1e-12:
            idx2 = np.where(h < 1.0)[0]
            if len(idx2) > 0:
                h[idx2] += remaining / len(idx2)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        per = diff / len(idx)
        h[idx] = np.maximum(h[idx] + per, 0.0)
        remaining = target_sum - np.sum(h)
        if remaining < -1e-12:
            idx2 = np.where(h > 0.0)[0]
            if len(idx2) > 0:
                h[idx2] += remaining / len(idx2)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lags, weights):
    """Compute gradient of weighted sum of correlations."""
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def smooth_lse_obj(x, alpha, target_sum, penalty):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    lse = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    sum_diff = np.sum(x) - target_sum
    return lse + penalty * sum_diff * sum_diff

def smooth_lse_grad(x, alpha, target_sum, penalty):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    grad = compute_grad_weighted(x, np.arange(2 * len(x) - 1), weights)
    sum_diff = np.sum(x) - target_sum
    grad += 2.0 * penalty * sum_diff * np.ones(len(x))
    return grad

def lbfgsb_refine(h_start, target_sum, alphas, penalty_schedule, maxiter, ftol):
    """Refine using L-BFGS-B with smooth LSE and penalty for sum constraint."""
    n = len(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    for i, alpha in enumerate(alphas):
        penalty = penalty_schedule[i] if i < len(penalty_schedule) else penalty_schedule[-1]
        try:
            res = minimize(smooth_lse_obj, h_cur,
                          args=(alpha, target_sum, penalty),
                          jac=smooth_lse_grad,
                          method='L-BFGS-B',
                          bounds=bounds,
                          options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

        h += amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
