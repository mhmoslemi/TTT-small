For a binary sequence h of length n with sum = n/2, the correlation at lag s is:
C(s) = sum_{i: 0≤i<n, 0≤i+s<n} h[i] * (1-h[i+s])

Let Z_s = {i ∈ A_s : h[i]=0, h[i+s]=0} (both 0)
Let C_s = {i ∈ A_s : h[i]=1, h[i+s]=0} (1→0)

Also |A_s| = n - |s| = P_s + Z_s + C_s + D_s.

We want to minimize max_s C_s (times dx = 2/n).

Note that C_s - D_s = |{i ∈ A_s : h[i]=1}| - |{i ∈ A_s : h[i+s]=1}|

For s > 0: |{i ∈ A_s : h[i]=1}| = number of 1s in positions 0 to n-1-s
For s > 0: |{i ∈ A_s : h[i+s]=1}| = number of 1s in positions s to n-1

So C_s - D_s = (total_1s - number of 1s in positions n-s to n-1) - (total_1s - number of 1s in positions 0 to s-1)

Let me think about this differently. For position i (changing from 1 to 0):

- As shifted element: affects lag s if 0 <= i-s < n, i.e., s in [i-(n-1), i]. Delta = h[i-s].

For position j (changing from 0 to 1):
- As base element: affects lag s if 0 <= j+s < n, i.e., s in [-j, n-1-j]. Delta = 1 - h[j+s].
- As shifted element: affects lag s if 0 <= j-s < n, i.e., s in [j-(n-1), j]. Delta = -h[j-s].

Actually, I realize there might be a simpler way to think about this. Let me define:
- For the swap i→0, j→1, the change in corr at lag s is:
  -corr change = -(contribution of i being 1) + (contribution of j being 1)
  
Where "contribution of position p being 1" at lag s is:
  - If p is the base: (1 - h[p+s]) if 0<=p+s<n, else 0
  - If p is the shifted: h[p-s] if 0<=p-s<n, else 0

So the total "value" of position p being 1 at lag s is:
  V(p, s) = [0<=p+s<n] * (1-h[p+s]) + [0<=p-s<n] * h[p-s]

And the delta at lag s from the swap is: -V(i, s) + V(j, s)

This is cleaner. For binary h, V(p, s) = [0<=p+s<n]*(1-h[p+s]) + [0<=p-s<n]*h[p-s].

s_lo = min(-i, i-n+1, -j, j-n+1)

                current_max = best_new_max
                improved = True
        if not improved:
            break
    return h, current_max

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Quick continuous optimization from initial
    elapsed = time.time() - start_time
    if budget_s - elapsed > 120:
        for eps in [0.01, 0.003, 0.001]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 60:
                break
            h_soft, c5 = slsqp_optimize(best_h, eps, 300)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_soft.copy()

    # Phase 2: Generate multiple diverse starting points and run SA
    elapsed = time.time() - start_time
    n_sa_budget = budget_s * 0.6
    if budget_s - elapsed > n_sa_budget + 100:
        # Generate diverse starts
        starts = []
        starts.append(to_binary(best_h))
        # Thue-Morse
        tm = thue_morse_sequence(n0).astype(float)
        tm_bin = to_binary(tm)
        starts.append(tm_bin)
        # Random
        for _ in range(4):
            h_rand = (rng.random(n0) < 0.5).astype(float)
            h_rand = project_onto_constraint(h_rand, target_sum)
            starts.append(to_binary(h_rand))
        # Perturbed best
        for _ in range(3):
            h_pert = best_h + 0.1 * rng.standard_normal(n0)
            h_pert = project_onto_constraint(h_pert, target_sum)
            starts.append(to_binary(h_pert))
        # Quasi-random (Weyl sequence based)
        h_weyl = np.zeros(n0)
        golden = (1.0 + np.sqrt(5.0)) / 2.0
        for i in range(n0):
            h_weyl[i] = 1.0 if np.floor((i + 1) * golden) - np.floor(i * golden) < 0.5 else 0.0
        h_weyl = project_onto_constraint(h_weyl, target_sum)
        starts.append(to_binary(h_weyl))

        # Evaluate all starts
        start_scores = []
        for s in starts:
            c5_s = compute_c5(s)
            start_scores.append(c5_s)
        start_scores = np.array(start_scores)
        order = np.argsort(start_scores)

        # Run SA on the best starts
        elapsed = time.time() - start_time
        time_for_sa = budget_s * 0.55 - (elapsed - start_time)
        if time_for_sa > 100:
            n_rounds = min(len(order), max(3, int(time_for_sa / 60)))
            for round_idx in range(n_rounds):
                elapsed = time.time() - start_time
                if budget_s - elapsed < 150:
                    break
                si = order[round_idx % len(order)]
                h_sa_start = starts[si].copy()
                if round_idx > 0:
                    # Add perturbation for diversity
                    n_flip = max(2, n0 // 20)
                    flip_idx = rng.choice(n0, size=n_flip, replace=False)
                    h_sa_start[flip_idx] = 1.0 - h_sa_start[flip_idx]
                    # Fix sum
                    s_sum = h_sa_start.sum()
                    if s_sum > target_sum:
                        ones_here = np.where(h_sa_start == 1)[0]
                        h_sa_start[ones_here[:int(s_sum - target_sum)]] = 0
                    elif s_sum < target_sum:
                        zeros_here = np.where(h_sa_start == 0)[0]
                        h_sa_start[zeros_here[:int(target_sum - s_sum)]] = 1

                t_s = 0.4 + 0.2 * (1.0 - round_idx / max(n_rounds, 1))
                t_e = 0.003
                n_steps_this = min(int((budget_s * 0.55 - (elapsed - start_time)) * 3000), 1500000)
                n_steps_this = max(n_steps_this, 50000)

                if round_idx % 2 == 0:
                    h_res, c5_res = binary_sa_fast(h_sa_start, n_steps_this, t_s, t_e, rng)
                else:
                    h_res, c5_res = worst_lag_sa(h_sa_start, n_steps_this, t_s, t_e, rng)

                c5_bin = (2.0 / n0) * c5_res
                if c5_bin < best_c5:
                    best_c5 = c5_bin
                    best_h = h_res.copy()
                    # Quick greedy polish
                    h_g, c5_g = greedy_improve_binary(h_res, n_passes=3)
                    c5_g_scaled = (2.0 / n0) * c5_g
                    if c5_g_scaled < best_c5:
                        best_c5 = c5_g_scaled
                        best_h = h_g.copy()
                    # Quick continuous polish
                    elapsed = time.time() - start_time
                    if budget_s - elapsed > 80:
                        h_polish = best_h + 0.01 * rng.standard_normal(n0)
                        h_polish = project_onto_constraint(h_polish, target_sum)
                        h_polish, c5_pol = slsqp_optimize(h_polish, 0.0005, 200)
                        if c5_pol < best_c5:
                            best_c5 = c5_pol
                            best_h = h_polish.copy()

    # Phase 3: Focused continuous refinement
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        h_final = best_h + 0.005 * rng.standard_normal(n0)
        h_final = project_onto_constraint(h_final, target_sum)
        for eps in [0.0003, 0.0001, 0.00003]:
            elapsed = time.time() - start_time
            if budget_s - elapsed < 20:
                break
            h_final, c5 = slsqp_optimize(h_final, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()

    # Phase 4: Final binary SA refinement from best continuous solution
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_bin_final = to_binary(best_h)
        n_steps_final = min(int((budget_s - elapsed) * 2000), 1000000)
        n_steps_final = max(n_steps_final, 30000)
        h_final_sa, c5_final_sa = binary_sa_fast(h_bin_final, n_steps_final, 0.2, 0.001, rng)
        c5_final_sa_scaled = (2.0 / n0) * c5_final_sa
        if c5_final_sa_scaled < best_c5:
            best_c5 = c5_final_sa_scaled
            best_h = h_final_sa.copy()
        # Final greedy
        h_g2, c5_g2 = greedy_improve_binary(best_h, n_passes=5)
        c5_g2_scaled = (2.0 / n0) * c5_g2
        if c5_g2_scaled < best_c5:
            best_c5 = c5_g2_scaled
            best_h = h_g2.copy()

    # Final polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 30:
        h_result = best_h + 0.003 * rng.standard_normal(n0)
        h_result = project_onto_constraint(h_result, target_sum)
        h_result, c5_res = slsqp_optimize(h_result, 0.0001, 500)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_result.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
