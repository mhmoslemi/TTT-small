def block_descent(h, target, deadline, rng, block_sizes=[3,4,5,6,8], n_rounds=5):

    h_cur = h.copy()
    cur_score = compute_c5(h_cur)
    
    def block_obj(x, s, e):
        h_full = h_cur.copy()
        h_full[s:e] = np.clip(x, 0.0, 1.0)
        h_full = project_feasible(h_full, target)
        corr = np.correlate(h_full, 1.0 - h_full, mode='full')
        c_max = float(np.max(corr))
        # LSE with high alpha
        alpha = 500.0
        shifted = alpha * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha
    
    for bs in block_sizes:
        if time.time() > deadline:
            break
        n_blocks = n // bs
        for rd in range(n_rounds):
            if time.time() > deadline:
                break
            order = rng.permutation(n_blocks)
            for bi in order:
                if time.time() > deadline:
                    break
                s = bi * bs
                e = s + bs
                x0 = h_cur[s:e].copy()
                try:
                    res = minimize(block_obj, x0, args=(s, e), method='Nelder-Mead',
                                  options={'maxiter': 3000, 'xatol': 1e-10, 'fatol': 1e-14})
                    h_test = h_cur.copy()
                    h_test[s:e] = np.clip(res.x, 0.0, 1.0)
                    h_test = project_feasible(h_test, target)
                    sc = compute_c5(h_test)
                    if sc < cur_score:
                        cur_score = sc
                        h_cur = h_test
                except:
                    pass
    
    return h_cur, cur_score

    dx = 2.0 / n
