def solve_lp_for_h(g, n):
    """Solve LP: min t s.t. A_s @ h <= t for all s, sum(h)=n/2, 0<=h<=1"""
    n_lags = 2 * n - 1
    A_ub = np.zeros((n_lags, n + 1))

        if shift >= 0:
            m = n - shift
            A_ub[s_idx, :m] = g[shift:shift+m]
        else:
            r = -shift
            m = n - r
            A_ub[s_idx, r:r+m] = g[:m]
        A_ub[s_idx, n] = -1.0

    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))

    b_eq = np.array([n / 2.0])

                        A_eq=A_eq, b_eq=b_eq,

        return time.time() - start_time

    def rem():
        return budget_s - elapsed()

    # Upsample to higher resolution for more degrees of freedom
    n_up = 240
    old_t = np.linspace(0.0, 1.0, n0)
    new_t = np.linspace(0.0, 1.0, n_up)
    h_up = np.interp(new_t, old_t, best_h)
    h_up = project_onto_constraint(h_up, n_up / 2.0)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
    n_w = len(best_h)

    # Phase 1: Coordinate descent (fast, exact per coordinate)
    if rem() > 300:
        h_cd, c5_cd = coord_descent(best_h, 30, min(rem() - 250, 120), start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 2: LP alternating
    if rem() > 250:
        h_lp, c5_lp = lp_alternating(best_h, 50, min(rem() - 200, 100), start_time)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()

    # Phase 3: Coordinate descent again after LP
    if rem() > 200:
        h_cd, c5_cd = coord_descent(best_h, 50, min(rem() - 150, 120), start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: SLSQP with medium smooth max
    if rem() > 150:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Projected gradient with top-k
    if rem() > 120:
        tl = min(rem() - 80, 80)
        h_pg, c5_pg = proj_grad_topk(best_h, 3, 0.01, 0.06, 300000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 6: Coordinate descent polish
    if rem() > 100:
        h_cd, c5_cd = coord_descent(best_h, 80, min(rem() - 60, 80), start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Fine SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Multi-start loop
    n_restart = 0
    eps_cycle = [0.003, 0.001, 0.0005]
    while rem() > 120 and n_restart < 20:
        strategy = n_restart % 5
        h_pert = gen_perturb(best_h, rng, n_w, n_w / 2.0, strategy)
        h_cd2, c5_cd2 = coord_descent(h_pert, 30, 30, start_time)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = h_cd2.copy()
            n_restart += 1
            continue
        eps_use = eps_cycle[min(n_restart // 7, len(eps_cycle) - 1)]
        if rem() > 80:
            h_opt2, c5_opt2 = slsqp_optimize(h_cd2, eps_use, 150)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
        if rem() > 50:
            h_cd3, c5_cd3 = coord_descent(best_h, 40, 30, start_time)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
        n_restart += 1

    # Phase 9: Very fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 10: Final coordinate descent
    if rem() > 40:
        h_cd, c5_cd = coord_descent(best_h, 100, min(rem() - 20, 40), start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 11: Final projected gradient
    if rem() > 30:
        tl = min(rem() - 15, 30)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.02, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 12: Resolution increase
    if rem() > 200:
        n2 = n_w + 60
        old_t2 = np.linspace(0.0, 1.0, n_w)
        new_t2 = np.linspace(0.0, 1.0, n2)
        h_v2 = np.interp(new_t2, old_t2, best_h)
        h_v2 = project_onto_constraint(h_v2, n2 / 2.0)
        h_cd_v, c5_cd_v = coord_descent(h_v2, 40, min(rem() - 100, 60), start_time)
        if rem() > 80:
            h_opt_v, c5_opt_v = slsqp_optimize(h_cd_v, 0.003, 300)
            if c5_opt_v < c5_cd_v:
                c5_cd_v = c5_opt_v
                h_cd_v = h_opt_v
        if rem() > 50:
            h_cd_v2, c5_cd_v2 = coord_descent(h_cd_v, 60, min(rem() - 30, 40), start_time)
            if c5_cd_v2 < c5_cd_v:
                c5_cd_v = c5_cd_v2
                h_cd_v = h_cd_v2
        if c5_cd_v < best_c5:
            best_c5 = c5_cd_v
            best_h = h_cd_v.copy()

    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
