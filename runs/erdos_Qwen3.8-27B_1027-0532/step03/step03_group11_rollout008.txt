def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0

        if s > target_sum:

        else:

def make_binary(h, n):
    order = np.argsort(np.abs(h - 0.5))
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def swap_delta_c5(h, oi, zi, s_max, n):
    omh = 1.0 - h
    old = 0.0
    nc = 0.0
    for (a, b) in [(oi, oi + s_max), (zi, zi + s_max)]:
        if 0 <= a < n and 0 <= b < n:
            old += h[a] * omh[b]
    for (a, b) in [(oi, oi - s_max), (zi, zi - s_max)]:
        if 0 <= a < n and 0 <= b < n:
            old += h[a] * omh[b]
    h_new_oi = 1.0 - h[oi]
    h_new_zi = 1.0 - h[zi]
    for (a, b, va, vb) in [(oi, oi + s_max, h_new_oi, h[oi + s_max] if 0 <= oi + s_max < n else 0),
                            (zi, zi + s_max, h_new_zi, h[zi + s_max] if 0 <= zi + s_max < n else 0),
                            (oi, oi - s_max, h_new_oi, h[oi - s_max] if 0 <= oi - s_max < n else 0),
                            (zi, zi - s_max, h_new_zi, h[zi - s_max] if 0 <= zi - s_max < n else 0)]:
        if 0 <= a < n and 0 <= b < n:
            nc += va * (1.0 - vb)
    return nc - old

def binary_local_search(h_bin, time_limit, start_time, max_iters=50000):
    n = len(h_bin)
    h = h_bin.copy()
    dx = 2.0 / n
    corr = compute_corr(h)
    max_idx = np.argmax(corr)
    current = corr[max_idx] * dx
    best_h = h.copy()
    best_c = current
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    s_max = max_idx - (n - 1)

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        sample_size = min(len(ones), 20)
        if sample_size == 0:
            break
        samp_o = rng_local.choice(ones, size=sample_size, replace=False) if sample_size < len(ones) else np.array(ones)
        samp_z = rng_local.choice(zeros, size=sample_size, replace=False) if sample_size < len(zeros) else np.array(zeros)
        for oi in samp_o:
            for zi in samp_z:
                if oi == zi:
                    continue
                delta = swap_delta_c5(h, oi, zi, s_max, n)
                if delta < best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        ones.remove(best_oi)
        zeros.remove(best_zi)
        corr = compute_corr(h)
        max_idx = np.argmax(corr)
        current = corr[max_idx] * dx
        s_max = max_idx - (n - 1)
        if current < best_c:
            best_c = current
            best_h = h.copy()
    return best_h, best_c

def binary_sa(h_init, n_pop, n_steps, time_limit, start_time, rng):
    n = len(h_init)
    pop = np.zeros((n_pop, n))
    scores = np.zeros(n_pop)
    pop[0] = h_init.astype(float)
    scores[0] = compute_c5(pop[0])
    for p in range(1, n_pop):
        if p == 1:
            pop[p] = make_binary(h_init + 0.1 * rng.standard_normal(n), n)
        elif p == 2:
            h_p = h_init.copy()
            idx = rng.choice(n, size=max(3, n // 10), replace=False)
            h_p[idx] = 1.0 - h_p[idx]
            pop[p] = make_binary(h_p, n)
        else:
            pop[p] = random_binary(n, rng)
        scores[p] = compute_c5(pop[p])
    best_h = pop[np.argmin(scores)].copy()
    best_c = float(scores.min())
    t0 = 0.005
    t1 = 0.00001
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t = t0 * (t1 / t0) ** (step / max(n_steps - 1, 1))
        p = step % n_pop
        r = rng.random()
        ones = np.where(pop[p] == 1.0)[0]
        zeros = np.where(pop[p] == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            continue
        if r < 0.6:
            oi = ones[rng.integers(len(ones))]
            zi = zeros[rng.integers(len(zeros))]
            pop[p][oi] = 0.0
            pop[p][zi] = 1.0
            new_s = compute_c5(pop[p])
            d = new_s - scores[p]
            if d < 0 or (t > 1e-9 and rng.random() < np.exp(-d / t)):
                scores[p] = new_s
                if new_s < best_c:
                    best_c = new_s
                    best_h = pop[p].copy()
            else:
                pop[p][oi] = 1.0
                pop[p][zi] = 0.0
        elif r < 0.85:
            blen = int(rng.integers(2, max(3, n // 8) + 1))
            a = int(rng.integers(0, n - blen))
            b = int(rng.integers(0, n - blen))
            if abs(a - b) < blen:
                continue
            ba = pop[p][a:a+blen].copy()
            bb = pop[p][b:b+blen].copy()
            pop[p][a:a+blen] = bb
            pop[p][b:b+blen] = ba
            new_s = compute_c5(pop[p])
            d = new_s - scores[p]
            if d < 0 or (t > 1e-9 and rng.random() < np.exp(-d / t)):
                scores[p] = new_s
                if new_s < best_c:
                    best_c = new_s
                    best_h = pop[p].copy()
            else:
                pop[p][a:a+blen] = ba
                pop[p][b:b+blen] = bb
        else:
            nf = int(rng.integers(1, 4))
            o_pool = list(ones)
            z_pool = list(zeros)
            if len(o_pool) < nf or len(z_pool) < nf:
                continue
            co = list(rng.choice(o_pool, size=nf, replace=False))
            cz = list(rng.choice(z_pool, size=nf, replace=False))
            for i in range(nf):
                pop[p][co[i]] = 0.0
                pop[p][cz[i]] = 1.0
            new_s = compute_c5(pop[p])
            d = new_s - scores[p]
            if d < 0 or (t > 1e-9 and rng.random() < np.exp(-d / t)):
                scores[p] = new_s
                if new_s < best_c:
                    best_c = new_s
                    best_h = pop[p].copy()
            else:
                for i in range(nf):
                    pop[p][co[i]] = 1.0
                    pop[p][cz[i]] = 0.0
    return best_h, best_c

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n_new):
        h_new[i] = h[i // factor]
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    global rng_local
    start_time = time.time()
    rng = np.random.default_rng(seed)
    rng_local = np.random.default_rng(seed + 1)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)
        h0 = project_onto_constraint(h0, n0 // 2)

    n0 = len(h0)
    target = n0 / 2.0
    h0 = project_onto_constraint(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        h_opt, c_opt = continuous_optimize(h0, 8000, budget_s - 250, start_time, K_base=4, mu=0.8)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
        h_opt2, c_opt2 = continuous_optimize(best_h, 5000, budget_s - 240, start_time, K_base=8, mu=0.7)
        if c_opt2 < best_c5:
            best_c5 = c_opt2
            best_h = h_opt2.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h_bin = make_binary(best_h, n0)
        h_bin, c_bin = binary_local_search(h_bin, budget_s - 200, start_time, max_iters=30000)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.astype(float).copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h_noise = best_h + 0.05 * rng.standard_normal(n0)
        h_noise = project_onto_constraint(h_noise, target)
        h_opt, c_opt = continuous_optimize(h_noise, 10000, budget_s - 200, start_time, K_base=6, mu=0.85)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        h_bin = make_binary(best_h, n0)
        h_bin, c_bin = binary_sa(h_bin, 8, 400000, budget_s - 300, start_time, rng)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.astype(float).copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        h_start = best_h + 0.03 * rng.standard_normal(n0)
        h_start = project_onto_constraint(h_start, target)
        h_opt, c_opt = continuous_optimize(h_start, 15000, budget_s - 250, start_time, K_base=10, mu=0.75)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    n_restarts = 0
    remaining = budget_s - (time.time() - start_time)
    while remaining > 150 and n_restarts < 15:
        strat = n_restarts % 5
        if strat == 0:
            h_pert = best_h + 0.04 * rng.standard_normal(n0)
        elif strat == 1:
            h_pert = best_h + 0.15 * rng.standard_normal(n0)
        elif strat == 2:
            h_bin_r = random_binary(n0, rng).astype(float)
            h_pert = 0.5 * best_h + 0.5 * h_bin_r
        elif strat == 3:
            h_pert = best_h.copy()
            idx = rng.choice(n0, size=max(3, n0 // 15), replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        else:
            h_pert = make_binary(best_h + 0.2 * rng.standard_normal(n0), n0).astype(float)
        h_pert = project_onto_constraint(h_pert, target)
        tl = min(remaining - 100, 120)
        if tl > 60:
            h_opt, c_opt = continuous_optimize(h_pert, 5000, start_time + (time.time() - start_time) + tl, start_time, K_base=5, mu=0.8)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
        n_restarts += 1
        remaining = budget_s - (time.time() - start_time)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n1 = 2 * n0
        h_up = upsample(best_h, 2)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.02 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_opt, c_opt = continuous_optimize(h_up, 8000, budget_s - 100, start_time, K_base=6, mu=0.8)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
            h_opt2, c_opt2 = continuous_optimize(best_h, 5000, budget_s - 50, start_time, K_base=10, mu=0.75)
            if c_opt2 < best_c5:
                best_c5 = c_opt2
                best_h = h_opt2.copy()

    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        h_final = continuous_optimize(best_h, 10000, budget_s - 10, start_time, K_base=15, mu=0.7)[0]
        h_final = project_onto_constraint(h_final, target_cur)
        c_final = compute_c5(h_final)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
