def subgradient_polish(h_init, n_steps, time_limit, start_time):
    """Polish using subgradient method with decreasing step size."""
    n = len(h_init)
    h = h_init.copy()
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = get_correlations(h)
        cmax = np.max(corr)
        
        if cmax < best_c5:
            best_c5 = cmax
            best_h = h.copy()
        
        # Find active constraints (near max)
        threshold = cmax * 0.999
        active_mask = corr >= threshold
        if not np.any(active_mask):
            active_mask = np.zeros_like(corr, dtype=bool)
            active_mask[np.argmax(corr)] = True
        
        # Average the subgradients of active constraints
        # Gradient of corr[k] w.r.t. h
        grad = np.zeros(n)
        dx = 2.0 / n
        omh = 1.0 - h
        k_indices = np.where(active_mask)[0]
        for k in k_indices:

                m = n - s
                grad[s:] += dx * omh[:m]
                grad[:m] -= dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += dx * omh[r:]
                grad[r:] -= dx * h[:m]

        h = project_onto_constraint(h, target_sum)
    
    return h, compute_c5(h)

    eps_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for eps in eps_schedule:
        if remaining() < 60:
            break
        maxit = 300 if eps <= 0.002 else 200
        h_opt, c5_opt = slsqp_optimize(best_h, eps, maxit)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Stage 2: Top-M gradient descent
    if remaining() > 120:
        h_opt, c5_opt = topm_descent(best_h, 8000, start_time + budget_s - 200, start_time, M=5)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Stage 3: Subgradient polish
    if remaining() > 100:
        h_opt, c5_opt = subgradient_polish(best_h, 10000, start_time + budget_s - 180, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Stage 4: Very fine SLSQP
    for eps in [0.0002, 0.0001, 0.00005]:
        if remaining() < 50:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, eps, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Stage 5: Top-M with larger M
    if remaining() > 100:
        h_opt, c5_opt = topm_descent(best_h, 6000, start_time + budget_s - 180, start_time, M=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Stage 6: Binary SA
    if remaining() > 300:
        sa_time = min(remaining() - 80, 400)
        bin_h, bin_c5 = binary_sa(best_h, 6, 400000, start_time + sa_time, start_time, rng)
        hr = bin_h.astype(float) + 0.008 * rng.standard_normal(n0)
        hr = project_onto_constraint(hr, target_sum)
        if remaining() > 60:
            h_opt, c5_opt = slsqp_optimize(hr, 0.005, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Stage 7: Upscale to larger n
    if remaining() > 250:
        n1 = 240
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[min(int(i * n0 / n1), n0 - 1)]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if remaining() < 40:
                break
            h_opt_up, c5_opt_up = slsqp_optimize(h_up, eps, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()

    # Stage 8: Final polish
    if remaining() > 30:
        n_cur = len(best_h)
        for eps in [0.0001, 0.00005, 0.00002]:
            if remaining() < 15:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, eps, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    if remaining() > 20:
        h_opt, c5_opt = subgradient_polish(best_h, 5000, start_time + budget_s - 5, start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h.copy()
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
