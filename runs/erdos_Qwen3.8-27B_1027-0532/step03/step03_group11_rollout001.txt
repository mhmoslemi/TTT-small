import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective_and_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    f = cmax + eps * np.log(sexp)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return f, grad

def adam_projected_optimize(h0, target_sum, lr, max_iter, eps_start, eps_end, beta1=0.85, beta2=0.95, time_limit=300, start_time=None):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_f = compute_c5(h)
    best_h = h.copy()
    best_iter = 0
    if start_time is None:
        start_time = time.time()
    for t in range(1, max_iter + 1):
        if time.time() - start_time > time_limit:
            break
        frac = (t - 1) / max(max_iter - 1, 1)
        eps = eps_start * (eps_end / eps_start) ** frac
        f, grad = smooth_objective_and_grad(h, eps)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad

    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    if start_time is None:
        start_time = time.time()
    remaining = time_limit - (time.time() - start_time)
    if remaining < 10:
        return h0, compute_c5(h0)
    maxiter_eff = min(maxiter, int(remaining / 1.5) + 10)
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target_sum),
             'jac': lambda x: np.ones(n)}]
    try:

                          jac=None, method='SLSQP',

                                               eps_end=0.0003, time_limit=t6,
                                               start_time=start_time)
        if c5_up < best_c5:
            # Also try SLSQP on higher resolution
            if time.time() - start_time < budget_s - 20:
                h_up2, c5_up2 = slsqp_optimize(h_up, 0.001, 100, time_limit=40,
                                               start_time=start_time)
                if c5_up2 < c5_up:
                    h_up = h_up2
                    c5_up = c5_up2
            best_c5 = c5_up
            best_h = h_up.copy()
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    # Phase 7: Final restart attempts
    n_cur = len(best_h)
    restart_count = 0
    while time.time() - start_time < budget_s - 20 and restart_count < 10:
        strategy = restart_count % 4
        if strategy == 0:
            h_pert = best_h + 0.03 * rng.standard_normal(n_cur)
        elif strategy == 1:
            h_pert = h_bin[:n_cur].astype(float) if n_cur == n0 else best_h + 0.05 * rng.standard_normal(n_cur)
        elif strategy == 2:
            h_pert = random_binary(n_cur, rng).astype(float) + 0.1 * rng.standard_normal(n_cur)
        else:
            alpha = 0.3
            h_rand = random_binary(n_cur, rng).astype(float)
            h_pert = alpha * best_h + (1 - alpha) * h_rand
        h_pert = project_onto_constraint(h_pert, n_cur / 2.0)
        t7 = min(40, budget_s - 20 - (time.time() - start_time))
        if t7 < 20:
            break
        h_opt, c5_opt = adam_projected_optimize(h_pert, n_cur / 2.0, lr=0.03,
                                                 max_iter=3000, eps_start=0.01,
                                                 eps_end=0.0005, time_limit=t7,
                                                 start_time=start_time)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        restart_count += 1
    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    if time.time() - start_time > budget_s - 2:
        pass
    return final_h, final_c5, len(final_h)
