        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_refine(h, eps, maxiter):
    """SLSQP refinement for continuous h."""
    n = len(h)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    def cjac(x):
        return np.ones(n)

                          jac=smooth_max_jac, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:

        else:

def binarize(h, target_ones):

    cur = int(h_b.sum())
    if cur > target_ones:
        idx = np.where(h_b == 1.0)[0]
        order = np.argsort(h[idx])[:cur - target_ones]
        h_b[idx[order]] = 0.0
