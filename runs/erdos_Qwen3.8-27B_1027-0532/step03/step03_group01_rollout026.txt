But these are for complex-valued or specific length sequences. For our problem with n=600 (not prime, not of the right form), we need a more general approach.

Let me think about what the optimal C₅ might be. If we could achieve perfectly uniform autocorrelation (A[k] = c for all k≠0), then:
- sum_{k=1}^{n-1} A[k] = n(n-2)/8 ≈ n²/8
- (n-1) values, so c = n(n-2)/(8(n-1)) ≈ n/8
- At lag k: overlap_k ≈ (n-k)/2 - (n-k)*c/n = (n-k)(1/2 - c/n) ≈ (n-k)(1/2 - 1/8) = (n-k)*3/8
- Maximum at k=1: ≈ (n-1)*3/8
- C₅ ≈ (n-1)*3/8 * 2/n ≈ 3/4 = 0.75

That's worse than 0.5! So even perfect uniform autocorrelation gives C₅ ≈ 0.75?

That can't be right. Let me recheck.

Actually, I think the issue is that the overlap at lag k is NOT simply (n-k)/2 - A[k]. Let me be more careful.

overlap_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])

Let S = {i : h[i] = 1}. Then:
overlap_k = |{i ∈ S : i ≤ n-1-k and h[i+k] = 0}|
= |{i ∈ S : i ≤ n-1-k}| - |{i ∈ S : i ≤ n-1-k and h[i+k] = 1}|
= |S ∩ [0, n-1-k]| - A[k]

If S is uniformly distributed, |S ∩ [0, n-1-k]| ≈ (n-k)/2.
And A[k] ≈ (n-k)/4 for a "random-like" sequence.

So overlap_k ≈ (n-k)/2 - (n-k)/4 = (n-k)/4.

C₅ = max_k overlap_k * (2/n) = max_k (n-k)/4 * 2/n = (n-1)/4 * 2/n ≈ 1/2.

OK so for a random-like sequence, C₅ ≈ 0.5. To get 0.38, we need to reduce the max overlap from ~n/4 to ~0.19n. That's a reduction by factor 0.19/(1/4) = 0.76.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    N = 2 * n
    corr = np.fft.ifft(np.fft.fft(h, N) * np.conj(np.fft.fft(b, N))).real
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        ones = np.where(h_r == 1)[0]
        # Flip the ones with smallest h value
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    # Final adjustment
    final_sum = h_r.sum()
    if final_sum > target_sum:
        extra = int(final_sum - target_sum)
        ones = np.where(h_r == 1)[0]
        h_r[ones[:extra]] = 0
    elif final_sum < target_sum:
        extra = int(target_sum - final_sum)
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[:extra]] = 1
    return h_r.astype(float)

def make_chirp(n, seed=0):
    rng = np.random.RandomState(seed)
    # Quadratic chirp: h[i] = 1 if (i^2 * a + b) mod n < n/2
    a = rng.randint(2, n)
    b = rng.randint(0, n)
    vals = (np.arange(n)**2 * a + b) % n
    h = (vals < n // 2).astype(float)
    # Adjust sum
    return project_binary(h, n // 2)

def make_random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:n//2]] = 1.0
    return h

def make_structured(n, seed=0):
    """Create structured initial guesses"""
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = 1.0
    
    # Try various structured patterns
    # 1. Chirp
    for _ in range(5):
        h = make_chirp(n, rng.randint(10000))
        c = compute_c5_fft(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # 2. Interleaved blocks
    for block_size in [7, 11, 13, 17, 19, 23]:
        pattern = np.zeros(block_size)
        pattern[:block_size//2] = 1
        # Randomize the pattern
        perm = rng.permutation(block_size)
        pattern2 = np.zeros(block_size)
        pattern2[perm[:block_size//2]] = 1
        h = np.tile(pattern2, n // block_size + 1)[:n]
        h = project_binary(h, n // 2)
        c = compute_c5_fft(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # 3. Random binary
    for _ in range(10):
        h = make_random_binary(n, rng.randint(10000))
        c = compute_c5_fft(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c

def compute_corr_array(h):
    """Return the full correlation array (not just the max)"""
    n = len(h)
    b = 1.0 - h
    N = 2 * n
    corr = np.fft.ifft(np.fft.fft(h, N) * np.conj(np.fft.fft(b, N))).real
    return corr

def sa_with_worst_lag_focus(h_start, target_sum, max_iters, t_init, t_final, rng,
                            time_limit=None, start_time=None, n_worst=5):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    corr = compute_corr_array(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = float(np.max(corr)) * dx
    current_c5 = best_c5
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    no_improve = 0
    max_no_improve = max(3000, max_iters // 10)
    
    # Precompute: for each position i, what lags does it affect?
    # Position i appears in corr at lag k when i+k-(n-1) is valid for the (1-h) side
    # and when i-(n-1)-k+... hmm this is complex.
    # Let's just use the full recomputation for simplicity but with FFT.
    
    for i in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
            else:
                break
        
        ratio = i / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        h_new = h.copy()
        moved = False
        
        # Choose move type
        move_type = rng.randint(5)
        
        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            # Simple swap: choose i from ones, j from zeros
            # Bias toward positions that are in "worst lag" regions
            i1 = ones[rng.randint(len(ones))]
            j0 = zeros[rng.randint(len(zeros))]
            h_new[i1] = 0
            h_new[j0] = 1
            moved = True
        elif move_type == 1 and len(ones) >= 2 and len(zeros) >= 2:
            # Double swap
            idx1 = rng.choice(len(ones), 2, replace=False)
            idx0 = rng.choice(len(zeros), 2, replace=False)
            h_new[ones[idx1[0]]] = 0
            h_new[ones[idx1[1]]] = 0
            h_new[zeros[idx0[0]]] = 1
            h_new[zeros[idx0[1]]] = 1
            moved = True
        elif move_type == 2 and len(ones) > 0 and len(zeros) > 0:
            # Targeted: choose i1 from positions that contribute to worst lags
            # Get worst lag positions
            worst_lags = np.argsort(corr)[::-1][:n_worst]
            # For each worst lag, find the positions contributing to it
            # And pick a 1-position to move
            i1 = ones[rng.randint(len(ones))]
            # Choose j0 near the "effective position" for the worst lag
            # This is a heuristic: prefer j0 that is at a distance from i1 equal to a worst lag
            target_dist = worst_lags[rng.randint(len(worst_lags))]
            # Convert lag to distance
            dist = int(abs(target_dist - (n-1)))  # relative to center
            candidates = zeros
            # Weight candidates by distance
            dists = np.abs(candidates - i1).astype(float)
            # Prefer candidates at the target distance
            weight = np.exp(-np.abs(dists - dist) / max(dist, 1))
            weights = weight / weight.sum()
            j0 = candidates[rng.choice(len(candidates), p=weights)]
            h_new[i1] = 0
            h_new[j0] = 1
            moved = True
        elif move_type == 3 and len(ones) >= 3 and len(zeros) >= 3:
            # Triple swap
            idx1 = rng.choice(len(ones), 3, replace=False)
            idx0 = rng.choice(len(zeros), 3, replace=False)
            for idx in idx1:
                h_new[ones[idx]] = 0
            for idx in idx0:
                h_new[zeros[idx]] = 1
            moved = True
        elif move_type == 4 and len(ones) > 0 and len(zeros) > 0:
            # Block move: move a contiguous block of 1s to a contiguous block of 0s
            block_len = rng.randint(2, min(6, len(ones)//4 + 1))
            # Find a run of 1s of at least block_len
            starts = np.where(np.diff(np.concatenate([[0], ones, [n]])) > block_len)[0]
            if len(starts) == 0:
                i1 = ones[rng.randint(len(ones))]
                j0 = zeros[rng.randint(len(zeros))]
                h_new[i1] = 0
                h_new[j0] = 1
            else:
                start_idx = starts[rng.randint(len(starts))]
                h_new[ones[start_idx]:ones[start_idx]+block_len] = 0
                # Find a run of 0s
                end_idx = np.where(np.diff(np.concatenate([[0], zeros, [n]])) >= block_len)[0]
                if len(end_idx) > 0:
                    end_s = end_idx[rng.randint(len(end_idx))]
                    h_new[zeros[end_s]:zeros[end_s]+block_len] = 1
                else:
                    # Just flip individual
                    h_new[ones[start_idx]:ones[start_idx]+block_len] = 0
                    for idx in rng.choice(len(zeros), block_len, replace=False):
                        h_new[zeros[idx]] = 1
            moved = True
        
        if not moved:
            continue
        
        # Evaluate
        c5_new = compute_c5_fft(h_new)
        delta = c5_new - current_c5
        
        if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
            h = h_new
            current_c5 = c5_new
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = compute_corr_array(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_c5

def fast_hill_climb(h, target_sum, max_iter, rng, n_attempts_per_iter=20):
    """Fast hill climbing using FFT evaluation"""
    n = len(h)
    h = project_binary(h, target_sum)
    best_c5 = compute_c5_fft(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for _ in range(max_iter):
        improved = False
        for _ in range(n_attempts_per_iter):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = ones[rng.randint(len(ones))]
            j0 = zeros[rng.randint(len(zeros))]
            h[i1] = 0
            h[j0] = 1
            c5_new = compute_c5_fft(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
            else:
                h[i1] = 1
                h[j0] = 0
        if not improved:
            break
    
    return h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def smooth_max_obj(h, dx, alpha):
    """Smooth approximation of max correlation"""
    corr = compute_corr_array(h)
    c = corr * dx
    # Log-sum-exp approximation of max
    shifted = alpha * (c - np.max(c))
    shifted = np.clip(shifted, -500, 0)
    return float(np.max(c) + np.log(np.sum(np.exp(shifted))) / alpha)

def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit=None, start_time=None):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    # Use log-sum-exp smooth max as objective
    # Constraint: sum(h) = target
    # Box: 0 <= h <= 1
    
    h_cur = project_feasible(h_start.copy(), target)
    curr_score = compute_c5_fft(h_cur)
    
    for alpha in alphas:
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        try:
            def obj(h):
                return smooth_max_obj(h, dx, alpha)
            
            const = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                     'jac': lambda h: np.ones(n)}]
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=const, options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5_fft(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score
