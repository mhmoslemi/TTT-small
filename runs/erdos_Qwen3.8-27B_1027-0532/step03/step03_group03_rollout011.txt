def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

    return np.sum(x) - _CONS_TARGET

    return np.ones(_CONS_N)

def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    if total < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)

def nesterov_subgradient(h_init, target_sum, max_iter, lr_init, topk_schedule,
                          alpha, noise_prob, noise_scale, patience, seed):
    # topk_schedule is a list of (start_iter, end_iter, topk) tuples
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    y = h.copy()  # Nesterov auxiliary variable
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    beta = 0.9  # momentum
    
    for it in range(max_iter):
        # Determine topk for this iteration
        topk = topk_schedule[0][2] if topk_schedule else 5
        for start, end, k in topk_schedule:
            if start <= it < end:
                topk = k
                break
        
        corr = overlap_corrs(y)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_weighted(y, top_indices, w)
        
        # Nesterov update
        v = beta * v + (1.0 - beta) * grad
        y_new = h - lr * v
        y_new = project_feasible(y_new, target_sum)
        
        if noise_prob > 0.0 and rng.random() < noise_prob:
            y_new = y_new + rng.randn(n) * noise_scale
            y_new = project_feasible(y_new, target_sum)
        
        # Accept/reject
        score = compute_c5(y_new)
        if score < best_score:
            best_score = score
            best_h = y_new.copy()
            stagnation = 0
            lr = min(lr * 1.005, 3.0)
        else:
            stagnation += 1
            lr *= 0.995
        h = y_new
        if stagnation > patience:
            break
    
    return best_h, best_score

def multi_stage_slsqp(h_start, target_sum, stages, maxiter_per_stage, ftol):

    stages: list of alpha values to use in sequence
    """
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)

    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N

            try:
                x0 = h_cur.copy()

                    x0 = project_feasible(x0, target_sum)

            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            h = h.copy()
            h[idx] += diff / len(idx)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            h = h.copy()
            h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    threshold = -15.0
    active_mask = shifted > threshold
    if not np.any(active_mask):
        active_mask = np.zeros(len(corr), dtype=bool)
        active_mask[np.argmax(corr)] = True
        active_mask[np.argmin(np.abs(np.arange(len(corr)) - (len(corr) - 1)))] = True
    exp_shifted = np.where(active_mask, np.exp(shifted), 0.0)
    total = np.sum(exp_shifted)
    if total < 1e-300:
        return np.zeros(len(x))
    weights = exp_shifted / total
    active_indices = np.where(active_mask)[0]
    active_weights = weights[active_indices]
    n = len(x)
    return compute_grad_weighted(x, active_indices, active_weights)


def nesterov_subgradient(h_init, target_sum, max_iter, lr_init, topk_schedule,
                          alpha, noise_prob, noise_scale, patience, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    beta = 0.92

    for it in range(max_iter):
        topk = 5
        for start, end, k in topk_schedule:
            if start <= it < end:
                topk = k
                break

        corr = overlap_corrs(h)
        n_corr = len(corr)
        if topk < n_corr:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        else:
            top_indices = np.arange(n_corr)
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)

        v = beta * v + (1.0 - beta) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)

        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)

        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.005, 3.0)
        else:
            stagnation += 1
            lr *= 0.995
        h = h_new
        if stagnation > patience:
            break

    return best_h, best_score


def multi_stage_slsqp(h_start, target_sum, stages, maxiter_per_stage, ftol, time_limit=None):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    start_t = time.time()

    for alpha in stages:
        if time_limit is not None and (time.time() - start_t) > time_limit:
            break
        for attempt in range(2):
            if time_limit is not None and (time.time() - start_t) > time_limit:
                break
            try:
                x0 = h_cur.copy()
                if attempt == 1:
                    pert_rng = np.random.RandomState(alpha * 13 + 7)
                    x0 = x0 + pert_rng.randn(_CONS_N) * 0.003
                    x0 = project_feasible(x0, target_sum)
                res = minimize(smooth_lse_obj, x0, args=(alpha,),
                               jac=smooth_lse_grad, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter_per_stage, 'ftol': ftol})
                h_new = project_feasible(res.x, target_sum)
                if compute_c5(h_new) < compute_c5(h_cur):
                    h_cur = h_new
            except Exception:
                pass
    return h_cur


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    topk_sched_1 = [(0, 1200, 25), (1200, 2800, 10), (2800, 4000, 5)]
    sg_configs_1 = [
        (0.14, 20, 0.05, 0.02, 300),
        (0.09, 40, 0.03, 0.01, 280),
        (0.11, 15, 0.07, 0.018, 250),
    ]

    for i, (lr, alpha, np_, ns, pat) in enumerate(sg_configs_1):
        if _remaining() < 50:
            break
        h_try = best_h + rng.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_new, score = nesterov_subgradient(h_try, target_init,
                                            max_iter=4000, lr_init=lr,
                                            topk_schedule=topk_sched_1,
                                            alpha=alpha, noise_prob=np_,
                                            noise_scale=ns, patience=pat,
                                            seed=seed + i * 100)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 100:
        h_ref = multi_stage_slsqp(best_h, target_init,
                                  stages=[40, 120, 300],
                                  maxiter_per_stage=1200, ftol=1e-14,
                                  time_limit=80)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    n_high = 1000
    target_high = n_high / 2.0
    h_high = interpolate_h(best_h, n_high, target_high)
    sc_high = compute_c5(h_high)
    if sc_high < best_score:
        best_score = sc_high
        best_h = h_high

    topk_sched_2 = [(0, 1500, 30), (1500, 3500, 12), (3500, 5000, 6)]
    sg_configs_2 = [
        (0.10, 25, 0.04, 0.015, 300),
        (0.06, 50, 0.02, 0.008, 280),
        (0.13, 12, 0.06, 0.02, 250),
    ]

    for i, (lr, alpha, np_, ns, pat) in enumerate(sg_configs_2):
        if _remaining() < 70:
            break
        h_try = h_high + rng.randn(n_high) * 0.008
        h_try = project_feasible(h_try, target_high)
        h_new, score = nesterov_subgradient(h_try, target_high,
                                            max_iter=5000, lr_init=lr,
                                            topk_schedule=topk_sched_2,
                                            alpha=alpha, noise_prob=np_,
                                            noise_scale=ns, patience=pat,
                                            seed=seed + 500 + i * 100)
        if score < best_score:
            best_score = score
            best_h = h_new
        else:
            pass
        if len(best_h) != n_high:
            h_high = interpolate_h(best_h, n_high, target_high)
        else:
            h_high = best_h

    if _remaining() > 200:
        h_work = h_high.copy()
        h_ref = multi_stage_slsqp(h_work, target_high,
                                  stages=[30, 80, 200, 500],
                                  maxiter_per_stage=1500, ftol=1e-14,
                                  time_limit=180)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 100:
        h_work = best_h.copy() if len(best_h) == n_high else interpolate_h(best_h, n_high, target_high)
        h_ref = multi_stage_slsqp(h_work, target_high,
                                  stages=[800, 1200],
                                  maxiter_per_stage=1200, ftol=1e-14,
                                  time_limit=90)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 300:
        n_vh = 1400
        target_vh = n_vh / 2.0
        h_vh = interpolate_h(best_h, n_vh, target_vh)
        sc = compute_c5(h_vh)
        if sc < best_score:
            best_score = sc
            best_h = h_vh

        if _remaining() > 150:

            h_try = h_vh + rng.randn(n_vh) * 0.005

            h_new, score = nesterov_subgradient(h_try, target_vh,

            if score < best_score:

        n_use = len(best_h)
        target_use = n_use / 2.0

        h_try = project_feasible(h_try, target_use)

            best_h = h_new

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
