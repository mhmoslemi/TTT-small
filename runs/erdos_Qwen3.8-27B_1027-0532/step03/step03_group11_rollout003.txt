```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def proj_gradient_optimize(h_init, eps, n_iters, time_limit, start_time):
    """Projected gradient with backtracking line search"""
    n = len(h_init)
    target = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target)
    best_h = h.copy()
    best_f = smooth_objective(h, eps)
    
    alpha = 1.0
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        g = smooth_gradient(h, eps)
        f_h = smooth_objective(h, eps)
        
        # Backtracking line search
        a = alpha
        improved = False
        for _ in range(30):
            h_try = project_onto_constraint(h - a * g, target)
            f_try = smooth_objective(h_try, eps)
            if f_try <= f_h + 0.3 * a * np.dot(g, h_try - h):
                h = h_try
                alpha = min(a * 1.5, 10.0)
                improved = True
                break
            a *= 0.5
        if not improved:
            # Try smaller alpha or break
            h_try = project_onto_constraint(h - a * g, target)
            h = h_try
        
        f_h = smooth_objective(h, eps)
        if f_h < best_f:
            best_f = f_h
            best_h = h.copy()
    
    return best_h, compute_c5(best_h)

def make_binary(h, rng=None):
    n = len(h)
    diffs = h - 0.5
    order = np.argsort(-diffs)
    result = np.zeros(n)
    result[order[:n // 2]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def binary_steepest(h, max_rounds=200, time_limit=60, start_time=0):
    """Steepest descent on binary solutions using all 1-swap moves"""
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                delta = sc - current
                if delta < best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                h[oi] = 1.0
                h[zi] = 0.0
        
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        current += best_delta
    
    return h, current
```

- So for each of the C(90,2) = 4005 pairs of 1s, try all C(90,2) = 4005 pairs of 0s
- Total: 4005 * 4005 = 16M evaluations at ~10μs each = 160 seconds per "round"
- That's too slow for a full round, but I can do partial rounds

Alternatively, I can use a greedy approach: for each oi in ones, find the best (zi1, zi2) pair to swap with another fixed oi2. This is O(|ones| * |zeros|² ) = 90 * 8100 = 729000 evaluations ≈ 7 seconds. More manageable.

But honestly, for the improvement we're looking for (0.0001), I think the binary steepest descent with 1-swaps is the most important. The 2-swaps and kicks are for escaping local optima, which might not be necessary if the 1-swap landscape is not too rugged.

Let me focus the code on:
1. Fast and correct continuous optimization
2. Thorough binary 1-swap steepest descent
3. A few random restarts
4. Final polishing

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def proj_grad_optimize(h_init, eps, n_iters, time_limit, start_time):
    n = len(h_init)
    target = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.5
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        g = smooth_gradient(h, eps)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-12:
            break
        f_h = smooth_objective(h, eps)
        a = alpha
        improved = False
        for _ in range(40):
            h_try = project_onto_constraint(h - a * g, target)
            f_try = smooth_objective(h_try, eps)
            if f_try <= f_h + 0.4 * a * np.dot(g, h_try - h):
                h = h_try
                alpha = min(a * 1.3, 5.0)
                improved = True
                break
            a *= 0.5
        if not improved:
            a = max(a, 1e-8)
            h = project_onto_constraint(h - a * g, target)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def make_binary(h, rng):
    n = len(h)
    diffs = h - 0.5
    order = np.argsort(-diffs)
    result = np.zeros(n)
    result[order[:n // 2]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def binary_steepest_1swap(h, max_rounds, time_limit, start_time):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        for oi in ones:
            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0
                sc = compute_c5(h)
                delta = sc - current
                if delta < best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                h[oi] = 1.0
                h[zi] = 0.0
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        current += best_delta
    return h, current

def binary_steepest_2swap(h, max_rounds, time_limit, start_time, rng):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_delta = 0.0
        best_move = None
        n_try = min(3000, len(ones) * (len(ones) - 1) // 2 * len(zeros) * (len(zeros) - 1) // 2)
        for _ in range(n_try):
            oi1 = ones[rng.integers(len(ones))]
            oi2 = ones[rng.integers(len(ones))]
            if oi1 == oi2:
                continue
            zi1 = zeros[rng.integers(len(zeros))]
            zi2 = zeros[rng.integers(len(zeros))]
            if zi1 == zi2:
                continue
            h[oi1] = 0.0
            h[oi2] = 0.0
            h[zi1] = 1.0
            h[zi2] = 1.0
            sc = compute_c5(h)
            delta = sc - current
            if delta < best_delta:
                best_delta = delta
                best_move = (oi1, oi2, zi1, zi2)
            h[oi1] = 1.0
            h[oi2] = 1.0
            h[zi1] = 0.0
            h[zi2] = 0.0
        if best_move is None:
            break
        oi1, oi2, zi1, zi2 = best_move
        h[oi1] = 0.0
        h[oi2] = 0.0
        h[zi1] = 1.0
        h[zi2] = 1.0
        current += best_delta
    return h, current

def iterated_binary_search(h_init, rng, time_limit, start_time, n_restarts=15):
    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    n_restarts_actual = 0
    for restart in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h_cur = best_h.copy()
        if restart > 0:
            if restart % 3 == 1:
                nflip = rng.integers(3, max(4, n // 15))
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]
                for _ in range(nflip):
                    oi = ones[rng.integers(len(ones))]
                    zi = zeros[rng.integers(len(zeros))]
                    h_cur[oi] = 0.0
                    h_cur[zi] = 1.0
                    ones = np.where(h_cur == 1.0)[0]
                    zeros = np.where(h_cur == 0.0)[0]
            elif restart % 3 == 2:
                nflip = rng.integers(2, max(3, n // 20))
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]
                for _ in range(nflip):
                    oi = ones[rng.integers(len(ones))]
                    zi = zeros[rng.integers(len(zeros))]
                    h_cur[oi] = 0.0
                    h_cur[zi] = 1.0
                    ones = np.where(h_cur == 1.0)[0]
                    zeros = np.where(h_cur == 0.0)[0]
        c5_cur = compute_c5(h_cur)
        hatch_time = min(120.0, max(10.0, (time_limit - (time.time() - start_time)) / max(n_restarts - restart, 1)))
        h_cur, c5_cur = binary_steepest_1swap(h_cur, 300, hatch_time, start_time)
        if c5_cur < best_c5 - 1e-10:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        remaining = time_limit - (time.time() - start_time)
        if remaining > 30 and c5_cur == best_c5:
            h_cur2, c5_cur2 = binary_steepest_2swap(h_cur, 50, min(30.0, remaining), start_time, rng)
            if c5_cur2 < best_c5 - 1e-10:
                best_c5 = c5_cur2
                best_h = h_cur2.copy()
                h_cur, c5_cur = binary_steepest_1swap(h_cur2, 200, min(60.0, time_limit - (time.time() - start_time)), start_time)
                if c5_cur < best_c5 - 1e-10:
                    best_c5 = c5_cur
                    best_h = h_cur.copy()
        n_restarts_actual += 1
    return best_h, best_c5

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n_new):
        h_new[i] = h[i // factor]
    return h_new
```
