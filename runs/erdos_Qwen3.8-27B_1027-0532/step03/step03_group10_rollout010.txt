import numpy as np
from scipy.optimize import minimize

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def evaluate_swap_effect(h, i, j, n):
    s_lo = max(-i, i - (n - 1), -j, j - (n - 1))
    s_hi = min((n - 1) - i, i, (n - 1) - j, j)
    if s_lo > s_hi:
        return np.array([], dtype=int), np.array([], dtype=float)
    s_range = np.arange(s_lo, s_hi + 1)
    k_range = s_range + (n - 1)
    h_i_s = np.zeros(s_hi - s_lo + 1)
    h_i_ms = np.zeros(s_hi - s_lo + 1)
    h_j_s = np.zeros(s_hi - s_lo + 1)
    h_j_ms = np.zeros(s_hi - s_lo + 1)
    mask_i_s = (i + s_range >= 0) & (i + s_range < n)
    mask_i_ms = (i - s_range >= 0) & (i - s_range < n)
    mask_j_s = (j + s_range >= 0) & (j + s_range < n)
    mask_j_ms = (j - s_range >= 0) & (j - s_range < n)
    h_i_s[mask_i_s] = h[i + s_range[mask_i_s]]
    h_i_ms[mask_i_ms] = h[i - s_range[mask_i_ms]]
    h_j_s[mask_j_s] = h[j + s_range[mask_j_s]]
    h_j_ms[mask_j_ms] = h[j - s_range[mask_j_ms]]
    delta = -(1.0 - h_i_s) + h_i_ms + (1.0 - h_j_s) - h_j_ms
    return k_range, delta

            zeros = [x for x in range(n0) if h_start[x] == 0.0]
            for _ in range(n_flip):
                i = ones[rng.integers(len(ones))]
                j = zeros[rng.integers(len(zeros))]
                h_start[i], h_start[j] = h_start[j], h_start[i]
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)

        tl = min(20, remaining * 0.08)
        mode = 'steepest' if n_iter % 3 != 0 else 'first'
        h_new, new_max = binary_local_search(h_start, tl, rng, mode)
        c5_new = (2.0 / n0) * new_max

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_polish = h_new + 0.008 * rng.standard_normal(n0)
            h_polish = project_onto_constraint(h_polish, target_sum)
            if time.time() - start_time < budget_s - 40:
                h_polish, c5p = slsqp_optimize(h_polish, 0.001, 200)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = h_polish.copy()

        remaining = budget_s - (time.time() - start_time) - 80

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_final = to_binary(best_h)
        h_final, fin_max = binary_local_search(h_final, min(30, budget_s - elapsed - 20), rng, 'steepest')
        c5_fin = (2.0 / n0) * fin_max
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_final.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 20:
        h_polish = best_h + 0.005 * rng.standard_normal(n0)
        h_polish = project_onto_constraint(h_polish, target_sum)
        for eps in [0.0005, 0.0001, 0.00003]:
            if time.time() - start_time > budget_s - 5:
                break
            h_polish, c5 = slsqp_optimize(h_polish, eps, 400)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_polish.copy()

    try:
        for n_try in [182, 184, 186]:
            if time.time() - start_time > budget_s - 60:
                break
            h_try = np.zeros(n_try)
            h_try[:n_try // 2] = 1.0
            h_try = np.roll(h_try, rng.integers(n_try))
            h_try, tmax = binary_local_search(h_try, 15, rng, 'steepest')
            c5_try = (2.0 / n_try) * tmax
            if c5_try < best_c5:
                h_polish = h_try + 0.01 * rng.standard_normal(n_try)
                h_polish = project_onto_constraint(h_polish, n_try / 2.0)
                h_polish, c5p = slsqp_optimize(h_polish, 0.001, 200)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = h_polish.copy()
            elif c5_try < best_c5 + 0.002:
                h_polish = h_try + 0.01 * rng.standard_normal(n_try)
                h_polish = project_onto_constraint(h_polish, n_try / 2.0)
                h_polish, c5p = slsqp_optimize(h_polish, 0.001, 200)
                if c5p < best_c5:
                    best_c5 = c5p
                    best_h = h_polish.copy()
    except Exception:
        pass

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
