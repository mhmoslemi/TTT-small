def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    if len(h) != n:
        x_new = np.linspace(0, 1, n)
        x_old = np.linspace(0, 1, len(h))
        h = np.interp(x_new, x_old, h)
    idx = np.argsort(h)[::-1][:n//2]
    out = np.zeros(n)
    out[idx] = 1.0
    return out

def make_thue_morse(n):
    # Thue-Morse sequence: t(k) = popcount(k) mod 2
    tm = np.zeros(n)
    for i in range(n):
        # Count bits
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = 1.0 if (c % 2 == 0) else 0.0
    # Ensure exactly n/2 ones (Thue-Morse has n/2 ones for n=2^k)
    n_ones = int(np.sum(tm))
    target_ones = n // 2
    if n_ones != target_ones:
        idx = np.argsort(tm)[::-1][:target_ones]
        tm = np.zeros(n)
        tm[idx] = 1.0
    return tm

def make_even_spaced(n, rng, jitter=0):
    # Place n/2 ones as evenly as possible
    h = np.zeros(n)
    n_ones = n // 2
    if jitter == 0:
        positions = np.linspace(0, n-1, n_ones, endpoint=False).astype(int)
    else:

        positions = np.sort((base + rng.randn(n_ones) * jitter).astype(int) % n)
        # Handle duplicates

        unique_pos = []

            if p not in seen:

                unique_pos.append(p)
        positions = unique_pos[:n_ones]

def compute_c5(h):

def corr_max(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
