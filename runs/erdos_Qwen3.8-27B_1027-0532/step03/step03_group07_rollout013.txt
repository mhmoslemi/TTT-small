import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_adaptive(h0, n, target_sum, n_iters, deadline, top_m=5, use_sg=True):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    alpha0 = 0.4
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        if use_sg:
            corr = np.correlate(y, 1.0 - y, mode='full')
            max_corr = np.max(corr)
            if max_corr < 1e-10:
                break
            weights = np.exp(80.0 * (corr - max_corr))
            total_w = weights.sum()
            if total_w < 1e-300:
                weights = np.zeros(len(corr))
                weights[np.argmax(corr)] = 1.0
            else:
                weights = weights / total_w
            top_m_actual = min(top_m, len(corr))
            top_m_idx = np.argsort(weights)[::-1][:top_m_actual]
            w_m = weights[top_m_idx].copy()
            if w_m.sum() > 1e-15:
                w_m = w_m / w_m.sum()
            g = np.zeros(n)
            for idx in range(len(top_m_idx)):
                g = g + w_m[idx] * _grad_corr_term(y, top_m_idx[idx], dx)
        else:
            corr = np.correlate(y, 1.0 - y, mode='full')
            k_star = int(np.argmax(corr))
            g = _grad_corr_term(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _pso_search(h0, n, target_sum, pop_size, n_iters, deadline, rng):
    dx = 2.0 / n
    particles = [h0.copy()]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        particles.append(h_rand)
    vs = [np.zeros(n) for _ in range(pop_size)]
    pbest = [p.copy() for p in particles]
    pbest_vals = [_c5_of(p) for p in particles]
    gbest_idx = int(np.argmin(pbest_vals))
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    w0 = 0.9
    w1 = 0.4
    c1 = 1.5
    c2 = 1.5
    for it in range(n_iters):
        if time.time() > deadline:
            break
        w = w0 + (w1 - w0) * it / max(n_iters - 1, 1)
        for i in range(pop_size):
            r1 = rng.rand()
            r2 = rng.rand()
            vs[i] = w * vs[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            new_p = particles[i] + 0.3 * vs[i]
            new_p = _project(new_p, target_sum)
            new_val = _c5_of(new_p)
            if new_val < pbest_vals[i]:
                pbest[i] = new_p.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = new_p.copy()
                    gbest_val = new_val
            particles[i] = new_p
            vs[i] = np.clip(vs[i], -0.5, 0.5)
    return gbest, gbest_val

def _block_cd(h0, n, target_sum, block_size, n_rounds, deadline, rng):
    dx = 2.0 / n
    h = h0.copy()
    best_val = _c5_of(h)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for block_start in range(0, n - block_size + 1, block_size):
            if time.time() > deadline:
                break
            block_end = min(block_start + block_size, n)
            block_size_actual = block_end - block_start
            if block_size_actual < 2:
                continue
            h_block = h[block_start:block_end].copy()
            h_fixed = h.copy()
            remaining_sum = target_sum - (h_fixed.sum() - h_fixed[block_start:block_end].sum())
            if remaining_sum < 0 or remaining_sum > block_size_actual:
                continue
            
            def block_obj(hb):
                h_trial = h_fixed.copy()
                h_trial[block_start:block_end] = hb
                return _c5_of(h_trial)
            
            try:
                bounds_b = [(0.0, 1.0)] * block_size_actual
                cons_b = [{'type': 'eq', 'fun': lambda hb: hb.sum() - remaining_sum,
                           'jac': lambda hb: np.ones(block_size_actual)}]
                res = minimize(block_obj, h_block, method='SLSQP',
                               bounds=bounds_b, constraints=cons_b,
                               options={'maxiter': 50, 'ftol': 1e-12})
                new_block = np.clip(res.x, 0.0, 1.0)
                s = new_block.sum()
                if abs(s - remaining_sum) > 1e-6:
                    new_block = h_block
                else:
                    h_fixed[block_start:block_end] = new_block
                    new_val = _c5_of(h_fixed)
                    if new_val < best_val - 1e-14:
                        h = h_fixed.copy()
                        best_val = new_val
                        improved = True
                    else:
                        h_fixed[block_start:block_end] = h_block
            except Exception:
                pass
        if not improved:
            break
    return h, best_val

def _pairwise_swap(h0, n, target_sum, n_rounds, deadline, rng, eps_list=None):
    if eps_list is None:
        eps_list = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    h = h0.copy()
    best_val = _c5_of(h)
    for r in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_swaps = min(n * 5, 500)
        for _ in range(n_swaps):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or h[i] <= h[j]:
                continue
            for eps in eps_list:
                d = min(eps, h[j] - h[i])
                if d < 1e-6:
                    continue
                h[i] += d
                h[j] -= d
                cv = _c5_of(h)
                if cv < best_val - 1e-14:
                    best_val = cv
                    improved = True
                    break
                else:
                    h[i] -= d
                    h[j] += d
        if not improved:
            break
    return h, best_val

def _coord_descent_frac(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    vals = [0.0, 0.1, 0.2, 0.25, 1.0/3.0, 0.4, 0.5, 0.6, 2.0/3.0, 0.7, 0.75, 0.8, 0.9, 1.0]
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-6:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return h

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.9 - 0.6 * t_frac
        cr = 0.5 + 0.3 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    trial[j] = mutant[j]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 20:
            break
    return best_h, best_c5

def _refine_local(h0, n, target_sum, deadline):
    h = h0.copy()
    best_val = _c5_of(h)
    alphas = [50, 200, 800, 3000, 10000, 50000]
    for a in alphas:
        if time.time() > deadline:
            break
        h_new = _slsqp_opt(h, n, target_sum, a, maxiter=400)
        cv = _c5_of(h_new)
        if cv < best_val:
            h = h_new
            best_val = cv
    if time.time() < deadline:
        for _ in range(3):
            if time.time() > deadline:
                break
            h_cd = _coord_descent_frac(h, n, target_sum, max_rounds=8)
            cv_cd = _c5_of(h_cd)
            if cv_cd < best_val:
                h = h_cd
                best_val = cv_cd
            h_new = _slsqp_opt(h, n, target_sum, 20000, maxiter=300)
            cv = _c5_of(h_new)
            if cv < best_val:
                h = h_new
                best_val = cv
    return h, best_val
