        # First term: (1 - h[m+l]) for valid m
        m_lo = max(0, -l)
        m_hi = min(n, n - l)

        m2_hi = min(n, l + n)
        if m2_lo < m2_hi:
            grad[m2_lo:m2_hi] -= wi * h[m2_lo - l:m2_hi - l]
    return grad

def project_feasible(h, target_sum):
