The derivative of corr[m] with respect to a[k] (which is h[k]):
- a[k] appears in corr[m] when i = k, contributing b[k + m - n + 1]
- a[k] also appears in b[j] where j = k, i.e., b[k] = 1 - a[k], so the derivative of b[k] wrt a[k] is -1
  - b[k] appears in corr[m] when i + m - n + 1 = k, i.e., i = k - m + n - 1
  - So the contribution is -a[k - m + n - 1]

Therefore:
d(corr[m])/d(h[k]) = b[k+m-n+1] * I[0≤k+m-n+1<n] + (-1) * a[k-m+n-1] * I[0≤k-m+n-1<n]
= (1-h[k+m-n+1]) * I[valid] - h[k-m+n-1] * I[valid]

The gradient of the LSE objective:
∇f[h][k] = dx * Σ_m w[m] * d(corr[m])/d(h[k])
= dx * Σ_m w[m] * (1-h[k+m-n+1]) * I[0≤k+m-n+1<n]
  - dx * Σ_m w[m] * h[k-m+n-1] * I[0≤k-m+n-1<n]

For the first sum, let j = m - n + 1, so m = j + n - 1:
= dx * Σ_j w[j+n-1] * (1-h[k+j]) * I[0≤k+j<n]

This is the cross-correlation of w (reversed or not?) with (1-h).

Let me define w_centered[j] = w[j+n-1] for j = -(n-1), ..., (n-1).

Then the first term is: Σ_j w_centered[j] * (1-h[k+j]) where 0 ≤ k+j < n.

This is like a convolution of w_centered with (1-h), evaluated at position k, but with the constraint that k+j is in [0,n).

If I pad (1-h) to length 4n with zeros, and w_centered to length 4n, then the full cross-correlation gives me what I want (the zero-padding ensures the boundary condition).

Let me use this FFT approach:

```python
def lse_gradient_fft(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500, 0)
    w = np.exp(shifted)
    w = w / np.sum(w)
    
    # w has length 2n-1, centered at index n-1
    # w_centered[j] = w[j + n - 1] for j = -(n-1), ..., (n-1)
    
    # Pad to length 4n for FFT
    N = 4 * n
    one_minus_h = 1.0 - h  # length n
    h_padded = np.zeros(N)
    h_padded[:n] = h
    omh_padded = np.zeros(N)
    omh_padded[:n] = one_minus_h
    
    # w_padded: place w centered in array of length N
    w_padded = np.zeros(N)
    w_padded[n-1 : n-1+(2*n-1)] = w  # w from index n-1 to 3n-3
    
    # First term: sum_j w_centered[j] * (1-h[k+j]) for 0<=k+j<n
    # = cross-correlation of omh_padded with w_padded
    # np.correlate(a, v)[k] = sum_i a[i] * v[i-k] (in full mode)
    # Hmm, I need to be careful about the convention.
    # 
    # Actually, let me just use the direct approach. For n=400, O(n^2) is fine.
    
    grad = np.zeros(n)
    offsets = np.arange(2*n-1) - (n-1)
    
    for k in range(n):
        idx1 = k + offsets
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[k] += dx * np.dot(w[valid1], 1.0 - h[idx1[valid1]])
        
        idx2 = k - offsets
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):

    return grad
```

Let me think about the Nesterov implementation more carefully.

For FISTA (Fast Iterative Shrinkage-Thresholding Algorithm):
- We want to minimize f(x) = LSE(x) subject to x ∈ C = {x : 0 ≤ x ≤ 1, Σx = target}
- f is smooth with Lipschitz constant L
- The proximal operator of g(x) = I_C(x) (indicator of C) is the projection onto C

FISTA iteration:
1. z_k = x_k + (t_k - 1)/t_{k+1} * (x_k - x_{k-1})
2. x_{k+1} = proj_C(z_k - (1/L) * ∇f(z_k))
3. t_{k+1} = (1 + sqrt(1 + 4t_k²)) / 2

The challenge is estimating L. For the LSE with parameter alpha:
f(x) = (1/alpha) * log(Σ exp(alpha * c(k)))
∇f(x) = Σ w(k) * ∇c(k) where w(k) = softmax(alpha * c(k))

The Hessian of f is:
∇²f = alpha * Σₖ w(k) * ∇²c(k) + alpha² * Σₖ w(k) * (∇c(k))(∇c(k))ᵀ ...

- L ≈ ||∇f(x1) - ∇f(x2)|| / ||x1 - x2||

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)
```


```python
def slsqp_obj(x, alpha, n):
```

```python
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)

def power_mean_obj(x, p, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr * dx
    return float(dx * 0.0 + np.sum(c ** p) ** (1.0 / p))


def lse_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_max = np.max(c)
    shifted = np.clip(alpha * (c - c_max), -500.0, 0.0)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    if w_sum < 1e-15:
        return np.zeros(n)
    w = w / w_sum
    grad = np.zeros(n)
    offsets = np.arange(2 * n - 1) - (n - 1)
    for k in range(n):
        idx1 = k + offsets
        valid1 = (idx1 >= 0) & (idx1 < n)
        if np.any(valid1):
            grad[k] += dx * np.dot(w[valid1], 1.0 - h[idx1[valid1]])
        idx2 = k - offsets
        valid2 = (idx2 >= 0) & (idx2 < n)
        if np.any(valid2):
            grad[k] -= dx * np.dot(w[valid2], h[idx2[valid2]])
    return grad

        else:
            no_improve += 1
            if no_improve > max_no_improve:
                no_improve = 0
                scale_idx += 1
                if scale_idx >= len(scales):
                    break

    return best, best_score


def block_anneal(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)
    current = h.copy()
    current_score = compute_c5(current)

    T = 0.008
    T_min = 1e-7
    cool_rate = 0.9995

    while time.time() - start < time_limit:
        T = max(T * cool_rate, T_min)
        move_type = rng.rand()

        if move_type < 0.4:
            i = rng.randint(n)
            d = rng.normal(0.0, 0.015)
            h_new = current.copy()
            val = current[i] + d
            if val < 0.0 or val > 1.0:
                continue
            h_new[i] = val
            h_new = project_feasible(h_new, target)
        elif move_type < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d = rng.normal(0.0, 0.02)
            vi = current[i] - d
            vj = current[j] + d
            if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                continue
            h_new = current.copy()
            h_new[i] = vi
            h_new[j] = vj
        else:
            blen = rng.randint(2, 6)
            i = rng.randint(max(1, n - blen))
            d = rng.normal(0.0, 0.008)
            h_new = current.copy()
            valid = True
            for b in range(blen):
                val = h_new[i + b] + d
                if val < 0.0 or val > 1.0:
                    valid = False
                    break
                h_new[i + b] = val
            if not valid:
                continue
            h_new = project_feasible(h_new, target)

        s = compute_c5(h_new)
        if s < current_score:
            current = h_new
            current_score = s
            if s < best_score:
                best_score = s
                best = h_new.copy()
        else:
            diff = s - current_score
            if diff < T * 50.0:
                accept_p = np.exp(-diff / max(T, 1e-10))
                if rng.rand() < accept_p:
                    current = h_new
                    current_score = s

    return best, best_score


def slsqp_refine(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    bounds = [(0.0, 1.0)] * n

    alphas = [20, 60, 200, 600]
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(100, remaining * 3))
        try:
            res = minimize(slsqp_obj, best, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': slsqp_cons,
                                         'jac': slsqp_cons_jac, 'args': (target,)}],
                           args=(alpha, n),
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    p_values = [10, 30, 100]
    for p in p_values:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        maxiter = int(max(50, remaining * 2))
        try:
            res = minimize(power_mean_obj, best, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': slsqp_cons,
                                         'jac': slsqp_cons_jac, 'args': (target,)}],
                           args=(p, n),
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass

    return best, best_score


def coordinate_fine(h, target, time_limit, seed=42):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(best)

    deltas = [0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002, 0.004, -0.004]
    no_improve_sweeps = 0

    while time.time() - start < time_limit:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() - start >= time_limit:
                break
            for d in deltas:
                j = rng.randint(n)
                if j == i:
                    continue
                vi = best[i] - d
                vj = best[j] + d
                if vi < 0.0 or vi > 1.0 or vj < 0.0 or vj > 1.0:
                    continue
                h_new = best.copy()
                h_new[i] = vi
                h_new[j] = vj
                s = compute_c5(h_new)
                if s < best_score - 1e-15:
                    best_score = s
                    best = h_new
                    improved = True
                    break
            if time.time() - start >= time_limit:
                break
        no_improve_sweeps = 0 if improved else no_improve_sweeps + 1
        if not improved:
            deltas = [d * 0.5 for d in deltas]
            if max(abs(d) for d in deltas) < 1e-6:
                break

    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    # Phase 1: SLSQP with LSE (proven effective)
    if tleft() > 250:
        h1, s1 = slsqp_refine(best_h, target, time_limit=min(250, tleft()))
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: FISTA with LSE (accelerated gradient)
    if tleft() > 200:
        h2, s2 = fista_phase(best_h, target, time_limit=min(200, tleft()),
                             alphas=(10, 30, 100, 300))
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Swap-based descent (preserves constraint exactly)
    if tleft() > 200:
        h3, s3 = swap_descent(best_h, target, time_limit=min(200, tleft()), seed=seed)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Block annealing (structured perturbation)
    if tleft() > 150:
        h4, s4 = block_anneal(best_h, target, time_limit=min(150, tleft()), seed=seed + 10)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: Fine coordinate descent with swaps
    if tleft() > 150:
        h5, s5 = coordinate_fine(best_h, target, time_limit=min(150, tleft()), seed=seed + 20)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Another FISTA round with higher alphas
    if tleft() > 150:
        h6, s6 = fista_phase(best_h, target, time_limit=min(150, tleft()),
                             alphas=(100, 300, 1000))
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Final swap polish
    if tleft() > 100:
        h7, s7 = swap_descent(best_h, target, time_limit=min(100, tleft()), seed=seed + 30)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Final SLSQP
    if tleft() > 80:
        h8, s8 = slsqp_refine(best_h, target, time_limit=min(80, tleft()))
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Final micro-adjustment
    if tleft() > 20:
        h9, s9 = coordinate_fine(best_h, target, time_limit=max(10, tleft()), seed=seed + 40)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
```
