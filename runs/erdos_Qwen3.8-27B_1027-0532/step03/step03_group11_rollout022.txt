def compute_swap_delta(h, oi, zi, n):
    """Compute the delta in correlation when h[oi] goes 1->0 and h[zi] goes 0->1."""
    delta = np.zeros(2*n-1)
    
    # oi going 1->0
    # Term 1: -(1-h[oi+s]) for -oi <= s <= n-1-oi
    s_vals = np.arange(max(-oi, -(n-1)), min(n-1-oi, n-1) + 1)
    delta[n-1+s_vals] -= (1.0 - h[oi + s_vals])
    
    # Term 2: +h[oi-s] for oi-(n-1) <= s <= oi
    s_vals2 = np.arange(max(oi-(n-1), -(n-1)), oi + 1)
    delta[n-1-s_vals2] += h[oi - s_vals2]
    
    # zi going 0->1

        return h0, compute_c5(h0)

def delta_at_lag(h, oi, zi, ki, n):
    s = ki - (n - 1)
    n_lo = max(0, -s) if s < 0 else 0
    n_hi = min(n - 1, n - 1 - s) if s > 0 else n - 1
    if s <= 0:
        n_lo = max(0, -s)
    else:
        n_lo = 0
    if s >= 0:
        n_hi = min(n - 1, n - 1 - s)
    else:
        n_hi = n - 1
    d = 0.0
    if n_lo <= oi - s <= n_hi:
        d -= (1.0 - h[oi - s])
    if n_lo <= oi <= n_hi:
        d += h[oi + s]
    if n_lo <= zi - s <= n_hi:
        d += (1.0 - h[zi - s])
    if n_lo <= zi <= n_hi:
        d -= h[zi + s]
    return d

def full_swap_delta(h, oi, zi):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    s_lo1 = max(-oi, -(n - 1))
    s_hi1 = min(n - 1 - oi, n - 1)
    s = np.arange(s_lo1, s_hi1 + 1)
    delta[n - 1 + s] -= (1.0 - h[oi + s])
    s_lo2 = max(oi - (n - 1), -(n - 1))
    s_hi2 = min(oi, n - 1)
    s2 = np.arange(s_lo2, s_hi2 + 1)
    delta[n - 1 - s2] += h[oi - s2]
    s_lo3 = max(-zi, -(n - 1))
    s_hi3 = min(n - 1 - zi, n - 1)
    s3 = np.arange(s_lo3, s_hi3 + 1)
    delta[n - 1 + s3] += (1.0 - h[zi + s3])
    s_lo4 = max(zi - (n - 1), -(n - 1))
    s_hi4 = min(zi, n - 1)
    s4 = np.arange(s_lo4, s_hi4 + 1)
    delta[n - 1 - s4] -= h[zi - s4]
    return delta

def fast_binary_local_search(h, time_limit, start_time, top_k=8):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    current_max = np.max(corr)
    rounds = 0
    while time.time() - start_time < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_indices = np.argsort(corr)[-top_k:]
        found = False
        best_improve = 1e-13
        best_oi = -1
        best_zi = -1
        for oi in ones:
            if found:
                break
            for zi in zeros:
                quick_ok = True
                for ki in top_indices:
                    d = delta_at_lag(h, oi, zi, ki, n)
                    if corr[ki] + d >= current_max - 1e-13:
                        quick_ok = False
                        break
                if not quick_ok:
                    continue
                d_full = full_swap_delta(h, oi, zi)
                new_corr = corr + d_full
                new_max = np.max(new_corr)
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_oi = oi
                    best_zi = zi
                    if improve > 0.005:
                        found = True
                        break
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = full_swap_delta(h, best_oi, best_zi)
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        current_max = np.max(corr)
        rounds += 1
        if rounds % 500 == 0 and time.time() - start_time > time_limit:
            break
    return h, current_max

def steepest_binary_search(h, time_limit, start_time, top_k=10):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    current_max = np.max(corr)
    while time.time() - start_time < time_limit:
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        top_indices = np.argsort(corr)[-top_k:]
        best_improve = 1e-13
        best_oi = -1
        best_zi = -1
        for oi in ones:
            for zi in zeros:
                quick_ok = True
                for ki in top_indices:
                    d = delta_at_lag(h, oi, zi, ki, n)
                    if corr[ki] + d >= current_max - 1e-13:
                        quick_ok = False
                        break
                if not quick_ok:
                    continue
                d_full = full_swap_delta(h, oi, zi)
                new_max = np.max(corr + d_full)
                improve = current_max - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_oi = oi
                    best_zi = zi
        if best_oi < 0:
            break
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = compute_corr(h)
        current_max = np.max(corr)
    return h, current_max

def make_binary_from_continuous(h, rng):
    n = len(h)
    target = n // 2
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n - target:]] = 1.0
    return result

def thue_morse_binary(n, rng):
    result = np.zeros(n)
    result[0] = 1.0
    for i in range(1, n):
        result[i] = 1.0 - result[i >> 1]
    ones_count = np.sum(result)
    if ones_count > n // 2:
        excess = int(ones_count) - n // 2
        one_indices = np.where(result == 1.0)[0]
        flip_idx = rng.choice(one_indices, size=excess, replace=False)
        result[flip_idx] = 0.0
    elif ones_count < n // 2:
        deficit = n // 2 - int(ones_count)
        zero_indices = np.where(result == 0.0)[0]
        flip_idx = rng.choice(zero_indices, size=deficit, replace=False)
        result[flip_idx] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def block_binary(n, rng, nblocks):
    result = np.zeros(n)
    block_size = n // nblocks
    for b in range(nblocks):
        start = b * block_size
        end = start + block_size if b < nblocks - 1 else n
        if rng.random() < 0.5:
            result[start:end] = 1.0
    ones_count = int(np.sum(result))
    if ones_count > n // 2:
        one_indices = np.where(result == 1.0)[0]
        flip = rng.choice(one_indices, size=ones_count - n // 2, replace=False)
        result[flip] = 0.0
    elif ones_count < n // 2:
        zero_indices = np.where(result == 0.0)[0]
        flip = rng.choice(zero_indices, size=n // 2 - ones_count, replace=False)
        result[flip] = 1.0
    return result

def upsample(h, factor):
    n = len(h)
    n_new = n * factor
    h_new = np.zeros(n_new)
    for i in range(n):
        h_new[i * factor:(i + 1) * factor] = h[i]
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None

    if h0 is not None:
        n0 = len(h0)
    else:
        n0 = 180
        h0 = project_onto_constraint(
            np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0, 1), n0 / 2.0)

    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        h_opt, c5_opt = slsqp_optimize(h0, 0.005, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    bin_candidates = []
    bin_candidates.append(make_binary_from_continuous(best_h, rng))
    bin_candidates.append(thue_morse_binary(n0, rng))
    bin_candidates.append(block_binary(n0, rng, 6))
    bin_candidates.append(block_binary(n0, rng, 12))
    h_pert = best_h + 0.2 * rng.standard_normal(n0)
    bin_candidates.append(make_binary_from_continuous(h_pert, rng))

    best_bin_h = None
    best_bin_c5 = float('inf')
    for bc in bin_candidates:
        c = compute_c5(bc)
        if c < best_bin_c5:
            best_bin_c5 = c
            best_bin_h = bc.copy()

    bin_time_limit = budget_s - 500
    if time.time() - start_time < bin_time_limit:
        best_bin_h, best_bin_c5 = fast_binary_local_search(
            best_bin_h, bin_time_limit, start_time, top_k=6)

    best_bin_h, best_bin_c5 = steepest_binary_search(
        best_bin_h, budget_s - 400, start_time, top_k=12)

    h_refine = best_bin_h.astype(float) + 0.02 * rng.standard_normal(n0)
    h_refine = project_onto_constraint(h_refine, target_sum)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        eps_list = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
        for eps in eps_list:
            if time.time() - start_time > budget_s - 200:
                break
            h_opt, c5_opt = slsqp_optimize(h_refine, eps, 300)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_refine = h_opt.copy()

    elapsed = time.time() - start_time
    restart_count = 0
    while budget_s - (time.time() - start_time) > 80 and restart_count < 20:
        strategy = restart_count % 5
        if strategy == 0:
            h_pert = best_h + 0.01 * rng.standard_normal(n0)
        elif strategy == 1:
            h_pert = best_bin_h.astype(float) + 0.03 * rng.standard_normal(n0)
        elif strategy == 2:
            h_pert = thue_morse_binary(n0, rng).astype(float) + 0.1 * rng.standard_normal(n0)
        elif strategy == 3:
            h_pert = block_binary(n0, rng, int(rng.integers(4, 16))).astype(float) + 0.1 * rng.standard_normal(n0)
        else:
            alpha = rng.uniform(0.3, 0.7)
            h_rand = random_binary(n0, rng).astype(float)
            h_pert = alpha * best_h + (1.0 - alpha) * h_rand

        h_pert = project_onto_constraint(h_pert, target_sum)
        h_opt, c5_opt = slsqp_optimize(h_pert, 0.001, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_opt2, c5_opt2 = slsqp_optimize(h_opt, 0.0001, 300)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
        restart_count += 1

    elapsed = time.time() - start_time
    if budget_s - elapsed > 200:
        n1 = 2 * n0
        h_up = upsample(best_h, 2)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.01 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        for eps in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            if time.time() - start_time > budget_s - 50:
                break
            h_opt_up, c5_opt_up = slsqp_optimize(h_up, eps, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()
        n_cur = len(best_h)
        bin_h_up = make_binary_from_continuous(best_h, rng)
        if budget_s - (time.time() - start_time) > 100:
            bin_h_up, _ = fast_binary_local_search(bin_h_up, budget_s - 100, start_time, top_k=8)
            h_ref_up = bin_h_up.astype(float) + 0.01 * rng.standard_normal(n_cur)
            h_ref_up = project_onto_constraint(h_ref_up, n_cur / 2.0)
            for eps in [0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
                if time.time() - start_time > budget_s - 30:
                    break
                h_opt_up, c5_opt_up = slsqp_optimize(h_ref_up, eps, 200)
                if c5_opt_up < best_c5:
                    best_c5 = c5_opt_up
                    best_h = h_opt_up.copy()
                h_ref_up = h_opt_up.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        n_cur = len(best_h)
        for eps in [0.00005, 0.00002, 0.00001]:
            if time.time() - start_time > budget_s - 10:
                break
            h_opt, c5_opt = slsqp_optimize(best_h, eps, 400)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
