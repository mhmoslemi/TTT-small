import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:s+m] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:s+m]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:r+m]
            grad[r:r+m] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

            t_size = 5
            t_idx = rng.integers(0, pop_size, size=t_size)
            best_t = t_idx[np.argmin(fitness[t_idx])]
            p1 = pop[best_t]

            t_idx2 = rng.integers(0, pop_size, size=t_size)
            best_t2 = t_idx2[np.argmin(fitness[t_idx2])]
            p2 = pop[best_t2]

            if rng.random() < crossover_rate:
                child = ga_crossover(p1, p2, rng)
            else:
                child = p1.copy()

            strat = int(rng.integers(0, 9))
            child = ga_mutate(child, rng, strat, n0)

            if rng.random() < 0.04:
                child = (rng.random(n0) < 0.5).astype(float)
                child = project_onto_constraint(child, target_sum)

            new_pop.append(child)

        fitness = np.array([compute_c5(p) for p in new_pop])

        gen_best = float(np.min(fitness))
        if gen_best < best_c5 - 1e-9:
            best_c5 = gen_best
            best_h = new_pop[int(np.argmin(fitness))].copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            for i in range(elite_size, len(new_pop)):
                strat = int(rng.integers(3, 9))
                new_pop[i] = ga_mutate(new_pop[i], rng, strat, n0)
            fitness = np.array([compute_c5(p) for p in new_pop])
            stall = 0
            stall_limit = max(12, stall_limit // 2)

        pop = new_pop
        gen += 1

    # Quick SLSQP polish on GA best
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Targeted SLSQP from multiple GA candidates
    if rem() > 120:
        order = np.argsort(fitness)
        tried = 0
        for i in order[:8]:
            if rem() < 40:
                break
            candidate = pop[i]
            h_opt, c5_opt = slsqp_optimize(candidate, 0.002, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            tried += 1

    # Phase 4: Projected gradient on best
    if rem() > 80:
        tl = min(rem() - 40, 150)
        h_pg, c5_pg = proj_grad_topk(best_h, 3, 0.005, 0.06, 300000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Phase 5: Fine SLSQP
    eps_list = [0.001, 0.0005, 0.0002, 0.0001]
    for eps in eps_list:
        if rem() > 45:
            h_opt, c5_opt = slsqp_optimize(best_h, eps, 250)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Resolution increase
    if rem() > 180:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.004 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.004, 250)
        if rem() > 80:
            h_opt_up2, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.001, 250)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
                h_opt_up = h_opt_up2
        if rem() > 50:
            tl_up = min(rem() - 30, 80)
            h_pg_up, c5_pg_up = proj_grad_topk(h_opt_up, 2, 0.003, 0.04, 200000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n0 = n1

    # Phase 7: Second resolution increase if time
    if rem() > 150:
        n2 = n0 + 30
        old_t = np.linspace(0.0, 1.0, len(best_h))
        new_t = np.linspace(0.0, 1.0, n2)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n2 / 2.0)
        h_up = h_up + 0.003 * rng.standard_normal(n2)
        h_up = project_onto_constraint(h_up, n2 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.003, 200)
        if rem() > 50:
            h_opt_up2, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.0005, 200)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
                h_opt_up = h_opt_up2

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()

    # Phase 8: Final polish
    if rem() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final projected gradient
    if rem() > 20:
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.02, 200000, rem() - 5, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
