def fast_sa_v2(h_start, n_steps, t_start, t_end, rng):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    
    # Use arrays for tracking
    is_one = (h == 1.0)
    
    half = n - 1
    
    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Pick random swap
        i = ones_idx[rng.integers(len(ones_idx))]
        j = zeros_idx[rng.integers(len(zeros_idx))]
        if i == j:
            continue
        
        # Compute range of affected correlations
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        
        if s_min > s_max:
            # No overlap, just swap
            h[i], h[j] = h[j], h[i]
            # Update tracking
            pos = np.where(ones_idx == i)[0][0]
            ones_idx[pos] = j
            pos = np.where(zeros_idx == j)[0][0]
            zeros_idx[pos] = i
            continue
        
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        
        # Compute delta
        # h[i] goes 1->0, h[j] goes 0->1
        # (1-h)[i] goes 0->1, (1-h)[j] goes 1->0
        # 
        # For correlation at lag s (s = k - half):
        # corr[s] = sum_m h[m] * (1-h)[m+s]
        # 
        # Change from h[i]: 1->0 removes h[i]*(1-h)[i+s] for each s
        # Change from h[j]: 0->1 adds (1-h)[j+s] for each s
        # Change from (1-h)[i]: 0->1 adds h[i-s]*1 for each s (where i-s is valid)
        # Change from (1-h)[j]: 1->0 removes h[j-s]*1 for each s (where j-s is valid)
        
        # For a given s:
        # delta_s = -h[i]*(1-h)[i+s] + (1-h)[j+s] + h[i-s] - h[j-s]
        #         = -(1-h)[i+s] + (1-h)[j+s] + h[i-s] - h[j-s]
        #         = -omh[i+s] + omh[j+s] + h[i-s] - h[j-s]
        # where omh = 1-h
        
        omh = 1.0 - h
        
        # Need to be careful with bounds
        # i+s valid when 0 <= i+s < n
        # j+s valid when 0 <= j+s < n
        # i-s valid when 0 <= i-s < n
        # j-s valid when 0 <= j-s < n
        
        delta = np.zeros(s_max - s_min + 1)
        for idx, s in enumerate(s_range):
            d = 0.0
            if 0 <= i + s < n:
                d -= omh[i + s]
            if 0 <= j + s < n:
                d += omh[j + s]
            if 0 <= i - s < n:
                d += h[i - s]
            if 0 <= j - s < n:
                d -= h[j - s]
            delta[idx] = d
        
        corr_test = corr[k_range].copy()
        corr_test += delta
        new_max = float(np.max(corr_test))
        # Also need to check max outside k_range
        if len(corr) - len(k_range) > 0:
            outside_max = float(np.max(np.concatenate([corr[:k_range[0]], corr[k_range[-1]+1:]]))) if not (k_range[0] == 0 and k_range[-1] == len(corr)-1) else 0.0
        else:
            outside_max = 0.0
        new_max = max(new_max, outside_max)
        
        delta_c5 = new_max - current_max
        if delta_c5 < 0 or (t > 1e-8 and rng.random() < np.exp(-delta_c5 / t)):
            # Accept
            corr[k_range] += delta
            h[i], h[j] = h[j], h[i]
            # Update tracking - this is the slow part
            pos_i = np.where(ones_idx == i)[0][0]
            pos_j = np.where(zeros_idx == j)[0][0]
            ones_idx[pos_i] = j
            zeros_idx[pos_j] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
    
    return best_h, best_c5_val

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp  # shape (2n-1,)
    grad = np.zeros(n)
    half = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - half
        if s >= 0:
            # omh term: m in [0, n-s)
            if n - s > 0:
                grad[:n-s] += wk * dx * omh[s:]
            # h term: m in [s, n)
            if n - s > 0:
                grad[s:] -= wk * dx * h[:n-s]
        else:
            r = -s
            # omh term: m in [r, n)
            if n - r > 0:
                grad[r:] += wk * dx * omh[:n-r]
            # h term: m in [0, n-r)
            if n - r > 0:
                grad[:n-r] -= wk * dx * h[r:]
    return grad

import numpy as np

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))
