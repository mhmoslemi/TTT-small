    ones = np.where(h_r == 1)[0]

Actually, let me think about this differently. The key challenge is that the "max" objective has many local optima. A single swap can only improve the max by a tiny amount (at most 1*dx = 2/n ≈ 0.0033 for n=600). So we need to accept many small improvements to make progress.

SA addresses this by accepting worse moves with some probability. Let me combine SA with ILS:

1. Do SA for a while (accepting worse moves with probability exp(-Δ/T))
2. Perturb
3. Repeat

The key is to choose T well. If T is too small, SA is like local search. If T is too large, SA is like random search.

For the swap move, Δ = change in C₅. Since C₅ is the max, a swap can change it by at most 2/n (if the swap affects the max lag) or 0 (if it doesn't). Typically Δ is either 0 or ±1/n or ±2/n.

So Δ/T is important. If T = 0.01 and Δ = 1/600 ≈ 0.0017, then Δ/T ≈ 0.17, so we accept with probability exp(-0.17) ≈ 0.84. That's reasonable.

        total_iter += 1
        
        if iter_in_phase >= phase_length:
            T *= T_ratio
            if T < T_end:
                T = T_end

        i0 = zeros[i0_idx]

            h[i0] = 0

def smooth_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    # Compute g(d) for all d
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Compute softmax weights
    shifted = alpha * corr - alpha * np.max(corr)  # for numerical stability
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Compute gradient
    grad = np.zeros(n)
    # For each d, the contribution to grad is:
    # grad[j] += weights[n-1+d] * ((1-h[j-d]) * [0<=j-d<n] - h[j+d] * [0<=j+d<n])
    
    # Let's do this by iterating over d
    for d_idx in range(len(corr)):
        d = d_idx - (n - 1)
        w = weights[d_idx]
        if w < 1e-15:
            continue
        # First part: (1-h[j-d]) for valid j
        # j-d ranges from 0 to n-1, so j ranges from d to n-1+d
        j_start = max(0, d)
        j_end = min(n, n + d)  # exclusive
        if j_end > j_start:
            valid_j = np.arange(j_start, j_end)
            grad[valid_j] += w * (1.0 - h[valid_j - d])
        # Second part: -h[j+d] for valid j
        # j+d ranges from 0 to n-1, so j ranges from -d to n-1-d
        j_start2 = max(0, -d)
        j_end2 = min(n, n - d)
        if j_end2 > j_start2:
            valid_j2 = np.arange(j_start2, j_end2)
            grad[valid_j2] -= w * h[valid_j2 + d]
    
    return grad
```

def smooth_gradient_fast(h, alpha, top_k=10):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k lags
    idx = np.argsort(corr)[-top_k:]
    c_max = np.max(corr)
    shifted = alpha * corr[idx] - alpha * c_max
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for i, d_idx in enumerate(idx):
        if weights[i] < 1e-15:
            continue
        d = d_idx - (n - 1)
        w = weights[i]
        j_start = max(0, d)
        j_end = min(n, n + d)
        if j_end > j_start:
            valid_j = np.arange(j_start, j_end)
            grad[valid_j] += w * (1.0 - h[valid_j - d])
        j_start2 = max(0, -d)
        j_end2 = min(n, n - d)
        if j_end2 > j_start2:
            valid_j2 = np.arange(j_start2, j_end2)
            grad[valid_j2] -= w * h[valid_j2 + d]
    
    return grad
```

def compute_c5(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:

        a = rng.randint(len(ones))
        b = rng.randint(len(zeros))
        i1 = ones[a]
        i0 = zeros[b]
        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5(h)
        delta = c5_new - curr_c5
        if delta < 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
            curr_c5 = c5_new
            ones[a] = i0
            zeros[b] = i1
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i1] = 1
            h[i0] = 0
            no_improve += 1

    return best_h, best_c5


def sa_ils_multiflag(h_start, target_sum, allowed_time, rng, start_time,
                     T_start=0.01, T_end=0.00005, perturb_size=30, phase_len=12000):
    n = len(h_start)
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    T = T_start
    iter_phase = 0
    no_improve = 0
    max_no_improve = 100000
    end_time = start_time + allowed_time

    while time.time() < end_time:
        if no_improve > max_no_improve:
            break
        iter_phase += 1
        if iter_phase >= phase_len:
            T = max(T * 0.5, T_end)
            iter_phase = 0
            for _ in range(perturb_size):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                a = rng.randint(len(ones))
                b = rng.randint(len(zeros))
                h[ones[a]] = 0
                h[zeros[b]] = 1
                ones[a] = zeros[b]
                zeros[b] = ones[a]
            curr_c5 = compute_c5(h)
            no_improve = 0
            T = max(T, 0.004)

        move_type = rng.randint(3)
        i1 = -1
        i0 = -1
        if move_type == 0 and len(ones) > 0 and len(zeros) > 0:
            a = rng.randint(len(ones))
            b = rng.randint(len(zeros))
            i1 = ones[a]
            i0 = zeros[b]
        elif move_type == 1 and len(ones) >= 2 and len(zeros) >= 2:
            a1 = rng.randint(len(ones))
            a2 = rng.randint(len(ones) - 1)
            if a2 >= a1:
                a2 += 1
            b1 = rng.randint(len(zeros))
            b2 = rng.randint(len(zeros) - 1)
            if b2 >= b1:
                b2 += 1
            h[ones[a1]] = 0
            h[ones[a2]] = 0
            h[zeros[b1]] = 1
            h[zeros[b2]] = 1
            c5_new = compute_c5(h)
            delta = c5_new - curr_c5
            if delta < 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
                curr_c5 = c5_new
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[ones[a1]] = 0
                h[ones[a2]] = 0
                h[zeros[b1]] = 1
                h[zeros[b2]] = 1
                h[ones[a1]] = 1
                h[ones[a2]] = 1
                h[zeros[b1]] = 0
                h[zeros[b2]] = 0
                no_improve += 1
            continue
        else:
            continue

        if i1 < 0:
            continue

        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5(h)
        delta = c5_new - curr_c5
        if delta < 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
            curr_c5 = c5_new
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i1] = 1
            h[i0] = 0
            no_improve += 1

    return best_h, best_c5


def smooth_gradient_vec(h, alpha, top_k=80):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * corr - alpha * c_max
    weights = np.exp(shifted)
    w_sum = np.sum(weights)
    if w_sum < 1e-300:
        return np.zeros(n)
    top_k = min(top_k, len(corr))
    idx = np.argsort(corr)[-top_k:]
    w_top = weights[idx]
    w_norm = np.sum(w_top)
    if w_norm < 1e-300:
        return np.zeros(n)
    w_top = w_top / w_norm
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = w_top[i]
        if w < 1e-15:
            continue
        d = idx[i] - (n - 1)
        j_min = max(0, d)
        j_max = min(n, n + d)
        if j_max > j_min:
            grad[j_min:j_max] += w * (1.0 - h[j_min - d:j_max - d])
        j_min2 = max(0, -d)
        j_max2 = min(n, n - d)
        if j_max2 > j_min2:
            grad[j_min2:j_max2] -= w * h[j_min2 + d:j_max2 + d]
    return grad


def continuous_pd(h_start, n, target, time_limit, start_time, alphas=None):
    if alphas is None:
        alphas = [30, 80, 200, 500, 1200, 3000, 8000]
    h = project_feasible(h_start.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    end_time = start_time + time_limit

    for alpha in alphas:
        if time.time() > end_time:
            break
        lr = 0.5
        top_k = min(80, max(20, int(1000.0 / alpha) + 10))
        for it in range(300):
            if time.time() > end_time:
                break
            grad = smooth_gradient_vec(h, alpha, top_k)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            if sc < compute_c5(h) + 1e-12:
                lr = min(lr * 1.05, 2.0)
            else:
                lr *= 0.6
            h = h_new
        h = best_h

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def local_search_fast(h_start, target_sum, max_iters, rng):
    n = len(h_start)
    h = h_start.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0

    for _ in range(max_iters):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = rng.randint(len(ones))
        b = rng.randint(len(zeros))
        i1 = ones[a]
        i0 = zeros[b]
        h[i1] = 0
        h[i0] = 1
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            ones = np.where(h == 1)[0].copy()
            zeros = np.where(h == 0)[0].copy()
            stall = 0
        else:
            h[i1] = 1
            h[i0] = 0
            stall += 1
            if stall > 3000:
                break
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    h_bin = project_binary(h_init, target_init)
    bin_score = compute_c5(h_bin)
    if bin_score < best_score:
        best_score = bin_score
        best_h = h_bin.copy()

    h_work = h_bin.copy()

    sa_configs = [
        (0.010, 0.00005, 20, 15000),
        (0.008, 0.00003, 30, 12000),
        (0.012, 0.00008, 15, 20000),
        (0.006, 0.00002, 35, 10000),
        (0.015, 0.0001, 25, 18000),
    ]

    for ci, (T_s, T_e, psize, plen) in enumerate(sa_configs):
        if tleft() < 80:
            break
        allowed = min(150.0, tleft() * 0.35)
        if ci % 2 == 0:
            h_new, sc = sa_ils(h_work, target_init, allowed, rng, start_time,
                               T_start=T_s, T_end=T_e, perturb_size=psize, phase_len=plen)
        else:
            h_new, sc = sa_ils_multiflag(h_work, target_init, allowed, rng, start_time,
                                         T_start=T_s, T_end=T_e, perturb_size=psize, phase_len=plen)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        h_work = h_new

        if tleft() > 60:
            h_ls, sc_ls = local_search_fast(h_new, target_init, 20000, rng)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls.copy()
                h_work = h_ls

    if tleft() > 100:
        for rs in range(3):
            if tleft() < 60:
                break
            h_rand = np.zeros(n_init)
            perm = rng.permutation(n_init)
            h_rand[perm[:n_init // 2]] = 1.0
            allowed = min(80.0, tleft() * 0.25)
            h_new, sc = sa_ils(h_rand, target_init, allowed, rng, start_time,
                               T_start=0.015, T_end=0.0005, perturb_size=40, phase_len=8000)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h_work = h_new

    n_candidates = [n_init - 50, n_init + 50, n_init + 100, n_init - 100, n_init + 150]
    n_candidates = [n for n in n_candidates if n >= 200 and n != n_init]
    seen_ns = set([n_init])

    for n_try in n_candidates:
        if tleft() < 100:
            break
        if n_try in seen_ns:
            continue
        seen_ns.add(n_try)
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        h_bin_try = project_binary(h_interp, target_try)
        allowed = min(100.0, tleft() * 0.3)
        h_new, sc = sa_ils(h_bin_try, target_try, allowed, rng, start_time,
                           T_start=0.008, T_end=0.00005, perturb_size=25, phase_len=12000)
        if sc < best_score:
            best_score = sc
            best_h = h_new.copy()
        if tleft() > 50:
            h_ls, sc_ls = local_search_fast(h_new, target_try, 15000, rng)
            if sc_ls < best_score:
                best_score = sc_ls
                best_h = h_ls.copy()

    if tleft() > 50:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_cont = project_feasible(best_h, target_use)
        time_rem = tleft() - 40
        alphas_set = [30, 100, 300, 1000, 3000, 8000, 20000]
        h_ref, sc = continuous_pd(h_cont, n_use, target_use, time_rem, start_time, alphas_set)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        for fa in range(3):
            if tleft() < 20:
                break
            noise_scale = [0.02, 0.05, 0.1][fa]
            h_pert = best_h + rng.randn(n_use) * noise_scale
            h_pert = project_feasible(h_pert, target_use)
            alphas_fine = [200, 1000, 5000, 20000]
            h_ref, sc = continuous_pd(h_pert, n_use, target_use, min(15.0, tleft() - 10),
                                      start_time, alphas_fine)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if tleft() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_bin_final = project_binary(best_h, target_use)
        max_it = int(max(5000, tleft() * 2000))
        h_sa_f, c5_f = sa_ils(h_bin_final, target_use, max(5.0, tleft() - 10),
                              rng, start_time, T_start=0.004, T_end=0.00002,
                              perturb_size=15, phase_len=8000)
        if c5_f < best_score:
            best_score = c5_f
            best_h = h_sa_f
        if tleft() > 10:
            h_ref, sc = continuous_pd(best_h, n_use, target_use, max(3.0, tleft() - 8),
                                      start_time, [1000, 5000, 20000])
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
