        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = pop[int(np.argmin(caches))].copy()
    best_c5 = min(caches)
    stall = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.85 - 0.45 * t_frac
        cr = 0.55 + 0.25 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            cands = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(cands), size=3, replace=False)
            r1, r2, r3 = cands[r1], cands[r2], cands[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.random(n) < cr
            trial[mask] = mutant[mask]
            trial = _fix_sum(trial, target_sum)
            ct = _c5_of(trial)
            if ct <= caches[i]:
                new_pop.append(trial.copy())
                new_caches.append(ct)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = trial.copy()
                    improved = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved:
            stall += 1
        else:
            stall = 0
        if stall >= 25:
            break
    return best_h, best_c5

def _pair_swap_cd(h0, n, target_sum, max_rounds=20, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
    best_c = _c5_of(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            if c0 < best_c - 1e-14:
                h[j] = 0.0
                best_c = c0
                improved = True
            elif c1 < best_c - 1e-14:
                h[j] = 1.0
                best_c = c1
                improved = True
        n_swaps = min(200, n * 2)
        for _ in range(n_swaps):
            i, j = rng.randint(0, n, size=2)
            if i == j:
                continue
            oi, oj = h[i], h[j]
            h[i], h[j] = oj, oi
            cs = _c5_of(h)
            if cs < best_c - 1e-14:
                best_c = cs
                improved = True
            else:
                h[i], h[j] = oi, oj
        if not improved:
            break
    return h, best_c

def _focal_smooth_obj(h, k, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    c = corr * dx
    topk = np.sort(c)[::-1][:k]
    scaled = alpha * topk
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / alpha

def _slsqp_focal(h0, n, target_sum, k, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_focal_smooth_obj, h0, args=(k, alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return _fix_sum(h0, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 18
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    grid_sizes = []
    for g in [n_init, n_init + 20, n_init + 40, n_init + 60, n_init + 80]:
        if g % 2 == 0 and 100 <= g <= 300 and g not in grid_sizes:
            grid_sizes.append(g)

    for n in grid_sizes:
        if time.time() > deadline - 100:
            break
        target_sum = n / 2.0
        if n == n_init:
            h_start = _fix_sum(h_init.copy(), target_sum)
        else:
            x_old = np.arange(n_init, dtype=float)
            x_new = np.linspace(0, n_init - 1, n)
            h_interp = np.interp(x_new, x_old, h_init)
            h_start = _fix_sum(np.clip(h_interp, 0.0, 1.0), target_sum)
        c5_start = _c5_of(h_start)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
            best_n = n

        t_rem = deadline - 90 - time.time()
        if t_rem > 40:
            pop_size = min(25, max(12, n // 8))
            n_gen = min(200, max(60, int(t_rem / (pop_size * 0.025))))
            h_de, c5_de = _de_optimize(h_start, n, target_sum, pop_size, n_gen,
                                       deadline - 90, rng)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n
        curr_h = best_h if best_n == n else h_start

        if time.time() < deadline - 50:
            h_n1, c5_n1 = _nesterov_adaptive(curr_h, n, target_sum,
                                             n_iters=250, alpha_start=30, alpha_end=800,
                                             deadline=deadline - 50)
            if c5_n1 < _c5_of(curr_h):
                curr_h = h_n1
            if _c5_of(curr_h) < best_c5:
                best_c5 = _c5_of(curr_h)
                best_h = curr_h.copy()
                best_n = n

        if time.time() < deadline - 35:
            h_n2, c5_n2 = _nesterov_adaptive(curr_h, n, target_sum,
                                             n_iters=200, alpha_start=200, alpha_end=3000,
                                             deadline=deadline - 35)
            if c5_n2 < _c5_of(curr_h):
                curr_h = h_n2
            if _c5_of(curr_h) < best_c5:
                best_c5 = _c5_of(curr_h)
                best_h = curr_h.copy()
                best_n = n

        for alpha in [200, 500, 1000, 3000, 8000]:
            if time.time() > deadline - 25:
                break
            h_sl = _slsqp_lse(curr_h, n, target_sum, alpha, maxiter=350)
            c5_sl = _c5_of(h_sl)
            if c5_sl < _c5_of(curr_h):
                curr_h = h_sl
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = curr_h.copy()
                best_n = n

        if time.time() < deadline - 20:
            for k in [3, 5]:
                if time.time() > deadline - 15:
                    break
                h_fk = _slsqp_focal(curr_h, n, target_sum, k=k, alpha=500, maxiter=200)
                c5_fk = _c5_of(h_fk)
                if c5_fk < _c5_of(curr_h):
                    curr_h = h_fk
                if c5_fk < best_c5:
                    best_c5 = c5_fk
                    best_h = curr_h.copy()
                    best_n = n

        if time.time() < deadline - 12:
            h_ps, c5_ps = _pair_swap_cd(curr_h, n, target_sum, max_rounds=15, rng=rng)
            if c5_ps < _c5_of(curr_h):
                curr_h = h_ps
            if c5_ps < best_c5:
                best_c5 = c5_ps
                best_h = curr_h.copy()
                best_n = n

        if time.time() < deadline - 8:
            for alpha in [5000, 12000]:
                if time.time() > deadline - 6:
                    break
                h_pol = _slsqp_lse(curr_h, n, target_sum, alpha, maxiter=400)
                c5_pol = _c5_of(h_pol)
                if c5_pol < _c5_of(curr_h):
                    curr_h = h_pol
                if c5_pol < best_c5:
                    best_c5 = c5_pol
                    best_h = curr_h.copy()
                    best_n = n

    if best_h is not None and time.time() < deadline - 5:
        nb = best_n
        tgt = nb / 2.0
        hf = _fix_sum(best_h, tgt)
        if time.time() < deadline - 4:
            hp = _slsqp_lse(hf, nb, tgt, 15000, maxiter=500)
            cp = _c5_of(hp)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()
        if time.time() < deadline - 3:
            hc, cc = _pair_swap_cd(best_h, best_n, tgt, max_rounds=25, rng=rng)
            if cc < best_c5:
                best_c5 = cc
                best_h = hc.copy()
        if time.time() < deadline - 2:
            hp2 = _slsqp_lse(best_h, best_n, tgt, 20000, maxiter=400)
            cp2 = _c5_of(hp2)
            if cp2 < best_c5:
                best_c5 = cp2
                best_h = hp2.copy()
        final_h = _fix_sum(best_h, best_n / 2.0)
        best_c5 = _c5_of(final_h)
        best_h = final_h

    return (best_h, best_c5, best_n)
