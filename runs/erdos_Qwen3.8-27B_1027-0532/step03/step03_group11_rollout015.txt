```python
import numpy as np
from scipy.optimize import minimize
import time

# === Helper functions ===

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum}"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 = max_k correlate(h, 1-h)[k] * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def lgbf_objective(h, eps, K, lam, target):
    """Objective: smooth top-K max of corr + penalty for sum constraint"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Top-K smooth max
    if K >= len(corr):
        vals = corr
    else:
        idx = np.argpartition(corr, len(corr) - K)[len(corr) - K:]
        vals = corr[idx]
    cmax = vals.max()
    x = (vals - cmax) / eps
    x -= x.max()
    obj = cmax + eps * np.log(np.sum(np.exp(x)))
    # Penalty
    sv = h.sum() - target
    obj += lam * sv * sv
    return obj

def lgbf_gradient(h, eps, K, lam, target):
    """Gradient of the objective"""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    # Top-K weights
    nc = len(corr)
    if K >= nc:
        idx = np.arange(nc)
    else:
        idx = np.argpartition(corr, nc - K)[nc - K:]
    vals = corr[idx]
    cmax = vals.max()
    x = (vals - cmax) / eps
    x -= x.max()
    ex = np.exp(x)
    sexp = ex.sum()
    w = np.zeros(nc)
    w[idx] = ex / sexp
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    
    # Gradient of penalty
    sv = h.sum() - target
    grad += 2.0 * lam * sv
    return grad

def lgbf_optimize(h_init, eps, K, lam, maxiter, time_limit, start_time):
    """Run L-BFGS-B optimization"""
    n = len(h_init)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return lgbf_objective(x, eps, K, lam, target)
    
    def grad(x):
        return lgbf_gradient(x, eps, K, lam, target)
    
    try:
        result = minimize(obj, h_init.copy(), jac=grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-9})
        h_opt = project_onto_constraint(result.x, target)
        c5 = compute_c5(h_opt)
        return h_opt, c5
    except Exception:
        h_init_p = project_onto_constraint(h_init, target)
        return h_init_p, compute_c5(h_init_p)
```
