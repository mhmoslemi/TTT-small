import time

def compute_corr(h):

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def active_gradient(h, shifts, n):
    """Compute gradient of sum_k corr[k] for given shifts (k - (n-1))."""
    grad = np.zeros(n)
    for shift in shifts:
        if shift == 0:
            grad += (1.0 - 2.0 * h)
        elif shift > 0:
            # (1-h[j-shift]) for j >= shift
            grad[shift:] += (1.0 - h[:n - shift])
            # -h[j+shift] for j < n-shift
            grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            # (1-h[j+s]) for j < n-s
            grad[:n - s] += (1.0 - h[s:])
            # -h[j-s] for j >= s
            grad[s:] -= h[:n - s]
    return grad

def get_active_shifts(corr, n, top_m=30, epsilon=0.0):

    max_val = np.max(corr)

        indices = np.where(mask)[0]
    else:
        top_idx = np.argsort(corr)[-top_m:]
        indices = top_idx
    shifts = indices - (n - 1)

def slsqp_refine(h, target, time_limit):

    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    alphas = [10, 30, 100, 300]
    n_alphas = len(alphas)
    
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (n_alphas - ai)
        maxiter = int(max(50, time_per * 2))
        
        def obj(x, a=alpha):
            corr = np.correlate(x, 1.0 - x, mode='full')
            c_max = np.max(corr)
            shifted = np.clip(a * (corr - c_max), -500.0, 0.0)
            return float(c_max + np.log(np.sum(np.exp(shifted))) / a)
        
        def cons_fun(x):
            return np.sum(x) - target
        
        def cons_jac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
        
        try:
            res = minimize(obj, best, method='SLSQP', bounds=bounds,

def lse_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights = weights / sw
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = shifts[k]
        if s == 0:
            grad += weights[k] * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += weights[k] * (1.0 - h[:n-s])
            grad[:n-s] -= weights[k] * h[s:]
        else:
            ss = -s
            grad[:n-ss] += weights[k] * (1.0 - h[ss:])
            grad[ss:] -= weights[k] * h[:n-ss]
    return grad
```

This loops over all 799 lags, but for large alpha, most weights are negligible, so the inner loop body is skipped quickly. For each active lag, the operation is O(n). Total cost is O(n * m_active) where m_active is the number of lags with weight > 1e-10.

For alpha=50 and a typical solution, m_active might be 5-20. So the total cost is O(n * 20) = O(8000), very fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def lse_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-15:
        return np.zeros(n)
    weights = weights / sw
    shifts = np.arange(len(corr)) - (n - 1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-10:
            continue
        s = int(shifts[k])
        w = weights[k]
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            ss = -s
            grad[:n - ss] += w * (1.0 - h[ss:])
            grad[ss:] -= w * h[:n - ss]
    return grad

def lse_objective(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

def lse_gradient_for_slsqp(x, alpha):
    return lse_gradient(x, alpha)

def sum_constraint(x, target):
    return np.sum(x) - target

def sum_constraint_jac(x, target):
    return np.ones(len(x))

def gradient_descent_phase(h, target, time_limit, alpha_init=20.0, eta_init=0.01, alpha_growth=1.1):
    n = len(h)
    start = time.time()
    alpha = alpha_init
    eta = eta_init
    best = h.copy()
    best_score = compute_c5(h)
    no_improve = 0
    
    while time.time() - start < time_limit:
        grad = lse_gradient(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        direction = -grad / gn  # normalized descent direction
        
        # Backtracking line search
        found_step = False
        for factor in [eta, eta * 0.5, eta * 0.25, eta * 0.1, eta * 2.0, eta * 4.0]:
            h_trial = h + factor * direction
            h_trial = project_feasible(h_trial, target)
            s = compute_c5(h_trial)
            if s < best_score:
                h = h_trial
                best = h.copy()
                best_score = s
                no_improve = 0
                eta = min(eta * 1.05, 0.05)
                found_step = True
                break
            elif s < compute_c5(h) + 1e-10:
                # Accept even if not better than best (for exploration)
                pass
        
        if not found_step:
            no_improve += 1
            if no_improve > 30:
                alpha *= alpha_growth
                eta *= 0.8
                no_improve = 0
                if alpha > 5000:
                    break
    
    return best, best_score

def pair_cd(h, target, time_limit, seed=42, deltas=None):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    if deltas is None:
        deltas = [0.002, 0.005, 0.01, 0.02, 0.05]
    best = h.copy()
    best_score = compute_c5(h)
    stall = 0
    
    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            vi = best[i] + d
            vj = best[j] - d
            if 0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0:
                trial = best.copy()
                trial[i] = vi
                trial[j] = vj
                s = compute_c5(trial)
                if s < best_score - 1e-15:
                    best = trial
                    best_score = s
                    stall = 0
                    break
            vi2 = best[i] - d
            vj2 = best[j] + d
            if 0.0 <= vi2 <= 1.0 and 0.0 <= vj2 <= 1.0:
                trial = best.copy()
                trial[i] = vi2
                trial[j] = vj2
                s = compute_c5(trial)
                if s < best_score - 1e-15:
                    best = trial
                    best_score = s
                    stall = 0
                    break
        else:
            stall += 1
            if stall > 500:
                # Shrink deltas
                deltas = [d * 0.7 for d in deltas]
                if max(deltas) < 1e-5:

                stall = 0
    
    return best, best_score

def slsqp_refine(h, target, time_limit):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac, 'args': (target,)}]
    
    alphas = [10, 30, 100, 300, 1000]
    n_alphas = len(alphas)
    
    for ai, alpha in enumerate(alphas):
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        time_per = remaining / (n_alphas - ai)
        maxiter = int(max(50, time_per * 3))
        
        try:
            res = minimize(lse_objective, best, args=(alpha,), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score

            vi2 = best[i] - d
            vj2 = best[j] + d
            if 0.0 <= vi2 <= 1.0 and 0.0 <= vj2 <= 1.0:
                trial = best.copy()
                trial[i] = vi2
                trial[j] = vj2
                s = compute_c5(trial)
                if s < best_score - 1e-15:
                    best = trial
                    best_score = s
                    stall = 0
                    break
        else:
            stall += 1
            if stall > 300:
                deltas = [d * 0.7 for d in deltas]
                if max(deltas) < 1e-5:
                    break
                stall = 0
    
    return best, best_score

def nudge_top_lags(h, target, time_limit, seed=42, top_m=3):
    """Target the top-m lags simultaneously."""
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    
    while time.time() - start < time_limit:
        corr = compute_corr(best)
        top_idx = np.argsort(corr)[-top_m:]
        shifts = top_idx - (n - 1)
        
        # Compute combined gradient for top lags
        grad = np.zeros(n)
        for s in shifts:
            s = int(s)
            if s == 0:
                grad += (1.0 - 2.0 * best)
            elif s > 0:
                grad[s:] += (1.0 - best[:n - s])
                grad[:n - s] -= best[s:]
            else:
                ss = -s
                grad[:n - ss] += (1.0 - best[ss:])
                grad[ss:] -= best[:n - ss]
        
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        # Find best index to decrease (most positive gradient) and increase (most negative)
        pos_idx = int(np.argmax(grad))
        neg_idx = int(np.argmin(grad))
        
        if grad[pos_idx] < 1e-8 and -grad[neg_idx] < 1e-8:
            break
        
        d = rng.uniform(0.001, 0.01)
        trial = best.copy()
        trial[pos_idx] = best[pos_idx] - d
        trial[neg_idx] = best[neg_idx] + d
        if 0.0 <= trial[pos_idx] <= 1.0 and 0.0 <= trial[neg_idx] <= 1.0:
            s = compute_c5(trial)
            if s < best_score:
                best_score = s
                best = trial
            else:
                for d2 in [d * 0.5, d * 0.1]:
                    t2 = best.copy()
                    t2[pos_idx] = best[pos_idx] - d2
                    t2[neg_idx] = best[neg_idx] + d2
                    if 0.0 <= t2[pos_idx] <= 1.0 and 0.0 <= t2[neg_idx] <= 1.0:
                        s2 = compute_c5(t2)
                        if s2 < best_score:
                            best_score = s2
                            best = t2
                            break
    
    return best, best_score

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)
    
    def tleft():
        return budget_s - 20.0 - (time.time() - start_time)
    
    # Phase 1: Gradient descent with exact LSE gradient
    if tleft() > 280:
        h1, s1 = gradient_descent_phase(h, target, time_limit=280, alpha_init=15.0, eta_init=0.015, alpha_growth=1.15)
        if s1 < best_score:
            best_score = s1
            best_h = h1
    
    # Phase 2: Structured pair coordinate descent
    if tleft() > 250:
        h2, s2 = structured_pair_cd(best_h, target, time_limit=250, seed=seed)
        if s2 < best_score:
            best_score = s2
            best_h = h2
    
    # Phase 3: Nudge top lags
    if tleft() > 200:
        h3, s3 = nudge_top_lags(best_h, target, time_limit=200, seed=seed, top_m=3)
        if s3 < best_score:
            best_score = s3
            best_h = h3
    
    # Phase 4: SLSQP refinement
    if tleft() > 200:
        h4, s4 = slsqp_refine(best_h, target, time_limit=200)
        if s4 < best_score:
            best_score = s4
            best_h = h4
    
    # Phase 5: Fine pair CD
    if tleft() > 200:
        h5, s5 = structured_pair_cd(best_h, target, time_limit=200, seed=seed + 1)
