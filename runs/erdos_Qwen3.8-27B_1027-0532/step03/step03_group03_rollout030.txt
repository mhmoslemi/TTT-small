            lr *= 0.98

def coordinate_polish(h, target_sum, n_iter=1000, eps=0.005):
    """Hill-climbing coordinate descent on the top lag."""
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for it in range(n_iter):
        corr = overlap_corrs(h_cur)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        
        if lag >= 0 and lag < n:
            n_active = n - lag
            contribs = h_cur[lag:lag+n_active] * (1.0 - h_cur[:n_active])
        elif lag < 0 and -lag < n:
            m = -lag
            n_active = n - m
            contribs = h_cur[:n_active] * (1.0 - h_cur[m:m+n_active])
        else:
            continue
        
        # Find worst contributors
        n_worst = min(20, n_active)
        worst_idx = np.argpartition(contribs, -n_worst)[-n_worst:]
        
        improved = False
        for wi in worst_idx:
            if lag >= 0:
                i = wi
                j = wi + lag
                # Move: decrease h[j], increase h[i]
                h_try = h_cur.copy()
                h_try[j] -= eps
                h_try[i] += eps
                if h_try[j] < 0 or h_try[i] > 1:
                    continue
                s = compute_c5(h_try)
                if s < best_score - 1e-12:
                    best_score = s
                    h_cur = h_try
                    improved = True
                    break
            else:
                m = -lag
                i = wi + m
                j = wi
                h_try = h_cur.copy()
                h_try[i] -= eps
                h_try[j] += eps
                if h_try[i] < 0 or h_try[j] > 1:
                    continue
                s = compute_c5(h_try)
                if s < best_score - 1e-12:
                    best_score = s
                    h_cur = h_try
                    improved = True
                    break
        
        if not improved:
            # Try smaller eps
            eps_half = eps * 0.5
            if eps_half < 1e-4:
                break
            # Try reversing the move direction for a few
            for wi in worst_idx[:5]:
                if lag >= 0:
                    i = wi
                    j = wi + lag
                    h_try = h_cur.copy()
                    h_try[j] += eps_half
                    h_try[i] -= eps_half
                    if h_try[j] > 1 or h_try[i] < 0:
                        continue
                    s = compute_c5(h_try)
                    if s < best_score - 1e-12:
                        best_score = s
                        h_cur = h_try
                        improved = True
                        break
                else:
                    m = -lag
                    i = wi + m
                    j = wi
                    h_try = h_cur.copy()
                    h_try[i] += eps_half
                    h_try[j] -= eps_half
                    if h_try[i] > 1 or h_try[j] < 0:
                        continue
                    s = compute_c5(h_try)
                    if s < best_score - 1e-12:
                        best_score = s
                        h_cur = h_try
                        improved = True
                        break
            
            if not improved:
                eps = eps_half
    
    return h_cur, best_score

def random_restart_polish(h, target_sum, n_trials, scale, seed):

    rng = np.random.RandomState(seed)

        if s < best_score:
            best_score = s
            best_h = h_try.copy()
    
    return best_h, best_score

    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        space = 1.0 - h[idx]
        total_space = np.sum(space)
        if total_space < 1e-15:
            return h
        h[idx] += diff * space / total_space
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        weights = h[idx]
        total_w = np.sum(weights)
        if total_w < 1e-15:
            return h
        h[idx] += diff * weights / total_w
    return np.clip(h, 0.0, 1.0)

    n = len(h)

            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if m < n:

            grad[m:] -= h[:n - m]
    return grad

    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)

        h_try = best_h + rng.randn(n_init) * 0.004
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                        max_iter=3000, lr_init=0.02, topk=3,
                                        alpha=200, noise_prob=0.0, noise_scale=0.0,
                                        patience=400, momentum=0.92, seed=seed + 5000)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 150:
        h_polish, score = coordinate_polish(best_h, target_init, n_iter=2000, eps_init=0.003)
        if score < best_score:
            best_score = score
            best_h = h_polish

    if _remaining() > 200:
        h_ref = slsqp_refine(best_h, target_init, [500, 1000, 2000, 5000], 1500, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 100:
        h_polish, score = coordinate_polish(best_h, target_init, n_iter=1500, eps_init=0.002)
        if score < best_score:
            best_score = score
            best_h = h_polish

    if _remaining() > 80:
        h_ref = slsqp_refine(best_h, target_init, [1000, 3000, 8000], 1200, 1e-15)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if _remaining() > 50:
        h_polish, score = coordinate_polish(best_h, target_init, n_iter=800, eps_init=0.001)
        if score < best_score:
            best_score = score
            best_h = h_polish

    h_final = project_feasible(best_h, target_init)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
