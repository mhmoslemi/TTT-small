    #   sum(h) = target (equality)

Let me also think about the convergence of the alternating scheme. The scheme is:
1. Given h, set g = 1-h
2. Solve LP: min t s.t. for all k, (g_k)^T h <= t, sum(h) = target, 0 <= h <= 1
3. Update h

This is a form of block coordinate descent on the non-convex problem min_h max_k (h * (1-h)_k). The convergence is not guaranteed to be fast, but in practice it works well.

One issue: the LP might have multiple optimal solutions, and the one found might not be the one that best leads to improvement. We could address this by adding a small regularization term that biases towards the previous h:
min t + ε * ||h - h_old||^2

But this makes it a QP, not an LP. Alternatively, we can add a small linear term: min t + ε * (h_old^T h) which biases towards h_old.

    start = time.time()
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    iter_count = 0
    stagnation = 0
    current_bias = bias_weight

    while time.time() - start < time_limit:
        g = 1.0 - h
        h_lp, t_lp = solve_overlap_lp(g, n, target, h, current_bias)
        if h_lp is None:
            break
        h_lp = project_feasible(h_lp, target)
        c5_new = compute_c5(h_lp)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_lp.copy()
            stagnation = 0
        else:
            stagnation += 1
        h = (1.0 - damping) * h + damping * h_lp
        h = project_feasible(h, target)
        iter_count += 1
        if iter_count % 200 == 0:
            current_bias = max(1e-6, current_bias * 0.7)
        if stagnation > 100:
            damping = max(0.2, damping * 0.9)
            stagnation = 0

    return best_h, best_c5, iter_count


def smooth_max_obj_fn(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


_global_n = 0
_global_target = 0.0


def _cons_fun(h):
    return np.sum(h) - _global_target


def _cons_jac(h):
    return np.ones(_global_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    best_score = compute_c5(h_cur)
    start = time.time()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(smooth_max_obj_fn, h_cur, args=(alpha, n),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Primary alternating LP with HiGHS at original resolution
    if tleft() > 250:
        lp_time = min(300, tleft() * 0.35)
        h_lp, c5_lp, iters = alternating_lp_hg(h_init, n, target,
                                                time_limit=lp_time,
                                                bias_weight=0.01,
                                                damping=0.6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Additional alternating LP runs with different parameters
    params_list = [
        (0.001, 0.8),
        (0.05, 0.4),
        (0.01, 0.3),
        (0.005, 0.7),
    ]
    for i, (bw, dm) in enumerate(params_list):
        if tleft() < 150:
            break
        lp_time = min(80, tleft() * 0.15)
        h_lp, c5_lp, iters = alternating_lp_hg(best_h, n, target,
                                                time_limit=lp_time,
                                                bias_weight=bw,
                                                damping=dm)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 3: Perturbed restarts
    for restart in range(4):
        if tleft() < 120:
            break
        noise_scale = 0.08 * (restart + 1)
        h_pert = best_h + rng.randn(n) * noise_scale
        h_pert = project_feasible(h_pert, target)
        lp_time = min(60, tleft() * 0.1)
        h_lp, c5_lp, iters = alternating_lp_hg(h_pert, n, target,
                                                time_limit=lp_time,
                                                bias_weight=0.005,
                                                damping=0.5)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 4: SLSQP refinement at original resolution
    if tleft() > 100:
        alphas = [50, 200, 500, 1000, 2000]
        h_ref, sc = slsqp_refine(best_h, n, target, alphas, 800, 1e-14,
                                  min(120, tleft() * 0.3))
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Higher resolution transfer and optimization
    if tleft() > 200:
        for n_try in [800, 1000]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(150, tleft() * 0.3)
            h_lp, c5_lp, iters = alternating_lp_hg(h_interp, n_try, target_try,
                                                    time_limit=lp_time,
                                                    bias_weight=0.005,
                                                    damping=0.6)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try
            else:
                best_h = h_interp
                n = n_try
                target = target_try

            # Second LP pass at higher resolution
            if tleft() > 80:
                lp_time2 = min(80, tleft() * 0.2)
                h_lp2, c5_lp2, _ = alternating_lp_hg(best_h, n_try, target_try,
                                                      time_limit=lp_time2,
                                                      bias_weight=0.001,
                                                      damping=0.5)
                if c5_lp2 < best_score:
                    best_score = c5_lp2
                    best_h = h_lp2

    # Phase 6: SLSQP at final resolution (if feasible)
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 60 and n_use <= 800:
        alphas = [200, 500, 1000, 2000]
        maxit = 400 if n_use > 600 else 600
        h_final_try, sc = slsqp_refine(best_h, n_use, target_use, alphas,
                                        maxit, 1e-14, min(80, tleft() * 0.5))
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 7: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alphas = [1000, 2000, 5000]
        maxit = 200 if n_use > 600 else 400
        h_polish, sc = slsqp_refine(best_h, n_use, target_use, alphas,
                                     maxit, 1e-15, min(50, tleft() * 0.6))
        if sc < best_score:
            best_score = sc
            best_h = h_polish

    # Final feasibility check
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
