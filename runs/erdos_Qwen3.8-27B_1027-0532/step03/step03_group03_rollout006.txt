    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    v = np.zeros(n)

    for it in range(max_iter):
        corr = overlap_corrs(h)
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        w = np.exp(alpha * (top_vals - np.max(top_vals)))
        wsum = float(np.sum(w))
        if wsum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / wsum
        grad = weighted_grad(h, top_idx, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.015, 5.0)
        else:
            stagnation += 1
            lr *= 0.985
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    return np.interp(x_new, x_old, h)


def binary_snap(h, target):
    h = h.copy()
    h[h < 0.06] = 0.0
    h[h > 0.94] = 1.0
    return project_feasible(h, target)


def multi_alpha_slsqp(h, target, alphas, max_iters_each):
    h_cur = h.copy()
    sc_cur = compute_c5(h_cur)
    for alpha in alphas:
        h_new, sc = slsqp_refine(h_cur, target, alpha, max_iters_each)
        if sc < sc_cur:
            sc_cur = sc
            h_cur = h_new
    return h_cur, sc_cur


def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n0 = len(h_init)
    t0 = n0 / 2.0

    best_h = project_feasible(h_init, t0)
    best_score = compute_c5(best_h)

    alphas_coarse = [10, 25, 60, 120, 250, 500, 1000]
    for i, alpha in enumerate(alphas_coarse):
        if _remaining() < 45:
            break
        iters = 500 if i < 3 else 350
        h_new, sc = slsqp_refine(best_h, t0, alpha, iters)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    subgrad_cfgs = [
        (3500, 0.10, 5, 30, 0.04, 0.012, 300, 0.90),
        (3000, 0.06, 8, 60, 0.02, 0.006, 350, 0.88),
        (2500, 0.18, 3, 12, 0.07, 0.025, 220, 0.92),
        (3500, 0.08, 5, 35, 0.03, 0.008, 300, 0.90),
        (3000, 0.14, 4, 20, 0.05, 0.018, 250, 0.91),
        (2500, 0.04, 10, 80, 0.01, 0.004, 350, 0.85),
    ]
    for i, cfg in enumerate(subgrad_cfgs):
        if _remaining() < 65:
            break
        noise_scale_start = 0.012 if i % 2 == 0 else 0.008
        h_try = best_h + rng.randn(n0) * noise_scale_start
        h_try = project_feasible(h_try, t0)
        h_new, sc = subgrad_optimize(h_try, t0, cfg[0], cfg[1], cfg[2],
                                      cfg[3], cfg[4], cfg[5], cfg[6], cfg[7],
                                      seed + i * 17)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if _remaining() > 60:
        for alpha in [200, 500, 1000]:
            if _remaining() < 35:
                break
            h_new, sc = slsqp_refine(best_h, t0, alpha, 300)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    for n_new in [600, 800, 1000, 1200]:
        if _remaining() < 90:
            break
        t_new = n_new / 2.0
        h_up = project_feasible(upsample(best_h, n_new), t_new)
        alphas_up = [20, 50, 120, 300, 600]
        for alpha in alphas_up:
            if _remaining() < 45:
                break
            iters = 350 if alpha < 200 else 250
            h_new, sc = slsqp_refine(h_up, t_new, alpha, iters)
            if sc < best_score:
                best_score = sc
                best_h = h_new
                h_up = h_new

        if _remaining() > 50:
            h_try = h_up + rng.randn(n_new) * 0.006
            h_try = project_feasible(h_try, t_new)
            h_new, sc = subgrad_optimize(h_try, t_new, 1200, 0.05, 5,
                                          30, 0.02, 0.008, 150, 0.90,
                                          seed + n_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    if _remaining() > 70:
        n_cur = len(best_h)
        t_cur = n_cur / 2.0
        h_snap = binary_snap(best_h, t_cur)
        sc_snap = compute_c5(h_snap)
        if sc_snap < best_score:
            best_score = sc_snap
            best_h = h_snap
        for alpha in [300, 800]:
            if _remaining() < 30:
                break
            h_new, sc = slsqp_refine(best_h, t_cur, alpha, 250)
            if sc < best_score:
                best_score = sc
                best_h = h_new

    n_fin = len(best_h)
    t_fin = n_fin / 2.0
    final_alphas = [800, 1500, 2500, 4000]
    for alpha in final_alphas:
        if _remaining() < 20:
            break
        h_new, sc = slsqp_refine(best_h, t_fin, alpha, 200)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    if _remaining() > 30:
        h_try = best_h + rng.randn(n_fin) * 0.004
        h_try = project_feasible(h_try, t_fin)
        h_new, sc = subgrad_optimize(h_try, t_fin, 1500, 0.04, 5,
                                      50, 0.0, 0.0, 200, 0.92, seed + 9999)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    best_h = project_feasible(best_h, t_fin)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
