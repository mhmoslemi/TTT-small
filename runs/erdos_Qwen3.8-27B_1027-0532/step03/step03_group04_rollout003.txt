    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_sl_n = 0
_sl_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def active_constraint_lp(h_init, n, target, max_iter, time_limit, n_active=40, damping_init=0.7):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        g = 1.0 - h

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k_idx in top_idx:
            k = int(k_idx) - (n - 1)
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=5000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 40:
            damping = max(0.08, damping * 0.8)
            stuck_count = 0
            if damping < 0.12:
                break

    return best_h, best_c5


def sl_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(corr))
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / _sl_n)


def sl_cons_fun(h):
    return np.sum(h) - _sl_target


def sl_cons_jac(h):
    return np.ones(_sl_n)


def slsqp_optimize(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    global _sl_n, _sl_target
    _sl_n = n
    _sl_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sl_cons_fun, 'jac': sl_cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(sl_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
