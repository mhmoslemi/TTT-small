def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-4, solver='SCS'):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)

                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:

    return best_h, best_c5

But we can do better. The gradient of the smooth max is:
∇f_α(h) = ∑_k p_k ∇f_k(h)

where p_k = softmax(α * f_k). And ∇f_k is a "shifted" version of (1-h) and h.

Specifically, if we define:
A_k = the vector with (A_k)[i] = (1-h[i+k]) if valid, 0 otherwise (this is a shifted/reversed version of 1-h)
B_k = the vector with (B_k)[i] = h[i-k] if valid, 0 otherwise

Then ∇f_k = dx * (A_k - B_k) and ∇f_α = dx * (∑ p_k A_k - ∑ p_k B_k)

Now, ∑ p_k A_k at position i = ∑_k p_k (1-h[i+k]) = ∑_k p_k - ∑_k p_k h[i+k] = 1 - ∑_k p_k h[i+k]

The sum ∑_k p_k h[i+k] is a convolution of p with h! Specifically, it's (p * h)[i] where * denotes appropriate cross-correlation.

Let me define: P = [p_{-n+1}, ..., p_{n-1}] (the weights indexed by lag).
Then ∑_{k} p_k h[i+k] = ∑_k P[k+n-1] * h[i+k]

This is a cross-correlation of P and h. We can compute it with np.correlate or FFT.

Similarly, ∑_k p_k h[i-k] is also a correlation (with P reversed).

So:
∇f_α(h) = dx * [(1 - corr(P, h, i)) - corr(rev(P), h, i)]

where corr(P, h, i) = ∑_k P[k] h[i+k] for valid indices.

Let me verify with a simple case. If P is a delta at k=0, then:
corr(P, h, i) = h[i]
corr(rev(P), h, i) = h[i]
∇f = dx * [(1-h[i]) - h[i]] = dx * (1-2h[i])

And f_0 = ∑ h[i](1-h[i]) dx, ∂f_0/∂h[i] = dx(1-2h[i]). ✓

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_smooth_max_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = np.clip(alpha * (corr - max_corr), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    smooth_val = max_corr + np.log(total) / alpha
    grad = np.zeros(n)
    center = n - 1
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        k = k_idx - center
        wd = w * dx
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += wd * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wd * h[lo2 - k:hi2 - k]
    return smooth_val, grad


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init, reg):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_damp = damping_init
    cur_reg = reg
    stuck = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        cons = [cp.sum(h_var) == target, h_var >= 0, h_var <= 1]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi - lo > 0:
                c = np.zeros(n)
                c[lo:hi] = g[lo - k:hi - k] * dx
                cons.append(c @ h_var <= t_var)
        prob = cp.Problem(cp.Minimize(t_var + cur_reg * cp.sum_squares(h_var - 0.5)), cons)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-9, verbose=False)
        except Exception:
            break
        if h_var.value is None:
            break
        h_new = np.clip(h_var.value.ravel().astype(float), 0.0, 1.0)
        h = (1.0 - cur_damp) * h + cur_damp * h_new
        h = project_feasible(h, target)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 500 == 0 and cur_reg > 1e-9:
            cur_reg *= 0.5
    return best_h, best_c5

    val, grad = compute_smooth_max_grad(xc, alpha)
    s = np.sum(xc) - target
    pen = 1e6 * s * s
    pgrad = np.full(n, 2e6 * s)
    return val + pen, grad + pgrad


def lbfgs_optimize(h_start, n, target, time_limit, alphas):

    for alpha in alphas:

                           jac=True, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 400, 'ftol': 1e-20, 'gtol': 1e-12, 'maxls': 50})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:

        except Exception:
