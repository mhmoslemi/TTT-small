def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    """Project y onto {x: 0<=x<=1, sum(x)=S} using bisection on threshold."""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:

import numpy as np
from scipy.optimize import minimize_scalar, minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _c5_raw(h):
    """Return raw max correlation (without dx scaling)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return _lse_max(corr, alpha) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:

                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _pair_descent(h, n, target_sum, deadline, rng, max_rounds=500, min_improve=1e-14):
    """Iterative pair coordinate descent. Preserves sum constraint exactly."""
    h = h.copy()
    best_c5 = _c5_raw(h)
    stall = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        # Randomize pair order
        indices = list(range(n))
        rng.shuffle(indices)
        
        improved = False
        for ai in range(n - 1):
            i = indices[ai]
            for j in indices[ai+1:]:
                s = h[i] + h[j]
                t_lo = max(0.0, s - 1.0)
                t_hi = min(1.0, s)
                if t_hi - t_lo < 1e-12:

                # 1D optimization
                orig_i = h[i]
                orig_j = h[j]
                
                def obj_1d(t):
                    h[i] = t
                    h[j] = s - t
                    return _c5_raw(h)
                
                try:
                    res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                         options={'xatol': 1e-10})
                    if res.fun < best_c5 - min_improve:
                        h[i] = res.x
                        h[j] = s - res.x
                        best_c5 = res.fun
                        improved = True
                    else:
                        h[i] = orig_i
                        h[j] = orig_j
                except Exception:
                    h[i] = orig_i
                    h[j] = orig_j
        
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    
    h = _fix_sum(h, target_sum)
    return h, _c5_raw(h) * (2.0 / n)

            d = c

            h[i] = c
            h[j] = s - c

            fd = _c5_raw(h)

    t_opt = c if fc <= fd else d
    h[i] = t_opt
    h[j] = s - t_opt
    final_c = _c5_raw(h)
    
    if final_c >= best_c:
        h[i] = orig_i
        h[j] = orig_j
        return h, 0.0
    
    return h, best_c - final_c
