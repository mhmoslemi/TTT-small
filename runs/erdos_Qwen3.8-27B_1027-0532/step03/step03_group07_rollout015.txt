    n = len(h)
    n_corr = 2 * n - 1
    delta = np.zeros(n_corr)
    lags = np.arange(n_corr) - (n - 1)
    
    for idx in range(n_corr):
        lag = lags[idx]
        d = 0.0
        if lag == 0:
            # Term h[j]*(1-h[j]) changes by (v_new - v_old) - (v_new^2 - v_old^2)
            # But only if 0 <= j < n (always true)
            d = (v_new - v_old) - (v_new*v_new - v_old*v_old)
        else:

            if 0 <= j + lag < n:

def _coord_optimize(h, j, corr, trial_values):
    """Find best v for coordinate j given current h and corr."""
    n = len(h)
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)
    v_old = h[j]
    
    # Compute deltas for all trial values at once
    # delta shape: (len(trial_values), n_corr)
    dv = trial_values - v_old  # shape: (T,)
    
    mask_nonzero = (lags != 0)
    
    # For non-zero lags: delta[t, idx] = dv[t] * (1-h[j+lag]) for valid first
    #                              + dv[t] * (-h[j-lag]) for valid second
    valid_first = (j + lags >= 0) & (j + lags < n) & mask_nonzero
    valid_second = (j - lags >= 0) & (j - lags < n) & mask_nonzero
    
    delta = np.zeros((len(trial_values), n_corr))
    
    if np.any(valid_first):
        coef_first = (1.0 - h[j + lags[valid_first]])  # shape: (n_valid,)
        delta[:, valid_first] = dv[:, None] * coef_first[None, :]
    if np.any(valid_second):
        coef_second = -h[j - lags[valid_second]]  # shape: (n_valid,)
        delta[:, valid_second] = delta[:, valid_second] + dv[:, None] * coef_second[None, :]
    
    # For lag = 0: exact
    delta[:, n-1] = trial_values * (1 - trial_values) - v_old * (1 - v_old)
    
    # New corr for each trial: corr + delta
    # new_corr shape: (T, n_corr)
    new_corr = corr[None, :] + delta
    # Objective: max over n_corr axis
    obj = np.max(new_corr, axis=1)
    
    best_idx = np.argmin(obj)
    return trial_values[best_idx], obj[best_idx]
