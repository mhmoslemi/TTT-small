```python
import numpy as np
from scipy.optimize import minimize, fmin_l_bfgs_b
import time

def project_onto_constraint(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def soft_max_corr(h, eps):
    """Compute soft max of correlations using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def soft_max_grad(h, eps):
    """Gradient of soft max of correlations."""
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def projected_gradient_method(h_init, eps, n_iters, lr, target_sum):
    """Projected gradient descent on soft max with projection onto constraint set."""
    h = h_init.copy()
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    for i in range(n_iters):
        grad = soft_max_grad(h, eps)
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c = compute_c5(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def l_bfgsb_optimize(h_init, eps, maxiter, target_sum):
    """L-BFGS-B optimization with penalty for sum constraint."""
    n = len(h_init)
    
    def obj(h):
        return soft_max_corr(h, eps)
    
    def grad(h):
        return soft_max_grad(h, eps)
    
    bounds = [(0.0, 1.0)] * n
    result = fmin_l_bfgs_b(obj, h_init, fprime=grad, args=(),
                            pgtol=1e-12, maxiter=maxiter, factr=1e12,
                            bounds=bounds)
    h_opt = project_onto_constraint(result[0], target_sum)
    return h_opt, compute_c5(h_opt)

def increment_corr(h, oi, zi):
    """Compute the change in correlation when h[oi]:=0 and h[zi]:=1.
    Returns the delta array of length 2n-1."""
    n = len(h)
    dx = 2.0 / n
    delta = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        s = k - (n - 1)
        d = 0.0
        # From h[oi]: 1 -> 0
        if s != 0 or True:  # always check
            # Term i=oi: h[oi+s]*(1-0) - h[oi+s]*(1-1) = h[oi+s]
            if 0 <= oi + s < n:

            # Term i=oi-s: 0*(1-h[oi-s]) - 1*(1-h[oi-s]) = -(1-h[oi-s])
            if s != 0 and 0 <= oi - s < n:

            elif s == 0:

                pass
        # From h[zi]: 0 -> 1

            # Term i=zi: h[zi+s]*(1-1) - h[zi+s]*(1-0) = -h[zi+s]
            if 0 <= zi + s < n:

            # Term i=zi-s: 1*(1-h[zi-s]) - 0*(1-h[zi-s]) = 1-h[zi-s]

                d += 1.0 - h[zi - s]
            elif s == 0:
                pass
        delta[k] = d * dx
    return delta
```

```python
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def soft_max_value(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def soft_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def projected_gradient(h0, eps, n_iters, target_sum, lr_init=0.1):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr_init
    f_prev = soft_max_value(h, eps)
    for i in range(n_iters):
        grad = soft_max_grad(h, eps)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        # Backtracking line search
        step = lr
        for _ in range(20):
            h_new = project_onto_constraint(h - step * grad, target_sum)
            f_new = soft_max_value(h_new, eps)
            if f_new < f_prev - 1e-4 * step * np.dot(grad, h - h_new):
                break
            step *= 0.5
        h = h_new
        f_prev = f_new
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Adaptive lr
        if i % 50 == 0 and i > 0:
            lr = step * 1.5
    return best_h, best_c
```

```python
    cmax = np.max(corr)
    # Find all k that achieve the max (within tolerance)

            grad[:m] -= h[s:]
        else:
            r = -s
            m = n - r

            grad[r:] -= h[:m]
```

```python
import numpy as np
from scipy.optimize import fmin_l_bfgs_b
import time


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def soft_max_value(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))


def soft_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad


def subgradient_step(h, target_sum, lr, tol=1e-10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    max_indices = np.where(corr >= cmax - tol)[0]
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in max_indices:
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += omh[:m]
            grad[:m] -= h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += omh[r:]
            grad[r:] -= h[:m]
    grad *= dx / len(max_indices)
    h_new = h - lr * grad
    return project_onto_constraint(h_new, target_sum)


def projected_gradient_cont(h0, eps_list, iters_per_eps, target_sum, start_time, deadline):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    momentum_np = np.zeros(n)
    beta = 0.7
    for eps in eps_list:
        if time.time() > deadline:
            break
        lr = 1.0 / (0.1 + np.linalg.norm(soft_max_grad(h, eps)))
        for it in range(iters_per_eps):
            if time.time() > deadline:
                break
            grad = soft_max_grad(h, eps)
            momentum_np = beta * momentum_np + (1 - beta) * grad
            h_new = h - lr * momentum_np
            h_new = project_onto_constraint(h_new, target_sum)
            c = compute_c5(h_new)
            if c < best_c:

                best_h = h_new.copy()
            h = h_new
            lr = lr * 0.995
    return best_h, best_c
```
