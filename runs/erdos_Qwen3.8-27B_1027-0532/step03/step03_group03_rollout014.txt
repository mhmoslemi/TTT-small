    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]

                grad[:n - lag] -= w * h[lag:]
        else:

import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    n = len(x)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * (2.0 / n)


def smooth_lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights) * (2.0 / n)


def run_subgradient_adaptive(h_init, target_sum, max_iter, lr_init, topk,
                             alpha_init, alpha_end, noise_prob, noise_scale,
                             patience, momentum, seed, sa_temp=0.0):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    burn_end = int(0.4 * max_iter)

    for it in range(max_iter):
        progress = it / max(max_iter - 1, 1)
        alpha = alpha_init + (alpha_end - alpha_init) * progress

        if sa_temp > 0.0:
            if it < burn_end:
                temp = sa_temp
            else:
                decay = 1.0 - (it - burn_end) / max(max_iter - burn_end, 1)
                temp = sa_temp * decay * decay
        else:
            temp = 0.0

        corr = overlap_corrs(h)
        top_indices = np.argpartition(corr, -topk)[-topk:]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.015, 2.0)
            h = h_new
        elif temp > 1e-8 and rng.random() < np.exp(min(-(score - best_score) / temp, 0.0)):
            h = h_new
            lr *= 0.995
            stagnation += 1
        else:
            stagnation += 1
            lr *= 0.99

                    h = h_try

        if stagnation > patience:
            break
    return best_h, best_score


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def coord_refine(h_start, target_sum, max_sweeps, seed):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    h = h_start.copy()
    best_score = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = False
        order = rng.permutation(n)
        for i in order:
            old_val = h[i]
            deltas = [0.02, -0.02, 0.05, -0.05, 0.01, -0.01]
            for d in deltas:
                new_val = np.clip(old_val + d, 0.0, 1.0)
                if abs(new_val - old_val) < 1e-9:
                    continue
                actual_d = new_val - old_val
                if actual_d > 0:
                    j_cands = np.where(h > 0.05)[0]
                else:
                    j_cands = np.where(h < 0.95)[0]
                if len(j_cands) == 0:
                    continue
                j = j_cands[rng.randint(len(j_cands))]
                h[i] = new_val
                h[j] = np.clip(h[j] - actual_d, 0.0, 1.0)
                score = compute_c5(h)
                if score < best_score - 1e-9:
                    best_score = score
                    improved = True
                else:
                    h[i] = old_val
                    h[j] += actual_d
                    h[j] = np.clip(h[j], 0.0, 1.0)
        if not improved and sweep >= 2:
            break
    return h, best_score
