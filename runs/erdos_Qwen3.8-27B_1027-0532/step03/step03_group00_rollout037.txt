def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    dx = 2.0 / n
    return float(np.max(compute_corr(h))) * dx

def project_feasible(h, target_sum):

    for _ in range(100):

        if abs(diff) < 1e-14:

        if diff > 0:

        cnt = mask.sum()

        h = h.copy()

c[k] = Σ_m h[m] - Σ_m h[m]*h[m+k]

The first sum: ∂/∂h[i] = 1
The second sum: terms involving h[i]:
- m = i: h[i] * h[i+k], derivative = h[i+k]
- m+k = i, i.e., m = i-k: h[i-k] * h[i], derivative = h[i-k]

So ∂c[k]/∂h[i] = 1 - h[i+k] - h[i-k] (with out-of-bounds = 0)

When we change h[i] by δ, the change in c[k] is:
Δc[k] = δ * (1 - h[i+k] - h[i-k])
