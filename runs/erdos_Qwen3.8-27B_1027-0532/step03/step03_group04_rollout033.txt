  So corr[m] = sum_i h[i+k] * (1-h[i]) for valid i.
  = sum_i (1-h[i]) * h[i+k]
  
  For k > 0: i ranges from 0 to n-1-k (so i+k <= n-1)
  corr = sum_{i=0}^{n-1-k} (1-h[i]) * h[i+k]
  
  For k < 0, j = -k > 0: i ranges from j to n-1 (so i-j >= 0)
  corr = sum_{i=j}^{n-1} (1-h[i]) * h[i-j]
  = sum_{i=0}^{n-1-j} (1-h[i+j]) * h[i]
  
  So in both cases, the form is:
  corr_k = sum (1-h[first]) * h[second] where |second - first| = |k|
  
  For k > 0: first = i, second = i+k
  For k < 0: first = i (which is i+j in the reindexing), second = i (which is i in the reindexing)
  
  Actually let me just use the formula:
  corr_k = sum over valid pairs (i, i+k) of h[i+k] * (1-h[i])
  
  For the gradient:
  d corr_k / d h[p]:
  - p appears in h[i+k] when i+k = p, i.e., i = p-k. Valid if 0 <= p-k and 0 <= p-k+k = p.
    Contribution: (1-h[p-k]) if 0 <= p-k <= n-1-k (for k>0) or appropriate range
  - p appears in (1-h[i]) when i = p. Valid if p-k is in range.
    Contribution: -h[p-k] if 0 <= p-k < n

  This is getting complicated. Let me just use the simpler approach: compute the gradient numerically or use a vectorized formula.

  Actually, for the subgradient method, I don't need the exact gradient. I can use:
  - Pick the worst shift k*
  - Compute grad of f_{k*} at h
  - Take a step in -grad direction

  For f_k(h) = sum_{valid i} h[i+k]*(1-h[i]):
  
  Using numpy, I can compute the gradient as:
  grad = np.zeros(n)
  if k >= 0:
      # h[i+k] terms: for i in [0, n-1-k], h[i+k] has coefficient (1-h[i])
      # So grad[i+k] += (1-h[i]) for i in [0, n-1-k]
      # i.e., grad[k:] += (1-h[:n-k])
      grad[k:] += (1 - h[:n-k])
      # (1-h[i]) terms: for i in [0, n-1-k], -h[i+k]
      # i.e., grad[:n-k] -= h[k:]
      grad[:n-k] -= h[k:]
  else:
      j = -k
      # h[i-j] terms: for i in [j, n-1], h[i-j] has coefficient (1-h[i])
      # i.e., grad[:n-j] += (1-h[j:])
      grad[:n-j] += (1 - h[j:])
      # (1-h[i]) terms: for i in [j, n-1], -h[i-j]
      # i.e., grad[j:] -= h[:n-j]
      grad[j:] -= h[:n-j]

  Then the actual gradient of dx * f_k is dx * grad.

  This is clean and vectorized! Let me use this.

Now, for the coordinate descent approach:

For a pair (i, j), I want to try changing h[i] by +δ and h[j] by -δ (maintaining sum).
The change in correlation at shift k is:
Δcorr_k = δ * [(1-h[i+k]) - h[j+k]] (if both i and i+k valid) 
        + δ * [-h[i-k] + h[j-k]] (if both i-k and i valid)
        ... similar for j with sign flipped

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_global_n = 0
_global_target = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def shift_gradient(h, k, n):
    """Compute gradient of corr_k w.r.t. h, where k is the shift."""
    grad = np.zeros(n)
    if k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    elif k < 0:
        j = -k
        grad[:n-j] += (1.0 - h[j:])
        grad[j:] -= h[:n-j]
    return grad

def multi_shift_subgrad(h, n, target, max_steps, time_limit, top_k=5):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_all_corr(h_cur)
        
        # Get top-k shifts
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        # Softmax weights
        weights = np.exp(corr[top_idx] * 100.0)
        weights /= np.sum(weights)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = int(idx) - n_half
            if k == 0:
                continue
            grad += w * shift_gradient(h_cur, k, n)
        
        grad *= dx
        
        # Try multiple step sizes (decreasing)
        found = False
        for lr_mult in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            base_lr = 0.05
            h_try = h_cur - base_lr * lr_mult * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                found = True
                break
        
        if not found:
            # Try larger step
            for lr_mult in [2.0, 4.0, 8.0]:
                base_lr = 0.05
                h_try = h_cur - base_lr * lr_mult * grad
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_try.copy()
                    h_cur = h_try
                    found = True
                    break
            if not found:
                # Perturb
                h_cur = h_cur + np.random.randn(n) * 0.03
                h_cur = project_feasible(h_cur, target)
    
    return best_h, best_c5

def coordinate_descent(h, n, target, max_iters, time_limit):
    """Pair-based coordinate descent."""
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(123)
    
    delta_options = np.array([0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5])
    
    no_improve = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Pick random pair
        i, j = rng.choice(n, size=2, replace=False)
        
        # Try various deltas
        a, b = h_cur[i], h_cur[j]
        
        best_delta = 0.0
        best_sc = best_c5
        
        for d in delta_options:
            # Try h[i] += d, h[j] -= d
            if a + d <= 1.0 + 1e-10 and b - d >= -1e-10:
                h_try = h_cur.copy()
                h_try[i] = min(1.0, max(0.0, a + d))
                h_try[j] = min(1.0, max(0.0, b - d))
                # Fix sum
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_sc:
                    best_sc = sc
                    best_delta = d
            
            # Try h[i] -= d, h[j] += d
            if a - d >= -1e-10 and b + d <= 1.0 + 1e-10:
                h_try = h_cur.copy()
                h_try[i] = min(1.0, max(0.0, a - d))
                h_try[j] = min(1.0, max(0.0, b + d))
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_sc:
                    best_sc = sc
                    best_delta = -d
        
        if best_delta != 0.0:
            h_try = h_cur.copy()
            h_try[i] = min(1.0, max(0.0, a + best_delta))
            h_try[j] = min(1.0, max(0.0, b - best_delta))
            h_try = project_feasible(h_try, target)
            h_cur = h_try
            best_c5 = compute_c5(h_cur)
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 2000:
                # Larger perturbation
                idx = rng.choice(n, size=10, replace=False)
                h_cur[idx] += rng.randn(10) * 0.1
                h_cur = project_feasible(h_cur, target)
                no_improve = 0
    
    return best_h, best_c5

def alternating_lp(h_init, n, target, max_iter, time_limit, damping=0.5):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in range(-(n-1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def slsqp_smooth(h, n, target, alphas, maxiter, ftol, time_limit):
    """SLSQP with log-sum-exp smooth approximation."""
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    start = time.time()
    
    bounds = [(0.0, 1.0)] * n
    
    def obj(x, alpha):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    h_cur = h.copy()
    for alpha in alphas:
        if time.time() - start > time_limit:
            break
        try:
            res = minimize(obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    
    return h_cur

1. Start with a set of "active" shifts (initially the top 5)
2. Solve: min t s.t. corr_k <= t for k in active set, sum(h) = target, 0 <= h <= 1

4. If active set is "too large" (> n+1 constraints), remove the one with smallest slack

def remez_alternating(h_init, n, target, max_rounds, time_limit):
    """Remez-like alternating VP with active set management."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    max_active = 10  # Max number of active constraints
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        # Compute all correlations
        corr = compute_all_corr(h)
        n_half = n - 1
        
        # Select active set: top shifts + some near-max
        sorted_corr = np.sort(corr)[::-1]
        threshold = sorted_corr[min(max_active-1, len(sorted_corr)-1)] * 0.995
        
        active_k = []
        for m in range(2*n-1):
            if corr[m] >= threshold * 0.95:
                k = m - n_half
                active_k.append(k)
        
        # Limit active set size
        if len(active_k) > max_active:
            # Keep only the top ones
            active_corr = [(corr[m], m - n_half) for m in range(2*n-1)]
            active_corr.sort(reverse=True)
            active_k = [kc for _, kc in active_corr[:max_active]]
        
        # Solve LP: min t s.t. sum g[i-k]*h[i]*dx <= t for k in active_k
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in active_k:
            if k == 0:
                continue  # corr_0 = 0 always
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, verbose=False)
        except Exception:
            pass
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        
        # Damped update
        for damping in [1.0, 0.7, 0.5, 0.3]:
            h_try = (1-damping) * h + damping * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                break
        else:
            # Try undamped
            h_try = project_feasible(h_new, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5

Now, let me also think about the **p-norm smooth approximation** as an alternative to log-sum-exp:

f_p(h) = (sum_k corr_k^p * dx^p)^(1/p)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h.copy()
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h

def shift_grad(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k > 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    elif k < 0:
        j = -k
        grad[:n-j] += (1.0 - h[j:])
        grad[j:] -= h[:n-j]
    return grad

def remez_lp(h_init, n, target, max_rounds, time_limit, max_active=8):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n_half = n - 1
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = compute_all_corr(h)
        
        # Active set: shifts with corr >= 0.99 * max
        c_max = np.max(corr)
        active_m = np.where(corr >= 0.99 * c_max)[0]
        
        # Limit to max_active
        if len(active_m) > max_active:
            sorted_active = np.argsort(corr[active_m])[::-1][:max_active]
            active_m = active_m[sorted_active]
        
        active_k = [int(m) - n_half for m in active_m]
        
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]
        
        for k in active_k:
            if k == 0:
                continue
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo-k:hi-k] * dx
                constraints.append(coeff @ h_var <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        
        # Try different dampings
        improved = False
        for damping in [1.0, 0.8, 0.5, 0.3]:
            h_try = (1.0 - damping) * h + damping * h_new
            h_try = project_feasible(h_try, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
                improved = True
                break
        
        if not improved:
            # Try without damping
            h_try = project_feasible(h_new, target)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h = h_try
    
    return best_h, best_c5

def subgrad_multi(h, n, target, max_steps, time_limit, top_k=5):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    rng = np.random.RandomState(7)
    base_lr = 0.03
    
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        
        corr = compute_all_corr(h_cur)
        top_idx = np.argsort(corr)[::-1][:top_k]
        
        # Softmax weights (temperature controlled)
        temp = 50.0 + step * 0.1
        w = np.exp(corr[top_idx] * temp)
        w /= np.sum(w)
        
        grad = np.zeros(n)
        for m, weight in zip(top_idx, w):
            k = int(m) - n_half
            if k != 0:
                grad += weight * shift_grad(h_cur, k)
        
        grad *= dx
        
        found = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0, 4.0]:
            h_try = h_cur - base_lr * mult * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                found = True
                break
        
        if not found:
            h_cur = h_cur + rng.randn(n) * 0.02
            h_cur = project_feasible(h_cur, target)
        
    return best_h, best_c5

def coord_descent(h, n, target, max_iters, time_limit, seed=0):
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    rng = np.random.RandomState(seed)
    
    deltas = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        i, j = rng.choice(n, size=2, replace=False)
        a, b = h_cur[i], h_cur[j]
        
        best_sc = best_c5
        best_h_try = None
        
        for d in deltas:
            for direction in [1, -1]:
                na = a + direction * d
                nb = b - direction * d
                if na < -1e-10 or na > 1.0 + 1e-10:
                    continue
                if nb < -1e-10 or nb > 1.0 + 1e-10:
                    continue
                h_try = h_cur.copy()
                h_try[i] = max(0.0, min(1.0, na))
                h_try[j] = max(0.0, min(1.0, nb))
                h_try = project_feasible(h_try, target)
                sc = compute_c5(h_try)
                if sc < best_sc:
                    best_sc = sc
                    best_h_try = h_try
        
        if best_h_try is not None:
            h_cur = best_h_try
            best_c5 = best_sc
            best_h = h_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5000:
                idx = rng.choice(n, size=min(20, n), replace=False)
                h_cur[idx] += rng.randn(len(idx)) * 0.05
                h_cur = project_feasible(h_cur, target)
                no_improve = 0
    
    return best_h, best_c5

def targeted_cd(h, n, target, max_iters, time_limit, seed=1):

    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    rng = np.random.RandomState(seed)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_all_corr(h_cur)
        top_m = np.argsort(corr)[::-1][:3]
        
        best_sc = best_c5
        best_h_try = None
        
        for m in top_m:
            k = int(m) - n_half
            if k == 0:
                continue
            
            # Find top contributing pairs
            if k > 0:
                contrib = h_cur[:n-k] * (1.0 - h_cur[k:])
            else:
                j = -k
                contrib = h_cur[j:] * (1.0 - h_cur[:n-j])
            
            # Top pairs to modify
            top_pairs = np.argsort(contrib)[::-1][:5]
            
            for p in top_pairs:
                if k > 0:
                    i1, i2 = p, p + k
                else:
                    i1, i2 = p - k, p  # i1 = p + j, i2 = p
                    
                # Try to reduce contrib: decrease h[i1] or increase h[i2]
                for delta in [0.05, 0.1, 0.15]:
                    # Decrease h[i1], increase h[i2]
                    h_try = h_cur.copy()
                    h_try[i1] = max(0.0, h_try[i1] - delta)
                    h_try[i2] = min(1.0, h_try[i2] + delta)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < best_sc:
                        best_sc = sc
                        best_h_try = h_try.copy()
        
        if best_h_try is not None:
            h_cur = best_h_try
            best_c5 = best_sc
            best_h = h_cur.copy()
    
    return best_h, best_c5

        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def _cons_fun(x):
        return np.sum(x) - target
    
    def _cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    
    h_cur = h.copy()
    for alpha in alphas:
