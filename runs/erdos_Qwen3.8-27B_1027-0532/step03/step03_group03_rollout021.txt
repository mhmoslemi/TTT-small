def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        # Distribute to coordinates with most room
        rooms = 1.0 - h[idx]
        if rooms.sum() < diff:
            h[idx] = 1.0
            return h
        # Water-filling
        frac = diff / rooms.sum()
        h[idx] = np.minimum(h[idx] + frac * rooms, 1.0)
    else:

            return h
        h = h.copy()
        vols = h[idx]
        if vols.sum() < -diff:
            h[idx] = 0.0
            return h
        frac = (-diff) / vols.sum()
        h[idx] = np.maximum(h[idx] - frac * vols, 0.0)

    """Project x onto {y: sum(y) = target, 0 <= y <= 1}"""
    n = len(x)
    def project_given_lambda(lam):
        return np.clip(x - lam, 0.0, 1.0)
    
    lo, hi = -1.0, 2.0  # range for lambda
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = project_given_lambda(mid)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return project_given_lambda((lo + hi) / 2)

This is a 2D optimization but with the constraint t + s = const, so it's really 1D (parametrize by t, then s = h[i]+h[j]-t).

The correlation change would be:
corr[m] + (t - h[i]) * slope_i[m] + (s - h[j]) * slope_j[m]
= corr[m] + (t - h[i]) * slope_i[m] + (h[i]+h[j]-t - h[j]) * slope_j[m]
= corr[m] + (t - h[i]) * (slope_i[m] - slope_j[m])

So it's still a 1D problem! And we can solve it the same way.

This is better because it maintains the sum constraint exactly without projection. Let me use this.

from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_sum_box(x, target):
    n = len(x)
    def proj(lam):
        return np.clip(x - lam, 0.0, 1.0)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = proj(mid).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj((lo + hi) * 0.5)


def compute_slopes(h, i):
    n = len(h)
    m_idx = np.arange(2 * n - 1)
    slope = np.zeros(2 * n - 1)
    idx1 = i + m_idx - (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    slope[valid1] += 1.0 - h[idx1[valid1]]
    idx2 = i - m_idx + (n - 1)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slope[valid2] -= h[idx2[valid2]]
    return slope


def coord_descent(h, target, n_sweeps=50, verbose=False):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        any_change = False
        for i in range(n):
            old_val = h_cur[i]
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            cur_max = np.max(corr)
            
            slope = compute_slopes(h_cur, i)
            
            # Minimize max(corr + (t - old_val) * slope) over t in [0,1]
            best_t = old_val
            best_obj = cur_max
            
            # Coarse grid
            t_grid = np.linspace(0.0, 1.0, 21)
            objs = np.array([np.max(corr + (t - old_val) * slope) for t in t_grid])
            best_idx = np.argmin(objs)
            best_t = t_grid[best_idx]
            best_obj = objs[best_idx]
            
            # Fine grid around best
            lo = max(0.0, t_grid[best_idx] - 0.05)
            hi = min(1.0, t_grid[best_idx] + 0.05)
            t_fine = np.linspace(lo, hi, 21)
            for t in t_fine:
                obj = np.max(corr + (t - old_val) * slope)
                if obj < best_obj:
                    best_obj = obj
                    best_t = t
            
            if abs(best_t - old_val) > 1e-6:
                h_cur[i] = best_t
                h_cur = project_sum_box(h_cur, target)
                any_change = True
        
        new_score = compute_c5(h_cur)
        if new_score < best_score:
            best_score = new_score
        
        if not any_change and verbose:
            break
    
    return h_cur, best_score

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_sum_box(x, target):
    n = len(x)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def compute_slopes(h, i):
    n = len(h)
    m_idx = np.arange(2 * n - 1)
    slope = np.zeros(2 * n - 1)
    idx1 = i + m_idx - (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    slope[v1] += 1.0 - h[idx1[v1]]
    idx2 = i - m_idx + (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    slope[v2] -= h[idx2[v2]]
    return slope

def coord_descent(h, target, n_sweeps=100, max_time=None, start_time=None):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if max_time is not None and time.time() - start_time > max_time:
            break
        improved = False
        for i in range(n):
            old_val = h_cur[i]
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            cur_max = np.max(corr)
            slope = compute_slopes(h_cur, i)
            best_t = old_val
            best_obj = cur_max
            t_grid = np.linspace(0.0, 1.0, 33)
            for t in t_grid:
                dt = t - old_val
                if abs(dt) < 1e-10:
                    continue
                obj = np.max(corr + dt * slope)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_t = t
            if best_t != old_val:
                t_lo = max(0.0, best_t - 0.03)
                t_hi = min(1.0, best_t + 0.03)
                for t in np.linspace(t_lo, t_hi, 25):
                    dt = t - old_val
                    if abs(dt) < 1e-10:
                        continue
                    obj = np.max(corr + dt * slope)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_t = t
            if abs(best_t - old_val) > 1e-8:
                h_cur[i] = best_t
                h_cur = project_sum_box(h_cur, target)
                improved = True
        new_score = compute_c5(h_cur)
        if new_score < best_score:
            best_score = new_score
        if not improved:
            break
    return h_cur, best_score

def pair_coord_descent(h, target, n_sweeps=80, max_time=None, start_time=None):
    n = len(h)
    h_cur = h.copy()
    best_score = compute_c5(h_cur)
    for sweep in range(n_sweeps):
        if max_time is not None and time.time() - start_time > max_time:
            break
        improved = False
        for i in range(0, n - 1):
            j = i + 1
            s = h_cur[i] + h_cur[j]
            if s <= 1e-10 or s >= 2.0 - 1e-10:
                continue
            corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            cur_max = np.max(corr)
            slope_i = compute_slopes(h_cur, i)
            slope_j = compute_slopes(h_cur, j)
            combined = slope_i - slope_j
            t_lo = max(0.0, s - 1.0)
            t_hi = min(1.0, s)
            if t_hi - t_lo < 1e-8:
                continue
            best_t = h_cur[i]
            best_obj = cur_max
            t_grid = np.linspace(t_lo, t_hi, 33)
            for t in t_grid:
                dt = t - h_cur[i]
                if abs(dt) < 1e-10:
                    continue
                obj = np.max(corr + dt * combined)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_t = t
            if best_t != h_cur[i]:
                t_lo2 = max(t_lo, best_t - 0.02)
                t_hi2 = min(t_hi, best_t + 0.02)
                for t in np.linspace(t_lo2, t_hi2, 21):
                    dt = t - h_cur[i]
                    if abs(dt) < 1e-10:
                        continue
                    obj = np.max(corr + dt * combined)
                    if obj < best_obj - 1e-12:
                        best_obj = obj
                        best_t = t
            if abs(best_t - h_cur[i]) > 1e-8:
                h_cur[i] = best_t
                h_cur[j] = s - best_t
                improved = True
        new_score = compute_c5(h_cur)
        if new_score < best_score:
            best_score = new_score
        if not improved:
            break
    return h_cur, best_score

def _grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def smooth_lse_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def smooth_lse_grad(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    exp_s = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_s)
    weights = exp_s / total
    n = len(x)
    return _grad_weighted(x, np.arange(2 * n - 1), weights)

def _slsqp_cons_fun(x, target):
    return np.sum(x) - target

def slsqp_refine(h_start, target, alphas, maxiter, ftol):
    n = len(h_start)
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > _RUN_DEADLINE:
            break
        try:
            res = minimize(
                smooth_lse_obj, h_cur, args=(alpha,),
                jac=smooth_lse_grad, method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': lambda x: np.ones(len(x)), 'args': (target,)}],
                options={'maxiter': maxiter, 'ftol': ftol}
            )
            h_new = project_sum_box(res.x, target)
            sc = compute_c5(h_new)
            if sc < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    for it in range(max_iter):
        if time.time() > _RUN_DEADLINE:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        if topk >= n_corr:
            top_indices = np.arange(n_corr)
        else:
            top_indices = np.argpartition(corr, -topk)[-topk:]
        top_values = corr[top_indices]
        cmax = np.max(top_values)
        w = np.exp(alpha * (top_values - cmax))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones(len(top_indices)) / len(top_indices)
        else:
            w = w / w_sum
        grad = _grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_sum_box(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_sum_box(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score

def freq_smooth(h, keep_frac=0.7):
    n = len(h)
    H = np.fft.rfft(h)
    n_keep = max(2, int(len(H) * keep_frac))
    H[n_keep:] *= 0.0
    H[1:n_keep] *= np.linspace(1.0, 0.3, n_keep - 1)
    return np.fft.irfft(H, n=n)

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_sum_box(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_sum_box(h_out, target_dst)
