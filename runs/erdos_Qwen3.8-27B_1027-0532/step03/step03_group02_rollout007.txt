def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def enforce_symmetry(h):
    n = len(h)
    return 0.5 * (h + 1.0 - h[::-1])

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def project_sym(h, target_sum):

def coord_descend_sym(h, target_sum, max_time, rng, patience=20, max_noise=0.05):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)

    corr = np.correlate(h, 1.0 - h, mode='full').copy()

    elapsed_start = time.time()

    while time.time() - elapsed_start < max_time:
        for j in range(half_n):
            old_val = h[j]
            js = j - s_arr
            js_plus = j + s_arr
            A = np.zeros(2 * n - 1)
            mask1 = (js >= 0) & (js < n)
            mask2 = (js_plus >= 0) & (js_plus < n)
            A[mask1] += (1.0 - h[js[mask1]])
            A[mask2] -= h[js_plus[mask2]]

            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(bisection_iters):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-10:
                    hi = mid
                elif grad < -1e-10:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
