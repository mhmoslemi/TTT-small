def slsqp_minimax(h_init, K, time_limit, start_time):
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # State: [h(0), h(1), ..., h(n-1), t]
    # Objective: minimize t
    # Constraints:
    #   C(s) <= t for top K active lags (recomputed iteratively)
    #   sum(h) = target_sum
    #   0 <= h[i] <= 1
    
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    corr0 = compute_corr(h0)
    active_lags = np.argsort(corr0)[-K:]
    
    def objective(x):
        return x[-1]  # minimize t
    
    def objective_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    # We need to re-solve with updated active lags
    h_cur = h0.copy()
    t_cur = float(np.max(compute_corr(h0)))
    best_h = h0.copy()
    best_c5 = t_cur
    
    for outer in range(20):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr(h_cur)
        active_lags = np.argsort(corr)[-K:]
        
        def constraints_fun(x):
            h_x = x[:-1]
            t_x = x[-1]
            c = compute_corr(h_x)
            return c[active_lags] - t_x
        
        def constraints_jac(x):
            h_x = x[:-1]
            n_x = len(h_x)
            j = np.zeros((K, n_x + 1))
            for i, lag in enumerate(active_lags):
                # Gradient of C(lag) w.r.t. h
                s = int(lag - (n_x - 1))
                # C(s) = dx * sum_i h[i]*(1-h[i+s])
                # dC(s)/dh[j] = dx * (1-h[j+s]) if j+s valid, -dx*h[j-s] if j-s valid
                if s >= 0:
                    # h[i] term: i from 0 to n-1-s
                    j[0:n-s, i] = dx * (1.0 - h_x[s:n])
                    # h[i+s] term: contributes to h[j] where j = i+s, i from 0 to n-1-s
                    # So h[j] for j from s to n-1: d/dh[j] of -h[i]*h[i+s] = -h[i] where i=j-s
                    j[s:n, i] += -dx * h_x[0:n-s]
                else:
                    r = -s
                    # h[i] term: i from r to n-1
                    j[r:n, i] = dx * (1.0 - h_x[0:n-r])
                    # h[i+s] = h[i-r] term: for i from r to n-1, h[i-r] is affected
                    # d/dh[j] where j = i-r, i from r to n-1: j from 0 to n-1-r
                    j[0:n-r, i] += -dx * h_x[r:n]
                # dt/dx[i] for i < n: 0
                j[-1, i] = -1.0  # d(C(lag) - t)/dt = -1
            
            return j
        
        x0 = np.concatenate([h_cur, [t_cur]])
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        # Sum constraint
        def sum_fun(x):
            return float(np.sum(x[:-1]) - target_sum)
        def sum_jac(x):
            g = np.zeros(len(x))
            g[:-1] = 1.0
            return g
        
        try:
            result = minimize(objective, x0, jac=objective_grad, method='SLSQP',
                             bounds=bounds,
                             constraints=[
                                {'type': 'ineq', 'fun': constraints_fun, 'jac': constraints_jac},
                                {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}
                             ],
                             options={'maxiter': 200, 'ftol': 1e-12})
            if result.success or result.x is not None:

                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h_cur = h_new.copy()
                    t_cur = c5_new
                else:
                    h_cur = result.x[:-1].copy()
                    t_cur = float(result.x[-1])
        except Exception:
            break
    
    return best_h, best_c5

                break

            cmax = np.max(corr)

                best_h = h_cur.copy()
            
            if cmax < 0.37:  # convergence threshold (adjustable)
                break

            active_shifts = (active_idx - (n - 1)).astype(int)
            active_vals = corr[active_idx]

            zeros_db = db[zeros]

            max_over_k = np.max(new_C, axis=2)
            min_val = np.min(max_over_k)

                min_pos = np.unravel_index(np.argmin(max_over_k), max_over_k.shape)

                b = zeros[min_pos[1]]
                h_cur[a] = 0.0
                h_cur[b] = 1.0
            else:
                break
        
        # Phase: random kick (do several random swaps)
        kick_size = max(2, n // 30)
        for _ in range(kick_size):
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)

def proj_grad_topk(h, k, temp, lr_init, n_steps, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    stall_limit = 300

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        grad = topk_subgrad(h_cur, k, temp)
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def precompute_deltas(h, active_shifts, n):
    K = len(active_shifts)
    da = np.zeros((n, K))
    db = np.zeros((n, K))
    for j in range(K):
        s = int(active_shifts[j])
        if s >= 0:
            i_lo, i_hi = 0, n - s
            h_lo, h_hi = s, n
        else:
            i_lo, i_hi = -s, n
            h_lo, h_hi = 0, n + s
        if i_hi > i_lo:
            da[i_lo:i_hi, j] = (h[h_lo:h_hi] - 1.0)
            db[i_lo:i_hi, j] = (1.0 - h[h_lo:h_hi])
        if s >= 0:
            i_lo2, i_hi2 = s, n
            h_lo2, h_hi2 = 0, n - s
        else:
            i_lo2, i_hi2 = 0, n + s
            h_lo2, h_hi2 = -s, n
        if i_hi2 > i_lo2:
            da[i_lo2:i_hi2, j] += h[h_lo2:h_hi2]
            db[i_lo2:i_hi2, j] += -h[h_lo2:h_hi2]
    return da, db

    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
