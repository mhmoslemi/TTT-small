    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - h)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=30000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=30000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 20:
            damping = max(0.1, damping * 0.85)
            stuck_count = 0
            if damping < 0.12:
                damping = min(0.9, damping * 1.5)
        if it % 300 == 0 and current_reg > 1e-8:
            current_reg = max(1e-8, current_reg * 0.7)

    return best_h, best_c5


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def cons_fun(h):
    return np.sum(h) - _global_target


def cons_jac(h):
    return np.ones(_global_n)


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def slsqp_round(h_start, n, target, alphas, maxiter_per_alpha, ftol):
    global _global_n, _global_target
    _global_n = n
    _global_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter_per_alpha, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = (n - 1)
    idxs = np.arange(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr)) - n_half

        grad = np.zeros(n)
        k = k_star
        idx_valid1 = (0 <= idxs - k) & ((idxs - k) < n)
        idx_valid2 = (0 <= idxs + k) & ((idxs + k) < n)
        grad[idx_valid1] += (1.0 - h_cur[idxs[idx_valid1] - k])
        grad[idx_valid2] -= h_cur[idxs[idx_valid2] + k]
        grad *= dx

        improved = False
        for lr in [0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-13:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                improved = True
                break

        if not improved and step > 100:
            break

    return best_h, best_c5


def local_perturbation_search(h_init, n, target, max_tries, time_limit):
    start = time.time()
    rng_local = np.random.RandomState(12345)
    h_cur = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)

    for step in range(max_tries):
        if time.time() - start > time_limit:
            break

        n_pert = rng_local.choice([1, 2, 3, 5, 8, 13, 21])
        n_pert = min(n_pert, n)
        idxs = rng_local.choice(n, size=n_pert, replace=False)

        scale = rng_local.choice([0.1, 0.2, 0.3, 0.5, 0.7])
        h_try = h_cur.copy()
        h_try[idxs] += rng_local.randn(n_pert) * scale
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)

        if sc < best_c5 - 1e-13:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    global _global_n, _global_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - 30.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Alternating LP with proximal regularization
    if tleft() > 200:
        lp_time = min(280, tleft() * 0.25)
        h_lp, c5_lp = alternating_lp(h_init, n, target,
                                      max_iter=8000, time_limit=lp_time,
                                      damping_init=0.7, reg=1e-6)
        if c5_lp < best_score:
            best_score = c5_lp
            best_h = h_lp

    # Phase 2: Aggressive SLSQP with very high alpha
    if tleft() > 120:
        h_refined = slsqp_round(best_h, n, target,
                                 [30, 100, 300, 1000, 3000, 10000, 30000, 100000],
                                 1500, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 3: Subgradient refinement
    if tleft() > 80:
        h_sg, c5_sg = subgradient_improve(best_h, n, target,
                                           max_steps=20000, time_limit=min(60, tleft() * 0.3))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 4: SLSQP after subgradient
    if tleft() > 100:
        h_refined = slsqp_round(best_h, n, target,
                                 [200, 1000, 5000, 15000, 50000],
                                 1500, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 5: Local perturbation search
    if tleft() > 60:
        h_pert, c5_pert = local_perturbation_search(best_h, n, target,
                                                     max_tries=5000, time_limit=min(50, tleft() * 0.4))
        if c5_pert < best_score:
            best_score = c5_pert
            best_h = h_pert
        if tleft() > 40:
            h_refined = slsqp_round(best_h, n, target,
                                     [500, 2000, 8000, 30000],
                                     1000, 1e-15)
            sc = compute_c5(h_refined)
            if sc < best_score:
                best_score = sc
                best_h = h_refined

    # Phase 6: Perturbed SLSQP restarts
    for noise in [0.02, 0.05, 0.1]:
        if tleft() < 80:
            break
        h_pert = best_h + rng.randn(n) * noise
        h_pert = project_feasible(h_pert, target)
        h_refined = slsqp_round(h_pert, n, target,
                                 [100, 500, 2000, 8000, 30000],
                                 1200, 1e-15)
        sc = compute_c5(h_refined)
        if sc < best_score:
            best_score = sc
            best_h = h_refined

    # Phase 7: Try higher resolution
    if tleft() > 180:
        for n_try in [500, 700]:
            if tleft() < 120:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(80, tleft() * 0.25)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try,
                                          max_iter=3000, time_limit=lp_time,
                                          damping_init=0.7, reg=1e-6)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try
            if tleft() < 80:
                break
            n_use = len(best_h)
            target_use = n_use / 2.0
            h_refined = slsqp_round(best_h, n_use, target_use,
                                     [200, 1000, 5000, 20000],
                                     1000, 1e-15)
            sc = compute_c5(h_refined)
            if sc < best_score:
                best_score = sc
                best_h = h_refined

    # Phase 8: Final SLSQP polish with extreme alpha
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_final_try = slsqp_round(best_h, n_use, target_use,
                                   [10000, 50000, 200000],
                                   1000, 1e-15)
        sc = compute_c5(h_final_try)
        if sc < best_score:
            best_score = sc
            best_h = h_final_try

    # Phase 9: Final subgradient
    if tleft() > 20:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_improve(best_h, n_use, target_use,
                                           max_steps=10000, time_limit=min(30, tleft() * 0.6))
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Final feasibility check and return
    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
