def fast_binary_sa_v2(h_start, n_steps, t_start, t_end, rng, kick_prob=0.05, kick_size=3):
    """Binary SA with kick moves"""
    n = len(h_start)

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

    best_c5_val = current_max
    best_h = h.copy()
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    n_one = len(ones_list)
    n_zero = len(zeros_list)
    half = (n - 1)
    
    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Occasionally do a kick
        if rng.random() < kick_prob and t < 0.1:
            kick_n = int(rng.integers(1, kick_size + 1))
            kick_idx = rng.choice(n, size=kick_n, replace=False)
            # Flip random positions
            h[kick_idx] = 1.0 - h[kick_idx]
            # Re-project to maintain sum
            h = project_onto_constraint(h, n / 2.0)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            current_max = float(np.max(corr))
            ones_list = np.where(h == 1.0)[0].tolist()
            zeros_list = np.where(h == 0.0)[0].tolist()
            step += 1
            continue
        
        i = ones_list[rng.integers(n_one)]
        j = zeros_list[rng.integers(n_zero)]
        if i == j:
            step += 1
            continue
        
        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            step += 1
            continue
        
        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_max_obj(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x = x - np.max(x)
    return m + eps * np.log(np.sum(np.ex

p(x)))

def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    
    grad = np.zeros(n)
    half = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - half
        if s >= 0:
            m = n - s
            grad[:m] += wk * dx * omh[s:]
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]

            zeros = np.where(h == 0.0)[0].tolist()

import numpy as np
from scipy.optimize import minimize
import time


def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))


def smooth_max_obj(h, eps, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    x = (corr - m) / eps
    x = x - np.max(x)
    return m + eps * np.log(np.sum(np.exp(x)))


def smooth_max_grad(h, eps, n):
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x = x - np.max(x)
    ex = np.exp(x)
    w = ex / ex.sum()
    grad = np.zeros(n)
    half = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - half
        if s >= 0:
            m = n - s
            grad[:m] += wk * dx * omh[s:]
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad


def to_binary_f(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def penalty_lbfgs(h_init, eps, rho, n, maxiter=300):
    h = h_init.astype(np.float64).copy()

    def obj_pen(x):
        val = smooth_max_obj(x, eps, n)
        s = x.sum() - n / 2.0
        return val + rho * s * s

    def grad_pen(x):
        g = smooth_max_grad(x, eps, n)
        g += 2.0 * rho * (x.sum() - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        result = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        h_out = project_onto_constraint(result.x, n / 2.0)
        return h_out
    except Exception:
        return project_onto_constraint(h, n / 2.0)


def slsqp_solve(h_init, eps, n, maxiter=300):
    h0 = h_init.astype(np.float64).copy()

    def obj_s(x):
        return smooth_max_obj(x, eps, n)

    def grad_s(x):
        return smooth_max_grad(x, eps, n)

    def con_fun(x):
        return x.sum() - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        result = minimize(obj_s, h0, jac=grad_s, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_onto_constraint(result.x, n / 2.0)
    except Exception:
        return project_onto_constraint(h0, n / 2.0)


def binary_sa_v2(h_start, n_steps, t_start, t_end, rng, kick_period=0):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(corr.max())
    best_val = current_max
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    half = n - 1

    for step in range(n_steps):
        prog = step / max(n_steps - 1, 1)
        t = t_start * (t_end / t_start) ** prog

        if kick_period > 0 and step > 0 and step % kick_period == 0:
            nk = int(rng.integers(2, 6))
            kick_idx = rng.choice(n, size=nk, replace=False)
            h[kick_idx] = 1.0 - h[kick_idx]
            h = project_onto_constraint(h, n / 2.0)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            current_max = float(corr.max())
            ones = np.where(h == 1.0)[0].tolist()
            zeros = np.where(h == 0.0)[0].tolist()
            continue

        i = ones[int(rng.integers(n // 2))]
        j = zeros[int(rng.integers(n // 2))]

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)

        if s_min > s_max:
            h[i], h[j] = 0.0, 1.0
            oi = ones.index(i)
            ones[oi] = j
            oz = zeros.index(j)
            zeros[oz] = i
            continue

        k0 = s_min + half
        k1 = s_max + half
        s_range = np.arange(s_min, s_max + 1)
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])

        corr[k0:k1 + 1] += delta
        new_max = float(corr.max())
        accept = (new_max < current_max) or (t > 1e-10 and rng.random() < np.exp(-(new_max - current_max) / t))
        if accept:
            h[i], h[j] = 0.0, 1.0
            oi = ones.index(i)
            ones[oi] = j
            oz = zeros.index(j)
            zeros[oz] = i
            current_max = new_max
            if current_max < best_val:
                best_val = current_max
                best_h = h.copy()
        else:
            corr[k0:k1 + 1] -= delta

    return best_h, best_val
