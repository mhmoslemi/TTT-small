def compute_c5(h):

def compute_overlaps(h):

def project_feasible(h, target_sum):

    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)

def smooth_max_obj(h, alpha, n):

    cmax = corr.max()

def subgradient_step(h, n, dx, lr):

    grad[imk_valid] -= dx * h[i_mk[imk_valid]]
