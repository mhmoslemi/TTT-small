```python
        grad[-k:] += w * dx * (1.0 - h[:n+k])
        # j-k in [0,n) means j in [0, n+k) = [0, n-k')
        grad[:n+k] -= w * dx * h[-k:]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_gs = {}


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h = h.copy()
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx2 = np.where(h > 1e-12)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0.0, 1.0)
    return h


def _obj_grad(h):
    alpha = _gs['alpha']
    mu = _gs['mu']
    n = _gs['n']
    target = _gs['target']
    dx = 2.0 / n

    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    max_val = np.max(values)
    shifted = alpha * (values - max_val)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    f_smooth = max_val + np.log(sum_exp) / alpha

    s = np.sum(h) - target
    f_penalty = mu * s * s
    f = f_smooth + f_penalty

    weights = exp_vals / sum_exp
    n_half = n - 1
    grad = np.zeros(n)

    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        k = m - n_half
        if k >= 0:
            seg_len = n - k
            if seg_len > 0:
                grad[:seg_len] += w * dx * (1.0 - h[k:k + seg_len])
                grad[k:] -= w * dx * h[:seg_len]
        else:
            kp = -k
            seg_len = n - kp
            if seg_len > 0:
                grad[kp:] += w * dx * (1.0 - h[:seg_len])
                grad[:seg_len] -= w * dx * h[kp:]

    grad += 2.0 * mu * s

    return f, grad


def lbfgsb_optimize(h_start, n, target, alpha_schedule, max_iter_per_alpha, time_limit, mu=1000.0):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n

    for alpha in alpha_schedule:
        if time.time() - start > time_limit:
            break
        _gs['alpha'] = alpha
        _gs['mu'] = mu
        _gs['n'] = n
        _gs['target'] = target
        try:
            res = minimize(_obj_grad, h_cur, method='L-BFGS-B', jac=True,
                           bounds=bounds,
                           options={'maxiter': max_iter_per_alpha,
                                    'ftol': 1e-18, 'gtol': 1e-12,
                                    'maxls': 50})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
            h_cur = h_new
        except Exception:
            pass

    return best_h, best_score


def subgradient_step(h, n, target, lower_bound=0.37):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    values = corr * dx
    f_val = float(np.max(values))
    k_star = int(np.argmax(values)) - (n - 1)

    grad = np.zeros(n)
    k = k_star
    if k >= 0:
        seg_len = n - k
        if seg_len > 0:
            grad[:seg_len] += dx * (1.0 - h[k:k + seg_len])
            grad[k:] -= dx * h[:seg_len]
    else:
        kp = -k
        seg_len = n - kp
        if seg_len > 0:
            grad[kp:] += dx * (1.0 - h[:seg_len])
            grad[:seg_len] -= dx * h[kp:]

    g_norm_sq = np.dot(grad, grad)
    if g_norm_sq < 1e-20:
        return h, f_val, 0.0

    gap = f_val - lower_bound
    if gap < 1e-10:
        step = 0.01
    else:
        step = gap / (g_norm_sq + 1e-15)
        step = min(step, 0.5 / (np.linalg.norm(grad) + 1e-15))

    h_new = project_feasible(h - step * grad, target)
    f_new = compute_c5(h_new)
    return h_new, f_new, step


def subgradient_descent(h_start, n, target, max_steps, time_limit, lower_bound=0.37, decay=0.9999):
    start = time.time()
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    lb = lower_bound

    for step_i in range(max_steps):
        if time.time() - start > time_limit:
            break
        lb = max(lb, best_score - 0.01)

        h_new, f_new, _ = subgradient_step(h_cur, n, target, lb)
        if f_new < best_score - 1e-12:
            best_score = f_new
            best_h = h_new.copy()
            h_cur = h_new
        elif step_i % 100 == 0:
            h_cur = best_h.copy()

    return best_h, best_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
