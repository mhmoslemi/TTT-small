def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def slsqp_optimize(h_init, eps, maxiter, target_sum):
    n = len(h_init)
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target_sum), 
             'jac': lambda x: np.ones(len(x))}]
    
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return smooth_max(corr, eps)
    
    def grad(x):
        dx = 2.0 / n
        omh = 1.0 - x
        corr = np.correlate(x, omh, mode='full') * dx
        cmax = np.max(corr)
        x_shifted = (corr - cmax) / eps
        x_shifted -= np.max(x_shifted)
        ex = np.exp(x_shifted)
        sexp = np.sum(ex)
        w = ex / sexp
        g = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-14:
                continue
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                g[s:] += wk * dx * omh[:m]
                g[:m] -= wk * dx * x[s:]
            else:
                r = -s
                m = n - r
                g[:m] += wk * dx * omh[r:]
                g[r:] -= wk * dx * x[:m]
        return g
    
    try:
        result = minimize(obj, h0, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

        elif r == 1:
            h_temp = h_init.copy()
            nflip = max(3, n // 12)
            idx = rng.choice(n, size=nflip, replace=False)
            h_temp[idx] = 1.0 - h_temp[idx]
            replicas[r] = h_temp
        else:
            h_rand = np.zeros(n)
            h_rand[rng.choice(n, size=n // 2, replace=False)] = 1.0
            replicas[r] = h_rand
        temps[r] = t_high * (t_low / t_high) ** (r / max(n_replicas - 1, 1))
        scores[r] = compute_c5(replicas[r])
        if scores[r] < best_c5:
            best_c5 = scores[r]
            best_h = replicas[r].copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        for r in range(n_replicas):
            if time.time() > deadline:
                break
            t = temps[r]
            h = replicas[r]
            r_move = rng.random()
            if r_move < 0.4:
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                oi = ones[rng.integers(len(ones))]
                zi = zeros[rng.integers(len(zeros))]
                h[oi] = 0.0
                h[zi] = 1.0
                new_score = compute_c5(h)
                delta = new_score - scores[r]
                if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
                    scores[r] = new_score
                    if new_score < best_c5:
                        best_c5 = new_score
                        best_h = h.copy()
                else:
                    h[oi] = 1.0
                    h[zi] = 0.0
            elif r_move < 0.65:
                blen = int(rng.integers(2, max(3, n // 8) + 1))
                a = int(rng.integers(0, n - blen + 1))
                old = h[a:a+blen].copy()
                h[a:a+blen] = 1.0 - old
                new_score = compute_c5(h)
                delta = new_score - scores[r]
                if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
                    scores[r] = new_score
                    if new_score < best_c5:
                        best_c5 = new_score
                        best_h = h.copy()
                else:
                    h[a:a+blen] = old
            else:
                blen = int(rng.integers(2, max(3, n // 10) + 1))
                a = int(rng.integers(0, n - blen + 1))
                b = int(rng.integers(0, n - blen + 1))
                if abs(a - b) < blen:
                    continue
                ba = h[a:a+blen].copy()
                bb = h[b:b+blen].copy()
                h[a:a+blen] = bb
                h[b:b+blen] = ba
                new_score = compute_c5(h)
                delta = new_score - scores[r]
                if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
                    scores[r] = new_score
                    if new_score < best_c5:
                        best_c5 = new_score
                        best_h = h.copy()
                else:
                    h[a:a+blen] = ba
                    h[b:b+blen] = bb
        fr = rng.integers(n_replicas - 1)
        swap_accept = scores[fr] < scores[fr+1] or rng.random() < np.exp((scores[fr+1] - scores[fr]) / max(min(temps[fr], temps[fr+1]), 1e-12))
        if swap_accept:
            replicas[fr], replicas[fr+1] = replicas[fr+1].copy(), replicas[fr].copy()
            scores[fr], scores[fr+1] = scores[fr+1], scores[fr]
            temps[fr], temps[fr+1] = temps[fr+1], temps[fr]
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1.0
        rng.shuffle(h0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 120:
        p_c5, p_pattern = periodic_search(n0, 100, start_time)
        if p_pattern is not None and p_c5 < best_c5:
            best_c5 = p_c5
            best_h = np.tile(p_pattern, n0 // len(p_pattern))

    elapsed = time.time() - start_time
    h_bin = best_h
    if (h_bin != h_bin.astype(int)).any():
        h_bin = make_binary_from_continuous(h_bin)
    h_bin = h_bin.copy()
    if h_bin.sum() != target_sum:
        diffs = np.abs(h_bin - 0.5)
        order = np.argsort(diffs)
        if h_bin.sum() > target_sum:
            h_bin[order[:int(h_bin.sum() - target_sum)]] = 0.0
        else:
            h_bin[order[int(target_sum - h_bin.sum()):]] = 1.0

    if compute_c5(h_bin) < best_c5:
        best_c5 = compute_c5(h_bin)
        best_h = h_bin.copy()

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    sa_budget = int(remaining * 0.45)
    deadline_sa = start_time + sa_budget
    if sa_budget > 50:
        n_repl = 4
        n_sa_steps = 300000
        h_pt, c5_pt = parallel_tempering_sa(h_bin, n_repl, n_sa_steps, 0.01, 0.00005, rng, deadline_sa)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()

    elapsed = time.time() - start_time
    if (time.time() - start_time) < budget_s - 80:
        h_cont = best_h.astype(float) + 0.02 * rng.standard_normal(n0)
        h_cont = project_onto_constraint(h_cont, target_sum)
        eps_seq = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        for eps in eps_seq:
            if time.time() - start_time > budget_s - 60:
                break
            h_opt, c5_opt = slsqp_optimize(h_cont, eps, 150, target_sum)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cont = h_opt.copy()

    elapsed = time.time() - start_time
    n_restarts = 0
    while (budget_s - (time.time() - start_time)) > 120 and n_restarts < 30:
        strategy = n_restarts % 5
        if strategy == 0:
            h_start = best_h.copy()
            nflip = max(3, n0 // 12)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
        elif strategy == 1:
            p = int(rng.choice([6, 10, 12, 15, 18, 20, 30, 36]))
            if n0 % p == 0 and p % 2 == 0:
                perm = rng.permutation(p)
                pattern = np.zeros(p)
                pattern[perm[:p // 2]] = 1.0
                h_start = np.tile(pattern, n0 // p)
            else:
                h_start = np.zeros(n0)
                h_start[rng.choice(n0, size=n0 // 2, replace=False)] = 1.0
        elif strategy == 2:
            h_pert = best_h + 0.15 * rng.standard_normal(n0)
            h_start = make_binary_from_continuous(h_pert)
        elif strategy == 3:
            h_start = best_h.copy()
            nflip = max(5, n0 // 6)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_start[idx] = 1.0 - h_start[idx]
        else:
            h_start = np.zeros(n0)
            h_start[rng.choice(n0, size=n0 // 2, replace=False)] = 1.0
            h_cont_p = project_onto_constraint(h_start.astype(float) + 0.1 * rng.standard_normal(n0), target_sum)
            h_start = make_binary_from_continuous(h_cont_p)

        tl = min(budget_s - (time.time() - start_time) - 60, 80)
        if tl > 30:
            h_new, c5_new = sa_single(h_start, 200000, 0.008, 0.0001, rng, start_time + tl)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        n_restarts += 1

    elapsed = time.time() - start_time
    if (budget_s - elapsed) > 50:
        h_cont = best_h.astype(float)
        for eps in [0.0001, 0.00005, 0.00002, 0.00001]:
            if time.time() - start_time > budget_s - 10:
                break
            h_opt, c5_opt = slsqp_optimize(h_cont, eps, 300, target_sum)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_cont = h_opt.copy()

    best_h = project_onto_constraint(best_h, target_sum)
    final_c5 = compute_c5(best_h)
    if final_c5 <= best_c5 + 1e-10:
        best_h = best_h
        best_c5 = final_c5
    else:
        best_c5 = final_c5

    return best_h, best_c5, n0
