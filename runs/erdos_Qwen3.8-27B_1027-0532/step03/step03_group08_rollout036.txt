def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:

        order = np.argsort(1.0 - h)

            if diff <= 1e-12:

            diff += sub

def _fix_sum(h, target_sum):

    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h

            if diff <= 1e-12:

            diff -= add
    else:

            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub

def _nesterov_smooth(h0, n, target_sum, n_iters=200, alpha_init=50.0, alpha_end=500.0, lr=0.1, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        alpha = alpha_init * (alpha_end / alpha_init) ** (it / max(n_iters-1, 1))
        
        # Compute smooth gradient
        corr = np.correlate(y, 1.0 - y, mode='full')
        scaled = alpha * corr
        max_s = np.max(scaled)
        exps = np.exp(scaled - max_s)
        w = exps / exps.sum()
        
        # grad = dx * sum_L w_L * coef(j, L) for each j
        # This is: dx * (w @ coefs_matrix) where coefs_matrix[j, L] = coef(j, L)
        # But we don't have the full matrix, so compute per-j:
        grad = np.zeros(n)
        L_vals = np.arange(-(n-1), n)
        for j in range(n):
            jp = j + L_vals
            jm = j - L_vals
            mask_p = (jp >= 0) & (jp < n)
            mask_m = (jm >= 0) & (jm < n)
            cj = np.zeros(2*n-1)
            cj[mask_p] += 1.0 - y[jp[mask_p]]
            cj[mask_m] -= y[jm[mask_m]]
            grad[j] = dx * np.dot(w, cj)
        
        step = lr / np.sqrt(float(it + 1))
        x_new = _project(y - step * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

            # δ ∈ [-h[i], 1-h[i]] ∩ [h[j]-1, h[j]]

            new_val = _c5_of(h)

def _slsqp_obj(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    scaled = alpha * corr
    m = np.max(scaled)
    return dx * (m + np.log(np.sum(np.exp(scaled - m))) / alpha)

def _slsqp_opt(h0, n, target_sum, alpha=300, maxiter=1000, deadline=None):

Actually, one more important consideration: the `_fast_coord_descent` function as I've designed it has an issue. When I change h[j], the coefs for subsequent coordinates are affected. Wait, but coef_j was computed BEFORE the change, using the old h. Is it still valid after the change? Yes! Because coef_j(L) = (1-h[j+L])_valid - h[j-L]_valid, and this is the DERIVATIVE of c(L) with respect to h[j]. Since c(L) is linear in h[j] (for fixed other coordinates), the derivative is constant regardless of the current value of h[j]. 

Wait, is c(L) linear in h[j]? Let's check:
c(L) = Σ_i h[i]*(1-h[i+L])

The terms involving h[j] are:
- i=j: h[j]*(1-h[j+L]) → linear in h[j] with coefficient (1-h[j+L])
- i+L=j, i=j-L: h[j-L]*(1-h[j]) → linear in h[j] with coefficient -h[j-L]

So yes, c(L) is linear in h[j] (for fixed other coordinates). The coefficient is (1-h[j+L]) - h[j-L], which depends on OTHER coordinates, not on h[j] itself. So the coefficient is constant as long as other coordinates don't change.

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add

            order = np.argsort(x)
            for i in order:

def _pairwise_exchange(h, target_sum, max_rounds=50, pairs_per_round=200, grid_size=15, deadline=None, seed=99):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    best_val = _c5_of(h)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for _ in range(pairs_per_round):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-10:
                continue
            ci = _coef_vector(h, i)
            cj = _coef_vector(h, j)
            diff_c = ci - cj
            best_d = 0.0
            best_c = np.max(corr) * dx
            for d in np.linspace(d_lo, d_hi, grid_size):
                if abs(d) < 1e-10:
                    continue
                c_val = np.max(corr + d * diff_c) * dx
                if c_val < best_c - 1e-13:
                    best_c = c_val
                    best_d = d
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                corr = corr + best_d * diff_c
                improved = True
        if improved:
            h = _fix_sum(h, target_sum)
            new_val = _c5_of(h)
            if new_val < best_val:
                best_val = new_val
        else:
            break
    return h, _c5_of(h)

def _nesterov_smooth(h0, n, target_sum, n_iters=200, alpha_init=30.0, alpha_end=400.0, lr=0.15, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    L_vals = np.arange(-(n - 1), n)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        alpha = alpha_init * (alpha_end / alpha_init) ** (float(it) / float(max(n_iters - 1, 1)))
        corr = np.correlate(y, 1.0 - y, mode='full')
        scaled = alpha * corr
        max_s = np.max(scaled)
        exps = np.exp(scaled - max_s)
        total = exps.sum()
        if total < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(scaled)] = 1.0
        else:
            w = exps / total
        grad = np.zeros(n)
        for j in range(n):
            jp = j + L_vals
            jm = j - L_vals
            mask_p = (jp >= 0) & (jp < n)
            mask_m = (jm >= 0) & (jm < n)
            cj = np.zeros(2 * n - 1)
            if mask_p.any():
                cj[mask_p] += 1.0 - y[jp[mask_p]]
            if mask_m.any():
                cj[mask_m] -= y[jm[mask_m]]
            grad[j] = dx * np.dot(w, cj)
        step = lr / np.sqrt(float(it + 1))
        x_new = _project(y - step * grad, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_obj(x, *args):
    n = _slsqp_n
    alpha = _slsqp_alpha
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = alpha * corr
    m = np.max(scaled)
    return dx * (m + np.log(np.sum(np.exp(scaled - m))) / alpha)

def _slsqp_eq_fun(x):
    return x.sum() - _slsqp_target

def _slsqp_eq_jac(x):
    return np.ones(_slsqp_n)

def _slsqp_opt(h0, n, target_sum, alpha=300, maxiter=1500):
    global _slsqp_n, _slsqp_target, _slsqp_alpha
    _slsqp_n = n
    _slsqp_target = target_sum
    _slsqp_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}]
    try:
        res = minimize(_slsqp_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        mod = i % 5
        if mod == 1:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif mod == 2:
            pop[i] = _fix_sum(1.0 - h0 + 0.1 * (rng.rand(n) - 0.5), target_sum)
        elif mod == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif mod == 4:
            shuffled = h0[rng.permutation(n)]
            pop[i] = _fix_sum(shuffled, target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (float(gen) / float(max_gen))
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > pop_size * 2:
            worst = np.argsort(pop_vals)[pop_size // 2:]
            for idx in worst:
                t_mix = rng.uniform(0.3, 0.7)
                pop[idx] = _fix_sum(t_mix * best_h + (1.0 - t_mix) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

        if best_n != n:
            h_start2 = np.interp(np.linspace(0, best_n - 1, n), np.arange(best_n), best_h)
            h_start2 = _fix_sum(np.clip(h_start2, 0.0, 1.0), target_sum)
            if _c5_of(h_start2) < _c5_of(h_start):
                h_start = h_start2

        if n == n_init or (best_c5 < 0.381 and ni < 4):
            h_de, c5_de = _de_optimize(h_start, n, target_sum, 30, 80, 0.8, 0.4, 0.7, phase_deadline - alloc * 0.4, rng)
            if c5_de < best_c5:
                best_c5 = c5_de
                best_h = h_de.copy()
                best_n = n
            if c5_de < _c5_of(h_start):
                h_start = h_de.copy()

        h_opt, c5_opt = _optimize_single(h_start, n, target_sum, phase_deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    if time.time() < deadline - 30:
        target = best_n / 2.0
        h_final = best_h.copy()
        h_cd_f, c5_cd_f = _fast_coord_descent(h_final, target, n_iters=30, grid_size=25, deadline=deadline - 25)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            h_final = h_cd_f
        if time.time() < deadline - 20:
            h_pw_f, c5_pw_f = _pairwise_exchange(h_final, target, max_rounds=50, pairs_per_round=200, deadline=deadline - 15)
            if c5_pw_f < best_c5:
                best_c5 = c5_pw_f
                h_final = h_pw_f
        if time.time() < deadline - 12:
            h_nes_f, c5_nes_f = _nesterov_smooth(h_final, best_n, target, n_iters=100, alpha_init=50.0, alpha_end=600.0, lr=0.1, deadline=deadline - 10)
            if c5_nes_f < best_c5:
                best_c5 = c5_nes_f
                h_final = h_nes_f
        if time.time() < deadline - 8:
            h_sl_f = _slsqp_opt(h_final, best_n, target, alpha=400, maxiter=2000)
            c5_sl_f = _c5_of(h_sl_f)
            if c5_sl_f < best_c5:
                best_c5 = c5_sl_f
                h_final = h_sl_f
        if time.time() < deadline - 3:
            h_cd_last, c5_last = _fast_coord_descent(h_final, target, n_iters=15, grid_size=30, deadline=deadline - 1)
            if c5_last < best_c5:
                best_c5 = c5_last
                h_final = h_cd_last
        best_h = h_final

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
