```python
def project_onto_simplex_box(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}."""
    n = len(h)
    # Bisection on lambda
    # We want: sum(clip(h - lambda, 0, 1)) = target_sum
    # clip(h_i - lambda, 0, 1) is:
    #   0 if h_i - lambda <= 0, i.e., lambda >= h_i
    #   1 if h_i - lambda >= 1, i.e., lambda <= h_i - 1
    #   h_i - lambda otherwise
    
    # The sum is a piecewise linear, decreasing function of lambda.
    # Find lambda by bisection.
    
    lo = np.min(h) - 1.0 - 1.0  # lambda this large, all clipped to 0
    hi = np.max(h) + 1.0  # lambda this small, all clipped to 1
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if abs(s - target_sum) < 1e-14:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    lambda_star = (lo + hi) / 2.0
    return np.clip(h - lambda_star, 0.0, 1.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)


def project_simplex_box(h, target_sum):
    n = len(h)
    lo = np.min(h) - 1.0
    hi = np.max(h) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def smax_val(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return cmax + float(np.log(np.sum(np.exp(shifted)))) / alpha


def smax_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-15:
        return np.zeros(n)
    w = w / sw
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        wk = w[k_idx]
        if wk < 1e-14:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            if L < n:
                grad[L:] += wk * (1.0 - h[:n - L])
            if L == 0:
                grad[:] -= wk * h[:]
            else:
                grad[:n - L] -= wk * h[L:]
        else:
            M = -L
            if M < n:
                grad[:n - M] += wk * (1.0 - h[M:])
            if M > 0:
                grad[M:] -= wk * h[:n - M]
    return grad * dx


def projected_gradient(h, alpha, target, time_limit, step_init=0.01, penalty_coeff=50.0):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    step = step_init
    stall = 0

    while time.time() - start < time_limit:
        g = smax_grad(h, alpha)
        sum_err = np.sum(h) - target
        g = g + penalty_coeff * sum_err / n
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        step = min(step, 0.5 / max(gnorm, 1e-10))
        h_trial = project_simplex_box(h - step * g, target)
        f_trial = smax_val(h_trial, alpha)
        f_curr = smax_val(h, alpha)

        if f_trial < f_curr:
            h = h_trial
            step = min(step * 1.1, 0.5 / max(gnorm, 1e-10))
            s = compute_c5(h)
            if s < best_score:
                best_score = s
                best = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            step *= 0.5
            stall += 1
            if stall > 200:
                step = step_init * 0.1
                stall = 0
                if np.linalg.norm(smax_grad(h, alpha)) < 1e-12:
                    break

    return best, best_score


def paired_coord(h, target, time_limit, seed=42, n_pairs=150, mode='medium'):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    stall = 0

    if mode == 'medium':
        fracs = [0.2, 0.4, 0.6, 0.8]
    elif mode == 'fine':
        fracs = [0.25, 0.5, 0.75]
    else:
        fracs = [0.4, 0.6]

    while time.time() - start < time_limit:
        improved = False
        for _ in range(n_pairs):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            d_lo = max(-best[i], best[j] - 1.0)
            d_hi = min(1.0 - best[i], best[j])
            if d_hi - d_lo < 1e-12:
                continue
            for frac in fracs:
                d = d_lo + frac * (d_hi - d_lo)
                if abs(d) < 1e-13:
                    continue
                h_trial = best.copy()
                h_trial[i] += d
                h_trial[j] -= d
                s = compute_c5(h_trial)
                if s < best_score - 1e-15:
                    best_score = s
                    best = h_trial
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

    return best, best_score

def lbfgs_smooth(h, alpha, target, maxiter=3000, time_limit=100):
    n = len(h)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    pc = 200.0

    def obj_pen(x):
        v = smax_val(x, alpha)
        se = np.sum(x) - target
        return v + pc * se * se

    def grad_pen(x):
        g = smax_grad(x, alpha) + 2.0 * pc * (np.sum(x) - target) * np.ones(n)
        return g

    try:
        res = minimize(obj_pen, h, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-14, 'maxls': 30})
        h_new = project_feasible(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)


def block_move(h, target, time_limit, seed=42, block_size=5):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    stall = 0

    while time.time() - start < time_limit:
        improved = False
        for _ in range(80):

            j = rng.randint(max(1, n - block_size))
            if abs(i - j) < block_size:
                continue

            d_hi = min(1.0 - best[i], best[j])

            for frac in [0.3, 0.5, 0.7]:
                d = d_lo + frac * (d_hi - d_lo)
                if abs(d) < 1e-13:
                    continue
                h_trial = best.copy()
                for offset in range(block_size):
                    ii = (i + offset) % n
                    jj = (j + offset) % n
                    val_i = h_trial[ii] + d / block_size
                    val_j = h_trial[jj] - d / block_size
                    if val_i < 0 or val_i > 1 or val_j < 0 or val_j > 1:
                        break
                    h_trial[ii] = val_i
                    h_trial[jj] = val_j
                else:
                    s = compute_c5(h_trial)
                    if s < best_score - 1e-15:
                        best_score = s
                        best = h_trial
                        improved = True
                        break

        if improved:

            stall += 1

                break
```
