def project_feasible(h, target_sum):

def lag_gradient(m, h, n):
    grad = np.zeros(n)
    if m >= 0:
        rem = n - m
        if rem > 0:
            grad[:rem] += (1.0 - h[m:])
            grad[m:] -= h[:rem]
    else:
        ma = -m
        rem = n - ma
        if rem > 0:
            grad[ma:] += (1.0 - h[:rem])
            grad[:rem] -= h[ma:]
    return grad


def subgrad_phase(h, n_iter, alpha0, decay, momentum0, momentum_end, time_limit_fn):

    target = n / 2.0

            break

        beta = momentum0 + (momentum_end - momentum0) * t / max(1, n_iter)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)

        active = np.where(corr >= max_val - tol)[0]

        grad /= len(active)

        if score < best_score:

            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_score


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])

def lbfgs_polish(h_start, n, target, alpha, maxiter, ftol):
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * n
    
    def obj_func(x):
        F, _ = smooth_obj_and_grad(x, alpha)
        return F
    
    def jac_func(x):
        _, g = smooth_obj_and_grad(x, alpha)
        return g
    
    try:
        res = minimize(obj_func, h_cur, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_cur

    if n_src == n_dst:

    x_new = np.linspace(0.0, 1.0, n_dst)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def lag_gradient(m, h, n):
    grad = np.zeros(n)
    if m >= 0:
        rem = n - m
        if rem > 0:
            grad[:rem] += (1.0 - h[m:])
            grad[m:] -= h[:rem]
    else:
        ma = -m
        rem = n - ma
        if rem > 0:
            grad[ma:] += (1.0 - h[:rem])
            grad[:rem] -= h[ma:]
    return grad

    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        term_a = np.sum(w[klo:khi + 1])
        j_idx = np.arange(klo, khi + 1) - (n - 1) + i
        term_b = np.sum(w[klo:khi + 1] * h[j_idx])
        klo_c = max(0, i)
        khi_c = min(2 * n - 2, i + n - 1)
        if klo_c <= khi_c:

            term_c = np.sum(w[klo_c:khi_c + 1] * h[j_idx_c])
        else:
            term_c = 0.0
        grad[i] = term_a - term_b - term_c
    return F, grad
