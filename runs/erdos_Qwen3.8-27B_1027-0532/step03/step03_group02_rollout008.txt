import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    for _ in range(200):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-10)[0]
        else:
            idx = np.where(h > 1e-10)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
    return h

def symmetrize(h):
    h_sym = (h.astype(float) + 1.0 - h[::-1].astype(float)) / 2.0
    return np.clip(h_sym, 0.0, 1.0)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def coord_descend(h, target_sum, max_time, rng, patience=30, noise_base=0.005):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = h.copy()
    h = project_feasible(h, target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    elapsed_start = time.time()

    while time.time() - elapsed_start < max_time:
        for j in range(n):
            old_val = h[j]
            js = j - s_arr
            js_plus = j + s_arr
            A = np.zeros(2 * n - 1)
            mask1 = (js >= 0) & (js < n)
            mask2 = (js_plus >= 0) & (js_plus < n)
            A[mask1] += (1.0 - h[js[mask1]])
            A[mask2] -= h[js_plus[mask2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2.0
                vals = A * mid + B
                k_star = np.argmax(vals)
                grad = A[k_star]
                if grad > 1e-10:
                    hi = mid
                elif grad < -1e-10:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) / 2.0
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val

        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            noise_scale = noise_base * (1.0 + float(no_improve) / patience)
            h = best_h + rng.randn(n) * noise_scale
            h = project_feasible(h, target_sum)
            corr = np.correlate(h, 1.0 - h, mode='full')
            no_improve = 0

    return best_h, best_score

def _smooth_obj(x, alpha):
    c = np.correlate(x, 1.0 - x, mode='full')
    cm = float(np.max(c))
    shifted = np.clip(alpha * (c - cm), -500.0, 0.0)
    return cm + np.log(np.sum(np.exp(shifted))) / alpha

def _cons_sum(x, target_sum, n):
    return float(np.sum(x) - target_sum)

def _cons_sum_jac(x, target_sum, n):
    return np.ones(n)

    n = len(h)
    bounds = [(0.0, 1.0)] * n
    best_score = compute_c5(h)
    best_h = h.copy()

    for alpha in [300, 800, 1500]:
        if time.time() - start > max_time:
            break
        try:
            res = minimize(_smooth_obj, best_h, args=(alpha,), method='SLSQP',
                           bounds=bounds,

                                        'jac': _cons_sum_jac, 'args': (target_sum, n)}],
                           options={'maxiter': 200, 'ftol': 1e-14})

            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new

            pass
