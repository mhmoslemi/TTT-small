def nesterov_subgradient(h, target, time_limit, alpha_init=10.0, eta_init=0.01, momentum=0.9):
    n = len(h)
    start = time.time()
    alpha = alpha_init
    eta = eta_init
    best = h.copy()
    best_score = compute_c5(h)
    y = h.copy()  # Nesterov lookahead
    v = np.zeros(n)  # velocity
    
    no_improve = 0
    while time.time() - start < time_limit:
        # Nesterov step: evaluate at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        weights = subgradient_weights(corr, alpha)
        grad = gradient_from_weights(y, weights, n)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        # Update velocity and position
        v = momentum * v + (1 - momentum) * grad / gn
        # Actually, standard Nesterov: v = momentum*v - eta*grad; y = x + v
        # But we need to handle the projection carefully.
        
        eta_curr = eta / np.sqrt(time.time() - start + 1)  # diminishing step size
        x_new = y + eta_curr * v
        y_new = x_new + momentum * (x_new - y)
        
        x_new = project_feasible(x_new, target)
        y_new = project_feasible(y_new, target)
        
        score = compute_c5(x_new)
        if score < best_score:
            best_score = score
            best = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                alpha *= 1.5
                eta *= 0.8
                no_improve = 0
        
        y = x_new
    
    return best, best_score

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def full_gradient(h, corr, weights, n):
    """Compute gradient of weighted sum of correlations."""
    grad = np.zeros(n)
    # corr has length 2n-1
    # corr[k] corresponds to lag L = k - (n-1)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-10:
            continue
        L = k - (n - 1)
        if L >= 0:
            # corr[L] = sum_{i=0}^{n-1-L} h[i+L] * (1-h[i])
            # d/dh[j] for j in [L, n-1]: contributes (1-h[j-L])
            # d/dh[j] for j in [0, n-1-L]: contributes -h[j+L]
            if L == 0:
                # corr[0] = sum h[i] * (1-h[i])
                # d/dh[j] = (1-h[j]) - h[j] = 1 - 2h[j]
                grad += w * (1.0 - 2.0 * h)
            else:
                # j from L to n-1: h[j] is the first factor
                idx = np.arange(L, n)
                grad[idx] += w * (1.0 - h[idx - L])
                # j from 0 to n-1-L: h[j] is the second factor (in 1-h[j])
                idx2 = np.arange(0, n - L)
                grad[idx2] -= w * h[idx2 + L]
        else:
            M = -L
            # corr[-M] = sum_{i=0}^{n-1-M} h[i] * (1-h[i+M])
            # d/dh[j] for j in [0, n-1-M]: contributes (1-h[j+M])
            # d/dh[j] for j in [M, n-1]: contributes -h[j-M]
            if M == 0:
                grad += w * (1.0 - 2.0 * h)
            else:
                idx = np.arange(0, n - M)
                grad[idx] += w * (1.0 - h[idx + M])
                idx2 = np.arange(M, n)
                grad[idx2] -= w * h[idx2 - M]

        h2, s2 = coord_descent_fine(best_h, target, time_limit=min(150, tleft() - 10),
                                     deltas=[0.004, -0.004, 0.009, -0.009,
                                            0.015, -0.015, 0.025, -0.025],
                                     seed=seed)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    if tleft() > 180:
        h3, s3 = slsqp_logsumexp(best_h, target, time_limit=min(180, tleft() - 10))
        if s3 < best_score:
            best_score = s3
            best_h = h3

    if tleft() > 150:
        h4, s4 = coord_descent_fine(best_h, target, time_limit=min(150, tleft() - 10),
                                     deltas=[0.002, -0.002, 0.005, -0.005,
                                            0.008, -0.008, 0.015, -0.015],
                                     seed=seed + 100)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    if tleft() > 200:
        h5, s5 = shake_and_refine(best_h, target, time_limit=min(200, tleft() - 10),
                                   seed=seed + 200, n_shakes=6)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    if tleft() > 150:
        h6, s6 = slsqp_pnorm(best_h, target, time_limit=min(150, tleft() - 10))
        if s6 < best_score:
            best_score = s6
            best_h = h6

    if tleft() > 150:
        h7, s7 = nesterov_subgrad(best_h, target, time_limit=min(150, tleft() - 10),
                                   alpha_init=60.0, eta_init=0.002, momentum_init=0.8)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    if tleft() > 120:
        h8, s8 = coord_descent_fine(best_h, target, time_limit=min(120, tleft() - 10),
                                     deltas=[0.001, -0.001, 0.003, -0.003,
                                            0.005, -0.005, 0.01, -0.01],
                                     seed=seed + 300)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    if tleft() > 100:
        h9, s9 = simulated_annealing(best_h, target, time_limit=min(100, tleft() - 10),
                                      seed=seed + 400, T_init=0.02)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    if tleft() > 80:
        h10, s10 = slsqp_logsumexp(best_h, target, time_limit=min(80, tleft() - 10),
                                    alphas=[15, 50, 120])
        if s10 < best_score:
            best_score = s10
            best_h = h10

    if tleft() > 40:
        h11, s11 = coord_descent_fine(best_h, target, time_limit=max(20, tleft() - 15),
                                       deltas=[0.0005, -0.0005, 0.001, -0.001,
                                              0.002, -0.002, 0.004, -0.004],
                                       seed=seed + 500)
        if s11 < best_score:
            best_score = s11
            best_h = h11

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
