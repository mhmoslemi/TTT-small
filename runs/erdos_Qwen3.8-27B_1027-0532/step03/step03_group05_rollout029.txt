    best = h.copy()
    best_score = compute_c5(h)

        for _ in range(1000):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            # Swap
            h_trial = best.copy()
            h_trial[i], h_trial[j] = best[j], best[i]
            s = compute_c5(h_trial)
            if s < best_score:
                best_score = s
                best = h_trial
        # Try small transfers
        for _ in range(1000):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:

            d = rng.uniform(-0.05, 0.05)

            vi = best[i] + d
            vj = best[j] - d
            if 0 <= vi <= 1 and 0 <= vj <= 1:
                h_trial[i] = vi
                h_trial[j] = vj

                if s < best_score:

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_weighted_gradient(h, corr_vals, alpha):
    """
    Compute gradient of the softmax-approximated max of correlation.
    corr_vals: the full correlation array (length 2n-1)
    alpha: log-sum-exp parameter
    """
    n = len(h)
    shifted = alpha * (corr_vals - np.max(corr_vals))
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-30:
        return np.zeros(n)
    weights = weights / sw
    
    grad = np.zeros(n)
    for lag_idx in range(2*n - 1):
        w = weights[lag_idx]
        if abs(w) < 1e-10:
            continue
        m = lag_idx - (n - 1)
        if m == 0:
            grad += w * (1 - 2*h)
        elif m > 0:
            grad[m:] += w * (1 - h[:n-m])
            grad[:n-m] += -w * h[m:]
        else:
            mm = -m
            grad[:n-mm] += w * (1 - h[mm:])
            grad[mm:] += -w * h[:n-mm]
    return grad

def subgradient_with_momentum(h, target, time_limit, alpha_init=20.0, eta_init=0.003, momentum=0.9):
    n = len(h)
    start = time.time()
    alpha = alpha_init
    eta = eta_init
    best_local = h.copy()
    best_local_score = compute_c5(h)
    v = np.zeros(n)  # momentum
    h_avg = h.copy()  # Polyak average
    count = 0
    no_improve = 0
    
    while time.time() - start < time_limit:
        corr = compute_full_corr(h)
        grad = compute_weighted_gradient(h, corr, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        
        # Momentum update
        v = momentum * v + (1 - momentum) * grad / gn
        vn = np.linalg.norm(v)
        if vn < 1e-14:
            break
        
        h_new = h + eta * v / vn
        h_new = project_feasible(h_new, target)
        
        count += 1
        h_avg = h_avg + (h_new - h_avg) / count
        
        score = compute_c5(h_new)

            best_local_score = score
            best_local = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.01)

            no_improve += 1
            if no_improve > 50:

                alpha *= 1.3

    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_weighted_gradient(h, corr_vals, alpha):
    n = len(h)
    shifted = alpha * (corr_vals - np.max(corr_vals))
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-30:
        return np.zeros(n)
    weights = weights / sw
    
    grad = np.zeros(n)
    center = n - 1
    for lag_idx in range(2 * n - 1):
        w = weights[lag_idx]
        if abs(w) < 1e-10:
            continue
        m = lag_idx - center
        if m == 0:
            grad += w * (1.0 - 2.0 * h)
        elif m > 0:
            grad[m:] += w * (1.0 - h[:n-m])

        else:
            mm = -m

    return grad
