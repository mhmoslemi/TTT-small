import numpy as np

_cons_target = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    n = len(h)
    h_r = np.clip(np.round(h), 0, 1).astype(float)
    current_sum = int(np.sum(h_r))
    if current_sum == target_sum:
        return h_r
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r = h_r.copy()
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r = h_r.copy()
            h_r[zeros[order]] = 1
    final_sum = int(np.sum(h_r))
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            h_r = h_r.copy()
            h_r[ones2[:final_sum - target_sum]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            h_r = h_r.copy()
            h_r[zeros2[:target_sum - final_sum]] = 1
    return h_r


def _apply_flip(h, corr, p, to_one, n):
    m = n - 1
    k_start = max(0, m - p)
    k_end = min(2 * n - 2, m - p + n - 1)
    ks = np.arange(k_start, k_end + 1)
    j1 = p + ks - m
    j2 = p + m - ks
    v1 = (j1 >= 0) & (j1 < n)
    v2 = (j2 >= 0) & (j2 < n)
    if to_one:
        if np.any(v1):
            corr[ks[v1]] += 1.0 - h[j1[v1]]
        if np.any(v2):
            corr[ks[v2]] -= h[j2[v2]]
    else:
        if np.any(v1):
            corr[ks[v1]] -= 1.0 - h[j1[v1]]
        if np.any(v2):
            corr[ks[v2]] += h[j2[v2]]


def fast_sa(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    m = n - 1
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    no_improve = 0
    max_stall = max(3000, max_iters // 15)
    n_in_ones = len(ones)
    n_in_zeros = len(zeros)

        if time.time() - start_time > time_limit:
            break
        if not ones or not zeros:
            break
        if no_improve > max_stall:
            break
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        mt = rng.randint(3)
        if mt == 0:

            q = zeros[rng.randint(n_in_zeros)]
            _apply_flip(h, corr, p, False, n)
            _apply_flip(h, corr, q, True, n)
            new_max = float(np.max(corr))

                ones.remove(p)
                ones.append(q)
                zeros.remove(q)
                zeros.append(p)
                if new_max < best_max:

                    best_h = h.copy()
                    no_improve = 0

                    no_improve += 1

                _apply_flip(h, corr, p, True, n)

        elif mt == 1 and n_in_ones >= 2 and n_in_zeros >= 2:

                    zeros.remove(q)
                    ones.append(q)

                    zeros.append(p)
                n_in_ones = len(ones)
                n_in_zeros = len(zeros)
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                _apply_flip(h, corr, q2, False, n)
                _apply_flip(h, corr, q1, False, n)
                _apply_flip(h, corr, p2, True, n)
                _apply_flip(h, corr, p1, True, n)
                current_max = float(np.max(corr))
                no_improve += 1
        else:
            p = ones[rng.randint(n_in_ones)]
            if n_in_zeros > 10:
                candidates = rng.randint(n_in_zeros, min(10, n_in_zeros))
                dists = np.array([abs(zeros[ci] - p) for ci in candidates])
                q = zeros[candidates[np.argmin(dists)]]
            else:
                q = zeros[rng.randint(n_in_zeros)]
            _apply_flip(h, corr, p, False, n)
            _apply_flip(h, corr, q, True, n)
            new_max = float(np.max(corr))
            delta = new_max - current_max
            if delta < 0 or (t > 1e-12 and rng.random() < np.exp(-delta / t)):
                current_max = new_max
                ones.remove(p)
                ones.append(q)
                zeros.remove(q)
                zeros.append(p)
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                _apply_flip(h, corr, q, False, n)
                _apply_flip(h, corr, p, True, n)
                current_max = float(np.max(corr))
                no_improve += 1
    return best_h, best_max


def hill_climb_fast(h_start, target_sum, max_iter, rng, time_limit, start_time):
    n = len(h_start)
    h = h_start.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    m = n - 1
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    ones = [int(x) for x in np.where(h == 1)[0]]
    zeros = [int(x) for x in np.where(h == 0)[0]]
    stall = 0
    max_stall = 500
    n_ones = len(ones)
    n_zeros = len(zeros)

    for _ in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if not ones or not zeros:
            break
        if stall > max_stall:
            break
        p = ones[rng.randint(n_ones)]
        q = zeros[rng.randint(n_zeros)]
        _apply_flip(h, corr, p, False, n)
        _apply_flip(h, corr, q, True, n)
        new_max = float(np.max(corr))
        if new_max < current_max:
            current_max = new_max
            ones.remove(p)
            ones.append(q)
            zeros.remove(q)
            zeros.append(p)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
            stall = 0
        else:
            _apply_flip(h, corr, q, False, n)
            _apply_flip(h, corr, p, True, n)
            stall += 1
    return best_h, best_max


def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)


def _cons_fun(h):
    return np.sum(h) - _cons_target


def _cons_jac(h):
    return np.ones(_cons_n)


def slsqp_refine(h_start, n, target, alphas, maxiter, ftol, time_limit, start_time):
    global _cons_n, _cons_target
    _cons_n = n
    _cons_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha,),
                           method='SLSQP', bounds=bounds,
                           constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src.astype(float))
    return project_feasible(h_out, target_dst)


def ils_loop(h_best, target_sum, max_rounds, rng, time_limit, start_time):
    n = len(h_best)
    best_h = h_best.copy()
    best_score = compute_c5(best_h)
    for r in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        perturb_frac = rng.choice([0.02, 0.03, 0.05, 0.07, 0.10])
        n_flip = max(2, int(n * perturb_frac))
        h_pert = best_h.copy()
        flip_idx = rng.choice(n, n_flip, replace=False)
        h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
        h_pert = project_binary(h_pert, target_sum)
        sa_time = min(30.0, (time_limit - (time.time() - start_time)) * 0.8)
        if sa_time < 5:
            break
        max_it = max(3000, int(sa_time * 8000))
        t_init = rng.choice([0.005, 0.008, 0.012, 0.02])
        t_final = 0.0001
        h_cand, score_cand = fast_sa(h_pert, target_sum, max_it, t_init, t_final,
                                      rng, sa_time, time.time())
        if compute_c5(h_cand) < best_score:
            h_hc, score_hc = hill_climb_fast(h_cand, target_sum, 3000, rng,
                                              10.0, time.time())
            if compute_c5(h_hc) < best_score:
                best_score = compute_c5(h_hc)
                best_h = h_hc.copy()
            else:
                best_score = compute_c5(h_cand)
                best_h = h_cand.copy()
    return best_h, best_score
