import numpy as np
from scipy.optimize import minimize
import time

_CONS_N = 0
_CONS_TARGET = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-14:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * one_minus_h[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * one_minus_h[m:]
                grad[m:] -= w * h[:n - m]
    return grad


def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET


def cons_eq_jac(x):
    return np.ones(_CONS_N)


def smooth_max_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha


def smooth_max_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    return compute_grad_weighted(x, np.arange(2 * len(x) - 1), weights)


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol, time_limit):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    t0 = time.time()
    for alpha in alphas:
        if time.time() - t0 > time_limit:
            break
        try:

                           jac=smooth_max_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

            break
        corr = overlap_corrs(h)
        sorted_idx = np.argsort(corr)[::-1]
        actual_topk = min(topk, len(corr))
        top_indices = sorted_idx[:actual_topk]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(actual_topk) / actual_topk
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-8:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 5.0)
            if stagnation == 0 and it > 100:
                topk = max(1, topk + 1)
        else:
            stagnation += 1
            lr *= 0.97
            if stagnation % 50 == 0:
                topk = max(1, topk - 1)
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def lag_chase_subgradient(h_init, target_sum, max_iter, lr_init, time_limit):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    t0 = time.time()

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = overlap_corrs(h)
        worst_idx = np.argmax(corr)
        grad = compute_grad_weighted(h, np.array([worst_idx]), np.array([1.0]))
        norm = np.linalg.norm(grad)
        if norm < 1e-10:
            break
        v = 0.9 * v + 0.1 * (grad / norm)
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score - 1e-9:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.005, 2.0)
        else:
            stagnation += 1
            lr *= 0.985
        h = h_new
        if stagnation > 300:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_refine(h_start, target_sum, block_size, n_sweeps, time_limit):
    n = len(h_start)
    h = h_start.copy()
    best_score = compute_c5(h)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * block_size
    global _CONS_N, _CONS_TARGET

    for sweep in range(n_sweps):
        if time.time() - t0 > time_limit:
            break
        for start in range(0, n - block_size + 1, block_size):
            if time.time() - t0 > time_limit:
                break
            block = h[start:start + block_size].copy()
            target_block = np.sum(block)
            if target_block < 1e-6 or target_block > block_size - 1e-6:
                continue
            _CONS_N = block_size
            _CONS_TARGET = target_block
            sub_bounds = [(0.0, 1.0)] * block_size
            cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]

            def block_obj(x, s=start, hh=h):
                hh_copy = hh.copy()
                hh_copy[s:s + block_size] = x
                return compute_c5(hh_copy)

            def block_grad(x, s=start, hh=h):
                hh_copy = hh.copy()
                hh_copy[s:s + block_size] = x
                g_full = smooth_max_grad(hh_copy, 50.0)
                return g_full[s:s + block_size]

            try:
                res = minimize(block_obj, block, jac=block_grad, method='SLSQP',
                               bounds=sub_bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12})
                new_block = project_feasible(res.x, target_block)
                h_new = h.copy()
                h_new[start:start + block_size] = new_block
                sc = compute_c5(h_new)
                if sc < best_score:
                    h = h_new
                    best_score = sc
            except Exception:
                pass
    return h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()

    def _remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    # Phase 1: Quick subgradient to escape any local issues
    if _remaining() > 80:
        h_try = best_h.copy()
        rng0 = np.random.RandomState(seed)
        h_try = h_try + rng0.randn(n_init) * 0.02
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient_adaptive(
            h_try, target_init, max_iter=2000, lr_init=0.12, alpha=30,
            noise_prob=0.05, noise_scale=0.02, patience=200, momentum=0.90,
            seed=seed, time_limit=max(0, _remaining() - 10))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: SLSQP with moderate alpha
    if _remaining() > 120:
        alphas = [30, 80, 200]
        h_ref = slsqp_refine(best_h, target_init, alphas, 2000, 1e-14,
                             time_limit=max(0, _remaining() - 30))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Lag-chasing subgradient
    if _remaining() > 60:
        h_chase, score = lag_chase_subgradient(
            best_h, target_init, max_iter=10000, lr_init=0.02,
            time_limit=max(0, _remaining() - 30))
        if score < best_score:
            best_score = score
            best_h = h_chase

    # Phase 4: SLSQP with high alpha
    if _remaining() > 100:
        alphas = [200, 500, 1000]
        h_ref = slsqp_refine(best_h, target_init, alphas, 3000, 1e-15,
                             time_limit=max(0, _remaining() - 20))
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 5: Another round of subgradient with different params
    if _remaining() > 80:
        rng1 = np.random.RandomState(seed + 100)
        h_try = best_h + rng1.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient_adaptive(
            h_try, target_init, max_iter=5000, lr_init=0.05, alpha=50,
            noise_prob=0.02, noise_scale=0.01, patience=300, momentum=0.92,
            seed=seed + 200, time_limit=max(0, _remaining() - 20))
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 6: Try higher resolution
    for n_try in [500, 600]:
        if _remaining() < 100:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        h_new, score = run_subgradient_adaptive(
            h_interp, target_try, max_iter=3000, lr_init=0.08, alpha=40,
            noise_prob=0.03, noise_scale=0.012, patience=250, momentum=0.90,
            seed=seed + 500 + n_try, time_limit=60)
        if score < best_score:
            best_score = score
            best_h = h_new
        if _remaining() > 80:
            alphas_hi = [100, 300, 800]
            h_ref = slsqp_refine(best_h, len(best_h) / 2.0, alphas_hi, 1500, 1e-14,
                                 time_limit=max(0, _remaining() - 50))
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    # Phase 7: Final SLSQP polish with very high alpha
    n_use = len(best_h)
    target_use = n_use / 2.0
    if _remaining() > 60:
        h_final = slsqp_refine(best_h, target_use, [500, 1500, 3000], 4000, 1e-16,
                               time_limit=max(0, _remaining() - 20))
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    # Phase 8: Final lag chase
    if _remaining() > 30:
        h_chase, score = lag_chase_subgradient(
            best_h, target_use, max_iter=20000, lr_init=0.01,
            time_limit=max(0, _remaining() - 10))
        if score < best_score:
            best_score = score
            best_h = h_chase

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
