def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):

    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def active_set_slsqp(h_init, n_active, maxiter, rng):
    """Optimize using only the top n_active lags in the objective"""
    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = h_init.astype(np.float64).copy()
    h = project_onto_constraint(h, target_sum)
    
    def objective(h_h):
        corr = np.correlate(h_h, 1.0 - h_h, mode='full') * dx
        top_vals = np.sort(corr)[-n_active:]
        return np.max(top_vals) + 0.01 * np.mean(top_vals)
    
    def gradient(h_h):
        corr = np.correlate(h_h, 1.0 - h_h, mode='full') * dx
        top_idx = np.argsort(corr)[-n_active:]
        n_sorted = n
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                for i in range(n):
                    g = 0.0
                    if i >= s:
                        g += (1.0 - h_h[max(0, i-s)])
                    if i < m:
                        g -= h_h[min(n-1, i+s)]
                    grad[i] += dx * g
            else:
                r = -s
                m = n - r
                for i in range(n):
                    g = 0.0
                    if i < m:
                        g += (1.0 - h_h[min(n-1, i+r)])
                    if i >= r:
                        g -= h_h[max(0, i-r)]
                    grad[i] += dx * g
        grad /= n_active
        return grad
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(objective, h, jac=gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h, compute_c5(h)
