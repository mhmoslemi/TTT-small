import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps_list, maxiter_per_eps):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    for eps in eps_list:
        try:
            result = minimize(smooth_objective, h, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_per_eps, 'ftol': 1e-16})
            h_new = project_onto_constraint(result.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = best_h.copy()
        except Exception:
            pass
    return best_h, best_c5

def to_binary(h):
    n = len(h)
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

        n_up = n0 * 2
        target_up = n_up / 2.0
        h_up = upsample(best_h, 2)
        h_up = project_onto_constraint(h_up, target_up)
        h_up_opt, c5_up = slsqp_optimize(h_up, [0.05, 0.01, 0.003, 0.001, 0.0003], 400, deadline)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up_opt.copy()
        
        n_tries = 0
        while n_tries < 4 and time.time() < deadline - 80:
            h_pert_up = best_h + 0.03 * rng.standard_normal(best_h.shape[0])
            h_pert_up = project_onto_constraint(h_pert_up, best_h.shape[0] / 2.0)
            h_new, c5_new = slsqp_optimize(h_pert_up, [0.005, 0.001, 0.0003, 0.0001], 400, deadline)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            n_tries += 1
    
    while time.time() < deadline - 10:
        n_final = len(best_h)
        h_final = best_h + 0.002 * rng.standard_normal(n_final)
        h_final = project_onto_constraint(h_final, n_final / 2.0)
        h_final, c5_final = slsqp_optimize(h_final, [0.0003, 0.0001, 0.00003], 500, deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
        else:
            break
    
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
