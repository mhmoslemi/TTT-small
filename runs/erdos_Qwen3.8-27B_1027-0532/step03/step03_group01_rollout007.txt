import numpy as np
from scipy.optimize import minimize
import time

_C_NS = 0
_C_TARGET = 0.0


def fft_cross_correlate(a, b):
    na = len(a)
    nb = len(b)
    size = 1 << (na + nb - 2).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    r = np.fft.irfft(A * np.conj(B), size)
    out = np.empty(na + nb - 1)
    if nb > 1:
        out[:nb - 1] = r[size - (nb - 1):size]
    out[nb - 1] = r[0]
    if na > 1:
        out[nb:] = r[1:na]
    return out


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_cross_correlate(h, 1.0 - h)
    return float(np.max(corr)) * dx


def compute_c5_verify(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)


def project_binary(h, target_sum):
    h_r = np.round(h).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = current_sum - target_sum
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        extra = final_sum - target_sum
        ones2 = np.where(h_r == 1)[0]
        h_r[ones2[:extra]] = 0
    elif final_sum < target_sum:
        extra = target_sum - final_sum
        zeros2 = np.where(h_r == 0)[0]
        h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

            i0 = zeros_arr[int(np.argmin(dist))]
            h[i1] = 0.0

        elif mt == 3 and len(ones_arr) >= 3 and len(zeros_arr) >= 3:

def rtt_optimize(h_start, target_sum, max_iters, rng, time_limit, start_time):

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def smooth_obj_func(h, alpha, n):
    dx = 2.0 / n
    corr = fft_cross_correlate(h, 1.0 - h)
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    val = c_max + np.log(np.sum(np.exp(shifted))) / alpha
    return val * dx


def _smooth_obj_wrapper(h, alpha, n):
    return smooth_obj_func(h, alpha, n)


def _cons_fun(h):
    return np.sum(h) - _C_TARGET


def _cons_jac(h):
    return np.ones(_C_NS)


def slsqp_refine(h_start, n, target, alphas, maxiter_per_alpha, ftol, time_limit, start_time):
    global _C_NS, _C_TARGET
    _C_NS = n
    _C_TARGET = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(
                lambda x: _smooth_obj_wrapper(x, alpha, n),
                h_cur,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': maxiter_per_alpha, 'ftol': ftol}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score


def multi_start_sa(h_best, target_sum, n_starts, rng, time_limit, start_time):

        n_flips = rng.randint(3, 15)
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        if len(ones) < n_flips or len(zeros) < n_flips:
            n_flips = min(n_flips, len(ones), len(zeros))
        if n_flips < 1:
            continue
        flips_1 = rng.choice(len(ones), n_flips, replace=False)
        flips_0 = rng.choice(len(zeros), n_flips, replace=False)
        for idx in flips_1:
            h_pert[ones[idx]] = 0.0
        for idx in flips_0:
            h_pert[zeros[idx]] = 1.0

        allowed = min(80.0, (time_limit - (time.time() - start_time)) * 0.4)
        h_sa, c5_sa = sa_optimize(h_pert, target_sum, 50000, 0.01, 0.0001,
                                  rng, time_limit=allowed, start_time=start_time)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    return best_h, best_c5
