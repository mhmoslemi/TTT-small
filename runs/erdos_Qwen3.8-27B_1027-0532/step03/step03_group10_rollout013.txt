    for step in range(n_steps):
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        if rng.random() < 0.6:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = int(ones[rng.integers(len(ones))])
            j = int(zeros[rng.integers(len(zeros))])
            if i == j:
                continue
            s_min, s_max, new_max, _ = swap_move_eval(h, i, j, corr, n)
            delta_c = new_max - current_max
            if delta_c < -1e-12 or (t > 1e-10 and rng.random() < np.exp(-max(delta_c, 0) / t)):
                if s_min <= s_max:
                    s_range = np.arange(s_min, s_max + 1)
                    k_range = s_range + half
                    delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
                    corr[k_range] += delta
                h[i], h[j] = h[j], h[i]
                current_max = new_max
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
        else:
            block_len = int(rng.integers(2, min(8, n // 4)))
            start_pos = int(rng.integers(0, max(1, n - 2 * block_len + 1)))
            ones_in = [p for p in range(start_pos, start_pos + block_len) if h[p] == 1.0]
            if not ones_in:
                continue
            i = int(rng.choice(ones_in))
            zero_positions = [p for p in range(start_pos - block_len, start_pos) if 0 <= p < n and h[p] == 0.0]
            zero_positions += [p for p in range(start_pos + block_len, start_pos + 2 * block_len) if p < n and h[p] == 0.0]
            if not zero_positions:
                continue
            j = int(rng.choice(zero_positions))
            s_min, s_max, new_max, _ = swap_move_eval(h, i, j, corr, n)
            delta_c = new_max - current_max
            if delta_c < -1e-12 or (t > 1e-10 and rng.random() < np.exp(-max(delta_c, 0) / t)):
                if s_min <= s_max:
                    s_range = np.arange(s_min, s_max + 1)
                    k_range = s_range + half
                    delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
                    corr[k_range] += delta
                h[i], h[j] = h[j], h[i]
                current_max = new_max
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()

    return best_h, best_max


def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))


def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad


def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)


def constraint_jac(h):
    return np.ones(len(h))


def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0_arr = np.array(initial_h_values, dtype=float)
    except Exception:
        h0_arr = None

    best_h = None
    best_c5 = 1.0

    n_values = []
    if h0_arr is not None:
        n_values.append(len(h0_arr))
    for nv in [180, 178, 176, 182, 200]:
        if nv not in n_values:
            n_values.append(nv)

    for n in n_values:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.85:
            break
        n_abs_budget = min(budget_s * 0.18, budget_s - elapsed - 40)
        if n_abs_budget < 20:
            continue

        target_sum = n / 2.0

        if n == len(h0_arr) if h0_arr is not None else False:
            h_init = h0_arr.copy()
        else:
            h_init = (rng.random(n) < 0.5).astype(float)

        h_bin = to_binary(h_init)
        h_bin = project_onto_constraint(h_bin, target_sum)
        c5_bin = compute_c5(h_bin)

        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

        t_remaining = n_abs_budget - (time.time() - start_time)
        if t_remaining < 5:
            continue

        h_cur = h_bin.copy()
        c5_cur = c5_bin

        t_sa = min(t_remaining * 0.4, 120)
        if t_sa > 5:
            h_cur, max_val = iterative_greedy_sa(h_cur, t_sa, rng)
            c5_cur = (2.0 / n) * max_val
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()

        t_remaining = n_abs_budget - (time.time() - start_time)
        if t_remaining < 5:
            continue

        n_rounds = max(3, int(t_remaining * 0.3 / 5))
        for r in range(n_rounds):
            elapsed = time.time() - start_time
            if elapsed > budget_s * 0.85:
                break
            t_round = min(40, t_remaining / max(n_rounds - r, 1))
            if t_round < 2:
                continue

            strategy = r % 4
            if strategy == 0:
                h_st = to_binary(h_cur)
            elif strategy == 1:
                h_pert = h_cur + 0.06 * rng.standard_normal(n)
                h_pert = project_onto_constraint(h_pert, target_sum)
                h_st = to_binary(h_pert)
            elif strategy == 2:
                h_st = h_cur.copy()
                n_flip = int(rng.integers(4, max(5, n // 8)))
                ones_idx = np.where(h_st == 1.0)[0]
                zeros_idx = np.where(h_st == 0.0)[0]
                for _ in range(n_flip):
                    if len(ones_idx) == 0 or len(zeros_idx) == 0:
                        break
                    a = int(ones_idx[rng.integers(len(ones_idx))])
                    b = int(zeros_idx[rng.integers(len(zeros_idx))])
                    h_st[a], h_st[b] = h_st[b], h_st[a]
                    ones_idx = np.where(h_st == 1.0)[0]
                    zeros_idx = np.where(h_st == 0.0)[0]
            else:
                alpha = rng.uniform(0.3, 0.6)
                h_rand = (rng.random(n) < 0.5).astype(float)
                h_st = to_binary(alpha * h_cur + (1 - alpha) * h_rand)

            if r % 2 == 0:
                h_try, max_val = block_move_sa(h_st, int(t_round * 3000), 0.2, 0.01, rng)
            else:
                h_try, max_val = iterative_greedy_sa(h_st, t_round, rng)
            c5_try = (2.0 / n) * max_val

            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
                h_cur = h_try.copy()
                c5_cur = c5_try

        t_remaining = n_abs_budget - (time.time() - start_time)
        if t_remaining > 10:
            h_pol = h_cur + 0.01 * rng.standard_normal(n)
            h_pol = project_onto_constraint(h_pol, target_sum)
            h_pol, c5_pol = slsqp_optimize(h_pol, 0.001, 150)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    elapsed = time.time() - start_time
    if elapsed < budget_s - 30 and best_h is not None:
        n_best = len(best_h)
        target_sum = n_best / 2.0
        h_final = best_h + 0.005 * rng.standard_normal(n_best)
        h_final = project_onto_constraint(h_final, target_sum)
        for eps in [0.0005, 0.0001, 0.00003]:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 20:
                break
            h_final, c5_f = slsqp_optimize(h_final, eps, 200)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
