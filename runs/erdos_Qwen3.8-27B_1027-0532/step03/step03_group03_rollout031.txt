import time

_CONS_N = 0
_CONS_TARGET = 0.0

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(20):
        s = np.sum(h)
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def cons_eq_fun(x):
    return np.sum(x) - _CONS_TARGET

def cons_eq_jac(x):
    return np.ones(_CONS_N)

def soft_max_obj(x, alpha):
    corr = compute_corr_array(x)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / alpha)

def soft_max_grad(x, alpha):
    corr = compute_corr_array(x)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    weights = exp_vals / total
    n = len(x)
    # Compute weighted sum of gradients of each correlation
    # corr[k] = sum_j h[j] * (1-h[j + k - (n-1)])
    # d(corr[k])/d(h[i]) = (1-h[i+lag]) if i+lag valid, minus h[i-lag] if i-lag valid, where lag = k-(n-1)
    grad = np.zeros(n)
    one_minus_h = 1.0 - x
    for ki in range(len(corr)):
        w = weights[ki]
        if w < 1e-14:
            continue

            upper = n - lag

            # For second term: i ranges lag to n-1, j=i-lag ranges 0 to n-1-lag

def subgradient_step(h, target_sum, lr, topk, alpha, momentum, v, seed):
    n = len(h)
    corr = compute_corr_array(h)
    # Get top k correlations
    top_idx = np.argpartition(corr, -topk)[-topk:]
    top_vals = corr[top_idx]
    w = np.exp(alpha * (top_vals - np.max(top_vals)))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(topk) / topk
    else:
        w = w / w_sum
    # Compute gradient
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(top_idx)):
        w_i = w[i]
        ki = top_idx[i]
        lag = ki - (n - 1)
        if lag >= 0:
            upper = n - lag
            grad[:upper] += w_i * one_minus_h[lag:]
            grad[lag:] -= w_i * h[:upper]
        else:
            m = -lag
            upper = n - m
            grad[m:] += w_i * one_minus_h[:upper]
            grad[:upper] -= w_i * h[m:]
    v = momentum * v + (1.0 - momentum) * grad
    h_new = h - lr * v
    h_new = project_feasible(h_new, target_sum)
    return h_new, v

                        j = np.argmin(h_try)
                    else:

                if sc < best_score - 1e-15:
                    h = h_try.copy()
                    best_score = sc
                    improved = True
        if not improved:

            delta *= 0.5

    return h, best_score

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(30):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = int(mask.sum())
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def corr_gradient(h):
    n = len(h)
    grad_all = np.zeros((n, 2 * n - 1))
    one_minus_h = 1.0 - h
    for ki in range(2 * n - 1):
        lag = ki - (n - 1)
        if lag >= 0:
            upper = n - lag
            grad_all[:upper, ki] = one_minus_h[lag:]
            grad_all[lag:, ki] -= h[:upper]
        else:
            m = -lag
            upper = n - m
            grad_all[m:, ki] = one_minus_h[:upper]
            grad_all[:upper, ki] -= h[m:]

                upper = n - lag
                grad[:upper] += wi * one_minus_h[lag:]
                grad[lag:] -= wi * h[:upper]
            else:
                m = -lag
                upper = n - m
                grad[m:] += wi * one_minus_h[:upper]
                grad[:upper] -= wi * h[m:]
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()

    def _remaining():
        return budget_s - 20.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0
    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    rng = np.random.RandomState(seed)

    # Phase 1: Subgradient with diverse configs
    subgrad_configs = [
        (5000, 0.12, 5, 30, 0.04, 0.015, 400, 0.91),
        (5000, 0.06, 8, 50, 0.02, 0.01, 350, 0.87),
        (4000, 0.20, 3, 15, 0.08, 0.03, 300, 0.92),
        (4000, 0.04, 12, 80, 0.00, 0.00, 250, 0.82),
        (4000, 0.10, 6, 40, 0.05, 0.02, 300, 0.89),
    ]

    for i, cfg in enumerate(subgrad_configs):
        if _remaining() < 100:
            break
        h_try = best_h + rng.randn(n_init) * 0.012
        h_try = project_feasible(h_try, target_init)
        alloc = min(_remaining() - 80, 200)
        h_new, score = subgradient_method(
            h_try, target_init,
            max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
            alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
            patience=cfg[6], momentum=cfg[7], seed=seed + i * 97,
            time_limit=alloc)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 2: Projected gradient with BB step size
    if _remaining() > 120:
        alloc = min(_remaining() - 100, 150)
        h_pg, sc_pg = projected_gradient_bb(
            best_h, target_init, alpha=200, max_iter=8000,
            time_limit=alloc, seed=seed + 500)
        if sc_pg < best_score:
            best_score = sc_pg
            best_h = h_pg

    # Phase 3: SLSQP with increasing alpha
    if _remaining() > 150:
        alloc = min(_remaining() - 100, 250)
        alphas = [100, 300, 800, 2000, 5000]
        h_ref, sc_ref = slsqp_refine(best_h, target_init, alphas, 8000, 1e-16, alloc)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 4: More subgradient from improved start
    subgrad_configs2 = [
        (6000, 0.04, 5, 40, 0.01, 0.008, 500, 0.92),
        (5000, 0.08, 3, 20, 0.06, 0.02, 400, 0.90),
        (5000, 0.03, 10, 60, 0.00, 0.00, 350, 0.85),
    ]
    for i, cfg in enumerate(subgrad_configs2):
        if _remaining() < 80:
            break
        h_try = best_h + rng.randn(n_init) * 0.005
        h_try = project_feasible(h_try, target_init)
        alloc = min(_remaining() - 60, 150)
        h_new, score = subgradient_method(
            h_try, target_init,
            max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
            alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
            patience=cfg[6], momentum=cfg[7], seed=seed + 1000 + i * 97,
            time_limit=alloc)
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 5: SLSQP high alpha
    if _remaining() > 120:
        alloc = min(_remaining() - 60, 200)
        alphas_hi = [500, 1500, 4000, 8000]
        h_ref, sc_ref = slsqp_refine(best_h, target_init, alphas_hi, 10000, 1e-16, alloc)
        if sc_ref < best_score:
            best_score = sc_ref
            best_h = h_ref

    # Phase 6: Try different resolutions
    for n_try in [350, 500, 600]:
        if _remaining() < 120:
            break
        target_try = n_try / 2.0
        h_interp = interpolate_h(best_h, n_try, target_try)
        alloc_sub = min(_remaining() - 80, 120)
        h_new, score = subgradient_method(
            h_interp, target_try,
            max_iter=3000, lr_init=0.08, topk=5,
            alpha=30, noise_prob=0.03, noise_scale=0.012,
            patience=250, momentum=0.90, seed=seed + 2000 + n_try,
            time_limit=alloc_sub)
        if _remaining() > 60:
            alloc_sls = min(_remaining() - 40, 80)
            h_ref, sc_ref = slsqp_refine(h_new, target_try, [300, 1000, 3000], 5000, 1e-16, alloc_sls)
            if sc_ref < score:
                score = sc_ref
                h_new = h_ref
        if score < best_score:
            best_score = score
            best_h = h_new

    # Phase 7: Coordinate polish
    if _remaining() > 60:
        alloc = min(_remaining() - 30, 50)
        best_h, best_score = coordinate_polish(
            best_h, len(best_h) / 2.0, delta=0.002,
            max_sweeps=50, time_limit=alloc)

    # Phase 8: Final SLSQP
    if _remaining() > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        alloc = min(_remaining() - 20, 60)
        alphas_final = [2000, 5000, 10000]
        h_final, sc_final = slsqp_refine(best_h, target_use, alphas_final, 8000, 1e-16, alloc)
        if sc_final < best_score:
            best_score = sc_final
            best_h = h_final

    # Final projection and evaluation
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, n_final
