            best_c5 = c5_l
            best_h = h_l.copy()
            best_n = n
        h_cd = _coord_descent_binary(h_l, target, max_rounds=8, deadline=phase1_deadline - 3)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            best_n = n

    # Phase 2: Basin hopping on the best solution
    n_best = best_n
    target = n_best / 2.0
    phase2_deadline = deadline - int(0.30 * (deadline - t_start))

    h_bh, c5_bh = _basin_hopping(best_h, n_best, target, phase2_deadline, rng, n_iters=80)
    if c5_bh < best_c5:
        best_c5 = c5_bh
        best_h = h_bh.copy()

    # Phase 2b: DE local search
    h_de, c5_de = _de_local(best_h, n_best, target, pop_size=14, n_gen=35,
                             deadline=phase2_deadline, rng=rng)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Phase 3: Intensive local refinement
    phase3_deadline = deadline - 60
    h_curr = best_h.copy()
    n_curr = n_best
    targ_curr = n_curr / 2.0

    for round_i in range(30):
        if time.time() > phase3_deadline:
            break
        alpha = 40 * (2.0 ** min(round_i, 8))
        mu = 2.0 * (1.5 ** min(round_i, 10))
        h_l, c5_l = _lbgfb_refine(h_curr, alpha_start=alpha, mu_start=mu,
                                   n_rounds=3, deadline=phase3_deadline)
        if c5_l < best_c5:
            best_c5 = c5_l
            best_h = h_l.copy()
        h_curr = h_l
        h_cd = _coord_descent_binary(h_curr, targ_curr, max_rounds=5, deadline=phase3_deadline)
        c5_cd = _c5_of(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
        h_curr = h_cd
        if round_i % 3 == 2:
            h_n, c5_n = _nesterov_smooth(h_curr, n_curr, targ_curr, n_iters=100,
                                          alpha0=0.3, sg_m=3, deadline=phase3_deadline)
            if c5_n < best_c5:
                best_c5 = c5_n
                best_h = h_n.copy()
            h_curr = h_n

    # Phase 4: Final polish
    h_final = best_h.copy()
    n_final = best_n
    t_final = n_final / 2.0

    h_f = _coord_descent_frac(h_final, t_final, max_rounds=10, deadline=deadline - 15)
    c5_f = _c5_of(h_f)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
        h_final = h_f

    h_f2 = _coord_descent_binary(h_final, t_final, max_rounds=25, deadline=deadline - 10)
    c5_f2 = _c5_of(h_f2)
    if c5_f2 < best_c5:
        best_c5 = c5_f2
        best_h = h_f2.copy()

    h_f3 = _lbgfb_refine(h_final, alpha_start=2000, mu_start=100.0, n_rounds=4,
                          deadline=deadline - 5)
    c5_f3 = _c5_of(h_f3)
    if c5_f3 < best_c5:
        best_c5 = c5_f3
        best_h = h_f3.copy()

    h_f4 = _coord_descent_binary(best_h, t_final, max_rounds=15, deadline=deadline - 3)
    c5_f4 = _c5_of(h_f4)
    if c5_f4 < best_c5:
        best_c5 = c5_f4
        best_h = h_f4.copy()

    h_f5 = _coord_descent_frac(best_h, t_final, max_rounds=6, deadline=deadline - 2)
    c5_f5 = _c5_of(h_f5)
    if c5_f5 < best_c5:
        best_c5 = c5_f5
        best_h = h_f5.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
