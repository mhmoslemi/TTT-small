- h[j] as h[i-r] where i=j+r (need r <= j+r <= n-1, i.e., j <= n-1-r = m-1): contributes -h[j+r]

So:
- For j in [r, n-1]: grad[j] += dx*(1-h[j-r])
- For j in [0, m-1]: grad[j] -= dx*h[j+r]

In code:
- For j in [r, n-1]: j-r is in [0, n-1-r] = [0, m-1]. So h[j-r] = h[0:m] indexed at j-r.
  `grad[r:] += dx * (1-h[:m])` → grad[r+j] += dx*(1-h[j]) for j=0..m-1. So grad[i] += dx*(1-h[i-r]) for i=r..r+m-1=n-1. ✓
  
- For j in [0, m-1]: h[j+r] = h[r:r+m] indexed at j.
  `grad[:m] -= dx * h[r:r+m]` → grad[j] -= dx*h[r+j] for j=0..m-1. So grad[i] -= dx*h[i+r] for i=0..m-1. ✓

Let me verify with example: n=4, s=-1 (r=1), m=3, h=[a,b,c,d].
corr_{-1} = dx * (h[1]*(1-h[0]) + h[2]*(1-h[1]) + h[3]*(1-h[2]))
           = dx * (b*(1-a) + c*(1-b) + d*(1-c))

d/dh[0] = -dx*b (from b*(1-h[0]))
d/dh[1] = dx*(1-a) - dx*c (from b*(1-a) and c*(1-b))
d/dh[2] = dx*(1-b) - dx*d
d/dh[3] = dx*(1-c)

My code:
- `grad[r:] += dx*(1-h[:m])` = `grad[1:] += dx*(1-h[:3])` = grad[1]+=dx*(1-a), grad[2]+=dx*(1-b), grad[3]+=dx*(1-c)
- `grad[:m] -= dx*h[r:r+m]` = `grad[:3] -= dx*h[1:4]` = grad[0]-=dx*b, grad[1]-=dx*c, grad[2]-=dx*d

Total:
- grad[0] = -dx*b ✓
- grad[1] = dx*(1-a) - dx*c = dx*(1-a-c) ✓
- grad[2] = dx*(1-b) - dx*d = dx*(1-b-d) ✓
- grad[3] = dx*(1-c) ✓

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += wk * dx * (1.0 - h[s:s+m])
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[r:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[r:r+m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def adam_optimize(h_init, eps, lr, n_steps, beta1, beta2, time_limit, start_time):
    n = len(h_init)
    target_sum = n / 2.0
    h = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 0
    
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t += 1
        grad = smooth_max_grad(h, eps)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
