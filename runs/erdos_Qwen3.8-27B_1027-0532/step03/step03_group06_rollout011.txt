    x0 = h[i]
    
    # Compute slopes s_k for all k
    # s_k = dx * [(1-h[i+k]) * 1_{0<=i+k<n} - h[i-k] * 1_{0<=i-k<n}]
    # k ranges from -(n-1) to (n-1), total 2n-1 shifts
    
    ks = np.arange(-(n-1), n)
    ip = i + ks  # i + k
    im = i - ks  # i - k
    
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    
    # (1-h[i+k]) for valid i+k
    one_minus_h = np.zeros(2*n-1)
    one_minus_h[valid_p] = 1.0 - h[ip[valid_p]]
    
    h_im = np.zeros(2*n-1)
    h_im[valid_m] = h[im[valid_m]]
    
    s = dx * (one_minus_h - h_im)
    
    # Compute c_k = O_k - x0 * s_k
    # O_k = sum_j h[j]*(1-h[j+k])*dx
    # This is just the correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    O = corr * dx
    
    c = O - x0 * s
    
    # Minimize max_k (c_k + s_k * x) over x in [0,1] using ternary search
    a, b = 0.0, 1.0
    for _ in range(50):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + s * m1)
        f2 = np.max(c + s * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    return (a + b) / 2.0

def coord_optimal_value(h, i, n, corr_dx):
    """Find optimal value of h[i] given all other coordinates fixed.
    corr_dx: precomputed correlation * dx (length 2n-1)"""
    dx = 2.0 / n
    x0 = h[i]
    
    # shifts from -(n-1) to (n-1)
    shifts = np.arange(-(n-1), n)  # length 2n-1
    ip = i + shifts
    im = i - shifts
    
    valid_p = (ip >= 0) & (ip < n)
    valid_m = (im >= 0) & (im < n)
    
    # s = dx * [(1-h[i+k]) * 1_{valid} - h[i-k] * 1_{valid}]
    term_p = np.zeros(2*n-1)
    term_p[valid_p] = 1.0 - h[ip[valid_p]]
    term_m = np.zeros(2*n-1)
    term_m[valid_m] = h[im[valid_m]]
    s = dx * (term_p - term_m)
    
    # c = corr_dx - x0 * s
    c = corr_dx - x0 * s
    
    # Minimize max(c + s*x) over x in [0,1] using ternary search
    a, b = 0.0, 1.0
    for _ in range(40):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(c + s * m1)
        f2 = np.max(c + s * m2)
        if f1 < f2:
            b = m2
        else:
            a = m1
    
    return (a + b) / 2.0

    if cd_score2 < best_score:
        best_score = cd_score2
        best_h = h_work

    for alpha_nes in [200, 500]:
        if tleft() < 60:
            break
        h_nes, nes_score = nesterov_stage(best_h, n_target, target_sum,
                                          alpha_nes, maxiter=400, time_limit_fn=tleft)
        if nes_score < best_score:
            best_score = nes_score
            best_h = h_nes

    h_work, cd_score3 = coordinate_descent(best_h, n_target, max_sweeps=15, time_limit_fn=tleft)
    if cd_score3 < best_score:
        best_score = cd_score3
        best_h = h_work

    for attempt in range(15):
        if tleft() < 80:
            break
        ptype = attempt % 5
        if ptype == 0:
            amp = 0.005 + 0.003 * attempt
            h_pert = project_box_sum(best_h + rng.randn(n_target) * amp, target_sum)
        elif ptype == 1:
            h_pert = best_h.copy()
            seg_len = rng.randint(10, max(11, n_target // 10))
            a = rng.randint(0, n_target - seg_len)
            b = rng.randint(0, n_target - seg_len)
            if abs(a - b) > seg_len:
                tmp = h_pert[a:a + seg_len].copy()
                h_pert[a:a + seg_len] = h_pert[b:b + seg_len]
                h_pert[b:b + seg_len] = tmp
            h_pert = project_box_sum(h_pert, target_sum)
        elif ptype == 2:
            shift_amt = rng.randint(2, max(3, n_target // 10))
            h_pert = np.roll(best_h, shift_amt)
            h_pert = project_box_sum(h_pert, target_sum)
        elif ptype == 3:
            edge = rng.randint(0, n_target)
            width = rng.randint(5, max(6, n_target // 20))
            h_pert = best_h.copy()
            lo = max(0, edge - width)
            hi = min(n_target, edge + width + 1)
            h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
            h_pert = project_box_sum(h_pert, target_sum)
        else:
            h_pert = best_h.copy()
            n_flip = rng.randint(5, 20)
            idx = rng.choice(n_target, size=n_flip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = project_box_sum(h_pert, target_sum)

        h_ref, sc = coordinate_descent(h_pert, n_target, max_sweeps=10, time_limit_fn=tleft)
        h_ref, sc = run_full_refine(h_ref, n_target, target_sum,
                                    [50, 200, 600], 200, 1e-13, tleft)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    if tleft() > 100:
        h_hires = interpolate_to(best_h, 1000, 500.0)
        h_hires, sc_h = coordinate_descent(h_hires, 1000, max_sweeps=10, time_limit_fn=tleft)
        h_hires, sc_h = run_full_refine(h_hires, 1000, 500.0,
                                        [20, 80, 300, 800], 200, 1e-14, tleft)
        h_hires, sc_h2 = coordinate_descent(h_hires, 1000, max_sweeps=10, time_limit_fn=tleft)
        sc_final_hires = min(sc_h, sc_h2)
        if sc_final_hires < best_score:
            best_score = sc_final_hires
            best_h = h_hires

    if tleft() > 50:
        h_final = project_box_sum(best_h, len(best_h) / 2.0)
        final_score = compute_c5(h_final)
    else:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, len(h_final)
