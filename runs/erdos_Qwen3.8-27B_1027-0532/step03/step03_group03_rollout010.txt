def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
        if len(idx) == 0:

        h[idx] += diff / len(idx)
    else:

        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lag_gradient(h, lag_offset, n):
    """Gradient of corr[lag_offset] w.r.t. h.
    lag_offset is the index into the full correlation array (0 to 2n-2).
    The actual lag is lag_offset - (n-1).
    """
    lag = lag_offset - (n - 1)
    grad = np.zeros(n)
    omh = 1.0 - h
    if lag >= 0:
        if lag < n:
            grad[:n-lag] += omh[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += omh[m:]
            grad[m:] -= h[:n-m]
    return grad

def weighted_lag_gradient(h, lag_offsets, weights, n):
    grad = np.zeros(n)
    for i, lo in enumerate(lag_offsets):
        if weights[i] < 1e-15:
            continue
        grad += weights[i] * lag_gradient(h, lo, n)
    return grad

    # Use top-k for efficiency

        top_corr = corr[top_idx]
        c_max = np.max(top_corr)
        exp_vals = np.exp(alpha * (top_corr - c_max))
        total = np.sum(exp_vals)
        obj = (c_max + np.log(total) / alpha) * (2.0 / len(h))
        weights = exp_vals / total
        grad = weighted_lag_gradient(h, top_idx, weights, len(h)) * (2.0 / len(h))

        grad = weighted_lag_gradient(h, np.arange(len(corr)), weights, len(h)) * (2.0 / len(h))
    return obj, grad

            stagnant = 0
        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            return h
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lag_gradient(h, lag_offset, n):
    lag = lag_offset - (n - 1)
    grad = np.zeros(n)
    omh = 1.0 - h
    if lag >= 0:
        if lag < n:
            grad[:n-lag] += omh[lag:]
            grad[lag:] -= h[:n-lag]
    else:
        m = -lag
        if m < n:
            grad[:n-m] += omh[m:]
            grad[m:] -= h[:n-m]
    return grad

def soft_max_obj_and_grad(h, alpha, topk):
    n = len(h)
    corr = get_corr(h)
    nc = len(corr)
    if topk < nc:
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_corr = corr[top_idx]
    else:
        top_idx = np.arange(nc)
        top_corr = corr
    c_max = np.max(top_corr)
    shifted = np.clip(alpha * (top_corr - c_max), -500.0, 0.0)
    exp_vals = np.exp(shifted)
    total = np.sum(exp_vals)
    obj = (c_max + np.log(total) / alpha) * (2.0 / n)
    weights = exp_vals / total
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if weights[i] > 1e-15:
            grad += weights[i] * lag_gradient(h, top_idx[i], n)
    grad *= (2.0 / n)
    return obj, grad

                  topk, beta1, beta2, eps, patience, cooldown_interval,
                  cooldown_scale, seed):

    n = len(h_init)
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    stagnant = 0
    t = 0
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        alpha = alpha_init * (alpha_end / alpha_init) ** frac
        
        _, grad = soft_max_obj_and_grad(h, alpha, topk)
        
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, target_sum)
        
        score = compute_c5(h_new)
        if score < best_score - 1e-12:
            best_score = score
            best_h = h_new.copy()
            stagnant = 0
        else:
            stagnant += 1

            h = best_h + rng.randn(n) * cooldown_scale
            h = project_feasible(h, target_sum)
            m *= 0.5
            v *= 0.5
            stagnant = 0
        
        if stagnant > patience:
            break
    
    return best_h, best_score

        if sc < best_score:
            best_score = sc
            best_h = h_a

    # Phase 2: SLSQP polish
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 120:
        h_ref = slp_refine(best_h, target_init, [50, 150, 400], 600, 1e-12, 25)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 3: Focused subgradient with different parameters
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 100:
        h_try = best_h + rng.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_s, sc_s = subgrad_focused(h_try, target_init, max_iter=3000,
                                     lr_init=0.08, topk=8, alpha=40,
                                     momentum=0.88, seed=seed + 500)
        if sc_s < best_score:
            best_score = sc_s
            best_h = h_s

    # Phase 4: Adam with higher alpha (sharper focus)
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 150:
        h_a2, sc2 = adam_optimize(best_h, target_init, n_iter=3000,
                                   alpha_init=20.0, alpha_end=500.0, lr=0.015,
                                   topk=15, beta1=0.9, beta2=0.999, eps=1e-8,
                                   patience=350, cooldown_interval=200,
                                   cooldown_scale=0.008, seed=seed + 1000)
        if sc2 < best_score:
            best_score = sc2
            best_h = h_a2

    # Phase 5: Try higher resolution
    for n_try in [500, 650]:
        remaining = budget_s - 20.0 - (time.time() - start_time)
        if remaining < 200:
            break
        target_try = n_try / 2.0
        h_interp = interp_to_resolution(best_h, n_try, target_try)
        h_a3, sc3 = adam_optimize(h_interp, target_try, n_iter=2500,
                                   alpha_init=5.0, alpha_end=120.0, lr=0.03,
                                   topk=25, beta1=0.9, beta2=0.999, eps=1e-8,
                                   patience=300, cooldown_interval=200,
                                   cooldown_scale=0.015, seed=seed + 2000 + n_try)
        if sc3 < best_score:
            best_score = sc3
            best_h = h_a3
        remaining = budget_s - 20.0 - (time.time() - start_time)
        if remaining > 100:
            h_ref2 = slp_refine(best_h, target_try, [80, 200, 500], 400, 1e-13, 20)
            sc = compute_c5(h_ref2)
            if sc < best_score:
                best_score = sc
                best_h = h_ref2

    # Phase 6: Multi-start at current best resolution
    n_use = len(best_h)
    target_use = n_use / 2.0
    for i in range(4):
        remaining = budget_s - 20.0 - (time.time() - start_time)
        if remaining < 80:
            break
        noise_scale = 0.008 + 0.005 * i
        h_try = best_h + rng.randn(n_use) * noise_scale
        h_try = project_feasible(h_try, target_use)
        h_a4, sc4 = adam_optimize(h_try, target_use, n_iter=1500,
                                   alpha_init=10.0, alpha_end=300.0, lr=0.02,
                                   topk=15, beta1=0.9, beta2=0.999, eps=1e-8,
                                   patience=250, cooldown_interval=150,
                                   cooldown_scale=0.005, seed=seed + 5000 + i * 77)
        if sc4 < best_score:
            best_score = sc4
            best_h = h_a4

    # Phase 7: Final focused subgradient
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 60:
        h_s2, sc_s2 = subgrad_focused(best_h, target_use, max_iter=2000,
                                       lr_init=0.03, topk=5, alpha=60,
                                       momentum=0.92, seed=seed + 9000)
        if sc_s2 < best_score:
            best_score = sc_s2
            best_h = h_s2

    # Phase 8: Final SLSQP polish
    remaining = budget_s - 20.0 - (time.time() - start_time)
    if remaining > 40:
        h_final = slp_refine(best_h, target_use, [300, 800, 1500], 300, 1e-14, 15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    h_out = project_feasible(best_h, target_use)
    final_score = compute_c5(h_out)
    return h_out, final_score, len(h_out)
