import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] = np.clip(h[idx] + adj, 0.0, 1.0)
    return h

def smooth_max_grad(h, alpha):
    """Compute smooth max of corr*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * dx
    m = np.max(scaled)
    exp_vals = np.exp(alpha * (scaled - m))
    # Avoid underflow/overflow issues
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    # Gradient
    one_minus_h = 1.0 - h
    first_sum = np.correlate(w, one_minus_h, mode='full')[n-1::-1]
    second_sum = np.convolve(h, w, mode='full')[n-1:2*n-1]
    grad = (first_sum - second_sum) * dx
    return F, grad

def lbfgs_refine(h_start, alpha, maxiter=200, ftol=1e-14):
    n = len(h_start)
    target = n / 2.0
    
    def obj(x):
        F, _ = smooth_max_grad(x, alpha)
        return F
    
    def jac(x):
        _, g = smooth_max_grad(x, alpha)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except:
        return h_start

def spsa_optimize(h_start, n_iter, c0=0.05, a0=0.3, kappa=0.6, alpha_gain=10, seed=42):
    n = len(h_start)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    
    h_best = h_start.copy()
    c5_best = compute_c5(h_best)
    
    for k in range(n_iter):
        a_k = a0 / (k + alpha_gain) ** kappa
        c_k = c0
        
        delta = rng.choice([-1.0, 1.0], size=n)
        
        h_plus = project_feasible(h_best + c_k * delta, target)
        h_minus = project_feasible(h_best - c_k * delta, target)
        
        f_plus = compute_c5(h_plus)
        f_minus = compute_c5(h_minus)
        
        # Gradient estimate
        g_est = (f_plus - f_minus) / (2 * c_k) * delta
        
        # Update
        h_new = project_feasible(h_best - a_k * g_est, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_best:
            c5_best = c5_new
            h_best = h_new
    
    return h_best, c5_best

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

    return F, grad

def project_feasible(h, target_sum):
