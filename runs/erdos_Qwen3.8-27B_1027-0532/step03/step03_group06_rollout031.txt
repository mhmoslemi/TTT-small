```python
import numpy as np
from scipy.optimize import minimize
import time

_alpha_current = 100.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def smooth_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    corr_shifted = corr - m
    exp_vals = np.exp(alpha * corr_shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        k_lo = max(-(n - 1), -i)
        k_hi = min(n - 1, n - 1 - i)
        if k_lo <= k_hi:
            js = i - np.arange(k_lo, k_hi + 1)
            first = np.dot(w[k_lo + n - 1: k_hi + n], 1.0 - h[js])
        else:
            first = 0.0
        if k_lo <= k_hi:
            js2 = i + np.arange(k_lo, k_hi + 1)
            second = np.dot(w[k_lo + n - 1: k_hi + n], h[js2])
        else:
            second = 0.0
        grad[i] = first - second
    return F, grad


def _smooth_obj(x):
    return smooth_obj_and_grad(x, _alpha_current)[0]


def _smooth_jac(x):
    return smooth_obj_and_grad(x, _alpha_current)[1]


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-14)[0]
        else:
            idx = np.where(h > 1e-14)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def lbfgs_refine(h_start, n, target, alpha, maxiter):
    global _alpha_current
    _alpha_current = alpha
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h_start.copy(), jac=_smooth_jac,
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start.copy()


def subgradient_polish(h_start, n, target, n_iter, momentum=0.9):
    h = h_start.copy()
    current_score = compute_c5(h)
    velocity = np.zeros(n)
    lr_schedule = [0.05, 0.03, 0.02, 0.01]
    seg_len = max(1, n_iter // len(lr_schedule))
    for it in range(n_iter):
        seg_idx = min(it // seg_len, len(lr_schedule) - 1)
        lr = lr_schedule[seg_idx] / (1.0 + 0.5 * (it / max(1, n_iter)))
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr)) - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            g = 0.0
            ik = i - k_star
            if 0 <= ik < n:
                g += (1.0 - h[ik])
            ik2 = i + k_star
            if 0 <= ik2 < n:
                g -= h[ik2]
            grad[i] = g
        velocity = momentum * velocity - lr * grad
        h_new = np.clip(h + velocity, 0.0, 1.0)
        h_new = project_feasible(h_new, target)
        new_score = compute_c5(h_new)
        if new_score < current_score:
            h = h_new
            current_score = new_score
        else:
            velocity = 0.5 * velocity
    return h, current_score


def coordinate_descent(h_start, n, target, n_passes, step_sizes=None):
    if step_sizes is None:
        step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002]
    h = h_start.copy()
    current_score = compute_c5(h)
    for p in range(n_passes):
        improved_any = False
        for i in range(n):
            for ds in step_sizes:
                for direction in [1.0, -1.0]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h[i] + direction * ds, 0.0, 1.0)
                    h_try = project_feasible(h_try, target)
                    sc = compute_c5(h_try)
                    if sc < current_score - 1e-14:
                        h = h_try
                        current_score = sc
                        improved_any = True
                        break
                if improved_any:
                    break
        if not improved_any:
            break
    return h, current_score


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```
