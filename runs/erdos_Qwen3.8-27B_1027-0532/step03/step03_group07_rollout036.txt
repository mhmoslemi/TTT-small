def _fix_sum(h, target_sum):

    if abs(diff) < 1e-12:

Let me also think about a key optimization: **the subgradient of the max function**.

The objective is f(h) = max_k corr_k(h), where corr_k(h) = sum_i h[i] * (1-h[i+k]) * dx.

The subgradient of f at h is the average of the gradients of all active corr_k (those achieving the max).

grad corr_k / grad h[i] = (1-h[i+k]) * dx if i+k is valid, plus -h[i-k] * dx if i-k is valid (from the second term in the product).

Wait, let me be more careful. corr_k(h) = sum_{valid i} h[i] * (1 - h[i+k]) * dx.

∂corr_k/∂h[j] = dx * [(1-h[j+k]) if j+k valid] - dx * [h[j-k] if j-k valid]

So the subgradient of f is:
∂f/∂h[j] = dx * sum_{k active} [(1-h[j+k]) * 1{j+k valid} - h[j-k] * 1{j-k valid}]

where "active" means corr_k = max(corr).

This gives us a direction to move h in the direction that reduces all active correlations simultaneously.

Let me use option 1: change h[j], then project. The projection uses a bisection on a threshold, which changes all values above/below the threshold by the same amount. This is a "soft" change that might not be ideal.

Alternative: change h[j] to new_val, then find the position k that needs the least adjustment to restore the sum, and adjust h[k] accordingly. This is more targeted.

```
def _adjust_sum_single(h, target_sum, j, new_val):
    """Set h[j] = new_val and adjust one other position to maintain sum."""
    old_val = h[j]
    diff = old_val - new_val  # amount we need to add back
    h[j] = new_val
    if abs(diff) < 1e-12:
        return h
    # Find position k ≠ j where we can add diff
    if diff > 0:
        # Need to increase some h[k] by diff
        k_candidates = np.where(h < 1.0 - 1e-6)[0]
        k_candidates = k_candidates[k_candidates != j]
        # Pick the one closest to 1.0 (least "disturbing")
        k = k_candidates[np.argmax(h[k_candidates])]
        h[k] = min(1.0, h[k] + diff)
    else:
        # Need to decrease some h[k] by |diff|
        k_candidates = np.where(h > 1e-6)[0]
        k_candidates = k_candidates[k_candidates != j]
        k = k_candidates[np.argmin(h[k_candidates])]
        h[k] = max(0.0, h[k] + diff)
    return h
```

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _equalizer_subgrad_step(h, n, target_sum, step_size):
    dx = 2.0 / n
    corr = _corr_of(h)
    max_val = corr.max()
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    for k_idx in active:
        offset = k_idx - (n - 1)
        for i in range(n):
            if 0 <= i + offset < n:
                grad[i] += (1.0 - h[i + offset]) * dx
            if 0 <= i - offset < n:
                grad[i] -= h[i - offset] * dx
    new_h = _project(h - step_size * grad, target_sum)
    return new_h, _c5_of(new_h)

def _binary_swap_search(h, n, deadline, rng, max_rounds=300):
    target_sum = n / 2.0
    h = (h >= 0.5).astype(float)
    h = _fix_sum(h, target_sum)
    max_val = float(_corr_of(h).max())
    best_h = h.copy()
    best_max = max_val
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = max_val
        best_i, best_j = -1, -1
        n_ones = len(ones)
        n_zeros = len(zeros)
        order_i = np.arange(n_ones)
        order_j = np.arange(n_zeros)
        rng.shuffle(order_i)
        rng.shuffle(order_j)
        for ia in order_i:
            i = ones[ia]
            for ib in order_j:
                j = zeros[ib]
                h[i] = 0.0
                h[j] = 1.0
                nm = float(_corr_of(h).max())
                h[i] = 1.0
                h[j] = 0.0
                if nm < best_improve - 1e-15:
                    best_improve = nm
                    best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            max_val = best_improve
            if max_val < best_max:
                best_max = max_val
                best_h = h.copy()
        else:
            break
    return best_h, best_max * (2.0 / n)

def _coord_descent_multi(h, n, target_sum, deadline, levels=None, max_rounds=200):
    if levels is None:
        levels = np.array([0.0, 0.1, 0.2, 0.25, 0.333, 0.4, 0.5, 0.6, 0.667,
                           0.7, 0.75, 0.8, 0.9, 1.0])
    h = h.copy()
    h = _fix_sum(h, target_sum)
    best_c5 = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old_val = h[j]
            best_local = _c5_of(h)
            best_v = old_val
            for v in levels:
                if abs(v - old_val) < 1e-9:
                    continue
                h[j] = v
                h_t = _fix_sum(h, target_sum)
                c = _c5_of(h_t)
                if c < best_local - 1e-14:
                    best_local = c
                    best_v = v
                    h[:] = h_t
            if best_v != old_val:
                improved = True
            else:
                h[j] = old_val
        if best_local < best_c5 - 1e-14:
            best_c5 = best_local
        if not improved:
            break
    return h, best_c5

def _nesterov_subgrad(h0, n, target_sum, n_iters, step0, deadline=None):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        step = step0 / np.sqrt(float(it + 1) + 1.0)
        new_x, cv = _equalizer_subgrad_step(y, n, target_sum, step)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = new_x + ((t - 1.0) / t_new) * (new_x - x)
        y_new = _project(y_new, target_sum)
        x = new_x
        y = y_new
        t = t_new
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        scaled = alpha * (corr - m)
        scaled = np.clip(scaled, -500, 500)
        return (m + np.log(np.sum(np.exp(scaled))) / alpha) * (2.0 / n)
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(corr)
        scaled = alpha * (corr - m)
        scaled = np.clip(scaled, -500, 500)
        w = np.exp(scaled)
        w = w / w.sum()
        dx = 2.0 / n
        grad = np.zeros(n)
        for k_idx in range(len(corr)):
            if w[k_idx] > 1e-10:
                grad += w[k_idx] * _grad_corr_term(x, k_idx, dx)
        return grad
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _kick_and_search(h, n, deadline, rng, n_kicks=10, rounds_per_kick=50):
    target_sum = n / 2.0
    best_h = (h >= 0.5).astype(float)
    best_h = _fix_sum(best_h, target_sum)
    best_c5 = _c5_of(best_h)
    for k in range(n_kicks):
        if time.time() > deadline:
            break
        h_kick = best_h.copy()
        n_flips = rng.randint(2, min(8, n // 10) + 1)
        ones = np.where(h_kick == 1)[0]
        zeros = np.where(h_kick == 0)[0]
        for _ in range(n_flips):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_kick[i] = 0.0
            h_kick[j] = 1.0
            ones = np.where(h_kick == 1)[0]
            zeros = np.where(h_kick == 0)[0]
        h_res, c5_res = _binary_swap_search(h_kick, n, deadline, rng,
                                             max_rounds=rounds_per_kick)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()
    return best_h, best_c5
