```python
def compute_lse_gradient(x, alpha, threshold=1e-8):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for m in range(0, n):
        w_pos = weights[n - 1 + m]
        w_neg = weights[n - 1 - m] if m > 0 else 0.0
        
        if m == 0:
            if w_pos > threshold:
                grad[:] += w_pos * (1.0 - 2.0 * x)
        else:
            if w_pos > threshold:
                grad[:n-m] += w_pos * (1.0 - x[m:])
                grad[m:] -= w_pos * x[:n-m]
            if w_neg > threshold:
                grad[m:] += w_neg * (1.0 - x[:n-m])
                grad[:n-m] -= w_neg * x[m:]
    
    return dx * grad
```

This should be faster when alpha is large (few active lags).

**Projected gradient method with LSE:**
1. Compute gradient of LSE
2. Take a step in the negative gradient direction
3. Project back to the feasible set (0 ≤ h ≤ 1, ∑h = n/2)

The projection onto {h : 0 ≤ h ≤ 1, ∑h = target} can be done in O(n log n) using a bisection on the Lagrange multiplier.

```python
def project_to_simplex(h, target):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target}."""
    n = len(h)
    # Use bisection on the threshold
    lo, hi = 0.0, 1.0
    
    def sum_at_threshold(t):
        return np.sum(np.clip(h - t, 0.0, 1.0))
    
    # We want: sum(clip(h - t, 0, 1)) = target
    # This is monotonically decreasing in t
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_at_threshold(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2
    return np.clip(h - t, 0.0, 1.0)
```

Wait, this is the projection onto the simplex {x : x ≥ 0, ∑x = target}. But we also need x ≤ 1. The projection onto {x : 0 ≤ x ≤ 1, ∑x = target} is:
x_i = clip(h_i - t, 0, 1) where t is chosen so that ∑x_i = target.

This is correct and can be found by bisection since ∑clip(h_i - t, 0, 1) is monotonically decreasing in t.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0.0, 1.0)

def compute_lse_gradient(x, alpha, threshold=1e-10):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    weights = np.exp(shifted)
    total_w = np.sum(weights)
    if total_w < 1e-15:
        return np.zeros(n)
    weights /= total_w
    
    grad = np.zeros(n)
    for m in range(0, n):
        w_pos = weights[n - 1 + m]
        if m == 0:
            if w_pos > threshold:
                grad[:] += w_pos * (1.0 - 2.0 * x)
        else:
            w_neg = weights[n - 1 - m]
            if w_pos > threshold:
                grad[:n-m] += w_pos * (1.0 - x[m:])
                grad[m:] -= w_pos * x[:n-m]
            if w_neg > threshold:
                grad[m:] += w_neg * (1.0 - x[:n-m])
                grad[:n-m] -= w_neg * x[m:]
    
    return dx * grad

def lse_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(dx * (c_max + np.log(np.sum(np.exp(shifted))) / alpha))

def projected_gradient_phase(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()

        alphas = [30, 100, 300, 1000]
    
    best = h.copy()
    best_score = compute_c5(h)
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        idx = alphas.index(alpha)
        t_per = remaining / (len(alphas) - idx)
        
        h_curr = best.copy()
        eta = 0.01
        momentum = np.zeros(n)
        beta_m = 0.9
        no_improve = 0
        
        while time.time() - start < time_limit:
            grad = compute_lse_gradient(h_curr, alpha)
            # Project gradient to constraint subspace
            rg = grad - np.mean(grad)
            
            # Momentum
            momentum = beta_m * momentum - eta * rg
            h_new = h_curr + momentum
            h_new = project_feasible(h_new, target)
            
            score = compute_c5(h_new)
            if score < best_score - 1e-13:
                best_score = score
                best = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 100:
                    eta *= 0.7
                    beta_m = 0.5
                    momentum = np.zeros(n)
                    h_curr = best.copy()
                    no_improve = 0
                    if eta < 1e-6:
                        break
            
            h_curr = h_new
    
    return best, best_score

def slsqp_phase(h, target, time_limit, alphas=None):
    n = len(h)
    start = time.time()
    if alphas is None:
        alphas = [200, 1000]
    best = h.copy()
    best_score = compute_c5(h)
    bounds = [(0.0, 1.0)] * n
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        remaining = time_limit - (time.time() - start)
        idx = alphas.index(alpha)
        t_per = remaining / (len(alphas) - idx)
        maxiter = max(30, int(t_per / 2.0))
        
        def obj(x, a=alpha):
            return lse_obj(x, a)
        
        def grad(x, a=alpha):
            return compute_lse_gradient(x, a, threshold=1e-8)
        
        def cfun(x):
            return np.sum(x) - target
        
        def cjac(x):
            return np.ones(n)
        
        cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
        
        try:
            res = minimize(obj, best, jac=grad, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            s = compute_c5(h_new)
            if s < best_score:
                best_score = s
                best = h_new
        except Exception:
            pass
    
    return best, best_score
