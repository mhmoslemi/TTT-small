```python
def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full') * dx  # actual c_k values
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    log_sum = np.log(np.sum(np.exp(shifted)))
    obj = c_max + log_sum / alpha
    
    # weights p_k
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # gradient
    # c_k = sum_i h[i](1-h[i+k]) * dx
    # dc_k/dh[j] = dx * [(1-h[j+k]) - h[j-k]] (boundary terms are 0)
    # grad[j] = sum_k p_k * dc_k/dh[j] = dx * sum_k p_k * [(1-h[j+k]) - h[j-k]]
    
    n_corr = 2*n - 1
    s_arr = np.arange(n_corr) - (n-1)
    
    grad = np.zeros(n)
    for k_idx in range(n_corr):
        if p[k_idx] < 1e-15:
            continue
        s = s_arr[k_idx]
        # dc_k/dh[j] involves h[j+s] and h[j-s]
        # For j where j+s is in [0,n): contribution is (1-h[j+s])
        # For j where j-s is in [0,n): contribution is -h[j-s]
        
        # First term: (1-h[j+s]) for j in [max(0,-s), min(n, n-s))
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_end > j_start:
            grad[j_start:j_end] += p[k_idx] * (1.0 - h[j_start+s:j_end+s])
        
        # Second term: -h[j-s] for j in [max(0,s), min(n, n+s))
        j_start2 = max(0, s)
        j_end2 = min(n, n + s)
        if j_end2 > j_start2:
            grad[j_start2:j_end2] += p[k_idx] * (-h[j_start2-s:j_end2-s])
    
    grad *= dx
    return obj, grad
```

Let me also think about the bisection in coordinate descent. The function we're minimizing is:
f(v) = max_k [A[k] * v + B[k]]

where A and B are fixed vectors (for a given coordinate j). This is a convex piecewise-linear function of v. The minimum occurs where the subdifferential contains 0, i.e., where there exist k1, k2 with A[k1] > 0 and A[k2] < 0 such that both are "active" (achieve the max) at the optimum.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    for _ in range(100):
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
    return h

def smooth_obj_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(corr))
    shifted = alpha * (corr - c_max)
    exp_s = np.exp(shifted)
    sum_exp = float(np.sum(exp_s))
    obj = c_max + np.log(sum_exp) / alpha
    p = exp_s / sum_exp
    n_corr = 2 * n - 1
    s_arr = np.arange(n_corr) - (n - 1)
    grad = np.zeros(n)
    for idx in range(n_corr):
        pv = p[idx]
        if pv < 1e-14:
            continue
        s = int(s_arr[idx])
        j1lo = max(0, -s)
        j1hi = min(n, n - s)
        if j1hi > j1lo:
            grad[j1lo:j1hi] += pv * (1.0 - h[j1lo + s:j1hi + s])
        j2lo = max(0, s)
        j2hi = min(n, n + s)
        if j2hi > j2lo:
            grad[j2lo:j2hi] += -pv * h[j2lo - s:j2hi - s]
    grad *= dx
    return obj, grad

_opt_state = {}

def _pen_fun(x):
    a = _opt_state['alpha']
    ts = _opt_state['target_sum']
    pen = _opt_state['penalty']
    hs = np.clip(x, 0.0, 1.0)
    obj, grad = smooth_obj_and_grad(hs, a)
    d = float(np.sum(hs) - ts)
    obj += pen * d * d
    return obj

def _pen_jac(x):
    a = _opt_state['alpha']
    ts = _opt_state['target_sum']
    pen = _opt_state['penalty']
    hs = np.clip(x, 0.0, 1.0)
    obj, grad = smooth_obj_and_grad(hs, a)
    d = float(np.sum(hs) - ts)
    grad = grad + 2.0 * pen * d
    return grad

def run_lbgfgs(h, target_sum, alpha, penalty, maxiter):
    _opt_state['alpha'] = alpha
    _opt_state['target_sum'] = target_sum
    _opt_state['penalty'] = penalty
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_pen_fun, h, jac=_pen_jac, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
    except Exception:
        return h, compute_c5(h)
    h_new = project_feasible(res.x, target_sum)
    return h_new, compute_c5(h_new)

def coord_descend(h, target_sum, max_time, rng, patience=25):
    n = len(h)
    dx = 2.0 / n
    s_arr = np.arange(2 * n - 1) - (n - 1)
    h = project_feasible(h.copy(), target_sum)
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    best_score = float(np.max(corr)) * dx
    best_h = h.copy()
    no_improve = 0
    start = time.time()
    while time.time() - start < max_time:
        for j in range(n):
            old_val = h[j]
            js = j - s_arr
            A = np.zeros(2 * n - 1)
            m1 = (js >= 0) & (js < n)
            A[m1] += (1.0 - h[js[m1]])
            js2 = j + s_arr
            m2 = (js2 >= 0) & (js2 < n)
            A[m2] -= h[js2[m2]]
            B = corr - A * old_val
            lo, hi = 0.0, 1.0
            for _ in range(22):
                mid = (lo + hi) * 0.5
                vals = A * mid + B
                k_star = np.argmax(vals)
                g = A[k_star]
                if g > 1e-12:
                    hi = mid
                elif g < -1e-12:
                    lo = mid
                else:
                    break
            new_val = (lo + hi) * 0.5
            delta = new_val - old_val
            if abs(delta) > 1e-15:
                corr += delta * A
                h[j] = new_val
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        score = float(np.max(corr)) * dx
        if score < best_score - 1e-13:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                scale = 0.002 + 0.003 * min(1.0, no_improve / (2.0 * patience))
                h = project_feasible(best_h + rng.randn(n) * scale, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full')
                no_improve = 0
    return best_h, best_score

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
