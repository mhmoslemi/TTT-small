import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        step = diff / len(idx)
        # Limit step to avoid overshooting
        if diff > 0:
            max_step = min(1.0 - h[idx]).max() if len(idx) > 0 else 0
        else:
            max_step = h[idx].min() if len(idx) > 0 else 0
        h[idx] += step
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_weights(corr, alpha):
    max_c = np.max(corr)
    shifted = alpha * (corr - max_c)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    return w

def compute_grad_lse(h, w):
    n = len(h)
    grad = np.zeros(n)
    # w is indexed from 0 to 2n-2, corresponding to lag t = idx - (n-1)
    for t in range(-(n-1), n):
        wt = w[t + (n-1)]
        if wt < 1e-12:
            continue
        # grad[j] += wt * (1 - h[j+t]) for valid j
        j_lo = max(0, -t)
        j_hi = min(n-1, n-1-t)
        if j_lo <= j_hi:
            grad[j_lo:j_hi+1] += wt * (1.0 - h[j_lo+t:j_hi+t])
        # grad[j] -= wt * h[j-t] for valid j
        j_lo2 = max(0, t)
        j_hi2 = min(n-1, t+n-1)
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2+1] -= wt * h[j_lo2-t:j_hi2-t]
    return grad

def nesterov_lse(h, target_sum, alpha_schedule, max_time, rng, momentum=0.9):
    n = len(h)
    dx = 2.0 / n
    h = project_feasible(h.copy(), target_sum)
    h_y = h.copy()  # Nesterov lookahead
    best_h = h.copy()
    best_score = compute_c5(h)
    
    elapsed_start = time.time()
    iter_count = 0
    
    while time.time() - elapsed_start < max_time:
        # Determine alpha based on progress
        frac = (time.time() - elapsed_start) / max_time
        # alpha increases over time
        alpha = alpha_schedule(frac)
        
        corr = compute_corr(h_y)
        w = lse_weights(corr, alpha)
        grad = compute_grad_lse(h_y, w)
        
        # Nesterov step
        h_new = h_y - 0.01 * grad  # step size
        h_new = project_feasible(h_new, target_sum)
        
        # Nesterov momentum update
        h_y_new = h_new + momentum * (h_new - h)
        h_y_new = project_feasible(h_y_new, target_sum)
        
        h = h_new
        h_y = h_y_new
        iter_count += 1
        
        # Check improvement every 10 iterations
        if iter_count % 10 == 0:
            score = compute_c5(h)
            if score < best_score - 1e-14:
                best_score = score
                best_h = h.copy()
    
    return best_h, best_score

2. Compute the intercepts: intercept_k = corr[k] - h[j] * slope_k (using current h[j])

- x = 0, or
- x = 1, or
- A point where two lines cross and both are on the upper envelope

1. Sort lines by their value at x = h[j] (current)
2. Check the top few lines
3. Find the crossing point of the top two (or top few)
4. Evaluate the objective at 0, 1, and the crossing points
5. Pick the best

def coord_optimal_j(h, j, target_sum, n):
    """Find optimal value of h[j] given all other h fixed."""
    dx = 2.0 / n
    s_arr = np.arange(2*n - 1) - (n-1)
    
    # For each k, compute slope and contribution to corr[k] from h[j]
    # corr[k] = base_k + h[j] * slope_k
    # where slope_k = (1-h[j+t_k]) - h[j-t_k] (with validity checks)
    
    # Compute slopes
    slopes = np.zeros(2*n-1)
    for k in range(2*n-1):
        t = s_arr[k]
        s = 0.0
        # Term (1-h[j+t]) if j+t valid
        if 0 <= j+t < n:
            s += (1.0 - h[j+t])
        # Term -h[j-t] if j-t valid
        if 0 <= j-t < n:
            s -= h[j-t]
        slopes[k] = s
    
    # Current corr values (with h[j] included)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # base_k = corr[k] - h[j] * slopes[k]
    base = corr - h[j] * slopes
    
    # Now minimize max_k (base_k + x * slopes_k) over x in [0,1]
    # This is a 1D convex piecewise linear optimization
    
    # Strategy: evaluate at x=0, x=1, and at crossing points of top lines
    candidates = [0.0, 1.0]
    
    # Get top lines at current point
    current_vals = base + h[j] * slopes
    # Sort by current value descending
    order = np.argsort(-current_vals)[:20]  # top 20
    
    # Find crossing points among top lines
    for i in range(len(order)):
        for j2 in range(i+1, len(order)):
            k1, k2 = order[i], order[j2]
            if abs(slopes[k1] - slopes[k2]) < 1e-15:
                continue

            if 0.0 <= x_cross <= 1.0:
                candidates.append(x_cross)
    
    # Also add the point where the max transitions
    best_x = h[j]
    best_val = np.max(current_vals) * dx
    
    for x in candidates:
        vals = base + x * slopes
        val = np.max(vals) * dx
        if val < best_val:
            best_val = val
            best_x = x
    
    return best_x, best_val

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    for _ in range(200):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        nm = int(np.sum(mask))
        h = h.copy()
        h[mask] += diff / nm
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
    return h


def interp_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    return np.interp(x_new, x_old, h_src)


def coord_descend(h, target_sum, max_time, n_top=20, patience=40, noise_scale_base=0.008):

    dx = 2.0 / n

    best_h = h.copy()

    elapsed_start = time.time()
    rng = np.random.RandomState(777)

    while time.time() - elapsed_start < max_time:
        for j in range(n):
            jt = j + s_arr
            j2t = j - s_arr
            m1 = (jt >= 0) & (jt < n)
            m2 = (j2t >= 0) & (j2t < n)
            slopes = np.zeros(2 * n - 1)
            slopes[m1] += (1.0 - h[jt[m1]])
            slopes[m2] -= h[j2t[m2]]

            hj = h[j]

            cur_vals = base + hj * slopes
            if n_top < 2 * n - 1:
                top_idx = np.argpartition(cur_vals, -n_top)[-n_top:]

                        continue

                    if -0.05 <= xc <= 1.05:

            if len(crossing_list) > 0:

def grad_lse_fast(h, w):

            corr0 = np.correlate(h_cur, 1.0 - h_cur, mode='full')

                return cm + np.log(np.sum(np.exp(shifted))) / alpha
