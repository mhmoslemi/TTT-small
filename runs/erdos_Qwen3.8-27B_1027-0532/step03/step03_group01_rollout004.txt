import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_obj(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    log_sum_exp = max_corr + np.log(np.sum(np.exp(shifted))) / alpha
    obj = log_sum_exp * dx
    constraint_violation = np.sum(h) - n/2.0
    obj += penalty * constraint_violation ** 2
    return obj

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h_r
    ones = np.where(h_r >= 0.5)[0]
    zeros = np.where(h_r < 0.5)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum + 0.5)
        if need > 0 and need <= len(ones):
            # Remove the ones with smallest h value
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum + 0.5)
        if need > 0 and need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # Verify
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            extra = int(final_sum - target_sum)
            ones2 = np.where(h_r == 1)[0]
            h_r[ones2[:extra]] = 0
        else:
            extra = int(target_sum - final_sum)
            zeros2 = np.where(h_r == 0)[0]
            h_r[zeros2[:extra]] = 1
    return h_r

def lbfgsb_optimize(h_start, n, target, alphas, penalty_steps, maxiter, time_limit, start_time):
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for penalty in penalty_steps:
            if time.time() - start_time > time_limit:
                break
            res = minimize(
                soft_max_obj, h_cur, args=(alpha, penalty),
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10}
            )
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
    return h_cur, curr_score

            else:

        h_new = h.copy()

                i1 = ones[rng.randint(len(ones))]
                i0 = zeros[rng.randint(len(zeros))]
                h_new[i1] = 0
                h_new[i0] = 1
                moved = True
        elif move_type == 1:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def soft_max_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    return float((m + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

def soft_max_pen_obj(h, alpha, penalty):
    n = len(h)
    base = soft_max_c5(h, alpha)
    sv = np.sum(h) - n / 2.0
    return base + penalty * sv * sv

def project_feasible(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target):
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0.0, 1.0)
    cs = h_r.sum()
    if abs(cs - target) < 1e-10:
        return h_r
    if cs > target:
        need = int(round(cs - target))
        ones = np.where(h_r == 1)[0]
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(round(target - cs))
        zeros = np.where(h_r == 0)[0]
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    # final fix
    fs = h_r.sum()
    if fs > target:
        extra = int(round(fs - target))
        ones = np.where(h_r == 1)[0]
        h_r[ones[:extra]] = 0
    elif fs < target:
        extra = int(round(target - fs))
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[:extra]] = 1
    return h_r

def continuous_optimize(h_start, target, alphas, penalties, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    best = compute_c5(h_cur)
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for pen in penalties:
            if time.time() - start_time > time_limit:
                break
            try:
                res = minimize(soft_max_pen_obj, h_cur, args=(alpha, pen),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-9})
                h_new = project_feasible(res.x, target)
                sc = compute_c5(h_new)
                if sc < best:
                    best = sc
                    h_cur = h_new
            except Exception:
                pass
    return h_cur, best
