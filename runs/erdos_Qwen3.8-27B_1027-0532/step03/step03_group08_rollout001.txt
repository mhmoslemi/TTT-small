def _c5_of(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Add to positions with most room
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_sum(y, S):

    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

# Global variables for SLSQP constraints
_g_sum = 0.0
_g_n = 0

def _con_fun(x):
    return x.sum() - _g_sum

def _con_jac(x):
    return np.ones(_g_n)

def _grad_lag(h, k_idx, dx):
    """Gradient of the overlap at correlation index k_idx with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    m = n - abs(lag)
    if m <= 0:
        return grad
    if lag >= 0:
        grad[:m] += (1.0 - h[lag:lag+m]) * dx
        grad[lag:lag+m] -= h[:m] * dx
    else:
        j = -lag
        grad[j:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[j:j+m] * dx
    return grad

def _slsqp_lse(h0, n, target, alpha, maxiter):
    global _g_sum, _g_n
    _g_sum = target
    _g_n = n
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mv = np.max(corr)
        # LSE
        shifted = alpha * (corr - mv)
        # Clip to avoid overflow
        shifted = np.clip(shifted, -100, 100)
        return float((mv + np.log(np.sum(np.exp(shifted))) / alpha) * dx)
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mv = np.max(corr)
        shifted = alpha * (corr - mv)
        shifted = np.clip(shifted, -100, 100)
        w = np.exp(shifted)
        w = w / w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-10:
                g += w[k] * _grad_lag(x, k, dx)
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _coord_descent_fine(h0, n, target, grid_size=21, max_rounds=10, deadline=None):
    """Coordinate descent with fine grid."""
    h = h0.copy()
    best_c = _c5_of(h)
    grid = np.linspace(0, 1, grid_size)
    
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_v = old
            best_local = best_c
            for v in grid:
                if v == old:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_local - 1e-14:
                    best_local = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
                best_c = best_local
        if not improved:
            break
    return h

def _nesterov_active(h0, n, target, n_iters, step0, m_active=5):
    """Nesterov accelerated gradient with multi-active lags."""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:m_active]
        top_vals = corr[top_idx]
        mv = np.max(top_vals)
        w = np.exp(50.0 * (top_vals - mv))
        w = w / w.sum()
        
        g = np.zeros(n)
        for i in range(m_active):
            g += w[i] * _grad_lag(y, top_idx[i], dx)
        
        step = step0 / np.sqrt(float(it + 1))
        x_new = _fix_sum(np.clip(y - step * g, 0.0, 1.0), target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _fix_sum(np.clip(y_new, 0.0, 1.0), target)
        x = x_new
        y = y_new
        t = t_new
        
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    
    return best, best_val

    # Various perturbation strategies
    strategies = []
    
    # Small Gaussian noise
    strategies.append(h0 + 0.1 * rng.randn(n))
    # Larger noise
    strategies.append(h0 + 0.3 * rng.randn(n))

    # Blend with complement
    strategies.append(0.7 * h0 + 0.3 * (1.0 - h0))

import time
import numpy as np
from scipy.optimize import minimize

_g_sum = 0.0
_g_n = 0

def _c5_of(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-14:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-14:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _con_fun(x):
    return x.sum() - _g_sum

def _con_jac(x):
    return np.ones(_g_n)

def _grad_lag(h, k_idx, dx):
    n = len(h)
    grad = np.zeros(n)
    lag = k_idx - (n - 1)
    m = n - abs(lag)
    if m <= 0:
        return grad
    if lag >= 0:
        grad[:m] += (1.0 - h[lag:lag + m]) * dx
        grad[lag:lag + m] -= h[:m] * dx
    else:
        j = -lag
        grad[j:] += (1.0 - h[:m]) * dx
        grad[:m] -= h[j:j + m] * dx
    return grad

def _slsqp_opt(h0, n, target, alpha, maxiter):
    global _g_sum, _g_n
    _g_sum = target
    _g_n = n
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mv = np.max(corr)
        shifted = np.clip(alpha * (corr - mv), -50, 50)
        return float((mv + np.log(np.sum(np.exp(shifted))) / alpha) * dx)

    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mv = np.max(corr)
        shifted = np.clip(alpha * (corr - mv), -50, 50)
        w = np.exp(shifted)
        w = w / w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] > 1e-10:
                g += w[k] * _grad_lag(x, k, dx)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _coord_descent_fine(h0, n, target, grid_size, max_rounds, deadline=None):
    h = h0.copy()
    best_c = _c5_of(h)
    grid = np.linspace(0, 1, grid_size)
    for _ in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_v = old
            best_local = best_c
            for v in grid:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_local - 1e-14:
                    best_local = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
                best_c = best_local
        if not improved:
            break
    return h

def _nesterov_active(h0, n, target, n_iters, step0, m_active):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:m_active]
        top_vals = corr[top_idx]
        mv = np.max(top_vals)
        w = np.exp(60.0 * (top_vals - mv))
        w = w / w.sum()
        g = np.zeros(n)
        for i in range(m_active):
            g += w[i] * _grad_lag(y, top_idx[i], dx)
        step = step0 / np.sqrt(float(it + 1))
        x_new = _fix_sum(np.clip(y - step * g, 0.0, 1.0), target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _fix_sum(np.clip(y_new, 0.0, 1.0), target)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_explore(h0, n, target, pop_size, max_gen, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.2:
            pop[i] = _fix_sum(h0 + 0.15 * rng.randn(n), target)
        elif r < 0.4:
            pop[i] = _fix_sum(h0 + 0.3 * rng.randn(n), target)
        elif r < 0.6:
            pop[i] = _fix_sum(1.0 - h0 + 0.05 * rng.randn(n), target)
        elif r < 0.7:
            sh = rng.randint(1, n // 4)
            pop[i] = _fix_sum(np.roll(h0, sh), target)
        elif r < 0.85:
            t = rng.uniform(0.5, 0.9)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target)
        else:
            bs = max(3, n // 8)
            s1 = rng.randint(0, n - bs)
            s2 = rng.randint(0, n - bs)
            pop[i] = h0.copy()
            pop[i][s1:s1 + bs] = h0[s2:s2 + bs]
            pop[i] = _fix_sum(pop[i], target)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = 0.5 + 0.3 * rng.random()
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            r3 = rng.randint(pop_size)
            if r1 == i:
                r1 = (r1 + 1) % pop_size
            if r2 == i or r2 == r1:
                r2 = (r2 + 2) % pop_size
            if r3 == i or r3 == r1 or r3 == r2:
                r3 = (r3 + 3) % pop_size
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2]) + 0.1 * (pop[r2] - pop[r3])
            trial = pop[i].copy()
            mask = rng.random(n) < 0.6
            trial[mask] = mutant[mask]
            mut_mask = rng.random(n) < 0.05
            trial[mut_mask] = rng.random(mut_mask.sum()) if mut_mask.sum() > 0 else trial[mut_mask]
            trial = _fix_sum(np.clip(trial, 0, 1), target)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
        worst = np.argsort(pop_vals)[-pop_size // 4:]
        if gen % 5 == 4:
            for idx in worst:
                t = rng.uniform(0.3, 0.7)
                pop[idx] = _fix_sum(t * best_h + (1.0 - t) * rng.rand(n), target)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
    return best_h, best_val
