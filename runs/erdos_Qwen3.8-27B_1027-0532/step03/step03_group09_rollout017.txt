import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1}"""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def subgrad_single(h, k_idx):
    """Exact subgradient of the k-th correlation value w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    omh = 1.0 - h
    if s >= 0:
        m = n - s
        grad[s:] += dx * omh[:m]
        grad[:m] -= dx * h[s:]
    else:
        r = -s
        m = n - r
        grad[:m] += dx * omh[r:]
        grad[r:] -= dx * h[:m]
    return grad

def subgrad_topk(h, k, temp):
    """Weighted subgradient using top-k correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx] - np.max(xc)
    w = np.exp(xc / temp)
    w = w / w.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        if w[i] < 1e-14:
            continue
        s = topk_idx[i] - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += w[i] * dx * omh[:m]
            grad[:m] -= w[i] * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += w[i] * dx * omh[r:]
            grad[r:] -= w[i] * dx * h[:m]
    return grad

def fista_subgrad(h_init, k, temp, lr_init, n_max_iters, time_limit, start_time):
    """FISTA with subgradient and projection"""
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    h_prev = h_init.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    t = 1.0
    lr = lr_init
    
    for it in range(n_max_iters):
        if time.time() - start_time > time_limit:
            break
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        h_mom = h_cur + (t - 1.0) / t_new * (h_cur - h_prev)
        h_prev = h_cur.copy()
        t = t_new
        
        # Subgradient at momentum point
        grad = subgrad_topk(h_mom, k, temp)
        
        # Gradient step
        h_new = h_mom - lr * grad
        
        # Project
        h_new = project_onto_constraint(h_new, target_sum)
        
        # Backtracking if no improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            lr = min(lr * 1.01, 0.5)
        else:
            lr = max(lr * 0.95, 1e-6)
            # Occasionally reset momentum
            if it % 50 == 0:
                h_prev = h_new.copy()
        
        h_cur = h_new
    
    return best_h, best_c5

Actually, I realize I need to be more careful about the FISTA implementation. The standard FISTA works for minimizing f(x) + g(x) where f is smooth and g is simple (proximable). Here, f is the max of linear functions (not smooth), and g is the indicator of the constraint set.

For non-smooth f, we can use the "subgradient FISTA" or "accelerated proximal gradient with subgradient". The convergence rate is O(1/k) for the objective gap, but the practical performance can be better than non-accelerated methods.

Let me also add a "shake" mechanism: if FISTA gets stuck (no improvement for many iterations), perturb the solution and reset momentum.

    target_sum = n / 2.0
    if strategy == 0:
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n), target_sum)
    elif strategy == 1:
        h_p = h_best.copy()
        idx = rng.choice(n, size=max(3, n // 8), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.15 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.4, 0.7))
        return project_onto_constraint(alpha * h_best + (1.0 - alpha) * h_bin, target_sum)
    elif strategy == 3:
        return gen_block_pattern(n, rng)
    elif strategy == 4:
        h_p = h_best.copy()
        max_len = max(2, n // 10)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            blk = h_p[a:a+blen].copy()
            h_p[a:a+blen] = h_p[b:b+blen]
            h_p[b:b+blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.005 * rng.standard_normal(n), target_sum)
    else:
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = gen_block_pattern(180, rng)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: Quick SLSQP at initial resolution
    if rem() > 200:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.01, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: FISTA polish at initial resolution
    if rem() > 150:
        tl = min(rem() - 50, 100)
        h_f, c5_f = fista_subgrad(best_h, 3, 0.01, 0.08, 100000, tl, start_time)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: Upsample to 2x resolution and optimize
    if rem() > 400:
        n1 = n0 * 2
        h_up = upsample(best_h, n1)
        h_up = project_onto_constraint(h_up + 0.005 * rng.standard_normal(n1), n1 / 2.0)

        # FISTA at new resolution
        tl = min(rem() - 200, 180)
        h_f, c5_f = fista_subgrad(h_up, 5, 0.02, 0.06, 80000, tl, start_time)
        if c5_f < c5_up := compute_c5(h_up):
            h_up = h_f

        # SLSQP at new resolution
        if rem() > 150:
            h_opt, c5_opt = slsqp_optimize(h_up, 0.01, 250)
            if c5_opt < compute_c5(h_up):
                h_up = h_opt

        # More FISTA
        if rem() > 80:
            tl = min(rem() - 30, 80)
            h_f, c5_f = fista_subgrad(h_up, 3, 0.005, 0.04, 60000, tl, start_time)
            if c5_f < compute_c5(h_up):
                h_up = h_f

        if compute_c5(h_up) < best_c5:
            best_c5 = compute_c5(h_up)
            best_h = h_up.copy()
            n0 = n1

    # Phase 4: Extended FISTA with varying parameters
    if rem() > 200:
        for k_val, temp_val, lr_val in [(5, 0.01, 0.05), (3, 0.005, 0.04), (1, 0.001, 0.03)]:
            if rem() < 60:
                break
            tl = min(rem() - 30, 60)
            h_f, c5_f = fista_subgrad(best_h, k_val, temp_val, lr_val, 50000, tl, start_time)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 5: Restart loop with diverse initializations
    n_restart = 0
    max_restarts = 30
    while rem() > 250 and n_restart < max_restarts:
        strategy = n_restart % 5
        h_pert = gen_perturb(best_h, rng, len(best_h), strategy)

        # FISTA from perturbed start
        tl_f = min(rem() - 120, 50)
        h_f, c5_f = fista_subgrad(h_pert, 3, 0.01, 0.05, 40000, tl_f, start_time)

        # SLSQP polish if promising
        if rem() > 100 and c5_f < best_c5 + 0.002:
            eps_use = [0.005, 0.002, 0.001, 0.0005][min(n_restart // 7, 3)]
            h_opt, c5_opt = slsqp_optimize(h_f, eps_use, 200)
            if c5_opt < c5_f:
                c5_f = c5_opt
                h_f = h_opt

        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

        n_restart += 1

    # Phase 6: Try higher resolution if time permits
    if rem() > 250 and len(best_h) <= 360:
        n2 = len(best_h) + 180
        h_up2 = upsample(best_h, n2)
        h_up2 = project_onto_constraint(h_up2 + 0.003 * rng.standard_normal(n2), n2 / 2.0)
        tl = min(rem() - 120, 120)
        h_f2, c5_f2 = fista_subgrad(h_up2, 5, 0.02, 0.05, 60000, tl, start_time)
        if rem() > 80:
            h_opt2, c5_opt2 = slsqp_optimize(h_f2, 0.005, 200)
            if c5_opt2 < c5_f2:
                c5_f2 = c5_opt2
                h_f2 = h_opt2
        if c5_f2 < best_c5:
            best_c5 = c5_f2
            best_h = h_f2.copy()

    # Phase 7: Final SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final FISTA polish
    if rem() > 40:
        tl = min(rem() - 15, 40)
        h_f, c5_f = fista_subgrad(best_h, 1, 0.001, 0.02, 50000, tl, start_time)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final projection and evaluation
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
