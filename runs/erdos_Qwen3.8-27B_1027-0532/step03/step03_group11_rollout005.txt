def fast_full_correlate(a, b):
    return fftconvolve(a, b[::-1], mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_correlate(h, 1.0 - h) * dx
    return float(np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = fast_full_correlate(h, 1.0 - h) * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - x.max()
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = fast_full_correlate(h, omh) * dx
    cmax = float(np.max(corr))
    x = (corr - cmax) / eps
    x = x - x.max()
    ex = np.exp(x)
    w = ex / ex.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[:m] += wk * dx * omh[s:s + m]
            grad[s:] -= wk * dx * h[:m]
        else:
            r = -s
            m = n - r
            grad[r:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[r:r + m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def make_binary_from_continuous(h):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def random_binary(n, rng):
    idx = np.arange(n)
    rng.shuffle(idx)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def binary_first_improvement(h, max_rounds=200, time_limit=30.0, start_time=0.0, max_pairs=None):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        found = False
        npair = len(ones) * len(zeros)
        if max_pairs is not None and npair > max_pairs:
            perm = np.random.default_rng(rnd).permutation(npair)
            perm = perm[:max_pairs]
        else:
            perm = np.arange(npair)
        for idx in perm:
            oi = ones[idx // len(zeros)]
            zi = zeros[idx % len(zeros)]
            h[oi] = 0.0
            h[zi] = 1.0
            sc = compute_c5(h)
            if sc < current - 1e-15:
                current = sc
                found = True
                break
            else:
                h[oi] = 1.0
                h[zi] = 0.0
        if not found:
            break
    return h, current

def binary_two_swap(h, max_rounds=100, time_limit=30.0, start_time=0.0):
    n = len(h)
    h = h.copy()
    current = compute_c5(h)
    rng_local = np.random.default_rng(int(start_time * 1000) % 10000)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        found = False
        for _ in range(50):
            o1, o2 = rng_local.choice(ones, 2, replace=False)
            z1, z2 = rng_local.choice(zeros, 2, replace=False)
            h[o1] = 0.0; h[o2] = 0.0
            h[z1] = 1.0; h[z2] = 1.0
            sc = compute_c5(h)
            if sc < current - 1e-15:
                current = sc
                found = True
                break
            else:
                h[o1] = 1.0; h[o2] = 1.0
                h[z1] = 0.0; h[z2] = 0.0
        if not found:
            break
    return h, current

def ills_search(h_init, budget_s=300, start_time=0.0, max_kicks=50):
    n = len(h_init)
    rng = np.random.default_rng(int(start_time * 1000) % 10000)
    h_best = h_init.copy()
    best_c5 = compute_c5(h_best)
    h_cur = h_init.copy()
    cur_c5 = best_c5
    for kick in range(max_kicks):
        if time.time() - start_time > budget_s:
            break
        h_cur, cur_c5 = binary_first_improvement(h_cur, max_rounds=100,
                                                   time_limit=15, start_time=start_time)
        h_cur, cur_c5 = binary_two_swap(h_cur, max_rounds=50,
                                         time_limit=10, start_time=start_time)
        h_cur, cur_c5 = binary_first_improvement(h_cur, max_rounds=100,
                                                   time_limit=15, start_time=start_time)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            h_best = h_cur.copy()
        nflip = rng.integers(2, max(3, n // 15) + 1)
        ones = np.where(h_cur == 1.0)[0]
        zeros = np.where(h_cur == 0.0)[0]
        flip_one = rng.choice(ones, nflip)
        flip_zero = rng.choice(zeros, nflip)
        h_cur = h_cur.copy()
        h_cur[flip_one] = 0.0
        h_cur[flip_zero] = 1.0
    return h_best, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0, 1)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    eps_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005]
    for eps in eps_schedule:
        if time.time() - start_time > budget_s * 0.25:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, eps, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    h_bin = make_binary_from_continuous(best_h)
    best_bin_c5 = compute_c5(h_bin)
    best_bin_h = h_bin.copy()

    if time.time() - start_time < budget_s * 0.5:
        h_ill, c5_ill = ills_search(h_bin, budget_s=budget_s * 0.3,
                                     start_time=start_time, max_kicks=40)
        if c5_ill < best_bin_c5:
            best_bin_c5 = c5_ill
            best_bin_h = h_ill.copy()

    h_from_bin = best_bin_h.astype(float) + 0.01 * rng.standard_normal(n0)
    h_from_bin = project_onto_constraint(h_from_bin, target_sum)

    eps_schedule2 = [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for eps in eps_schedule2:
        if time.time() - start_time > budget_s * 0.65:
            break
        h_opt, c5_opt = slsqp_optimize(h_from_bin, eps, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_from_bin = h_opt.copy()

    n_starts = 8
    for si in range(n_starts):
        if time.time() - start_time > budget_s * 0.85:
            break
        if si == 0:
            h_pert = best_h + 0.03 * rng.standard_normal(n0)
        elif si == 1:
            h_pert = best_bin_h.astype(float) + 0.05 * rng.standard_normal(n0)
        elif si == 2:
            h_pert = best_h.copy()
            nflip = max(3, n0 // 12)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
        elif si == 3:
            h_pert = random_binary(n0, rng).astype(float) + 0.1 * rng.standard_normal(n0)
        elif si == 4:
            alpha = rng.uniform(0.5, 0.8)
            h_rand = random_binary(n0, rng).astype(float)
            h_pert = alpha * best_h + (1.0 - alpha) * h_rand
        elif si == 5:
            h_pert = best_h + 0.15 * rng.standard_normal(n0)
        elif si == 6:
            h_pert = best_bin_h.astype(float)
            nflip = max(2, n0 // 20)
            idx = rng.choice(n0, size=nflip, replace=False)
            h_pert[idx] = 1.0 - h_pert[idx]
            h_pert = h_pert + 0.03 * rng.standard_normal(n0)
        else:
            h_pert = best_h + 0.08 * rng.standard_normal(n0)

        h_pert = project_onto_constraint(h_pert, target_sum)
        h_opt, c5_opt = slsqp_optimize(h_pert, 0.001, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h_opt2, c5_opt2 = slsqp_optimize(h_opt, 0.0001, 300)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2.copy()
            h_bin2 = make_binary_from_continuous(best_h)
            h_ill2, c5_ill2 = ills_search(h_bin2, budget_s=30,
                                            start_time=start_time, max_kicks=15)
            if c5_ill2 < best_bin_c5:
                best_bin_c5 = c5_ill2
                best_bin_h = h_ill2.copy()
            h_fb = c5_ill2
            if h_fb < best_c5:
                best_c5 = h_fb
                best_h = best_bin_h.astype(float)

    if time.time() - start_time < budget_s - 60:
        n1 = 2 * n0
        h_up = np.zeros(n1)
        for i in range(n1):
            h_up[i] = best_h[i // 2]
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        eps_up = [0.005, 0.002, 0.001, 0.0005]
        for eps in eps_up:
            if time.time() - start_time > budget_s - 20:
                break
            h_opt_up, c5_opt_up = slsqp_optimize(h_up, eps, 200)
            if c5_opt_up < best_c5:
                best_c5 = c5_opt_up
                best_h = h_opt_up.copy()
            h_up = h_opt_up.copy()

    n_cur = len(best_h)
    for eps in [0.00005, 0.00002, 0.00001]:
        if time.time() - start_time > budget_s - 5:
            break
        h_fin, c5_fin = slsqp_optimize(best_h, eps, 300)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_h = best_h
        final_c5 = best_c5
    return final_h, final_c5, len(final_h)
