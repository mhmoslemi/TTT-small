import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _grad_lag_raw(h, k_idx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[:m] += (1.0 - h[lag:n])
            grad[lag:n] += -h[:m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[j:n] += (1.0 - h[:m])
            grad[:m] += -h[j:n]
    return grad

def _grad_lag(h, k_idx):
    n = len(h)
    return _grad_lag_raw(h, k_idx) * (2.0 / n)

def _top_m_grad(h, m):
    n = len(h)
    dx = 2.0 / n
    corr = _corr_vec(h)
    idx = np.argsort(corr)[::-1][:m]
    weights = corr[idx].copy()
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(idx)) / len(idx)
    else:
        weights = weights / wsum
    grad = np.zeros(n)
    for i in range(len(idx)):
        if weights[i] > 1e-8:
            grad += weights[i] * _grad_lag_raw(h, idx[i])
    return grad * dx

def _lp_obj(h, p, n):
    corr = _corr_vec(h)
    s = np.sum(corr**p)
    if s < 1e-300:
        return 0.0
    return s**(1.0/p) * (2.0/n)

def _lp_grad_exact(h, p, n):
    corr = _corr_vec(h)
    dx = 2.0 / n
    corr_p_sum = np.sum(corr**p)
    if corr_p_sum < 1e-300:
        return np.zeros(n)
    lp = corr_p_sum**(1.0/p)
    coeff = lp**(p-1) * dx
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        c = corr[k_idx]
        if c > 1e-14:
            w = c**(p-1)
            if w > 1e-14:
                grad += w * _grad_lag_raw(h, k_idx)
    return grad * coeff

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _nesterov_correct(h0, n, target, n_iters, alpha0, top_m, deadline):
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        g = _top_m_grad(y, top_m)
        alpha = alpha0 / (1.0 + 0.05 * it)
        x_new = _project(y - alpha * g, target)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _slsqp_lp(h0, n, target, p, maxiter, deadline):
    t_chk = time.time()
    if t_chk > deadline:
        return h0

    def _obj_fn(x):
        return _lp_obj(x, p, n)

    def _grad_fn(x):
        return _lp_grad_exact(x, p, n)

    def _con_fun(x):
        return x.sum() - target

    def _con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_fn, h0, jac=_grad_fn, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _slsqp_lse(h0, n, target, alpha, maxiter, deadline):
    if time.time() > deadline:
        return h0

    def _obj_fn(x):
        corr = _corr_vec(x)
        m = np.max(corr)
        return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

    def _con_fun(x):
        return x.sum() - target

    def _con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_fn, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target)
    except Exception:
        return h0

def _coord_descent(h0, n, target, max_rounds):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            best_c = c0 if c0 < c1 else c1
            if best_c < co - 1e-14:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_descent_cont(h0, n, target, max_rounds):
    h = h0.copy()
    vals = np.array([0.0, 0.1, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.75, 0.8, 0.9, 1.0])
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if abs(best_v - old) > 1e-10:
                improved = True
        if not improved:
            break
    return h

def _discrete_ls(h0, n, target, deadline, rng, max_iters=5000):
    h = np.zeros(n)
    ones = h0 > 0.5
    n_ones = ones.sum()
    if n_ones != n // 2:

        ones = np.zeros(n, dtype=bool)

    h[ones] = 1.0
    h[~ones] = 0.0
    current_c5 = _c5_of(h)
    best_h = h.copy()
    best_c5 = current_c5
    one_indices = np.where(ones)[0]

    for it in range(max_iters):

            one_indices = np.where(h > 0.5)[0]

def _active_set_slsqp(h0, n, target, top_m, maxiter, deadline):

    corr = _corr_vec(h0)

def _multistart_de(h0, n, target, pop_size, n_gen, deadline, rng):
    target_sum = target
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_r = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_r)
        caches.append(_c5_of(h_r))
    best_h = h0.copy()
    best_c5 = min(caches)
    stall = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        improved = False
        new_pop = [pop[int(np.argmin(caches))].copy()]
        new_caches = [min(caches)]
        for i in range(pop_size - 1):
            cands = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(cands), size=3, replace=False)
            mutant = pop[cands[r1]] + 0.8 * (pop[cands[r2]] - pop[cands[r3]])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.rand(n) < 0.5
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            ct = _c5_of(trial)
            if ct <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(ct)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = trial.copy()
                    improved = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved:
            stall += 1
        else:
            stall = 0

            break
    return best_h, best_c5
