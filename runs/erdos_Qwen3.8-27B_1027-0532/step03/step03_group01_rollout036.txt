import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def linear_corr_fft(h):
    n = len(h)
    size = 2 * n - 1
    H = np.fft.rfft(h, n=size)
    Wh = np.fft.rfft(1.0 - h, n=size)
    return np.fft.irfft(H * np.conj(Wh), n=size)

def circular_autocorr(h):
    n = len(h)
    H = np.fft.fft(h)
    return np.real(np.fft.ifft(H * np.conj(H)))

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = int(h_r.sum())
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum > target_sum:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    else:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    final_sum = int(h_r.sum())
    if final_sum > target_sum:
        extra = final_sum - target_sum
        ones = np.where(h_r == 1)[0]
        h_r[ones[:extra]] = 0
    elif final_sum < target_sum:
        extra = target_sum - final_sum
        zeros = np.where(h_r == 0)[0]
        h_r[zeros[:extra]] = 1
    return h_r.astype(float)

def spectral_grad(h):
    n = len(h)
    P = circular_autocorr(h)
    g = P - n / 4.0
    H = np.fft.fft(h)
    G = np.fft.fft(g)
    conv = np.real(np.fft.ifft(G * H))
    return 4.0 * conv, P

def smooth_obj(h, alpha, target_sum):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    smax = float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    penalty = 0.01 * (np.sum(h) - target_sum) ** 2
    return smax * (2.0 / n) + penalty

def slsqp_refine(h_start, target, alphas, maxiter, ftol, time_limit, start_time):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    h_cur = h_start.copy()
    curr_score = compute_c5(h_cur)

    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        try:
            res = minimize(smooth_obj, h_cur, args=(alpha, target),
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass
    return h_cur, curr_score

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)

    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

            if c5_3b < best_score:
                best_score = c5_3b
                best_h = h3b.copy()

    # Phase 4: Try at larger resolution
    n_large = n_init + 200
    if tleft() > 200:
        target_large = n_large / 2.0
        h_large = interpolate_to(best_h, n_large, target_large)
        h_large_bin = project_binary(h_large, target_large)

        h4, c5_4 = spectral_descent(h_large_bin, target_large, min(250.0, tleft() * 0.5), start_time, rng)
        if tleft() > 100:
            h4b, c5_4b = direct_c5_search(h4, target_large, min(150.0, tleft() * 0.5), start_time, rng)
            if c5_4b < c5_4:
                h4 = h4b
                c5_4 = c5_4b
        if c5_4 < best_score:
            best_score = c5_4
            best_h = h4.copy()

    # Phase 5: SA from best for diverse exploration
    if tleft() > 200:
        for sa_i in range(3):
            if tleft() < 80:
                break
            h_sa, c5_sa = sa_binary(best_h, len(best_h) / 2.0, 50000, 0.01, 0.0001,
                                    rng, min(120.0, tleft() * 0.4), start_time)
            if c5_sa < best_score:
                best_score = c5_sa
                best_h = h_sa.copy()
            if tleft() > 60:
                h_dc, c5_dc = direct_c5_search(h_sa, len(h_sa) / 2.0,
                                                min(80.0, tleft() * 0.4), start_time, rng)
                if c5_dc < best_score:
                    best_score = c5_dc
                    best_h = h_dc.copy()

    # Phase 6: SLSQP continuous refinement
    n_use = len(best_h)
    target_use = n_use / 2.0
    if tleft() > 80:
        alphas = [100, 300, 800, 2000, 5000]
        h_ref, sc = slsqp_refine(best_h, target_use, alphas, 500, 1e-14,
                                 tleft() - 40, start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Phase 7: Final direct search on binary projection of best
    if tleft() > 60:
        h_bin_final = project_binary(best_h, len(best_h) / 2.0)
        h_fd, c5_fd = direct_c5_search(h_bin_final, len(h_bin_final) / 2.0,
                                        min(60.0, tleft() * 0.6), start_time, rng)
        if c5_fd < best_score:
            best_score = c5_fd
            best_h = h_fd.copy()

    # Phase 8: Final SLSQP polish
    if tleft() > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_pert = best_h + rng.randn(n_use) * 0.01
        h_pert = project_feasible(h_pert, target_use)
        h_ref, sc = slsqp_refine(h_pert, target_use, [1000, 3000, 8000], 300, 1e-15,
                                 min(25.0, tleft() - 10), start_time)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    # Final feasibility
    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score:
        h_final = best_h
        final_score = best_score

    return h_final, final_score, n_final
