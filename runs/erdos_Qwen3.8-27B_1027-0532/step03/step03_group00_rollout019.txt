        frac = float(it) / float(max_iters)
        T = T_init * (T_final / T_init) ** frac
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        p = int(ones_idx[rng.randint(len(ones_idx))])
        q = int(zeros_idx[rng.randint(len(zeros_idx))])
        h[p] = 0.0
        h[q] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_score = float(np.max(new_corr))
        delta = new_score - cur_score
        if delta <= 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            h[p] = 1.0
            h[q] = 0.0
    return best_h, best_score
