import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0
_global_alpha = 100.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_grad(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp(alpha * (corr - m))
    weights /= weights.sum()
    g = np.zeros(n)
    # Only compute for top terms to save time
    top_k = min(n, 20)
    idx = np.argsort(weights)[::-1][:top_k]
    w_top = weights[idx]
    w_top /= w_top.sum()
    for i in range(len(idx)):
        g += w_top[i] * _grad_corr_term(h, idx[i], dx)
    return g

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),

                       options={'maxiter': maxiter, 'ftol': 1e-14})

def _nesterov_topm(h0, n, target_sum, n_iters, alpha0, m_list, deadline=None):
    """Nesterov with adaptive top-m gradient"""
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    n_actual = len(m_list)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        m_current = m_list[min(it * n_actual // n_iters, n_actual - 1)]
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(30.0 * (corr - max_corr))
        total = weights.sum()
        if total < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= total
        top_m = np.argsort(weights)[::-1][:m_current]
        w_m = weights[top_m].copy()
        w_m /= w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):

def _coord_descent_binary(h0, n, target_sum, max_rounds=30):

def _optimal_coord_value(h, j, n):

    h0[j] = 0.0

    h1[j] = 1.0

    # For k ≠ 0 (i.e., k_idx ≠ n-1): corr[k](t) = corr0[k] + t*(corr1[k] - corr0[k])
    # For k = 0 (k_idx = n-1): 
    #   h0 has h[j]=0, so h0[j]*(1-h0[j]) = 0
    #   h1 has h[j]=1, so h1[j]*(1-h1[j]) = 0
    #   The actual term is t*(1-t)
    #   So corr[0](t) = corr0[0] + t*(1-t) ...

    # corr1 = corr with h[j]=1

    #   corr0 at index n-1 includes the term 0*(1-0) = 0 at position j
    #   corr1 at index n-1 includes the term 1*(1-1) = 0 at position j

            # The quadratic change is t*(1-t) - old*(1-old)

                c = corr + delta * sens
                c[n-1] += (t*(1-t) - old*(1-old))
                return np.max(c)

                                  options={'xatol': 1e-10})

                delta = t_opt - old

                corr[n-1] += (t_opt*(1-t_opt) - old*(1-old))

                improved = True

    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov_topm(h0, n, target_sum, n_iters, alpha0, m_list, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    nml = len(m_list)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        m_cur = m_list[min(it * nml // n_iters, nml - 1)]
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = np.max(corr)
        weights = np.exp(40.0 * (corr - max_c))
        tw = weights.sum()
        if tw < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights /= tw
        top_m = np.argsort(weights)[::-1][:m_cur]
        w_m = weights[top_m].copy()
        w_m /= w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_m)):
            g += w_m[idx] * _grad_corr_term(y, top_m[idx], dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        tf = gen / max(n_gen - 1, 1)
        fm = 0.8 - 0.4 * tf
        cr = 0.5 + 0.3 * (1.0 - tf)
        bi = int(np.argmin(caches))
        imp = False
        np2 = [pop[bi].copy()]
        nc2 = [caches[bi]]
        for i in range(pop_size - 1):
            cands = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(cands), size=3, replace=False)
            r1, r2, r3 = cands[r1], cands[r2], cands[r3]
            mut = pop[r1] + fm * (pop[r2] - pop[r3])
            mut = _project(mut, target_sum)
            tri = pop[i].copy()
            for j in range(n):
                if rng.random() < cr:
                    tri[j] = mut[j]
            tri = _project(tri, target_sum)
            ct = _c5_of(tri)
            if ct <= caches[i]:
                tri = _fix_sum(tri, target_sum)
                np2.append(tri)
                nc2.append(ct)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = tri.copy()
                    imp = True
            else:
                np2.append(pop[i].copy())
                nc2.append(caches[i])
        pop = np2
        caches = nc2
        if not imp:
            stall += 1
        else:
            stall = 0
        if stall >= 20:
            break
    return best_h, best_c5

def _opt_coord_cd(h0, n, target_sum, max_rounds=20, deadline=None):
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)
    k0 = n - 1
    for r in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            sens = np.zeros(n_corr)
            for k_idx in range(n_corr):
                lag = abs(k_idx - k0)
                if lag > 0:
                    s = 0.0
                    if j + lag < n:
                        s += (1.0 - h[j + lag])
                    if j - lag >= 0:
                        s += (-h[j - lag])
                    sens[k_idx] = s
            base_corr = corr.copy()
            old_quad = old * (1.0 - old)

            def obj1d(t):
                d = t - old
                c_tmp = base_corr + d * sens
                c_tmp[k0] += t * (1.0 - t) - old_quad
                return np.max(c_tmp)

            try:
                res = minimize_scalar(obj1d, bounds=(0.0, 1.0), method='bounded',
                                      options={'xatol': 1e-10})
                t_opt = res.x
            except Exception:
                t_opt = old
            if abs(t_opt - old) > 1e-9:
                d = t_opt - old
                corr += d * sens
                corr[k0] += t_opt * (1.0 - t_opt) - old_quad
                h[j] = t_opt
                improved = True
        if not improved:
            break
    return h

def _run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 100
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_h = h_init.copy()
    best_n = n_init
    best_c5 = _c5_of(h_init)

    # Phase 1: DE at n_init
    t1 = deadline - 450
    pop_sz = 15
    h_de, c5_de = _de_optimize(h_init, n_init, n_init / 2.0, pop_sz, 50, t1, rng)
    if c5_de < best_c5:
        best_c5 = c5_de
        best_h = h_de.copy()

    # Phase 2: Nesterov at n_init
    if time.time() < deadline - 350:
        h_n, c5_n = _nesterov_topm(h_de, n_init, n_init / 2.0, 250, 0.4, [3, 5, 8, 12], deadline - 350)
        if c5_n < best_c5:
            best_c5 = c5_n
            best_h = h_n.copy()
        h_curr = h_n
    else:
        h_curr = best_h

    # Phase 3: SLSQP at n_init
    for alpha in [50, 200, 800]:
        if time.time() > deadline - 250:
            break
        h_sl = _slsqp_opt(h_curr, n_init, n_init / 2.0, alpha, 300)
        c5_sl = _c5_of(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl.copy()
            best_n = n_init
        h_curr = h_sl

    # Phase 4: Optimal coord descent at n_init
    if time.time() < deadline - 150:
        h_ocd = _opt_coord_cd(h_curr, n_init, n_init / 2.0, 15, deadline - 140)
        c5_ocd = _c5_of(h_ocd)
        if c5_ocd < best_c5:
            best_c5 = c5_ocd
            best_h = h_ocd.copy()
            best_n = n_init
        h_curr = h_ocd

    # Phase 5: Upsample to higher n and refine
    n_target = n_init + 60
    if n_target % 2 != 0:
        n_target += 1
    if n_target > 20:
        ts = n_target / 2.0
        if time.time() < deadline - 120:
            h_up = np.clip(np.interp(np.linspace(0, n_init - 1, n_target),
                                      np.arange(n_init), best_h), 0, 1)
            h_up = _fix_sum(h_up, ts)
            c5_up = _c5_of(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
                best_n = n_target

            h_n2, c5_n2 = _nesterov_topm(h_up, n_target, ts, 200, 0.3, [3, 5, 8, 12],
                                          deadline - 120)
            if c5_n2 < best_c5:
                best_c5 = c5_n2
                best_h = h_n2.copy()
                best_n = n_target
            h_curr2 = h_n2
        else:
            h_curr2 = best_h if best_n == n_target else h_up

        for alpha in [100, 400, 1500]:
            if time.time() > deadline - 60:
                break
            h_sl2 = _slsqp_opt(h_curr2, n_target, ts, alpha, 300)
            c5_sl2 = _c5_of(h_sl2)
            if c5_sl2 < best_c5:
                best_c5 = c5_sl2
                best_h = h_sl2.copy()
                best_n = n_target
            h_curr2 = h_sl2

        if time.time() < deadline - 40:
            h_ocd2 = _opt_coord_cd(h_curr2, n_target, ts, 12, deadline - 30)
            c5_ocd2 = _c5_of(h_ocd2)
            if c5_ocd2 < best_c5:
                best_c5 = c5_ocd2
                best_h = h_ocd2.copy()
                best_n = n_target
            h_curr2 = h_ocd2

    # Phase 6: Try even higher n if time allows
    n_hi = n_target + 40
    if n_hi % 2 != 0:
        n_hi += 1
    if time.time() < deadline - 60 and n_hi <= 280:
        ts_hi = n_hi / 2.0
        h_hi = np.clip(np.interp(np.linspace(0, best_n - 1, n_hi),
                                  np.arange(best_n), best_h), 0, 1)
        h_hi = _fix_sum(h_hi, ts_hi)
        if time.time() < deadline - 50:
            h_n3, c5_n3 = _nesterov_topm(h_hi, n_hi, ts_hi, 150, 0.3, [3, 5, 8],
                                          deadline - 50)
            if c5_n3 < best_c5:
                best_c5 = c5_n3
                best_h = h_n3.copy()
                best_n = n_hi
            h_curr3 = h_n3
        else:
            h_curr3 = h_hi
        if time.time() < deadline - 30:
            h_sl3 = _slsqp_opt(h_curr3, n_hi, ts_hi, 500, 250)
            c5_sl3 = _c5_of(h_sl3)
            if c5_sl3 < best_c5:
                best_c5 = c5_sl3
                best_h = h_sl3.copy()
                best_n = n_hi
            h_curr3 = h_sl3
        if time.time() < deadline - 20:
            h_ocd3 = _opt_coord_cd(h_curr3, n_hi, ts_hi, 10, deadline - 15)
            c5_ocd3 = _c5_of(h_ocd3)
            if c5_ocd3 < best_c5:
                best_c5 = c5_ocd3
                best_h = h_ocd3.copy()
                best_n = n_hi

    # Phase 7: Final aggressive polish at best_n
    target = best_n / 2.0
    h_fin = best_h.copy()
    for alpha in [500, 2000, 8000]:
        if time.time() > deadline - 15:
            break
        h_fin = _slsqp_opt(h_fin, best_n, target, alpha, 400)
        c5_f = _c5_of(h_fin)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fin.copy()

    if time.time() < deadline - 8:
        h_fd = _opt_coord_cd(best_h, best_n, target, 15, deadline - 5)
        c5_fd = _c5_of(h_fd)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_fd.copy()

    if time.time() < deadline - 2:
        h_pf = _slsqp_opt(best_h, best_n, target, 20000, 300)
        c5_pf = _c5_of(h_pf)
        if c5_pf < best_c5:
            best_c5 = c5_pf
            best_h = h_pf.copy()

    if time.time() < deadline:
        h_fd2 = _opt_coord_cd(best_h, best_n, target, 10, deadline)
        c5_fd2 = _c5_of(h_fd2)
        if c5_fd2 < best_c5:
            best_c5 = c5_fd2
            best_h = h_fd2.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)

def run(seed=42, budget_s=1300, **kwargs):
    return _run(seed=seed, budget_s=budget_s, **kwargs)
