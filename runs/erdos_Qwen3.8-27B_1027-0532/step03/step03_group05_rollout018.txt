```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_weighted_gradient(h, weights):
    """
    weights: array of length 2n-1 (same as corr)
    Returns gradient of sum_k weights[k] * corr[k]
    """
    n = len(h)
    grad = np.zeros(n)
    # corr[l] where l = 0..2n-2, lag = l - (n-1)
    for m in range(n):
        g = 0.0
        for lag in range(1, n):
            l_pos = (n - 1) + lag  # index in corr for positive lag
            l_neg = (n - 1) - lag  # index in corr for negative lag
            # Positive lag contribution to h[m]:
            # h[m] appears as h[i+lag] when i = m-lag, valid if 0 <= m-lag <= n-1-lag => lag <= m <= n-1
            if m >= lag:
                g += weights[l_pos] * (1.0 - h[m - lag])
            # h[m] appears as h[i] when i = m, valid if 0 <= m <= n-1-lag
            if m <= n - 1 - lag:
                g -= weights[l_pos] * h[m + lag]
            # Negative lag contribution to h[m]:
            # For lag' = -lag, corr[n-1-lag] = sum_{i=lag}^{n-1} h[i-lag] * (1-h[i])
            # h[m] appears as h[i-lag] when i = m+lag, valid if lag <= m+lag <= n-1 => 0 <= m <= n-1-lag
            if m <= n - 1 - lag:
                g += weights[l_neg] * (1.0 - h[m + lag])
            # h[m] appears as h[i] when i = m, valid if lag <= m <= n-1
            if m >= lag:
                g -= weights[l_neg] * h[m - lag]
        grad[m] = g
    return grad
```

```python
from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_box_sum(h, target_sum):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}"""
    h = h.copy()
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Then adjust sum
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference to coordinates that can accept it
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def weighted_gradient(h, corr_weights):
    """
    Compute gradient of sum_l corr_weights[l] * corr[l]
    where corr = np.correlate(h, 1-h, 'full')
    """
    n = len(h)
    grad = np.zeros(n)
    for lag in range(1, n):
        w_pos = corr_weights[n - 1 + lag]
        w_neg = corr_weights[n - 1 - lag]
        if abs(w_pos) > 1e-15:
            grad[lag:] += w_pos * (1.0 - h[:n - lag])
            grad[:n - lag] -= w_pos * h[lag:]
        if abs(w_neg) > 1e-15:
            grad[:n - lag] += w_neg * (1.0 - h[lag:])
            grad[lag:] -= w_neg * h[:n - lag]
    return grad

    v = beta2 * v + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    return lr * m_hat / (np.sqrt(v_hat) + eps)
```

```python
def project_to_box_sum(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)

    if diff > 0:

    else:

        return h

def slsqp_phase(h, target, time_limit, alpha=100):

    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}]

        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    remaining = time_limit - (time.time() - start)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_to_box_sum(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        count = np.sum(mask)
    else:
        mask = h > 0.0
        count = np.sum(mask)
    if count == 0:
        return h
    h[mask] += diff / count
    return np.clip(h, 0.0, 1.0)

def weighted_gradient(h, corr_weights):
    n = len(h)
    grad = np.zeros(n)
    for lag in range(1, n):
        w_pos = corr_weights[n - 1 + lag]
        w_neg = corr_weights[n - 1 - lag]
        if abs(w_pos) > 1e-15:
            grad[lag:] += w_pos * (1.0 - h[:n - lag])
            grad[:n - lag] -= w_pos * h[lag:]
        if abs(w_neg) > 1e-15:
            grad[:n - lag] += w_neg * (1.0 - h[lag:])
            grad[lag:] -= w_neg * h[:n - lag]
    return grad

def coord_gradient_row(h, i):
    """Compute the i-th row of the Jacobian: d(corr[k])/dh[i] for all k"""
    n = len(h)
    g = np.zeros(2 * n - 1)
    for lag in range(1, n):
        d_pos = 0.0
        if i >= lag:
            d_pos += (1.0 - h[i - lag])
        if i <= n - 1 - lag:
            d_pos -= h[i + lag]
        g[n - 1 + lag] = d_pos
        
        d_neg = 0.0
        if i <= n - 1 - lag:
            d_neg += (1.0 - h[i + lag])
        if i >= lag:
            d_neg -= h[i - lag]
        g[n - 1 - lag] = d_neg
    return g

def gradient_adam_phase(h, target, time_limit, alpha_start=15.0, lr_start=0.005, alpha_end=None):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    alpha = alpha_start
    lr = lr_start
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    patience = 0
    max_patience = 200
    if alpha_end is None:
        alpha_end = alpha_start * 5
    
    while time.time() - start < time_limit:
        t += 1
        # Gradually increase alpha
        progress = min(1.0, (time.time() - start) / time_limit)
        alpha_current = alpha_start + (alpha_end - alpha_start) * progress
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(corr)
        shifted = alpha_current * (corr - c_max)
        shifted = np.clip(shifted, -500.0, 0.0)
        weights = np.exp(shifted)
        sw = np.sum(weights)
        if sw < 1e-15:
            break
        weights = weights / sw
        
        grad = weighted_gradient(h, weights)
        grad -= np.mean(grad)
        
        # Clip gradient to prevent large steps
        gnorm = np.linalg.norm(grad)
        if gnorm > 10.0:
            grad *= 10.0 / gnorm
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        step -= np.mean(step)
        
        h_new = h - step
        h_new = project_to_box_sum(h_new, target)
        
        score = compute_c5(h_new)
        h = h_new
        
        if score < best_score - 1e-14:
            best_score = score
            best = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            lr = max(lr * 0.5, 1e-7)
            patience = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            if lr < 1e-7:
                break
    
    return best, best_score

def slsqp_phase(h, target, time_limit, alpha=100):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(corr)
        shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
        return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)
    
    def cons_fun(x):
        return np.sum(x) - target
    
    def cons_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    remaining = time_limit - (time.time() - start)
    maxiter = int(max(50, remaining * 2))
    
    try:
        res = minimize(obj, best, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_to_box_sum(res.x, target)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best = h_new
    except Exception:
        pass
    
    return best, best_score

def fast_coord_descent(h, target, time_limit, seed=42):

    start = time.time()

    no_improve_sweeps = 0

        improved = False
        order = rng.permutation(n)

            if no_improve_sweeps > 3:

    return best, best_score
```
