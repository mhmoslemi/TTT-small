    corr = np.correlate(h_current, 1-h_current, mode='full') * dx

    # For shift s (from -(n-1) to n-1):
    # grad_s[j] = dx * (1 - h_current[j+s]) if 0 <= j+s < n else 0

    #   dC(s)/dh[i+s] = dx * (-h[i]) for i in [0, n-1-s], i.e., h[j] for j in [s, n-1] gets -dx*h[j-s]
    # For s < 0 (let r = -s > 0): C(-r) = sum_{i: 0<=i<n, 0<=i-r<n} h[i] * (1-h[i-r]) * dx
    #   i ranges from r to n-1
    #   dC(-r)/dh[i] = dx * (1-h[i-r]) for i in [r, n-1]
    #   dC(-r)/dh[i-r] = dx * (-h[i]) for i in [r, n-1], i.e., h[j] for j in [0, n-1-r] gets -dx*h[j+r]

    #   For j in [0, n-1-s]: grad += dx*(1-h[j+s])
    #   For j in [s, n-1]: grad += -dx*h[j-s]

    #   For j in [r, n-1]: grad += dx*(1-h[j-r])
    #   For j in [0, n-1-r]: grad += -dx*h[j+r]
    
    # Let me compute all gradients at once
    n_shifts = 2*n - 1
    gradients = np.zeros((n_shifts, n))
    
    for idx in range(n_shifts):
        s = idx - (n - 1)

from scipy.optimize import minimize, linprog
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    best_h = h_current.copy()

        if time.time() - start_time > time_limit:

        # Minimize: C
        # Constraints:
        #   For each shift idx: const[idx] + grad[idx] @ h <= C
        #     => grad[idx] @ h - C <= -const[idx]
        #   sum(h) = n/2
        #   0 <= h_i <= 1
        #   C >= 0

        # Compute gradient matrix and constants
        # grad[idx, j] for shift s = idx - (n-1):

        # For each shift s:
        #   First part: (1-h[j+s]) for j in [0, n-1-s] (when s >= 0) or j in [r, n-1] (when s < 0, r=-s)
        #   Second part: -h[j-s] for j in [s, n-1] (when s >= 0) or j in [0, n-1-r] (when s < 0)
        
        A_rows = []
        b_rows = []
        
        for idx in range(n_shifts):
            s = idx - (n - 1)
            grad = np.zeros(n)
            const = corr[idx]

                grad[:m] += (1.0 - h_work[s:s+m])
                # Second part: for j in [s, n-1], -h[j-s]
                grad[s:] -= h_work[:m]
            else:
                r = -s
                m = n - r
                # First part: for j in [r, n-1], (1-h[j-r])
                grad[r:] += (1.0 - h_work[:m])
                # Second part: for j in [0, m-1], -h[j+r]
                grad[:m] -= h_work[r:r+m]
            
            grad *= dx
            const -= np.dot(grad, h_work)

            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0

            b_rows.append(-const)

        b_ub = np.array(b_rows)

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

            best_c5 = true_c5
            best_h = h_lp.copy()
            h_work = h_lp.copy()
        else:
            # Line search: try moving partway
            improved = False
            for alpha in [1.0, 0.5, 0.25, 0.1]:
                h_try = (1 - alpha) * h_work + alpha * h_lp

        # Check convergence

    target_sum = n / 2.0
    n_shifts = 2 * n - 1

    h_work = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    no_improve = 0
    max_no_improve = 200

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        corr = compute_corr_array(h_work)
        curr_c5 = float(np.max(corr))

        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_work.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                break

        A_rows = np.empty((n_shifts, n + 1))
        b_rows = np.empty(n_shifts)

        for idx in range(n_shifts):
            s = idx - (n - 1)
            grad = np.zeros(n)
            const = corr[idx]

            if s >= 0:
                m = n - s
                grad[:m] += (1.0 - h_work[s:s + m])
                grad[s:] -= h_work[:m]
            else:
                r = -s
                m = n - r
                grad[r:] += (1.0 - h_work[:m])
                grad[:m] -= h_work[r:r + m]

            grad *= dx
            const -= np.dot(grad, h_work)

            A_rows[idx, :n] = grad
            A_rows[idx, n] = -1.0
            b_rows[idx] = -const

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target_sum])

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0

        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        res = linprog(c_obj, A_ub=A_rows, b_ub=b_rows, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res.status != 0:
            break

        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = project_onto_constraint(h_lp, target_sum)

        best_alpha = 0.0
        best_alpha_c5 = curr_c5
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            h_try = (1.0 - alpha) * h_work + alpha * h_lp
            h_try = project_onto_constraint(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < best_alpha_c5:
                best_alpha_c5 = c5_try
                best_alpha = alpha

        if best_alpha > 0:
            h_work = (1.0 - best_alpha) * h_work + best_alpha * h_lp
            h_work = project_onto_constraint(h_work, target_sum)

    return best_h, best_c5


def slsqp_smooth_max(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))


def slsqp_smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * (1.0 - h[:m])
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * (1.0 - h[r:])
            grad[r:] -= wk * dx * h[:m]
    return grad


def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(slsqp_smooth_max, h0, args=(eps,),
                          jac=slsqp_smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)


def binary_search_from_continuous(h_cont, time_limit, start_time, rng, max_iters=500000):
    n = len(h_cont)
    dx = 2.0 / n
    target_sum = n / 2.0

    h_bin = (h_cont >= 0.5).astype(float)
    ones_count = int(h_bin.sum())
    if ones_count > target_sum:
        excess = ones_count - target_sum
        one_idx = np.where(h_bin == 1)[0]
        to_flip = one_idx[:excess]
        h_bin[to_flip] = 0
    elif ones_count < target_sum:
        deficit = target_sum - ones_count
        zero_idx = np.where(h_bin == 0)[0]
        to_flip = zero_idx[:deficit]
        h_bin[to_flip] = 1

    corr = compute_corr_array(h_bin)
    best_c5 = float(np.max(corr))
    best_h = h_bin.copy()
    best_corr = corr.copy()

    temperature = 0.002
    ones = np.where(h_bin == 1)[0].tolist()
    zeros = np.where(h_bin == 0)[0].tolist()

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break

        if it % 5000 == 0 and it > 0:
            temperature = max(0.0001, temperature * 0.95)

        worst_idx = int(np.argmax(corr))
        worst_s = worst_idx - (n - 1)

        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break

        i = ones[int(rng.integers(n_ones))]
        j = zeros[int(rng.integers(n_zeros))]

        delta_val = 0.0
        if 0 <= i - worst_s < n:
            delta_val += h_bin[i - worst_s]
        if 0 <= i + worst_s < n:
            delta_val += h_bin[i + worst_s]
        if 0 <= j + worst_s < n:
            delta_val -= h_bin[j + worst_s]
        if 0 <= j - worst_s < n:
            delta_val -= h_bin[j - worst_s]

        if delta_val < 0 or (delta_val == 0 and rng.random() < 0.3):
            new_corr = corr.copy()
            for s_idx in range(2 * n - 1):
                s = s_idx - (n - 1)
                v = 0.0
                if 0 <= i - s < n:
                    v += h_bin[i - s]
                if 0 <= i + s < n:
                    v += h_bin[i + s]
                if 0 <= j + s < n:
                    v -= h_bin[j + s]
                if 0 <= j - s < n:
                    v -= h_bin[j - s]
                new_corr[s_idx] += dx * v

            new_c5 = float(np.max(new_corr))

            if new_c5 <= best_c5 or np.exp((best_c5 - new_c5) / temperature) > rng.random():
                h_bin[i] = 0.0
                h_bin[j] = 1.0
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
                    best_corr = new_corr.copy()

    return best_h, best_c5


def gen_structured(n, rng, seed_offset):
    patterns = []
    p1 = n // 3
    h = np.zeros(n)
    for i in range(n):
        if (i % p1) < p1 // 2:
            h[i] = 1.0
    patterns.append(h)

    p2 = n // 5
    h2 = np.zeros(n)
    for i in range(n):
        if (i % p2) < p2 * 2 // 5:
            h2[i] = 1.0
    patterns.append(h2)

    tm = 0
    h3 = np.zeros(n)
    for i in range(n):
        v = bin(i).count('1')
        if v % 2 == 0:
            h3[i] = 1.0
    patterns.append(h3)

    h4 = (rng.random(n) < 0.5).astype(float)
    patterns.append(h4)

    h5 = np.zeros(n)
    periods = [7, 11, 13, 17]
    sub = periods[rng.integers(len(periods))]
    for i in range(n):
        if (i % sub) < sub // 2:
            h5[i] = 1.0
    patterns.append(h5)

    idx = (seed_offset) % len(patterns)
    return patterns[idx]


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = (rng.random(180) < 0.5).astype(float)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def elapsed():
        return time.time() - start_time

    def rem():
        return budget_s - elapsed()

    # Phase 1: SLP optimization
    if rem() > 100:
        tl = min(rem() - 50, 300)
        h_slp, c5_slp = slp_optimize(best_h, tl, start_time, max_iters=8000)
        if c5_slp < best_c5:
            best_c5 = c5_slp
            best_h = h_slp.copy()

    # Phase 2: SLSQP refinement
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: SLP again from refined point
    if rem() > 80:
        tl = min(rem() - 40, 200)
        h_slp2, c5_slp2 = slp_optimize(best_h, tl, start_time, max_iters=6000)
        if c5_slp2 < best_c5:
            best_c5 = c5_slp2
            best_h = h_slp2.copy()

    # Phase 4: Binary local search
    if rem() > 60:
        tl_bin = min(rem() - 30, 120)
        h_bin, c5_bin = binary_search_from_continuous(best_h, tl_bin, start_time, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        else:
            h_interp = project_onto_constraint(0.5 * best_h + 0.5 * h_bin, target_sum)
            h_opt_m, c5_m = slsqp_optimize(h_interp, 0.002, 200)
            if c5_m < best_c5:
                best_c5 = c5_m
                best_h = h_opt_m.copy()

    # Phase 5: SLP from binary result
    if rem() > 60:
        tl = min(rem() - 30, 150)
        h_slp3, c5_slp3 = slp_optimize(best_h, tl, start_time, max_iters=5000)
        if c5_slp3 < best_c5:
            best_c5 = c5_slp3
            best_h = h_slp3.copy()

    # Phase 6: Multiple SLSQP with decreasing eps
    if rem() > 50:
        for eps_val in [0.001, 0.0005, 0.0002]:
            if rem() < 30:
                break
            h_opt_e, c5_e = slsqp_optimize(best_h, eps_val, 200)
            if c5_e < best_c5:
                best_c5 = c5_e
                best_h = h_opt_e.copy()

    # Phase 7: Restart with structured initializations
    n_restarts = 0
    while rem() > 80 and n_restarts < 10:
        h_struct = gen_structured(n0, rng, seed + n_restarts)
        h_pert = 0.6 * best_h + 0.4 * h_struct
        h_pert = project_onto_constraint(h_pert, target_sum)

        h_opt_r, c5_r = slp_optimize(h_pert, min(rem() - 40, 60), start_time, max_iters=3000)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_opt_r.copy()

        h_opt_r2, c5_r2 = slsqp_optimize(h_opt_r, 0.001, 150)
        if c5_r2 < best_c5:
            best_c5 = c5_r2
            best_h = h_opt_r2.copy()

        n_restarts += 1

    # Phase 8: Resolution increase
    if rem() > 150:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_up_slp, c5_up_slp = slp_optimize(h_up, min(rem() - 80, 80), start_time, max_iters=3000)
        if rem() > 50:
            h_up_opt, c5_up_opt = slsqp_optimize(h_up_slp, 0.002, 200)
            if c5_up_opt < c5_up_slp:
                c5_up_slp = c5_up_opt
                h_up_slp = h_up_opt

        if c5_up_slp < best_c5:
            best_c5 = c5_up_slp
            best_h = h_up_slp.copy()
            n0 = n1

    # Phase 9: Final SLP polish
    if rem() > 30:
        tl_final = min(rem() - 10, 60)
        h_final_slp, c5_final = slp_optimize(best_h, tl_final, start_time, max_iters=3000)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final_slp.copy()

    # Phase 10: Final SLSQP
    if rem() > 20:
        h_last, c5_last = slsqp_optimize(best_h, 0.0001, 150)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
