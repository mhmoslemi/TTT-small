def project_box_hyperplane(h, target):
    """Project h onto {0 ≤ x ≤ 1, sum(x) = target}."""
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-14:
            break
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, alpha, n_active=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    c_max = np.max(C)
    shifted = np.clipalpha * (C - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = c_max + np.log(sum_exp) / alpha
    weights = exp_s / (sum_exp * alpha)
    
    if n_active < 2 * n - 1:
        top_idx = np.argsort(weights)[-n_active:]
    else:
        top_idx = np.arange(2 * n - 1)
    
    grad = np.zeros(n)
    i_arr = np.arange(n)
    for lag in top_idx:
        w = weights[lag]
        if w < 1e-15:
            continue
        k = lag - (n - 1)
        j_up = i_arr + k
        m_up = (j_up >= 0) & (j_up < n)
        j_dn = i_arr - k
        m_dn = (j_dn >= 0) & (j_dn < n)
        d = np.zeros(n)
        d[m_up] += (1.0 - h[j_up[m_up]])
        d[m_dn] -= h[j_dn[m_dn]]
        grad += w * dx * d
    return obj, grad

def proj_grad_method(h_init, n, target, alphas, time_limit, max_iter=10000, n_active=80):
    start_time = time.time()
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = 0.01
    
    for alpha in alphas:
        if time.time() - start_time > time_limit:
            break
        for it in range(max_iter):
            if time.time() - start_time > time_limit:
                break
            obj, grad = smooth_obj_grad(h, alpha, n_active)
            h_new = h - lr * grad
            h_new = project_feasible(h_new, target)
            sc = compute_c5(h_new)
            if sc < best_score:
                best_score = sc
                best_h = h_new.copy()
                h = h_new
                lr = min(0.1, lr * 1.05)
            else:
                lr = max(0.001, lr * 0.7)
                h_new = h - lr * grad
                h_new = project_feasible(h_new, target)
                sc = compute_c5(h_new)
                if sc < best_score:
                    best_score = sc
                    best_h = h_new.copy()
                    h = h_new
    return best_h, best_score
