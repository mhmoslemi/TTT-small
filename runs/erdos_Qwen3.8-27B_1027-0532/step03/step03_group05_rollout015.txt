from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cmax = np.max(corr)
    shifted = np.clip(alpha * (corr - cmax), -500.0, 0.0)
    exp_s = np.exp(shifted)
    sw = np.sum(exp_s)
    weights = exp_s / sw
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        w = weights[n-1+k]
        if w < 1e-15:
            continue
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += w * (1.0 - h[lo1+k:hi1+k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] -= w * h[lo2-k:hi2-k]
    return dx * grad

def gradient_descent_smooth(h, target, time_limit, alpha_init=10.0, alpha_end=200.0, eta_init=0.01):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)
    alpha = alpha_init
    eta = eta_init
    no_improve = 0
    
    while time.time() - start < time_limit:
        g = smooth_grad(h, alpha)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        # Descent direction is -g, but we need to maintain sum constraint
        # Project gradient onto the constraint space
        g_proj = g - np.mean(g)  # project to sum-zero subspace
        h_new = h - eta * g_proj / (np.linalg.norm(g_proj) + 1e-15)
        h_new = project_feasible(h_new, target)
        s = compute_c5(h_new)
        if s < best_score:
            best_score = s
            best = h_new.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve > 30:
                eta *= 0.8
                no_improve = 0
            if no_improve > 100:
                break
        h = h_new
        # Slowly increase alpha
        alpha = min(alpha * 1.01, alpha_end)
    
    return best, best_score

                h_trial = best.copy()
                h_trial[i] = val
                h_trial = project_feasible(h_trial, target)
                s = compute_c5(h_trial)
                if s < best_score - 1e-15:
                    best_score = s
                    best = h_trial
                    improved = True
                    break
        if not improved:
            delta_values = [d * 0.5 for d in delta_values]
            if max(abs(d) for d in delta_values) < 5e-6:
                break
    return best, best_score


def phase_annealing(h, target, time_limit, seed):
    n = len(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    best = h.copy()
    best_score = compute_c5(h)
    T = 0.01

    while time.time() - start < time_limit:
        T *= 0.998
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.uniform(0.001, 0.05)
        if best[i] < d or 1.0 - best[j] < d:
            if best[j] < d or 1.0 - best[i] < d:
                continue
            i, j = j, i
        h_trial = best.copy()
        h_trial[i] -= d
        h_trial[j] += d
        s = compute_c5(h_trial)
        if s < best_score:
            best_score = s
            best = h_trial
        elif rng.rand() < np.exp(-(s - best_score) / max(T * 0.3, 1e-12)):
            best = h_trial

    return best, best_score


def slsqp_smooth(h, target, time_limit, alpha):
    n = len(h)
    start = time.time()
    best = h.copy()
    best_score = compute_c5(h)

    def obj_func(x):
        return smooth_obj(x, alpha)

    def cons_fun(x):
        return np.sum(x) - target

    def cons_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]

    remaining = time_limit - (time.time() - start)
    maxiter = max(50, int(remaining * 2))

    try:
        res = minimize(obj_func, best, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, target)
        s = compute_c5(h_new)
        if s < best_score:
            best = h_new
            best_score = s
    except Exception:
        pass

    return best, best_score


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    h = np.array(initial_h_values, dtype=float)
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(best_h)

    def tleft():
        return budget_s - 25.0 - (time.time() - start_time)

    # Phase 1: Coarse gradient descent
    if tleft() > 200:
        h1, s1 = phase_gradient(best_h, target, min(200, tleft()),
                                alpha_start=5.0, alpha_end=50.0, eta_init=0.03)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    # Phase 2: Coarse pair moves
    if tleft() > 250:
        h2, s2 = phase_coarse_pair(best_h, target, min(250, tleft()), seed=seed)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    # Phase 3: Fine gradient descent with higher alpha
    if tleft() > 200:
        h3, s3 = phase_gradient(best_h, target, min(200, tleft()),
                                alpha_start=30.0, alpha_end=200.0, eta_init=0.015)
        if s3 < best_score:
            best_score = s3
            best_h = h3

    # Phase 4: Fine pair moves
    if tleft() > 300:
        h4, s4 = phase_pair_moves(best_h, target, min(300, tleft()), seed=seed + 1)
        if s4 < best_score:
            best_score = s4
            best_h = h4

    # Phase 5: SLSQP with smooth objective
    if tleft() > 150:
        h5, s5 = slsqp_smooth(best_h, target, min(150, tleft()), alpha=100.0)
        if s5 < best_score:
            best_score = s5
            best_h = h5

    # Phase 6: Fine coordinate descent
    if tleft() > 200:
        h6, s6 = phase_fine_coord(best_h, target, min(200, tleft()), seed=seed + 2)
        if s6 < best_score:
            best_score = s6
            best_h = h6

    # Phase 7: Very fine pair moves
    if tleft() > 150:
        fine_deltas = [0.0005, 0.001, 0.002, 0.005, 0.01]
        h7, s7 = phase_pair_moves(best_h, target, min(150, tleft()),
                                  seed=seed + 3, delta_list=fine_deltas)
        if s7 < best_score:
            best_score = s7
            best_h = h7

    # Phase 8: Simulated annealing for exploration
    if tleft() > 100:
        h8, s8 = phase_annealing(best_h, target, min(100, tleft()), seed=seed + 4)
        if s8 < best_score:
            best_score = s8
            best_h = h8

    # Phase 9: Final gradient polish
    if tleft() > 80:
        h9, s9 = phase_gradient(best_h, target, min(80, tleft()),
                                alpha_start=100.0, alpha_end=500.0, eta_init=0.005)
        if s9 < best_score:
            best_score = s9
            best_h = h9

    # Phase 10: Final fine coordinate descent
    if tleft() > 40:
        h10, s10 = phase_fine_coord(best_h, target, max(20, tleft() - 5), seed=seed + 5)
        if s10 < best_score:
            best_score = s10
            best_h = h10

    h_final = project_feasible(best_h, target)
    final_score = compute_c5(h_final)
    if final_score < best_score:
        best_score = final_score
        best_h = h_final

    return best_h, best_score, n
