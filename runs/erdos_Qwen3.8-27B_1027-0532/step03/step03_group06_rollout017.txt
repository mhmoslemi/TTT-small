```python
import time

    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_gradient(h, top_k=5):
    """Compute gradient of max_k corr_k using top-k lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top k lags
    if top_k >= len(corr):
        top_indices = np.arange(len(corr))
    else:
        top_indices = np.argsort(corr)[-top_k:]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        s = k_idx - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] -= h[:n-s]
        else:
            t = -s
            if n - t > 0:
                grad[t:] += (1.0 - h[:n-t])
                grad[:n-t] -= h[t:]
    return corr, grad / len(top_indices)

def projected_gradient_step(h, grad, step_size, target_sum):
    """Take a projected gradient step."""
    h_new = h - step_size * grad
    return project_feasible(h_new, target_sum)

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def fast_gradient(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    k_use = min(top_k, m)
    top_indices = np.argsort(corr)[-k_use:]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        s = int(k_idx) - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] -= h[:n-s]
        else:
            t = -s
            if n - t > 0:
                grad[t:] += (1.0 - h[:n-t])
                grad[:n-t] -= h[t:]
    return corr, grad / k_use

def objective_with_penalty(h, target, penalty):
    score = compute_c5(h)
    sum_viol = (np.sum(h) - target) ** 2
    return score + penalty * sum_viol

def obj_grad_with_penalty(h, target, penalty, top_k=3):
    corr, grad = fast_gradient(h, top_k)
    score = float(np.max(corr)) * (2.0 / len(h))
    sum_viol = np.sum(h) - target
    grad = grad * (2.0 / len(h)) + 2 * penalty * sum_viol * np.ones(len(h))
    return score + penalty * sum_viol**2, grad

def projected_momentum(h, target, n_iter, top_k_schedule, lr_schedule, time_check):
    n = len(h)
    h_cur = h.copy()
    velocity = np.zeros(n)
    momentum = 0.9
    best_local = compute_c5(h_cur)
    best_h_local = h_cur.copy()
    
    for i in range(n_iter):
        if time_check() < 30:
            break
        top_k = top_k_schedule
        lr = lr_schedule(i)
        
        corr, grad = fast_gradient(h_cur, top_k)
        # Scale gradient to match objective scale
        grad = grad * (2.0 / n)
        
        # Momentum update
        velocity = momentum * velocity - lr * grad
        h_new = project_feasible(h_cur + velocity, target)
        
        sc = compute_c5(h_new)
        if sc < best_local:
            best_local = sc
            best_h_local = h_new.copy()
        h_cur = h_new
    
    return best_h_local, best_local

def lbfgs_polish(h_start, target, top_k, maxiter, time_check):
    n = len(h_start)
    penalty = 100.0
    
    def obj(x):
        return objective_with_penalty(x, target, penalty)
    
    def jac(x):
        _, g = obj_grad_with_penalty(x, target, penalty, top_k)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, jac=jac, method='L-BFGS-B',
                       bounds=bounds, 
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        h_new = project_feasible(res.x, target)
        return h_new
    except:
        return h_start

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
```

```python
    grad = np.zeros(n)
    if s >= 0:

            grad[:n-s] += (1.0 - h[s:n])  # h[j+s] for j in [0, n-1-s]
```

```python
def fast_gradient(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)  # 2n-1
    k_use = min(top_k, m)
    top_indices = np.argsort(corr)[-k_use:]
    
    grad = np.zeros(n)
    for k_idx in top_indices:
        s = int(k_idx) - (n - 1)
        if s >= 0:
            if s < n:
                grad[:n-s] += (1.0 - h[s:n])
                grad[s:n] -= h[0:n-s]
        else:
            t = -s
            if t < n:
                grad[t:n] += (1.0 - h[0:n-t])
                grad[0:n-t] -= h[t:n]
    grad /= k_use
    return corr, grad
```

If we could achieve a "perfectly balanced" sequence where for every k, the number of 1-to-0 and 0-to-1 transitions at shift k is exactly n/4 - something, that would be ideal.

Actually, for a random binary sequence with 200 ones and 200 zeros, the expected number of 1-to-0 pairs at shift k is:
E[Σ_i h[i](1-h[i+k])] = (n-|k|) * (200/400) * (200/400) = (n-|k|) * 1/4

So the expected max is roughly n/4 = 100 (at k=0 it's 0, but at small k it's close to 100).

C₅ for random binary ≈ 100 * (2/400) = 0.5. That's worse than our current 0.38.

So binary solutions are not good. The optimal solution must use non-binary values to balance the correlations more evenly.

This makes sense: by using values in (0,1), we can "spread out" the correlation more evenly across shifts.

```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(50):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)
    return h


def fast_gradient(h, top_k=3):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    k_use = min(top_k, m)
    top_indices = np.argsort(corr)[-k_use:]
    grad = np.zeros(n)
    for ki in range(len(top_indices)):
        k_idx = int(top_indices[ki])
        s = k_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[:n - s] += (1.0 - h[s:n])
                grad[s:n] -= h[0:n - s]
        else:
            t = -s
            if t < n:
                grad[t:n] += (1.0 - h[0:n - t])
                grad[0:n - t] -= h[t:n]
    grad /= k_use
    return corr, grad


def obj_and_grad(h, target, penalty, top_k):
    n = len(h)
    corr, grad = fast_gradient(h, top_k)
    score = float(np.max(corr)) * (2.0 / n)
    sv = np.sum(h) - target
    obj = score + penalty * sv * sv
    grad = grad * (2.0 / n) + 2.0 * penalty * sv
    return obj, grad


def lbfgs_optimize(h_start, target, top_k, penalty, maxiter, time_check):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    penalty_list = []

    def obj_func(x):
        o, _ = obj_and_grad(x, target, penalty, top_k)
        return o

    def jac_func(x):
        _, g = obj_and_grad(x, target, penalty, top_k)
        return g

    try:
        res = minimize(obj_func, h_start, jac=jac_func, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9,
                                'maxls': 50})
        h_new = project_feasible(res.x, target)
        return h_new
    except Exception:
        return h_start


def projected_grad_descent(h_start, target, n_iter, top_k, lr, time_check):
    n = len(h_start)
    h_cur = h_start.copy()
    best_h = h_cur.copy()
    best_sc = compute_c5(h_cur)
    vel = np.zeros(n)
    mom = 0.7
    stall = 0
    for i in range(n_iter):
        if time_check() < 25:
            break
        _, grad = fast_gradient(h_cur, top_k)
        grad = grad * (2.0 / n)
        vel = mom * vel - lr * grad
        h_new = project_feasible(h_cur + vel, target)
        sc = compute_c5(h_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                vel = np.zeros(n)
                mom = min(0.9, mom + 0.05)
                stall = 0
        h_cur = h_new
    return best_h, best_sc


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_refine(h, target, block_size, n_rounds, time_check):
    n = len(h)
    best_sc = compute_c5(h)
    h_cur = h.copy()
    for rnd in range(n_rounds):
        if time_check() < 25:
            break
        improved = False
        for b_start in range(0, n, block_size):
            b_end = min(b_start + block_size, n)
            h_block = h_cur[b_start:b_end].copy()
            for it in range(50):
                _, grad_full = fast_gradient(h_cur, top_k=3)
                gb = grad_full[b_start:b_end] * (2.0 / n)
                h_block = project_feasible(h_block - 0.05 * gb,
                                           np.sum(h_block))
                h_cur[b_start:b_end] = h_block
            h_cur = project_feasible(h_cur, target)
            sc = compute_c5(h_cur)
            if sc < best_sc - 1e-12:
                best_sc = sc
                improved = True
        if not improved:
            break
    return h_cur, best_sc
```
