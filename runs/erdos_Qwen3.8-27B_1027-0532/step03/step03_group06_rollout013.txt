def compute_grad_direct(h, alpha):
    """Compute gradient of smooth objective using direct O(n^2) method."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))
    sum_exp = np.sum(exp_vals)

    # T1[j] = conv((1-h), w[::-1], mode='full')[n-1+j]
    # T2[j] = conv(h, w, mode='full')[n-1+j]
    t1 = np.convolve(1.0 - h, w[::-1], mode='full')[n-1 : 2*n-1]
    t2 = np.convolve(h, w, mode='full')[n-1 : 2*n-1]
    grad = t1 - t2
    return w, grad, m

def compute_grad_fft(h, alpha):

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_vals = np.exp(alpha * (corr - m))

    # Use FFT for convolution

    H = np.fft.rfft(1.0 - h, size)
    W_rev = np.fft.rfft(w[::-1], size)

    t1 = t1_full[n-1 : 2*n-1]
    
    H2 = np.fft.rfft(h, size)
    W = np.fft.rfft(w, size)
    t2_full = np.fft.irfft(H2 * W, size)
    t2 = t2_full[n-1 : 2*n-1]
    
    grad = t1 - t2
    return w, grad, m

    return float(np.max(corr)) * dx


def project_feasible_fast(x, target_sum):
    n = len(x)
    if n == 0:
        return x.copy()
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def compute_smooth_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = alpha * (corr - m)
    max_shifted = np.max(shifted)
    exp_vals = np.exp(shifted - max_shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + max_shifted / alpha + np.log(sum_exp) / alpha
    if n <= 600:
        t1 = np.convolve(1.0 - h, w[::-1], mode='full')
        t2 = np.convolve(h, w, mode='full')
        grad = t1[n - 1:2 * n - 1] - t2[n - 1:2 * n - 1]
    else:
        size = 1
        needed = n + 2 * n - 1
        while size < needed:
            size *= 2
        H1 = np.fft.rfft(1.0 - h, size)
        Wr = np.fft.rfft(w[::-1], size)
        t1 = np.fft.irfft(H1 * Wr, size)[n - 1:2 * n - 1]
        H2 = np.fft.rfft(h, size)
        W = np.fft.rfft(w, size)
        t2 = np.fft.irfft(H2 * W, size)[n - 1:2 * n - 1]
        grad = t1 - t2
    return F, grad


def gradient_descent_loop(h_start, target, alpha, max_iters, time_limit_fn):
    h = h_start.copy()
    n = len(h)
    F, grad = compute_smooth_and_grad(h, alpha)
    step = 1.0
    for it in range(max_iters):
        if time_limit_fn():
            break
        if np.linalg.norm(grad) < 1e-12:
            break
        improved = False
        for ls in range(40):
            h_trial = project_feasible_fast(h - step * grad, target)
            F_trial, _ = compute_smooth_and_grad(h_trial, alpha)
            if F_trial < F:
                h = h_trial
                F = F_trial
                improved = True
                break
            step *= 0.5
        if not improved:
            step = 1.0
            break
        _, grad = compute_smooth_and_grad(h, alpha)
    return h, F


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible_fast(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible_fast(h_out, target_dst)


def refine_solution(h_start, target, alpha_list, iters_per_alpha, time_limit_fn):
    h = h_start.copy()
    for alpha, maxit in zip(alpha_list, iters_per_alpha):
        if time_limit_fn():
            break
        h, _ = gradient_descent_loop(h, target, alpha, maxit, time_limit_fn)
    return h


def try_perturbation(best_h, target, idx, rng, time_limit_fn, alpha_list, maxit):
    n = len(best_h)
    ptype = idx % 6
    if ptype == 0:
        amp = 0.003 + 0.003 * (idx % 10)
        h_pert = project_feasible_fast(best_h + rng.randn(n) * amp, target)
    elif ptype == 1:
        h_pert = best_h.copy()
        seg_len = max(3, n // 15)
        a = rng.randint(0, n - seg_len)
        shift_amt = rng.uniform(-0.4, 0.4)
        h_pert[a:a + seg_len] += shift_amt
        h_pert = project_feasible_fast(h_pert, target)
    elif ptype == 2:
        h_pert = best_h.copy()
        edge = rng.randint(0, n)
        width = max(2, n // 25)
        lo = max(0, edge - width)
        hi = min(n, edge + width + 1)
        h_pert[lo:hi] = 1.0 - h_pert[lo:hi]
        h_pert = project_feasible_fast(h_pert, target)
    elif ptype == 3:
        shift = rng.choice([-2, -1, 1, 2, 3, -3])
        h_pert = np.roll(best_h, shift)
        h_pert = project_feasible_fast(h_pert, target)
    elif ptype == 4:
        h_pert = best_h.copy()
        seg_len = max(4, n // 10)
        a = rng.randint(0, n - seg_len)
        b = rng.randint(0, n - seg_len)
        if abs(a - b) > seg_len:
            tmp = h_pert[a:a + seg_len].copy()
            h_pert[a:a + seg_len] = h_pert[b:b + seg_len]
            h_pert[b:b + seg_len] = tmp
        h_pert = project_feasible_fast(h_pert, target)
    else:
        h_pert = best_h.copy()
        k = rng.randint(1, 4)
        noise = rng.randn(n) * 0.01
        smooth = np.convolve(noise, np.ones(k) / k, mode='same')
        h_pert = project_feasible_fast(best_h + smooth, target)
    h_new = refine_solution(h_pert, target, alpha_list, [maxit] * len(alpha_list), time_limit_fn)
    return h_new


def coordinate_descent_pass(h, target, max_passes, time_limit_fn):
    n = len(h)
    for passes in range(max_passes):
        if time_limit_fn():
            break
        improved = False
        perm = np.arange(n)
        rng_cd = np.random.RandomState(passes)
        rng_cd.shuffle(perm)
        for j in perm:
            if time_limit_fn():
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            m = np.max(corr)
            alpha = 200.0
            shifted = alpha * (corr - m)
            max_sh = np.max(shifted)
            exp_v = np.exp(shifted - max_sh)
            w = exp_v / np.sum(exp_v)
            grad_j = 0.0
            s_lo = 0
            s_hi = 2 * n - 2
            for k in range(s_lo, s_hi + 1):
                s = k - (n - 1)
                if j + s >= 0 and j + s < n:
                    grad_j += w[k] * (1.0 - h[j + s])
                if j - s >= 0 and j - s < n:
                    grad_j -= w[k] * h[j - s]
            if abs(grad_j) < 1e-10:
                continue
            direction = -np.sign(grad_j)
            best_val = h[j]
            best_obj = compute_c5(h)
            for step_size in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
                trial = h.copy()
                trial[j] = np.clip(h[j] + direction * step_size, 0.0, 1.0)
                trial = project_feasible_fast(trial, target)
                obj = compute_c5(trial)
                if obj < best_obj:
                    best_obj = obj
                    best_val = trial[j]
                    h = trial
                    improved = True
                    break
    return h, improved


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target = n_init / 2.0

    best_h = project_feasible_fast(h_init.copy(), target)
    best_score = compute_c5(best_h)

    def time_ok():
        return time.time() - start_time < budget_s - 20

    # Phase 1: Main gradient descent with increasing alpha
    alpha_main = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    iters_main = [200, 300, 400, 500, 600, 700, 700, 600, 500]
    for i in range(len(alpha_main)):
        if not time_ok():
            break
        best_h, _ = gradient_descent_loop(best_h, target, alpha_main[i], iters_main[i], time_ok)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    # Phase 2: Multi-start perturbations
    perturb_alpha = [200, 1000, 5000, 20000]
    perturb_iters = 300
    num_perturb = 60
    for attempt in range(num_perturb):
        if not time_ok():
            break
        h_new = try_perturbation(best_h, target, attempt, rng, time_ok, perturb_alpha, perturb_iters)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    # Phase 3: Coordinate descent on best solution
    if time_ok():
        best_h, _ = coordinate_descent_pass(best_h, target, 20, time_ok)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    # Phase 4: Higher resolution
    for n_try in [600, 800, 1000]:
        if not time_ok():
            break
        target_try = n_try / 2.0
        h_interp = interpolate_to(best_h, n_try, target_try)
        alpha_hires = [50, 200, 1000, 5000, 20000, 80000]
        iters_hires = [200, 300, 400, 500, 500, 400]
        for i in range(len(alpha_hires)):
            if not time_ok():
                break
            h_interp, _ = gradient_descent_loop(h_interp, target_try, alpha_hires[i], iters_hires[i], time_ok)
        sc = compute_c5(h_interp)
        if sc < best_score:
            best_score = sc
            best_h = h_interp

    # Phase 5: Final polish with very high alpha
    n_use = len(best_h)
    target_use = n_use / 2.0
    for alpha in [50000, 200000, 500000]:
        if not time_ok():
            break
        best_h, _ = gradient_descent_loop(best_h, target_use, alpha, 800, time_ok)
        sc = compute_c5(best_h)
        if sc < best_score:
            best_score = sc

    # Phase 6: Micro-perturbations at the end
    for _ in range(15):
        if not time_ok():
            break
        amp = rng.uniform(0.001, 0.004)
        h_pert = project_feasible_fast(best_h + rng.randn(n_use) * amp, target_use)
        h_new = refine_solution(h_pert, target_use, [50000, 200000], [400, 400], time_ok)
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new

    best_h = project_feasible_fast(best_h, len(best_h) / 2.0)
    final_score = compute_c5(best_h)
    return best_h, final_score, len(best_h)
