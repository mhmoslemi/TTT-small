```python
def fix_sum(h, target_sum):

    if abs(diff) < 1e-12:

        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:

        for i in order:
            if diff >= -1e-12:

            diff += sub
    return h
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -100:
        return m * (2.0 / n)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / n)

def smooth_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    weights = exps / exps.sum()
    dx_val = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mm = n - lag
            g[lag:lag+mm] += weights[k] * (1.0 - h[:mm])
            g[:mm] += weights[k] * (-h[lag:lag+mm])
        else:
            j = -lag
            mm = n - j
            g[:mm] += weights[k] * (1.0 - h[j:])
            g[j:] += weights[k] * (-h[:mm])
    return g * dx_val

def nesterov_topm(h0, n, target_sum, n_iters, alpha0, top_m=5, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_m]
        g = np.zeros(n)
        w = 1.0 / top_m
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            if lag >= 0:
                m = n - lag
                g[lag:lag+m] += w * (1.0 - y[:m])
                g[:m] += w * (-y[lag:lag+m])
            else:
                j = -lag
                m = n - j
                g[:m] += w * (1.0 - y[j:])
                g[j:] += w * (-y[:m])
        g *= dx
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def slsqp_smooth(h0, n, target_sum, alpha, maxiter, deadline=None):
    def obj(x):
        return smooth_obj(x, alpha, n)
    def obj_grad(x):
        return smooth_grad(x, alpha, n)
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        if deadline is not None and time.time() > deadline:
            return h0
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return fix_sum(res.x, target_sum)
    except:
        return h0

def slew_lbfgs(h0, n, target_sum, alpha, penalty, maxiter=1000, deadline=None):
    """L-BFGS-B with penalty for sum constraint."""
    def obj(x):
        return smooth_obj(x, alpha, n) + penalty * (x.sum() - target_sum) ** 2
    def obj_grad(x):
        return smooth_grad(x, alpha, n) + 2 * penalty * (x.sum() - target_sum) * np.ones(n)
    bounds = [(0.0, 1.0)] * n
    try:
        if deadline is not None and time.time() > deadline:
            return h0
        res = minimize(obj, h0, jac=obj_grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return fix_sum(res.x, target_sum)
    except:
        return h0

def binary_round(h, n, target_sum):
    """Round h to binary, adjusting to maintain sum."""
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    n_ones = int(h_bin.sum())
    target_ones = int(target_sum)  # should be n/2
    if n_ones > target_ones:
        # Need to flip some 1s to 0s, prefer those with lowest h value
        ones_mask = h_bin == 1
        ones_indices = np.where(ones_mask)[0]
        # Sort by h value (ascending) - flip the ones that are closest to 0.5
        order = np.argsort(h[ones_indices])
        flip_count = n_ones - target_ones
        for idx in order[:flip_count]:
            h_bin[ones_indices[idx]] = 0
    elif n_ones < target_ones:
        zeros_mask = h_bin == 0
        zeros_indices = np.where(zeros_mask)[0]
        order = np.argsort(-h[zeros_indices])  # flip those closest to 1
        flip_count = target_ones - n_ones
        for idx in order[:flip_count]:
            h_bin[zeros_indices[idx]] = 1
    return h_bin

def binary_local_search(h_bin, n, target_sum, max_rounds=20, deadline=None):
    """Local search on binary solutions: flip single bits that maintain the sum constraint."""
    h = h_bin.copy()
    current_c5 = c5_of(h)
    dx = 2.0 / n
    n_ones = int(h.sum())
    
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        improved = False
        # Try flipping a 1 to 0 and a 0 to 1 (maintains sum)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try each 1→0 flip paired with each 0→1 flip
        # This is O(n²/4) which is about 1225 for n=140
        # For each pair, evaluate the change
        for i in ones:
            if deadline and time.time() > deadline:
                break
            for j in zeros:
                h[i] = 0
                h[j] = 1
                c5 = c5_of(h)
                h[i] = 1
                h[j] = 0
                if c5 < current_c5 - 1e-14:
                    h[i] = 0
                    h[j] = 1
                    current_c5 = c5
                    improved = True
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    break
            if improved and rnd > 5:
                # Restart the inner loop after improvement
                break
        if not improved:
            break
    return h, current_c5

def sa_optimize(h0, n, target_sum, n_iter, t0, t1, deadline, rng):

    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    curr_c5 = best_c5
    target_ones = int(target_sum)

            break
        t = t0 * (t1 / t0) ** (float(it) / n_iter)

        elif move_type == 2:
            # Block reverse
            i = rng.randint(max(1, n - 10))
            length = rng.randint(3, min(11, n - i))
            h_new[i:i+length] = h_new[i:i+length][::-1]
        elif move_type == 3:

            i = rng.randint(max(1, n - 20))

            tmp = h_new[idx1].copy()
            h_new[idx1] = h_new[idx2]
            h_new[idx2] = tmp
        elif move_type == 4:

            h_new = project(h + noise, target_sum)

            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new

    return best_h, best_c5
```

But the "cost" depends on the interaction between i and j, so we can't separate them. Unless... we use a first-order approximation.

First-order approximation:
The change in max corr when flipping h[i] (1→0) and h[j] (0→1) is approximately:
Δ ≈ -∂f/∂h[i] + ∂f/∂h[j] (using the subgradient at the current point)

Where ∂f/∂h[i] is the component of the subgradient at position i.

This gives us a fast way to find promising pairs! We just need the subgradient, which we compute once per round.

Let me implement this:

```python
def subgradient_at(h, n):
    """Compute a subgradient of C5 at h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    # If multiple k have the same max, use the average
    max_val = np.max(corr)
    top_k = np.where(corr == max_val)[0]
    g = np.zeros(n)
    for k_idx in top_k:
        w = 1.0 / len(top_k)
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            g[lag:lag+m] += w * (1.0 - h[:m])
            g[:m] += w * (-h[lag:lag+m])
        else:
            j = -lag
            m = n - j
            g[:m] += w * (1.0 - h[j:])
            g[j:] += w * (-h[:m])
    return g * (2.0 / n)
```

Then for the binary local search:
- Compute subgradient g
- For the best flip pair: we want to maximize g[i] - g[j] where h[i]=1, h[j]=0
  (because flipping h[i] from 1 to 0 changes f by approximately -g[i], and flipping h[j] from 0 to 1 changes f by approximately +g[j], so total change ≈ -g[i] + g[j])
  
  Wait, we want to MINIMIZE f, so we want the most negative change.
  Change ≈ g[i] * (-1) + g[j] * (+1) = g[j] - g[i]
  We want this to be most negative, so we want g[i] - g[j] to be positive and large.
  So: pick i with h[i]=1 and large g[i], and j with h[j]=0 and small (negative) g[j].

This gives us the best single move in O(n) time!

Let me implement a "steepest descent" binary local search:

```python
def binary_steepest(h_bin, n, target_sum, max_rounds=50, deadline=None):
    h = h_bin.copy()
    current_c5 = c5_of(h)
    for rnd in range(max_rounds):
        if deadline and time.time() > deadline:
            break
        g = subgradient_at(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Find i in ones with max g[i], j in zeros with min g[j]
        best_i = ones[np.argmax(g[ones])]
        best_j = zeros[np.argmin(g[zeros])]
        h[best_i] = 0
        h[best_j] = 1
        new_c5 = c5_of(h)
        if new_c5 < current_c5 - 1e-15:
            current_c5 = new_c5
        else:
            h[best_i] = 1
            h[best_j] = 0
            # Try next best
            # Remove best_i from consideration and try again
            # For simplicity, just try all pairs sorted by g[i]-g[j]
            improved = False
            gs = g[ones][:, None] - g[zeros][None, :]
            order = np.argsort(-gs.ravel())
            for idx in order[:50]:
                ii = idx // len(zeros)
                jj = idx % len(zeros)
                hi = ones[ii]
                hj = zeros[jj]
                h[hi] = 0
                h[hj] = 1
                c5_try = c5_of(h)
                h[hi] = 1
                h[hj] = 0
                if c5_try < current_c5 - 1e-15:
                    h[hi] = 0
                    h[hj] = 1
                    current_c5 = c5_try
                    improved = True
                    break
            if not improved:
                break
    return h, current_c5
```

This is much more efficient! Each round is O(n²) for the subgradient + O(50*n) for the local evaluation = O(n²). For n=140, that's about 20K + 10K = 30K ops per round. With 50 rounds, that's 1.5M ops. Very fast.

```python
    best_h = h.copy()
```

```python
import time
import numpy as np
from scipy.optimize import minimize

def c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    if m < -100:
        return m * (2.0 / n)
    val = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    return val * (2.0 / n)

def smooth_grad(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    total = exps.sum()
    if total < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = exps / total
    dx_val = 2.0 / n
    g = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            mm = n - lag
            g[lag:lag+mm] += weights[k] * (1.0 - h[:mm])
            g[:mm] += weights[k] * (-h[lag:lag+mm])
        else:
            j = -lag
            mm = n - j
            g[:mm] += weights[k] * (1.0 - h[j:])
            g[j:] += weights[k] * (-h[:mm])
    return g * dx_val

def subgradient_at(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    top_k = np.where(corr >= max_val - 1e-10)[0]
    g = np.zeros(n)
    w = 1.0 / len(top_k)
    for k_idx in top_k:
        lag = k_idx - (n - 1)
        if lag >= 0:
            m = n - lag
            g[lag:lag+m] += w * (1.0 - h[:m])
            g[:m] += w * (-h[lag:lag+m])
        else:
            j = -lag
            m = n - j
            g[:m] += w * (1.0 - h[j:])
            g[j:] += w * (-h[:m])
    return g * (2.0 / n)

def nesterov_topm(h0, n, target_sum, n_iters, alpha0, top_m, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_m]
        g = np.zeros(n)
        w = 1.0 / top_m
        for k_idx in top_idx:
            lag = k_idx - (n - 1)
            if lag >= 0:
                m = n - lag
                g[lag:lag+m] += w * (1.0 - y[:m])
                g[:m] += w * (-y[lag:lag+m])
            else:
                j = -lag
                m = n - j
                g[:m] += w * (1.0 - y[j:])
                g[j:] += w * (-y[:m])
        g *= dx
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def slsqp_smooth(h0, n, target_sum, alpha, maxiter, deadline=None):
    def obj(x):
        return smooth_obj(x, alpha, n)
    def obj_grad(x):
        return smooth_grad(x, alpha, n)
    def con_fun(x):
        return x.sum() - target_sum
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        if deadline is not None and time.time() > deadline:
            return h0
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return fix_sum(res.x, target_sum)
    except Exception:
        return h0

def binary_round(h, n, target_sum):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target_ones = int(round(target_sum))
    if n_ones > target_ones:
        ones_indices = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_indices])
        flip_count = n_ones - target_ones
        for idx in order[:flip_count]:
            h_bin[ones_indices[idx]] = 0.0
    elif n_ones < target_ones:
        zeros_indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_indices])
        flip_count = target_ones - n_ones
        for idx in order[:flip_count]:
            h_bin[zeros_indices[idx]] = 1.0
    return h_bin

def binary_steepest(h_bin, n, target_sum, max_rounds=50, deadline=None):
    h = h_bin.copy()
    current_c5 = c5_of(h)
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        g = subgradient_at(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        gs = g[ones][:, None] - g[zeros][None, :]
        flat_order = np.argsort(-gs.ravel())
        improved = False
        for idx in flat_order[:80]:
            ii = int(idx) // len(zeros)
            jj = int(idx) % len(zeros)
            hi = ones[ii]
            hj = zeros[jj]
            h[hi] = 0.0
            h[hj] = 1.0
            c5_try = c5_of(h)
            if c5_try < current_c5 - 1e-14:
                current_c5 = c5_try
                improved = True
                break
            else:
                h[hi] = 1.0
                h[hj] = 0.0
        if not improved:
            break
    return h, current_c5

def sa_optimize(h0, n, target_sum, n_iter, t0, t1, deadline, rng):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    curr_c5 = best_c5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        t = t0 * (t1 / t0) ** (float(it) / max(n_iter, 1))
        move_type = rng.randint(6)
        h_new = h.copy()
        if move_type == 0:
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if len(ones) > 0 and len(zeros) > 0:
                i = int(rng.choice(ones))
                j = int(rng.choice(zeros))
                h_new[i], h_new[j] = h_new[j], h_new[i]
        elif move_type == 1:
            i, j = rng.choice(n, size=2, replace=False)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif move_type == 2:
            i = int(rng.randint(max(1, n - 12)))
            length = int(rng.randint(3, min(11, n - i)))
            h_new[i:i+length] = h_new[i:i+length][::-1]
        elif move_type == 3:
            i = int(rng.randint(max(1, n - 20)))
            length = int(rng.randint(2, min(6, n - i)))
            shift = int(rng.randint(1, min(5, n - i - length)))
            h_new[i:i+shift+length] = np.concatenate([h_new[i+shift:i+shift+length], h_new[i:i+shift]])
        elif move_type == 4:
            noise = rng.normal(0, 0.12, n)
            h_new = project(h + noise, target_sum)
        else:
            i = int(rng.randint(n))
            length = int(rng.randint(4, min(12, n - i)))
            h_new[i:i+length] = np.clip(h_new[i:i+length] + rng.normal(0, 0.2, length), 0, 1)
            h_new = fix_sum(h_new, target_sum)
        c5_new = c5_of(h_new)
        delta = c5_new - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            h = h_new
            curr_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5
```
