    n = len(h)
    dx = 2.0 / n
    return float(np.max(corr)) * dx, corr

def project_feasible(h, target_sum):

def smoothed_objective(h, alpha, corr):

    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def compute_grad_smoothed(h, corr, alpha):
    """Compute gradient of the log-sum-exp smoothed max of corr."""
    n = len(h)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)  # weights sum to 1

    # = 1 - sum_k w[k]*h[j-k+n-1] - sum_k w[k]*h[j+k-n-1]

        w_circ[N-l] = w[n-1-l]  # w_{-l}

    # = ifft(conj(fft(w_circ)) * fft(h))[j]
    # Let's verify: ifft(conj(W)*H)[j] = sum_k I(conj(W[k])*H[k]) * exp(2πijk/N)

def projected_gradient_descent(h, target, time_limit, alphas=None, eta_init=0.01, seed=42):

        t = 1.0

            grad = compute_grad_smoothed(z, corr, alpha)
            
            # Gradient step

            u = project_feasible(u, target)

            z = u + ((t - 1.0) / t_new) * (u - z)
            z = np.clip(z, 0.0, 1.0)
            t = t_new
            
            score = compute_c5(u)
            if score < best_score:
                best_score = score
                best = u.copy()
            
            # Adaptive step size (simplified backtracking)
            if it % 100 == 0:
                # Check if we're making progress

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr_full_fft(h):
    n = len(h)
    b = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    circ = np.fft.irfft(H * np.conj(B), N)
    full = np.zeros(2 * n - 1)
    full[:n] = circ[n-1::-1]
    full[n:] = circ[N-n+1:N][::-1]
    return full

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    if not np.any(mask):
        return h
    h[mask] += diff / np.sum(mask)
    return np.clip(h, 0.0, 1.0)

def compute_grad(h, corr, alpha):
    n = len(h)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    N = 1
    while N < 2 * n - 1:
        N *= 2
    
    w_circ = np.zeros(N)
    w_circ[0] = w[n-1]
    w_circ[1:n] = w[n:]
    w_circ[N-n+1:N] = w[:n-1][::-1]
    
    H = np.fft.rfft(h, N)
    W = np.fft.rfft(w_circ, N)
    
    S1 = np.fft.irfft(W * H, N)[:n]
    S2 = np.fft.irfft(np.conj(W) * H, N)[:n]
    
    return 1.0 - S1 - S2

    start = time.time()
    best = h.copy()
    best_score = compute_c5(best)
    
    if alphas is None:
        alphas = [5, 10, 20, 50, 100, 200, 500, 1000]
    
    for alpha in alphas:
        if time.time() - start >= time_limit:
            break
        eta = eta_init
        z = best.copy()
        t_k = 1.0
        
        for it in range(50000):
            if time.time() - start >= time_limit:
                break
            
            corr = get_corr_full_fft(z)
            grad = compute_grad(z, corr, alpha)
            
            if binary_weight > 0:
                grad = grad + binary_weight * (2.0 * z - 1.0)
            
            # FISTA step
            u = z - eta * grad
            u = project_feasible(u, target)
            
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            z = u + ((t_k - 1.0) / t_new) * (u - z)
            z = np.clip(z, 0.0, 1.0)
            t_k = t_new
            
            score = compute_c5(u)
            if score < best_score:
                best_score = score
                best = u.copy()
            
            # Adaptive: reduce eta if no improvement for a while
            if it % 500 == 0 and it > 0:
                pass
    
    return best, best_score

    n = len(h)
