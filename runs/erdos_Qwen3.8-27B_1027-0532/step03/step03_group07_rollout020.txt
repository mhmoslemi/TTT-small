import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

    lo = y.min() - 1.0
    hi = y.max() + 1.0

        else:

        if m > 0:

            grad[:m] += -h[lag:lag+m]
    else:

        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])

    return grad * dx

def _coord_descent_binary(h0, n, target_sum, max_rounds=20):

    for _ in range(max_rounds):

            elif c1 < _c5_of(h) - 1e-13:

def _coord_descent_frac(h0, n, target_sum, max_rounds=8):

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])

                diff += sub

def _grad_for_lag(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

    m = np.max(corr)
    if m < -500:
        return m * (2.0 / n)
    return (m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha) * (2.0 / n)

def _slsqp_smooth_grad(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    exps = np.exp(alpha * (corr - m))
    w = exps / exps.sum()
    n_top = min(12, len(corr))

    for idx in range(len(top_idx)):
        g += w_top[idx] * _grad_for_lag(x, top_idx[idx], dx)
    return g * (2.0 / n)

                    c5_rs = _c5_of(h_rs)
                    if c5_rs < best_c5:
                        best_c5 = c5_rs
                        best_h = h_rs.copy()
                        best_n = n
                    if time.time() < n_deadline - 10:
                        h_cd_rs = _coord_descent_binary(h_rs, n, target_sum, max_rounds=15)
                        c5_cd_rs = _c5_of(h_cd_rs)
                        if c5_cd_rs < best_c5:
                            best_c5 = c5_cd_rs
                            best_h = h_cd_rs.copy()
                            best_n = n

    # Final aggressive polish
    n_best = best_n
    target = n_best / 2.0
    h_final = best_h.copy()

    if time.time() < deadline - 12:
        for alpha in [10000, 30000, 80000]:
            if time.time() > deadline - 5:
                break
            h_final = _slsqp_opt(h_final, n_best, target, alpha, maxiter=350)
            c5_f = _c5_of(h_final)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_final.copy()

    if time.time() < deadline - 8:
        h_cd_f = _coord_descent_binary(best_h, n_best, target, max_rounds=35)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    if time.time() < deadline - 5:
        h_cf_f = _coord_descent_frac(best_h, n_best, target, max_rounds=12)
        c5_cf_f = _c5_of(h_cf_f)
        if c5_cf_f < best_c5:
            best_c5 = c5_cf_f
            best_h = h_cf_f.copy()

    if time.time() < deadline - 3:
        h_pol = _slsqp_opt(best_h, n_best, target, 50000, maxiter=500)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # One final cycle
    if time.time() < deadline - 2:
        h_last = _coord_descent_binary(best_h, n_best, target, max_rounds=20)
        c5_last = _c5_of(h_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = h_last.copy()

    best_h = _fix_sum(best_h, n_best / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, n_best)
