    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(100):
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
        else:

        h = np.clip(h, 0.0, 1.0)

    n = len(x)
    dx = 2.0 / n
    n_corr = 2 * n - 1

    weights = np.exp(shifted) / np.sum(np.exp(shifted))

    return val, grad

                    best_h = h_sl
