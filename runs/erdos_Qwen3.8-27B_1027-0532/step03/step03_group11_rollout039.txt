```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def compute_corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= x.max()
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= x.max()
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)
```

Let me define: for binary h, let comp[x] = 1 - h[x].

Then corr[s] = sum_i h[i] * comp[i+s]

When we flip oi from 1 to 0: h[oi] becomes 0, comp[oi] becomes 1
When we flip zi from 0 to 1: h[zi] becomes 1, comp[zi] becomes 0

Change in corr[s]:
- From h[oi] 1→0: terms where i=oi: was h[oi]*comp[oi+s] = comp[oi+s], now 0. Change: -comp[oi+s]
- From comp[oi] 0→1: terms where i+s=oi (i=oi-s): was h[oi-s]*comp[oi] = 0, now h[oi-s]*1. Change: +h[oi-s]
- From h[zi] 0→1: terms where i=zi: was 0, now comp[zi+s]. Change: +comp[zi+s]
- From comp[zi] 1→0: terms where i+s=zi (i=zi-s): was h[zi-s]*1, now 0. Change: -h[zi-s]

So Δcorr[s] = -comp[oi+s] + h[oi-s] + comp[zi+s] - h[zi-s]
(validity conditions apply)

For binary h, comp[x] = 1-h[x], so:
Δcorr[s] = -(1-h[oi+s]) + h[oi-s] + (1-h[zi+s]) - h[zi-s]

This is valid when the relevant indices are in [0, n-1].

Now, for the max, we need to track which s achieves the maximum. If we can quickly compute the new max after a swap, we can do a very fast local search.

The maximum of the correlation array changes only at the lags affected by the swap. Specifically, the lags affected are those s where any of the four terms is non-zero, which is s ∈ [-oi, n-1-oi] ∪ [oi-n+1, oi] ∪ [-zi, n-1-zi] ∪ [zi-n+1, zi].

The union of these ranges is at most [-n+1, n-1] (all of them), so in the worst case, all lags are affected. But in practice, for a swap of nearby positions, only a subset is affected.

For the local search, the strategy is:
1. Maintain the full correlation array
2. For each candidate swap (oi, zi), compute Δcorr[s] for all affected s
3. Compute the new max
4. Accept if new max < old max

The cost per candidate is O(n) (to update and find max of the affected lags). With n=180 and trying all n²/4 ≈ 8100 candidate swaps per round, that's 8100 * 180 ≈ 1.4M operations per round. Very fast!

```python
def binary_best_improvement(h, max_rounds=100, time_limit=60, start_time=0):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')  # raw, without dx
    current_max = float(np.max(corr))
    
    for round_idx in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        best_corr = None
        
        for oi in ones:
            for zi in zeros:
                # Compute change in correlation
                # Δcorr[s] for affected s
                new_corr = corr.copy()
                
                # From oi: 1->0
                # s in [-oi, n-1-oi]: new_corr[s+n-1] -= (1-h[oi+s])
                s_start = max(-oi, -(n-1))
                s_end = min(n-1-oi, n-1)
                for s in range(s_start, s_end + 1):
                    new_corr[s + n - 1] -= (1.0 - h[oi + s])
                
                # s in [oi-n+1, oi]: new_corr[s+n-1] += h[oi-s]
                s_start = max(oi - n + 1, -(n-1))
                s_end = min(oi, n-1)
                for s in range(s_start, s_end + 1):
                    new_corr[s + n - 1] += h[oi - s]
                
                # From zi: 0->1
                # s in [-zi, n-1-zi]: new_corr[s+n-1] += (1-h[zi+s])
                s_start = max(-zi, -(n-1))
                s_end = min(n-1-zi, n-1)
                for s in range(s_start, s_end + 1):
                    new_corr[s + n - 1] += (1.0 - h[zi + s])
                
                # s in [zi-n+1, zi]: new_corr[s+n-1] -= h[zi-s]
                s_start = max(zi - n + 1, -(n-1))
                s_end = min(zi, n-1)
                for s in range(s_start, s_end + 1):
                    new_corr[s + n - 1] -= h[zi - s]
                
                new_max = float(np.max(new_corr))
                delta = new_max - current_max
                if delta < best_delta:
                    best_delta = delta
                    best_oi = oi
                    best_zi = zi
                    best_corr = new_corr
        
        if best_oi < 0:
            break
        
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = best_corr
        current_max = best_max = float(np.max(corr))
    
    return h, current_max * (2.0 / n)
```

```python
def fast_binary_local_search(h, max_rounds=500, time_limit=100, start_time=0):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Precompute flip effects
        flip_eff = np.zeros((n, 2*n-1))
        for i in range(n):
            if h[i] == 1.0:
                # 1 -> 0
                s1 = np.arange(-i, n - i)
                flip_eff[i, s1 + n - 1] -= (1.0 - h[i + s1])
                s2 = np.arange(i - n + 1, i + 1)
                flip_eff[i, s2 + n - 1] += h[i - s2]
            else:
                # 0 -> 1
                s1 = np.arange(-i, n - i)
                flip_eff[i, s1 + n - 1] += (1.0 - h[i + s1])
                s2 = np.arange(i - n + 1, i + 1)
                flip_eff[i, s2 + n - 1] -= h[i - s2]
        
        # For each pair (oi, zi), compute new max
        best_delta = 0.0
        best_oi = -1
        best_zi = -1
        
        for oi in ones:
            # new corr for all zi: corr + flip_eff[oi] + flip_eff[zi]
            # = (corr + flip_eff[oi]) + flip_eff[zi]
            base = corr + flip_eff[oi]
            # For each zi, new_max = max(base + flip_eff[zi])
            # Vectorize: create matrix of shape (n_zeros, corrlen)
            new_corrs = base[np.newaxis, :] + flip_eff[zeros][:, np.newaxis, :]  # (n_zeros, corrlen)
            new_maxes = new_corrs.max(axis=1)
            deltas = new_maxes - current_max
            best_local = np.argmin(deltas)
            if deltas[best_local] < best_delta:
                best_delta = deltas[best_local]
                best_oi = oi
                best_zi = zeros[best_local]
        
        if best_oi < 0:
            break
        
        h[best_oi] = 0.0
        h[best_zi] = 1.0
        corr = corr + flip_eff[best_oi] + flip_eff[best_zi]
        current_max = float(np.max(corr))
    
    return h, current_max * dx
```

```python
def compute_flip_effect(h, i):
    """Compute the change in corr_full array when flipping h[i]."""
    n = len(h)
    corrlen = 2 * n - 1
    effect = np.zeros(corrlen)
    
    if h[i] == 1.0:
        # Flipping 1 -> 0
        # Part 1: m in [n-1-i, 2n-2-i], m != n-1
        m_start = n - 1 - i
        m_end = 2 * n - 2 - i
        for m in range(m_start, m_end + 1):
            if m == n - 1:
                continue
            s = m - n + 1
            effect[m] -= (1.0 - h[i + s])
        
        # Part 2: m in [i, i+n-1], m != n-1
        m_start = i
        m_end = i + n - 1
        for m in range(m_start, m_end + 1):
            if m == n - 1:
                continue
            s = m - n + 1
            idx = i - s  # = i - (m-n+1) = i + n - 1 - m
            effect[m] += h[idx]
    else:
        # Flipping 0 -> 1
        m_start = n - 1 - i
```

Actually, one more thought. I want to make sure the "corrections" in the swap formula are handled correctly in the vectorized version.

For a swap (oi, zi), the correction is +1 at m = n-1+(zi-oi) and m = n-1+(oi-zi).

In the vectorized version, for a fixed oi and varying zi:
- m1 = n-1+(zi-oi) varies with zi
- m2 = n-1+(oi-zi) = n-1-(zi-oi) also varies with zi

So for each zi, the corrections are at different m values. I can't easily vectorize this as a single operation. But I can handle it by:
1. Compute new_corrs = (corr + flip_eff[oi])[None,:] + flip_eff[zeros][:,None,:]
2. For each zi_j at index j: new_corrs[j, n-1+(zeros[j]-oi)] += 1 and new_corrs[j, n-1+(oi-zeros[j])] += 1

This is O(n_zeros) scalar operations, which is negligible.
