```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)

def swap_delta(h, a, b, n):
    """Compute delta_corr[k] for swapping h[a]:1->0 and h[b]:0->1.
    Returns array of length 2*n-1 for the full correlate output."""

    # corr_full[j] = sum_i h[i] * (1-h[i+k]) where k = j - (n-1)
    # So shift k = j - (n-1)
    
    # For shift k, delta = h[a+k] - h[b+k] + h[a-k] - h[b-k]

    # where h[x] = 0 if x not in [0, n-1]
    
    k_range = np.arange(-(n-1), n)
    
    # h[a+k] for valid a+k
    a_plus = a + k_range
    valid_ap = (a_plus >= 0) & (a_plus < n)
    
    b_plus = b + k_range
    valid_bp = (b_plus >= 0) & (b_plus < n)
    
    a_minus = a - k_range

    b_minus = b - k_range

    delta[valid_am] += h[a_minus[valid_am]]
    delta[valid_bm] -= h[b_minus[valid_bm]]
    
    return delta
```

```python
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h)
    h_r = np.clip(h_r, 0, 1).astype(int)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    ones = np.where(h_r == 1)[0]
    zeros = np.where(h_r == 0)[0]
    if current_sum > target_sum:
        need = int(current_sum - target_sum)
        if need <= len(ones):
            order = np.argsort(h[ones])[:need]
            h_r[ones[order]] = 0
    else:
        need = int(target_sum - current_sum)
        if need <= len(zeros):
            order = np.argsort(-h[zeros])[:need]
            h_r[zeros[order]] = 1
    final_sum = h_r.sum()
    if final_sum != target_sum:
        if final_sum > target_sum:
            ones2 = np.where(h_r == 1)[0]
            extra = int(final_sum - target_sum)
            h_r[ones2[:extra]] = 0
        else:
            zeros2 = np.where(h_r == 0)[0]
            extra = int(target_sum - final_sum)
            h_r[zeros2[:extra]] = 1
    return h_r.astype(float)
```

```python
def sa_fast(h_start, target_sum, max_iters, t_init, t_final, rng, time_limit=None, start_time=None):
    n = len(h_start)
    h = project_binary(h_start, target_sum)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    # Precompute k_range for delta computation
    k_range = np.arange(-(n-1), n)  # shape (2n-1,)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    
    no_improve = 0
    max_no_improve = max(3000, max_iters // 30)
    
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if (time.time() - start_time) > time_limit:
                break
        if no_improve > max_no_improve:
            if rng.random() < 0.05:
                no_improve = 0
            else:
                break
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        ratio = it / max(max_iters - 1, 1)
        t = t_init * (t_final / t_init) ** ratio
        
        # Choose swap
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        
        # Fast delta computation
        # delta[k_shift] = h[a+k] - h[b+k] + h[a-k] - h[b-k]
        a_plus = a + k_range
        valid_ap = (a_plus >= 0) & (a_plus < n)
        b_plus = b + k_range
        valid_bp = (b_plus >= 0) & (b_plus < n)
        a_minus = a - k_range
        valid_am = (a_minus >= 0) & (a_minus < n)
        b_minus = b - k_range
        valid_bm = (b_minus >= 0) & (b_minus < n)
        
        # Instead of creating a full delta array, compute new_max directly
        # new_corr = corr + delta
        # We need max(new_corr)
        
        # Create delta efficiently
        delta = np.zeros(2*n - 1)
        delta[valid_ap] += h[a_plus[valid_ap]]
        delta[valid_bp] -= h[b_plus[valid_bp]]
        delta[valid_am] += h[a_minus[valid_am]]
        delta[valid_bm] -= h[b_minus[valid_bm]]
        
        new_max = float(np.max(corr + delta))
        delta_val = new_max - current_max
        
        if delta_val < -1e-15 or (t > 1e-12 and rng.random() < np.exp(-delta_val / t)):
            h[a] = 0
            h[b] = 1
            corr += delta
            current_max = new_max
            # Update ones and zeros
            # a moves from ones to zeros, b moves from zeros to ones
            ones = ones[ones != a]
            zeros = zeros[zeros != b]
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
    
    return best_h, best_max
```
