def _binary_local_search(h0, n, max_rounds, deadline):
    h = h0.copy()
    # Ensure binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _fix_sum(h, n/2.0)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = np.max(corr) * (2.0/n)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Compute Δcorr
                # For each k from -(n-1) to n-1 (mapped to index k+n-1 in corr array)
                # Δc_k = (h[i-k]-1)*[0≤i-k<n] + h[i+k]*[0≤i+k<n] + (1-h[j-k])*[0≤j-k<n] - h[j+k]*[0≤j+k<n]
                # Correction: if k=i-j (k≠0): +1; if k=j-i (k≠0): +1
                
                # Vectorized computation:
                # Term 1: h[i-k]-1 for k in range where 0≤i-k<n, i.e., k in (i-n+1, i]
                # Term 2: h[i+k] for k in range where 0≤i+k<n, i.e., k in [-i, n-1-i]
                # Term 3: 1-h[j-k] for k in range where 0≤j-k<n, i.e., k in (j-n+1, j]
                # Term 4: -h[j+k] for k in range where 0≤j+k<n, i.e., k in [-j, n-1-j]
                
                delta = np.zeros(2*n-1)
                # k ranges from -(n-1) to n-1, index = k + (n-1)
                
                # Term 1: k from max(-(n-1), i-n+1) to min(n-1, i)
                k_lo1 = max(-(n-1), i-n+1)
                k_hi1 = min(n-1, i)
                if k_lo1 <= k_hi1:
                    ks = np.arange(k_lo1, k_hi1+1)
                    delta[ks + (n-1)] += h[i - ks] - 1.0
                
                # Term 2: k from max(-(n-1), -i) to min(n-1, n-1-i)
                k_lo2 = max(-(n-1), -i)
                k_hi2 = min(n-1, n-1-i)
                if k_lo2 <= k_hi2:
                    ks = np.arange(k_lo2, k_hi2+1)
                    delta[ks + (n-1)] += h[i + ks]
                
                # Term 3: k from max(-(n-1), j-n+1) to min(n-1, j)
                k_lo3 = max(-(n-1), j-n+1)
                k_hi3 = min(n-1, j)
                if k_lo3 <= k_hi3:
                    ks = np.arange(k_lo3, k_hi3+1)
                    delta[ks + (n-1)] += 1.0 - h[j - ks]
                
                # Term 4: k from max(-(n-1), -j) to min(n-1, n-1-j)
                k_lo4 = max(-(n-1), -j)
                k_hi4 = min(n-1, n-1-j)
                if k_lo4 <= k_hi4:
                    ks = np.arange(k_lo4, k_hi4+1)
                    delta[ks + (n-1)] -= h[j + ks]
                
                # Corrections
                k_corr1 = i - j
                if k_corr1 != 0 and abs(k_corr1) <= n-1:
                    delta[k_corr1 + (n-1)] += 1.0
                k_corr2 = j - i
                if k_corr2 != 0 and abs(k_corr2) <= n-1:
                    delta[k_corr2 + (n-1)] += 1.0
                
                new_corr = corr + delta
                new_max = np.max(new_corr) * (2.0/n)
                if new_max < current_max - 1e-14:
                    # Apply swap
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    current_max = new_max
                    improved = True
        
        if not improved:
            break
    
    return h, current_max

import time

from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _binary_steepest_descent(h0, n, max_rounds, deadline):
    h = h0.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _fix_sum(h, n / 2.0)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    s_arr = np.arange(-(n - 1), n)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_improvement = 0.0
        best_swap = None
        best_new_corr = None
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                delta = np.zeros(2 * n - 1)
                mask1 = (i - s_arr >= 0) & (i - s_arr < n) & (s_arr != 0)
                if np.any(mask1):
                    delta[mask1] += h[i - s_arr[mask1]] - 1.0
                mask2 = (i + s_arr >= 0) & (i + s_arr < n) & (s_arr != 0)
                if np.any(mask2):
                    delta[mask2] += h[i + s_arr[mask2]]
                mask3 = (j - s_arr >= 0) & (j - s_arr < n) & (s_arr != 0)
                if np.any(mask3):
                    delta[mask3] += 1.0 - h[j - s_arr[mask3]]
                mask4 = (j + s_arr >= 0) & (j + s_arr < n) & (s_arr != 0)
                if np.any(mask4):
                    delta[mask4] -= h[j + s_arr[mask4]]
                d1 = i - j
                if d1 != 0 and abs(d1) <= n - 1:
                    delta[d1 + n - 1] += 1.0
                d2 = j - i
                if d2 != 0 and abs(d2) <= n - 1:
                    delta[d2 + n - 1] += 1.0
                new_corr = corr + delta
                nm = float(np.max(new_corr))
                if nm < current_max - 1e-12:
                    improvement = current_max - nm
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_swap = (i, j)
                        best_new_corr = new_corr
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0.0
            h[j] = 1.0
            corr = best_new_corr
            current_max -= best_improvement
        else:
            break
    return h, current_max * (2.0 / n)

def _binary_sa(h0, n, max_iters, T_start, T_end, deadline, rng):
    h = h0.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    h = _fix_sum(h, n / 2.0)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_val = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()
    best_val = current_val
    s_arr = np.arange(-(n - 1), n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (float(it) / float(max_iters))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        mask1 = (i - s_arr >= 0) & (i - s_arr < n) & (s_arr != 0)
        if np.any(mask1):
            delta[mask1] += h[i - s_arr[mask1]] - 1.0
        mask2 = (i + s_arr >= 0) & (i + s_arr < n) & (s_arr != 0)
        if np.any(mask2):
            delta[mask2] += h[i + s_arr[mask2]]
        mask3 = (j - s_arr >= 0) & (j - s_arr < n) & (s_arr != 0)
        if np.any(mask3):
            delta[mask3] += 1.0 - h[j - s_arr[mask3]]
        mask4 = (j + s_arr >= 0) & (j + s_arr < n) & (s_arr != 0)
        if np.any(mask4):
            delta[mask4] -= h[j + s_arr[mask4]]
        d1 = i - j
        if d1 != 0 and abs(d1) <= n - 1:
            delta[d1 + n - 1] += 1.0
        d2 = j - i
        if d2 != 0 and abs(d2) <= n - 1:
            delta[d2 + n - 1] += 1.0
        new_corr = corr + delta
        new_val = float(np.max(new_corr)) * (2.0 / n)
        diff = new_val - current_val
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-10)):
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
    return best_h, best_val

def _nesterov_smooth(h0, n, target_sum, n_iters, top_m, top_alpha, step_init, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(top_alpha * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            idx_list = np.array([np.argmax(corr)])
            w_list = np.array([1.0])
        else:
            weights = weights / total_w
            top_idx = np.argsort(weights)[::-1][:top_m]
            idx_list = top_idx
            w_list = weights[top_idx].copy()
            w_list = w_list / w_list.sum()
        g = np.zeros(n)
        for idx in range(len(idx_list)):
            g += w_list[idx] * _grad_corr_term(y, idx_list[idx], dx)
        step = step_init / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

_global_target_sum = 0.0
_global_n = 0

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _coord_descent(h0, n, max_rounds, deadline):
    h = h0.copy()
    vals = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    target = n / 2.0
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_c = _c5_of(h)
            best_v = old
            for v in vals:
                if v == old:
                    continue
                h[j] = v
                h_t = _fix_sum(h, target)
                c = _c5_of(h_t)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h[j] = best_v
            h = _fix_sum(h, target)
            if best_v != old:
                improved = True
        if not improved:
            break
    return h, _c5_of(h)

def _block_swap_search(h, n, corr, current_max, deadline):
    s_arr = np.arange(-(n - 1), n)
    for block_len in [2, 3, 4]:
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        for i in range(len(ones) - block_len + 1):
            if time.time() > deadline:
                break
            i_start = ones[i]
            if ones[i + block_len - 1] != i_start + block_len - 1:
                continue
            for j in range(len(zeros) - block_len + 1):
                j_start = zeros[j]
                if zeros[j + block_len - 1] != j_start + block_len - 1:
                    continue
                delta = np.zeros(2 * n - 1)
                for ii in range(i_start, i_start + block_len):
                    mask1 = (ii - s_arr >= 0) & (ii - s_arr < n) & (s_arr != 0)
                    if np.any(mask1):
                        delta[mask1] += h[ii - s_arr[mask1]] - 1.0
                    mask2 = (ii + s_arr >= 0) & (ii + s_arr < n) & (s_arr != 0)
                    if np.any(mask2):
                        delta[mask2] += h[ii + s_arr[mask2]]
                for jj in range(j_start, j_start + block_len):
                    mask3 = (jj - s_arr >= 0) & (jj - s_arr < n) & (s_arr != 0)
                    if np.any(mask3):
                        delta[mask3] += 1.0 - h[jj - s_arr[mask3]]
                    mask4 = (jj + s_arr >= 0) & (jj + s_arr < n) & (s_arr != 0)
                    if np.any(mask4):
                        delta[mask4] -= h[jj + s_arr[mask4]]
                for ii in range(i_start, i_start + block_len):
                    for jj in range(j_start, j_start + block_len):
                        d = ii - jj
                        if d != 0 and abs(d) <= n - 1:

                new_corr = corr + delta
                nm = float(np.max(new_corr))
                if nm < current_max - 1e-12:

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
