def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h > 1e-12)[0]
        if len(idx) == 0:
            break
        h = h.copy()
        h[idx] += diff / len(idx)
        h = np.clip(h, 0.0, 1.0)

def solve_lp(h, n, target, solver_pref='PDLP'):

        if not found and step % 100 == 0:
            h_try = h_cur + np.random.randn(n) * 0.01
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
    return best_h, best_c5

def altern_schedule(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = project_feasible(h.copy(), target)
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    ar = np.arange(n)
    for step in range(max_steps):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)
        grad = np.zeros(n)
        v1 = (ar - k >= 0) & (ar - k < n)
        if np.any(v1):
            grad[v1] += (1.0 - h_cur[ar[v1] - k]) * dx
        v2 = (ar + k >= 0) & (ar + k < n)
        if np.any(v2):
            grad[v2] += -h_cur[ar[v2] + k] * dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        lr = 0.3 * dx / (gnorm + 1e-10)
        h_try = h_cur - lr * grad
        h_try = project_feasible(h_try, target)
        sc = compute_c5(h_try)
        if sc < best_c5:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try
    return best_h, best_c5

def get_time_left(budget_s, start_time):
    return budget_s - 25.0 - (time.time() - start_time)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_target
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    best_h = project_feasible(h_init.copy(), target)
    best_score = compute_c5(best_h)

    tl = get_time_left(budget_s, start_time)
    if tl > 200:
        lp_time = min(280, tl * 0.35)
        h1, s1 = alternating_lp(best_h, n, target, max_iter=5000,
                                time_limit=lp_time, damping=0.7)
        if s1 < best_score:
            best_score = s1
            best_h = h1

    tl = get_time_left(budget_s, start_time)
    if tl > 80:
        h2 = slsqp_refine(best_h, n, target,
                          [5, 10, 20, 50, 100, 200, 500, 1000, 3000], 800, 1e-15)
        s2 = compute_c5(h2)
        if s2 < best_score:
            best_score = s2
            best_h = h2

    tl = get_time_left(budget_s, start_time)
    if tl > 60:
        h3, s3 = subgradient_refine(best_h, n, target,
                                     max_steps=5000, time_limit=min(90, tl * 0.5))
        if s3 < best_score:
            best_score = s3
            best_h = h3

    tl = get_time_left(budget_s, start_time)
    if tl > 120:
        lp_time = min(120, tl * 0.4)
        h4, s4 = alternating_lp(best_h, n, target, max_iter=3000,
                                time_limit=lp_time, damping=0.5)
        if s4 < best_score:
            best_score = s4
            best_h = h4
        tl = get_time_left(budget_s, start_time)
        if tl > 60:
            h4b = slsqp_refine(best_h, n, target,
                               [30, 100, 300, 1000, 3000, 8000], 600, 1e-15)
            s4b = compute_c5(h4b)
            if s4b < best_score:
                best_score = s4b
                best_h = h4b

    tl = get_time_left(budget_s, start_time)
    for noise_lvl in [0.03, 0.08, 0.15]:
        if tl < 100:
            break
        h_pert = best_h + rng.randn(n) * noise_lvl
        h_pert = project_feasible(h_pert, target)
        dmp = 0.5 + 0.2 * noise_lvl
        lp_time = min(90, tl * 0.3)
        h5, s5 = alternating_lp(h_pert, n, target, max_iter=2000,
                                time_limit=lp_time, damping=dmp)
        tl = get_time_left(budget_s, start_time)
        if tl > 40 and s5 < best_score + 0.001:
            h5b = slsqp_refine(h5, n, target,
                               [20, 50, 100, 300, 1000], 400, 1e-15)
            s5b = compute_c5(h5b)
            if s5b < s5:
                h5 = h5b
                s5 = s5b
        if s5 < best_score:
            best_score = s5
            best_h = h5
        tl = get_time_left(budget_s, start_time)

    tl = get_time_left(budget_s, start_time)
    if tl > 180:
        for n_try in [500, 700, 1000]:
            if tl < 130:
                break
            target_try = n_try / 2.0
            x_old = np.linspace(0.0, 1.0, n)
            x_new = np.linspace(0.0, 1.0, n_try)
            h_interp = np.interp(x_new, x_old, best_h)
            h_interp = project_feasible(h_interp, target_try)
            lp_time = min(100, tl * 0.35)
            h_hr, s_hr = alternating_lp(h_interp, n_try, target_try,
                                         max_iter=2000, time_limit=lp_time,
                                         damping=0.6)
            tl = get_time_left(budget_s, start_time)
            if tl > 50 and s_hr < best_score:
                h_hr2 = slsqp_refine(h_hr, n_try, target_try,
                                      [20, 50, 100, 300, 1000], 500, 1e-15)
                s_hr2 = compute_c5(h_hr2)
                if s_hr2 < s_hr:
                    h_hr = h_hr2
                    s_hr = s_hr2
            if s_hr < best_score:
                best_score = s_hr
                best_h = h_hr
                n = n_try
                target = target_try
            tl = get_time_left(budget_s, start_time)

    tl = get_time_left(budget_s, start_time)
    if tl > 40:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_fin = slsqp_refine(best_h, n_use, target_use,
                              [200, 500, 1000, 3000, 10000], 600, 1e-16)
        s_fin = compute_c5(h_fin)
        if s_fin < best_score:
            best_score = s_fin
            best_h = h_fin

    tl = get_time_left(budget_s, start_time)
    if tl > 30:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg2, s_sg2 = subgradient_refine(best_h, n_use, target_use,
                                           max_steps=3000, time_limit=min(40, tl * 0.6))
        if s_sg2 < best_score:
            best_score = s_sg2
            best_h = h_sg2

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
