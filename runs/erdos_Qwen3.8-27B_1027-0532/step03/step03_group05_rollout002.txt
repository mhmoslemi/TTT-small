def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    count = np.sum(mask)
    if count == 0:
        return h
    step = diff / count
    h[mask] += step
    return np.clip(h, 0.0, 1.0)

def compute_full_grad(h, p=100):
    """Compute gradient of power mean objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    y = h.copy()
    t = 1.0

            best = h_new.copy()
        
        # Update
        y = h_new
        h = h_new
        t = t_new
        
        # Adaptive step size
        if score >= best_score:
            eta *= 0.9
        else:
            eta = min(eta * 1.01, 0.05)
    
    return best, best_score

    if abs(diff) < 1e-14:

    else:

    if count == 0:
        return h

    np.clip(h, 0.0, 1.0, out=h)
    return h

def soft_subgradient(h, corr, temperature=0.05):
    """Compute softmax-weighted subgradient of max(corr)."""
    n = len(h)
    dx = 2.0 / n
    max_val = np.max(corr)
    shifted = (corr - max_val) / temperature
    shifted = np.minimum(shifted, 0.0)  # numerical stability
    w = np.exp(shifted)
    sw = np.sum(w)
    if sw < 1e-30:
        # Only the max contributes
        idx = int(np.argmax(corr))
        w = np.zeros(len(corr))
        w[idx] = 1.0
    else:
        w /= sw
    
    grad = np.zeros(n)
    n_corr = len(corr)
    for idx in range(n_corr):
        wi = w[idx]
        if wi < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] += wi * (-h[:n - lag])
        else:
            m = -lag
            grad[m:] += wi * (1.0 - h[:n - m])
            grad[:n - m] += wi * (-h[m:])
