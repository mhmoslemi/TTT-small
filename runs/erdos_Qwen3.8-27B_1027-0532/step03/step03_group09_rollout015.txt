def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr_scaled(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_scaled(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-16})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def nesterov_proj_grad(h, time_limit, start_time):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    y = h_cur.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t = 1
    lr_base = 0.1
    stall = 0
    stall_limit = 300

    for step in range(500000):
        if time.time() - start_time > time_limit:
            break
        
        # Adaptive top-k
        corr = compute_corr_scaled(y)
        cmax = np.max(corr)
        near_max = np.sum(corr > cmax - 0.003)
        k = max(3, min(10, int(near_max)))
        temp = 0.005 + 0.003 * (k / 5.0)
        
        grad = topk_subgrad(y, k, temp)
        
        # Nesterov step
        lr = lr_base / (1.0 + 0.1 * t)
        y_new = project_onto_constraint(y - lr * grad, target_sum)
        
        momentum = (t - 1) / (t + 2) if t > 1 else 0.0
        h_new = momentum * h_cur + (1 - momentum) * y_new
        h_new = project_onto_constraint(h_new, target_sum)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h_cur = h_new
        y = y_new
        t += 1
        
        if stall > stall_limit:
            # Reset with perturbation
            rng_local = np.random.default_rng(step + 12345)
            h_cur = project_onto_constraint(best_h + 0.02 * rng_local.standard_normal(n), target_sum)
            y = h_cur.copy()
            stall = 0
            stall_limit = max(100, stall_limit // 2)
            t = 1

    return best_h, best_c5

def gen_perturb(h_best, rng, target_sum, method):
    n = len(h_best)
    if method == 0:
        return project_onto_constraint(h_best + 0.015 * rng.standard_normal(n), target_sum)
    elif method == 1:
        max_len = max(2, n // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n - blen + 1))
        b = int(rng.integers(0, n - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.01 * rng.standard_normal(n), target_sum)
    elif method == 2:
        h_bin = (rng.random(n) < 0.5).astype(float)
        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.4, 0.7))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)
    elif method == 3:
        period = int(rng.integers(5, 25))
        amp = float(rng.uniform(0.15, 0.4))
        phase = float(rng.uniform(0, 2 * np.pi))
        h_pat = 0.5 + amp * np.sin(2 * np.pi * np.arange(n) / period + phase)
        h_pat = project_onto_constraint(h_pat, target_sum)
        alpha = float(rng.uniform(0.6, 0.85))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_pat, target_sum)
    elif method == 4:
        center = int(rng.integers(0, n))
        width = int(rng.integers(4, n // 3))
        h_p = h_best.copy()
        for i in range(n):
            dist = min(abs(i - center), n - abs(i - center))
            if dist < width:
                w = 1.0 - dist / width
                h_p[i] = np.clip(h_p[i] + 0.08 * w * rng.standard_normal(), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif method == 5:
        h_rev = h_best[::-1].copy()
        alpha = float(rng.uniform(0.7, 0.9))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_rev, target_sum)
    elif method == 6:
        # Cyclic shift
        shift = int(rng.integers(1, n // 3))
        h_shift = np.roll(h_best, shift)
        h_shift = project_onto_constraint(h_shift, target_sum)
        alpha = float(rng.uniform(0.7, 0.95))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_shift, target_sum)
    else:
        # Two-block swap
        h_p = h_best.copy()
        for _ in range(2):
            blen = int(rng.integers(1, n // 8 + 1))
            a = int(rng.integers(0, n - blen + 1))
            b = int(rng.integers(0, n - blen + 1))
            if abs(a - b) >= blen:
                blk = h_p[a:a + blen].copy()
                h_p[a:a + blen] = h_p[b:b + blen]
                h_p[b:b + blen] = blk
        return project_onto_constraint(h_p, target_sum)

        if rem() > 430:

            h_ng, c5_ng = nesterov_proj_grad(h_opt if c5_opt < best_c5 else best_h, tl, start_time)
            if c5_ng < best_c5:
                best_c5 = c5_ng
                best_h = h_ng.copy()
        
        n_restarts += 1

    # Phase 3: Extended Nesterov projected gradient
    if rem() > 100:
        tl = min(rem() - 30, 100)
        h_ng, c5_ng = nesterov_proj_grad(best_h, tl, start_time)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    # Phase 4: Fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Resolution increase
    if rem() > 200:

        old_t = np.linspace(0.0, 1.0, n0)

        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_up = project_onto_constraint(h_up, n1 / 2.0)
        
        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.003, 200)
        
        improved = False
        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n0 = n1
            target_sum = n0 / 2.0
            improved = True
        
        # Further refine
        if improved or rem() > 150:
            if rem() > 100:
                h_opt_up2, c5_opt_up2 = slsqp_optimize(best_h, 0.001, 150)
                if c5_opt_up2 < best_c5:
                    best_c5 = c5_opt_up2
                    best_h = h_opt_up2.copy()
            
            if rem() > 60:
                tl_up = min(rem() - 20, 50)
                h_ng_up, c5_ng_up = nesterov_proj_grad(best_h, tl_up, start_time)
                if c5_ng_up < best_c5:
                    best_c5 = c5_ng_up
                    best_h = h_ng_up.copy()

    # Phase 6: Second resolution increase if possible
    if rem() > 120 and len(best_h) < 260:
        n2 = len(best_h) + 30
        old_t = np.linspace(0.0, 1.0, len(best_h))
        new_t = np.linspace(0.0, 1.0, n2)
        h_up2 = np.interp(new_t, old_t, best_h)
        h_up2 = project_onto_constraint(h_up2, n2 / 2.0)
        h_up2 = h_up2 + 0.003 * rng.standard_normal(n2)
        h_up2 = project_onto_constraint(h_up2, n2 / 2.0)
        
        h_opt_up2, c5_opt_up2 = slsqp_optimize(h_up2, 0.002, 150)
        if c5_opt_up2 < best_c5:
            best_c5 = c5_opt_up2
            best_h = h_opt_up2.copy()
        
        if rem() > 40:
            h_opt_up3, c5_opt_up3 = slsqp_optimize(best_h, 0.0005, 100)
            if c5_opt_up3 < best_c5:
                best_c5 = c5_opt_up3
                best_h = h_opt_up3.copy()

    # Phase 7: Final fine SLSQP
    if rem() > 35:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0002, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final Nesterov
    if rem() > 20:
        tl = min(rem() - 8, 40)
        h_ng, c5_ng = nesterov_proj_grad(best_h, tl, start_time)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    # Final
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)

But wait, I'm being too optimistic about SLSQP time. Let me be more careful. SLSQP with n=180 variables, 180 bounds, 1 equality constraint, and a smooth max objective... The gradient computation involves a Python loop of 359 iterations, each doing a few numpy ops on arrays of size ~180. Let's say each gradient eval takes about 1ms. SLSQP typically does about 2-5 gradient evaluations per iteration. So 150 iterations * 3 * 1ms = 0.45s. That's fast!

Actually, I think the bottleneck is the constraint handling in SLSQP, not the gradient. With 180 bounds and 1 equality constraint, SLSQP solves a QP subproblem at each iteration, which is O(n^3) = O(5.8M) ops. That's maybe 1-5ms per subproblem. So 150 iterations * 5ms = 0.75s. Still fast.

But in practice, SLSQP can be slower due to numerical issues and line searches. Let me estimate 5-15 seconds per SLSQP call with maxiter=150.

OK let me also think about whether the Nesterov method is actually appropriate here. The objective is the max of linear functions (each correlation value is linear in h), so it's a convex function. Wait, is it?

corr[k] = sum_i h[i] * (1-h[i+k]) = sum_i h[i] - sum_i h[i]*h[i+k]

The second term is NOT linear in h (it's bilinear). So the objective is NOT convex! This means Nesterov's convergence guarantees don't directly apply. However, the method can still work heuristically.

Actually wait, let me reconsider. For a fixed lag s:
corr[s] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i]) = sum_{i} h[i+s] - sum_i h[i+s]*h[i]

This is not linear in h because of the h[i+s]*h[i] term. So the problem is non-convex.

This means:
1. The smooth max is also non-convex
2. SLSQP can get stuck in local minima
3. Nesterov's method is just a heuristic here
4. Diversity in restarts is crucial

Given this, I think the most important thing is to have diverse starting points and to be efficient in finding good local minima from each starting point.

                pop.append((best_c5, best_h.copy()))
                if len(pop) > 5:
                    pop.pop(0)

        n_restarts += 1

    # Phase 3: Extended Nesterov
    if rem() > 80.0:
        tl = min(rem() - 20.0, 80.0)
        h_ng, c5_ng = nesterov_proj_grad(best_h, tl, start_time)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    # Phase 4: Fine SLSQP
    if rem() > 50.0:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 5: Resolution increase
    if rem() > 180.0:
        n1 = len(best_h) + 40
        old_t = np.linspace(0.0, 1.0, len(best_h))
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.003, 150)

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
        else:
            h_up2 = project_onto_constraint(h_up + 0.02 * rng.standard_normal(n1), n1 / 2.0)
            h_opt_up2, c5_opt_up2 = slsqp_optimize(h_up2, 0.003, 150)
            if c5_opt_up2 < best_c5:
                best_c5 = c5_opt_up2
                best_h = h_opt_up2.copy()

        if rem() > 100.0:
            h_opt_r, c5_opt_r = slsqp_optimize(best_h, 0.001, 150)
            if c5_opt_r < best_c5:
                best_c5 = c5_opt_r
                best_h = h_opt_r.copy()

        if rem() > 50.0:
            tl_r = min(rem() - 15.0, 40.0)
            h_ng_r, c5_ng_r = nesterov_proj_grad(best_h, tl_r, start_time)
            if c5_ng_r < best_c5:
                best_c5 = c5_ng_r
                best_h = h_ng_r.copy()

    # Phase 6: Second resolution increase
    if rem() > 100.0 and len(best_h) < 260:
        n2 = len(best_h) + 30
        old_t2 = np.linspace(0.0, 1.0, len(best_h))
        new_t2 = np.linspace(0.0, 1.0, n2)
        h_up2 = np.interp(new_t2, old_t2, best_h)
        h_up2 = project_onto_constraint(h_up2, n2 / 2.0)
        h_up2 = h_up2 + 0.003 * rng.standard_normal(n2)
        h_up2 = project_onto_constraint(h_up2, n2 / 2.0)

        h_opt_up2, c5_opt_up2 = slsqp_optimize(h_up2, 0.002, 120)
        if c5_opt_up2 < best_c5:
            best_c5 = c5_opt_up2
            best_h = h_opt_up2.copy()

        if rem() > 40.0:
            h_opt_up3, c5_opt_up3 = slsqp_optimize(best_h, 0.0005, 100)
            if c5_opt_up3 < best_c5:
                best_c5 = c5_opt_up3
                best_h = h_opt_up3.copy()

    # Phase 7: Final fine SLSQP
    if rem() > 30.0:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0002, 150)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 8: Final Nesterov
    if rem() > 15.0:
        tl = min(rem() - 5.0, 30.0)
        h_ng, c5_ng = nesterov_proj_grad(best_h, tl, start_time)
        if c5_ng < best_c5:
            best_c5 = c5_ng
            best_h = h_ng.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
