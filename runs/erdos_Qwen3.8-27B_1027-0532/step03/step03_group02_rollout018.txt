```python
def project_feasible(h, target_sum):

        return h

def grad_smoothed(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    s_arr = np.arange(2 * n - 1) - (n - 1)
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        s = s_arr[idx]
        w = weights[idx]
        if w < 1e-10:

        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_hi > j_lo:

    # Phase 7: Final SLSQP
    if tleft() > 45:
        h_sl_f, sc = slsqp_refine(best_h, tgt_fin, [800, 1500], 300, max(25, tleft() - 45))
        if sc < best_score:
            best_score = sc
            best_h = h_sl_f

    # Final feasibility
    n_out = len(best_h)
    best_h = project_feasible(best_h, n_out / 2.0)
    final_score = compute_c5(best_h)
    if final_score < best_score:
        best_score = final_score

    return best_h, best_score, n_out
```
