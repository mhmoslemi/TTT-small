import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def lsp_obj_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -50.0/alpha, 0.0)
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for i in range(n):
        klo = max(0, n - 1 - i)
        khi = min(2 * n - 2, 2 * n - 2 - i)
        if klo <= khi:
            j_idx = np.arange(klo, khi + 1) - (n - 1) + i
            term1 = np.sum(w[klo:khi + 1] * (1.0 - h[j_idx]))
        else:
            term1 = 0.0
        klo2 = max(0, i)
        khi2 = min(2 * n - 2, i + n - 1)
        if klo2 <= khi2:
            j_idx2 = i - (np.arange(klo2, khi2 + 1) - (n - 1))
            term2 = np.sum(w[klo2:khi2 + 1] * h[j_idx2])
        else:
            term2 = 0.0
        grad[i] = term1 - term2
    return F, grad

    """One SPSA step."""
    n = len(h)
    delta = rng.choice([-1.0, 1.0], size=n)
    h_plus = project_feasible(h + c * delta, target)
    h_minus = project_feasible(h - c * delta, target)
    f_plus = compute_c5(h_plus)
    f_minus = compute_c5(h_minus)
    grad_est = (f_plus - f_minus) / (2.0 * c) * delta
    return h, grad_est, f_plus, f_minus

def srsa_optimize(h_init, target, n_iter, c0, c_decay, a0, a_decay, rng, time_check):
    """SPSA optimization."""
    h = h_init.copy()
    best_h = h.copy()

    c = c0

        h_new = h - a * grad_est
        h_new = project_feasible(h_new, target)
        f_new = compute_c5(h_new)
        if f_new < best_f:

            # else keep h as is

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)

def lbfgs_refine(h_start, target, alpha_schedule, maxiter_per_stage, ftol, time_check):
    h_cur = project_feasible(h_start.copy(), target)
    n = len(h_cur)
    current_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_schedule:
        if time_check() < 30:
            break
        def obj(x):
            F, _ = lsp_obj_grad(x, alpha)
            return F
        def jac(x):
            _, g = lsp_obj_grad(x, alpha)
            return g
        try:

                           bounds=bounds, options={'maxiter': maxiter_per_stage, 'ftol': ftol, 'gtol': 1e-8})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc <= current_score:
                h_cur = h_new
                current_score = sc
        except Exception:
            pass
    return h_cur, current_score

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        adj = diff / len(idx)
        h[idx] += adj
        h = np.clip(h, 0.0, 1.0)
    return h


def corr_grad_at_k(h, k):
    n = len(h)
    grad = np.zeros(n)
    shift = k - (n - 1)
    idx1 = np.arange(n) + shift
    mask1 = (idx1 >= 0) & (idx1 < n)
    if np.any(mask1):
        grad[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = np.arange(n) - shift
    mask2 = (idx2 >= 0) & (idx2 < n)
    if np.any(mask2):
        grad[mask2] -= h[idx2[mask2]]
    return grad


def lsp_obj_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(corr - m, -60.0 / alpha, 0.0)
    exp_vals = np.exp(alpha * shifted)
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        gk = corr_grad_at_k(h, k)
        grad += w[k] * gk
    return F, grad


def srsa_with_momentum(h, target, c, a, mu, rng):
    n = len(h)
    delta = rng.choice([-1.0, 1.0], size=n)
    h_plus = project_feasible(h + c * delta, target)
    h_minus = project_feasible(h - c * delta, target)
    f_plus = compute_c5(h_plus)
    f_minus = compute_c5(h_minus)
    g_est = (f_plus - f_minus) / (2.0 * c) * delta
    return g_est, f_plus, f_minus


def srsa_phase(h_init, target, n_iter, c0, a0, mu, c_gamma, a_gamma, c_dec, a_dec, rng, time_check):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    velocity = np.zeros(n)
    stall_count = 0
    for i in range(n_iter):
        if time_check() < 30:
            break
        c = c0 / (1.0 + i * c_gamma) ** c_dec
        a = a0 / (1.0 + i * a_gamma) ** a_dec
        g_est, f_plus, f_minus = srsa_with_momentum(h, target, c, a, mu, rng)
        velocity = mu * velocity + a * g_est
        h_new = h - velocity
        h_new = project_feasible(h_new, target)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            h = h_new
            stall_count = 0
        else:
            if rng.random() < 0.05:
                h = h_new
            stall_count += 1
            if stall_count > 500:
                stall_count = 0
                h = best_h + rng.randn(n) * 0.005
                h = project_feasible(h, target)
    return best_h, best_f


def lbfgs_refine(h_start, target, alpha_schedule, maxiter_per_stage, ftol, time_check):
    h_cur = project_feasible(h_start.copy(), target)
    n = len(h_cur)
    current_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    for alpha in alpha_schedule:
        if time_check() < 25:
            break
        try:
            def obj(x):
                F, _ = lsp_obj_and_grad(x, alpha)
                return F
            def jac(x):
                _, g = lsp_obj_and_grad(x, alpha)
                return g
            res = minimize(obj, h_cur, jac=jac, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': maxiter_per_stage, 'ftol': ftol, 'gtol': 1e-9})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc <= current_score:
                h_cur = h_new
                current_score = sc
        except Exception:
            pass
    return h_cur, current_score


def subgradient_phase(h_init, target, n_iter, step0, step_decay, rng, time_check):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    dx = 2.0 / n
    for i in range(n_iter):
        if time_check() < 25:
            break
        step = step0 / (1.0 + i * step_decay)
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        active = np.where(corr > m - 1e-8)[0]
        n_active = len(active)
        grad = np.zeros(n)
        for k in active:
            grad += corr_grad_at_k(h, k)
        grad /= n_active
        h_new = h - step * dx * grad
        h_new = project_feasible(h_new, target)
        f_new = compute_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
            h = h_new
    return best_h, best_f


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    x_new = np.linspace(0.0, 2.0, n_dst, endpoint=False)
    x_old = np.linspace(0.0, 2.0, n_src, endpoint=False)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def block_coordinate_descent(h_init, target, n_blocks, time_check, rng):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_f = compute_c5(h)
    block_size = n // n_blocks
    for _ in range(200):
        if time_check() < 25:
            break
        improved = False
        for b in range(n_blocks):
            lo = b * block_size
            hi = lo + block_size if b < n_blocks - 1 else n
            for trial in range(30):
                if time_check() < 25:
                    break
                h_trial = h.copy()
                delta = rng.randn(hi - lo) * 0.01
                h_trial[lo:hi] += delta
                h_trial = project_feasible(h_trial, target)
                f_trial = compute_c5(h_trial)
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = h_trial.copy()
                    h = h_trial
                    improved = True
        if not improved:
            break
    return best_h, best_f
