import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def lag_gradient(h, lag_idx):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    s = lag_idx - (n - 1)
    if s >= 0:
        m = n - s
        grad[s:] += dx * omh[:m]
        grad[:m] -= dx * h[s:]
    else:
        r = -s
        m = n - r
        grad[:m] += dx * omh[r:]
        grad[r:] -= dx * h[:m]
    return grad

def topk_gradient(h, k, temp):
    n = len(h)
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        grad += wk * lag_gradient(h, topk_idx[i])
    return grad

def lag_focused_optimize(h, n_steps, time_limit, start_time, top_k=3, lr_init=0.015):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0
    stall_limit = 400

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h_cur)
        cmax = np.max(corr)
        if cmax < best_c5:
            best_c5 = cmax
            best_h = h_cur.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            h_cur = best_h.copy()
            lr = lr_init * max(0.1, 1.0 - stall / (stall_limit * 3.0))
            stall = 0
            stall_limit = max(100, stall_limit // 2)

        topk_idx = np.argsort(corr)[-top_k:]
        for lag in topk_idx:
            grad = lag_gradient(h_cur, lag)
            h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_cur = h_new
            if time.time() - start_time > time_limit:
                return best_h, best_c5

        if step % 50 == 49 and best_c5 < cmax * 1.0001:
            h_cur = best_h.copy()
            lr = lr_init * 0.6

    return best_h, best_c5

        h_lag, c5_lag = lag_focused_optimize(best_h, 200000, tl, start_time, top_k=3, lr_init=0.012)
        if c5_lag < best_c5:
            best_c5 = c5_lag
            best_h = h_lag.copy()

    # Phase 3: Medium SLSQP
    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.003, 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: More lag-focused
    if rem() > 60:
        tl = min(rem() - 20, 150)
        h_lag, c5_lag = lag_focused_optimize(best_h, 200000, tl, start_time, top_k=2, lr_init=0.008)
        if c5_lag < best_c5:
            best_c5 = c5_lag
            best_h = h_lag.copy()

    # Phase 5: Fine SLSQP
    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Single worst lag gradient
    if rem() > 40:
        tl = min(rem() - 15, 120)
        h_lag, c5_lag = lag_focused_optimize(best_h, 200000, tl, start_time, top_k=1, lr_init=0.005)
        if c5_lag < best_c5:
            best_c5 = c5_lag
            best_h = h_lag.copy()

    # Phase 7: Restart loop with diverse strategies
    n_restart = 0
    while rem() > 90 and n_restart < 35:
        strategy = n_restart % 5
        h_pert = gen_perturb(best_h, rng, n0, target_sum, strategy)

        if n_restart < 12:
            eps_use = 0.002
            maxit_use = 300
        elif n_restart < 24:
            eps_use = 0.0008
            maxit_use = 400
        else:
            eps_use = 0.0003
            maxit_use = 500

        h_opt, c5_opt = slsqp_optimize(h_pert, eps_use, maxit_use)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        if rem() > 50 and c5_opt <= best_c5 + 0.002:
            tl_pg = min(rem() - 20, 35)
            h_lag, c5_lag = lag_focused_optimize(h_opt, 80000, tl_pg, start_time, top_k=2, lr_init=0.006)
            if c5_lag < best_c5:
                best_c5 = c5_lag
                best_h = h_lag.copy()

        n_restart += 1

    # Phase 8: Ultra-fine SLSQP
    if rem() > 50:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 9: Final lag polish
    if rem() > 30:
        tl = min(rem() - 10, 80)
        h_lag, c5_lag = lag_focused_optimize(best_h, 150000, tl, start_time, top_k=1, lr_init=0.003)
        if c5_lag < best_c5:
            best_c5 = c5_lag
            best_h = h_lag.copy()

    # Phase 10: Resolution increase
    if rem() > 180:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.004, 300)
        if rem() > 80:
            h_opt_up, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.001, 300)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
        if rem() > 40:
            tl_up = min(rem() - 20, 60)
            h_lag_up, c5_lag_up = lag_focused_optimize(h_opt_up, 100000, tl_up, start_time, top_k=2, lr_init=0.005)
            if c5_lag_up < c5_opt_up:
                c5_opt_up = c5_lag_up
                h_opt_up = h_lag_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()

    # Final
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
