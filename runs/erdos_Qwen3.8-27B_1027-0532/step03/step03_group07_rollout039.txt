    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    m = np.max(corr)
    if m > -500:
        weights = np.exp(alpha * (corr - m))
        total = weights.sum()
        obj = (m + np.log(total) / alpha) * dx
        weights = weights / total
    else:
        obj = m * dx
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    
    # Gradient: sum of weights[k] * grad corr[k]
    grad = np.zeros(n)
    for k_idx in range(len(corr)):

        if lag >= 0:

- n + 1 variables (h_1,...,h_n, t)
- 2n-1 inequality constraints (one per lag)

- 1 equality constraint

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):

                if m2 > 0:
                    g[:m2] += (1.0 - h[j2:])
                    g[j2:] += -h[:m2]

            A_ub[k_idx, n] = -1.0
            b_ub[k_idx] = dx * np.dot(g, h) - dx * corr[k_idx]
        A_eq = np.zeros((1, n + 1))

            if c5_slr < best_c5:
                best_c5 = c5_slr
                best_h = h_slr.copy()

    # Final aggressive polish
    if time.time() < deadline - 8:
        target = best_n / 2.0
        h_pol = _slp_refine(best_h, best_n, target, max_iters=50, deadline=deadline - 8)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    if time.time() < deadline - 5:
        h_pol2 = _slsqp_analytical(best_h, best_n, best_n / 2.0, 10000, maxiter=400)
        c5_pol2 = _c5_of(h_pol2)
        if c5_pol2 < best_c5:
            best_c5 = c5_pol2
            best_h = h_pol2.copy()

    if time.time() < deadline - 3:
        h_cd_f = _coord_descent_fast(best_h, best_n, best_n / 2.0, max_rounds=30, deadline=deadline - 3)
        c5_cd_f = _c5_of(h_cd_f)
        if c5_cd_f < best_c5:
            best_c5 = c5_cd_f
            best_h = h_cd_f.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
