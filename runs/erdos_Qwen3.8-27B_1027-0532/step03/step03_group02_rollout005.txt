    h[(x >= shift) & (x < shift + 0.5)] = 1.0

    h[(x >= shift - 1.0) & (x < shift - 0.5)] = 1.0
    h[(x >= shift - 2.0) & (x < shift - 1.5)] = 1.0
    return np.clip(h, 0.0, 1.0)

def make_sine(n, freq=1.0, phase=0.0):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.clip(0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase), 0.0, 1.0)

def make_tri(n):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.abs(2.0 * x - np.floor(2.0 * x + 0.5))
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - 25.0 - (time.time() - t_start)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_score = compute_c5(best_h)

    # Phase 1: Anti-periodic CD at initial resolution
    if tl() > 250:
        h_a = project_anti(h_init, n_init)
        h_r, s_r = coord_anti(h_a, n_init, min(350, tl() - 250), rng, patience=35)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    # Phase 2: Multi-resolution with anti-periodic
    res_list = [n_init + 100, n_init - 100, n_init + 200, 300, 500, 600, 800]
    res_list = sorted(set([int(r) for r in res_list if r >= 100 and r != n_init and r % 2 == 0]))
    for n_try in res_list:
        if tl() < 130:
            break
        h_i = interp_n(best_h, n_try)
        h_cp = project_anti(h_i, n_try)
        alloc = min(100, tl() - 130)
        if alloc < 20:
            continue
        h_r, s_r = coord_anti(h_cp, n_try, alloc, rng, patience=25)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    # Phase 3: Different initializations
    inits = []
    for n_try in [n_init, 400, 600]:
        if n_try % 2 != 0:
            n_try += 1
        inits.append(make_block(n_try, 0.0))
        inits.append(make_block(n_try, 0.25))
        inits.append(make_sine(n_try, 1.0, 0.0))
        inits.append(make_sine(n_try, 1.0, np.pi / 4))
        inits.append(make_sine(n_try, 2.0, 0.0))
        inits.append(make_tri(n_try))
        inits.append(project_anti(rng.rand(n_try), n_try))

    for h_ini in inits:
        if tl() < 80:
            break
        n_try = len(h_ini)
        h_cp = project_anti(h_ini, n_try)
        alloc = min(70, tl() - 80)
        if alloc < 15:
            continue
        h_r, s_r = coord_anti(h_cp, n_try, alloc, rng, patience=20)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    # Phase 4: Unconstrained CD on best
    n_best = len(best_h)
    if tl() > 120:
        h_r, s_r = coord_uncon(best_h, n_best, min(150, tl() - 120), rng, patience=35)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    # Phase 5: Anti-periodic polish on best
    n_final = len(best_h)
    if tl() > 60:
        h_cp = project_anti(best_h, n_final)
        h_r, s_r = coord_anti(h_cp, n_final, min(90, tl() - 60), rng, patience=50)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    # Phase 6: Final unconstrained polish
    n_final = len(best_h)
    if tl() > 40:
        h_r, s_r = coord_uncon(best_h, n_final, min(50, tl() - 40), rng, patience=50)
        if s_r < best_score:
            best_score = s_r
            best_h = h_r

    h_out = project_feasible(best_h, len(best_h) / 2.0)
    final_score = compute_c5(h_out)
    if final_score < best_score:
        best_score = final_score
    return h_out, best_score, len(h_out)
