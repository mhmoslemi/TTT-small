import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse(x, alpha):
    m = np.max(x)
    shifted = np.clip(alpha * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / alpha

_gs_sum = 0.0
_gs_n = 0

def _eq_fun(x):
    return x.sum() - _gs_sum

def _eq_jac(x):
    return np.ones(_gs_n)

def _smooth_obj(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _lse(corr, alpha) * dx

def _smooth_grad(x, alpha):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    shifted = np.clip(alpha * (corr - m), -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()
    g = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m2 = n - lag
            if m2 > 0:
                g[lag:lag+m2] += w[k] * (1.0 - x[:m2])
                g[:m2] += w[k] * (-x[lag:lag+m2])
        else:
            j = -lag
            m2 = n - j
            if m2 > 0:
                g[:m2] += w[k] * (1.0 - x[j:])
                g[j:] += w[k] * (-x[:m2])
    return g * dx

def _slsqp_refine(h0, n, target_sum, alpha, maxiter):
    global _gs_sum, _gs_n
    _gs_sum = target_sum
    _gs_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha,),
                       jac=lambda x: _smooth_grad(x, alpha),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_active(h0, n, target_sum, n_iters, top_m, wtemp, lr, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = np.max(corr)
        w = np.exp(np.clip(wtemp * (corr - mc), -500, 500))
        tw = w.sum()
        if tw < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w = w / tw
        top = np.argsort(w)[::-1][:top_m]
        wm = w[top].copy()
        wm = wm / wm.sum()
        g = np.zeros(n)
        for ii in range(len(top)):
            k = top[ii]
            lag = k - (n - 1)
            if lag >= 0:
                m2 = n - lag
                if m2 > 0:
                    g[lag:lag+m2] += wm[ii] * (1.0 - y[:m2])
                    g[:m2] += wm[ii] * (-y[lag:lag+m2])
            else:
                j = -lag
                m2 = n - j
                if m2 > 0:
                    g[:m2] += wm[ii] * (1.0 - y[j:])
                    g[j:] += wm[ii] * (-y[:m2])
        g *= dx
        a = lr / (1.0 + 0.008 * it)
        x_new = _project(y - a * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_desc(h0, n, target_sum, max_rounds, nvals):
    h = h0.copy()
    vals = np.unique(np.concatenate([
        np.linspace(0.0, 1.0, nvals),
        np.array([1.0/3.0, 2.0/3.0, 0.05, 0.95])
    ]))
    current_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_c = current_c5
            best_v = old
            for v in vals:
                if abs(v - old) < 1e-10:
                    continue
                h[j] = v
                c = _c5_of(h)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = v
            h[j] = best_v
            if best_v != old:
                improved = True
                current_c5 = best_c
        if not improved:
            break
    return h

def _pair_cd(h0, n, target_sum, max_rounds):
    h = h0.copy()
    vals = np.array([0.0, 0.15, 0.3, 0.4, 0.5, 0.6, 0.7, 0.85, 1.0])
    current_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n - 1):
            o1, o2 = h[j], h[j+1]
            best_c = current_c5
            bp = (o1, o2)
            changed = False
            for v1 in vals:
                h[j] = v1
                for v2 in vals:
                    if v1 == o1 and v2 == o2:
                        continue
                    h[j+1] = v2
                    c = _c5_of(h)
                    if c < best_c - 1e-15:
                        best_c = c
                        bp = (v1, v2)
                        changed = True
            h[j], h[j+1] = bp
            if changed:
                improved = True
                current_c5 = best_c
        if not improved:
            break
    return h

def _triple_cd(h0, n, target_sum, max_rounds):
    h = h0.copy()
    vals = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    current_c5 = _c5_of(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(0, n - 2, 1):
            o1, o2, o3 = h[j], h[j+1], h[j+2]
            best_c = current_c5
            bp = (o1, o2, o3)
            changed = False
            for v1 in vals:
                h[j] = v1
                for v2 in vals:
                    h[j+1] = v2
                    for v3 in vals:
                        if v1 == o1 and v2 == o2 and v3 == o3:
                            continue
                        h[j+2] = v3
                        c = _c5_of(h)
                        if c < best_c - 1e-15:
                            best_c = c
                            bp = (v1, v2, v3)
                            changed = True
            h[j], h[j+1], h[j+2] = bp
            if changed:
                improved = True
                current_c5 = best_c
        if not improved:
            break
    return h

def _interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.arange(n_src, dtype=float)
    x_dst = np.linspace(0, n_src - 1, n_dst)
    return np.interp(x_dst, x_src, h_src)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 20

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 140
        h_init = np.full(n_init, 0.5)

    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = n_init

    n_list = []
    for d in [0, 2, -2, 4, -4, 6, -6, 8, -8, 10, -10]:
        nn = n_init + d
        if nn % 2 == 0 and 100 <= nn <= 200 and nn not in n_list:
            n_list.append(nn)
    n_list.sort(key=lambda x: abs(x - n_init))

    time_per_n = (deadline - t0) / len(n_list)

    for n in n_list:
        t_n_start = time.time()
        if t_n_start > deadline - 70:
            break
        ts = n / 2.0

        if n == n_init:
            h = _fix_sum(h_init.copy(), ts)
        else:
            h = _fix_sum(_interp_to_n(h_init, n), ts)

        c5 = _c5_of(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

        phase_deadline = min(t_n_start + time_per_n, deadline - 50)

        # Phase 1: Nesterov
        if time.time() < phase_deadline - 60:
            h, c5 = _nesterov_active(h, n, ts, 200, 10, 40.0, 0.35,
                                      min(time.time() + 30, phase_deadline - 60))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

        # Phase 2: Multi-alpha SLSQP (homotopy)
        alpha_schedule = [50, 100, 250, 500, 1000, 2500, 5000, 10000]
        for alpha in alpha_schedule:
            if time.time() > phase_deadline - 20:
                break
            maxit = 200 if alpha <= 500 else 300
            h2 = _slsqp_refine(h, n, ts, alpha, maxit)
            c52 = _c5_of(h2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()
                best_n = n
            h = h2

        # Phase 3: Coordinate descent
        if time.time() < phase_deadline - 15:
            h = _coord_desc(h, n, ts, max_rounds=25, nvals=17)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

        # Phase 4: Pair CD
        if time.time() < phase_deadline - 10:
            h = _pair_cd(h, n, ts, max_rounds=10)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

        # Phase 5: More SLSQP at high alpha
        for alpha in [15000, 30000]:
            if time.time() > phase_deadline - 8:
                break
            h2 = _slsqp_refine(h, n, ts, alpha, 350)
            c52 = _c5_of(h2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2.copy()
                best_n = n
            h = h2

        # Phase 6: Final CD
        if time.time() < phase_deadline - 5:
            h = _coord_desc(h, n, ts, max_rounds=20, nvals=21)
            c5 = _c5_of(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n

    # Global final polish on the best solution found
    if time.time() < deadline - 15 and best_h is not None:
        nb = best_n
        tb = nb / 2.0
        hp = best_h.copy()

        for alpha in [10000, 30000, 60000]:
            if time.time() > deadline - 12:
                break
            hp = _slsqp_refine(hp, nb, tb, alpha, 500)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        if time.time() < deadline - 10:
            hp = _coord_desc(hp, nb, tb, max_rounds=25, nvals=21)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        if time.time() < deadline - 8:
            hp = _pair_cd(hp, nb, tb, max_rounds=12)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        if time.time() < deadline - 6:
            hp = _coord_desc(hp, nb, tb, max_rounds=15, nvals=25)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        if time.time() < deadline - 4:
            hp = _slsqp_refine(hp, nb, tb, 80000, 500)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

        if time.time() < deadline - 2:
            hp = _coord_desc(hp, nb, tb, max_rounds=10, nvals=25)
            c5 = _c5_of(hp)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hp.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
