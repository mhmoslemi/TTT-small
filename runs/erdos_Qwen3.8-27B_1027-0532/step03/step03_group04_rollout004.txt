def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.5, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        g = 1.0 - h
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == target, h_var >= 0.0, h_var <= 1.0]
        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)
        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=20000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=20000)
            except Exception:
                break
        if h_var.value is None:
            break
        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val

            stuck_count = 0

def smooth_obj(h, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return float(c_max + np.log(np.sum(np.exp(shifted))) / alpha)

    h_rand = np.zeros(n)
    ones_idx = rng.choice(n, size=n // 2, replace=False)
    h_rand[ones_idx] = 1.0
    binary_starts.append(h_rand)

    h_qp = make_quasi_periodic(n, 8, rng)
    h_qp_bin = (h_qp > 0.5).astype(float)
    n1 = int(np.sum(h_qp_bin))
    if n1 != n // 2:
        if n1 > n // 2:
            excess = np.where(h_qp_bin == 1)[0]
            h_qp_bin[excess[rng.randint(0, len(excess), size=n1 - n // 2)]] = 0
        else:
            deficit = np.where(h_qp_bin == 0)[0]
            h_qp_bin[deficit[rng.randint(0, len(deficit), size=n // 2 - n1)]] = 1
    binary_starts.append(h_qp_bin)

    h_qp2 = make_quasi_periodic(n, 16, rng)
    h_qp2_bin = (h_qp2 > 0.5).astype(float)
    n1 = int(np.sum(h_qp2_bin))
    if n1 != n // 2:
        if n1 > n // 2:
            excess = np.where(h_qp2_bin == 1)[0]
            h_qp2_bin[excess[rng.randint(0, len(excess), size=n1 - n // 2)]] = 0
        else:
            deficit = np.where(h_qp2_bin == 0)[0]
            h_qp2_bin[deficit[rng.randint(0, len(deficit), size=n // 2 - n1)]] = 1
    binary_starts.append(h_qp2_bin)

    for i, bstart in enumerate(binary_starts):
        if tleft() < 150:
            break
        sa_time = min(200, tleft() * 0.2)
        h_sa, c5_sa = binary_sa(bstart, n, sa_time, rng, T0=0.006)
        if c5_sa < best_score:
            best_score = c5_sa
            best_h = h_sa.copy()

    if tleft() > 200:
        h_best_bin = (best_h > 0.5).astype(float)
        n1 = int(np.sum(h_best_bin))
        if n1 != n // 2:
            if n1 > n // 2:
                excess = np.where(h_best_bin == 1)[0]
                h_best_bin[excess[rng.randint(0, len(excess), size=n1 - n // 2)]] = 0
            else:
                deficit = np.where(h_best_bin == 0)[0]
                h_best_bin[deficit[rng.randint(0, len(deficit), size=n // 2 - n1)]] = 1

        gre_time = min(200, tleft() * 0.2)
        h_gre, c5_gre = binary_greedy_improve(h_best_bin, n, gre_time, rng)
        if c5_gre < best_score:
            best_score = c5_gre
            best_h = h_gre.copy()

        sa_time2 = min(150, tleft() * 0.15)
        h_sa2, c5_sa2 = binary_sa(h_gre, n, sa_time2, rng, T0=0.003)
        if c5_sa2 < best_score:
            best_score = c5_sa2
            best_h = h_sa2.copy()

    if tleft() > 150:
        lp_time = min(200, tleft() * 0.25)
        h_lp2, c5_lp2 = alternating_lp(best_h, n, target,
                                        max_iter=4000, time_limit=lp_time,
                                        damping_init=0.5, reg=1e-4)
        if c5_lp2 < best_score:
            best_score = c5_lp2
            best_h = h_lp2

    if tleft() > 100:
        h_ref, c5_ref = slsqp_refine(best_h, n, target,
                                      [50, 200, 500, 1000], 2000, 1e-14,
                                      time_limit=min(120, tleft() * 0.3))
        if c5_ref < best_score:
            best_score = c5_ref
            best_h = h_ref

    if tleft() > 80:
        for noise in [0.03, 0.08]:
            if tleft() < 60:
                break
            h_pert = best_h + rng.randn(n) * noise
            h_pert = project_feasible(h_pert, target)
            h_pert_bin = (h_pert > 0.5).astype(float)
            n1 = int(np.sum(h_pert_bin))
            if n1 != n // 2:
                if n1 > n // 2:
                    excess = np.where(h_pert_bin == 1)[0]
                    h_pert_bin[excess[rng.randint(0, len(excess), size=n1 - n // 2)]] = 0
                else:
                    deficit = np.where(h_pert_bin == 0)[0]
                    h_pert_bin[deficit[rng.randint(0, len(deficit), size=n // 2 - n1)]] = 1
            sa_t = min(80, tleft() * 0.2)
            h_s, c5_s = binary_sa(h_pert_bin, n, sa_t, rng, T0=0.004)
            if c5_s < best_score:
                best_score = c5_s
                best_h = h_s.copy()
            if tleft() > 60:
                h_r, c5_r = slsqp_refine(best_h, n, target,
                                          [100, 400, 800], 1500, 1e-14,
                                          time_limit=min(60, tleft() * 0.3))
                if c5_r < best_score:
                    best_score = c5_r
                    best_h = h_r

    if tleft() > 40:
        h_final_try, c5_ft = slsqp_refine(best_h, n, target,
                                           [500, 1000, 2000], 800, 1e-15,
                                           time_limit=min(50, tleft() * 0.6))
        if c5_ft < best_score:
            best_score = c5_ft
            best_h = h_final_try

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)

    return h_final, final_score, len(h_final)
