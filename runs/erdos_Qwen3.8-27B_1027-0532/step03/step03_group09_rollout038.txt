import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def top_lags(corr, k):
    return np.argsort(corr)[-k:]

def smooth_max_obj_active(h, eps, active_idx):
    n = len(h)
    corr = compute_corr(h)
    corr_use = corr[active_idx]
    cmax = np.max(corr_use)
    x = (corr_use - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad_active(h, eps, active_idx):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    corr_use = corr[active_idx]
    cmax = np.max(corr_use)
    x = (corr_use - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for i in range(len(active_idx)):
        kidx = active_idx[i]
        wk = w[i]
        if wk < 1e-14:
            continue
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def smooth_max_obj_all(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad_all(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_active(h_init, eps, active_idx, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj_active, h0, args=(eps, active_idx),
                          jac=lambda x: smooth_max_grad_active(x, eps, active_idx),
                          method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_all_lags(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj_all, h0, args=(eps,),
                          jac=lambda x: smooth_max_grad_all(x, eps),
                          method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})

        return h_opt, compute_c5(h_opt)

        return h0, compute_c5(h0)

def active_set_optimize(h_init, n_active, n_outer, eps_init, t_start, t_budget):
    n = len(h_init)
    target_sum = n / 2.0
    h_cur = h_init.copy()
    c5_cur = compute_c5(h_cur)

    for outer in range(n_outer):
        if time.time() - t_start > t_budget:
            break
        corr = compute_corr(h_cur)
        active = top_lags(corr, n_active)
        episode_idx = outer % 3
        eps = eps_init * (0.5 ** episode_idx)
        maxiter = 300 - 50 * episode_idx
        h_new, c5_new = slsqp_active(h_cur, eps, active, maxiter)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
    return h_cur, c5_cur

def differential_evolution(h_seed, n, target_sum, rng, t_start, t_budget):
    pop_size = 44
    pop = np.zeros((pop_size, n))
    pop[0] = h_seed.copy()
    for i in range(1, pop_size):
        r = i % 5
        if r == 0:
            pop[i] = project_onto_constraint(0.5 + 0.35 * rng.standard_normal(n), target_sum)
        elif r == 1:
            pop[i] = project_onto_constraint(h_seed + 0.08 * rng.standard_normal(n), target_sum)
        elif r == 2:
            h_bin = (rng.random(n) < 0.5).astype(float)
            alpha = 0.6 + 0.3 * rng.random()
            pop[i] = project_onto_constraint(alpha * h_seed + (1 - alpha) * h_bin, target_sum)
        elif r == 3:
            h_p = h_seed.copy()
            blen = max(3, n // 10)
            a = int(rng.integers(0, n - blen))
            b = int(rng.integers(0, n - blen))
            if abs(a - b) >= blen:
                tmp = h_p[a:a + blen].copy()
                h_p[a:a + blen] = h_p[b:b + blen]
                h_p[b:b + blen] = tmp
            pop[i] = project_onto_constraint(h_p + 0.02 * rng.standard_normal(n), target_sum)
        else:
            h_p = 1.0 - h_seed
            h_p = project_onto_constraint(h_p, target_sum)
            pop[i] = project_onto_constraint(0.5 * h_seed + 0.5 * h_p + 0.05 * rng.standard_normal(n), target_sum)

    fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    best_idx = np.argmin(fitness)
    best_c5 = fitness[best_idx]
    best_h = pop[best_idx].copy()
    F = 0.7
    CR = 0.35
    no_improve = 0
    max_no_improve = 25

    while time.time() - t_start < t_budget:
        for i in range(1, pop_size):
            if time.time() - t_start > t_budget:
                break
            idxs = rng.choice(pop_size, size=3, replace=False)
            a, b, c = idxs[0], idxs[1], idxs[2]
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = project_onto_constraint(mutant, target_sum)
            cross_mask = rng.random(n) < CR
            if not cross_mask.any():
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = project_onto_constraint(trial, target_sum)
            fit_trial = compute_c5(trial)
            if fit_trial < fitness[i]:
                pop[i] = trial
                fitness[i] = fit_trial

        new_best = np.argmin(fitness)
        if fitness[new_best] < best_c5 - 1e-9:
            best_c5 = fitness[new_best]
            best_h = pop[new_best].copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                F = min(F + 0.15, 1.4)
                CR = min(CR + 0.1, 0.75)
                n_mut = max(3, pop_size // 6)
                for _ in range(n_mut):
                    mi = int(rng.integers(1, pop_size))
                    pop[mi] = project_onto_constraint(pop[mi] + 0.12 * rng.standard_normal(n), target_sum)
                    fitness[mi] = compute_c5(pop[mi])
                no_improve = 0
                max_no_improve = max(15, max_no_improve - 3)

    return best_h, best_c5

def projected_grad_topk(h, k, temp, lr_init, n_steps, t_start, t_budget):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    stall = 0

    for step in range(n_steps):
        if time.time() - t_start > t_budget:
            break
        corr = compute_corr(h_cur)
        topk_idx = np.argsort(corr)[-k:]
        xc = corr[topk_idx]
        xc = xc - np.max(xc)
        w = np.exp(xc / temp)
        w = w / np.sum(w)
        grad = np.zeros(n)
        omh = 1.0 - h_cur
        dx = 2.0 / n
        for i in range(len(topk_idx)):
            wk = w[i]
            if wk < 1e-14:
                continue
            kidx = topk_idx[i]
            s = kidx - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:] += wk * dx * omh[:m]
                grad[:m] -= wk * dx * h[s:]
            else:
                r = -s
                m = n - r
                grad[:m] += wk * dx * omh[r:]
                grad[r:] -= wk * dx * h[:m]
        h_new = project_onto_constraint(h_cur - lr * grad, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.01, 0.25)
        else:
            stall += 1
            lr = max(lr * 0.97, 1e-8)
        h_cur = h_new
        if stall > 800:
            h_cur = best_h.copy()
            lr = lr_init * 0.6
            stall = 0

    return best_h, best_c5
