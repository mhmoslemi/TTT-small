        h_bin = project_onto_constraint(h_bin, target_sum)
        alpha = float(rng.uniform(0.55, 0.85))
        return project_onto_constraint(alpha * h + (1.0 - alpha) * h_bin, target_sum)
    elif strategy == 4:
        h_p = h.copy()
        half = n // 2
        n_swap = max(3, n // 8)
        idx = rng.choice(half, size=min(n_swap, half), replace=False)
        h_p[idx] = 1.0 - h[half - 1 - idx]
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 5:
        h_p = h.copy()
        sigma = float(rng.uniform(0.01, 0.04))
        h_p += sigma * rng.standard_normal(n)
        threshold = float(np.sort(h_p)[int(n * 0.3)])
        h_p[h_p > threshold] = np.clip(h_p[h_p > threshold] + 0.05, 0, 1)
        h_p[h_p <= threshold] = np.clip(h_p[h_p <= threshold] - 0.05, 0, 1)
        return project_onto_constraint(h_p, target_sum)
    else:
        return project_onto_constraint(h + 0.008 * rng.standard_normal(n), target_sum)

def ga_crossover(h1, h2, rng, target_sum):
    n = len(h1)
    cp = int(rng.integers(n // 4, 3 * n // 4))
    child = np.concatenate([h1[:cp], h2[cp:]])
    return project_onto_constraint(child, target_sum)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time

    def rem():
        return budget_s - elapsed()

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = np.clip(0.5 + 0.3 * rng.standard_normal(n0), 0.0, 1.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.008, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    pop_size = 36
    population = [best_h.copy()]
    fitness = [best_c5]

    for i in range(pop_size - 1):
        strat = i % 6
        h_init = ga_mutation(best_h, rng, target_sum, strat)
        c5_init = compute_c5(h_init)
        population.append(h_init)
        fitness.append(c5_init)

    gen = 0
    stagnant = 0
    max_gen = 400
    slsqp_counter = 0

    while rem() > 250 and gen < max_gen:
        gen += 1
        order = np.argsort(fitness)
        population = [population[i] for i in order]
        fitness = [fitness[i] for i in order]

        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnant = 0
        else:
            stagnant += 1

        elite_size = max(4, pop_size // 5)
        new_pop = population[:elite_size][:]
        new_fit = fitness[:elite_size][:]

        while len(new_pop) < pop_size:
            t_size = min(5, len(population))
            t_idx = rng.choice(len(population), size=t_size, replace=False)
            winner_idx = t_idx[np.argmin([fitness[i] for i in t_idx])]
            winner = population[winner_idx]

            if rng.random() < 0.55 and len(new_pop) >= 2:
                t_idx2 = rng.choice(len(population), size=t_size, replace=False)
                winner2_idx = t_idx2[np.argmin([fitness[i] for i in t_idx2])]
                winner2 = population[winner2_idx]
                child = ga_crossover(winner, winner2, rng, target_sum)
            else:
                child = winner.copy()

            strat = int(rng.integers(0, 7))
            child = ga_mutation(child, rng, target_sum, strat)
            new_pop.append(child)
            new_fit.append(compute_c5(child))

        population = new_pop
        fitness = new_fit

        if gen % 12 == 0 and rem() > 300:
            slsqp_counter += 1
            eps_seq = [0.003, 0.001, 0.0005, 0.0002]
            eps_use = eps_seq[min(slsqp_counter - 1, len(eps_seq) - 1)]
            h_pol, c5_pol = slsqp_optimize(population[0], eps_use, 150)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()
                population[0] = h_pol.copy()
                fitness[0] = c5_pol
            if rem() > 60:
                h_pol2, c5_pol2 = proj_grad_topk(population[0], 2, 0.003, 0.04, 50000, min(rem() - 40, 30), start_time)
                if c5_pol2 < best_c5:
                    best_c5 = c5_pol2
                    best_h = h_pol2.copy()
                    population[0] = h_pol2.copy()
                    fitness[0] = c5_pol2

        if stagnant > 25:
            for i in range(pop_size // 3):
                strat = int(rng.integers(2, 7))
                h_new = ga_mutation(best_h, rng, target_sum, strat)
                population[i] = h_new
                fitness[i] = compute_c5(h_new)
            stagnant = 0

    if rem() > 100:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.002, 350)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 80:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 350)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 60:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0005, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 50:
        tl = min(rem() - 20, 150)
        h_pg, c5_pg = proj_grad_topk(best_h, 3, 0.005, 0.06, 500000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 40:
        tl = min(rem() - 15, 100)
        h_pg, c5_pg = proj_grad_topk(best_h, 1, 0.001, 0.03, 500000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    if rem() > 40:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0002, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 30:
        h_opt, c5_opt = slsqp_optimize(best_h, 0.0001, 250)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 120:
        n1 = n0 + 40
        old_t = np.linspace(0.0, 1.0, n0)
        new_t = np.linspace(0.0, 1.0, n1)
        h_up = np.interp(new_t, old_t, best_h)
        h_up = project_onto_constraint(h_up, n1 / 2.0)
        h_up = h_up + 0.005 * rng.standard_normal(n1)
        h_up = project_onto_constraint(h_up, n1 / 2.0)

        h_opt_up, c5_opt_up = slsqp_optimize(h_up, 0.004, 250)
        if rem() > 60:
            h_opt_up, c5_opt_up2 = slsqp_optimize(h_opt_up, 0.001, 200)
            if c5_opt_up2 < c5_opt_up:
                c5_opt_up = c5_opt_up2
        if rem() > 40:
            tl_up = min(rem() - 20, 50)
            h_pg_up, c5_pg_up = proj_grad_topk(h_opt_up, 2, 0.004, 0.04, 100000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up
        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
