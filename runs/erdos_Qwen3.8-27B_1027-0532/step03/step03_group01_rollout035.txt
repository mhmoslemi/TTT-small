from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible_exact(h, target):
    n = len(h)
    # Binary search for threshold
    lo, hi = h.min() - 0.1, h.max() + 0.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_proj = np.clip(h - tau, 0.0, 1.0)
    # Fix tiny errors
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h_proj > 1e-12) & (h_proj < 1.0 - 1e-12))[0]
        if len(idx) > 0:
            h_proj[idx] += diff / len(idx)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(float)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if abs(current_sum - target_sum) < 1e-10:
        return h_r
    if current_sum > target_sum:
        excess = int(round(current_sum - target_sum))
        # Set the 'most fractional' ones to 0
        ones = np.where(h_r == 1)[0]
        # Sort by how much they "want" to be 1 (lower h value = less wanted)
        order = np.argsort(h[ones])[:excess]
        h_r[ones[order]] = 0
    else:
        deficit = int(round(target_sum - current_sum))
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        h_r[zeros[order]] = 1
    # Final fix
    final_sum = h_r.sum()
    if final_sum != target_sum:
        diff = target_sum - final_sum
        if diff > 0:
            zeros = np.where(h_r == 0)[0]
            h_r[zeros[:int(diff)]] = 1
        else:
            ones = np.where(h_r == 1)[0]
            h_r[ones[:int(-diff)]] = 0
    return h_r

def c5_grad_fft(h, alpha):
    """Compute smoothed C5 and its gradient at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)  # 2n-1
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    val = (c_max + np.log(sum_exp) / alpha) * dx
    
    # Gradient
    one_mh = 1.0 - h
    A = fftconvolve(one_mh, w, mode='full')[:n]
    
    M = 2 * m + 2
    h_ft = np.fft.rfft(h, M)
    w_ft = np.fft.rfft(w, M)
    r = np.fft.irfft(h_ft * np.conj(w_ft), M)
    B = r[:n]
    
    grad = (A - B) * dx
    return val, grad

def grad_descent(h0, target, alpha_seq, lr, max_iters_per_alpha, rng, time_limit, t0):
    """Run gradient descent with sequence of alpha values."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    
    for alpha in alpha_seq:
        iters = 0
        for it in range(max_iters_per_alpha):
            if time.time() - t0 > time_limit:
                return best_h, best_val
            
            val, grad = c5_grad_fft(h, alpha)
            h_new = h - lr * grad
            h_new = project_feasible_exact(h_new, target)
            
            actual_val = compute_c5(h_new)
            if actual_val < best_val:
                best_val = actual_val
                best_h = h_new.copy()
            
            h = h_new
            iters += 1
        
    return best_h, best_val

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def project_feasible_exact(h, target):
    n = len(h)
    lo, hi = h.min() - 0.1, h.max() + 0.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_proj = np.clip(h - tau, 0.0, 1.0)
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        idx = np.where((h_proj > 1e-12) & (h_proj < 1.0 - 1e-12))[0]

def c5_val_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    exp_s = np.exp(shifted)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    val = (c_max + np.log(sum_exp) / alpha) * dx
    one_mh = 1.0 - h
    A = fftconvolve(one_mh, w, mode='full')[:n]
    M = 2 * m + 2
    h_ft = np.fft.rfft(h, M)
    w_ft = np.fft.rfft(w, M)
    r = np.fft.irfft(h_ft * np.conj(w_ft), M)
    B = r[:n]
    grad = (A - B) * dx
    return val, grad

                             rng, max(8.0, tleft() - 10), t0)
        if sc < best_score:
            best_score = sc
            best_h = h_sa
        # One more gradient pass
        if tleft() > 10:
            h_gd, sc2 = grad_descent(best_h, target_use, [1000.0, 3000.0], lr=1.0,
                                     max_iters=1000, time_limit=max(5.0, tleft() - 5), t0=t0)
            if sc2 < best_score:
                best_score = sc2
                best_h = h_gd

    n_final = len(best_h)
    target_final = n_final / 2.0
    h_final = project_feasible_exact(best_h, target_final)
    final_score = compute_c5(h_final)
    if final_score > best_score + 1e-10:
        h_final = project_feasible_simple(best_h, target_final)
        final_score = compute_c5(h_final)
    return h_final, final_score, n_final
