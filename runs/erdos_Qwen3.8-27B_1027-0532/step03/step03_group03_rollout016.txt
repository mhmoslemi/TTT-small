from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0: return h
        per = diff / len(idx)
        h[idx] = np.minimum(h[idx] + per, 1.0)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0: return h
        per = diff / len(idx)
        h[idx] = np.maximum(h[idx] + per, 0.0)
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    return (c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx

    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -50.0, 0.0)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    return compute_grad_v2(x, weights) * dx

_GLOBAL_TARGET = 0.0

    return np.sum(x) - _GLOBAL_TARGET

    return np.ones(_GLOBAL_N)

            if sc < best_score:

Here's a "constraint activation" algorithm:
1. Start with K=1 (just the maximum lag)
2. Solve the QP: minimize t s.t. corr_{k*}(h) ≤ t, 0≤h≤1, sum(h)=n/2
3. Check if any other lag now exceeds t
4. If yes, add it to the constraint set and repeat
5. If no, we're optimal (for this constraint set)

This is essentially the "active set" method for the min-max problem. Each iteration solves a QP with one more constraint.

The QP is: minimize t
subject to: Σ_i h[i]·(1-h[i+L_k]) ≤ t/dx for each active k
            Σ h[i] = n/2
            0 ≤ h[i] ≤ 1

The objective is linear (t), the first set of constraints are quadratic (in h), and the rest are linear. This is a QCQP, which SLSQP can handle.

            g[0] = 1.0
            return g

                    return j
