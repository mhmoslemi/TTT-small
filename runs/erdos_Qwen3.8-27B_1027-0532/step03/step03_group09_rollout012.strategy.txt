```python
def repair_binary(h, target_sum):
    h = np.array(h)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) <= 0.1:
        return h
    while current_sum != target_sum:
        if current_sum < target_sum:
            # Flip a 0 to 1
            indices = np.where(h == 0)[0]
            flip_pos = np.random.choice(indices)
            h[flip_pos] = 1
            current_sum += 1
        else:
            indices = np.where(h == 1)[0]
            flip_pos = np.random.choice(indices)
            h[flip_pos] = 0
            current_sum -= 1
    return h
```
