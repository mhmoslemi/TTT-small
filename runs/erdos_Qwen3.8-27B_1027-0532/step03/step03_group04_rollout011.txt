    lo, hi = h.min() - 1, h.max() + 1

        s = h_proj.sum()
        if s > target:

For coordinate i, the objective becomes a function of h[i] alone:
f(h[i]) = max_k c_k(h[i])

where c_k depends on h[i] linearly (through the terms involving h[i]).

Actually, c_k = sum_j h[j](1-h[j+k])*dx. The terms involving h[i] are:
- j = i: h[i]*(1-h[i+k])
- j+k = i (j = i-k): h[i-k]*(1-h[i])

So c_k = (terms not involving h[i]) + h[i]*(1-h[i+k]) + h[i-k]*(1-h[i])
       = (const) + h[i] - h[i]*h[i+k] + h[i-k] - h[i-k]*h[i]
       = (const') + h[i]*(1-h[i+k]) + h[i-k]*(1-h[i])

This is a quadratic in h[i]:
c_k = A_k + B_k * h[i] - C_k * h[i]²

where A_k, B_k, C_k depend on the other coordinates and the specific lag k.

Actually wait, it's:
c_k = (const) + h[i]*(1-h[i+k]) + h[i-k]*(1-h[i])

    = (const + h[i-k]) + h[i]*(1 - h[i+k] - h[i-k]) + 0*h[i]²

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

_n_global = 0
_target_global = 0.0


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx


def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_p = np.clip(h - mid, 0.0, 1.0)
        if h_p.sum() > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    res = target - h_p.sum()
    if abs(res) > 1e-12:
        free = (h_p > 1e-10) & (h_p < 1.0 - 1e-10)
        if free.any():
            h_p[free] += res / free.sum()
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p


def smooth_max_value(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_shift = c - c.max()
    return float(c.max() + np.log(np.sum(np.exp(alpha * c_shift))) / alpha)


def smooth_max_grad(h, alpha, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    c_shift = c - c.max()
    log_w = alpha * c_shift
    log_w_shifted = log_w - log_w.max()
    w = np.exp(log_w_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w /= w_sum

    if top_k < len(w):
        top_idx = np.argpartition(w, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(w))

    grad = np.zeros(n)
    for m in top_idx:
        wm = w[m]
        if wm < 1e-15:
            continue
        lag = m - (n - 1)
        wm_dx = wm * dx
        if lag >= 0:
            l = n - lag
            if l > 0:
                grad[:l] += wm_dx * (1.0 - h[lag:])
                grad[lag:] -= wm_dx * h[:l]
        else:
            blog = -lag
            l = n - blog
            if l > 0:
                grad[blog:] += wm_dx * (1.0 - h[:l])
                grad[:l] -= wm_dx * h[blog:]

    return grad


def nesterov_projected(h_start, n, target, alpha, max_iter, time_limit, lr_init=0.03, decay=0.003):
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    v = np.zeros(n)
    count_no_improve = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = smooth_max_grad(h, alpha, top_k=80)
        lr = lr_init / (1.0 + it * decay)
        v = 0.9 * v + lr * g
        h_new = project_feasible(h - v, target)

        c5 = compute_c5(h_new)
        if c5 < best_val - 1e-14:
            best_val = c5
            best_h = h_new.copy()
            count_no_improve = 0
        else:
            count_no_improve += 1

        h = h_new

        if count_no_improve > 300:
            v = np.zeros(n)
            count_no_improve = 0
            h = best_h.copy()

    return best_h, best_val


def _cons_fun(x):
    return np.sum(x) - _target_global


def _cons_jac(x):
    return np.ones(_n_global)


def _smooth_val_wrapper(x, alpha):
    return smooth_max_value(x, alpha)


def slsqp_optimize(h_start, n, target, alphas, maxiter=2000):
    global _n_global, _target_global
    _n_global = n
    _target_global = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    h_cur = h_start.copy()

    for alpha in alphas:
        try:
            res = minimize(_smooth_val_wrapper, h_cur, args=(alpha,),
                          method='SLSQP', bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_feasible(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass

    return h_cur


def penalty_lbfgs(h_start, n, target, alphas, time_limit, max_iter_per=5000):
    start = time.time()
    h = h_start.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    for mu in [1.0, 5.0, 20.0, 100.0, 500.0, 2000.0]:
        if time.time() - start > time_limit:
            break
        for alpha in alphas:
            if time.time() - start > time_limit:
                break

            def _obj(x):
                return smooth_max_value(x, alpha) + mu * (np.sum(x) - target) ** 2

            def _grad(x):
                g = smooth_max_grad(x, alpha, top_k=60)
                g += 2.0 * mu * (np.sum(x) - target) * np.ones(n)
                return g

            try:
                res = minimize(_obj, h, jac=_grad, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n,
                              options={'maxiter': max_iter_per, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h):
                    h = h_new
                if c5_new < best_val:
                    best_val = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    return best_h, best_val


def alternating_lp(h_init, n, target, max_iter, time_limit, damping_init=0.6, reg=1e-4):
    start = time.time()
    dx = 2.0 / n
    h = h_init.copy()
    h = project_feasible(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    damping = damping_init
    stuck_count = 0
    current_reg = reg

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        g = 1.0 - h

        h_var = cp.Variable(n)
        t_var = cp.Variable()

        constraints = [
            cp.sum(h_var) == target,
            h_var >= 0.0,
            h_var <= 1.0
        ]

        for k in range(-(n - 1), n):
            lo = max(0, k)
            hi = min(n, k + n)
            if hi > lo:
                coeff = np.zeros(n)
                coeff[lo:hi] = g[lo - k:hi - k] * dx
                constraints.append(coeff @ h_var <= t_var)

        objective = t_var + current_reg * cp.sum_squares(h_var - 0.5)
        prob = cp.Problem(cp.Minimize(objective), constraints)

        try:
            prob.solve(solver=cp.SCS, max_iters=50000, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iter=50000)
            except Exception:
                break

        if h_var.value is None:
            break

        h_new = np.clip(np.asarray(h_var.value).ravel(), 0.0, 1.0)
        h = (1.0 - damping) * h + damping * h_new
        h = project_feasible(h, target)

        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        if stuck_count > 30:
            damping = max(0.05, damping * 0.8)
            stuck_count = 0

        if it % 100 == 0 and current_reg > 1e-6:
            current_reg = max(1e-6, current_reg * 0.6)

    return best_h, best_c5


def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def subgradient_improve(h, n, target, max_steps, time_limit):
    start = time.time()
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    n_half = n - 1
    v = np.zeros(n)

    for step in range(max_steps):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - n_half

        grad = np.zeros(n)
        if lag >= 0:
            l = n - lag
            if l > 0:
                grad[:l] += (1.0 - h_cur[lag:]) * dx
                grad[lag:] -= h_cur[:l] * dx
        else:
            blog = -lag
            l = n - blog
            if l > 0:
                grad[blog:] += (1.0 - h_cur[:l]) * dx
                grad[:l] -= h_cur[blog:] * dx

        lr = 0.05 / np.sqrt(1.0 + step * 0.01)
        v = 0.9 * v + lr * grad
        h_try = project_feasible(h_cur - v, target)
        sc = compute_c5(h_try)
        if sc < best_c5 - 1e-14:
            best_c5 = sc
            best_h = h_try.copy()
            h_cur = h_try

    return best_h, best_c5
