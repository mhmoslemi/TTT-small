- t = 0 or t = 1 (boundary)
- Two (or more) of the linear functions are equal and all others are below (a "kink" where the subgradient contains 0)

The subgradient of max_k (A_k * t + B_k) at point t is [min_k A_k * I[k active], max_k A_k * I[k active]] where "active" means the k that achieves the max.

The minimum is where 0 is in the subgradient, i.e., where there exist active k's with A_k >= 0 and A_k <= 0 (or a single active k with A_k = 0).

So the algorithm for one coordinate is:
1. Compute all (A_k, B_k) pairs
2. Sort by A_k
3. For each possible "cut" in the sorted list, check if the max of the upper group equals the max of the lower group at the intersection point
4. Also check t=0 and t=1

This is O(n log n) per coordinate (dominated by sorting), which is fine.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        for _ in range(50):
            diff = target_sum - np.sum(h)
            if abs(diff) < 1e-14:
                break
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), 1.0 - h[mask].max())
            if step <= 0:
                break
            h[mask] += step
    else:
        for _ in range(50):
            diff = np.sum(h) - target_sum
            if abs(diff) < 1e-14:
                break
            mask = h > 1e-14
            if not np.any(mask):
                break
            step = min(diff / np.sum(mask), h[mask].min())
            if step <= 0:
                break
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def compute_grad_weighted(h, lags, weights):
    n = len(h)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-15:
            continue
        lag = lags[i] - (n - 1)
        if lag >= 0:
            if lag < n:
                grad[lag:] += w * omh[:n - lag]
                grad[:n - lag] -= w * h[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += w * omh[m:]
                grad[m:] -= w * h[:n - m]
    return grad

def coord_descent_step(h, j, target_sum):

def coord_descent_sweep(h, target_sum, n_sweeps=3, start_idx=0, n_coords=None):

def lse_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def lse_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    if total < 1e-300:
        weights = np.ones(len(corr)) / len(corr)
    else:
        weights = exp_shifted / total
    return compute_grad_weighted(x, np.arange(len(corr)), weights)

def slsqp_refine(h_start, target_sum, alphas, maxiter=1000):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(n)}]
    h_cur = h_start.copy()
    for alpha in alphas:
        try:
            res = minimize(lse_obj, h_cur, args=(alpha,),
                           jac=lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': 1e-16})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur
