import time
import numpy as np
from scipy.optimize import minimize

# Global variables for constraint functions
_g_target_sum = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _eq_con_fun(x):
    return x.sum() - _g_target_sum

def _eq_con_jac(x):
    return np.ones(_g_n)

def _lse(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _nesterov_exact(h0, n, target_sum, n_iters, alpha0):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        k_star = int(np.argmax(corr))
        g = _grad_corr_term(y, k_star, dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _nesterov_multi(h0, n, target_sum, n_iters, alpha0, top_m):

    x = h0.copy()
    y = h0.copy()
    t = 1.0

    best_val = _c5_of(x)

        max_corr = np.max(corr)

        total_w = weights.sum()
        if total_w < 1e-300:

    _g_target_sum = target_sum
    _g_n = n

def _coord_line_search(h0, n, target_sum, max_rounds=10, max_iters_ls=30):

import numpy as np

_g_target_sum = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_corr_term(h, k_idx, dx):

    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

    except Exception:
        return h0

def _nesterov_multi(h0, n, target_sum, n_iters, alpha0, top_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(100.0 * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:

        top_indices = np.argsort(weights)[::-1][:top_m]

        step = alpha0 / np.sqrt(float(it + 1))

        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:

            best = x.copy()
    return best, best_val

        # We want: f_k(h) + ∇f_k · (h_new - h) ≤ t
        # ∇f_k · h_new - t ≤ f_k(h) - ∇f_k · h
        # The RHS is the Taylor remainder, which for a bilinear function is 0 (since we're linearizing a bilinear function, the linearization is exact in the sense that f_k(h + δ) = f_k(h) + ∇f_k · δ + (quadratic terms in δ))
        
        # Actually for f_k(h) = ∑ h[i]+h[i+k] - ∑ h[i]h[i+k], the gradient is:
        # ∂f_k/∂h[j] = (contribution from linear terms) - (contribution from quadratic terms)
        # The linearization is: f_k(h) + ∇f_k(h) · δ
        # Constraint: f_k(h) + ∇f_k(h) · (h_new - h) ≤ t
        # => ∇f_k(h) · h_new - t ≤ f_k(h) - ∇f_k(h) · h
        
        rhs = corr[k_idx] - np.dot(_grad_corr_term(h, k_idx, dx) / dx, h)
        # Hmm, this is getting complicated. Let me just use the direct formula.
        
        # f_k(h) = corr[k_idx] * dx (since corr is the sum, and we multiply by dx for the integral)
        # Actually, corr[k_idx] is the sum ∑ h[i+k]*(1-h[i]), and the integral approx is corr[k_idx] * dx
        # So f_k = corr[k_idx] * dx
        # ∇f_k[j] = * dx * (the discrete gradient)
        # _grad_corr_term returns the gradient * dx already
        
        # Let me just directly compute:
        # f_k(h_new) ≈ corr_k + ∇corr_k · (h_new - h)
        # where ∇corr_k[j] is the partial derivative of corr_k w.r.t. h[j]
        # Constraint: corr_k + ∇corr_k · (h_new - h) * dx ≤ t
        # i.e., ∇corr_k · h_new * dx - t ≤ t - corr_k * dx - ∇corr_k · h * dx + corr_k * dx
        # = (corr_k - ∇corr_k · h) * dx
        
        # This is getting messy. Let me just compute the linear constraint directly.
        
        g_k = _grad_corr_term(h, k_idx, dx)  # This is dx * ∂(corr_k)/∂h, so it's ∂f_k/∂h
        # f_k = corr_k * dx, so ∂f_k/∂h[j] = g_k[j]
        # Linear constraint: f_k(h) + g_k · (h_new - h) ≤ t
        # g_k · h_new - t ≤ t - f_k(h) - g_k · h + f_k(h) = t - g_k · h
        # Wait: f_k(h) + g_k · h_new - g_k · h ≤ t
        # g_k · h_new - t ≤ t - g_k · h
        # The RHS should be: f_k(h) - g_k · h = corr_k * dx - g_k · h
        
        rhs_i = corr[k_idx] * dx - np.dot(g_k, h)
        A_ub[i, :n] = g_k
        A_ub[i, n] = -1.0
        b_ub[i] = rhs_i
    
    # Box constraints: h ≤ 1
    for j in range(n):
        A_ub[M + j, j] = 1.0
        b_ub[M + j] = 1.0
    # -h ≤ 0
    for j in range(n):
        A_ub[M + n + j, j] = -1.0
        b_ub[M + n + j] = 0.0
    
    # Equality: sum(h) = target
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    
    bounds_h = [(0.0, 1.0)] * n + [(0.0, None)]
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds_h, method='highs')
    
    if res.success:
        h_new = res.x[:n]
        h_new = _fix_sum(h_new, target_sum)
        return h_new
    return h

import time
import numpy as np
from scipy.optimize import minimize, linprog

_g_target_sum = 0.0
_g_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _corr_of(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _eq_con_fun(x):
    return x.sum() - _g_target_sum

def _eq_con_jac(x):
    return np.ones(_g_n)

def _lse(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    shifted = alpha * (x - m)
    mask = shifted > -700.0
    if not np.any(mask):
        return m
    return m + np.log(np.sum(np.exp(shifted[mask])) + (len(x) - np.sum(mask)) * 0.0) / alpha

def _smooth_obj(h, alpha, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return _lse(corr, alpha) * (2.0 / n)

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _slsqp_opt(h0, n, target_sum, alpha, maxiter, ftol=1e-14):
    global _g_target_sum, _g_n
    _g_target_sum = target_sum
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _nesterov_multi(h0, n, target_sum, n_iters, alpha0, top_m=5):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(200.0 * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        else:
            weights = weights / total_w
        top_indices = np.argsort(weights)[::-1][:top_m]
        w_m = weights[top_indices].copy()
        w_m = w_m / w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_indices)):
            g += w_m[idx] * _grad_corr_term(y, int(top_indices[idx]), dx)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        mod = i % 5
        if mod == 1:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif mod == 2:
            pop[i] = _fix_sum(1.0 - h0 + 0.1 * (rng.rand(n) - 0.5), target_sum)
        elif mod == 3:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif mod == 4:
            shuffled = h0[rng.permutation(n)]
            pop[i] = _fix_sum(shuffled, target_sum)
        else:
            pop[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = float(pop_vals[best_idx])
    best_h = pop[best_idx].copy()
    stall = 0
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (gen / max_gen)
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            while r1 == i:
                r1 = rng.randint(pop_size)
            while r2 == i or r2 == r1:
                r2 = rng.randint(pop_size)
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        if stall > 2 * pop_size:
            worst = np.argsort(pop_vals)[pop_size // 2:]
            for idx in worst:
                t_mix = rng.uniform(0.1, 0.9)
                pop[idx] = _fix_sum(t_mix * best_h + (1.0 - t_mix) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = float(pop_vals[idx])
                    best_h = pop[idx].copy()
                    best_idx = int(idx)
            stall = 0
    return best_h, best_val

def _sca_step(h, n, target_sum, top_m=5):
    dx = 2.0 / n
    corr = _corr_of(h)
    top_indices = np.argsort(corr)[::-1][:top_m]
    nv = n + 1
    c = np.zeros(nv)
    c[n] = 1.0
    A_ub = np.zeros((top_m + 2 * n, nv))
    b_ub = np.zeros(top_m + 2 * n)
    for i, k_idx in enumerate(top_indices):
        k_idx = int(k_idx)
        g_k = _grad_corr_term(h, k_idx, dx)
        rhs = corr[k_idx] * dx - np.dot(g_k, h)
        A_ub[i, :n] = g_k
        A_ub[i, n] = -1.0
        b_ub[i] = rhs
    for j in range(n):
        A_ub[top_m + j, j] = 1.0
        b_ub[top_m + j] = 1.0
        A_ub[top_m + n + j, j] = -1.0
        b_ub[top_m + n + j] = 0.0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_lp, method='highs')
        if res.success:
            h_new = _fix_sum(res.x[:n], target_sum)
            return h_new
    except Exception:
        pass
    return h

def _sca_loop(h0, n, target_sum, max_iters=200, top_m=5, deadline=None):
    h = h0.copy()
    c5 = _c5_of(h)
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        h_new = _sca_step(h, n, target_sum, top_m)
        c5_new = _c5_of(h_new)
        if c5_new < c5 - 1e-12:
            c5 = c5_new
            h = h_new
        else:
            top_m2 = min(top_m + 3, 15)
            h_new2 = _sca_step(h, n, target_sum, top_m2)
            c5_new2 = _c5_of(h_new2)
            if c5_new2 < c5 - 1e-12:
                c5 = c5_new2
                h = h_new2
            else:
                break
    return h, c5

def _adaptive_polish(h0, n, target_sum, deadline):
    h = h0.copy()
    c5 = _c5_of(h)
    alphas = [50, 100, 200, 400, 800, 1500, 3000, 6000, 12000, 20000]
    for a in alphas:
        if time.time() > deadline:

        h_new = _slsqp_opt(h, n, target_sum, a, 500, ftol=1e-14)
        c5_new = _c5_of(h_new)
        if c5_new < c5:

            h = h_new
        if time.time() > deadline:
            break
        h_nes, c5_nes = _nesterov_multi(h, n, target_sum, 50, 0.3, top_m=5)
        if c5_nes < c5:
            c5 = c5_nes
            h = h_nes
    return h, c5

def _perturb_reoptimize(h0, n, target_sum, deadline, rng):
    h_best = h0.copy()
    c5_best = _c5_of(h0)
    for attempt in range(6):
        if time.time() > deadline:
            break
        sigma = 0.05 / (attempt + 1)
        h_pert = h0 + sigma * (rng.rand(n) - 0.5)
        h_pert = _fix_sum(np.clip(h_pert, 0.0, 1.0), target_sum)
        h_opt = _slsqp_opt(h_pert, n, target_sum, 500, 300)
        c5_opt = _c5_of(h_opt)
        if c5_opt < c5_best:
            c5_best = c5_opt
            h_best = h_opt.copy()
    return h_best, c5_best
