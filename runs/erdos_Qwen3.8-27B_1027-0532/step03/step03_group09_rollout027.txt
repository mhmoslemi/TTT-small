import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def logsumexp_all(h, eps):
    corr = compute_corr(h)
    m = corr.max()
    return m + eps * np.log(np.sum(np.exp((corr - m) / eps)))

def logsumexp_all_grad(h, eps):

    m = corr.max()

    w = w / w.sum()

    for k_idx in range(2 * n - 1):
        wk = w[k_idx]

        if k >= 0:

        else:

def slsqp_lse(h_init, eps, maxiter, top_k=0):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def obj_fun(x):
        if top_k > 0:
            return focused_max(x, top_k, eps)
        else:
            return logsumexp_all(x, eps)

    def obj_grad(x):
        if top_k > 0:
            return focused_max_grad(x, top_k, eps)
        else:
            return logsumexp_all_grad(x, eps)

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(obj_fun, h0, jac=obj_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def proj_subgrad_optimize(h_init, top_k, eps, lr_init, n_steps, time_limit, start_time):

    target_sum = n / 2.0
    h_cur = h_init.copy()
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    lr = lr_init
    stall = 0
    stall_limit = 300

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        corr = compute_corr(h_cur)
        idx = np.argsort(corr)[-top_k:]
        vals = corr[idx]
        m = vals.max()
        w = np.exp((vals - m) / eps)
        w = w / w.sum()

        dx = 2.0 / n
        grad = np.zeros(n)
        for i in range(len(idx)):
            if w[i] < 1e-15:
                continue
            k = idx[i] - (n - 1)
            if k >= 0:
                grad[k:n] += w[i] * dx * (1.0 - h_cur[0:n - k])
                grad[0:n - k] -= w[i] * dx * h_cur[k:n]
            else:
                r = -k
                grad[0:n - r] += w[i] * dx * (1.0 - h_cur[r:n])
                grad[r:n] -= w[i] * dx * h_cur[0:n - r]

        c5_new = compute_c5(h_new)

            stall = 0

def gen_random_start(n, rng, target_sum):
    strategy = rng.integers(0, 5)
    if strategy == 0:
        h = (rng.random(n) < 0.5).astype(float)
    elif strategy == 1:
        h = 0.5 + 0.4 * rng.standard_normal(n)
    elif strategy == 2:
        n_blocks = rng.integers(3, 8)
        h = np.zeros(n)
        block_size = n // n_blocks
        for b in range(n_blocks):
            if rng.random() < 0.5:
                h[b * block_size:(b + 1) * block_size] = 1.0
    elif strategy == 3:
        period = rng.integers(3, 9)
        pattern = (rng.random(period) < 0.5).astype(float)
        h = np.tile(pattern, n // period + 1)[:n]
    else:
        h = np.sin(np.linspace(0, 4 * np.pi, n, rng=rng)) * 0.3 + 0.5
    return project_onto_constraint(h, target_sum)

def blend_crossover(h1, h2, rng):
    n = len(h1)
    alpha = rng.uniform(0.3, 0.7)
    return alpha * h1 + (1 - alpha) * h2

def mutate(h, rng, sigma):
    return h + sigma * rng.standard_normal(len(h))

def ga_optimize(h_init, n_pop, n_gen, sigma_init, time_limit, start_time, rng):
    n = len(h_init)
    target_sum = n / 2.0
    population = [h_init.copy()]
    for _ in range(n_pop - 1):
        pop_member = gen_random_start(n, rng, target_sum)
        population.append(pop_member)

    fitness = [compute_c5(p) for p in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    sigma = sigma_init
    stall = 0

    for gen in range(n_gen):
        if time.time() - start_time > time_limit:
            break

        # Elitism: keep top 2
        sorted_idx = np.argsort(fitness)
        new_pop = [population[sorted_idx[0]].copy(), population[sorted_idx[1]].copy()]
        new_fit = [fitness[sorted_idx[0]], fitness[sorted_idx[1]]]

        # Generate offspring
        while len(new_pop) < n_pop:
            # Tournament selection
            t_size = min(5, len(population))
            t_indices = rng.integers(0, len(population), size=t_size)
            t_fitness = [fitness[i] for i in t_indices]
            winner1 = population[t_indices[np.argmin(t_fitness)]]

            t_indices2 = rng.integers(0, len(population), size=t_size)
            t_fitness2 = [fitness[i] for i in t_indices2]
            winner2 = population[t_indices2[np.argmin(t_fitness2)]]

            # Crossover

                child = winner1.copy()

            child = project_onto_constraint(child, target_sum)

            new_fit.append(compute_c5(child))

        fitness = new_fit

        gen_best_idx = np.argmin(fitness)
        if fitness[gen_best_idx] < best_c5:
            best_c5 = fitness[gen_best_idx]
            best_h = population[gen_best_idx].copy()
            stall = 0

            h_pg_up, c5_pg_up = proj_subgrad_optimize(h_opt_up, 3, 0.003, 0.04, 100000, tl_up, start_time)
            if c5_pg_up < c5_opt_up:
                c5_opt_up = c5_pg_up
                h_opt_up = h_pg_up

        if c5_opt_up < best_c5:
            best_c5 = c5_opt_up
            best_h = h_opt_up.copy()
            n_best = n1

        # Second resolution increase
        if rem() > 150 and n_best < 300:
            n2 = n_best + 30
            old_t2 = np.linspace(0.0, 1.0, n_best)
            new_t2 = np.linspace(0.0, 1.0, n2)
            h_up2 = np.interp(new_t2, old_t2, best_h)
            h_up2 = project_onto_constraint(h_up2, n2 / 2.0)
            h_up2 = project_onto_constraint(h_up2 + 0.003 * rng.standard_normal(n2), n2 / 2.0)

            h_opt_up2, c5_opt_up2 = slsqp_lse(h_up2, 0.003, 250, top_k=5)
            if rem() > 60:
                h_opt_up3, c5_opt_up3 = slsqp_lse(h_opt_up2, 0.001, 250, top_k=2)
                if c5_opt_up3 < c5_opt_up2:
                    c5_opt_up2 = c5_opt_up3
                    h_opt_up2 = h_opt_up3

            if c5_opt_up2 < best_c5:
                best_c5 = c5_opt_up2
                best_h = h_opt_up2.copy()
                n_best = n2

    # Phase 11: Final polish
    if rem() > 40:
        h_opt, c5_opt = slsqp_lse(best_h, 0.0003, 300, top_k=2)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    if rem() > 30:
        tl = min(rem() - 10, 60)
        h_pg, c5_pg = proj_subgrad_optimize(best_h, 1, 0.001, 0.02, 200000, tl, start_time)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

    # Final projection and evaluation
    final_n = len(best_h)
    final_h = project_onto_constraint(best_h, final_n / 2.0)
    final_c5 = compute_c5(final_h)

    return final_h, final_c5, final_n
