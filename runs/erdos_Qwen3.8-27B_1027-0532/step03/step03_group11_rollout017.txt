import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: float(np.sum(h) - n/2.0),
             'jac': lambda h: np.ones(n)}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def subgradient_optimize(h_init, n_iters, time_limit, start_time, initial_lr=0.05):
    n = len(h_init)
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = initial_lr
    omh = 1.0 - h
    
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, omh, mode='full') * dx
        cmax = np.max(corr)
        
        # Active shifts: within 15% of max
        threshold = cmax - 0.05
        active = np.where(corr >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Compute gradient (average over active)
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                m = n - s
                grad[s:n] += dx * omh[:m]
                grad[:m] -= dx * h[s:n]
            else:
                r = -s
                m = n - r
                grad[:m] += dx * omh[r:n]
                grad[r:n] -= dx * h[:m]
        grad /= len(active)
        
        # Step
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        omh = 1.0 - h
        lr *= 0.9995
    
    return best_h, best_c5

def make_binary_from_continuous(h, rng):
    n = len(h)
    diffs = np.abs(h - 0.5)
    order = np.argsort(diffs)
    result = np.zeros(n)
    result[order[n // 2:]] = 1.0
    return result

def tabu_binary_search(h_init, time_limit, start_time, tabu_size=100):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_c5 = best_c5
    
    tabu_list = {}  # position -> expiry iteration
    it = 0
    stagnant = 0
    
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    
    while time.time() - start_time < time_limit:
        it += 1
        
        # Choose move type
        r = np.random.random()
        if r < 0.7:
            # Single swap
            candidates_ones = [oi for oi in ones if tabu_list.get(oi, -1) < it]
            candidates_zeros = [zi for zi in zeros if tabu_list.get(zi, -1) < it]
            if not candidates_ones or not candidates_zeros:
                continue
            oi = candidates_ones[np.random.randint(len(candidates_ones))]
            zi = candidates_zeros[np.random.randint(len(candidates_zeros))]
            h[oi] = 0.0
            h[zi] = 1.0
            tabu_list[oi] = it + tabu_size
            tabu_list[zi] = it + tabu_size
            new_c5 = compute_c5(h)
            accepted = new_c5 < current_c5 or (new_c5 < best_c5 and np.random.random() < 0.3)
            if not accepted:
                h[oi] = 1.0
                h[zi] = 0.0
                del tabu_list[oi]
                del tabu_list[zi]
                stagnant += 1
            else:
                if not (new_c5 < current_c5):
                    stagnant += 1
                current_c5 = new_c5
                ones.remove(oi)
                zeros.remove(zi)
                ones.append(zi)
                zeros.append(oi)
        elif r < 0.85:
            # Block swap (length 2-5)
            blen = int(np.random.randint(2, 6))
            a = int(np.random.randint(0, n - blen))
            b = int(np.random.randint(0, n - blen))
            if abs(a - b) < blen:
                continue
            block_a = h[a:a+blen].copy()
            block_b = h[b:b+blen].copy()
            h[a:a+blen] = block_b
            h[b:b+blen] = block_a
            new_c5 = compute_c5(h)
            if new_c5 >= current_c5 and new_c5 >= best_c5:
                h[a:a+blen] = block_a
                h[b:b+blen] = block_b
                stagnant += 1
            else:
                current_c5 = new_c5
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
        else:
            # Invert a block
            blen = int(np.random.randint(2, 8))
            a = int(np.random.randint(0, n - blen))
            h[a:a+blen] = 1.0 - h[a:a+blen]
            new_c5 = compute_c5(h)
            if new_c5 >= current_c5 and new_c5 >= best_c5:
                h[a:a+blen] = 1.0 - h[a:a+blen]
                stagnant += 1
            else:
                current_c5 = new_c5
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnant = 0
        
        # Restart if too stagnant
        if stagnant > 200:
            h = best_h.copy()
            nflip = int(np.random.randint(2, max(3, n // 20)))
            flip_idx = np.random.choice(n, nflip, replace=False)
            h[flip_idx] = 1.0 - h[flip_idx]
            # Fix sum
            n_ones = int(np.sum(h))
            target = n // 2
            diff = target - n_ones
            if diff > 0:
                z = np.where(h == 0.0)[0]
                to_flip = np.random.choice(z, min(diff, len(z)), replace=False)
                h[to_flip] = 1.0
            elif diff < 0:
                o = np.where(h == 1.0)[0]
                to_flip = np.random.choice(o, min(-diff, len(o)), replace=False)
                h[to_flip] = 0.0
            current_c5 = compute_c5(h)
            stagnant = 0
            tabu_list = {}
            ones = list(np.where(h == 1.0)[0])
            zeros = list(np.where(h == 0.0)[0])
        
        # Clean tabu
        if it % tabu_size == 0:
            tabu_list = {k: v for k, v in tabu_list.items() if v >= it}
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        x = np.linspace(0, 2, n0, endpoint=False)
        h0 = np.clip(0.5 + 0.4 * np.sin(2 * np.pi * 5 * x / 2.0), 0, 1)
    
    n0 = len(h0)
    h0 = project_onto_constraint(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Phase 1: SLSQP coarse
    for eps in [0.02, 0.01, 0.005]:
        if time.time() - start_time > budget_s * 0.15:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, eps, 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: SLSQP fine
    for eps in [0.002, 0.001, 0.0005, 0.0002]:
        if time.time() - start_time > budget_s * 0.3:
            break
        h_opt, c5_opt = slsqp_optimize(best_h, eps, 300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 3: Subgradient refinement
    if time.time() - start_time < budget_s * 0.4:
        h_sg, c5_sg = subgradient_optimize(best_h, 3000, budget_s * 0.45, start_time)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg
        # One more SLSQP after subgradient
        if time.time() - start_time < budget_s * 0.48:
            h_opt, c5_opt = slsqp_optimize(best_h, 0.001, 200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Phase 4: Binary search
    if time.time() - start_time < budget_s * 0.5:
        h_bin = make_binary_from_continuous(best_h, rng)
        h_bin, c5_bin = tabu_binary_search(h_bin, budget_s * 0.75, start_time)
        # Use binary result to inform continuous
        h_from_bin = h_bin.astype(float)
        # Blend with current best

        h_blend = (1 - alpha) * best_h + alpha * h_from_bin
        h_blend = project_onto_constraint(h_blend, n0 / 2.0)
