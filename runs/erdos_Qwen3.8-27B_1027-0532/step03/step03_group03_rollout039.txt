import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h.copy()
    if diff > 0:
        idx = np.where(h < 1.0)[0]
        if len(idx) == 0:
            return h.copy()
        h = h.copy()
        h[idx] += diff / len(idx)
    else:
        idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            return h.copy()
        h = h.copy()
        h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

def overlap_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def two_coord_step(h, j, m, corr):
    n = len(h)
    lags = np.arange(-(n - 1), n)
    A = np.zeros(2 * n - 1)
    jk = j - lags
    jkp = j + lags
    mk = m - lags
    mkp = m + lags
    mask1 = (jk >= 0) & (jk < n)
    mask2 = (jkp >= 0) & (jkp < n)
    mask3 = (mk >= 0) & (mk < n)
    mask4 = (mkp >= 0) & (mkp < n)
    if np.any(mask1):
        A[mask1] += (1.0 - h[jk[mask1]])
    if np.any(mask2):
        A[mask2] -= h[jkp[mask2]]
    if np.any(mask3):
        A[mask3] -= (1.0 - h[mk[mask3]])
    if np.any(mask4):
        A[mask4] += h[mkp[mask4]]
    delta_lo = max(-h[j], h[m] - 1.0)
    delta_hi = min(1.0 - h[j], h[m])
    if delta_hi - delta_lo < 1e-14:
        return h, corr, False
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = delta_lo, delta_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = float(np.max(corr + A * c_pt))
    fd = float(np.max(corr + A * d_pt))
    for _ in range(45):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = float(np.max(corr + A * c_pt))
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = float(np.max(corr + A * d_pt))
    delta_best = (a + b) / 2.0
    if abs(delta_best) < 1e-13:
        return h, corr, False
    h_new = h.copy()
    h_new[j] = np.clip(h[j] + delta_best, 0.0, 1.0)
    h_new[m] = np.clip(h[m] - delta_best, 0.0, 1.0)
    corr_new = corr + A * delta_best
    return h_new, corr_new, True

def two_coord_descent(h, max_sweeps, seed, elapsed_fn, time_limit):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    best_score = compute_c5(h)
    corr = overlap_corrs(h)
    for sweep in range(max_sweeps):
        if elapsed_fn() > time_limit:
            break
        improved = False
        step = 2
        for j in range(0, n - 1, step):
            m = j + 1
            h_new, corr_new, changed = two_coord_step(h, j, m, corr)
            if changed:
                sc = float(np.max(corr_new)) * (2.0 / n)
                if sc < best_score - 1e-13:
                    h = h_new
                    corr = corr_new
                    best_score = sc
                    improved = True
        for j in range(1, n - 1, step):
            m = j + 1
            h_new, corr_new, changed = two_coord_step(h, j, m, corr)
            if changed:
                sc = float(np.max(corr_new)) * (2.0 / n)
                if sc < best_score - 1e-13:
                    h = h_new
                    corr = corr_new
                    best_score = sc
                    improved = True
        if not improved:
            for _ in range(n // 2):
                if elapsed_fn() > time_limit:
                    break
                j, m = rng.choice(n, 2, replace=False)
                h_new, corr_new, changed = two_coord_step(h, int(j), int(m), corr)
                if changed:
                    sc = float(np.max(corr_new)) * (2.0 / n)
                    if sc < best_score - 1e-13:
                        h = h_new
                        corr = corr_new
                        best_score = sc
                        improved = True
        if not improved:
            break
    h = project_feasible(h, target)
    return h, compute_c5(h)

def subgradient_optimize(h_init, target_sum, max_iter, lr_init, topk, alpha,
                         noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0
    dx = 2.0 / n
    for it in range(max_iter):
        corr = overlap_corrs(h)
        top_indices = np.argpartition(corr, -topk)[-topk:]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        lags_arr = top_indices - (n - 1)
        grad = np.zeros(n)
        for i in range(topk):
            wi = w[i]
            if wi < 1e-14:
                continue
            lag = lags_arr[i]
            if lag >= 0:
                if lag < n:
                    grad[lag:] += wi * (1.0 - h[:n - lag])
                    grad[:n - lag] -= wi * h[lag:]
            else:
                m = -lag
                if m < n:
                    grad[:n - m] += wi * (1.0 - h[m:])
                    grad[m:] -= wi * h[:n - m]
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = float(np.max(overlap_corrs(h_new))) * dx
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.01, 3.0)
        else:
            stagnation += 1
            lr *= 0.985
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score

_GLOBAL_N = 0
_GLOBAL_TARGET = 0.0

def _slsqp_eq_fun(x):
    return np.sum(x) - _GLOBAL_TARGET

def _slsqp_eq_jac(x):
    return np.ones(_GLOBAL_N)

def _slsqp_obj(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    shifted = np.clip(alpha * (corr - c_max), -500.0, 0.0)
    return c_max + np.log(np.sum(np.exp(shifted))) / alpha

def _slsqp_grad(x, alpha):
    corr = overlap_corrs(x)
    c_max = np.max(corr)
    exp_shifted = np.exp(np.clip(alpha * (corr - c_max), -500.0, 0.0))
    total = np.sum(exp_shifted)
    weights = exp_shifted / total
    n = len(x)
    lags_all = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(len(weights)):
        wi = weights[i]
        if wi < 1e-14:
            continue
        lag = lags_all[i]
        if lag >= 0:
            if lag < n:
                grad[lag:] += wi * (1.0 - x[:n - lag])
                grad[:n - lag] -= wi * x[lag:]
        else:
            m = -lag
            if m < n:
                grad[:n - m] += wi * (1.0 - x[m:])
                grad[m:] -= wi * x[:n - m]
    return grad

def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _GLOBAL_N, _GLOBAL_TARGET
    _GLOBAL_N = len(h_start)
    _GLOBAL_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _GLOBAL_N
    cons = [{'type': 'eq', 'fun': _slsqp_eq_fun, 'jac': _slsqp_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(_slsqp_obj, h_cur, args=(alpha,),
                           jac=_slsqp_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur

def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
