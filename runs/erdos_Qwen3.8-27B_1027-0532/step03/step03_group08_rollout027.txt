import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x : 0 <= x <= 1, sum(x) = S}"""
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any rounding issues
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _pso_optimize(h0, n, target_sum, pop_size, max_iter, deadline, rng):
    cx = 2.0 / n
    # Initialize particles
    X = np.zeros((pop_size, n))
    V = np.zeros((pop_size, n))
    X[0] = h0.copy()
    for i in range(1, pop_size):
        X[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
    
    F = np.array([_c5_of(X[i]) for i in range(pop_size)])
    P = X.copy()  # personal best
    PF = F.copy()
    gbest_idx = int(np.argmin(F))
    GF = F[gbest_idx]
    
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        w = w_start + (w_end - w_start) * (it / max_iter)
        for i in range(pop_size):
            r1 = rng.rand(n)
            r2 = rng.rand(n)
            V[i] = w * V[i] + c1 * r1 * (P[i] - X[i]) + c2 * r2 * (X[gbest_idx] - X[i])
            # Clamp velocity
            V[i] = np.clip(V[i], -0.1, 0.1)
            X[i] = X[i] + V[i]
            X[i] = _project_box_sum(X[i], target_sum)
            f = _c5_of(X[i])
            if f < PF[i]:
                PF[i] = f
                P[i] = X[i].copy()
            if f < GF:
                GF = f
                gbest_idx = i
        # Reseed worst particles if stuck
        if it % 50 == 0 and it > 0:

            X[worst] = _fix_sum(P[gbest_idx] + 0.3 * (rng.rand(n) - 0.5), target_sum)
            PF[worst] = _c5_of(X[worst])

def _coord_descent_continuous(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    current_c5 = _c5_of(h)
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            # Try to find best value for h[j] given rest fixed
            # The C5 as a function of h[j] involves multiple correlation terms
            # where j appears
            # We can do a 1D search
            def obj_1d(v):
                h[j] = v
                return _c5_of(h)
            # Use golden section search
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-4})
            h[j] = res.x
            new_c5 = _c5_of(h)
            if new_c5 < current_c5 - 1e-10:
                current_c5 = new_c5
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h, current_c5

                current_c5 = new_c5
                improved = True
            else:
                h[j] = old
        if not improved:
            break
    return h, current_c5

def _slsqp_lse(h0, n, target_sum, alpha, maxiter, deadline):
    global _g_ts, _g_n
    _g_ts = target_sum
    _g_n = n
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smoothed_obj, h0, args=(alpha, n),
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _eq_fun(x):
    return x.sum() - _g_ts

def _eq_jac(x):
    return np.ones(_g_n)

def _multi_resolution(h_best, n_best, target_best, deadline, rng):
    global _g_ts, _g_n
    best_h = h_best.copy()
    best_c5 = _c5_of(h_best)
    best_n = n_best
    n_candidates = set()
    for d in [-20, -10, -5, 0, 5, 10, 20]:
        nc = n_best + d
        if nc % 2 == 0 and 80 <= nc <= 250:
            n_candidates.add(nc)
    for nc in sorted(n_candidates):
        if time.time() > deadline - 30:
            break
        ts = nc / 2.0
        if nc == n_best:
            h_start = h_best.copy()
        else:
            h_start = np.interp(np.linspace(0, n_best - 1, nc), np.arange(n_best), h_best)
            h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), ts)
        if _c5_of(h_start) > best_c5 + 0.01:
            continue
        h_sub, c5_sub = _subgradient_method(h_start, nc, ts, 80, deadline - 60, rng)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()
            best_n = nc
        if time.time() < deadline - 40:
            h_cd, c5_cd = _coord_descent_cont(h_sub, nc, ts, 3, deadline - 20)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                best_n = nc
    return best_h, best_c5, best_n

def _fourier_init(n, K, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    for k in range(1, K + 1):
        a_k = rng.uniform(-0.15, 0.15)
        b_k = rng.uniform(-0.15, 0.15)
        h += a_k * np.cos(np.pi * k * x) + b_k * np.sin(np.pi * k * x)
    return _fix_sum(h, n / 2.0)

def _nesterov_smooth(h0, n, target_sum, n_iters, alpha0, sg_m, sg_alpha, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        corr = _corr_full(y)
        max_corr = np.max(corr)
        weights = np.exp(sg_alpha * (corr - max_corr))
        total_w = weights.sum()
        if total_w < 1e-300:
            weights = np.zeros(len(corr))
            weights[int(np.argmax(corr))] = 1.0
        else:
            weights = weights / total_w
        top_m = np.argsort(weights)[::-1][:sg_m]
        w_m = weights[top_m].copy()
        w_m = w_m / w_m.sum()
        g = np.zeros(n)
        for idx in range(len(top_m)):
            lag = int(top_m[idx]) - (n - 1)
            g = g + w_m[idx] * _grad_at_lag(y, lag)
        step = alpha0 / np.sqrt(float(it + 1))
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

_g_ts = 0.0
_g_n = 0

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 30
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = np.full(n_init, 0.5)

    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    h_fourier = _fourier_init(n_init, 5, rng)
    if _c5_of(h_fourier) < best_c5:
        best_c5 = _c5_of(h_fourier)
        best_h = h_fourier.copy()

    t_p1 = deadline - 800
    h_pso, c5_pso = _pso_optimize(best_h, best_n, best_n / 2.0, 40, 400, t_p1, rng)
    if c5_pso < best_c5:
        best_c5 = c5_pso
        best_h = h_pso.copy()
        best_n = n_init

    t_p2 = deadline - 650
    h_sub, c5_sub = _subgradient_method(best_h, best_n, best_n / 2.0, 200, t_p2, rng)
    if c5_sub < best_c5:
        best_c5 = c5_sub
        best_h = h_sub.copy()

    t_p3 = deadline - 500
    h_nest, c5_nest = _nesterov_smooth(best_h, best_n, best_n / 2.0, 200, 0.4, 8, 60.0, t_p3)
    if c5_nest < best_c5:
        best_c5 = c5_nest
        best_h = h_nest.copy()

    t_p4 = deadline - 350
    h_sl1 = _slsqp_lse(best_h, best_n, best_n / 2.0, 100, 300, t_p4)
    c5_sl1 = _c5_of(h_sl1)
    if c5_sl1 < best_c5:
        best_c5 = c5_sl1
        best_h = h_sl1.copy()

    t_p5 = deadline - 250
    h_cd1, c5_cd1 = _coord_descent_cont(best_h, best_n, best_n / 2.0, 5, t_p5)
    if c5_cd1 < best_c5:
        best_c5 = c5_cd1
        best_h = h_cd1.copy()

    t_p6 = deadline - 150
    h_sl2 = _slsqp_lse(best_h, best_n, best_n / 2.0, 200, 500, t_p6)
    c5_sl2 = _c5_of(h_sl2)
    if c5_sl2 < best_c5:
        best_c5 = c5_sl2
        best_h = h_sl2.copy()

    t_p7 = deadline - 80
    h_cd2, c5_cd2 = _coord_descent_cont(best_h, best_n, best_n / 2.0, 4, t_p7)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    if time.time() < deadline - 60:
        h_mr, c5_mr, n_mr = _multi_resolution(best_h, best_n, best_n / 2.0, deadline - 40, rng)
        if c5_mr < best_c5:
            best_c5 = c5_mr
            best_h = h_mr.copy()
            best_n = n_mr

    if time.time() < deadline - 30:
        ts = best_n / 2.0
        h_final_nest, c5_fn = _nesterov_smooth(best_h, best_n, ts, 150, 0.2, 10, 80.0, deadline - 20)
        if c5_fn < best_c5:
            best_c5 = c5_fn
            best_h = h_final_nest.copy()

    if time.time() < deadline - 15:
        h_final_sl = _slsqp_lse(best_h, best_n, best_n / 2.0, 300, 2000, deadline - 10)
        c5_fs = _c5_of(h_final_sl)
        if c5_fs < best_c5:
            best_c5 = c5_fs
            best_h = h_final_sl.copy()

    if time.time() < deadline - 8:
        h_final_cd, c5_fc = _coord_descent_cont(best_h, best_n, best_n / 2.0, 3, deadline - 5)
        if c5_fc < best_c5:
            best_c5 = c5_fc
            best_h = h_final_cd.copy()

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
