Actually, one important thing I want to add: **fractional coordinate descent** that's also fast.

For fractional CD, instead of trying just 0 and 1, try a few intermediate values. The key insight is that for each coordinate, the objective as a function of h[j0] is:
max_k corr_k(h) where corr_k depends on h[j0] linearly (for k where j0 is involved in the sum).

Wait, corr_k = Σ_i h[i]*(1-h[i+L_k]). The terms involving h[j0] are:
- h[j0] * (1 - h[j0+L_k]) if j0+L_k valid → linear in h[j0]
- h[j0-L_k] * (1 - h[j0]) if j0-L_k valid → linear in h[j0]

So corr_k as a function of h[j0] = v is:
corr_k(v) = corr_k(0) + v * [ (1-h[j0+L_k])*1_{valid} - h[j0-L_k]*1_{valid} ]

Wait no. Let me be more careful.

corr_k = Σ_i h[i]*(1-h[i+k-(n-1)]) (using the full mode indexing)

Let L = k - (n-1). Then:
corr(L) = Σ_i h[i]*(1-h[i+L]) for valid i, i+L.

Terms involving h[j0]:
1. i = j0: h[j0]*(1-h[j0+L]) → if j0+L valid, this is h[j0]*(1-h[j0+L]) which is linear in h[j0]
2. i+L = j0, i.e., i = j0-L: h[j0-L]*(1-h[j0]) → if j0-L valid, this is h[j0-L]*(1-h[j0]) which is linear in h[j0]

So corr(L) as a function of v = h[j0] is:
corr(L) = corr(L)_base + v * (1-h[j0+L])*1_{j0+L valid} - v * h[j0-L]*1_{j0-L valid}

where corr(L)_base is the part not involving h[j0].

Let a_L = (1-h[j0+L])*1_{j0+L valid} - h[j0-L]*1_{j0-L valid} (for L≠0)
For L=0: a_0 = (1-v) - v = 1-2v (this is quadratic, not linear!)

Wait, for L=0:
Term: h[j0]*(1-h[j0]) = v*(1-v) = v - v². This IS quadratic in v.

For L≠0:
Terms: v*(1-h[j0+L]) + h[j0-L]*(1-v) = v - v*h[j0+L] + h[j0-L] - v*h[j0-L]
= v*(1 - h[j0+L] - h[j0-L]) + h[j0-L]
This is LINEAR in v.

So for L≠0, corr(L) is linear in v, and for L=0, it's quadratic.

This means the max of {corr(L)} as a function of v is the max of a set of linear functions (plus one quadratic). The optimum v is where two of these functions cross (or at a boundary 0 or 1).

This is a 1D optimization! I can solve it exactly by finding the maximum of a set of lines (and one parabola) over [0,1].

For the linear functions: corr(L) = b_L + a_L * v where b_L is the intercept and a_L is the slope.
The max of lines is a piecewise linear convex function. Its maximum over [0,1] occurs at either v=0, v=1, or at a crossing point of two lines.

For the quadratic (L=0): f_0(v) = v - v² = v(1-v), which is a downward parabola with max at v=0.5.

So the overall max is the max of:
- Several lines: max_L (b_L + a_L * v) for L≠0
- One parabola: v(1-v) (for L=0, plus the base)

The maximum of a set of lines over [0,1] can be found by computing the upper envelope. But that's complex.

A simpler approach: since we're in 1D, we can just evaluate at a few points:
- v = 0
- v = 1
- v = 0.5 (peak of the parabola)
- Crossing points of the top lines

Actually, for practical purposes, trying v ∈ {0, 0.25, 0.5, 0.75, 1} should be good enough. And since the update is O(n) per evaluation, trying 5 values is O(5n) per coordinate, which is still much better than O(n²) for recomputation.

import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0
_global_alpha = 0.0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_at_lag(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

            if old <= 1e-10 or old >= 1.0 - 1e-10:
                if not use_frac:

            current_max = corr.max()

            best_v = old
            for v in vals:

                        if 0 <= jm < n:

                c_new = corr.max()

def _nesterov_m(h0, n, target_sum, n_iters, alpha0, top_m=5, deadline=None):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_corr = np.max(corr)
        if top_m > 1:
            sorted_idx = np.argsort(corr)[::-1]
            top_idx = sorted_idx[:top_m]
            top_vals = corr[top_idx]
            w = np.exp(40.0 * (top_vals - top_vals[0]))
            w = w / w.sum()
            g = np.zeros(n)
            for mi in range(top_m):
                g += w[mi] * _grad_at_lag(y, top_idx[mi], dx)
        else:
            k_star = int(np.argmax(corr))
            g = _grad_at_lag(y, k_star, dx)
        alpha = alpha0 / np.sqrt(float(it + 1))
        x_new = _project(y - alpha * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _lsq_obj(x):
    n = _global_n
    alpha = _global_alpha
    corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
    m = np.max(corr)
    if m < -500
        return m
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

def _lsq_eq_fun(x):
    return x.sum() - _global_target_sum

def _lsq_eq_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter=300):
    global _global_target_sum, _global_n, _global_alpha
    _global_target_sum = target_sum
    _global_n = n
    _global_alpha = alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _lsq_eq_fun, 'jac': _lsq_eq_jac}]
    try:
        res = minimize(_lsq_obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _de_optimize(h0, n, target_sum, pop_size, n_gen, deadline, rng):
    pop = [h0.copy()]
    caches = [_c5_of(h0)]
    for i in range(pop_size - 1):
        h_rand = _project(h0 + 0.5 * (rng.rand(n) * 2 - 1), target_sum)
        pop.append(h_rand)
        caches.append(_c5_of(h_rand))
    best_h = h0.copy()
    best_c5 = caches[0]
    stall_count = 0
    for gen in range(n_gen):
        if time.time() > deadline:
            break
        t_frac = gen / max(n_gen - 1, 1)
        f_mut = 0.95 - 0.6 * t_frac
        cr = 0.3 + 0.5 * (1.0 - t_frac)
        best_idx = int(np.argmin(caches))
        improved_this_gen = False
        new_pop = [pop[best_idx].copy()]
        new_caches = [caches[best_idx]]
        for i in range(pop_size - 1):
            candidates = [x for x in range(pop_size) if x != i]
            r1, r2, r3 = rng.choice(len(candidates), size=3, replace=False)
            r1 = candidates[r1]
            r2 = candidates[r2]
            r3 = candidates[r3]
            mutant = pop[r1] + f_mut * (pop[r2] - pop[r3])
            mutant = _project(mutant, target_sum)
            trial = pop[i].copy()
            mask = rng.rand(n) < cr
            trial[mask] = mutant[mask]
            trial = _project(trial, target_sum)
            c_trial = _c5_of(trial)
            if c_trial <= caches[i]:
                trial = _fix_sum(trial, target_sum)
                new_pop.append(trial)
                new_caches.append(c_trial)
                if c_trial < best_c5:
                    best_c5 = c_trial
                    best_h = trial.copy()
                    improved_this_gen = True
            else:
                new_pop.append(pop[i].copy())
                new_caches.append(caches[i])
        pop = new_pop
        caches = new_caches
        if not improved_this_gen:
            stall_count += 1
        else:
            stall_count = 0
        if stall_count >= 20:
            break
    return best_h, best_c5

def _local_refine(h0, n, target_sum, deadline):

    h = h0.copy()
    c5 = _c5_of(h)
    if time.time() < t_end - 60:

        c5_cd = _c5_of(h_cd)
        if c5_cd < c5:
            c5 = c5_cd
            h = h_cd
    if time.time() < t_end - 40:

        if c5_n < c5:
            c5 = c5_n
            h = h_n
    for alpha in [200, 800, 2000, 5000]:
        if time.time() > t_end - 20:

        h_s = _slsqp_opt(h, n, target_sum, alpha, maxiter=400)
        c5_s = _c5_of(h_s)
        if c5_s < c5:
            c5 = c5_s
            h = h_s
    if time.time() < t_end - 15:
        h_cd2 = _fast_cd(h, n, target_sum, max_rounds=30, deadline=t_end - 10)
        c5_cd2 = _c5_of(h_cd2)
        if c5_cd2 < c5:
            c5 = c5_cd2
            h = h_cd2
    if time.time() < t_end - 8:
        h_f = _fast_cd(h, n, target_sum, max_rounds=15, deadline=t_end - 5, use_frac=True)
        c5_f = _c5_of(h_f)
        if c5_f < c5:
            c5 = c5_f
            h = h_f
    return h, c5

def _gen_diverse(n, rng, base_h):
    starts = []
    starts.append(base_h.copy())
    h_rand = _project(rng.rand(n), n / 2.0)
    starts.append(h_rand)
    h_alt = np.zeros(n)
    step = max(1, n // 70)
    for i in range(n):
        h_alt[i] = 1.0 if (i // step) % 2 == 0 else 0.0
    starts.append(_fix_sum(h_alt, n / 2.0))
    h_sine = 0.5 + 0.45 * np.sin(np.pi * np.arange(n) * 3.0 / n)
    starts.append(_fix_sum(h_sine, n / 2.0))
    h_step2 = np.zeros(n)
    h_step2[:n // 2] = 1.0
    starts.append(_fix_sum(h_step2, n / 2.0))
    return starts
