def compute_c5(h):

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current_sum = h_r.sum()
    if current_sum == target_sum:
        return h_r.astype(float)
    if current_sum < target_sum:
        need = target_sum - current_sum
        zeros = np.where(h_r == 0)[0]
        # Pick the ones with highest h value
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current_sum - target_sum
        ones = np.where(h_r == 1)[0]

def qr_sequence(n, rng):
    """Generate a quadratic residue-inspired binary sequence of length n with n/2 ones."""
    # Find a prime near n
    def is_prime(x):
        if x < 2: return False
        if x % 2 == 0: return x == 2
        for f in range(3, int(np.sqrt(x)) + 1, 2):
            if x % f == 0: return False
        return True
    
    p = n
    while not is_prime(p):
        p += 1
    
    # Legendre symbol based sequence
    h = np.zeros(n)
    for i in range(1, n):
        # i is a QR mod p if i^((p-1)/2) ≡ 1 (mod p)
        if pow(i, (p-1)//2, p) == 1:
            h[i] = 1.0
    
    # Adjust to get exactly n/2 ones
    target = n // 2
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        h[zeros[:target - current]] = 1.0
    elif current > target:
        ones = np.where(h == 1)[0]
        rng.shuffle(ones)
        h[ones[:current - target]] = 0.0
    
    return h

def compute_delta(h, i, j, n):
    """Compute change in correlation when moving 1 from position i to position j."""
    # corr[m] = sum_k h[k] * (1 - h[k + m - n + 1])
    # Change: remove contribution of h[i]=1, add contribution of h[j]=1
    # Also: (1-h[i]) changes from 0 to 1, (1-h[j]) changes from 1 to 0
    
    m = np.arange(2 * n - 1)
    # Effective indices for each m:
    # h appears at position k in the sum, and (1-h) is evaluated at k + m - n + 1
    # When we change h[i] from 1 to 0:
    #   - Direct term: h[i] was 1, now 0. Affects corr[m] by -(1 - h[i + m - n + 1])
    #   - Indirect term: (1-h[i]) was 0, now 1. Affects corr[m] by +h[k] where k + m - n + 1 = i, i.e., k = i - m + n - 1
    # When we change h[j] from 0 to 1:
    #   - Direct term: h[j] was 0, now 1. Affects corr[m] by +(1 - h[j + m - n + 1])
    #   - Indirect term: (1-h[j]) was 1, now 0. Affects corr[m] by -h[k] where k + m - n + 1 = j, i.e., k = j - m + n - 1
    
    idx_forward_i = i + m - n + 1  # where (1-h) is evaluated for direct term of i
    idx_forward_j = j + m - n + 1  # where (1-h) is evaluated for direct term of j
    idx_backward_i = i - m + n - 1  # where h is evaluated for indirect term of i
    idx_backward_j = j - m + n - 1  # where h is evaluated for indirect term of j
    
    delta = np.zeros(2 * n - 1)
    
    # Direct term for i: -(1 - h[idx_forward_i]) where in range
    mask = (idx_forward_i >= 0) & (idx_forward_i < n)
    delta[mask] -= 1.0 - h[idx_forward_i[mask]]
    
    # Direct term for j: +(1 - h[idx_forward_j]) where in range
    mask = (idx_forward_j >= 0) & (idx_forward_j < n)
    delta[mask] += 1.0 - h[idx_forward_j[mask]]
    
    # Indirect term for i: +h[idx_backward_i] where in range
    mask = (idx_backward_i >= 0) & (idx_backward_i < n)
    delta[mask] += h[idx_backward_i[mask]]
    
    # Indirect term for j: -h[idx_backward_j] where in range
    mask = (idx_backward_j >= 0) & (idx_backward_j < n)
    delta[mask] -= h[idx_backward_j[mask]]
    
    return delta

def project_binary(h, target_sum):
    n = len(h)
    h_r = np.round(h).astype(int)
    h_r = np.clip(h_r, 0, 1)
    current = int(h_r.sum())
    target = int(target_sum)
    if current == target:
        return h_r.astype(float)
    if current < target:
        need = target - current
        zeros = np.where(h_r == 0)[0]
        order = np.argsort(-h[zeros])[:need]
        h_r[zeros[order]] = 1
    else:
        need = current - target
        ones = np.where(h_r == 1)[0]
        order = np.argsort(h[ones])[:need]
        h_r[ones[order]] = 0
    return h_r.astype(float)

    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)

        return h
    if diff > 0:
        idx = np.where(h < 1.0)[0]
    else:
        idx = np.where(h > 0.0)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    return np.clip(h, 0.0, 1.0)

            i2 = int(ones[rng.randint(len(ones))])
            if i1 == i2:
                continue
            j1 = int(zeros[rng.randint(len(zeros))])
            j2 = int(zeros[rng.randint(len(zeros))])
            if j1 == j2:
                continue
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            one_minus_h[i1] = 1
            one_minus_h[i2] = 1
            one_minus_h[j1] = 0
            one_minus_h[j2] = 0
            saved_change = [(i1, 1), (i2, 1), (j1, 0), (j2, 0)]
            valid = True
        elif move_type == 2 and len(ones) > 0 and len(zeros) > 0:
            worst_idx = int(np.argmax(corr))
            worst_lag = worst_idx - (n - 1)
            contrib_idx = []
            for ci in range(len(ones)):
                oi = int(ones[ci])
                pos = oi + worst_lag
                if 0 <= pos < n and h[pos] == 0:
                    contrib_idx.append(ci)
            if len(contrib_idx) > 0:
                ci = contrib_idx[rng.randint(len(contrib_idx))]
                ii = int(ones[ci])
            else:
                ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0
            h[jj] = 1
            one_minus_h[ii] = 1
            one_minus_h[jj] = 0
            saved_change = [(ii, 1), (jj, 0)]
            valid = True
        elif move_type == 3 and len(ones) >= 3 and len(zeros) >= 3:
            idx1 = rng.choice(len(ones), 3, replace=False)
            idx0 = rng.choice(len(zeros), 3, replace=False)
            saved_change = []
            for idx in idx1:
                pos = int(ones[idx])
                h[pos] = 0
                one_minus_h[pos] = 1
                saved_change.append((pos, 1))
            for idx in idx0:
                pos = int(zeros[idx])
                h[pos] = 1
                one_minus_h[pos] = 0
                saved_change.append((pos, 0))
            valid = True
        elif move_type == 4 and len(ones) > 0 and len(zeros) > 0:
            block = rng.randint(1, max(2, n // 100) + 1)
            ii = int(ones[rng.randint(len(ones))])
            jj = int(zeros[rng.randint(len(zeros))])
            h[ii] = 0
            h[jj] = 1
            one_minus_h[ii] = 1
            one_minus_h[jj] = 0
            saved_change = [(ii, 1), (jj, 0)]
            for _ in range(block - 1):
                if len(ones) > 0 and len(zeros) > 0:
                    oi = int(ones[rng.randint(len(ones))])
                    oz = int(zeros[rng.randint(len(zeros))])
                    if oi == ii or oz == jj:
                        continue
                    h[oi] = 0
                    h[oz] = 1
                    one_minus_h[oi] = 1
                    one_minus_h[oz] = 0
                    saved_change.append((oi, 1))
                    saved_change.append((oz, 0))
            valid = True

        if not valid:
            continue

        new_corr = np.correlate(h, one_minus_h, mode='full')
        new_c5 = float(np.max(new_corr)) * 2.0 / n
        delta = new_c5 - cur_c5

        if delta < -1e-15 or (temp > 1e-14 and rng.random() < np.exp(-delta / temp)):
            corr = new_corr.astype(np.float64)
            cur_c5 = new_c5
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            for pos, old_val in saved_change:
                h[pos] = old_val
                one_minus_h[pos] = 1.0 - old_val
            no_imp += 1

    return best_h, best_c5


def smooth_objective(h, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max) / dx
    shifted = np.clip(shifted, -500.0, 0.0)
    return float((c_max + np.log(np.sum(np.exp(shifted))) / alpha) * dx)


def _cons_fun(h, target):
    return np.sum(h) - target


def _cons_jac(h, target):
    return np.ones(len(h))


def continuous_refine(h_init, n, target, time_limit, t_start):
    h_cur = project_feasible(h_init.copy(), target)
    curr_score = compute_c5(h_cur)
    bounds = [(0.0, 1.0)] * n
    alphas = [20, 60, 150, 400, 800, 1500]

    for alpha in alphas:
        if time.time() - t_start > time_limit:
            break
        try:
            def obj(x):
                return smooth_objective(x, alpha, n)

            def cfun(x):
                return np.sum(x) - target

            def cjac(x):
                return np.ones(n)

            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                           options={'maxiter': 500, 'ftol': 1e-14})
            h_new = project_feasible(res.x, target)
            sc = compute_c5(h_new)
            if sc < curr_score:
                curr_score = sc
                h_cur = h_new
        except Exception:
            pass

    return h_cur, curr_score


def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    remaining = budget_s - 30.0

    h_init_arr = np.array(initial_h_values, dtype=float)

        best_c5 = c5_ls
        best_h = h_ls.copy()

                                min(120.0, remaining), start, 0.015, 0.00005)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()

    remaining = budget_s - 30.0 - (time.time() - start)

    for nt in n_try_list[1:]:
        if remaining < 80.0:
            break
        h_qr = make_qr_sequence(nt, rng)
        h_ls, c5_ls = local_search(h_qr, nt / 2.0, 150000, rng,
                                    min(100.0, remaining), start, 0.015, 0.00005)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 30.0 - (time.time() - start)

    if c5_ls < best_c5:

        best_h = h_ls.copy()

    remaining = budget_s - 30.0 - (time.time() - start)

    for it in range(6):

        h_pert_bin = project_binary(h_pert, target_use)
        h_ls, c5_ls = local_search(h_pert_bin, target_use, 120000, rng,
                                    min(80.0, remaining), start, 0.012, 0.00005)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
        remaining = budget_s - 30.0 - (time.time() - start)

    remaining = budget_s - 30.0 - (time.time() - start)

    n_res_list = [n_init - 20, n_init + 20, n_init + 40, n_init + 60, n_init + 100]
    for nt in n_res_list:
        if remaining < 70.0:
            break
        if nt < 200:
            continue
        target_nt = nt / 2.0
        x_old = np.linspace(0.0, 1.0, len(best_h))
        x_new = np.linspace(0.0, 1.0, nt)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_try = project_binary(h_interp, target_nt)
        h_ls, c5_ls = local_search(h_bin_try, target_nt, 100000, rng,

        h_final = best_h
        final_c5 = best_c5

    return h_final, final_c5, n_final
