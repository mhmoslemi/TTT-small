import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    """Project y onto {x: 0 <= x <= 1, sum(x) = S}"""
    n = len(y)
    # Use bisection on the threshold
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix small numerical errors
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:

            for i in order:

                x[i] += add
                diff -= add

                diff += sub

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_lag(h, L, dx):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -L)
    hi1 = min(n, n - L)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+L:hi1+L])
    lo2 = max(0, L)
    hi2 = min(n, n + L)
    if lo2 < hi2:
        grad[lo2:hi2] += -h[lo2-L:hi2-L]
    return grad * dx

def _smooth_gradient(h, alpha, top_m, dx):
    """Compute the softmax-weighted gradient using top M lags."""
    ov = _overlaps(h)
    # Get top M lags
    n_ov = len(ov)
    if top_m >= n_ov:
        top_indices = np.arange(n_ov)
    else:
        top_indices = np.argsort(ov)[::-1][:top_m]
    # Compute softmax weights
    vals = ov[top_indices]
    max_v = vals.max()
    weights = np.exp(alpha * (vals - max_v))
    weights = weights / weights.sum()
    # Compute weighted gradient
    grad = np.zeros(len(h))
    for idx in range(len(top_indices)):
        k_idx = top_indices[idx]
        L = k_idx - (len(h) - 1)
        grad += weights[idx] * _grad_lag(h, L, dx)
    return grad, ov

def _nesterov(h0, n, target_sum, n_iters, alpha, step0, top_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Adaptive step size
        step = step0 / np.sqrt(float(it + 1))
        # Compute gradient at y
        g, _ = _smooth_gradient(y, alpha, top_m, dx)
        # Nesterov step
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        # Evaluate
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_descent_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    current = _c5_of(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            # Try h[j] = 0
            h[j] = 0.0
            c0 = _c5_of(h)
            # Try h[j] = 1
            h[j] = 1.0
            c1 = _c5_of(h)
            # Restore
            h[j] = old
            if c0 < current - 1e-12:
                h[j] = 0.0
                current = c0
                improved = True
            elif c1 < current - 1e-12:
                h[j] = 1.0
                current = c1
                improved = True
        if not improved:
            break
    return h, current

def _coord_descent_frac(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    current = _c5_of(h)
    step = 0.1
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            best_v = old
            best_c = current
            # Try a few values around current
            for dv in [-step, step, -0.5, 0.5, -0.25, 0.25]:
                v = old + dv
                if 0.0 <= v <= 1.0:
                    h[j] = v
                    c = _c5_of(h)
                    if c < best_c - 1e-12:
                        best_c = c
                        best_v = v
            # Also try 0 and 1
            h[j] = 0.0
            c = _c5_of(h)
            if c < best_c - 1e-12:
                best_c = c
                best_v = 0.0
            h[j] = 1.0
            c = _c5_of(h)
            if c < best_c - 1e-12:
                best_c = c
                best_v = 1.0
            h[j] = best_v
            if best_v != old:
                current = best_c
                improved = True
        if not improved:
            break
        step *= 0.7
    return h, current

def _slsqp_opt(h0, n, target_sum, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        ov = _overlaps(x)
        m = ov.max()
        return (m + np.log(np.sum(np.exp(alpha * (ov - m)))) / alpha)
    
    def obj_grad(x):
        g, _ = _smooth_gradient(x, alpha, n*2-1, dx)  # use all lags
        return g
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

import time
import numpy as np
from scipy.optimize import minimize

def _c5_of(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project_box_sum(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _grad_lag(h, L, dx):
    n = len(h)
    grad = np.zeros(n)
    lo1 = max(0, -L)
    hi1 = min(n, n - L)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 + L:hi1 + L])
    lo2 = max(0, L)
    hi2 = min(n, n + L)
    if lo2 < hi2:
        grad[lo2:hi2] += -h[lo2 - L:hi2 - L]
    return grad * dx

def _smooth_grad(h, alpha, top_m, dx):
    n = len(h)
    ov = _overlaps(h)
    n_ov = len(ov)
    if top_m >= n_ov:
        top_idx = np.arange(n_ov)
    else:
        top_idx = np.argsort(ov)[::-1][:top_m]
    vals = ov[top_idx]
    mx = vals.max()
    exps = np.exp(alpha * (vals - mx))
    w = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        L = int(top_idx[idx]) - (n - 1)
        grad += w[idx] * _grad_lag(h, L, dx)
    return grad, ov, mx

def _nesterov_run(h0, n, target_sum, n_iters, alpha, step0, top_m, deadline):
    dx = 2.0 / n
    x = h0.copy()
    y = h0.copy()
    t = 1.0
    best = x.copy()
    best_val = _c5_of(x)
    for it in range(n_iters):
        if time.time() > deadline:
            break
        step = step0 / np.sqrt(float(it + 1))
        g, ov, mx = _smooth_grad(y, alpha, top_m, dx)
        x_new = _project_box_sum(y - step * g, target_sum)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = x_new + ((t - 1.0) / t_new) * (x_new - x)
        y_new = _project_box_sum(y_new, target_sum)
        x = x_new
        y = y_new
        t = t_new
        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_binary(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    cur = _c5_of(h)
    for _ in range(max_rounds):
        if time.time() > deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            if c0 < cur - 1e-13:
                h[j] = 0.0
                cur = c0
                imp = True
            elif c1 < cur - 1e-13:
                h[j] = 1.0
                cur = c1
                imp = True
        if not imp:
            break
    return h, cur

def _coord_frac(h0, n, target_sum, max_rounds, deadline):
    h = h0.copy()
    cur = _c5_of(h)
    offsets = [0.05, 0.1, 0.15, 0.25, 0.4, 0.6, -0.05, -0.1, -0.15, -0.25, -0.4, -0.6]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        imp = False
        scale = 1.0 / (1.0 + 0.5 * rnd)
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            bv = old
            bc = cur
            for dv in offsets:
                v = old + dv * scale
                if 0.0 <= v <= 1.0:
                    h[j] = v
                    c = _c5_of(h)
                    if c < bc - 1e-13:
                        bc = c
                        bv = v
            h[j] = 0.0
            c = _c5_of(h)
            if c < bc - 1e-13:
                bc = c
                bv = 0.0
            h[j] = 1.0
            c = _c5_of(h)
            if c < bc - 1e-13:
                bc = c
                bv = 1.0
            h[j] = bv
            if bv != old:
                cur = bc
                imp = True
        if not imp:
            break
    return h, cur

def _slsqp_polish(h0, n, target_sum, alpha, maxiter, deadline):
    dx = 2.0 / n
    n_ov = 2 * n - 1

    def obj_fn(x):
        ov = _overlaps(x)
        m = ov.max()
        return float(m + np.log(np.sum(np.exp(alpha * (ov - m)))) / alpha)

    def obj_grad(x):
        g, _, _ = _smooth_grad(x, alpha, n_ov, dx)
        return g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj_fn, h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _perturb_reopt(h0, n, target_sum, rng, alpha, n_iters, step0, deadline):
    h_new = h0 + rng.normal(0, 0.05, n)
    h_new = _fix_sum(np.clip(h_new, 0.0, 1.0), target_sum)
    best, bv = _nesterov_run(h_new, n, target_sum, n_iters, alpha, step0, 15, deadline)
    h_cd, cv = _coord_binary(best, n, target_sum, 5, deadline)
    if cv < bv:
        best, bv = h_cd, cv
    return best, bv
