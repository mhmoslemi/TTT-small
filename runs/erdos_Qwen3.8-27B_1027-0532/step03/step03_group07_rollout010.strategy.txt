```python
def _fix_sum(h, target_sum):
    """From provided code, clip and adjust to enforce sum exactly target_sum."""
    h_array = numpy.array(h)
    h_array = numpy.clip(h_array, 0.0, 1.0)
    diff = target_sum - h_array.sum()
    if abs(diff) < 1e-12:
        return h_array
    order = numpy.argsort(1.0 - h_array) if diff > 0 else numpy.argsort(h_array)
    if diff > 0:
        for i in order:
            if abs(diff) < 1e-12:
                break
            add = min(diff, 1.0 - h_array[i])
            h_array[i] += add
            diff -= add
    else:
        for i in order:
            if abs(diff) < 1e-12:
                break  # Since `-diff` was the original diff (negative now subtracted)
            sub = min(-diff, h_array[i])
            h_array[i] -= sub
            diff += sub
    return h_array.clip(0.0, 1.0)
```

```python
def objective_function(h):
    """Evaluate the objective: maximum of full correlation times dx."""
    corr = numpy.correlate(h, 1.0 - h, mode='full')
    max_corr = numpy.max(corr)
    return max_corr * (2.0 / len(h))

def sinitial_ga_population(n, seed=None, parent=None):
    """Initialize GA population of size 100."""
    numpy.random.seed(seed)
    population = []
    if parent is not None:
        interpolated = interp_parent(parent, n)
        fixed = _fix_sum(interpolated, n/2)
        population.append(fixed)
        # Additionally, create another individual by mutation/remixing parent
        if len(population) < 100:
            variation = fixed + 0.1 * (numpy.random.rand(len(fixed)) * 2 - 1)
            fixed_var = _fix_sum(variation, n/2)
            population.append(fixed_var)
    # Fill the rest with random justified samples
    while len(population) < 100:
        h_random = numpy.clip(numpy.random.rand(len(parent)) * 2 - 1, 0, 1)
        fixed_random = _fix_sum(h_random, n/2)
        population.append(fixed_random)
    return population

def tournament_selection(population, fitness, tournament_size=5):
    """Select parents with tournament selection."""
    parents = []
    for _ in range(len(population) // 2):
        candidates = numpy.random.choice(range(len(population)), tournament_size, replace=False)
        best_index = candidates[fitness[candidates].argmin()]
        parents.append(population[best_index])
    return parents

def crossover(individual1, individual2, eta_cx=20):
    """Single-point crossover with polynomial perturbation."""
    # Single-point crossover with probability eta_cx? Standard SBX is non-parametric.
    # For simplicity, use one-point crossover
    crossover_point = numpy.random.randint(len(individual1))
    child1 = numpy.concatenate([individual1[:crossover_point], individual2[crossover_point:]])
    child2 = numpy.concatenate([individual2[:crossover_point], individual1[crossover_point:]])
    return child1, child2

def mutate(individual, eta=20):
    """Mutate an individual using polynomial mutation."""
    mutated = numpy.clip(individual + numpy.random.randn(len(individual)) * (eta=20), 0, 1.0)
    return mutated

def repair(individual, target_sum):
    """Repair an individual to meet the integral constraint."""
    current_sum = numpy.sum(individual)
    diff = target_sum - current_sum
    result = individual.copy()
    if diff > 0:
        # Can add up to (1 - result[i]) for each item
        order = numpy.argsort(1 - result)
        for i in order:
            if diff <= 0 or (1 - result[i]) == 0:
                break
            add = min(diff, 1 - result[i])
            result[i] += add
            diff -= add
    else:
        # Can subtract up to result[i]
        order = numpy.argsort(result)
        for i in order:
            if diff >= 0:  # Correction is done
                break
            sub = min(-diff, result[i])
            result[i] -= sub
            diff += sub
    return _fix_sum(result, target_sum)  # Ensure bounds in fix_sum

def ga_phase(n, budget_remaining):
    """Main GA execution."""
    target_sum = n / 2
    population = sinitial_ga_population(n, seed=42, parent=initial_h_values)
    best_fitness = float('inf')
    best_solution = None
    start_time = time.time()
    while time.time() - start_time < budget_remaining:
        fitness = [objective_function(ind) for ind in population]
        best_idx = numpy.argmin(fitness)
        current_best = population[best_idx]
        best_fitness = fitness[best_idx]
        best_solution = current_best.copy()

        parents = tournament_selection(population, fitness)
        children = []

        for p in parents:
            child1, child2 = crossover(p, p)  # Not implemented technically, using asexual for simplicity
            child1 = mutate(child1)
            child2 = mutate(child2)
            child1_repaired = repair(child1, target_sum)
            child2_repaired = repair(child2, target_sum)
            children.append(child1_repaired)
            children.append(child2_repaired)

        # Keep the best (n-1) individuals and one random child replacement
        combined = population + children
        fitness_combined = [objective_function(ind) for ind in combined]
        # Sort by fitness and keep top `n-1` individuals
        args = numpy.argsort(fitness_combined)
        to_keep = args[:len(population) - 1]  # Keep top n-1
        population = [combined[i] for i in to_keep]
        # Append one more randomly chosen child
        population.append(children[numpy.random.randint(len(children))])
```

```python
def objective_slsqp(h):
    """Objective function for SLSQP with numerical gradient."""
    return objective_function(h)

def grad_slsqp_estimate(h):
    """Numerical gradient for the original objective."""
    grad = numpy.zeros(len(h))
    eps = numpy.sqrt(numpy.finfo(float).eps)
    for j in range(len(h)):
        perturbed = h.copy()
        perturbed[j] += eps
        obj_plus = objective_function(perturbed)
        perturbed[j] -= 2 * eps  # Second perturbation for finite difference
        obj_minus = objective_function(perturbed)
        # Central difference (more accurate)
        grad[j] = (obj_plus - obj_minus) / (2 * eps)
    return grad
```
