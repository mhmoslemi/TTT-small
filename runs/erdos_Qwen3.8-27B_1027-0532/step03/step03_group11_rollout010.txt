        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)) + 1e-300)

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h, target_sum):
    return float(np.sum(h) - target_sum)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter=500, ftol=1e-14):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: constraint_fun(h, target_sum),
             'jac': lambda h: constraint_jac(h)}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

```python
def multiplicative_weights(h_init, max_iters=5000, tau_init=0.02, tau_end=0.0005, momentum=0.8):
    n = len(h_init)
    h = project_onto_constraint(h_init.astype(float).copy(), n / 2.0)
    target_sum = n / 2.0
    n_lags = 2 * n - 1
    w = np.ones(n_lags) / n_lags
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        # Compute A[i] = sum_k w[k] * h[i + k - (n-1)]
        A = np.zeros(n)
        for k in range(n_lags):
            wk = w[k]
            if wk < 1e-15:
                continue
            s = k - (n - 1)
            if s >= 0:
                A[:n-s] += wk * h[s:]
            else:
                A[-s:] += wk * h[:n+s]
        
        # h_new = project(A, target_sum)
        h_new = project_onto_constraint(A, target_sum)
        
        # Momentum
        h = momentum * h + (1.0 - momentum) * h_new
        h = project_onto_constraint(h, target_sum)
        
        # Update weights
        corr = compute_corr(h)
        tau = tau_init * (tau_end / tau_init) ** (it / max(max_iters - 1, 1))
        x = corr / max(tau, 1e-12)
        x -= np.max(x)
        w = np.exp(x)
        sw = np.sum(w)
        if sw > 0:
            w /= sw
        else:
            w = np.ones(n_lags) / n_lags
        
        # Track best
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
```

For n=180, that's 32,400 operations per A computation, times 5000 iterations = 162M operations. In Python with numpy vector ops, each inner iteration is fast (just a few numpy operations on arrays of length ≤ 180), so maybe 5-10μs each. Total: 5000 * 359 * 7μs ≈ 12.5 seconds. That's fine.
