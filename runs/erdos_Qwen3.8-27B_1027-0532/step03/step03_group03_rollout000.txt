    weights = exp_shifted / total
    n = len(x)
    return compute_grad_weighted(x, np.arange(2 * n - 1), weights)


def slsqp_refine(h_start, target_sum, alphas, maxiter, ftol):
    global _CONS_N, _CONS_TARGET
    _CONS_N = len(h_start)
    _CONS_TARGET = target_sum
    h_cur = h_start.copy()
    bounds = [(0.0, 1.0)] * _CONS_N
    cons = [{'type': 'eq', 'fun': cons_eq_fun, 'jac': cons_eq_jac}]
    for alpha in alphas:
        try:
            res = minimize(smooth_lse_obj, h_cur, args=(alpha,),
                           jac=smooth_lse_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': maxiter, 'ftol': ftol})
            h_new = project_feasible(res.x, target_sum)
            sc = compute_c5(h_new)
            if sc < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
    return h_cur


def interpolate_h(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    x_new = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)


def run_es(start_h, target_sum, budget_s, seed, mu=15, lam=40,
           sigma_init=0.35, sigma_min=0.02, sigma_max=2.5, patience=60):
    rng = np.random.RandomState(seed)
    n = len(start_h)
    t0 = time.time()

    pop = np.zeros((lam, n))
    pop[0] = start_h.copy()
    for i in range(1, lam):
        frac = i / lam
        s = sigma_init * (1.0 - frac) + sigma_init * 0.2 * frac
        s *= rng.uniform(0.5, 1.5)
        pop[i] = project_feasible(start_h + s * rng.randn(n), target_sum)

    scores = np.array([compute_c5(pop[i]) for i in range(lam)])
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    sigma = sigma_init
    no_improve = 0
    success_count = 0
    window = 0
    gen = 0

    while time.time() - t0 < budget_s:
        gen += 1
        order = np.argsort(scores)[:mu]
        parents = pop[order]

        for i in range(lam):
            pidx = rng.randint(0, mu)
            parent = parents[pidx]
            rank_boost = 1.0 + 0.3 * (1.0 - pidx / max(mu - 1, 1))

            r = rng.random()
            if r < 0.08:
                s = sigma * rng.uniform(2.0, 4.0) * rank_boost
            elif r < 0.25:
                s = sigma * rng.uniform(0.3, 0.7) * rank_boost
            else:
                s = sigma * rng.uniform(0.6, 1.4) * rank_boost

            child = parent + s * rng.randn(n)
            if rng.random() < 0.15:
                bstart = rng.randint(0, n // 2)
                blen = rng.randint(n // 10, n // 3)
                blen = min(blen, n - bstart)
                amplitude = rng.uniform(-0.3, 0.3) * sigma
                child[bstart:bstart + blen] += amplitude * np.cos(
                    np.pi * np.arange(blen) / max(blen, 1))
            pop[i] = project_feasible(child, target_sum)

        for i in range(lam):
            scores[i] = compute_c5(pop[i])

        gen_best = int(np.argmin(scores))
        window += 1
        if scores[gen_best] < best_score + 1e-10:
            if scores[gen_best] < best_score - 1e-10:
                success_count += 1
            best_score = scores[gen_best]
            best_h = pop[gen_best].copy()
            no_improve = 0

        if window >= mu:
            if success_count / window > 0.4:
                sigma = min(sigma * 1.1, sigma_max)
            elif success_count / window < 0.1:
                sigma = max(sigma * 0.8, sigma_min)
            success_count = 0
            window = 0

        pop[0] = best_h.copy()
        scores[0] = best_score

        if no_improve > patience and rng.random() < 0.1:
            px = rng.randint(1, lam)
            pop[px] = project_feasible(best_h + rng.uniform(0.8, 2.0) * rng.randn(n), target_sum)
            scores[px] = compute_c5(pop[px])
            no_improve = 0

    return best_h, best_score, gen


def run_subgradient(h_init, target_sum, max_iter, lr_init, topk, alpha,
                    noise_prob, noise_scale, patience, momentum, seed):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_score = compute_c5(h)
    lr = lr_init
    stagnation = 0

    for it in range(max_iter):
        corr = overlap_corrs(h)
        n_corr = len(corr)
        top_indices = np.argpartition(corr, -topk)[-topk:]
        top_values = corr[top_indices]
        w = np.exp(alpha * (top_values - np.max(top_values)))
        w_sum = np.sum(w)
        if w_sum < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = w / w_sum
        grad = compute_grad_weighted(h, top_indices, w)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h_new = project_feasible(h_new, target_sum)
        if noise_prob > 0.0 and rng.random() < noise_prob:
            h_new = h_new + rng.randn(n) * noise_scale
            h_new = project_feasible(h_new, target_sum)
        score = compute_c5(h_new)
        if score < best_score:
            best_score = score
            best_h = h_new.copy()
            stagnation = 0
            lr = min(lr * 1.03, 5.0)
        else:
            stagnation += 1
            lr *= 0.97
        h = h_new
        if stagnation > patience:
            break
    return best_h, best_score


def run(seed=42, budget_s=1300, **kwargs):
    global _CONS_N, _CONS_TARGET
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _remaining():
        return budget_s - 25.0 - (time.time() - start_time)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    target_init = n_init / 2.0

    best_h = project_feasible(h_init, target_init)
    best_score = compute_c5(best_h)

    es_b1 = min(350, max(100, _remaining() * 0.3))
    h_es, sc_es, _ = run_es(best_h, target_init, es_b1, seed,
                            mu=15, lam=45, sigma_init=0.4,
                            sigma_min=0.04, sigma_max=2.5, patience=70)
    if sc_es < best_score:
        best_score = sc_es
        best_h = h_es

    if _remaining() > 120:
        h_ref = slsqp_refine(best_h, target_init, [40, 100, 300, 600], 1200, 1e-14)
        sc = compute_c5(h_ref)
        if sc < best_score:
            best_score = sc
            best_h = h_ref

    sub_configs = [
        (3000, 0.12, 5, 25, 0.04, 0.02, 250, 0.90),
        (3000, 0.06, 8, 50, 0.02, 0.01, 250, 0.85),
        (4000, 0.20, 3, 12, 0.08, 0.03, 300, 0.92),
    ]
    for i, cfg in enumerate(sub_configs):
        if _remaining() < 60:
            break
        h_try = best_h + rng.randn(n_init) * 0.01
        h_try = project_feasible(h_try, target_init)
        h_new, score = run_subgradient(h_try, target_init,
                                       max_iter=cfg[0], lr_init=cfg[1], topk=cfg[2],
                                       alpha=cfg[3], noise_prob=cfg[4], noise_scale=cfg[5],
                                       patience=cfg[6], momentum=cfg[7], seed=seed + i * 77)
        if score < best_score:
            best_score = score
            best_h = h_new

    if _remaining() > 250:
        n2 = 640
        target2 = n2 / 2.0
        h_640 = interpolate_h(best_h, n2, target2)
        es_b2 = min(300, max(100, _remaining() * 0.35))
        h_es2, sc_es2, _ = run_es(h_640, target2, es_b2, seed + 111,
                                  mu=12, lam=35, sigma_init=0.3,
                                  sigma_min=0.03, sigma_max=1.8, patience=55)
        if sc_es2 < best_score:
            best_score = sc_es2
            best_h = h_es2
        if _remaining() > 100:
            h_ref = slsqp_refine(best_h, target2, [50, 150, 400, 800], 1000, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    if _remaining() > 250:
        n3 = 800
        target3 = n3 / 2.0
        h_800 = interpolate_h(best_h, n3, target3)
        es_b3 = min(300, max(100, _remaining() * 0.4))
        h_es3, sc_es3, _ = run_es(h_800, target3, es_b3, seed + 222,
                                  mu=10, lam=30, sigma_init=0.22,
                                  sigma_min=0.02, sigma_max=1.2, patience=50)
        if sc_es3 < best_score:
            best_score = sc_es3
            best_h = h_es3
        if _remaining() > 100:
            h_ref = slsqp_refine(best_h, target3, [80, 200, 500], 800, 1e-14)
            sc = compute_c5(h_ref)
            if sc < best_score:
                best_score = sc
                best_h = h_ref

    n_use = len(best_h)
    target_use = n_use / 2.0

    if _remaining() > 80:
        h_final = slsqp_refine(best_h, target_use, [150, 400, 800], 700, 1e-15)
        sc = compute_c5(h_final)
        if sc < best_score:
            best_score = sc
            best_h = h_final

    if _remaining() > 50:
        h_fes, sc_fes, _ = run_es(best_h, target_use,
                                  min(35.0, _remaining() - 15.0),
                                  seed + 500, mu=8, lam=20,
                                  sigma_init=0.06, sigma_min=0.008,
                                  sigma_max=0.6, patience=30)
        if sc_fes < best_score:
            best_score = sc_fes
            best_h = h_fes

    if _remaining() > 30:
        h_try = best_h + rng.randn(n_use) * 0.004
        h_try = project_feasible(h_try, target_use)
        h_new, score = run_subgradient(h_try, target_use,
                                       max_iter=2000, lr_init=0.03, topk=5,
                                       alpha=60, noise_prob=0.0, noise_scale=0.0,
                                       patience=150, momentum=0.92, seed=seed + 999)
        if score < best_score:
            best_score = score
            best_h = h_new

    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
