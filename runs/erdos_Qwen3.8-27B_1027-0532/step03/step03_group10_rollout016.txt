import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def smooth_objective(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def constraint_fun(h):
    return float(np.sum(h) - len(h) / 2.0)

def constraint_jac(h):
    return np.ones(len(h))

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        result = minimize(smooth_objective, h0, args=(eps,),
                          jac=smooth_gradient, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def greedy_local_search(h_bin, n_passes=10):
    """Greedy improvement: try all single swaps, keep best."""
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    
    for p in range(n_passes):
        improved = False
        ones_arr = np.where(h == 1.0)[0]
        zeros_arr = np.where(h == 0.0)[0]
        for i in ones_arr:
            if h[i] != 1.0:
                continue
            best_j = -1
            best_max = current_max
            for j_idx in range(len(zeros_arr)):
                j = zeros_arr[j_idx]
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    new_max = current_max  # no change in corr
                else:
                    s_range = np.arange(s_min, s_max + 1)
                    k_range = s_range + half
                    delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
                    new_corr_k = corr[k_range] + delta
                    new_max = max(float(np.max(new_corr_k)), 
                                  float(np.max(np.delete(corr, k_range))))
                if new_max < best_max - 1e-12:
                    best_max = new_max
                    best_j = j
            if best_j >= 0:
                j = best_j
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min <= s_max:
                    s_range = np.arange(s_min, s_max + 1)
                    k_range = s_range + half
                    delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
                    corr[k_range] += delta
                h[i] = 0.0
                h[j] = 1.0
                current_max = best_max
                improved = True
        if not improved:
            break
    
    return h, current_max

                is_prime = False
                break

        improved = False
        ones_arr = np.where(h == 1.0)[0]

                    new_max = current_max
                else:

                    delta = h[i + sr] + h[i - sr] - h[j + sr] - h[j - sr]

                    new_max = max(float(np.max(corr[kr] + delta)), old_max_excl)
                gain = current_max - new_max

                    best_i, best_j = i, j

            i, j = best_i, best_j
            s_min = max(-i, i - half, -j, j - half)

            h[i] = 0.0
            h[j] = 1.0
            current_max -= best_gain
            improved = True

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:

    n = len(h)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

        corr[k_range] += delta
        new_max = float(np.max(corr))
        dc = new_max - current_max

        accepted = False
        if dc < -1e-15:
            accepted = True
        elif t > 1e-10:
            accepted = rng.random() < np.exp(-dc / t)

        if accepted:
            h[i], h[j] = h[j], h[i]
            oi = np.flatnonzero(ones_arr == i)
            oj = np.flatnonzero(zeros_arr == j)
            if len(oi) > 0 and len(oj) > 0:
                ones_arr[oi[0]] = j
                zeros_arr[oj[0]] = i
            current_max = new_max
            if current_max < best_c5_val:
                best_c5_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta

        step += 1

    return best_h, best_c5_val

def fast_sa_segment(h_start, n_steps, t_start, t_end, rng):
    n = len(h_start)
    h = h_start.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_c5_val = current_max
    best_h = h.copy()
    half = n - 1

    ones_arr = np.where(h == 1.0)[0].copy()
    zeros_arr = np.where(h == 0.0)[0].copy()
    n_one = len(ones_arr)
    n_zero = len(zeros_arr)

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        r = rng.random()

        if r < 0.7:
            i = int(ones_arr[rng.integers(n_one)])
            j = int(zeros_arr[rng.integers(n_zero)])
            if i == j:
                step += 1
                continue
            s_min = max(-i, i - half, -j, j - half)
            s_max = min(half - i, i, half - j, j)
            if s_min > s_max:
                h[i], h[j] = h[j], h[i]
                oi = np.flatnonzero(ones_arr == i)
                oj = np.flatnonzero(zeros_arr == j)
                if len(oi) > 0 and len(oj) > 0:
                    ones_arr[oi[0]] = j
                    zeros_arr[oj[0]] = i
                step += 1
                continue
            s_range = np.arange(s_min, s_max + 1)
            k_range = s_range + half
            delta = h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range]
            corr[k_range] += delta
            new_max = float(np.max(corr))
            dc = new_max - current_max
            if dc < -1e-15 or (t > 1e-10 and rng.random() < np.exp(-dc / t)):
                h[i], h[j] = h[j], h[i]
                oi = np.flatnonzero(ones_arr == i)
                oj = np.flatnonzero(zeros_arr == j)
                if len(oi) > 0 and len(oj) > 0:
                    ones_arr[oi[0]] = j
                    zeros_arr[oj[0]] = i
                current_max = new_max
                if current_max < best_c5_val:
                    best_c5_val = current_max
                    best_h = h.copy()
            else:
                corr[k_range] -= delta
        else:
            seg_len = rng.integers(3, min(15, n // 4 + 1))
            pos = rng.integers(0, n - seg_len + 1)
            seg = h[pos:pos + seg_len].copy()
            h[pos:pos + seg_len] = seg[::-1]
            new_corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
            new_max = float(np.max(new_corr))
            dc = new_max - current_max
            if dc < -1e-15 or (t > 1e-10 and rng.random() < np.exp(-dc / t)):
                corr = new_corr
                current_max = new_max
                ones_arr = np.where(h == 1.0)[0].copy()
                zeros_arr = np.where(h == 0.0)[0].copy()
                n_one = len(ones_arr)
                n_zero = len(zeros_arr)
                if current_max < best_c5_val:
                    best_c5_val = current_max
                    best_h = h.copy()
            else:
                h[pos:pos + seg_len] = seg

        step += 1

    return best_h, best_c5_val

def greedy_improve(h_bin, n_passes=10):
    n = len(h_bin)
    h = h_bin.astype(np.float64).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    half = n - 1
    corr_len = 2 * n - 1

    for p in range(n_passes):
        improved = False
        ones_arr = np.where(h == 1.0)[0]
        zeros_arr = np.where(h == 0.0)[0]
        best_gain = 0.0
        best_i, best_j = -1, -1

        for ii in range(len(ones_arr)):
            i = ones_arr[ii]
            if h[i] != 1.0:
                continue
            for jj in range(len(zeros_arr)):
                j = zeros_arr[jj]
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    new_max = current_max
                else:
                    sr = np.arange(s_min, s_max + 1)
                    kr = sr + half
                    delta = h[i + sr] + h[i - sr] - h[j + sr] - h[j - sr]
                    in_max = float(np.max(corr[kr] + delta))
                    out_parts = []
                    if kr[0] > 0:
                        out_parts.append(float(np.max(corr[:kr[0]])))
                    if kr[-1] < corr_len - 1:
                        out_parts.append(float(np.max(corr[kr[-1] + 1:])))
                    out_max = max(out_parts) if out_parts else -np.inf
                    new_max = max(in_max, out_max)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j

        if best_i >= 0 and best_gain > 1e-12:
            i, j = best_i, best_j
            s_min = max(-i, i - half, -j, j - half)
            s_max = min(half - i, i, half - j, j)
            if s_min <= s_max:
                sr = np.arange(s_min, s_max + 1)
                kr = sr + half
                delta = h[i + sr] + h[i - sr] - h[j + sr] - h[j - sr]
                corr[kr] += delta
            h[i] = 0.0
            h[j] = 1.0
            current_max -= best_gain
            improved = True

        if not improved:
            break

    return h, current_max

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 180
        h0 = (rng.random(n0) < 0.5).astype(float)
        h0 = project_onto_constraint(h0, n0 / 2.0)

    n0 = len(h0)
    target_sum = n0 / 2.0
    h0 = project_onto_constraint(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    elapsed = time.time() - start_time
    if budget_s - elapsed > 20 and n0 <= 200:
        h_s, c5_s = slsqp_optimize(best_h, 0.01, 60)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    starts = [to_binary(best_h)]
    if n0 <= 200:
        starts.append(make_quadratic_start(n0, rng))
        starts.append(make_golden_start(n0, rng))
        h_rand = (rng.random(n0) < 0.5).astype(float)
        starts.append(h_rand)

    remaining = budget_s - (time.time() - start_time) - 80
    sa_round = 0

    while remaining > 120:
        used = time.time() - start_time
        if used >= budget_s - 80:
            break

        if sa_round < len(starts):
            h_start = starts[sa_round].copy()
        else:
            r = sa_round % 6
            if r == 0:
                h_start = to_binary(best_h).copy()
            elif r == 1:
                h_p = best_h + 0.1 * rng.standard_normal(n0)
                h_p = project_onto_constraint(h_p, target_sum)
                h_start = to_binary(h_p)
            elif r == 2:
                h_start = to_binary(best_h).copy()
                nf = rng.integers(4, min(20, n0 // 5 + 1))
                idx = rng.choice(n0, size=nf, replace=False)
                h_start[idx] = 1.0 - h_start[idx]
                s = h_start.sum()
                if s > target_sum:
                    ex = int(s - target_sum)
                    ones = np.where(h_start == 1)[0]
                    h_start[ones[:ex]] = 0
                elif s < target_sum:
                    nd = int(target_sum - s)
                    zeros = np.where(h_start == 0)[0]
                    h_start[zeros[:nd]] = 1
            elif r == 3:
                alpha = rng.uniform(0.2, 0.6)
                hr = (rng.random(n0) < 0.5).astype(float)
                h_start = to_binary(alpha * best_h + (1.0 - alpha) * hr)
            elif r == 4:
                h_start = to_binary(best_h).copy()
                ns = rng.integers(2, 8)
                idx = rng.choice(n0, size=ns, replace=False)
                h_start[idx] = 1.0 - h_start[idx]
                s = h_start.sum()
                if s > target_sum:
                    ex = int(s - target_sum)
                    ones = np.where(h_start == 1)[0]
                    h_start[ones[:ex]] = 0
                elif s < target_sum:
                    nd = int(target_sum - s)
                    zeros = np.where(h_start == 0)[0]
                    h_start[zeros[:nd]] = 1
            else:
                h_start = make_quadratic_start(n0, rng) if rng.random() < 0.5 else make_golden_start(n0, rng)

        frac = min(sa_round, 40) / 40.0
        t_start = 0.35 * (1.0 - 0.7 * frac) + 0.05
        t_end = 0.0003
        n_steps = max(50000, min(int((remaining - 80) * 2000), 4000000))

        if sa_round % 3 == 2:
            h_new, c5_raw = fast_sa_segment(h_start, n_steps, t_start, t_end, rng)
        else:
            h_new, c5_raw = fast_sa(h_start, n_steps, t_start, t_end, rng)

        c5_new = (2.0 / n0) * c5_raw
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            rem = budget_s - (time.time() - start_time) - 80
            if rem > 40:
                h_gr, c5_gr_raw = greedy_improve(h_new, 5)
                c5_gr = (2.0 / n0) * c5_gr_raw
                if c5_gr < best_c5:
                    best_c5 = c5_gr
                    best_h = h_gr.copy()

        sa_round += 1
        remaining = budget_s - (time.time() - start_time) - 80

    rem = budget_s - (time.time() - start_time) - 15
    if rem > 20:
        h_fin = to_binary(best_h)
        h_fin, c5_raw = greedy_improve(h_fin, 12)
        c5_fin = (2.0 / n0) * c5_raw
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    rem = budget_s - (time.time() - start_time) - 10
    if rem > 25 and n0 <= 200:
        h_pol = best_h + 0.01 * rng.standard_normal(n0)
        h_pol = project_onto_constraint(h_pol, target_sum)
        for eps in (0.0001, 0.00003):
            if time.time() - start_time > budget_s - 8:
                break
            h_pol, c5 = slsqp_optimize(h_pol, eps, 150)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pol.copy()

    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
