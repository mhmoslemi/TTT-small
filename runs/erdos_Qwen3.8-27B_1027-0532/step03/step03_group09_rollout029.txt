def generate_block_start(n, rng, target_sum):
    # Generate a block-structured function
    n_blocks = int(rng.integers(3, max(4, n // 40) + 1))
    block_sizes = np.zeros(n_blocks, dtype=int)
    remaining = int(target_sum)
    for i in range(n_blocks - 1):
        bs = int(rng.integers(1, max(2, remaining // (n_blocks - i) + 1)))
        block_sizes[i] = bs
        remaining -= bs
    block_sizes[-1] = remaining
    h = np.zeros(n)
    pos = 0
    for i in range(n_blocks):
        if i % 2 == 0:  # "on" block
            end = min(pos + int(block_sizes[i]), n)
            h[pos:end] = 1.0
            pos = end
        else:
            pos += int(block_sizes[i])
        if pos >= n:
            break
    return project_onto_constraint(h, target_sum)

def gen_perturb(h_best, rng, n0, target_sum, strategy):

        # Small smooth perturbation
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:
        # Subset perturbation
        h_p = h_best.copy()
        idx = rng.choice(n0, size=max(3, n0 // 8), replace=False)
        h_p[idx] = np.clip(h_p[idx] + 0.2 * rng.standard_normal(len(idx)), 0, 1)
        return project_onto_constraint(h_p, target_sum)
    elif strategy == 2:
        # Block swap
        max_len = max(2, n0 // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
        # Binary mix
        h_bin = generate_binary_start(n0, rng, target_sum)
        alpha = float(rng.uniform(0.3, 0.7))
        return project_onto_constraint(alpha * h_best + (1 - alpha) * h_bin, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_obj(h, eps):
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_max_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h)
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def topk_subgrad(h, k, temp):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    topk_idx = np.argsort(corr)[-k:]
    xc = corr[topk_idx]
    xc = xc - np.max(xc)
    w = np.exp(xc / temp)
    w = w / np.sum(w)
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(len(topk_idx)):
        wk = w[i]
        if wk < 1e-14:
            continue
        kidx = topk_idx[i]
        s = kidx - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

def slsqp_optimize(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n

    def cfun(x):
        return float(np.sum(x) - target_sum)

    def cjac(x):
        return np.ones(n)

    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def generate_binary_start(n, rng, target_sum):
    h = np.zeros(n)
    indices = rng.choice(n, size=int(target_sum), replace=False)
    h[indices] = 1.0
    return h

def gen_perturb(h_best, rng, n0, target_sum, strategy):
    if strategy == 0:
        return project_onto_constraint(h_best + 0.02 * rng.standard_normal(n0), target_sum)
    elif strategy == 1:

    elif strategy == 2:
        max_len = max(2, n0 // 6)
        blen = int(rng.integers(2, max_len + 1))
        a = int(rng.integers(0, n0 - blen + 1))
        b = int(rng.integers(0, n0 - blen + 1))
        if abs(a - b) >= blen:
            h_p = h_best.copy()
            blk = h_p[a:a + blen].copy()
            h_p[a:a + blen] = h_p[b:b + blen]
            h_p[b:b + blen] = blk
            return h_p
        return project_onto_constraint(h_best + 0.03 * rng.standard_normal(n0), target_sum)
    elif strategy == 3:
        h_bin = generate_binary_start(n0, rng, target_sum)

    elif strategy == 4:

def proj_grad_anneal(h, n_steps, time_limit, start_time, k_init, k_final,
                     temp_init, temp_final, lr_init, lr_final, rng):
    n = len(h)
    target_sum = n / 2.0
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5

    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        t = step / max(1, n_steps - 1)
        k_use = max(1, int(round(k_init + (k_final - k_init) * t)))
        temp_use = temp_init * (temp_final / temp_init) ** t
        lr_use = lr_init * (lr_final / lr_init) ** t

        grad = topk_subgrad(h_cur, k_use, temp_use)
        h_new = project_onto_constraint(h_cur - lr_use * grad, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < cur_c5:
            h_cur = h_new
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            delta = c5_new - cur_c5
            anneal_temp = 0.002 * (1.0 - t) + 1e-7
            if rng.random() < np.exp(-delta / anneal_temp):
                h_cur = h_new
                cur_c5 = c5_new
            else:
                lr_small = lr_use * 0.3
                h_cur = project_onto_constraint(h_cur - lr_small * grad, target_sum)

    return best_h, best_c5
