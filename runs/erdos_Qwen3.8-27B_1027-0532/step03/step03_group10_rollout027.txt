    # Actually: d/dh[j] sum_i h[i+s] * (1-h[i])
    # Term where h[i+s] = h[j]: i = j-s, so contribution is (1-h[j-s]) if 0<=j-s<n
    # Term where (1-h[i]) = (1-h[j]): i = j, contribution is -h[j+s] if 0<=j+s<n
    # So: d corr[k]/d h[j] = (1-h[j-s]) * [0<=j-s<n] - h[j+s] * [0<=j+s<n]
    # where s = k - (n-1)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        # d corr[k] / d h[j] = (1-h[j-s]) if 0<=j-s<n minus h[j+s] if 0<=j+s<n
        # For all j: contribution from first term is (1-h[j-s]) for j in [s, s+n-1] (i.e., j-s in [0,n-1])
        # Contribution from second term is -h[j+s] for j in [-s, n-1-s] (i.e., j+s in [0,n-1])
        if s >= 0:
            # j-s in [0,n-1] means j in [s, n-1], so j from s to n-1, count = n-s
            m = n - s
            # (1-h[j-s]) for j = s,...,n-1: this is (1-h[0]), (1-h[1]), ..., (1-h[n-1-s])
            grad[s:] += wk * (1.0 - h[:m])
            # -h[j+s] for j in [0, n-1-s], count = n-s
            grad[:m] -= wk * h[s:]
        else:
            r = -s
            m = n - r
            # j-s = j+r in [0,n-1] means j in [0, n-1-r], count = n-r = m
            # (1-h[j+r]) for j = 0,...,m-1: (1-h[r]), (1-h[r+1]), ..., (1-h[n-1])
            grad[:m] += wk * (1.0 - h[r:])
            # -h[j+s] = -h[j-r] for j in [r, n-1], count = n-r = m
            grad[r:] -= wk * h[:m]
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def compute_c5_raw(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def to_binary(h, target_ones=None):
    n = len(h)
    if target_ones is None:
        target_ones = n // 2
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:target_ones]] = 1.0
    return h_bin

def upsample_rep(h, factor):
    return np.repeat(h, factor)

def downsample_block(h, factor):
    n = len(h)
    m = n // factor
    return h[:m * factor].reshape(m, factor).mean(axis=1)

def smooth_objective_fine(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_gradient_fine(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full') * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * (1.0 - h[r:])
            grad[r:] -= wk * h[:m]
    return grad * dx

    h_fine = upsample_rep(x, r)
    return smooth_objective_fine(h_fine, eps)

def coarse_gradient(x, n_fine, r, eps):
    h_fine = upsample_rep(x, r)
    gf = smooth_gradient_fine(h_fine, eps)
    m = len(x)
    grad_coarse = np.zeros(m)
    for j in range(m):
        grad_coarse[j] = np.sum(gf[j * r:(j + 1) * r])
    return grad_coarse

def coarse_constraint_fun(x, m):
    return float(np.sum(x) - m / 2.0)

def coarse_constraint_jac(x, m):
    return np.ones(m)

def slsqp_optimize_fine(h_init, eps, maxiter):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    def cfun(h):
        return float(np.sum(h) - target_sum)
    def cjac(h):
        return np.ones(len(h))
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        result = minimize(smooth_objective_fine, h0, args=(eps,),
                          jac=smooth_gradient_fine, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        h_opt = project_onto_constraint(result.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def slsqp_optimize_coarse(h_coarse_init, n_fine, r, eps, maxiter):

        return np.ones(m)

    # Phase 4: Try n=240 again from best n=180
    if remaining_time() > 150 and len(best_h) == n0:
        old_x = np.linspace(0, 2, n0, endpoint=False) + 2.0 / (2 * n0)
        new_x = np.linspace(0, 2, n240, endpoint=False) + 2.0 / (2 * n240)
        h240b = np.interp(new_x, old_x, best_h)
        h240b = project_onto_constraint(h240b, n240 / 2.0)
        h240b_bin = to_binary(h240b, n240 // 2)
        if remaining_time() > 100:
            h240b_sa, raw_max240 = fast_sa(h240b_bin, 200000, 0.3, 0.005, rng, use_2swap=True)
            c5_240b = (2.0 / n240) * raw_max240
            if c5_240b < best_c5:
                best_c5 = c5_240b
                best_h = h240b_sa.copy()
    
    # Phase 5: Continuous refinement
    if remaining_time() > 80:
        n_best = len(best_h)
        h_polish = best_h.copy()
        if np.all((h_polish == 0) | (h_polish == 1)):
            h_polish = h_polish + 0.03 * rng.standard_normal(n_best)
            h_polish = project_onto_constraint(h_polish, n_best / 2.0)
        else:
            h_polish = h_polish + 0.01 * rng.standard_normal(n_best)
            h_polish = project_onto_constraint(h_polish, n_best / 2.0)
        
        for eps in [0.005, 0.001, 0.0003, 0.0001]:
            if remaining_time() < 15:
                break
            h_polish, c5_p = slsqp_optimize_fine(h_polish, eps, 200)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_polish.copy()
    
    # Final: ensure feasibility
    final_h = project_onto_constraint(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(final_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    
    return final_h, final_c5, len(final_h)
