def project_feasible(h, target_sum):

def smoothed_c5_and_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n

    n_half = n - 1
    
    c_max = np.max(corr)
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    log_sum = c_max + np.log(np.sum(exp_s)) / alpha
    obj = log_sum * dx
    
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)

        if w < 1e-14:

            grad[m:] += wk * (1.0 - h[:n-m])
