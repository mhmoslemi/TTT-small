```python
def compute_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / np.sum(exp_c)  # length 2n-1
    
    # Build the gradient matrix: G[i, m] = dc[m]/dh[i]
    # Then grad = G^T @ p
    # dc[m]/dh[i] where k = m - (n-1):
    #   (1-h[i+k]) if 0<=i+k<n else 0
    #   -h[i-k] if 0<=i-k<n else 0
    
    # Vectorized: for each m, k = m-(n-1)
    # G[i, m] = (1-h[i+k_m]) * 1[0<=i+k_m<n] - h[i-k_m] * 1[0<=i-k_m<n]
    
    # M = k values, shape (2n-1,)
    k_vals = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    
    # For each m (with k = k_vals[m]):
    # Term 1: (1-h[i+k]) for i where 0<=i+k<n, i.e., -k <= i < n-k
    # Term 2: -h[i-k] for i where 0<=i-k<n, i.e., k <= i < n+k
    
    grad = np.zeros(n)
    for m in range(2*n-1):
        if p[m] < 1e-15:
            continue
        k = k_vals[m]
        # Term 1: (1-h[i+k])
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += p[m] * (1.0 - h[lo1+k:hi1+k])
        # Term 2: -h[i-k]
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] += p[m] * (-h[lo2-k:hi2-k])
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    lo = -max(h.max(), 0.0) - 1.0
    hi = 1.0 - min(h.min(), 1.0) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)

def smooth_max_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    return float(c_max + np.log(np.sum(np.exp(alpha * (c - c_max)))) / alpha)

def smooth_max_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    p = exp_c / np.sum(exp_c)
    
    k_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        if p[m] < 1e-16:
            continue
        k = k_vals[m]
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += p[m] * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] += p[m] * (-h[lo2 - k:hi2 - k])
    return grad

def fista(h0, n, target, alpha, max_iter, L=None, tol=1e-14):
    if L is None:
        L = 2.0 + alpha * 0.5
    h = h0.copy()
    h_prev = h.copy()
    t = 1.0
    
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad = smooth_max_grad(y, alpha)
        h_new = project_feasible(y - grad / L, target)
        
        # Check for improvement (optional, for safety)
        h_prev = h
        h = h_new
        t = t_new
    
    return h

def slsqp_smooth(h, n, target, alpha, maxiter=2000, ftol=1e-16):
    def obj(x):
        return smooth_max_obj(x, alpha)
    def cons(x):
        return np.sum(x) - target
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=constraints, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except:
        return project_feasible(h.copy(), target)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n = len(h_init)
    target = n / 2.0
    
    best_h = project_feasible(h_init, target)
    best_score = compute_c5(best_h)

        iters = min(3000, int(tleft() / 0.5))
        h_cur = fista(best_h, n, target, a, iters)
        sc = compute_c5(h_cur)
        if sc < best_score:

            best_h = h_cur.copy()
    
    # Phase 2: Higher alpha FISTA
    alphas_p2 = [1000, 3000, 10000]
    for a in alphas_p2:
        if tleft() < 40:

        iters = min(5000, int(tleft() / 0.3))
        h_cur = fista(best_h, n, target, a, iters)
        sc = compute_c5(h_cur)
        if sc < best_score:
            best_score = sc
            best_h = h_cur.copy()
    
    # Phase 3: Random perturbations

    for trial in range(n_trials):
        if tleft() < 60:

        scale = 0.05 + 0.1 * (trial / n_trials)
        h_pert = best_h + rng.randn(n) * scale
        h_pert = project_feasible(h_pert, target)
        # Quick FISTA

        sc = compute_c5(h_pert)
        if sc < best_score:
```

```python
def compute_c5(h):
    n = len(h)
```

    return float(np.max(c)) * dx


def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-13:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h + lam, 0.0, 1.0)


def smooth_max_obj(h, alpha):
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    return c_max + float(np.log(np.sum(np.exp(shifted)))) / alpha


def smooth_max_grad(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = float(np.max(c))
    shifted = alpha * (c - c_max)
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_c = np.exp(shifted)
    p = exp_c / np.sum(exp_c)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-16:
            continue
        k = k_vals[m]
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if lo1 < hi1:
            grad[lo1:hi1] += pm * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if lo2 < hi2:
            grad[lo2:hi2] += pm * (-h[lo2 - k:hi2 - k])
    return grad

    if L is None:

    n = len(h0)
    h = h0.copy()
    h_prev = h.copy()
    t = 1.0
    for it in range(max_iter):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        grad = smooth_max_grad(y, alpha)
        h_new = project_feasible(y - grad / L, target)
        h_prev = h
        h = h_new
        t = t_new
    return h
