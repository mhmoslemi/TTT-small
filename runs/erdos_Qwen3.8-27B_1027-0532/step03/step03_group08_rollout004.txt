def _fix_sum(h, target_sum):

    """Gradient of the correlation at index k_idx with respect to h."""
    n = len(h)

            #           - h[j+lag] (from -h[j]*h[j+lag]) for j < m  
            #           - h[j-lag] (from h[j-lag]*(-h[j])) for j >= lag
            if lag == 0:
                grad[:m] = 1.0 - 2.0 * h[:m]

    else:

            grad[:m] = grad[:m] - h[j:j+m]  # from -h[i+j]
    return grad * dx

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    """Differential Evolution optimizer."""
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.3:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif r < 0.5:
            pop[i] = _fix_sum(h0 + 0.2 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.7:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif r < 0.85:
            pop[i] = _fix_sum(1.0 - h0 + 0.05 * (rng.rand(n) - 0.5), target_sum)
        else:
            # Block swap
            pop[i] = h0.copy()
            bs = max(2, n // 8)
            s1 = rng.randint(0, n - 2 * bs)
            s2 = rng.randint(0, n - 2 * bs)
            pop[i][s1:s1+bs], pop[i][s2:s2+bs] = pop[i][s2:s2+bs].copy(), pop[i][s1:s1+bs].copy()
            pop[i] = _fix_sum(pop[i], target_sum)
    
    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()
    
    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (gen / max(1, max_gen))
        improved_this_gen = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            while r1 == i:
                r1 = rng.randint(pop_size)
            while r2 == i or r2 == r1:
                r2 = rng.randint(pop_size)
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            cross_mask = (rng.random(n) < CR)
            cross_mask[j_rand] = True
            trial[cross_mask] = mutant[cross_mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            trial_val = _c5_of(trial)
            if trial_val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = trial_val
                improved_this_gen = True
                if trial_val < best_val:
                    best_val = trial_val
                    best_h = trial.copy()
                    best_idx = i
        # Diversity maintenance
        if not improved_this_gen:
            for idx in range(pop_size // 2):
                t_mix = rng.uniform(0.1, 0.5)
                pop[idx] = _fix_sum(t_mix * best_h + (1.0 - t_mix) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = idx
    
    return best_h, best_val

def _sa_optimize(h0, n, target_sum, max_iters, T_start, T_end, deadline, rng):
    """Simulated annealing with adaptive perturbation size."""
    h = h0.copy()
    h_val = _c5_of(h)
    best_h = h.copy()
    best_val = h_val
    n_pert_start = max(2, n // 6)
    n_pert_end = max(1, n // 40)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = float(it) / float(max_iters)
        T = T_start * (T_end / T_start) ** frac
        n_pert = int(n_pert_start + (n_pert_end - n_pert_start) * frac)
        n_pert = max(1, n_pert)
        
        h_new = h.copy()
        indices = rng.choice(n, n_pert, replace=False)
        for idx in indices:
            r = rng.random()
            if r < 0.25:
                h_new[idx] = 0.0
            elif r < 0.5:
                h_new[idx] = 1.0
            else:
                h_new[idx] = rng.random()
        h_new = _fix_sum(h_new, target_sum)
        new_val = _c5_of(h_new)
        
        if new_val < h_val or rng.random() < np.exp(-(new_val - h_val) / max(T, 1e-12)):
            h = h_new
            h_val = new_val
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()

def _corr_vals(h):

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

            else:
                grad[:m] = 1.0 - h[lag:lag + m]

        alpha = min(alpha * 1.3, 3.0)

        cv = _c5_of(x)
        if cv < best_val:
            best_val = cv
            best = x.copy()
    return best, best_val

def _coord_binary(h0, n, target_sum, max_rounds=10):
    h = h0.copy()
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            h[j] = 0.0
            c0 = _c5_of(h)
            h[j] = 1.0
            c1 = _c5_of(h)
            h[j] = old
            co = _c5_of(h)
            if min(c0, c1) < co - 1e-12:
                h[j] = 0.0 if c0 <= c1 else 1.0
                improved = True
        if not improved:
            break
    return h

def _coord_continuous(h0, n, target_sum, max_rounds=4):
    h = h0.copy()
    d = 2.0 / n
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            co = _c5_of(h)
            best_v = old
            best_c = co
            candidates = []
            candidates.append(0.0)
            candidates.append(1.0)
            if old > 0.05:
                candidates.append(old * 0.5)
                candidates.append(old - 0.1)
                candidates.append(old - 0.2)
            if old < 0.95:
                candidates.append(old * 1.5)
                candidates.append(old + 0.1)
                candidates.append(old + 0.2)
            candidates.append(0.5)
            for cv in candidates:
                cv = max(0.0, min(1.0, cv))
                h[j] = cv
                c = _c5_of(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cv
            h[j] = best_v
            if best_v != old:
                improved = True
        if not improved:
            break
    return _fix_sum(h, target_sum)

def _de_optimize(h0, n, target_sum, pop_size, max_gen, F_start, F_end, CR, deadline, rng):
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        r = rng.random()
        if r < 0.25:
            pop[i] = _fix_sum(rng.rand(n), target_sum)
        elif r < 0.45:
            pop[i] = _fix_sum(h0 + 0.15 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.6:
            t = rng.uniform(0.3, 0.7)
            pop[i] = _fix_sum(t * h0 + (1.0 - t) * rng.rand(n), target_sum)
        elif r < 0.75:
            pop[i] = _fix_sum(1.0 - h0 + 0.03 * (rng.rand(n) - 0.5), target_sum)
        elif r < 0.9:
            pop[i] = h0.copy()
            bs = max(2, n // 10)
            s1 = rng.randint(0, max(1, n - 2 * bs))
            s2 = rng.randint(0, max(1, n - 2 * bs))
            tmp = pop[i][s1:s1 + bs].copy()
            pop[i][s1:s1 + bs] = pop[i][s2:s2 + bs]
            pop[i][s2:s2 + bs] = tmp
            pop[i] = _fix_sum(pop[i], target_sum)
        else:
            freq = rng.randint(2, max(3, n // 20))
            pop[i] = 0.5 + 0.4 * np.sign(np.sin(np.arange(n) * 2.0 * np.pi * freq / n + rng.uniform(0, 2 * np.pi)))
            pop[i] = _fix_sum(np.clip(pop[i], 0, 1), target_sum)

    pop_vals = np.array([_c5_of(pop[i]) for i in range(pop_size)])
    best_idx = int(np.argmin(pop_vals))
    best_val = pop_vals[best_idx]
    best_h = pop[best_idx].copy()

    for gen in range(max_gen):
        if time.time() > deadline:
            break
        F = F_start + (F_end - F_start) * (gen / max(1, max_gen))
        improved = False
        for i in range(pop_size):
            if time.time() > deadline:
                break
            r1 = rng.randint(pop_size)
            r2 = rng.randint(pop_size)
            while r1 == i or r1 == r2:
                r1 = rng.randint(pop_size)
            while r2 == i or r2 == r1:
                r2 = rng.randint(pop_size)
            mutant = pop[best_idx] + F * (pop[r1] - pop[r2])
            trial = pop[i].copy()
            j_rand = rng.randint(n)
            mask = (rng.random(n) < CR)
            mask[j_rand] = True
            trial[mask] = mutant[mask]
            trial = _fix_sum(np.clip(trial, 0.0, 1.0), target_sum)
            tv = _c5_of(trial)
            if tv < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = tv
                improved = True
                if tv < best_val:
                    best_val = tv
                    best_h = trial.copy()
                    best_idx = i
        if not improved:
            for idx in range(pop_size // 2):
                tm = rng.uniform(0.1, 0.4)
                pop[idx] = _fix_sum(tm * best_h + (1.0 - tm) * rng.rand(n), target_sum)
                pop_vals[idx] = _c5_of(pop[idx])
                if pop_vals[idx] < best_val:
                    best_val = pop_vals[idx]
                    best_h = pop[idx].copy()
                    best_idx = idx
    return best_h, best_val

def _sa_optimize(h0, n, target_sum, max_iters, T_start, T_end, deadline, rng):
    h = h0.copy()
    h_val = _c5_of(h)
    best_h = h.copy()
    best_val = h_val
    np_start = max(2, n // 5)
    np_end = max(1, n // 30)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = float(it) / float(max_iters)
        T = T_start * (T_end / T_start) ** frac
        np_pert = max(1, int(np_start + (np_end - np_start) * frac))
        h_new = h.copy()
        indices = rng.choice(n, np_pert, replace=False)
        for idx in indices:
            r = rng.random()
            if r < 0.2:
                h_new[idx] = 0.0
            elif r < 0.4:
                h_new[idx] = 1.0
            else:
                h_new[idx] = rng.random()
        h_new = _fix_sum(h_new, target_sum)
        nv = _c5_of(h_new)
        if nv < h_val or rng.random() < np.exp(-(nv - h_val) / max(T, 1e-12)):
            h = h_new
            h_val = nv
            if h_val < best_val:
                best_val = h_val
                best_h = h.copy()
    return best_h, best_val

def _refine_aggressive(h0, n, target_sum, deadline, rng):
    h_best = h0.copy()
    c5_best = _c5_of(h0)

    if time.time() < deadline - 80:
        h1, c1 = _nesterov_topm(h_best, n, target_sum, 400, 0.5, m=8, deadline=deadline - 70)
        if c1 < c5_best:
            c5_best = c1
            h_best = h1

    if time.time() < deadline - 65:
        h2 = _slsqp_active(h_best, n, target_sum, alpha=200, top_m=5, maxiter=3000)
        c2 = _c5_of(h2)
        if c2 < c5_best:
            c5_best = c2
            h_best = h2

    if time.time() < deadline - 50:
        h3, c3 = _nesterov_bt(h_best, n, target_sum, 300, deadline=deadline - 45, m=7)
        if c3 < c5_best:
            c5_best = c3
            h_best = h3

    if time.time() < deadline - 40:
        h4 = _slsqp_active(h_best, n, target_sum, alpha=800, top_m=5, maxiter=4000)
        c4 = _c5_of(h4)
        if c4 < c5_best:
            c5_best = c4
            h_best = h4

    if time.time() < deadline - 25:
        h5 = _coord_binary(h_best, n, target_sum, max_rounds=8)
        c5_ = _c5_of(h5)
        if c5_ < c5_best:
            c5_best = c5_
            h_best = h5

    if time.time() < deadline - 15:
        h6 = _slsqp_active(h_best, n, target_sum, alpha=2000, top_m=4, maxiter=5000)
        c6 = _c5_of(h6)
        if c6 < c5_best:
            c5_best = c6
            h_best = h6

    if time.time() < deadline - 8:
        h7 = _coord_continuous(h_best, n, target_sum, max_rounds=3)
        c7 = _c5_of(h7)
        if c7 < c5_best:
            c5_best = c7
            h_best = h7

    if time.time() < deadline - 4:
        h8 = _slsqp_full(h_best, n, target_sum, alpha=5000, maxiter=5000)
        c8 = _c5_of(h8)
        if c8 < c5_best:
            c5_best = c8
            h_best = h8

    if time.time() < deadline - 1:
        h9 = _coord_binary(h_best, n, target_sum, max_rounds=5)
        c9 = _c5_of(h9)
        if c9 < c5_best:
            c5_best = c9
            h_best = h9

    return h_best, c5_best

def _perturb_refine(h0, n, target_sum, deadline, rng, n_perturb=10):
    best_h = h0.copy()
    best_c5 = _c5_of(h0)
    for _ in range(n_perturb):
        if time.time() > deadline:
            break
        h_p = h0.copy()
        np_ = rng.randint(3, max(4, n // 10))
        indices = rng.choice(n, np_, replace=False)
        for idx in indices:
            h_p[idx] = rng.random()
        h_p = _fix_sum(h_p, target_sum)
        h_r = _slsqp_active(h_p, n, target_sum, alpha=500, top_m=5, maxiter=2000)
        cr = _c5_of(h_r)
        if cr < best_c5:
            best_c5 = cr
            best_h = h_r
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 15
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    else:
        n_init = 150
        h_init = _fix_sum(0.5 + 0.3 * np.sin(np.arange(n_init) * 2 * np.pi * 3 / n_init), n_init / 2.0)

    best_c5 = _c5_of(h_init)
    best_h = h_init.copy()
    best_n = n_init

    n_list = []
    for d in [0, -2, 2, -4, 4, -6, 6, -10, 10, -14, 14, -20, 20, -30, 30, -40, 40, 50]:
        nt = n_init + d
        if nt % 2 == 0 and 80 <= nt <= 250:
            n_list.append(nt)
    n_list = sorted(set(n_list), key=lambda x: abs(x - n_init))

    for n in n_list:
        if time.time() > deadline - 150:
            break
        target_sum = n / 2.0

        if n == n_init:
            h_start = h_init.copy()
        else:
            h_start = np.interp(np.linspace(0, n_init - 1, n), np.arange(n_init), h_init)
            h_start = _fix_sum(np.clip(h_start, 0.0, 1.0), target_sum)

        if best_n != n and best_h is not None:
            h_start2 = np.interp(np.linspace(0, best_n - 1, n), np.arange(best_n), best_h)
            h_start2 = _fix_sum(np.clip(h_start2, 0.0, 1.0), target_sum)
            if _c5_of(h_start2) < _c5_of(h_start):
                h_start = h_start2

        c5_start = _c5_of(h_start)

        # Quick SLSQP to see potential
        if time.time() < deadline - 100:
            h_q = _slsqp_active(h_start, n, target_sum, alpha=300, top_m=5, maxiter=2000)
            c5_q = _c5_of(h_q)
            if c5_q < c5_start:
                h_start = h_q
                c5_start = c5_q

        # DE exploration
        if time.time() < deadline - 80 and c5_start < best_c5 + 0.005:
            h_de, c5_de = _de_optimize(h_start, n, target_sum,
                                       pop_size=35, max_gen=80,
                                       F_start=0.8, F_end=0.3, CR=0.6,
                                       deadline=deadline - 80, rng=rng)
            if c5_de < c5_start:
                h_start = h_de
                c5_start = c5_de

        # Full refinement
        if c5_start < best_c5 + 0.003 and time.time() < deadline - 50:
            h_ref, c5_ref = _refine_aggressive(h_start, n, target_sum, deadline, rng)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n

        # Perturbation refinement on best
        if best_n == n and time.time() < deadline - 40:
            h_pr, c5_pr = _perturb_refine(best_h, n, target_sum, deadline - 30, rng, n_perturb=5)
            if c5_pr < best_c5:
                best_c5 = c5_pr
                best_h = h_pr.copy()

    # Final push on the best solution
    n_best = best_n
    target = n_best / 2.0

    if time.time() < deadline - 40:
        h_sa, c5_sa = _sa_optimize(best_h, n_best, target, 4000, 0.04, 0.0005, deadline - 30, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa

    if time.time() < deadline - 15:
        h_fin, c5_fin = _refine_aggressive(best_h, n_best, target, deadline - 5, rng)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    if time.time() < deadline - 3:
        h_pol = _slsqp_active(best_h, n_best, target, alpha=8000, top_m=3, maxiter=3000)
        c5_pol = _c5_of(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    best_h = _fix_sum(best_h, best_n / 2.0)
    best_c5 = _c5_of(best_h)
    return (best_h, best_c5, best_n)
