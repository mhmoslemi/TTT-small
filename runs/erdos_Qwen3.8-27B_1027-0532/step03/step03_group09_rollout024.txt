```python
    ex = np.exp(x)
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            m = n - s
            grad[s:] += wk * dx * omh[:m]
            grad[:m] -= wk * dx * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * dx * omh[r:]
            grad[r:] -= wk * dx * h[:m]
    return grad

    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)
    bounds = [(0.0, 1.0)] * n
    
    def cfun(x):
        return float(np.sum(x) - target_sum)
    
    def cjac(x):
        return np.ones(n)
    
    try:
        result = minimize(smooth_max_obj, h0, args=(eps,),
                          jac=smooth_max_grad, method='SLSQP',
                          bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': maxiter, 'ftol': 1e-15})
        if time.time() - start_time > time_limit:
            return h0, compute_c5(h0)
        h_opt = project_onto_constraint(result.x, target_sum)

def round_to_binary(h, target_sum, rng=None):
    """Round h to binary with exact sum = target_sum"""
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    current_sum = h_bin.sum()
    
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        too_many = int(current_sum - target_sum)
        ones = np.where(h_bin == 1)[0]
        # Flip the ones with smallest h values (most "uncertain")
        sorted_ones = ones[np.argsort(h[ones])]
        h_bin[sorted_ones[:too_many]] = 0
    elif current_sum < target_sum:
        # Need to flip some 0s to 1s
        too_few = int(target_sum - current_sum)
        zeros = np.where(h_bin == 0)[0]
        sorted_zeros = zeros[np.argsort(-h[zeros])]
        h_bin[sorted_zeros[:too_few]] = 1
    
    return h_bin
```

```python
def swap_delta(h, a, b, n):
    """Compute the change in correlation for swapping h[a]:0->1 and h[b]:1->0"""
    # Returns a vector of length 2n-1 (same as np.correlate output)
    # corr[k] where k=0 corresponds to lag -(n-1), k=2n-2 corresponds to lag n-1
    
    # For lag s (0-indexed as k = s + n - 1):
    # ΔC_k = [1-h[a+s]]·[0≤a+s<n] - [h[a-s]]·[0≤a-s<n] - [1-h[b+s]]·[0≤b+s<n] + [h[b-s]]·[0≤b-s<n]
    
    # Let's use the index directly:
    # k goes from 0 to 2n-2, corresponding to s = k - (n-1) which goes from -(n-1) to n-1
    # For position a:
    #   "a+s valid" means 0 ≤ a + (k-(n-1)) < n, i.e., n-1-a ≤ k ≤ n-1-a+n-1 = 2n-2-a
    #   "a-s valid" means 0 ≤ a - (k-(n-1)) < n, i.e., n-1-a+n-1 ≤ k ≤ n-1-a... 
    #   Wait: a - s = a - (k-(n-1)) = a + n - 1 - k
    #   0 ≤ a + n - 1 - k < n means 1-a+n-n ≤ k ≤ 1-a+n-0... 
    #   Hmm, let me just compute it directly.
    
    k = np.arange(2*n - 1)
    s = k - (n - 1)
    
    delta = np.zeros(2*n - 1)
    
    a_plus = a + s  # a+s
    a_minus = a - s  # a-s
    b_plus = b + s
    b_minus = b - s
    
    mask_ap = (a_plus >= 0) & (a_plus < n)
    mask_am = (a_minus >= 0) & (a_minus < n)
    mask_bp = (b_plus >= 0) & (b_plus < n)
    mask_bm = (b_minus >= 0) & (b_minus < n)
    
    delta[mask_ap] += 1.0 - h[a_plus[mask_ap]]
    delta[mask_am] -= h[a_minus[mask_am]]
    delta[mask_bp] -= 1.0 - h[b_plus[mask_bp]]
    delta[mask_bm] += h[b_minus[mask_bm]]
    
    return delta
```

```python
def compute_swap_delta_c(h, a, b, s, n):
    """Compute ΔC(s) for swapping h[a]:0->1, h[b]:1->0"""
    # Affected j values
    candidates = set()
    for j in [a, b, a+s, b+s]:
        if 0 <= j < n and 0 <= j-s < n:
            candidates.add(j)
    
    delta = 0.0
    for j in candidates:
        # f(j) = h(j-s) * (1-h(j))
        # f'(j) = h'(j-s) * (1-h'(j))
        # h' differs from h at a and b
        
        old_val = h[j-s] * (1.0 - h[j])
        
        # Compute new value
        js = j - s
        new_h_js = h[js]
        new_h_j = h[j]
        if js == a:
            new_h_js = 1.0
        elif js == b:
            new_h_js = 0.0
        if j == a:
            new_h_j = 1.0
        elif j == b:
            new_h_j = 0.0
        
        new_val = new_h_js * (1.0 - new_h_j)
        delta += new_val - old_val
    
    return delta
```
