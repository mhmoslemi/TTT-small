from scipy.optimize import linprog, minimize
import cvxpy as cp
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h = h.copy()
    h[idx] += diff / len(idx)
    h = np.clip(h, 0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx2 = np.where(h < 1.0)[0]
        else:
            idx2 = np.where(h > 0.0)[0]
        if len(idx2) > 0:
            h[idx2] += residual / len(idx2)
            h = np.clip(h, 0, 1.0)
    return h

def build_constraint_matrix(g, n):
    """Build the A matrix where A[k_idx, i] = g[i+k] if valid.
    k_idx goes from 0 to 2n-2, corresponding to k = k_idx - (n-1)."""
    A = np.zeros((2*n-1, n))
    n_half = n - 1
    for k_idx in range(2*n-1):
        k = k_idx - n_half
        if k >= 0:
            # valid i: 0 ≤ i < n-k
            A[k_idx, :n-k] = g[k:]
        else:
            kk = -k
            # valid i: kk ≤ i < n
            A[k_idx, kk:] = g[:n-kk]
    return A

def alternating_lp_hig(h_init, n, target, max_iter, time_limit, damping=0.5):
    """Alternating LP using scipy HiGHS solver."""
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h_init.copy(), target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        g = 1.0 - h
        A = build_constraint_matrix(g, n)
        
        # LP: min t
        # s.t. A @ h - t ≤ 0  (A @ h ≤ t)
        #      1^T h = target
        #      0 ≤ h ≤ 1
        
        # Variables: [h_0, ..., h_{n-1}, t]
        # Objective: [0, ..., 0, 1]
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints: A @ h - t ≤ 0
        # i.e., [A, -1] @ [h; t] ≤ 0
        A_ub = np.hstack([A, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        
        # Equality: 1^T h = target
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([target])
        
        # Bounds: 0 ≤ h_i ≤ 1, t unbounded (or t ≥ 0)
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'presolve': True})
        except Exception:
            break
        
        if res.status != 0:
            break
        
        h_new = res.x[:n]
        h_new = np.clip(h_new, 0, 1)
        
        h = (1 - damping) * h + damping * h_new
        h = project_feasible(h, target)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
    
    return best_h, best_c5

def compute_smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = alpha * corr
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)
    n_full = 2 * n - 1
    n_half = n - 1

    if time_left() > 20:
        sg_time = min(20, time_left() * 0.5)
        h_sg, c5_sg = subgradient_polish(best_h, n, target, time_limit=sg_time,
                                          max_steps=10000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    # Phase 8: Try higher resolution
    if time_left() > 100:
        for n_try in [600, 800]:
            if time_left() < 80:
                break
            target_try = n_try / 2.0
            h_interp = interpolate_to(best_h, n_try, target_try)
            lp_time = min(60, time_left() * 0.3)
            h_lp, c5_lp = alternating_lp(h_interp, n_try, target_try, time_limit=lp_time)
            if c5_lp < best_score:
                best_score = c5_lp
                best_h = h_lp
                n = n_try
                target = target_try
            if time_left() < 40:
                break

    # Final: subgradient polish at final resolution
    if time_left() > 15:
        n_use = len(best_h)
        target_use = n_use / 2.0
        h_sg, c5_sg = subgradient_polish(best_h, n_use, target_use,
                                          time_limit=min(15, time_left() * 0.6),
                                          max_steps=5000)
        if c5_sg < best_score:
            best_score = c5_sg
            best_h = h_sg

    n_use = len(best_h)
    target_use = n_use / 2.0
    h_final = project_feasible(best_h, target_use)
    final_score = compute_c5(h_final)
    return h_final, final_score, len(h_final)
