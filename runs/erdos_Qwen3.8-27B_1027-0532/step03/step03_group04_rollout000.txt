                    grad[ka:] += w * (1.0 - h_cur[:n-ka])
                    grad[:n-ka] -= w * h_cur[ka:]
        
        total_w = np.sum(weights)
        if total_w > 0:
            grad /= total_w
        
        # Try different step sizes
        found = False
        for lr in [0.005, 0.01, 0.02, 0.05, 0.1]:
            h_try = h_cur - lr * grad
            h_try = project_feasible(h_try, target)
            sc = compute_c5(h_try)
            if sc < best_c5 - 1e-14:
                best_c5 = sc
                best_h = h_try.copy()
                h_cur = h_try
                found = True
                break
        
        if not found:
            break
    
    return best_h, best_c5

def coordinate_descent(h, target, n_sweeps=20, time_limit=60):
    """Coordinate descent: for each coordinate, try to find the optimal value."""
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    
    for sweep in range(n_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            # Compute the effect of changing h[i]
            # For each k, the change in c[k] when h[i] changes to t:
            # dc[k]/dt = (1-h[i+k])*dx (if i+k valid) - h[i-k]*dx (if i-k valid)
            # f(t) = max_k (c[k] + (t - h[i]) * dc[k]/dt)
            # We want to minimize f(t) over t in [0,1]
            
            # Compute dc[k]/dt for all k
            dc = np.zeros(2*n-1)
            for k in range(-(n-1), n):
                coeff = 0.0
                if 0 <= i+k < n:
                    coeff += (1.0 - h_cur[i+k]) * dx
                if 0 <= i-k < n:
                    coeff -= h_cur[i-k] * dx
                dc[k + n - 1] = coeff
            
            # f(t) = max_k (corr[k] + (t - h_cur[i]) * dc[k])
            corr_cur = compute_corr(h_cur)
            # Try several candidate values of t
            candidates = [0.0, 1.0, h_cur[i]]
            # Also try the point where the argmax changes
            # For now, just try a few values around the current
            if 0.01 < h_cur[i] < 0.99:
                candidates.extend([h_cur[i] - 0.1, h_cur[i] + 0.1])
                candidates.extend([h_cur[i] * 0.5, h_cur[i] * 1.5])
            
            best_t = h_cur[i]
            best_f = np.inf
            for t_cand in candidates:
                if t_cand < 0 or t_cand > 1:
                    continue
                f_val = np.max(corr_cur + (t_cand - h_cur[i]) * dc)
                if f_val < best_f:
                    best_f = f_val
                    best_t = t_cand
            
            if best_t != h_cur[i]:
                h_cur[i] = best_t
                h_cur = project_feasible(h_cur, target)
                sc = compute_c5(h_cur)
                if sc < best_c5:
                    best_c5 = sc
                    best_h = h_cur.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def interpolate_to(h_src, n_dst, target_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_feasible(h_src.copy(), target_dst)
    x_new = np.linspace(0.0, 1.0, n_dst)
    x_old = np.linspace(0.0, 1.0, n_src)
    h_out = np.interp(x_new, x_old, h_src)
    return project_feasible(h_out, target_dst)
