```python
import time
import numpy as np
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:
                if diff >= 0:
                    break
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
    return x

def _lse_max(x, alpha):
    m = np.max(x)
    if m < -500:
        return m
    return m + np.log(np.sum(np.exp(alpha * (x - m)))) / alpha

def _grad_corr_term(h, k_idx, dx):
    n = len(h)
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

def _smooth_obj_func(x, alpha, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return _lse_max(corr, alpha) * (2.0 / n)

def _smooth_grad_func(x, alpha, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    total_w = weights.sum()
    if total_w < 1e-300:
        weights = np.zeros(len(corr))
        weights[np.argmax(corr)] = 1.0
    else:
        weights = weights / total_w
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] > 1e-8:
            g += weights[k] * _grad_corr_term(x, k, dx)
    return g * (2.0 / n)

def _eq_con_fun(x):
    return x.sum() - _global_target_sum

def _eq_con_jac(x):
    return np.ones(_global_n)

def _slsqp_opt(h0, n, target_sum, alpha, maxiter):
    global _global_target_sum, _global_n
    _global_target_sum = target_sum
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_con_fun, 'jac': _eq_con_jac}]
    try:
        res = minimize(_smooth_obj_func, h0, jac=_smooth_grad_func,
                       args=(alpha, n),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return _fix_sum(res.x, target_sum)
    except Exception:
        return h0

def _golden_section_2coord(h, j, k, target_sum, n_iters=35):
    """Optimize h[j] with h[k] as balancing coordinate using golden section search."""
    other_sum = target_sum - h[j] - h[k]
    a = max(0.0, target_sum - other_sum - 1.0)
    b = min(1.0, target_sum - other_sum)
    if b - a < 1e-10:
        return h, _c5_of(h)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    h[j] = c
    h[k] = target_sum - other_sum - c
    fc = _c5_of(h)
    
    h[j] = d
    h[k] = target_sum - other_sum - d
    fd = _c5_of(h)
    
    for _ in range(n_iters - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            h[j] = c
            h[k] = target_sum - other_sum - c
            fc = _c5_of(h)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            h[j] = d
            h[k] = target_sum - other_sum - d
            fd = _c5_of(h)
    
    v_best = (a + b) / 2.0
    h[j] = v_best
    h[k] = target_sum - other_sum - v_best
    return h, _c5_of(h)

def _cs_gd_round(h, n, target_sum, k_offset, n_iters, deadline):
    """One round of coordinate descent using golden section on pairs (j, j+k_offset)."""
    for j in range(n):
        if deadline is not None and time.time() > deadline:
            break
        k = (j + k_offset) % n
        if k == j:
            continue
        h, _ = _golden_section_2coord(h, j, k, target_sum, n_iters)
    return h

def _cso_explore(h_best, n, target_sum, pop_size, n_iter, deadline, rng):
    """Cat Swarm Optimization exploration phase."""
    cats = [h_best.copy()]
    fitness = [_c5_of(h_best)]
    
    for i in range(pop_size - 1):
        h_new = h_best + 0.3 * (rng.rand(n) * 2 - 1)
        h_new = _project(h_new, target_sum)
        cats.append(h_new)
        fitness.append(_c5_of(h_new))
    
    best_idx = int(np.argmin(fitness))
    best_h = cats[best_idx].copy()
    best_f = fitness[best_idx]
    stall = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        mode_probs = rng.rand(pop_size)
        new_cats = []
        new_fitness = []
        improved = False
        
        for i in range(pop_size):
            if mode_probs[i] < 0.55:
                # Exploration: Cauchy-based random jump
                jumps = 1.0 / np.abs(rng.cauchy(1.0, size=n)) - 1.0
                jumps = np.clip(jumps, -1.0, 1.0) * 0.2
                h_new = cats[i] + jumps
                h_new = _project(h_new, target_sum)
            else:
                # Recognition: move towards best
                h_new = cats[i] + 0.3 * (best_h - cats[i]) + 0.08 * (rng.rand(n) * 2 - 1)
                h_new = _project(h_new, target_sum)
            
            f_new = _c5_of(h_new)
            
            if f_new < fitness[i]:
                new_cats.append(h_new)
                new_fitness.append(f_new)
                if f_new < best_f:
                    best_f = f_new
                    best_h = h_new.copy()
                    improved = True
            else:
                new_cats.append(cats[i].copy())
                new_fitness.append(fitness[i])
        
        cats = new_cats
        fitness = new_fitness
        best_idx = int(np.argmin(fitness))
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        if stall >= 25:
            break
    
    return best_h, best_f

def _local_perturb(h0, n, target_sum, n_perturbs, deadline, rng):
    """Local perturbation search."""
    best_h = h0.copy()
    best_f = _c5_of(h0)
    
    for _ in range(n_perturbs):
        if time.time() > deadline:
            break
        h_pert = h0.copy()
        n_change = rng.randint(1, max(2, n // 10))
        idx = rng.choice(n, size=n_change, replace=False)
        for i in idx:
            h_pert[i] = np.clip(h_pert[i] + rng.normal(0, 0.06), 0.0, 1.0)
        h_pert = _fix_sum(h_pert, target_sum)
        f_pert = _c5_of(h_pert)
        if f_pert < best_f:
            best_f = f_pert
            best_h = h_pert.copy()
    
    return best_h, best_f
```

```python
from scipy.optimize import minimize

_global_target_sum = 0.0
_global_n = 0

def _c5_of(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / len(h)))

def _fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
        for i in order:
            if diff <= 1e-12:
                break
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    else:
        order = np.argsort(h)
        for i in order:
            if diff >= -1e-12:
                break
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
    return h

def _project(y, S):
    n = len(y)
    lo = float(y.min()) - 1.0
    hi = float(y.max()) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S
    if abs(diff) > 1e-10:
        diff = S - x.sum()
        if diff > 0:
            order = np.argsort(1.0 - x)
            for i in order:
                if diff <= 0:
                    break
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
        else:
            order = np.argsort(x)
            for i in order:

                sub = min(-diff, x[i])

    m = np.max(x)
    if m < -500:

        m = n - lag
        if m > 0:
            grad[lag:lag+m] += (1.0 - h[:m])
            grad[:m] += -h[lag:lag+m]
    else:
        j = -lag
        m = n - j
        if m > 0:
            grad[:m] += (1.0 - h[j:])
            grad[j:] += -h[:m]
    return grad * dx

    dx = 2.0 / n

    max_c = np.max(corr)
    weights = np.exp(alpha * (corr - max_c))
    total_w = weights.sum()
    if total_w < 1e-300:
        g = _grad_corr_term(x, int(np.argmax(corr)), dx)
    else:
        weights = weights / total_w
        g = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] > 1e-7:
                g += weights[k] * _grad_corr_term(x, k, dx)
    return g * (2.0 / n)
```
