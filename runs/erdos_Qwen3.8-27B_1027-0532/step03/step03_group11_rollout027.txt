    # Type 2 (s≠0, 0<=b+s<n): h[b+s]*(1-1) - h[b+s]*(1-0) = 0 - h[b+s] = -h[b+s]

    #
    # BUT: we need to be careful when a-s = b or b-s = a (the "cross terms")
    # When computing the change for h[a], we use h[a-s] which might be h[b] if a-s=b (s=a-b)
    # When computing the change for h[b], we use h[b-s] which might be h[a] if b-s=a (s=b-a)
    #
    # Since we're using the PRE-swap values, we should use h[b]=0 for h[a-s] when a-s=b,
    # and h[a]=1 for h[b-s] when b-s=a.
    # This is correct because we're computing the change from the pre-swap state.
    
    # So the algorithm is:
    # 1. For s in valid range for a (s≠0, 0<=a-s<n): corr[n-1+s] -= (1-h[a-s])
    # 2. For s in valid range for a (s≠0, 0<=a+s<n): corr[n-1+s] += h[a+s]
    # 3. For s in valid range for b (s≠0, 0<=b-s<n): corr[n-1+s] += (1-h[b-s])
    # 4. For s in valid range for b (s≠0, 0<=b+s<n): corr[n-1+s] -= h[b+s]
    # Using pre-swap h values throughout.
    
    # The valid range for s in step 1: 0 <= a-s < n => a-n+1 <= s <= a, s≠0
    # So s ranges from max(1-n, a-n+1) to min(n-1, a), excluding 0
    # Actually s must also satisfy |s| <= n-1 for the correlation index to be valid.
    # corr index = n-1+s, must be in [0, 2n-2], so s in [-(n-1), n-1]
    
    # Let me just iterate over valid s values:
    
    # Step 1: h[a] 1->0, as first factor
    s_min1 = max(1 - n, a - n + 1)  # a-s >= 0 => s <= a; a-s <= n-1 => s >= a-n+1
    s_max1 = min(n - 1, a)
    # Need 0 <= a-s < n => a-n+1 <= s <= a
    # Also need s in [-(n-1), n-1]
    lo1 = max(a - n + 1, -(n-1))
    hi1 = min(a, n-1)
    for s in range(lo1, hi1 + 1):
        if s == 0:
            continue
        idx = n - 1 + s
        new_corr[idx] -= (1.0 - h[a - s])
    
    # Step 2: h[a] 1->0, as second factor
    # Need 0 <= a+s < n => 0 <= s <= n-1-a
    lo2 = max(0, -(n-1))  # s >= 0 from constraint, and s >= -(n-1)
    hi2 = min(n - 1 - a, n - 1)
    for s in range(lo2, hi2 + 1):
        if s == 0:
            continue
        idx = n - 1 + s
        new_corr[idx] += h[a + s]
    
    # Step 3: h[b] 0->1, as first factor
    lo3 = max(b - n + 1, -(n-1))
    hi3 = min(b, n-1)
    for s in range(lo3, hi3 + 1):
        if s == 0:
            continue
        idx = n - 1 + s
        new_corr[idx] += (1.0 - h[b - s])
    
    # Step 4: h[b] 0->1, as second factor
    lo4 = max(0, -(n-1))
    hi4 = min(n - 1 - b, n - 1)
    for s in range(lo4, hi4 + 1):
        if s == 0:
            continue
        idx = n - 1 + s
        new_corr[idx] -= h[b + s]
    
    return new_corr

            for zi in zeros:
                h[oi] = 0.0
                h[zi] = 1.0

                sc = float(np.max(new_corr))
                delta = sc - current
                if delta < best_delta:

                    best_oi = oi
                    best_zi = zi
                    best_new_corr = new_corr.copy()
                h[oi] = 1.0
                h[zi] = 0.0

        h[best_zi] = 1.0
        corr = best_new_corr
        current += best_delta

def corr_delta(h, a, b):
    """
    Compute the change in the full correlation vector when h[a]: 1->0 and h[b]: 0->1.
    Uses pre-swap values of h.
    """
    n = len(h)
    delta = np.zeros(2 * n - 1)
    
    # f[a] 1->0:
    #   As first factor (i=a): delta[n-1+s] -= (1-h[a-s]) for valid s≠0
    #   As second factor (i-s=a): delta[n-1+s] += h[a+s] for valid s≠0
    # f[b] 0->1:
    #   As first factor (i=b): delta[n-1+s] += (1-h[b-s]) for valid s≠0
    #   As second factor (i-s=b): delta[n-1+s] -= h[b+s] for valid s≠0
    
    # First factor for a: s ranges where 0<=a-s<n and s≠0
    # s in [a-n+1, a] excluding 0
    s = np.arange(max(a-n+1, -(n-1)), min(a, n-1) + 1)
    s = s[s != 0]
    if len(s) > 0:
        idx = n - 1 + s
        delta[idx] -= (1.0 - h[a - s])
    
    # Second factor for a: s ranges where 0<=a+s<n and s≠0
    s = np.arange(1, min(n - a, n) )  # s from 1 to n-1-a
    if len(s) > 0:
        idx = n - 1 + s
        delta[idx] += h[a + s]
    
    # First factor for b: s ranges where 0<=b-s<n and s≠0
    s = np.arange(max(b-n+1, -(n-1)), min(b, n-1) + 1)
    s = s[s != 0]
    if len(s) > 0:
        idx = n - 1 + s
        delta[idx] += (1.0 - h[b - s])
    
    # Second factor for b: s ranges where 0<=b+s<n and s≠0
    s = np.arange(1, min(n - b, n))
    if len(s) > 0:
        idx = n - 1 + s
        delta[idx] -= h[b + s]
    
    return delta
