```python
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

                    if k_range[-1] < len(corr) - 1:
```

Actually, minimizing max_s corr_s where corr_s = sum h[i+s]*(1-h[i]) is equivalent to minimizing the maximum "cross-correlation" between h and (1-h) at all shifts. 

For binary h with exactly n/2 ones:
corr_s = sum h[i+s]*(1-h[i]) = sum h[i+s] - sum h[i+s]*h[i]

The second term is the autocorrelation of h at shift s. So:
corr_s = (number of 1s in the overlap region) - (autocorrelation at shift s)

For shift s (0 ≤ s < n), the overlap region has n-s elements. The number of 1s depends on the structure.

For a "random-like" sequence, corr_s ≈ (n-s)/4 for small s, which gives max ≈ n/4.
For a "perfect" sequence, we'd want corr_s to be as equal as possible across all s.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(dx * np.max(corr))

def project_onto_constraint(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h.copy()
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def smooth_max_val(corr, eps):
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    return cmax + eps * np.log(np.sum(np.exp(x)))

def smooth_obj(h, eps):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h) * dx
    return smooth_max_val(corr, eps)

def smooth_grad(h, eps):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    corr = compute_corr(h) * dx
    cmax = np.max(corr)
    x = (corr - cmax) / eps
    x -= np.max(x)
    ex = np.exp(x)
    sexp = np.sum(ex)
    w = ex / sexp
    grad = np.zeros(n)
    half = n - 1
    for k in range(2 * n - 1):
        wk = w[k] * dx
        if abs(wk) < 1e-15:
            continue
        s = k - half
        if s >= 0:
            m = n - s
            grad[s:] += wk * (1.0 - h[:m])
            grad[:m] -= wk * h[s:]
        else:
            r = -s
            m = n - r
            grad[:m] += wk * (1.0 - h[r:])
            grad[r:] -= wk * h[:m]
    return grad

def lbfgs_optimize(h_init, eps, maxiter=400, penalty=500.0):
    n = len(h_init)
    target_sum = n / 2.0
    h0 = project_onto_constraint(h_init.astype(float).copy(), target_sum)

    def obj_pen(h):
        v = smooth_obj(h, eps)
        cv = h.sum() - target_sum
        return v + penalty * cv * cv

    def grad_pen(h):
        g = smooth_grad(h, eps)
        cv = h.sum() - target_sum
        return g + 2.0 * penalty * cv

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h0, jac=grad_pen, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
        h_opt = project_onto_constraint(res.x, target_sum)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

def targeted_binary_sa(h_start, n_steps, t_start, t_end, rng):
    n = len(h_start)
    half = n - 1
    h = h_start.copy().astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_val = current_max
    best_h = h.copy()
    ones_list = list(np.where(h == 1.0)[0])
    zeros_list = list(np.where(h == 0.0)[0])
    n_one = len(ones_list)
    n_zero = len(zeros_list)

    step = 0
    while step < n_steps:
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        worst_k = int(np.argmax(corr))
        worst_s = worst_k - half

        if rng.random() < 0.6 and worst_s != 0:
            if worst_s > 0:
                m = n - worst_s
                cand_i = [x for x in ones_list if x >= worst_s and x - worst_s < m]
                cand_j = [x for x in zeros_list if x < m]
                if not cand_i:
                    cand_i = ones_list
                if not cand_j:
                    cand_j = zeros_list
            else:
                r = -worst_s
                m = n - r
                cand_i = [x for x in ones_list if x < m]
                cand_j = [x for x in zeros_list if x >= r]
                if not cand_i:
                    cand_i = ones_list
                if not cand_j:
                    cand_j = zeros_list
            i = cand_i[rng.integers(len(cand_i))]
            j = cand_j[rng.integers(len(cand_j))]
        else:
            i = ones_list[rng.integers(n_one)]
            j = zeros_list[rng.integers(n_zero)]

        if i == j:
            step += 1
            continue

        s_min = max(-i, i - half, -j, j - half)
        s_max = min(half - i, i, half - j, j)
        if s_min > s_max:
            h[i], h[j] = h[j], h[i]
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            step += 1
            continue

        s_range = np.arange(s_min, s_max + 1)
        k_range = s_range + half
        delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
        corr[k_range] += delta
        new_max = float(np.max(corr))
        d_c5 = new_max - current_max
        if d_c5 < 0 or (t > 1e-12 and rng.random() < np.exp(-d_c5 / t)):
            h[i] = 0.0
            h[j] = 1.0
            oi = ones_list.index(i)
            ones_list[oi] = j
            oz = zeros_list.index(j)
            zeros_list[oz] = i
            current_max = new_max
            if current_max < best_val:
                best_val = current_max
                best_h = h.copy()
        else:
            corr[k_range] -= delta
        step += 1
    return best_h, best_val

def local_search_binary(h_in, max_passes=5, time_limit=5.0):
    n = len(h_in)
    half = n - 1
    h = h_in.copy().astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    t0 = time.time()

    for _pass in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        ones = list(np.where(h == 1.0)[0])
        zeros = list(np.where(h == 0.0)[0])
        rng_local = np.random.default_rng(0)
        rng_local.shuffle(ones)
        rng_local.shuffle(zeros)

        for i in ones:
            if h[i] != 1.0:
                continue
            for j in zeros:
                if h[j] != 0.0:
                    continue
                s_min = max(-i, i - half, -j, j - half)
                s_max = min(half - i, i, half - j, j)
                if s_min > s_max:
                    continue
                s_range = np.arange(s_min, s_max + 1)
                k_range = s_range + half
                delta = (h[i + s_range] + h[i - s_range] - h[j + s_range] - h[j - s_range])
                new_aff_max = float(np.max(corr[k_range] + delta))
                if k_range[0] > 0:
                    unaff_parts = [corr[:k_range[0]]]
                else:
                    unaff_parts = []
                if k_range[-1] < len(corr) - 1:
                    unaff_parts.append(corr[k_range[-1] + 1:])
                if unaff_parts:
                    unaff_max = float(np.max(np.concatenate(unaff_parts)))
                else:
                    unaff_max = -1e18
                new_max = max(new_aff_max, unaff_max)
                if new_max < current_max - 1e-12:
                    corr[k_range] += delta
                    h[i] = 0.0
                    h[j] = 1.0
                    current_max = new_max
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, current_max
```
