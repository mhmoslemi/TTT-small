```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_onto_constraint(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def solve_1d(C, D):
    """Solve min_{x in [0,1]} max_k (C[k]*x + D[k])
    C, D are 1D arrays of same length.
    Returns optimal x.
    """
    # g(x) = max(C*x + D) is convex piecewise-linear on [0,1]
    # Use grid + golden section refinement
    
    # Evaluate on a coarse grid
    n_grid = 32
    x_grid = np.linspace(0.0, 1.0, n_grid)
    # g_vals[i] = max_k (C[k]*x_grid[i] + D[k])
    # = max_k (C[k]*x_grid[i] + D[k]) for each i
    # Vectorized: C[:, None] * x_grid[None, :] + D[:, None], then max over axis 0
    g_vals = np.max(C[:, None] * x_grid[None, :] + D[:, None], axis=0)
    idx = np.argmin(g_vals)
    
    if idx == 0:
        return 0.0
    if idx == n_grid - 1:
        return 1.0
    
    # Golden section in [x_grid[idx-1], x_grid[idx+1]]
    lo = x_grid[idx - 1]
    hi = x_grid[idx + 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ≈ 0.618
    
    for _ in range(40):
        m1 = hi - gr * (hi - lo)
        m2 = lo + gr * (hi - lo)
        g1 = np.max(C * m1 + D)
        g2 = np.max(C * m2 + D)
        if g1 < g2:
            hi = m2
        else:
            lo = m1
    
    return (lo + hi) / 2.0

def coord_descent(h, lam, n_sweeps=10):
    """Coordinate descent with Lagrange multiplier lam.
    Minimize max_k corr_k(h) + lam * sum(h)
    """
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    for sweep in range(n_sweeps):
        for j in range(n):
            # Save current value
            h[j] = 0.0
            # Compute B = corr when h[j] = 0
            B = np.correlate(h, 1.0 - h, mode='full') * dx
            
            # Compute A = corr(h[j]=1) - corr(h[j]=0)
            h[j] = 1.0
            B1 = np.correlate(h, 1.0 - h, mode='full') * dx
            A = B1 - B
            
            # Solve 1D: min_{x in [0,1]} max_k (A[k]*x + B[k]) + lam*x
            C = A + lam  # slope
            D = B         # intercept
            x_opt = solve_1d(C, D)
            h[j] = x_opt
    
    return h

def bisection_lam(h_init, target_sum, n_outer=30, n_inner_sweeps=3):
    """Find λ such that coordinate descent gives sum(h) ≈ target_sum."""
    n = len(h_init)
    h = h_init.copy()
    lam_lo = -5.0
    lam_hi = 5.0
    
    # First, get sum at lam_lo and lam_hi
    h_lo = coord_descent(h, lam_lo, n_inner_sweeps)
    s_lo = np.sum(h_lo)
    h_hi = coord_descent(h, lam_hi, n_inner_sweeps)
    s_hi = np.sum(h_hi)
    
    # We want sum = target_sum
    # Larger λ → smaller sum (since we're adding λ*sum(h) to the objective)
    # So if s_lo > target_sum > s_hi, we're good
    # If both > target, need larger λ
    # If both < target, need smaller λ
    
    for _ in range(n_outer):
        lam = (lam_lo + lam_hi) / 2.0
        h = coord_descent(h, lam, n_inner_sweeps)
        s = np.sum(h)
        if s > target_sum:
            lam_lo = lam
        else:
            lam_hi = lam
    
    return h, (lam_lo + lam_hi) / 2.0

def binary_local_search(h, max_rounds=1000, time_limit=60, start_time=0):
    """First-improvement binary local search via swaps."""

    h = h.copy().astype(float)
    # Ensure binary

    order = np.argsort(np.abs(h - 0.5))
    h = np.zeros(n)
    h[order[n//2:]] = 1.0
    
    current = compute_c5(h)
    
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        improved = False
        # Try random swaps
        for _ in range(200):
            oi = ones[np.random.default_rng().integers(len(ones))]
            zi = zeros[np.random.default_rng().integers(len(zeros))]
            h[oi] = 0.0
            h[zi] = 1.0
            sc = compute_c5(h)
            if sc < current - 1e-15:
                current = sc
                # Update lists
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
                improved = True
                break
            else:
                h[oi] = 1.0
                h[zi] = 0.0
        if not improved:
            break
    return h, current
```

Wait, I have a concern. The coordinate descent with Lagrange dual might not be as effective as SLSQP for this problem. The issue is that the Lagrange dual approach might have a duality gap, and the coordinate descent might converge slowly.

Let me think about alternatives:

1. **Direct approach without Lagrange**: Instead of using a Lagrange multiplier, we can enforce the constraint directly by "pairing" coordinates. When we update h[j], we also update h[k] (for some fixed k ≠ j) to maintain the sum. This is "paired coordinate descent."

2. **Feasible direction method**: At each step, find a feasible direction (one that maintains the sum constraint) and take a step in that direction.

3. **Projection-based**: Do coordinate descent without the constraint, then project back to the constraint after each sweep.

Option 3 is simplest. Let me try that:
- Run coordinate descent ignoring the sum constraint (λ=0)
- After each sweep, project to the constraint
- Repeat

This might work well because the projection is cheap (O(n log n) via the bisection method).

Actually, let me reconsider. The objective max_k corr_k(h) is already quite "flat" in the direction that changes the sum. The constraint sum(h) = n/2 is not very active at the optimum (the optimal h likely has values well within [0,1]). So the projection approach should work fine.

```python
def coord_descent_feasible(h, n_sweeps=20):
    """Coordinate descent with projection to maintain sum constraint."""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    h = project_onto_constraint(h, target_sum)
    
    for sweep in range(n_sweeps):
        for j in range(n):
            h[j] = 0.0
            B = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            h[j] = 1.0
            B1 = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
            A = B1 - B
            # min_{x in [0,1]} max_k (A[k]*x + B[k])
            x_opt = solve_1d(A, B)
            h[j] = x_opt
        # Project to maintain constraint
        h = project_onto_constraint(h, target_sum)
    
    return h
```

This is simpler and should work well. The projection after each sweep ensures we stay feasible.
