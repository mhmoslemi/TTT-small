    """Project x onto {h: 0<=h_i<=1, sum(h_i)=S} using bisection"""
    lo, hi = -max(x), -min(x) + 1  # bounds on lambda

    # sum(clip(x_i - lambda, 0, 1)) = S

    # At lambda = -min(x) + 1: all h_i = x_i - lambda <= min(x) + min(x) - 1...

    # We need f(lambda) = S.

    # If lambda >= max(x_i): all h_i <= 0, sum = 0

        else:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr


def project_sum_bound(x, S):
    lo = np.min(x) - 1.0 - 1.0
    hi = np.max(x) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        h = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-14:
            break
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)


def grad_lag_array(h, corridor):
    n = len(h)
    grads = np.zeros((len(corridor), n))
    for idx in range(len(corridor)):
        k_idx = corridor[idx]
        lag = k_idx - (n - 1)
        g = np.zeros(n)
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        elif lag > 0:
            g[:n - lag] += 1.0 - h[lag:]
            g[lag:] += -h[:n - lag]
        else:
            m = -lag
            g[m:] += 1.0 - h[:n - m]
            g[:n - m] += -h[m:]
        grads[idx] = g
    return grads


def smooth_max_and_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    max_shifted = 0.0
    exp_vals = np.exp(alpha * (shifted - max_shifted))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = m + np.log(sum_exp) / alpha
    corridor = np.where(w > 1e-10)[0]
    if len(corridor) == 0:
        return F, np.zeros(n)
    grads = grad_lag_array(h, corridor)
    grad = np.tensordot(w[corridor], grads, axes=(0, 0))
    return F, grad


def subgradient_step(h, corr, active_frac=0.05, max_active=40):
    n = len(h)
    max_val = np.max(corr)
    threshold = max_val - active_frac * (max_val - np.min(corr) + 1e-10)
    active = np.where(corr >= threshold)[0]
    if len(active) > max_active:
        top_idx = np.argsort(corr[active])[::-1][:max_active]
        active = active[top_idx]
    grads = grad_lag_array(h, active)
    g = grads.mean(axis=0)
    return g, active


def nesterov_subgrad(h0, S, n_iter, alpha0, decay, momentum, time_check):
    n = len(h0)
    h = h0.copy()
    y = h0.copy()
    best_h = h0.copy()
    best_F, _ = compute_c5_and_corr(best_h)
    beta = momentum
    for t in range(n_iter):
        if t % 50 == 0:
            F_curr, _ = compute_c5_and_corr(h)
            if F_curr < best_F:
                best_F = F_curr
                best_h = h.copy()
        if time_check() < 40:
            break
        alpha = alpha0 / (1.0 + 0.01 * t) * decay ** (t % 200)
        corr = np.correlate(y, 1.0 - y, mode='full')
        g, _ = subgradient_step(y, corr)
        h_new = y - alpha * g
        h_new = project_sum_bound(h_new, S)
        beta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y = h_new + (beta / beta_new) * (h_new - h)
        y = project_sum_bound(y, S)
        h = h_new
        if t % 100 == 0:
            F_check, _ = compute_c5_and_corr(h)
            if F_check < best_F:
                best_F = F_check
                best_h = h.copy()
    F_final, _ = compute_c5_and_corr(h)
    if F_final < best_F:
        best_F = F_final
        best_h = h.copy()
    return best_h, best_F


def lbfgs_smooth(h_start, S, alpha, maxiter, time_check):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        F, _ = smooth_max_and_grad(x, alpha)
        return F

    def jac(x):
        _, g = smooth_max_and_grad(x, alpha)
        return g

    try:
        res = minimize(obj, h_start.copy(), jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        h_new = project_sum_bound(res.x, S)
        return h_new
    except Exception:
        return h_start.copy()


def active_set_refine(h, S, n_rounds, time_check):
    n = len(h)
    best_h = h.copy()
    best_F, _ = compute_c5_and_corr(best_h)
    for rnd in range(n_rounds):
        if time_check() < 30:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_val = np.max(corr)
        threshold = max_val - 0.005 * (max_val - np.min(corr) + 1e-10)
        active = np.where(corr >= threshold)[0]
        if len(active) > 30:
            top_idx = np.argsort(corr[active])[::-1][:30]
            active = active[top_idx]
        grads = grad_lag_array(best_h, active)
        g_mean = grads.mean(axis=0)
        g_max_contrib = np.zeros(n)
        for i in range(len(active)):
            g_max_contrib += np.abs(grads[i])
        g_max_contrib /= len(active)
        step_sizes = [0.001, 0.005, 0.01, 0.02, 0.05]
        best_step_h = best_h
        best_step_F = best_F
        for ss in step_sizes:
            h_try = best_h - ss * g_mean
            h_try = project_sum_bound(h_try, S)
            F_try, _ = compute_c5_and_corr(h_try)
            if F_try < best_step_F:
                best_step_F = F_try
                best_step_h = h_try
        if best_step_F < best_F:
            best_F = best_step_F
            best_h = best_step_h
        else:
            break
    return best_h, best_F


def interpolate_to(h_src, n_dst, S_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return project_sum_bound(h_src.copy(), S_dst)
    x_src = np.linspace(0.0, 1.0, n_src)
    x_dst = np.linspace(0.0, 1.0, n_dst)
    h_out = np.interp(x_dst, x_src, h_src)
    return project_sum_bound(h_out, S_dst)


def basin_hopping_step(best_h, S, rng, alpha, n_perturb):
    n = len(best_h)
    h_pert = best_h + rng.randn(n) * 0.02
    h_pert = project_sum_bound(h_pert, S)
    h_new = lbfgs_smooth(h_pert, S, alpha, 150, lambda: 9999)
    return h_new
