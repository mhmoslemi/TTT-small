```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    return float(np.max(corr)) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, k, n):
    """Compute gradient of corr_k w.r.t. h.
    corr_k = sum_i h[i]*(1-h[i+k]) for valid i
    d/dh[j] = (1-h[j+k]) if 0<=j+k<n else 0
              - h[j-k] if 0<=j-k<n else 0
    """
    grad = np.zeros(n)
    # First term: (1-h[j+k]) for j=0..n-1-k-1 (if k>0) or j=0..n-1-k (if k>0)
    # Actually: valid j for first term: j+k in [0,n-1], i.e., j in [-k, n-1-k] ∩ [0,n-1]
    if k >= 0:
        # j ranges from 0 to n-1-k
        grad[0:n-k] += (1.0 - h[k:n])
        # Second term: j-k in [0,n-1], i.e., j in [k, n-1]
        # -h[j-k] for j=k..n-1
        grad[k:n] -= h[0:n-k]
    else:
        k2 = -k
        # First term: j+k in [0,n-1], i.e., j-k2 in [0,n-1], j in [k2, n-1]
        grad[k2:n] += (1.0 - h[0:n-k2])
        # Second term: j-k in [0,n-1], j+k2 in [0,n-1], j in [0, n-1-k2]
        grad[0:n-k2] -= h[k2:n]
    return grad

def subgradient_step(h, target, alpha, lr, momentum=0.9):
    """One step of subgradient method with momentum."""
    n = len(h)
    corr = compute_corr_full(h)
    c_max = np.max(corr)
    # Softmax weights
    shifted = alpha * (corr - c_max)
    shifted = np.clip(shifted, -500, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    ks = np.arange(-(n-1), n)
    for i, w in enumerate(weights):
        if w < 1e-10:
            continue
        k = ks[i]
        grad += w * corr_gradient(h, int(k), n)
    
    grad *= 2.0 / n  # multiply by dx
    
    # Projected gradient step with momentum (handled externally)
    return grad, c_max * (2.0/n)

def accelerated_subgrad(h, target, max_iter, deadline):
    """Nesterov accelerated subgradient method."""
    n = len(h)
    h_cur = h.copy()
    v = np.zeros(n)
    best_h = h_cur.copy()
    best_score = compute_c5(h_cur)
    
    T = max_iter
    for t in range(1, T+1):
        if time.time() > deadline:
            break
        
        alpha = 150.0
        lr = 0.5 / (t + 1)  # step size
        
        # Nesterov lookahead
        h_nes = h_cur + momentum_factor(t) * v
        h_nes = project_feasible(h_nes, target)
        
        grad, score = subgradient_step(h_nes, target, alpha, lr)
        
        v = 0.9 * v - lr * grad
        h_new = h_nes + v
        h_new = project_feasible(h_new, target)
        
        sc = compute_c5(h_new)
        if sc < best_score:
            best_score = sc
            best_h = h_new
        h_cur = h_new
    
    return best_h

def momentum_factor(t):
    return (beta(t-1) - 1) / (beta(t) + 1)

def beta(t):
    return (1 + 2*np.sqrt(t)) / 3
```

```python
def active_set_slsqp(h, n, target, tol=1e-5, max_outer=50, deadline=None):
    """Active set method: identify active lags, optimize with explicit constraints."""
    h_cur = h.copy()
    dx = 2.0 / n
    
    for outer in range(max_outer):
        if deadline and time.time() > deadline:
            break
        
        corr = compute_corr_full(h_cur) * dx
        c_max = np.max(corr)
        
        # Active lags: those within tol of the max
        active = np.where(corr >= c_max - tol)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        # Build constraint functions for active lags
        # g_k(h) - t <= 0 for each active k
        ks = np.arange(-(n-1), n)[active]
        
        def objective(x, t_idx=0):
            # Minimize t (last variable)
            return x[-1]
        
        def make_constraint(k_val):
            def constraint(x):
                h_part = x[:-1]
                corr_k = np.correlate(h_part, 1.0 - h_part, mode='full')[k_val + n - 1]
                return x[-1] - corr_k * dx  # should be >= 0, so constraint is x[-1] - corr_k*dx >= 0
            return constraint
        
        # Actually, SLSQP constraints are of the form fun(x) >= 0 (ineq) or fun(x) == 0 (eq)
        # We want: t - g_k(h) >= 0, i.e., t >= g_k(h)
        # In SLSQP inequality form: fun(x) >= 0
        
        cons = []
        for k_val in ks:
            idx = k_val + n - 1
            def constraint(x, idx=idx):
                h_part = x[:-1]
                corr_k = np.correlate(h_part, 1.0 - h_part, mode='full')[idx]
                return x[-1] - corr_k * dx
            cons.append({'type': 'ineq', 'fun': constraint})
        
        # Equality constraint: sum(h) = target
        def sum_constraint(x):
            return np.sum(x[:-1]) - target
        cons.append({'type': 'eq', 'fun': sum_constraint})
        
        # Variables: [h_0, ..., h_{n-1}, t]
        x0 = np.append(h_cur, c_max)
        bounds = [(0, 1)] * n + [(0, 2)]
        
        res = minimize(objective, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-12})
        
        if res.success:
            h_new = project_feasible(res.x[:-1], target)
            sc = compute_c5(h_new)
            if sc < compute_c5(h_cur):
                h_cur = h_new
            else:
                break
        else:
            break
    
    return h_cur
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h))) * dx

def project_feasible(h, target_sum):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h = h.copy()
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_init, n, target, deadline, seed=123):
    """Fast binary local search with incremental correlate updates."""
    rng = np.random.RandomState(seed)
    h = (h_init >= 0.5).astype(float)
    h = project_feasible(h, target)
    # Ensure it's truly binary
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    # Fix sum
    sum_h = h.sum()
    if sum_h != target:
        if sum_h < target:
            idx = np.where(h == 0)[0]
            for i in idx[:int(target - sum_h)]:
                h[i] = 1.0
        else:
            idx = np.where(h == 1)[0]
            for i in idx[:int(sum_h - target)]:
                h[i] = 0.0
    
    corr = compute_corr_full(h)
    c_max = np.max(corr)
    best_h = h.copy()
    best_cmax = c_max
    dx = 2.0 / n
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    n_trials_per_iter = 500
    max_iter = 10000
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        improved = False
        for trial in range(n_trials_per_iter):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            
            # Compute delta for all lags (O(n))
            delta = np.zeros(2 * n - 1)
            for k in range(-(n-1), n):
                val = 0.0
                if 0 <= i + k < n:
                    val += h[i + k] - 1.0
                if 0 <= j + k < n:
                    val += 1.0 - h[j + k]
                if 0 <= i - k < n:
                    val -= h[i - k]
                if 0 <= j - k < n:
                    val -= h[j - k]
                delta[k + n - 1] = val
            
            new_cmax = np.max(corr + delta)
            
            if new_cmax < c_max - 1e-12:
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + delta
                c_max = new_cmax
                improved = True
                if c_max < best_cmax:
                    best_cmax = c_max
                    best_h = h.copy()
                # Update lists
                ones = list(np.where(h == 1)[0])
                zeros = list(np.where(h == 0)[0])
                
                if time.time() > deadline:
                    return best_h, best_cmax * dx
        
        if not improved:
            break
    
    return best_h, best_cmax * dx
```
