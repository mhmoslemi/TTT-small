def fft_conv(a, fsz):

def grad_topk(a, cv, k):

    k = min(k, 2 * n - 1)
    idx = np.argpartition(cv, -k)[-k:]

        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)

    g /= float(k)
    return g
