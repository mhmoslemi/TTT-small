def _peak_shave(a, n_shave=50, eta=0.01):
    """Reduce the peak of conv(a) by redistributing mass."""
    a = a.copy()
    n = a.size
    c = _conv(a)
    k_star = c.argmax()
    # Contribution of a_i to c_{k_star}
    # c_{k*} = sum_{i=max(0,k*-n+1)}^{min(n-1,k*)} a_i * a_{k*-i}
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    contrib = np.zeros(n)
    for i in range(lo, hi + 1):
        contrib[i] = a[k_star - i]
    # Reduce top contributors, increase bottom
    idx = np.argsort(contrib)
    n_red = min(n_shave, n // 4)
    red_set = idx[-n_red:]  # top contributors
    inc_set = idx[:n_red]   # bottom contributors
    red_mass = a[red_set].sum() * eta
    a[red_set] *= (1.0 - eta)
    a[inc_set] += red_mass / n_red
    a = np.clip(a, 0, None)
    return a
