                               options={'maxiter': 800, 'gtol': 1e-14,
                                        'maxls': 50, 'ftol': 1e-18})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * scale
                sc = _score(nx)
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Third LP pass with more constraints
    if time.monotonic() < dl - 80.0:
        cur = best.copy()
        for step in range(200):
            if time.monotonic() > dl - 80.0:
                break
            g = _lp_dir(cur, K=max(10, n // 100))
            if g is None:
                break
            g = g / g.sum() * scale
            for t in [0.1, 0.05, 0.02]:
                nx = (1.0 - t) * cur + t * g
                nx = np.clip(nx, 0.0, 1000.0)
                if nx.sum() < 0.01:
                    continue
                sc = _score(nx)
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                    cur = nx.copy()
                    break

    # Local random search
    while time.monotonic() < dl - 10.0:
        cd = cur.copy()
        nw = max(1, int(rng.integers(1, n // 12)))
        s = int(rng.integers(0, n - nw))
        cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.03)))
        if rng.random() < 0.08:
            cd *= float(np.exp(rng.normal(0.0, 0.008)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = _score(cd)
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2
                cur = cd.copy()
        if rng.random() < 0.03:
            cur = best.copy()

    # Peak-targeting: reduce elements contributing to top peaks
    if time.monotonic() < dl - 5.0:
        for _ in range(100):
            if time.monotonic() > dl - 5.0:
                break
            c = _conv(best)
            topk = np.argpartition(c, -5)[-5:]
            cd = best.copy()
            for pk in topk:
                pk = int(pk)
                lo = max(0, pk - n + 1)
                hi = min(n - 1, pk)
                for j in range(lo, hi + 1):
                    if rng.random() < 0.5:
                        cd[j] = max(0.0, cd[j] * float(np.exp(rng.normal(-0.005, 0.02))))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc2 = _score(cd)
                if np.isfinite(sc2) and sc2 < best_sc:
                    best = cd.copy()
                    best_sc = sc2

    # Final verification with exact evaluator
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n) / n * scale
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n) / n * scale
        final_sc = float(evaluate_sequence(best.tolist()))
    return [float(v) for v in best]
