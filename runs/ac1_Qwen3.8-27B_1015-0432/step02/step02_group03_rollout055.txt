import numpy as np
import time

def _fft_conv(a):
    """Full autoconvolution via FFT."""
    n = len(a)
    fs = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:2*n-1]

def _grad_active(a, lags, fs):
    """Compute subgradient using FFT cross-correlation."""
    n = len(a)
    h = np.zeros(fs)
    for k in lags:
        h[int(k)] = 1.0
    # g_i = 2 * sum_j a_j * h[j+i]
    # = 2 * c[n-1+i] where c = conv(flip(a), h)
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, fs)
    fh = np.fft.rfft(h, fs)
    c = np.fft.irfft(fa * fh, fs)
    g = 2.0 * c[n-1 : 2*n-1]
    m = len(lags)
    if m > 0:
        g /= m
    return g

def _descent(a, sum_target, dl, rng, mode='euro', eta_init=0.5, beta=0.85,
             frac=3e-4, cap=50, smax=3000, max_iter=200000):
    """Subgradient descent with FFT gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    a *= sum_target / a.sum()
    fs = 1 << (2*n - 1).bit_length()
    
    c = _fft_conv(a)
    sc = 2.0 * n * float(c.max()) / (a.sum()**2)
    best = a.copy()
    best_sc = sc
    mom = np.zeros(n)
    stall = 0
    prev_sc = sc
    it = 0
    
    while it < max_iter and time.monotonic() < dl:

        mx = float(c.max())
        thr = mx * (1.0 - frac)
        act = np.where(c >= thr)[0]
        act = act[(act >= 0) & (act < 2*n-1)]
        if len(act) > cap:
            idx = np.argpartition(c[act], -cap)[-cap:]
            act = act[idx]
        
        # Compute gradient via FFT
        g = _grad_active(a, act, fs)
        g -= g.mean()  # mean-zero for scale invariance
        
        # Update
        if mode == 'multi':
            gm = np.abs(g).max()
            if gm > 1e-14:
                a *= np.exp(-eta_init * g / gm)
        else:
            mom = beta * mom + (1-beta) * g
            a -= eta_init * mom
            a = np.clip(a, 0, None)
        
        s = a.sum()
        if s < 1e-10:

        a *= sum_target / s

def _fc(a):
    """Full autoconvolution via FFT."""
    n = len(a)
    fs = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:2 * n - 1]

        eta0 *= 0.99997
