import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def norm1(f):
    s = f.sum()
    return f / s if s > 1e-15 else f.copy()

def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    return np.maximum(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1))), 1e-15)

def init_gauss(n, sf):
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)
    return np.maximum(np.exp(-0.5 * ((x - c) / (sf * n)) ** 2), 1e-15)

def init_cospow(n, p):
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)
    a = (0.5 + 0.5 * np.cos(np.pi * (x - c) / (0.5 * n))) ** p
    return np.maximum(a, 1e-15)

    f = f.copy()
    b, e = b0, e0

        M = bb.max()

            break
