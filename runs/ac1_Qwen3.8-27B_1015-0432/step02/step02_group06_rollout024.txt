def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    if size < 2:
        size = 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def compute_grad(f, beta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), 0.0
    w = np.exp(beta * (b / M - 1.0))
    f_rev = f[::-1]
    gc = fft_conv(w, f_rev)
    return gc[n - 1:2 * n - 1], M


def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    h = np.maximum(h, 1e-15)
    return h / h.sum()

    x = np.arange(n, dtype=np.float64)
    g = np.exp(-0.5 * ((x - center * n) / (sigma * n)) ** 2)
    return g / g.sum()

        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = trial.copy()

    # Final validation
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0] * n
        final_sc = evaluate_sequence(result)

    return result
