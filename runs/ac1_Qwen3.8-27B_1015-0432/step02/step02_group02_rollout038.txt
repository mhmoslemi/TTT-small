import numpy as np
import time
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv_full(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _grad_max(a, tau):
    n = a.size
    c = _conv_full(a)
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    p = e / e.sum()
    L = _fpow2(3 * n)
    rev = a[::-1]
    g = ifft(fft(p, L) * fft(rev, L)).real[n - 1:2 * n - 1]
    return g, cm


def _obj_unit(a, n):
    return 2.0 * n * _conv_full(a).max()
