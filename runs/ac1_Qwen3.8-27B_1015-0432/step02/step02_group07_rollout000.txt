def fft_conv(a, N):

    """Main optimization loop."""
    local_best = best.copy()
    local_score = best_score
    # Initialize Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    t = 0

    stall = 0
    top_k = 12

        t += 1
        conv = fft_conv(a, N)
        M = float(np.max(conv))
        score = 2.0 * n * M / (s * s)
        
        if score < local_score:
            local_score = score
            local_best = a.copy()
        
        if M < prev_M - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_M = M
        
        if stall > 3000:
            # Add noise to escape
            a += 0.01 * s * np.random.standard_normal(n)
            a = np.clip(a, 0, None)
            s_new = a.sum()
            if s_new < 1e-10:
                break
            a = a * (s / s_new)
            s = s_new  # keep s the same for normalization
            stall = 0
            continue
        
        # Compute weighted subgradient
        # Use top-k peaks with smooth weighting
        tk = min(top_k, 2*n - 1)
        idx = np.argpartition(conv, -tk)[-tk:]
        # Weight by peak height (normalized)
        peak_vals = conv[idx]
        # Soft weights: proportional to peak height
        w = peak_vals / peak_vals.sum()
        
        grad = np.zeros(n)
        for ii in range(tk):
            k = int(idx[ii])
            ww = float(w[ii])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2.0 * ww * a[k-hi:k-lo+1][::-1]
        
        # Subtract mean to maintain sum
        grad -= grad.mean()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_try = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_try = np.clip(a_try, 0, None)
        s_try = a_try.sum()

            top_k += 2
        lr = max(0.02, lr * 0.9999)
        
    return a, local_best, local_score
