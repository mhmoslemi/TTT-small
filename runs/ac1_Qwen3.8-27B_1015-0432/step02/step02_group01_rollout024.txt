    f /= f.sum()
    best_f = f.copy()
    best_sc = score_norm(f)
    eta = eta0
    stall = 0
    use_lp = False

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        sc = 2.0 * n * b.max()
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
            stall = 0
            use_lp = False
        else:
            stall += 1
            if stall == 40:
                use_lp = True
            if stall > 80:
                if use_lp:
                    g_dir = lp_direction(best_f)
                    f = (1 - 0.3) * best_f + 0.3 * g_dir
                    f = np.maximum(f, 1e-20)
                    f /= f.sum()
                else:
                    f = best_f * np.exp(0.04 * rng.standard_normal(n))
                    f = np.maximum(f, 1e-20)
                    f /= f.sum()
                eta = min(eta * 2.0, 5.0)
                use_lp = not use_lp
                stall = 0
                continue

        g = compute_grad(f, b, beta)
        if g is None:
            break
        gm = g.max()
        g /= gm
        trial = f * np.exp(-eta * g)
        trial = np.maximum(trial, 1e-20)
        trial /= trial.sum()
        sc_t = score_norm(trial)
        if sc_t < best_sc:
            f = trial
            eta = min(eta * 1.08, 8.0)
        else:
            eta *= 0.5
            if eta < 0.002:
                stall += 3
            continue
        if it % 3000 == 0 and it > 0:
            beta *= 1.04

    return best_f, best_sc


def make_inits(n):
    inits = []
    x = np.arange(n, dtype=np.float64) + 1.0
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        r = x ** p
        inits.append(r / r.sum())
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        s = np.maximum((1 + np.cos(t)) ** alpha, 1e-15)
        inits.append(s / s.sum())
    for c in [0.25, 0.3, 0.35, 0.4, 0.45]:
        u = np.arange(n) / max(n - 1, 1)
        a = np.exp(-((u - c) / 0.12) ** 2) + np.exp(-((u - (1 - c)) / 0.12) ** 2)
        inits.append(a / a.sum())
    return inits


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p /= p.sum()

    best_f = p.copy()
    best_sc = score_norm(p)

    inits = [p, p[::-1].copy()] + make_inits(n)
    for ini in inits:
        sc = score_norm(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    phases = [
        (10.0, 1.5, 40000),
        (20.0, 0.8, 40000),
        (40.0, 0.5, 40000),
        (80.0, 0.3, 40000),
        (160.0, 0.15, 40000),
        (320.0, 0.08, 40000),
        (640.0, 0.04, 40000),
        (1280.0, 0.02, 40000),
    ]

    for beta_p, eta_p, miters in phases:
        if time.monotonic() > deadline - 35:
            break
        f_opt, sc_opt = optimize_phase(best_f, deadline - 18, rng, beta_p, eta_p, miters)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()
        if time.monotonic() > deadline - 30:
            break
        for t in [0.25, 0.5, 0.75]:
            bl = (1 - t) * best_f + t * best_f[::-1]
            bl = np.maximum(bl, 1e-20)
            bl /= bl.sum()
            sc_b = score_norm(bl)
            if sc_b < best_sc:
                best_sc = sc_b
                best_f = bl.copy()

    if time.monotonic() < deadline - 45:
        f_fin, sc_fin = optimize_phase(best_f, deadline - 12, rng, 3000.0, 0.01, 60000)
        if sc_fin < best_sc:
            best_sc = sc_fin
            best_f = f_fin.copy()

    # Final micro-perturbations
    if time.monotonic() < deadline - 25:
        for _ in range(20):
            if time.monotonic() > deadline - 22:
                break
            trial = best_f * np.exp(0.008 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc_t = score_norm(trial)
            if sc_t < best_sc:
                best_sc = sc_t
                best_f = trial.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
