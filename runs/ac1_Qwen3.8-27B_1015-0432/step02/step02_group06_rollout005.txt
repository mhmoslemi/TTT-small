        a = 1.0 + param * np.cos(2.0 * np.pi * param * x / n)
    elif kind == 4:
        a = np.exp(-param * (x - (n - 1) / 2.0) ** 2 / (2.0 * n * n))
    elif kind == 5:
        a = 1.0 / (1.0 + param * np.exp(-x / 50.0))
    else:
        a = 1.0
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def mirror_descend(f, deadline, rng, beta, eta, check_every, stall_limit):
    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0

    while time.monotonic() < deadline - 3.0:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        w = np.exp(beta * (b / M - 1.0))
        ws = w.sum()
        if ws < 1e-15:
            break
        w /= ws
        f_rev = f[::-1]
        gc = fft_conv(w, f_rev)
        grad = 2.0 * gc[n - 1:2 * n - 1]
        grad = np.maximum(grad, 0.0)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        it += 1

        if it % check_every == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall >= stall_limit:
                break
            if stall % 40 == 0 and stall < stall_limit:
                f = best * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
            if it % 400 == 0:
                beta = min(600.0, beta * 1.06)
                eta = max(0.002, eta * 0.99)

    return best, best_score(f) if (best_sc := best_sc) > 0 else fast_score(best)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p /= p.sum()

    inits = [p]
    for kind, param in [(0, 0), (1, 0.3), (1, 0.5), (1, 0.7), (2, 0.2),
                        (3, 1.0), (3, 2.0), (3, 3.0), (4, 1.0), (4, 0.5), (5, 1.0)]:
        inits.append(make_init(n, kind, float(param)))
    for s in [0.1, 0.25]:
        t = p * np.exp(s * rng.standard_normal(n))
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())
    psym = 0.5 * (p + p[::-1])
    inits.append(np.maximum(psym, 1e-25) / np.maximum(psym, 1e-25).sum())

    best_f = None
    best_score = np.inf
    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = ini.copy()

    if best_f is None:
        best_f, best_score = p, fast_score(p)

    t_avail = deadline - t0
    phases = [(0.40, 100.0, 0.10, 15, 300),
              (0.30, 250.0, 0.035, 10, 250),
              (0.22, 450.0, 0.012, 5, 200)]

    elapsed = 0.0
    for frac, b0, e0, ce, sl in phases:
        phase_dl = t0 + elapsed + frac * t_avail
        if time.monotonic() > phase_dl - 5.0:
            break
        f_opt, sc_opt = mirror_descend(best_f, phase_dl - 2.0, rng, b0, e0, ce, sl)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
        elapsed += frac * t_avail

    # Polish phase with tiny steps
    if time.monotonic() < deadline - 30.0:
        f_opt, sc_opt = mirror_descend(best_f, deadline - 30.0, rng, 500.0, 0.005, 3, 100)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

    # Final micro-perturbations
    for _ in range(80):
        if time.monotonic() > deadline - 15.0:
            break
        for scale in [0.0008, 0.002, 0.006]:
            trial = best_f * np.exp(scale * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-25)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    # Symmetric mix attempts
    for mix in [0.05, 0.1, 0.15, 0.25]:
        if time.monotonic() > deadline - 12.0:
            break
        trial = (1.0 - mix) * best_f + mix * best_f[::-1]
        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = trial.copy()
            best_f = best_f / best_f.sum()

    final_score = evaluate_sequence([float(x) for x in best_f])
    if not np.isfinite(final_score):
        best_f = p.copy()

    return [float(x) for x in best_f]
