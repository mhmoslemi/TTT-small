def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def subgrad(v, pk, w):
    n = v.shape[0]
    h = np.zeros(2 * n - 1)
    h[pk] = w
    c = fft_conv(v, h[::-1])
    g = 2.0 * c[n - 1:n + n - 1][::-1]
    return np.maximum(g, 0.0)
