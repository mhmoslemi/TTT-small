            else:
                eta = max(eta * 0.78, 0.002)
        if it % 350 == 0:
            T = max(0.02, T * 0.91)
        if no_imp > 400:
            s = 0.06 + 0.08 * rng.random()
            f = best_f * np.exp(s * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.1
            T = max(0.5, T * 0.6)
    return best_f, best_sc

def init_flat(n, asym=0.0):
    x = np.linspace(-np.pi, np.pi, n)
    w = 1.0 - 0.42 * np.cos(x) + 0.21 * np.cos(2 * x) - 0.07 * np.cos(3 * x)
    if abs(asym) > 1e-9:
        w *= np.exp(asym * (2.0 * np.arange(n) / max(n - 1, 1) - 1.0))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def init_pow(n, a, b):
    j = np.arange(n, dtype=np.float64)
    w = (j + 1.0) ** a * (n - j) ** b
    w = np.maximum(w, 1e-15)
    return w / w.sum()

def init_bump(n, c, w, asym):
    j = np.arange(n, dtype=np.float64)
    v = np.exp(-0.5 * ((j - c) / w) ** 2)
    v *= np.exp(asym * (2.0 * j / max(n - 1, 1) - 1.0))
    v = np.maximum(v, 1e-15)
    return v / v.sum()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 18.0
    n = len(height_sequence_1)
    pp = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = []
    if pp.size == n and np.all(np.isfinite(pp)) and pp.sum() > 1e-10:
        inits.append(pp / pp.sum())
    inits.append(init_flat(n, 0.0))
    inits.append(init_flat(n, 0.4))
    inits.append(init_flat(n, -0.4))
    inits.append(init_flat(n, 0.8))
    inits.append(init_flat(n, -0.8))
    inits.append(init_pow(n, 0.0, 0.0))
    inits.append(init_pow(n, 0.4, 0.0))
    inits.append(init_pow(n, 0.0, 0.4))
    inits.append(init_pow(n, 0.3, 0.3))
    inits.append(init_pow(n, 0.6, 0.2))
    inits.append(init_bump(n, 0.35 * n, 0.3 * n, 0.5))
    inits.append(init_bump(n, 0.5 * n, 0.2 * n, -0.5))
    inits.append(init_bump(n, 0.5 * n, 0.35 * n, 0.0))
    inits.append(np.ones(n) / n)
    best_f, best_sc = inits[0].copy(), fast_score(inits[0])
    for ini in inits[1:]:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_f = sc, ini.copy()
    phase = 0
    while time.monotonic() < deadline - 30.0:
        phase += 1
        to = time.monotonic() + min(220.0, deadline - time.monotonic() - 25.0)
        f1, sc1 = optimize(best_f, to, rng, do_lp=(phase >= 2))
        if sc1 < best_sc:
            best_sc, best_f = sc1, f1
        if phase % 2 == 0 and time.monotonic() < deadline - 55.0:
            for sc in [0.07, 0.15, 0.25]:
                tr = best_f * np.exp(sc * rng.standard_normal(n))
                tr = np.maximum(tr, 1e-20)
                tr /= tr.sum()
                to2 = time.monotonic() + min(70.0, deadline - time.monotonic() - 25.0)
                if to2 <= time.monotonic():
                    break
                f2, sc2 = optimize(tr, to2, rng, do_lp=False)
                if sc2 < best_sc:
                    best_sc, best_f = sc2, f2
    result = [float(x) for x in np.maximum(best_f, 0.0)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
