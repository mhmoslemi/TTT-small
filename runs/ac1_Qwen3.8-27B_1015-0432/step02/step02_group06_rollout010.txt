import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def init_bump(n, taper_frac):
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    edge = taper_frac * n / 2.0
    d = np.abs(x - c)
    a = np.ones(n)
    mask = d > (n / 2.0 - edge)
    a[mask] = 0.5 * (1.0 + np.cos(np.pi * (d[mask] - (n / 2.0 - edge)) / edge))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def init_power(n, p):
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    u = 1.0 - np.abs(x - c) / (c + 1.0)
    a = np.power(np.maximum(u, 1e-10), p)
    return a / a.sum()


def init_gauss(n, sf):
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - c) / (sf * n)) ** 2)
    return a / a.sum()


def optimize_mult(f, deadline, rng, beta0=40.0, eta0=0.12):

    f = f.copy()
    beta = beta0
    eta = eta0
    best = f.copy()
    best_sc = fast_score(f)
    it = 0
    stall = 0
    while time.monotonic() < deadline - 2.0:
        b = fft_conv(f, f)
        M = b.max()
        ratio = b / M
        w = np.power(ratio, beta)
        f_rev = f[::-1]
        gc = fft_conv(w, f_rev)
        grad = gc[n - 1:2 * n - 1]
        grad = np.maximum(grad, 1e-30)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        it += 1
        if it % 15 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 250:
                break
            if it % 150 == 0:
                beta = min(600.0, beta * 1.05)
                eta = max(0.004, eta * 0.982)
            if 60 <= stall < 250 and stall % 60 == 0:
                f = best * np.exp(0.012 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
    return best, best_sc

    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    eta = 0.008
    beta = 200.0
    it = 0

        ratio = b / M
