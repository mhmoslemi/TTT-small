def optimize(f, deadline, rng, beta_start, beta_end, eta_start, eta_end, lam, eval_ever, clip_c):
    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    while time.monotonic() < deadline - 3.0:
        it += 1
        p = min(1.0, it / 40000.0)
        beta = beta_start + (beta_end - beta_start) * p
        eta = eta_start + (eta_end - eta_start) * p
        grad = compute_grad(f, beta, lam)
        gmax = np.abs(grad).max()
        if gmax < 1e-30:
            break
        g = np.clip(grad / gmax, -clip_c, clip_c)
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        if it % eval_ever == 0:
            sc = fast_score(f)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 500:
                f = best * np.exp(0.1 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
            elif stall >= 150 and stall % 60 == 0:
                f = best * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = p / p.sum()

    best_f, best_score = None, np.inf
    inits = [p.copy()]
    for s in [0.04, 0.1, 0.2]:
        t = p * np.exp(s * rng.standard_normal(n))
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())
    for mix in [0.15, 0.3, 0.5]:
        t = (1.0 - mix) * p + mix * p[::-1]
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())
    for c in inits:
        sc = fast_score(c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = c.copy()

    # Phase 1: variance-aware broad shaping
    dl1 = t0 + 0.22 * (budget_s - 20)
    f, sc = optimize(best_f, dl1, rng, 8.0, 40.0, 0.10, 0.04, 0.5, 15, 5.0)
    if sc < best_score:
        best_score, best_f = sc, f.copy()

    # Phase 2: peak-focused with mild variance
    dl2 = t0 + 0.50 * (budget_s - 20)
    f, sc = optimize(best_f, dl2, rng, 40.0, 150.0, 0.04, 0.012, 0.1, 12, 4.0)
    if sc < best_score:
        best_score, best_f = sc, f.copy()

    # Phase 3: ultra-sharp peak minimization
    dl3 = t0 + 0.80 * (budget_s - 20)
    f, sc = optimize(best_f, dl3, rng, 150.0, 400.0, 0.012, 0.004, 0.02, 10, 3.0)
    if sc < best_score:
        best_score, best_f = sc, f.copy()

    # Phase 4: final polish
    f, sc = optimize(best_f, deadline - 10, rng, 400.0, 600.0, 0.004, 0.0015, 0.0, 8, 2.5)
    if sc < best_score:
        best_score, best_f = sc, f.copy()

    # Micro-perturbations
    if time.monotonic() < deadline - 5:
        for _ in range(400):
            if time.monotonic() > deadline - 5:
                break
            trial = best_f * np.exp(0.002 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-25)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    # Mirror blend as last resort
    if time.monotonic() < deadline - 3:
        for mix in [0.05, 0.1, 0.2]:
            trial = (1.0 - mix) * best_f + mix * best_f[::-1]
            trial = np.maximum(trial, 1e-25)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    out = [float(x) for x in best_f]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs):
        out = [1.0 / n] * n
    return out
