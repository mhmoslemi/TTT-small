import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def peak_subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def peak_subgrad_w(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.06, n_peaks=160, sym=False, max_iter=800000):
    n = f.shape[0]
    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    restart_count = 0
    momentum = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        frac = 0.99 - 0.02 * min(1.0, it / 5000.0)
        thresh = M * frac
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > n_peaks:
            idx = np.argsort(b)[::-1]
            peaks = idx[:n_peaks]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])

        vals = b[peaks]
        w = vals / vals.max()
        g = peak_subgrad_w(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:
            break

        momentum = 0.85 * momentum + 0.15 * g
        mmax = momentum.max()
        if mmax < 1e-25:
            break

        f = f * np.exp(-eta * momentum / mmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f = norm_pos(f)

        if it % 40 == 0:
            sc = obj_val(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1

            if no_improv > 600:
                scale = 0.03 + 0.06 * min(1.0, restart_count / 8.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-15)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = norm_pos(f)
                momentum = np.zeros(n)
                no_improv = 0
                restart_count += 1
                eta = min(0.3, eta * 1.35)

        eta *= 0.9995

    return best_f, best_sc
