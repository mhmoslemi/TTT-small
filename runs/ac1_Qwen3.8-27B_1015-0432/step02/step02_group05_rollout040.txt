def _fft_conv_and_grad(a, L, temp):
    """Compute convolution and softmax-weighted gradient."""
    n = a.size
    FA = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(FA * FA, n=L)[:2*n-1]
    cm = conv.max()
    if temp > 1e-12:
        w = np.exp((conv - cm) / temp)
        w /= w.sum()
    else:

        w = np.zeros_like(conv)
        w[conv >= cm - 1e-15] = 1.0
        w /= w.sum()
    FW = np.fft.rfft(w, n=L)
    g = np.fft.irfft(np.conj(FW) * FA, n=L)[:n]
