import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score_norm(f):
    n = f.size
    b = fft_conv(f, f)
    return 2.0 * n * b.max()


def grad_direction(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        k = max(1, b.size // 100)
        w[np.argsort(b)[-k:]] = 1.0
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n - 1:2 * n - 1].copy()
    return np.maximum(grad, 0.0)


def lp_direction(f, b, K, dl):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    top_idx = np.sort(np.argsort(b)[-K:])
    rows, cols, vals = [], [], []
    for ri, x in enumerate(top_idx):
        xi = int(x)
        jlo = max(0, xi - n + 1)
        jhi = min(xi, n - 1)
        if jhi < jlo:
            continue
        js = np.arange(jlo, jhi + 1)
        rows.extend([ri] * (jhi - jlo + 1))
        cols.extend(js.tolist())
        vals.extend((f[xi - js]).tolist())
    if not rows:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_idx), n))
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'time_limit': 25, 'presolve': True})
    if time.monotonic() > dl:
        return None
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def optimize(f, deadline, rng, beta0, eta0, use_lp, lp_gate):
    n = f.size
    f = f.astype(np.float64).copy()
    f = np.maximum(f, 0.0)
    S = f.sum()
    if S < 1e-10:
        return f, np.inf
    f /= S
    best_f = f.copy()
    best_sc = score_norm(f)
    beta = beta0
    eta = eta0
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        if it % 50 == 0 and time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        sc = 2.0 * n * M

        if sc < best_sc - 1e-13:
            best_sc = sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > (300 if use_lp else 150):
                f = best_f * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                beta = beta0
                eta = eta0
                it += 1
                continue

        g = None
        if use_lp and it % lp_gate == 0:
            K = min(120, max(30, n // 60))
            g = lp_direction(f, b, K, deadline - 5.0)
        if g is None:
            g = grad_direction(f, b, beta)
        if g is None:
            break
        gmax = g.max()
        if gmax < 1e-30:
            break
        g = g / gmax

        if use_lp and it % lp_gate == 0:
            step = eta * 0.6
        else:
            step = eta
        f = f * np.exp(-step * g)
        f = np.maximum(f, 1e-20)
        f /= f.sum()

        it += 1
        if it % 200 == 0:
            beta = min(beta * 1.06, 600.0)
            eta = max(0.002, eta * 0.992)

    return best_f, best_sc


def make_ramp(n, pwr, rev):
    x = np.arange(n, dtype=np.float64) + 1.0
    r = x ** pwr
    if rev:
        r = r[::-1]
    return r / r.sum()


def make_arc(n, center, width):
    u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    v = (u - center) / width
    a = np.maximum(0.0, 1.0 - v * v)
    a **= 0.7
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_cos(n, k, ph):
    u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * u + ph)
    a = np.maximum(a, 1e-15)
    return a / a.sum()
