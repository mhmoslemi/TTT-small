import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_flat_top(n, asym):
    x = np.linspace(-np.pi, np.pi, n)
    w = 1.0 - 0.42 * np.cos(x) + 0.21 * np.cos(2 * x) - 0.07 * np.cos(3 * x)
    if abs(asym) > 1e-9:
        w *= np.exp(asym * (2.0 * np.arange(n) / max(n - 1, 1) - 1.0))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def make_powlaw(n, a, b):
    j = np.arange(n, dtype=np.float64)
    w = (j + 1.0) ** a * (n - j) ** b
    return w / w.sum()

def make_gauss(n, center, width, asym):
    j = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * ((j - center) / width) ** 2)
    w *= np.exp(asym * (2.0 * j / max(n - 1, 1) - 1.0))
    w = np.maximum(w, 1e-15)
    return w / w.sum()

def grad_peak(f, k_star):
    n = f.size
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    grad = np.zeros(n)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * f[k_star - hi:k_star - lo + 1][::-1]
    return grad

def grad_smooth(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def grad_topk(f, C, K):
    n = f.size
    idx = np.argpartition(C, -K)[-K:]
    p = np.zeros_like(C)
    p[idx] = C[idx]
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

    best_sc = fast_score(f)
    no_imp = 0
