def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    h = np.maximum(h, 1e-15)
    return h / h.sum()

def init_sym_freq(n, rng):
    # Design in frequency domain for flat |F|^2
    # F(omega_k) = exp(i * theta_k) gives flat |F|^2 = 1
    # f = ifft(F) might have negative values, so clip
    N = 1 << (2 * n - 2).bit_length()
    theta = rng.standard_normal(N) * 0.5
    F = np.exp(1j * theta)
    f = np.fft.irfft(F, N)[:n]
    f = np.maximum(f, 1e-15)
    return f / f.sum()

def symmetric_map(f):
    n = len(f)
    return (f + f[::-1]) / 2.0
