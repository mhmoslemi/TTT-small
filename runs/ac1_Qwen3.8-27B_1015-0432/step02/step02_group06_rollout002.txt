import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def mult_step(f, beta, eta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return f
    w = np.exp(beta * (b / M - 1.0))
    w /= w.sum()
    f_rev = f[::-1]
    wc = fft_conv(w, f_rev)
    grad = 2.0 * wc[n - 1:2 * n - 1]
    grad = np.maximum(grad, 1e-30)
    gmax = grad.max()
    if gmax < 1e-30:
        return f
    f_new = f * np.exp(-eta * grad / gmax)
    f_new = np.maximum(f_new, 1e-25)
    f_new /= f_new.sum()
    return f_new


def argmax_step(f, eta):
    n = f.size
    b = fft_conv(f, f)
    k = int(np.argmax(b))
    grad = np.zeros(n)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
    gmax = grad.max()
    if gmax < 1e-30:
        return f
    f_new = f * np.exp(-eta * grad / gmax)
    f_new = np.maximum(f_new, 1e-25)
    f_new /= f_new.sum()
    return f_new
