import numpy as np
import time

def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def grad_topk(a, cv, tk):
    n = a.size
    if tk >= cv.size:
        idx = np.arange(cv.size)
    else:
        idx = np.argpartition(cv, -tk)[-tk:]
    g = np.zeros(n)
    cnt = 0
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
            cnt += 1
    if cnt > 0:
        g /= float(cnt)
    return g
