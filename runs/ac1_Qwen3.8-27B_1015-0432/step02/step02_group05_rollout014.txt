    x = np.arange(n, dtype=np.float64)
    r2 = np.clip(1.0 + 0.25 * np.sin(2.0 * np.pi * f1 * x + p1), 0.01, None)
    inits.append(r2 / r2.sum())
    mid = (n - 1) / 2.0
    r3 = np.exp(-0.5 * ((x - mid) / (n * 0.18)) ** 2)
    inits.append(r3 / r3.sum())
    r4 = np.clip(1.0 + 0.2 * np.cos(2.0 * np.pi * (n // 7) * x / n + rng.uniform(0, 6.28)), 0.01, None)
    inits.append(r4 / r4.sum())

    n_inits = len(inits)
    for k in range(n_inits):
        rem = dl - time.monotonic()
        if rem < 50.0:
            break
        cap = 400.0 if k == 0 else 200.0
        per = min(rem / (n_inits - k), cap)
        if per < 10.0:
            break
        a, _ = run_multi_phase(inits[k].copy(), n, fsz, time.monotonic() + per, rng)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < dl - 15.0:
        a = best.copy()
        a, _ = run_multi_phase(a, n, fsz, dl - 4.0, rng)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < dl - 3.0:
        base = best * s0
        while time.monotonic() < dl - 1.0:
            c = base.copy()
            w = max(3, n // 250)
            n_p = int(rng.integers(1, 4))
            for _ in range(n_p):
                st = int(rng.integers(0, max(1, n - w)))
                c[st:min(n, st + w)] *= float(np.exp(rng.normal(0.0, 0.012)))
            c = np.clip(c, 0.0, 1000.0)
            if c.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(c.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                base = c
                best = (c / s0).copy()

    out = np.clip(best * s0, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(x) for x in out]
