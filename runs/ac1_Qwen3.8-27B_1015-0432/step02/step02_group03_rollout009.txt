import numpy as np
import time

def _fft_conv(a, fs):
    """Full autocorrelation via FFT."""
    n = a.size
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _grad_sub(a, lags, weights):
    """Weighted subgradient of max conv over given lags."""
    n = a.size
    g = np.zeros(n)
    for idx, k in enumerate(lags):
        w = weights[idx]
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    return g
