The problem: minimize F(a) = 2n * max(conv(a,a)) subject to a >= 0, sum(a) = 1.

- Update: a <- a * exp(-eta * g) / (a * exp(-eta * g)).sum()
- This is multiplicative weight update (exponentiated gradient)

The key is choosing eta (step size) and T (temperature for smoothing).

Starting with T large (nearly uniform gradient) and gradually decreasing T (focusing on the peak) is a standard annealing approach.

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def grad_smooth_max(f, T):
    """Gradient of T*logsumexp(b/T) where b = conv(f,f)."""
    n = len(f)
    b = fft_conv(f, f)  # length 2n-1
    b_shift = b - b.max()
    p = np.exp(b_shift / T)
    p /= p.sum()
    # grad[i] = sum_k p[k] * f[k-i]
    # = cross-correlation of p and f
    # Use FFT: 
    # Compute conv(f, p[::-1]) and pick right indices
    p_rev = p[::-1]
    conv = fft_conv(f, p_rev)  # length n + (2n-1) - 1 = 3n-2
    # grad[i] = conv[2n-2-i]
    grad = conv[n-1:2*n-1][::-1].copy()
    return grad, b.max()

def optimize(f, deadline, T_schedule, eta_schedule):
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 1e-30)
    f /= f.sum()
    best_f = f.copy()
    best_sc = fast_score(f)
    it = 0
    stall = 0
    
    while time.monotonic() < deadline - 2.0:
        T = T_schedule(it)
        eta = eta_schedule(it)
        grad, bmax = grad_smooth_max(f, T)
        # Normalize gradient
        gmax = grad.max()
        if gmax < 1e-30:
            break
        grad = grad / gmax
        # Multiplicative update
        f = f * np.exp(-eta * grad)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        it += 1
        
        if it % 20 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 100:
                break
    
    return best_f, best_sc
