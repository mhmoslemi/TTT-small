import numpy as np

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _act_grad(a, lags):
    n = a.size
    m = lags.size
    G = np.zeros((m, n))
    for i in range(m):
        k = int(lags[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            G[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return G

def _proj_simplex(v):
    n = v.size
    if n <= 1:
        return np.ones(max(n, 1), dtype=float)
    mu = np.sort(v)[::-1]
    sv = np.cumsum(mu)
    tmp = mu * np.arange(1, n + 1) - (sv - 1)
    rho = np.nonzero(tmp > 0)[0]
    if len(rho) == 0:
        return np.full(n, 1.0 / n)
    r = rho[-1]
    theta = (sv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)
