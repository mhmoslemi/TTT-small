            best_sc, best_f = s3, f3.copy()
    # Phase 4: fine random perturbations
    if time.monotonic() < deadline - 8.0:
        for _ in range(60):
            if time.monotonic() > deadline - 4.0:
                break
            for sc_try in [0.003, 0.007, 0.014, 0.025]:
                trial = best_f * np.exp(sc_try * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = compute_score(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial.copy()
                    break
    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
