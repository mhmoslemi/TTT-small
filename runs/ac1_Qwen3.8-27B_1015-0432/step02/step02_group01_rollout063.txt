import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

def score_of(a):
    """Score with sum(a) not necessarily 1: 2n*max(conv)/sum^2"""
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fft_conv(a, a)
    return 2.0 * len(a) * b.max() / (s * s)

    """Gradient of sum_{k in active} conv(a,a)_k w.r.t. a.
    Returns vector of length len(a)."""
    n = len(a)
    f_r = a[::-1]
    c = fft_conv(active.astype(np.float64), f_r)
    return 2.0 * c[n-1:2*n-1].copy()

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

    n = len(a)

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n+m-1]


def eval_score_ft(a):
    n = len(a)
    b = fft_conv(a, a)
    return 2.0 * n * b.max()


def compute_grad(a, w):

    c = fft_conv(w, a[::-1])
    return 2.0 * c[n-1:2*n-1].copy()

def mk(n, k):
    if k == 0:
        return np.ones(n) / n
    if k == 1:
        x = np.arange(n, dtype=np.float64)
        a = np.exp(-1.8 * ((x - n * 0.5) / (n * 0.15)) ** 2)
        return a / a.sum()
    if k == 2:
        u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = np.maximum(np.clip(1.0 - (2 * u - 1) ** 2, 0, None) ** 2.5, 1e-14)
        return a / a.sum()
    if k == 3:
        x = np.arange(n, dtype=np.float64) + 1
        return x / x.sum()
    if k == 4:
        x = np.arange(n, dtype=np.float64) + 1
        return x[::-1] / x.sum()
    if k == 5:
        u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
        a = np.maximum(np.cos(np.pi * (2 * u - 1)) ** 2, 1e-14)
        return a / a.sum()
    u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    ctr = 0.3 + 0.08 * (k - 6)
    wid = 0.2 + 0.04 * (k - 6)
    a = np.maximum(np.exp(-0.5 * ((u - ctr) / wid) ** 2), 1e-14)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    if p.sum() < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / p.sum()
    inits = [p, p[::-1].copy()]
    for k in range(6, -1, -1):
        inits.append(mk(n, k))
    inits.append(rng.random(n) / rng.random(n).sum())
    scores = [sc0(iv) for iv in inits]
    order = np.argsort(scores)
    best_a, best_sc = inits[order[0]], scores[order[0]]

    d1 = min(dl - 200, t0 + 0.55 * (budget_s - 20))
    a1, s1 = do_opt(best_a, d1, rng)
    if s1 < best_sc:
        best_a, best_sc = a1, s1

    for idx in order[1:4]:
        if time.monotonic() > dl - 150:
            break
        d2 = min(dl - 100, time.monotonic() + 0.08 * (budget_s - 20))
        a2, s2 = do_opt(inits[idx], d2, rng)
        if s2 < best_sc:
            best_a, best_sc = a2, s2

    if time.monotonic() < dl - 15:
        a3, s3 = do_opt(best_a, dl - 10, rng)
        if s3 < best_sc:
            best_a, best_sc = a3, s3

    if time.monotonic() < dl - 8:
        rev = best_a[::-1].copy()
        for t in np.linspace(0.05, 0.5, 8):
            bl = (1 - t) * best_a + t * rev
            bl = np.maximum(bl, 1e-15)
            bl /= bl.sum()
            sb = sc0(bl)
            if sb < best_sc:
                best_sc, best_a = sb, bl.copy()

    if time.monotonic() < dl - 8:
        for _ in range(15):
            if time.monotonic() > dl - 8:
                break
            tr = best_a * np.exp(0.004 * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-15)
            tr /= tr.sum()
            st = sc0(tr)
            if st < best_sc:
                best_sc, best_a = st, tr.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
