```python
def lp_update(f, M, top_k=80, time_limit=30):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)[k] <= M for top-k peaks."""
    n = f.shape[0]
    b = fft_conv(f, f)
    peaks = np.argsort(b)[-top_k:]
    
    # Build constraint matrix sparsely
    # Row i corresponds to peak peaks[i]
    # Constraint: sum_j f[peaks[i] - j] * b[j] <= M (for valid j)
    
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(peaks):
        j_range = range(max(0, k - n + 1), min(n, k + 1))
        for j in j_range:
            idx = k - j
            if 0 <= idx < n:
                rows.append(i)
                cols.append(j)
                vals.append(f[idx])
    
    A = csc_array((vals, (rows, cols)), shape=(len(peaks), n))
    
    b_var = cp.Variable(n, nonneg=True)
    constraint = A @ b_var <= M
    obj = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(obj, [constraint])
    
    try:
        prob.solve(solver=cp.SCS, max_iter=10000, eps=1e-6)
        if b_var.value is not None:
            return np.maximum(b_var.value, 0)
    except:
        pass
    return None
```

```python
        M = b.max()
        g = subgrad(f, b, M * 0.995, max_p=600)
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-18)
        f /= f.sum()
        if it % 80 == 0:
            sc = eval_score(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                eta = min(0.03, eta * 1.03)
            else:
                eta *= 0.92
        if it > 50:
            eta *= 0.999
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = gen_inits(n, rng)
    best_f, best_sc = None, np.inf
    for init in inits:
        sc = eval_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    if not np.isfinite(best_sc) or best_f is None:
        best_f = np.ones(n) / n
        best_sc = eval_score(best_f)
    p1_end = min(deadline, t0 + 350.0)
    f1, s1 = grad_opt(best_f.copy(), p1_end, rng, use_lp=False)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()
    if time.monotonic() < deadline - 200.0:
        p2_end = min(deadline, t0 + 800.0)
        f2, s2 = grad_opt(best_f.copy(), p2_end, rng, use_lp=True, lp_every=4000)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()
    if time.monotonic() < deadline - 80.0:
        p3_end = min(deadline, t0 + 1040.0)
        f3, s3 = grad_opt(best_f.copy(), p3_end, rng, use_lp=False)
        if s3 < best_sc:
            best_sc, best_f = s3, f3.copy()
    if time.monotonic() < deadline - 40.0:
        f4, s4 = polish(best_f.copy(), min(deadline, t0 + 1070.0), rng)
        if s4 < best_sc:
            best_sc, best_f = s4, f4.copy()
    for _ in range(40):
        if time.monotonic() > deadline - 10.0:
            break
        trial = best_f * np.exp(0.005 * rng.standard_normal(n))
        trial = normalize(trial)
        sc = eval_score(trial)
        if sc < best_sc:
            best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
```
