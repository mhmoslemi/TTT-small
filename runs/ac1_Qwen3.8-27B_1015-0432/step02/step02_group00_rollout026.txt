import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return 1e18
    return 2.0 * len(f) * b.max() / (s * s)

def subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
