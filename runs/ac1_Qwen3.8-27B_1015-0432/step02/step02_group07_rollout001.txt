        if time.monotonic() > dl:
            break
        c = fft_conv(a, fs)
        M = float(np.max(c[:2*n-1]))
        if M < bM - 1e-16:
            bM = M
            ba = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            a = ba.copy()
            # perturb
            for _ in range(4):
                p = rng.integers(0, n)
                w = int(rng.integers(3, n // 8))
                l = max(0, p - w // 2)
                r = min(n, p + w // 2)
                a[l:r] *= 1.0 + rng.normal(0, 0.01)
            # cosine perturbation
            a *= 1.0 + 0.02 * np.cos(2.0 * np.pi * rng.integers(1, 5) * np.arange(n) / n)
            a = np.maximum(a, 0)
            a /= a.sum()
            v[:] = 0
            stall = 0
            continue
        g = soft_grad(a, c, n, tk, sb)
        g -= g.mean()
        v = beta * v + (1.0 - beta) * g
        acc = False
        for sh in range(8):
            stp = lr / (2.0 ** sh)
            at = a - stp * v
            at = np.maximum(at, 0)
            sa = at.sum()
            if sa < 1e-14:
                break
            at /= sa
            ct = fft_conv(at, fs)
            if float(np.max(ct[:2*n-1])) <= M + 1e-15:
                a = at
                acc = True
                break
        if not acc:
            a = np.maximum(a - 0.005 * lr * v, 0)
            a /= a.sum()
        lr = max(0.02, lr * 0.99997)
        beta = min(0.96, beta + 1e-6)
        sb = min(1000.0, sb * 1.0005)
        if it % 8000 == 0 and tk < 40:
            tk += 2
    return ba, bM

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 2 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        parent = np.ones(1024, dtype=np.float64)
        n = 1024
        s0 = 1024.0
    fs = 1 << (2 * n - 1).bit_length()
    best = parent.copy()
    bs = float(evaluate_sequence(best.tolist()))
    
    # Phase 1: main optimization (65% budget)
    t1 = time.monotonic() + 0.65 * (dl - time.monotonic())
    a1, _ = run_opt(parent / s0, n, fs, t1)
    c1 = np.clip(a1 * s0, 0.0, 1000.0)
    s1 = float(evaluate_sequence(c1.tolist()))
    if np.isfinite(s1) and s1 < bs:
        bs, best = s1, c1
    
    # Phase 2: refine (15% budget)
    rem = dl - time.monotonic()
    if rem > 30:
        t2 = time.monotonic() + 0.15 * (dl - time.monotonic())
        a2, _ = run_opt(best / float(best.sum()), n, fs, t2)
        c2 = np.clip(a2 * s0, 0.0, 1000.0)
        s2 = float(evaluate_sequence(c2.tolist()))
        if np.isfinite(s2) and s2 < bs:
            bs, best = s2, c2
    
    # Phase 3: local perturbations (remaining)
    base = best.copy()
    while time.monotonic() < dl - 5.0:
        c = base.copy()
        w = max(2, n // 100)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            c[st:e] *= float(np.exp(rng.normal(0, 0.008)))
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < bs:
            bs, best, base = sc, c, c
    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
