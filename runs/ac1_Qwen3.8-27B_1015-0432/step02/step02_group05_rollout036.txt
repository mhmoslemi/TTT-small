import numpy as np
from scipy.optimize import linprog
import time

def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def lp_dir(f, fsz, K, eps, tl):
    n = len(f)
    c = fft_conv(f, fsz)
    M = float(c.max())
    if K >= 2 * n - 1:
        act = np.arange(2 * n - 1)
    else:
        act = np.sort(np.argpartition(c, -K)[-K:])
    A = np.zeros((len(act), n))
    for ii in range(len(act)):
        k = int(act[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[ii, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    b = c[act].copy() + (M - c[act]) / (2.0 * eps)
    mf = float(f.max())
    cap = 10.0 * mf + 1e-10
    res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                  bounds=[(0.0, cap)] * n,
                  method='highs', options={'time_limit': tl})
    if res.status != 0:
        return None, M
    y = np.maximum(res.x, 0.0)
    return (1.0 - eps) * f + eps * y, M

def grad_topk(f, c, tk, fsz):
    n = f.size
    idx = np.argpartition(c, -tk)[-tk:]
    g = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi+1] += f[kk-hi:kk-lo+1][::-1]
    return g / tk

def polish(f, fsz, dl, n_iter):
    n = len(f)
    f = np.maximum(f, 1e-15)
    f = f / f.sum()
    best = f.copy()
    best_m = fft_conv(f, fsz).max()
    eta = 0.1
    for it in range(n_iter):
        if time.monotonic() > dl:
            break
        c = fft_conv(f, fsz)
        m = c.max()
        if m < best_m:
            best_m = m
            best = f.copy()
        T = min(64, 2*n-1)
        g = grad_topk(f, c, T, fsz)
        eta *= 0.9995
        f *= np.exp(-eta * g)
        f = np.maximum(f, 1e-18)
        f /= f.sum()
    return best

def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def build_A(f, act, n):
    K = len(act)
    A = np.zeros((K, n))
    for ii in range(K):
        k = int(act[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[ii, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A

def lp_step(f, fsz, Kc, eps, tl):
    n = len(f)
    c = fft_conv(f, fsz)
    M = float(c.max())
    if Kc >= 2 * n - 1:
        act = np.arange(2 * n - 1)
    else:
        act = np.sort(np.argpartition(c, -Kc)[-Kc:])
    A = build_A(f, act, n)
    b = c[act].astype(np.float64) + (M - c[act]) / (2.0 * eps)
    cap = 10.0 * float(f.max()) + 1e-8
    bnds = Bounds(lb=0.0, ub=cap)

                  method='highs', options={'time_limit': max(1.0, tl)})

        return None, M
    y = np.maximum(res.x, 0.0)
    if y.sum() < 1e-15:
        return None, M
    return (1.0 - eps) * f + eps * y, M
