        (0.50, 5e-5, 0.85, 5e-4, 55, 3000, False, 345),
        (0.30, 5e-5, 0.00, 3e-4, 60, 2500, True,  245),
        (0.20, 8e-5, 0.88, 2e-4, 45, 2000, False, 145),
        (0.15, 8e-5, 0.00, 2e-4, 50, 2000, True,   75),
        (0.10, 1.2e-4, 0.90, 1e-4, 35, 1500, False, 20),
    ]
    for eta0, decay, beta, frac, cap, smx, multi, reserve in phases:
        if time.monotonic() >= t_end - reserve:
            break
        a, s = _desc(best, fs, ts, t_end - reserve + 5,
                     eta0, decay, beta, frac, cap, smx, multi, rng)
        if s < best_sc:
            best_sc = s
            best = a

    # Local search with exact evaluator, slowly shrinking perturbation
    if time.monotonic() < t_end - 3:
        base = best.copy()
        w = max(3, n // 80)
        scale = 0.08
        while time.monotonic() < t_end - 2:
            cand = base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - w)))
                e = min(n, st + w)
                cand[st:e] *= float(np.exp(rng.normal(0.0, scale)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand
            scale = max(0.005, scale * 0.9995)

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = p
    return [float(x) for x in out]
