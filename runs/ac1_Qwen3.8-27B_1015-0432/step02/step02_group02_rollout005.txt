    # g[1] = 2*(p1+2p2) = 2p1+4p2 ✓
    
    # Great, the formula is correct!
    
    L = _fpow2(3 * n - 2)
    pr = p[::-1]
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    
    # Full objective gradient:
    # F(a) = 2n * mv / S^2
    # dF/da = (2n/S^2) * g - 4n*mv/S^3 * ones
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
    return (2.0 * n * mv) / (S * S), grad
