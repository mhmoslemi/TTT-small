    A = csr_matrix((vals, (rows, cols)), shape=(len(idx), n))
    try:
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(len(idx), M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 4, 'presolve': True})
    except Exception:
        return None
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def optimize_phase(f, deadline, rng, beta0, eta0, use_lp, lp_k):
    n = f.size
    f = f.astype(np.float64).copy()
    f /= f.sum()
    beta = beta0
    eta = eta0
    best_f = f.copy()
    best_sc = score_norm(f)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        sc = 2.0 * n * M
        if sc < best_sc - 1e-11:
            best_sc = sc
            best_f = f.copy()
            stall = 0
        elif it % 80 == 0:
            stall += 1
            if stall > 35:
                f = best_f * np.exp(0.06 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                beta = beta0
                eta = eta0
                it = 0
                continue
        g = grad_top(f, b, beta)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        it += 1
        if it % 150 == 0:
            beta *= 1.05
            eta = max(eta * 0.992, 0.002)
        if use_lp and it % 400 == 200 and time.monotonic() < deadline - 8:
            g_lp = lp_step(f, b, M, lp_k)
            if g_lp is not None:
                for t in [0.2, 0.4]:
                    fn = (1.0 - t) * f + t * g_lp
                    fn = np.maximum(fn, 1e-20)
                    fn /= fn.sum()
                    scn = score_norm(fn)
                    if scn < best_sc:
                        best_sc = scn
                        best_f = fn.copy()
                        f = fn.copy()
    return best_f, best_sc


def make_ramp(n, pwr, rev):
    x = np.arange(n, dtype=np.float64) + 1.0
    r = x ** pwr
    if rev:
        r = r[::-1]
    return r / r.sum()


def make_arc(n, center, width, power=0.7):
    x = np.arange(n, dtype=np.float64)
    u = x / max(n - 1, 1)
    v = (u - center) / width
    a = np.maximum(0.0, 1.0 - v * v) ** power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_bump(n, center, width):
    x = np.arange(n, dtype=np.float64)
    u = x / max(n - 1, 1)
    v = (u - center) / width
    a = np.exp(-v * v * 2.0)
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n, dtype=np.float64)
    p = p / p.sum()

    inits = [p, p[::-1].copy()]
    for pwr in [0.3, 0.5, 0.8, 1.2, 1.5, 2.0, 3.0]:
        inits.append(make_ramp(n, pwr, False))
        inits.append(make_ramp(n, pwr, True))
    for c, w in [(0.25, 0.8), (0.35, 0.7), (0.5, 0.6), (0.6, 0.5), (0.7, 0.4)]:
        inits.append(make_arc(n, c, w))
        inits.append(make_bump(n, c, w * 0.5))

    best_f = inits[0]
    best_sc = np.inf
    for ini in inits:
        sc = score_norm(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    # Phase 1: broad exploration
    d1 = t0 + budget_s * 0.30
    f1, s1 = optimize_phase(best_f, d1, rng, 8.0, 0.10, False, 0)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()

    # Phase 2: medium focus
    d2 = t0 + budget_s * 0.55
    f2, s2 = optimize_phase(best_f, d2, rng, 40.0, 0.04, False, 0)
    if s2 < best_sc:
        best_sc, best_f = s2, f2.copy()

    # Phase 3: high focus + LP
    d3 = t0 + budget_s * 0.78
    f3, s3 = optimize_phase(best_f, d3, rng, 120.0, 0.015, True, 60)
    if s3 < best_sc:
        best_sc, best_f = s3, f3.copy()

    # Phase 4: fine tuning + LP
    d4 = deadline - 15.0
    f4, s4 = optimize_phase(best_f, d4, rng, 250.0, 0.006, True, 100)
    if s4 < best_sc:
        best_sc, best_f = s4, f4.copy()

    # Mirror blend
    if time.monotonic() < deadline - 10:
        flip = best_f[::-1].copy()
        for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
            bl = (1.0 - t) * best_f + t * flip
            bl = np.maximum(bl, 1e-20)
            bl /= bl.sum()
            sc_b = score_norm(bl)
            if sc_b < best_sc:
                best_sc, best_f = sc_b, bl.copy()

    # Small perturbations
    if time.monotonic() < deadline - 5:
        for _ in range(15):
            trial = best_f * np.exp(0.008 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc_t = score_norm(trial)
            if sc_t < best_sc:
                best_sc, best_f = sc_t, trial.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc) or not result:
        result = [1.0 / n] * n
    return result
