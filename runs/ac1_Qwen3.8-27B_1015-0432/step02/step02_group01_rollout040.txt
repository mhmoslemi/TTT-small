import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def compute_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    # Smooth-max weighting
    x = beta * (b / M - 1.0)
    # Clip for numerical stability
    x = np.clip(x, -50.0, 0.0)
    w = np.exp(x)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        idx = np.argmax(b)
        w[idx] = 1.0
        ws = 1.0
    w /= ws
    # Gradient: grad[k] = 2 * sum_j w_j * a_{j-k} for valid j-k
    # Using the code's approach:
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    n = f.size
    grad = gc[n-1:2*n-1].copy()
    return np.maximum(grad, 0.0)

def optimize(f, deadline, rng, beta0, eta0, patience=200):
    n = f.size
    f = f.astype(np.float64).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    
    best_f = f.copy()
    best_sc = score_norm(f)
    beta = beta0
    eta = eta0
    no_improve = 0
    
    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        sc = 2.0 * n * b.max()
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= patience:
                # Restart with perturbation
                f = best_f * np.exp(0.1 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                m_buf = np.zeros(n)
                v_buf = np.zeros(n)
                beta = beta0
                eta = eta0
                no_improve = 0
                continue
        
        g = compute_grad(f, b, beta)
        if g is None or g.max() < 1e-30:
            break
        g /= g.max()
        
        t += 1
        lr = eta / (1 + t * 0.001)
        m_buf = 0.9 * m_buf + 0.1 * g

        m_hat = m_buf / (1 - 0.9**t)

        # Slowly increase beta, decrease eta
        if it % 200 == 0:
            beta *= 1.03
            eta *= 0.995
    
    return best_f, best_sc
