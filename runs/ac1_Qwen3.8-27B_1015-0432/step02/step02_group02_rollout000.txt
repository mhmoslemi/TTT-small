    return m

    L = _fpow2(2 * n - 1)

    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    S = a.sum()
    if S < 1e-8:
