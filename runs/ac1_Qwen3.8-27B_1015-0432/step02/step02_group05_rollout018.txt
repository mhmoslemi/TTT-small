def grad_topk(a, cv, T):
    n = a.size
    T = min(T, 2 * n - 1)
    idx = np.argpartition(cv, -T)[-T:]
    g = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    g /= T
    return g

def run_mu(a, n, fsz, dl, T0, Tf, e0, ef, rng, sm=0.25, stall_lim=3000):
    a = np.maximum(a, 1e-18).copy()
    a /= a.sum()
    best = a.copy()
    best_m = fft_conv(a, fsz).max()
    stall = 0
    gs = np.zeros(n)
    it = 0
    while time.monotonic() < dl and it < 80000:
        it += 1
        f = min(1.0, it / 15000)
        T = max(2, int(round(T0 + (Tf - T0) * f)))
        eta = e0 + (ef - e0) * f
        cv = fft_conv(a, fsz)
        m = cv.max()
        if m < best_m - 1e-14:
            best_m = m
            best = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_lim:
            a = best.copy()
            a *= np.exp(rng.normal(0, 0.008, n))
            a = np.maximum(a, 1e-18)
            a /= a.sum()
            gs[:] = 0.0
            stall = 0
            continue
        g = grad_topk(a, cv, T)
        gs += sm * (g - gs)
        a *= np.exp(-eta * gs)
        a = np.maximum(a, 1e-18)
        a /= a.sum()
    return best, best_m

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 12.0)
    par = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if par.size < 2 or not np.isfinite(par).all() or par.sum() < 0.01:
        par = np.ones(1024, dtype=np.float64)
    n = par.size
    S = float(par.sum())
    fsz = 1 << (2 * n - 1).bit_length()
    a0 = par / S
    best = a0.copy()
    best_sc = float(evaluate_sequence(par.tolist()))

    inits = [a0.copy()]
    x = np.arange(n, dtype=np.float64)
    gm = np.exp(-((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits.append(gm / gm.sum())
    r = np.abs(rng.standard_normal(n))
    for _ in range(4):
        r = 0.5 * (r + np.roll(r, 1))
    inits.append(r / r.sum())
    t1 = np.clip(1.0 - 0.7 * np.abs(2.0 * x / n - 1.0), 0.01, None)
    inits.append(t1 / t1.sum())

    # Primary: longest run on parent
    per1 = min(dl - time.monotonic(), 500.0)
    a_opt, _ = run_mu(inits[0], n, fsz, time.monotonic() + per1,
                       16, 4, 0.04, 0.003, rng)
    sc = float(evaluate_sequence((a_opt * S).tolist()))
    if np.isfinite(sc) and sc < best_sc:
        best_sc = sc
        best = a_opt.copy()

    # Secondary initializations
    for k in range(1, len(inits)):
        rem = dl - time.monotonic()
        if rem < 40.0:
            break
        per = min(rem / (len(inits) - k), 150.0)
        T0 = [16, 8, 12][k - 1]
        Tf = [4, 4, 4][k - 1]
        e0 = [0.05, 0.06, 0.04][k - 1]
        a_opt, _ = run_mu(inits[k], n, fsz, time.monotonic() + per,
                           T0, Tf, e0, 0.002, rng)
        sc = float(evaluate_sequence((a_opt * S).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a_opt.copy()

    # Continue polishing the best
    rem = dl - time.monotonic()
    if rem > 60.0:
        a_opt, _ = run_mu(best, n, fsz, time.monotonic() + rem * 0.7,
                           8, 2, 0.02, 0.001, rng, sm=0.15, stall_lim=4000)
        sc = float(evaluate_sequence((a_opt * S).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a_opt.copy()

    # Local search perturbation
    rem = dl - time.monotonic()
    if rem > 10.0:
        base = best.copy()
        w = max(2, n // 400)
        while time.monotonic() < dl - 2.0:
            c = base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - w)))
                c[st:min(n, st + w)] *= float(np.exp(rng.normal(0, 0.004)))
            c = np.maximum(c, 0.0)
            if c.sum() < 0.01:
                continue
            sc = float(evaluate_sequence((c * S).tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = c.copy()
                base = c.copy()

    out = np.clip(best * S, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(v) for v in out]
