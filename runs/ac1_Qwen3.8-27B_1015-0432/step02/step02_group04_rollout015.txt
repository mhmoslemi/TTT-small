import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def div_freq(f, rng, k=6):
