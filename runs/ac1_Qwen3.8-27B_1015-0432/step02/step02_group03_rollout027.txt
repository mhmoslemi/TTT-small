    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(512)
    n = p.size
    fs = 1 << (2 * n - 1).bit_length()

    best = p.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    # Phase 1: Wide Euclidean descent
    dl1 = min(dl, t0 + 220)
    a, s = _desc(p.copy(), fs, dl1, 0.5, 5e-4, 60, False, rng, 4000, 800)
    if s < best_sc:
        best_sc, best = s, a

    # Phase 2: Wide Multiplicative descent
    if time.monotonic() < dl - 150:
        dl2 = min(dl, time.monotonic() + 280)
        a, s = _desc(best.copy(), fs, dl2, 0.3, 3e-4, 90, True, rng, 5000, 1000)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 3: Moderate Euclidean polish
    if time.monotonic() < dl - 120:
        dl3 = min(dl, time.monotonic() + 230)
        a, s = _desc(best.copy(), fs, dl3, 0.2, 1.5e-4, 35, False, rng, 5000, 800)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 4: Narrow Multiplicative polish
    if time.monotonic() < dl - 70:
        dl4 = min(dl, time.monotonic() + 170)
        a, s = _desc(best.copy(), fs, dl4, 0.12, 1e-5, 10, True, rng, 5000, 600)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 5: Very narrow Euclidean final polish
    if time.monotonic() < dl - 30:
        dl5 = min(dl, time.monotonic() + 100)
        a, s = _desc(best.copy(), fs, dl5, 0.08, 5e-6, 5, False, rng, 4000, 500)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 6: Exact local perturbation search
    if time.monotonic() < dl - 3:
        base = best.copy()
        w = max(2, n // 300)
        while time.monotonic() < dl - 1:
            cand = base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - w)))
                e = min(n, st + w)
                cand[st:e] *= np.exp(rng.normal(0.0, 0.015))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = np.clip(p, 0.0, 1000.0)
    return [float(x) for x in out]
