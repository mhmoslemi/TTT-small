            e = min(n, st + w)
            cand[st:e] *= np.exp(float(rng.normal(0.0, 0.012)))
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s < 1e-15:
            continue
        cand = cand / s
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_arr, base = sc, cand, cand

    out = np.clip(best_arr, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
