    for c in [-1.0, -0.5, 0.0, 0.5, 1.0]:
        j = np.arange(n, dtype=np.float64)
        w = np.exp(c * (2.0 * j / max(n - 1, 1) - 1.0))
        w = np.maximum(w, 1e-10)
        inits.append(w / w.sum())
    for cn, w in [(0.35, 0.2), (0.5, 0.15), (0.65, 0.25), (0.5, 0.4)]:
        j = np.arange(n, dtype=np.float64)
        w2 = np.exp(-0.5 * ((j - n * cn) / (n * w)) ** 2)
        w2 = np.maximum(w2, 1e-15)
        inits.append(w2 / w2.sum())
    if pp.size == n and pp.sum() > 1e-10:
        for sc in [0.05, 0.12, 0.25]:
            t = pp * np.exp(sc * rng.standard_normal(n)) / pp.sum()
            t = np.maximum(t, 1e-20)
            inits.append(t / t.sum())
    return inits

def l2step(f, C, loss, base, rn2, eps, it):
    n = f.size
    idx = np.argpartition(C, -base)[-base:]
    s = np.zeros(C.size)
    s[idx] = 1.0 / base
    g = topl_grad(f, C, base) / (2.0 * len(idx))
    g -= g.mean()
    al = min(rn2 / (np.dot(g, g) + 1e-30), 0.06)
    cand = f - al * g
    cand = np.maximum(cand, eps)
    if it == 0 or it % 20 == 0:
        cand -= cand.min()
        cand = np.maximum(cand, eps)
    cand /= cand.sum()
    return cand

def l2phase(f, dl, rng):
    n = f.size
    C = fft_conv(f, f)
    loss = float(C.max())
    best_f, best_l = f.copy(), loss
    f = f.copy()
    eta = 0.8
    base = 8
    rn2 = (0.02 * f.max()) ** 2
    eps = 1e-30
    base_l = base
    plate = 0
    for it in range(80000):
        if time.monotonic() > dl:
            break
        if it % 8 == 0:
            C = fft_conv(f, f)
            loss = float(C.max())
            if loss < best_l - 1e-12:
                best_l = loss
                best_f = f.copy()
                plate = 0
            else:
                plate += 1
                eta = max(0.3, eta * 0.96)
        g = topl_grad(f, C, base) / (2.0 * base)
        f = f * np.exp(-eta * g / (g.max() + 1e-30))
        f = np.maximum(f, eps)
        f /= f.sum()
        if it % 15 == 0:
            al = min(rn2 / ((2.0 * base) ** 2 * np.dot(g, g) + 1e-30), 0.05)
            cand = np.maximum(f - al * g, eps)
            cand /= cand.sum()
            nl = float(fft_conv(cand, cand).max())
            if nl < loss:
                f = cand
                loss = nl
        if plate > 250:
            f = best_f.copy()
            C = fft_conv(f, f)
            loss = best_l
            f = f * np.exp(0.04 * rng.standard_normal(n))
            f = np.maximum(f, eps)
            f /= f.sum()
            eta = 0.7
            base = base_l
            plate = 0
        if it % 300 == 0:
            base = max(4, int(base * 0.97))
            rn2 *= 0.98
    return best_f, best_l

def lp_dir(f, C):
    n = f.size
    M = float(C.max())
    g = np.ones(n) / n
    y = np.ones(C.size) / C.size
    N = 1 << (C.size + n - 1).bit_length()
    Fd = np.fft.rfft(f, N)
    cF = np.conj(Fd)
    for _ in range(60):
        fg = fft_conv(f, g)
        da = 0.4 * (fg - M)
        da -= da.max()
        y = y * np.exp(da)
        Aty = np.fft.irfft(cF * np.fft.rfft(y, N), N)[:n]
        pa = 0.4 * (1.0 - Aty)
        pa -= pa.max()
        g = np.maximum(g * np.exp(pa), 1e-30)
        g /= g.sum()
    return 0.4 * f + 0.6 * g

def lp_phase(f, dl, rng):
    n = f.size
    best_f, best_s = f.copy(), fast_score(f)
    for _ in range(30):
        if time.monotonic() > dl:
            break
        C = fft_conv(best_f, best_f)
        d = lp_dir(best_f, C)
        for t in [0.1, 0.25, 0.4, 0.6]:
            c = np.maximum((1.0 - t) * best_f + t * d, 1e-20)
            c /= c.sum()
            sc = fast_score(c)
            if np.isfinite(sc) and sc < best_s - 1e-12:
                best_s = sc
                best_f = c.copy()
        tl = min(30.0, dl - time.monotonic() - 5.0)
        if tl > 5.0:
            f2, l2 = l2phase(best_f, time.monotonic() + tl, rng)
            s2 = 2.0 * n * l2
            if s2 < best_s:
                best_s, best_f = s2, f2
    return best_f, best_s

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 12.0
    n = len(height_sequence_1)
    inits = make_initials(n, height_sequence_1, rng)
    best_f, best_s = None, np.inf
    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_s:
            best_s, best_f = sc, ini.copy()
    if best_f is None:
        best_f, best_s = np.ones(n) / n, 2.0
    phase = 0
    while time.monotonic() < dl - 25.0:
        phase += 1
        tl = min(250.0, dl - time.monotonic() - 10.0)
        f1, l1 = l2phase(best_f, time.monotonic() + tl, rng)
        if 2.0 * n * l1 < best_s:
            best_s, best_f = 2.0 * n * l1, f1
        if time.monotonic() > dl - 35.0:
            break
        tl2 = min(180.0, dl - time.monotonic() - 10.0)
        f2, s2 = lp_phase(best_f, time.monotonic() + tl2, rng)
        if s2 < best_s:
            best_s, best_f = s2, f2
        if phase % 2 == 0 and time.monotonic() < dl - 60.0:
            for sc in [0.05, 0.12, 0.2]:
                tr = best_f * np.exp(sc * rng.standard_normal(n))
                tr = np.maximum(tr, 1e-20)
                tr /= tr.sum()
                tl3 = min(35.0, dl - time.monotonic() - 10.0)
                if tl3 > 8.0:
                    f3, l3 = l2phase(tr, time.monotonic() + tl3, rng)
                    if 2.0 * n * l3 < best_s:
                        best_s, best_f = 2.0 * n * l3, f3
        if phase >= 14 and time.monotonic() > dl - 120.0:
            break
    fb = best_f.copy()
    for it in range(80):
        if time.monotonic() > dl - 5.0:
            break
        C = fft_conv(fb, fb)
        base = max(4, 40 - 2 * (it // 10))
        g = topl_grad(fb, C, base) / (2.0 * base)
        g -= g.mean()
        gs = np.dot(g, g) + 1e-30
        al = min(0.02 / np.sqrt(gs), 0.03)
        c = np.maximum(fb - al * g, 1e-20)
        c /= c.sum()
        if float(fft_conv(c, c).max()) < float(C.max()):
            fb = c
    fb = fb / fb.sum()
    result = [float(x) for x in fb]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs < 0.01:
        result = [float(x) for x in np.ones(n)]
    return result
