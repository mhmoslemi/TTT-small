import numpy as np
import time

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def active_set_grad(a, conv, thresh_frac=0.999):
    """Compute subgradient at the active set (peaks near max)."""
    n = a.size
    M = conv.max()
    thresh = M * thresh_frac
    # Find indices of conv near the max
    active = np.where(conv >= thresh)[0]
    # Limit to top 32 for efficiency
    if len(active) > 32:
        # Get the 32 largest
        idx = np.argpartition(conv, -32)[-32:]
        active = idx
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # d(conv[k])/d(a[j]) = a[k-j] for j in [lo, hi]
            grad[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return grad

def run_descent(a, n, fft_size, deadline, best, best_score):
    """Main descent loop."""
    v = np.zeros(n)
    beta = 0.85
    lr = 0.3
    stall = 0
    prev_M = np.inf
    local_best = best.copy()
    local_score = best_score
    
    # Normalize a to sum=1 for the descent
    a = a / a.sum()
    
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        conv = fft_conv(a, fft_size)
        M = float(conv[n-1])  # For nonneg, peak is at n-1

        M = float(conv.max())
        
        if M < prev_M - 1e-14:

        else:

        prev_M = M
        
        if stall > 8000 and it > 1000:

        grad = active_set_grad(a, conv)
        v = beta * v + (1 - beta) * grad
        # Don't mean-center (that would change the sum)
        # Instead, project: after update, renormalize
        
        # Line search
        accepted = False
        for bs in range(8):

            if s < 1e-15:

import numpy as np
import time


def _fft_convolve(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def _active_gradient(a, conv, n, active):
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad


def _descent_phase(a, n, fft_size, deadline, best, best_score, stall_limit, forker):
    v = np.zeros(n, dtype=np.float64)
    momentum = 0.82
    lr = 0.25
    stall = 0
    prev_peak = np.inf
    local_best = best.copy()
    local_score = best_score
    conv_len = 2 * n - 1
    norm = a.sum()

    for it in range(500000):
        if time.monotonic() > deadline:
            break
        conv = _fft_convolve(a, fft_size)
        peak = float(conv[conv_len].max()) if conv_len <= conv.size else float(np.max(conv[:conv_len]))
        if peak < prev_peak - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_peak = peak
        if stall > stall_limit and it > 200:
            break
        thresh = peak * (1.0 - 1e-6)
        active = np.where(conv[:conv_len] >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv[:conv_len]))])
        if len(active) > 24:
            active = np.argpartition(conv[:conv_len], -24)[-24:]
        grad = _active_gradient(a, conv, n, active)

            if np.isfinite(sc) and sc < local_score:

                local_best = cand.copy()

def _perturb(base, rng, n, w_lo, w_hi):

    w = int(rng.integers(w_lo, w_hi + 1))
    for _ in range(int(rng.integers(1, 4))):
        st = int(rng.integers(0, max(1, n - w)))
        e = min(n, st + w)
        cand[st:e] *= np.exp(float(rng.normal(0.0, 0.03)))
    return np.clip(cand, 0.0, 1000.0)
