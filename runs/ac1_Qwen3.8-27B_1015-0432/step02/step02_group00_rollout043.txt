def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    return float(2.0 * f.size * b.max() / (s * s))

def make_grad(f, b, K, tau):

    # Soft weights: how close each b_k is to the max

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    return float(2.0 * f.size * b.max() / (s * s))

def grad(f, b, K, tau):
    n = f.shape[0]

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
