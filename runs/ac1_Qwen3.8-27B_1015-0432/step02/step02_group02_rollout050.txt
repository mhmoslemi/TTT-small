        inits.append(_normalize(g, scale))

    for sig in [0.06, 0.1, 0.15]:
        g = (np.exp(-0.5 * ((x - 0.25 * n) / (sig * n)) ** 2) +
             np.exp(-0.5 * ((x - 0.75 * n) / (sig * n)) ** 2))
        inits.append(_normalize(g, scale))

    bx = np.zeros(n)
    bx[n // 5:4 * n // 5] = 1.0
    inits.append(_normalize(bx, scale))

    inits.append(np.ones(n))

    # Sine-modulated parent
    sm = np.clip(par / par.sum(), 0, None) * (1.0 + 0.4 * np.sin(4.0 * np.pi * x / n))
    inits.append(_normalize(sm, scale))

    for inp in inits:
        sc = float(evaluate_sequence(inp.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = inp.copy()
            best_sc = sc
            cur = inp.copy()

    # --- Main L-BFGS-B annealing ---
    bounds_list = [(0.0, 1000.0)] * n
    taus = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 90.0:
            break
        iters = 600 if i < 3 else 400
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': iters, 'gtol': 1e-14,
                                    'maxls': 60, 'ftol': 1e-19,
                                    'maxfun': iters * 5})
            nx = _normalize(res.x, scale)
            sc = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # --- Second pass: finer tau from best ---
    if time.monotonic() < dl - 60.0:
        cur = best.copy()
        taus2 = [0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
        for tv in taus2:
            if time.monotonic() > dl - 30.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 400, 'gtol': 1e-14,
                                        'maxls': 60, 'ftol': 1e-19,
                                        'maxfun': 2000})
                nx = _normalize(res.x, scale)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # --- Peak-targeted local search ---
    stall = 0
    while time.monotonic() < dl - 5.0:
        strategy = rng.choice(5)
        cd = best.copy()

        if strategy == 0:
            nw = int(rng.integers(1, max(2, n // 25)))
            s = int(rng.integers(0, n - nw))
            cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.04)))
        elif strategy == 1:
            c = _conv(cd)
            kmax = int(c.argmax())
            ctr = kmax // 2
            rad = max(1, n // 60)
            lo, hi = max(0, ctr - rad), min(n, ctr + rad + 1)
            cd[lo:hi] *= float(np.exp(rng.normal(-0.015, 0.01)))
            mask = np.ones(n, dtype=bool)
            mask[lo:hi] = False
            idx = np.where(mask)[0]
            if len(idx) > 0:
                pick = idx[rng.integers(0, len(idx), size=min(20, len(idx)))]
                cd[pick] *= float(np.exp(rng.normal(0.015, 0.01, size=len(pick))))
        elif strategy == 2:
            c = _conv(cd)
            top_k = np.argsort(c)[-5:]
            for kk in top_k:
                ctr = int(kk) // 2
                rad = max(1, n // 80)
                lo, hi = max(0, ctr - rad), min(n, ctr + rad + 1)
                cd[lo:hi] *= 1.0 - 0.01 * rng.random()
        elif strategy == 3:
            sigma = 0.005 + 0.02 * (1.0 - (time.monotonic() - t0) / max(1, dl - t0))
            cd *= float(np.exp(rng.normal(0.0, sigma)))
        else:
            nw = int(rng.integers(1, max(2, n // 15)))
            s = int(rng.integers(0, n - nw))
            cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.08)))

        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2
                stall = 0
            else:
                stall += 1
        if stall > 5000:
            cur = best.copy()
            stag = 0.02
            cd = cur * np.exp(rng.normal(0.0, stag, size=n))
            cd = np.clip(cd, 0.0, 1000.0)
            sc = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = cd.copy()
                best_sc = sc
            stall = 0

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n)
    return [float(v) for v in best]
