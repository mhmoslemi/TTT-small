        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand.copy()
            best_a = a.copy()
    
    # Main optimization from best initialization
    a = best_a.copy()
    v = np.zeros(n)
    beta = 0.5
    lr = 0.1
    stall = 0
    prev_M = np.inf
    
    while time.monotonic() < deadline - 60.0:
        conv = fft_convolve(a, fft_size)
        M = float(np.max(conv))
        
        if M < prev_M - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_M = M
        
        if stall > 5000:
            break
        
        grad = grad_max(a, conv, fft_size, n)
        v = beta * v + (1 - beta) * grad
        
        # Step with projection onto simplex
        for _ in range(5):
            a_try = a - lr * v
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-14:
                break
            a_try = a_try / s_try
            conv_try = fft_convolve(a_try, fft_size)
            if float(np.max(conv_try)) <= M * (1 + 1e-10):
                a = a_try
                # Update best
                cand = np.clip(a * s0, 0.0, 1000.0)
