                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Final projected gradient
    if time.monotonic() < dl - 40.0:
        tau = 1e-4
        alpha = 0.02
        for it in range(2000):
            if time.monotonic() > dl - 15.0:
                break
            if it % 300 == 0:
                alpha *= 0.92
            cur = _pg_step(cur, tau, alpha)
            if it % 400 == 0:
                sc = float(evaluate_sequence(cur.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc

    # Local perturbation search
    while time.monotonic() < dl - 3.0:
        cd = best.copy()
        nw = int(rng.integers(1, max(2, n // 30)))
        s = int(rng.integers(0, n - nw))
        cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.015)))
        if rng.random() < 0.08:
            cd *= float(np.exp(rng.normal(0.0, 0.008)))
        if rng.random() < 0.05:
            sh = int(rng.integers(-n // 10, n // 10))
            cd = np.roll(cd, sh)
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2

    best = np.clip(best, 0.0, 1000.0)
    if best.size == 0 or best.sum() < 0.01 or not np.all(np.isfinite(best)):
        best = np.ones(n) if n > 0 else np.array([1.0])
    return [float(v) for v in best]
