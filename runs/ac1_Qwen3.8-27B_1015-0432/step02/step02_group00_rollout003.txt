def fft_conv(a, b):
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:na + nb - 1]

def norm1(f):
    f = np.clip(f, 1e-30, None)
    s = f.sum()
    return f / s if s > 1e-15 else np.ones(len(f)) / len(f)

    s = f.sum()
    if s < 0.01:
        return np.inf
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))
