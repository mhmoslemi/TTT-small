def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def grad_topk(a, cv, tk):
    n = a.size
    T = min(tk, 2 * n - 1)
    idx = np.argpartition(cv, -T)[-T:]
    g = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi+1] += a[kk-hi:kk-lo+1][::-1]
    g /= T
    return g
