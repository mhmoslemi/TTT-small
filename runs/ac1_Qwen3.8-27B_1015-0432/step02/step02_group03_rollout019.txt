def _fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _score(a):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    fs = 1 << (2 * n - 1).bit_length()
    c = _fft_conv(a, fs)
    return 2.0 * n * float(c.max()) / (s * s)

def _subgrad_fft(a, active, fs):
    n = a.size
    ind = np.zeros(fs)
    ind[active] = 1.0
    F_ind = np.fft.rfft(ind, n=fs)
    F_a = np.fft.rfft(a, n=fs)
    g = np.fft.irfft(np.conj(F_ind) * F_a, n=fs)[:n]
    return 2.0 * g
