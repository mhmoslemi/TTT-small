import numpy as np
import time

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _grad_contig(a, k_lo, k_hi):
    """O(n) subgradient for contiguous active lag range [k_lo, k_hi]."""
    n = a.size
    m = k_hi - k_lo + 1
    if m <= 0:
        return np.zeros(n)
    ps = np.concatenate(([0.0], np.cumsum(a)))
    j = np.arange(n)
    i_lo = np.maximum(0, k_lo - j)
    i_hi = np.minimum(n - 1, k_hi - j)
    valid = i_lo <= i_hi
    g = np.zeros(n)
    g[valid] = 2.0 * (ps[i_hi[valid] + 1] - ps[i_lo[valid]])
    g /= m
    return g

def _grad_general(a, lags):
    """Subgradient for arbitrary active lag set."""
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    m = len(lags)
    if m:
        g /= m
    return g

def _score(a, fs):
    c = _fc(a, fs)
    return 2.0 * a.size * float(c.max()) / (a.sum()**2)

def _desc(a, fs, ts, dl, eta0, beta, frac0, cap, smx, multi, rng, gamma):
    n = a.size
    mom = np.zeros(n)
    bst = a.copy()
    bsc = _score(bst, fs)
    stall = 0
    prev_sc = bsc
    frac = frac0
    
    for it in range(2000000):
        if time.monotonic() > dl:
            break
        c = _fc(a, fs)
        sc = 2.0 * n * float(c.max()) / (a.sum()**2)
        if sc < bsc - 1e-13:
            bsc = sc
            bst = a.copy()
        stall = 0 if sc < prev_sc - 1e-14 else stall + 1
        prev_sc = sc
        if stall > smx:
            break
        
        mx = float(c.max())
        # Adaptive fraction: widen as we stall
        frac_eff = min(frac * (1.0 + 0.3 * stall / max(1, smx//4)), 0.5)
        act = np.where(c >= mx * (1.0 - frac_eff))[0]
        act = act[(act >= 0) & (act < 2*n-1)]
        
        if len(act) > cap:
            ii = np.argpartition(c[act], -cap)[-cap:]
            act = act[ii]
        
        # Check if active set is roughly contiguous
        if len(act) > 2:
            gaps = np.diff(np.sort(act))
            if gaps.max() <= max(2, len(act) // 20):
                # Contiguous: use fast gradient
                k_lo, k_hi = int(act.min()), int(act.max())
                g = _grad_contig(a, k_lo, k_hi)
            else:
                g = _grad_general(a, act)
        else:
            g = _grad_general(a, act)
        
        g -= g.mean()
        
        if stall > smx // 3:
            gm0 = np.abs(g).max()
            if gm0 > 1e-14:
                g += 0.05 * rng.standard_normal(n) / gm0
        
        if multi:
            gm = np.abs(g).max()
            if gm > 1e-12:
                a = a * np.exp(-eta0 * g / gm)
        else:

            a = np.clip(a - eta0 * mom, 0.0, None)
        
        sa = a.sum()
        if sa < 1e-10:
            break
        a *= ts / sa
        eta0 *= gamma
    
    return bst, bsc
