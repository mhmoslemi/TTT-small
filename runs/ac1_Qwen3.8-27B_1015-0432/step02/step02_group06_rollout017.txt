        templates.append(g)

        templates.append(np.exp(-r * (n - 1 - x) / n))

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    L = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-14:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def gen_inits(n, rng, parent):
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    c = (n - 1) / 2.0
    raw = [parent.copy(),
           np.ones(n),
           0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1))),
           0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t),
           (1.0 - (2.0 * t - 1.0)**2)**2,
           (1.0 - (2.0 * t - 1.0)**2)**4,
           np.exp(-0.5 * ((x - c) / (0.25 * n))**2),
           np.exp(-0.5 * ((x - c) / (0.15 * n))**2),
           np.exp(-1.0 * x / n),
           np.exp(-0.5 * ((x - 0.3*c) / (0.12*n))**2) + np.exp(-0.5 * ((x - 0.7*c) / (0.12*n))**2)]
    out = []
    for r in raw:
        r = np.maximum(r.astype(np.float64), 1e-15)
        out.append(r / r.sum())
    for s_ in [0.08, 0.15, 0.25]:
        p = parent * np.exp(s_ * rng.standard_normal(n))
        p = np.maximum(p, 1e-15)
        out.append(p / p.sum())
    return out

def opt_mult(f, deadline, rng, beta0, eta0, sym=False):
    n = f.size
    f = f.astype(np.float64).copy()
    if sym:
        f = 0.5 * (f + f[::-1])
        f /= f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    beta = float(beta0)
    eta = float(eta0)
    stall = 0
    it = 0
    while time.monotonic() < deadline - 4.0:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        w = np.exp(np.clip(beta * (b / M - 1.0), -60.0, 0.0))
        w -= w.min()
        if w.max() < 1e-25:
            break
        gc = fft_conv(w, f[::-1])
        grad = np.maximum(gc[n - 1:2 * n - 1], 1e-30)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        it += 1
        if it % 15 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 0 and stall % 25 == 0:
                beta = min(5000.0, beta * 1.2)
                eta = max(0.008, eta * 0.94)
            if stall >= 100:
                f = best * np.exp(0.04 * rng.standard_normal(n))
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
                beta = max(beta0 * 0.5, beta * 0.5)
    return best, best_sc

def try_variants(f, rng, n, deadline, label=""):
    best = f.copy()
    bs = fast_score(f)
    # Power re-shaping
    for alpha in [0.85, 0.9, 1.1, 1.3, 1.6, 2.0]:
        if time.monotonic() > deadline - 5.0:
            break
        trial = np.power(f, alpha)
        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if sc < bs:
            bs = sc; best = trial.copy()
    # Small perturbations
    for _ in range(60):
        if time.monotonic() > deadline - 5.0:

        trial = best * np.exp(0.002 * rng.standard_normal(n))
        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if sc < bs:
            bs = sc; best = trial.copy()
    # Mirror blends

        if time.monotonic() > deadline - 5.0:

        trial = (1.0 - mix) * best + mix * best[::-1]
        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if sc < bs:
            bs = sc; best = trial.copy()
    return best, bs

    avail = deadline - t0

    p1_end = t0 + 0.35 * avail

        if sc_opt < best_sc:

            best_f = f_opt.copy()
    # Phase 2: continue best asymmetric + symmetric (35%)

    if time.monotonic() < p2_end - 15.0:
        half = (p2_end - time.monotonic()) / 2.0
        dl1 = min(time.monotonic() + half, p2_end)
        f_o1, s1 = opt_mult(best_f.copy(), dl1, rng, 80.0, 0.08, sym=False)
        if s1 < best_sc:

        if s4 < best_sc:
            best_sc = s4; best_f = f_o4.copy()
    result = [float(v) for v in best_f]

    return result
