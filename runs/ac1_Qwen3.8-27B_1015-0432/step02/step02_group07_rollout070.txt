import numpy as np
import time

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def conv_and_fa(a, nfft):
    fa = np.fft.rfft(a, n=nfft)
    c = np.fft.irfft(fa * fa, n=nfft)
    return fa, c

def norm_nonneg(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-14 or a.size != n:
        # if size mismatch, pad/trunc
        b = np.zeros(n, dtype=np.float64)
        m = min(a.size, n)
        if m > 0: b[:m] = a[:m]
        a = b
        s = float(a.sum())
    if s < 1e-14:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s

    x = norm_nonneg(start, n)

        vals = seg[idx]
        vmax_val = float(np.max(vals))
        if vals.size > 1:
            ptp = vmax_val - float(np.min(vals))
            temp = max(ptp * 0.75, 1e-18)
            u = np.empty(vals.size, dtype=np.float64)

            w = np.exp(u)

import numpy as np

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fft_convolve(a, nfft):
    fa = np.fft.rfft(a, n=nfft)
    return np.fft.irfft(fa * fa, n=nfft)


def score_fft(a, n, nfft):
    c = fft_convolve(a, nfft)
    return float(2.0 * float(n) * float(np.max(c[:2 * n - 1])))


def exact_score_safe(a, limit):
    if a.size > limit:
        return float("inf")

    except Exception:

def norm_nonneg(a, n):
    a = np.asarray(a, dtype=np.float64)

        b = np.zeros(n, dtype=np.float64)
        m = min(a.size, n)

    if side == 0:

    if n <= 8:
        return out
    i = np.arange(n, dtype=np.float64)

    s1 = max(1, n // 8)
    s2 = max(1, n // 4)
    for s in (s1, s2):
        if 0 < s < n:
            out.append(shift_pad(base, s, 0))
            out.append(shift_pad(base, s, 1))

    t = i / float(max(1, n - 1))
    for amp in (0.75, -0.75):
        out.append(norm_nonneg(base * np.exp(amp * (t - 0.5)), n))

    for f in (1, 2):
        out.append(norm_nonneg(base * np.exp(0.25 * np.sin(2.0 * np.pi * float(f) * i / float(n))), n))

    out.append(norm_nonneg(base * np.exp(0.15 * np.cos(np.pi * i)), n))
    out.append(norm_nonneg(base * np.exp(rng.normal(0.0, 0.03, size=n)), n))
    return out[:5]


def search_one(start, n, nfft, deadline, mode, lr_init, top_k_init):
    x = norm_nonneg(start, n)
    L = 2 * n - 1
    conv = fft_convolve(x, nfft)
    seg = conv[:L]
    Mx = float(np.max(seg))

    best = x.copy()
    best_M = Mx
    v = np.zeros(n, dtype=np.float64)

    beta = 0.9
    lr = float(lr_init)
    top_k = int(max(1, top_k_init))
    lr_cap = 2.5 if mode == "add" else 1.2
    lr_floor = 0.02

    it = 0
    max_it = 200000
    best_stall = 0

    while it < max_it and time.monotonic() < deadline:
        it += 1

        if top_k > L:
            top_k = L
        if top_k <= 0:
            top_k = 1

        if top_k >= L:
            idx = np.arange(L)
        else:
            idx = np.argpartition(seg, -top_k)[-top_k:]

        vals = seg[idx]
        v0 = float(np.max(vals))
        if vals.size > 1:
            ptp = v0 - float(np.min(vals))
            temp = max(ptp * 0.75, 1e-18)
            u = np.empty(vals.size, dtype=np.float64)
            u[:] = (vals - v0) / temp
            np.clip(u, -50.0, 0.0, out=u)
            w = np.exp(u)
            sw = float(w.sum())
            if sw <= 0.0 or not np.isfinite(sw):
                w = np.full(vals.size, 1.0 / vals.size)
            else:
                w /= sw
        else:
            w = np.ones(1, dtype=np.float64)

        g = np.zeros(n, dtype=np.float64)
        for ii in range(idx.size):
            wt = float(w[ii])
            if wt <= 0.0:
                continue
            k = int(idx[ii])
            lo = k - n + 1
            if lo < 0:
                lo = 0
            hi = k
            if hi >= n:
                hi = n - 1
            if lo <= hi:
                g[lo:hi + 1] += (2.0 * wt) * x[k - hi:k - lo + 1][::-1]

        g -= float(np.mean(g))
        gmax = float(np.max(np.abs(g)))
        if gmax < 1e-15:
            best_stall += 1
            v *= 0.5
            if best_stall > 20000:
                break
            continue

        v = beta * v + (1.0 - beta) * g
        v -= float(np.mean(v))
        vabs = float(np.max(np.abs(v)))
        if vabs < 1e-15:
            v = g
            vabs = gmax
        d = v / vabs

        base = lr / float(n) if mode == "add" else lr
        accepted = False

        for bl in range(5):
            step = base * (0.6 ** bl)
            if step <= 0.0:
                break

            if mode == "mul":
                expo = -step * d
                np.clip(expo, -5.0, 5.0, out=expo)
                z = (x + 1e-12) * np.exp(expo)
            else:
                z = x - step * d

            z = np.clip(z, 0.0, None)
            sz = float(z.sum())
            if sz < 1e-12:
                continue
            z = z / sz

            conv_z = fft_convolve(z, nfft)
            mz = float(np.max(conv_z[:L]))
            if not np.isfinite(mz):
                continue

            if mz < Mx - 1e-14:
                x = z
                Mx = mz
                conv = conv_z
                seg = conv[:L]
                accepted = True
                break

        if accepted:
            best = x.copy()
            best_M = Mx
            best_stall = 0
            lr = min(lr_cap, lr * 1.04)
            beta = min(0.98, beta + 0.0004)
        else:
            best_stall += 1
            lr = max(lr_floor, lr * 0.85)
            beta = max(0.80, beta - 0.01)
            v *= 0.5
            if best_stall % 1000 == 0 and top_k < 96:
                top_k = min(96, top_k + 8)
            if best_stall > 60000:
                break

    return best, best_M


def update_states(best_fft, best_m, best_exact, best_exact_score, cand, cand_m, limit):
    if cand_m < best_m:
        best_fft = cand
        best_m = cand_m
    if cand.size <= limit:
        sc = exact_score_safe(cand, limit)
        if np.isfinite(sc) and sc < best_exact_score:
            best_exact = cand
            best_exact_score = sc
    return best_fft, best_m, best_exact, best_exact_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    if n < 2:
        parent = np.ones(1024, dtype=np.float64)
        n = 1024

    base = norm_nonneg(parent, n)

    if n < 16:
        return [float(x) for x in base]

    exact_limit = 12000 if n <= 12000 else 0
    nfft = next_pow2(2 * n - 1)

    best_fft = base.copy()
    best_fft_m = score_fft(best_fft, n, nfft)
    best_exact = base.copy()
    best_exact_score = exact_score_safe(best_exact, exact_limit)

    final_deadline = time.monotonic() + max(1.0, float(budget_s) - 18.0)

    now = time.monotonic()
    if now < final_deadline - 20.0:
        dl = now + 0.50 * (final_deadline - now)
        cand, cand_m = search_one(base, n, nfft, dl, "add", 0.70, 8)
        cand = norm_nonneg(cand, n)
        best_fft, best_fft_m, best_exact, best_exact_score = update_states(
            best_fft, best_fft_m, best_exact, best_exact_score, cand, cand_m, exact_limit
        )

    now = time.monotonic()
    if now < final_deadline - 30.0:
        start = best_exact if np.isfinite(best_exact_score) else best_fft
        dl = now + min(0.35 * (final_deadline - now), final_deadline - 60.0)
        cand, cand_m = search_one(start, n, nfft, dl, "mul", 0.15, 8)
        cand = norm_nonneg(cand, n)
        best_fft, best_fft_m, best_exact, best_exact_score = update_states(
            best_fft, best_fft_m, best_exact, best_exact_score, cand, cand_m, exact_limit
        )

    variants = make_variants(base, n, rng)
    if variants:
        now = time.monotonic()
        if now < final_deadline - 30.0:
            dl_total = now + min(0.30 * (final_deadline - now), final_deadline - 20.0)
            per = max(2.0, (dl_total - now) / float(max(1, len(variants))))
            for vv in variants:
                now = time.monotonic()
                if now >= dl_total - 2.0:
                    break
                dl_i = min(dl_total, now + per * 1.5)
                cand, cand_m = search_one(vv, n, nfft, dl_i, "add", 0.35, 6)
                cand = norm_nonneg(cand, n)
                best_fft, best_fft_m, best_exact, best_exact_score = update_states(
                    best_fft, best_fft_m, best_exact, best_exact_score, cand, cand_m, exact_limit
                )

    now = time.monotonic()
    if now < final_deadline - 12.0:
        start = best_exact if np.isfinite(best_exact_score) else best_fft
        cand, cand_m = search_one(start, n, nfft, final_deadline - 10.0, "add", 0.25, 16)
        cand = norm_nonneg(cand, n)
        best_fft, best_fft_m, best_exact, best_exact_score = update_states(
            best_fft, best_fft_m, best_exact, best_exact_score, cand, cand_m, exact_limit
        )

    now = time.monotonic()
    if now < final_deadline - 2.0:
        local_base = best_exact if np.isfinite(best_exact_score) else best_fft
        w = max(3, n // 100)
        while time.monotonic() < final_deadline - 2.0:
            cand = local_base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - w + 1)))
                e = min(n, st + w)
                cand[st:e] *= np.exp(float(rng.normal(0.0, 0.02)))
            cand = norm_nonneg(cand, n)
            cm = score_fft(cand, n, nfft)
            if cm < best_fft_m - 1e-15:
                best_fft, best_fft_m, best_exact, best_exact_score = update_states(
                    best_fft, best_fft_m, best_exact, best_exact_score, cand, cm, exact_limit
                )
                local_base = cand

    out = best_exact if np.isfinite(best_exact_score) else best_fft
    out = norm_nonneg(out, n)
    out = np.clip(out, 0.0, 1000.0)

    s = float(out.sum())
    if not np.all(np.isfinite(out)) or s < 0.01:
        out = norm_nonneg(base, n)

    return [float(x) for x in out]
