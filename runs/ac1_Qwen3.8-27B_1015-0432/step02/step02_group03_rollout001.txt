def _autoconv(a, fs):
    """Autoconvolution via FFT."""
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _subgrad(a, peaks):
    """Subgradient of max(C) at given peak indices. O(n * len(peaks))."""
    n = a.size
    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        # g[j] += 2*a[k-j] if 0 <= k-j < n
        lo = max(0, k - (n - 1))  # j >= k - (n-1)
        hi = min(n - 1, k)        # j <= k
        if lo <= hi:
            # g[lo:hi+1] += 2*a[k-hi : k-lo+1][::-1]
            g[lo:hi+1] += 2.0 * a[k - hi : k - lo + 1][::-1]
    return g

            best = a.copy()

import numpy as np
import time

def _autoconv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _subgrad(a, peaks):
    n = a.size
    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g

def _descend(a, fs, dl, eta0, peak_frac, peak_cap, multi, mom_b, seed, stall_max):
    n = a.size
    rng = np.random.default_rng(seed)
    mom = np.zeros(n)
    best = a.copy()
    best_sc = 2.0 * n * float(_autoconv(best, fs).max()) / (best.sum() ** 2)
    p_sc = best_sc
    stall = 0
    eta = eta0
    while time.monotonic() < dl:
        C = _autoconv(a, fs)
        s = a.sum()
        sc = 2.0 * n * float(C.max()) / (s * s)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best = a.copy()
        stall = 0 if sc < p_sc - 1e-15 else stall + 1
        p_sc = sc
        if stall > stall_max:
            break
        mx = float(C.max())
        th = mx * (1.0 - peak_frac)
        pk = np.where(C >= th)[0]
        pk = pk[(pk >= 0) & (pk < 2 * n - 1)]
        if len(pk) > peak_cap:
            ii = np.argpartition(C[pk], -peak_cap)[-peak_cap:]
            pk = pk[ii]
        g = _subgrad(a, pk)
        g -= g.mean()
        if stall > stall_max // 2:
            ng = np.abs(g).max()
            if ng > 1e-14:
                g += 0.015 * rng.standard_normal(n) / ng
        if multi:
            gm = np.abs(g).max()
            if gm > 1e-14:
                a = a * np.exp(-eta * g / gm)
        else:
            mom = mom_b * mom + (1.0 - mom_b) * g
            a = np.maximum(a - eta * mom, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a *= n / s
        eta *= 0.99995
    return best, best_sc
