    a **= power
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def make_gauss(n, center, width):
    x = np.arange(n, dtype=np.float64)
    u = (x - center) / width
    a = np.exp(-0.5 * u * u)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()

    inits = [p, p[::-1].copy()]
    for pwr in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0]:
        inits.append(make_ramp(n, pwr, False))
        inits.append(make_ramp(n, pwr, True))
    for c, w, pw in [(0.2, 0.6, 0.5), (0.3, 0.5, 0.7), (0.4, 0.4, 0.5),
                     (0.5, 0.3, 1.0), (0.6, 0.4, 0.7), (0.7, 0.5, 0.5),
                     (0.8, 0.6, 0.5), (0.35, 0.45, 1.0), (0.65, 0.45, 1.0)]:
        inits.append(make_arc(n, c, w, pw))
    for c, w in [(0.3, 0.15), (0.4, 0.2), (0.5, 0.25), (0.5, 0.12),
                 (0.6, 0.18), (0.7, 0.22)]:
        inits.append(make_gauss(n, c * n, w * n))

    best_f = inits[0].copy()
    best_sc = np.inf
    for ini in inits:
        sc = score_norm(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    phases = [
        (10.0, 0.12, 40000),
        (25.0, 0.08, 40000),
        (50.0, 0.05, 40000),
        (100.0, 0.035, 40000),
        (200.0, 0.022, 60000),
        (400.0, 0.014, 60000),
        (800.0, 0.009, 60000),
        (1600.0, 0.006, 80000),
    ]

    for beta_s, eta_s, max_it in phases:
        if time.monotonic() > deadline - 40.0:
            break
        f_opt, sc_opt = optimize_phase(best_f, deadline - 25.0, rng, beta_s, eta_s, max_it)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()

        if time.monotonic() > deadline - 40.0:
            break

        # Mirror blends
        flip = best_f[::-1].copy()
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            bl = (1.0 - t) * best_f + t * flip
            bl = np.maximum(bl, 1e-20)
            bl /= bl.sum()
            sc_b = score_norm(bl)
            if sc_b < best_sc:
                best_sc = sc_b
                best_f = bl.copy()

        # Random perturbations
        for _ in range(10):
            if time.monotonic() > deadline - 40.0:
                break
            sigma = 0.015 * (1.0 + 0.5 * rng.random())
            trial = best_f * np.exp(sigma * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc_t = score_norm(trial)
            if sc_t < best_sc:
                best_sc = sc_t
                best_f = trial.copy()

    # Final fine polish with very high beta
    if time.monotonic() < deadline - 20.0:
        f_fin, sc_fin = optimize_phase(best_f, deadline - 5.0, rng, 3000.0, 0.004, 150000)
        if sc_fin < best_sc:
            best_sc = sc_fin
            best_f = f_fin.copy()

    # Second polish if time remains
    if time.monotonic() < deadline - 60.0:
        f_fin2, sc_fin2 = optimize_phase(best_f, deadline - 5.0, rng, 5000.0, 0.003, 100000)
        if sc_fin2 < best_sc:
            best_sc = sc_fin2
            best_f = f_fin2.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc) or final_sc >= 2.0:
        result = [float(max(0.0, min(1000.0, x))) for x in p]
        final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
