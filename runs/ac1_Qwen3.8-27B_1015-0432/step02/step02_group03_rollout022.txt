    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _xconv(u, v, fs):
    """Full convolution via FFT."""
    return np.fft.irfft(np.fft.rfft(u, n=fs) * np.fft.rfft(v, n=fs), n=fs)

def _sg(a, c, tau, fs):
    """Soft-max subgradient of max(c) w.r.t. a, O(n log n)."""
    n = a.size
    cm = c.max()
    if cm < 1e-30:
        return np.zeros(n)

    else:
        w /= ws
