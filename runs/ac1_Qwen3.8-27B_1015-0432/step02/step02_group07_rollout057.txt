    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def subgrad(a, conv, nk):
    n = len(a)
    idx = np.argpartition(conv, -min(nk, 2*n-1))[-min(nk, 2*n-1):]
    g = np.zeros(n)

        k = int(k)
        lo1 = max(0, k - n + 1)
        hi1 = min(k, n - 1)
