import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def make_ramp(n, alpha):
    j = np.arange(n, dtype=np.float64)
    w = np.power(j + 1.0, alpha)
    w = np.maximum(w, 1e-15)
    return w / w.sum()

def make_step(n, frac):
    w = np.zeros(n)
    w[int(n * (1.0 - frac)):] = 1.0
    return w / w.sum()

def make_bump(n, center, width):
    j = np.arange(n, dtype=np.float64)
    w = np.exp(-0.5 * np.square((j - center) / width))
    return w / w.sum()

def make_wind(n, k):
    w = np.ones(n) * 1e-10
    start = n // 2 - k // 2
    w[start:start + k] = 1.0
    return w / w.sum()
