            # d/d(a[i]) for i in [lo, hi]: 2*a[k-i]
            # For i in [k-hi, k-lo] (which is [k-lo, k-hi] rearranged): 2*a[k-i]
            # Actually both ranges are [lo, hi] since lo = max(0, k-n+1) and hi = min(n-1, k)
            # And k-i for i in [lo,hi] gives [k-hi, k-lo] = [k-min(n-1,k), k-max(0,k-n+1)]
            # If k < n: k-hi = k-k = 0, k-lo = k-0 = k. So k-i ranges over [0, k] = [0, hi]. OK.
            # If k >= n: k-hi = k-(n-1), k-lo = k-(k-n+1) = n-1. So k-i ranges over [k-n+1, n-1].
            # In both cases, k-i is in [0, n-1] for i in [lo, hi].
            # So the gradient contribution is: for i in [lo, hi], g[i] += 2*a[k-i]
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * a[k - hi:k - lo + 1][::-1]
    m = len(lags)
    if m > 0:
        g /= m
    return g
