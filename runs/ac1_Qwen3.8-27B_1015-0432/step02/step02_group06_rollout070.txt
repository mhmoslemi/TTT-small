        a_new = np.exp(log_a - amax)
        ss = float(a_new.sum())
        if (not np.isfinite(ss)) or ss < 1e-15:
            a = best_a.copy()
            log_a = np.log(a + 1e-300)
            log_a = log_a - log_a.max()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
        else:
            a = a_new / ss
            la = np.log(a)
            log_a = la - la.max()

        if it % 400 == 0:
            beta = min(5000.0, beta * 1.25)
            eta = max(0.008, eta * 0.97)

    out = [float(x) for x in best_a]
    final_sc = evaluate_sequence(out)
    if (not np.isfinite(final_sc)) or final_sc <= 0.0:
        fb = np.ones(n, dtype=np.float64) / float(n)
        out = [float(x) for x in fb]
    return out
