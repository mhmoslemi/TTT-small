def _fcc(a, L):
    fa = np.fft.rfft(a, n=L)
    return np.fft.irfft(fa * fa, n=L)

def _lsm_grad(C, a, L, tau):
    n = a.size
    Ck = C[:2 * n - 1]
    cm = float(Ck.max())
    e = np.exp(np.clip((Ck - cm) / tau, -50.0, 0.0))
    ws = float(e.sum())
    if ws < 1e-300:
        return np.zeros(n)
    w = e / ws
    Lc = 1 << (3 * n - 2).bit_length()
    fw = np.fft.rfft(w, n=Lc)
    fa = np.fft.rfft(a[::-1], n=Lc)
    cr = np.fft.irfft(fw * fa, n=Lc)
    return 2.0 * cr[n - 1:2 * n - 1]

def _desc_lsm(a, L, ts, dl, f0, f1, e0, e1, mx):
    n = a.size
