import numpy as np
import time

FLOOR = 1e-30

def fft_conv(a, b, fft_len=None):
    na = a.size; nb = b.size
    if fft_len is None:
       fft_len = 1 << (na + nb - 2).bit_length()
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    c = np.fft.irfft(A * B, fft_len)
    c = c[:na + nb - 1]
    # Clip tiny negatives. Maybe if negative large due numerical? use np.maximum.
    np.maximum(c, 0.0, out=c)
    return c

def fast_score(f, fft_len=None):
    s = f.sum()
    if not np.isfinite(s) or s <= 0:
        return np.inf
    b = fft_conv(f, f, fft_len)
    mb = b.max()
    if not np.isfinite(mb) or mb <= 0:
        return np.inf
    return float(2.0 * f.size * mb / (s * s))

def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, FLOOR)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s

import numpy as np
import time

FLOOR = 1e-30

def fft_conv(a, b, fft_len=None):
    na = len(a)
    nb = len(b)
    if fft_len is None:
        need = na + nb - 1
        fft_len = 1 << (need - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    c = np.fft.irfft(A * B, fft_len)
    c = c[:na + nb - 1]
    np.maximum(c, 0.0, out=c)
    return c

def fast_score(f, fft_len=None):
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    b = fft_conv(f, f, fft_len)
    mb = b.max()
    if not np.isfinite(mb) or mb <= 0.0:
        return np.inf
    return float(2.0 * f.size * mb / (s * s))

def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, FLOOR)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s

def make_inits(n, parent, rng):
    inits = []
    cands = []
    cands.append(parent)
    cands.append(np.ones(n, dtype=np.float64))
    if n > 1:
        x = np.arange(n, dtype=np.float64)
        scale = float(n)
        mid = 0.5 * (n - 1)
        cands.append(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1.0))))
        cands.append(0.5 * (1.0 + np.cos(np.pi * x / (n - 1.0))))
        cands.append(np.exp(-0.5 * ((x - mid) / (scale / 6.0)) ** 2))
        cands.append(np.exp(-0.5 * ((x - mid) / (scale / 9.0)) ** 2))
        cands.append(np.exp(-1.2 * x / scale) + 0.4 * np.exp(-1.2 * (n - 1 - x) / scale))
        cands.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x / scale))
        cands.append(0.5 + 0.4 * np.sin(2.0 * np.pi * x / scale))
        cands.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x / scale + np.pi / 3.0))
    for s in (0.03, 0.08, 0.18, 0.35):
        cands.append(np.exp(s * rng.standard_normal(n)))
    for m in (0.2, 0.45):
        cands.append((1.0 - m) * parent + m * parent[::-1])
    for a in cands:
        z = normalize_sequence(a)
        if z is not None:
            inits.append(z)
    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / float(max(n, 1)))
    return inits

    f = f.copy()
    n = f.size
    if sym_mode:
        f = 0.5 * (f + f[::-1])
    f = np.maximum(f, FLOOR)
    s = f.sum()
    if s <= 0.0:
        return f, np.inf
    f /= s
    best_f = f.copy()
    best_score = fast_score(f, fft_len)
    beta_cur = min(beta_first, beta_limit)
    eta_cur = min(eta_first, eta_limit) if eta_first >= eta_limit else eta_first
    # Actually ensure eta_cur between limits: if eta_first > eta_limit, start first then decay to limit; else start limit.
    if eta_first > eta_limit:
        eta_cur = eta_first
    else:
        eta_cur = eta_limit
    adapt_every = max(6, check_interval * 4)
    stall = 0
    it = 0
    while time.monotonic() < stop_time - 4.0:
        b = fft_conv(f, f, fft_len)
        M = b.max()
        if not np.isfinite(M) or M <= 0.0:
            break
        r = b / M - 1.0
        r = np.minimum(r, 0.0)
        w = np.exp(beta_cur * r)
        w = w + 1e-5
        sw = w.sum()
        if not np.isfinite(sw) or sw <= 0.0:
            break
        w = w / sw
        grad = fft_conv(w, f[::-1], fft_len)[n - 1:2 * n - 1]
        if sym_mode:
            grad = 0.5 * (grad + grad[::-1])
        grad = np.maximum(grad, 1e-300)
        gmax = grad.max()
        if not np.isfinite(gmax) or gmax <= 0.0:
            break
        step = eta_cur * (grad / gmax)
        f = f * np.exp(-step)
        if sym_mode:
            f = 0.5 * (f + f[::-1])
        f = np.maximum(f, FLOOR)
        s = f.sum()
        if not np.isfinite(s) or s <= 0.0:
            break
        f /= s
        it += 1
        if it % check_interval == 0:
            sc = fast_score(f, fft_len)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
            if stall >= max_stall:
                break
            if it % adapt_every == 0:
                beta_cur = min(beta_limit, beta_cur * 1.04)
                if eta_cur > eta_limit:
                    eta_cur = max(eta_limit, eta_cur * 0.985)
            if 70 < stall < max_stall and stall % 30 == 0:

                nz = normalize_sequence(f)
                if nz is None:
                    f = best_f.copy()
                else:
                    f = nz
                stall = 0
    return best_f, best_score

import numpy as np
import time

np.seterr(over="ignore", under="ignore", invalid="ignore", divide="ignore")

FLOOR = 1e-30


def fft_conv(a, b, fft_len=None):
    na = len(a)
    nb = len(b)
    if fft_len is None:
        need = na + nb - 1
        fft_len = 1 << (need - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    B = np.fft.rfft(b, fft_len)
    c = np.fft.irfft(A * B, fft_len)
    c = c[:na + nb - 1]
    np.maximum(c, 0.0, out=c)
    return c


def fast_score(f, fft_len=None):
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    b = fft_conv(f, f, fft_len)
    mb = b.max()
    if not np.isfinite(mb) or mb <= 0.0:
        return np.inf
    return float(2.0 * f.size * mb / (s * s))


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, FLOOR)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        return None
    return a / s


def make_inits(n, parent, rng):
    inits = []
    cands = []
    cands.append(parent)
    cands.append(np.ones(n, dtype=np.float64))

    if n > 1:
        x = np.arange(n, dtype=np.float64)
        scale = float(n)
        mid = 0.5 * (n - 1)
        cands.append(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1.0))))
        cands.append(0.5 * (1.0 + np.cos(np.pi * x / (n - 1.0))))
        cands.append(np.exp(-0.5 * ((x - mid) / (scale / 6.0)) ** 2))
        cands.append(np.exp(-0.5 * ((x - mid) / (scale / 9.0)) ** 2))
        cands.append(np.exp(-1.2 * x / scale) + 0.4 * np.exp(-1.2 * (n - 1 - x) / scale))
        cands.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x / scale))
        cands.append(0.5 + 0.4 * np.sin(2.0 * np.pi * x / scale))
        cands.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x / scale + np.pi / 3.0))

    for s in (0.03, 0.08, 0.18, 0.35):
        cands.append(np.exp(s * rng.standard_normal(n)))

    for m in (0.2, 0.45):
        cands.append((1.0 - m) * parent + m * parent[::-1])

    for a in cands:
        z = normalize_sequence(a)
        if z is not None:
            inits.append(z)

    if not inits:
        inits.append(np.ones(n, dtype=np.float64) / float(max(n, 1)))
    return inits


def optimize_sequence(
    f,
    stop_time,
    rng,
    beta_first,
    beta_limit,
    eta_first,
    eta_limit,
    check_interval,
    max_stall,
    sym_mode,
    fft_len,
):
    f = f.copy()
    n = f.size

    if sym_mode:
        f = 0.5 * (f + f[::-1])

    f = np.maximum(f, FLOOR)
    s = f.sum()
    if not np.isfinite(s) or s <= 0.0:
        return f, np.inf
    f /= s

    best_f = f.copy()
    best_score = fast_score(f, fft_len)

    beta_cur = min(beta_first, beta_limit)
    if eta_first > eta_limit:
        eta_cur = eta_first
    else:
        eta_cur = eta_limit

    adapt_every = max(6, check_interval * 4)
    stall = 0
    it = 0

    while time.monotonic() < stop_time - 4.0:
        b = fft_conv(f, f, fft_len)
        M = b.max()
        if not np.isfinite(M) or M <= 0.0:
            break

        r = b / M - 1.0
        r = np.minimum(r, 0.0)
        w = np.exp(beta_cur * r)
        w = w + 1e-6
        sw = w.sum()
        if not np.isfinite(sw) or sw <= 0.0:
            break
        w = w / sw

        grad = fft_conv(w, f[::-1], fft_len)[n - 1:2 * n - 1]
        if sym_mode:
            grad = 0.5 * (grad + grad[::-1])

        grad = np.maximum(grad, 1e-300)
        gmax = grad.max()
        if not np.isfinite(gmax) or gmax <= 0.0:
            break

        step = eta_cur * (grad / gmax)
        f = f * np.exp(-step)

        if sym_mode:
            f = 0.5 * (f + f[::-1])

        f = np.maximum(f, FLOOR)
        s = f.sum()
        if not np.isfinite(s) or s <= 0.0:
            break
        f /= s

        it += 1
        if it % check_interval == 0:
            sc = fast_score(f, fft_len)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_f = f.copy()
                stall = 0
            else:
                stall += 1

            if stall >= max_stall:
                break

            if it % adapt_every == 0:
                beta_cur = min(beta_limit, beta_cur * 1.04)
                if eta_cur > eta_limit:
                    eta_cur = max(eta_limit, eta_cur * 0.985)

            if 70 < stall < max_stall and stall % 30 == 0:
                scale = 0.02 + 0.05 * (float(stall) / float(max_stall))
                f = best_f * np.exp(scale * rng.standard_normal(n))
                if sym_mode:
                    f = 0.5 * (f + f[::-1])
                nz = normalize_sequence(f)
                if nz is None:
                    f = best_f.copy()
                else:
                    f = nz
                stall = 0

    return best_f, best_score

                if nz is not None:
                    sc = fast_score(nz, fft_len)

    candidates = []
    nz = normalize_sequence(best_f)

    trial = 0.5 * (best_f + best_f[::-1])
