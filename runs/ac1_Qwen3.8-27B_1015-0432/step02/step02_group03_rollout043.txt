import numpy as np

def _conv_fft(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def _grad_lags(a, lags, weights):
    n = a.size
    g = np.zeros(n)
    wsum = weights.sum()
    if wsum < 1e-30:
        return g
    for j in range(len(lags)):
        k = int(lags[j])
        w = weights[j] / wsum
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    return g


def _score_fft(a, fs):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _conv_fft(a, fs)
    return 2.0 * n * float(c.max()) / (s * s)
