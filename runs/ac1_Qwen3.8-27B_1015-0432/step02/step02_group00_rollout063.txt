import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def normalize(f):
    s = f.sum()
    if s < 1e-30:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def score_of(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def build_A(f, peaks, n):
    K = len(peaks)
    rv, cv, vv = [], [], []
    for i, k in enumerate(peaks):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        js = np.arange(lo, hi + 1)
        rv.append(np.full(js.size, i, np.int32))
        cv.append(js.astype(np.int32))
        vv.append(f[k - js])
    if not vv:
        return csr_matrix((K, n))
    return csr_matrix((np.concatenate(vv), (np.concatenate(rv), np.concatenate(cv))), shape=(K, n))

def lp_dir(f, peaks, M):
    n = len(f)
    A = build_A(f, peaks, n)
    if A.nnz < n:
        return None
    try:
        r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(A.shape[0], M),
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': 2.5, 'presolve': True})
        if r.status == 0:
            g = np.maximum(r.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

def subgrad(f, peaks):
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return np.maximum(2.0 * c[n - 1:2 * n - 1][::-1], 0.0)
