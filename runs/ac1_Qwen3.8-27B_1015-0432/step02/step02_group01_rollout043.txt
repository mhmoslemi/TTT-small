import numpy as np

def fft_conv(a, b):
    n1, n2 = len(a), len(b)
    size = 1 << (n1 + n2 - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n1 + n2 - 1]

def score_norm(f):
    return 2.0 * f.size * fft_conv(f, f).max()

def smoothed_grad(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(-beta * (1.0 - b / M))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        top_k = np.argsort(b)[-max(1, n // 100):]
        w[top_k] = 1.0
    g = fft_conv(w, f[::-1])
    return np.maximum(g[n - 1:2 * n - 1], 0.0)

def exact_grad(f, b):
    n = f.size
    k = int(np.argmax(b))
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    sg = np.zeros(n)
    sg[lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return sg

def run_descent(f, deadline, rng, beta0, eta0, max_iter, use_exact_frac=0.3):

    f = f.astype(np.float64).copy()
    f /= f.sum()

        b = fft_conv(f, f)

            stall = 0

            t = rng.uniform(0.1, 0.5)
            f = (1.0 - t) * best_f + t * flip
            f = np.maximum(f, 1e-20)

            beta = beta0

def fft_conv(a, b):
    n1, n2 = len(a), len(b)

    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n1 + n2 - 1]

    return 2.0 * f.size * fft_conv(f, f).max()

    n = f.size
    M = b.max()
    if M < 1e-30:
        return None

import numpy as np
import time


def fft_conv(a, b):
    n1, n2 = len(a), len(b)
    size = 1 << (n1 + n2 - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[: n1 + n2 - 1]


def sc(f):
    return 2.0 * f.size * fft_conv(f, f).max()


def sg_smooth(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(-beta * (1.0 - b / M))
    s = w.sum()
    if s < 1e-30:
        w = (b > 0.95 * M).astype(np.float64)
        s = max(w.sum(), 1.0)
    g = fft_conv(w, f[::-1])[n - 1: 2 * n - 1].copy()
    return np.maximum(g, 0.0) / s


def sg_exact(f, b):
    n = f.size
    k = int(np.argmax(b))
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    sg = np.zeros(n)
    sg[lo: hi + 1] = f[k - hi: k - lo + 1][::-1]
    return sg


def make_win(n, kind):
    u = np.linspace(0.0, 1.0, n)
    if kind == 0:
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * u))
    elif kind == 1:
        w = np.sin(np.pi * u)
    elif kind == 2:
        w = 1.0 - np.abs(2.0 * u - 1.0)
    else:
        w = np.exp(-5.0 * (u - 0.5) ** 2)
    w = np.maximum(w, 1e-10)
    return w / w.sum()

            stall = 0

            eta = e0
