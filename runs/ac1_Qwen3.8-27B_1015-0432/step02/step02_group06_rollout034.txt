        ini = np.maximum(ini, 1e-25)
        ini /= ini.sum()
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = ini.copy()

    if best_f is None:
        best_f = p.copy()
        best_score = fast_score(p)

    f = best_f.copy()
    time_avail = deadline - t0

    ph1_dl = t0 + 0.40 * time_avail
    f_opt, sc = gd_optimize(f, ph1_dl, rng, 20.0, 0.10, eval_interval=8)
    if sc < best_score:
        best_score = sc
        best_f = f_opt.copy()
    f = best_f.copy()

    f = f * np.exp(0.02 * rng.standard_normal(n))
    f = np.maximum(f, 1e-25)
    f /= f.sum()

    ph2_dl = min(t0 + 0.75 * time_avail, deadline - 80.0)
    f_opt, sc = gd_optimize(f, ph2_dl, rng, 45.0, 0.05, eval_interval=6)
    if sc < best_score:
        best_score = sc
        best_f = f_opt.copy()
    f = best_f.copy()

    f = f * np.exp(0.01 * rng.standard_normal(n))
    f = np.maximum(f, 1e-25)
    f /= f.sum()

    ph3_dl = deadline - 10.0
    f_opt, sc = gd_optimize(f, ph3_dl, rng, 80.0, 0.025, eval_interval=5)
    if sc < best_score:
        best_score = sc
        best_f = f_opt.copy()

    if time.monotonic() < deadline - 8.0:
        for _ in range(200):
            if time.monotonic() > deadline - 8.0:
                break
            for scale in [0.001, 0.003, 0.008]:
                trial = best_f * np.exp(scale * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-25)
                trial /= trial.sum()
                sc = fast_score(trial)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_f = trial.copy()

        sym = 0.5 * (best_f + best_f[::-1])
        sym = np.maximum(sym, 1e-25)
        sym /= sym.sum()
        sc = fast_score(sym)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = sym.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
        fs = evaluate_sequence(result)
    return result
