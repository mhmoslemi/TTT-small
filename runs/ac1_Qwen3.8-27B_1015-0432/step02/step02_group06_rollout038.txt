    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def make_inits(n, rng):
    inits = [init_parent(n), np.ones(n) / n]
    x = np.arange(n, dtype=np.float64) / n
    for c in [0.15, 0.3, 0.45, 0.55, 0.7, 0.85]:
        a = np.exp(-((x - c) ** 2) / 0.06)
        inits.append(a / a.sum())
    for r in [0.8, 1.5, 3.0, 5.0]:
        a = np.exp(-r * x)
        inits.append(a / a.sum())
        a = np.exp(-r * (1.0 - x))
        inits.append(a / a.sum())
    for p in [0.3, 1.5, 3.0]:
        a = np.maximum(x, 1e-8) ** p
        inits.append(a / a.sum())
    a = np.exp(-((x - 0.2) ** 2) / 0.03) + np.exp(-((x - 0.8) ** 2) / 0.03)
    inits.append(a / a.sum())
    for s in [0.12, 0.25, 0.4]:
        t = init_parent(n) * np.exp(s * rng.standard_normal(n))
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())
    return inits


def gd_optimize(f, deadline, rng, beta, eta, topk_frac=1.0):
    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    stall = 0
    it = 0
    topk = max(1, int(2 * n - 1)) if topk_frac >= 1.0 else max(1, int((2 * n - 1) * topk_frac))

    while time.monotonic() < deadline - 2.0:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        ratio = b / M
        w = np.exp(beta * (ratio - 1.0))
        if topk < 2 * n - 1:
            idx = np.argpartition(b, -topk)[-topk:]
            w[:topk] = 0.0
            w[:] = 0.0
            w[idx] = np.exp(beta * (b[idx] / M - 1.0))
        gc = fft_conv(w, f[::-1])
        grad = gc[n - 1:2 * n - 1]
        grad = np.maximum(grad, 1e-30)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        it += 1

        if it % 10 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-11:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if it % 100 == 0:
                wp = w / (w.sum() + 1e-30)
                H = -np.sum(wp * np.log(np.maximum(wp, 1e-30)))
                Hn = H / np.log(len(w))
                if Hn > 0.4:
                    beta = min(600.0, beta * 1.12)
                elif Hn < 0.01:
                    beta = max(5.0, beta * 0.85)
                eta = max(0.002, eta * 0.996)
            if 60 <= stall < 250 and stall % 60 == 0:
                f = best * np.exp(0.012 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
            if stall >= 250:
                break
    return best, best_sc


def peak_shave(f, deadline, rng, iters=200):
    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    for _ in range(iters):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        k = int(np.argmax(b))
        M = b[k]
        contrib = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] = 2.0 * f[j]
        contrib = np.maximum(contrib, 1e-30)
        cmax = contrib.max()
        f = f * np.exp(-0.05 * contrib / cmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        sc = fast_score(f)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = f.copy()
    return best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    inits = make_inits(n, rng)
    best_f = None
    best_score = np.inf
    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = ini.copy()
    if best_f is None:
        best_f = init_parent(n)
        best_score = fast_score(best_f)

    total = deadline - t0
    phases = [
        (0.22, 15.0, 0.2, 1.0),
        (0.22, 40.0, 0.12, 0.7),
        (0.22, 90.0, 0.06, 0.4),
        (0.22, 200.0, 0.03, 0.2),
    ]
    elapsed = 0.0
    for frac, b0, e0, tkf in phases:
        pdl = t0 + elapsed + frac * total
        elapsed += frac * total
        if time.monotonic() > pdl - 5.0:
            break
        f_opt, sc_opt = gd_optimize(best_f, pdl, rng, b0, e0, topk_frac=tkf)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

    if time.monotonic() < deadline - 40.0:
        f_opt, sc_opt = gd_optimize(best_f, deadline - 40.0, rng, 350.0, 0.012, 0.1)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

    if time.monotonic() < deadline - 20.0:
        f_opt, sc_opt = peak_shave(best_f, deadline - 20.0, rng, iters=300)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

    if time.monotonic() < deadline - 10.0:
        for _ in range(150):
            if time.monotonic() > deadline - 10.0:
                break
            trial = best_f * np.exp(0.002 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-25)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    if time.monotonic() < deadline - 5.0:
        for mix in [0.05, 0.1, 0.15]:
            trial = (1.0 - mix) * best_f + mix * best_f[::-1]
            trial = np.maximum(trial, 1e-25)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    final = [float(max(0.0, min(1000.0, x))) for x in best_f]
    fs = evaluate_sequence(final)
    if not np.isfinite(fs):
        final = [1.0 / n] * n
        fs = evaluate_sequence(final)
    return final
