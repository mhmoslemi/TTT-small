def _fcc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, c, frac, cap):
    n = a.size
    mx = float(np.max(c[:2 * n - 1]))
    act = np.where(c[:2 * n - 1] >= mx * (1.0 - frac))[0]
    if act.size > cap:
        idx = np.argpartition(c[:2 * n - 1][act], -cap)[-cap:]
        act = act[idx]
    g = np.zeros(n)
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    m = act.size
    if m:
        g *= 2.0 / m
    return g
