    t2 = t0 + 0.70 * eff
    a2 = best / s0
    a2 *= np.exp(rng.normal(0.0, 0.008, n))
    a2 = np.clip(a2, 1e-15, None)
    a2 /= a2.sum()
    _, lb2, ls2 = run_mult(a2, n, fs, s0, t2)
    if np.isfinite(ls2) and ls2 < best_score:
        best_score, best = ls2, lb2

    # Phase 3: Fine additive polish (15%)
    t3 = t0 + 0.88 * eff
    a3 = best / s0
    _, lb3, ls3 = run_add(a3, n, fs, s0, t3, 0.92, 0.12, 16)
    if np.isfinite(ls3) and ls3 < best_score:
        best_score, best = ls3, lb3

    # Phase 4: Local perturbations
    base = best.copy()
    while time.monotonic() < deadline - 4.0:
        cand = base.copy()
        w = max(3, n // 35)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= np.exp(float(rng.normal(0.0, 0.012)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best, base = sc, cand, cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = np.ones(1024, dtype=np.float64)
    return [float(x) for x in out]
