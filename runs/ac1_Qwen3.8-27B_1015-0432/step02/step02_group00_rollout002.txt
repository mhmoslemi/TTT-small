            best_sc = sc_s
            best_f = f_s.copy()
        phase2_end = min(deadline, t0 + 850.0)
        f_so, sc_so = optimize(f_s, phase2_end, rng, sym=True)
        if sc_so < best_sc:
            best_sc = sc_so
            best_f = f_so.copy()

    if time.monotonic() < deadline - 40.0:
        f_ref, sc_ref = optimize(best_f.copy(), deadline - 12.0, rng, sym=False)
        if sc_ref < best_sc:
            best_sc = sc_ref
            best_f = f_ref.copy()

    if time.monotonic() < deadline - 5.0:
        for _ in range(60):
            if time.monotonic() > deadline - 2.0:
                break
            trial = best_f * np.exp(0.004 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [float(1.0 / n)] * n
    return result
