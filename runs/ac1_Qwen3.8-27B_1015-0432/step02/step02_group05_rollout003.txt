def _optimize(a, n, fsz, deadline, k_hi, k_lo, eta0, decay, mom):
    a = np.maximum(a.copy(), 1e-15)
    a /= a.sum()
    best_a = a.copy()
    best_m = float(_fft_conv(a, fsz)[:2 * n - 1].max())
    g_m = np.zeros(n)
    stall = 0
    prev_m = best_m
    it = 0
    total = 2 * n - 1
    while time.monotonic() < deadline:
        cv = _fft_conv(a, fsz)[:total]
        m = float(cv.max())
        if m < prev_m - 1e-15:
            stall = 0
            prev_m = m
        else:
            stall += 1
        if m < best_m:
            best_m = m
            best_a = a.copy()
        if stall > 12000:
            break
        frac = min(1.0, it / 30000.0)
        k = max(2, int(k_hi - (k_hi - k_lo) * frac))
        g = _grad_topk(a, cv, k, n)
        g_m = mom * g_m + (1.0 - mom) * g
        eta = eta0 * (decay ** it)
        a = np.exp(np.log(np.maximum(a, 1e-20)) - eta * g_m)
        a = np.maximum(a, 1e-20)
        a /= a.sum()
        it += 1
    return best_a, best_m

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 12.0)

    par = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if not (par.ndim == 1 and par.size >= 2 and np.isfinite(par).all() and par.sum() >= 0.01):
        par = np.ones(512, dtype=np.float64)
    n = par.size
    S = float(par.sum())
    total = 2 * n - 1
    fsz = 1 << total.bit_length()

    best_arr = par.copy()
    best_sc = float(evaluate_sequence(best_arr.tolist()))

    inits = []
    inits.append(par / S)

    inits.append(np.ones(n, dtype=np.float64) / n)

    r = np.abs(rng.standard_normal(n)) ** 0.35
    r = (r + np.roll(r, 1) + np.roll(r, -1)) / 3.0
    inits.append(np.maximum(r, 1e-4) / np.maximum(r, 1e-4).sum())

    f1 = rng.uniform(0.02, 0.4)
    p1 = rng.uniform(0.0, 6.28)
    r2 = 0.4 + 0.6 * np.abs(np.sin(2.0 * np.pi * f1 * np.arange(n) + p1))
    r2 = np.maximum(r2, 0.01)
    inits.append(r2 / r2.sum())

    c1 = int(n * 0.25)
    c2 = int(n * 0.75)
    sig = max(5, int(n * 0.15))
    x = np.arange(n, dtype=np.float64)
    r3 = np.maximum(
        np.exp(-((x - c1) ** 2) / (2.0 * sig * sig)),
        np.exp(-((x - c2) ** 2) / (2.0 * sig * sig))
    ) + 0.03
    inits.append(r3 / r3.sum())

    r4 = np.abs(rng.standard_normal(n))
    for _ in range(5):
        r4 = (r4 + np.roll(r4, 1)) * 0.5
    inits.append(r4 / r4.sum())

    weights = [0.30, 0.10, 0.15, 0.15, 0.15, 0.15]
    k_his = [max(8, total // 80), 64, max(8, total // 100), max(8, total // 100), max(8, total // 90), max(8, total // 80)]
    k_lows = [max(2, total // 400), 4, max(2, total // 400), max(2, total // 400), max(2, total // 400), max(2, total // 400)]
    etas = [n * 0.035, n * 0.05, n * 0.04, n * 0.04, n * 0.045, n * 0.04]
    decays = [0.99994, 0.99996, 0.99995, 0.99995, 0.99994, 0.99995]
    moms = [0.82, 0.75, 0.80, 0.80, 0.78, 0.80]

    for ii in range(len(inits)):
        rem = deadline - time.monotonic()
        if rem < 25.0:
            break
        alloc = rem * weights[ii]
        t_end = time.monotonic() + alloc
        a_opt = _optimize(inits[ii], n, fsz, t_end, k_his[ii], k_lows[ii],
                          etas[ii], decays[ii], moms[ii])
        cand = np.clip(a_opt * S, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_arr = cand.copy()

    # Second pass: re-optimize from best with tighter params
    if time.monotonic() < deadline - 80.0 and best_sc < 2.0:
        rem2 = deadline - time.monotonic()
        a_start = best_arr / best_arr.sum()
        a_opt2 = _optimize(a_start, n, fsz, time.monotonic() + rem2 * 0.6,
                           max(2, total // 200), max(2, total // 600),
                           n * 0.02, 0.99998, 0.88)
        cand2 = np.clip(a_opt2 * S, 0.0, 1000.0)
        if cand2.sum() >= 0.01:
            sc2 = float(evaluate_sequence(cand2.tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best_arr = cand2.copy()

    # Local perturbation polish
    if time.monotonic() < deadline - 4.0:
        base = best_arr.copy()
        w = max(4, n // 250)
        while time.monotonic() < deadline - 2.0:
            c = base.copy()
            npert = int(rng.integers(1, 4))
            for _ in range(npert):
                st = int(rng.integers(0, max(1, n - w)))
                c[st:min(n, st + w)] *= float(np.exp(rng.normal(0.0, 0.012)))
            c = np.clip(c, 0.0, 1000.0)
            if c.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(c.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                base = c

    out = np.clip(best_arr, 0.0, 1000.0)
    if not (out.size >= 1 and np.isfinite(out).all() and out.sum() >= 0.01):
        out = par.copy()
    return [float(x) for x in out]
