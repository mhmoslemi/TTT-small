```python
def compute_grad(a, N, t):
    n = len(a)
    # Convolution
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:2*n-1]
    mx = conv.max()
    # Softmax weights
    w = np.exp(t * (conv - mx))
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2*n-1, 1.0/(2*n-1))
    else:
        w /= ws
    # Gradient
    w_rev = np.zeros(N)
    w_rev[:2*n-1] = w[::-1]
    w_fwd = np.zeros(N)
    w_fwd[:2*n-1] = w
    a_pad = np.zeros(N)
    a_pad[:n] = a
    f_a = np.fft.rfft(a_pad)
    f_wr = np.fft.rfft(w_rev)
    f_wf = np.fft.rfft(w_fwd)
    part1 = np.fft.irfft(f_wr * f_a, n=N)[2*n-2 : 2*n-2+n]
    part2 = np.fft.irfft(f_wf * f_a, n=N)[:n]
    return part1 + part2, conv, mx
```
