import time

def fft_convolve(a, b, fft_size):
    """Full convolution of a and b via FFT."""
    fa = np.fft.rfft(a, n=fft_size)
    fb = np.fft.rfft(b, n=fft_size)
    return np.fft.irfft(fa * fb, n=fft_size)

def compute_grad_softmax(a, conv, T, fft_size):
    """Compute gradient of T*logsumexp(conv/T) w.r.t. a using FFT."""
    n = a.size
    # Safe logsumexp
    c_max = np.max(conv)
    shifted = (conv - c_max) / T
    exp_vals = np.exp(shifted)
    total = exp_vals.sum()
    if total < 1e-30:
        return np.zeros(n)
    w = exp_vals / total  # soft-max weights, length 2n-1

def full_conv(a, b, fs):
    fa = np.fft.rfft(a, n=fs)
    fb = np.fft.rfft(b, n=fs)
    return np.fft.irfft(fa * fb, n=fs)


def softmax_grad(a, conv, T, fs, n):
    n1 = 2 * n - 1
    cmax = float(np.max(conv[:n1]))
    w = np.exp((conv[:n1] - cmax) / T)
    sw = float(w.sum())
    if sw < 1e-30:
        return np.zeros(n)
    w /= sw
    wr = w[::-1]
    c1 = full_conv(w, a, fs)[:n]
    c2 = full_conv(a, wr, fs)[n - 1:2 * n - 1][::-1]
    return 2.0 * (c1 + c2)
