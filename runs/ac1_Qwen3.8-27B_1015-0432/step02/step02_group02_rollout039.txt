    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 20.0)
    
    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256)
    n = par.size
    if par.sum() < 0.01:
        par = np.ones(n)
    
    scale = np.sqrt(2.0 * n)
    a0 = par / par.sum() * scale
    
    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    
    x = np.arange(n, dtype=np.float64)
    inits = [a0]
    for k, amp in [(1, 0.4), (1, 0.7), (2, 0.4), (3, 0.3), (5, 0.2)]:
        c = 1.0 + amp * np.cos(2.0 * np.pi * k * x / n)
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            inits.append(c / c.sum() * scale)
    for sig, ctr in [(0.2, 0.5), (0.35, 0.3), (0.35, 0.7), (0.12, 0.5)]:
        g = np.exp(-0.5 * ((x - ctr * n) / (sig * n)) ** 2)
        if g.sum() > 0.01:
            inits.append(g / g.sum() * scale)
    bx = np.zeros(n)
    bx[n//5:4*n//5] = 1.0
    inits.append(bx / bx.sum() * scale)
    
    cur = a0.copy()
    for inp in inits:
        s2 = float(evaluate_sequence(inp.tolist()))
        if np.isfinite(s2) and s2 < best_sc:
            best = inp.copy()
            best_sc = s2
            cur = inp.copy()
    
    # Main optimization with decreasing tau
    bl = [(0.0, 1000.0)] * n
    taus = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
    
    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 150.0:
            break
        iters = 800
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': iters, 'gtol': 1e-16,
                                    'maxls': 50, 'ftol': 1e-18})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                nx = nx / nx.sum() * scale
            s2 = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(s2) and s2 < best_sc:
                best = nx.copy()
                best_sc = s2
            cur = nx.copy()
        except Exception:
            pass
    
    # Sparsification
    for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
        if time.monotonic() > dl - 80.0:
            break
        a = best.copy()
        thresh = np.sort(a)[max(1, int(frac * n))]
        a[a < thresh] = 0.0
        if a.sum() > 0.01:
            a = a / a.sum() * scale
            s2 = float(evaluate_sequence(a.tolist()))
            if np.isfinite(s2) and s2 < best_sc:
                best = a.copy()
                best_sc = s2

    if time.monotonic() < dl - 60.0:
        cur = best.copy()
        for tv in [0.01, 0.005, 0.002, 0.001, 5e-4]:
            if time.monotonic() > dl - 30.0:
                break

                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 500, 'gtol': 1e-16,
                                        'maxls': 50, 'ftol': 1e-18})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * scale
                s2 = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(s2) and s2 < best_sc:
                    best = nx
