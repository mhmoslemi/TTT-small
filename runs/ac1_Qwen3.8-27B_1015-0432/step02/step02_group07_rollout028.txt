For index i: g[i] += w[k] * a[k-i]. ✓

And the return is 2.0 * g / ws, which gives the weighted average subgradient scaled by 2. ✓

def fcov(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def subgrad(a, c, n, topk, soft_a):
    M = float(c[:2 * n - 1].max())
    cc = c[:2 * n - 1]
    if soft_a > 0.0:
        sp = max(M - float(cc.min()), 1e-18)
        w = np.exp(-soft_a * (M - cc) / sp)
        w[cc < M - 3.0 * sp] = 0.0
    else:
        idx = np.argpartition(cc, -topk)[-topk:]
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[idx] = 1.0
    ws = w.sum()
    if ws < 1e-20:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(cc))] = 1.0
        ws = 1.0
    act = np.where(w > 1e-12)[0]
    g = np.zeros(n, dtype=np.float64)
    for k in act:
        kk = int(k)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += w[kk] * a[hi - kk + lo:hi + 1][::-1]
    return 2.0 * g / ws
