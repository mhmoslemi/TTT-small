def fft_conv(a, size):
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)

    A = np.fft.rfft(a, n=N)
    W = np.fft.rfft(w, n=N)
    return 2.0 * np.fft.irfft(np.conj(A) * W, n=N)[:a.size]
