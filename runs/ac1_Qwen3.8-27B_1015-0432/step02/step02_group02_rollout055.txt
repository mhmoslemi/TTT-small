def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

    cm = c.max()
    d = np.maximum((c - cm) / tau, -80.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    n = a.size
    S = a.sum()
    if S < 1e-8:

    c = _conv(a)
    m, _ = _smax(c, tau)
    return 2.0 * n * m / (S * S)
