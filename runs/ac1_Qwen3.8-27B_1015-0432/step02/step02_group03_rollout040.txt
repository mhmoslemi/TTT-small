def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, lags, n):
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    m = len(lags)
    if m > 0:
        g /= float(m)
    return g

def _obj(a, fs, n):
    c = _fc(a, fs)
    s = a.sum()
    return 2.0 * n * float(c[:2*n - 1].max()) / (s * s)

def _desc(a, fs, n, ts, eta0, beta, dl, frac, cap, smx, multi, rng, edec):
    a = a.copy().astype(np.float64)
    mom = np.zeros(n)
    bst = a.copy()
    bsc = _obj(bst, fs, n)
    stall = 0
    psc = bsc
    it = 0
    while it < 1000000 and time.monotonic() < dl:
        it += 1
        c = _fc(a, fs)
        s = a.sum()
        sc = 2.0 * n * float(c[:2*n - 1].max()) / (s * s)
        if sc < bsc - 1e-13:
            bsc = sc
            bst = a.copy()
        stall = 0 if sc < psc - 1e-14 else stall + 1
        psc = sc
        if stall > smx:
            break
        mx = float(c[:2*n - 1].max())
        f = frac * (1.0 + 0.5 * min(3.0 * stall / smx, 1.0))
        act = np.where(c[:2*n - 1] >= mx * (1.0 - f))[0]

            act = act[ii]
        if len(act) == 0:

        g = _sg(a, act, n)
        g -= g.mean()
        if stall > smx // 3:

        sc = float(evaluate_sequence(cand.tolist()))
