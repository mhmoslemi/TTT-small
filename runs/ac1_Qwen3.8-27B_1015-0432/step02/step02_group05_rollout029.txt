        a, _ = run_opt(inits[k].copy(), n, fsz, time.monotonic() + per, T_s, T_d, e0, e_d)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < deadline - 60.0:
        rem = deadline - time.monotonic()
        per = min(rem * 0.6, 200.0)
        a, _ = run_opt(best.copy(), n, fsz, time.monotonic() + per, 0.3, 0.9998, 1.0, 0.99995)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < deadline - 5.0:
        base = (best * s0).copy()
        while time.monotonic() < deadline - 3.0:
            c = base.copy()
            wp = max(1, n // 300)
            np_ = int(rng.integers(1, 5))
            for _ in range(np_):
                st = int(rng.integers(0, max(1, n - wp)))
                c[st:min(n, st + wp)] *= float(np.exp(rng.normal(0.0, 0.008)))
            c = np.clip(c, 0.0, 1000.0)
            if c.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(c.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                base = c

    out = np.clip(best * s0, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(v) for v in out]
