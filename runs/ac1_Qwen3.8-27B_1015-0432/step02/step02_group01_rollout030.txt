```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

This is O(n log n) due to sorting.

```python
            trial = f * np.exp(-te * g)
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            tb = fft_conv(trial, trial)
            tsc = 2.0 * n * tb.max()
            if tsc < best_trial_sc:
                best_trial_sc = tsc
                best_trial = trial
        f = best_trial
        # Schedule
        if it % 300 == 0 and it > 0:
            beta *= 1.025
            eta = max(0.002, eta * 0.992)
    return best_f, best_sc


def make_init(n, kind, par, rng):
    x = np.arange(n, dtype=np.float64)
    if kind == 'parent':
        return par.copy()
    if kind == 'parent_rev':
        return par[::-1].copy()
    if kind == 'pow':
        p = (x + 1.0) ** par
        if rng.random() < 0.5:
            p = p[::-1]
        return p / p.sum()
    if kind == 'cos':
        p = (0.5 + 0.5 * np.cos(np.pi * x / (n - 1))) ** par
        p = np.maximum(p, 1e-15)
        return p / p.sum()
    if kind == 'bump':
        c, w = par
        p = np.exp(-((x - c * (n - 1)) ** 2) / (2.0 * (w * n) ** 2))
        p = np.maximum(p, 1e-15)
        return p / p.sum()
    if kind == 'sym':
        h = (n + 1) // 2
        v = np.arange(h, dtype=np.float64)
        p = np.sin(np.pi * (v + 1) / (h + 1)) ** par
        p = np.maximum(p, 1e-15)
        f = np.zeros(n)
        f[:h] = p
        f[h:] = p[-1:0:-1][:n - h]
        return f / f.sum()
    if kind == 'smooth_rand':
        r = rng.standard_normal(n)
        fc = np.fft.rfft(r)
        cutoff = max(2, len(fc) // 6)
        fc[cutoff:] *= 0.01
        r = np.fft.irfft(fc, n)
        r = np.maximum(r, 0.01)
        return r / r.sum()
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()

    inits = [make_init(n, 'parent', p, rng), make_init(n, 'parent_rev', p, rng)]
    for pw in (0.3, 0.5, 0.8, 1.2, 2.0, 3.0):
        inits.append(make_init(n, 'pow', pw, rng))
    for pw in (0.5, 1.0, 1.5, 2.0, 3.0):
        inits.append(make_init(n, 'cos', pw, rng))
    for c, w in ((0.3, 0.4), (0.5, 0.3), (0.7, 0.4), (0.4, 0.55), (0.6, 0.5)):
        inits.append(make_init(n, 'bump', (c, w), rng))
    for pw in (0.7, 1.0, 1.5, 2.5):
        inits.append(make_init(n, 'sym', pw, rng))
    for _ in range(5):
        inits.append(make_init(n, 'smooth_rand', None, rng))

    best_f = inits[0].copy()
    best_sc = np.inf
    for ini in inits:
        sc = score_of(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    # Multi-phase optimization
    n_phases = 5
    frac = np.array([0.32, 0.26, 0.20, 0.14, 0.08])
    frac /= frac.sum()
    remaining = deadline - t0

    for ph in range(n_phases):
        if time.monotonic() > deadline:
            break
        phase_time = remaining * frac[ph]
        phase_dl = time.monotonic() + phase_time - 1.5

        # Re-seed rng for diversity
        sub_rng = np.random.default_rng(seed + ph * 1000)
        f_opt, sc_opt = run_optimize(best_f, phase_dl, sub_rng)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()

        # Symmetrization and blending (quick checks)
        if time.monotonic() < phase_dl + 0.5:
            sym = 0.5 * (best_f + best_f[::-1])
            sym = np.maximum(sym, 1e-20)
            sym /= sym.sum()
            ssc = score_of(sym)
            if ssc < best_sc:
                best_sc = ssc
                best_f = sym.copy()
            for t in (0.15, 0.3, 0.5):
                if time.monotonic() > deadline - 5.0:
                    break
                bl = (1.0 - t) * best_f + t * best_f[::-1]
                bl = np.maximum(bl, 1e-20)
                bl /= bl.sum()
                blsc = score_of(bl)
                if blsc < best_sc:
                    best_sc = blsc
                    best_f = bl.copy()

        # Small random perturbation attempts
        for _ in range(6):
            if time.monotonic() > deadline - 5.0:
                break
            trial = best_f * np.exp(0.01 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            tsc = score_of(trial)
            if tsc < best_sc:
                best_sc = tsc
                best_f = trial.copy()

    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
        final_sc = evaluate_sequence(result)
    return result
```
