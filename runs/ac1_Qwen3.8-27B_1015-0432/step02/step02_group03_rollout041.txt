    return np.fft.irfft(fa * fa, n=fs)[:2*n-1]

    if a.size == 0 or a.sum() < 0.01:
        return np.inf
    c = _conv(a)
    return 2.0 * len(a) * c.max() / (a.sum() ** 2)

    For the max, the subgradient is the mean of gradients at active lags."""

    fi = np.fft.rfft(ind, n=fs)
    g = np.fft.irfft(fa * np.conj(fi), n=fs)[:n]
    return g

import numpy as np
import time

def _fft_conv(a):
    n = a.size
    fs = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[:2 * n - 1]

def _fse_grad(a, c, fs, tau):
    n = a.size
    mc = float(c.max())
    z = tau * (c - mc)
    z = np.where(z < -40.0, -40.0, z)
    w = np.exp(z)
    sw = float(w.sum())
    if sw < 1e-300:
        return np.zeros(n)
    w /= sw
    fa = np.fft.rfft(a, n=fs)
    fw = np.fft.rfft(w, n=fs)
    g = np.fft.irfft(fa * np.conj(fw), n=fs)[:n]
    return g

def _score_fast(a):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (s * s)
