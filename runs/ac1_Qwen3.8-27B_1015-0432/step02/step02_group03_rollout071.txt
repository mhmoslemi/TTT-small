def _window(c, p, thr, maxw, minw):
    L = len(c)
    # boolean maybe c >= thr
    # To get contiguous around p within [p-maxw, p+maxw]
    lo0 = max(0, p - maxw)
    hi0 = min(L - 1, p + maxw)
    # start from p expand while above threshold
    lo = p; hi = p
    while lo > lo0 and c[lo - 1] >= thr:
        lo -= 1
    while hi < hi0 and c[hi + 1] >= thr:
        hi += 1
    # enforce min width
    cur = hi - lo + 1
    if cur < minw:
        # expand left/right sym within bounds favor similar slope
        lo = max(lo0, lo - (minw - cur)//2)
        hi = min(hi0, hi + (minw - cur) - ((minw - cur)//2))
    # cap max width
    return lo,hi

def _make_ws(n, L):
    j = np.arange(n, dtype=np.int64)
    ps = np.empty(n + 1, dtype=np.float64)
    ilo = np.empty(n, dtype=np.int64)
    ihi = np.empty(n, dtype=np.int64)
    valid = np.empty(n, dtype=bool)
    out = np.empty(n, dtype=np.float64)
    rev = (L - 1) - j
    return j, ps, ilo, ihi, valid, out, rev

def _grad_window(a, lo, hi, j, ps, ilo, ihi, valid, out):
    n = a.size
    ps[0] = 0.0
    np.cumsum(a, out=ps[1:])
    np.subtract(lo, j, out=ilo)
    np.maximum(ilo, 0, out=ilo)
    np.subtract(hi, j, out=ihi)
    np.minimum(ihi, n - 1, out=ihi)
    np.less_equal(ilo, ihi, out=valid)
    out.fill(0.0)
    out[valid] = ps[ihi[valid] + 1] - ps[ilo[valid]]
    out *= 2.0
    return out

def _window(c, p, thr, maxw, minw, L):
    lo0 = p - maxw
    if lo0 < 0:
        lo0 = 0
    hi0 = p + maxw
    if hi0 >= L:
        hi0 = L - 1
    lo = p
    hi = p
    while lo > lo0 and c[lo - 1] >= thr:
        lo -= 1
    while hi < hi0 and c[hi + 1] >= thr:
        hi += 1
    cur = hi - lo + 1
    if cur < minw:
        need = minw - cur
        left = need // 2
        if left > lo - lo0:
            left = lo - lo0
        lo -= left
        need -= left
        right = need
        if right > hi0 - hi:
            right = hi0 - hi
        hi += right
    return lo, hi

def _conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    c = np.fft.irfft(fa * fa, n=fs)
    return fa, c

def _soft_grad(fa, c, L, n, fs, temp, rev):
    mx = float(np.max(c[:L]))
    if mx <= 0.0 or not np.isfinite(mx):
        return np.zeros(n, dtype=np.float64)
    z = (c[:L] - mx) / temp
    np.clip(z, -700.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws <= 0.0 or not np.isfinite(ws):
        w.fill(0.0)
        w[L // 2] = 1.0
    else:
        w /= ws
    wr = w[::-1]
    fw = np.fft.rfft(wr, n=fs)
    g = np.fft.irfft(fa * fw, n=fs)
    g = g[rev]
    g *= 2.0
    return g

def _score(a, fs, L):
    n = a.size
    if n == 0:
        return np.inf
    fa, c = _conv(a, fs)
    mx = float(np.max(c[:L])) if L > 0 else 0.0
    if mx < 0.0 and mx > -1e-18:
        mx = 0.0
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s) or not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))

def _desc(a, fs, L, rng, dl, grad_soft, use_multi, eta0, beta, temp_rel, frac, maxw, minw, noise, stall_max, gamma, ws):
    j, ps, ilo, ihi, valid, out, rev = ws
    n = a.size
    s0 = float(np.sum(a))
    if s0 <= 0.0 or not np.isfinite(s0):
        return a, np.inf
    a = a / s0
    bst = a.copy()
    bsc = _score(a, fs, L)
    if not np.isfinite(bsc):
        return a, np.inf
    mom = np.zeros(n, dtype=np.float64)
    stall = 0
    prev = bsc
    eta = eta0
    for _ in range(10000000):
        if time.monotonic() > dl:
            break
        fa, c = _conv(a, fs)
        mx = float(np.max(c[:L]))
        sa = float(np.sum(a))
        if sa <= 1e-15 or not np.isfinite(sa) or not np.isfinite(mx) or mx <= 0.0:
            break
        sc = 2.0 * n * mx / (sa * sa)
        if not np.isfinite(sc):
            break
        if sc < bsc - 1e-13:
            bsc = sc
            bst = a.copy()
        if sc < prev - 1e-14:
            stall = 0
        else:
            stall += 1
        prev = sc
        if stall > stall_max:
            break
        if grad_soft:
            temp = mx * temp_rel
            if temp <= 0.0 or not np.isfinite(temp):
                temp = 1e-300
            prog = min(1.0, 0.5 * stall / max(1, stall_max // 4))
            temp *= (1.0 + prog)
            g = _soft_grad(fa, c, L, n, fs, temp, rev)
        else:
            p = int(np.argmax(c[:L]))
            prog = min(1.0, 0.5 * stall / max(1, stall_max // 4))
            frac_eff = frac * (1.0 + prog)
            if frac_eff > 0.25:
                frac_eff = 0.25
            maxw_eff = int(maxw + 4 * min(stall, 100))
            if maxw_eff < minw:
                maxw_eff = minw
            if maxw_eff > L - 1:
                maxw_eff = L - 1
            thr = mx * (1.0 - frac_eff)
            lo, hi = _window(c, p, thr, maxw_eff, minw, L)
            width = hi - lo + 1
            if width <= 0:
                g = np.zeros(n, dtype=np.float64)
            else:
                g = _grad_window(a, lo, hi, j, ps, ilo, ihi, valid, out) / float(width)
        dp = g - float(np.mean(g))
        deg = float(np.max(np.abs(dp)))
        if deg <= 1e-18:
            if stall > 20 and noise > 0.0:
                dp = noise * rng.standard_normal(n)
                dp -= float(np.mean(dp))
                deg = float(np.max(np.abs(dp)))
                if deg <= 1e-18:
                    break
            else:
                break
        if stall > stall_max // 3 and noise > 0.0:
            dm = float(np.max(np.abs(dp)))
            if dm > 1e-18:
                dp = dp + (noise * rng.standard_normal(n)) / dm
                dp -= float(np.mean(dp))
        if use_multi:
            dm = float(np.max(np.abs(dp)))
            if dm <= 1e-18:
                break
            step = np.clip(-eta * dp / dm, -2.0, 2.0)
            a = a * np.exp(step)
        else:
            a_scale = max(float(np.mean(a)), 1e-300)
            dp = dp / a_scale
            mom = beta * mom + (1.0 - beta) * dp
            a = np.clip(a - eta * mom, 0.0, None)
        sa = float(np.sum(a))
        if sa <= 1e-12 or not np.isfinite(sa):
            a = bst.copy()
            break
        a = a / sa
        eta *= gamma
    return bst, bsc

def _local_search(base, fs, L, rng, dl, use_exact, best_exact_seq, best_exact_score, best_search_score):
    n = base.size
    a = np.asarray(base, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s > 0.0:
        a = a / s
    search_best = a.copy()
    search_score = best_search_score if np.isfinite(best_search_score) else _score(a, fs, L)
    exact_best = np.asarray(best_exact_seq, dtype=np.float64).copy()
    exact_score = best_exact_score if np.isfinite(best_exact_score) else np.inf
    w1 = max(2, n // 400)
    w2 = max(3, n // 100)
    w3 = max(5, n // 30)
    sig = 0.15
    no_imp = 0
    exact_calls = 0
    max_exact = 250 if n <= 8000 else 80
    while time.monotonic() < dl:
        cand = a.copy()
        cnt = int(rng.integers(1, 4))
        for _ in range(cnt):
            w = int((w1, w2, w3)[int(rng.integers(0, 3))])
            if w > n:
                w = n
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            sc = sig * (0.5 + 0.5 * float(w) / float(max(1, n)))
            factor = float(np.exp(rng.normal(0.0, sc)))
            cand[st:e] = cand[st:e] * factor
        cand = np.clip(cand, 0.0, 1000.0)
        cs = float(np.sum(cand))
        if cs <= 0.01 or not np.isfinite(cs):
            continue
        sc = _score(cand, fs, L)
        if np.isfinite(sc) and sc < search_score - 1e-11:
            search_score = sc
            search_best = cand.copy()
            no_imp = 0
            sig = min(0.55, sig * 1.05)
            if use_exact and exact_calls < max_exact:
                exact_calls += 1
                exc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(exc) and exc < exact_score:
                    exact_score = exc
                    exact_best = cand.copy()
                a = cand / float(np.sum(cand))
            else:
                a = cand / float(np.sum(cand))
        else:
            no_imp += 1
        if no_imp > 300:
            sig = min(0.55, sig * 1.15)
            no_imp = 0
    return exact_best, exact_score, search_best, search_score

import numpy as np

def _safe_parent():
    obj = globals().get("height_sequence_1")
    if obj is None:
        p = np.ones(512, dtype=np.float64)
    else:
        p = np.asarray(obj, dtype=np.float64)
    p = np.atleast_1d(p).ravel().astype(np.float64, copy=False)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    if p.size < 2 or float(np.sum(p)) < 0.01:
        p = np.ones(512, dtype=np.float64)
    s = float(np.sum(p))
    if s > 0.0:
        p = p / s
    return p


def _make_ws(n, L):
    j = np.arange(n, dtype=np.int64)
    ps = np.empty(n + 1, dtype=np.float64)
    ilo = np.empty(n, dtype=np.int64)
    ihi = np.empty(n, dtype=np.int64)
    valid = np.empty(n, dtype=bool)
    out = np.empty(n, dtype=np.float64)
    rev = (L - 1) - j
    return j, ps, ilo, ihi, valid, out, rev


def _grad_window(a, lo, hi, j, ps, ilo, ihi, valid, out):
    n = a.size
    ps[0] = 0.0
    np.cumsum(a, out=ps[1:])
    np.subtract(lo, j, out=ilo)
    np.maximum(ilo, 0, out=ilo)
    np.subtract(hi, j, out=ihi)
    np.minimum(ihi, n - 1, out=ihi)
    np.less_equal(ilo, ihi, out=valid)
    out.fill(0.0)
    out[valid] = ps[ihi[valid] + 1] - ps[ilo[valid]]
    out *= 2.0
    return out


def _window(c, p, thr, maxw, minw, L):
    lo0 = p - maxw
    if lo0 < 0:
        lo0 = 0
    hi0 = p + maxw
    if hi0 >= L:
        hi0 = L - 1
    lo = p
    hi = p
    while lo > lo0 and c[lo - 1] >= thr:
        lo -= 1
    while hi < hi0 and c[hi + 1] >= thr:
        hi += 1
    cur = hi - lo + 1
    if cur < minw:
        need = minw - cur
        left = need // 2
        if left > lo - lo0:
            left = lo - lo0
        lo -= left
        need -= left
        right = need
        if right > hi0 - hi:
            right = hi0 - hi
        hi += right
    return lo, hi


def _conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    c = np.fft.irfft(fa * fa, n=fs)
    return fa, c


def _soft_grad(fa, c, L, n, fs, temp, rev):
    mx = float(np.max(c[:L]))
    if mx <= 0.0 or not np.isfinite(mx):
        return np.zeros(n, dtype=np.float64)
    z = (c[:L] - mx) / temp
    np.clip(z, -700.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(np.sum(w))
    if ws <= 0.0 or not np.isfinite(ws):
        w.fill(0.0)
        w[L // 2] = 1.0
    else:
        w /= ws
    wr = w[::-1]
    fw = np.fft.rfft(wr, n=fs)
    g = np.fft.irfft(fa * fw, n=fs)
    g = g[rev]
    g *= 2.0
    return g


def _score(a, fs, L):
    n = a.size
    if n == 0:
        return np.inf
    fa, c = _conv(a, fs)
    mx = float(np.max(c[:L])) if L > 0 else 0.0
    if mx < 0.0 and mx > -1e-18:
        mx = 0.0
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s) or not np.isfinite(mx):
        return np.inf
    return float(2.0 * n * mx / (s * s))

    n = a.size
    s0 = float(np.sum(a))
    if s0 <= 0.0 or not np.isfinite(s0):
        return a, np.inf
    a = a / s0
    bst = a.copy()
    bsc = _score(a, fs, L)
    if not np.isfinite(bsc):
        return a, np.inf

    mom = np.zeros(n, dtype=np.float64)
    stall = 0
    prev = bsc
    eta = eta0

    for _ in range(10000000):
        if time.monotonic() > dl:
            break

        fa, c = _conv(a, fs)
        mx = float(np.max(c[:L]))
        sa = float(np.sum(a))
        if sa <= 1e-15 or not np.isfinite(sa) or not np.isfinite(mx) or mx <= 0.0:
            break

        sc = 2.0 * n * mx / (sa * sa)
        if not np.isfinite(sc):
            break

        if sc < bsc - 1e-13:
            bsc = sc
            bst = a.copy()

        if sc < prev - 1e-14:
            stall = 0
        else:
            stall += 1
        prev = sc

        if stall > stall_max:
            break

        if grad_soft:
            temp = mx * temp_rel
            if temp <= 0.0 or not np.isfinite(temp):
                temp = 1e-300
            prog = min(1.0, 0.5 * stall / max(1, stall_max // 4))
            temp *= 1.0 + prog
            g = _soft_grad(fa, c, L, n, fs, temp, rev)
        else:
            p = int(np.argmax(c[:L]))
            prog = min(1.0, 0.5 * stall / max(1, stall_max // 4))
            frac_eff = frac * (1.0 + prog)
            if frac_eff > 0.25:
                frac_eff = 0.25
            maxw_eff = int(maxw + 4 * min(stall, 100))
            if maxw_eff < minw:
                maxw_eff = minw
            if maxw_eff > L - 1:
                maxw_eff = L - 1
            thr = mx * (1.0 - frac_eff)
            lo, hi = _window(c, p, thr, maxw_eff, minw, L)
            width = hi - lo + 1
            if width <= 0:
                g = np.zeros(n, dtype=np.float64)
            else:
                g = _grad_window(a, lo, hi, j, ps, ilo, ihi, valid, out) / float(width)

        dp = g - float(np.mean(g))
        deg = float(np.max(np.abs(dp)))
        mean_a = max(float(np.mean(a)), 1e-300)

        if deg <= 1e-18:
            if noise > 0.0:
                dp = 0.05 * mean_a * rng.standard_normal(n)
                dp -= float(np.mean(dp))
                deg = float(np.max(np.abs(dp)))
            if deg <= 1e-18:
                break

        if stall > stall_max // 3 and noise > 0.0:
            dm = float(np.max(np.abs(dp)))
            if dm > 1e-18:
                dp = dp + (noise * rng.standard_normal(n)) / dm
                dp -= float(np.mean(dp))

        if use_multi:
            dm = float(np.max(np.abs(dp)))
            if dm <= 1e-18:
                break
            step = np.clip(-eta * dp / dm, -2.0, 2.0)
            a = a * np.exp(step)
        else:
            if stall > 10 and noise > 0.0:
                amp = 0.02 * mean_a
                if float(np.max(np.abs(dp))) < amp:
                    dp = dp + amp * rng.standard_normal(n)
                    dp -= float(np.mean(dp))
            mom = beta * mom + (1.0 - beta) * dp
            a = np.clip(a - eta * mom, 0.0, None)

        sa = float(np.sum(a))
        if sa <= 1e-12 or not np.isfinite(sa):
            a = bst.copy()
            break
        a = a / sa
        eta *= gamma

    return bst, bsc


def _evaluate_safe(seq):
    try:
        return float(evaluate_sequence(seq.tolist()))
    except Exception:
        return np.inf


def _update_state(search_best, search_score, exact_best, exact_score,
                  a, s, use_exact, hard_dl):
    improved = False
    if np.isfinite(s) and s < search_score:
        search_best = a
        search_score = s
        improved = True
    if use_exact and improved and time.monotonic() < hard_dl - 25.0:
        exc = _evaluate_safe(search_best)
        if np.isfinite(exc) and exc < exact_score:
            exact_best = search_best.copy()
            exact_score = exc
    return search_best, search_score, exact_best, exact_score


def _local_search(base, fs, L, rng, dl, use_exact,
                  exact_best, exact_score, search_score):
    n = base.size
    a = np.asarray(base, dtype=np.float64).copy()
    s = float(np.sum(a))
    if s > 0.0:
        a = a / s
    a = np.maximum(a, 1e-12)
    s = float(np.sum(a))
    if s > 0.0:
        a = a / s

    search_best = a.copy()
    if not np.isfinite(search_score):
        search_score = _score(a, fs, L)

    exact_best = np.asarray(exact_best, dtype=np.float64).copy()
    if not np.isfinite(exact_score):
        exact_score = np.inf

    wins = (max(2, n // 400), max(3, n // 100), max(5, n // 30))
    sig = 0.25
    no_imp = 0
    exact_calls = 0
    max_exact = 250 if n <= 8000 else 80

    while time.monotonic() < dl:
        cand = a.copy()
        cnt = int(rng.integers(1, 4))
        for _ in range(cnt):
            w = int(wins[int(rng.integers(0, 3))])
            if w > n:
                w = n
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            sg = sig * (0.5 + 0.5 * float(w) / float(max(1, n)))
            factor = float(np.exp(rng.normal(0.0, sg)))
            cand[st:e] = cand[st:e] * factor

        cand = np.clip(cand, 0.0, 1000.0)
        cs = float(np.sum(cand))
        if cs <= 0.01 or not np.isfinite(cs):
            continue

        sc = _score(cand, fs, L)
        if np.isfinite(sc) and sc < search_score - 1e-11:
            search_best = cand.copy()
            search_score = sc
            no_imp = 0
            sig = min(0.55, sig * 1.05)
            a = cand / float(np.sum(cand))
            a = np.maximum(a, 1e-12)
            a = a / float(np.sum(a))

            if use_exact and exact_calls < max_exact:
                exact_calls += 1
                exc = _evaluate_safe(cand)
                if np.isfinite(exc) and exc < exact_score:
                    exact_best = cand.copy()
                    exact_score = exc
        else:
            no_imp += 1

        if no_imp > 300:
            sig = min(0.55, sig * 1.15)
            no_imp = 0

    return exact_best, exact_score, search_best, search_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    parent = _safe_parent()
    n = parent.size
    L = 2 * n - 1
    fs = 1 << max(1, (L - 1).bit_length())
    ws = _make_ws(n, L)

    best_search = parent.copy()
    best_search_score = _score(best_search, fs, L)

    use_exact = n <= 12000
    if use_exact:
        best_exact = parent.copy()
        best_exact_score = _evaluate_safe(best_exact)
        if not np.isfinite(best_exact_score):
            best_exact_score = best_search_score
    else:
        best_exact = parent.copy()
        best_exact_score = best_search_score

    hard_dl = t0 + max(20.0, float(budget_s) - 20.0)

    if n <= 8000:
        local_time = 55.0
    elif n <= 20000:
        local_time = 25.0
    else:
        local_time = 8.0

    total_window = hard_dl - t0
    if total_window < local_time + 60.0:
        local_time = max(1.0, 0.10 * total_window)

    local_dl = hard_dl - local_time
    total_opt = max(0.0, local_dl - t0)

    if total_opt > 20.0:
        d1 = t0 + 0.22 * total_opt
        d2 = t0 + 0.45 * total_opt
        d3 = t0 + 0.65 * total_opt
        d4 = t0 + 0.82 * total_opt

        if time.monotonic() < d1 - 1.0:
            a, s = _desc(best_search, fs, L, rng, d1,
                         False, False, 0.40, 0.86, 0.0, 2e-3,
                         90, 3, 0.04, 4000, 0.99996, ws)
            best_search, best_search_score, best_exact, best_exact_score = _update_state(
                best_search, best_search_score, best_exact, best_exact_score,
                a, s, use_exact, hard_dl)

        if time.monotonic() < d2 - 1.0:
            a, s = _desc(best_search, fs, L, rng, d2,
                         False, True, 0.25, 0.0, 0.0, 4e-3,
                         120, 5, 0.05, 3500, 0.99996, ws)
            best_search, best_search_score, best_exact, best_exact_score = _update_state(
                best_search, best_search_score, best_exact, best_exact_score,
                a, s, use_exact, hard_dl)

        if time.monotonic() < d3 - 1.0:
            a, s = _desc(best_search, fs, L, rng, d3,
                         True, True, 0.18, 0.0, 8e-4, 0.0,
                         0, 0, 0.04, 3000, 0.99997, ws)
            best_search, best_search_score, best_exact, best_exact_score = _update_state(
                best_search, best_search_score, best_exact, best_exact_score,
                a, s, use_exact, hard_dl)

        if time.monotonic() < d4 - 1.0:
            a, s = _desc(best_search, fs, L, rng, d4,
                         False, False, 0.30, 0.90, 0.0, 8e-4,
                         50, 2, 0.02, 4500, 0.99998, ws)
            best_search, best_search_score, best_exact, best_exact_score = _update_state(
                best_search, best_search_score, best_exact, best_exact_score,
                a, s, use_exact, hard_dl)

        if time.monotonic() < local_dl - 1.0:
            a, s = _desc(best_search, fs, L, rng, local_dl,
                         True, True, 0.10, 0.0, 1e-4, 0.0,
                         0, 0, 0.02, 4000, 0.99998, ws)
            best_search, best_search_score, best_exact, best_exact_score = _update_state(
                best_search, best_search_score, best_exact, best_exact_score,
                a, s, use_exact, hard_dl)

    if time.monotonic() < local_dl - 25.0:
        pert = best_search.copy()
        pert = pert * np.exp(rng.normal(0.0, 0.03, n))
        pert = np.clip(pert, 0.0, 1000.0)
        if float(np.sum(pert)) > 0.01:
            dl_r = min(local_dl - 15.0, time.monotonic() + 40.0)
            if dl_r > time.monotonic() + 4.0:
                a, s = _desc(pert, fs, L, rng, dl_r,
                             True, True, 0.12, 0.0, 4e-4, 0.0,
                             0, 0, 0.03, 3000, 0.99998, ws)
                best_search, best_search_score, best_exact, best_exact_score = _update_state(
                    best_search, best_search_score, best_exact, best_exact_score,
                    a, s, use_exact, hard_dl)

    if local_dl > t0 + 3.0:
        best_exact, best_exact_score, best_search, best_search_score = _local_search(
            best_search, fs, L, rng, local_dl, use_exact,
            best_exact, best_exact_score, best_search_score)

    if use_exact and np.isfinite(best_search_score) and time.monotonic() < hard_dl - 5.0:
        exc = _evaluate_safe(best_search)
        if np.isfinite(exc) and exc < best_exact_score:
            best_exact = best_search.copy()
            best_exact_score = exc

    if use_exact and np.isfinite(best_exact_score):
        out = best_exact
    else:
        out = best_search

    out = np.asarray(out, dtype=np.float64).ravel()
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(np.sum(out))
    if out.size == 0 or s < 0.01:
        out = parent
        s = float(np.sum(out))

    if s > 0.0:
        out = out / s
    out = np.clip(out, 0.0, 1000.0)

    if out.size == 0 or float(np.sum(out)) < 0.01 or not np.all(np.isfinite(out)):
        out = np.ones(512, dtype=np.float64)

    return [float(x) for x in out]
