    f, s = run_opt(bf.copy(), t1, rng, 0.10, 200, True, True)
    if s < bs:
        bs, bf = s, f

    # Phase 2: free, weighted
    t2 = min(T, t0 + 660.0)
    f, s = run_opt(bf.copy(), t2, rng, 0.07, 250, False, True)
    if s < bs:
        bs, bf = s, f

    # Phase 3: fine-tuning with more peaks
    t3 = min(T, T - 25.0)
    f, s = run_opt(bf.copy(), t3, rng, 0.03, 500, False, True)
    if s < bs:
        bs, bf = s, f

    # Phase 4: micro perturbations
    amps = [0.002, 0.005, 0.012, 0.03, 0.06]
    while time.monotonic() < T - 6.0:
        for amp in amps:
            if time.monotonic() > T - 6.0:
                break
            tr = norm(bf * np.exp(amp * rng.standard_normal(n)))
            sc = score_of(tr)
            if sc < bs:
                bs, bf = sc, tr
        # periodic tiny symmetric tweak
        tr2 = norm(0.5 * (bf + bf[::-1]) * np.exp(0.003 * rng.standard_normal(n)))
        sc2 = score_of(tr2)
        if sc2 < bs:
            bs, bf = sc2, tr2

    result = [float(v) for v in bf]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
