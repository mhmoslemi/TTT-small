import time

def _conv_full(a, fs):
    """Full autoconvolution via FFT."""
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _eval_score(a):
    """Compute 2n * max(conv) / sum^2."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    fs = 1 << (2 * n - 1).bit_length()
    c = _conv_full(a, fs)
    return 2.0 * n * c.max() / (s * s)

def _active_lags(c, frac=0.0, cap=100):
    """Get active lags where conv is near max."""
    mx = c.max()
    if frac > 0:
        thresh = mx * (1.0 - frac)
    else:
        thresh = mx
    lags = np.where(c >= thresh)[0]
    lags = lags[(lags >= 0) & (lags < 2 * len(c) - 1)]
    if len(lags) > cap:
        idx = np.argpartition(c[lags], -cap)[-cap:]
        lags = lags[idx]
    return lags

def _subgrad(a, lags):
    """Subgradient of max(conv) w.r.t. a at active lags."""
    n = a.size
    g = np.zeros(n)

        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:

            g[k-hi:k-lo+1] += a[lo:hi+1]
    g /= max(1, len(lags))
