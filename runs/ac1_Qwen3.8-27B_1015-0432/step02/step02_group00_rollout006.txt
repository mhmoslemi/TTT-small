import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def lp_direction(f, M, top_k, deadline):
    """Find LP direction to reduce autocorrelation at top lags."""
    n = f.size
    L = 2 * n - 1
    b = fft_conv(f, f)
    
    # Get top K lags
    thresh = M * 0.98
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > top_k:
        idx = np.argsort(b[peaks])[-top_k:]
        peaks = peaks[idx]
    
    # Build sparse T matrix for these lags
    # T[k_row, j] = f[k - j] if valid
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(peaks):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    
    if not rows:
        return None
    
    T = csr_matrix((vals, (rows, cols)), shape=(len(peaks), n))
    
    # LP: max sum(g) s.t. T@g <= M, g >= 0
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=T, b_ub=np.full(len(peaks), M), bounds=[(0, None)]*n, 
                  method='highs', options={'time_limit': 30, 'presolve': True})
    
    if res.success and res.x is not None:
        g = res.x
        if g.sum() > 1e-10:
            return g
    return None

from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.size) / f.size
    return f / s

def subgradient(f, peaks):
    n = f.size
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    b = fft_conv(f, f)
    thresh = M * 0.97

        idx = np.argsort(b[peaks])[-top_k:]
        peaks = peaks[idx]

        peaks = np.array([np.argmax(b)])

    all_cols = []

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)


def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.size) / f.size
    return f / s


def get_peaks(f, frac, max_k):
    b = fft_conv(f, f)
    M = b.max()
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_k:
        peaks = np.argsort(b)[-max_k:]
    if len(peaks) < 1:
        peaks = np.array([np.argmax(b)])
    return peaks, M


def subgrad_dir(f, peaks):
    n = f.size
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)


def lp_direction(f, M, top_k, time_limit):
    n = f.size
    b = fft_conv(f, f)
    thresh = M * 0.96
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > top_k:
        peaks = peaks[np.argsort(b[peaks])[-top_k:]]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(b)])

    K = len(peaks)
    all_rows, all_cols, all_vals = [], [], []
    for i, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        cnt = je - js + 1
        if cnt <= 0:
            continue
        all_rows.extend([i] * cnt)
        all_cols.extend(range(js, je + 1))
        all_vals.extend(f[k - je:k - js + 1][::-1])

    if not all_rows:
        return None

    T = csr_matrix((all_vals, (all_rows, all_cols)), shape=(K, n))
    obj = -np.ones(n)
    try:
        res = linprog(obj, A_ub=T, b_ub=np.full(K, M),
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None and res.x.sum() > 1e-10:
            return res.x
    except Exception:
        pass
    return None


def multiplicative_update(f, eta, peaks, sym):
    g = subgrad_dir(f, peaks)
    gmax = g.max()
    if gmax < 1e-30:
        return f
    f = f * np.exp(-eta * g / gmax)
    f = np.maximum(f, 1e-20)
    if sym:
        f = 0.5 * (f + f[::-1])
    return normalize(f)


def self_optimize(f, deadline, rng, use_lp, lp_every, sym, lp_topk, lp_tl):
    n = f.size
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.1
    no_improv = 0
    restarts = 0

    for it in range(200000):
        if time.monotonic() > deadline - 3.0:
            break

        frac = 0.99 - 0.03 * min(1.0, it / 2000.0)
        peaks, M = get_peaks(f, frac, 150)

        did_lp = False
        if use_lp and it % lp_every == 0:
            g_dir = lp_direction(f, M, lp_topk, lp_tl)
            if g_dir is not None:
                g_dir = normalize(g_dir)
                best_t, best_sc_t = 0.0, score(f)
                for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6, 0.8):
                    trial = (1.0 - t) * f + t * g_dir
                    trial = normalize(np.maximum(trial, 1e-20))
                    if sym:
                        trial = 0.5 * (trial + trial[::-1])
                    sc_t = score(trial)
                    if sc_t < best_sc_t:
                        best_sc_t = sc_t
                        best_t = t
                if best_t > 0.0:

                    if sym:
