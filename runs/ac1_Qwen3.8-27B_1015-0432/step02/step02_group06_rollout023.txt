import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * fft_conv(f, f).max() / (s * s)

def compute_grad(f, beta, lam2):
    b = fft_conv(f, f)
    M = b.max()
    w = np.zeros_like(b)
    if M > 1e-30:
        w += np.exp(beta * (b / M - 1.0))
    if lam2 > 0.0:
        w += 2.0 * lam2 * b
    gc = fft_conv(w, f[::-1])
    return gc[f.size - 1:2 * f.size - 1]

def norm_pos(x):
    np.maximum(x, 1e-25, out=x)
