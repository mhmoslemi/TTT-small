import numpy as np
import time

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def sgrad(a, c, tk):
    n = a.size
    idx = np.argpartition(c, -tk)[-tk:]
    w = (c[idx] / (c[idx].max() + 1e-30)) ** 8
    g = np.zeros(n)
    for ii in range(tk):
        k = int(idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * w[ii] * a[k - hi:k - lo + 1][::-1]
    ts = w.sum()
    if ts > 1e-30:
        g /= ts
    g -= g.mean()
    return g

def lp_solve(a, c, M, tk, tl):
    from scipy.optimize import linprog
    n = a.size
    idx = np.argpartition(c, -tk)[-tk:]
    A = np.zeros((tk, n))
    for i in range(tk):
        k = int(idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[i, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    r = linprog(-np.ones(n), A_ub=A, b_ub=np.full(tk, M),
                bounds=[(0, None)] * n, method='highs',
                options={'time_limit': tl, 'presolve': True})
    if r.success and r.x is not None:
        g = np.maximum(r.x, 0.0)
        s = g.sum()
        if s > 1e-14:
            return g / s
    return None

def descend(a, n, fs, end, lk=5):
    v = np.zeros(n)
    mom = 0.82
    lr = 0.45
    tk = 6
    stall = 0
    pM = np.inf
    ba = a.copy()
    bM = np.inf
    for it in range(900000):
        if time.monotonic() > end:
            break
        c = fft_conv(a, fs)
        M = float(c.max())
        if M < bM:
            bM = M
            ba = a.copy()
        stall = 0 if M < pM - 1e-15 else stall + 1
        pM = M
        if stall > 5000 and it > 400:
            break
        g = sgrad(a, c, min(tk, 2 * n - 1))
        v = mom * v + (1.0 - mom) * g
        v -= v.mean()
        ok = False
        for bs in range(7):
            st = lr / (2.0 ** bs)
            at = np.clip(a - st * v, 0.0, None)
            s = at.sum()
            if s < 1e-14:
                break
            at /= s
            Mt = float(fft_conv(at, fs).max())
            if Mt <= M + 1e-14:
                a = at
                M = Mt
                if M < bM:
                    bM = M
                    ba = a.copy()
                ok = True
                break
        if not ok:
            a = np.clip(a - 0.04 * lr * v, 0.0, None)
            s = a.sum()
            if s < 1e-14:
                break
            a /= s
        if it % 700 == 0 and tk < 56:
            tk += 1
        if it > 0 and it % (lk * 700) == 0 and time.monotonic() < end - 8.0:
            gg = lp_solve(a, c, M, min(12, tk), 2.5)
            if gg is not None:
                for t in [0.15, 0.35, 0.6]:
                    bl = (1.0 - t) * a + t * gg
                    bl = np.clip(bl, 0.0, None)
                    s = bl.sum()
                    if s < 1e-14:
                        continue
                    bl /= s
                    mb = float(fft_conv(bl, fs).max())
                    if mb < M - 1e-15:
                        a = bl
                        M = mb

                            ba = a.copy()
                        break
        lr = max(0.006, lr * 0.99996)
        mom = min(0.96, mom + 0.000002)
    return ba, bM
