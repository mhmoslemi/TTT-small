import time

def fft_conv(a, b):
    """Compute full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def compute_score(f):
    """Compute the evaluator score."""
    n = len(f)
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def subgradient(f, peaks, L):
    """Compute subgradient at given peaks using FFT."""
    n = len(f)
    h = np.zeros(L)  # will use rfft with size L
    h[peaks] = 1.0
    F = np.fft.rfft(f, L)
    H = np.fft.rfft(h, L)
    g = np.fft.irfft(H * np.conj(F), L)[:n]
    return np.maximum(g, 0.0)

                no_improve = 0

def fft_conv(a, b):
    n, m = len(a), len(b)
