import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def grad_topk(f, C, k_list):
    n = f.size
    grad = np.zeros(n)
    for k in k_list:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]
    return grad
