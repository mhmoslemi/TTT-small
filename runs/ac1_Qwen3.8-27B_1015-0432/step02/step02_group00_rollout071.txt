```
def psize(n):

def norm_vec(f):
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.clip(f, 0.0, 1000.0)
    s = f.sum()
    if s <= 1e-15:
        n = f.size
        if n == 0: return None
        return np.ones(n)/n
    return f/s

def autocorr_from_F(F, n):
    c = np.fft.irfft(F*F)
    c = c[:2*n-1]
```

```
import numpy as np
import time

def psize(n):
    m = 3*n - 2
    if m < 1:
        m = 1
    return 1 << m.bit_length()

def norm_vec(f):
    f = np.asarray(f, dtype=np.float64).copy()
    if f.size == 0:
        f = np.ones(1)
    np.clip(f, 0.0, 1000.0, out=f)
    s = f.sum()
    if s <= 1e-15:
        f = np.ones(f.size)
    else:
        f *= 1.0/s
    return f

def autocorr_F(F, n):
    c = np.fft.irfft(F*F)
    c = c[:2*n-1]
    c = 0.5*(c + c[::-1])
    return c

def score_vec(f):
    n = f.size
    if n == 0:
        return np.inf
    size = psize(n)
    F = np.fft.rfft(f, size)
    c = autocorr_F(F, n)
    return 2.0*n*c.max()

def grad_F(F, w, size, n):
    W = np.fft.rfft(w[::-1], size)
    t = np.fft.irfft(F*W)
    g = 2.0 * t[n-1:2*n-1][::-1]
    np.maximum(g, 0.0, out=g)
    return g

def make_hard_weights(c, frac, maxk):

    M = c.max()

        w = np.zeros(L); i=int(np.argmax(c)); w[i]=1; w[L-1-i]=1; return w

    cand = np.unique(np.concatenate((cand, L-1-cand)))
```

```python
import numpy as np
import time

def psize(n):
    m = 3 * n - 2
    if m < 1:
        m = 1
    return 1 << m.bit_length()

def norm_vec(f):
    f = np.asarray(f, dtype=np.float64).copy()
    if f.size == 0:
        f = np.ones(1)
    f = np.nan_to_num(f, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f, 0.0, 1000.0, out=f)
    s = f.sum()
    if s <= 1e-15:
        f = np.ones(f.size)
    else:
        f *= 1.0 / s
    return f

def autocorr_F(F, n):
    c = np.fft.irfft(F * F)
    c = c[:2 * n - 1]
    c = 0.5 * (c + c[::-1])
    return c

def score_vec(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.inf
    size = psize(n)
    F = np.fft.rfft(f, size)
    c = autocorr_F(F, n)
    return 2.0 * n * c.max()

def grad_F(F, w, size, n):
    W = np.fft.rfft(w[::-1], size)
    t = np.fft.irfft(F * W)
    g = 2.0 * t[n - 1:2 * n - 1][::-1]
    np.maximum(g, 0.0, out=g)
    return g

def make_hard_weights(c, frac, maxk):
    L = c.size
    M = c.max()
    w = np.zeros(L)
    im = int(np.argmax(c))
    w[im] = 1.0
    w[L - 1 - im] = 1.0
    if M <= 1e-300 or frac >= 1.0:
        return w
    cand = np.nonzero(c >= M * frac)[0]
    if cand.size > maxk:
        cand = np.argpartition(c, -maxk)[-maxk:]
    if cand.size == 0:
        return w
    cand = np.unique(np.concatenate((cand, L - 1 - cand)))
    w[cand] = 1.0
    return w

def make_soft_weights(c, alpha):
    L = c.size
    M = c.max()
    w = np.zeros(L)
    if M <= 1e-300:
        im = int(np.argmax(c))
        w[im] = 1.0
        w[L - 1 - im] = 1.0
        return 0.5 * w
    z = alpha * (c - M)
    np.clip(z, -70.0, 0.0, out=z)
    w = np.exp(z)
    sw = w.sum()
    if sw <= 1e-300:
        return w
    w *= 1.0 / sw
    return w

def search(f, deadline, rng, mode, alpha0, alpha1, frac0, frac1, maxk, eta0, check, patience, ramp):
    f = norm_vec(f)
    n = f.size
    size = psize(n)
    best = f.copy()
    c0 = autocorr_F(np.fft.rfft(best, size), n)
    best_sc = 2.0 * n * c0.max()
    eta = float(eta0)
    it = 0
    no = 0
    rescale = 0
    pos = np.linspace(-0.5, 0.5, n)
    while True:
        if time.monotonic() > deadline:
            break
        F = np.fft.rfft(f, size)
        c = autocorr_F(F, n)
        M = c.max()
        sc = 2.0 * n * M
        improved = sc < best_sc - 1e-14
        if improved:
            best_sc = sc
            best = f.copy()
            no = 0
        L = 2 * n - 1
        progress = min(1.0, it / max(1, ramp))
        if mode == 0:
            frac = frac0 + (frac1 - frac0) * progress
            w = make_hard_weights(c, frac, maxk)
        else:
            base = alpha0 + (alpha1 - alpha0) * progress
            alpha = base / max(M, 1e-300)
            w = make_soft_weights(c, alpha)
        g = grad_F(F, w, size, n)
        gmax = g.max()
        if gmax <= 1e-300:
            break
        f = f * np.exp(-eta * g / (gmax + 1e-300))
        np.maximum(f, 1e-300, out=f)
        s = f.sum()
        if s <= 1e-15:
            f = best.copy()
        else:
            f *= 1.0 / s
        it += 1
        if check and it % check == 0:
            if not improved:
                no += 1
            if no > patience:
                scale = 0.05 + 0.02 * rescale
                if scale > 0.35:
                    scale = 0.35
                f = best * np.exp(scale * rng.standard_normal(n))
                tilt = rng.uniform(-0.2, 0.2)
                f = f * np.exp(tilt * pos)
                np.maximum(f, 1e-300, out=f)
                s = f.sum()
                if s <= 1e-15:
                    f = best.copy()
                else:
                    f *= 1.0 / s
                no = 0
                rescale += 1
                if rescale % 3 == 0 and eta < 0.35:
                    eta = min(0.45, eta * 1.25)
                else:
                    eta = max(0.03, eta * 0.85)
            if improved:
                eta = min(0.45, eta * 1.02)
            else:
                eta = max(0.03, eta * 0.995)
    return best, best_sc

def make_starts(parent, n, rng):
    starts = []
    p = np.asarray(parent, dtype=np.float64)
    if p.size == n:
        p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        s = p.sum()
        if s > 1e-15:
            starts.append(p / s)
    if n <= 0:
        starts.append(np.ones(1))
        return starts
    starts.append(np.ones(n) / n)
    pos = np.linspace(-0.5, 0.5, n)
    for rho in (0.25, 0.6, 0.85):
        r = 1.0 + rho * np.clip(rng.standard_normal(n), -1.0, 1.0)
        np.maximum(r, 1e-3, out=r)
```

```python
import numpy as np
import time


def psize(n):
    m = 3 * n - 2
    if m < 1:
        m = 1
    return 1 << m.bit_length()


def norm_vec(f):
    f = np.asarray(f, dtype=np.float64).copy()
    if f.size == 0:
        f = np.ones(1)
    f = np.nan_to_num(f, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f, 0.0, 1000.0, out=f)
    s = f.sum()
    if s <= 1e-15:
        f = np.ones(f.size)
    else:
        f *= 1.0 / s
    return f


def autocorr_F(F, n):
    c = np.fft.irfft(F * F)
    c = c[:2 * n - 1]
    c = 0.5 * (c + c[::-1])
    return c


def score_vec(f):
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return np.inf
    size = psize(n)
    F = np.fft.rfft(f, size)
    c = autocorr_F(F, n)
    return 2.0 * n * c.max()


def grad_F(F, w, size, n):
    W = np.fft.rfft(w[::-1], size)
    t = np.fft.irfft(F * W)
    g = 2.0 * t[n - 1:2 * n - 1][::-1]
    np.maximum(g, 0.0, out=g)
    return g


def make_hard_weights(c, frac, maxk):
    L = c.size
    M = c.max()
    w = np.zeros(L)
    im = int(np.argmax(c))
    w[im] = 1.0
    w[L - 1 - im] = 1.0
    if M <= 1e-300 or frac >= 1.0:
        return w
    cand = np.nonzero(c >= M * frac)[0]
    if cand.size > maxk:
        cand = np.argpartition(c, -maxk)[-maxk:]
    if cand.size == 0:
        return w
    cand = np.unique(np.concatenate((cand, L - 1 - cand)))
    w[cand] = 1.0
    return w


def make_soft_weights(c, alpha):
    L = c.size
    M = c.max()
    w = np.zeros(L)
    if M <= 1e-300:
        im = int(np.argmax(c))
        w[im] = 1.0
        w[L - 1 - im] = 1.0
        return 0.5 * w
    z = alpha * (c - M)
    np.clip(z, -70.0, 0.0, out=z)
    w = np.exp(z)
    sw = w.sum()
    if sw <= 1e-300:
        return w
    w *= 1.0 / sw
    return w


def search(f, deadline, rng, mode, alpha0, alpha1, frac0, frac1, maxk, eta0, check, patience, ramp):
    f = norm_vec(f)
    n = f.size
    size = psize(n)
    best = f.copy()
    c0 = autocorr_F(np.fft.rfft(best, size), n)
    best_sc = 2.0 * n * c0.max()
    eta = float(eta0)
    it = 0
    no = 0
    rescale = 0
    pos = np.linspace(-0.5, 0.5, n)

    while True:
        if time.monotonic() > deadline:
            break

        F = np.fft.rfft(f, size)
        c = autocorr_F(F, n)
        M = c.max()
        sc = 2.0 * n * M
        improved = sc < best_sc - 1e-14
        if improved:
            best_sc = sc
            best = f.copy()
            no = 0

        L = 2 * n - 1
        progress = min(1.0, it / max(1, ramp))
        if mode == 0:
            frac = frac0 + (frac1 - frac0) * progress
            w = make_hard_weights(c, frac, maxk)
        else:
            base = alpha0 + (alpha1 - alpha0) * progress
            alpha = base / max(M, 1e-300)
            w = make_soft_weights(c, alpha)

        g = grad_F(F, w, size, n)
        gmax = g.max()
        if gmax <= 1e-300:
            break

        f = f * np.exp(-eta * g / (gmax + 1e-300))
        np.maximum(f, 1e-300, out=f)
        s = f.sum()
        if s <= 1e-15:
            f = best.copy()
        else:
            f *= 1.0 / s

        it += 1
        if check and it % check == 0:
            if not improved:
                no += 1
            if no > patience:
                scale = 0.05 + 0.02 * rescale
                if scale > 0.35:
                    scale = 0.35
                f = best + (1e-3 / n)
                f = f * np.exp(scale * rng.standard_normal(n))
                tilt = rng.uniform(-0.2, 0.2)
                f = f * np.exp(tilt * pos)
                np.maximum(f, 1e-300, out=f)
                s = f.sum()
                if s <= 1e-15:
                    f = best.copy()
                else:
                    f *= 1.0 / s
                no = 0
                rescale += 1
                if rescale % 3 == 0 and eta < 0.35:
                    eta = min(0.45, eta * 1.25)
                else:
                    eta = max(0.03, eta * 0.85)

            if improved:
                eta = min(0.45, eta * 1.02)
            else:
                eta = max(0.03, eta * 0.995)

    return best, best_sc


def make_starts(parent, n, rng):
    starts = []
    p = np.asarray(parent, dtype=np.float64)
    if p.size == n:
        p = np.nan_to_num(p, nan=0.0, posinf=1000.0, neginf=0.0)
        p = np.clip(p, 0.0, 1000.0)
        s = p.sum()
        if s > 1e-15:
            starts.append(p / s)
    if n <= 0:
        starts.append(np.ones(1))
        return starts

    starts.append(np.ones(n) / n)
    pos = np.linspace(-0.5, 0.5, n)

    for rho in (0.25, 0.6, 0.85):
        r = 1.0 + rho * np.clip(rng.standard_normal(n), -1.0, 1.0)
        np.maximum(r, 1e-3, out=r)
        starts.append(r / r.sum())

    for a in (0.5, 1.0, 2.0, -0.5, -1.0):
        r = np.exp(a * pos)
        starts.append(r / r.sum())

    r = np.exp(0.4 * rng.standard_normal(n))
    starts.append(r / r.sum())

    for q in (1, 2, 3, 5):
        phase = rng.uniform(0.0, 6.28318)
        r = 1.0 + 0.6 * np.sin(2.0 * np.pi * q * pos + phase)
        np.maximum(r, 1e-3, out=r)
        starts.append(r / r.sum())

    return starts


def polish(f, deadline, rng):
    f = norm_vec(f)
    n = f.size
    if n == 0:
        return f, np.inf
    size = psize(n)
    best = f.copy()
    c0 = autocorr_F(np.fft.rfft(best, size), n)
    best_sc = 2.0 * n * c0.max()
    pos = np.linspace(-0.5, 0.5, n)
    cnt = 0

    while time.monotonic() < deadline:
        scale = 0.002 + 0.02 * abs(rng.standard_normal())
        if cnt % 37 == 0:
            scale = max(scale, 0.04)

        trial = best + (1e-6 / n)
        trial = trial * np.exp(scale * rng.standard_normal(n))
        tilt = rng.uniform(-0.1, 0.1)
        trial = trial * np.exp(tilt * pos)
        np.maximum(trial, 1e-300, out=trial)
        s = trial.sum()
        if s <= 1e-15:
            trial = best.copy()
        else:
            trial *= 1.0 / s

        c = autocorr_F(np.fft.rfft(trial, size), n)
        sc = 2.0 * n * c.max()
        if sc < best_sc - 1e-15:
            best = trial
            best_sc = sc
        cnt += 1

    return best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    try:
        parent = height_sequence_1
    except NameError:
        parent = []

    n = len(parent)
    if n <= 0:
        return [1.0]

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    avail = max(5.0, float(budget_s) - 25.0)
    deadline = t0 + avail

    starts = make_starts(parent, n, rng)
    f = None
    fsc = np.inf
    start_dl = min(deadline, t0 + min(60.0, 0.05 * avail))
    for g in starts:
        if time.monotonic() > start_dl:
            break
        sc = score_vec(g)
        if np.isfinite(sc) and sc < fsc:
            fsc = sc
            f = g.copy()

    if f is None:
        f = np.ones(n) / n
        fsc = score_vec(f)

    dl = t0 + 0.25 * avail
    f1, sc1 = search(f, min(deadline, dl), rng, 1, 1.5, 12.0, 0.0, 0.0, 1200, 0.18, 20, 120, 6000)
    if sc1 < fsc:
        f, fsc = f1, sc1

    dl = t0 + 0.55 * avail
    f1, sc1 = search(f, min(deadline, dl), rng, 1, 8.0, 60.0, 0.0, 0.0, 1200, 0.14, 20, 140, 8000)
    if sc1 < fsc:
        f, fsc = f1, sc1

    dl = t0 + 0.80 * avail
    f1, sc1 = search(f, min(deadline, dl), rng, 0, 0.0, 0.0, 0.995, 0.975, 1200, 0.10, 20, 160, 7000)
    if sc1 < fsc:
        f, fsc = f1, sc1

    dl = t0 + 0.93 * avail
    f1, sc1 = search(f, min(deadline, dl), rng, 0, 0.0, 0.0, 0.998, 0.985, 1200, 0.06, 20, 200, 6000)
    if sc1 < fsc:
        f, fsc = f1, sc1

    polish_dl = min(deadline, t0 + max(1.0, avail - 2.0))
    f, fsc = polish(f, polish_dl, rng)

    result = [float(v) for v in f]
    val = evaluate_sequence(result)
    if not np.isfinite(val):
        result = [float(1.0 / n) for _ in range(n)]
        val2 = evaluate_sequence(result)
        if np.isfinite(val2):
            return result

    return result
```
