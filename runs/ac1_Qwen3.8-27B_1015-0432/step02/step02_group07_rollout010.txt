import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)

def subgradient(a, conv, top_k, threshold_frac):
    n = a.size
    M = float(np.max(conv))
    # Find active set: lags within threshold_frac of max
    active = np.where(conv >= M * threshold_frac)[0]
    if len(active) > top_k:
        idx = np.where(conv >= M * threshold_frac)[0]
        # Take the top_k among active
        active = idx[np.argpartition(conv[idx], -top_k)[-top_k:]]
    elif len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    grad /= float(len(active))
    return grad

def project_nonneg_sum(v, s_target):
    """Project v onto {x >= 0, sum(x) = s_target}"""
    # First, clip negatives
    v = np.clip(v, 0, None)
    s = v.sum()
    if s < 1e-15:
        v = np.ones_like(v) / v.size * s_target
        return v
    v = v / s * s_target
    return v

def lp_refine(a, conv, M, active_lags, deadline):
    """Solve LP to find better direction. Returns refined a."""
    n = a.size
    # Build sparse constraint matrix for convolution constraints
    # A * b <= M for each active lag k
    # A[k, j] = a[k-j] for valid j
    
    rows = []
    cols = []
    vals = []
    for ii, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    
    # Maximize sum(b) subject to A @ b <= M, b >= 0
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_dir = linprog(c, A_ub=A, b_ub=np.full(len(active_lags), M), 
                    bounds=[(0, None)] * n, method='highs',
                    options={'time_limit': max(1, deadline - time.monotonic())})
    
    if b_dir.success:
        b = b_dir.x
        s_b = b.sum()
        if s_b > 1e-10:
            b = b / s_b * a.sum()
            return b
    return None

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def subgradient(a, conv, top_k, thresh_frac):
    n = a.size
    M = float(np.max(conv))
    active = np.where(conv >= M * thresh_frac)[0]
    if len(active) > top_k:
        c_act = conv[active]
        sel = active[np.argpartition(c_act, -top_k)[-top_k:]]
        active = sel
    elif len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad /= float(len(active))
    return grad


def lp_refine(a, conv, M, active_lags, timeout_s):
    n = a.size
    rows, cols, vals = [], [], []
    for ii, k in enumerate(active_lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(ii)
            cols.append(j)
            vals.append(a[k - j])
    if not vals:
        return None
    A = csr_matrix((vals, (rows, cols)), shape=(len(active_lags), n))
    c_obj = -np.ones(n)
    t_end = time.monotonic() + max(0.5, timeout_s)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(len(active_lags), M),
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': max(0.5, timeout_s)})
    if res.success and res.x is not None:
        b = np.asarray(res.x, dtype=np.float64)
        sb = float(b.sum())
        if sb > 1e-12:
            return b / sb * float(a.sum())
    return None
