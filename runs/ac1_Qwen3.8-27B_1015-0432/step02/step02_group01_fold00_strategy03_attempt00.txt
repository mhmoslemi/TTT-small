        total = np.sum(cand_next)
        if total < 0.01: cand_next = 0.01 * np.ones_like(cand_next)
        else: cand_next = cand_next / total
        
        # Evaluate candidate
        new_obj = evaluate_safe(cand_next[:10])
        if new_obj <= obj_initial:
            cand_best = cand_next
            obj_initial = new_obj
        
        iter_count += 1
        if iter_count > 10000: break
        if grad is None: grad = np.random.randn(len(cand_best)) * 0.1
        eta = max(0.001, eta * 0.9)
        beta *= beta_decay  # Increase focus over phases
    
    # Final candidate for return
    final_candidate = cand_best.tolist()
    return final_candidate
