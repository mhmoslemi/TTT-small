import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _sm(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _sm(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _sm(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _polish(a, steps=50, lr=0.001):
    """Polish using exact max and subgradient."""
    n = a.size
    scale = np.sqrt(2.0 * n)
    for s in range(steps):
        S = a.sum()
        if S < 1e-8:
            break
        c = _conv(a)
        kmax = c.argmax()
        maxv = c[kmax]
        # Gradient of conv at lag kmax: 2*a[kmax-j] for valid j
        lo = max(0, kmax - (n - 1))
        hi = min(kmax, n - 1)
        g = np.zeros(n)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[kmax - idx]
        grad = (2.0 * n / (S * S)) * g - (4.0 * n * maxv / (S ** 3))
        a = a - lr * grad
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() > 1e-8:
            a = a / a.sum() * scale
    return a

def _adam_optimize(a, tau, t0, dl, max_iter=2000):
    n = a.size
    m_v = np.zeros(n)  # first moment
    v_v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01
    scale = np.sqrt(2.0 * n)
    cur = a.copy()
    t = 0
    best_local = cur.copy()
    best_sc_local = float(evaluate_sequence(cur.tolist()))
    while t < max_iter and time.monotonic() < dl - 10:
        g = _grad(cur, tau)
        t += 1
        m_v = beta1 * m_v + (1 - beta1) * g
        v_v = beta2 * v_v + (1 - beta2) * g * g
        m_hat = m_v / (1 - beta1**t)
        v_hat = v_v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = cur - step
        cur = np.clip(cur, 0.0, 1000.0)
        if cur.sum() > 1e-8:
            cur = cur / cur.sum() * scale
        sc = float(evaluate_sequence(cur.tolist()))
        if np.isfinite(sc) and sc < best_sc_local:
            best_local = cur.copy()
            best_sc_local = sc
        # Decay learning rate
        lr *= 0.999
    return best_local, best_sc_local
