A better approach might be:
1. Use a **smoothed objective**: replace max_k c_k with log-sum-exp(c_k / T) * T for small T. This gives a smooth approximation.
2. The gradient of the smoothed objective is: (1/T) * sum_k c_k * exp(c_k/T) / sum_k exp(c_k/T) * (∂c_k/∂a_i)
3. This is a weighted sum over ALL lags, not just the top ones.

But computing this gradient requires summing over all 2n-1 lags, which is O(n²) per gradient evaluation. For n=4498, that's 20M operations per gradient. At 1M ops/sec, that's 20 seconds per gradient. Way too slow.

Alternative: use a **sparse softmax** - only include the top K lags in the softmax. This gives O(K*n) per gradient, same as before, but with better weighting.

    n = a.size
    # Pick top K lags
    top_k = np.argpartition(c, -K)[-K:]
    w = c[top_k]
    w_exp = np.exp((w - w.max()) / T)
    w_norm = w_exp / w_exp.sum()
    g = np.zeros(n)
    for idx, k in enumerate(top_k):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += w_norm[idx] * 2.0 * a[k-hi:k-lo+1][::-1]
    return g

def _cv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def _sgw(a, act, c):
    n = a.size
    m = len(act)
    if m == 0:
        return np.zeros(n)
    w = c[act].astype(np.float64)
    w -= w.max()
    np.exp(w, out=w)
    w /= w.sum()
    g = np.zeros(n)
    for j in range(m):
        k = int(act[j])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += w[j] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _descent(a0, n, fs, S, eta, mu, frac, cap, smx, multi, dt):
    a = a0.copy()
    mo = np.zeros(n)
    bst = a.copy()
    c = _cv(a, fs)
    bsc = 2.0 * n * float(c.max()) / (a.sum() ** 2)
    psc = bsc
    st = 0
    e = eta
    for _ in range(2000000):
        if time.monotonic() > dt:
            break
        c = _cv(a, fs)
        sc = 2.0 * n * float(c.max()) / (a.sum() ** 2)
        if sc < bsc - 1e-14:
            bsc = sc
            bst = a.copy()
        st = 0 if sc < psc - 1e-15 else st + 1
        psc = sc
        if st > smx:
            break
        mx = float(c.max())
        if mx < 1e-30:
            break
        act = np.where(c >= mx * (1.0 - frac))[0]
        act = act[(act >= 0) & (act < 2 * n - 1)]
        if len(act) > cap:
            ii = np.argpartition(c[act], -cap)[-cap:]
            act = act[ii]
        g = _sgw(a, act, c)
        g -= g.mean()
        if multi:
            gm = np.abs(g).max()
            if gm > 1e-14:
                a = np.maximum(a * np.exp(-e * g / gm), 0.0)
        else:
            mo = mu * mo + (1.0 - mu) * g
            gm = np.abs(mo).max()
            if gm > 1e-14:
                a = np.maximum(a - e * mo / gm, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a *= S / s
        e *= 0.99999
    return bst, bsc
