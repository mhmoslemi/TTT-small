    if par.size < 2 or not np.isfinite(par).all() or par.sum() < 0.01:
        par = np.ones(512, dtype=np.float64)
    n = par.size
    S0 = float(par.sum())
    fsz = 1 << (2 * n - 1).bit_length()
    a0 = par / S0
    best_a = a0.copy()
    best_sc = float(evaluate_sequence(par.tolist()))
    if not np.isfinite(best_sc):
        best_sc = 1e10

    # Main optimization on parent with large top-k
    rem = dl - time.monotonic()
    if rem > 120:
        t = min(rem - 60, 700.0)
        a1, _ = run_opt(a0, n, fsz, time.monotonic() + t, 128, 0.08, 0.82)
        sc = float(evaluate_sequence((a1 * S0).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a1.copy()

    # Secondary: alternate initializations
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(np.exp(-0.5 * ((x - 0.5*n) / (0.2*n))**2))
    inits.append(np.exp(-0.5*((x-0.3*n)/(0.12*n))**2) + np.exp(-0.5*((x-0.7*n)/(0.12*n))**2))
    r = np.abs(rng.standard_normal(n))
    for _ in range(8):
        r = 0.5 * (r + np.roll(r, 3))
    inits.append(r)
    for i, init in enumerate(inits):
        rem = dl - time.monotonic()
        if rem < 120:
            break
        t = min(rem / (len(inits) - i) - 20, 280.0)
        if t < 30:
            break
        ao, _ = run_opt(init / init.sum(), n, fsz, time.monotonic() + t, 96, 0.06, 0.85)
        sc = float(evaluate_sequence((ao * S0).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = ao.copy()

    # Fine polishing with small perturbations
    if time.monotonic() < dl - 4:
        w = max(5, n // 60)
        while time.monotonic() < dl - 2:
            c = best_a.copy()
            for _ in range(int(rng.integers(1, 5))):
                st = int(rng.integers(0, max(1, n - w)))
                c[st:st+w] *= float(np.exp(rng.normal(0.0, 0.005)))
            c = np.maximum(c, 0.0)
            if c.sum() < 1e-6:
                continue
            c /= c.sum()
            sc = float(evaluate_sequence((c * S0).tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = c

    out = np.clip(best_a * S0, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(v) for v in out]
