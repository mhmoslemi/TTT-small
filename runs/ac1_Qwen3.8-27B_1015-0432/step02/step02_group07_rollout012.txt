import numpy as np
import time


def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def smooth_grad(a, conv, p, fs):
    n = a.shape[0]
    w = conv[:2 * n - 1] ** (p - 1)
    wr = np.zeros(fs)
    wr[:2 * n - 1] = w[::-1]
    A = np.zeros(fs)
    A[:n] = a
    c = np.fft.irfft(np.fft.rfft(A) * np.fft.rfft(wr), n=fs)
    return 2.0 * p * c[n - 1:2 * n - 1][::-1]


def subgrad_topk(a, conv, tk):
    n = a.shape[0]
    idx = np.argpartition(conv[:2 * n - 1], -tk)[-tk:]
    g = np.zeros(n)
    for ii in range(tk):
        k = int(idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g / float(tk)

    rows_list = []
    cols_list = []
    vals_list = []

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cnt = hi - lo + 1
        rows_list.append(np.full(cnt, k, dtype=np.int32))
        cols_list.append(np.arange(lo, hi + 1, dtype=np.int32))
        vals_list.append(an[lo:hi + 1])

        return a, np.inf

            t = 0.25

            an = an / an.sum()
            return an, 0.0
    return a, np.inf
