def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def compute_grad(a, cv, n, max_act, alpha):
    m = cv.max()
    if m <= 0:
        return np.zeros(n)
    thr = (1.0 - alpha) * m
    idx = np.where(cv >= thr)[0]
    if len(idx) > max_act:
        idx = np.argpartition(cv, -max_act)[-max_act:]
    g = np.zeros(n)
    for p in idx:
        p = int(p)
        lo = max(0, p - n + 1)
        hi = min(n - 1, p)
        if lo <= hi:
            g[lo:hi + 1] += a[p - hi:p - lo + 1][::-1]
    g /= max(1, len(idx))
    return g

def run_opt(a, n, fsz, dl, topk, alpha, eta0, decay, mom):
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    ba = a.copy()
    bm = fft_conv(a, fsz).max()
    gavg = np.zeros(n)
