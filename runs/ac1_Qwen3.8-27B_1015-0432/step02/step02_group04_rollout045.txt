        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        grad = make_smooth_grad(f, C, T)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        if it % 4 == 0:
            ns = fast_score(f)
            if ns < sc - 1e-10:
                eta = min(eta * 1.08, 5.0)
            else:
                eta = max(eta * 0.8, 0.02)
        if it % 80 == 0:
            T = max(0.1, T * 0.95)
        if no_imp > 200:
            f = best_f * np.exp(0.05 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.1
            T = 5.0
    return best_f, best_sc
