import numpy as np

from numpy.fft import fft, ifft

def _fpow2(s):
    m=1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n=a.size
    if n == 1:
        return np.array([a[0]*a[0]])
    L=_fpow2(2*n-1)
    fa=fft(a,L)
    return ifft(fa*fa).real[:2*n-1]

def _smax(c, tau):
    cm = float(c.max())
    if tau <= 0 or not np.isfinite(tau):
        p = np.zeros_like(c, dtype=np.float64)
        p[int(c.argmax())] = 1.0
        return cm, p
    d = (c - cm) / tau
    # d <=0
    np.clip(d, -745.0, 0.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _cross2(a, p):
    # returns 2 * sum_j a_j p[i+j]
    n = a.size
    pr = p[::-1]
    L = _fpow2(3*n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    return 2.0 * cv[n-1:2*n-1][::-1]

import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = float(c.max())
    if tau <= 0.0 or not np.isfinite(tau):
        p = np.zeros_like(c, dtype=np.float64)
        p[int(c.argmax())] = 1.0
        return cm, p
    d = (c - cm) / tau
    np.clip(d, -745.0, 0.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _cross2(a, p):
    n = a.size
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    return 2.0 * cv[n - 1:2 * n - 1][::-1]


def _iscore(a):
    n = a.size
    if n == 0:
        return 1e100
    S = a.sum()
    if S <= 1e-12:
        return 1e100
    c = _conv(a)
    return float(2.0 * n * c.max() / (S * S))


def _exact(a):
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return 1e100
    return float(evaluate_sequence(a.tolist()))


def _normalize(a, S):
    s = a.sum()
    if s <= 1e-12:
        return np.full(a.size, S / max(1, a.size), dtype=np.float64)
    return a / s * S


def _z_to_a(z, S):
    n = z.size + 1
    if n == 1:
        return np.array([S], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.clip(np.nan_to_num(z, nan=0.0, posinf=40.0, neginf=-40.0), -40.0, 40.0)
    m = float(np.max(z)) if z.size else 0.0
    if m < 0.0:
        m = 0.0
    ex = np.empty(n, dtype=np.float64)
    ex[:-1] = np.exp(z - m)
    ex[-1] = np.exp(-m)
    tot = ex.sum()
    if tot <= 0.0 or not np.isfinite(tot):
        ex = np.ones(n, dtype=np.float64)
        tot = float(n)
    return S * ex / tot


def _a_to_z(a, S):
    n = a.size
    if n <= 1:
        return np.empty(0, dtype=np.float64)
    small = 1e-300
    arr = np.maximum(a, small)
    base = max(float(arr[-1]), small)
    z = np.log(arr[:-1] / base)
    return np.clip(z, -40.0, 40.0)


def _z_obj(z, tau, S):
    a = _z_to_a(z, S)
    n = a.size
    c = _conv(a)
    mv, _ = _smax(c, tau)
    return float(2.0 * n * mv / (S * S))


def _z_grad(z, tau, S):
    a = _z_to_a(z, S)
    n = a.size
    c = _conv(a)
    mv, p = _smax(c, tau)
    cross = _cross2(a, p)
    S2 = S * S
    S3 = S2 * S
    d = (2.0 * n / S2) * cross - (4.0 * n * mv / S3)
    t = (d @ a) / S
    g = S * d[:-1] - a[:-1] * t
    return g


def _run_z(z, tau, S, bounds, iters):
    res = minimize(
        _z_obj, z, args=(tau, S), jac=_z_grad,
        method='L-BFGS-B', bounds=bounds,
        options={'maxiter': iters, 'ftol': 1e-20, 'gtol': 1e-5,
                 'maxls': 20, 'm': 30, 'disp': False}
    )
    return res.x


def _active_grad(a, S, tol):
    n = a.size
    c = _conv(a)
    mv = float(c.max())
    if not np.isfinite(mv) or mv <= 0.0:
        return None
    if tol <= 0.0:
        tol = 0.001 * max(1.0, mv)
    mask = c >= mv - tol
    if not np.any(mask):
        mask = np.zeros_like(c, dtype=np.float64)
        mask[int(c.argmax())] = 1.0
        mv = float(mask.max())
    p = mask.astype(np.float64)
    p /= p.sum()
    cross = _cross2(a, p)
    S2 = S * S
    S3 = S2 * S
    return (2.0 * n / S2) * cross - (4.0 * n * mv / S3)


def _make_inits(par, n, S):
    x = np.arange(n, dtype=np.float64)
    out = []

    r = np.clip(par, 0.0, 1000.0)
    if r.ndim == 1 and r.size == n and r.sum() > 0.01:
        out.append(r / r.sum() * S)
        sym = 0.5 * (r + r[::-1])
        if sym.sum() > 0.01:
            out.append(sym / sym.sum() * S)
        for k in (5, 25):
            if n > 2:
                sm = np.convolve(r, np.ones(k) / k, mode='same').clip(0.0, 1000.0)
                if sm.sum() > 0.01:
                    out.append(sm / sm.sum() * S)

    out.append(np.full(n, S / n, dtype=np.float64))

    for sig in (0.1, 0.25):
        e1 = np.exp(-x / (sig * n))
        out.append(e1 / e1.sum() * S)
        e2 = np.exp(-((n - 1 - x) / (sig * n)))
        out.append(e2 / e2.sum() * S)
        for ctr in (0.30, 0.50, 0.70):
            g = np.exp(-0.5 * ((x - ctr * n) / (sig * n)) ** 2)
            out.append(g / g.sum() * S)

    for freq, amp in ((1, 0.5), (2, 0.7), (3, 0.9)):
        c = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() > 0.01:
            out.append(c / c.sum() * S)

    bw = max(2, n // 2)
    b = np.zeros(n, dtype=np.float64)
    b[(n - bw) // 2:(n + bw) // 2 + 1] = 1.0
    out.append(b / b.sum() * S)

    rng = np.random.default_rng(7)
    klp = max(3, n // 100)
    ker = np.ones(klp) / klp
    for _ in range(3):
        rr = rng.random(n)
        sm = np.convolve(rr, ker, mode='same').clip(0.0, 1000.0)
        if sm.sum() > 0.01:
            out.append(sm / sm.sum() * S)

    g2 = (
        np.exp(-0.5 * ((x - 0.33 * n) / (0.1 * n)) ** 2) +
        np.exp(-0.5 * ((x - 0.67 * n) / (0.1 * n)) ** 2)
    )
    out.append(g2 / g2.sum() * S)

    if r.ndim == 1 and r.size == n and r.sum() > 0.01:
        pn = r * np.exp(rng.normal(0.0, 0.10, size=n))
        pn = np.clip(pn, 0.0, 1000.0)
        if pn.sum() > 0.01:
            out.append(pn / pn.sum() * S)

    good = []
    for a in out:
        a = np.asarray(a, dtype=np.float64)
        if a.ndim != 1 or a.size != n:
            continue
        if not np.all(np.isfinite(a)):
            continue
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            continue
        good.append(a / s * S)
    return good


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 25.0)

    try:
        raw = height_sequence_1
    except NameError:
        raw = np.ones(256, dtype=np.float64)

    par = np.asarray(raw, dtype=np.float64)
    if par.ndim != 1 or par.size < 1 or not np.all(np.isfinite(par)):
        par = np.ones(256, dtype=np.float64)
    par = np.clip(par, 0.0, 1000.0)
    n = par.size
    S = float(np.sqrt(2.0 * n))

    inits = _make_inits(par, n, S)
    best = None
    best_exact = 1e100
    best_fft = 1e100

    for a in inits:
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            continue
        a = _normalize(a, S)
        isc = _iscore(a)
        if not np.isfinite(isc):
            continue
        ex = _exact(a)
        if np.isfinite(ex) and ex < best_exact:
            best_exact = ex
            best_fft = isc
            best = a.copy()

    if best is None:
        best = np.full(n, S / max(1, n), dtype=np.float64)
        best_fft = _iscore(best)
        best_exact = _exact(best)

    bounds_z = [(-40.0, 40.0)] * max(0, n - 1)

    if n > 1 and time.monotonic() < dl - 260.0:
        z = _a_to_z(best, S)
        taus = [
            2.0, 1.0, 0.5, 0.25,
            0.12, 0.06, 0.03, 0.016,
            0.008, 0.004, 0.002, 0.001,
            5e-4, 2e-4
        ]
        for tau in taus:
            if time.monotonic() > dl - 260.0:
                break
            if tau < 0.005:
                iters = 500
            elif tau < 0.1:
                iters = 350
            else:
                iters = 250
            try:
                znew = _run_z(z, tau, S, bounds_z, iters)
            except Exception:
                break
            znew = np.asarray(znew, dtype=np.float64)
            if znew.size != n - 1 or not np.all(np.isfinite(znew)):
                break
            a = _z_to_a(znew, S)
            a = np.clip(a, 0.0, 1000.0)
            a = _normalize(a, S)
            isc = _iscore(a)
            if not np.isfinite(isc):
                break
            ex = _exact(a)
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                best_fft = isc
                best = a.copy()
            z = _a_to_z(a, S)

        if time.monotonic() < dl - 300.0:
            z = _a_to_z(best, S)
            z = z + rng.normal(0.0, 0.05, size=n - 1)
            z = np.clip(z, -40.0, 40.0)
            for tau in [1.0, 0.2, 0.05, 0.01, 0.003, 0.001, 3e-4]:
                if time.monotonic() > dl - 300.0:
                    break
                try:
                    znew = _run_z(z, tau, S, bounds_z, 350)
                except Exception:
                    break
                znew = np.asarray(znew, dtype=np.float64)
                if znew.size != n - 1 or not np.all(np.isfinite(znew)):
                    break
                a = _z_to_a(znew, S)
                a = np.clip(a, 0.0, 1000.0)
                a = _normalize(a, S)
                isc = _iscore(a)
                if not np.isfinite(isc):
                    break
                ex = _exact(a)
                if np.isfinite(ex) and ex < best_exact:
                    best_exact = ex
                    best_fft = isc
                    best = a.copy()
                z = _a_to_z(a, S)

    loc = best.copy()
    loc_score = _iscore(loc)
    best_fft = _iscore(best)
    max_block = max(2, n // 40)
    last_exact = 0.0
    exact_calls = 0
    moves = 0

    while time.monotonic() < dl - 8.0:
        moves += 1
        cand = loc.copy()
        u = rng.random()

        if u < 0.70:
            width = int(rng.integers(1, max_block + 1))
            start = int(rng.integers(0, n - width + 1))
            sigma = 0.03 + 0.15 * rng.random()
            cand[start:start + width] *= float(np.exp(rng.normal(0.0, sigma)))
        elif u < 0.85:
            idx = int(rng.integers(0, n))
            cand[idx] *= float(np.exp(rng.normal(0.0, 0.30)))
        elif u < 0.95:
            idx = int(rng.integers(0, n))
            full = S / max(1, n)
            cand[idx] = max(0.0, cand[idx] + float(rng.normal(0.0, 0.20 * full)))
        else:
            dgrad = _active_grad(loc, S, 0.0)
            if dgrad is not None:
                step = 0.02 + 1.0 * rng.random()
                cand = loc - step * dgrad

        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s <= 1e-12:
            continue
        cand = cand / s * S
        isc = _iscore(cand)
        if not np.isfinite(isc):
            continue

        improved = False
        if isc < loc_score - 1e-12:
            loc = cand
            loc_score = isc
            improved = True

        if improved:
            now = time.monotonic()
            if (
                exact_calls < 120 and
                (
                    (isc < best_fft - 1e-6 and now - last_exact > 2.0) or
                    (isc < best_fft - 1e-8 and now - last_exact > 0.7) or
                    (moves % 2000 == 0 and now - last_exact > 5.0)
                )
            ):
                ex = _exact(cand)
                last_exact = now
                exact_calls += 1
                if np.isfinite(ex) and ex < best_exact:
                    best_exact = ex
                    best_fft = isc
                    best = cand.copy()
                    loc = best.copy()
                    loc_score = best_fft

        if exact_calls < 120 and (moves % 1000 == 0 or time.monotonic() - last_exact > 25.0):
            ex = _exact(loc)
            last_exact = time.monotonic()
            exact_calls += 1
            if np.isfinite(ex) and ex < best_exact:
                best_exact = ex
                best_fft = _iscore(loc)
                best = loc.copy()
                loc_score = best_fft

        if rng.random() < 0.02:
            loc = best.copy()
            loc_score = best_fft

    ex = _exact(loc)
    if np.isfinite(ex) and ex < best_exact:
        best_exact = ex
        best = loc.copy()

    out = np.clip(best, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.full(n, 1.0, dtype=np.float64)
    return [float(v) for v in out]
