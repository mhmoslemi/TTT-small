def fft_conv(a, b):
    n, m = len(a), len(b)

    return 2.0 * f.size * b.max()

        hi = min(k, n - 1)

import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()


def hard_grad(f, b, ntop):
    n = f.size
    idx = np.argsort(b, kind="stable")[-ntop:]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo > hi:
            continue
        g[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]
    return g


def soft_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    wsum = w.sum()
    if wsum < 1e-30:
        return None
    fr = f[::-1]
    gc = fft_conv(w, fr)
    g = (gc[f.size - 1:2 * f.size - 1]).copy()
    return np.maximum(g, 0.0)

def make_ramp(n, pw, rev):
    x = np.arange(1.0, n + 1.0)
    r = x ** pw
    if rev:
        r = r[::-1]
    return r / r.sum()


def make_arc(n, c, w, sp):
    u = np.linspace(0.0, 1.0, n)
    v = (u - c) / w
    a = np.maximum(0.0, 1.0 - v * v) ** sp
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    u = np.linspace(0.0, 1.0, n)
    a = np.exp(-((u - c) / w) ** 2)
    return a / a.sum()
