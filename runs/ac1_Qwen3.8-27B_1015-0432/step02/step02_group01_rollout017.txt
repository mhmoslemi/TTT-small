def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def gradient_active(f, b, top_k):
    """Gradient using top-k active lags."""
    n = f.size
    k = min(top_k, b.size)
    idx = np.argsort(b)[-k:]
    w = np.zeros_like(b)
    w[idx] = 1.0
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n-1:2*n-1].copy()
    return np.maximum(grad, 0.0)

    """Gradient using soft top-lag weighting."""
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))

    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n-1:2*n-1].copy()
    return np.maximum(grad, 0.0)

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def sym(f):
    return (f + f[::-1]) * 0.5

def grad_dir(f, b, beta, n):
    M = b.max()
    if M < 1e-30:
        return None
    arg = beta * (b - M) / M
    w = np.exp(arg)
    ws = w.sum()

        k = min(200, b.size)
        w = np.zeros(b.size)
        w[np.argpartition(b, -k)[-k:]] = 1.0

    gc = fft_conv(w, f_r)
