import numpy as np
import time
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a, L):
    n = a.size
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _score_fft(a, L):
    c = _conv(a, L)
    return 2.0 * a.size * c.max()


def _grad_sm(a, tau, L2):
    n = a.size
    fa = fft(a, L2)
    c = ifft(fa * fa).real[:2 * n - 1]
    cm = c.max()
    d = np.maximum((c - cm) / tau, -40.0)
    e = np.exp(d)
    s = e.sum()
    mv = cm + tau * np.log(s)
    p = e / s
    fp = fft(p, L2)
    corr = ifft(np.conj(fa) * fp).real[:n]
    return 4.0 * n * corr, mv


def _proj(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.full(a.shape, 1.0 / a.size)
    return a / s
