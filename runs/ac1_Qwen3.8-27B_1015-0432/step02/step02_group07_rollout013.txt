import numpy as np
import time
from scipy import sparse, optimize


def fft_conv(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def build_conv_matrix(a):
    n = a.size
    cidx = np.repeat(np.arange(n, dtype=np.int64), n)
    ridx = np.tile(np.arange(n, dtype=np.int64), n) + cidx
    vals = np.repeat(a, n)
    return sparse.csr_matrix((vals, (ridx, cidx)), shape=(2 * n - 1, n))


def lp_direction(a, M, tlimit=20):
    n = a.size
    if n > 12000:
        return None
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M * 1.000001)
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n,
                           method='highs', options={'time_limit': tlimit})
    if res.status == 0:
        return np.clip(res.x, 0.0, None)
    return None


def subgrad_direction(a, conv, n, tk=24):
    M = float(conv.max())
    tol = M * 1e-8
    active = np.where(conv >= M - tol)[0]
    if len(active) > tk:
        active = np.argpartition(conv, -tk)[-tk:]
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    if len(active) > 0:
        grad /= float(len(active))
    else:
        grad[:] = 1.0 / n
    return grad


def try_step(a, candidate, n, fft_size, M):

    if s < 1e-15:

    Mc = float(conv_c.max())
