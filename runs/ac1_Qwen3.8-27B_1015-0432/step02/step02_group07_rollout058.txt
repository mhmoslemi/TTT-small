def _rfft_conv(a, b, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    fb = np.fft.rfft(b, n=fft_size)
    return np.fft.irfft(fa * fb, n=fft_size)


def _self_conv(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def _smooth_grad(a, conv, T, fft_size_g):
    n = a.size
    cs = conv - conv.max()
    w = np.exp(T * cs)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws
    a_rev = a[::-1]
    g_full = _rfft_conv(a_rev, w, fft_size_g)
    return g_full[n - 1:2 * n - 1].copy()


def _lp_proj(a, M, fft_size, iters=10):
    n = a.size
    b = np.full(n, 1.0 / n)
    eta = 0.3 / n
    a_rev = a[::-1]

        viol = np.maximum(0.0, ab - M)

        b = np.maximum(b, 0.0)
        bs = b.sum()
        if bs < 1e-300:

        b /= bs
