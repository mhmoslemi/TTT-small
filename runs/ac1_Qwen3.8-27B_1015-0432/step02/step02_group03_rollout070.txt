def _next_fft_len(n):
    need = max(1, 3 * int(n) - 2)
    return 1 << max(0, (need - 1).bit_length())

def _full_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[: 2 * a.size - 1]

def _fast_score(a, fs):
    n = a.size
    if n == 0:
       return np.inf
    s = float(np.sum(a))
    if s <= 0.0:
       return np.inf
    c = _full_conv(a, fs)
    return float(2.0 * n * np.max(c) / (s * s))

def _smooth_grad(a, c, tau, fs):
    n = a.size
    c = c[:2*n-1]
    cmax = float(np.max(c))
    if not np.isfinite(cmax): return np.zeros(n)
    tau = float(tau)
    floor = 1e-10 * max(1.0, abs(cmax))
    if tau < floor: tau = floor
    cm = c - cmax
    ec = np.exp(cm / tau)
    z = float(np.sum(ec))
    if not np.isfinite(z) or z <= 0.0:
        p = np.ones_like(c) / c.size
    else:
        p = ec / z
    fp = np.fft.rfft(p, n=fs)
    fa = np.fft.rfft(a, n=fs)
    xc = np.fft.irfft(fp * np.conj(fa), n=fs)
    g = 2.0 * xc[:n]
    g -= float(np.mean(g))
    return g

def _quad_grad(a, c, fs):
    n = a.size
    cc = c[:2*n-1]
    fc = np.fft.rfft(cc, n=fs)
    fa = np.fft.rfft(a, n=fs)
    xc = np.fft.irfft(fc * np.conj(fa), n=fs)
    g = 4.0 * xc[:n]
    g -= float(np.mean(g))
    return g

def _active_grad(a, c, frac, cap):
    n = a.size
    m = 2 * n - 1
    c = c[:m]
    g = np.zeros(n)
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx <= 0.0:
       g -= 0
       return g
    thr = mx * (1.0 - frac)
    act = np.where(c >= thr)[0]
    if act.size > cap:
        part = np.argpartition(c[act], -cap)[-cap:]
        act = act[part]
    if act.size == 0:
        act = np.array([int(np.argmax(c))], dtype=np.int64)
    for k in act:
        k = int(k)
        lo = k - (n - 1)
        if lo < 0: lo = 0
        hi = k
        if hi >= n: hi = n - 1
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= float(act.size)
    g -= float(np.mean(g))
    return g

    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[:2 * a.size - 1]

def _fast_score(a, fs):
    n = a.size
    if n <= 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 0.0:
        return np.inf
    c = _full_conv(a, fs)
    mx = float(np.max(c))
    sc = 2.0 * n * mx / (s * s)
    return float(sc) if np.isfinite(sc) else np.inf

def _project(a):
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return None
    return a / s

def _tau_of(a, fs, factor):
    c = _full_conv(a, fs)
    cmx = float(np.max(c))
    cmin = float(np.min(c))
    sp = max(cmx - cmin, cmx * 1e-3, 1e-12)
    return float(factor) * sp

def _smooth_grad(a, c, tau, fs):
    n = a.size
    m = 2 * n - 1
    c = c[:m]
    cmx = float(np.max(c))
    tau = max(1e-15, float(tau))
    w = np.exp((c - cmx) / tau)
    z = float(np.sum(w))
    if z <= 0.0 or not np.isfinite(z):
        p = np.ones(m, dtype=np.float64) / m
    else:
        p = w / z
    fp = np.fft.rfft(p, n=fs)
    fa = np.fft.rfft(a, n=fs)
    xc = np.fft.irfft(fp * np.conj(fa), n=fs)
    g = 2.0 * xc[:n]
    g -= float(np.mean(g))
    return g

def _quad_grad(a, c, fs):
    n = a.size
    m = 2 * n - 1
    cc = c[:m]
    fc = np.fft.rfft(cc, n=fs)
    fa = np.fft.rfft(a, n=fs)
    xc = np.fft.irfft(fc * np.conj(fa), n=fs)
    g = 4.0 * xc[:n]
    g -= float(np.mean(g))
    return g

def _active_grad(a, c, frac, cap):
    n = a.size
    m = 2 * n - 1
    cc = c[:m]
    g = np.zeros(n, dtype=np.float64)
    mx = float(np.max(cc))
    if not np.isfinite(mx) or mx <= 0.0:
        return g
    act = np.where(cc >= mx * (1.0 - frac))[0]
    if act.size > cap:
        act = act[np.argpartition(cc[act], -cap)[-cap:]]
    if act.size == 0:
        act = np.array([int(np.argmax(cc))])
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= float(act.size)
    g -= float(np.mean(g))
    return g

def _exact_score(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0)
    if a.size == 0:
        return np.inf
    try:
        sc = evaluate_sequence([float(x) for x in a])
    except Exception:
        return np.inf
    return float(sc) if np.isfinite(sc) else np.inf

def _active_euclidean(a0, fs, t_limit, eta0, frac, cap, beta, max_stall, rng):
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    pr = _project(a)
    if pr is None:
        return np.ones(64, dtype=np.float64), np.inf
    n = a.size
    a = pr
    best = a.copy()
    best_sc = _fast_score(a, fs)
    mom = np.zeros(n, dtype=np.float64)
    eta = float(eta0)
    stall = 0
    prev = best_sc
    while time.monotonic() < t_limit:
        c = _full_conv(a, fs)
        s = float(np.sum(a))
        if s <= 0.0:
            break
        sc = 2.0 * n * float(np.max(c)) / (s * s)
        if not np.isfinite(sc):
            break
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        if sc > prev + 1e-12:
            stall += 1
        prev = sc
        if stall > max_stall:
            break
        g = _active_grad(a, c, frac, cap)
        gm = float(np.max(np.abs(g)))
        if gm <= 1e-16:
            if rng is not None and stall > max(50, max_stall // 8):
                g = rng.standard_normal(n)
                g -= float(np.mean(g))
                gm = float(np.max(np.abs(g)))
            if gm <= 1e-16:
                break
        d = g / gm
        d -= float(np.mean(d))
        mom = beta * mom + (1.0 - beta) * d
        mom -= float(np.mean(mom))
        ma = max(1e-15, s / float(n))
        a = a - eta * ma * mom
        pr = _project(a)
        if pr is None:
            break
        a = pr
        eta *= 0.999985
    return best, best_sc

def _active_multiplicative(a0, fs, t_limit, eta0, frac, cap, max_stall, rng):
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    pr = _project(a)
    if pr is None:
        return np.ones(64, dtype=np.float64), np.inf

    eta = float(eta0)

        c = _full_conv(a, fs)
        s = float(np.sum(a))
        if s <= 0.0:
            break
        sc = 2.0 * n * float(np.max(c)) / (s * s)
        if not np.isfinite(sc):
            break
        if sc < best_sc - 1e-13:

            stall = 0
        else:

        prev = sc

        g = _active_grad(a, c, frac, cap)
        gm = float(np.max(np.abs(g)))
        if gm <= 1e-16:

                g = rng.standard_normal(n)
                g -= float(np.mean(g))
                gm = float(np.max(np.abs(g)))
            if gm <= 1e-16:
                break
        a = a * np.exp(-eta * (g / gm))
        pr = _project(a)
        if pr is None:

        a = pr
        eta *= 0.99995
    return best, best_sc

def _combined_descent(a0, fs, t_limit, eta0, tau0, beta, max_stall, smooth_w, quad_w, rng):
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    pr = _project(a)
    if pr is None:
        return np.ones(64, dtype=np.float64), np.inf
    n = a.size
    a = pr
    best = a.copy()
    best_sc = _fast_score(a, fs)
    mom = np.zeros(n, dtype=np.float64)
    eta = float(eta0)
    tau = float(tau0)
    stall = 0
    prev = best_sc
    while time.monotonic() < t_limit:
        c = _full_conv(a, fs)
        s = float(np.sum(a))
        if s <= 0.0:
            break
        sc = 2.0 * n * float(np.max(c)) / (s * s)
        if not np.isfinite(sc):
            break
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        if sc > prev + 1e-12:
            stall += 1
        prev = sc
        if sc > best_sc + 0.01:
            if rng is not None:
                q = _project(best * (1.0 + 0.02 * rng.standard_normal(n)))
                if q is None:
                    break
                a = q
                mom = np.zeros(n, dtype=np.float64)
                eta *= 0.5
                tau = max(tau * 2.0, 1e-15)
                prev = sc
                stall = 0
                continue
            break
        if stall > max_stall:
            break
        d = np.zeros(n, dtype=np.float64)
        used = False
        if smooth_w > 0.0:
            st = _smooth_grad(a, c, tau, fs)
            sm = float(np.max(np.abs(st)))
            if sm > 1e-16:
                d += float(smooth_w) * (st / sm)
                used = True
        if quad_w > 0.0:
            qd = _quad_grad(a, c, fs)
            qm = float(np.max(np.abs(qd)))
            if qm > 1e-16:
                d += float(quad_w) * (qd / qm)
                used = True
        if not used:
            if rng is not None and stall > max(50, max_stall // 8):
                d = rng.standard_normal(n)
                used = True
            else:
                break
        dm = float(np.max(np.abs(d)))
        if dm <= 1e-16:
            break
        d -= float(np.mean(d))
        dm = float(np.max(np.abs(d)))
        if dm <= 1e-16:
            break
        d /= dm
        mom = beta * mom + (1.0 - beta) * d
        mom -= float(np.mean(mom))
        ma = max(1e-15, s / float(n))
        a = a - eta * ma * mom
        pr = _project(a)
        if pr is None:
            break
        a = pr
        cmx = float(np.max(c))
        cmin = float(np.min(c))
        sp = max(cmx - cmin, cmx * 1e-6, 1e-14)
        floor = max(1e-14, 0.005 * sp, 1e-9 * cmx)
        tau = max(tau * 0.99999, floor)
        eta *= 0.999985
    return best, best_sc

def _local_search_fast(base, fs, t_limit, rng):
    b = np.maximum(np.asarray(base, dtype=np.float64), 0.0).copy()
    pr = _project(b)
    if pr is None:
        pr = _project(np.ones(64, dtype=np.float64))
    best = pr
    best_sc = _fast_score(best, fs)
    n = best.size
    w = max(2, n // 120)
    sig = 0.025
    stuck = 0
    while time.monotonic() < t_limit:
        cand = best.copy()
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= float(np.exp(rng.normal(0.0, sig)))
        pr = _project(cand)
        if pr is None:
            continue
        sc = _fast_score(pr, fs)
        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best = pr
            best_sc = sc
            stuck = 0
            sig = max(0.004, sig * 0.97)
        else:
            stuck += 1
            if stuck > 150:
                sig = min(0.12, sig * 1.3)
                stuck = 0
    return best, best_sc

def _local_search_exact(base, t_limit, rng):
    b = np.clip(np.asarray(base, dtype=np.float64), 0.0, 1000.0).copy()
    pr = _project(b)
    if pr is None:
        pr = _project(np.ones(64, dtype=np.float64))
    best = pr

    n = best.size

    stuck = 0

            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= float(np.exp(rng.normal(0.0, sig)))
        pr = _project(cand)
        if pr is None:
            continue
        sc = _exact_score(pr)
        if np.isfinite(sc) and sc < best_sc - 1e-14:

            best_sc = sc
            stuck = 0
            sig = max(0.004, sig * 0.97)
        else:
            stuck += 1
            if stuck > 150:
                sig = min(0.12, sig * 1.3)
                stuck = 0
    return best, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    now = time.monotonic()
    t_end = now + max(10.0, float(budget_s) - 35.0)
    t_hard = now + max(25.0, float(budget_s) - 8.0)

    try:
        arr = height_sequence_1
    except NameError:
        arr = np.ones(512, dtype=np.float64)

    p = np.asarray(arr, dtype=np.float64).ravel()
    p = np.clip(p, 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or float(np.sum(p)) < 1e-8:
        p = np.ones(512, dtype=np.float64)
    p = _project(p)
    if p is None:
        p = _project(np.ones(512, dtype=np.float64))
    n = p.size
    fs = _next_fft_len(n)

    fast_best = p.copy()
    fast_sc = _fast_score(fast_best, fs)
    exact_ok = n <= 12000
    exact_best = p.copy()
    exact_sc = _exact_score(p) if exact_ok else np.inf
    if not np.isfinite(exact_sc):
        exact_sc = fast_sc

    c0 = _full_conv(fast_best, fs)
    sp0 = max(float(np.max(c0) - np.min(c0)), float(np.max(c0)) * 1e-3, 1e-12)
    span = max(1.0, t_end - now)

    e1 = now + 0.14 * span
    if time.monotonic() < e1:
        cand, sc = _active_euclidean(fast_best, fs, e1, 0.25, 5e-3, 80, 0.85, 2500, rng)
        if np.isfinite(sc) and sc < fast_sc:
            fast_sc = sc
            fast_best = cand
        if exact_ok and (t_end - time.monotonic()) > 10.0:
            sx = _exact_score(fast_best)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = fast_best.copy()

    e2 = now + 0.42 * span
    if time.monotonic() < e2:
        cand, sc = _combined_descent(fast_best, fs, e2, 0.18, 0.15 * sp0, 0.85, 6000, 1.0, 1.0, rng)
        if np.isfinite(sc) and sc < fast_sc:
            fast_sc = sc
            fast_best = cand
        if exact_ok and (t_end - time.monotonic()) > 10.0:
            sx = _exact_score(fast_best)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = fast_best.copy()

    e3 = now + 0.62 * span
    if time.monotonic() < e3:
        cand, sc = _active_multiplicative(fast_best, fs, e3, 0.22, 1e-3, 50, 3000, rng)
        if np.isfinite(sc) and sc < fast_sc:
            fast_sc = sc
            fast_best = cand
        if exact_ok and (t_end - time.monotonic()) > 10.0:
            sx = _exact_score(fast_best)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = fast_best.copy()

    e4 = now + 0.80 * span
    if time.monotonic() < e4:
        tau = _tau_of(fast_best, fs, 0.03)
        cand, sc = _combined_descent(fast_best, fs, e4, 0.10, tau, 0.90, 4000, 1.0, 1.25, rng)
        if np.isfinite(sc) and sc < fast_sc:
            fast_sc = sc
            fast_best = cand
        if exact_ok and (t_end - time.monotonic()) > 10.0:
            sx = _exact_score(fast_best)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = fast_best.copy()

    e5 = now + 0.92 * span
    if time.monotonic() < e5:
        cand, sc = _active_euclidean(fast_best, fs, e5, 0.08, 2e-4, 28, 0.88, 2500, rng)
        if np.isfinite(sc) and sc < fast_sc:
            fast_sc = sc
            fast_best = cand
        if exact_ok and (t_end - time.monotonic()) > 10.0:
            sx = _exact_score(fast_best)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = fast_best.copy()

    local_end = min(t_end, t_hard - 12.0)
    if exact_ok:
        if time.monotonic() < local_end:
            base = exact_best if np.isfinite(exact_sc) else fast_best
            pr = _project(base)
            if pr is None:
                pr = _project(fast_best)
            cand, sx = _local_search_exact(pr, local_end, rng)
            if np.isfinite(sx) and sx < exact_sc:
                exact_sc = sx
                exact_best = cand
        out = exact_best if np.isfinite(exact_sc) else fast_best
    else:

        out = fast_best

    out = _project(out)
    if out is None:

    if exact_ok and time.monotonic() < t_hard - 3.0:
        sc_out = _exact_score(out)
        sc_par = _exact_score(p)
        if not np.isfinite(sc_out) or sc_out > sc_par + 1e-12:
            out = p

    out = np.asarray(out, dtype=np.float64)
    if out.size < 1 or not np.all(np.isfinite(out)) or float(np.sum(out)) < 0.01:
        out = p

    return [float(x) for x in out]
