        if gm < 1e-30:
            break
        f *= np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-25)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 40 == 39:
            sc = cs(f)
            if sc < bs - 1e-12:
                bs = sc
                bf = f.copy()
                stall = 0
            else:
                stall += 1
            if stall == 20:
                eta *= 0.7
            elif stall >= 80:
                f = bf * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                eta = 0.05
                stall = 0
            if it > 3000:
                eta = max(0.003, eta * 0.998)
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)
    p = np.clip(height_sequence_1, 0.0, 1000.0).astype(np.float64)

    # Diverse initializations
    inits = []
    if p.sum() > 1e-10:
        pn = p / p.sum()
        inits.append(pn.copy())
        inits.append(0.5 * (pn + pn[::-1]))
    inits.append(np.ones(n) / n)
    tri = np.minimum(x + 1.0, n - x)
    inits.append(tri / tri.sum())
    for a in [0.4, 0.7, 1.0, 1.5, 2.0]:
        c = np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * x / n)), 1e-12) ** a
        c = 0.5 * (c + c[::-1])
        c = np.maximum(c, 1e-15)
        inits.append(c / c.sum())
    for _ in range(3):
        r = np.maximum(1.0 + 0.3 * rng.standard_normal(n), 0.01)
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    bf, bs = None, 1e18
    for ins in inits:
        sc = cs(ins)
        if sc < bs:
            bs = sc
            bf = ins.copy()

    # Phase 1: symmetric optimization
    p1 = min(dl - 200.0, t0 + 500.0)
    bf, bs = opt(bf.copy(), p1, rng, True)

    # Phase 2: asymmetric refinement
    if time.monotonic() < dl - 80.0:
        fr, sr = opt(bf.copy(), min(dl - 40.0, t0 + 900.0), rng, False)
        if sr < bs:
            bf, bs = fr, sr

    # Phase 3: re-symmetrize and polish
    if time.monotonic() < dl - 40.0:
        sp = 0.5 * (bf + bf[::-1])
        sp = np.maximum(sp, 1e-25)
        sp /= sp.sum()
        fr, sr = opt(sp, min(dl - 15.0, t0 + 1030.0), rng, True)
        if sr < bs:
            bf, bs = fr, sr

    # Phase 4: fine perturbations (symmetric and asymmetric)
    for _ in range(60):
        if time.monotonic() > dl - 2.0:
            break
        for s2 in [0.003, 0.006, 0.012]:
            tr = bf * np.exp(s2 * rng.standard_normal(n))
            tr = np.maximum(tr, 0.0)
            tr = 0.5 * (tr + tr[::-1])
            tr /= tr.sum()
            v = cs(tr)
            if v < bs:
                bs, bf = v, tr.copy()
            ta = np.maximum(bf + s2 * rng.standard_normal(n), 0.0)
            ta /= ta.sum()
            v2 = cs(ta)
            if v2 < bs:
                bs, bf = v2, ta.copy()

    res = [float(v) for v in bf]
    if len(res) != n:
        res = [1.0 / n] * n
    for v in res:
        if not np.isfinite(v):
            res = [1.0 / n] * n
            break
    fv = evaluate_sequence(res)
    if not np.isfinite(fv):
        res = [1.0 / n] * n
    return res
