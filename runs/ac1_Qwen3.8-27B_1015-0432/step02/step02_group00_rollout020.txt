    inits.append(tri / tri.sum())
    hat = np.maximum(0.0, 1.0 - 2.0 * np.abs(x - ctr) / n)
    if hat.sum() > 1e-10:
        inits.append(hat / hat.sum())
    for _ in range(2):
        r = np.abs(rng.standard_normal(n)) + 0.1
        inits.append(r / r.sum())

    best_f = None
    best_sc = np.inf
    for idx in range(len(inits)):
        sc = do_score(inits[idx])
        if sc < best_sc:
            best_sc = sc
            best_f = inits[idx].copy()

    ph1_end = min(dl, t0 + 380.0)
    f1, s1 = md_opt(best_f.copy(), ph1_end, rng, tf=0.95, eta0=0.06, decay=0.998, check=20, rst=250)
    if s1 < best_sc:
        best_sc, best_f = s1, f1

    if time.monotonic() < dl - 200.0:
        ph2_end = min(dl, t0 + 680.0)
        f2, s2 = md_opt(best_f.copy(), ph2_end, rng, tf=0.98, eta0=0.03, decay=0.999, check=15, rst=350)
        if s2 < best_sc:
            best_sc, best_f = s2, f2

    if time.monotonic() < dl - 50.0:
        ph3_end = dl - 5.0
        f3, s3 = md_opt(best_f.copy(), ph3_end, rng, tf=0.995, eta0=0.015, decay=0.9995, check=10, rst=500)
        if s3 < best_sc:
            best_sc, best_f = s3, f3

    if time.monotonic() < dl - 3.0:
        for _ in range(40):
            if time.monotonic() > dl - 2.0:
                break
            t = best_f * np.exp(0.004 * rng.standard_normal(n))
            t = normalize(t)
            sc = do_score(t)
            if sc < best_sc:
                best_sc, best_f = sc, t

    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
