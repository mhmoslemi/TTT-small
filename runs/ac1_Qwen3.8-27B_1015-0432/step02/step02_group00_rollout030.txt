import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def normalize(f):
    f = np.maximum(f, 1e-20)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s


def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)


def build_conv_matrix(f):
    n = len(f)
    L = 2 * n - 1
    k_vals = np.arange(L, dtype=np.int64)
    j_starts = np.maximum(0, k_vals - (n - 1))
    j_ends = np.minimum(n - 1, k_vals) + 1
    counts = j_ends - j_starts
    total = int(counts.sum())
    rows = np.repeat(k_vals, counts)
    cols = np.empty(total, dtype=np.int64)
    pos = 0
    for i in range(L):
        c = int(counts[i])
        if c > 0:
            cols[pos:pos + c] = np.arange(j_starts[i], j_ends[i])
            pos += c
    vals = f[rows - cols]
    return sparse.csc_matrix((vals, (rows, cols)), shape=(L, n))

                  bounds=[(0, None)] * n,
