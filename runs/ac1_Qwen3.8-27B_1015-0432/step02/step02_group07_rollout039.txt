    """Compute weighted subgradient of max(conv) using soft-max approximation."""
    n = a.size
    M = np.max(conv)
    if delta is None:

    weights = np.maximum(conv - M + delta, 0) ** p
    total = weights.sum()
    if total < 1e-30:
        # Fall back to single max
        k = int(np.argmax(conv))
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)

        return np.zeros(n)

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def subgradient(a, conv, top_k):
    n = a.size
    idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros(n, dtype=np.float64)
    for ii in range(top_k):
        k = int(idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    return grad / float(top_k)


def soft_subgradient(a, conv, p, top_k):

    M = np.max(conv)
    delta = 0.05 * M
    weights = np.maximum(conv - M + delta, 0.0) ** p
