                no_imp += 1
            if no_imp > 1200:
                ssc = 0.006 + 0.02 * min(1.0, rst / 12.0)
                a = best_a * np.exp(ssc * rng.standard_normal(n))
                a = np.maximum(a, 1e-10)
                if sym:
                    a = 0.5 * (a + a[::-1])
                a /= a.sum()
                no_imp = 0
                rst += 1
                eta = min(0.3, eta * 1.3)
            eta *= 0.998
        if it > 3000000:
            break
    return best_a, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)
    inits = []
    p = np.clip(np.asarray(height_sequence_1, np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    rc = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(rc / rc.sum())
    irc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1))) + 1e-8
    inits.append(irc / irc.sum())
    c0 = n / 2.0
    g1 = np.exp(-0.5 * ((x - c0) ** 2) / (n / 5.5) ** 2)
    inits.append(g1 / g1.sum())
    d = np.minimum(x, n - 1 - x) + 1.0
    for pw in (0.5, 1.5):
        v = d ** pw
        inits.append(v / v.sum())
    tp = np.exp(-2.0 * ((x - 0.3 * n) / (n / 6.0)) ** 2) + np.exp(-2.0 * ((x - 0.7 * n) / (n / 6.0)) ** 2)
    inits.append(tp / tp.sum())
    rv = np.abs(rng.standard_normal(n)) + 0.15
    inits.append(rv / rv.sum())
    best_a, best_sc = inits[0].copy(), 1e30
    for ini in inits:
        sc = _score(ini)
        if sc < best_sc:
            best_sc, best_a = sc, ini.copy()
    e1 = min(dl, t0 + 350.0)
    a1, s1 = _improve(best_a.copy(), e1, rng, False)
    if s1 < best_sc:
        best_sc, best_a = s1, a1
    if time.monotonic() < dl - 80:
        as_ = 0.5 * (best_a + best_a[::-1])
        as_ /= as_.sum()
        e2 = min(dl, t0 + 650.0)
        a2, s2 = _improve(as_.copy(), e2, rng, True)
        if s2 < best_sc:
            best_sc, best_a = s2, a2
    if time.monotonic() < dl - 20:
        a3, s3 = _improve(best_a.copy(), dl - 5, rng, False)
        if s3 < best_sc:
            best_sc, best_a = s3, a3
    if time.monotonic() < dl - 3:
        for _ in range(40):
            if time.monotonic() > dl - 2:
                break
            e = 0.004 + 0.008 * rng.random()
            tr = best_a * np.exp(e * rng.standard_normal(n))
            tr = np.maximum(tr, 0.0)
            tr /= tr.sum()
            sc = _score(tr)
            if sc < best_sc:
                best_sc, best_a = sc, tr.copy()
    res = [float(v) for v in best_a]
    fs = evaluate_sequence(res)
    if not np.isfinite(fs):
        res = [1.0 / n] * n
        fs = evaluate_sequence(res)
    return res
