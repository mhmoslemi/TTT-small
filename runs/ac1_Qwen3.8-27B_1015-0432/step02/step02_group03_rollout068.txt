import numpy as np

def _conv(a):
    n = len(a)
    fs = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _subgrad(a, lags):
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    m = len(lags)
    if m > 0:
        g /= m
    return g

def _flatten_grad(a, c, n, target):
    L = 1 << (3 * n - 1).bit_length()
    C = np.fft.rfft(c[:2 * n - 1], n=L)
    A = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(np.conj(C) * A, n=L)[:n]
    S = a.sum()
    return 2.0 * (corr - target * S)
