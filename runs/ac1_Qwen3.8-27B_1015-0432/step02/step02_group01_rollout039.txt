                f = best_f * np.exp(cfg['pt'] * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                continue
        g = compute_grad(f, b, beta, kh)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
    return best_f, best_sc

def make_init(n, kind, rng):
    x = np.arange(n, dtype=np.float64)
    u = x / max(n - 1, 1)
    if kind == 0:
        return np.ones(n)
    if kind == 1:
        return np.maximum(1.0 - np.abs(2.0 * u - 1.0), 1e-10)
    if kind == 2:
        return np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * u)), 1e-10)
    if kind == 3:
        return np.exp(-4.0 * (2.0 * u - 1.0) ** 2)
    if kind == 4:
        return (x + 1.0) ** 1.5
    if kind == 5:
        return np.maximum(1.0 / (1.0 + 50.0 * (u - 0.35) ** 2), 1e-10)
    if kind == 6:
        return np.maximum(1.0 / (1.0 + 50.0 * (u - 0.65) ** 2), 1e-10)
    return np.maximum(rng.standard_gamma(2.0, size=n), 1e-10)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    span = deadline - t0

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)

    inits = [p, p[::-1].copy()]
    for k in range(8):
        inits.append(make_init(n, k, rng))

    best_f, best_sc = inits[0].copy(), np.inf
    for ini in inits:
        sc = score_of(ini)
        if sc < best_sc:
            best_sc, best_f = sc, ini.copy()

    # Phase 1: broad, soft weight
    d1 = t0 + span * 0.35
    c1 = {'iters': 50000, 'b0': 12.0, 'b1': 5.0, 'k0': 0, 'k1': 0,
          'e0': 0.12, 'e1': 0.04, 'sm': 8000, 'pt': 0.03}
    if time.monotonic() < d1:
        f, sc = descent(best_f, d1, rng, c1)
        if sc < best_sc:
            best_sc, best_f = sc, f

    # Mirror blends between phases
    if time.monotonic() < deadline - 90:
        flip = best_f[::-1].copy()
        for t in [0.1, 0.2, 0.3, 0.5]:
            bl = np.maximum((1.0 - t) * best_f + t * flip, 1e-20)
            bl /= bl.sum()
            s = score_of(bl)
            if s < best_sc:
                best_sc, best_f = s, bl

    # Phase 2: medium, sharper soft weight
    d2 = t0 + span * 0.60
    c2 = {'iters': 40000, 'b0': 50.0, 'b1': 25.0, 'k0': 0, 'k1': 0,
          'e0': 0.06, 'e1': 0.015, 'sm': 6000, 'pt': 0.02}
    if time.monotonic() < d2:
        f, sc = descent(best_f, d2, rng, c2)
        if sc < best_sc:
            best_sc, best_f = sc, f

    # Mirror blends again
    if time.monotonic() < deadline - 60:
        flip = best_f[::-1].copy()
        for t in [0.1, 0.2, 0.3]:
            bl = np.maximum((1.0 - t) * best_f + t * flip, 1e-20)
            bl /= bl.sum()
            s = score_of(bl)
            if s < best_sc:
                best_sc, best_f = s, bl

    # Phase 3: narrow, hard top-k subgradient
    d3 = t0 + span * 0.82
    c3 = {'iters': 40000, 'b0': 0.0, 'b1': 0.0,
          'k0': max(50.0, n / 25.0), 'k1': max(10.0, n / 300.0),
          'e0': 0.025, 'e1': 0.004, 'sm': 5000, 'pt': 0.015}
    if time.monotonic() < d3:
        f, sc = descent(best_f, d3, rng, c3)
        if sc < best_sc:
            best_sc, best_f = sc, f

    # Phase 4: fine polish with very few active lags
    c4 = {'iters': 80000, 'b0': 0.0, 'b1': 0.0,
          'k0': max(10.0, n / 500.0), 'k1': 5.0,
          'e0': 0.003, 'e1': 0.0005, 'sm': 4000, 'pt': 0.008}
    if time.monotonic() < deadline - 15:
        f, sc = descent(best_f, deadline - 15, rng, c4)
        if sc < best_sc:
            best_sc, best_f = sc, f

    # Final micro-perturbations
    if time.monotonic() < deadline - 5:
        for _ in range(15):
            if time.monotonic() > deadline - 5:
                break
            tr = best_f * np.exp(0.003 * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-20)
            tr /= tr.sum()
            s = score_of(tr)
            if s < best_sc:
                best_sc, best_f = s, tr

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
