import numpy as np
import time

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _true_score(a, fs):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = _fc(a, fs)
    return 2.0 * n * b.max() / (s * s)

def _grad_soft(a, w_rev, fs):
    """Gradient of soft-max score: g[i] = 2 * (a * w_rev)[2n-2-i]"""
    n = a.size
    fa = np.fft.rfft(a, n=fs)
    fw = np.fft.rfft(w_rev, n=fs)
    conv = np.fft.irfft(fa * fw, n=fs)
    idx = 2 * n - 2 - np.arange(n)
    return 2.0 * conv[idx]

def _soft_descent(a0, fs, S, dl, T0, Tmin, anneal, step0, decay, beta, rng):
    n = a0.size
    a = a0.copy()
    a *= S / max(a.sum(), 1e-15)
    best = a.copy()
    best_sc = _true_score(a, fs)
    T = T0
    step = step0
    mom = np.zeros(n)
    stall = 0
    prev_sc = best_sc
    
    for it in range(200000):
        if time.monotonic() > dl:
            break
        
        b = _fc(a, fs)
        sc = 2.0 * n * b.max() / (a.sum()**2)
        if sc < best_sc - 1e-15:
            best_sc = sc
            best = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            break
        
        # Soft-max weights
        b_pos = np.maximum(b, 0)
        Tb = T * b_pos
        Tb -= Tb.max()
        e = np.exp(Tb)
        w = e / e.sum()
        
        # Gradient via FFT
        w_rev = w[::-1]
        g = _grad_soft(a, w_rev, fs)
        
        # Normalize and apply momentum
        gn = np.sqrt((g*g).sum())
        if gn > 1e-15:
            g /= gn
        mom = beta * mom + (1.0 - beta) * g
        a -= step * mom
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            break
        a *= S / s
        
        # Anneal
        T = max(Tmin, T * anneal)
        step *= decay
        
    return best, best_sc

    return best, best_sc

    rng = np.random.default_rng(seed)
    dl_total = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Initialize from parent
    p = np.clip(np.array(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(512)
    n = p.size
    fs = 1 << (2 * n - 1).bit_length()
    S = p.sum()  # normalization level (score is scale-invariant)
    
    best = p.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    
    # Phase 1: Soft-max descent (warm)
    dl1 = time.monotonic() + (dl_total - time.monotonic()) * 0.30
    a, sc = _soft_descent(p.copy(), fs, S, dl1, 
                          T0=5.0, Tmin=50.0, anneal=1.002, 
                          step0=0.4, decay=0.99995, beta=0.85, rng=rng)
    if sc < best_sc:
        best_sc, best = sc, a
    
    # Phase 2: Soft-max descent (hot, exploration)
    if time.monotonic() < dl_total - 200:
        dl2 = time.monotonic() + (dl_total - time.monotonic()) * 0.25
        a, sc = _soft_descent(best.copy(), fs, S, dl2,
                              T0=2.0, Tmin=30.0, anneal=1.003,
                              step0=0.3, decay=0.99993, beta=0.75, rng=rng)
        if sc < best_sc:
            best_sc, best = sc, a
    
    # Phase 3: Subgradient hard-descent (cold)
    if time.monotonic() < dl_total - 100:
        dl3 = time.monotonic() + (dl_total - time.monotonic()) * 0.20
        a, sc = _soft_descent(best.copy(), fs, S, dl3,
                              T0=80.0, Tmin=200.0, anneal=1.002,
                              step0=0.25, decay=0.99997, beta=0.90, rng=rng)
