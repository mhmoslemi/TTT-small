            g[lo:hi+1] += w[ii] * a[kk-hi:kk-lo+1][::-1]
    return g

def optimize(a_in, n, fsz, dl, tk, es):
    a = np.maximum(a_in, 1e-15).copy()
    a /= a.sum()
    ba = a.copy()
    bm = float(fft_conv(a, fsz).max())
    gavg = np.zeros(n)
    stall = 0
    for it in range(500000):
        if time.monotonic() > dl:
            break
        cv = fft_conv(a, fsz)
        m = float(cv.max())
        if m < bm - 1e-16:
            bm = m
            ba = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 10000:
            break
        g = grad_topk(a, cv, tk, n)
        gavg = 0.8 * gavg + 0.2 * g
        eta = es * n * (1.0 / (1.0 + 0.001 * it))
        np.multiply(a, np.exp(-eta * gavg), out=a)
        np.maximum(a, 1e-18, out=a)
        a /= a.sum()
    return ba, bm

def local_refine(seq, n, dl, rng):
    best = np.maximum(seq, 0.0).copy()
    if best.sum() < 0.01:
        return best, np.inf
    best_sc = float(evaluate_sequence(best.tolist()))
    w = max(3, n // 300)
    it = 0
    while time.monotonic() < dl and it < 100000:
        it += 1
        c = best.copy()
        n_p = int(rng.integers(1, 5))
        for _ in range(n_p):
            st = int(rng.integers(0, max(1, n - 1)))
            ln = int(rng.integers(max(1, w // 4), w + 1))
            st = min(st, n - 1)
            ln = min(ln, n - st)
            c[st:st+ln] *= float(np.exp(rng.normal(0.0, 0.01)))
        c = np.clip(c, 0.0, 1e6)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = c.copy()
    return best, best_sc
