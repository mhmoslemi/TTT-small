from scipy.optimize import linprog

def fft_convolve(a, b, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    fb = np.fft.rfft(b, n=fft_size)
    return np.fft.irfft(fa * fb, n=fft_size)

def solve_lp(f, M, active_set, fft_size):
    """Solve max sum(b) s.t. conv(f,b)[k] <= M for k in active_set, b >= 0"""
    n = len(f)
    m = len(active_set)
    if m == 0:
        # No constraints, optimum is unbounded. Return uniform.
        return np.ones(n) / n * M  # arbitrary feasible
    
    # Build constraint matrix A (m x n)
    # A[k_idx, j] = f[k - j] if 0 <= k - j < n, else 0
    A = np.zeros((m, n))
    for ii, k in enumerate(active_set):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[ii, lo:hi+1] = f[k - hi : k - lo + 1][::-1]
    
    # Solve: min -sum(b) s.t. A b <= M, b >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

import numpy as np
import time
from scipy.optimize import linprog

def fft_conv(a, b, fs):
    if a is b:
        fa = np.fft.rfft(a, n=fs)
        return np.fft.irfft(fa*fa, n=fs)
    fa = np.fft.rfft(a, n=fs)
    fb = np.fft.rfft(b, n=fs)
    return np.fft.irfft(fa*fb, n=fs)

def build_A(f, k_list):
    """Build constraint matrix for LP. Shape: (len(k_list), len(f))"""
    n = len(f)
    m = len(k_list)
    A = np.zeros((m, n))
    for ii, k in enumerate(k_list):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[ii, lo:hi+1] = f[k - hi : k - lo + 1][::-1]
    return A

def lp_direction(f, M, active_k):
    """Solve LP: max sum(b) s.t. conv(f,b)[k] <= M for k in active_k, 0 <= b <= 100"""
    n = len(f)
    m = len(active_k)
    if m == 0:
        return np.ones(n) / n
    A = build_A(f, active_k)
    c = -np.ones(n)
    b_ub = np.full(m, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, 100.0)]*n, method='highs', options={'maxiter': 5000})
    if res.success:
        return res.x
    return None

def score_norm(a):
    """Score for normalized a (sum=1): 2n*max(conv(a,a))"""
    n = len(a)
    fs = 1 << (2*n-1).bit_length()
    conv = fft_conv(a, a, fs)
    return 2.0 * n * float(np.max(conv[:2*n-1]))

def run_lp_opt(a, n, fs, deadline, max_lp_iters=200):

    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-10:
        return a
    a = a / s
    
    conv = fft_conv(a, a, fs)[:2*n-1]
    M = float(np.max(conv))
    best_score = 2.0 * n * M
    best_a = a.copy()
    
    # Initial active set: top lags
    n_top = min(50, 2*n-1)
    top_idx = np.argpartition(conv, -n_top)[-n_top:]
    active_k = sorted(set(int(x) for x in top_idx))
    
    for it in range(max_lp_iters):
        if time.monotonic() > deadline:
            break
        
        b = lp_direction(a, M, active_k)
        if b is None or b.sum() <= 1e-10:
            break
        
        # Check full feasibility
        fb = fft_conv(a, b, fs)[:2*n-1]
        max_fb = float(np.max(fb))
        
        if max_fb > M * (1 + 1e-4):
            # Infeasible - expand active set
            viol = np.where(fb > M * (1 + 1e-4))[0]
            if len(viol) > 0:
                # Add top violators to active set
                top_v = np.argpartition(fb, -min(100, 2*len(active_k)+50))[-min(100, 2*len(active_k)+50):]
                new_set = set(int(x) for x in top_v)
                active_k = sorted(new_set)
                if len(active_k) > 300:
                    active_k = sorted(set(int(x) for x in np.argpartition(fb, -300)[-300:]))
            continue
        
        # Feasible - use as direction
        g = b / b.sum()
        
        # Line search
        best_t = 0
        best_M_new = M
        for t in [0.3, 0.5, 0.7, 1.0, 0.1, 0.2]:

            if M_try < best_M_new - 1e-10:

            a_try = np.clip(a_try, 0, None)
            if a_try.sum() < 1e-10:

            conv_try = fft_conv(a_try, a_try, fs)[:2*n-1]
            M_try = float(np.max(conv_try))
            if M_try >= M:

def subgrad_refine(a, n, fs, deadline, max_iters=200000):
    a = np.clip(a, 0, None) / a.sum()
    conv = fft_conv(a, a, fs)[:2*n-1]
    M = float(np.max(conv))
    best_a = a.copy()
    best_M = M
    v = np.zeros(n)
    beta = 0.9
    lr = 0.5
    top_k = 16
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Subgradient: average of active terms
        idx = np.argpartition(conv, -top_k)[-top_k:]
        grad = np.zeros(n)
        for ii in range(top_k):
            k = int(idx[ii])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
        grad /= top_k  # 2*average (the factor of 2 is absorbed in lr)
        
        v = beta * v + (1 - beta) * grad
        v -= v.mean()  # preserve sum
        
        # Line search
        accepted = False
        for bs in range(6):
            step = lr / (2**bs)
            a_try = np.clip(a - step * v, 0, None)
            s_try = a_try.sum()
            if s_try < 1e-10:
                break
            a_try = a_try / s_try
            conv_try = fft_conv(a_try, a_try, fs)[:2*n-1]
            M_try = float(np.max(conv_try))
            if M_try <= M * (1 + 1e-12):
                a = a_try
                M = M_try
                accepted = True
                break
        
        if accepted:
            if M < best_M - 1e-12:
                best_M = M
                best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 5000:
                break
        
        lr = max(0.01, lr * 0.99998)
        beta = min(0.97, beta + 0.00001)
        if it % 5000 == 0 and top_k < 128:
            top_k += 8
    
    return best_a, 2.0 * n * best_M

from scipy.optimize import linprog

    fb = np.fft.rfft(b, n=fs)
    return np.fft.irfft(fa * fb, n=fs)
