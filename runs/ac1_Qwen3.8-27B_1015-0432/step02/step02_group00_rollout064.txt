def fft_conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = fa if b is a else np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb)[:n + m - 1]

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s
