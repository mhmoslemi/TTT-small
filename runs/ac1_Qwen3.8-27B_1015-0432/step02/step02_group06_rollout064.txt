import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def multigra_obj(f, beta):
    b = fft_conv(f, f)
    M = b.max()
    w = np.exp(beta * (b - M))
    f_rev = f[::-1]
    gc = fft_conv(w, f_rev)
    grad = gc[f.size - 1: 2 * f.size - 1]
    return grad, M
