  4. The "cheapest" reduction is to reduce the smaller of a_j and a_{k*-j}.
  5. This gives a "water-filling" type update.
  
  More concretely:
  - Identify the top contributing pairs (j, k*-j) to c_{k*}.
  - Reduce a at those positions proportionally to their contribution.
  - Redistribute the saved mass to positions that don't contribute to c_{k*}.
  
  This is essentially what the gradient does, but the "directed" version might be more efficient.

import numpy as np
import time
from numpy.fft import fft, ifft
from scipy.optimize import minimize


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -45.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _soft_obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _soft_grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _subgrad(a, c, K):
    n = a.size
    S = a.sum()
    M = c.max()
    topk = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in topk:
        idx = k - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g[valid] += 2.0 * a[idx[valid]]
    g *= (2.0 * n) / (S * S)
    g -= np.full(n, 4.0 * n * M / (S ** 3))
    return g
