                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                to3 = time.monotonic() + min(70.0, deadline - time.monotonic() - 25.0)
                if to3 > time.monotonic() + 5.0:
                    f3, sc3 = optimize(trial, to3, rng)
                    if sc3 < best_sc:
                        best_sc, best_f = sc3, f3
    while time.monotonic() < deadline - 5.0:
        f_fin, sc_fin = optimize(best_f, deadline - 5.0, rng)
        if sc_fin < best_sc:
            best_sc, best_f = sc_fin, f_fin
        else:
            break
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [float(x) for x in parent]
    return result
