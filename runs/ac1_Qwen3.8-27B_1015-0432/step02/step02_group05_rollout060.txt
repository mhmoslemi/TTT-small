        (4, 2, 0.10, 0.85, 0.93),
        (3, 2, 0.08, 0.85, 0.92),
    ]

    best = a0.copy()
    best_sc = float(evaluate_sequence((a0 * s0).tolist()))

    n_inits = len(inits)
    for k in range(n_inits):
        rem = deadline - time.monotonic()
        if rem < 45.0:
            break
        cap = 280.0 if k == 0 else 170.0
        per = min(rem / max(1, n_inits - k), cap)
        per = min(per, rem - 20.0)
        if per < 10.0:
            break
        k0, k1, e0, ml, mh = configs[k % len(configs)]
        a, _ = optimize(inits[k].copy(), n, fsz, time.monotonic() + per, k0, k1, e0, ml, mh)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < deadline - 40.0:
        rem2 = deadline - time.monotonic() - 30.0
        per2 = min(rem2 * 0.6, 200.0)
        a, _ = optimize(best.copy(), n, fsz, time.monotonic() + per2, 2, 1, 0.05, 0.90, 0.95)
        cand = a * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()

    if time.monotonic() < deadline - 5.0:
        base = best * s0
        w_sm = max(2, n // 500)
        w_md = max(3, n // 100)
        while time.monotonic() < deadline - 2.0:
            c = base.copy()
            np_ = int(rng.integers(1, 5))
            for _ in range(np_):
                w = w_sm if rng.random() < 0.5 else w_md
                st = int(rng.integers(0, max(1, n - w)))
                sigma = 0.008 + 0.02 * rng.random()
                c[st:min(n, st + w)] *= float(np.exp(rng.normal(0.0, sigma)))
            c = np.clip(c, 0.0, 1000.0)
            if c.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(c.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                base = c

    out = np.clip(best * s0, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(x) for x in out]
