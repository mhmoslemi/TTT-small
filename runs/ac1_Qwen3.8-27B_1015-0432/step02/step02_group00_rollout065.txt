def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def normalize(f):
    f = np.maximum(f, 1e-30)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s) if s > 1e-15 else np.inf

def onehot(b):
    w = np.zeros_like(b)
    w[np.argmax(b)] = 1.0
    return w

def grad_sw(f, b, T):

    return np.maximum(2.0 * c[n - 1:2 * n - 1][::-1], 0.0)
