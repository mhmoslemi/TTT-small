def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_window(n, alpha, beta, gamma):
    """Parametric window: alpha * raised_cosine + beta * parabolic + gamma * flat"""
    x = np.linspace(0, 1, n)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))  # raised cosine (Hann)
    par = 1 - (2 * x - 1)**2  # parabolic
    flat = np.ones(n)
    w = alpha * rc + beta * par + gamma * flat
    w = np.maximum(w, 1e-15)
    return w / w.sum()
