                lb = cand.copy()
    return a, lb, ls


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = par.size
    if n < 2 or not np.all(np.isfinite(par)) or par.sum() < 0.01:
        par = np.ones(1024, dtype=np.float64)
        n = 1024
    s0 = float(par.sum())
    fs = 1 << (2 * n - 1).bit_length()
    ap = par / s0
    best = par.copy()
    bsc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(bsc):
        bsc = 2.0
        best = np.ones(n, dtype=np.float64)

    # Phase 1: main descent from parent (50%)
    t1 = time.monotonic() + 0.50 * (dl - time.monotonic())
    _, lb, ls = optimize(ap.copy(), n, fs, t1, best, bsc, s0, rng)
    if ls < bsc:
        bsc, best = ls, lb

    # Phase 2: refinement (25%)
    t2 = time.monotonic() + 0.25 * (dl - time.monotonic())
    _, lb2, ls2 = optimize(best / s0, n, fs, t2, best, bsc, s0, rng)
    if ls2 < bsc:
        bsc, best = ls2, lb2

    # Phase 3: alternative initializations (12%)
    ix = np.arange(n, dtype=np.float64)
    inits = [
        np.cos(np.pi * (ix + 0.5) / n) ** 2,
        np.sin(np.pi * (ix + 0.5) / n) ** 2,
        1.0 + 0.5 * np.cos(2.0 * np.pi * ix / n),
    ]
    for init in inits:
        if time.monotonic() > dl - 18.0:
            break
        init = np.clip(init, 0.0, None)
        if init.sum() < 1e-10:
            continue
        init = init / init.sum()
        t3 = time.monotonic() + 0.04 * (dl - time.monotonic())
        _, lb3, ls3 = optimize(init, n, fs, t3, best, bsc, s0, rng)
        if ls3 < bsc:
            bsc, best = ls3, lb3

    # Phase 4: local perturbations
    base = best.copy()
    while time.monotonic() < dl - 3.0:
        c = base.copy()
        w = max(3, n // 100)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            c[st:st + w] *= np.exp(float(rng.normal(0.0, 0.035)))
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < bsc:
            bsc, best, base = sc, c, c.copy()

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = par
    return [float(x) for x in out]
