from scipy.sparse import lil_matrix, csr_matrix
from scipy.optimize import linprog
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def compute_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30 or not np.isfinite(ws):
        w = np.zeros_like(b)
        k = max(1, b.size // 50)
        w[np.argsort(b)[-k:]] = 1.0
    gc = fft_conv(w, f[::-1])
    grad = gc[f.size - 1:2 * f.size - 1].copy()
    return np.maximum(grad, 0.0)

def solve_lp(f, b, M, K, time_limit=20):
    n = f.size
    top = np.argsort(b)[-K:][::-1]
    A = lil_matrix((K, n), dtype=np.float64)
    for idx, k in enumerate(top):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        for j in range(lo, hi + 1):
            A[idx, j] = f[k - j]
    A = A.tocsr()
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(K, M), method='highs',

    if res.status == 0:
        g = np.maximum(res.x, 0)
        s = g.sum()
        return g / s if s > 1e-10 else None
    return None

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score_of(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()


def compute_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30 or not np.isfinite(ws):
        w = np.zeros_like(b)
        k = max(1, b.size // 50)
        w[np.argsort(b)[-k:]] = 1.0
    gc = fft_conv(w, f[::-1])
    grad = gc[f.size - 1:2 * f.size - 1].copy()
    return np.maximum(grad, 0.0)


def solve_lp(f, b, M, K, time_limit=25):
    n = f.size
    top = np.argsort(b)[-K:][::-1]
    A = lil_matrix((K, n), dtype=np.float64)
    for idx, k in enumerate(top):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if hi >= lo:
            A[idx, lo:hi + 1] = f[k - hi:k - lo + 1]
    A = A.tocsr()
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n

                  options={'time_limit': time_limit})
    if res.status == 0:
