import numpy as np
import time
from scipy.optimize import linprog

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_flat_top(n, asym):
    x = np.linspace(-np.pi, np.pi, n)
    w = (1.0 - np.cos(x)) / 2.0
    # Blackman-like
    w = 1.0 - 0.42 * np.cos(x) + 0.21 * np.cos(2*x) - 0.07 * np.cos(3*x)
    if abs(asym) > 1e-9:
        w *= np.exp(asym * (2.0 * np.arange(n) / max(n-1, 1) - 1.0))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def make_various(n, idx):
    """Generate various initializations algorithmically."""
    j = np.arange(n, dtype=np.float64)
    x = j / max(n-1, 1)
    if idx == 0:
        return make_flat_top(n, 0.0)
    elif idx == 1:
        return make_flat_top(n, 0.5)
    elif idx == 2:
        return make_flat_top(n, -0.5)
    elif idx == 3:
        w = (1-x)**2 * x**2  # original
        return np.maximum(w, 1e-10) / np.maximum(w, 1e-10).sum()
    elif idx == 4:
        w = np.exp(-4*x*(1-x))  # concentrated
        return w / w.sum()
    elif idx == 5:
        w = np.exp(-10*x*(1-x))  # more concentrated
        return w / w.sum()
    elif idx == 6:

    elif idx == 7:
        w = 1.0 / (1.0 + 50.0 * x * (1-x))
        return w / w.sum()
    else:
        w = 1.0 - 0.5*np.cos(2*np.pi*x)
        return np.maximum(w, 1e-10) / np.maximum(w, 1e-10).sum()

def grad_smooth(f, C, T):
    """Smooth gradient of max(C) w.r.t. f."""
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def mirror_optimize(f, deadline, rng):
    n = f.size
    f = f.copy()
    f = np.maximum(f, 1e-30)
    f /= f.sum()
    eta = 0.05
    T = 2.0
    best_f = f.copy()
    best_sc = fast_score(f)
    no_imp = 0
    for it in range(100000):
        if time.monotonic() > deadline - 10.0:
            break
        C = fft_conv(f, f)
        sc = 2.0 * n * C.max() / (f.sum()**2)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Adaptive temperature
        C_range = C.max() - C.min()
        if C_range < 1e-30:
            break
        
        grad = grad_smooth(f, C, T)
        gmax = grad.max()
        if gmax < 1e-30:

        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-30)
        f /= f.sum()

            if ns < sc - 1e-10:

            else:

            no_imp = 0

            hi = min(k, n - 1)
            if lo <= hi:
                # A[row, j] = f[k-j] for j in [lo, hi]
                A[row, lo:hi+1] = f[k - np.arange(lo, hi+1)]
        
        # Solve: max sum(g) s.t. A@ag <= M, g >= 0
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(m, M * s / n),  # scale M
                      bounds=bounds, method='highs',
                      options={'time_limit': min(30, deadline - time.monotonic() - 20)})
        if not res.success:
            break
        g_new = res.x
        if g_new.sum() < 1e-15:
            break
        
        # Check for violations
        fg = fft_conv(f, g_new)
        thresh = (fg.max() / M + 1e-8) * M
        violations = np.where(fg > thresh)[0]
        if len(violations) == 0:

            break
        # Add violations to active set
        new_active = list(set(active + list(violations[:30])))
        new_active = new_active[:200]  # cap
        active = new_active
        g = g_new
    return g

def lp_mix_step(f, gradient_dir, t, deadline, rng):

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
