    g = np.zeros(n)
    g[j_valid] = 2.0 * a[k_star - j_valid]
    
    # Full objective gradient (subgradient): d(2n*max/S²)/da = (2n/S²)*g - (4n*max/S³)*1
    max_val = c[k_star]
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * max_val / (S ** 3)) * np.ones(n)
    
    # Gradient descent step with projection
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 0.01:
        return a
    return a_new
