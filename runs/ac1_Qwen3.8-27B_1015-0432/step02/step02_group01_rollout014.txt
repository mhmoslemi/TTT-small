import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    result = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))
    return result[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max() / f.sum()**2

def grad_for_weighted_max(f, b, top_frac=0.01):
    """Compute gradient direction for weighted max of convolution."""
    M = b.max()
    if M < 1e-30:
        return None
    n = f.size
    # Weight top lags
    k = max(1, int(n * top_frac))
    idx = np.argsort(b)[-k:]
    # Temperature weighting
    w = np.zeros_like(b)
    w[idx] = (b[idx] / M) ** 20  # sharp weighting
    w /= w.sum()
    # Gradient: conv(w, f_reversed)
    f_rev = f[::-1]
    grad_full = fft_conv(w, f_rev)
    grad = grad_full[n-1 : 2*n-1]
    return np.maximum(grad, 0)

def optimize(f_init, deadline, rng, max_iters=50000):

    f = f_init.astype(np.float64).copy()
    f = np.maximum(f, 1e-10)
    f /= f.sum()
    
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.1
    beta = 30.0
    stall_count = 0
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        if it % 500 == 0 and time.monotonic() > deadline - 60:
            break
            
        b = fft_conv(f, f)
        sc = 2.0 * n * b.max() / f.sum()**2
        
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_f = f.copy()
            no_improve = 0
            stall_count = 0
        else:
            no_improve += 1
            if no_improve % 500 == 0:
                stall_count += 1
                if stall_count > 5:
                    # Restart with perturbation
                    f = best_f * np.exp(0.1 * rng.standard_normal(n))
                    f = np.maximum(f, 1e-10)
                    f /= f.sum()
                    eta = 0.1
                    beta = 30.0
                    stall_count = 0
                    no_improve = 0
                    continue
        
        # Compute gradient
        M = b.max()
        k = max(1, int(n * 0.005))
        top_idx = np.argsort(b)[-k:]
        w = np.zeros(b.size)
        vals = b[top_idx] / M
        w[top_idx] = vals ** (20 + beta)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(b)
            w[top_idx[:max(1,k//10)]] = 1.0
            w_sum = w.sum()
        w /= w_sum
        
        f_rev = f[::-1]
        grad_full = fft_conv(w, f_rev)
        grad = grad_full[n-1 : 2*n-1].copy()
        grad = np.maximum(grad, 0)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        grad /= gmax
        
        # Multiplicative update
        f = f * np.exp(-eta * grad)
        f = np.maximum(f, 1e-15)
        f /= f.sum()
        
        # Decay
        if it % 1000 == 0:
            eta *= 0.98
            beta *= 1.02
            eta = max(eta, 0.002)
    
    return best_f, best_sc
