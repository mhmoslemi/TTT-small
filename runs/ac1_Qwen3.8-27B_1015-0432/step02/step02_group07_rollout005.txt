import numpy as np
import time


def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def compute_grad(a, conv, tau, tk):
    n = a.size
    idx = np.argpartition(conv, -tk)[-tk:]
    order = np.argsort(conv[idx])[::-1]
    idx = idx[order]
    mx = conv[idx[0]]
    w = np.exp((conv[idx] - mx) / tau)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(tk)
        w[0] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for ii in range(tk):
        k = int(idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += w[ii] * 2.0 * a[k - hi:k - lo + 1][::-1]
    return grad
