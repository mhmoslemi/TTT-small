            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return g

def _run(a, n, ts, dl, rng, mode, eta0, beta, cap, frac, smx, dec):
    a = a.astype(np.float64).copy()
    mom = np.zeros(n)
    bst, bsc, psc, stall = a.copy(), _sc(a, n), bsc, 0
    eta = eta0
    v = np.zeros(n)  # for Adam-like
    m = np.zeros(n)
    t = 0
    while time.monotonic() < dl:
        t += 1
        c = _cv(a)
        s = a.sum()
        sc = 2.0 * n * float(c.max()) / (s * s)
        if sc < bsc - 1e-12:
            bsc, bst = sc, a.copy()
        stall = 0 if sc < psc - 1e-13 else stall + 1
        psc = sc
        if stall > smx:
            break
        mx = float(c.max())
        act = np.where(c >= mx * (1.0 - frac))[0]
        act = act[(act >= 0) & (act <= 2*n-2)]
        if len(act) > cap:
            act = act[np.argpartition(c[act], -cap)[-cap:]]
        if len(act) == 0:
            act = np.array([n-1])
        # Weighted gradient: weight by c[k]
        wts = c[act] / c[act].sum()
        g = np.zeros(n)
        for k, w in zip(act, wts):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                g[lo:hi+1] += w * a[k-hi:k-lo+1][::-1]
        g -= g.mean()
        # Noise injection for plateau escape
        if stall > smx // 3:
            nz = rng.standard_normal(n)
            g += 0.04 * nz / (np.abs(nz).max() + 1e-14)
        if mode == 'mu':
            gm = np.abs(g).max()
            if gm > 1e-14:
                a *= np.exp(-eta * g / gm)
        else:
            mom = beta * mom + (1 - beta) * g
            a -= eta * mom
        a = np.clip(a, 0.0, 1000.0)
        s2 = a.sum()
        if s2 < 1e-10:
            break
        a *= ts / s2
        eta *= dec
    return bst, bsc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a0.size < 2 or not np.all(np.isfinite(a0)) or a0.sum() < 0.01:
        a0 = np.ones(256)
    n = a0.size
    ts = float(a0.sum())
    
    best, bs = a0.copy(), float(evaluate_sequence(a0.tolist()))
    
    # Phase 1: Euclidean descent
    a, s = _run(a0, n, ts, dl - 280, rng, 'eu', 0.5, 0.85, 60, 3e-4, 4500, 0.99998)
    if s < bs: best, bs = a, s
    
    # Phase 2: Multiplicative descent
    if time.monotonic() < dl - 180:
        a, s = _run(best, n, ts, dl - 160, rng, 'mu', 0.3, 0, 70, 2.5e-4, 4000, 0.99999)
        if s < bs: best, bs = a, s
