import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def grad_smooth(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def grad_argmax(f, k_star):
    n = f.size
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    grad = np.zeros(n)
    if lo <= hi:
        grad[lo:hi + 1] = 2.0 * f[k_star - lo:k_star - hi + 1][::-1]
    return grad

def mirror_optimize(f, deadline, rng):
    n = f.size
    f = f.copy()
    f /= f.sum()
    eta = 0.15
    T = 0.5
    best_f = f.copy()
    best_sc = fast_score(f)
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        if it % 5000 == 0:
            C = fft_conv(f, f)
            sc = 2.0 * n * C.max() / (f.sum() ** 2)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
            if time.monotonic() > deadline:
                break
        # Choose gradient
        if it % 40 >= 10:
            grad = grad_smooth(f, C, T)
        else:
            grad = grad_argmax(f, int(np.argmax(C)))
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        # Adapt eta
        if it % 20 == 0:
            ns = fast_score(f)
            eta = min(eta * 1.05, 2.0) if ns < sc - 1e-12 else max(eta * 0.9, 0.01)
        # Cool down
        if it % 50 == 0:
            T = max(0.02, T * 0.97)
        # Restart
        if no_imp > 150:
            f = best_f * np.exp(0.05 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.15
            T = 0.5
        it += 1
    return best_f, best_sc
