        e = int(n * (1.0 + fr) / 2.0)
        bx[s:e] = 1.0
        if bx.sum() > 0.01:
            inits.append(_normalize(bx.copy(), scale))
    for pw in [1.5, 3.0]:
        d = np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1.0)
        pl = np.clip(1.0 - d, 0.0, None) ** pw
        if pl.sum() > 0.01:
            inits.append(_normalize(pl.copy(), scale))
    for _ in range(2):
        r = np.clip(rng.uniform(0.0, 1.0, n), 0.0, None)
        if r.sum() > 0.01:
            inits.append(_normalize(r.copy(), scale))

    cur = a0.copy()
    for inp in inits:
        sc = float(evaluate_sequence(inp.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = inp.copy()
            best_sc = sc
            cur = inp.copy()

    # Main L-BFGS-B with annealed temperature
    bounds_list = [(0.0, 1000.0)] * n
    taus = [5.0, 2.5, 1.5, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 100.0:
            break
        rem = dl - time.monotonic()
        iters = min(800, int(rem * 14))
        if iters < 30:
            break
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': iters, 'gtol': 1e-14,
                                    'maxls': 40, 'ftol': 1e-20})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _normalize(nx, scale)
            best, best_sc = _eval_and_update(nx, best, best_sc)
            cur = nx.copy()
        except Exception:
            pass

    # Symmetrization check
    if time.monotonic() < dl - 60.0:
        sym = 0.5 * (cur + cur[::-1])
        sym = _normalize(np.clip(sym, 0.0, 1000.0), scale)
        sc = float(evaluate_sequence(sym.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = sym.copy()
            best_sc = sc
            cur = sym.copy()

    # Targeted local search: perturb near peak contributors
    if time.monotonic() < dl - 25.0:
        wmax = max(2, n // 25)
        for _ in range(800):
            if time.monotonic() > dl - 25.0:
                break
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            w1 = int(rng.integers(1, wmax + 1))
            lo = max(0, min(pk, n - 1) - w1)
            hi = min(n - w1, max(0, min(pk, n - 1)))
            s1 = int(rng.integers(lo, hi + 1)) if hi >= lo else int(rng.integers(0, n - w1))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.04, 0.05)))
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.04, 0.05)))
            if rng.random() < 0.12:
                cd *= float(np.exp(rng.normal(0.0, 0.012)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.06:
                cur = best.copy()

    # Final polish at very low temperature
    if time.monotonic() < dl - 5.0:
        cur = best.copy()
        for tv in [0.001, 3e-4, 1e-4]:
            if time.monotonic() > dl - 5.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 700, 'gtol': 1e-15,
                                        'maxls': 25, 'ftol': 1e-22})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _normalize(nx, scale)
                best, best_sc = _eval_and_update(nx, best, best_sc)
                cur = nx.copy()
            except Exception:
                pass

    # Verify final result
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = a0
        for v in best:
            if not np.isfinite(v):
                best[:] = 1.0
    return [float(v) for v in best]
