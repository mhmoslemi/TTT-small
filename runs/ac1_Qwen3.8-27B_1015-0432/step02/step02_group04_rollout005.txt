import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def smooth_grad(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def topk_grad(f, C, k):
    """Gradient using top-k largest values of C."""
    n = f.size
    k = min(k, C.size)
    idx = np.argpartition(C, -k)[-k:]
    w = C[idx]
    w = w - w.min()
    p = w / w.sum()
    N = 1 << (C.size + n - 1).bit_length()
    # Build p as full vector
    pv = np.zeros(C.size)
    pv[idx] = p
    P = np.fft.rfft(pv, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def init_vars(n, rng):
    inits = [np.ones(n) / n]
    x = 2.0 * np.arange(n) / max(n - 1, 1) - 1.0
    # Raised cosine
    w = 0.5 * (1 + np.cos(np.pi * x))
    inits.append(np.maximum(w, 1e-10) / max(w.max(), 1e-10))
    inits[-1] /= inits[-1].sum()
    # Asymmetric exponentials
    for a in [0.5, -0.5, 1.0, -1.0]:
        w = np.exp(a * x)
        inits.append(w / w.sum())
    # Bimodal
    w = np.exp(-((x - 0.5) / 0.3) ** 2) + np.exp(-((x + 0.5) / 0.3) ** 2)
    inits.append(w / w.sum())
    # Power law
    j = np.arange(1, n + 1, dtype=float)
    for a in [0.5, -0.5]:
        w = j ** a
        inits.append(w / w.sum())
    return inits

    no_imp = 0
    for it in range(max_it):

def lp_direction_approx(f, M):
    """Approximate LP direction: g proportional to 1/f, projected to satisfy constraints."""
    n = f.size
    g = 1.0 / np.maximum(f, 1e-15)
    # Iterate: reduce g where (f*g) > M
    for _ in range(20):
        fg = fft_conv(f, g)
        viol = fg[fg > M]
        if viol.size == 0:
            break
        # Reduce g uniformly (simple projection)
        ratio = M / fg.max()
        g *= ratio
        g = np.maximum(g, 1e-30)
    s = g.sum()
    if s < 1e-15:
        return None
    return g / s
