    taus = [2.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]

    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 200.0:
            break
        if i < 3:
            iters = 1000
        elif i < 7:
            iters = 800
        else:
            iters = 600
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': iters, 'gtol': 1e-16,
                                    'maxls': 50, 'ftol': 1e-22})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                nx = nx / nx.sum()
            else:
                nx = cur.copy()
            sc = float(evaluate_sequence((nx * best.sum()).tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx * best.sum()
                best_sc = sc
                cur = nx.copy()
            else:
                cur = nx.copy()
        except Exception:
            pass

    # Manual projected gradient refinement
    if time.monotonic() < dl - 80.0:
        cur = best / best.sum()
        step = 0.5 / n
        tau_ref = 1e-4
        for it in range(30000):
            if time.monotonic() > dl - 80.0:
                break
            a_new = _proj_grad_step(cur, step, tau_ref)
            if a_new.sum() < 1e-8:
                break
            a_new = a_new / a_new.sum()
            sc2 = float(evaluate_sequence((a_new * best.sum()).tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = a_new * best.sum()
                best_sc = sc2
                cur = a_new.copy()
                step = min(step * 1.01, 5.0 / n)
            else:
                step *= 0.85
            if step < 1e-12:
                break

    # Subgradient descent targeting exact max
    if time.monotonic() < dl - 40.0:
        cur = best / best.sum()
        n_cur = cur.size
        step = 0.3 / n_cur
        for it in range(20000):
            if time.monotonic() > dl - 40.0:
                break
            c = _conv(cur)
            p = int(np.argmax(c))
            S = cur.sum()
            sf = np.zeros(n_cur)
            lo = max(0, p - n_cur + 1)
            hi = min(n_cur - 1, p)
            if lo <= hi:
                js = np.arange(lo, hi + 1)
                sf[js] = cur[p - js]
            g = (2.0 * n_cur / (S * S)) * (2.0 * sf) - (4.0 * n_cur * c[p] / (S ** 3))
            a_new = cur - step * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() < 1e-8:
                break
            a_new = a_new / a_new.sum()
            sc2 = float(evaluate_sequence((a_new * best.sum()).tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = a_new * best.sum()
                best_sc = sc2
                cur = a_new.copy()
                step = min(step * 1.01, 3.0 / n_cur)
            else:
                step *= 0.85
            if step < 1e-12:
                break

    # Local multiplicative perturbation
    while time.monotonic() < dl - 5.0:
        cd = cur.copy()
        nw = max(1, n // 60)
        s = int(rng.integers(0, max(1, n - nw)))
        cd[s:s + nw] *= np.exp(rng.normal(0.0, 0.02, size=nw))
        if rng.random() < 0.08:
            cd *= np.exp(rng.normal(0.0, 0.008))
        cd = np.clip(cd, 0.0, None)
        if cd.sum() > 0.01:
            cd = cd / cd.sum()
            sc2 = float(evaluate_sequence((cd * best.sum()).tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd * best.sum()
                best_sc = sc2
                cur = cd.copy()
        if rng.random() < 0.04:
            cur = best / best.sum()

    final = np.clip(best, 0.0, 1000.0)
    if not np.all(np.isfinite(final)) or final.sum() < 0.01:
        final = np.ones(n) * 1.0
    return [float(v) for v in final]
