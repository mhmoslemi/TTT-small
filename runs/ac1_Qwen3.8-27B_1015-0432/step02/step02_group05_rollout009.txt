    rem = dl - time.monotonic()
    if rem > 90:
        r3 = np.abs(rng.standard_normal(n))
        for _ in range(3):
            r3 = (r3 + np.roll(r3, 1)) * 0.5
        r3 /= r3.sum()
        cap = min(rem * 0.15, 120.0)
        a, _ = run_phase(r3, n, fsz, time.monotonic() + cap, 24, 0.02, 15000, 0.99994)
        sc = float(evaluate_sequence((a * s0).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best = sc, a.copy()

    # Phase F: Sine modulation init
    rem = dl - time.monotonic()
    if rem > 90:
        f1 = rng.uniform(0.04, 0.12)
        p1 = rng.uniform(0.0, 6.28)
        r2 = np.clip(1.0 + 0.35 * np.sin(2.0 * np.pi * f1 * np.arange(n) + p1), 0.01, None)
        r2 /= r2.sum()
        cap = min(rem * 0.15, 120.0)
        a, _ = run_phase(r2, n, fsz, time.monotonic() + cap, 32, 0.02, 15000, 0.99993)
        sc = float(evaluate_sequence((a * s0).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best = sc, a.copy()

    # Final ultra-fine polish
    rem = dl - time.monotonic()
    if rem > 50:
        cap = min(rem * 0.4, 100.0)
        a, _ = run_phase(best, n, fsz, time.monotonic() + cap, 3, 0.003, 50000, 0.99999)
        sc = float(evaluate_sequence((a * s0).tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best = sc, a.copy()

    # Perturbation phase
    if time.monotonic() < dl - 3.0:
        base = best.copy()
        n_pat = 0
        while time.monotonic() < dl - 2.0 and n_pat < 2000:
            n_pat += 1
            c = base.copy()
            w = max(3, n // 400)
            n_p = int(rng.integers(1, 4))
            for _ in range(n_p):
                st = int(rng.integers(0, max(1, n - w)))
                c[st:min(n, st + w)] *= float(np.exp(rng.normal(0.0, 0.008)))
            c = np.clip(c, 0.0, None)
            if c.sum() < 0.01:
                continue
            c /= c.sum()
            sc = float(evaluate_sequence((c * s0).tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                base = c.copy()

    out = best * s0
    out = np.clip(out, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(x) for x in out]
