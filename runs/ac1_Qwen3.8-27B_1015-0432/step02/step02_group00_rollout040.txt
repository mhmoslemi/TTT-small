def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_f(f):
    f = np.maximum(f, 1e-30)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / len(f)
    return f / s

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * len(f) * b.max() / (s * s)

def soft_grad(f, t):
    n = f.size
    c = fft_conv(f, f)
    M = c.max()
    shifted = c - M
    w = np.exp(t * shifted)

    if ws < 1e-30:
