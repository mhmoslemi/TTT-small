import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    h = np.maximum(h, 1e-15)
    return h / h.sum()


def init_causal(n, r):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-r * x / n)
    return a / a.sum()


def init_sym(n, r):
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-r * np.abs(x - c) / n)
    return a / a.sum()


def init_blackman(n):
    x = np.arange(n, dtype=np.float64)
    a = 0.42 + 0.5 * np.cos(2.0 * np.pi * x / (n + 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n + 1))
    a = np.maximum(a, 1e-15)
    return a / a.sum()

    stall = 0
    it = 0
    f_rev = f[::-1]

        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break

def lp_blend_step(f, mix, deadline):
    """Approximate LP target and blend toward it."""
    n = len(f)
    b = fft_conv(f, f)
    M = b.max()
    g = np.ones(n) / n
    f_rev = f[::-1]
    for _ in range(40):
        if time.monotonic() > deadline - 10.0:
            break
        c = fft_conv(f, g)
        viol = np.maximum(0.0, c - M)
        if viol.max() < 1e-15:
            break
        grad = fft_conv(viol, f_rev)[n - 1:2 * n - 1]
        grad = np.maximum(grad, 1e-30)
        g = g * np.exp(-0.8 * grad / grad.max())
        g = np.maximum(g, 1e-25)
        g = g / g.sum()
    blended = (1.0 - mix) * f + mix * g
    blended = np.maximum(blended, 1e-25)
    blended = blended / blended.sum()
    return blended
