import numpy as np
import time

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _grad_active(a, act):
    n = a.size
    g = np.zeros(n)
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g / max(1, len(act))

def _grad_soft(a, c, fs, gamma):
    n = a.size
    cl = c[:2 * n - 1]
    cx = cl.max()
    with np.errstate(over='ignore'):
        ec = np.exp(gamma * (cl - cx))
    s = ec.sum()
    if not np.isfinite(s) or s < 1e-300:
        ec = np.zeros_like(cl)
        ec[np.argmax(cl)] = 1.0
    else:
        ec /= s
    fa = np.fft.rfft(a, n=fs)
    fw = np.fft.rfft(ec, n=fs)
    return 2.0 * np.fft.irfft(fa.conj() * fw, n=fs)[:n]
