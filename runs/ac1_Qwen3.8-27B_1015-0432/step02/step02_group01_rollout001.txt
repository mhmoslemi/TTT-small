```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max(), b

def get_grad(f, b, beta):
    n = f.size
    M = b.max()
    avg = b.mean()
    spread = M - avg
    if spread < 1e-30:
        return None
    # Softmax weights: emphasize lags above average
    w = np.exp(beta * (b - avg) / spread)
    # Remove baseline (don't push down lags at average)
    w = w - 1.0
    w = np.maximum(w, 0.0)
    if w.max() < 1e-30:
        # Fallback: use raw softmax
        w = np.exp(beta * (b - avg) / spread)
    wmax = w.max()
    if wmax < 1e-30:
        return None
    w /= wmax
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n - 1: 2 * n - 1].copy()
    return np.maximum(grad, 0.0)

def run_opt(f0, deadline, rng, beta_lo, beta_hi, eta_lo, eta_hi, n_iter):
    f = np.maximum(f0.astype(np.float64), 1e-20)
    f /= f.sum()
    n = f.size
    best_f = f.copy()
    best_sc, _ = get_score(f)
    stall = 0
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        sc, b = get_score(f)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            f = best_f * np.exp(0.04 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            stall = 0
        # Geometric schedule
        frac = min(1.0, (it + 1) / n_iter)
        beta = beta_lo * (beta_hi / beta_lo) ** frac
        eta = eta_lo * (eta_hi / eta_lo) ** frac
        g = get_grad(f, b, beta)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
    return best_f, best_sc

def ramp(n, p, rev):
    x = np.arange(n, dtype=np.float64) + 1.0
    r = x ** p
    if rev:
        r = r[::-1]
    return r / r.sum()

def bell(n, c, w, p):
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    a = np.maximum(0.0, 1.0 - ((x - c) / w) ** 2)
    a **= p
    a = np.maximum(a, 1e-30)
    return a / a.sum()

    p /= p.sum()

    for pp in [0.3, 0.6, 1.0, 1.5, 2.5]:
        inits.append(ramp(n, pp, False))
        inits.append(ramp(n, pp, True))
    for cc, ww, pp in [(0.3, 0.8, 0.5), (0.5, 1.0, 0.7), (0.7, 0.6, 1.0), (0.5, 0.5, 2.0)]:
        inits.append(bell(n, cc, ww, pp))
    # Choose best

    for ini in inits:
        sc, _ = get_score(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()
    # Multi-phase optimization
    phases = [
        (3.0, 20.0, 0.3, 0.03, 30000),
        (15.0, 80.0, 0.1, 0.008, 30000),
        (50.0, 200.0, 0.04, 0.003, 30000),
        (150.0, 600.0, 0.012, 0.0008, 30000),
        (400.0, 1500.0, 0.004, 0.0003, 30000),

    for bl, bh, el, eh, ni in phases:
        if time.monotonic() > dl - 10.0:

        fo, sc = run_opt(best_f, dl - 5.0, rng, bl, bh, el, eh, ni)
        if sc < best_sc:
            best_sc = sc
            best_f = fo
    # Final validation
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs > 2.0:
        result = [1.0 / n] * n
    return result
```

```python
import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def get_score(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max(), b


def get_grad(f, b, beta):
    n = f.size
    M = b.max()
    avg = b.mean()
    spread = M - avg
    if spread < 1e-30:
        return None
    exponent = beta * (b - avg) / spread
    exponent -= exponent.max()
    w = np.exp(exponent)
    base = np.exp(-beta) if beta < 30.0 else 0.0
    w = np.maximum(w - base, 0.0)
    wmax = w.max()
    if wmax < 1e-30:
        w = np.exp(exponent)
        wmax = w.max()
        if wmax < 1e-30:
            return None
    w /= wmax
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n - 1:2 * n - 1].copy()
    return np.maximum(grad, 0.0)
```
