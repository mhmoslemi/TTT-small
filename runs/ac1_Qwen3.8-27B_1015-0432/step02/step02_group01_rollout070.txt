import numpy as np
import time


def _fc(a, b):
    sz = 1 << (len(a) + len(b) - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:len(a) + len(b) - 1]


def _sc(a):
    return 2.0 * a.size * _fc(a, a).max()


def _grd(a, b, k_top, beta):
    M = b.max()
    if M < 1e-30:
        return None
    thr = np.partition(b, b.size - k_top)[b.size - k_top]
    w = np.zeros_like(b)
    m = b >= thr
    w[m] = np.exp(beta * (b[m] / M - 1.0))
    s = w.sum()
    if s < 1e-30:
        w[m] = 1.0
        s = m.sum()
    w /= s
    gc = _fc(w, a[::-1])
    g = 2.0 * gc[a.size - 1:2 * a.size - 1].copy()
    return np.maximum(g, 0.0)


def _run(f, dl, rng, eta0, beta0, kfrac):
    n = f.size
    f = f.astype(np.float64).copy()
    f /= f.sum()
    bf = f.copy()
    bs = _sc(f)
    eta = eta0
    beta = beta0
    kk = max(3, int(n * kfrac))
    stall = 0
    for it in range(5000000):
        if time.monotonic() > dl:
            break
        b = _fc(f, f)
        sc = 2.0 * n * b.max()
        if sc < bs - 1e-15:
            bs = sc
            bf = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 3000:
                f = bf * np.exp(0.025 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                eta = eta0
                beta = max(5.0, beta * 0.4)
                continue
        g = _grd(f, b, kk, beta)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        ok = False
        for s_ in (1.0, 0.5, 0.25, 0.1, 0.05):
