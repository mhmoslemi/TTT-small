import scipy.sparse as sp

def _fft_conv(a):

    fs = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[:2*n-1]

def _subgrad_step(f, C, M, n, eta, frac):
    """One subgradient step on the active set."""
    act = np.where(C >= M * (1.0 - frac))[0]
    if len(act) > 60:

        act = act[ii]
    g = np.zeros(n)

def _fft_conv(a):

    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[:2 * n - 1]
