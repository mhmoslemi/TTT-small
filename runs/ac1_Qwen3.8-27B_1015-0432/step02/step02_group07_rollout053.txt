    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc

    # Phase 2: Refined topk pass
    if time.monotonic() < deadline - 10.0:
        a2 = best / s0
        t2 = deadline - 0.15 * (deadline - start)
        a2 = run_descent(a2, n, fft_size, fft_size2, t2, 10.0, 80.0, False)
        cand2 = np.clip(a2 * s0, 0.0, 1000.0)
        sc2 = float(evaluate_sequence(cand2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best, best_score = cand2, sc2

    # Phase 3: Additional smoothed pass with lower T
    if time.monotonic() < deadline - 10.0:
        a3 = best / s0
        t3 = deadline - 6.0
        a3 = run_descent(a3, n, fft_size, fft_size2, t3, 1.5, 15.0, True)
        cand3 = np.clip(a3 * s0, 0.0, 1000.0)
        sc3 = float(evaluate_sequence(cand3.tolist()))
        if np.isfinite(sc3) and sc3 < best_score:
            best, best_score = cand3, sc3

    # Phase 4: Local perturbations (finer scale)
    base = best.copy()
    while time.monotonic() < deadline - 2.0:
        cand = base.copy()
        nw = max(2, n // 40)
        for _ in range(int(rng.integers(1, 6))):
            st = int(rng.integers(0, max(1, n - nw)))
            e = min(n, st + nw)
            scale = np.exp(float(rng.normal(0.0, 0.015)))
            cand[st:e] *= scale
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score, base = sc, cand, cand
        elif sc < best_score + 1e-6 and time.monotonic() < deadline - 5.0:
            a_p = best / s0
            a_p = run_descent(a_p, n, fft_size, fft_size2, time.monotonic() + 3.0, 5.0, 30.0, True)
            cand_p = np.clip(a_p * s0, 0.0, 1000.0)
            sc_p = float(evaluate_sequence(cand_p.tolist()))
            if np.isfinite(sc_p) and sc_p < best_score:
                best, best_score, base = sc_p, cand_p, cand_p

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
