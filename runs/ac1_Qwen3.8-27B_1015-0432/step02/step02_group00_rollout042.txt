    s = f.sum()

        return np.ones(f.shape[0]) / f.shape[0]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

    n = f.shape[0]
    w = np.exp(T * (b - b.max()))
    w /= w.sum()
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n-1:2*n-1][::-1]

def hard_grad(f, b, thresh_frac, max_peaks):
    n = f.shape[0]
    M = b.max()
    thresh = M * thresh_frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_peaks:

    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n-1:2*n-1][::-1]
    return np.maximum(g, 0.0)

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def normalize(f):
    f = np.maximum(f, 1e-30)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def grad_soft(f, b, T):
    """Soft gradient: weight peaks by softmax(T*(b-M))."""
    n = f.shape[0]
    w = np.exp(T * (b - b.max()))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def grad_hard(f, b, frac, kmax):
    """Hard gradient: weight top-k peaks uniformly."""
    n = f.shape[0]
    M = b.max()
    peaks = np.where(b >= M * frac)[0]

    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]

    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.3, 0.7, 1.0, 1.5, 2.0, 3.0]:
        inits.append(normalize(d ** pw))
    inits.append(normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1.0)))))
    inits.append(normalize(0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1.0))) + 1e-12))
    for s in [0.15, 0.4]:
        r = np.ones(n) + s * rng.standard_normal(n)
        inits.append(normalize(np.maximum(0.5 * (r + r[::-1]), 0.01)))
    
    bf, bs = None, np.inf
    for ini in inits:
        s = sc(ini)
        if s < bs:
            bs, bf = s, ini.copy()
    
    f, s = opt(bf.copy(), min(dl, t0 + 300.0), rng, 60.0)
    if s < bs:
        bf, bs = f.copy(), s
