    t0 = time.monotonic()
    hard_deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()

    x = np.arange(n, dtype=np.float64)
    inits = [p.copy()]

    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(np.maximum(h, 1e-15) / np.maximum(h, 1e-15).sum())

    for a1, a2 in [(1.5, 1.5), (2.5, 0.8), (0.8, 2.5), (3.0, 0.4), (1.0, 3.0)]:
        e = np.exp(-a1 * x / n) + np.exp(-a2 * (n - 1 - x) / n)
        inits.append(e / e.sum())

    for sh in [-0.12, 0.12, -0.2, 0.2]:
        c = np.cos(np.pi * (x / n + sh)) ** 2
        inits.append(np.maximum(c, 1e-15) / np.maximum(c, 1e-15).sum())

    for s in [0.08, 0.2, 0.4]:
        t = p * np.exp(s * rng.standard_normal(n))
        inits.append(np.maximum(t, 1e-15) / np.maximum(t, 1e-15).sum())

    best_f = None
    best_score = np.inf
    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = ini.copy()

    if best_f is None:
        best_f = p.copy()
        best_score = fast_score(p)

    total = hard_deadline - t0

    # Phase 1: broad search
    d1 = t0 + 0.30 * total
    f1, sc1 = optimize(best_f, d1, rng, 40, 150, 0.12)
    if sc1 < best_score:
        best_score, best_f = sc1, f1.copy()

    # Phase 2: refinement
    d2 = t0 + 0.55 * total
    f2, sc2 = optimize(best_f, d2, rng, 80, 400, 0.06)
    if sc2 < best_score:
        best_score, best_f = sc2, f2.copy()

    # Phase 3: high-beta fine tuning
    d3 = t0 + 0.80 * total
    f3, sc3 = optimize(best_f, d3, rng, 200, 1500, 0.03)
    if sc3 < best_score:
        best_score, best_f = sc3, f3.copy()

    # Phase 4: exact peak shaving
    d4 = t0 + 0.92 * total
    f4, sc4 = peak_shave(best_f, d4, rng, iters=100)
    if sc4 < best_score:
        best_score, best_f = sc4, f4.copy()

    # Phase 5: final polish with tiny perturbations
    d5 = hard_deadline
    if time.monotonic() < d5:
        for _ in range(80):
            if time.monotonic() > d5:
                break
            trial = best_f * np.exp(0.002 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-30)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score, best_f = sc, trial.copy()

    # Mirror-mix attempts
    if time.monotonic() < d5:
        for mix in [0.05, 0.1, 0.15]:
            if time.monotonic() > d5:
                break
            trial = (1.0 - mix) * best_f + mix * best_f[::-1]
            trial = np.maximum(trial, 1e-30)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score, best_f = sc, trial.copy()

    result = [float(v) for v in best_f]
    final = evaluate_sequence(result)
    if not np.isfinite(final):
        result = [1.0 / n] * n
        final = evaluate_sequence(result)

    return result
