def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def compute_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    # Soft-max weighting on convolution peaks
    w = np.exp(beta * (b / M - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:

        w = np.zeros_like(b)
        w[idx] = 1.0

(sum_k w_k f_{k-i} = lambda)

For a single peak at lag m: f_{i+m} = lambda for all i where f_i > 0. That means f is constant (for large enough support), which gives the uniform solution.

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def get_grad(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30:
        k = max(1, b.size // 100)
        idx = np.argsort(b)[-k:]
        w = np.zeros_like(b)
        w[idx] = 1.0
        ws = float(k)
    w /= ws
    gc = fft_conv(w, f[::-1])
    grad = gc[f.size - 1:2 * f.size - 1].copy()
    return np.maximum(grad, 0.0)

def optimize(f0, deadline, rng, beta0, eta0, max_steps=200000):
    n = f0.size
    f = f0.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
