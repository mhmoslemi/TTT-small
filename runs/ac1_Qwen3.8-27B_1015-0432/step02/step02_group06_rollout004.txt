def lp_step(f, M, g_init, iters=200):
    """Find g maximizing sum(g) s.t. g >= 0, ||f*g||_inf <= M."""
    n = len(f)
    g = g_init.copy()
    g = np.maximum(g, 0)
    
    for _ in range(iters):
        # Check constraint violation
        fg = fft_conv(f, g)
        k_star = int(np.argmax(fg))
        violation = fg[k_star] - M
        
        if violation < 1e-10:
            break
        
        # Project: reduce g to satisfy (f*g)[k_star] <= M
        # (f*g)[k_star] = sum_i f[i] * g[k_star - i]
        # The "active" indices are i where g[k_star - i] > 0
        # Simple projection: scale down g at positions contributing to the violation
        
        # The gradient of (f*g)[k_star] w.r.t. g[j] is f[k_star - j] (for valid j)
        # So we want to reduce g[j] proportionally to f[k_star - j]
        
        # For the constraint: sum_i f[i]*g[k_star-i] <= M
        # The "shadow" of g at position k_star in the convolution with f:
        # Let h[j] = f[k_star - j] for j = max(0, k_star-n+1) to min(k_star, n-1)
        # Then constraint is h @ g <= M (with h having zeros outside range)
        
        # Project g onto h @ g <= M, g >= 0:
        # If h @ g > M, scale g: g_new = g - alpha * h where alpha < 0
        # Or: g_new = g * (1 - t) where t chosen so h @ (g*(1-t)) = M
        # t = 1 - M / (h @ g)
        
        # But we also need g >= 0, so:
        h = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        for i in range(lo, hi + 1):
            if i < n:
                h[i] = f[k_star - i]  # Wait, need to be careful
        # Actually: (f * g)[k] = sum_i f[i] * g[k-i]
        # d/dg[j] (f*g)[k] = f[k-j] if 0 <= k-j < n, else 0
        # So h[j] = f[k_star - j] if 0 <= k_star - j < n
        
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                h[j] = f[idx]
        
        h @ g is the violation metric
        current = np.dot(h, g)
        if current > M + 1e-15:
            scale = M / (current + 1e-30)
            g = g * scale  # This works only if h is uniform...

        # Minimize ||g - g_old||^2 s.t. h @ g <= M, g >= 0
        # This is a QP. Simple approach:
        # If h @ g > M and max(h) > 0:
        #   g_new = g - lambda * h where lambda = max(0, (h@g - M) / (h@h))
