import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:n + m - 1]

def sc(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * len(f) * fft_conv(f, f).max() / (s * s))

def grad_smooth(f, C, T):
    n = f.size
    p = np.exp((C - C.max()) / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    return 2.0 * np.fft.irfft(np.fft.rfft(p, N) * np.conj(np.fft.rfft(f, N)), N)[:n]
