                 (1, 0.7), (2, 0.0), (2, 0.5), (2, -0.5), (3, 0.0), (3, 0.5)]:
        inits.append(_mi(n, k, p))
    inits.append(np.ones(n) / n)

    bf, bs = None, np.inf
    for ini in inits:
        sc = _sc(ini)
        if np.isfinite(sc) and sc < bs:
            bs, bf = sc, ini.copy()
    if bf is None:
        bf, bs = np.ones(n) / n, 2.0

    f1, s1 = _opt(bf, min(t0 + 600, ddl - 200), rng, 1.0, 0.15)
    if s1 < bs:
        bs, bf = s1, f1

    if time.monotonic() < ddl - 120:
        f2, s2 = _opt(bf, min(t0 + 900, ddl - 80), rng, 0.05, 0.06)
        if s2 < bs:
            bs, bf = s2, f2

    for _ in range(4):
        if time.monotonic() > ddl - 60:
            break
        trial = bf * np.exp(0.04 * rng.standard_normal(n))
        trial = np.maximum(trial, 1e-12)
        trial /= trial.sum()
        d3 = min(time.monotonic() + 35, ddl - 30)
        f3, s3 = _opt(trial, d3, rng, 0.15, 0.1)
        if s3 < bs:
            bs, bf = s3, f3

    if time.monotonic() < ddl - 10:
        f4, s4 = _opt(bf, ddl - 5, rng, 0.01, 0.015)
        if s4 < bs:
            bs, bf = s4, f4

    result = [float(x) for x in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0] * n
    return result
