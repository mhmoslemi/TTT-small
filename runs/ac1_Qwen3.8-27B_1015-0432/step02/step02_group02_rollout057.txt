                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 350, 'gtol': 1e-15,
                                        'maxls': 30, 'ftol': 1e-20})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = _normalize(nx, scale)
                sc = _eval(nx)
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Third pass: very fine tau
    if time.monotonic() < dl - 80.0:
        cur = best.copy()
        for tv in [5e-4, 2e-4, 1e-4]:
            if time.monotonic() > dl - 40.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 250, 'gtol': 1e-15,
                                        'maxls': 30, 'ftol': 1e-20})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = _normalize(nx, scale)
                sc = _eval(nx)
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Local search: multiplicative block noise + targeted peak shaving
    while time.monotonic() < dl - 5.0:
        cd = cur.copy()
        mode = rng.random()
        if mode < 0.6:
            nw = int(rng.integers(max(1, n // 60), max(2, n // 12)))
            s = int(rng.integers(0, n - nw))
            cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.035)))
            if rng.random() < 0.2:
                cd *= float(np.exp(rng.normal(0.0, 0.01)))
        elif mode < 0.8:
            # Targeted peak shave
            c = _conv(cd)
            ks = int(c.argmax())
            lo = max(0, ks - n + 1)
            hi = min(n, ks + 1)
            contribs = []
            for i in range(lo, hi):
                j = ks - i
                if 0 <= j < n:
                    contribs.append((i, j, cd[i] * cd[j]))
            contribs.sort(key=lambda t: -t[2])
            if len(contribs) >= 2:
                i1, j1, _ = contribs[0]
                if i1 != j1:
                    red = 0.04 * min(cd[i1], cd[j1])
                    cd[i1] -= red
                    cd[j1] -= red
                    idx = int(rng.integers(0, n))
                    while idx in (i1, j1):
                        idx = int(rng.integers(0, n))
                    cd[idx] += red * 2.0
        else:
            # Spike: add small amount at random, subtract from another
            i_add = int(rng.integers(0, n))
            i_sub = int(rng.integers(0, n))
            while i_sub == i_add:
                i_sub = int(rng.integers(0, n))
            amt = 0.02 * max(cd.max(), 1e-6)
            cd[i_add] += amt
            cd[i_sub] = max(0.0, cd[i_sub] - amt)

        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            cd = _normalize(cd, scale)
            sc2 = _eval(cd)
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2
                cur = cd.copy()
        if rng.random() < 0.04:
            cur = best.copy()

    final_sc = _eval(best)
    if not np.isfinite(final_sc):
        best = a0.copy()

    return [float(v) for v in best]
