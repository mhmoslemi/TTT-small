            g[lo:hi + 1] += 2.0 * f[ki - hi:ki - lo + 1][::-1]
    return g

def mirror_opt(f, deadline, rng, eta0=0.12):
    n = f.size
    f = np.maximum(f.astype(np.float64).copy(), 1e-22)
    f /= f.sum()
    eta, T = eta0, 1.5
    best_f, best_sc = f.copy(), fast_score(f)
    no_imp = 0
    for it in range(300000):
        if time.monotonic() > deadline - 4.0:
            break
        C = fft_conv(f, f)
        sc = 2.0 * n * C.max() / (f.sum() ** 2)
        if sc < best_sc - 1e-12:
            best_sc, best_f, no_imp = sc, f.copy(), 0
        else:
            no_imp += 1
        frac = it / max(it + 1, 200)
        T = max(0.04, 1.5 * (0.35) ** (it / 200.0))
        if frac < 0.5:
            g = grad_smooth(f, C, T)
        elif frac < 0.8:
            g = 0.5 * grad_smooth(f, C, T) + 0.5 * grad_topk(f, C, min(3, n // 100))
        else:
            g = grad_topk(f, C, min(5, n // 100))
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-25)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            break
        f /= s
        if it % 8 == 0:
            ns = fast_score(f)
            eta = min(eta * 1.07, 2.5) if ns < sc - 1e-10 else max(eta * 0.82, 0.003)
        if no_imp > 180:
            f = best_f * np.exp(0.04 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp, eta, T = 0, 0.12, 1.5
    return best_f, best_sc

def lp_greedy_direction(f, C):
    n = f.size
    M = C.max() * 1.01
    f_safe = np.maximum(f, 1e-15)
    g = 1.0 / f_safe
    g /= g.sum()
    for _ in range(3):
        h = fft_conv(f, g)
        mask = h > M
        if not mask.any():
            break
        idx = np.where(mask)[0]
        for ki in idx[:n]:
            lo, hi = max(0, ki - n + 1), min(ki, n - 1)
            if lo <= hi:
                g[lo:hi + 1] *= 0.5
        g = np.maximum(g, 0)
        if g.sum() < 1e-15:
            break
        g /= g.sum()
    return 0.4 * f + 0.6 * g

def lp_refine(f, deadline, rng):
    best_f, best_sc = f.copy(), fast_score(f)
    for _ in range(20):
        if time.monotonic() > deadline - 8.0:
            break
        C = fft_conv(best_f, best_f)
        d = lp_greedy_direction(best_f, C)
        for t in [0.1, 0.25, 0.4, 0.6, 0.8]:
            c = np.maximum((1 - t) * best_f + t * d, 1e-20)
            c /= c.sum()
            sc = fast_score(c)
            if np.isfinite(sc) and sc < best_sc:
                best_sc, best_f = sc, c
        to = time.monotonic() + min(40.0, deadline - time.monotonic() - 10.0)
        if to > time.monotonic():
            f2, sc2 = mirror_opt(best_f, to, rng, 0.08)
            if sc2 < best_sc:
                best_sc, best_f = sc2, f2
    return best_f, best_sc

def make_init(n, kind, p1=0.0, p2=0.0):
    j = np.arange(n, dtype=np.float64)
    xc = 2.0 * np.pi * j / n
    if kind == 0:
        w = 1 - 0.42 * np.cos(xc) + 0.21 * np.cos(2 * xc) - 0.07 * np.cos(3 * xc)
    elif kind == 1:
        w = (j + 1) ** p1 * (n - j) ** p2
    else:
        w = np.exp(-0.5 * ((j - n * p1) / (n * 0.25)) ** 2)
    if abs(p1 - round(p1)) > 0.01 or abs(p2 - round(p2)) > 0.01:
        if kind == 0:
            w *= np.exp(p1 * (2 * j / max(n - 1, 1) - 1))
    return np.maximum(w, 1e-10)

def diversify(f, rng, n):
    out = []
    for s in [0.03, 0.08, 0.15]:
        t = np.maximum(f * np.exp(s * rng.standard_normal(n)), 1e-20)
        out.append(t / t.sum())
    for a in [-0.6, 0.6, -1.2]:
        t = np.maximum(f * np.exp(a * (2 * np.arange(n) / max(n - 1, 1) - 1)), 1e-20)
        out.append(t / t.sum())
    return out

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 12.0
    n = len(height_sequence_1)
    inits = []
    for a in [0.0, 0.5, 1.0, -0.5, 1.5, -1.0, 2.0, 3.0]:
        inits.append(make_init(n, 0, a))
    for a, b in [(0, 0), (0.1, 0), (0.5, 0), (1, 0), (0, 1), (0.3, 0.5), (1, 0.3), (0.5, 0.5)]:
        inits.append(make_init(n, 1, a, b))
    for c, w in [(0.3, 0.3), (0.5, 0.2), (0.4, 0.25), (0.6, 0.25), (0.5, 0.35)]:
        inits.append(make_init(n, 2, c, w))
    inits.append(np.ones(n) / n)
    pp = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if pp.size == n and pp.sum() > 1e-10:
        inits.append(pp / pp.sum())
    best_f, best_sc = None, np.inf
    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_f = sc, ini.copy()
    if best_f is None:
        best_f, best_sc = np.ones(n) / n, 2.0
    phase = 0
    while time.monotonic() < dl - 25.0:
        phase += 1
        to = time.monotonic() + min(280.0, dl - time.monotonic() - 12.0)
        f1, sc1 = mirror_opt(best_f, to, rng)
        if sc1 < best_sc:
            best_sc, best_f = sc1, f1
        if time.monotonic() > dl - 50.0:
            break
        to2 = time.monotonic() + min(150.0, dl - time.monotonic() - 12.0)
        f2, sc2 = lp_refine(best_f, to2, rng)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2
        if phase % 2 == 0 and time.monotonic() < dl - 80.0:
            for tr in diversify(best_f, rng, n):
                sc_tr = fast_score(tr)
                if sc_tr < best_sc:
                    best_sc, best_f = sc_tr, tr
                to3 = time.monotonic() + min(50.0, dl - time.monotonic() - 12.0)
                if to3 <= time.monotonic():
                    break
                f3, sc3 = mirror_opt(tr, to3, rng, 0.08)
                if sc3 < best_sc:
                    best_sc, best_f = sc3, f3
                if time.monotonic() > dl - 40.0:
                    break
    while time.monotonic() < dl - 5.0:
        imp = False
        for tr in diversify(best_f, rng, n):
            sc = fast_score(tr)
            if np.isfinite(sc) and sc < best_sc:
                best_sc, best_f, imp = sc, tr, True
        if not imp:
            break
    result = [float(x) for x in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [float(1.0)] * n
    return result
