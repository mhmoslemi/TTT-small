def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def compute_grad(f, w):
    """Compute gradient of sum(w * conv(f,f)) w.r.t. f using FFT."""
    n = f.size
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad1 = gc[n-1:2*n-1].copy()
    gc2 = fft_conv(w, f)
    grad2 = gc2[:n].copy()
    return grad1 + grad2
