```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v, kind='mergesort')[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

This is the standard algorithm. O(n log n) per call.

Actually, for the AC1 bound, the relevant quantity is max b_k. The sum of all b_k is (sum a)² = 1. There are 2n-1 values. The average is 1/(2n-1). The peak b_{n-1} is typically the largest. For a flat distribution, b_{n-1} ≈ 1/(2n-1) ≈ 1/(2n). But we also have b_0 = a_0², b_1 = 2a_0 a_1, etc.

For the uniform sequence: b_k = (k+1)/n² for k = 0, ..., n-1 and (2n-1-k)/n² for k = n, ..., 2n-2. Peak at k = n-1: b_{n-1} = n/n² = 1/n. Score = 2n * 1/n = 2.

For a "flat" sequence (where b_k ≈ 1/(2n) for all k): score ≈ 2n * 1/(2n) = 1. But this is impossible because b_0 = a_0² and sum b_k = 1.

The theoretical minimum is related to the Welch bound or similar. For our problem, the target is 1.5027, meaning max b_k ≈ 0.7514/n.

Let me think about whether the subgradient method can reach this. The issue is that the max function is non-smooth, and the subgradient is a convex combination of the active gradients. The subgradient method converges slowly for non-smooth functions.

One improvement: use a "smoothed max" or "soft max" approach. Replace max(b_k) with logsumexp(b_k / tau) * tau for small tau. This gives a smooth approximation with a well-defined gradient everywhere.

Soft max: S(a) = tau * log(sum_k exp(b_k / tau))
∂S/∂a_i = sum_k (exp(b_k/tau) / sum_j exp(b_j/tau)) * ∂b_k/∂a_i

This is a weighted average of all the ∂b_k/∂a_i, with weights proportional to exp(b_k/tau). For small tau, this concentrates on the max. For larger tau, it distributes more evenly.

The advantage is that the gradient is smooth and dense (no need to track the active set). The disadvantage is that it's only an approximation.

I can use a "continuation" approach: start with large tau (smooth, fast convergence) and gradually decrease tau (sharper, slower convergence). This is like a homotopy method.

```python
import numpy as np
import time

def _fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _proj_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    idx = np.where(u * np.arange(1, n+1) > css - 1)[0]
    if len(idx) == 0:
        return np.ones(n) / n
    r = idx[-1]
    theta = (css[r] - 1) / (r + 1)
    return np.maximum(v - theta, 0)

def _soft_grad(a, c, n, cs, tau):
    w = np.exp((c - c.max()) / tau)
    w /= w.sum() + 1e-30
    Wr = w[::-1]
    h = np.fft.irfft(np.fft.rfft(Wr, cs) * np.fft.rfft(a, cs), cs)
    return 2.0 * h[n-1:2*n-1][::-1] - w[::2] * a

def _hard_grad(a, c, n):
    M = c.max()
    tol = max(M * 1e-7, 1e-15)
    act = np.where(c >= M - tol)[0]
    g = np.zeros(n)
    for k in act:
        k = int(k)
        lo, hi = max(0, k-n+1), min(n-1, k)
        if lo > hi:
            continue
        seg = a[k-hi:k-lo+1][::-1]
        g[lo:hi+1] += 2.0 * seg
        if k % 2 == 0 and lo <= k//2 <= hi:
            g[k//2] -= seg[k//2 - lo]
    return g

def _eval(a, s0, ev_fn):
    c = np.clip(a * s0, 0.0, 1000.0)
    if c.sum() < 0.01:
        return np.inf
    s = ev_fn(c.tolist())
    return s if np.isfinite(s) else np.inf

def _descent(a, n, fs, gcs, dl, ba, bs, s0):
    vm = np.zeros(n)
    beta = 0.8
    lr = 0.3
    tau_cur = 1.0 / (2.0 * n) * 5.0  # initial tau
    stall = 0
    pM = np.inf
    while time.monotonic() < dl:
        conv = _fft_conv(a, fs)
        c = conv[:2*n-1]
        M = float(c.max())
        if M < pM - 1e-14:
            stall = 0
        else:
            stall += 1
        pM = M
        if stall > 3000:
            break
        # Choose gradient method based on tau
        if tau_cur > 1e-6 / (2*n):
            g = _soft_grad(a, c, n, gcs, tau_cur)
        else:
            g = _hard_grad(a, c, n)
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        vm = beta * vm + (1 - beta) * g / gn
        # Backtracking line search
        ok = False
        for s in range(15):
            a_t = _proj_simplex(a - (lr / (2.0**s)) * vm)
            ct = _fft_conv(a_t, fs)[:2*n-1]
            Mt = float(ct.max())
            if Mt < M:
                a, M, conv = a_t, Mt, ct
                ok = True
                break
        if not ok:
            a = _proj_simplex(a - 0.01 * lr * vm)
            conv = _fft_conv(a, fs)
            M = float(conv[:2*n-1].max())
        # Decay tau
        tau_cur *= 0.9995
        sc = 2.0 * n * M
        if sc < bs:
            bs, ba = sc, a.copy()
        lr = max(0.02, lr * 0.9997)
        beta = min(0.94, beta + 5e-6)
    return ba, bs
```

I compute this as:
Wr = w[::-1] (length 2n-1, Wr_k = w_{2n-2-k})
conv_result = conv(Wr, a) (linear convolution, length 3n-2)
conv_result_m = sum_{j} Wr_j * a_{m-j} = sum_j w_{2n-2-j} * a_{m-j}

Let k = 2n-2-j, j = 2n-2-k:
conv_result_m = sum_k w_k * a_{m-(2n-2-k)} = sum_k w_k * a_{m-2n+2+k}

We want h_i = sum_k w_k * a_{k-i}, so we need m-2n+2+k = k-i, i.e., m = 2n-2-i.

So h_i = conv_result_{2n-2-i}.

For i=0: h_0 = conv_result_{2n-2}
For i=n-1: h_{n-1} = conv_result_{n-1}

So h = conv_result[n-1 : 2n-1][::-1]

conv_result has length 3n-2 (indices 0 to 3n-3).
n-1 ≥ 0 ✓
2n-2 ≤ 3n-3 for n ≥ 1 ✓

So the slice conv_result[n-1 : 2n-1] has n elements (indices n-1 to 2n-2).
Reversed: [conv_result_{2n-2}, conv_result_{2n-3}, ..., conv_result_{n-1}]
= [h_0, h_1, ..., h_{n-1}] ✓

Now, the FFT size for this convolution: Wr has length 2n-1, a has length n. The linear convolution has length (2n-1) + n - 1 = 3n - 2. So the FFT size should be a power of 2 >= 3n - 2.

gcs = 1 << (3*n - 1).bit_length()
