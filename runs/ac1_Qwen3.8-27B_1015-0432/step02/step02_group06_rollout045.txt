def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

import numpy as np
import time


def fft_conv(a, b):
    n, m = a.size, b.size
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * f.size * b.max() / (s * s))


def make_inits(n, rng):
    x = np.arange(n, dtype=np.float64)
    u = 2.0 * (x + 0.5) / n - 1.0
    inits = []
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(np.maximum(h, 1e-15) / np.maximum(h, 1e-15).sum())
    for p in [1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        w = np.maximum(1.0 - np.abs(u) ** p, 1e-15)
        inits.append(w / w.sum())
    for c in [1.0, 2.0, 3.0, 5.0]:
        w = np.exp(-0.5 * c * u ** 2)
        inits.append(w / w.sum())
    for r in [0.5, 1.0, 1.5]:
        w = np.exp(-r * x / n) + 0.4 * np.exp(-r * (n - 1 - x) / n)
        inits.append(w / w.sum())
    for sh in [-0.2, -0.1, 0.1, 0.2]:
        w = np.maximum(np.cos(np.pi * (u + sh)) ** 2, 1e-15)
        inits.append(w / w.sum())
    for s in [0.15, 0.4]:
        w = np.ones(n) * np.exp(s * rng.standard_normal(n))
        inits.append(w / w.sum())
    return inits

        M = b.max()
        w = np.exp(beta * (b / M - 1.0))
        w /= w.sum()
        f_rev = f[::-1]
        gc = fft_conv(w, f_rev)
        grad = gc[n - 1:2 * n - 1].copy()
        grad = np.maximum(grad, 1e-25)
        gmax = grad.max()
        if gmax < 1e-25:
            break
        f *= np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        it += 1
        if it % 15 == 0:
            sc = fast_score(f)

                stall = 0

                stall = 0
