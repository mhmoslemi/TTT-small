import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2 * len(f) * fft_conv(f, f).max() / (s * s))


def xcorr(p, a, N):
    return np.fft.irfft(np.fft.rfft(p, N) * np.conj(np.fft.rfft(a, N)), N)


def soft_weights(C, T):
    cs = C - C.max()
    p = np.exp(cs / T)
    return p / p.sum()


def mm_opt(f0, dl, rng):
    n = f0.size
    N = 1 << (3 * n - 1).bit_length()
    f = f0.astype(np.float64).copy()
    eta, T = 0.12, 3.0
    best, bsc = f.copy(), fast_score(f)
    stall = 0
    for it in range(500000):
        if time.monotonic() > dl - 3.0:
            break
        C = fft_conv(f, f)
        sc = 2.0 * n * C.max() / (f.sum() ** 2)
        if sc < bsc - 1e-12:
            best, bsc, stall = f.copy(), sc, 0
        else:
            stall += 1
        p = soft_weights(C, T)
        g = 2.0 * xcorr(p, f, N)
        gm = np.abs(g).max()
        if gm < 1e-30:
            break
        f *= np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        if it % 16 == 0:
            ns = fast_score(f)
            eta = min(eta * 1.04, 1.5) if ns < sc - 1e-10 else max(eta * 0.82, 0.008)
        if it % 200 == 0:

        if it % 400 == 200:
            nf = np.maximum(f ** 0.7, 1e-30)

            f = np.maximum(f, 1e-20)
            f /= f.sum()
            stall, eta, T = 0, 0.12, 3.0
    return best, bsc


def gen_inits(n, parent):
    out = []
    pp = np.clip(np.asarray(parent, dtype=np.float64), 0.0, 1000.0)
    if pp.size == n and np.all(np.isfinite(pp)) and pp.sum() > 1e-10:
        out.append(pp / pp.sum())
    j = np.arange(n, dtype=np.float64)
    x = np.linspace(-np.pi, np.pi, n)
    for co in ([1.0, -0.42, 0.21, -0.07], [1.0, -0.46, 0.36, -0.13, 0.014]):
        w = np.zeros(n)
        for ci in range(len(co)):
            w += co[ci] * np.cos(ci * x)
        w = np.maximum(w, 1e-12)
        out.append(w / w.sum())
    w = 0.54 + 0.46 * np.cos(np.linspace(0, np.pi, n))
    out.append(w / w.sum())
    out.append(np.ones(n) / n)
    for a in (1.0, 2.0, 0.5):
        w = (j + 1.0) ** a * (n - j) ** a
        out.append(w / w.sum())
    for sg in (0.15, 0.3, 0.45):
        w = np.exp(-0.5 * ((j - n / 2.0) / (sg * n)) ** 2)
        out.append(w / w.sum())
    return out


def spec_perturb(f, rng, al):
    n = f.size
    N = 1 << (n - 1).bit_length()
    F = np.fft.rfft(f, N)
    ph = np.angle(F) + al * rng.standard_normal(len(F))
    F2 = np.abs(F) * np.exp(1j * ph)
    F2[0] = F[0]
    nf = np.fft.irfft(F2, N)[:n]
    nf = np.maximum(nf, 1e-20)
    return nf / nf.sum()


def diversify(f, rng, j):
    n = f.size
    out = []
    for sc in (0.03, 0.1):
        t = f * np.exp(sc * rng.standard_normal(n))
        t = np.maximum(t, 1e-20)
        out.append(t / t.sum())
    for al in (1.5, 2.5):
        t = np.maximum(f ** (1.0 / al), 1e-20)
        out.append(t / t.sum())
    for a in (-0.4, 0.4):
        s = np.exp(a * (2.0 * j / max(n - 1, 1) - 1.0))
        t = np.maximum(f * s, 1e-20)
        out.append(t / t.sum())
    out.append(spec_perturb(f, rng, 0.3))
    out.append(spec_perturb(f, rng, 0.8))
    return out

    N = 1 << (3 * n - 1).bit_length()
    best, bsc = f.copy(), fast_score(f)
    for _ in range(2000):

            break
        C = fft_conv(best, best)
        ks = int(np.argmax(C))
        lo, hi = max(0, ks - n + 1), min(ks, n - 1)
        if lo > hi:
            break
        g = np.zeros(n)
        g[lo:hi + 1] = 2.0 * best[ks - hi:ks - lo + 1][::-1]
        p = soft_weights(C, 0.5)

        nf /= nf.sum()

        if sc < bsc - 1e-12:
            bsc, best = sc, nf
        else:
            break
    return best, bsc
