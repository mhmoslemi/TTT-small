    bounds_list = [(0.0, 1000.0)] * n

    # Main tau annealing loop
    taus = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.025, 0.01, 0.005, 0.002, 0.001]
    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 90.0:
            break
        iters = 1000 if i < 3 else 600 if i < 7 else 350
        try:
            res = minimize(_obj_grad, cur, args=(tv,), method='L-BFGS-B',
                          bounds=bounds_list,
                          options={'maxiter': iters, 'gtol': 1e-11,
                                   'maxls': 50, 'ftol': 1e-16})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _normalize(nx, scale)
            sc = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # Second refinement pass
    if time.monotonic() < dl - 70.0:
        cur = best.copy()
        for tv in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if time.monotonic() > dl - 55.0:
                break
            try:
                res = minimize(_obj_grad, cur, args=(tv,), method='L-BFGS-B',
                              bounds=bounds_list,
                              options={'maxiter': 500, 'gtol': 1e-11,
                                       'maxls': 40, 'ftol': 1e-16})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _normalize(nx, scale)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Edge-shrink refinement: reduce endpoints to lower c[0], c[2n-2]
    if time.monotonic() < dl - 40.0:
        for shrink_try in range(5):
            if time.monotonic() > dl - 35.0:
                break
            for edge_frac in [0.02, 0.04, 0.06]:
                cd = best.copy()
                k = max(1, int(n * edge_frac / 2))
                taper = np.linspace(1.0, 0.3, k)
                cd[:k] *= taper
                cd[-k:] *= taper[::-1]
                cd = np.clip(cd, 0.0, 1000.0)
                sc2 = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc2) and sc2 < best_sc:
                    best = cd.copy()
                    best_sc = sc2
                    cur = cd.copy()

    # Final local random search
    while time.monotonic() < dl - 3.0:
        cd = cur.copy()
        nw = int(rng.integers(1, max(2, n // 20)))
        s = int(rng.integers(0, n - nw))
        cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.03)))
        if rng.random() < 0.08:
            cd *= float(np.exp(rng.normal(0.0, 0.008)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2
                cur = cd.copy()
        if rng.random() < 0.04:
            cur = best.copy()

    # Final edge taper attempt
    for eps in [0.5, 0.7, 0.85]:
        cd = best.copy()
        k = max(1, int(n * 0.03))
        taper = np.linspace(1.0, eps, k)
        cd[:k] *= taper
        cd[-k:] *= taper[::-1]
        cd = np.clip(cd, 0.0, 1000.0)
        sc2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(sc2) and sc2 < best_sc:
            best = cd.copy()
            best_sc = sc2

    return [float(v) for v in best]
