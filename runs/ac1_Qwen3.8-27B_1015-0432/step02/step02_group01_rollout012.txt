def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()


def grad_dir(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros_like(b)
        idx = np.argsort(b)[-max(1, b.size // 50):]
        w[idx] = 1.0
        ws = w.sum()
    w /= ws
    gc = fft_conv(w, f[::-1])
    g = gc[f.size - 1:2 * f.size - 1]
    return np.maximum(g, 0.0)


def run_opt(f, deadline, rng, beta, eta, max_it, b_g, e_d):
    n = f.size
    f = f.astype(np.float64).copy() / f.sum()
    bf = f.copy()
    bs = score_norm(f)
    stall = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        s = 2.0 * n * b.max()
        if s < bs - 1e-13:
            bs = s
            bf = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 400:
                f = bf * np.exp(0.05 * rng.standard_normal(n))
                f = np.maximum(f, 1e-18)
                f /= f.sum()
                stall = 0
                continue
        g = grad_dir(f, b, beta)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        bt = f.copy()
        bts = s
        for m in (0.6, 1.0, 1.6):
            t = f * np.exp(-eta * m * g)
            t = np.maximum(t, 1e-18)
            t /= t.sum()
            ts = 2.0 * n * fft_conv(t, t).max()
            if ts < bts:
                bts = ts
                bt = t
        f = bt
        if it % 400 == 0 and it > 0:
            beta *= b_g
            eta *= e_d
    return bf, bs


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n, dtype=np.float64)
    p /= p.sum()

    x = np.arange(n, dtype=np.float64)
    u = x / max(n - 1, 1)
    c1 = np.cos(np.pi * (x + 0.5) / n)
    c1 = c1 * c1 + 1e-15
    c1 /= c1.sum()
    a1 = np.maximum(1.0 - (2.0 * u - 1.0) ** 2, 1e-15)
    a1 = a1 ** 1.3
    a1 /= a1.sum()

    inits = [p, p[::-1].copy(), c1, a1, (p + p[::-1]) * 0.5]
    bf, bs = inits[0], np.inf
    for ini in inits:
        s = score_norm(ini)
        if s < bs:
            bs = s
            bf = ini.copy()

    phases = [
        (10.0, 0.08, 1.04, 0.99),
        (25.0, 0.04, 1.05, 0.99),
        (50.0, 0.02, 1.05, 0.99),
        (100.0, 0.01, 1.06, 0.99),
        (200.0, 0.005, 1.05, 0.995),
        (400.0, 0.002, 1.04, 0.997),
        (600.0, 0.001, 1.03, 0.998),
    ]
    for b0, e0, bg, ed in phases:
        if time.monotonic() > dl - 30:
            break
        fo, so = run_opt(bf, dl - 10, rng, b0, e0, 150000, bg, ed)
        if so < bs:
            bs = so
            bf = fo.copy()

    if time.monotonic() < dl - 5:
        flip = bf[::-1].copy()
        for t in (0.2, 0.4, 0.5, 0.6, 0.8):
            bl = (1.0 - t) * bf + t * flip
            bl = np.maximum(bl, 0.0)
            if bl.sum() < 1e-10:
                continue
            bl /= bl.sum()
            s_b = score_norm(bl)
            if s_b < bs:
                bs = s_b
                bf = bl.copy()

    res = [float(max(0.0, min(1000.0, v))) for v in bf]
    if not np.isfinite(evaluate_sequence(res)):
        res = [1.0 / n] * n
    return res
