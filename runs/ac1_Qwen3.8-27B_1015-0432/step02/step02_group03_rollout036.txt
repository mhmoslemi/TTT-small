import numpy as np
import time

def _fc(a, fs):
    """Full autoconvolution via FFT."""
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, lags):
    """Subgradient of max C at given lags, averaged."""
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    m = len(lags)
    if m:
        g /= m
    return g

def _desc(a, fs, ts, eta, beta, dl, frac, cap, smx, multi, rng):
    """Subgradient descent on 2n*max(C)/S^2."""
    n = a.size
    mom = np.zeros(n)
    bst = a.copy()
    bsc = 2.0 * n * float(_fc(bst, fs).max()) / (bst.sum()**2)
    stall = 0
    psc = bsc
    for _ in range(600000):
        if time.monotonic() > dl:
            break
        c = _fc(a, fs)
        sc = 2.0 * n * float(c.max()) / (a.sum()**2)
        if sc < bsc - 1e-12:
            bsc = sc
            bst = a.copy()
        stall = 0 if sc < psc - 1e-13 else stall + 1
        psc = sc
        if stall > smx:
            break

        act = np.where(c >= mx * (1.0 - frac))[0]

            act = act[ii]
        g = _sg(a, act)
        g -= g.mean()
        if stall > smx//3:
            gm0 = np.abs(g).max()
            if gm0 > 1e-14:

                a = a * np.exp(-eta * g / gm)
                a = np.clip(a, 0.0, None)

        s = a.sum()
