import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_conv(a, b):
    size = 1 << (len(a) + len(b) - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:len(a) + len(b) - 1]


def score_of(f):
    c = fft_conv(f, f)
    return 2.0 * f.size * c.max()


def build_conv_A(f):
    n = len(f)
    jj = np.repeat(np.arange(n), n).astype(np.int32)
    rows = (jj + np.tile(np.arange(n), n).astype(np.int32)).astype(np.int32)
    vals = np.tile(f, n).astype(np.float64)
    return sp.csr_matrix((vals, (rows, jj)), shape=(2 * n - 1, n))


def lp_step(f, M, tlimit):
    n = len(f)
    A = build_conv_A(f)
    c_vec = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_vec, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'time_limit': tlimit})
    if not res.success or res.x is None:
        return None
    return np.maximum(res.x, 0.0)


def subgradient(f, c, frac):
    n = len(f)
    M = c.max()
    if M < 1e-30:
        return None
    k_act = max(1, int(frac * (2 * n - 1)))
    top_idx = np.argpartition(c, -k_act)[-k_act:]
    u = np.zeros(2 * n - 1)
    u[top_idx] = 1.0
    fu = fft_conv(f[::-1], u)
    grad = 2.0 * fu[n - 1:2 * n - 1]
    return np.maximum(grad, 0.0)


def run_gradient(f, deadline, rng, frac0, eta0):
    n = len(f)
    f = f.copy()
    f /= f.sum()
    best_f = f.copy()
    best_sc = score_of(f)
    eta = eta0
    frac = frac0
    stall = 0
    while time.monotonic() < deadline:
        c = fft_conv(f, f)
        sc = 2.0 * n * c.max()

            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                f = best_f * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                eta = eta0
                frac = max(0.003, frac * 0.6)
                continue
        g = subgradient(f, c, frac)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        eta *= 0.9998
        if stall % 80 == 0:
            frac = max(0.001, frac * 0.92)
    return best_f, best_sc

    inits = []
    x = np.arange(n, dtype=np.float64) + 1.0
    for pw in [0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
