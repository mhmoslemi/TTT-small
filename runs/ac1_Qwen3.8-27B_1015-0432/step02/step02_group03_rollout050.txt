def _conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _subgrad(a, c, frac, cap):
    n = a.size
    mx = float(c.max())
    thresh = mx * (1.0 - frac)
    lags = np.where(c >= thresh)[0]
    lags = lags[(lags >= 0) & (lags < 2*n-1)]
    if len(lags) > cap:
        idx = np.argpartition(c[lags], -cap)[-cap:]
        lags = lags[idx]
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    if len(lags) > 0:
        g /= len(lags)
    return g

def _descend(a, fs, dl, frac, cap, eta0, decay, multi, mom_beta, stall_max, rng):

    S = a.sum()
    mom = np.zeros(n)

    c_best = float(_conv(best, fs).max())
    S_best = float(best.sum())
    score_best = 2.0 * n * c_best / (S_best ** 2)
    stall = 0
    prev_score = score_best
    eta = eta0
    t = 0
    while time.monotonic() < dl:
        c = _conv(a, fs)
        S = a.sum()
        score = 2.0 * n * float(c.max()) / (S ** 2)
        if score < score_best - 1e-12:
            score_best = score
            best = a.copy()
        if score < prev_score - 1e-14:
            stall = 0
        else:
            stall += 1
        prev_score = score
        if stall > stall_max:
            break
        g = _subgrad(a, c, frac, cap)
        g -= g.mean()  # keep sum roughly constant
        # Add noise if stalled
        if stall > stall_max // 3:
            gm = np.abs(g).max()
            if gm > 1e-14:
