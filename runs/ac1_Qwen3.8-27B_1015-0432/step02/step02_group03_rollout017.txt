import numpy as np
import time

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, lags):
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return g

def _run(a, fs, dl, eta, cap, frac, multi, beta, smx, rng):
    n = a.size
    a = a.copy().astype(np.float64)
    a /= a.sum()
    ba = a.copy()
    bs = 2.0 * n * float(_fc(a, fs).max())
    mom = np.zeros(n)
    st = 0
    for _ in range(500000):
        if time.monotonic() > dl:
            break
        b = _fc(a, fs)
        mx = float(b.max())
        sc = 2.0 * n * mx
        if sc < bs - 1e-13:
            bs = sc
            ba = a.copy()
            st = 0
        else:
            st += 1
        if st > smx:
            break
        thr = mx * (1.0 - frac)
        act = np.where(b >= thr)[0]
        act = act[(act >= 0) & (act < 2 * n - 1)]
        if len(act) > cap:
            ii = np.argpartition(b[act], -cap)[-cap:]
            act = act[ii]
        g = _sg(a, act)
        if multi:
            gm = np.abs(g).max()
            if gm > 1e-14:
                a *= np.exp(-eta * g / gm)
        else:
            mom = beta * mom + (1.0 - beta) * g
            a -= eta * mom
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = ba.copy()
            break
        a /= s
        eta *= 0.99997
    return ba, bs
