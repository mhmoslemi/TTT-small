        best_sc_eval = float(evaluate_sequence(best.tolist()))

    if time.monotonic() < dl - 80.0:
        cur2 = best.copy()
        taus_minor = [0.002, 0.001, 5e-4, 2e-4, 1e-4]
        cur2, sc2 = _ptg_optimize(cur2, S, dl, taus_minor)
        if sc2 < best_sc:
            best = cur2.copy()
            best_sc = sc2
            best_sc_eval = float(evaluate_sequence(best.tolist()))

    # Random local search polish
    while time.monotonic() < dl - 5.0:
        cd = best.copy()
        nw = int(rng.integers(1, max(2, n // 40)))
        s = int(rng.integers(0, n - nw))
        cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.02)))
        if rng.random() < 0.1:
            cd *= float(np.exp(rng.normal(0.0, 0.008)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = _fast_eval(cd)
            if np.isfinite(sc2) and sc2 < best_sc:
                best = cd.copy()
                best_sc = sc2

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = a0.copy()
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = a0.copy()
    return [float(v) for v in best]
