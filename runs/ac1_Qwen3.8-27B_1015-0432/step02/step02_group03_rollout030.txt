import numpy as np
import time

def _fc(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, lags):
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    m = len(lags)
    if m:
        g /= m
    return g

def _smooth_g(a, conv, tau, fs):
    n = a.size
    cm = float(conv.max())
    ec = np.exp(tau * (conv - cm))
    w = ec / (ec.sum() + 1e-300)
    a_r = a[::-1].copy()
    fa = np.fft.rfft(a_r, n=fs)
    fw = np.fft.rfft(w, n=fs)
    gc = np.fft.irfft(fa * fw, n=fs)
    return 2.0 * gc[n-1:2*n-1]

def _descent(a, fs, ts, dl, eta, beta, frac, cap, smx, multi
