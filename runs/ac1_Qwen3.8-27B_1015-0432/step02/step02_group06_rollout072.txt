def state_from_arr(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    z = np.log(a)
    z -= float(z.max())
    return softmax_z(z)

def noise_shift(rng, L, n, mask):
    w = rng.standard_normal(n)
    W = np.fft.rfft(w, L)
    y = np.fft.irfft(W * mask, L)[:n]
    y -= float(y.mean())
    sd = float(y.std())
    if sd < 1e-300:
        return np.zeros(n, dtype=np.float64)
    return y / sd

import numpy as np
import time


def next_pow_len(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def auto_conv(x, L, n):
    X = np.fft.rfft(x, L)
    b = np.fft.irfft(X * X, L)[:2 * n - 1]
    np.maximum(b, 0.0, out=b)
    M = float(b.max())
    return b, X, M


def grad_fft(w, X, phase, L, n):
    W = np.fft.rfft(w, L)
    Xr = phase * np.conj(X)
    g = np.fft.irfft(W * Xr, L)[n - 1:2 * n - 1]
    np.maximum(g, 0.0, out=g)
    return g


def softmax_z(z):
    m = float(z.max())
    if not np.isfinite(m):
        n = len(z)
        return np.ones(n, dtype=np.float64) / n, np.zeros(n, dtype=np.float64)
    x = np.exp(z - m)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 0.0:
        n = len(z)
        return np.ones(n, dtype=np.float64) / n, np.zeros(n, dtype=np.float64)
    x /= s
    return x, z - m


def state_from_arr(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-15)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a /= s
    z = np.log(a)
    z -= float(z.max())
    return softmax_z(z)


def noise_shift(rng, L, n, mask):
    w = rng.standard_normal(n)
    W = np.fft.rfft(w, L)
    y = np.fft.irfft(W * mask, L)[:n]
    y -= float(y.mean())
    sd = float(y.std())
    if sd < 1e-300:
        return np.zeros(n, dtype=np.float64)
    return y / sd


def restart_state(best_z, rng, L, n, mask, no):
    best = None
    best_M = np.inf
    for r in range(2):
        c = (no + r) % 5
        sigma = 0.05 + 0.05 * c + 0.005 * ((no + r) % 10)
        sigma = min(0.7, sigma)
        mix = 0.0
        if c == 3:
            mix = 0.1
        if c == 4:
            mix = 0.2
        base = best_z
        if r == 1 and c == 0:
            base = base[::-1]
        if mix > 0.0:
            base = (1.0 - mix) * base
        nz = noise_shift(rng, L, n, mask)
        z = base + sigma * nz
        x, z = softmax_z(z)
        b, X, M = auto_conv(x, L, n)
        if M < best_M:
            best_M = M
            best = (x, z, b, X, M)
    return best
