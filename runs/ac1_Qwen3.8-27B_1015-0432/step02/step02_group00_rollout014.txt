def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_fn(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def soft_subgrad(f, b, T):
    n = f.shape[0]
    L = 2 * n - 1
    bs = b - b.max()
    w = np.exp(T * bs)
    ws = w.sum()

    else:

    c = fft_conv(f, w[::-1])
    g = c[n-1:2*n-1][::-1]
