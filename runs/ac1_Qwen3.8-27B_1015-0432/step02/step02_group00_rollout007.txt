def optimize(f, deadline, rng, sym=False, max_iter=200000):
    n = f.size
    f = f / f.sum()
    best_f = f.copy()
    best_sc = score(f)
    
    eta = 0.1
    no_improv = 0
    restart_count = 0
    eval_every = max(20, n // 100)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        b = fft_conv(f, f)
        M = b.max()
        
        # Select top-K peaks
        K = min(50, 2*n-1)
        # Use top K peaks by value
        thresh = M * 0.999
        peaks_all = np.where(b >= thresh)[0]
        if len(peaks_all) > K:
            # Take top K
            idx = np.argpartition(b, -K)[-K:]
            peaks = np.sort(idx)
        else:
            peaks = peaks_all
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        
        g = subgrad(f, peaks)
        gmax = g.max()
        if gmax < 1e-15:
            break
        
        # Multiplicative update
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-30)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        
        if (it+1) % eval_every == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            
            if no_improv > 500:
                # Restart with perturbation
                scale = 0.03 + 0.04 * min(1.0, restart_count / 5.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-30)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                restart_count += 1
                eta = min(0.3, eta * 1.5)
            elif it > 100:
                eta *= 0.998
    
    return best_f, best_sc
