The projection onto the probability simplex {x >= 0, 1'x = 1}:
Algorithm (Michelot 2011):
1. Sort v in descending order: u_1 >= u_2 >= ... >= u_n
2. Compute prefix sums: S_k = sum_{i=1}^k u_i
3. Find largest k such that u_k > (S_k - 1)/k
4. theta = (S_k - 1)/k
5. Result: max(v - theta, 0)

This is O(n log n) due to sorting. For n=4498, this is fast (~1-2 ms).

                break
    return a, local_best, local_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 12.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 2 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        parent = np.ones(1024, dtype=np.float64)
        n = 1024
        s0 = 1024.0
    size = 1 << (3 * n - 2).bit_length()
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    a = parent / s0
    # Phase 1: main smooth-max projected descent
    t1 = time.monotonic() + 0.55 * (deadline - time.monotonic())
    a, lb, ls = run_descent(a, n, size, t1, best, best_score, s0)
    if ls < best_score:
        best_score, best = ls, lb
    # Phase 2: refined second pass
    remain = deadline - time.monotonic()
    if remain > 25:
        t2 = time.monotonic() + 0.35 * remain
        s2 = float(best.sum())
        a2 = best / s2
        a2, lb2, ls2 = run_descent(a2, n, size, t2, best, best_score, s2)
        if ls2 < best_score:
            best_score, best = ls2, lb2
    # Phase 3: perturbation + short descent
    base = best.copy()
    while time.monotonic() < deadline - 5.0:
        cand = base.copy()
        w = max(2, n // 120)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= np.exp(float(rng.normal(0.0, 0.012)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best, base = sc, cand, cand
            continue
        if np.isfinite(sc) and sc < best_score + 0.004 and time.monotonic() < deadline - 12.0:
            t_sh = time.monotonic() + min(5.0, deadline - time.monotonic() - 4.0)
            sp = float(cand.sum())
            a_p = cand / sp
            _, lb_p, ls_p = run_descent(a_p, n, size, t_sh, cand, sc, sp)
            if ls_p < best_score:
                best_score, best, base = ls_p, lb_p, cand
    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
