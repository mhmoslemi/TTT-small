`d(b_{k*})/d(a_i) = a_{k*-i} + a_{i-k*}` (with appropriate bounds)

More precisely, for full convolution `b_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}`:

import numpy as np

def _fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), n + m - 1)

def _score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = _fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def _init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    h = np.maximum(h, 1e-15)
    return h / h.sum()
