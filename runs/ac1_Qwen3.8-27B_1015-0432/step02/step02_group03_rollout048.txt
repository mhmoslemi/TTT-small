def _conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _grad(a, lags):
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return g

def _subgrad_descent(a, fs, total_sum, deadline, eta, lags_frac, lags_max, stall_lim, use_multi, rng, momentum):

    mom = np.zeros(n)
    best = a.copy()
    best_c = 2.0 * n * float(_conv(best, fs).max()) / (best.sum()**2)
    stall = 0
    prev_c = best_c
    while time.monotonic() < deadline:
        c = _conv(a, fs)
        cur_c = 2.0 * n * float(c.max()) / (a.sum()**2)
        if cur_c < best_c - 1e-14:
            best_c = cur_c
            best = a.copy()
        if cur_c > prev_c + 1e-14:

        if stall > stall_lim:

        thr = mx * (1.0 - lags_frac)

        act = act[(act >= 0) & (act < 2*n - 1)]
        if len(act) > lags_max:
