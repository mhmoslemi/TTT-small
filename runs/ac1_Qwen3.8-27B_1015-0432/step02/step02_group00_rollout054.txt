import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def cscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, h_vec):
    n = f.shape[0]
    c = fft_conv(f, h_vec[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, k_start, k_end, eta0, eta1, use_weight=False):
    n = f.shape[0]
    f = np.maximum(f, 1e-20).copy()
    f /= f.sum()
    best_f = f.copy()
    best_sc = cscore(f)
    no_improv = 0
    L = 2 * n - 1

        if time.monotonic() > deadline:
            break
        t = min(1.0, it / 8000.0)
        k = max(1, int(k_start + (k_end - k_start) * t))
        eta = eta0 + (eta1 - eta0) * t

        b = fft_conv(f, f)
        if use_weight:
            mean_b = b.sum() / L
            thresh = mean_b + 0.3 * (b.max() - mean_b)
            h_vec = np.maximum(b - thresh, 0.0)
            h_vec = h_vec / (h_vec.max() + 1e-30)
            peaks_all = h_vec > 1e-12
        else:

                peaks = np.argpartition(b, -k)[-k:]
            else:
                peaks = np.arange(L)
            h_vec = np.zeros(L)
            h_vec[peaks] = 1.0

        g = subgrad(f, h_vec)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-25)
