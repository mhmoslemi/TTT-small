import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def smooth_grad(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(np.conj(F) * P, N)[:n]

def peak_grad(f, k_star):
    n = f.size
    lo = max(0, k_star - n + 1)
    hi = min(k_star, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * f[k_star - hi:k_star - lo + 1][::-1]
    return g

def make_init(n, kind, **kw):
    if kind == 'flat':
        a = kw.get('asym', 0.0)
        x = np.linspace(-np.pi, np.pi, n)
        w = 1.0 - 0.42 * np.cos(x) + 0.21 * np.cos(2 * x) - 0.07 * np.cos(3 * x)
        if abs(a) > 1e-9:
            w *= np.exp(a * (2.0 * np.arange(n) / max(n - 1, 1) - 1.0))
        return np.maximum(w, 1e-10)
    elif kind == 'powlaw':
        j = np.arange(n, dtype=np.float64)
        return (j + 1.0) ** kw['a'] * (n - j) ** kw['b']
    elif kind == 'gauss':
        j = np.arange(n, dtype=np.float64)
        c = kw.get('c', 0.5)
        w_val = kw.get('w', 0.2)
        return np.exp(-0.5 * ((j - n * c) / max(n * w_val, 1)) ** 2)
    else:
        return np.ones(n)

def run_mwo(f, deadline, rng):
    n = f.size
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    eta = 0.12
    T = max(2.0, n * 0.025)
    T_min = 0.25
    best_f = f.copy()
    best_sc = fast_score(f)
    no_imp = 0
    it = 0
    while time.monotonic() < deadline - 5.0:
        it += 1
        C = fft_conv(f, f)
        s = f.sum()
        sc = 2.0 * n * C.max() / (s * s)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        if T > T_min + 0.15:
            g = smooth_grad(f, C, T)
        else:
            ks = np.argsort(C)[-3:][::-1]
            g = np.zeros(n)
            for k in ks:
                g += peak_grad(f, int(k))
            g /= len(ks)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        if it % 8 == 0:
            ns = fast_score(f)
            if ns < sc - 1e-9:
                eta = min(eta * 1.09, 2.5)
            else:
                eta = max(eta * 0.82, 0.015)
        if it % 60 == 0:
            T = max(T_min, T * 0.92)
        if no_imp > 200:
            f = best_f * np.exp(0.05 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.12
            T = max(2.0, n * 0.025)
    return best_f, best_sc

def lp_dir(f):
    inv_f = 1.0 / np.maximum(f, 1e-20)
    rmax_f = np.maximum.accumulate(inv_f)
    rmax_r = np.maximum.accumulate(inv_f[::-1])[::-1]
    rmax_all = np.maximum(rmax_f, rmax_r)
    g = (inv_f < rmax_all - 1e-12).astype(np.float64)
    if g.sum() < 1:
        g = np.ones_like(f)
    return g / g.sum()

def lp_mix(f, deadline, rng):
    best_f = f.copy()
    best_sc = fast_score(f)
    for _ in range(25):
        if time.monotonic() > deadline - 10.0:
            break
        d = lp_dir(best_f)
        for t in [0.15, 0.3, 0.5, 0.7]:
            cand = np.maximum((1 - t) * best_f + t * d, 1e-20)
            cand /= cand.sum()
            sc = fast_score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_f = cand
        to = time.monotonic() + min(60.0, deadline - time.monotonic() - 12.0)
        if to > time.monotonic() + 8.0:
            f2, sc2 = run_mwo(best_f, to, rng)
            if sc2 < best_sc:
                best_sc = sc2
                best_f = f2
    return best_f, best_sc
