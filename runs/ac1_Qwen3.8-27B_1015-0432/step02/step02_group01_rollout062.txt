import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def compute_grad(f, w):
    n = f.size
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n - 1:2 * n - 1].copy()

        sc = 2.0 * n * M

            best_f = f.copy()

        else:

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score_norm(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def make_grad(f, w):
    n = f.size
    g = fft_conv(w, f[::-1])[n - 1:2 * n - 1].copy()
    return np.maximum(g, 0.0)

def ramp(n, p, rev):
    x = np.arange(n, dtype=np.float64) + 1.0
    r = x ** p
    if rev: r = r[::-1]
    return r / r.sum()

def arc(n, c, w):
    u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    v = np.clip((u - c) / w, -1.5, 1.5)
    a = np.maximum(0.0, 1.0 - v * v) ** 0.7

        inits.append(ramp(n, pw, True))
    for c, w in ((0.25, 0.8), (0.35, 0.7), (0.45, 0.6), (0.6, 0.5)):
        inits.append(arc(n, c, w))

    bf, bs = inits[0].copy(), np.inf
    for ini in inits:
        s = score_norm(ini)
        if s < bs:
            bs, bf = s, ini.copy()

    phases = [(15, 0.12), (30, 0.08), (60, 0.05), (100, 0.03),
              (200, 0.018), (400, 0.01), (800, 0.006), (1500, 0.003)]

    for be, te in phases:
        if time.monotonic() > dl - 30:
            break
        fo, so = run_phase(bf, dl - 15, rng, be, te, 30000)
        if so < bs:
            bs, bf = so, fo.copy()
        if time.monotonic() > dl - 25:
            break
        flip = bf[::-1].copy()
        for t in (0.1, 0.2, 0.3, 0.5):
            bl = (1.0 - t) * bf + t * flip
            bl = np.maximum(bl, 1e-20)
            bl /= bl.sum()
            sb = score_norm(bl)
            if sb < bs:
                bs, bf = sb, bl.copy()
        for _ in range(4):
            if time.monotonic() > dl - 20:
                break
            tr = bf * np.exp(0.015 * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-20)
            tr /= tr.sum()
            st = score_norm(tr)
            if st < bs:
                bs, bf = st, tr.copy()

    if time.monotonic() < dl - 35:
        ff, sf = run_phase(bf, dl - 10, rng, 2000.0, 0.002, 50000)
        if sf < bs:
            bs, bf = sf, ff.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in bf]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
