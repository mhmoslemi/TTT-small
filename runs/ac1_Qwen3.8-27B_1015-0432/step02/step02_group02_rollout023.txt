    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    L = _fpow2(2 * n - 1)
    c = ifft(fft(a, L) ** 2).real[:2 * n - 1]
    mv, _ = _smax(c, tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return np.ones(n)
    L = _fpow2(2 * n - 1)
    c = ifft(fft(a, L) ** 2).real[:2 * n - 1]
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L2 = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L2) * fft(pr, L2)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 10.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(256)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256)
    n = par.size

    # Candidate pool: uniform (theoretical min for nonneg), parent, structured
    cands = [np.ones(n, dtype=np.float64)]
    if par.sum() > 0.01:
        cands.append(par.copy())
    x = np.arange(n, dtype=np.float64)
    for amp in (0.3, 0.6):
        c = np.clip(1.0 + amp * np.cos(2.0 * np.pi * x / n), 0, None)
        if c.sum() > 0.01:
            cands.append(c / c.sum() * np.sqrt(2.0 * n))
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    if g.sum() > 0.01:
        cands.append(g / g.sum() * np.sqrt(2.0 * n))

    best = cands[0].copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    cur = best.copy()
    for inp in cands[1:]:
        sc = float(evaluate_sequence(inp.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = inp.copy()
            cur = inp.copy()

    # Gradient-based refinement with decreasing soft-max temperature
    if time.monotonic() < dl - 60.0:
        bl = [(0.0, 1000.0)] * n
        for tv in (1.0, 0.5, 0.1, 0.05, 0.01):
            if time.monotonic() > dl - 60.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 200, 'gtol': 1e-12,
                                        'maxls': 20, 'ftol': 1e-18})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    sc = float(evaluate_sequence(nx.tolist()))
                    if np.isfinite(sc) and sc < best_sc:
                        best_sc = sc
                        best = nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    # Randomised local search in remaining time
    while time.monotonic() < dl - 3.0:
        cd = cur.copy()
        nw = int(rng.integers(1, max(2, n // 50)))
        s = int(rng.integers(0, n - nw))
        cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.03)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc2 = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc2) and sc2 < best_sc:
                best_sc = sc2
                best = cd.copy()
                cur = cd.copy()

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n)
    return [float(v) for v in best]
