def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * len(f) * fft_conv(f, f).max() / (s * s))

def grad_at_peak(f, C):
    n = f.size
    k = int(np.argmax(C))
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
    return g

def grad_smooth(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)

    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

    j = np.arange(n, dtype=np.float64)

def make_poly(n, p, q):

    w = (j + 1.0) ** p * (n - j) ** q
    w = np.maximum(w, 1e-15)
    return w / w.sum()
