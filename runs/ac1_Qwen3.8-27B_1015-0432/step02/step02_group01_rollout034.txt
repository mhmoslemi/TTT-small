def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_unit(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max(), b

def grad_direction(f, b, beta, top_frac):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    k = max(1, int(top_frac * b.size))
    idx = np.argsort(b)[-k:]
    wm = np.zeros_like(w)
    wm[idx] = w[idx]
    if wm.sum() < 1e-30:
        return None
    fr = f[::-1]
    size = 1 << ((2 * len(w) + len(fr) - 2).bit_length())
    gc = np.fft.irfft(np.fft.rfft(wm, size) * np.fft.rfft(fr, size))[:len(wm) + len(fr) - 1]
    g = gc[f.size - 1:2 * f.size - 1].copy()
    g = np.maximum(g, 0.0)
    return g

def lp_direction(f, M):
    """Solve LP to find g >= 0, sum(g) = 1, conv(f,g)_k <= M for all k."""
    n = f.size
    # Build sparse Toeplitz-like matrix A where A[k,j] = f[k-j] if 0<=k-j<n
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # Convert to min: min -sum(g) s.t. A @ g <= M, g >= 0
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    g /= s
    return g

        gmax = g.max()

        f /= f.sum()
        
        if it % 100 == 0 and it > 0:
            beta *= 1.03
            eta = max(0.001, eta * 0.992)
    
    return best_f, best_sc

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_unit(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max(), b

def grad_dir(f, b, beta, top_frac):
    M = b.max()
    if M < 1e-30:
        return None
    w = np.exp(beta * (b / M - 1.0))
    k = max(1, int(top_frac * b.size))
    idx = np.argsort(b)[-k:]
    wm = np.zeros_like(w)
    wm[idx] = w[idx]
    if wm.sum() < 1e-30:
        return None
    fr = f[::-1]
    sz = 1 << ((2*len(w) + len(fr) - 2).bit_length())
    gc = np.fft.irfft(np.fft.rfft(wm, sz) * np.fft.rfft(fr, sz))[:len(wm)+len(fr)-1]
    g = gc[f.size-1 : 2*f.size-1].copy()
    return np.maximum(g, 0.0)

def build_A(f):

    js = np.arange(n, dtype=np.int64)

    rows = cols + within
    data = f[within.astype(np.intp)]
    return csc_matrix((data, (rows.astype(np.int32), cols.astype(np.int32))), shape=(2*n-1, n))
