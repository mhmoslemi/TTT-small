    exp_vals = np.exp((conv - mx) / T)

        hi = min(n - 1, k)

import numpy as np
import time


def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def grad_at_lags(a, conv, lags, weights=None):
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    m = len(lags)
    if m == 0:
        return grad
    if weights is None:
        weights = np.full(m, 1.0 / m)
    for i in range(m):
        k = int(lags[i])
        w = float(weights[i])
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * w * a[k - hi:k - lo + 1][::-1]
    return grad


def project_simplex(y):
    n = y.size
    s = y.sum()
    if s <= 0:
        return np.ones(n) / n
    lo = 0.0
    hi = float(y.max())
    tau = 0.0
    for _ in range(50):
        tau = 0.5 * (lo + hi)
        s2 = np.maximum(0.0, y - tau).sum()
        if abs(s2 - 1.0) < 1e-13:
            break
        if s2 > 1.0:
            lo = tau
        else:
            hi = tau
    return np.maximum(0.0, y - tau)

def local_search(best, best_score, n, s0, deadline, rng):
    base = best.copy()
    while time.monotonic() < deadline:
        cand = base.copy()
        w = max(3, n // 80)
        nsegs = int(rng.integers(1, 5))
        for _ in range(nsegs):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            scale = np.exp(float(rng.normal(0.0, 0.012)))
            cand[st:e] *= scale
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            base = cand.copy()
    return best_score, base
