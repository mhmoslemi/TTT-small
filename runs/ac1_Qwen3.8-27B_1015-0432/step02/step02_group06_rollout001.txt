        grad = np.maximum(grad, 1e-30)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        it += 1
        if it % 25 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if stall >= 150:
                f = best * (1.0 + 0.02 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                stall = 0
            if it % 250 == 0:
                beta = min(300.0, beta * 1.06)
                eta = max(0.01, eta * 0.97)
    return best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    # Theoretical minimum is 2 (uniform), by Cauchy-Schwarz on b[n-1]=sum(a_i^2)
    uniform = np.ones(n)
    uniform_sc = fast_score(uniform)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 0.01:
        p = uniform.copy()
    else:
        p = p / p.sum()
    p_sc = fast_score(p)

    best_f = uniform.copy() if uniform_sc <= p_sc else p.copy()
    best_score = min(uniform_sc, p_sc)

    # Try optimizing from parent (briefly) in case structure helps
    if time.monotonic() < deadline - 120:
        f_opt, sc_opt = gd_loop(p.copy(), min(deadline - 60, t0 + 300), rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

    # Try smoothed parent
    if time.monotonic() < deadline - 120:
        sm = init_smooth(n, p)
        sm_sc = fast_score(sm)
        if sm_sc < best_score:
            best_score = sm_sc
            best_f = sm.copy()
        else:
            f_opt2, sc_opt2 = gd_loop(sm.copy(), min(deadline - 60, t0 + 500), rng, beta=40, eta=0.12)
            if sc_opt2 < best_score:
                best_score = sc_opt2
                best_f = f_opt2.copy()

    # Try a second optimization round from current best (if not already near-uniform)
    if best_score > uniform_sc + 1e-9 and time.monotonic() < deadline - 30:
        f_opt3, sc_opt3 = gd_loop(best_f.copy(), deadline, rng, beta=100, eta=0.05)
        if sc_opt3 < best_score:
            best_score = sc_opt3
            best_f = f_opt3.copy()

    # Final verification via exact evaluator
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)

    # Compare against uniform as absolute safety net
    uniform_result = [1.0] * n
    uniform_final = evaluate_sequence(uniform_result)
    if uniform_final < final_sc:
        result = uniform_result
        final_sc = uniform_final

    if not np.isfinite(final_sc):
        result = [1.0] * n

    return result
