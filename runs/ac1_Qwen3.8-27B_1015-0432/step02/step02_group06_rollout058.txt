import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_of(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def subgrad_at(f, k):
    n = len(f)
    j = np.arange(n)
    idx = k - j
    valid = (idx >= 0) & (idx < n)
    g = np.zeros(n)
    g[valid] = 2.0 * f[idx[valid]]
    return g

        M = b.max()
        
        # Weighted subgradient: use exponential weighting over b
        # w_k = (b_k / M)^beta, then grad = sum_k w_k * subgrad_at(f, k) / sum_k w_k
        # But this is expensive. Use just the argmax for efficiency.
        k_star = np.argmax(b)
        grad = subgrad_at(f_cur, k_star)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        
        # Also add a small amount of "all-k" gradient for stability
        # grad_smooth[j] = sum_k (b[k]/M) * 2*f[k-j] (over valid k,j)
        # This is 2 * (b_conj * f_rev) reversed... it's a cross-correlation
        # Actually, let's just use the argmax for speed
        
        f_cur = f_cur * np.exp(-eta * grad / gmax)
        f_cur = np.maximum(f_cur, 1e-30)
        f_cur /= f_cur.sum()
        it += 1
        
        if it % 100 == 0:
            sc = score_of(f_cur)
            if sc < best_sc:
                best_sc = sc
                best = f_cur.copy()
                stall = 0
            else:
                stall += 1
                if stall > 100:
                    break
                if stall % 20 == 0:
                    eta *= 0.9
                    if stall > 60:
                        f_cur = best * np.exp(0.01 * rng.standard_normal(n))
                        f_cur = np.maximum(f_cur, 1e-30)
                        f_cur /= f_cur.sum()
                        stall = 0
        if it % 500 == 0:
            eta = max(0.01, eta * 0.95)
    
    return best, best_sc
