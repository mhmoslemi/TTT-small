def optimize(f, deadline, rng, max_iters=100000):
    n = f.size
    f = f.copy()
    f /= f.sum()
    eta = 0.1  # initial step size
    best_f = f.copy()
    best_sc = 2.0 * n * fft_conv(f, f).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = fft_conv(f, f)
        M = b.max()
        sc = 2.0 * n * M
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
        
        # Compute subgradient
        # Find active set (within tolerance)
        tol = max(1e-10, M * 1e-6)
        S_mask = (b >= M - tol).astype(np.float64)
        S_count = S_mask.sum()
        if S_count < 1:
            S_mask[M.argmax()] = 1.0
            S_count = 1
        
        # Compute subgradient via FFT
        mask_rev = S_mask[::-1]
        c = fft_conv(f, mask_rev)
        g = c[n-1:2*n-1][::-1] * (2.0 / S_count)
        
        # Normalize gradient
        g_max = g.max()
        if g_max < 1e-30:
            break
        g /= g_max
        
        # Multiplicative update
        f_new = f * np.exp(-eta * g)
        f_new = np.maximum(f_new, 1e-20)
        f_new /= f_new.sum()
        
        # Check if score improved
        b_new = fft_conv(f_new, f_new)
        sc_new = 2.0 * n * b_new.max()
        
        if sc_new < sc:
            f = f_new
            eta = min(eta * 1.02, 0.5)  # slight increase
        else:
            eta *= 0.7  # reduce step
            if eta < 1e-6:
                # Try a perturbation
                f = best_f * np.exp(0.05 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                eta = 0.05
    
    return best_f, best_sc
