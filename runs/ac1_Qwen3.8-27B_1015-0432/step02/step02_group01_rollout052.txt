        else:
            stall += 1
        if stall > 350:
            f = best_f * np.exp(0.025 * rng.standard_normal(n))
            f = np.maximum(f, 1e-30)
            f /= f.sum()
            eta, beta, stall = eta0, beta0, 0
            continue
        g = grad_dir(f, b, beta)
        if g is None:
            break
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        f = np.maximum(f * np.exp(-eta * g), 1e-30)
        f /= f.sum()
        cnt += 1
        if cnt % 200 == 0:
            beta *= 1.05
            eta = max(0.001, eta * 0.985)
    return best_f, best_sc

def gen_inits(n, p, rng):
    pool = [p, p[::-1].copy()]
    sym = (p + p[::-1]) / 2.0
    pool.append(sym / sym.sum())
    x = np.arange(n, dtype=np.float64) + 1.0
    for a in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0]:
        r = x ** a
        pool.append(r / r.sum())
        pool.append(r[::-1] / r.sum())
    u = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    for c, w in [(0.2, 0.8), (0.3, 0.6), (0.5, 0.4), (0.7, 0.5)]:
        arc = np.maximum(0.0, 1.0 - ((u - c) / w) ** 2) ** 0.7
        arc = np.maximum(arc, 1e-15)
        pool.append(arc / arc.sum())
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.1
        k = max(3, n // 200)
        r = np.convolve(r, np.ones(k) / k, mode='same')
        r = np.maximum(r, 1e-10)
        pool.append(r / r.sum())
    return pool

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 12.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p /= p.sum()

    best_f, best_sc = None, np.inf
    for ini in gen_inits(n, p, rng):
        sc = score_norm(ini)
        if sc < best_sc:
            best_sc, best_f = sc, ini.copy()

    phases = [(0.06, 15), (0.04, 40), (0.02, 100), (0.01, 250), (0.005, 600), (0.003, 1200)]
    for eta0, beta0 in phases:
        if time.monotonic() > dl - 20:
            break
        rem = dl - time.monotonic()
        pdl = time.monotonic() + rem * 0.35
        f2, s2 = optimize(best_f, pdl, rng, eta0, beta0)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()
        if time.monotonic() > dl - 15:
            break
        for t in [0.1, 0.2, 0.3, 0.5]:
            bl = (1.0 - t) * best_f + t * best_f[::-1]
            bl = np.maximum(bl, 1e-30)
            bl /= bl.sum()
            sb = score_norm(bl)
            if sb < best_sc:
                best_sc, best_f = sb, bl.copy()

    if time.monotonic() < dl - 25:
        f3, s3 = optimize(best_f, dl - 8, rng, 0.002, 2000)
        if s3 < best_sc:
            best_sc, best_f = s3, f3.copy()
        sym2 = (best_f + best_f[::-1]) / 2.0
        sym2 = np.maximum(sym2, 1e-30)
        sym2 /= sym2.sum()
        ss = score_norm(sym2)
        if ss < best_sc:
            best_sc, best_f = ss, sym2

    result = [float(max(0.0, min(1000.0, v))) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
