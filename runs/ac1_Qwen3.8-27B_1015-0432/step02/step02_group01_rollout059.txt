def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def softmax_weights(b, beta):
    """Compute softmax weights over lags."""
    M = b.max()
    if M < 1e-30:
        return None
    exponents = beta * (b / M - 1.0)
    exponents -= exponents.max()  # numerical stability
    w = np.exp(exponents)

        w[b >= threshold] = 1.0
        if w.sum() == 0:

    else:
        w /= s
