import time


def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def subgrad(a, conv, topk):
    n = a.size
    nlags = 2 * n - 1
    if topk >= nlags:
        idx = np.arange(nlags)
    else:
        idx = np.argpartition(conv, -topk)[-topk:]
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    g /= float(len(idx))
    return g


def mirror_descend(a, n, fft_size, s0, best, best_score, deadline,

def local_perturb(base, n, rng, best_score, best, deadline):

        cand = base.copy()
