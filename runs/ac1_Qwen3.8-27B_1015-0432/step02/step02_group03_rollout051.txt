    bsc = 2.0 * n * float(_fc(bst, fs).max()) / (bst.sum() ** 2)
    stall = 0
    psc = bsc
    while time.monotonic() < deadline:
        c = _fc(a, fs)
        sv = a.sum()
        sc = 2.0 * n * float(c.max()) / (sv * sv)
        if sc < bsc - 1e-12:
            bsc = sc
            bst = a.copy()
        stall = 0 if sc < psc - 1e-13 else stall + 1
        psc = sc
        if stall > smx:
            break
        f = frac * (1.0 + 0.5 * min(stall / float(smx), 2.0))
        mx = float(c.max())
        act = np.where(c >= mx * (1.0 - f))[0]
        act = act[(act >= 0) & (act < 2 * n - 1)]
        if act.size == 0:
            act = np.array([int(np.argmax(c))])
        if act.size > cap:
            ii = np.argpartition(c[act], -cap)[-cap:]
            act = act[ii]
        g = _sg(a, act)
        g -= g.mean()
        if stall > max(1, smx // 3):
            gm0 = np.abs(g).max()
            if gm0 > 1e-14:
                g += noise_amp * rng.standard_normal(n) / gm0
        if multi:
            gm = np.abs(g).max()
            if gm > 1e-12:
                a = a * np.exp(-eta * g / gm)
        else:
            mom = beta * mom + (1.0 - beta) * g
            a = np.clip(a - eta * mom, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            break
        a *= ts / s
        eta *= 0.9999

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(0.0, float(budget_s) - 15.0)
    p = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(512)
    n = p.size
    fs = 1 << (2 * n - 1).bit_length()
    ts = float(p.sum())

    best = p.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    # Phase 1: Broad Euclidean descent (wide active set, larger steps)
    d1 = dl - 480.0
    a, s = _desc(p.copy(), fs, ts, 0.5, 0.85, d1, 6e-4, 80, 5000,
                 False, rng, n, 0.03)
    if s < best_sc:
        best_sc, best = s, a

    # Phase 2: Multiplicative descent (different geometry)
    d2 = dl - 240.0
    if time.monotonic() < d2:
        a, s = _desc(best.copy(), fs, ts, 0.3, 0.0, d2, 4e-4, 72,
                     4000, True, rng, n, 0.025)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 3: Fine Euclidean polish (tight active set, small steps)
    d3 = dl - 100.0
    if time.monotonic() < d3:
        a, s = _desc(best.copy(), fs, ts, 0.2, 0.90, d3, 2e-4, 55,
                     3500, False, rng, n, 0.02)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 4: Fine multiplicative polish
    d4 = dl - 45.0
    if time.monotonic() < d4:
        a, s = _desc(best.copy(), fs, ts, 0.1, 0.0, d4, 1.5e-4, 45,
                     3000, True, rng, n, 0.015)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 5: Annealed restart with short descent
    d5 = dl - 28.0
    if time.monotonic() < d5:
        cand = best * np.exp(rng.normal(0.0, 0.012, n))
        cand = np.clip(cand, 0.0, None)
        if cand.sum() > 0.01:
            a, s = _desc(cand, fs, ts, 0.15, 0.87, d5, 3e-4, 50,
                         1200, False, rng, n, 0.02)
            if s < best_sc:
                best_sc, best = s, a

    # Phase 6: Local search with exact evaluator
    d6 = dl - 3.0
    if time.monotonic() < d6:
        base = best.copy()
        w = max(3, n // 70)
        while time.monotonic() < d6:
            cand = base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - w)))
                e = min(n, st + w)
                cand[st:e] *= float(np.exp(rng.normal(0.0, 0.025)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = np.clip(p, 0.0, 1000.0)
    return [float(x) for x in out]
