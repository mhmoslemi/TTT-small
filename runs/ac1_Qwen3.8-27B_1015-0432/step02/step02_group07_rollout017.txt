            best_sc, best = s2, c2

    # Phase 3: alternative initialisation (raised-cosine bump)
    if time.monotonic() < dl - 35.0 and best_sc > 1.503:
        t3 = t0 + 0.86 * (total - 18.0)
        ii = np.arange(n, dtype=np.float64)
        ag = 0.5 * (1.0 + np.cos(2.0 * np.pi * (ii / (n - 1) - 0.5)))
        ag /= ag.sum()
        ag = descent(ag, n, fs, t3, rng)
        cg = np.clip(ag * s0, 0.0, 1000.0)
        sg = float(evaluate_sequence(cg.tolist()))
        if np.isfinite(sg) and sg < best_sc:
            best_sc, best = sg, cg

    # Phase 4: local perturbations
    while time.monotonic() < dl - 4.0:
        c = best.copy()
        w = max(3, n // 50)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            c[st:st + w] *= np.exp(float(rng.normal(0.0, 0.012)))
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best = sc, c

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
