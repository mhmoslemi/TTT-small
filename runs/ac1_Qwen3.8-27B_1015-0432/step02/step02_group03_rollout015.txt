import numpy as np
import time


def _fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def _descent(a0, fs, n, eta0, beta_sm, frac, cap, stall_max, t_dl, multi):
    a = a0.copy()
    c = _fft_conv(a, fs)[:2 * n - 1]
    best_l = a.copy()
    best_s = 2.0 * n * float(c.max())
    stall = 0
    prev_s = best_s * 2
    eta = eta0
    for _ in range(2000000):
        if time.monotonic() > t_dl:
            break
        fa = np.fft.rfft(a, n=fs)
        c = np.fft.irfft(fa * fa, n=fs)[:2 * n - 1]
        mx = float(c.max())
        sc = 2.0 * n * mx
        if sc < best_s - 1e-10:
            best_s = sc
            best_l = a.copy()
            stall = 0
        elif sc < prev_s - 1e-13:
            stall = 0
        else:
            stall += 1
        prev_s = sc
        if stall > stall_max:
            break
        # Build weight vector for gradient
        c_sh = c - mx
        if beta_sm > 0:
            w = np.exp(beta_sm * c_sh)
            w /= w.sum()
        else:
            thr = mx * (1.0 - frac)
            act = np.where(c >= thr)[0]
            if len(act) > cap:
                idx = np.argpartition(c[act], -cap)[-cap:]
                act = act[idx]
            w = np.zeros(2 * n - 1)
            w[act] = 1.0 / len(act)
        # Cross-correlation gradient via FFT
        b = np.zeros(fs)
        b[:2 * n - 1] = w
        fb = np.fft.rfft(b, n=fs)
        g = np.fft.irfft(fa * np.conj(fb), n=fs)[:n]
        g -= g.mean()
        if multi:
            gm = float(np.abs(g).max())
            if gm > 1e-15:
                a = a * np.exp(-eta * g / gm)
                a = np.maximum(a, 1e-15)
                a /= a.sum()
        else:
            a = np.maximum(a - eta * g, 0.0)
            s = a.sum()
            if s < 1e-10:
                break
            a /= s
        eta *= 0.99998
    return best_l, best_s
