import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    ca = np.fft.rfft(a, size)
    cb = np.fft.rfft(b, size)
    return np.fft.irfft(ca * cb, size)[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    C = fft_conv(f, f)
    return float(2 * len(f) * C.max() / (s * s))

def mirror_opt(f, deadline, rng, eta0=0.15, T0=1.0):
    n = f.size
    f = f.copy()
    best_f = f.copy()
    best_sc = score(f)
    eta = eta0
    T = T0
    no_imp = 0
    
    for it in range(500000):
        if time.monotonic() > deadline - 5:
            break
        C = fft_conv(f, f)
        Cmax = C.max()
        sc = 2.0 * n * Cmax / (f.sum() ** 2)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Gradient: combine argmax with smooth
        k_star = int(np.argmax(C))
        # Gradient at argmax
        lo = max(0, k_star - n + 1)
        hi = min(k_star, n - 1)
        grad = np.zeros(n)
        if lo <= hi:
            grad[lo:hi+1] = 2.0 * f[k_star - hi : k_star - lo + 1][::-1]
        
        # Smooth gradient (softmax over C)
        cs = C - Cmax
        # Only use top portion for efficiency
        p = np.exp(cs / T)
        p /= p.sum()
        N = 1 << (C.size + n - 1).bit_length()
        P = np.fft.rfft(p, N)
        F = np.fft.rfft(f, N)
        grad_smooth = 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]
        
        # Combine
        if T > 0.1:
            grad = 0.5 * grad + 0.5 * grad_smooth
        else:
            grad = grad
        
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        
        # Adaptive eta
        if it % 10 == 0:
            ns = score(f)
            if ns < sc - 1e-10:
                eta = min(eta * 1.05, 2.0)
            else:
                eta = max(eta * 0.8, 0.005)
        
        if it % 200 == 0:
            T = max(0.01, T * 0.9)
        
        if no_imp > 300:
            # Restart with perturbation
            f = best_f * np.exp(0.05 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.05
            T = 0.5
    
    return best_f, best_sc
