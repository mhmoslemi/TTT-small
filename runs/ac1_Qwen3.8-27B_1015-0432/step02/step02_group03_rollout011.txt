        npert = int(rng.integers(1, 5))
        for _ in range(npert):
            st = int(rng.integers(0, max(1, n - w)))
            ln = int(rng.integers(max(2, w // 3), w + 1))
            e = min(n, st + ln)
            cand[st:e] *= np.exp(rng.normal(0.0, 0.035))
        cand = np.clip(cand, 0.0, None)
        s = float(cand.sum())
        if s < 1e-10:
            continue
        sc = 2.0 * n * float(_conv(cand, fs).max()) / (s * s)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
    return best, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + max(1.0, float(budget_s)) - 12.0
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(512)
    n = p.size
    fs = 1 << (2 * n - 1).bit_length()
    ts = float(p.sum())
    a0 = p / ts
    best = a0.copy()
    best_sc = 2.0 * n * float(_conv(best, fs).max())
    # Phase 1: Euclidean descent with momentum from parent
    a, sc = _desc(a0, fs, t_end - 420, 0.55, 0.85, 0.02, 300, 12.0, False, 12000, rng)
    if sc < best_sc:
        best_sc, best = sc, a
    # Phase 2: Multiplicative descent for geometry change
    if time.monotonic() < t_end - 300:
        a, sc = _desc(best, fs, t_end - 260, 0.35, 0.0, 0.03, 400, 18.0, True, 8000, rng)
        if sc < best_sc:
            best_sc, best = sc, a
    # Phase 3: Fine Euclidean polish with tight threshold
    if time.monotonic() < t_end - 150:
        a, sc = _desc(best, fs, t_end - 80, 0.25, 0.9, 0.008, 250, 25.0, False, 15000, rng)
        if sc < best_sc:
            best_sc, best = sc, a
    # Phase 4: Fine multiplicative polish
    if time.monotonic() < t_end - 40:
        a, sc = _desc(best, fs, t_end - 15, 0.12, 0.0, 0.004, 350, 35.0, True, 10000, rng)
        if sc < best_sc:
            best_sc, best = sc, a
    # Phase 5: Local search with exact scoring
    if time.monotonic() < t_end - 2:
        best, best_sc = _local_search(best, n, fs, t_end - 0.5, rng, best_sc)
    # Final output
    out = np.clip(best * ts, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = np.clip(p, 0.0, 1000.0)
    fs_final = evaluate_sequence(out.tolist())
    if not np.isfinite(fs_final):
        out = np.clip(p, 0.0, 1000.0)
    return [float(x) for x in out]
