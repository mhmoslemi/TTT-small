        if gmax < 1e-30:
            break

        f_new = f * np.exp(-eta * grad / gmax)
        f_new = np.maximum(f_new, 1e-25)
        f_new /= f_new.sum()
        f = f_new
        it += 1

        if it % 30 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1

            if stall % 50 == 0 and stall > 0:
                eta *= 0.96
                beta *= 1.03

            if stall >= 600:
                f = best * np.exp(0.03 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                beta = max(10.0, beta0)
                eta = min(0.3, eta0)
                stall = 0

            if stall >= 2000:
                break

    return best, best_sc


def make_init_sym(n, func):
    x = np.arange(n, dtype=np.float64)
    a = func(x, n)
    a = np.maximum(a, 1e-15)
    a = (a + a[::-1]) / 2.0
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def init_hann(x, n):
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))


def init_cos(x, n):
    return 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))


def init_exp_sym(x, n, r):
    return np.exp(-r * x / n) + np.exp(-r * (n - 1 - x) / n)


def init_blackman(x, n):
    return 0.42 - 0.5 * np.cos(2.0 * np.pi * x / n) + 0.08 * np.cos(4.0 * np.pi * x / n)


def init_sinc_like(x, n):
    t = 2.0 * x / n - 1.0
    return 1.0 / (1.0 + 4.0 * t * t)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = p / p.sum()
    ps = (p + p[::-1]) / 2.0
    ps = np.maximum(ps, 1e-25)
    ps /= ps.sum()

    best_f = ps.copy()
    best_score = fast_score(ps)

    inits = [ps, p]
    x = np.arange(n, dtype=np.float64)

    inits.append(make_init_sym(n, init_hann))
    inits.append(make_init_sym(n, init_cos))
    inits.append(make_init_sym(n, init_blackman))
    inits.append(make_init_sym(n, init_sinc_like))
    for r in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        inits.append(make_init_sym(n, lambda xv, nv, _r=r: init_exp_sym(xv, nv, _r)))

    for s in [0.1, 0.2, 0.35]:
        t = ps * np.exp(s * rng.standard_normal(n))
        t = np.maximum(t, 1e-25)
        t = (t + t[::-1]) / 2.0
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())

    for ini in inits:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = ini.copy()

    f = best_f.copy()
    time_left = deadline - t0
    n_runs = 0
    max_runs = 8

    while time.monotonic() < deadline - 30.0 and n_runs < max_runs:
        run_time = min(120.0, (deadline - t0 - 30.0) / (max_runs - n_runs))
        run_dl = time.monotonic() + run_time

        if n_runs > 0:
            f = best_f * np.exp(0.02 * rng.standard_normal(n))
            f = np.maximum(f, 1e-25)
            f = (f + f[::-1]) / 2.0
            f = np.maximum(f, 1e-25)
            f /= f.sum()

        beta0 = 15.0 + 10.0 * n_runs
        eta0 = 0.25 / (1.0 + 0.1 * n_runs)
        f_opt, sc_opt = grad_descent(f, run_dl, rng, beta0, eta0, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
        n_runs += 1

    # Fine polish phase
    if time.monotonic() < deadline - 20.0:
        polish_dl = deadline - 5.0
        best_f, best_score = grad_descent(best_f, polish_dl, rng, 80.0, 0.05, n)

    # Final symmetrization check
    fs = (best_f + best_f[::-1]) / 2.0
    fs = np.maximum(fs, 1e-25)
    fs /= fs.sum()
    sc_s = fast_score(fs)
    if sc_s < best_score:
        best_f = fs
        best_score = sc_s

    final_score = evaluate_sequence([float(v) for v in best_f])
    if not np.isfinite(final_score):
        best_f = np.ones(n) / n
        final_score = evaluate_sequence([float(v) for v in best_f])

    return [float(v) for v in best_f]
