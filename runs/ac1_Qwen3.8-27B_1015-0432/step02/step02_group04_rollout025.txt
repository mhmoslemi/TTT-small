    eta = 0.08
    T = 2.0
    best_f = f.copy()
    best_sc = fast_score(f)
    no_imp = 0
    for it in range(max_it):
        if time.monotonic() > deadline - 4.0:
            break
        C = fft_conv(f, f)
        sc = 2.0 * n * C.max() / (f.sum()**2)
        if sc < best_sc - 1e-9:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
        r = rng.random()
        if r < 0.35:
            grad = grad_argmax(f, int(np.argmax(C)))
        elif r < 0.70:
            T_use = max(0.1, T * 0.4)
            grad = grad_smooth(f, C, T_use)
        elif r < 0.85:
            T_use = max(0.3, T)
            grad = grad_smooth(f, C, T_use)
        else:
            g1 = grad_argmax(f, int(np.argmax(C)))
            g2 = grad_smooth(f, C, max(0.2, T*0.5))
            gm = max(g1.max(), g2.max(), 1e-30)
            grad = 0.5*(g1 + g2)
        gmax = grad.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        if it % 12 == 0:
            ns = fast_score(f)
            if ns < sc - 1e-8:
                eta = min(eta * 1.06, 2.5)
            else:
                eta = max(eta * 0.82, 0.003)
        if it % 250 == 0:
            T = max(0.04, T * 0.92)
        if no_imp > 350:
            f = best_f * np.exp(0.06 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            no_imp = 0
            eta = 0.08
            T = 1.5
    return best_f, best_sc

def ridgeline_step(f, deadline, rng):
    n = f.size
    best_f = f.copy()
    best_sc = fast_score(f)
    directions = []
    inv_f = 1.0 / np.maximum(f, 1e-20)
    thr = np.percentile(inv_f, 92)
    g_lp = np.maximum(inv_f - thr, 0.0)
    if g_lp.sum() > 1e-15:
        directions.append(g_lp / g_lp.sum())
    deficit = 1.0/n - f
    g_sp = np.maximum(deficit, 0.0)
    if g_sp.sum() > 1e-15:
        directions.append(g_sp / g_sp.sum())
    j = np.arange(n, dtype=np.float64)
    for cw in [0.1, 0.2, 0.35]:
        gb = np.exp(-((j - n*0.5)/(n*cw))**2)
        gb /= gb.sum()
        directions.append(gb)
    for g in directions:
        if time.monotonic() > deadline - 8.0:
            break
        for t in [0.05, 0.1, 0.2, 0.3, 0.45]:
            cand = (1.0 - t)*best_f + t*g
            cand = np.maximum(cand, 1e-20)
            cand /= cand.sum()
            sc = fast_score(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_f = cand.copy()
        to = time.monotonic() + min(100.0, deadline - time.monotonic() - 10.0)
        if to > time.monotonic():
            f2, sc2 = mirror_optimize(best_f, to, rng, max_it=60000)
            if sc2 < best_sc:
                best_sc, best_f = sc2, f2
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 14.0
    n = len(height_sequence_1)
    inits = []
    pp = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if pp.size == n and np.all(np.isfinite(pp)) and pp.sum() > 1e-10:
        inits.append(pp / pp.sum())
    for a in [0.0, 0.4, -0.4, 0.8, -0.8]:
        inits.append(make_flat_top(n, a))
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        inits.append(make_ushape(n, alpha, 0.0))
    inits.append(make_ushape(n, 2.0, 0.5))
    inits.append(make_ushape(n, 2.0, -0.5))
    for a, b in [(0.5, 0.5), (1.0, 0.0), (0.0, 1.0), (0.3, 0.7)]:
        inits.append(make_powlaw(n, a, b))
    inits.append(np.ones(n) / n)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    if best_f is None:
        best_f = np.ones(n) / n
        best_sc = 2.0
    # Phase 1: main mirror descent
    to1 = t0 + min(420.0, deadline - t0 - 40.0)
    f1, sc1 = mirror_optimize(best_f, to1, rng, max_it=300000)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1
    # Phase 2: ridge-line
    if time.monotonic() < deadline - 50.0:
        to2 = time.monotonic() + min(280.0, deadline - time.monotonic() - 20.0)
        f2, sc2 = ridgeline_step(best_f, to2, rng)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2
    # Phase 3: iterate mirror + perturb
    while time.monotonic() < deadline - 22.0:
        it_time = min(160.0, deadline - time.monotonic() - 14.0)
        if it_time < 25.0:
            break
        to3 = time.monotonic() + it_time
        f3, sc3 = mirror_optimize(best_f, to3, rng, max_it=120000)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3
        if time.monotonic() < deadline - 70.0:
            trial = best_f * np.exp(0.07 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            to4 = time.monotonic() + min(90.0, deadline - time.monotonic() - 14.0)
            if to4 > time.monotonic():
                f4, sc4 = mirror_optimize(trial, to4, rng, max_it=60000)
                if sc4 < best_sc:
                    best_sc, best_f = sc4, f4
    # Final polish
    if time.monotonic() < deadline - 8.0:
        to5 = deadline - 6.0
        f5, sc5 = mirror_optimize(best_f, to5, rng, max_it=100000)
        if sc5 < best_sc:
            best_sc, best_f = sc5, f5
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0] * n
    return result
