```python
import scipy.sparse as sp
import cvxpy as cp

def solve_lp(f, M):
    n = len(f)
    L = 2*n - 1
    # Build A matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    rows, cols, vals = [], [], []
    for k in range(L):
        for j in range(max(0, k-n+1), min(n, k+1)):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[idx])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(L, n))
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, max_time=30)
    return np.asarray(g.value).flatten()
```

```python
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_fn(f):
    s = f.sum()
    if s < 0.01:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def build_conv_matrix(f):
    """Build sparse matrix A such that A @ g = convolve(f, g)"""
    n = len(f)
    L = 2 * n - 1
    # Column j: A[k, j] = f[k-j] for k in [j, j+n-1]
    # Total nonzeros: n * n
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return sp.csr_matrix((data, (row_idx, col_idx)), shape=(L, n))

def solve_lp(f, M, time_limit=30):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0"""
    n = len(f)
    L = 2 * n - 1
    A = build_conv_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(L, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.status not in (0, 2):  # 2 = time limit
        return None
    return res.x
```

```python
def lp_refine(f, deadline, rng, max_iters=30):
    """LP-based refinement (SOTA style)."""
    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score_fn(f)
    
    # Build sparse structure once
    col_idx = np.repeat(np.arange(n), n)
    row_idx = col_idx + np.tile(np.arange(n), n)
    
    M = np.max(fft_conv(f, f))
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Build A matrix (values change, structure fixed)
        data = np.tile(f, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        
        # Solve: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        
        # Time limit for this LP
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        t_limit = min(30.0, remaining * 0.8)
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                     bounds=[(0, np.inf)] * n,
                     method='highs',
                     options={'time_limit': t_limit, 'presolve': True})
        
        if res.x is None:
            break
        g = np.maximum(res.x, 0)
        gsum = g.sum()
        if gsum < 1e-15:
            break
        g = g / gsum
        
        # Blend with current f
        alpha = 0.5 / (1.0 + 0.1 * it)  # decreasing blend
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()
        
        sc = score_fn(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
        f = f_new
        M = np.max(fft_conv(f, f))
    
    return best_f, best_sc

        r = np.convolve(r, kernel, 'same')
        inits.append(r / r.sum())

    best_f = None
    best_sc = np.inf
    for init in inits:
        init = normalize(init)
        sc = score_fn(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    
    # Phase 1: Subgradient optimization
    p1_end = min(deadline, t0 + 350.0)
    f1, sc1 = subgrad_opt(best_f.copy(), p1_end, rng, eta_init=0.04)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()
    
    # Phase 2: LP refinement (if time permits)
    if time.monotonic() < deadline - 200.0:
        p2_end = min(deadline, t0 + 750.0)
        f2, sc2 = lp_refine(best_f.copy(), p2_end, rng, max_iters=25)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2.copy()
    
    # Phase 3: Final subgradient polish
    if time.monotonic() < deadline - 30.0:
        f3, sc3 = subgrad_opt(best_f.copy(), deadline - 10.0, rng, eta_init=0.02)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3.copy()

    if time.monotonic() < deadline - 5.0:
        for _ in range(50):

            trial = best_f * np.exp(0.005 * rng.standard_normal(n))
            trial = normalize(trial)

        fin = evaluate_sequence(result)
    return result
```

```python
import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score_fn(f):
    s = f.sum()
    if s < 1e-6:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def normalize(f):
    f = np.clip(f, 0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def subgrad_opt(f, deadline, rng, eta_init=0.04, sym=False):
    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score_fn(f)
    eta = eta_init
    no_improv = 0
    restarts = 0
    it = 0
    while time.monotonic() < deadline and it < 600000:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        frac = 0.999 - 0.015 * min(1.0, it / 3000.0)
        thresh = M * frac
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 250:
            peaks = np.argsort(b)[::-1][:250]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])
        h = np.zeros(2 * n - 1)
        h[peaks] = 1.0
        c = fft_conv(f, h[::-1])
        g = 2.0 * c[n - 1:2 * n - 1][::-1]
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f = f / f.sum()
        if it % 40 == 0:
            sc = score_fn(f)
            if sc < best_sc - 5e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            eta *= 0.9985
            if no_improv > 400:
                scale = 0.03 + 0.06 * min(1.0, restarts / 8.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = f / f.sum()
                no_improv = 0
                restarts += 1
                eta = eta_init * (1.0 + 0.3 * min(restarts, 6))
    return best_f, best_sc

    n = len(f)
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score_fn(f)
    for it in range(max_iters):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        b = fft_conv(f, f)
        M = b.max()
        thresh = M * 0.92
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > K:
            peaks = np.argsort(b)[::-1][:K]
        if len(peaks) < 10:
            peaks = np.arange(2 * n - 1)
        rows_parts = []
        cols_parts = []
        vals_parts = []
        for k in peaks:
            j_lo = max(0, k - n + 1)
            j_hi = min(n - 1, k)
            jj = np.arange(j_lo, j_hi + 1)

        rows = np.concatenate(rows_parts)

        c_obj = -np.ones(n)
```
