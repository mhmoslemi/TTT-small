def fft_conv(a, b):
    L = 1 << (len(a) + len(b) - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L))[:len(a)+len(b)-1]

def sc(f):
    s = f.sum()
    if s < 1e-15:
        return 1e18
    return float(2*f.size*fft_conv(f, f).max()/(s*s))
