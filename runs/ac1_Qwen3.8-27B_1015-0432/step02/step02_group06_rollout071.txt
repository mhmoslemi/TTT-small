import numpy as np

def _self_conv(a):
    n = int(a.size)
    L = 2 * n - 1
    size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:L]

def _cross_conv(a, b):
    L = int(a.size) + int(b.size) - 1
    size = 1 << (L - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:L]

import numpy as np
import time


def _self_conv(a):
    n = int(a.size)
    L = 2 * n - 1
    size = 1 << (L - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:L]


def _cross_conv(a, b):
    L = int(a.size) + int(b.size) - 1
    size = 1 << (L - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:L]


def _score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = _self_conv(f)
    return float(2.0 * len(f) * b.max() / (s * s))


def _normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.size, dtype=np.float64) / float(f.size)
    return f / s


def _init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(n - 1, 1)))
    return _normalize(h)


def _init_supergauss(n, p):
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-((x - c) / (0.35 * n)) ** p)
    return _normalize(a)


def _init_two_lobe(n):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-5.0 * ((x / n - 0.33) / 0.23) ** 2)
    a += 0.35 * np.exp(-5.0 * ((x / n - 0.67) / 0.23) ** 2)
    return _normalize(a)


def _optimize(f, t_stop, rng, beta, eta0, score_every=5):
    n = int(f.size)
    f = _normalize(np.array(f, dtype=np.float64, copy=True))
    best = f.copy()
    b = _self_conv(f)
    best_sc = float(2.0 * n * b.max() / (f.sum() * f.sum()))
    if not np.isfinite(best_sc):
        return best, np.inf

    eta = float(eta0)
    eta_min = max(1e-5, eta * 0.1)
    stall = 0
    it = 0
    ls = np.empty(n, dtype=np.float64)

    while time.monotonic() < t_stop:
        M = b.max()
        if not np.isfinite(M) or M <= 1e-300:
            break

        b *= 1.0 / M
        b -= 1.0
        b *= float(beta)
        np.clip(b, -60.0, 0.0, out=b)
        ws = b.sum()
        if ws <= 1e-300:
            break
        b *= 1.0 / ws

        gc = _cross_conv(b, f[::-1])
        grad = gc[n - 1:2 * n - 1]
        np.maximum(grad, 1e-30, out=grad)
        gm = grad.max()
        if gm <= 1e-30:
            break

        np.multiply(grad, -eta / gm, out=ls)
        np.clip(ls, -12.0, 0.0, out=ls)
        np.exp(ls, out=ls)

        f *= ls
        np.maximum(f, 1e-25, out=f)
        f /= f.sum()

        it += 1
        if time.monotonic() >= t_stop:
            break

        b = _self_conv(f)
        if it % score_every == 0:
            ss = f.sum()
            sc = float(2.0 * n * b.max() / (ss * ss))
            if np.isfinite(sc) and sc < best_sc - 1e-11:
                best_sc = sc
                best = f.copy()
                stall = 0
                eta = min(eta * 1.04, eta0 * 2.0)
            else:
                f = best.copy()
                b = _self_conv(f)
                eta = max(eta_min, eta * 0.95)
                stall += 1
                if stall >= 40:
                    f = best * np.exp(0.015 * rng.standard_normal(n))
                    f = _normalize(f)
                    b = _self_conv(f)
                    eta = max(eta_min, eta0 * 0.6)
                    stall = 0
                elif stall == 20:

                    f = _normalize(f)
                    b = _self_conv(f)
                    eta = max(eta_min, eta0 * 0.8)

    return best, best_sc

def _local_search(f, t_stop, rng):

    best = _normalize(np.array(f, dtype=np.float64, copy=True))

    st_blk = int(max(8, min(128, n // 32)))
    it = 0

    while time.monotonic() < t_stop:
        trial = best.copy()
        r = rng.random()

        if r < 0.35:
            trial *= np.exp(0.0015 * rng.standard_normal(n))
        elif r < 0.70:
            st = int(rng.integers(0, max(1, n - st_blk)))

            it = 0
