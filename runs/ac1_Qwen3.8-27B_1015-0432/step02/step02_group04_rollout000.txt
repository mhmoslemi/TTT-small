    inits.append(make_flat_top(n, -0.4))
    inits.append(make_two_hump(n))
    inits.append(np.ones(n) / n)
    best_f, best_sc = inits[0], fast_score(inits[0])
    for ini in inits[1:]:
        sc = fast_score(ini)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best_f = sc, ini.copy()
    # Main loop: alternate optimization and diversification
    phase = 0
    while time.monotonic() < deadline - 30.0:
        phase += 1
        to = time.monotonic() + min(200.0, deadline - time.monotonic() - 20.0)
        f1, sc1 = mirror_opt(best_f, to, rng)
        if sc1 < best_sc:
            best_sc, best_f = sc1, f1
        if time.monotonic() > deadline - 50.0:
            break
        # Try diversified restarts
        for trial in diversify(best_f, rng):
            to2 = time.monotonic() + min(60.0, deadline - time.monotonic() - 15.0)
            if to2 <= time.monotonic():
                break
            f2, sc2 = mirror_opt(trial, to2, rng)
            if sc2 < best_sc:
                best_sc, best_f = sc2, f2
            if time.monotonic() > deadline - 40.0:
                break
        # Periodically try fresh parametric inits
        if phase % 3 == 0 and time.monotonic() < deadline - 80.0:
            for ini in [make_flat_top(n, rng.uniform(-1, 1)), make_two_hump(n)]:
                to3 = time.monotonic() + min(80.0, deadline - time.monotonic() - 15.0)
                if to3 <= time.monotonic():
                    break
                f3, sc3 = mirror_opt(ini, to3, rng)
                if sc3 < best_sc:
                    best_sc, best_f = sc3, f3
    # Final polish: fine-tune with small T
    to_final = time.monotonic() + min(60.0, deadline - time.monotonic() - 5.0)
    if to_final > time.monotonic():
        f4, sc4 = mirror_opt(best_f, to_final, rng, T0=0.5, Tcool=0.95, Tmax=1.0)
        if sc4 < best_sc:
            best_sc, best_f = sc4, f4
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    if evaluate_sequence(result) == np.inf:
        result = [1.0] * n
    return result
