def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    sz = max(sz, 2)
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb)[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def softmax_grad(f, b, tau):
    n = f.shape[0]
    L = 2 * n - 1
    mb = b.max()
    if mb < 1e-30:
        return np.zeros(n)
    e = np.exp((b - mb) / tau)
    w = e / e.sum()
    wr = w[::-1]

    g = np.fft.irfft(np.fft.rfft(f, sz) * np.fft.rfft(wr, sz))[:n + L - 1]
