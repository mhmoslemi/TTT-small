def _pnorm_descent(a, fs, p, eta0, beta, max_iter, dl, p_anneal):

    a is normalized to sum=1.

    n = a.size
    best_a = a.copy()
    best_score = 2.0 * n * float(_conv(a, fs).max())

    eta = eta0

        C = np.fft.irfft(A * A, n=fs)
        c_max = float(C.max())

        # Compute p-norm gradient

        w = cw ** pm1  # C_k^(p-1)
        denom = float(np.sum(w * cw))  # = sum C_k^p

        eta *= (1.0 - eta0 * 0.1 / max_iter)  # gentle decay

import time

def _pnorm_desc(a, n, fs, dl, p0, p1, eta0, beta, max_stall):
    """P-norm gradient descent with Nesterov momentum. a has sum=1."""
    best_a = a.copy()
    best_sc = 2.0 * n * float(np.fft.irfft(np.fft.rfft(a, n=fs)**2, n=fs).max())
    v = np.zeros(n)
    eta = eta0
    p = p0
    stall = 0
    n_iter = 0
    while time.monotonic() < dl:
        A = np.fft.rfft(a, n=fs)
        C = np.fft.irfft(A * A, n=fs)
        sc = 2.0 * n * float(C.max())
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            break
        n_iter += 1
        # P-norm gradient
        pm1 = max(0.5, p - 1.0)
        cw = C[:2*n-1]
        w = cw ** pm1
        s = float(np.sum(w * cw))
        if s < 1e-30:
            break
        denom = s ** (pm1 / p)
        w_rev = w[::-1]
        W = np.fft.rfft(w_rev, n=fs)
        cv = np.fft.irfft(A * W, n=fs)
        num = cv[n-1:2*n-1][::-1]
        grad = 2.0 * num / denom
        # Nesterov momentum
        v = beta * v - eta * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            a = np.ones(n) / n
        else:
            a = a_new / sn
        eta *= 0.999999
        p += (p1 - p) * 0.0001
    return best_a, best_sc

    """Subgradient descent on max(C). a has sum=1."""
    best_a = a.copy()
    best_sc = 2.0 * n
    best_sc = 2.0 * n * float(np.fft.irfft(np.fft.rfft(a, n=fs)**2, n=fs).max())

        C = np.fft.irfft(A * A, n=fs)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > max_stall:
            break
        mx = float(C.max())
        act = np.where(C >= mx * (1.0 - frac))[0]
        act = act[(act >= 0) & (act < 2*n-1)]
        if len(act) > cap:
            idx = np.argpartition(C[act], -cap)[-cap:]
            act = act[idx]
        g = np.zeros(n)
        for k in act:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n-1, k)
            if lo <= hi:
                g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
        if act.size > 0:
            g /= act.size
        g -= g.mean()

            gm = np.abs(g).max()
            if gm > 1e-12:
                a = a * np.exp(-eta * g / gm)
        else:
            v = beta * v + (1.0 - beta) * g
            a = np.maximum(a - eta * v, 0.0)
        s = a.sum()
        if s < 1e-10:

        else:
            a /= s
        eta *= 0.99999
