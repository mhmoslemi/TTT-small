import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_nof(fn):
    """Score with sum=1 assumption."""
    b = fft_conv(fn, fn)
    return 2.0 * fn.size * b.max()

def smooth_convolutions(fn, t):
    """Compute softmax(t * conv) weights."""
    conv = fft_conv(fn, fn)
    m = conv.max()
    shifted = t * (conv - m)
    exp_v = np.exp(shifted)
    s = exp_v.sum()
    if s < 1e-300:
        p = np.zeros_like(conv)
        p[conv.argmax()] = 1.0
    else:
        p = exp_v / s
    return p, conv

def grad_smooth(fn, p):
    """Gradient of smooth max, given softmax weights p."""
    n = fn.size
    h = p
    c = fft_conv(fn, h[::-1])
    g = 2.0 * c[n-1:2*n-1][::-1]
    return g

def optimize_smooth(fn, deadline, rng, t_start=50, t_end=500, n_iter=5000):
    """Annealed smooth max minimization."""
    n = fn.size
    fn = np.maximum(fn, 1e-20)
    fn = fn / fn.sum()
    best_f = fn.copy()
    best_sc = score_nof(fn)
    eta = 0.05
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        # Annealing schedule
        frac = min(1.0, it / n_iter * 1.5)
        t = t_start * (t_end / t_start) ** frac
        
        p, conv = smooth_convolutions(fn, t)
        g = grad_smooth(fn, p)
        gmax = g.max()
        if gmax < 1e-30:
            break
        
        # Gradient step with multiplicative update
        fn = fn * np.exp(-eta * g / gmax)
        fn = np.maximum(fn, 1e-20)
        fn /= fn.sum()
        
        # Periodic evaluation
        if it % 200 == 0:
            sc = score_nof(fn)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = fn.copy()
            # Decay eta slowly
            eta *= 0.998
    
    return best_f, best_sc

Actually, for a "Barker-like" nonnegative sequence... this is a well-studied problem. The optimal nonnegative sequence for minimizing peak sidelobe level (PSL) has been studied in radar/communications.

One known good construction is the "Thomas-Furmargent" sequence or similar, but those are typically for complex sequences.

For real nonnegative sequences, the problem is equivalent to designing a window function with minimal peak sidelobe. The optimal window in this sense tends to be somewhat like a raised cosine or prolate spheroidal function.

    return w

def anneal_opt(fn, deadline, rng, t_start=5.0, t_end=500.0, sym=False, max_it=200000):
    n = fn.size
    fn = np.maximum(fn, 1e-15).astype(np.float64)
    fn /= fn.sum()
    bf = fn.copy()
    bs = score_raw(fn)
    eta = 0.06
    stall = 0
    for it in range(max_it):
        if time.monotonic() > deadline - 5.0:
            break
        frac = min(1.0, it / (max_it * 0.7))
        t = t_start * (t_end / t_start) ** frac
        conv = fft_conv(fn, fn)
        w = soft_weights(conv, t)
        g = grad_w(fn, w)
        gm = g.max()
        if gm < 1e-28:
            break
        fn = fn * np.exp(-eta * g / gm)
        fn = np.maximum(fn, 1e-15)
        if sym:
            fn = 0.5 * (fn + fn[::-1])
        fn /= fn.sum()
        if it % 128 == 0:
            sc = score_raw(fn)
            if sc < bs - 1e-13:
                bs = sc
                bf = fn.copy()
                stall = 0
            else:
                stall += 1
        if stall > 256:
            noise = 0.03 + 0.07 * min(1.0, stall / 2000.0)
            fn = bf * np.exp(noise * rng.standard_normal(n))
            fn = np.maximum(fn, 1e-15)
            if sym:
                fn = 0.5 * (fn + fn[::-1])
            fn /= fn.sum()
            stall = 0
            eta = min(0.35, eta * 1.6)
        eta *= 0.9992
    return bf, bs
