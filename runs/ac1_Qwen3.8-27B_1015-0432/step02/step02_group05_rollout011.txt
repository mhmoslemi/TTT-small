```python
def fft_conv(a):
    n = len(a)
    fsz = 1 << (2*n - 1).bit_length()
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)[:2*n-1]

def score(a):
    n = len(a)
    if a.sum() < 0.01:
        return float('inf')
    cv = fft_conv(a)
    return 2.0 * n * cv.max() / (a.sum()**2)

def optimize(a, deadline, rng):
    n = len(a)
    a = np.maximum(a, 1e-15).astype(np.float64)
    a /= a.sum()
    
    best_a = a.copy()
    cv = fft_conv(a)
    best_cv = cv.max()
    
    # Adaptive parameters
    eta = 0.3  # learning rate
    p = 0.9997  # decay
    patience = 0
    max_patience = 3000
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        cv = fft_conv(a)
        m = cv.max()
        
        if m < best_cv - 1e-14:
            best_cv = m
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
        
        # Active set: lags near the maximum
        thresh = m * (1.0 - 0.05)  # 5% below max
        active = np.where(cv > thresh)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(cv)])
        
        # Gradient at active lags
        n_active = len(active)
        g = np.zeros(n)

            hi = min(n - 1, k)
            if lo <= hi:

        g /= n_active
        
        # Mirror descent update

        eta *= p
    
    return best_a
```

```python
def smooth_max_grad(a, beta):
    cv = fft_conv(a)
    cv -= cv.max()
    w = np.exp(beta * cv)
    w /= w.sum()
    # Gradient is weighted average of individual gradients
    n = len(a)
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] > 1e-15:
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            g[lo:hi+1] += w[k] * a[k-hi:k-lo+1][::-1]
    return g
```

But this loops over all 2n-1 lags, which is 9000 iterations with vectorized operations. Each iteration is O(n) in the worst case, so total is O(n²) ≈ 20M operations. That's about 10-20ms in numpy. For 10,000 iterations, that's 100-200 seconds. Feasible but tight.

I can optimize by only computing the gradient for lags where w[k] is significant (above some threshold).

Actually, let me reconsider. The smooth max gradient is:
∇ log-sum-exp(β*cv) / β = (1/w_sum) * sum_k w[k] * ∇cv[k]

where w[k] = exp(β*cv[k]). For large β, only the top few lags contribute significantly. So I only need to compute the gradient for the top ~20 lags.

Let me implement this:

```python
def SmoothMaxGrad(a, beta, n):
    cv = fft_conv(a)
    cv_shift = cv - cv.max()
    w = np.exp(beta * cv_shift)
    total = w.sum()
    if total < 1e-300:
        return np.zeros(n), cv.max()
    # Only top lags contribute
    top_k = min(50, 2*n-1)
    top_idx = np.argsort(w)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        wk = w[k] / total
        if wk < 1e-12:
            continue
        k = int(k)
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        if lo <= hi:
            g[lo:hi+1] += wk * a[k-hi:k-lo+1][::-1]
    return g, cv.max()
```

This is O(n * top_k) per evaluation, which is O(n * 50) ≈ 225K operations. Very fast!

Then I can use a simple projected gradient descent:
```
a -= lr * SmoothMaxGrad(a)
a = np.maximum(a, 0)
a /= a.sum()
```

or the mirror descent version:
```
a *= np.exp(-lr * SmoothMaxGrad(a))
a /= a.sum()
```

The mirror descent version naturally maintains non-negativity.

Let me combine both approaches: mirror descent with smooth max gradient, plus the threshold-based active set for the sharp max.

Actually, I think the most effective approach is:

1. Start with mirror descent using a smooth max (log-sum-exp with moderate β) for the first phase
2. Switch to the sharp max (threshold-based active set) for the refinement phase
3. Do local perturbations at the end

One more consideration: the `Score` function calls `evaluate_sequence` which does the full convolution. This is O(n log n) but with a Python→NumPy call overhead. For the local polish phase, I should call it sparingly.

def fft_conv(a):
    n = len(a)
    fsz = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)[:2 * n - 1]

def ActiveGrad(a, active, n):
    g = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return g / max(1, len(active))

def LSEGrad(a, beta, n):
    cv = fft_conv(a)
    c = cv.max()
    w = np.exp(beta * (cv - c))
    s = w.sum()
    if s < 1e-300:
        return np.zeros(n), c
    top = np.argsort(w)[-min(60, 2*n-1):]
    g = np.zeros(n)
    for idx in top:
        wi = w[idx] / s
        if wi < 1e-10:
            continue
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += wi * a[k-hi:k-lo+1][::-1]
    return g, c
