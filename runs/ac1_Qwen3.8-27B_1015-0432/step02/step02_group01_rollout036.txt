import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def obj_val(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * max(b.max(), 0.0)


def grad_sm(f, b, beta):
    M = b.max()
    if M < 1e-30:
        return None
    e = np.exp(beta * (b - M) / M)
    es = e.sum()
    if es < 1e-30:
        e = np.zeros_like(b)
        e[np.argmax(b)] = 1.0
    else:
        e /= es
    gc = fft_conv(e, f[::-1])
    g = gc[f.size - 1 : 2 * f.size - 1]
    return np.maximum(g, 0.0)
