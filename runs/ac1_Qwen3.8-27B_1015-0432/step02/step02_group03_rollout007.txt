        a, s = _des(p.copy(), fs, ts, 1.2, 0.86, dl_end - 0.70 * T,
                    0.12, 0.0008, 90, 9000, False, rng)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 2: Multiplicative descent
    if time.monotonic() < dl_end - 0.45 * T:
        a, s = _des(best.copy(), fs, ts, 0.5, 0.0, dl_end - 0.47 * T,
                    0.06, 0.0006, 110, 8000, True, rng)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 3: Fine Euclidean
    if time.monotonic() < dl_end - 0.28 * T:
        a, s = _des(best.copy(), fs, ts, 0.35, 0.90, dl_end - 0.30 * T,
                    0.015, 0.0004, 55, 7000, False, rng)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 4: Fine multiplicative
    if time.monotonic() < dl_end - 0.15 * T:
        a, s = _des(best.copy(), fs, ts, 0.18, 0.0, dl_end - 0.16 * T,
                    0.006, 0.0003, 45, 6000, True, rng)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 5: Very fine Euclidean polish
    if time.monotonic() < dl_end - 0.07 * T:
        a, s = _des(best.copy(), fs, ts, 0.09, 0.93, dl_end - 0.08 * T,
                    0.003, 0.0002, 30, 5000, False, rng)
        if s < best_sc:
            best_sc, best = s, a

    # Phase 6: Exact-evaluator local search
    if time.monotonic() < dl_end - 1.5:
        base = best.copy()
        w = max(2, n // 130)
        while time.monotonic() < dl_end - 0.5:
            cand = base.copy()
            nb = int(rng.integers(1, 4))
            for _ in range(nb):
                if rng.random() < 0.35:
                    i2 = int(rng.integers(0, n))
                    j2 = int(rng.integers(0, n))
                    d = 0.008 * cand.mean()
                    if cand[i2] > d:
                        cand[i2] -= d
                        cand[j2] += d
                else:
                    st2 = int(rng.integers(0, max(1, n - w)))
                    e2 = min(n, st2 + w)
                    cand[st2:e2] *= float(np.exp(rng.normal(0.0, 0.012)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = np.clip(p, 0.0, 1000.0)
    return [float(x) for x in out]
