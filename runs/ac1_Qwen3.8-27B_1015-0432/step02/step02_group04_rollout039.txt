import time

_LP_CACHE = {}

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def build_lp_matrix(n):
    """Build sparse structure for the LP constraint matrix."""
    key = n
    if key in _LP_CACHE:
        return _LP_CACHE[key]
    col_idx = np.tile(np.arange(n), n)
    row_offset = np.tile(np.arange(n), n)
    row_idx = np.repeat(np.arange(n), n) + row_offset
    valid = row_idx < (2 * n - 1)
    data = {
        'row_idx': row_idx[valid].astype(np.int32),
        'col_idx': col_idx[valid].astype(np.int32),
        'valid': valid,
        'nnz': int(valid.sum()),

    M = C.max()
    data = build_lp_matrix(n)
    val = np.repeat(f, n)[data['valid']]
    A = scipy.sparse.csr_array((val, (data['row_idx'], data['col_idx'])), shape=(2*n-1, n))

import numpy as np
import scipy.sparse
import scipy.optimize
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def grad_smooth(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def grad_topk(f, C, k):
    n = f.size
    idx = np.argpartition(C, -k)[-k:]
    p = np.zeros_like(C)
    p[idx] = 1.0 / k
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]
