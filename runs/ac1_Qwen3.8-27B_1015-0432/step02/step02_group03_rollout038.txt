    td = time.monotonic() + max(0.0, float(budget_s) - 14.0)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.size < 2 or not np.all(np.isfinite(p)) or p.sum() < 0.01:
        p = np.ones(512)
    n = p.size
    clip = 2 * n - 1
    fs = 1 << clip.bit_length()
    S = float(p.sum())

    best = p.copy()
    c0 = _fc(best, fs)
    best_sc = _scen(c0, best, n)
    bl = S

    # Phase 1: smooth soft-max descent, beta ramps 3->25
    a = p.copy()
    mom = np.zeros(n)
    eta = 0.6
    stall = 0
    psc = best_sc
    it = 0
    p1_end = td - 380.0
    while time.monotonic() < p1_end:
        it += 1
        c = _fc(a, fs)
        sc = _scen(c, a, n)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
        stall = 0 if sc < psc - 1e-14 else stall + 1
        psc = sc
        if stall > 4000:
            break
        ph = min(1.0, it / 25000.0)
        beta = 3.0 + 22.0 * ph ** 1.3
        g = _gsm(a, c, beta, n, fs, clip)
        gm = float(np.abs(g).max())
        if gm < 1e-14:
            break
        g -= g.mean()
        mom = 0.82 * mom + 0.18 * g
        a = np.maximum(a - eta * mom, 0.0)
        a = np.minimum(a, 2000.0)
        s = a.sum()
        if s < 1e-10:
            break
        a *= bl / s
        eta *= 0.99996
        if stall > 1200 and stall < 3500:
            a += 0.02 * (float(a.max()) + 1e-6) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a *= bl / a.sum()

    # Phase 2: multiplicative soft-max, beta 25->80
    if time.monotonic() < td - 220.0:
        a = best.copy()
        eta = 0.35
        stall = 0
        psc = best_sc
        p2_end = td - 210.0
        it2 = 0
        while time.monotonic() < p2_end:
            it2 += 1
            c = _fc(a, fs)
            sc = _scen(c, a, n)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = a.copy()
            stall = 0 if sc < psc - 1e-14 else stall + 1
            psc = sc
            if stall > 3000:
                break
            ph = min(1.0, it2 / 15000.0)
            beta = 25.0 + 55.0 * ph
            g = _gsm(a, c, beta, n, fs, clip)
            gm = float(np.abs(g).max())
            if gm < 1e-14:
                break
            g -= g.mean()
            a = np.maximum(a * np.exp(-eta * g / gm), 1e-12)
            a *= bl / a.sum()
            eta *= 0.99997
            if stall > 800 and stall < 2500:
                a += 0.01 * (float(a.max()) + 1e-6) * rng.standard_normal(n)
                a = np.maximum(a, 1e-12)
                a *= bl / a.sum()

    # Phase 3: fine additive polish, high beta
    if time.monotonic() < td - 100.0:
        a = best.copy()
        mom = np.zeros(n)
        eta = 0.18
        stall = 0
        psc = best_sc
        p3_end = td - 90.0
        while time.monotonic() < p3_end:
            c = _fc(a, fs)
            sc = _scen(c, a, n)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = a.copy()
            stall = 0 if sc < psc - 1e-14 else stall + 1
            psc = sc
            if stall > 2000:
                break
            beta = 80.0
            g = _gsm(a, c, beta, n, fs, clip)
            gm = float(np.abs(g).max())
            if gm < 1e-14:
                break
            g -= g.mean()
            mom = 0.85 * mom + 0.15 * g
            a = np.maximum(a - eta * mom, 0.0)
            a *= bl / a.sum()
            eta *= 0.99999

    # Phase 4: second multiplicative pass from best
    if time.monotonic() < td - 60.0:
        a = best.copy()
        eta = 0.2
        stall = 0
        psc = best_sc
        p4_end = td - 50.0
        while time.monotonic() < p4_end:
            c = _fc(a, fs)
            sc = _scen(c, a, n)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best = a.copy()
            stall = 0 if sc < psc - 1e-14 else stall + 1
            psc = sc
            if stall > 1500:
                break
            beta = 100.0
            g = _gsm(a, c, beta, n, fs, clip)
            gm = float(np.abs(g).max())
            if gm < 1e-14:
                break
            g -= g.mean()
            a = np.maximum(a * np.exp(-eta * g / gm), 1e-12)
            a *= bl / a.sum()
            eta *= 0.99999

    # Phase 5: local search with exact evaluator
    if time.monotonic() < td - 3.0:
        base = best.copy()
        wl = max(3, n // 50)
        while time.monotonic() < td - 2.0:
            cand = base.copy()
            for _ in range(int(rng.integers(1, 4))):
                st = int(rng.integers(0, max(1, n - wl)))
                e = min(n, st + wl)
                cand[st:e] *= float(np.exp(rng.normal(0.0, 0.03)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = p.copy()
    return [float(x) for x in out]
