import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def grad_smooth(f, C, T):
    n = f.size
    cs = C - C.max()
    p = np.exp(cs / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    P = np.fft.rfft(p, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(P * np.conj(F), N)[:n]

def mirror_optimize(f, deadline, rng, max_iters=500000):

    f = f.copy()
    eta = 0.05
    T = max(1.0, n * 0.01)
    best_f = f.copy()
    best_sc = fast_score(f)
    no_imp = 0

            break
        C = fft_conv(f, f)
        sc = 2.0 * n * C.max() / (f.sum() ** 2)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            no_imp = 0
        else:

        grad = grad_smooth(f, C, T)
        gmax = grad.max()
        if gmax < 1e-30:

        f = np.maximum(f, 1e-15)

            ns = fast_score(f)
