            best_sc = sc
            cur = inp.copy()

    bounds_list = [(0.0, 1000.0)] * n

    # Phase 1: L-BFGS-B with annealed tau
    tau_schedule = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for tv in tau_schedule:
        if time.monotonic() > dl - 90.0:
            break
        remaining = dl - time.monotonic()
        iters = int(min(8000, max(200, remaining * 300)))
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': iters, 'gtol': 1e-16,
                                    'maxls': 50, 'ftol': 1e-20})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                nx = nx / nx.sum() * scale
            sc = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # Phase 2: Second L-BFGS-B pass from best
    if time.monotonic() < dl - 60.0:
        cur = best.copy()
        for tv in [0.005, 0.002, 0.001, 0.0005, 0.0002]:
            if time.monotonic() > dl - 30.0:
                break
            iters = int(min(6000, max(200, (dl - time.monotonic()) * 200)))
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': iters, 'gtol': 1e-16,
                                        'maxls': 50, 'ftol': 1e-20})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * scale
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Phase 3: Direct subgradient on exact max
    if time.monotonic() < dl - 10.0:
        cur = best.copy()
        step = 0.002
        for it in range(200000):
            if time.monotonic() > dl - 5.0:
                break
            cur = _subgrad_step(cur, step)
            if it % 50 == 0:
                sc = float(evaluate_sequence(cur.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc
                if it % 500 == 0:
                    step *= 0.95
                    step = max(step, 0.00005)
            if it % 200 == 0:
                sc = float(evaluate_sequence(cur.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc

    # Phase 4: Random local search with gradient info
    if time.monotonic() < dl - 3.0:
        cur = best.copy()
        c = _conv(cur)
        kmax = int(c.argmax())
        # Focus perturbations near the kmax-relevant positions
        for it in range(50000):
            if time.monotonic() > dl - 2.0:
                break
            cd = cur.copy()
            nw = max(1, n // 200)
            if rng.random() < 0.3:
                center = kmax % n
                s = int(rng.integers(max(0, center - nw), min(n, center + nw)))
            else:
                s = int(rng.integers(0, n - nw))
            s = max(0, min(s, n - nw))
            cd[s:s + nw] *= float(np.exp(rng.normal(0.0, 0.03)))
            if rng.random() < 0.1:
                cd *= float(np.exp(rng.normal(0.0, 0.01)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc2 = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc2) and sc2 < best_sc:
                    best = cd.copy()
                    best_sc = sc2
                    cur = cd.copy()
            if it % 50 == 0:
                cur = best.copy()

    return [float(v) for v in best]
