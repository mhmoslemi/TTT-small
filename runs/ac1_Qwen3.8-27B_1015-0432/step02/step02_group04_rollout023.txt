The key insight for improvement: instead of using a pure softmax (which spreads the gradient too thin when T is high, or too concentrated when T is low), use a **power mean** of the top K peaks:

S_alpha = (sum_{k in topK} C[k]^alpha)^(1/alpha)

For alpha -> inf, this approaches the max. For alpha = 2, it's the RMS. For alpha = 1, it's the mean.

The gradient of S_alpha w.r.t. f[j] involves C[k]^(alpha-1) weighting for the top K peaks.

This gives a more controlled gradient than pure softmax.

```python
def power_mean_grad(f, C, K, alpha):
    n = f.size
    # Get top K indices
    if K < len(C):
        top_idx = np.argpartition(C, -K)[-K:]
        top_vals = C[top_idx]
        top_mask = np.zeros(len(C))
        top_mask[top_idx] = 1.0
        w = (top_vals ** (alpha - 1)) * top_mask
    else:
        w = C ** (alpha - 1)
    w /= w.sum()
    # Now compute gradient using w as the "probability"
    N = 1 << (2 * n - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(np.conj(W) * F, N)[:n]
```

```python
import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    C = fft_conv(f, f)
    return float(2.0 * len(f) * C.max() / (s * s))

def compute_grad(f, C, K, alpha):
    n = f.size
    nc = len(C)
    if K < nc:
        top_idx = np.argpartition(C, -K)[-K:]
        mask = np.zeros(nc)
        mask[top_idx] = 1.0
        top_vals = C * mask
        w = np.power(np.maximum(top_vals, 1e-30), alpha - 1.0)
    else:
        w = np.power(np.maximum(C, 1e-30), alpha - 1.0)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    N = 1 << (nc + n - 1).bit_length()
    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    return 2.0 * np.fft.irfft(np.conj(W) * F, N)[:n]
```
