The issue is that a "bump" shape gives a triangular autocorrelation with peak proportional to the bump area.

Actually, let me reconsider the problem from the frequency domain:
- The full convolution b = conv(a, a) has DFT: B(ω) = A(ω)^2 (not |A|^2, since it's not the autocorrelation but the convolution of a with itself, not with its conjugate).

Wait, for real sequences, conv(a, a) has DFT A(ω)^2, not |A(ω)|^2. The autocorrelation (conv with reversed) has DFT |A(ω)|^2.

So we're looking at the "aliquot" of the power spectrum, not the power spectrum itself. This is important!

For the score: we want max_k b[k] to be small relative to (sum a_j)^2 = A(0)^2.

In the frequency domain: b[k] = (1/2π) ∫ B(ω) e^{ikω} dω = (1/2π) ∫ A(ω)^2 e^{ikω} dω

And sum(a) = A(0) = (1/2π) ∫ A(ω) dω.

This is quite different from the standard autocorrelation problem.

import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def make_inits(n, rng, parent):
    x = np.arange(n, dtype=np.float64)
    inits = [parent.copy()]
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(np.maximum(h, 1e-15) / np.maximum(h, 1e-15).sum())
    for r in [1.0, 2.5, 5.0]:
        e = np.exp(-r * x / n) + 0.4 * np.exp(-r * (n - 1 - x) / n)
        inits.append(e / e.sum())
    for sh in [-0.2, 0.2, -0.4]:
        c = np.cos(np.pi * (x / n + sh)) ** 2
        inits.append(np.maximum(c, 1e-15) / np.maximum(c, 1e-15).sum())
    for s in [0.2, 0.5, 1.0]:
        t = parent * np.exp(s * rng.standard_normal(n))
        t = np.maximum(t, 1e-25)
        inits.append(t / t.sum())
    r = np.exp(rng.standard_normal(n) * 0.5)
    inits.append(r / r.sum())
    return inits


def mw_pass(f, deadline, rng, beta0, eta0, se, stall_limit):
    n = f.size
    f = np.maximum(f, 1e-25).copy()
    f /= f.sum()
    best = f.copy()
    best_sc = fast_score(f)
    beta = beta0
    eta = eta0
    stall = 0
    step = 0
    while time.monotonic() < deadline - 3.0:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        w = np.exp(beta * (b / M - 1.0))
        f_r = f[::-1]
        gc = fft_conv(w, f_r)
        grad = gc[n - 1:2 * n - 1]
        gm = grad.max()
        if gm < 1e-25:
            break
        f = f * np.exp(-eta * grad / gm)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        step += 1
        if step % se == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
            if step % (se * 8) == 0:
                beta = min(600.0, beta * 1.08)
                eta = max(0.03, eta * 0.99)
            if stall >= stall_limit:
                nb = int(rng.integers(5, 25))
                for _ in range(nb):
                    bl = int(rng.integers(max(2, n // 60), max(3, n // 8)))
                    st = int(rng.integers(0, n - bl))
                    best[st:st + bl] *= np.exp(0.1 * rng.standard_normal(bl))
                best = np.maximum(best, 1e-25)
                best /= best.sum()
                f = best.copy()
                stall = 0
                eta = min(0.2, eta * 1.3)
            if stall >= stall_limit * 3:
                break
    return best, best_sc


def block_refine(f, deadline, rng, n_trials):
    n = f.size
    best = f.copy()
    best_sc = fast_score(f)
    for _ in range(n_trials):
        if time.monotonic() > deadline - 5.0:
            break
        bl = int(rng.integers(max(5, n // 100), max(10, n // 15)))
        st = int(rng.integers(0, n - bl))
        scale = 1.0 + 0.05 * rng.standard_normal()
        trial = best.copy()
        trial[st:st + bl] *= scale
        trial = np.maximum(trial, 1e-25)
        trial /= trial.sum()
        sc = fast_score(trial)
        if sc < best_sc:
            best_sc = sc
            best = trial.copy()
    return best, best_sc
