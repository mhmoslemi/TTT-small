import time

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)

def active_set_gradient(a, conv, tol=1e-10):
    """Compute subgradient using exact active set."""
    n = a.size
    M = float(np.max(conv))
    active = np.where(conv >= M - tol)[0]
    if len(active) == 0:
        active = np.where(conv >= M * 0.9999)[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad from position k: d/d a[j] of sum a[j]*a[k-j]
        # = a[k-j] for j in range
        if lo <= hi:
            grad[lo:hi+1] += a[k - hi:k - lo + 1][::-1]
    grad /= len(active)
    return grad, M, len(active)

def run_descent(a, n, fft_size, deadline, max_it=200000):
    """Main optimization loop."""
    M_cur = float(np.max(fft_convolve(a, fft_size)))
    v = np.zeros(n, dtype=np.float64)
    beta = 0.85
    lr = 0.3
    best_a = a.copy()
    best_M = M_cur
    stall = 0
    
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        conv = fft_convolve(a, fft_size)
        M_cur = float(np.max(conv))
        
        if M_cur < best_M - 1e-15:
            best_M = M_cur
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            break
        
        grad, M_cur, n_active = active_set_gradient(a, conv)
        # Subtract mean to maintain normalization
        grad -= grad.mean()
        
        v = beta * v + (1 - beta) * grad
        v -= v.mean()
        
        # Line search
        accepted = False
        for trial in range(4):
            step = lr / (2.0 ** trial)
            a_try = a - step * v
            a_try = np.clip(a_try, 0, None)
            s = a_try.sum()
            if s < 1e-14:
