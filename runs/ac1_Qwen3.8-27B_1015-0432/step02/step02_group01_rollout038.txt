def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n+m-1]

def score_of(f):
    """Score for a sequence (not necessarily normalized)."""
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def active_lags(b, M, tol=0.01):
    """Return lags where b[k] > M*(1-tol)."""
    return np.where(b > M * (1 - tol))[0]

def subgradient(f, lags):
    """Compute (averaged) subgradient of max(f*f)_k at given lags."""
    n = len(f)
    sg = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            sg[lo:hi+1] += f[k-hi:k-lo+1][::-1]
    sg /= max(len(lags), 1)
    return 2.0 * sg

def optimize(f0, deadline, rng):
    """Main optimization loop."""
    n = f0.size
    f = f0.astype(np.float64).copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    best_f = f.copy()
    best_sc = 2.0 * n * fft_conv(f, f).max()
    
    eta = 0.1
    it = 0
    stall = 0

        M = b.max()
        sc = 2.0 * n * M
        
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f.copy()
            stall = 0

                f = best_f * np.exp(0.05 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                stall = 0
                eta = 0.1
                continue

        tol = max(0.005, 0.05 * np.exp(-it * 0.01))
        lags = np.where(b >= M * (1 - tol))[0]

        sg_max = sg.max()
        if sg_max < 1e-30:

        eta *= 0.9995
        eta = max(eta, 0.002)
        
        it += 1
        if it % 500 == 0:

            pass

Wait, but the subgradient approach (exact, at active lags only) might be too "focused" and cause oscillation. The temperature-weighted approach smooths this out. Let me combine: use the exact subgradient at active lags but add a small temperature-smoothed component.

Actually, let me think about this more carefully. The objective is:
minimize max_k (f*f)_k / (sum f)^2

Since the score is scale-invariant, we can fix sum(f) = 1 and minimize max_k (f*f)_k.

The subgradient of max_k (f*f)_k at a point where lags K = {k1, k2, ...} are active is:
any convex combination of the individual subgradients at k_i.

The natural choice is the average (uniform convex combination). This gives the "balanced" descent direction.

For the multiplicative update f_new = f * exp(-eta * sg / max(sg)), we're doing a form of mirror descent (with the negative entropy as the mirror map). This is well-suited for the simplex constraint (f >= 0, sum f = 1).

Let me also think about the LP refinement. After the gradient method converges, we can do LP refinement:
1. Given f, compute M = max(f*f).
2. Solve LP: max sum(g) s.t. (f*g)_k <= M, g >= 0.
3. Get g from LP. Rescale to sum=1.
4. Blend: f_new = (1-t)*f + t*g for small t.
5. Check if M decreased. If yes, update f.

The LP tells us the "best" direction to move. If the optimal g is very close to f, we're at a local (or global) optimum for the LP relaxation.

Let me implement this. The key challenge is building the sparse matrix efficiently.

Actually, I just realized: for the LP, I don't need to handle the 2n-1 constraints separately. I can note that the constraint (f*g)_k <= M is equivalent to: for all k, sum_j f_j * g_{k-j} <= M.

Using FFT: (f*g) = fft_conv(f, g). So the constraint is that all entries of fft_conv(f, g) are <= M. This is 2n-1 linear constraints on g.

The matrix-free computation of A@g = fft_conv(f, g) is O(n log n). But scipy linprog needs the matrix explicitly...

Hmm, let me be more careful with the subgradient indexing.

For lag k: sg_k[j] = f[k-j] if 0 <= k-j < n and 0 <= j < n, else 0.

Valid j range: max(0, k-n+1) <= j <= min(k, n-1). Call these lo and hi.

For j = lo, lo+1, ..., hi:
sg_k[j] = f[k-j]

So the vector [sg_k[lo], sg_k[lo+1], ..., sg_k[hi]] = [f[k-lo], f[k-lo-1], ..., f[k-hi]]

This is f[k-hi : k-lo+1] reversed, i.e., f[lo': hi'+1][::-1] where lo' = k-hi, hi' = k-lo.

Check: k-hi <= k-lo since lo <= hi. And k-lo >= 0 (since lo <= k when k < n, but if k >= n, then lo = k-n+1, so k-lo = n-1 >= 0). And k-hi >= 0 (since hi <= k). And k-lo <= n-1 (since lo >= k-n+1, so k-lo <= n-1). ✓

So: sg_k[lo:hi+1] = f[k-hi : k-lo+1][::-1]

import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_unit(f):
    """Score for sum(f)=1: 2n * max(f*f)."""
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def subgrad(f, lags):
    n = f.size
    sg = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            sg[lo:hi+1] += f[k-hi:k-lo+1][::-1]
    return 2.0 * sg / max(1, len(lags))

def build_A(f):
    """Build sparse convolution matrix: A[k,j] = f[j] for 0 <= k-j < n."""
    n = f.size
    ci = np.repeat(np.arange(n, dtype=np.int32), n)
    ri = (np.repeat(np.arange(n, dtype=np.int32), n) + np.tile(np.arange(n, dtype=np.int32), n))
    vals = np.repeat(f.astype(np.float64), n)
    return sparse.csr_matrix((vals, (ri, ci)), shape=(2*n-1, n))

def lp_refine(f, deadline):
    """LP refinement step."""
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    
    t_start = time.monotonic()
    A = build_A(f)
    
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)] * n, method='highs',
                  options={'time_limit': max(5, deadline - time.monotonic() - 20)})
    
    A = None  # free memory
    
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0)
        gsum = g.sum()
        if gsum > 1e-10:
            g /= gsum
            return g
    return None

        ini /= ini.sum()
        sc = score_unit(ini)
