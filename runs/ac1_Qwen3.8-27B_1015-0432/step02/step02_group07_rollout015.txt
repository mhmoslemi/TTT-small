            beta = min(beta + 0.003, 0.94)
    return a, best_M, best_a


def run_mirror(a, best_M, best_a, n, fs, limit, deadline):
    conv = fft_conv(a, fs)
    prev_M = float(np.max(conv[:limit]))
    topk = 4
    lr_md = 0.08
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        grad = topk_subgrad(a, conv, n, topk)
        gc = grad - grad.mean()
        a_try = a * np.exp(-lr_md * gc)
        a_try = np.maximum(a_try, 1e-20)
        a_try /= a_try.sum()
        conv_try = fft_conv(a_try, fs)
        M_try = float(np.max(conv_try[:limit]))
        if M_try <= prev_M * (1.0 + 1e-10):
            a = a_try
            conv = conv_try
            prev_M = M_try
            stall = 0
        else:
            lr_md *= 0.4
            stall += 1
            if stall > 800:
                a = best_a.copy()
                conv = fft_conv(a, fs)
                prev_M = float(np.max(conv[:limit]))
                lr_md = 0.08
                stall = 0
        if prev_M < best_M:
            best_M = prev_M
            best_a = a.copy()
        if it % 3000 == 0:
            topk = min(topk + 1, 16)
    return a, best_M, best_a


def run_fine(a, best_M, best_a, n, fs, limit, deadline):
    conv = fft_conv(a, fs)
    prev_M = float(np.max(conv[:limit]))
    topk = 3
    lr = 0.025
    v = np.zeros(n)
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        grad = topk_subgrad(a, conv, n, topk)
        v = 0.4 * v + 0.6 * grad
        v -= v.mean()
        accepted = False
        for b in range(5):
            step = lr / (2.0 ** b)
            a_try = a - step * v
            a_try = np.maximum(a_try, 0.0)
            s = a_try.sum()
            if s < 1e-14:
                break
            a_try /= s
            conv_try = fft_conv(a_try, fs)
            M_try = float(np.max(conv_try[:limit]))
            if M_try <= prev_M * (1.0 + 1e-12):
                a = a_try
                conv = conv_try
                prev_M = M_try
                accepted = True
                break
        if not accepted:
            stall += 1
            lr = max(0.005, lr * 0.85)
            if stall > 3000:
                a = best_a.copy()
                conv = fft_conv(a, fs)
                prev_M = float(np.max(conv[:limit]))
                v[:] = 0.0
                lr = 0.02
                stall = 0
        else:
            stall = 0
        if prev_M < best_M:
            best_M = prev_M
            best_a = a.copy()
    return a, best_M, best_a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 15.0)

    parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 2 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        parent = np.ones(1024)
        n = 1024
        s0 = float(n)

    fs = 1 << (2 * n - 1).bit_length()
    limit = 2 * n - 1

    a = parent / s0
    conv = fft_conv(a, fs)
    best_M = float(np.max(conv[:limit]))
    best_a = a.copy()

    # Phase 1: Additive subgradient descent (45% of budget)
    t1 = time.monotonic() + 0.45 * (deadline - time.monotonic())
    a, best_M, best_a = run_additive(
        a, conv, best_M, best_a, n, fs, limit, t1,
        lr0=0.28, topk0=8, beta0=0.82, stall_limit=3000
    )

    # Phase 2: Mirror descent (25% of budget)
    if time.monotonic() < deadline:
        t2 = time.monotonic() + 0.25 * (deadline - time.monotonic())
        a, best_M, best_a = run_mirror(
            best_a, best_M, best_a, n, fs, limit, t2
        )

    # Phase 3: Fine additive polish (20% of budget)
    if time.monotonic() < deadline:
        t3 = time.monotonic() + 0.20 * (deadline - time.monotonic())
        a, best_M, best_a = run_fine(
            best_a, best_M, best_a, n, fs, limit, t3
        )

    # Phase 4: Local perturbations (remaining)
    base = best_a * s0
    best_sc = float(evaluate_sequence(base.tolist()))
    if not np.isfinite(best_sc):
        base = parent.copy()
        best_sc = float(evaluate_sequence(base.tolist()))

    while time.monotonic() < deadline - 1.0:
        cand = base.copy()
        w = max(4, n // 120)
        for _ in range(int(rng.integers(1, 3))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= float(np.exp(rng.normal(0.0, 0.012)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            base = cand

    out = np.clip(base, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
