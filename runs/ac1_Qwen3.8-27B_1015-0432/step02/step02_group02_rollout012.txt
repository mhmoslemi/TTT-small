```python
from scipy.optimize import minimize

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = np.fft.fft(a, L)
    return np.fft.ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    S = S_target  # fixed sum for shape optimization
    mv, _ = _smax(_conv(a), tau)
```

```python
def _subgrad_step(a, alpha, S_target):
    n = a.size
    c = _conv(a)
    M = c.max()
    # Find argmax positions
    idx = np.where(c >= M - 1e-10)[0]
    # Subgradient: average over argmax positions
    L = _fpow2(3 * n - 2)
    fa = np.fft.fft(a, L)
    g = np.zeros(n)
    for k in idx:
        # grad_j = 2*a_{k-j} for valid j
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += 2.0 * a[k - (np.arange(lo, hi+1))]
    g /= len(idx)
    # Step
    a_new = a - alpha * g
    # Project to {a >= 0, sum(a) = S_target}
    a_new = _project_simplex(a_new, S_target)
    return a_new, M

def _project_simplex(v, S):
    n = v.size
    # Binary search for theta: sum(max(v_i - theta, 0)) = S
    lo = v.min() - 1
    hi = v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(v - mid, 0).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) / 2, 0)
```

```python
def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = np.fft.fft(a, L)
    return np.fft.ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    n = a.size
    S = a.sum()

        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)

    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
```

```python
    r = idx[-1]
    theta = (cs[r] - S) / (r + 1.0)
```
