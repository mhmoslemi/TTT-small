```python
def proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} in Euclidean norm."""
    n = v.size
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.5, float(budget_s) - 20.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 2 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        n = 1024
        parent = np.ones(n, dtype=np.float64)
        s0 = float(n)

    fsz = 1 << (2 * n - 1).bit_length()

    best = parent.copy()
    best_score = float(evaluate_sequence([float(x) for x in best]))

    xx = np.arange(n, dtype=np.float64)
    inits = [parent / s0]

    h = 0.5 * (1.0 + np.cos(2.0 * np.pi * xx / (n - 1)))
    inits.append(h / h.sum())

    t = 1.0 - np.abs(2.0 * xx / (n - 1) - 1.0)
    inits.append(t / t.sum())

    g = np.exp(-2.0 * ((xx - n / 2.0) / (n / 4.0)) ** 2)
    inits.append(g / g.sum())

    inits.append(np.ones(n, dtype=np.float64) / n)

    b = (np.exp(-3.0 * ((xx - 0.22 * n) / (n / 8.0)) ** 2) +
         np.exp(-3.0 * ((xx - 0.78 * n) / (n / 8.0)) ** 2))
    inits.append(b / b.sum())

    tr = np.ones(n, dtype=np.float64)
    w = max(1, n // 5)
    tr[:w] = xx[:w] / w
    tr[-w:] = xx[-w:][::-1] / w
    inits.append(tr / tr.sum())

    g2 = np.exp(-4.0 * ((xx - n / 2.0) / (n / 6.0)) ** 2)
    inits.append(g2 / g2.sum())

    n_init = len(inits)
    for ii in range(n_init):
        remaining = deadline - time.monotonic()
        if remaining < 25.0:
            break
        frac = (n_init - ii) / (n_init * (n_init + 1) / 2.0)
        t_end = time.monotonic() + frac * remaining - 5.0
        if t_end < time.monotonic() + 20.0:
            break
        res, _ = run_descent(inits[ii].copy(), n, fsz, t_end)
        sc = float(evaluate_sequence([float(x) for x in res]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = res.copy()

    if time.monotonic() < deadline - 40.0:
        a_pol = best.copy()
        a_pol /= a_pol.sum()
        t_end = deadline - 15.0
        res, _ = run_descent(a_pol, n, fsz, t_end)
        sc = float(evaluate_sequence([float(x) for x in res]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = res.copy()

    base = best.copy()
    while time.monotonic() < deadline - 8.0:
        cand = base.copy()
        w = max(3, n // 35)
        st = int(rng.integers(0, max(1, n - w)))
        cand[st:st + w] *= np.exp(float(rng.normal(0.0, 0.004)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence([float(x) for x in cand]))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = cand.copy()
            base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
```
