import numpy as np
import time


def fft_conv(a, fsize):
    fa = np.fft.rfft(a, n=fsize)
    return np.fft.irfft(fa * fa, n=fsize)


def smooth_grad(conv, a, T, gsize):
    n = a.size
    cmax = conv.max()
    exp_vals = np.exp(-(cmax - conv) / T)
    S = exp_vals.sum()
    if S < 1e-300:
        return np.zeros(n)
    f = exp_vals[::-1]
    F = np.fft.rfft(f, n=gsize)
    A = np.fft.rfft(a, n=gsize)
    convol = np.fft.irfft(F * A, n=gsize)
    result = convol[2 * n - 2: 3 * n - 2]
    grad = (2.0 / S) * result
    grad -= grad.mean()
    return grad
