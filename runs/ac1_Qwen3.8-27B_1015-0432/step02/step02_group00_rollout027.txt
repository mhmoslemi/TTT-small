import time

def fft_conv(a, b):
    n, m = int(len(a)), int(len(b))
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def eval_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    m = 2 * n - 1
    f = f.copy()
    best_f = f.copy()

        if time.monotonic() > deadline - 30:
            break
        b = fft_conv(f, f)
        M = b.max()
        # Build sparse conv matrix
        rows_list, cols_list, vals_list = [], [], []
        for j in range(n):
            ks = np.arange(j, min(j + n, m), dtype=np.int32)
            rows_list.append(ks)
            cols_list.append(np.full(len(ks), j, dtype=np.int32))
            vals_list.append(f[ks - j])

            (np.concatenate(vals_list), (np.concatenate(rows_list), np.concatenate(cols_list))),
