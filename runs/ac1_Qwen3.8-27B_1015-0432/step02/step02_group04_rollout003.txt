            trial2 /= trial2.sum()
            for tr in [trial, trial2]:
                if time.monotonic() > deadline - 35:
                    break
                sc_tr = fast_score(tr)
                if sc_tr < best_sc - 1e-9:
                    best_sc = sc_tr
                    best_f = tr.copy()
                    improved = True
                to = min(time.monotonic() + 50.0, deadline - 20)
                if to > time.monotonic() + 5:
                    f2, sc2 = mirror_opt(tr, to, rng)
                    if sc2 < best_sc:
                        best_sc, best_f = sc2, f2
                        improved = True
        if not improved and round_num >= 3:
            break
    # Final polish
    if time.monotonic() < deadline - 10:
        fp, scp = mirror_opt(best_f, deadline - 8, rng, max_iter=50000)
        if scp < best_sc:
            best_sc, best_f = scp, fp
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    if sum(result) < 0.01:
        result = [1.0 / n] * n
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
