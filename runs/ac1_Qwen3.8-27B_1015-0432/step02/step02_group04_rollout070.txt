def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1

def fallback_uniform(n):
    n = int(n)
    if n <= 0:

    return np.ones(n, dtype=np.float64) / float(n)


def normalize_nonneg(g):
    g = np.asarray(g, dtype=np.float64)
    if g.size == 0:
        return None
    g = np.maximum(g, TINY)
    s = float(g.sum())
    if not np.isfinite(s) or s <= 1e-15:

def self_conv_fft(f, N):
    F = np.fft.rfft(f, N)
    C = np.fft.irfft(F * np.conj(F), N)
    C = C[:2 * int(f.size) - 1]
    C[C < 0.0] = 0.0

        n = int(arr.size)

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if float(arr.sum()) < 1e-10:
        parent = make_flat_top(n, 0.0)
    else:
        parent = normalize_nonneg(arr)
        if parent is None:
            parent = make_flat_top(n, 0.0)

    best_f = parent.copy()
    best_exact = candidate_exact_score(best_f)
    if not np.isfinite(best_exact):
        best_exact = np.inf

    starts = make_starts(parent, rng, n)
    top = []
    for cand in starts:
        if cand is None:
            continue
        if int(cand.size) != n:
            continue
        sc = fast_score(cand)
        top_update(top, sc, cand)

    top.sort(key=start_key)
    weights = (1.0, 0.65, 0.50, 0.40, 0.32, 0.26, 0.22, 0.18, 0.15, 0.12)
    reserve_exact = 50.0 if n <= 12000 else 10.0

    max_starts = min(len(top), 8)
    total_w = float(sum(weights[:max_starts])) if max_starts > 0 else 1.0
    for i in range(max_starts):
        if time.monotonic() > deadline - reserve_exact - 10.0:
            break
        w = weights[i] if i < len(weights) else 0.10
        remaining = max(5.0, deadline - time.monotonic() - reserve_exact - 10.0)
        chunk = max(5.0, min(240.0, remaining * w / max(total_w, 1e-12)))
        local_deadline = time.monotonic() + chunk
        if local_deadline > deadline - reserve_exact - 10.0:
            local_deadline = deadline - reserve_exact - 10.0
        cand, _sc = optimize_sequence(top[i][1], local_deadline, rng)
        if cand is None:
            continue
        ext = candidate_exact_score(cand)
        if np.isfinite(ext) and ext < best_exact:
            best_exact = ext
            best_f = cand.copy()

    if time.monotonic() < deadline - reserve_exact - 15.0:
        local_deadline = deadline - reserve_exact - 5.0
        cand, _sc = optimize_sequence(best_f, local_deadline, rng)
        if cand is not None:
            ext = candidate_exact_score(cand)
            if np.isfinite(ext) and ext < best_exact:
                best_exact = ext
                best_f = cand.copy()

    if n <= 12000 and time.monotonic() < deadline - 15.0:
        cand, exact_sc = exact_polish(best_f, deadline - 2.0, rng)
        if cand is not None and np.isfinite(exact_sc) and exact_sc < best_exact:
            best_exact = exact_sc
            best_f = cand.copy()

    out = np.asarray(best_f, dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    if out.size == 0 or float(out.sum()) < 0.01:
        out = fallback_uniform(n)

    result = [float(x) for x in out]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [float(x) for x in fallback_uniform(n)]
    return result
