def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    """Score with sum(f)=1: 2n*max(conv(f,f))"""
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

def grad_of_max(f, b):
    """Subgradient of max(conv(f,f)) w.r.t. f.
    Returns 2*(f reversed/shifted) at argmax lag."""
    k_star = b.argmax()
    # (grad b[k*])[m] = 2*f[k*-m] for valid m
    # f has indices 0..n-1, k*-m must be in 0..n-1
    # m in [max(0, k*-(n-1)), min(n-1, k*)]
    n = f.size
    m_lo = max(0, k_star - (n - 1))
    m_hi = min(n - 1, k_star)
    g = np.zeros(n)

        idx = k_star - m_range  # indices into f
        g[m_range] = 2.0 * f[idx]
    return g, k_star

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()

    a = np.maximum(0.0, 1.0 - v * v)
