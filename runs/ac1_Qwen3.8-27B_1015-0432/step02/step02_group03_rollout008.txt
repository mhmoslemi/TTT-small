        w = max(3, n // 150)
        while time.monotonic() < dl - 3:
            cand = base.copy()
            for _ in range(int(rng.integers(1, 3))):
                st = int(rng.integers(0, max(1, n - w)))
                e2 = min(n, st + w)
                cand[st:e2] *= float(np.exp(rng.normal(0.0, 0.012)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand / cand.sum()
                base = cand

    out = np.clip(best, 0.0, None) * ts
    out = np.clip(out, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = p
    return [float(x) for x in out]
