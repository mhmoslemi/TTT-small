def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def subgrad_topk(a, conv, tk):
    n = a.size
    tk = min(tk, 2 * n - 1)
    idx = np.argpartition(conv, -tk)[-tk:]
    grad = np.zeros(n, dtype=np.float64)
    for ii in range(tk):
        k = int(idx[ii])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    grad /= float(tk)
    return grad
