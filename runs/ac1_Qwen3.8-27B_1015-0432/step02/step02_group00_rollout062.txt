def fft_conv(a, b):
    n, m = len(a), len(b)
    s = 1 << max(1, (n + m - 2).bit_length())
    return np.fft.irfft(np.fft.rfft(a, s) * np.fft.rfft(b, s))[:n+m-1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(f) * b.max() / (s*s)

def subgrad(f, pk, w):
    n = len(f)
    h = np.zeros(2*n-1)
    h[pk] = w
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n-1:2*n-1][::-1]
