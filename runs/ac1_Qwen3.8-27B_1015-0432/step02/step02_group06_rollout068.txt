import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def norm_pos(f, floor=1e-25):
    f = np.maximum(f, floor)
    f /= f.sum()
    return f


def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    return norm_pos(h, 1e-15)


def init_ramp(n, alpha, sym=0.0):
    x = np.arange(n, dtype=np.float64)
    f = (x + 1.0) ** alpha
    if sym > 0:
        f = f + sym * (n - x) ** alpha
    return norm_pos(f, 1e-15)


def init_exp(n, r, sym=0.0):
    x = np.arange(n, dtype=np.float64)
    f = np.exp(-r * x / n)
    if sym > 0:
        f = f + sym * np.exp(-r * (n - 1 - x) / n)
    return norm_pos(f, 1e-15)
