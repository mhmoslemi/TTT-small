    par = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size < 2 or not np.isfinite(par).all() or par.sum() < 0.01:
        par = np.ones(1024, dtype=np.float64)
    n = par.size
    s0 = float(par.sum())
    fsz = 1 << (2 * n - 1).bit_length()
    best = par / s0
    best_sc = float(evaluate_sequence(par.tolist()))

    inits = []
    inits.append(par / s0)
    r = np.abs(rng.standard_normal(n)) ** 0.6
    inits.append(r / r.sum())
    r2 = np.abs(rng.standard_normal(n))
    for _ in range(3):
        r2 = (r2 + np.roll(r2, 1)) * 0.5
    inits.append(r2 / r2.sum())
    fc = max(1, n // 20)
    r3 = np.clip(0.5 + 0.4 * np.cos(2.0 * np.pi * fc * np.arange(n) / n), 0.01, None)
    inits.append(r3 / r3.sum())
    x = np.arange(n) - n / 2.0
    r4 = np.exp(-x * x / (2.0 * (n / 5.0) ** 2))
    inits.append(r4 / r4.sum())

    configs = [(24, 0.06), (12, 0.09), (8, 0.11), (16, 0.07), (10, 0.10)]
    ni = len(inits)

    for k in range(ni):
        rem = deadline - time.monotonic()
        if rem < 50.0:
            break
        cap = 280.0 if k == 0 else 160.0
        per = min(rem / (ni - k), cap)
        tk, lr = configs[k % len(configs)]
        a_opt, _ = run_opt(inits[k].copy(), n, fsz, time.monotonic() + per, tk, lr)
        cand = a_opt * s0
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a_opt.copy()

    # Polishing with small steps top-2
    if time.monotonic() < deadline - 4.0:
        base = best.copy()
        while time.monotonic() < deadline - 2.0:
            a2 = base.copy()
            cv = fft_conv(a2, fsz)
            idx = topk_idx(cv, 2)
            g = np.zeros(n)
            for kk in idx:
                grad_lag_into(g, a2, int(kk), n)
            g /= 2.0
            step = 0.15 * np.exp(rng.normal(0.0, 0.03))
            a2 *= np.exp(-step * n * g)
            np.maximum(a2, 1e-18, out=a2)
            a2 /= a2.sum()
            cand = a2 * s0
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = a2.copy()
                base = a2.copy()
            else:
                # small random perturbation
                w = max(2, n // 300)
                st = int(rng.integers(0, max(1, n - w)))
                base[st:st + w] *= float(np.exp(rng.normal(0, 0.015)))
                base = np.maximum(base, 1e-15)
                base /= base.sum()

    # Final local random improvements
    if time.monotonic() < deadline - 1.0:
        base = best.copy()
        while time.monotonic() < deadline - 0.5:
            c = base.copy()
            for _ in range(3):
                w = int(rng.integers(2, max(3, n // 100)))
                st = int(rng.integers(0, max(1, n - w)))
                c[st:st + w] *= float(np.exp(rng.normal(0, 0.025)))
            c = np.clip(c, 0.0, None)
            if c.sum() < 0.01:
                continue
            c = c / c.sum()
            cand = c * s0
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = c
                base = c

    out = np.clip(best * s0, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = par.copy()
    return [float(x) for x in out]
