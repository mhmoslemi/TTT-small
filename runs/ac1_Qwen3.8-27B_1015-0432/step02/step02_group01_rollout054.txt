        active = np.argsort(b)[-max_active:]
    grad = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += f[k - hi:k - lo + 1][::-1]
    grad *= 2.0 / len(active)
    return grad


def optimize_phase(f, deadline, rng, eps, eta0, eta_min, iter_max, restart_freq):
    n = f.size
    f = np.maximum(f, 1e-15)
    f /= f.sum()
    best_f = f.copy()
    best_sc = score_norm(f)
    eta = eta0
    stall = 0
    for it in range(iter_max):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        sc = 2.0 * n * b.max()
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
            if stall >= restart_freq:
                f = best_f * np.exp(0.02 * rng.standard_normal(n))
                f = np.maximum(f, 1e-15)
                f /= f.sum()
                eta = eta0
                stall = 0
        g = subgradient(f, b, eps, 20)
        if g is None:
            break
        gmax = g.max()
        if gmax < 1e-30:
            break
        g /= gmax
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-15)
        f /= f.sum()
        if it % 1000 == 0 and it > 0:
            eta = max(eta_min, eta * 0.96)
    return best_f, best_sc


def make_ramp(n, pwr, reverse):
    x = np.arange(n, dtype=np.float64) + 1.0
    r = x ** pwr
    if reverse:
        r = r[::-1]
    return r / r.sum()


def make_arc(n, center, width):
    x = np.arange(n, dtype=np.float64)
    u = x / max(n - 1, 1)
    v = (u - center) / width
    a = np.maximum(0.0, 1.0 - v * v)
    a **= 0.7
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()

    inits = [p, p[::-1].copy()]
    for pwr in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(make_ramp(n, pwr, False))
        inits.append(make_ramp(n, pwr, True))
    for c, w in [(0.25, 0.6), (0.35, 0.5), (0.5, 0.4), (0.65, 0.5), (0.75, 0.6)]:
        inits.append(make_arc(n, c, w))
    inits.append(np.ones(n) / n)

    best_f = inits[0]
    best_sc = np.inf
    for ini in inits:
        sc = score_norm(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    d1 = deadline - (budget_s - 20.0) * 0.25
    d2 = deadline - (budget_s - 20.0) * 0.45
    d3 = deadline - (budget_s - 20.0) * 0.65
    d4 = deadline - (budget_s - 20.0) * 0.80
    d5 = deadline - (budget_s - 20.0) * 0.90

    if t0 + time.monotonic() - t0 < d1:
        f, sc = optimize_phase(best_f, d1, rng, 0.02, 0.15, 0.005, 300000, 3000)
        if sc < best_sc:
            best_sc, best_f = sc, f.copy()

    if time.monotonic() < d2:
        f, sc = optimize_phase(best_f, d2, rng, 0.005, 0.08, 0.003, 300000, 2500)
        if sc < best_sc:
            best_sc, best_f = sc, f.copy()

    if time.monotonic() < d3:
        f, sc = optimize_phase(best_f, d3, rng, 0.002, 0.05, 0.002, 300000, 2000)
        if sc < best_sc:
            best_sc, best_f = sc, f.copy()

    if time.monotonic() < d4:
        flip = best_f[::-1].copy()
        for t in np.arange(0.1, 0.9, 0.1):
            bl = (1.0 - t) * best_f + t * flip
            bl = np.maximum(bl, 1e-15)
            bl /= bl.sum()
            sc_b = score_norm(bl)
            if sc_b < best_sc:
                best_sc, best_f = sc_b, bl.copy()
        for _ in range(8):
            if time.monotonic() > d4:
                break
            trial = best_f * np.exp(0.01 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc_t = score_norm(trial)
            if sc_t < best_sc:
                best_sc, best_f = sc_t, trial.copy()

    if time.monotonic() < d5:
        f, sc = optimize_phase(best_f, d5, rng, 0.001, 0.03, 0.001, 300000, 1500)
        if sc < best_sc:
            best_sc, best_f = sc, f.copy()

    if time.monotonic() < deadline - 5:
        f, sc = optimize_phase(best_f, deadline - 5, rng, 0.0005, 0.015, 0.0005, 300000, 1200)
        if sc < best_sc:
            best_sc, best_f = sc, f.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc) or final_sc > 3.0:
        result = [1.0 / n] * n
    return result
