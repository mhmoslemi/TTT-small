import time
from scipy import sparse
from scipy.optimize import linprog

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

    cols_parts = []
    vals_parts = []

        lo = max(0, k - n + 1)

        shape=(2*n-1, n)
    )

def fft_conv(a, b):
