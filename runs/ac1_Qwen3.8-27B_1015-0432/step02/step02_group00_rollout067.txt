import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def calc_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, b, thresh):
    n = f.shape[0]
    L = 2 * n - 1
    mask = b > thresh
    h = np.zeros(L)
    h[mask] = b[mask] - thresh
    if h.max() < 1e-30:
        h[np.argmax(b)] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
