def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def conv_peak(a, fsz):
    cv = fft_conv(a, fsz)
    return cv[:2 * a.size - 1].max()

def grad_topk(a, cv, n, tk):
    T = min(tk, 2 * n - 1)
    idx = np.argpartition(cv, -T)[-T:]

        hi = min(n - 1, kk)

    return g
