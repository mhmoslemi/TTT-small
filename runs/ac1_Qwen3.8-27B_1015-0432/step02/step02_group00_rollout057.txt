```python
def peel(f, iters=100, alpha=0.3):
    n = f.size
    f = f.copy()
    for _ in range(iters):
        b = fft_conv(f, f)
        p = np.argmax(b)
        M = b[p]
        # Reduce f at positions that contribute to peak p
        lo = max(0, p - n + 1)
        hi = min(n, p + 1)
        contrib = np.zeros(n)
        for j in range(lo, hi):
            contrib[j] = f[p - j] if 0 <= p - j < n else 0
        # Actually contrib[j] = f[p-j]
        # Reduce: f[j] -= alpha * contrib[j] for j in [lo, hi)
        f[lo:hi] -= alpha * contrib[lo:hi]
        f = np.maximum(f, 1e-20)
        f = f / f.sum()
    return f
```
