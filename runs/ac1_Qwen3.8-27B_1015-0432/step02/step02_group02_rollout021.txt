import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _norm(a, scale):
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-10:
        a = np.ones_like(a)
    return a / a.sum() * scale


def _obj(a, tau, scale):
    a = _norm(a, scale)
    n = a.size
    c = _conv(a)
    sm, _ = _smax(c, tau)
    return 2.0 * n * sm


def _grad(a, tau, scale):
    a = _norm(a, scale)
    n = a.size
    c = _conv(a)
    sm, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    # Project gradient to tangent space of sum(a)=scale
    g = g - g.mean()
    return 2.0 * n * g


def _proj_subgrad(a, scale, lr, n_peaks):
    a = a.copy()
    c = _conv(a)
    cm = c.max()
    idx = np.argsort(c)[-n_peaks:]

    pr = subsum[::-1]
