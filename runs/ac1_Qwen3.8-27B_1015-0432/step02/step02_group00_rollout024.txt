def fft_conv(a, b):

    size = 1 << (n + m - 2).bit_length()

    if s < 1e-15:

        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 150:
            peaks = np.argsort(b)[-150:]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])

        w = b[peaks] ** 2
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        g = compute_grad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-30:
            break

        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-18)
        if sym:
            f = 0.5 * (f + f[::-1])
        f = f / f.sum()

        if it % 30 == 0:
            sc = obj_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1

            if no_improv > 500:
                scale = 0.02 + 0.04 * min(1.0, restarts / 10.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-18)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = normalize(f)
                no_improv = 0
                restarts += 1
                eta = min(0.15, 0.04 * (1.0 + 0.3 * restarts))

        eta *= 0.9997

    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(normalize(h))
    d = np.minimum(x, n - 1.0 - x) + 1.0
    inits.append(normalize(d ** 0.6))
    ah = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1))) + 1e-12
    inits.append(normalize(ah))

    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = obj_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    p1_end = min(deadline, t0 + 450.0)
    f_opt, sc_opt = grad_opt(best_f.copy(), p1_end, rng)
    if sc_opt < best_sc:
        best_sc = sc_opt
        best_f = f_opt.copy()

    if time.monotonic() < deadline - 80.0:
        f_s = 0.5 * (best_f + best_f[::-1])
        f_s = normalize(f_s)
        f_s_opt, sc_s = grad_opt(f_s, min(deadline, t0 + 850.0), rng, sym=True)
        if sc_s < best_sc:
            best_sc = sc_s
            best_f = f_s_opt.copy()

    if time.monotonic() < deadline - 20.0:
        f_ref, sc_ref = grad_opt(best_f.copy(), deadline - 8.0, rng)
        if sc_ref < best_sc:
            best_sc = sc_ref
            best_f = f_ref.copy()

    if time.monotonic() < deadline - 5.0:
        for _ in range(40):
            if time.monotonic() > deadline - 3.0:
                break
            trial = best_f * np.exp(0.004 * rng.standard_normal(n))
            trial = normalize(trial)
            sc = obj_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
