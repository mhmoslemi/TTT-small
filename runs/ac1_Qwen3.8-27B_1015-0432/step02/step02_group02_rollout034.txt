def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -60.0)
    e = np.exp(d)
    s = e.sum()
    if s < 1e-300:
        return cm, np.zeros_like(c)
    return cm + tau * np.log(s), e / s
