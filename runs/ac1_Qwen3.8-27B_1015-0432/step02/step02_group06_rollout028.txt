import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def build_conv_csc(f):
    n = f.size
    N2 = 2 * n - 1
    indptr = np.zeros(2 * n, dtype=np.int64)
    for k in range(N2):
        indptr[k + 1] = indptr[k] + ((k + 1) if k < n else (N2 - k))
    total = int(indptr[-1])
    col_idx = np.empty(total, dtype=np.int64)
    data = np.empty(total, dtype=np.float64)
    for k in range(N2):
        s, e = indptr[k], indptr[k + 1]
        if k < n:
            col_idx[s:e] = np.arange(k + 1)
            data[s:e] = f[:k + 1]
        else:
            col_idx[s:e] = np.arange(k - n + 1, n)
            data[s:e] = f[k - n + 1:]
    return sp.csc_matrix((data, col_idx, indptr), shape=(N2, n))


def lp_solve(f, M, tlimit):
    n = f.size
    T = build_conv_csc(f).tocsr()
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=T, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': tlimit, 'presolve': True})
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None
