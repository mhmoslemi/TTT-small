```python
def softgrad(a, c, T, fsz2):
    """Gradient of log-sum-exp(c/T) w.r.t. a, via FFT."""
    n = a.size
    # softmax weights
    c_scaled = c / T
    c_scaled = c_scaled - c_scaled.max()  # for numerical stability
    w = np.exp(c_scaled)
    w /= w.sum()
    # w[k] is the weight for lag k
    # grad[j] = sum_k w[k] * a[k-j] (for valid k-j)
    # = same as before but with fractional weights
    # conv(a[::-1], w)[n-1:2n-1]
    ar = a[::-1]
    fa = np.fft.rfft(ar, n=fsz2)
    fw = np.fft.rfft(w, n=fsz2)
    conv = np.fft.irfft(fa * fw, n=fsz2)
    return conv[n-1:2*n-1]
```

import time

def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def soft_gradient(a, c, T, fsz2):
    n = a.size
    cs = c / T
    cs -= cs.max()
    w = np.exp(cs)
    wsum = float(w.sum())
    if wsum < 1e-30:
        w = np.zeros(2 * n - 1)
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    ar = a[::-1]
    fa = np.fft.rfft(ar, n=fsz2)
    fw = np.fft.rfft(w, n=fsz2)
    conv_g = np.fft.irfft(fa * fw, n=fsz2)
