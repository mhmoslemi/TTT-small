def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def grad_peak(a, cv, k):
    n = a.size
    m = min(k, 2 * n - 1)
    idx = np.argpartition(cv[:2 * n - 1], -m)[-m:]
    g = np.zeros(n)

        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    return g / m
