def _rfc(f, N):
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)

def _score(f, N=None):
    n = len(f)
    if N is None:
        N = 1 << (2 * n - 1).bit_length()
    c = _rfc(f, N)[:2 * n - 1]
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def _mk_cos(n, tilt):
    j = np.arange(n, dtype=np.float64)
    w = np.cos(np.pi * j / (n + 1)) + 0.5
    if tilt != 0.0:
        w *= np.exp(tilt * (2.0 * j / max(n - 1, 1) - 1.0))
    w = np.maximum(w, 1e-12)
    return w / w.sum()

def _mk_gauss(n, cf, wf, tilt=0.0):

        w *= np.exp(tilt * (2.0 * j / max(n - 1, 1) - 1.0))
