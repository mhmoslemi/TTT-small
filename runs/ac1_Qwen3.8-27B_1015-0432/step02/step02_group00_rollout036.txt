def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

    h = np.zeros(L)
    h[peaks] = 1.0

    g = 2.0 * c[n - 1:2 * n - 1][::-1]

        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()

    if time.monotonic() < deadline - 2.0:
        for _ in range(400):
            if time.monotonic() > deadline - 1.0:
                break
            a = 0.002 + 0.006 * rng.random()
            tr = best_f * np.exp(a * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-20)
            tr /= tr.sum()
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()

    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
