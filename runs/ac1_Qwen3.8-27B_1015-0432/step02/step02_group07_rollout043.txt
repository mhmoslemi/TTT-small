import numpy as np
import time

def fft_convolve(a, b=None, fft_size=None):
    if b is None:
        b = a
    if fft_size is None:
        fft_size = 1 << (len(a) + len(b) - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b, fft_size)
    return np.fft.irfft(fa * fb, fft_size)

def compute_lr(a, conv, tau, fft_size):
    """Compute the gradient of log-sum-exp(tau * conv) w.r.t. a."""
    n = len(a)
    c = conv[:2*n-1]  # full convolution
    max_c = np.max(c)
    exp_c = np.exp(tau * (c - max_c))
    p = exp_c / np.sum(exp_c)
    a_rev = a[::-1]
    full = fft_convolve(p, a_rev, fft_size)
    g = 2.0 * full[n-1:2*n-1]
    return g
