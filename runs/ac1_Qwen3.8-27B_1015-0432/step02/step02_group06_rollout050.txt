import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def compute_grad(f, b, beta):
    n = len(f)
    M = b.max()
    w = np.exp(beta * (b - M))
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        w /= wsum
    f_rev = f[::-1]
    gc = fft_conv(w, f_rev)
    return gc[n - 1:2 * n - 1]


def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    return np.maximum(h, 1e-15) / np.maximum(h, 1e-15).sum()


def init_exp_pair(n, r1, r2):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-r1 * x / n) + np.exp(-r2 * (n - 1 - x) / n)
    return a / a.sum()


def init_cos2(n, shift):
    x = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * (x / n + shift)) ** 2
    a = np.maximum(a, 1e-15)
    return a / a.sum()


def init_bump(n, center, width):
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return a / a.sum()

    inits.append(init_exp_pair(n, 2.0, 2.0))

    inits.append(init_exp_pair(n, 0.5, 0.5))

    inits.append(init_cos2(n, 0.0))
    inits.append(init_cos2(n, -0.15))
    inits.append(init_cos2(n, 0.15))
    inits.append(init_bump(n, n * 0.3, n * 0.2))
    inits.append(init_bump(n, n * 0.7, n * 0.2))
    p = parent / parent.sum()
    for s in [0.05, 0.15, 0.3, 0.5]:

    for _ in range(4):
        t = np.abs(rng.standard_normal(n)) + 0.1
        inits.append(t / t.sum())
    return inits


def optimize(f, deadline, rng, beta_start, beta_end, eta_start):
    n = len(f)
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)
    beta = beta_start
    eta = eta_start
    it = 0
    stall = 0

    while time.monotonic() < deadline - 5.0:
        b = fft_conv(f, f)
        grad = compute_grad(f, b, beta)
        gmax = grad.max()
        if gmax < 1e-30:
            break

        f_new = f * np.exp(-eta * grad / gmax)
        f_new = np.maximum(f_new, 1e-25)
        f_new /= f_new.sum()
        f = f_new
        it += 1

        if it % 10 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-11:
                best_sc = sc
                best = f.copy()
                stall = 0
            else:
                stall += 1
                if stall % 10 == 0:
                    eta = max(0.002, eta * 0.85)

            progress = min(1.0, it / 8000.0)
            beta = beta_start * (beta_end / beta_start) ** progress

                best_sc = fast_score(f)
                stall = 0
