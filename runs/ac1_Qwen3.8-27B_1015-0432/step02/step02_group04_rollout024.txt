    return float(2 * len(f) * b.max() / (s * s))

def g_sm(f, C, T):
    n = f.size
    p = np.exp((C - C.max()) / T)
    p /= p.sum()
    N = 1 << (C.size + n - 1).bit_length()
    return 2.0 * np.fft.irfft(np.fft.rfft(p, N) * np.conj(np.fft.rfft(f, N)), N)[:n]

def g_hd(f, C):
    n = f.size
    k = int(np.argmax(C))
    lo, hi = max(0, k - n + 1), min(k, n - 1)
    g = np.zeros(n)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
    return g

def opt(f, dl, rng, mx=100000):
    n = f.size
    f = f.copy()
    bf, bs = f.copy(), sc(f)
    eta, T, ni, psc = 0.3, 1.5, 0, bs
    for it in range(mx):
        if time.monotonic() > dl - 3:
            break
        C = fft_conv(f, f)
        s = 2.0 * n * C.max() / (f.sum() ** 2)
        if s < bs - 1e-11:
            bs, bf, ni = s, f.copy(), 0
        else:
            ni += 1
        g = g_sm(f, C, T) if it % 4 < 3 else g_hd(f, C)
        gm = g.max()
        if gm < 1e-30:
            break
        f *= np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-30)
        f /= f.sum()
        if it % 20 == 0:
            eta = min(eta * 1.06, 2.0) if s < psc - 1e-10 else max(eta * 0.85, 0.01)
            psc = s
        if it % 200 == 0:
            T = max(0.03, T * 0.88)
        if ni > 250:
            f = bf * np.exp(0.05 * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)
            f /= f.sum()
            ni, eta, T, psc = 0, 0.3, 1.5, s
    return bf, bs

def diversify(f, rng):
    n = f.size
    x = 2.0 * np.arange(n) / max(n - 1, 1) - 1.0
    R = []
    for a in [0.04, 0.1, 0.2]:
        t = f * np.exp(a * rng.standard_normal(n))
        t = np.maximum(t, 1e-20)
        R.append(t / t.sum())
    for c in [-1.0, 1.0]:
        t = f * np.exp(c * x)
        t = np.maximum(t, 1e-20)
        R.append(t / t.sum())
    t = 1.0 / np.maximum(f, 1e-8)
    R.append(t / t.sum())
    return R

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 10.0
    n = len(height_sequence_1)

    inits = [np.ones(n) / n]
    pp = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if pp.size == n and np.all(np.isfinite(pp)) and pp.sum() > 1e-10:
        inits.append(pp / pp.sum())
    j = np.arange(n, dtype=np.float64) + 1.0
    for a in [0.5, 1.0, 2.0]:
        w = j ** a
        inits.append(w / w.sum())
        w2 = (n + 1.0 - j) ** a
        inits.append(w2 / w2.sum())
    for _ in range(3):
        w = rng.random(n) + 0.01
        inits.append(w / w.sum())

    bf, bs = None, np.inf
    for ini in inits:
        s = sc(ini)
        if np.isfinite(s) and s < bs:
            bs, bf = s, ini.copy()
    if bf is None:
        bf, bs = np.ones(n) / n, 2.0

    while time.monotonic() < dl - 25:
        to = time.monotonic() + min(200.0, dl - time.monotonic() - 15)
        f1, s1 = opt(bf, to, rng)
        if s1 < bs:
            bs, bf = s1, f1
        if time.monotonic() > dl - 30:
            break
        for tri in diversify(bf, rng):
            if time.monotonic() > dl - 35:
                break
            s = sc(tri)
            if s < bs:
                bs, bf = s, tri.copy()
            to2 = time.monotonic() + min(120.0, dl - time.monotonic() - 15)
            f2, s2 = opt(tri, to2, rng)
            if s2 < bs:
                bs, bf = s2, f2
            if time.monotonic() > dl - 35:
                break

    res = [float(x) for x in bf]
    fs = evaluate_sequence(res)
    if not np.isfinite(fs):
        res = [1.0] * n
    return res
