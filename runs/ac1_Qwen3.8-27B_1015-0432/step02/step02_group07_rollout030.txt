        conv = fft_autoconv(a, fft_size)
        M = float(np.max(conv))
        if M < prev_M - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall > 4000 and it > 300:
            break
        tk = min(top_k, 2 * n - 1)
        grad = subgrad_topk(a, conv, n, tk)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break
        grad /= gnorm
        v = beta * v + (1.0 - beta) * grad
        vn = float(np.linalg.norm(v))
        if vn < 1e-16:
            break
        v /= vn
        accepted = False
        step = lr
        for _ in range(7):
            a_try = a - step * v
            a_try = np.clip(a_try, 0.0, None)
            s_try = float(a_try.sum())
            if s_try < 1e-14:
                step *= 0.5
                continue
            a_try = a_try / s_try
            conv_try = fft_autoconv(a_try, fft_size)
            if float(np.max(conv_try)) <= M + 1e-13:
                a = a_try
                accepted = True
                break
            step *= 0.5
        if not accepted and step < 1e-12:
            break
        if it % 400 == 0:
            lr *= 0.96
        if it % 10 == 0:
            beta = min(0.93, beta + 0.00005)
        if it % 1500 == 0 and top_k < 48:
            top_k += 2
        if it % 300 == 0:
            cand = np.clip(a * s0, 0.0, 1000.0)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < local_score:
                local_score = sc
                local_best = cand.copy()
    return a, local_best, local_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 4 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        n = 1024
        parent = np.ones(n, dtype=np.float64)
        s0 = float(n)

    fft_size = 1 << (2 * n - 1).bit_length()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    uni = np.ones(n, dtype=np.float64)
    uni_score = float(evaluate_sequence(uni.tolist()))
    if np.isfinite(uni_score) and uni_score < best_score:
        best_score, best = uni_score, uni.copy()

    j = np.arange(n, dtype=np.float64)
    cos_w = np.clip(np.cos(np.pi * j / (n - 1) + np.pi / 2), 0.05, None)
    sc = float(evaluate_sequence(cos_w.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score, best = sc, cos_w.copy()

    remaining = deadline - time.monotonic()
    t1 = time.monotonic() + 0.60 * remaining
    a = best / best.sum()
    a, lb, ls = run_descent(a, n, fft_size, s0, best, best_score, t1)
    if ls < best_score:
        best_score, best = ls, lb

    remaining = deadline - time.monotonic()
    if remaining > 20:
        t2 = time.monotonic() + 0.25 * remaining
        a2 = best / best.sum()
        a2, lb2, ls2 = run_descent(a2, n, fft_size, s0, best, best_score, t2)
        if ls2 < best_score:
            best_score, best = ls2, lb2

    base = best.copy()
    while time.monotonic() < deadline - 5.0:
        cand = base.copy()
        for _ in range(int(rng.integers(1, 4))):
            w = int(rng.integers(max(1, n // 80), max(2, n // 15)))
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= float(np.exp(rng.normal(0.0, 0.04)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best, base = sc, cand, cand.copy()

    uni_s = float(evaluate_sequence(np.ones(n, dtype=np.float64).tolist()))
    if np.isfinite(uni_s) and uni_s < best_score:
        best = np.ones(n, dtype=np.float64)
        best_score = uni_s

    out = np.clip(best, 0.0, 1000.0)
    final_sc = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(final_sc) or out.sum() < 0.01:
        out = np.ones(n, dtype=np.float64)
    return [float(x) for x in out]
