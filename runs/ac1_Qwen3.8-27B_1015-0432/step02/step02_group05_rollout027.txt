import numpy as np
import time

def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def grad_topk(a, cv, tk):
    n = a.size
    idx = np.argpartition(cv, -tk)[-tk:]
    g = np.zeros(n)
    for kk in idx:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    g /= tk
    return g

def grad_softmax(a, cv, tau):
    n = a.size
    cmax = cv.max()
    w = np.exp((cv - cmax) / tau)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    th = 1e-6 / n
    active = np.where(w > th)[0]
    g = np.zeros(n)
    for kk in active:
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            g[lo:hi + 1] += w[kk] * a[kk - hi:kk - lo + 1][::-1]
    return g

def run_opt(a, n, fsz, dl, mode, val, eta_f, beta, stall_lim):
    a = np.maximum(a, 1e-15).astype(np.float64)
    a /= a.sum()
    ba = a.copy()
    bm = fft_conv(a, fsz).max()
    g_avg = np.zeros(n)
    stall = 0
    pm = bm
    it = 0
    while time.monotonic() < dl:
        cv = fft_conv(a, fsz)
        m = cv.max()
        stall = 0 if m < pm - 1e-15 else stall + 1
        pm = m
        if stall > stall_lim:
            break
        if m < bm:
            bm = m
            ba = a.copy()
        if mode == 'topk':
