def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _fft_cross_corr(a, w, n_out):
    """Compute cross-correlation C[m] = sum_j a[j]*w[j+m] for m=0..n_out-1"""
    N = 1 << (len(a) + len(w) - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    return np.fft.irfft(A * np.conj(W), N)[:n_out]

def _grad_obj(a, soft_beta):
    """Compute gradient of 2n*max(a*a)/s^2 using soft-softmax."""
    n = len(a)
    c = _fft_conv(a, a)
    cmax = c.max()
    # Clamp for numerical stability
    w = np.exp(soft_beta * (c - cmax))
    Z = w.sum()
    if Z < 1e-300:
        # All mass on the max elements
        w = np.zeros_like(c)
        w[c >= cmax - 1e-10] = 1.0 / np.sum(c >= cmax - 1e-10)
    else:
        w /= Z
    s = a.sum()
    C = _fft_cross_corr(a, w, n)  # C[m] = sum_j a[j]*w[j+m]
    M = cmax
    grad = 4.0 * n / (s * s) * C - 4.0 * n * M / (s ** 3)
    return grad, c

def _eval_coord_at(x, a, b, k, s_val, n_val):
    """Evaluate objective with a[k] replaced by x (b is unchanged convolution)."""
    old = a[k]
    d = x - old
    if abs(d) < 1e-15:
        return 2.0 * n_val * b.max() / (s_val * s_val)
    s_new = s_val + d
    if s_new < 0.01:
        return 1e10
    lo = k
    hi = min(k + n_val - 1, 2 * n_val - 2)
    seg_len = hi - lo + 1
    b_aff = b[lo:hi + 1] + 2.0 * d * a[:seg_len]
    idx_2k = 2 * k - lo
    b_aff[idx_2k] += d * d
    if lo > 0:
        max_ua = b[:lo].max()
    else:
        max_ua = -1e10
    if hi + 1 < 2 * n_val - 1:
        max_ua = max(max_ua, b[hi + 1:].max())
    max_b = max(b_aff.max(), max_ua)
    return 2.0 * n_val * max_b / (s_new * s_new)

def _optimize_coord(k, a, b, s_val, n_val, deadline):
    """Optimize coordinate k using trial points + bisection."""
    old = a[k]
    cur_f = _eval_coord_at(old, a, b, k, s_val, n_val)
    # Try several candidates
    trial_ps = [0.0, old * 0.25, old * 0.5, old * 0.75, old * 1.25,
                old * 1.5, old * 2.0, s_val / n_val]
    best_x = old
    best_f = cur_f
    for tx in trial_ps:
        tx = max(0.0, min(1000.0, tx))
        if abs(tx - old) < 1e-10:
            continue
        tf = _eval_coord_at(tx, a, b, k, s_val, n_val)
        if tf < best_f:
            best_f = tf
            best_x = tx
    # Bisection: find bracket
    lo_b = max(0.0, best_x * 0.5 - s_val / n_val)
    hi_b = min(1000.0, best_x * 1.5 + s_val / n_val)
    # Actually, let me do a simple local search with bisection
    # Try to bracket: expand around best_x
    step = max(0.01, s_val / n_val * 0.1)
    for _ in range(8):
        x1 = max(0.0, best_x - step)
        x2 = min(1000.0, best_x + step)
        f1 = _eval_coord_at(x1, a, b, k, s_val, n_val)
        f2 = _eval_coord_at(x2, a, b, k, s_val, n_val)
        if f1 < best_f:
            best_x, best_f = x1, f1
        if f2 < best_f:
            best_x, best_f = x2, f2
        if f1 > f2:
            # Minimum is between x1 and x2
            # Bisection
            for _ in range(12):
                mid = 0.5 * (x1 + x2)
                fm = _eval_coord_at(mid, a, b, k, s_val, n_val)
                if fm < f1:
                    x2, f2 = mid, fm
                else:
                    x1, f1 = mid, fm
                if time.monotonic() > deadline:
                    break
            best_x = 0.5 * (x1 + x2)
            best_f = min(f1, f2, _eval_coord_at(best_x, a, b, k, s_val, n_val))
            break
        step *= 0.7
        if step < 1e-6:
            break
        if time.monotonic() > deadline:
            break
    return best_x, best_f
