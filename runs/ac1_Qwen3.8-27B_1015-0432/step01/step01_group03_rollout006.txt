- M ≈ 1/n (at the center)
- corr[i] = sum_j a[j]*p[j+i]. If p is concentrated at j=n-1, then corr[i] ≈ a[i-n+1] ≈ 1/n.
- g_tan[i] = 4n * (1/n - 1/n) = 0 (uniform sequence has zero gradient, as expected by symmetry)

For a perturbed sequence, the gradient will be small (O(1/n^2) or so). The step size needs to be large enough to make meaningful changes.

Let me normalize the gradient: g_dir = g_tan / ||g_tan||_2 (unit direction), then step = eta * a (multiplicative) with eta ~ 0.1.

Actually, the simplest effective approach:
new_a = cur * np.exp(-eta * g_tan / max(1, ||g_tan||_inf))

This normalizes the gradient so the step is proportional to eta regardless of scale.

import numpy as np
import time
from numpy.fft import fft, ifft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _smooth_max_grad(a, tau, L1, L2):
    n = a.size
    A = fft(a, L1)
    c = ifft(A * A).real[:2 * n - 1]
    cm = c.max()
    e = np.exp((c - cm) / tau)
    se = e.sum()
    p = e / se
    M = cm + tau * np.log(se)
    P = fft(p, L2)
    Ach = fft(a, L2)
    corr = ifft(P * np.conj(Ach)).real[:n]
    return M, corr


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(budget_s)) - 10.0

    try:
        a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if a.ndim != 1 or a.size == 0 or not np.all(np.isfinite(a)):
            raise ValueError("bad init")
    except Exception:
        a = np.ones(1024, dtype=np.float64)

    n = a.size
    S = a.sum()
    if S < 1e-8:
        a[:] = 1.0
        S = float(n)
    a = a / S

    best = a.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_sc):
        a[:] = 1.0
        best = a.copy()
        best_sc = float(evaluate_sequence(best.tolist()))

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n - 2)

    cur = a.copy()
    step = 0.5
    max_step = 5.0

    taus = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
    steps_per_tau = 3000

    total_plan_steps = len(taus) * steps_per_tau
    eval_interval = max(50, total_plan_steps // 30)
    step_count = 0

    for tv in taus:
        if time.monotonic() > deadline - 120:
            break
        step = 0.5
        for it in range(steps_per_tau):
            if time.monotonic() > deadline - 120:
                break
            S = cur.sum()
            if S < 1e-10:
                break

            M, corr = _smooth_max_grad(cur, tv, L1, L2)
            g_tan = corr - corr.mean()
            g_scale = np.max(np.abs(g_tan))
            if g_scale < 1e-15:
                break
            g_norm = g_tan / g_scale

            new_a = cur * np.exp(-step * g_norm)
            new_a = np.clip(new_a, 0.0, 1000.0)
            ns = new_a.sum()
            if ns < 1e-10:
                step *= 0.5
                continue
            new_a = new_a / ns

            dot = np.dot(g_tan, new_a - cur)
            if dot < 0:
                cur = new_a
                step = min(step * 1.03, max_step)
            else:
                step *= 0.5
                if step < 0.01:
                    step = 0.5
                    break

            step_count += 1
            if step_count % eval_interval == 0:
                sc = float(evaluate_sequence(cur.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc

    # Final gradient polish with small tau
    if time.monotonic() < deadline - 120:
        cur = best.copy()
        tv = 0.005
        step = 0.1
        for it in range(10000):

                break

            if S < 1e-10:

            g_scale = np.max(np.abs(g_tan))

            new_a = cur * np.exp(-step * g_norm)
            new_a = np.clip(new_a, 0.0, 1000.0)
            ns = new_a.sum()
            if ns < 1e-10:
                step *= 0.5
                continue
            new_a /= ns
            dot = np.dot(g_tan, new_a - cur)
            if dot < 0:
                cur = new_a
                step = min(step * 1.01, 2.0)
            else:
                step *= 0.5
                if step < 0.005:
                    break
            if it % 200 == 0:
                sc = float(evaluate_sequence(cur.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cur.copy()
                    best_sc = sc

    # Stochastic local search
    if time.monotonic() < deadline - 3:

            for _ in range(4):
                if time.monotonic() > deadline - 3:
                    break

                s = int(rng.integers(0, max(1, n - win)))
                w = int(rng.integers(max(2, win // 4), win + 1))
                if s + w > n:
                    w = n - s
                cand[s:s + w] *= np.exp(rng.normal(0.0, 0.04, size=w))
                if rng.random() < 0.25:
                    s2 = int(rng.integers(0, max(1, n - win)))
                    w2 = int(rng.integers(max(2, win // 8), win // 2 + 1))
                    if s2 + w2 > n:
                        w2 = n - s2
                    cand[s2:s2 + w2] *= np.exp(rng.normal(0.0, 0.08, size=w2))
                cand = np.clip(cand, 0.0, 1000.0)
                cs = cand.sum()
                if cs > 0.01:
                    sc = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc) and sc < best_sc:
                        best = cand.copy()
                        best_sc = sc
                        batch_improved = True
                if batch_improved:
                    break

            if not batch_improved and rng.random() < 0.1:
                # Occasional larger perturbation
                cand = best.copy()
                nseg = int(rng.integers(3, 10))
                for _ in range(nseg):
                    s = int(rng.integers(0, max(1, n - win)))
                    w = int(rng.integers(max(2, win // 4), win))
                    if s + w > n:
                        w = n - s
                    cand[s:s + w] *= np.exp(rng.normal(0.0, 0.15, size=w))
                cand = np.clip(cand, 0.0, 1000.0)
                if cand.sum() > 0.01:
                    sc = float(evaluate_sequence(cand.tolist()))
                    if np.isfinite(sc) and sc < best_sc:
                        best = cand.copy()
                        best_sc = sc

    # Final verification
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        return [1.0] * n

    return [float(v) for v in best]
