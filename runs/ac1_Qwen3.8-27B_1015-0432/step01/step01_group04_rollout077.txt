Let me also make sure the spectral step is correct. The idea:
1. Compute c = conv(a, a) (autocorrelation, length 2n-1)
2. Normalize: c = c / sum(c)
3. Raise to power alpha: p = c^alpha (this "softens" the peak)
4. Normalize: p = p / sum(p)
5. Fourier transform: T = FFT(p)
6. Take sqrt: sqrt(T) (this gives the "amplitude" of a signal whose autocorrelation is p)
7. Inverse FFT: a_new = IFFT(sqrt(T))[:n]
8. Project to nonnegative

def _proj(a, fb):
    a = np.asarray(a, dtype=np.float64)
    fb = np.asarray(fb, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size != fb.size or a.size < 1 or float(a.sum()) < 0.01:
        return fb.copy()
    return a

def _scf(a, N):
    n = a.size
    s = float(a.sum())
    if n < 1 or s < 0.01:
        return np.inf
    L = 2 * n - 1
    if N < L:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if mc < 0.0 or not np.isfinite(mc):
        return np.inf
    return 2.0 * n * mc / (s * s)

def _spec(a, N, al):
    n = a.size
    L = 2 * n - 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cs = float(c.sum())
    if cs <= 0.0:
        return a.copy()
    p = np.power(c / cs + 1e-15, al)
    ps = float(p.sum())
    if ps <= 0.0 or not np.isfinite(ps):
        return a.copy()
    p = p / ps
    T = np.fft.rfft(p, N)
    a2 = np.fft.irfft(np.sqrt(np.maximum(T, 0.0)), N)[:n]
    return np.where(np.isfinite(a2), a2, 0.0)
