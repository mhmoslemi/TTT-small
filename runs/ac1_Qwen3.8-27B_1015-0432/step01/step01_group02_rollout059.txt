            st = int(rng.integers(0, n - w_len))
            a[st:st + w_len] *= float(np.exp(rng.normal(0.0, 0.6)))
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 > 1e-12:
                a /= s2
            stuck = 0
            alpha = 0.10
            topK = int(rng.integers(18, 40))

        if stuck > 1200:
            a = best_a.copy()
            w_len = int(rng.integers(n // 20, n // 5))
            st = int(rng.integers(0, n - w_len))
            a[st:st + w_len] *= float(np.exp(rng.normal(0.0, 0.8)))
            a = np.maximum(a, 0.0)
            s2 = a.sum()
            if s2 > 1e-12:
                a /= s2
            stuck = 0
            alpha = 0.08
            topK = int(rng.integers(20, 50))

        if iters % 500 == 0:
            topK = int(rng.integers(18, 45))

    remaining = deadline - time.monotonic()
    if remaining > 15:
        a = best_a.copy()
        for _ in range(3000):
            if time.monotonic() > deadline:
                break
            c = _conv(a)
            score = 2.0 * n * c.max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
            K5 = min(5, len(c))
            idx5 = np.argpartition(c, -K5)[-K5:]
            h = np.zeros(n)
            for j in range(K5):
                kj = int(idx5[j])
                lo = max(0, kj - n + 1)
                hi = min(n - 1, kj)
                if hi >= lo:
                    h[lo:hi + 1] += a[kj - np.arange(lo, hi + 1)]
            hmax = h.max()
            if hmax > 1e-15:
                a *= np.exp(-0.015 * h / hmax)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = best_a.copy()
                break
            a /= s

    result = [float(x) for x in best_a]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs):
        result = [1.0] * n
        fs = float(evaluate_sequence(result))
    return result
