import numpy as np
import time

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-14:
        return None
    return a / s

def init_seq(k, n, rng, par):

        a = np.abs(rng.standard_normal(n)) ** 2

        a = np.maximum(np.exp(-0.5 * ((np.arange(n) - n * 0.5) / s) ** 2), 0.01)
    elif k == 5:

def wgrad(a, conv, n, nl):
    nl = min(nl, conv.size)
    idx = np.argpartition(conv, -nl)[-nl:]
    vals = conv[idx]
    ref = float(np.median(conv))
    exc = np.maximum(vals - ref, 1e-30)
    w = exc / exc.sum()
    g = np.zeros(n)
    for i in range(nl):
        k = int(idx[i])
        wi = 2.0 * w[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[k - hi:k - lo + 1][::-1]
    return g
