import time

def fft_conv(a, size):
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)

def topm_grad(a, idx, w, top_m):
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for i in range(top_m):
        k = int(idx[i])
        wi = float(w[i])
        if wi < 1e-16:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[k - hi:k - lo + 1][::-1]
    return 2.0 * g

def norm1(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a / s

    n = a_init.size
    a = norm1(a_init)
    if a is None:
        return None
    beta = beta0
    lr = lr0
    prev_M = np.inf
    stall = 0

            stall = 0
        else:
