    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _local_search(best, best_sc, rng, n, dl):
    amp = 0.05
    while time.monotonic() < dl:
        cd = best.copy()
        r = rng.random()
        if r < 0.35:
            w = max(2, n // (30 + int(50 * rng.random())))
            s = int(rng.integers(0, n - w))
            cd[s:s + w] *= float(np.exp(rng.normal(0.0, amp)))
        elif r < 0.60:
            for _ in range(int(rng.integers(3, 25))):
                j = int(rng.integers(0, n))
                cd[j] *= float(np.exp(rng.normal(0.0, amp * 1.2)))
        elif r < 0.85:
            cd *= float(np.exp(rng.normal(0.0, amp * 0.3)))
            w = max(2, n // 120)
            for _ in range(4):
                s = int(rng.integers(0, n - w))
                cd[s:s + w] *= float(np.exp(rng.normal(0.0, amp * 0.7)))
        else:
            w1 = max(2, n // 200)
            s1 = int(rng.integers(0, n - w1))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(0.0, amp * 0.5)))
            w2 = max(2, n // 200)
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.0, amp * 0.5)))
        cd = np.clip(cd, 1e-10, 1000.0)
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < best_sc:
            best = cd.copy()
            best_sc = s2
            amp = min(amp, 0.08)
        else:
            amp = max(0.005, amp * 0.995)
    return best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 12.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)
    a0 = par.copy()

    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    cur = a0.copy()
    bounds = [(0.0, 1000.0)] * n

    taus = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0004]
    L = _fft_pow2(3 * n - 2)

    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 200.0:
            break
        maxit = 100 if i < 6 else 150
        try:
            res = minimize(
                _obj, cur, args=(tv,), jac=_grad,
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': maxit, 'gtol': 1e-12,
                         'maxls': 25, 'ftol': 1e-16})
            nx = np.clip(res.x, 0.0, 1000.0)
            sc = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

        if i >= 4 and time.monotonic() < dl - 200.0:
            try:
                A = fft(cur, L)
                A *= np.exp(1j * rng.normal(0, 0.12, A.size))
                pc = np.clip(ifft(A).real[:n], 0.0, 1000.0)
                psc = float(evaluate_sequence(pc.tolist()))
                if np.isfinite(psc) and psc < best_sc:
                    best = pc.copy()
                    best_sc = psc
                cur = pc.copy()
                if time.monotonic() < dl - 200.0:
                    res2 = minimize(
                        _obj, cur, args=(tv,), jac=_grad,
                        method='L-BFGS-B', bounds=bounds,
                        options={'maxiter': 50, 'gtol': 1e-12,
                                 'maxls': 20, 'ftol': 1e-16})
                    nx2 = np.clip(res2.x, 0.0, 1000.0)
                    sc2 = float(evaluate_sequence(nx2.tolist()))
                    if np.isfinite(sc2) and sc2 < best_sc:
                        best = nx2.copy()
                        best_sc = sc2
                    cur = nx2.copy()
            except Exception:
                pass

    best, best_sc = _local_search(best, best_sc, rng, n, dl - 3.0)
    return [float(v) for v in best]
