import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _p2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _smax(c, tau):
    cm = c.max()
    d = np.exp((c - cm) / tau)
    s = d.sum()
    if s < 1e-300:
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
        return cm, p
    return cm + tau * np.log(s), d / s


def _conv(a):
    n = a.size
    L = _p2(2 * n)
    A = fft(a, L)
    return ifft(A * A).real[:2 * n - 1]


def _fobj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _fjac(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _p2(3 * n)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    r = cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * (r - mv)


def _ls_window(bv, sc, rng, n, dl):
    w = max(2, n // 60)

        cd = bv.copy()
        s = int(rng.integers(0, n - w))
        cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.02)))
        if rng.random() < 0.15:
            cd *= float(np.exp(rng.normal(0.0, 0.005)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() < 1e-8:
            continue
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc:

            sc = s2
    return bv, sc


def _coord_step(bv, sc, rng, n, dl, n_iter=5000):

            break
        j = int(rng.integers(0, n))

                best_s = s2
                best_x = base * fac
        bv[j] = best_x
        sc = best_s
    return bv, sc

   - Else: c_new[k] = c[k]
3. max_new = max(c_new)
4. S_new = S_old + v
5. obj = 2n * max_new / S_new^2

Step 2 is O(n) for the affected indices. Step 3 is O(n) to find the max. So the total per trial is O(n) after the initial O(n log n) convolution.

For the coordinate descent:
- Compute c once: O(n log n)
- For each of M trials: O(n) to update and check
- Total: O(n log n + M*n)

With M=6 trials per coordinate and 6733 coordinates: O(6733 * (n log n + 6n)) ≈ O(6733 * 7n) ≈ O(47n^2) ≈ O(2.1 * 10^8) for n=6733. That's a few seconds.
