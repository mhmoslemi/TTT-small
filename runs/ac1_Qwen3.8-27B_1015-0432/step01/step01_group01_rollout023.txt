import time

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)

def compute_grad(a, conv, top_k, temp):
    n = a.size
    conv_len = conv.size
    k = min(top_k, conv_len)
    idx = np.argpartition(conv, -k)[-k:]
    vals = conv[idx]
    w = np.exp((vals - vals.max()) / temp)
    w /= w.sum()
    grad = np.zeros(n, dtype=np.float64)
    for i in range(k):
        kv = int(idx[i])
        lo = max(0, kv - n + 1)
        hi = min(n - 1, kv)
        if lo <= hi:
            grad[lo:hi+1] += w[i] * a[kv-hi:kv-lo+1][::-1]
    return grad

    if s < 1e-14:

    return a * (target / s)

            (64, 0.5, 0.3), (16, 0.3, 0.45), (40, 0.25, 0.35)]
    for i in range(n_inits):
        if time.monotonic() > hard_dl - 60.0:
            break
        tk, tp, tl = cfgs[i % 6]
        a0 = inits[i].copy()
        dl_i = time.monotonic() + t_alloc
        a_opt, sc = optimize(a0, n, fs, dl_i, tk, tp, tl)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best = np.clip(a_opt * psum, 0.0, 1000.0)
        # Second pass with reduced k from best
        if time.monotonic() < hard_dl - 120.0 and np.isfinite(sc):
            a2, sc2 = optimize(a_opt, n, fs, time.monotonic() + t_alloc * 0.6, max(4, tk // 3), tp * 0.7, tl * 0.5)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best = np.clip(a2 * psum, 0.0, 1000.0)
    if time.monotonic() < hard_dl - 15.0:
        best, best_score = polish(best, n, hard_dl - 3.0, rng)
    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
    return [float(x) for x in out]
