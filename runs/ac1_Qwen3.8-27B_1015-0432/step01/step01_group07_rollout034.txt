            at = np.maximum(a - t * g, 0.0)
            s = float(np.sum(at))
            if s < 1e-8:
                continue
            at /= s
            ts = _score(at)
            if np.isfinite(ts) and ts < bt:
                bt = ts
                ba = at.copy()
                ok = True

        if ok:
            a = ba
            stall = 0
            if bt < best_score:
                best_score = bt
                best = a.copy()
            lr = min(lr * 1.04, 0.5)
            eps = max(eps * 0.98, 0.002)
        else:
            stall += 1
            lr *= 0.85
            if stall > 40:
                a += rng.normal(0.0, 0.005 / np.sqrt(n), n)
                a = np.maximum(a, 0.0)
                ss = float(np.sum(a))
                if ss > 1e-8:
                    a /= ss
                stall = 0
                lr = 0.05
                eps = 0.02
                spec_left = 10
        it += 1

    out = (best * S0).astype(np.float64)
    out = np.clip(out, 0.0, 1000.0)
    fv = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(fv):
        out2 = np.clip(parent, 0.0, 1000.0)
        fv2 = float(evaluate_sequence(out2.tolist()))
        if not np.isfinite(fv2):
            return [1.0] * n
        out = out2
    return [float(x) for x in out]
