import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def run_opt(f, deadline, rng):
    """Main optimization loop."""
    n = f.size
    f = f.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    beta = 20.0
    eta = 0.1
    best_f = f.copy()
    best_score = fast_score(f)
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.5  # momentum coefficient
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break

        # w_j = exp(beta * (b_j - M) / M) to focus on peaks
        w = np.exp(beta * (b / M - 1.0))
        w /= w.sum()
        
        # Gradient: for each i, d(max b)/d(a_i) = sum_j w_j * (∂b_j/∂a_i)
        # ∂b_j/∂a_i = a_{j-i} (for valid indices) i.e., correlation of w with reversed f

        gc = fft_conv(w, f_r)
        grad = gc[n-1:2*n-1].copy()
        grad = np.maximum(grad, 0)
        gmax = grad.max()
        
        if gmax < 1e-30:
            break
        
        grad /= gmax
        
        # Momentum update
        momentum = mu * momentum + (1-mu) * grad
        momentum /= momentum.max()
        
        # Multiplicative update
        f_new = f * np.exp(-eta * momentum)
        f_new = np.maximum(f_new, 1e-20)
        f_new /= f_new.sum()
        
        # Check improvement
        if it % 50 == 0:
            sc = fast_score(f_new)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            f = f_new
        else:
            f = f_new
        
        # Adaptive parameters
        if it % 500 == 0 and it > 0:
            beta = min(200.0, beta * 1.05)
            eta = max(0.01, eta * 0.995)
            mu = min(0.8, mu * 1.01)
