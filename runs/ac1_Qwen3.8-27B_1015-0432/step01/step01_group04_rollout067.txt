def _pow2(x):
    m = 1
    x = int(max(x, 1))
    while m < x:
        m <<= 1
    return m

def _score_fft(a, N):
    n = a.size
    s = a.sum()
    if s < 1e-12:
        return np.inf, None
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = np.maximum(c[:L], 0.0)
    sc = 2.0 * n * c.max() / (s * s)
    return sc, c
