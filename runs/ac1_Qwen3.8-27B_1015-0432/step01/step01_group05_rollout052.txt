    fs = float(evaluate_sequence(a.tolist()))
    if np.isfinite(fs) and fs < best_s:
        best = a.copy()

    result = [float(min(1000.0, max(0.0, x))) for x in best]
    if not result or sum(result) < 0.01:

    return result
