def uniform(n):

def safe_norm(a):
    a = np.asarray(a, dtype=np.float64).ravel()

    a = np.maximum(a, 0.0)

def make_inits(n, rng, parent):

    f = np.fft.rfft(a, n=fs)

    return c[:2 * a.size - 1], f


def smooth_grad(a, f, x, p, fs):
    xmax = float(x.max())
    if xmax <= 1e-300:
        return np.zeros_like(a, dtype=np.float64)
    q = max(0.0, float(p) - 1.0)
    ratio = x / xmax
    ratio = np.clip(ratio, 1e-300, 1.0)
    if q <= 0.0:
        w = np.ones_like(ratio, dtype=np.float64)
    else:
        w = ratio ** q
    w += 1e-10
    ws = float(w.sum())
    if ws <= 1e-300:
        return np.zeros_like(a, dtype=np.float64)
    w /= ws
    fw = np.fft.rfft(w, n=fs)
    res = np.fft.irfft(np.conj(fw) * f, n=fs)
    g = np.empty(a.size, dtype=np.float64)
    g[0] = res[0]
    if a.size > 1:

    return g
