def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _quick_score(corr, n, s):
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(corr.max()) / (s * s)


def _subgrad(corr, a, n, k):
    m = len(corr)
    if k >= m:
        idx = np.arange(m)
    else:
        idx = np.argpartition(corr, -k)[-k:]
    g = np.zeros(n)
    for ki in idx:
        ki = int(ki)

        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += a[lo:hi + 1][::-1] * 0 + a[ki - np.arange(lo, hi + 1)]
    return g / k
