import numpy as np
import time

def npow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def autocorr(a):
    n = a.size
    L = 2*n - 1
    N = npow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:L]
    return np.maximum(c, 0.0)

def score(a):
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    c = autocorr(a)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0*a.size*mc/(s*s))

def score_grad(a):
    """Returns (score, grad) where grad = d(score)/da"""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf, np.zeros(n)
    c = autocorr(a)
    mc = float(c.max())
    mstar = int(np.argmax(c))
    sc = float(2.0*n*mc/(s*s))
    # grad of mc w.r.t. a
    gmc = np.zeros(n, dtype=np.float64)
    lo = max(0, mstar - n + 1)
    hi = min(n-1, mstar)
    if hi >= lo:
        gmc[lo:hi+1] = 2.0*a[mstar-hi:mstar-lo+1][::-1]
    if mstar % 2 == 0:
        jm = mstar // 2
        if 0 <= jm < n:
            gmc[jm] = a[jm]
    # d(score)/da = (2n/s^2)*(gmc - 2*mc/s)
    grad = (2.0*n/(s*s)) * (gmc - 2.0*mc/s)
    return sc, grad

def project(a):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 1e-30, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones_like(a)
    return a

def spectral_sharpen(a, N, alpha, n_iter=50, deadline=None):
    """Iterative spectral sharpening to reduce peak autocorrelation."""
    n = a.size
    best = a.copy()
    best_sc = score(best)
    cur = a.copy()
    for it in range(n_iter):
        if deadline and time.monotonic() > deadline:
            break
        c = autocorr(cur)
        cs = float(c.sum())
        if cs <= 0:
            break
        cn = c / cs
        cn_alpha = np.power(cn + 1e-20, alpha)
        sct = float(cn_alpha.sum())
        if sct <= 0:
            break
        cn_alpha /= sct
        T = np.fft.rfft(cn_alpha, N)
        A2 = np.sqrt(np.maximum(T, 0))
        a2 = np.fft.irfft(A2, N)[:n]
        a2 = np.maximum(a2, 0.0)
        sa2 = float(a2.sum())
        if sa2 > 0:
            a2 /= sa2
        mix = 0.7
        cand = (1-mix)*cur + mix*a2
        cand = project(cand)
        sc = score(cand)
        if np.isfinite(sc) and sc < best_sc:
            best = cand.copy()
            best_sc = sc
        if np.isfinite(sc) and sc <= score(cur)*1.05:
            cur = cand
        if it % 10 == 0:
            alpha *= 0.98  # slowly decrease
    return best, best_sc

def run_adam(a0, deadline, seed):
    """Adam optimization in log-amplitude space."""
    n = a0.size
    L = 2*n - 1
    N = npow2(L)
    rng = np.random.default_rng(seed)
    
    # Initialize in log space
    a0 = project(a0)

    beta1, beta2 = 0.9, 0.999
    t = 0
    lr = 0.05
    
    best = a0.copy()
    best_sc = score(best)
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Compute a from y
        y_shift = y - float(y.max())
        a = np.exp(np.clip(y_shift, -50, 50))
        a = project(a)
        s = float(a.sum())
        if s < 0.01:
            a = best.copy()
            s = float(a.sum())
        
        # Score and gradient
        sc, grad = score_grad(a)
        if not np.isfinite(sc):
            break
        
        if sc < best_sc - 1e-10:
            best = a.copy()
            best_sc = sc
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart logic
        if no_improve > 500:
            # Perturb from best
            y = np.log(project(best) + 1e-30)
            y -= y.mean()
            y += rng.normal(0, 0.3, n)
            y = np.clip(y, -20, 20)
            m.fill(0); v.fill(0)
            t = 0
            lr = max(0.01, lr * 0.5)
            no_improve = 0
            continue
        
        # Convert gradient to log-space
        log_grad = grad * a  # chain rule: d(score)/dy = d(score)/da * da/dy = grad * a
        
        # Adam update
        t += 1
        m = beta1*m + (1-beta1)*log_grad
        v = beta2*v + (1-beta2)*log_grad*log_grad
        b1c = 1 - beta1**t
        b2c = 1 - beta2**t
        step = lr * (m/b1c) / (np.sqrt(v/b2c) + 1e-8)
        step = np.clip(step, -0.5, 0.5)
        y = y - step
        y -= y.mean()
        y = np.clip(y, -20, 20)
        
        # Adaptive learning rate
        if t % 100 == 0:
            if no_improve > 100:
                lr *= 0.9
            elif no_improve == 0:
                lr = min(lr * 1.05, 0.15)
    
    return best, best_sc

    n = a0.size
