def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def topk_grad(a, conv, k):
    n = a.size
    idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros(n)
    for lag in idx:
        lag = int(lag)
        # d b[lag]/d a[j] = 2*a[lag-j] if 0<=lag-j<n

        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        g[lo:hi+1] += 2.0 * a[lag - hi : lag - lo + 1][::-1]
    return g / float(k)
