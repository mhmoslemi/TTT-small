    while time.monotonic() < deadline - 10.0:
        nsteps = 200
        for si in range(nsteps):
            t_adam += 1
            ca = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), L)[:2 * n - 1]
            cm = ca.max()
            ea = sbeta * (ca - cm)
            mx = ea.max()
            w = np.exp(ea - mx)
            Z = w.sum()
            if Z < 1e-30:
                w = np.zeros(2 * n - 1)
                w[np.argmax(ca)] = 1.0
            else:
                w /= Z
            buf_ar[:n] = a[::-1]
            buf_ar[n:] = 0.0
            buf_w[:2 * n - 1] = w
            buf_w[2 * n - 1:] = 0.0
            conv_g = np.fft.irfft(np.fft.rfft(buf_ar, L) * np.fft.rfft(buf_w, L), L)
            grad = 2.0 * conv_g[n - 1:2 * n - 1]
            grad -= grad.mean()

            m_buf *= b1
            m_buf += (1.0 - b1) * grad
            v_buf *= b2
            v_buf += (1.0 - b2) * grad * grad
            tc1 = 1.0 - b1 ** t_adam
            tc2 = 1.0 - b2 ** t_adam
            a -= lr * (m_buf / tc1) / (np.sqrt(v_buf / tc2) + eps)
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa < 1e-12:
                a = best.copy()
                m_buf[:] = 0.0
                v_buf[:] = 0.0
                t_adam = 0
            else:
                a *= S0 / sa

        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_s:
            best = a.copy()
            best_s = sc
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                sbeta = min(sbeta * 1.5, 500.0)
                lr = max(lr * 0.7, 0.0005)
                a = a * (1.0 + 0.02 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                sa2 = a.sum()
                if sa2 > 1e-12:
                    a *= S0 / sa2
                else:
                    a = best.copy()
                m_buf[:] = 0.0
                v_buf[:] = 0.0
                t_adam = 0
                stall = 0
            if stall >= 8:
                a = best * (1.0 + 0.05 * rng.standard_normal(n))
                a = np.maximum(a, 0.0)
                sa3 = a.sum()
                if sa3 > 1e-12:
                    a *= S0 / sa3
                else:
                    a = best.copy()
                sbeta = 20.0
                lr = 0.003
                m_buf[:] = 0.0
                v_buf[:] = 0.0
                t_adam = 0
                stall = 0

        if t_adam % 1000 < nsteps:
            sbeta = min(sbeta * 1.1, 500.0)
            lr = max(lr * 0.99, 0.0005)

    fs = float(evaluate_sequence(a.tolist()))
    if np.isfinite(fs) and fs < best_s:
        best = a.copy()

    result = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    if not result or sum(result) < 0.01:
        result = [1.0]
    return result
