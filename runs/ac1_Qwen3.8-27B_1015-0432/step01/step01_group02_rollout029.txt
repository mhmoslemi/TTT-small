    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real

def _make_h(c, a, n, K):
    K = min(K, 2 * n - 1)
    idx = np.argpartition(c, -K)[-K:]
    peaks = c[idx]
    pmax = peaks.max()
    if pmax > 1e-15:
        weights = peaks / pmax
    else:
        weights = np.ones(K) / K
    h = np.zeros(n)
    for j in range(K):
        kj = int(idx[j])
        wj = float(weights[j])
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += wj * a[kj - np.arange(lo, hi + 1)]
    return h
