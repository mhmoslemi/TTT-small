def _fp2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _smax(c, tau):
    cm = c.max()
    e = np.exp(np.clip((c - cm) / tau, -60.0, 60.0))
    s = e.sum()
    if s < 1e-300:
        s = 1e-300
    return cm + tau * np.log(s), e / s

    S = a.sum()

        return np.inf

    return 2.0 * n * cv.max() / (S * S)
