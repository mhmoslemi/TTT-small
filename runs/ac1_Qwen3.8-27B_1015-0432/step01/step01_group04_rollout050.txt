def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def project_simplex(a):

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.full(a.size, 1.0 / max(1, a.size))

def conv_full(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * a.size - 1
    c = c[:L].copy()
    c = np.maximum(c, 0.0)
    return c

    return g
