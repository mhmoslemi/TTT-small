import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def compute_grad_softmax(f, beta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), M
    w = np.exp(beta * (b / M - 1.0))
    ws = w.sum()
    if ws < 1e-30:
        return np.full(n, 1.0 / n), M
    w /= ws
    ar = f[::-1]
    gc = fft_conv(w, ar)
    grad = gc[n - 1:2 * n - 1]
    return grad, M


def compute_grad_l2(f):

    ar = f[::-1]

    grad = gc[n - 1:2 * n - 1]
