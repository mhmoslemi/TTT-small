                                  'ftol': 1e-16, 'gtol': 1e-11})
            if r.fun < 1e9:
                nx = np.clip(r.x, 0.0, 1000.0)
                s2 = _score(nx)
                if s2 < sc:
                    a, sc = nx.copy(), s2
        except Exception:
            pass
    return a, sc


def _polish(a, dl):
    n = a.size
    sc = _score(a)
    a = a.copy()
    lr = 0.05
    for it in range(600):
        if time.monotonic() > dl - 4:
            break
        b = _fft_conv(a)
        S = a.sum()
        k = int(np.argmax(b))
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        g = np.zeros(n)
        g[j0:j1 + 1] = a[k - j1:k - j0 + 1][::-1]
        g_obj = (2.0 * n / S**2) * g - (4.0 * n * b[k] / S**3)
        improved = False
        for _ in range(30):
            trial = np.maximum(0.0, a - lr * g_obj)
            s2 = _score(trial)
            if s2 < sc:
                a, sc = trial, s2
                improved = True
                break
            lr *= 0.5
        if not improved:
            lr = max(lr, 0.02)
            if lr < 1e-8:
                break
    return a, sc


def _local_search(a, sc, rng, dl):
    n = a.size
    stall = 0
    while time.monotonic() < dl:
        c = a.copy()
        w = max(2, n // 50)
        s = int(rng.integers(0, n - w))
        c[s:s + w] *= np.exp(rng.normal(0.0, 0.012, w))
        c *= np.exp(rng.normal(0.0, 0.003))
        c = np.clip(c, 0.0, 1000.0)
        s2 = _score(c)
        if s2 < sc:
            a, sc, stall = c, s2, 0
        else:
            stall += 1
            if stall > 300:
                c = a * np.exp(rng.normal(0.0, 0.06, n))
                c = np.clip(c, 0.0, 1000.0)
                s2 = _score(c)
                if s2 < sc:
                    a, sc = c, s2
                stall = 0
    return a, sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 10.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except NameError:
        par = np.ones(512)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(512)
    n = par.size

    t = np.linspace(0.0, 1.0, n)
    inits = [
        np.ones(n),
        np.maximum(0.0, 1.0 - np.abs(2.0 * t - 1.0)),
        0.5 * (1.0 + np.cos(np.pi * (2.0 * t - 1.0))),
        np.exp(-6.0 * (t - 0.5) ** 2),
        np.maximum(0.0, 1.0 - (2.0 * t - 1.0) ** 2),
        par.copy(),
    ]

    best_a, best_sc = None, np.inf
    for i, init in enumerate(inits):
        sub_dl = min(t0 + (float(budget_s) - 40.0) * (i + 1) / len(inits), dl - 10.0)
        a, sc = _optimize(init.copy(), sub_dl)
        if sc < best_sc:
            best_a, best_sc = a.copy(), sc

    if best_a is not None:
        best_a, best_sc = _polish(best_a, dl - 12.0)
    else:
        best_a, best_sc = par.copy(), _score(par)

    best_a, best_sc = _local_search(best_a, best_sc, rng, dl - 3.0)

    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = np.ones(n)

    return [float(v) for v in best_a]
