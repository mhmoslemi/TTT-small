import time

def _fft_self_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _subgrad(c, a, n, K):
    K = min(K, c.size)
    top = np.argpartition(c, -K)[-K:]
    thresh = c[top].min()
    sel = [int(k) for k in top if c[k] >= thresh - 1e-12]
    if not sel:
        sel = [int(np.argmax(c))]
    g = np.zeros(n)
    for k in sel:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
    g /= len(sel)
    return g

def _score_from_conv(c, n, s):
    mx = c.max()
    if s < 1e-15:
        return 1e30
    return 2.0 * n * mx / (s * s)
