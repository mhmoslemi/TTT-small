This is `2 * (Lr * a)_{2n-2-j}` where `*` is full convolution.

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)[: 2 * a.size - 1]

    ix = np.argpartition(cv, -k)[-k:].astype(np.intp)
    g = np.zeros(n)
    for lg in ix:
        lg = int(lg)
        lo = lg - n + 1
        if lo < 0:
            lo = 0
        hi = lg if lg < n else n - 1
        if lo <= hi:
            g[lo : hi + 1] += a[lg - hi : lg - lo + 1][::-1]
    return (2.0 / k) * g

        a = an
        t = tn
