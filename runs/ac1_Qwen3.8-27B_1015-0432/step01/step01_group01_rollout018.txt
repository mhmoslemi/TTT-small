def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def subgrad(a, conv, tk):
    n = a.size
    peaks = np.argpartition(conv, -tk)[-tk:]
    g = np.zeros(n, dtype=np.float64)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g *= 2.0 / float(len(peaks))
    return g


def normalize(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a * (1.0 / s)
