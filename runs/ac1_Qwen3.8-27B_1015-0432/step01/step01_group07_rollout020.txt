    a_work, s2 = _optimize_from(a_work, n, rng, sub_deadline2, 0.1)
    if s2 < best_score:
        best_score = s2
        best = a_work.copy()

    # Final random-restart attempts if time remains
    for _ in range(20):
        if time.monotonic() > deadline - 15.0:
            break
        a_try = _project_simplex(best + rng.normal(0.0, 1.0 / n, n))
        a_try, s_try = _optimize_from(a_try, n, rng, min(deadline - 12.0, time.monotonic() + 3.0), 0.3)
        if s_try < best_score:
            best_score = s_try
            best = a_try.copy()

    # Verify with authoritative evaluator
    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        best = np.ones(n) / n
    return [float(x) for x in best]
