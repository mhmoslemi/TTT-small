import time

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)

def top_k_subgradient(a, conv, top_k):
    n = a.size
    idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[lo:hi+1][::-1]  # a[k-hi : k-lo+1] reversed
            grad[lo:hi+1] += seg
    grad /= float(top_k)
    return grad

def normalize(a, target_sum):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a * (target_sum / s)

    new_fa = phase * np.sqrt(new_power)
