import numpy as np
import time

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def proj_pos(a, fb):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.clip(np.where(np.isfinite(fb), fb, 0.0), 0.0, 1000.0)
        s = a.sum()
    if s < 0.01:
        a = np.ones(fb.size) / fb.size
    return a

def autocorr(a, N):
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = c[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c

def score_from_c(c, a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))

def compute_grad_entropic(a, c, N2, alpha, n):
    """Compute gradient of entropic proxy in log-space."""
    max_c = c.max()
    if max_c <= 1e-30:
        return np.zeros(n)
    # Compute p_k = softmax(alpha * c_k / max_c)
    sc = alpha * (c / max_c - 1.0)
    sc = np.clip(sc, -500.0, 0.0)
    e = np.exp(sc)
    zsum = e.sum()
    if zsum < 1e-30:
        p = np.zeros(len(c))
        p[int(np.argmax(c))] = 1.0
    else:
        p = e / zsum
    # h_j = (p * a_rev)[n-1+j] for j=0..n-1
    a_rev = a[::-1]
    P = np.fft.rfft(p, N2)
    AR = np.fft.rfft(a_rev, N2)
    h_full = np.fft.irfft(P * AR, N2)
    h = h_full[n-1:2*n-1]
    h = np.where(np.isfinite(h), h, 0.0)
    # Gradient of entropic proxy w.r.t. a: g = 2*h / max_c
    g = 2.0 * h / max_c
    # Convert to log-space gradient
    s = a.sum()
    ga = g / max(s, 1e-30)  # normalize to sum=1 scale
    gdot = float(np.dot(a, ga))
    grad_log = a * (ga - gdot)
    grad_log = np.where(np.isfinite(grad_log), grad_log, 0.0)
    return grad_log

def run_adam(y_init, n, N, N2, deadline, seed):
    rng = np.random.default_rng(seed)
    y = np.asarray(y_init, dtype=np.float64).copy()
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - y.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    b1p, b2p = 1.0, 1.0
    t = 0
    lr = 0.5
    best_a = None
    best_score = np.inf
    
    while time.monotonic() < deadline:
        # Compute a from y
        y_shift = y - y.max()
        a = np.exp(np.clip(y_shift, -700, 700))
        s = a.sum()
        if s < 1e-30:
            a = np.ones(n) / n
            s = 1.0
        a = a / s  # normalize sum to 1
        
        # Compute autocorrelation
        c = autocorr(a, N)
        max_c = c.max()
        score = float(2.0 * n * max_c)  # since sum=1, (sum)^2 = 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Compute gradient
        alpha = 10.0 + 50.0 * min(1.0, t / 3000.0)
        grad = compute_grad_entropic(a, c, N2, alpha, n)
        
        # Adam step
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        b1p *= b1
        b2p *= b2
        mb = m / (1 - b1p)
        vb = v / (1 - b2p)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.5, 0.5)
        y = y - step
        y = y - y.mean()
        y = np.clip(y, -50.0, 50.0)
        
        if t % 500 == 0:
            # Reduce lr
            lr *= 0.9
            if lr < 0.01:
                lr = 0.01
            # Check for stagnation
            if t % 2000 == 0 and t > 2000:
                # Perturb
                y = y + rng.normal(0, 0.1, n)
                y = y - y.mean()
                lr = max(0.1, lr)
    
    if best_a is None:

    return best_a, best_score

def make_inits(n, seed, parent, N):

    n = a.size
    A = np.fft.rfft(a, N)

    if time.monotonic() < adam1_dl:

            a1, sc1 = adam_loop(y0, n, N, N2, adam1_dl, seed + 1)
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1.copy()
        except Exception:
            pass

    # Second Adam with perturbation
    if time.monotonic() < deadline - 40.0:
        rng2 = np.random.default_rng(seed + 99)
        y1 = np.log(np.maximum(best_a, 1e-30)) + rng2.normal(0.0, 0.3, n)
        adam2_dl = min(deadline - 30.0, time.monotonic() + 0.2 * (deadline - time.monotonic()))
        if time.monotonic() < adam2_dl:
            try:
                a2, sc2 = adam_loop(y1, n, N, N2, adam2_dl, seed + 2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_a = a2.copy()
            except Exception:
                pass

    # Final exact evaluation and return
    best_a = np.maximum(best_a, 0.0)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    s = best_a.sum()
    if s < 0.01:
        best_a = np.ones(n) / n
    out = [float(x) for x in best_a]
    try:
        es = evaluate_sequence(out)
        if not np.isfinite(es):
            out = [1.0 / n] * n
    except Exception:
        out = [1.0 / n] * n
    return out
