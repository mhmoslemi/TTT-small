            if np.isfinite(sc) and sc < best_score:
                best = cand.copy()
                best_score = sc
                no_improve = 0
            if np.isfinite(sc) and sc <= current_score:
                current = cand
                current_score = sc
            elif np.isfinite(sc) and sc < current_score * 1.01:
                # Accept slightly worse to escape local minima
                current = cand
                current_score = sc
        it += 1
        # Periodic restart
        if it % 200 == 0 and no_improve > 100:
            noise_scale = 0.05 * float(np.max(best))
            current = project_positive(best + rng.normal(0, noise_scale, n), best)
            current_score = score_fft(current, N)
            no_improve = 0
    return best, best_score

def run_adam(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    n = a.size
    # Work in log-space
    a_clip = np.clip(a, 1e-300, 1000.0)
    y = np.log(a_clip)
    y = y - float(np.mean(y))
    y = np.clip(y, -30.0, 30.0)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2 = 0.9, 0.999
    b1p, b2p = 1.0, 1.0
    t = 0
    lr_base = 0.05
    L = 2 * n - 1
    best = a.copy()
    best_score = score_fft(a, N)
    no_improve = 0
    while time.monotonic() < deadline:
        # Compute a from y
        y_shift = y - np.max(y)
        a_curr = np.exp(np.clip(y_shift, -700, 700))
        a_curr = a_curr / a_curr.sum()
        
        # Compute c = a*a
        A = np.fft.rfft(a_curr, N)
        c = np.fft.irfft(A * A, N)
        c = c[:L].astype(np.float64)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        score = float(2.0 * n * max_c)  # sum(a)=1
        
        if np.isfinite(score) and score < best_score:
            best = a_curr.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        
        if not np.isfinite(score) or max_c <= 0:
            break
        
        # Restart if stuck
        if no_improve > 1500:
            y_best = np.log(np.clip(best, 1e-300, 1000))
            y = y_best + rng.normal(0, 0.05, n)
            y = y - np.mean(y)
            y = np.clip(y, -30, 30)
            m.fill(0)
            v.fill(0)
            b1p, b2p = 1.0, 1.0
            t = 0
            no_improve = 0
            continue
        
        # Compute gradient
        # sharpness increases over time
        sharp = 50.0 + 200.0 * min(1.0, t / 5000.0)
        z = np.exp(sharp * (c - max_c))
        zsum = float(z.sum())
        if not np.isfinite(zsum) or zsum <= 0:
            z = np.zeros(L)
            z[int(np.argmax(c))] = 1.0
        else:
            z = z / zsum
        # G = 2 * (a * z_rev)
        W = np.fft.rfft(z[::-1], N)
        conv = np.fft.irfft(A * W, N)
        G = 2.0 * conv[L - n:L][::-1]
        G = np.where(np.isfinite(G), G, 0.0)
        # Subtract mean to ensure sum-preserving
        gdot = float(np.dot(a_curr, G))
        grad = a_curr * (G - gdot)
        
        if not np.all(np.isfinite(grad)):
            break
        
        t += 1
        lr = lr_base / (1.0 + 0.01 * t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        b1p *= beta1
        b2p *= beta2
        mb = m / (1 - b1p)
        vb = v / (1 - b2p)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.05, 0.05)
        y = y - step
        y = y - float(np.mean(y))
        y = np.clip(y, -30, 30)
    
    return best, best_score

def local_search(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_score = score_fft(best, N)
    while time.monotonic() < deadline:
        width = int(rng.integers(max(1, n // 200), max(2, n // 50)))
        start = int(rng.integers(0, n - width))
        cand = best.copy()
        factor = float(np.exp(rng.normal(0, 0.1)))
        cand[start:start+width] *= factor
        cand = project_positive(cand, best)
        sc = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return best, best_score
