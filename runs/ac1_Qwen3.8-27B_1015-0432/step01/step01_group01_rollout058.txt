def fft_conv(a, N):

    return np.fft.irfft(fa * fa, N)

def subgrad_topk(a, b, k, T):
    """Subgradient of max(b) wrt a, using top-k lags with optional softmax."""
    n = a.size
    if n < 2:
        return np.zeros(n)

    s = a.sum()
    if s < 1e-14:

import numpy as np
import time


def fft_conv(a, N):
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)


def subgrad(a, b, k, T):
    n = a.size
    idx = np.argpartition(b, -k)[-k:]
    if T > 1e-6:
        bk = b[idx]
        w = np.exp((bk - bk.max()) / T)
        w /= w.sum()
    else:
        w = np.full(k, 1.0 / k)
    g = np.zeros(n)
    for j in range(k):
        kl = int(idx[j])
        wi = w[j]
        lo = max(0, kl - n + 1)
        hi = min(n - 1, kl)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[kl - hi:kl - lo + 1][::-1]
    return g


def normalize(a, target):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a * target / s


def run_descent(a0, N, n, deadline, rng, top_k, T0, lr0, beta0):
    a = normalize(a0, 1.0)
    if a is None:
        a = np.ones(n) / n
    v = np.zeros(n)
    beta = beta0
    lr = lr0
    T = T0
    stall = 0
    best_a = a.copy()
    best_M = np.inf
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(a, N)
        M = float(b.max())
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            break
        g = subgrad(a, b, top_k, T)
        g -= g.mean()
        v = beta * v + (1.0 - beta) * g
        a_new = np.clip(a - lr * v, 0.0, None)
        a_new = normalize(a_new, 1.0)
        if a_new is None:
            lr *= 0.5
            v *= 0.5
            continue
        a = a_new
        lr = max(0.03, lr * 0.99997)
        T *= 0.99995
        beta = min(0.96, beta + 0.000005)
        it += 1
    return best_a, it
