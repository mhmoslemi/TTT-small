        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s))


def stable_softmax(y):
    y = np.asarray(y, dtype=np.float64)
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.max(y))
    e = np.exp(np.clip(y, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return e / s


def run_spectral(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_score = score_fft(a, N)
    current = a.copy()
    cur_score = best_score
    L = 2 * n - 1
    it = 0
    no_imp = 0
    while time.monotonic() < deadline:
        alpha = max(0.5, 1.0 - 0.003 * it)
        mix = max(0.25, 0.85 - 0.004 * it)
        A = np.fft.rfft(current, N)
        c = np.fft.irfft(A * A, N)[:L]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        cs = float(c.sum())
        if cs <= 0 or not np.isfinite(cs):
            break
        cn = c / cs
        ct = np.power(cn + 1e-18, alpha)
        sct = float(ct.sum())
        if sct <= 0 or not np.isfinite(sct):
            break
        ct = ct / sct
        T = np.fft.rfft(ct, N)
        A2 = np.sqrt(np.maximum(T, 0.0))
        a2 = np.fft.irfft(A2, N)[:n]
        a2 = np.where(np.isfinite(a2), a2, 0.0)
        a2 = np.maximum(a2, 0.0)
        cand = (1.0 - mix) * current + mix * a2
        cand = project_positive(cand, current)
        sc = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
        if np.isfinite(sc) and sc <= cur_score * 1.005:
            current = cand
            cur_score = sc
            no_imp = 0 if sc < best_score else no_imp + 1
        else:
            no_imp += 1
        it += 1
        if no_imp > 50:
            current = project_positive(best * (1.0 + 0.08 * rng.normal(0, 1, n)), best)
            cur_score = score_fft(current, N)
            no_imp = 0
    return best, best_score


def run_adam(y, n, Ns, Na, deadline, seed):
    rng = np.random.default_rng(seed)
    y = np.asarray(y, dtype=np.float64).copy()
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - float(np.mean(y))
    y = np.clip(y, -30.0, 30.0)
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1 = 0.9
    b2 = 0.999
    b1p = 1.0
    b2p = 1.0
    t = 0
    lr = 0.12
    L = 2 * n - 1
    best = stable_softmax(y)
    best_score = score_fft(best, Ns)
    no_imp = 0
    while time.monotonic() < deadline:
        a = stable_softmax(y)
        A_s = np.fft.rfft(a, Ns)
        c = np.fft.irfft(A_s * A_s, Ns)[:L].astype(np.float64)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        s = float(a.sum())
        score = 2.0 * n * max_c / (s * s) if s > 0 else np.inf
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            no_imp = 0
        else:
            no_imp += 1
        if not np.isfinite(score) or max_c <= 0:
            break
        if no_imp > 1500:
            y = np.log(np.maximum(best, 1e-300))
            y = y - np.mean(y)
            y += rng.normal(0, 0.04, n)
            y = np.clip(y, -30, 30)
            m.fill(0)
            v.fill(0)
            b1p = 1.0
            b2p = 1.0
            t = 0
            lr = max(0.005, lr * 0.5)
            no_imp = 0
            continue
        progress = min(1.0, t / 12000.0)
        alpha_base = 2.0 + 150.0 * progress
        if no_imp > 200:
            alpha_base += 80.0 * min(1.0, (no_imp - 200) / 800.0)
        alpha = alpha_base / max(max_c, 1e-12)
        z = np.exp(np.clip(alpha * (c - max_c), -500.0, 0.0))
        zsum = float(z.sum())
        if not np.isfinite(zsum) or zsum <= 0:
            z = np.zeros(L, dtype=np.float64)
            z[int(np.argmax(c))] = 1.0
        else:
            z = z / zsum
        A_n = np.fft.rfft(a, Na)
        W = np.fft.rfft(z[::-1], Na)
        conv = np.fft.irfft(A_n * W, Na)
        G = 2.0 * conv[L - n:L][::-1]
        G = np.where(np.isfinite(G), G, 0.0)
        gdot = float(np.dot(a, G))
        grad = a * (G - gdot)
        if not np.all(np.isfinite(grad)):
            break
        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        b1p *= b1
        b2p *= b2
        mb = m / (1.0 - b1p)
        vb = v / (1.0 - b2p)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        y -= np.clip(step, -0.12, 0.12)
        y -= np.mean(y)
        y = np.clip(y, -30, 30)
    return best, best_score


def local_refine(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_score = score_fft(best, N)
    lo = max(1, n // 600)
    hi = max(lo + 1, n // 16)
    while time.monotonic() < deadline:
        width = int(rng.integers(lo, hi + 1))
        start = int(rng.integers(0, n - width + 1))
        cand = best.copy()
        factor = float(np.exp(rng.normal(0, 0.08)))
        cand[start:start + width] *= factor
        if rng.random() < 0.3:
            block = cand[start:start + width].copy()
            block.reverse()
            cand[start:start + width] = block
        cand = project_positive(cand, best)
        sc = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return best, best_score


def exact_score(a):
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return np.inf


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    headroom = max(10.0, 0.05 * budget)
    deadline = start + max(0.0, budget - headroom)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        parent = np.ones(5582, dtype=np.float64)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(5582, dtype=np.float64)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 0.01:
        parent = np.ones(max(parent.size, 5582), dtype=np.float64)

    n = parent.size
    ones = np.ones(n, dtype=np.float64) / n
    Ns = next_pow2(2 * n - 1)
    Na = next_pow2(3 * n - 2)
    Na = max(Na, Ns)

    pool = [ones.copy(), project_positive(parent, ones)]
    for alpha in (0.7, 0.5, 0.35):
        A0 = np.fft.rfft(ones, Ns)
        c0 = np.fft.irfft(A0 * A0, Ns)[:2 * n - 1]
        c0 = np.maximum(np.where(np.isfinite(c0), c0, 0.0), 0.0)
        cs = float(c0.sum())
        if cs > 0:
            cn = c0 / cs
            ct = np.power(cn + 1e-18, alpha)
            ct /= max(ct.sum(), 1e-30)
            T = np.fft.rfft(ct, Ns)
            a2 = np.fft.irfft(np.sqrt(np.maximum(T, 0)), Ns)[:n]
            a2 = np.maximum(a2, 0.0)
            pool.append(project_positive(a2, ones))

    best_a = ones.copy()
    best_score = np.inf
    for cand in pool:
        sc = score_fft(cand, Ns)
        if np.isfinite(sc) and sc < best_score:
            best_a = cand.copy()
            best_score = sc

    rem = deadline - time.monotonic()
    if rem > 30:
        spec_dl = time.monotonic() + max(10.0, 0.22 * rem)
        try:
            sa, ss = run_spectral(best_a, n, Ns, spec_dl, seed + 1)
            if np.isfinite(ss) and ss < best_score:
                best_a = sa.copy()
                best_score = ss
            pool.append(sa)
        except Exception:
            pass

    rem = deadline - time.monotonic()
    if rem > 40:
        ad_dl = time.monotonic() + max(20.0, 0.52 * rem)
        try:
            y0 = np.log(np.maximum(best_a, 1e-300))
            y0 = y0 - float(np.mean(y0))
            y0 = np.clip(y0, -30, 30)
            ya, ys = run_adam(y0, n, Ns, Na, ad_dl, seed + 2)
            if np.isfinite(ys) and ys < best_score:
                best_a = ya.copy()
                best_score = ys
            pool.append(ya)
        except Exception:
            pass

    rem = deadline - time.monotonic()
    if rem > 15:
        ref_dl = time.monotonic() + max(5.0, 0.25 * rem)
        try:
            ra, rs = local_refine(best_a, n, Ns, ref_dl, seed + 3)
            if np.isfinite(rs) and rs < best_score:
                best_a = ra.copy()
                best_score = rs
            pool.append(ra)
        except Exception:
            pass

    best_exact_a = best_a.copy()
    best_exact = float(best_score)
    for cand in pool:
        if time.monotonic() > deadline - 5:
            break
        if cand is None or cand.size != n:
            continue
        sc = exact_score(cand)
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_exact_a = cand.copy()

    result = project_positive(best_exact_a, ones)
    if result.size == 0:
        result = ones.copy()
    return [float(x) for x in result.tolist()]
