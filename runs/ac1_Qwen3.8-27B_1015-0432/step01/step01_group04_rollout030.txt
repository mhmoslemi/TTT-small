    if a.size != fb.size:
        a = fb.copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = fb.copy()
        s = float(a.sum())
    if not np.isfinite(s) or s <= 0:
        a = np.ones(max(1, a.size))
    return a / max(float(a.sum()), 0.01)

def score_fft(a, N):
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))

def spec_step(a, N, alpha, mix, phase):
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = np.where(np.isfinite(c[:L]), c[:L], 0.0)
    c = np.maximum(c, 0.0)
    cs = c.sum()
    if cs <= 0:
        return a
    c = c / cs
    ct = np.power(c + 1e-20, alpha)
    sct = ct.sum()
    if sct <= 0:
        return a
    ct /= sct
    T = np.fft.rfft(ct, N)
    T = np.maximum(np.where(np.isfinite(T), T, 0.0), 0.0)
    A2 = np.sqrt(T)
    if phase:
        A2 = A2 * (A / (np.abs(A) + 1e-30))
    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.maximum(np.where(np.isfinite(a2), a2, 0.0), 0.0)
    s2 = a2.sum()
    if s2 <= 0:
        return a
    return project_positive((1 - mix) * a / a.sum() + mix * a2 / s2, a)

def run_spectral(a, N, deadline, seed):
    best = a.copy()
    best_sc = score_fft(a, N)
    cur = a.copy()
    cur_sc = best_sc
    it = 0
    stall = 0
    rng = np.random.default_rng(seed)
    n = a.size
    while time.monotonic() < deadline:
        alpha = max(0.3, 0.9 - 0.6 * min(1.0, it / 300.0))
        mix = max(0.1, 0.6 - 0.4 * min(1.0, it / 200.0))
        for ph in (1, 0):
            cand = spec_step(cur, N, alpha, mix, ph == 1)
            sc = score_fft(cand, N)
            if np.isfinite(sc) and sc < best_sc:
                best = cand.copy()
                best_sc = sc
                stall = 0
            elif np.isfinite(sc) and sc <= cur_sc * 1.02:
                cur = cand
                cur_sc = sc
                stall += 1
            else:
                stall += 1
        it += 1
        if stall > 50:
            cur = project_positive(best * (1 + 0.05 * rng.standard_normal(n)), best)
            cur_sc = score_fft(cur, N)
            stall = 0
    return best, best_sc

def run_adam(a, N, deadline, seed):
    n = a.size
    L = 2 * n - 1
    best = a.copy()
    best_sc = score_fft(a, N)
    y = np.log(np.maximum(a, 1e-30))
    y -= y.mean()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    b1p, b2p = 1.0, 1.0
    t = 0
    lr = 0.05
    stall = 0
    rng = np.random.default_rng(seed)
    while time.monotonic() < deadline:
        a_c = project_positive(np.exp(np.clip(y, -30, 30)), best)
        sc = score_fft(a_c, N)
        if np.isfinite(sc) and sc < best_sc:
            best = a_c.copy()
            best_sc = sc
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            y = np.log(np.maximum(best, 1e-30))
            y -= y.mean()
            m.fill(0)
            v.fill(0)
            b1p, b2p = 1.0, 1.0
            t = 0
            lr = max(0.005, lr * 0.5)
            stall = 0
            continue
        A = np.fft.rfft(a_c, N)
        c = np.fft.irfft(A * A, N)[:L]
        c = np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)
        cm = c.max()
        if cm <= 0:
            break
        aw = 8.0 + 40.0 * min(1.0, t / 5000.0)
        z = np.exp(aw * (c - cm))
        zs = z.sum()
        if not np.isfinite(zs) or zs <= 0:
            z = np.zeros(L)
            z[int(np.argmax(c))] = 1.0
        else:
            z /= zs
        zr = z[::-1]
        W = np.fft.rfft(zr, N)
        conv = np.fft.irfft(A * W, N)
        G = 2.0 * conv[L - n:L][::-1]
        G = np.where(np.isfinite(G), G, 0.0)
        t += 1
        m = b1 * m + (1 - b1) * G
        v = b2 * v + (1 - b2) * G * G
        b1p *= b1
        b2p *= b2
        mb = m / (1 - b1p)
        vb = v / (1 - b2p)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.05, 0.05)
        y -= step
        y -= y.mean()
        y = np.clip(y, -30, 30)
    return best, best_sc

def local_search(a, N, deadline, seed):
    n = a.size
    best = a.copy()
    best_sc = score_fft(a, N)
    rng = np.random.default_rng(seed)
    lo = max(1, n // 400)
    hi = max(lo + 1, n // 6)
    while time.monotonic() < deadline:
        w = int(rng.integers(lo, hi + 1))
        s = int(rng.integers(0, n - w + 1))
        c = best.copy()
        c[s:s+w] *= float(np.exp(rng.normal(0, 0.12)))
        if rng.random() < 0.2:
            c[s:s+w] = c[s:s+w][::-1]
        c = project_positive(c, best)
        sc = score_fft(c, N)
        if np.isfinite(sc) and sc < best_sc:
            best = c
            best_sc = sc
    return best, best_sc
