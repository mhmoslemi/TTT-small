import numpy as np
import time
import math
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _smooth_max_val(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * math.log(s)


def _smooth_max_probs(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * math.log(s), e / s


def _conv_fft(a, L):
    fa = fft(a, L)
    return ifft(fa * fa).real


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10
    L = _next_pow2(2 * n - 1)
    cv = _conv_fft(a, L)[:2 * n - 1]
    mv, _ = _smooth_max_probs(cv, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    L = _next_pow2(2 * n - 1)
    cv = _conv_fft(a, L)[:2 * n - 1]
    mv, p = _smooth_max_probs(cv, tau)
    pr = p[::-1]
    L2 = _next_pow2(n + 2 * n - 1 - 1)  # need length >= n + (2n-1) - 1 = 3n-2
    # Actually for conv of a (len n) and pr (len 2n-1), need L >= n + 2n-1 - 1 = 3n-2
    fa = fft(a, L2)
    fp = fft(pr, L2)
    h = ifft(fa * fp).real
    # h[i] for i = 0..L2-1, we need h[n-1 .. 2n-2] reversed
    g = 2.0 * h[n - 1:2 * n - 1][::-1]
    coef1 = 2.0 * n / (S * S)
    coef2 = 4.0 * n * mv / (S ** 3)
    return coef1 * g - coef2

import numpy as np
import time
import math
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _npow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _smax(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * math.log(s), e / s


def _sobj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10
    L = _npow2(2 * n - 1)
    fa = fft(a, L)
    cv = ifft(fa * fa).real[:2 * n - 1]
    mv, _ = _smax(cv, tau)
    return 2.0 * n * mv / (S * S)


def _sgrad(a, tau):
    n = a.size
    S = a.sum()
    L1 = _npow2(2 * n - 1)
    fa = fft(a, L1)
    cv = ifft(fa * fa).real[:2 * n - 1]
    mv, p = _smax(cv, tau)
    pr = p[::-1]
    L2 = _npow2(3 * n - 2)
    h = ifft(fft(a, L2) * fft(pr, L2)).real
    g = 2.0 * h[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _local_search(bv, sc, rng, n, dl):
    step = 0
    while time.monotonic() < dl:
        step += 1
        # Choose perturbation type

        cd = bv.copy()
        if r < 0.35:

            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] *= np.exp(float(rng.normal(0.0, 0.03)))
        elif r < 0.6:
            # Additive perturbation
            w = max(1, n // int(rng.integers(4, 80)))
            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] += float(rng.normal(0.0, 0.01 * bv[s:s + w].max() + 0.001))
        elif r < 0.8:

            F = fft(cd, L)
            k = int(rng.integers(0, n // 2))
            dk = int(max(1, rng.integers(0, n // 50)))
            F[k:k+dk] += complex(float(rng.normal(0, 0.1)), float(rng.normal(0, 0.1)))
            # Maintain conjugate symmetry
            F[(-k):(-k+dk)] = np.conj(F[k:k+dk])
            cd = ifft(F).real
        else:
            # Global scaling
            cd *= float(np.exp(rng.normal(0.0, 0.02)))

        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc:
