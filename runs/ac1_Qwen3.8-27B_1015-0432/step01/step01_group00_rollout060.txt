    f = f.astype(np.float64).copy()

    beta = 40.0
    eta = 0.06
    stall = 0
    for it in range(400000):
        if time.monotonic() > dl:
            break

        M = b.max()
        s2 = f.sum()
        cur = 2.0 * n * M / s2 / s2

            best_sc = cur
            best = f.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            p = 0.05 + 0.5 * rng.random()
