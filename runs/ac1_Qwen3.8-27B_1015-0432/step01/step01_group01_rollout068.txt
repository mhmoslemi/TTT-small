import numpy as np
import time


def fft_autocorr(a, fsize):
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)


def subgrad(a, conv, k):
    n = a.shape[0]
    idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros(n)
    for m in idx:
        m = int(m)
        lo = max(0, m - n + 1)
        hi = min(m, n - 1)
        if lo <= hi:
            g[lo:hi + 1] += a[m - hi:m - lo + 1][::-1]
    return g / float(k)


def proj(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return None
    return a / s
