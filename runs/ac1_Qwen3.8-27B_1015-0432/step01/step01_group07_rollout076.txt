def _fft_len(n):

import numpy as np
import time


def _fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()


def _fan(f, L):
    F = np.fft.rfft(f, L)
    return np.fft.irfft(F * F, L)[: len(f) * 2 - 1]


def _score(f, L):
    n = len(f)
    s = f.sum()
    if s <= 0.0:
        return np.inf
    fan = _fan(f, L)
    return float(2.0 * n * fan.max() / (s * s))


def _normalize(f, n):
    s = f.sum()
    if s <= 1e-12:
        return np.ones(n)
    return np.clip(f / s * n, 0.0, 1000.0)


def _init_from_parent(n, rng):
    g = globals().get("height_sequence_1", None)
    if g is not None:
        try:
            p = np.asarray(g, dtype=np.float64)
        except Exception:
            p = None

            p = np.clip(p, 0.0, 1000.0)
            if p.sum() > 1e-9:
                return p
    t = np.linspace(0.0, 1.0, n)
    base = 1.0 + 0.5 * np.cos(2.0 * np.pi * t)
    base = base + rng.normal(0.0, 0.05, n)
    return np.clip(base, 0.0, 1000.0)


def _descend(f, L, t0, deadline, rng):

        S = f.sum()
