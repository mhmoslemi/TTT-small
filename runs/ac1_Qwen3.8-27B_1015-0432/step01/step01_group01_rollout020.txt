        s0 = float(n)
    fs = 1 << (2 * n - 1).bit_length()

    best = np.clip(parent, 0.0, 1000.0).copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best_score = 999.0

    # Phase 1: LP-based optimization (K=100, t=0.5)
    a = parent / s0
    K1, t1 = 100, 0.5
    stall = 0
    prev_M = np.inf
    for it in range(200000):
        if time.monotonic() > deadline - 350:
            break
        a, M = lp_step(a, fs, K1, t1)
        sc = score_of(a, s0, n, M)
        if M < prev_M - 1e-14:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall > 2000:
            break
        if it % 500 == 0:
            cand = np.clip(a * s0, 0.0, 1000.0)
            if cand.sum() >= 0.01:
                sc_ex = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc_ex) and sc_ex < best_score:
                    best_score = sc_ex
                    best = cand.copy()

    # Phase 2: LP with more lags, smaller step
    K2, t2 = 200, 0.3
    stall = 0
    prev_M = np.inf
    for it in range(200000):
        if time.monotonic() > deadline - 200:
            break
        a, M = lp_step(a, fs, K2, t2)
        sc = score_of(a, s0, n, M)
        if M < prev_M - 1e-14:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall > 3000:
            break
        if it % 500 == 0:
            cand = np.clip(a * s0, 0.0, 1000.0)
            if cand.sum() >= 0.01:
                sc_ex = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc_ex) and sc_ex < best_score:
                    best_score = sc_ex
                    best = cand.copy()

    # Phase 3: Subgradient polish
    a = best / s0
    v = np.zeros(n)
    beta = 0.85
    lr = 0.4
    stall = 0
    prev_M = np.inf
    for it in range(300000):
        if time.monotonic() > deadline - 5:
            break
        a, M, v = subgrad_step(a, fs, 200, lr, v, beta)
        if M < prev_M - 1e-14:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall > 5000:
            break
        lr = max(0.05, lr * 0.99997)
        if it % 500 == 0:
            cand = np.clip(a * s0, 0.0, 1000.0)
            if cand.sum() >= 0.01:
                sc_ex = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc_ex) and sc_ex < best_score:
                    best_score = sc_ex
                    best = cand.copy()

    # Final: verify and return
    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
    final_score = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(final_score):
        out = np.clip(parent, 0.0, 1000.0)
    return [float(x) for x in out]
