        it += 1
    return best, best_sc


def local_search(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_sc, _ = score_fft(a, N)
    lo = max(1, n // 1000)
    hi = max(lo + 1, n // 16)
    while time.monotonic() < deadline:
        w = int(rng.integers(lo, hi + 1))
        st = int(rng.integers(0, n - w + 1))
        cand = best.copy()
        f = float(np.exp(rng.normal(0, 0.2)))
        cand[st:st + w] *= f
        cand = safe_clip(cand, best)
        cs = float(cand.sum())
        if cs > 0:
            cand = cand / cs
        sc, _ = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
    return best, best_sc


def make_inits(n, seed, parent, N):
    rng = np.random.default_rng(seed)
    ones = np.ones(n) / n
    out = [ones.copy(), safe_clip(parent, ones).copy()]
    x = np.linspace(0, 1, n)
    for A in (1.0, 2.0, 4.0, 8.0):
        out.append(np.exp(A * x) / float(n))
        out.append(np.exp(A * (x - 1)) / float(n))
        out.append(np.exp(-A * (x - 0.5)) / float(n))
    for t in (0.3, 0.5, 0.7):
        v = np.ones(n)
        v[: int(n * t)] = t
        v[int(n * t):] = 1 - t
        out.append(v / v.sum())
    if n > 4:
        for _ in range(3):
            y = np.zeros(n)
            j = np.arange(n, dtype=np.float64)
            for kk in range(1, 8):
                y += np.cos(2 * np.pi * kk * j / n + rng.normal(0, 6.28)) / kk
            v = np.exp(0.5 * np.clip(y, -20, 20))
            out.append(v / v.sum())
    return out

    best_sc = score_fft(best_a, N)[0]

    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()

    return [float(x) for x in best_exact]
