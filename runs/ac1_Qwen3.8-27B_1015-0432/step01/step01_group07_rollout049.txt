            sc = 2.0 * n * float(np.max(_fft_conv(a_new))) / (S_t * S_t)
            if np.isfinite(sc) and sc < cur_score - 1e-12:
                a = a_new
                cur_score = sc
                found = True
                break

        if found:
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
            stall = 0
            lr = min(lr * 1.04, 20.0)
        else:
            stall += 1
            lr *= 0.75
            if stall > 40:
                a = np.clip(a + rng.normal(0, 0.02 * S_t / n, n), 0, 1000)
                S_p = float(np.sum(a))
                a = a / max(S_p, 0.01) * S_t
                stall = 0
                lr = 0.3
                t = max(0.05, t * 0.4)

        t = max(0.005, t * 0.9997)

        if time.monotonic() - t0 > 3.0:
            break

    # Polish with exact subgradient
    for _ in range(300):
        if time.monotonic() > deadline - 15.0:
            break
        g, M = _subgrad(a)
        S = float(np.sum(a))
        cur_sc = 2.0 * n * M / (S * S)
        found = False
        for tr in (0.02, 0.01, 0.005, 0.002, 0.04):
            a_new = np.clip(a - tr * g, 0.0, 1000.0)
            S_new = float(np.sum(a_new))
            if S_new < 0.01:
                continue
            a_new = a_new / S_new * S_t
            sc = 2.0 * n * float(np.max(_fft_conv(a_new))) / (S_t * S_t)
            if np.isfinite(sc) and sc < cur_sc - 1e-12:
                a = a_new
                cur_sc = sc
                found = True
                break
        if found and cur_sc < best_score:
            best = a.copy()
            best_score = cur_sc
        elif not found:
            continue

    # Final validation and fallback
    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        best = np.ones(n, dtype=np.float64)
    elif final > 2.0:
        psc = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(psc) and psc < final:
            best = parent

    out = [float(x) for x in best]
    return out
