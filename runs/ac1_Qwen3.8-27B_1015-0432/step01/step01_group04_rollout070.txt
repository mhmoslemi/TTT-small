def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def proj_simplex(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def softmax_vec(y):
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - y.max()
    e = np.exp(np.clip(y, -700.0, 700.0))
    s = float(e.sum())
    if s < 1e-30:
        return np.ones(y.size, dtype=np.float64) / y.size
    return e / s


def autocorr(a, N):
    A = np.fft.rfft(a, N)
    L = 2 * a.size - 1
    c = np.fft.irfft(A * A, N)[:L]
    return np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)


def score_norm(a, N):
    n = a.size
    c = autocorr(a, N)
    return 2.0 * n * float(c.max())

def fft_grad_a(w, a, n, Ng):
    A = np.fft.rfft(a, Ng)
    W = np.fft.rfft(w, Ng)
    conv_aw = np.fft.irfft(A * W, Ng)
    Wr = np.fft.rfft(w[::-1], Ng)
    conv_awr = np.fft.irfft(A * Wr, Ng)
    t1 = conv_awr[n - 1:2 * n - 1][::-1]
    t2 = conv_aw[:n]
    return t1 + t2
