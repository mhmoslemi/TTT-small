import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s

def make_init(n, kind, rng=None):
    if kind == 'uniform':
        return np.ones(n) / n
    if kind == 'hann':
        x = np.arange(n, dtype=np.float64)
        return normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1))))
    if kind.startswith('pow'):
        p = float(kind[3:])
        x = np.arange(n, dtype=np.float64)
        d = np.minimum(x, n - 1.0 - x) + 1.0
        return normalize(d ** (-p))
    if kind == 'bilateral':
        x = np.arange(n, dtype=np.float64)
        c = (n - 1) / 2.0
        return normalize(1.0 / (1.0 + ((x - c) / (n / 6.0)) ** 4))
    if kind == 'gauss':
        x = np.arange(n, dtype=np.float64)
        c = (n - 1) / 2.0
        return normalize(np.exp(-2.0 * ((x - c) / (n / 4.0)) ** 2))
    if kind == 'cos_bump':
        x = np.arange(n, dtype=np.float64)
        w = np.pi * np.minimum(x, n - 1 - x) / (n / 2)
        return normalize(np.maximum(np.cos(w), 1e-10))
    if kind == 'rand_sym':
        u = 1.0 + 0.5 * rng.standard_normal(n)
        u = np.maximum(u, 0.05)
        u = 0.5 * (u + u[::-1])
        return normalize(u)
    if kind == 'rand_asym':
        u = 1.0 + 0.4 * rng.standard_normal(n)
