def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def score_norm(f):
    """Score for sum(f)=1: returns 2n*max(conv(f,f))"""
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max()


def grad_direction(f, b, beta_n):
    """Compute multiplicative gradient direction using top-lag emphasis."""
    M = b.max()
    if M < 1e-30:
        return None

    w = np.exp(beta_n * (b / M - 1.0))
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        top_k = np.argsort(b)[-max(1, b.size // 100):]
        w[top_k] = 1.0

    gc = fft_conv(w, f_r)
    grad = gc[f.size - 1:2 * f.size - 1].copy()
