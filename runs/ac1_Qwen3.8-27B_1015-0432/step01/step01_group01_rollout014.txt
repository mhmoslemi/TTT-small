def fft_conv(a, fs):

    fa = np.fft.rfft(a, n=fs)

def lag_grad(a, conv, k, n):
    """Gradient contribution of a single lag k."""
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return None
    return a[k - hi:k - lo + 1][::-1], lo, hi + 1

def topk_subgradient(a, conv, tk, n):
    """Subgradient using top-k lags."""
    idx = np.argpartition(conv, -tk)[-tk:]
    g = np.zeros(n)
    for k in idx:
        r = lag_grad(a, conv, int(k), n)
        if r:
            seg, lo, hi = r
            g[lo:hi] += seg
    g /= tk
    return g

def softmax_subgradient(a, conv, n, alpha, max_lags=256):
    """Smoothed subgradient using softmax over lags."""
    c = conv - conv.max()
    w = np.exp(alpha * c)
    total = w.sum()
    if total < 1e-30:
        w = np.ones_like(conv) / len(conv)
    else:

    # Use only top lags

        top_idx = np.argpartition(w, -max_lags)[-max_lags:]
        mask = np.zeros(len(conv), dtype=bool)

        w = np.where(mask, w, 0.0)

            g[lo:hi] += w[k] * seg
    return g

import time


def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def compute_grad_topk(a, conv, tk, n):
    idx = np.argpartition(conv, -tk)[-tk:]
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g /= float(tk)
    return g


def compute_grad_softmax(a, conv, n, alpha, max_lags):
    c = conv - conv.max()
    w = np.exp(alpha * c)
    s = w.sum()
    if s < 1e-30:
        w = np.ones(len(conv)) / len(conv)
    else:
        w /= s
    if len(conv) > max_lags:
        top = np.argpartition(w, -max_lags)[-max_lags:]
        mask = np.zeros(len(conv), dtype=bool)

        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:

    return g
