def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _grad(a, conv, K):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return None
    M = float(np.max(conv))
    k_idx = np.argsort(conv)[::-1][:K]
    vals = conv[k_idx]
    temp = M / max(K, 1)
    w = np.exp((vals - vals[0]) / temp)
    w /= w.sum()
    dconv = np.zeros(n)
    for wi, ki in zip(w, k_idx):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)

    M_app = float(np.dot(w, vals))
    g = (2.0 * n / (S * S)) * dconv - 4.0 * n * M_app / (S * S * S)
    return g
