def _topk_grad(a, b, K, p, n):

    top_idx = np.argpartition(b, -K)[-K:]  # last K are largest (approximately)
    top_b = b[top_idx]

    w_full = np.zeros(2*n-1)
    w_full[top_idx] = top_b ** p
    w_full /= w_full.sum()
    # Compute 4n * sum_k w_full[k] * a[k-j]
    # Same as before but with sparse w_full
    # Since w_full has at most K nonzero entries, we can compute directly:
    grad = np.zeros(n)
    for k in top_idx:
        wk = w_full[k]
        # grad[j] += 4n * wk * a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            jj = np.arange(js, je + 1)
            grad[jj] += 4.0 * n * wk * a[k - jj]
    return grad

    w = np.zeros(len(b2n))
    w[idx] = top_b ** p
    tw = w.sum()
    if tw < 1e-300:
        return np.zeros(n)
    w /= tw
    grad = np.zeros(n)
    for k in idx:
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            jj = np.arange(js, je + 1)
            grad[jj] += wk * a[k - jj]
    return 4.0 * n * grad


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n < 2:
        return [1.0, 1.0]
    S0 = float(parent.sum())
    a = parent / S0

    def _sc(x):
        b = _fft_conv(x)
        S = float(np.sum(x))
        if S < 1e-10:
            return np.inf
        return float(2 * n * np.max(b) / (S * S))

    # Try structured initializations
    idx = np.arange(n, dtype=np.float64)
    cands = [a]
    cands.append(np.abs(2.0 * idx / (n - 1) - 1.0))
    cands.append((idx + 1.0) * (n - idx))
    cands.append(0.5 * (1.0 + np.cos(np.pi * idx / max(n - 1, 1))))
    for fp in (3, 7, 11, 17, 23):
        cands.append(np.abs(np.sin(np.pi * fp * idx / n)))
    best = a.copy()
    best_sc = _sc(a)
    for c in cands:
        c = np.maximum(c, 0.0)
        cs = float(c.sum())
        if cs < 1e-10:
            continue
        c = c / cs
        sc_c = _sc(c)
        if sc_c < best_sc:
            best = c.copy()
            best_sc = sc_c
    a = best.copy()

    # Also try blending parent with structured
    for c in cands:
        c = np.maximum(c, 0.0)
        cs = float(c.sum())
        if cs < 1e-10:
            continue
        c = c / cs
        for mix in (0.3, 0.5, 0.7):
            ab = (1 - mix) * a + mix * c
            ab = np.maximum(ab, 0.0)
            ab_s = float(ab.sum())
            if ab_s < 1e-10:
                continue
            ab = ab / ab_s
            sc_ab = _sc(ab)
            if sc_ab < best_sc:
                best = ab.copy()
                best_sc = sc_ab
    a = best.copy()

    tau_K = 20
    tau_p = 2.5
    eta = 1.0
    no_improve = 0
    restart_count = 0

    while time.monotonic() < deadline:
        b = _fft_conv(a)
        M = float(b.max())
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)

        # Adapt K: more lags when peak is flat
        sort_b = np.sort(b)[::-1]
        gap = sort_b[0] - sort_b[min(199, len(b) - 1)]
        if gap < M * 0.03:
            tau_K = min(tau_K + 5, 100)
            tau_p = max(tau_p - 0.3, 1.0)
        elif gap > M * 0.2:
            tau_K = max(tau_K - 3, 10)
            tau_p = min(tau_p + 0.2, 4.0)

        g = _topk_grad(a, b, tau_K, tau_p)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-18:
            break

        # Line search
        best_trial = sc
        best_a_t = a.copy()
        best_te = 0.0
        for te in (eta, eta * 0.5, eta * 0.25, eta * 2.0, eta * 4.0):
            at = a - te * g
            at = np.maximum(at, 0.0)
            St = float(at.sum())
            if St < 1e-10:
                continue
            at = at / St
            st = _sc(at)
            if st < best_trial:
                best_trial = st
                best_a_t = at.copy()
                best_te = te

        if best_te > 0:
            a = best_a_t
            eta = best_te * 1.04
            no_improve = 0
            if best_trial < best_sc:
                best = a.copy()
                best_sc = best_trial
        else:
            no_improve += 1
            eta *= 0.75
            if no_improve > 40:
                restart_count += 1
                if restart_count > 25:
                    break
                a = best + rng.normal(0.0, 0.0003 * best.mean(), n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                eta = 1.0
                no_improve = 0
                tau_K = 20
                tau_p = 2.5

    # Final 2-coordinate refinement
    if time.monotonic() < deadline - 15.0:
        b = _fft_conv(best)
        peak_lags = np.argpartition(b, -10)[-10:]
        step_base = best.mean() * 0.005
        remaining = deadline - 15.0 - time.monotonic()
        max_refine = int(min(500, remaining / 0.02))
        for _ in range(max_refine):
            if time.monotonic() > deadline - 15.0:
                break
            j = int(rng.integers(0, n))
            k_lag = int(peak_lags[int(rng.integers(0, len(peak_lags)))])
            m = k_lag - j
            if m < 0 or m >= n or m == j:
                continue
            for sgn in (1, -1):
                s_val = sgn * step_base
                old_j, old_m = best[j], best[m]
                best[j] = max(0.0, old_j + s_val)
                best[m] = max(0.0, old_m - s_val)
                if best[j] < 1e-15 and best[m] < 1e-15:
                    best[j], best[m] = old_j, old_m
                    continue
                sc_r = _sc(best)
                if sc_r < best_sc:
                    best_sc = sc_r
                else:
                    best[j], best[m] = old_j, old_m
        b = _fft_conv(best)
        peak_lags = np.argpartition(b, -10)[-10:]

    # Return best at original scale
    result = [float(x) for x in best]
    try:
        ev = evaluate_sequence(result)
        if not np.isfinite(ev):
            result = [float(x) for x in parent]
    except Exception:
        result = [float(x) for x in parent]
    return result
