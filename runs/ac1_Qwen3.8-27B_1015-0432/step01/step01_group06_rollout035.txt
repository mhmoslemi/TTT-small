def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score(corr, n):

    return s  # We'll divide by (sum_a)^2 outside, but since we normalize sum=n, it's s/n^2

def _subgrad(corr, a, n, K):
    # Active set: top K positions in corr
    ntop = min(K, 2 * n - 1)
    idx = np.argpartition(corr, -ntop)[-ntop:]
    thresh = corr[idx].min()
    active = idx[corr[idx] >= thresh]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # d(a*a)_k / da_j = 2*a_{k-j} for j in [lo, hi]

        g[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi + 1)]

                best = a.copy()
                stuck = 0

        y = a + beta * (a - prev)
        prev = a.copy()
        y = np.clip(y, 0.0, 1000.0)
        ys = y.sum()
        y = y * n / ys if ys > 1e-10 else np.ones(n)

        corr_y = _fft_conv(y)
        g = _subgrad(corr_y, y, n, K)

        v = beta * v - lr * g
        step = y + v
        a = _project(step, n)

        corr_a = _fft_conv(a)
        sc_a = 2.0 * n * corr_a.max() / (n * n)
        if sc_a < sc + 1e-14:
            lr = min(lr * 1.015, 200.0)
            beta = min(beta * 1.0005, 0.94)
        else:
            lr = max(lr * 0.97, 0.005)
            beta = max(beta * 0.998, 0.4)
            v *= 0.5

        if stuck > 600:
            a = _gen(n, rng, int(rng.integers(0, 8)))
            v[:] = 0.0
            prev = a.copy()
            beta = 0.82
            lr = 2.0
            stuck = 0
        elif stuck > 250:
            noise = rng.normal(0.0, 0.04 * max(a.std(), 0.05), n)
            a = _project(a + noise, n)
            v[:] = 0.0
            prev = a.copy()
            stuck = 0

    out = np.clip(best, 0.0, 1000.0)
    s = out.sum()
    if s < 0.01:
        out = np.ones(n)
    return [float(x) for x in out]
