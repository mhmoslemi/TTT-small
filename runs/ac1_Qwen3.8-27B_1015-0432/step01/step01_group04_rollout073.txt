    pool.append(_proj(0.5 + 0.5 * np.cos(2.0 * np.pi * x), flat))
    pool.append(_proj(0.5 + 0.5 * np.sin(np.pi * x) ** 2, flat))
    j = np.arange(n, dtype=np.float64)
    for kk in (1, 3):
        pool.append(_proj(1.0 + 0.4 * np.cos(2.0 * np.pi * kk * j / n), flat))

    best_a = flat.copy()
    best_sc = _score(best_a, N)
    for p in pool:
        if p.size == n:
            sc = _score(p, N)
            if np.isfinite(sc) and sc < best_sc:
                best_a = p.copy()
                best_sc = sc

    rem1 = deadline - time.monotonic()
    if rem1 > 30.0:
        d1 = time.monotonic() + min(250.0, 0.30 * rem1)
        try:
            sa, ss = _spectral_phase(best_a, n, N, d1, seed + 1)
            if np.isfinite(ss) and ss < best_sc:
                best_a = sa.copy()
                best_sc = ss
            pool.append(sa.copy())
        except Exception:
            pass
    best_a = _proj(best_a, flat)

    rem2 = deadline - time.monotonic()
    if rem2 > 30.0:
        d2 = time.monotonic() + min(600.0, 0.50 * rem2)
        try:
            aa, as_ = _adam_phase(best_a, n, N, d2, seed + 2)
            if np.isfinite(as_) and as_ < best_sc:
                best_a = aa.copy()
                best_sc = as_
            pool.append(aa.copy())
        except Exception:
            pass

    rem3 = deadline - time.monotonic()
    if rem3 > 20.0:
        d3 = time.monotonic() + min(150.0, 0.25 * rem3)
        try:
            ra, rs = _local_search(best_a, n, N, d3, seed + 3)
            if np.isfinite(rs) and rs < best_sc:
                best_a = ra.copy()
                best_sc = rs
            pool.append(ra.copy())
        except Exception:
            pass

    rem4 = deadline - time.monotonic()
    if rem4 > 25.0:
        d4 = time.monotonic() + min(200.0, 0.40 * rem4)
        try:
            aa2, as2 = _adam_phase(best_a, n, N, d4, seed + 4)
            if np.isfinite(as2) and as2 < best_sc:
                best_a = aa2.copy()
                best_sc = as2
            pool.append(aa2.copy())
        except Exception:
            pass

    best_exact = flat.copy()
    be_best = 1e18
    for p in pool:
        if p is None or p.size != n:
            continue
        if time.monotonic() > deadline - 5.0:
            break
        try:
            es = float(evaluate_sequence(p.tolist()))
        except Exception:
            es = 1e18
        if np.isfinite(es) and es < be_best:
            be_best = es
            best_exact = p.copy()

    if not np.all(np.isfinite(best_exact)) or float(best_exact.sum()) < 0.01:
        best_exact = flat

    return [float(x) for x in best_exact.tolist()]
