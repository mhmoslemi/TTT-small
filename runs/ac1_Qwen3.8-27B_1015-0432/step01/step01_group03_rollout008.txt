import time
from scipy.optimize import minimize


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smooth_max(np.convolve(a, a), tau)
    return 2.0 * n * mv / (S * S)
