Actually, for the LP:

- M = max(convolve(a,a))
- Solve: max sum(b) s.t. convolve(a,b) <= M, b >= 0
- The optimal b* has sum(b*) >= sum(a) (since a itself is feasible: convolve(a,a) <= M)
- Rescale: g = b* * sum(a) / sum(b*)
- Update: a = (1-t)*a + t*g

```python
import scipy.sparse as sp
import scipy.optimize as opt

def _solve_lp(a, M):
    n = len(a)
    # Build sparse constraint matrix: (2n-1) x n
    # T[k, i] = a[i] if max(0, k-n+1) <= i <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[i])
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c = -np.ones(n)  # minimize -sum(b)
    A_ub = T
    b_ub = np.full(2*n-1, M)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

import numpy as np
import time

def _fft_conv(a, b):
    n = len(a)
    sb = len(b)
    sz = 1
    while sz < n + sb - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    fb = np.fft.rfft(b, sz)
    return np.fft.irfft(fa * fb, sz)[:n + sb - 1]

def _project_nn_sum(a, target_sum):
    # Project a onto {x >= 0, sum(x) = target_sum}
    n = len(a)
    if n == 0:
        return a
    # Water filling
    sorted_a = np.sort(a)[::-1]
    cumsum = np.cumsum(sorted_a)
    rho = np.arange(n, 0, -1)
    candidates = (cumsum - target_sum) / rho
    # Find the largest rho where candidate > sorted_a[rho-1]
    valid = candidates >= sorted_a
    if not np.any(valid):
        # All zero, spread evenly
        return np.full(n, target_sum / n)
    idx = np.max(np.where(valid)[0])
    alpha = candidates[idx]
    return np.maximum(a - alpha, 0.0)

def _subgrad_with_active(c, a, K):
    n = len(a)
    # Get top K indices of c
    if K >= len(c):
        active = np.arange(len(c))
    else:
        # Find top K
        active = np.argpartition(c, -K)[-K:]
    active = np.sort(active)
    
    g = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # grad_j c[k] = a[k-j] + a[j-k] for j in [0, n-1]

        # Actually for k in [0, 2n-2]:

        #   a[j-k] non-zero for j in [max(0,k), min(n-1,k+n-1)]
        # So:
        for j in range(max(0, k - n + 1), min(n, k + 1)):

            if 0 <= j - k < n:

    g /= len(active)
