import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s

    """One gradient step with exponential weighting and optional top-k focus."""
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return f, M
    
    # Exponential weighting
    w = np.exp(beta * b / M)
    wmax = w.max()
    if wmax > 0:
        w /= wmax
    
    # Optional top-k focus
    if k_top is not None and k_top < len(b):
        thr = np.partition(b, -k_top)[-k_top]
        w = np.where(b >= thr * 0.9, w * 10.0, w)
    
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n-1:2*n-1]
    grad = np.maximum(grad, 0.0)
    gmax = grad.max()
    if gmax < 1e-30:
        return f, M
    
    f_new = f * np.exp(-eta * grad / gmax)
    f_new = np.maximum(f_new, 1e-20)
    f_new /= f_new.sum()
    return f_new, M
