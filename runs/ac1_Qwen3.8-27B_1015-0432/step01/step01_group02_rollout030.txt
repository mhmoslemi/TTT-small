import numpy as np
import time


def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _grad_at_peaks(a, c, peak_idx, weights, n):
    h = np.zeros(n, dtype=np.float64)
    for ii in range(len(peak_idx)):
        kj = int(peak_idx[ii])
        wj = float(weights[ii])
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += wj * a[kj - np.arange(lo, hi + 1)]
    return h
