import time

def fft_autocorr(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def simplex_project(p):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(p)
    if p.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(p)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][0]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(p - theta, 0)

def topk_grad(a, corr, top_k):
    n = a.size
    vc = corr[:n]
    idx = np.argpartition(vc, -top_k)[-top_k:]
    idx = np.sort(idx)
    tk = len(idx)
    if tk == 0:
        return np.zeros(n)
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    g /= tk
    return g

    if s < 1e-15:

    return a / s

        M = float(corr.max())
        if M < best_M:

            stall = 0
        else:

        if stall > 3000:
            break
        g = topk_grad(a, corr, top_k)

        v -= v.mean() / v.sum() * 0  # just subtract mean for the constraint

        a = a - lr * v
        a = simplex_project(a)
        lr = max(0.05, lr * 0.9999)
        beta = min(0.95, beta + 0.000005)
    # Scale back to original sum
    return best_a * s0, best_a, best_M
