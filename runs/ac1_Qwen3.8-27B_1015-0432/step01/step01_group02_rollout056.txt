```python
    c = np.fft.irfft(F * F, Nconv).real
```

```python
import numpy as np
import time


def _fftconv(a):
    n = int(len(a))
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _pressure_fft(a, c, thresh_frac):
    n = int(len(a))
    max_c = float(c.max())
    if max_c < 1e-30:
        return np.zeros(n)
    thresh = thresh_frac * max_c
    w = np.zeros(len(c))
    mask = c > thresh
    w[mask] = (c[mask] - thresh) / (max_c - thresh + 1e-30)
    L = 4 * n
    a_full = np.fft.fft(a, L)
    w_pad = np.fft.fft(w, L)
    h_full = np.fft.ifft(np.conj(a_full) * w_pad).real
    return h_full[:n]


def _normalize(a):
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)
    n = len(height_sequence_1)
    t = np.linspace(0.0, 1.0, n)

    a = np.array(height_sequence_1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    a = _normalize(a)
    c = _fftconv(a)
    best_score = 2.0 * n * float(c.max())
    best_a = a.copy()

    inits = [
        np.maximum(np.sin(np.pi * t), 1e-12),
        np.maximum(np.sin(np.pi * t) ** 2, 1e-12),
        np.ones(n),
        np.maximum(1.0 - np.abs(2.0 * t - 1.0), 1e-12),
        np.maximum(np.cos(np.pi * t) ** 2, 1e-12),
        np.exp(-8.0 * (t - 0.5) ** 2),
        np.maximum(np.sin(2.0 * np.pi * t) ** 2 + 0.1, 1e-12),
    ]
    for ic in inits:
        ic = _normalize(ic)
        sc = 2.0 * n * float(_fftconv(ic).max())
        if sc < best_score:
            best_score = sc
            best_a = ic.copy()
            a = ic.copy()

    alpha = 0.10
    mom = 0.6
    h_mom = np.zeros(n)
    stuck = 0
    it = 0
    thresh_frac = 0.60

    while time.monotonic() < deadline:
        c = _fftconv(a)
        max_c = float(c.max())
        sc = 2.0 * n * max_c

        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        h = _pressure_fft(a, c, thresh_frac)
        h_mom = mom * h_mom + (1.0 - mom) * h
        hm = float(h_mom.max())
        if hm > 1e-14:
            a *= np.exp(-alpha * (h_mom / hm))
            a = _normalize(a)
        else:
            a = best_a.copy()

        it += 1

        if it % 120 == 0:
            f = rng.uniform(0.3, n / 3.0)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            eps = min(0.35, 0.04 + 0.008 * stuck)
            a *= 1.0 + eps * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
            a = _normalize(a)
            h_mom *= 0.5

        if stuck > 350:
            alpha *= 0.93
            mom = min(0.8, mom + 0.01)
            if alpha < 0.012:
                a = best_a * np.exp(rng.normal(0.0, 0.12, n))
                a = _normalize(a)
                h_mom = np.zeros(n)
                alpha = 0.10
                mom = 0.6
                stuck = 0

        if stuck > 3500:
            a = best_a.copy()
            k_mut = max(1, n // 6)
            idx = rng.integers(0, n, size=k_mut)
            a[idx] *= np.exp(rng.normal(0.0, 0.7, size=k_mut))
            a = _normalize(a)
            h_mom = np.zeros(n)
            stuck = 0
            alpha = 0.10
            mom = 0.6
            thresh_frac = rng.choice([0.5, 0.6, 0.7, 0.8])

    # Polishing: try small perturbations on best
    time_left = deadline - time.monotonic()
    if time_left > 5.0:
        trial = best_a.copy()
        for _ in range(50):
            if time.monotonic() > deadline:
                break
            kc = max(1, n // 20)
            idx = rng.integers(0, n, size=kc)
            trial = best_a.copy()
            trial[idx] *= np.exp(rng.normal(0.0, 0.05, size=kc))
            trial = _normalize(trial)
            tc = _fftconv(trial)
            tsc = 2.0 * n * float(tc.max())
            if tsc < best_score:
                best_score = tsc
                best_a = trial.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
        fs = evaluate_sequence(result)
    return result
```
