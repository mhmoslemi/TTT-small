    if S < 0.01:
        return 1e10
    conv = _self_conv(a)
    return float(2.0 * n * conv.max() / (S * S))


def _local_search(a, best_sc, rng, dl):
    n = a.size
    cur = a.copy()
    sc = best_sc
    w_med = max(10, n // 30)
    w_sml = max(3, n // 120)

    while time.monotonic() < dl:
        cd = cur.copy()
        r = rng.random()

        if r < 0.30:
            w = w_sml if rng.random() < 0.4 else w_med
            s = int(rng.integers(0, n - w))
            noise = rng.normal(0.0, 0.025, w)
            if w > 5:
                k = np.ones(3) / 3.0
                noise = np.convolve(noise, k, 'same')
            cd[s:s + w] *= np.exp(noise)
        elif r < 0.55:
            freq = int(rng.integers(1, 10))
            phase = rng.uniform(0.0, 2.0 * np.pi)
            xx = np.arange(n)
            amp = 0.012 + 0.008 * rng.random()
            shape = 1.0 + amp * np.cos(2.0 * np.pi * freq * xx / n + phase)
            cd *= shape
        elif r < 0.75:
            bw = max(5, n // 60)
            c1 = int(rng.integers(bw, n - bw))
            c2 = int(rng.integers(bw, n - bw))
            if abs(c1 - c2) < bw + 1:
                c2 = (c1 + n // 2) % max(1, n - 2 * bw) + bw
                c2 = min(max(c2, bw), n - bw)
            xx = np.linspace(-3.0, 3.0, bw)
            amp = 0.008 * max(cur.mean(), 1e-6) * rng.random()
            bump = amp * np.exp(-0.5 * xx * xx)
            cd[c1 - bw // 2:c1 + bw // 2 + 1] += bump
            cd[c2 - bw // 2:c2 + bw // 2 + 1] -= bump
        else:
            w = max(5, n // 50)
            half = n // 2 - w
            if half > 0:
                s = int(rng.integers(0, half))
                t = float(rng.uniform(0.05, 0.4))
                seg_a = cur[s:s + w].copy()
                seg_b = cur[n - 1 - s - w:n - 1 - s].copy()
                cd[s:s + w] = (1.0 - t) * seg_a + t * seg_b
                cd[n - 1 - s - w:n - 1 - s] = (1.0 - t) * seg_b + t * seg_a

        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() < 0.01:
            continue
        s2 = _fast_eval(cd)
        if s2 < sc - 1e-13:
            cur = cd
            sc = s2

    return cur, sc


def _grad_phase(a, tau, bounds, rem, maxiter_cap):
    if rem < 10.0:
        return a
    try:
        mi = max(20, min(maxiter_cap, int(rem / 8.0)))
        res = minimize(
            _obj_grad, a, args=(tau,),
            jac=True, method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': mi, 'gtol': 1e-10,
                     'maxls': 30, 'ftol': 1e-14})
        if res.x is not None:
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                return nx
    except Exception:
        pass
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 20.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
            raise ValueError
    except (NameError, ValueError):
        par = np.ones(256, dtype=np.float64)

    n = par.size
    ps = par.sum()
    if ps < 1e-8:
        par = np.ones(n)
        ps = float(n)

    a0 = par / ps
    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    cur = a0.copy()
    bounds = [(0.0, 1000.0)] * n

    # Phase 1: gradient with decreasing tau
    taus = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]
    for tv in taus:
        rem = dl - time.monotonic()
        if rem < 25.0:
            break
        cur = _grad_phase(cur, tv, bounds, rem, 200)
        fv = _fast_eval(cur)
        if fv < _fast_eval(best):
            sc = float(evaluate_sequence(cur.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = cur.copy()
                best_sc = sc

    # Phase 2: local search
    rem = dl - time.monotonic()
    if rem > 40.0:
        cur, sc = _local_search(cur, best_sc, rng, dl - 60.0)
        if sc < best_sc:
            best = cur.copy()
            best_sc = sc

    # Phase 3: fine gradient refinement
    rem = dl - time.monotonic()
    if rem > 25.0:
        cur = _grad_phase(cur, 0.005, bounds, rem, 100)
        sc = float(evaluate_sequence(cur.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = cur.copy()
            best_sc = sc

    # Phase 4: final local polish
    rem = dl - time.monotonic()
    if rem > 20.0:
        cur, sc = _local_search(cur, best_sc, rng, dl - 10.0)
        if sc < best_sc:
            best = cur.copy()
            best_sc = sc

    # Verify with exact evaluator
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n)
        final_sc = float(evaluate_sequence(best.tolist()))

    return [float(v) for v in best]
