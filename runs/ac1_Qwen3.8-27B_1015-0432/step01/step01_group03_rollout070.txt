import numpy as np
import time
from numpy.fft import rfft, irfft


def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m


def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1)
    idx = np.nonzero(u * cs > cssv - 1.0)
    if idx[0].size == 0:
        return np.full(n, 1.0 / n)
    rho = idx[0][0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _sm_val(a, tau, L):
    fa = rfft(a, L)
    c = irfft(fa * fa, L)[:2 * a.size - 1]
    cm = c.max()
    return cm + tau * np.log(np.exp((c - cm) / tau).sum())


def _sm_grad(a, tau, L1, L2):
    n = a.size
    fa = rfft(a, L1)
    c = irfft(fa * fa, L1)[:2 * n - 1]
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    p = e / s
    sm = cm + tau * np.log(s)
    fp = rfft(p, L2)
    fa2 = rfft(a, L2)
    R = irfft(fp * np.conj(fa2), L2)
    return sm, 2.0 * R[:n]


def _eval_fast(a, L1):
    n = a.size
    s = a.sum()
    if s < 1e-8:
        return np.inf
    fa = rfft(a, L1)
    c = irfft(fa * fa, L1)[:2 * n - 1]
    return 2.0 * n * c.max() / (s * s)


def _gd(a, tau, lr, iters, dl, L1, L2):
    a = a.copy().astype(np.float64)
    best_a = a.copy()
    best_sm = _sm_val(a, tau, L1)
    lr = float(max(lr, 1e-14))
    stall = 0
    for _ in range(iters):
        if time.monotonic() > dl:
            break
        sm, g = _sm_grad(a, tau, L1, L2)
        step = lr
        found = False
        for _b in range(14):
            a_new = _project_simplex(a - step * g)
            sm2 = _sm_val(a_new, tau, L1)
            if sm2 < sm:
                found = True
                break
            step *= 0.5
        if not found:
            break
        a = a_new
        sm = sm2
        if sm < best_sm - 1e-15:
            best_sm = sm
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 60:
            break
    return best_a


def _local_search(a, rng, dl, L1):
    n = a.size
    best = a.copy()
    best_sc = _eval_fast(best, L1)
    w = max(2, n // 40)
    count = 0
    while time.monotonic() < dl:
        a2 = best.copy()
        for _ in range(2):
            s = int(rng.integers(0, n - w))
            a2[s:s + w] *= np.exp(rng.normal(0.0, 0.025, w))
        a2 = _project_simplex(a2)
        sc = _eval_fast(a2, L1)
        if np.isfinite(sc) and sc < best_sc:
            best = a2.copy()
            best_sc = sc
            count = 0
        else:
            count += 1
        if count > 500:
            a2 = best.copy()
            s = int(rng.integers(0, n - w))
            a2[s:s + w] *= np.exp(rng.normal(0.0, 0.15, w))
            a2 = _project_simplex(a2)
            count = 0
    return best


def _try_init(rng, n, par):
    a = par / par.sum()
    a = a * np.exp(rng.normal(0.0, 0.3, n))
    a = _project_simplex(a)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 8.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256, dtype=np.float64)
    n = par.size
    ps = par.sum()
    if ps < 1e-8:
        par = np.ones(n)
        ps = float(n)

    L1 = _next_pow2(2 * n - 1)
    L2 = _next_pow2(3 * n)

    a0 = par / ps
    best = a0.copy()
    best_sc = _eval_fast(best, L1)

    starts = [a0.copy()]
    for i in range(3):
        starts.append(_try_init(rng, n, par))

    for a_start in starts:
        if time.monotonic() > dl - 120:
            break
        cur = a_start.copy()
        taus = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        for tau in taus:
            if time.monotonic() > dl - 100:
                break
            mx = max(float(cur.max()), 1e-10)
            lr = tau / (2.0 * n * mx + 1e-10)
            lr = min(lr, 1.0)
            rem = dl - time.monotonic() - 80
            iters = max(30, min(500, int(rem * 80)))
            cur = _gd(cur, tau, lr, iters, dl - 80, L1, L2)
            sc = _eval_fast(cur, L1)
            if np.isfinite(sc) and sc < best_sc:
                best = cur.copy()
                best_sc = sc
        sc_final = _eval_fast(best, L1)
        if not np.isfinite(sc_final):
            break

    if time.monotonic() < dl - 10:
        best = _local_search(best, rng, dl - 5.0, L1)

    result = [float(v) for v in best]
    return result
