def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def compute_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def compute_grad(f, beta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), b, M
    w = np.exp(beta * b / M)
    wsum = w.sum()
    w /= wsum
    h1 = fft_conv(f[::-1], w)[n - 1:2 * n - 1]
    h2 = fft_conv(w, f)[:n]
    return h1 + h2, b, M
