def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s
