This is a quadratic form. For the "soft" objective J = sum_k z[k] * b[k]:
∂J/∂a[j] = sum_k z[k] * ∂b[k]/∂a[j]

Now ∂b[k]/∂a[j]: b[k] involves a[j] in two places:
1. As a[i] when i = j (requires j >= 0, j <= n-1, and k-j >= 0, k-j <= n-1)
2. As a[k-i] when k-i = j, i.e., i = k-j (requires 0 <= k-j <= n-1)

Wait, that's only if the two appearances are distinct. If i = j AND k-i = j, then i = j and i = k-j, so j = k-j, k = 2j. In that case, a[j] appears squared: a[j]^2. The derivative is 2*a[j].

In general: ∂b[k]/∂a[j] = 2*a[k-j] if 0 <= k-j <= n-1, else 0.
(When k = 2j, both terms give a[j], so total is 2*a[j] = 2*a[k-j]. Same formula.)

So: ∂J/∂a[j] = 2 * sum_{k: 0<=k-j<=n-1} z[k] * a[k-j]
= 2 * sum_{m=0}^{n-1} z[j+m] * a[m]
= 2 * cross_correlation(z, a)[j]

Where cross_correlation(z, a)[j] = sum_m z[j+m] * a[m] (cross-correlation, not convolution).

Using FFT: pad a to length N (N >= 2n-1), pad z to length N.
A = rfft(a, N), Z = rfft(z, N).
Circular cross-correlation: r = ifft(A * conj(Z)) gives r[k] = sum_t a[t] * z[(t-k) mod N].
We want corr[j] = sum_m a[m] * z[j+m] = sum_t a[t] * z[(t+j) mod N] = r[(-j) mod N] = r[N-j] for j>0, r[0] for j=0.

So G[j] = 2 * r[(N-j) % N] for j = 0, ..., n-1.

    idx[1:] = N - np.arange(1, n)
    G = 2.0 * r[idx]
    return np.where(np.isfinite(G), G, 0.0)

def adam_run(a0, n, N, deadline, seed, temp0=0.06):
    rng = np.random.default_rng(seed)
    a = clip_nn(a0.copy())
    a = norm_sum(a)
    if a is None:
        a = np.ones(n) / n
    best = a.copy()
    best_sc = score_fast(a, N)
    L = 2 * n - 1
    mv = np.zeros(n)
    vv = np.zeros(n)
    t = 0
    temp = temp0
    no_imp = 0
    while time.monotonic() < deadline:
        t += 1
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        z = make_weights(c, max_c, temp)
        G = grad_conv(a, z, N, n)
        aG = float(a.dot(G))
        G = G - a * aG
        b1, b2 = 0.9, 0.999
        mv = b1 * mv + (1.0 - b1) * G
        vv = b2 * vv + (1.0 - b2) * G * G
        lr = 0.04 / (1.0 + 0.0003 * t)
        mb = mv / (1.0 - b1 ** t)
        vb = vv / (1.0 - b2 ** t)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        a = a - step
        a = clip_nn(a)
        a = norm_sum(a)
        if a is None:
            a = best.copy()
            break
        sc = score_fast(a, N)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = sc
            no_imp = 0
        else:
            no_imp += 1
            temp = min(1.0, temp * 1.005)
        if no_imp > 400:
            a = best * (1.0 + 0.15 * rng.normal(0, 1, n))
            a = clip_nn(a)
            a = norm_sum(a)
            if a is None:
                a = best.copy()
            mv.fill(0.0)
            vv.fill(0.0)
            t = 0
            no_imp = 0
            temp = temp0
    return best, best_sc

def block_perturb(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = clip_nn(a.copy())
    best = norm_sum(best)
    if best is None:
        best = np.ones(n) / n
    best_sc = score_fast(best, N)
    lo = max(1, n // 400)
    hi = max(lo + 1, n // 16)
    while time.monotonic() < deadline:
        w = int(rng.integers(lo, hi + 1))
        st = int(rng.integers(0, n - w))
        c = best.copy()
        c[st:st+w] *= float(np.exp(rng.normal(0, 0.2)))
        c = clip_nn(c)
        c = norm_sum(c)
        if c is None:
            continue
        sc = score_fast(c, N)
        if np.isfinite(sc) and sc < best_sc:
            best = c
            best_sc = sc
    return best, best_sc

def exact_score(a):
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return np.inf

def propose_candidate(seed=42, budget_s=1100, **kw):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    deadline = t0 + max(0.0, budget * 0.94)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        parent = np.ones(5582)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        parent = np.ones(5582)
    parent = clip_nn(parent)
    n = parent.size
    if n <= 0:
        n = 5582
        parent = np.ones(n)

    N = next_pow2(2 * n - 1)
    ones = np.ones(n) / n

    # Build candidate pool
    pool = [ones.copy(), clip_nn(parent)]
    if n > 4:
        j = np.arange(n, dtype=np.float64)
        for fr in (2, 3, 5):
            x = 1.0 + 0.4 * np.cos(2.0 * np.pi * fr * j / n)
            p = norm_sum(np.maximum(x, 0.01))
            if p is not None:
                pool.append(p)
        for fr in (1, 2):
            x = 1.0 + 0.3 * np.sin(2.0 * np.pi * fr * j / n)
            p = norm_sum(np.maximum(x, 0.01))
            if p is not None:
                pool.append(p)

    # Pick best init
    best_a = ones.copy()
    best_sc = score_fast(ones, N)
    for p in pool:
        sc = score_fast(p, N)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = p.copy()

    # Phase 1: Adam from best init
    rem = deadline - time.monotonic()
    if rem > 15:
        d1 = time.monotonic() + min(rem * 0.55, budget * 0.5)
        try:
            a1, s1 = adam_run(best_a, n, N, d1, seed + 10, temp0=0.05)
            if np.isfinite(s1) and s1 < best_sc:
                best_a, best_sc = a1, s1
        except Exception:
            pass

    # Phase 2: Adam from parent with different temp
    rem = deadline - time.monotonic()
    if rem > 15:
        d2 = time.monotonic() + min(rem * 0.35, budget * 0.25)
        try:
            a2, s2 = adam_run(clip_nn(parent), n, N, d2, seed + 20, temp0=0.12)
            if np.isfinite(s2) and s2 < best_sc:
                best_a, best_sc = a2, s2
        except Exception:
            pass

    # Phase 3: Block perturbations
    rem = deadline - time.monotonic()
    if rem > 10:
        d3 = time.monotonic() + min(rem * 0.35, budget * 0.15)
        try:
            a3, s3 = block_perturb(best_a, n, N, d3, seed + 30)
            if np.isfinite(s3) and s3 < best_sc:
                best_a, best_sc = a3, s3
        except Exception:
            pass

    # Phase 4: Final Adam polish
    rem = deadline - time.monotonic()
    if rem > 10:
        d4 = time.monotonic() + min(rem * 0.6, budget * 0.1)
        try:
            a4, s4 = adam_run(best_a, n, N, d4, seed + 40, temp0=0.03)
            if np.isfinite(s4) and s4 < best_sc:
                best_a, best_sc = a4, s4
        except Exception:
            pass

    # Exact evaluation of top candidates
    cands = [best_a]
    for p in pool[:4]:
        cands.append(p)
    best_final = best_a
    best_final_sc = np.inf
    for c in cands:
        if time.monotonic() > deadline - 5:
            break
        c = clip_nn(c)
        c = norm_sum(c)
        if c is None:
            continue
        sc = exact_score(c)
        if np.isfinite(sc) and sc < best_final_sc:
            best_final_sc = sc
            best_final = c

    if best_final is None or best_final_sc >= np.inf:
        best_final = ones

    return [float(x) for x in best_final]
