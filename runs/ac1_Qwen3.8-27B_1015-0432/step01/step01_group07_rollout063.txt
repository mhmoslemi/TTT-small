import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[: 2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _smooth_max_and_weights(conv, tau):
    c = conv - conv.max()
    ec = np.exp(tau * c)
    s = ec.sum()
    p = ec / s
    val = conv.max() + np.log(s) / tau
    return val, p


def _xcorr_for_grad(a, p, xsize):
    F = np.fft.rfft(a, xsize)
    P = np.fft.rfft(p, xsize)
    return np.fft.irfft(np.conj(F) * P, xsize)[: len(a)]

import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[: 2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _smooth_max_w(conv, tau):
    """Return (smooth_max_value, softmax_weights) of conv."""
    c = conv - conv.max()
    ec = np.exp(np.maximum(tau * c, -700.0))  # prevent underflow issues
    s = float(ec.sum())
    if s < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
        return float(conv.max()), p
    p = ec / s
    val = float(conv.max() + np.log(s) / tau)
    return val, p


def _xcorr_grad(a, p, sz):
    """h[i] = sum_k p[k] * a[k - i] via FFT cross-correlation."""
    F = np.fft.rfft(a, sz)
    P = np.fft.rfft(p, sz)
    return np.fft.irfft(np.conj(F) * P, sz)[: len(a)]

    while xsz < 3 * n + 8:
        xsz <<= 1

    best = a0.copy()
    best_score = _score(a0)

    lr = 0.5
    tau = 50.0

    v_a = np.zeros(n)

        S = float(a.sum())
        if S < 0.01 or not np.isfinite(S):
            a = a0.copy()
            m_a[:] = 0.0
            v_a[:] = 0.0
            t_a = 0

        sm_val, p = _smooth_max_w(conv, tau)
        h = _xcorr_grad(a, p, xsz)
        g = (4.0 * n / (S * S)) * h - (4.0 * n * sm_val / (S * S * S))

import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[: 2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _smooth_max_w(conv, tau):
    c = conv - conv.max()
    ec = np.exp(np.maximum(tau * c, -700.0))
    s = float(ec.sum())
    if s < 1e-300:
        p = np.zeros_like(conv)
        p[int(np.argmax(conv))] = 1.0
        return float(conv.max()), p
    p = ec / s
    val = float(conv.max() + np.log(s) / tau)
    return val, p


def _xcorr_grad(a, p, sz):
    F = np.fft.rfft(a, sz)
    P = np.fft.rfft(p, sz)
    return np.fft.irfft(np.conj(F) * P, sz)[: len(a)]
