def _fft_conv_sq(a):
    n = len(a)
    sz = 1 << (2 * n - 2).bit_length()
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _active_subgrad(c, a, n, k):
    m = len(c)
    if k >= m:
        idx = np.arange(m)
    else:
        part = np.argpartition(c, -k)[-k:]
        idx = part[np.argsort(c[part])[::-1]]
    g = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[ki - hi:ki - lo + 1]
    g /= len(idx)
    return g


def _gen_init(n, rng, mode):
    j = np.arange(n, dtype=np.float64)
    if mode == 0:
        a = 0.5 + 0.5 * np.cos(np.pi * j / max(1, n - 1))
    elif mode == 1:
        a = 0.5 * (1.0 + np.cos(np.pi * j / max(1, n - 1))) ** 2
    elif mode == 2:
        blk = max(3, n // 200)
        a = np.repeat(rng.random(n // blk + 2), blk)[:n]
    elif mode == 3:
        a = np.exp(-((j - 0.3 * n) / (0.15 * n)) ** 2) + np.exp(-((j - 0.7 * n) / (0.15 * n)) ** 2)
    elif mode == 4:
        a = np.abs(np.sin(np.pi * j / n)) ** 0.5
    else:
        r = rng.integers(1, max(2, n // 4))
        a = np.abs(np.cos(2.0 * np.pi * r * j / n)) + 0.1
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    return a * n / s if s > 1e-10 else np.ones(n)

    # Nesterov-style momentum subgradient descent
    a = best.copy()
    v = np.zeros(n)
    beta = 0.88
    lr = 1.5
    it = 0
    stuck = 0
    total_iters = 0

    while time.monotonic() < dl:
        it += 1
        total_iters += 1
        c = _fft_conv_sq(a)
        sc = 2.0 * n * c.max() / (n * n)

        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 3000 == 0:
            se = float(evaluate_sequence(a.tolist()))
            if se < best_sc:
                best_sc = se
                best = a.copy()
                stuck = 0

        # Annealed active set size: 32 -> 128
        frac = min(1.0, it / 10000.0)
        topk = int(32 + 96 * frac)

        # Subgradient
        g = _active_subgrad(c, a, n, topk)

        # Nesterov momentum
        v = beta * v + (1.0 - beta) * g

        # Backtracking line search
        accepted = False
        for f in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15):
            step = lr * f
            a_new = a - step * v
            a_new = np.clip(a_new, 0.0, 1000.0)
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new * n / sn
            else:
                continue
            sc_new = 2.0 * n * _fft_conv_sq(a_new).max() / (n * n)
            if sc_new < sc - 1e-14:
                a = a_new
                accepted = True
                break

        if accepted:
            lr = min(lr * 1.04, 25.0)
        else:
            lr = max(lr * 0.72, 0.002)
            stuck += 1

        # Soft perturb when moderately stuck
        if stuck > 1500 and stuck <= 5000:
            sigma = 0.01 * max(1.0, a.std())
            a = a + sigma * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
            a = a * n / a.sum()
            v[:] = 0.0
            lr = 1.5
            stuck = 0
            it = max(1, it - 6000)

        # Restart from best + noise when heavily stuck
        if stuck > 5000 and stuck <= 12000:
            a = best + 0.04 * max(1.0, best.std()) * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
            a = a * n / a.sum()
            v[:] = 0.0
            lr = 1.5
            stuck = 0
            it = 1

        # Full restart with new initialization
        if stuck > 12000:
            m = int(rng.integers(0, 5))
            a = _gen_init(n, rng, m)
            sc_r = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_r) and sc_r < best_sc:
                best_sc = sc_r
                best = a.copy()
            v[:] = 0.0
            lr = 1.5
            stuck = 0
            it = 1

    out = np.clip(best, 0.0, 1000.0)
    if out.sum() < 0.01:
        out = np.ones(n)
    return [float(x) for x in out]
