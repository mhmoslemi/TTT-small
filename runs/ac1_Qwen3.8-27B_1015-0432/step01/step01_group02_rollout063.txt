        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a /= s

        iters += 1

        if iters % 200 == 0:

            st = int(rng.integers(0, n - ws))
            a[st:st + ws] *= np.exp(rng.normal(0.0, 0.3, ws))
            a = np.clip(a, 0.0, 1000.0)
            s = a.sum()
            if s > 1e-12:

            if alpha < 0.03:

                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s > 1e-12:
                    a /= s
                alpha = 0.12
                stuck = 0

    result = [float(x) for x in best_a]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * n
    return result
