import numpy as np
import time


def fft_conv(a, S):
    n = a.size
    fa = np.fft.rfft(a, S)
    return np.fft.irfft(fa * fa, S)[:2 * n - 1]


def cross_grad(a, p, S):
    n = a.size
    P = np.fft.rfft(p, S)
    A = np.fft.rfft(a, S)
    return np.fft.irfft(np.conj(P) * A, S)[:n]


def normalize(nn):

    if s < 1e-15:
        return None
    return nn / s

def run_soft(a, S, deadline, max_it=200000):

    a = normalize(a)
    if a is None:
        return None, np.inf
    v = np.zeros(n)
    bs_a = a.copy()
    bs_sc = np.inf
    stall = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        conv = fft_conv(a, S)
        M = float(conv.max())
        sc = 2.0 * n * M
        if sc < bs_sc - 1e-12:
            bs_sc = sc
            bs_a = a.copy()
            stall = 0

            stall += 1
        if stall > 4000:

        spread = M - float(conv.mean())

        prog = min(1.0, it / float(max_it))

        if ps < 1e-300:

        beta = 0.78 + 0.12 * prog
        lr = 0.25 * (1.0 + np.cos(np.pi * min(1.0, 1.2 * prog)))

        a = np.clip(a - lr * v, 0.0, None)
        a = normalize(a)
        if a is None:
            break
    return bs_a, bs_sc
