import numpy as np
import time

def _fft_corr(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score_fast(corr, n):
    s = float(n)
    return 2.0 * n * float(corr.max()) / (s * s)

def _active_grad(corr, a, n, topk):
    if topk >= corr.size:
        idx = np.arange(corr.size)
    else:
        thresh = np.partition(corr, -topk)[-topk]
        idx = np.where(corr >= thresh)[0]
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        if lo >= n:
            continue
        js = np.arange(lo, n)
        diffs = np.abs(k - js)
        m = diffs < n
        if m.any():
            g[lo:n] += a[diffs[m]]
    if len(idx) > 0:
        g /= len(idx)
    return g

def _gen_init(n, rng, m):
    j = np.arange(n, dtype=np.float64)
    if m == 0:
        a = np.ones(n)
    elif m == 1:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * j / n)
    elif m == 2:
        a = np.abs(np.sin(np.pi * j / n)) ** 0.7
    elif m == 3:
        a = rng.random(n)
    elif m == 4:
        a = 0.3 + 0.7 * np.abs(np.cos(np.pi * j * (n - 1) / (2.0 * n)))
    elif m == 5:
        blk = max(2, n // 200)
        a = np.repeat(rng.random(n // blk + 2), blk)[:n]
    else:
        a = 1.0 + 0.3 * np.sin(2.0 * np.pi * j * 7.0 / n) + 0.2 * np.cos(2.0 * np.pi * j * 13.0 / n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    return a * n / s if s > 1e-10 else np.ones(n)

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:

    return a * n / s
