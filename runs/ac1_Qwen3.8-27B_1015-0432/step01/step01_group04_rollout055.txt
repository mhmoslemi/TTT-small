import numpy as np
import time

def convolve_fft(a, N):
    """Compute full linear convolution a*a via FFT."""
    Af = np.fft.rfft(a, N)
    return np.fft.irfft(Af * Af, N)

def score_fft(a, N):
    n = a.size
    s = a.sum()
    if s <= 0: return np.inf
    b = convolve_fft(a, N)
    b = b[:2*n-1]
    return 2.0 * n * b.max() / (s*s)

def active_set_gradient(a, N, beta):
    """Compute subgradient of max(b) using soft-max weighting."""
    n = a.size
    b = convolve_fft(a, N)
    b = b[:2*n-1]
    b = np.maximum(b, 0)
    bmax = b.max()
    # Soft weights
    if bmax <= 0:
        return np.zeros(n), bmax, b
    w = np.exp(beta * (b - bmax))

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def project(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s

def conv_full(a, N):
    Af = np.fft.rfft(a, N)
    return np.fft.irfft(Af * Af, N)

def score(a, N):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = conv_full(a, N)
    b = b[:2*n-1]
    b = np.maximum(b, 0.0)
    return float(2.0 * n * b.max() / (s*s))

def grad_softmax(a, N, beta):
    n = a.size
    b = conv_full(a, N)
    b = b[:2*n-1].copy()
    b = np.maximum(b, 0.0)
    bmax = float(b.max())

    w = np.exp(beta * (b - bmax))
    ws = float(w.sum())
