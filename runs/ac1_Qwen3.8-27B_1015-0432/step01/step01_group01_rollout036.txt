import numpy as np
import time


def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def calc_score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    fs = 1 << (2 * n - 1).bit_length()
    conv = fft_convolve(a, fs)
    return float(2.0 * n * conv.max() / (s * s))


def normalize(a, target=1.0):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a * (target / s)

    v_buf = np.zeros(n)
    bc1, bc2, eps = 0.85, 0.999, 1e-8
    best_a = a.copy()
    best_sc = calc_score(a)
    prev_M = np.inf
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        conv = fft_convolve(a, fft_size)
        M = float(conv.max())
        stall = 0 if M < prev_M - 1e-15 else stall + 1
        prev_M = M
        if stall > 2500:
            break
        tk = min(n // 2, init_top_k + it // 200)
        g = subgrad_peak(a, conv, tk)
        g -= g.mean()
        t = it + 1
        m_buf = bc1 * m_buf + (1.0 - bc1) * g
        v_buf = bc2 * v_buf + (1.0 - bc2) * g * g
        mh = m_buf / (1.0 - bc1 ** t)
        vh = v_buf / (1.0 - bc2 ** t)
        step = lr * (0.9997 ** it) * mh / (np.sqrt(vh) + eps)
        a = np.clip(a - step, 0.0, None)

        if it % 300 == 0:

            a2, la2, ls2 = adam_optimize(la, n, fft_size, t_end2, tk0 // 2, lr0 * 0.5)
            if ls2 < best_score:
                best_score = ls2
                best = la2 * s0

    # Final polish: greedy multiplicative block perturbations
    if time.monotonic() < deadline - 3.0:
        base = best / s0
        w = max(2, n // 80)
        while time.monotonic() < deadline - 2.0:
            cand = base.copy()
            n_perts = int(rng.integers(1, 4))
            for _ in range(n_perts):
                st = int(rng.integers(0, max(1, n - w)))
                ln = int(np.clip(w * (1.0 + rng.normal(0, 0.4)), 1, n - st))
                e = min(n, st + ln)
                cand[st:e] *= float(np.exp(rng.normal(0, 0.035)))
            cand = normalize(cand, 1.0)
            if cand is None:
                continue
            ccand = np.clip(cand * s0, 0.0, 1000.0)
            sc = float(evaluate_sequence(ccand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = ccand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
    return [float(x) for x in out]
