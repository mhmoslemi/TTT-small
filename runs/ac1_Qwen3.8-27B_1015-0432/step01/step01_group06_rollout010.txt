        w[idx] = corr[idx]
        w /= w.sum()
    g = np.zeros(n)
    for k in idx:
        if w[k] == 0:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi + 1] += w[k] * a[k - np.arange(lo, hi + 1)]
    g -= g.mean()  # project onto tangent space
    return g

    return 2.0 * n * corr.max() / (n * n)
