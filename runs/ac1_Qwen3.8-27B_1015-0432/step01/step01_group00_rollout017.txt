    return float(2 * len(f) * b.max() / (s * s))

def symmetrize(f):
    return 0.5 * (f + f[::-1])

def make_init(n, kind, rng=None):
    x = np.arange(n, dtype=np.float64)
    if kind == 'uniform':
        f = np.ones(n)
    elif kind == 'hann':
        f = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    elif kind == 'powlaw05':
        d = np.minimum(x, n - 1.0 - x) + 1.0
        f = 1.0 / np.sqrt(d)
    elif kind == 'powlaw1':
        d = np.minimum(x, n - 1.0 - x) + 1.0
        f = 1.0 / d
    elif kind == 'powlaw15':
        d = np.minimum(x, n - 1.0 - x) + 1.0
        f = 1.0 / d ** 1.5
    elif kind == 'gauss':
        c = (n - 1.0) / 2.0
        f = np.exp(-4.0 * ((x - c) / c) ** 2)
    elif kind == 'cos2':
        f = np.cos(np.pi * x / (n + 1)) ** 2
    elif kind == 'tri':
        f = 1.0 - 2.0 * np.abs(x - (n-1)/2.0) / (n - 1)
    elif kind == 'perturb1':
        f = np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n))
    elif kind == 'perturb2':
        f = np.ones(n) * (1.0 + 0.6 * rng.standard_normal(n))
    else:
        f = np.ones(n)
    f = np.maximum(f, 1e-10)
    f = symmetrize(f)
    return f / f.sum()
