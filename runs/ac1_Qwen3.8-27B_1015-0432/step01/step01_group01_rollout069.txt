import numpy as np
import time


def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def topk_grad(a, conv, k):
    n = a.size
    idx = np.argpartition(conv, -k)[-k:]
    g = np.zeros(n)
    for lag in idx:
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            g[lo:hi + 1] += a[lag - hi:lag - lo + 1][::-1]
    return g / float(k)
