def npow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def proj(a, n):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01 or not np.isfinite(s):
        return np.ones(n) / n
    return a / s

def ffscore(a, N):
    n = len(a)
    s = a.sum()
    if s < 0.01 or not np.finite(s):
        return np.inf
    A = np.fft.rfft(a, N)
    b = np.fft.irfft(A * A, N)[:2 * n - 1]
    b = np.maximum(b, 0.0)
    mb = b.max()
    if not np.finite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s)), b

def xcorr(p, a, N):
    P = np.fft.rfft(p, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(np.conj(P) * A, N)[:len(a)]
