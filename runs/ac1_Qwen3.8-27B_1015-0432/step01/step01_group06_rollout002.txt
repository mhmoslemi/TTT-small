def _fft_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)

import numpy as np
import time


def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    r = rho[-1]
    theta = (cssv[r] - 1) / (r + 1.0)
    return np.maximum(v - theta, 0)


def _gen_init(n, rng, m):
    j = np.arange(n)
    if m == 0:
        a = np.ones(n)
    elif m == 1:
        a = 0.5 + 0.5 * np.cos(2.0 * np.pi * j / n)
    elif m == 2:
        a = np.abs(np.sin(np.pi * j / n)) ** 0.7
    elif m == 3:
        a = rng.random(n) + 0.01
    elif m == 4:
        a = np.cos(3.0 * np.pi * j / n) ** 2
    elif m == 5:
        a = np.abs(np.sin(2.0 * np.pi * j / n)) ** 0.5 + 0.01
    else:
        blk = max(2, n // 300)
        a = np.repeat(rng.random(n // blk + 2), blk)[:n] + 0.01
    s = a.sum()
    return a / s if s > 1e-10 else np.ones(n) / n
