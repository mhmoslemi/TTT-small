import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _coord_eval(x, conv_sl, base, j2i, ilen, m_unaff, S, v, n):
    nS = S - v + x
    if nS < 0.01:
        return 1e30
    d = base * (x - v)
    if 0 <= j2i < ilen:
        d = d.copy()
        d[j2i] = x * x - v * v
    mx = np.max(conv_sl + d)
    if mx < m_unaff:
        mx = m_unaff
    return 2.0 * n * mx / (nS * nS)

def _opt_coord(conv, a, i, n, S, m_unaff):
    v = a[i]
    j_lo = i
    j_hi = min(2 * n - 2, i + n - 1)

    # Golden section refinement
    pos = int(np.argmin(np.abs(xs - best_x)))
    lo = float(xs[max(0, pos - 1)])
    hi = float(xs[min(11, pos + 1)])
    if lo == hi:

    if f0 <= fx and f0 <= fv:
