def _sc(a, c=None):
    n = a.size
    if c is None:
        c = _fft_conv(a)
    s = a.sum()
    if s < 0.01:
        return 1e30
    return 2.0*n*c.max()/(s*s)

def _mul_update(a, alpha, n, rng, stuck):
    c = _fft_conv(a)
    M = c.max()
    if M < 1e-15:
        return a
    # exponential peak weights
    w = np.exp(3.0*(c/M - 1.0))
    wsum = w.sum()
    if wsum < 1e-15:
        w = np.zeros(2*n-1)
        w[np.argmax(c)] = 1.0
    else:
        w /= wsum
    # pressure
    p = np.zeros(n)
    # only use significant peaks
    wmax = w.max()
    peak_ids = np.where(w > 0.3*wmax)[0]
    for k in peak_ids:
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        p[lo:hi+1] += w[k]*a[k-np.arange(lo,hi+1)]
    # centered update
    pc = p - p.mean()
    pm = np.abs(pc).max()
    if pm > 1e-15:
        a *= np.exp(-alpha*pc/pm)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a.fill(1.0/n)
    else:
        a /= s
    return a

def _run(a, n, rng, tlimit, alpha0=0.15):
    best = a.copy()
    bs = _sc(best)
    alpha = alpha0
    stuck = 0
    it = 0

        sc = _sc(a, c)

            best = a.copy()
            stuck = 0
        else:

        it += 1

            st = int(rng.integers(0, max(1, n-pw)))

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a /= s
        if stuck > 800:
            alpha *= 0.92
            if alpha < alpha0*0.08:

                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-12:
                    a /= s
                alpha = alpha0
                stuck = 0
    return best, bs

def _flat_spec_init(n, rng):

def _fft_conv(a):
    n = a.size
    N = 1 << (2 * n - 2).bit_length()

    return np.fft.irfft(F * F, N)[:2 * n - 1]

    n = a.size

    if s < 0.01:
