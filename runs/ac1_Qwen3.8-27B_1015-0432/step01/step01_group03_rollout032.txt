def _project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cs = np.arange(1, n + 1)
    idx = np.nonzero(u * cs > (cssv - 1.0))[0]
    if len(idx) == 0:
        return np.full(n, 1.0 / n)
    rho = idx[0]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
