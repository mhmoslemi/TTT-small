import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score_raw(a, conv):
    S = float(np.sum(a))
    n = len(a)
    if S < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(conv)) / (S * S)


def _weighted_subgrad(a, conv, top_k=25, frac=0.85):
    n = len(a)
    M = float(np.max(conv))
    if M <= 0:
        return np.zeros(n)
    if len(conv) <= top_k:
        idx = np.where(conv >= M * frac)[0]
    else:
        part = np.argpartition(-conv, min(top_k, len(conv) - 1))[:top_k]
        idx = part[conv[part] >= M * frac]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(conv))])
    w = conv[idx].astype(np.float64)
    w = np.maximum(w - w.min(), 1e-15)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i])
        wi = float(w[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        jj = np.arange(js, je + 1)
        ki = k - jj
        g[jj] += wi * a[ki]
        g[ki] += wi * a[jj]
    return g
