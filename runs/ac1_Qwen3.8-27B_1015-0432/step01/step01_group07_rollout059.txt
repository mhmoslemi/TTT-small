def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

    n = len(a)
    conv = _fft_conv(a)
    return 2.0 * n * float(np.max(conv))

def _subgrad(a, lags):
    n = len(a)
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            g[js:je + 1] += a[k - np.arange(js, je + 1)]
    if len(lags) > 0:
        g /= len(lags)
    return g
