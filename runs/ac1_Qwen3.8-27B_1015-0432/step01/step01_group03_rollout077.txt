def _sv_obj(a, tau, lam, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e12
    tau = max(1e-15, float(tau))
    A = rfft(a, L)
    c = irfft(A * A, L)[:2*n-1]
    cm = c.max()
    if not np.isfinite(cm):
        return 1e12
    e = np.exp((c - cm) / tau)
    se = e.sum()
    if not np.isfinite(se) or se <= 0.0:
        return 1e12
    M = cm + tau * np.log(se)
    return float((2.0 * n * M) / (S * S) + lam * (S - 1.0) * (S - 1.0))

def _sv_grad(a, tau, lam, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return np.zeros(n, dtype=np.float64)
    tau = max(1e-15, float(tau))
    A = rfft(a, L)
    c = irfft(A * A, L)[:2*n-1]
    cm = c.max()
    if not np.isfinite(cm):
        return np.zeros(n, dtype=np.float64)
    e = np.exp((c - cm) / tau)
    se = e.sum()
    if not np.isfinite(se) or se <= 0.0:
        p = np.zeros(2*n-1, dtype=np.float64); p[0]=1.0; M=cm
    else:
        p = e / se
        M = cm + tau * np.log(se)
    P = rfft(p, L)
    h = irfft(np.conj(P) * A, L)[:n]
    g = (4.0 * n * h) / (S * S) - (4.0 * n * M) / (S * S * S)
    if lam > 0.0:
        g = g + 2.0 * lam * (S - 1.0)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return g

import time
import numpy as np
from numpy.fft import rfft, irfft
from scipy.optimize import minimize

_deadline = None


def _p2(m):
    s = 1
    while s < m:
        s <<= 1
    return s


def _item_score(x):
    return x[0]


def _timeout_cb(x):
    if _deadline is not None and time.monotonic() > _deadline - 3.0:
        raise TimeoutError
    return False


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def _score_l(a, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    S = a.sum()
    if S < 1e-8:
        return np.inf
    A = rfft(a, L)
    c = irfft(A * A, L)
    mc = c[:2 * n - 1].max()
    if not np.isfinite(mc):
        return np.inf
    if mc < 0.0:
        mc = 0.0
    return float((2.0 * n * mc) / (S * S))


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    return _score_l(a, _p2(2 * a.size - 1))


def _true_score(a):
    try:
        vals = [float(v) for v in np.asarray(a, dtype=np.float64)]
        return float(evaluate_sequence(vals))
    except Exception:
        return np.inf


def _central_binomial(n):
    a = np.empty(n, dtype=np.float64)
    if n == 0:
        return a
    a[0] = 1.0
    v = 1.0
    for k in range(1, n):
        v *= (2.0 * k - 1.0) / (2.0 * k)
        a[k] = v
    return a


def _sv_obj(a, tau, lam, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e12
    tau = max(1e-15, float(tau))
    A = rfft(a, L)
    c = irfft(A * A, L)[:2 * n - 1]
    cm = c.max()
    if not np.isfinite(cm):
        return 1e12
    e = np.exp((c - cm) / tau)
    se = e.sum()
    if not np.isfinite(se) or se <= 0.0:
        return 1e12
    M = cm + tau * np.log(se)
    return float((2.0 * n * M) / (S * S) + lam * (S - 1.0) * (S - 1.0))


def _sv_grad(a, tau, lam, L):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = a.sum()
    if S < 1e-4:
        return np.zeros(n, dtype=np.float64)
    tau = max(1e-15, float(tau))
    A = rfft(a, L)
    c = irfft(A * A, L)[:2 * n - 1]
    cm = c.max()
    if not np.isfinite(cm):
        return np.zeros(n, dtype=np.float64)
    e = np.exp((c - cm) / tau)
    se = e.sum()
    if not np.isfinite(se) or se <= 0.0:
        p = np.zeros(2 * n - 1, dtype=np.float64)
        p[int(c.argmax())] = 1.0
        M = cm
    else:
        p = e / se
        M = cm + tau * np.log(se)
    P = rfft(p, L)
    h = irfft(np.conj(P) * A, L)[:n]
    g = (4.0 * n * h) / (S * S) - (4.0 * n * M) / (S * S * S)
    if lam > 0.0:
        g = g + 2.0 * lam * (S - 1.0)
    if not np.all(np.isfinite(g)):
        g = np.zeros(n, dtype=np.float64)
    return g


def _refine_lbmfgs(a, tau, maxiter, lam, L, bounds):
    if _deadline is None or time.monotonic() > _deadline - 5.0:

    a = np.asarray(a, dtype=np.float64).copy()

                       method='L-BFGS-B', bounds=bounds, options=opts,
                       callback=_timeout_cb)
        x = res.x if hasattr(res, 'x') and res.x is not None else a
    except Exception:
        x = a
    return _normalize(x)

    cand = _normalize(cand)
    sc = _internal_score(cand)
