import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

# Global FFT length (computed once)
_L = 0

def _setup_fft(n):
    global _L
    _L = _next_pow2(3 * n - 2)

def _next_pow2(n):
    m = 1
    while m < n:
        m <<= 1
    return m

def _conv(a):
    """Full convolution a*a via FFT, length 2n-1."""
    return (ifft(fft(a, _L) * fft(a, _L)).real)[:2 * a.size - 1]

def _smmax(a, tau):
    """Smooth max of convolve(a,a) and its gradient w.r.t. a."""
    n = a.size
    c = _conv(a)
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    p = e / s
    mv = cm + tau * np.log(s)
    # grad = 2 * correlation of p with a
    pr = p[::-1]
    g = 2.0 * (ifft(fft(a, _L) * fft(pr, _L)).real)[n - 1:2 * n - 1]
    return mv, p, g

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _, _ = _smmax(a, tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    mv, _, g = _smmax(a, tau)
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S * S * S)) * a

def _active_grad(a, c, top_k):

    n = a.size
    idx = np.argsort(c)[-top_k:]
    w = 1.0 / top_k
    g = np.zeros(n)

        ki = c  # reuse
