    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score_internal(a, corr):
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * corr.max() / (s * s)

def _subgrad_peak(corr, a, k):
    """Gradient of (a*a)[k] w.r.t. a."""
    n = a.size
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = k - np.arange(lo, hi + 1)
        grad[lo:hi+1] = 2.0 * a[idx]
    return grad
