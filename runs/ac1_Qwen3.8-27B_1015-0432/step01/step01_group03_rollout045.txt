def _fft_pow2(size):

def _smmax(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj_fn(a, tau, n, L):
    S = a.sum()
    if S < 1e-8:

    A = fft(a, L)
    b = ifft(A * A).real[:2 * n - 1]

def _grad_fn(a, tau, n, L):
