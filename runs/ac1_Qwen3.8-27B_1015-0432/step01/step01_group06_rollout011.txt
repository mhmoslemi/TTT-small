import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score(corr, n):
    return 2.0 * n * corr.max() / (n * n)

def _grad_active(corr, a, n, top_k):
    idx = np.argpartition(corr, -top_k)[-top_k:]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
    g /= len(idx)
    return g - g.mean()

def _gen_init(n, rng, variant):
    j = np.arange(n)
    if variant == 0:
        a = np.ones(n)
    elif variant == 1:
        a = 1.0 + 0.25 * np.cos(2.0 * np.pi * j / n)
    elif variant == 2:
        a = 1.0 + 0.25 * np.cos(4.0 * np.pi * j / n + 0.7)
    elif variant == 3:
        a = 1.0 + 0.15 * np.sin(3.0 * 2.0 * np.pi * j / n + 1.3)
    elif variant == 4:
        a = 1.0 + 0.2 * np.cos(2.0 * np.pi * 5.0 * j / n) + 0.1 * np.cos(2.0 * np.pi * 11.0 * j / n)
    elif variant == 5:

    a = np.clip(a, 0.01, 10.0)
    return a * n / a.sum()
