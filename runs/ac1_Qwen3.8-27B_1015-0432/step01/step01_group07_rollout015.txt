import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _score_of(a):
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def _subgrad(a, c, K=5):
    """Return subgradient of f(a) = 2n*max(c)/S^2, using top-K smoothing."""
    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return None
    top_idx = np.argsort(c)[-K:][::-1]
    top_vals = c[top_idx]
    w = np.exp((top_vals - top_vals[0]) * 20.0 / n)

            km = k - m
            g[m] += wk * (4.0 * n * a[km] / (S * S) - 4.0 * n * c[k] / (S ** 3))
