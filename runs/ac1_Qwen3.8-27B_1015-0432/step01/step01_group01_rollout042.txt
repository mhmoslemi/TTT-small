def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def topk_grad(a, conv, tk):
    n = a.size
    idx = np.argpartition(conv, -tk)[-tk:]
    g = np.zeros(n, dtype=np.float64)
    for k in idx:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += a[ki - hi:ki - lo + 1][::-1]
    return g / tk
