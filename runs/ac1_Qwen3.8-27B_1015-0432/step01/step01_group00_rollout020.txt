def fft_convolve(a, b):

    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_convolve(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def calc_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def grad_smooth(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.exp(np.clip(beta * (b - M) / M, -50.0, 0.0))
    raw = fft_conv(w, f[::-1])[n - 1:2 * n - 1]
    return np.maximum(raw, 0.0)
