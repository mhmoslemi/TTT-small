import numpy as np
import time


def fft_conv(a, S):
    n = a.size
    fa = np.fft.rfft(a, S)
    return np.fft.irfft(fa * fa, S)[:2 * n - 1]


def cross_grad(a, p, S):
    n = a.size
    P = np.fft.rfft(p, S)
    A = np.fft.rfft(a, S)
    return np.fft.irfft(np.conj(P) * A, S)[:n]


def normalize_nn(x):
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-15:
        return None
    return x / s


def gen_init(kind, n, rng, pnorm):
    if kind == 0:
        a = pnorm.copy()
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        f = rng.uniform(0.05, 0.4)
        ph = rng.uniform(0.0, 6.2832)
        a = np.clip(1.0 + 0.5 * np.sin(2.0 * np.pi * f * np.arange(n) + ph), 0.01, None)
    elif kind == 3:
        r = rng.uniform(0.2, 0.7)
        tt = (np.arange(n) - n * 0.5) / (n * r)
        a = np.exp(-0.5 * tt * tt)
    elif kind == 4:
        f1 = rng.uniform(0.03, 0.2)
        f2 = rng.uniform(0.3, 0.8)
        t = np.arange(n)
        a = np.clip(1.0 + 0.4 * np.cos(2*np.pi*f1*t + rng.uniform(0, 6.28))
                    + 0.2 * np.cos(2*np.pi*f2*t + rng.uniform(0, 6.28)), 0.01, None)
    elif kind == 5:
        a = pnorm * (1.0 + 0.15 * rng.standard_normal(n))
    else:
        a = np.ones(n)
    return normalize_nn(a)


def run_soft(a, S, deadline, max_it=200000):
    n = a.size
    a = normalize_nn(a)
    if a is None:
        return None, np.inf
    v = np.zeros(n)
    bs_a = a.copy()
    bs_sc = np.inf
    stall = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        conv = fft_conv(a, S)
        M = float(conv.max())
        sc = 2.0 * n * M
        if sc < bs_sc - 1e-12:
            bs_sc = sc
            bs_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
        spread = M - float(conv.mean())
        if spread < 1e-14:
            break
        prog = min(1.0, it / float(max_it))
        tau = spread * (0.02 + 0.98 * (1.0 - prog))
        shifted = (conv - M) / tau
        p = np.exp(shifted)
        ps = p.sum()
        if ps < 1e-300:
            break
        p /= ps
        g = cross_grad(a, p, S) * (2.0 / tau)
        g -= g.mean()
        beta = 0.78 + 0.12 * prog
        lr = 0.25 * (1.0 + np.cos(np.pi * min(1.0, 1.2 * prog)))
        v = beta * v + (1.0 - beta) * g
        a = np.clip(a - lr * v, 0.0, None)
        a = normalize_nn(a)
        if a is None:
            break
    return bs_a, bs_sc
