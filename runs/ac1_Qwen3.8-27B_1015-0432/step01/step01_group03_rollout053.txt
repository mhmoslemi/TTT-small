def _fft_conv(a, b=None):
    if b is None:
        b = a
    L = 1
    total = a.size + b.size - 1
    while L < total:
        L <<= 1
    return ifft(fft(a, L) * fft(b, L)).real

def _smooth_max(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    b = _fft_conv(a)
