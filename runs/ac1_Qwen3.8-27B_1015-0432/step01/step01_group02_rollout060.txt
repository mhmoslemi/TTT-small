import numpy as np
import time


def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _compute_h(a, c, n, topK, use_fft):
    cmax = c.max()
    if cmax < 1e-15:
        return None
    idx = np.argpartition(c, -topK)[-topK:]
    peaks = c[idx]
    wsum = peaks.sum()
    if wsum < 1e-15:
        return None
    weights = peaks / wsum
    if use_fft:
        L = 1
        while L < 3 * n - 1:
            L <<= 1
        w_ext = np.zeros(L)
        for j, k in enumerate(idx):
            w_ext[int(k)] = weights[j]
        a_ext = np.zeros(L)
        a_ext[:n] = a
        Cw = np.fft.rfft(w_ext, L)
        Ca = np.fft.rfft(a_ext, L)
        h_full = np.fft.irfft(Cw * Ca, L)
        return h_full[:n].copy()
    else:
        h = np.zeros(n)
        for j, k in enumerate(idx):
            k = int(k)
            wk = weights[j]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if hi >= lo:
                h[lo:hi + 1] += wk * a[k - np.arange(lo, hi + 1)]
        return h
