import time
from scipy.optimize import linprog

def fft_conv(a, b, fs):
    fa = np.fft.rfft(a, fs)
    fb = np.fft.rfft(b, fs)
    return np.fft.irfft(fa * fb, fs)

def full_autoconv(a, fs):
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)

def top_peaks_subgrad(a, conv, t):
    n = a.size
    idx = np.argpartition(-conv, min(t, len(conv)))[:t]
    idx = sorted(idx.tolist(), key=lambda k: -conv[k])  # sort by value
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1]
            grad[lo:hi + 1] += seg[::-1]
    return grad / max(1, len(idx))

def gen_init(kind, n, rng, parent):
    if kind == 0:
        a = parent.copy()
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        alpha = rng.uniform(0.5, 3.0)

    elif kind == 3:
        a = np.abs(rng.standard_normal(n)) ** 2
    elif kind == 4:
        f1 = rng.uniform(0.05, 0.3)
        f2 = rng.uniform(0.3, 0.8)
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.sin(2 * np.pi * f1 * x + rng.uniform(0, 2 * np.pi))
        a += 0.3 * np.cos(2 * np.pi * f2 * x + rng.uniform(0, 2 * np.pi))
        a = np.clip(a, 0.01, None)
    elif kind == 5:

        a = 1.0 / (1.0 + np.exp(rng.uniform(1, 5) * (np.arange(n) / n - 0.3)))
    elif kind == 6:

        a = np.clip(a, 0.01, None)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

    beta = 0.8
    lr = 0.3
    prev_M = np.inf
    stall = 0
    local_best = best.copy()
    local_score = best_score
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        conv = full_autoconv(a, fs)
        M = float(conv[n-1] * 0 + conv.max())  # max over full conv
        if M < prev_M - 1e-18:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall >= 2000:
            break
        grad = top_peaks_subgrad(a, conv, top_k)
        v = beta * v + (1 - beta) * grad
        v -= v.mean()
        a = np.clip(a - lr * v, 1e-10, None)
        a /= a.sum()
        lr = max(0.01, lr * 0.9999)

            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < local_score:
                local_best = cand
                local_score = sc
    return a, local_best, local_score

def lp_update(a, n, fs, s0, top_t=30):

    conv = full_autoconv(a, fs)
    M = float(conv.max())
    # Get top-t lags
    idx = np.argpartition(-conv, min(top_t, len(conv)))[:top_t]
    idx = sorted(idx.tolist(), key=lambda k: -conv[k])
    
    # Build constraint matrix: t x n
    C = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            C[i, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    
    # LP: maximize sum(g) subject to C g <= M, eps <= g <= U
    eps_g = 1e-8
    U_g = 10.0 / n  # reasonable upper bound
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = np.vstack([C, np.eye(n), -np.eye(n)])
    b_ub = np.concatenate([np.full(len(idx), M), np.full(n, U_g), np.full(n, -eps_g)])
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 10})
    if res.status != 0 or res.x is None:
        return None
    
    g = np.clip(res.x, eps_g, 1000)
    g_sum = g.sum()
    if g_sum < 1e-14:
        return None
    return g / g_sum

    for k in idx:
        k = int(k)
        lo, hi = max(0, k - n + 1), min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    grad /= max(1, len(idx))
    return grad

def make_init(kind, n, rng, parent):
    x = np.arange(n) / max(1, n)
    if kind == 0:
        a = parent / parent.sum()
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        a = np.exp(-rng.uniform(1.0, 4.0) * x)
    elif kind == 3:
        a = np.abs(rng.standard_normal(n)) ** 2 + 0.01
    elif kind == 4:
        f = rng.uniform(0.1, 0.6)
        ph = rng.uniform(0, 6.28)
        a = np.clip(1 + 0.7 * np.sin(2 * np.pi * f * x + ph), 0.01, None)
    elif kind == 5:
        a = 1 / (1 + np.exp(rng.uniform(2, 8) * (x - 0.4)))
    elif kind == 6:
        c1, c2 = rng.uniform(0.15, 0.35), rng.uniform(0.6, 0.85)
        w = rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((x - c1) / w) ** 2) + 0.7 * np.exp(-0.5 * ((x - c2) / w) ** 2)
    elif kind == 7:
        a = 0.5 + 0.5 * np.cos(np.pi * x)
        a = np.abs(a + 0.1 * rng.standard_normal(n)) + 0.01
    else:
        a = 1.0 + 0.5 * rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    s = a.sum()
    if s < 1e-14:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def run_subgrad(a, n, fs, s0, best, best_sc, rng, deadline, tk):
    v = np.zeros(n)
    beta, lr = 0.82, 0.35
    prev_M, stall = np.inf, 0
    lb, lsc = best.copy(), best_sc
    for it in range(10**7):
        if time.monotonic() > deadline:
            break
        conv = fft_autoconv(a, fs)
        M = float(conv.max())
        stall = 0 if M < prev_M - 1e-18 else stall + 1
        prev_M = M
        if stall > 3000:
            break
        g = subgrad_step(a, conv, tk)
        v = beta * v + (1 - beta) * g
        v -= v.mean()
        a = np.clip(a - lr * v, 0, None)
        s = a.sum()
        if s < 1e-14:
            break
        a /= s
        lr = max(0.01, lr * 0.99992)
        if it % 800 == 0:
            cand = np.clip(a * s0, 0, 1000)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < lsc:
                lb, lsc = cand, sc
    return a, lb, lsc

def lp_refine(a, n, fs, s0, top_t, deadline):
    conv = fft_autoconv(a, fs)
    M = float(conv.max())
    idx = np.argpartition(-conv, min(top_t, conv.size))[:top_t]
    C = np.zeros((len(idx), n))
    for i, k in enumerate(idx):
        k = int(k)
        lo, hi = max(0, k - n + 1), min(n - 1, k)
        if lo <= hi:
            C[i, lo:hi+1] = a[k-hi:k-lo+1][::-1]
    eps_g, U_g = 1e-10, 2.0
    c_obj = -np.ones(n)
    A = np.vstack([C, np.eye(n), -np.eye(n)])
    b = np.concatenate([np.full(len(idx), M), np.full(n, U_g), np.full(n, -eps_g)])
    res = linprog(c_obj, A_ub=A, b_ub=b, method='highs',
                  options={'time_limit': 15, 'presolve': True})
    if res.status != 0:
        return None
    g = np.clip(res.x, 0, 1000)
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    parent = np.clip(height_sequence_1.astype(np.float64), 0, 1000)
    if parent.size < 2 or parent.ndim != 1 or not np.all(np.isfinite(parent)) or parent.sum() < 0.01:
        parent = np.ones(1024)
    n = parent.size

    fs = 1 << (2 * n - 1).bit_length()
    best = parent.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    
    # Phase 1: subgradient from diverse inits
    n_init = min(8, max(3, int((dl - time.monotonic()) / 120)))
    kinds = list(range(n_init))
    tks = [16, 8, 32, 12, 24, 6, 20, 10]
    for k in range(n_init):
        remain = dl - time.monotonic()
        if remain < 120:
            break
        a = make_init(kinds[k], n, rng, parent)
        phase_dl = dl - remain * 0.4
        a, lb, lsc = run_subgrad(a, n, fs, s0, best, best_sc, rng, phase_dl, tks[k % len(tks)])
        if lsc < best_sc:
            best, best_sc = lb, lsc
    
    # Phase 2: LP refinement
    lp_end = dl - (dl - time.monotonic()) * 0.3
    while time.monotonic() < lp_end:
        a = best / s0
        t_t = min(40, max(10, n // 100))
        g = lp_refine(a, n, fs, s0, t_t, dl)
        if g is None:
            break
        for alpha in [0.4, 0.7, 1.0]:
            a_new = (1 - alpha) * a + alpha * g
            a_new = np.clip(a_new, 0, None)
            if a_new.sum() < 1e-14:
                continue
            cand = np.clip(a_new * s0, 0, 1000)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best, best_sc = cand, sc
                a = a_new
    
    # Phase 3: Polish
    base = best.copy()
    while time.monotonic() < dl - 3:
        cand = base.copy()
        w = max(3, n // 80)
        for _ in range(rng.integers(1, 4)):

            cand[st:min(n, st+w)] *= float(np.exp(rng.normal(0, 0.035)))

            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best, base, best_sc = cand, cand, sc
    
    out = np.clip(best, 0, 1000)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
