import numpy as np

def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score_norm(conv, n):
    return 2.0 * conv.max() / n

def _subgrad(conv, a, n, topk, beta):
    kws = np.argpartition(conv, -topk)[-topk:]
    cv = conv[kws]
    w = np.exp(cv * beta - cv.max())
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(kws)):
        wi = w[i]
        if wi < 1e-12:
            continue
        k = int(kws[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[k - np.arange(lo, hi + 1)]
    return g
