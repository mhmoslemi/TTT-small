        je = min(n - 1, k)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += w * 2.0 * n * a[k - jj] / (S * S)
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n < 2:
        return [1.0, 1.0]

    a = parent.copy()
    best = a.copy()
    best_score = _score(a)

    lr = 200.0
    beta = 0.82
    mom = np.zeros(n)
    stall = 0
    restarts = 0
    K = 5

    while time.monotonic() < deadline:
        if deadline - time.monotonic() < 8.0:
            break

        conv = _fft_conv(a)
        S = float(np.sum(a))
        if S < 0.01:
            a = parent.copy()
            S = float(np.sum(a))

        M = float(np.max(conv))
        cur_score = 2.0 * n * M / (S * S)

        g = _multi_peak_grad(a, conv, K)
        if g is None:
            break

        mom = beta * mom + (1.0 - beta) * g
        direction = mom

        best_trial = cur_score
        best_a = a.copy()
        found = False

        for mult in (1.0, 0.5, 0.25, 0.1, 2.0):
            a_t = np.clip(a - lr * mult * direction, 0.0, 1000.0)
            st = _score(a_t)
            if np.isfinite(st) and st < best_trial:
                best_trial = st
                best_a = a_t
                found = True

        if found:
            a = best_a
            if best_trial < best_score:
                best = a.copy()
                best_score = best_trial
            lr = min(lr * 1.04, 3000.0)
            stall = 0
        else:
            stall += 1
            lr *= 0.8
            mom *= 0.4
            if stall > 55:
                scale = 0.04 * S / n
                a = np.clip(a + rng.normal(0.0, scale, n), 0.0, 1000.0)
                mom = np.zeros(n)
                lr = 200.0
                stall = 0
                restarts += 1
                if restarts > 5:
                    idx = np.arange(n, dtype=np.float64)
                    mod = 1.0 + 0.05 * np.sin(2.0 * np.pi * 3.0 * idx / n) \
                          + 0.03 * np.sin(2.0 * np.pi * 7.0 * idx / n)
                    a = np.clip(parent * mod, 0.0, 1000.0)
                    restarts = 0

    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        best = np.ones(n, dtype=np.float64)

    return [float(x) for x in best]
