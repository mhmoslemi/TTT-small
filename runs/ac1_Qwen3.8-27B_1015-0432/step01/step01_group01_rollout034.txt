            seg = a[k - hi : k - lo + 1]
            g[lo : hi + 1] += (w[i] / ws) * seg[::-1]
    return g

def norm1(a):
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-14:
        return None
    return a / s

def gen_init(kind, n, rng, parent):
    if kind == 0:
        a = parent.copy()
    elif kind == 1:
        a = np.abs(rng.standard_normal(n)) ** 0.5
    elif kind == 2:
        x = np.arange(n) / max(1, n)
        a = (0.5 * (1.0 + np.cos(2.0 * np.pi * x))) ** 0.7 + 0.1
    elif kind == 3:
        x = np.arange(n)
        a = 1.0 + 0.3 * np.cos(6.0 * np.pi * x / n) + 0.2 * np.cos(14.0 * np.pi * x / n)
        a = np.clip(a, 0.05, None)
    elif kind == 4:
        a = np.abs(np.sin(np.pi * np.arange(n) / n)) ** 0.4 + 0.1
    else:
        ctx = np.arange(n) / max(1, n)
        a = 0.8 + 0.4 * np.cos(10.0 * np.pi * ctx) + 0.2 * np.sin(22.0 * np.pi * ctx)
        a = np.clip(a, 0.05, None)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-14:
        a = np.ones(n)
        s = float(n)
    return a / s

def run_opt(a, n, fs, deadline, top_k, tf, lr0):
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    pa = a.copy()
    bM = np.inf
    st = 0
    while time.monotonic() < deadline:
        conv = fft_conv(a, fs)
        M = float(conv.max())
        if M < bM:
            bM = M
            pa = a.copy()
            st = 0
        else:
            st += 1
        if st > 3000:
            break
        g = subgrad(a, conv, n, top_k, tf)
        g -= g.mean()
        t += 1
        lr_t = lr0 / (1.0 + 5e-4 * t)
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * g * g
        a = a - lr_t * (m / (1.0 - 0.9 ** t)) / (np.sqrt(v / (1.0 - 0.999 ** t)) + 1e-8)
        a = np.clip(a, 0.0, None)
        a = norm1(a)
        if a is None:
            break
    return pa

def fft_score(a, n, fs):
    conv = fft_conv(a, fs)
    s = float(a.sum())
    if s < 1e-14:
        return np.inf
    return 2.0 * n * float(conv.max()) / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    T = max(0.0, float(budget_s) - 10.0)
    deadline = time.monotonic() + T
    parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if parent.size < 2 or parent.ndim != 1 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(1024, dtype=np.float64)
    n = int(parent.size)
    s0 = float(parent.sum())
    fs = 1 << (2 * n - 1).bit_length()
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    kinds = [0, 1, 2, 3, 4, 5]
    cfgs = [(16, 0.10, 0.20), (32, 0.05, 0.15), (8, 0.15, 0.25),
            (24, 0.07, 0.12), (12, 0.12, 0.18), (20, 0.06, 0.14)]
    for k in range(len(kinds)):
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        a0 = gen_init(kinds[k], n, rng, parent / s0)
        tk, tf, lr0 = cfgs[k]
        d1 = time.monotonic() + min(rem * 0.35, 200.0)
        pa = run_opt(a0, n, fs, d1, tk, tf, lr0)
        sc = fft_score(pa, n, fs)
        if sc < best_score * 1.03:
            rem2 = deadline - time.monotonic()
            if rem2 > 40:
                d2 = time.monotonic() + min(rem2 * 0.4, 150.0)
                pa2 = run_opt(pa, n, fs, d2, max(4, tk // 3), max(0.02, tf * 0.5), lr0 * 0.3)
                sc2 = fft_score(pa2, n, fs)
                if sc2 < sc:
                    pa, sc = pa2, sc2
        if sc < best_score:
            cand = np.clip(pa * s0, 0.0, 1000.0)
            sc_ex = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc_ex) and sc_ex < best_score:
                best = cand
                best_score = sc_ex
    if time.monotonic() < deadline - 8:
        base = best.copy()
        while time.monotonic() < deadline - 3:
            cand = base.copy()
            nb = int(rng.integers(1, 4))
            for _ in range(nb):
                w = max(2, n // int(rng.integers(40, 200)))
                st_ = int(rng.integers(0, max(1, n - w)))
                e = min(n, st_ + w)
                cand[st_ : e] *= float(np.exp(rng.normal(0.0, 0.025)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best = cand
                base = cand
                best_score = sc
    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
        if out.sum() < 0.01:
            out = np.ones(10)
    return [float(x) for x in out]
