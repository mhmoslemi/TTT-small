    deadline = time.monotonic() + max(0.5, float(budget_s) - 12.0)

    parent = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    n = parent.size
    s0 = float(parent.sum())
    if n < 2 or s0 < 0.01 or not np.all(np.isfinite(parent)):
        parent = np.ones(1024, dtype=np.float64)
        n = 1024
        s0 = float(n)
    fft_size = 1 << (2 * n - 1).bit_length()

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Phase 1: main descent from parent (70% of budget)
    t1 = deadline - 0.30 * (deadline - time.monotonic())
    a = parent / s0
    a, lb, ls = run_descent(a, n, fft_size, s0, best, best_score, t1, "parent")
    if ls < best_score:
        best_score, best = ls, lb

    # Phase 2: refined pass from best (20% of budget)
    if time.monotonic() < deadline - 0.20 * (deadline - time.monotonic()):
        t2 = deadline - 0.12 * (deadline - time.monotonic())
        a2 = best / s0
        a2, lb2, ls2 = run_descent(a2, n, fft_size, s0, best, best_score, t2, "refine")
        if ls2 < best_score:
            best_score, best = ls2, lb2

    # Phase 3: local perturbations
    base = best.copy()
    while time.monotonic() < deadline - 3.0:
        cand = base.copy()
        w = max(2, n // 60)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            cand[st:e] *= np.exp(float(rng.normal(0.0, 0.02)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score, best, base = sc, cand, cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.all(np.isfinite(out)) or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
