import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

    best_sc = fast_score(f)
    beta = 15.0
    eta = 0.12
    stall = 0

        if time.monotonic() > deadline - min_left:
            break
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        s = beta * b / M
        s -= s.max()
        w = np.exp(s)
        wm = w.max()
        if wm < 1e-30:
            break
        w /= wm
        gc = fft_conv(w, f[::-1])
        grad = np.maximum(gc[n - 1:2 * n - 1], 0.0)
        gm = grad.max()
        if gm < 1e-30:
            break

        f = np.maximum(f, 1e-30)
        f /= f.sum()
        if (it + 1) % ev == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-13:

                best_f = f.copy()
                stall = 0
