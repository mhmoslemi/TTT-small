import numpy as np
import time


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _clean_positive(v):
    v = np.asarray(v, dtype=np.float64)
    v = np.ravel(v)
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    v = v + 1e-14
    s = float(v.sum())
    if not np.isfinite(s) or s <= 0.0:
        v = np.ones(v.size, dtype=np.float64)
        s = float(v.size)
    return v / s


def _score_p(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.ravel(p)
    n = p.size
    if n == 0:
        return np.inf
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(p.sum())
    if s <= 1e-12:
        return np.inf
    L = 2 * n - 1
    N = _next_pow2(L)
    fac = np.fft.rfft(p, N)
    b = np.fft.irfft(fac * fac, N)[:L]
    b = np.where(np.isfinite(b), b, 0.0)
    mb = float(np.max(b))
    if mb <= 0.0:
        return 2.0
    return float(2.0 * n * mb / (s * s))


def _score_unit(p, N):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.inf
    fac = np.fft.rfft(p, N)
    b = np.fft.irfft(fac * fac, N)[:2 * n - 1]
    b = np.where(np.isfinite(b), b, 0.0)
    mb = float(np.max(b))
    if mb <= 0.0:
        return 2.0
    return float(2.0 * n * mb)


def _direct_exp(p, idx, w):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    e = np.zeros(n, dtype=np.float64)
    idx = np.asarray(idx, dtype=np.int64)
    w = np.asarray(w, dtype=np.float64)
    for ii in range(idx.size):
        k = int(idx[ii])
        wt = float(w[ii])
        if wt == 0.0:
            continue
        lo = k - n + 1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if lo <= hi:
            e[lo:hi + 1] += wt * p[k - hi:k - lo + 1][::-1]
    return e


def _exposure_fft(p, w, N):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    P = np.fft.rfft(p, N)
    W = np.fft.rfft(w[::-1], N)
    c = np.fft.irfft(W * P, N)
    return c[n - 1:2 * n - 1][::-1]


def _top_active(b, k, beta):
    L = b.size
    if L == 0 or k <= 0:
        return np.empty(0, dtype=np.int64), np.empty(0, dtype=np.float64)
    beta = float(beta)
    if k >= L:
        idx = np.arange(L, dtype=np.int64)
    else:
        idx = np.argpartition(b, L - k)[-k:].astype(np.int64)
    if idx.size == 0:
        return idx, np.empty(0, dtype=np.float64)
    order = np.argsort(-b[idx])
    idx = idx[order]
    vals = b[idx]
    maxv = float(vals[0])
    z = beta * (vals - maxv)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 0.0:
        w = np.ones(idx.size, dtype=np.float64) / idx.size
    else:
        w = w / sw
    return idx, w


def _greedy_support(n, m, rng):
    if n <= 1:
        return np.arange(n, dtype=np.int64)
    m = min(int(m), n)
    L = 2 * n - 1
    used = np.zeros(L, dtype=np.int32)
    selected = []
    xvals = np.arange(n, dtype=np.int64)
    for _ in range(m):

        for y in selected:
            score += used[y:y + n].astype(np.int64)
        for y in selected:

        scoref = score.astype(np.float64) + 1e-9 * np.arange(n, dtype=np.float64)

        selected.append(x)
        for y in selected[:-1]:

            lr = max(0.08, lr * 0.992)
        if it % 2500 == 0:
            K_target = 128
        if it % 1000 == 0 and init_pool:
            worst = 0
            worst_sc = init_pool[0][0]
            for kk in range(1, len(init_pool)):
                if init_pool[kk][0] > worst_sc:
                    worst = kk
                    worst_sc = init_pool[kk][0]
            if best_score < worst_sc - 1e-12:
                init_pool[worst] = (float(best_score), 9999, best.copy())

        if stall > 2500:
            choices = []
            for kk, entry in enumerate(init_pool):
                if entry[0] <= best_score * 1.5 + 1e-9:
                    choices.append(kk)
            alt = None
            if choices:
                alt = init_pool[choices[int(rng.integers(len(choices)))]][2]
            if alt is None:
                alt = best
            p_cur = np.asarray(alt, dtype=np.float64) * (1.0 + 0.01 * rng.standard_normal(n))
            p_cur = _clean_positive(p_cur)
            lr = max(0.35, lr)
            K = max(12, K // 2)
            beta_top = max(1.0, beta_top * 0.7)
            stall = 0

    if time.monotonic() < hard_deadline - 12.0:
        polish_deadline = hard_deadline - 5.0
        best, best_score = _polish_sequence(best, n, polish_deadline, rng)

    result = [float(x) for x in np.asarray(best, dtype=np.float64)]
    if not result:
        result = [1.0]

    try:
        es = float(evaluate_sequence(result))
        if not np.isfinite(es):
            result = [1.0]
    except Exception:
        pass

    return result
