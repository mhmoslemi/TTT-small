def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def project_positive(a, fallback):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        a = np.asarray(fallback, dtype=np.float64).copy()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        a = np.ones(max(1, len(a)), dtype=np.float64)
        s = float(a.size)

        return np.inf
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf

    b = np.fft.irfft(Af * Af, N)

    b = np.maximum(b, 0.0)
    mb = float(b.max())
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))

    b = np.fft.irfft(Af * Af, N)
    L = 2 * a.size - 1
    b = b[:L].astype(np.float64, copy=True)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    return b

def soft_grad(b, a, beta):
    """Compute soft-max gradient of max(b). Returns (2*h, bmax)."""
    n = a.size
    bmax = float(b.max())
    if bmax <= 0:
        return np.zeros(n, dtype=np.float64), 0.0
    w = np.exp(beta * (b - bmax))
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / ws
    # h[i] = sum_j a[j] * w[i+j]
    # h[i] = conv(a, w_rev)[2n-2-i] where w_rev = w[::-1]
    w_rev = w[::-1]
    N2 = next_pow2(3 * n - 1)
    Af = np.fft.rfft(a, N2)
    Wf = np.fft.rfft(w_rev, N2)
    conv = np.fft.irfft(Af * Wf, N2)
    h = conv[n - 1:2 * n - 1][::-1].copy()
    h = np.where(np.isfinite(h), h, 0.0)
    return 2.0 * h, bmax

def spectral_step(a, N, alpha):
    """Spectral flattening step."""
    n = a.size
    b = compute_conv(a, N)
    bs = float(b.sum())
    if bs <= 0 or not np.isfinite(bs):
        return a.copy()
    bn = b / bs
    bc = np.power(bn + 1e-15, alpha)
    sc = float(bc.sum())
    if not np.isfinite(sc) or sc <= 0:
        bc = bn
    else:
        bc = bc / sc
    T = np.fft.rfft(bc, N)
    # sqrt of T (element-wise, handling complex)
    mag = np.abs(T)
    ph = np.angle(T)
    smag = np.sqrt(mag)
    S = smag * np.exp(1j * ph)
    a2 = np.fft.irfft(S, N)
    a2 = np.real(a2[:n])
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    return project_positive(a2, a)

import numpy as np
import time

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def project_positive(a, fb):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        a = np.asarray(fb, dtype=np.float64).copy()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        a = np.ones(max(1, a.size), dtype=np.float64)
        s = float(a.size)
    return a / s

def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    Af = np.fft.rfft(a, N)
    b = np.fft.irfft(Af * Af, N)
    L = 2 * n - 1
    if b.size < L:
        return np.inf
    b = np.where(np.isfinite(b[:L]), b[:L], 0.0)
    b = np.maximum(b, 0.0)
    mb = float(b.max())
    if not np.isfinite(mb):
        return np.inf
    return float(2.0 * n * mb / (s * s))

def compute_conv(a, N):
    Af = np.fft.rfft(a, N)
    b = np.fft.irfft(Af * Af, N)
    L = 2 * a.size - 1
    b = b[:L].astype(np.float64, copy=True)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    return b

def soft_grad(a, b, beta, N2):
    n = a.size
    bmax = float(b.max())
    if bmax <= 0:
        return np.zeros(n, dtype=np.float64), 0.0
    w = np.exp(beta * (b - bmax))
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0:
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / ws
    w_rev = w[::-1]
    Af = np.fft.rfft(a, N2)
    Wf = np.fft.rfft(w_rev, N2)
    conv = np.fft.irfft(Af * Wf, N2)
    h = conv[n - 1:2 * n - 1][::-1].copy()
    h = np.where(np.isfinite(h), h, 0.0)
    return 2.0 * h, bmax

def spectral_step(a, N, alpha):
    n = a.size
    b = compute_conv(a, N)
    bs = float(b.sum())
    if bs <= 0 or not np.isfinite(bs):
        return a.copy()
    bn = b / bs
    bc = np.power(bn + 1e-15, alpha)
    sc = float(bc.sum())
    if not np.isfinite(sc) or sc <= 0:
        bc = bn
    else:
        bc = bc / sc
    T = np.fft.rfft(bc, N)
    mag = np.abs(T)
    ph = np.angle(T)
    a2 = np.fft.irfft(np.sqrt(mag) * np.exp(1j * ph), N)
    a2 = np.real(a2[:n])
    return project_positive(a2, a)
