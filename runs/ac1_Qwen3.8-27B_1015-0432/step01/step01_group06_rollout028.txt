def _fft_self_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _smooth_grad(y, eps, n):
    sz = 1
    while sz < 3 * n - 2:
        sz <<= 1
    corr = _fft_self_conv(y)
    c = corr - corr.max()

    fft_p = np.fft.rfft(p, sz)
