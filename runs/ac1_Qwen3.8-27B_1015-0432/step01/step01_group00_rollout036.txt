import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def gradient_descent(f, deadline, rng, min_time=5.0):

    n = f.size
    f = f.copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    
    best_f = f.copy()
    best_score = fast_score(f)
    
    beta = 20.0  # temperature for softmax
    eta = 0.1    # step size
    no_improve = 0
    patience = 2000
    
    it = 0
    while time.monotonic() < deadline - min_time:

        b = fft_conv(f, f)
        M = b.max()
        
        # Softmax weights focused on peaks
        scaled = beta * (b - M) / max(M, 1e-30)

        w = np.exp(scaled)
        w_sum = w.sum()
        
        if w_sum < 1e-30:
            break
        
        # Gradient
        f_r = f[::-1]
        gc = fft_conv(w, f_r)
        grad = gc[n-1:2*n-1]
        grad = np.maximum(grad, 0)
        gmax = grad.max()
        
        if gmax < 1e-30:
            break
        
        # Multiplicative update
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        
        # Adaptive parameters
        if it % 100 == 0:
            sc = fast_score(f)
            if sc < best_score - 1e-10:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > patience:
                # Restart with perturbation
                f = best_f * np.exp(0.1 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improve = 0
                eta = 0.1
                beta = 20.0
        
        if it % 2000 == 0:
            beta = min(200.0, beta * 1.05)
            eta = max(0.01, eta * 0.98)
    
    return best_f, best_score

import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_init(n, kind, rng):
    if kind == 0:  # uniform
        return np.ones(n) / n
    elif kind == 1:  # Hann
        x = np.arange(n)
        h = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
        h[0] = max(h[0], 1e-10)
        h[-1] = max(h[-1], 1e-10)
        return h / h.sum()
    elif kind == 2:  # power law
        x = np.arange(n)
        d = np.minimum(x, n - 1 - x).astype(float) + 1
        p = d ** -1.0
        return p / p.sum()
    elif kind == 3:  # Gaussian
        x = np.arange(n)
        c = (n - 1) / 2
        h = np.exp(-0.5 * ((x - c) / (n * 0.25))**2)
        return h / h.sum()
    elif kind == 4:  # symmetric random
        u = 1.0 + 0.5 * rng.standard_normal(n)
        u = np.maximum(u, 0.1)
        u = (u + u[::-1]) / 2
        return u / u.sum()
    elif kind == 5:  # cos^2
        x = np.arange(n)
        h = np.cos(np.pi * x / (n - 1))**2
        h[0] = max(h[0], 1e-10)
        h[-1] = max(h[-1], 1e-10)
        return h / h.sum()
    elif kind == 6:  # sinc-squared approximation
        x = np.arange(n)
        c = (n - 1) / 2
        t = (x - c) / (n - 1)
        h = np.sinc(t * 2)**2  # sinc(s) = sin(pi*s)/(pi*s)
        h = np.maximum(h, 1e-10)

    else:

        if w_sum < 1e-30:

        # Normalized weights (only for convolution, the actual sum matters for gradient scale)
        # Gradient: G[j] = sum_m f[m] * w[m+j]

        grad = gc[n-1:2*n-1].copy()
        grad = np.maximum(grad, 0.0)
