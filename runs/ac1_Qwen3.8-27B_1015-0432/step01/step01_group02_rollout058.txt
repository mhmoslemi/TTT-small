import numpy as np

def _fft_conv(a):
    n = len(a)
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _norm(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    a /= s
    return a


def _press(a, b, K):
    n = len(a)
    if K < len(b):
        idx = np.argpartition(b, -K)[-K:]
    else:
        idx = np.arange(len(b))
    h = np.zeros(n)
    for kj in idx:
        kj = int(kj)
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += a[kj - np.arange(lo, hi + 1)]
    return h
