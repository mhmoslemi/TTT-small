def _fft_conv(a):
    """Compute linear convolution a*a using FFT."""
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score(corr, n):
    """Compute the AC1 score from the convolution."""
    return 2.0 * n * corr.max() / (n * n)  # normalized by sum=n

    """Compute subgradient using top ntop active constraints."""
    cmax = corr.max()
    # Get indices where corr is close to max
    idx = np.argsort(corr)[-ntop:]
