from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = np.convolve(a, a)
    mv, _ = _smooth_max(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = np.convolve(a, a)
    mv, p = _smooth_max(c, tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = np.convolve(a, a)
    mv, _ = _smooth_max(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = np.convolve(a, a)
    mv, p = _smooth_max(c, tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _coord_pass(a, rng, n):
    b = np.convolve(a, a)
    pm = np.maximum.accumulate(b)
    sm = np.maximum.accumulate(b[::-1])[::-1]
    S = a.sum()
    improved = 0
    order = rng.permutation(n)
    fracs = (0.0, 0.25, 0.5, 0.75, 0.9, 1.1, 1.4)
    for i in order:
        old = a[i]
        k0 = i
        k1 = min(i + n, 2 * n - 1)
        aff_len = k1 - k0
        if aff_len <= 0:
            continue
        left_m = pm[k0 - 1] if k0 > 0 else -1e30
        right_m = sm[k1] if k1 < len(b) else -1e30
        out_m = left_m if left_m > right_m else right_m
        aff = b[k0:k1]
        wa = a[:aff_len]
        cur_nm = aff.max()
        if out_m > cur_nm:
            cur_nm = out_m
        best_sy = 2.0 * n * cur_nm / (S * S)
        best_tv = old
        if old > 1e-14:
            for fr in fracs:
                tv = old * fr
                if tv > 1000.0:
                    continue
                d = tv - old
                if abs(d) < 1e-14:
                    continue
                ns = S + d
                if ns < 0.01:
                    continue
                nm = (aff + 2.0 * d * wa).max()
                if out_m > nm:
                    nm = out_m
                sy = 2.0 * n * nm / (ns * ns)
                if sy < best_sy - 1e-15:
                    best_sy = sy
                    best_tv = tv
        else:
            tv = S / (20.0 * n)
            if tv > 1e-6 and tv <= 1000.0:
                d = tv
                ns = S + d
                nm = (aff + 2.0 * d * wa).max()
                if out_m > nm:
                    nm = out_m
                sy = 2.0 * n * nm / (ns * ns)
                if sy < best_sy - 1e-15:
                    best_sy = sy
                    best_tv = tv
        if best_tv != old:
            a[i] = max(0.0, min(1000.0, best_tv))
            improved += 1
    return improved


def _lgb_step(cur, best, best_sc, tv, n):
    try:
        res = minimize(
            _obj, cur, args=(tv,), jac=_grad,
            method='L-BFGS-B',
            bounds=[(0.0, 1000.0)] * n,
            options={'maxiter': 400, 'gtol': 1e-10,
                     'maxls': 40, 'ftol': 1e-15})
        old_obj = _obj(cur, tv)
        if res.fun < old_obj - 1e-13:
            nx = np.clip(res.x, 0.0, 1000.0)
            nxs = nx.sum()
            if nxs > 1e-8:
                nx = nx / nxs
            sc = float(evaluate_sequence(nx.tolist()))

                cur = nx.copy()
            elif np.isfinite(sc) and sc < _obj(cur, tv):
                cur = nx.copy()
    except Exception:
        pass
    return cur, best, best_sc


def _local_search(bv, sc, rng, n, dl):
    wmin = max(1, n // 400)
    wmax = max(wmin + 2, n // 12)
    stall_count = 0
    pert_scale = 1.0
    while time.monotonic() < dl:
        mode = rng.random()
        cd = bv.copy()
        if mode < 0.35:
            w = int(rng.integers(wmin, wmax))
            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.04 * pert_scale)))
        elif mode < 0.55:
            i = int(rng.integers(0, n))
            if cd[i] > 1e-10:
                cd[i] *= float(np.exp(rng.normal(0.0, 0.12 * pert_scale)))
            else:
                cd[i] = abs(float(rng.normal(0.0, max(1e-5, bv.max() * 0.02))))
        elif mode < 0.7:
            w = int(rng.integers(1, max(2, n // 250)))
            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] *= 0.05 + 0.5 * rng.random()
        elif mode < 0.85:
            w1 = int(rng.integers(wmin, wmax))
            w2 = int(rng.integers(wmin, wmax))
            s1 = int(rng.integers(0, max(1, n - w1)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(0.0, 0.05 * pert_scale)))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.0, 0.05 * pert_scale)))
        else:
            w = int(rng.integers(max(4, 2 * wmin), max(8, wmax)))
            s = int(rng.integers(0, max(1, n - w)))
            h = w // 2
            tmp1 = cd[s:s + h].copy()
            cd[s:s + h] = cd[s + h:s + w]
            cd[s + h:s + w] = tmp1
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() < 0.01:
            continue
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc - 1e-12:
            bv = cd.copy()
            sc = s2
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 600:
                pert_scale = min(3.0, pert_scale * 1.5)
                stall_count = 0
    return bv, sc
