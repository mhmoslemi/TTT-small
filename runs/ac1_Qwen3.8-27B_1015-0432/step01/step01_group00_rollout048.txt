import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    f = np.maximum(f.copy(), 1e-20)
    f /= f.sum()
    b = fft_conv(f, f)
    M = b.max()
    target = M * 0.995

    for _ in range(200):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        peak_idx = np.where(b >= M * 0.999)[0]
        if peak_idx.size == 0:
            break

        for ki in peak_idx[:8]:
            shift = ki - (n - 1)
            lo = max(0, shift)
            hi = min(n, ki + 1)
            seg = f[lo:hi]
            if seg.sum() < 1e-20:
                continue
            reduce = np.minimum(seg, f[lo:hi] * 0.97)
            diff = seg - reduce
            f[lo:hi] = reduce
            spread = np.clip(np.arange(-n + 1, n), 0, n - 1)
            f += diff.sum() * spread / (2 * (n - 1) * n)

        f = np.maximum(f, 1e-20)
        f /= f.sum()

    return f


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    n = len(height_sequence_1)

    inits = make_inits(n, rng)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    if best_f is None:
        best_f = np.ones(n)
        best_sc = 2.0

    f = best_f.copy()
    while time.monotonic() < deadline - 20.0:
        local_dl = min(deadline - 5.0, time.monotonic() + 180.0)
        f_opt, opt_M = run_descent(f, local_dl, rng)
        sc_opt = fast_score(f_opt)
        if np.isfinite(sc_opt) and sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()

        if time.monotonic() < deadline - 30.0:
            f_flat = run_flatten(best_f, deadline - 10.0, rng)
            sc_flat = fast_score(f_flat)
            if np.isfinite(sc_flat) and sc_flat < best_sc:
                best_sc = sc_flat
                best_f = f_flat.copy()

        f = best_f.copy()

        n_perturb = max(5, int((deadline - time.monotonic()) * 0.5))
        for i in range(n_perturb):
            if time.monotonic() > deadline - 20.0:
                break
            scale = 0.02 + 0.03 * (i % 5)
            trial = best_f * np.exp(scale * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = [float(x) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(1.0 / n)] * n
    return result
