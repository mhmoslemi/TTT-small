    return np.fft.irfft(fa*fa, sz)[:2*n-1]

            sc_try = float(_score_norm(_fft_conv(a_try), n))
            if sc_try < sc - 1e-12:
                a = a_try
                accepted = True
                break

        if accepted:
            lr = min(lr * 1.01, 2.5)
        else:
            lr = max(lr * 0.72, 0.004)

        if it % 150 == 0:
            sc_exact = float(evaluate_sequence(a.tolist()))
            if sc_exact < best_sc - 1e-12:
                best_sc = sc_exact
                best = a.copy()
                stuck = 0

        # Periodic dual step to escape local minima
        if stuck > 300 and time.monotonic() < deadline - 30.0:
            M = float(corr.max())
            g_dual = _dual_step(a, n, M, 8, 0.5)
            sc_dual = float(_score_norm(_fft_conv(g_dual), n))
            t_blend = min(1.0, 0.3 + 0.1 * (stuck // 300))
            a_new = (1.0 - t_blend) * a + t_blend * g_dual
            a_new = np.clip(a_new, 0.0, 1000.0)
            sn = a_new.sum()
            if sn > 1e-10:
                a_new = a_new * n / sn
                sc_new = float(_score_norm(_fft_conv(a_new), n))
                if sc_new < sc:
                    a = a_new
                    v[:] = 0.0
                    lr = 0.12
                stuck = 0

        if stuck > 800:
            a = np.clip(a + rng.standard_normal(n) * 0.08 * n, 0.0, 1000.0)
            a = a * n / a.sum()
            v[:] = 0.0
            lr = 0.12
            stuck = 0
        elif stuck > 1500:
            kind = int(rng.integers(0, 9))
            a = _make_init(n, kind, rng)
            sc_new = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_new) and sc_new < best_sc:
                best = a.copy()
                best_sc = sc_new
            v[:] = 0.0
            lr = 0.12
            stuck = 0

    # Final local search
    for _ in range(50):
        if time.monotonic() > deadline - 3.0:
            break
        a_try = np.clip(best + rng.standard_normal(n) * 0.008 * n, 0.0, 1000.0)
        a_try = a_try * n / a_try.sum()
        sc_try = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(sc_try) and sc_try < best_sc:
            best = a_try.copy()
            best_sc = sc_try

    out = np.clip(best, 0.0, 1000.0)
    final_sc = float(evaluate_sequence(out.tolist()))
    if not np.isfinite(final_sc):
        out = np.ones(n)
    return [float(x) for x in out]
