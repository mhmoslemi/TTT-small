def _subgrad(corr, a, ntop):
    n = a.size
    idx = np.argsort(corr)[-ntop:]
    thresh = corr[int(idx[0])]
    sel = [int(k) for k in idx if corr[k] >= thresh]
    if not sel:
        sel = [int(np.argmax(corr))]
    g = np.zeros(n)
    for k in sel:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi+1)]
    g /= len(sel)
    return g

import numpy as np
import time

def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _score(corr, n):
    return 2.0 * float(corr.max()) / n

def _subgrad(corr, a, ntop):
    n = a.size
    idx = np.argsort(corr)[-ntop:]
    thresh = float(corr[int(idx[0])])
    sel = [int(k) for k in idx if corr[k] >= thresh]
    if not sel:
        sel = [int(np.argmax(corr))]
    g = np.zeros(n)
    for k in sel:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
    g /= len(sel)
    return g

def _normalize(a, n):

    s = a.sum()
    if s < 1e-10:
        return np.ones(n)
    return a * n / s
