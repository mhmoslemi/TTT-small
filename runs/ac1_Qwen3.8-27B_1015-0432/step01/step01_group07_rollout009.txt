import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _grad_smooth(a, conv, n, S, K):
    M = float(conv.max())
    idx = np.argpartition(conv, -K)[-K:]
    idx = np.sort(idx)
    w = conv[idx]
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    h = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i])
        wk = float(w[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            jj = np.arange(js, je + 1)
            h[jj] += wk * a[k - jj]
    return (4.0 * n / (S * S)) * (h - M / S)
