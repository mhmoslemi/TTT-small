from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _pow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _fconv(a, b):
    L = _pow2(len(a) + len(b) - 1)
    return ifft(fft(a, L) * fft(b, L)).real[:len(a) + len(b) - 1]

def _fconv2(a):
    L = _pow2(2 * len(a) - 1)
    F = fft(a, L)
    return ifft(F * F).real[:2 * len(a) - 1]

def _sm_and_p(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _fconv2(a)
    sm, _ = _sm_and_p(c, tau)

import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _pow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _fconv2(a):
    L = _pow2(2 * len(a) - 1)
    F = fft(a, L)
    return ifft(F * F).real[:2 * len(a) - 1]


def _sm_p(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _fconv2(a)
    sm, _ = _sm_p(c, tau)
    return 2.0 * n * sm / (S * S)


def _obj_grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _fconv2(a)
    sm, p = _sm_p(c, tau)
    pr = p[::-1]
    L = _pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real[:3 * n - 2]
    r = cv[n - 1:2 * n - 1][::-1]
    f = 2.0 * n * sm / (S * S)
    g = 4.0 * n * r / (S * S) - 4.0 * n * sm / (S ** 3)
    return f, g


def _refine(a, taus, n, dl, maxit=200):
    best = a.copy()
    best_s = float(evaluate_sequence(best.tolist()))
    for tv in taus:
        if time.monotonic() > dl:
            break
        try:
            tgt = _obj(a, tv)
            res = minimize(
                _obj, a, args=(tv,), jac=_obj_grad,
                method='L-BFGS-B',
                bounds=[(0.0, 1000.0)] * n,
                options={'maxiter': maxit, 'gtol': 1e-9,
                         'maxls': 40, 'ftol': 1e-14})
            if res.fun < tgt - 1e-13:
                nx = np.clip(res.x, 0.0, 1000.0)
                nxs = nx.sum()
                if nxs > 1e-8:
                    nx = nx / nxs
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_s:
                    best = nx.copy()
                    best_s = sc
                a = nx.copy()
        except Exception:
            pass
    return best, best_s


def _perturb_freq(a, rng, str_):
    n = a.size
    L = _pow2(n)
    A = fft(a, L)
    mask = rng.random(L) < 0.25
    A[mask] *= 1.0 + str_ * rng.normal(0, 1, int(mask.sum()))
    return np.clip(ifft(A).real[:n], 0.0, 1000.0)


def _perturb_block(a, rng):
    n = a.size
    cd = a.copy()
    w = max(1, int(rng.integers(n // 400, n // 25)))
    s = int(rng.integers(0, n - w))
    cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.03)))
    if rng.random() < 0.3:
        cd *= float(np.exp(rng.normal(0.0, 0.005)))
    return np.clip(cd, 0.0, 1000.0)


def _peak_shave(a, dl):
    n = a.size
    best = a.copy()
    best_s = float(evaluate_sequence(best.tolist()))
    eps = 1e-3 / n
    for _ in range(500):
        if time.monotonic() > dl - 2.0:
            break
        c = _fconv2(best)
        k = int(np.argmax(c))
        # gradient of c[k] w.r.t. a[i] = 2*a[k-i] if 0<=k-i<n
        g = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            g[i] = 2.0 * a[k - i]
        cd = best - eps * g
        cd = np.clip(cd, 0.0, 1000.0)
        cds = cd.sum()

            continue
        sc = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(sc) and sc < best_s:

            best_s = sc
    return best, best_s


def _local(bv, sc, rng, n, dl):

        if cd.sum() < 0.01:
            continue
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc:
            bv = cd.copy()
            sc = s2
    return bv, sc
