    cols = []
    vals = []

        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            rows.append(k)  # constraint index
            cols.append(j)
            vals.append(a[k - j])

import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog

def _fft_conv(a, b=None):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    if b is None:
        return np.fft.irfft(fa * fa, sz)[:2 * n - 1]
    return np.fft.irfft(fa * np.fft.rfft(b, sz), sz)[:2 * n - 1]

def _subgrad(c, a, n, ntop):
    idx = np.argsort(c)[::-1][:ntop]
    g = np.zeros(n)
    cnt = 0
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - np.arange(lo, hi + 1)]
            cnt += 1
    if cnt:
        g /= cnt
    return g

def _build_A(a, active, n):
    m = len(active)
    R, C, V = [], [], []
    for i, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        jj = np.arange(lo, hi + 1)
        R.append(np.full(jj.size, i, dtype=np.int32))
        C.append(jj.astype(np.int32))
        V.append(a[k - jj])
    if not R:
        return scipy.sparse.csr_matrix((m, n))
    return scipy.sparse.csr_matrix(
        (np.concatenate(V), (np.concatenate(R), np.concatenate(C))), shape=(m, n))
