    g = np.zeros(n, dtype=np.float64)
    for lag in idx:
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            g[lo:hi + 1] += a[lag - hi:lag - lo + 1][::-1]
    return g / float(tk)

def score_normalized(a, fs):
    """Score for a with sum=1: 2*n*max(conv(a,a))"""
    n = a.size
    conv = fft_conv(a, fs)
    return 2.0 * n * float(conv.max())

def make_init(n, rng, kind, parent):
    if kind == 0 and parent is not None:
        a = parent.copy()
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        f = rng.uniform(0.2, 0.7)
        ph = rng.uniform(0, 2*np.pi)
        a = np.clip(1.0 + 0.5*np.sin(2*np.pi*f*np.arange(n)/n + ph), 0.01, None)
    elif kind == 3:
        # Bowl shape
        x = np.linspace(-1, 1, n)
        a = 1.0 + 0.8*x*x + 0.1*rng.standard_normal(n)
        a = np.clip(a, 0.01, None)
    elif kind == 4:
        # Two-peak
        c = rng.integers(n//4, 3*n//4)
        x = np.abs(np.arange(n) - c) / n
        a = np.exp(-x*x*8) + 0.3*np.exp(-((np.arange(n)-0)*(n-0)/n)**2 * 8)
        a += 0.05*np.abs(rng.standard_normal(n))
    elif kind == 5:
        # Smooth random via low-freq FFT
        spec = np.abs(rng.standard_normal(n//2+1)) + 0.5
        spec[len(spec)//2:] *= 0.3  # low-pass
        a = np.fft.irfft(spec, n=n)
        a = np.clip(a - a.min() + 0.1, 0.01, None)
    else:
        a = 1.0 + 0.5*np.cos(2*np.pi*rng.integers(1, n//2)*np.arange(n)/n)
        a = np.clip(a + 0.05*np.abs(rng.standard_normal(n)), 0.01, None)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n)
    return a / a.sum()

def descent(a, n, fs, deadline, tk_init, tk_min, it_max, rng):
    """Momentum subgradient descent with cooling top_k."""
    v = np.zeros(n, dtype=np.float64)
    beta = 0.85
    lr = 0.5
    prev_M = np.inf
    stall = 0
    best_local = a.copy()
    best_local_sc = score_normalized(a, fs)
    
    for it in range(it_max):
        if time.monotonic() > deadline:
            break
        frac = min(1.0, it / max(1, it_max * 0.8))
        tk = max(tk_min, int(tk_init * (1.0 - 0.9 * frac)))
        
        conv = fft_conv(a, fs)
        M = float(conv.max())
        sc = 2.0 * n * M
        if sc < best_local_sc:
            best_local_sc = sc
            best_local = a.copy()
        
        if M < prev_M - 1e-13:
            stall = 0
        else:
            stall += 1
        prev_M = M
        if stall > 2500:
            # Try a kick
            if rng.random() < 0.3:
                a = a + 0.01 * rng.standard_normal(n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                stall = 0
                continue
            else:
                break
        
        g = peak_subgrad(a, conv, tk)
        v = beta * v + (1.0 - beta) * g
        v -= v.mean()
        
        a = a - lr * v
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            a = best_local.copy()
            break
        a = a / s
        
        lr = max(0.03, lr * 0.99992)
    
    return best_local, best_local_sc

One more thought: the LP approach from the state-of-the-art is interesting. The idea is:
- Given current f, find g >= 0 that maximizes sum(g) subject to conv(f*g) <= M (current peak)
- This gives a direction that "spreads out" the convolution
