def fft_conv(a, b):
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:na + nb - 1]


def fast_score(f):
    s = f.sum()

        return np.inf

    return float(2 * len(f) * b.max() / (s * s))
