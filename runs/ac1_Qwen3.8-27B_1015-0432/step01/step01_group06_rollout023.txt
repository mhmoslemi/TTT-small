    sz = 1

    return np.fft.irfft(fa*fa, sz)[:2*n-1]

def _subgrad(c, a, n, K):

    mx = c.max()
    # Find positions where c is close to max
    thresh = mx - 1e-10 * max(mx, 1.0)

    for v in range(9):
        if time.monotonic() > dl - 20:
            break
        a_try = _gen_init(n, rng, v)
        sc = 2.0 * _fft_conv(a_try).max() / n
        if sc < best_sc:
            best_sc = sc
            best = a_try.copy()

    a = best.copy()
    sc = best_sc
    v = np.zeros(n)
    lr = 0.3
    stuck = 0
    K = min(16, max(4, n // 256))
    beta = 0.75
    it = 0

    while time.monotonic() < dl:
        it += 1
        c = _fft_conv(a)
        sc = 2.0 * c.max() / n
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 2000 == 0:
            sc_e = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_e) and sc_e < best_sc:
                best_sc = sc_e
                best = a.copy()
                stuck = 0

        g = _subgrad(c, a, n, K)
        g -= g.mean()
        v = beta * v + (1.0 - beta) * g
        vn = np.linalg.norm(v)
        if vn < 1e-15:
            break
        d = -v / vn

        ok = False
        for f in (1.0, 0.6, 0.4, 0.2, 0.1):
            a_new = a + lr * f * d
            a_new = np.clip(a_new, 0.0, 1000.0)
            sn = a_new.sum()
            if sn < 1e-10:
                a_new = np.ones(n)
            else:
                a_new = a_new * n / sn
            sc_new = 2.0 * _fft_conv(a_new).max() / n
            if sc_new < sc - 1e-14:
                a = a_new
                ok = True
                break

        if ok:
            lr = min(lr * 1.005, 3.0)
        else:
            lr *= 0.85
            if lr < 0.01:
                if stuck > 2000:
                    nv = int(rng.integers(0, 9))
                    a = _gen_init(n, rng, nv)
                    sc_chk = 2.0 * _fft_conv(a).max() / n
                    if sc_chk < best_sc:
                        best_sc = sc_chk
                        best = a.copy()
                    else:
                        a = best.copy()
                        sc = best_sc
                    v[:] = 0.0
                    lr = 0.3
                    stuck = 0
                elif stuck > 500:
                    a = np.clip(a + rng.normal(0, 0.02 * max(a.std(), 0.1), n), 0, 1000)
                    a = a * n / a.sum()
                    v[:] = 0.0
                    lr = 0.3
                    stuck = 0

    if time.monotonic() < dl:
        sc_e = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc_e) and sc_e < best_sc:
            best_sc = sc_e
            best = a.copy()

    par_sc = float(evaluate_sequence(par.tolist()))
    if np.isfinite(par_sc) and par_sc < best_sc:
        best = par.copy()

    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n)
    return [float(x) for x in best]
