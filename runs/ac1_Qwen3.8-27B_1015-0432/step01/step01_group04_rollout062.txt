def np2(x):
    m = 1
    while m < x:
        m <<= 1
    return m

def proj(a, fb):
    a = np.clip(np.where(np.isfinite(a), a, 0.0), 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.clip(np.where(np.isfinite(fb), fb, 0.0), 0.0, 1000.0)
        s = float(a.sum())
    if s < 0.01:
        a = np.ones(max(1, a.size), dtype=np.float64)
        s = float(a.size)
    return a / s

def sc_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[: 2 * n - 1]
    mc = float(np.maximum(c, 0.0).max())
    if not np.isfinite(mc):
        return np.inf
    return 2.0 * n * mc / (s * s)

def conv_aa(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    n = a.size
    return np.maximum(c[: 2 * n - 1], 0.0)

def grad_G(a, z, N):
    n = a.size
    a_r = a[::-1]
    conv = np.fft.irfft(np.fft.rfft(z, N) * np.fft.rfft(a_r, N), N)
    return 2.0 * conv[n - 1 : 2 * n - 1]
