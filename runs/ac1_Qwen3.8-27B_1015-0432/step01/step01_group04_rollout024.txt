        return a.copy()
    c = c / cs
    ct = np.power(c + 1e-15, alpha)
    T = np.fft.rfft(ct, N)
    T = np.where(np.isfinite(T), T, 0.0)
    T = np.maximum(T, 0.0)
    a2 = np.fft.irfft(np.sqrt(T), N)[:n]
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    a2 = np.maximum(a2, 0.0)
    s = float(a2.sum())
    return a2 / s if s > 0 else a.copy()
