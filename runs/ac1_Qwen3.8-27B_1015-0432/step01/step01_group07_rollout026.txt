        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2*n-1]

    a = np.clip(np.array(height_sequence_1, dtype=float), 0, 1000).astype(float)
    
    # Normalize to sum=1 (scale invariance)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(a)
    
    # Phase 1: LP-based optimization
    consecutive_failures = 0
    while time.monotonic() < deadline - 120:
        conv_a = _fft_conv(a)
        M = float(conv_a.max())
        
        b = _lp_step(a, M, time_limit=3)
        if b is None:
            consecutive_failures += 1
            if consecutive_failures > 5:
                break
            continue
        
        consecutive_failures = 0
        b = np.clip(b, 0, None)
        Sb = b.sum()
        if Sb < 0.01:
            break
        b = b / Sb  # normalize
        
        # Mix
        t = 0.5  # mixing parameter
        a_new = (1-t)*a + t*b
        a_new = np.clip(a_new, 0, None)
        S_new = a_new.sum()
        if S_new < 0.01:
            break
        a_new /= S_new
        
        score_new = _score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            # Try increasing t
        else:
            a = a_new  # still accept (non-increasing peak)
    
    # Phase 2: Subgradient fine-tuning
    lr = 100.0 / n
    K = 5
    momentum = np.zeros(n)
    stall = 0
    while time.monotonic() < deadline:
        conv = _fft_conv(a)
        top_K = np.argpartition(conv, -K)[-K:]
        M = float(conv.max())
        S = float(a.sum())
        
        # Subgradient of max_k (a*a)[k]
        g = np.zeros(n)
        for k in top_K:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                jj = np.arange(js, je+1)
                g[jj] += 2.0 * a[k - jj]
        g /= K
        
        # Subgradient of 2n*M/S^2
        h = (2.0*n/S**2) * g - (4.0*n*M/S**3)
        
        # Momentum
        momentum = 0.85 * momentum - lr * h
        a = np.clip(a + momentum, 0, None)
        a /= a.sum()
        
        trial_score = _score(a)
        if trial_score < best_score:
            best_score = trial_score
            best_a = a.copy()
            stall = 0
            lr *= 1.02
        else:
            stall += 1
            if stall > 100:
                a = best_a + rng.normal(0, 0.5/n, n)
                a = np.clip(a, 0, None)
                a /= a.sum()
                momentum = np.zeros(n)
                stall
