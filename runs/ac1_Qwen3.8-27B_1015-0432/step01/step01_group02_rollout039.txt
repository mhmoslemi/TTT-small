import numpy as np
import time


def _fft_conv_sq(a):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real[:2 * n - 1]


def _fft_corr(w, a):
    nw, na = len(w), len(a)
    N = 1
    while N < nw + na - 1:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(W * np.conj(A), N).real[:na]


def _score(a):
    c = _fft_conv_sq(a)
    return 2.0 * len(a) * c.max()

    for ic in inits[1:]:
