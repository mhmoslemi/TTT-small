import numpy as np
import time

def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real

def _score(a, n):
    return 2.0 * n * _conv(a, n).max()

def _update(a, n, topK, alpha):
    c = _conv(a, n)
    sc = 2.0 * n * c.max()
    if topK < len(c):
        idx = np.argpartition(c, -topK)[-topK:]
    else:
        idx = np.arange(len(c))
    px = c[idx]
    pm = px.max()
    if pm > 1e-15:
        wts = (px / pm) ** 2
        wts /= wts.sum()
    else:
        wts = np.ones(len(idx)) / len(idx)
    h = np.zeros(n)
    for j in range(len(idx)):
        w = wts[j]
        if w < 1e-12:
            continue

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            h[lo:hi + 1] += w * a[k - hi:k - lo + 1][::-1]
    hm = h.max()
    if hm > 1e-15:
        a = a * np.exp(-alpha * h / hm)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-12:
        a /= s
    else:
        a = np.ones(n) / n
    return a, sc
