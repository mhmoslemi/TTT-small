            cand = (1.0 - t_mix) * best_f + t_mix * g_lp
            cand = np.maximum(cand, 1e-20)
            cand = sym(cand)
            cand /= cand.sum()
            sc = score_func(cand)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_f = cand.copy()
        phase += 1
        scale = max(0.02, 0.06 * np.exp(-0.25 * phase))
        f = best_f * np.exp(scale * rng.standard_normal(n))
        f = np.maximum(f, 1e-20)
        f = sym(f)
        f /= f.sum()
    if time.monotonic() < deadline - 5.0:
        f_fin, sc_fin = grad_opt(best_f, deadline - 3.0, rng, beta_init=80.0, eta_init=0.04)
        if sc_fin < best_sc:
            best_sc = sc_fin
            best_f = f_fin.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * n
    return result
