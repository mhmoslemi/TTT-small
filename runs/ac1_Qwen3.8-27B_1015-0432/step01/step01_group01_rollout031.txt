            hi = min(n - 1, k)
            data[p:p + lengths[k]] = a[lo:hi + 1]
            p += lengths[k]
        A = sparse.csr_matrix((data, cols, indptr.copy()), shape=(2 * n - 1, n))
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      bounds=[(0, None)] * n,
                      options={'time_limit': 25, 'presolve': True})
        if res.status != 0 or res.x is None:
            break
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb < 1e-14:
            break
        b = b / sb
        for tstep in [0.3, 0.15]:
            trial = (1 - tstep) * a + tstep * b
            trial = project(trial)
            cand = np.clip(trial * s0, 0, 1000)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = trial.copy()
                a = trial
                break
    return best_a, best_sc

def polish(base, best_score, n, deadline, rng):
    b = base.copy()
    bs = best_score
    w = max(3, n // 60)
    while time.monotonic() < deadline:
        c = b.copy()
        nb = int(rng.integers(1, 4))
        for _ in range(nb):
            st = int(rng.integers(0, max(1, n - w)))
            c[st:st + w] *= float(np.exp(rng.normal(0, 0.025)))
        c = np.clip(c, 0, 1000)
        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < bs:
            b = c.copy()
            bs = sc
    return b, bs
