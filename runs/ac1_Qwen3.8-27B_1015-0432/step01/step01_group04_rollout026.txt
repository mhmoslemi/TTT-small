3. Compute g = 2 * (a ⋆ p) (cross-correlation, via FFT)
4. Compute L = sum(p * b) (or logsumexp)
5. Full gradient: grad[j] = 2n/S^2 * (2*g[j] - 2*L/S)
6. Project to tangent of simplex: grad_proj = grad - mean(grad) (or in log-space)
7. Update

For the log-space parameterization:
a_j = exp(y_j) / sum(exp)
da_j/dy_k = a_j * (delta_jk - a_k)

So dy_k = sum_j (df/da_j) * (da_j/dy_k) = sum_j grad_j * a_j * (delta_jk - a_k)
= a_k * grad_k - a_k * sum_j(a_j * grad_j)
= a_k * (grad_k - sum(a*grad))

This is the "natural gradient" or "Fisher gradient".

For Adam, I apply this in log-space.

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def clip_nn(a):
    a = np.zeros_like(a)
    np.clip(a, 0, 1000, out=a)
    a[~np.isfinite(a)] = 0
    return a

def score_a(a, N):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = np.maximum(c[:L], 0)
    return 2.0 * n * c.max() / (s * s)

    n = a.size
    # Normalize
    s = a.sum()
    a = a / s
    # Log-space parameterization
    a = np.clip(a, 1e-20, None)
    y = np.log(a)
    y -= y.mean()
    
    m1 = np.zeros(n)
    v1 = np.zeros(n)
    t = 0
    b1, b2 = 0.9, 0.999
    lr = 0.05
    
    best = a.copy()
    best_sc = score_a(a, N)
    no_imp = 0
    gamma = 2.0
    beta = 10.0
    
    while time.monotonic() < deadline:
        t += 1
        # Compute convolution
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2*n-1]
        c = np.maximum(c, 0)
        max_c = c.max()
        
        # Weighted p
        if max_c > 0:
            p = np.power(c / max_c, gamma)
            p /= p.sum()
        else:
            p = np.full(2*n-1, 1.0/(2*n-1))
        
        # Cross-correlation: g[j] = sum_m a[m] * p[j+m]
        N_g = N2
        A_g = np.fft.rfft(a, N_g)
        P_g = np.fft.rfft(p, N_g)
        cross = np.fft.irfft(np.conj(A_g) * P_g, N_g)
        g = 2.0 * cross[:n]
        
        # Full gradient
        L = np.dot(p, c)
        sc = 2.0 * n * max_c  # since S=1
        grad = (g - L)  # proportional to actual gradient
        
        # Project to tangent (sum = 0)
        grad -= grad.mean()
        
        # Adam update in log-space
        dy = a * grad
        m1 = b1 * m1 + (1 - b1) * dy
        v1 = b2 * v1 + (1 - b2) * dy * dy
        m1_hat = m1 / (1 - b1**t)
        v1_hat = v1 / (1 - b2**t)
        step = lr * m1_hat / (np.sqrt(v1_hat) + 1e-8)
        y -= step
        y -= y.mean()
        y = np.clip(y, -50, 50)
        
        # Convert back
        a = np.exp(y - y.max())
        a /= a.sum()
        
        sc = score_a(a, N)
        if np.isfinite(sc) and sc < best_sc:
            best = a.copy()
            best_sc = sc
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp > 500:
            # Restart
            y = np.log(np.clip(best, 1e-20, None))
            y -= y.mean()
            m1[:] = 0
            v1[:] = 0
            t = 0
            lr = max(0.005, lr * 0.5)
            gamma = min(10.0, gamma * 1.5)
            no_imp = 0
    
    return best, best_sc
