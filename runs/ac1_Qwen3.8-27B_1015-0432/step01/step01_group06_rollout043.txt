import numpy as np
import time

def _autocorr(a):
    n = a.size
    sz = 1
    while sz < 2*n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa*fa, sz)[:2*n-1]

def _sgrad(corr, a, eps):
    n = a.size
    cx = corr.max()
    w = np.exp((corr - cx) / eps)
    s = w.sum()
    if s < 1e-30:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= s
    sz = 1
    while sz < corr.size + n:
        sz <<= 1
    W = np.fft.rfft(w, sz)
    A = np.fft.rfft(a, sz)
    g = 2.0 * np.fft.irfft(W * np.conj(A), sz)[:n]
    g -= g.mean()
    return g
