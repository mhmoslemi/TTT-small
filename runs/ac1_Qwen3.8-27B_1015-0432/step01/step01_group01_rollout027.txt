def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)

def topk_grad(a, conv, k):
    n = a.size
    idx = np.argpartition(conv, -k)[-k:]
    grad = np.zeros(n, dtype=np.float64)
    for lag in idx:
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            seg = a[lag - hi : lag - lo + 1]
            grad[lo : hi + 1] += seg[::-1]
    grad /= float(k)
    return grad

    if s < 1e-14:

def fft_conv(a, N):
    fa = np.fft.rfft(a, n=N)
    return np.fft.irfft(fa * fa, n=N)

def topk_grad(a, conv, K):
    n = a.size
    idx = np.argpartition(conv, -K)[-K:]
    g = np.zeros(n, dtype=np.float64)
    for lag in idx:
        lag = int(lag)
        lo = max(0, lag - n + 1)
        hi = min(n - 1, lag)
        if lo <= hi:
            g[lo:hi+1] += a[lag-hi:lag-lo+1][::-1]
    return g * (1.0 / float(K))

    s = a.sum()
