def solve_lp(a, n):
    M = _conv(a, n).max()
    # Variables: g[0..n-1], g >= 0
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (a * g)_k <= M for k = 0..2n-2

    rows, cols, vals = [], [], []

        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = M * np.ones(2*n - 1)
    
    # Rescale: sum(g) = 1 (equality constraint)

    if res.success:

        s = g.sum()
        if s > 1e-12:

    return None
