    n = len(a)
    S = float(np.sum(a))
    conv = _fft_conv(a)
    c_max = float(np.max(conv))
    exp_vals = np.exp(T * (conv - c_max))
    sum_exp = float(np.sum(exp_vals))
    smooth_max = c_max + np.log(sum_exp) / T
    p = exp_vals / sum_exp
    p_rev = p[::-1]
    size = 1
    while size < 3 * n - 1:
        size <<= 1
    c = np.fft.irfft(np.fft.rfft(p_rev, size) * np.fft.rfft(a, size), size)[:3 * n - 2]
    g = c[n - 1:2 * n - 1][::-1]
    grad = 4.0 * n * (g * S - smooth_max) / (S * S * S)
    return grad


def _subgrad_step(a, lr_s):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return a
    conv = _fft_conv(a)
    k_star = int(np.argmax(conv))
    M = float(conv[k_star])
    base = -4.0 * n * M / (S * S * S)
    g = np.full(n, base)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    if js <= je:
        jj = np.arange(js, je + 1)
        g[jj] += 4.0 * n * a[k_star - jj] / (S * S)
    return np.clip(a - lr_s * g, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = a.size
    if n < 2:
        return [1.0, 1.0]

    best = a.copy()
    best_score = _exact_score(best)

    # Phase 1: Adam on smoothed objective
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = 0.8
    T = 30.0
    t = 0
    stall = 0

    while time.monotonic() < deadline - 60:
        t += 1
        grad = _grad_smooth(a, T)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_c = m / (1.0 - b1 ** t)
        v_c = v / (1.0 - b2 ** t)
        step = lr * m_c / (np.sqrt(v_c) + eps)
        a = np.clip(a - step, 0.0, 1000.0)
        if a.sum() < 1.0:
            a = a + 0.01

        if t % 80 == 0:
            sc = _exact_score(a)
            if np.isfinite(sc) and sc < best_score - 1e-8:
                best = a.copy()
                best_score = sc
                stall = 0
                lr = max(lr * 0.92, 0.05)
            else:
                stall += 1

            if stall > 40:
                a = best + rng.normal(0.0, 0.008 * best.sum() / n, n)
                a = np.clip(a, 0.0, 1000.0)
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = 0.4
                stall = 0

            T = min(T * 1.03, 300.0)

    # Phase 2: Subgradient polish from best
    a = best.copy()
    lr_s = 0.5
    for _ in range(3000):
        if time.monotonic() >= deadline - 20:
            break
        a_new = _subgrad_step(a, lr_s)
        if a_new.sum() > 0.01:
            sc = _exact_score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best = a_new.copy()
                best_score = sc
                a = a_new
            else:
                lr_s *= 0.85
                if lr_s < 0.01:
                    break
        else:
            break

    # Final Adam polish with high T
    if time.monotonic() < deadline - 20:
        a = best.copy()
        m = np.zeros(n)
        v = np.zeros(n)
        t2 = 0
        T = 200.0
        lr = 0.2
        while time.monotonic() < deadline - 15:
            t2 += 1
            grad = _grad_smooth(a, T)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_c = m / (1.0 - b1 ** t2)
            v_c = v / (1.0 - b2 ** t2)
            step = lr * m_c / (np.sqrt(v_c) + eps)
            a = np.clip(a - step, 0.0, 1000.0)
            if a.sum() > 0.01 and t2 % 50 == 0:
                sc = _exact_score(a)
                if np.isfinite(sc) and sc < best_score:
                    best = a.copy()
                    best_score = sc

    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        best = np.ones(n)
    return [float(x) for x in best]
