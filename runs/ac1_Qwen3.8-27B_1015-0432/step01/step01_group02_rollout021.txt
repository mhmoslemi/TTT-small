def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _score(a, n):
    c = _conv(a, n)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

    threshold = cmax * thresh_frac
    w = np.maximum(c - threshold, 0.0)

import numpy as np
import time


def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _maxc(a, n):
    return _conv(a, n).max()


def _score(a, n):
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * _conv(a, n).max() / (s * s)


def _subgrad(c, a, n, tf):
    cmax = c.max()
    w = np.maximum(c - tf * cmax, 0.0)
    ws = w.sum()
    if ws < 1e-15:
        k = min(10, len(c) - 1)
        idx = np.argpartition(c, -k)[::-1][:k]
        w = np.zeros(len(c))
        v = c[idx]
        w[idx] = v / v.sum()
    af = a[::-1]
    L = n + len(w) - 1
    h = np.fft.irfft(np.fft.rfft(af, L) * np.fft.rfft(w, L), L).real
    return h[n - 1:n - 1 + n]


def _optimize(a0, n, rng, deadline, al0):
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s if s > 1e-12 else np.ones(n) / n
    bsc = _score(a, n)
    ba = a.copy()
    al = al0
    tf = 0.999
    stk = 0
    while time.monotonic() < deadline:
        c = _conv(a, n)
        h = _subgrad(c, a, n, tf)
        hm = h.max()
        if hm < 1e-15:

        done = False

            a2 /= s2
            nw = _conv(a2, n).max()
            if nw * 2.0 * n < old - 1e-13:

                done = True
                break

            if s2 > 1e-12:

        sc = 2.0 * n * _conv(a, n).max()

            ba = a.copy()
            stk = 0

            s2 = a.sum()
            if s2 > 1e-12:
