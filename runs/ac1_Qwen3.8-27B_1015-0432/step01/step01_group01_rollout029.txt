import numpy as np
import time

def fft_convolve(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def subgrad_fn(a, b, top_k=10):
    n = len(a)
    M = b.max()
    idx = np.argpartition(b, -top_k)[-top_k:]
    peaks = idx[b[idx] >= M * 0.999]
    if len(peaks) == 0:
        peaks = idx[:1]
    elif len(peaks) > top_k:
        peaks = np.sort(peaks)[::-1][0:top_k]
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad /= len(peaks)
    return grad

def adam_opt(a, n, deadline, lr=0.05, top_k=10):
    m_n = np.zeros(n)
    v_n = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_M = np.inf
    stall = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    cur_lr = lr
    while time.monotonic() < deadline:
        b = fft_convolve(a)
        M = float(b.max())
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
        if stall > 1000 and cur_lr > 0.005:
            cur_lr *= 0.95
        grad = subgrad_fn(a, b, top_k)
        grad -= grad.mean()
        t += 1
        m_n = b1 * m_n + (1 - b1) * grad
        v_n = b2 * v_n + (1 - b2) * grad * grad
        m_hat = m_n / (1 - b1 ** t)
        v_hat = v_n / (1 - b2 ** t)
        a = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        np.maximum(a, 0, out=a)
        s = a.sum()
        if s < 1e-12:
            return best_a
        a /= s
    return best_a

    else:
        a = np.abs(rng.standard_normal(n)) ** 0.5
    return np.maximum(a, 0.01)

    while time.monotonic() < deadline:

import numpy as np
import time


def fft_convolve(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def subgrad_fn(a, b, top_k=10):
    n = len(a)
    M = b.max()
    idx = np.argpartition(b, -top_k)[-top_k:]
    peaks = idx[b[idx] >= M - 1e-12]
    if len(peaks) == 0:
        peaks = idx[:1]
    elif len(peaks) > top_k:
        peaks = np.sort(peaks)[::-1][:top_k]
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad /= len(peaks)
    return grad


def adam_opt(a, n, deadline, lr=0.05, top_k=10):
    m_n = np.zeros(n)
    v_n = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_M = np.inf
    stall = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    while time.monotonic() < deadline:
        b = fft_convolve(a)
        M = float(b.max())
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if stall > 8000:
            break
        grad = subgrad_fn(a, b, top_k)
        grad -= grad.mean()
        t += 1
        m_n = b1 * m_n + (1 - b1) * grad
        v_n = b2 * v_n + (1 - b2) * grad * grad
        m_hat = m_n / (1 - b1 ** t)
        v_hat = v_n / (1 - b2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        np.maximum(a, 0, out=a)
        s = a.sum()
        if s < 1e-12:
            return best_a
        a /= s
    return best_a
