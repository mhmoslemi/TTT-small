    # Soft gradient using log-sum-exp weighting
    cmax = corr.max()
    weights = np.exp((corr - cmax) / T)
    weights /= weights.sum()

    # This is like a weighted sum of the gradients

    # For efficiency, only iterate over indices with significant weight
    thresh = weights.max() * 1e-4
    active = np.where(weights > thresh)[0]
    for k in active:

        if lo <= hi:
            g[lo:hi+1] += weights[k] * a[int(k)-hi : int(k)-lo+1][::-1]
    g -= g.mean()
    return g

import numpy as np
import time

def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _proj(a, n):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n)
    return a * n / s

def _grad(corr, a, n, K):
    idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for k in idx:
        ki = int(k)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[ki - hi:ki - lo + 1][::-1]
    g /= float(K)
    g -= g.mean()
    return g
