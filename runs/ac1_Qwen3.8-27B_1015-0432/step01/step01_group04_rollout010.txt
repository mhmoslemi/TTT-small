        c2 = 1 - b2**t
        st = lr * (m1/c1) / (np.sqrt(m2/c2) + 1e-8)
        st = np.clip(st, -0.3, 0.3)
        y -= st
        y -= y.mean()
        y = np.clip(y, -30, 30)
        # Decay LR
        if t % 500 == 0:
            lr = max(0.01, 0.3 * (1 - 0.95 * min(1.0, t/50000.0)))
    return ba, bsc
