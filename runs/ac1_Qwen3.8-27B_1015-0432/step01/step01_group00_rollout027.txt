def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def autoref(f):
    return fft_conv(f, f)

def normalize_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / len(f)
