import numpy as np
import time


def _pow2(m):
    p = 1
    m = int(m)
    while p < m:
        p <<= 1
    return p


def _conv_score(x, n, L):
    F = np.fft.rfft(x, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    s = float(x.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf, c
    m = 0.0 if c.size == 0 else float(c.max())
    return 2.0 * float(n) * m / (s * s), c


def _make_pos(raw, n):
    y = np.asarray(raw, dtype=np.float64)
    if y.size != n:
        y = np.resize(y, n)
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    y = np.clip(y, 0.0, None)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 1e-300:
        y = np.ones(n, dtype=np.float64)
    else:
        y = y / s
    y = np.maximum(y, 1e-12)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 0.0:
        y = np.ones(n, dtype=np.float64) / n
    else:
        y = y * (float(n) / s)
    return y


def _normalize(x, S):
    y = np.asarray(x, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=0.0, neginf=0.0)
    y = np.maximum(y, 1e-12)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    return y * (S / s)


def _perturb(x, rng, n, sigma):
    y = x.copy()
    if n <= 1:
        y[0] = max(float(y[0]), 1e-12)
        return y * (float(n) / max(float(y.sum()), 1e-300))
    w = max(4, min(n, n // 64))
    s = int(rng.integers(0, n - w))
    y[s:s + w] *= np.exp(rng.normal(0.0, sigma, size=w))
    y *= np.exp(rng.normal(0.0, sigma * 0.15, size=n))
    return _normalize(y, float(n))


def _bank_add(bank, x, score):
    bank.append((float(score), x.copy()))
    if len(bank) > 14:
        worst = 0
        for i in range(len(bank)):
            if bank[i][0] > bank[worst][0]:
                worst = i
        del bank[worst]


def _direction_h(x, c, n, Lg, gap):
    cm = 0.0 if c.size == 0 else float(c.max())
    if cm <= 0.0:
        return np.zeros(n, dtype=np.float64), 0

    tol = max(gap * cm, 1e-12)
    active = np.nonzero(c >= cm - tol)[0]
    if active.size < 4:
        k = min(c.size, 4)
        if k > 1:
            active = np.argpartition(c, -k)[-k:]
        else:
            active = np.array([int(np.argmax(c))], dtype=np.int64)

    cnt = int(active.size)
    vals = c[active]
    wsum = float(vals.sum())
    if wsum > 0.0 and np.isfinite(wsum):
        weights = vals / wsum
    else:
        weights = np.full(cnt, 1.0 / max(1, cnt), dtype=np.float64)

    h = np.zeros(n, dtype=np.float64)
    if cnt <= 64:
        for ii in range(cnt):
            wt = float(weights[ii])
            if wt <= 0.0:
                continue
            kj = int(active[ii])
            lo = kj - n + 1
            if lo < 0:
                lo = 0
            hi = kj
            if hi >= n:
                hi = n - 1
            if hi >= lo:
                h[lo:hi + 1] += wt * x[kj - hi:kj - lo + 1][::-1]
        return h, cnt

    w = np.zeros(c.size, dtype=np.float64)
    w[active] = weights
    Fx = np.fft.rfft(x, Lg)
    Fw = np.fft.rfft(w[::-1], Lg)
    g = np.fft.irfft(Fx * Fw, Lg)
    h = 2.0 * g[n - 1:2 * n - 1][::-1]
    h = np.maximum(h, 0.0)
    return h, cnt


def _shape_guesses(t):
    n = t.size
    out = [np.ones(n)]
    s = np.sin(np.pi * t)
    for p in (0.3, 0.5, 1.0, 2.0, 4.0, 8.0):
        out.append(np.maximum(s ** p, 0.0))
        out.append(np.maximum((1.0 - np.cos(np.pi * t)) ** p, 0.0))
    out.append(0.54 - 0.46 * np.cos(2.0 * np.pi * t))
    out.append(0.5 - 0.5 * np.cos(2.0 * np.pi * t))
    out.append(0.42 - 0.5 * np.cos(2.0 * np.pi * t) + 0.08 * np.cos(4.0 * np.pi * t))
    u = 2.0 * t - 1.0
    y = np.maximum(1.0 - u * u, 0.0)
    out.append(y)
    out.append(np.sqrt(y))
    pq = (0.2, 0.5, 1.0, 2.0, 3.0)
    for p in pq:
        a = np.power(t, p)
        for q in pq:
            out.append(a * np.power(1.0 - t, q))
    for mu in (0.3, 0.5, 0.7):
        for sg in (0.15, 0.3, 0.5):
            out.append(np.exp(-0.5 * ((t - mu) / sg) ** 2))
    for lam in (1.0, 2.0, 4.0):
        out.append(np.exp(-lam * t))
        out.append(np.exp(-lam * (1.0 - t)))
    return out


def _random_poly_guesses(t, rng, count, m, scale):
    n = t.size
    out = []
    for _ in range(count):
        y = np.ones(n)
        for k in range(1, m + 1):
            amp = scale / float(np.sqrt(k))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            y += amp * np.cos(float(k) * np.pi * t + ph)
        out.append(y)
    return out

    it = 0

    while time.monotonic() < deadline:

        else:

        hmax = float(h.max())
        if not np.isfinite(hmax) or hmax <= 1e-300:

            x = _perturb(base, rng, n, 0.3)
