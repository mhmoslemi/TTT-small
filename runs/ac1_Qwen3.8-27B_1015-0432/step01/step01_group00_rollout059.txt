def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_inits(n, rng, parent):
    inits = []
    # Parent
    p = np.clip(parent.astype(np.float64), 0, 1000)

    # Uniform
    inits.append(np.ones(n))
    # Symmetric tapers
    x = np.linspace(0, np.pi, n)
    inits.append(np.maximum(0.5 * (1 + np.cos(x)), 0.01))
    inits.append(np.maximum(np.cos(x / 2)**2, 0.01))
    # Power laws
    d = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1

    n, m = len(a), len(b)

            best_f = f_fin.copy()

    # Validate
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0] * n
    return result
