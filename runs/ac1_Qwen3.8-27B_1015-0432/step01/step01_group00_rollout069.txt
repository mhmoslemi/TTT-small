import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_inits(n, rng):
    inits = []
    pp = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if pp.size == n and np.isfinite(pp).all() and pp.sum() > 1e-10:
        inits.append(pp / pp.sum())
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    inits.append(np.ones(n) / n)
    inits.append(((1 - t) + 0.1) / ((1 - t) + 0.1).sum())
    inits.append((t + 0.1) / (t + 0.1).sum())
    inits.append(np.exp(-2*t) / np.exp(-2*t).sum())
    inits.append(np.exp(2*t) / np.exp(2*t).sum())
    inits.append((np.cos(np.pi*t/2)**0.7) / (np.cos(np.pi*t/2)**0.7).sum())
    inits.append((np.cos(np.pi*(1-t)/2)**0.7) / (np.cos(np.pi*(1-t)/2)**0.7).sum())
    inits.append((1 + np.sin(np.pi*t)) / (1 + np.sin(np.pi*t)).sum())
    for sc in [0.3, 0.6, 1.0]:
        r = np.exp(sc * rng.standard_normal(n))
        inits.append(r / r.sum())
    return inits

def gd_step(f, beta, eta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return f, b
    z = beta * b / M
    z -= z.max()
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-30:
        return f, b
    w /= ws
    f_r = f[::-1]
    gc = fft_conv(w, f_r)
    grad = gc[n-1:2*n-1].copy()
    grad = np.maximum(grad, 0.0)
    gm = grad.max()
    if gm < 1e-30:
        return f, b
    f_new = f * np.exp(-eta * grad / gm)
    f_new = np.maximum(f_new, 1e-20)
    f_new /= f_new.sum()
    return f_new, b

def run_gd(f, deadline, rng):
    n = f.size
    f = f.copy()
    beta = 20.0
    eta = 0.08
    best = f.copy()
    best_sc = fast_score(f)
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        f, _ = gd_step(f, beta, eta)
        it += 1
        if it % 100 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best = f.copy()
                no_imp = 0
            else:
                no_imp += 1
            if no_imp >= 40:
                f = best * np.exp(0.1 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_imp = 0
                eta = 0.08
            if no_imp >= 20:
                beta = min(500.0, beta * 1.3)
            if it > 500:
                eta = max(0.003, eta * 0.995)
    return best, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    if best_f is None:
        best_f = np.ones(n) / n
        best_sc = 2.0
    
    # Multi-start GD
    top_inits = []
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc):
            top_inits.append((sc, init))
    top_inits.sort(key=lambda x: x[0])
    
    for i in range(min(3, len(top_inits))):
        if time.monotonic() > deadline - 60:
            break
        phase_end = time.monotonic() + max(30, (deadline - time.monotonic()) * 0.2)
        f_opt, sc_opt = run_gd(top_inits[i][1], phase_end, rng)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()
    
    # Iterate refinement
    while time.monotonic() < deadline - 30:
        f_opt, sc_opt = run_gd(best_f, deadline - 30, rng)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_f = f_opt.copy()
        if time.monotonic() < deadline - 40:
            # small perturbations
            for _ in range(10):
                trial = best_f * np.exp(0.03 * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = fast_score(trial)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_f = trial.copy()
    
    # Final high-beta refinement
    if time.monotonic() < deadline - 5:
        f = best_f.copy()
        for _ in range(200):
            if time.monotonic() > deadline - 5:
                break
            b = fft_conv(f, f)
            M = b.max()
            z = 200.0 * b / M
            z -= z.max()
            w = np.exp(z)
            w /= w.sum()
            f_r = f[::-1]
            gc = fft_conv(w, f_r)
            grad = gc[n-1:2*n-1]
            grad = np.maximum(grad, 0.0)
            gm = grad.max()
            if gm < 1e-30:
                break
            f = f * np.exp(-0.01 * grad / gm)
            f = np.maximum(f, 1e-20)
            f /= f.sum()
        sc = fast_score(f)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = f
    
    result = [float(x) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(1.0/n)] * n
    return result
