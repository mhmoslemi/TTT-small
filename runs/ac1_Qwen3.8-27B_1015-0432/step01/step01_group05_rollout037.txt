def _fft_conv_auto(a):
    """Compute linear autocorrelation using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def _fft_gradient(c, a, beta):
    """Compute gradient of soft-max of c with respect to a, using FFT.
    c: autocorrelation of a (length 2n-1)
    a: input (length n)
    Returns gradient of (1/beta)*log(sum exp(beta*c)) w.r.t. a, length n.
    """
    n = len(a)
    cmax = c.max()
    w = np.exp(beta * (c - cmax))
    Z = w.sum()
    if Z < 1e-300:
        w = np.zeros_like(c)
        w[(2 * n - 1) // 2] = 1.0  # center
        Z = 1.0
    w /= Z
    
    # grad[i] = 2 * sum_j w[i+j] * a[j] for j=0..n-1, i+j in [0, 2n-2]

    size = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    corr = np.fft.irfft(W * np.conj(A), size)[:n]
    return 2.0 * corr

def _fft_conv_auto(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

def _fft_soft_grad(c, a, beta):
    n = len(a)
    cmax = c.max()
    w = np.exp(beta * (c - cmax))
    Z = w.sum()
    if Z < 1e-300:

        w[(2 * n - 1) // 2] = 1.0
        Z = 1.0
    w /= Z
    N = 1 << (3 * n - 1).bit_length()
