            lr *= 0.85
            if stall > 50:
                # Perturb and restart
                sigma = 0.05 * float(np.sum(a)) / n * 10
                a = np.clip(a + rng.normal(0.0, sigma, n), 0.0, None)
                a = _normalize(a)
                lr = 0.02
                stall = 0
                best_streak = 0

        # Periodic: try a front-loaded mixture to explore different basins
        if best_streak > 200 and time.monotonic() < deadline - 60:
            idx = np.arange(n, dtype=np.float64)
            shape = 1.0 / (1.0 + idx / (n * 0.3))
            shape = shape - shape.min()
            shape = shape / shape.sum()
            mix = 0.3 * a + 0.7 * shape
            mix = _normalize(mix)
            s_mix = _score(mix)
            if s_mix < best_score:
                best_score = s_mix
                best = mix.copy()
                a = mix.copy()
            best_streak = 0

    # Phase 2: Local refinement with smaller steps
    if time.monotonic() < deadline - 30:
        a = best.copy()
        lr = 0.005
        for _ in range(200):
            if time.monotonic() > deadline - 30:
                break
            conv = _fft_conv(a)
            k_star = int(np.argmax(conv))
            M = float(conv[k_star])
            S = float(np.sum(a))
            cur_s = 2.0 * n * M / (S * S)
            g = _subgrad_fast(a, k_star)
            g -= 2.0 * M
            g_norm = float(np.sqrt(np.sum(g * g)))
            if g_norm < 1e-15:
                break
            g = g / g_norm
            a_new = np.clip(a - lr * g, 0.0, None)
            a_new = _normalize(a_new)
            s_new = 2.0 * n * float(np.max(_fft_conv(a_new)))
            if s_new < cur_s:
                a = a_new
                lr = min(lr * 1.1, 0.05)
                if s_new < best_score:
                    best_score = s_new
                    best = a.copy()
            else:
                lr *= 0.7

    # Final validation
    final = evaluate_sequence(best.tolist())
    if not np.isfinite(final):
        best = np.ones(n) / n

    return [float(x) for x in best]
