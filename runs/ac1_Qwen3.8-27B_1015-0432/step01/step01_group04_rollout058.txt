            m1.fill(0); v2.fill(0); b1t, b2t = 1.0, 1.0; t, stall = 0, 0
            lr = max(0.01, lr * 0.5)
            continue
        A = np.fft.rfft(av, Ng)
        c = np.fft.irfft(A * A, Ng)[:L]
        c = np.maximum(c, 0.0)
        mc = float(c.max())
        if mc <= 0:
            break
        # Soft-max weighting on lags
        tau = max(0.3, 3.0 - 0.0005 * t)
        c_shift = c / mc
        z = np.exp(-(c_shift - 1.0) / tau)
        z = z / z.sum()
        W = np.fft.rfft(z[::-1], Ng)
        conv_g = np.fft.irfft(A * W, Ng)
        G = 2.0 * conv_g[L - n:L][::-1]
        G = np.nan_to_num(G, nan=0.0, posinf=0.0, neginf=0.0)
        gdot = float(av @ G)
        grad = av * (G - gdot)
        if not np.all(np.isfinite(grad)):
            break
        t += 1
        m1 = b1 * m1 + (1 - b1) * grad
        v2 = b2 * v2 + (1 - b2) * grad * grad
        b1t *= b1
        b2t *= b2
        mb = m1 / (1 - b1t)
        vb = v2 / (1 - b2t)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.12, 0.12)
        y = y - step
    return best, best_sc
