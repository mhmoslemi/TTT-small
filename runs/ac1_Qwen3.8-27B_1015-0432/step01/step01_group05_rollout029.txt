                    sa2 = a.sum()
                    if sa2 > 1e-12:
                        a = a / sa2 * S
                    else:
                        a = best.copy()
                    soft_beta = min(soft_beta * 1.6, 800.0)
                    lr = max(lr * 0.5, 1.0 / (n * 200.0))
                    m_buf *= 0.3
                    v_buf *= 0.3
                    stall = 0

        if t_adam % 800 == 0:
            soft_beta = min(soft_beta * 1.18, 800.0)
            lr = max(lr * 0.96, 1.0 / (n * 200.0))

    # Late refinement: exact max gradient with small steps
    refine_time = min(30.0, max(0.0, deadline - time.monotonic()))
    if refine_time > 5.0:
        a = best.copy()
        lr2 = 0.02 / n
        for _ in range(2000):
            if time.monotonic() > deadline:
                break
            c = _fft_conv(a, a)
            cmax = c.max()
            idx = np.where(c == cmax)[0]
            if len(idx) == 0:
                break
            grad = np.zeros(n)
            for k in idx[:50]:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                if lo <= hi:
                    j = np.arange(lo, hi + 1)
                    grad[lo:hi + 1] += a[k - j]
                    grad[lo:hi + 1] += a[j - lo + k - k + k - j * 0 + (k - j) * 0]
            # Correct: dc[k]/da[i] = a[k-i] + a[i-k]
            grad = np.zeros(n)
            for k in idx[:80]:
                for i in range(n):
                    v1 = a[k - i] if (0 <= k - i < n) else 0.0
                    v2 = a[i - k] if (0 <= i - k < n) else 0.0
                    grad[i] += (v1 + v2) / len(idx)
            grad -= grad.mean()
            a = a - lr2 * grad
            a = np.maximum(a, 0.0)
            sa = a.sum()
            if sa > 1e-12:
                a = a / sa * S
            sc = _score_fft(a)
            if np.isfinite(sc) and sc < best_s:
                best = a.copy()
                best_s = sc

    result = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs) or fs > 2.0:
        result = [1.0 / n] * n
    return result
