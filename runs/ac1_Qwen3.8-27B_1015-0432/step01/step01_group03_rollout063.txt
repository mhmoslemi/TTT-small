import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _p2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _smax(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    L = _p2(2 * n - 1)
    C = ifft(fft(a, L) * fft(a, L)).real[:2 * n - 1]
    mv, _ = _smax(C, tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    L = _p2(3 * n)
    A = fft(a, L)
    C = ifft(A * A).real[:2 * n - 1]
    mv, p = _smax(C, tau)
    P = fft(p, L)
    T1 = ifft(fft(a[::-1], L) * P).real[n - 1:2 * n - 1]
    T2 = ifft(A * P).real[:n]
    return (2.0 * n / (S * S)) * (T1 + T2) - (4.0 * n * mv / (S ** 3)) * np.ones(n)

def _lbgf(a, tau, n, mx):
    a = a.copy()
    S = a.sum()
    if S > 1e-10:
        a /= S
    try:
        r = minimize(_obj, a, args=(tau,), jac=_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1000.0)] * n,
                     options={'maxiter': mx, 'gtol': 1e-8, 'maxls': 20, 'ftol': 1e-15})
        if r.fun < _obj(a, tau) - 1e-12:
            nx = np.clip(r.x, 0, 1000)
            ns = nx.sum()
            if ns > 1e-10:
                nx /= ns
            return nx
    except Exception:
        pass
    return a

def _ls(bv, sc, rng, n, dl):
    best, bs = bv.copy(), sc
    cool = 0.02
    while time.monotonic() < dl:
        cd = best.copy()
        st = int(rng.integers(0, 4))
        if st == 0:
            w = int(rng.integers(2, max(3, n // 30)))
            s = int(rng.integers(0, n - w))
            cd[s:s+w] *= float(np.exp(rng.normal(0.0, cool)))
        elif st == 1:
            w = int(rng.integers(3, max(4, n // 15)))
            s = int(rng.integers(0, n - w))
            cd[s:s+w] *= rng.normal(1.0, cool, w)
        elif st == 2:
            for _ in range(5):
                j = int(rng.integers(0, n - 1))
                av = 0.5 * (cd[j] + cd[j+1])
                cd[j] = cd[j+1] = av
        else:
            C = np.convolve(cd, cd)
            ks = int(np.argmax(C))
            M = float(C[ks])
            S = cd.sum()
            if S < 1e-8:
                continue
            i1 = ks - np.arange(n)
            v1 = np.where((i1 >= 0) & (i1 < n), cd[np.clip(i1, 0, n-1)], 0.0)
            i2 = np.arange(n) - ks
            v2 = np.where((i2 >= 0) & (i2 < n), cd[np.clip(i2, 0, n-1)], 0.0)
            sg = (2.0*n/(S*S))*(v1+v2) - (4.0*n*M/(S**3))
            lr = cool / (np.max(np.abs(sg)) + 1e-10)
            cd -= lr * sg
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() < 1e-10:
            continue
        sc2 = evaluate_sequence(cd.tolist())
        if np.isfinite(sc2) and sc2 < bs:
            best, bs = cd.copy(), sc2
            cool = min(cool * 1.03, 0.06)
        else:
            cool = max(0.003, cool * 0.995)
            if rng.random() < 0.005:
                cool = 0.02
    return best, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 15.0)
    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(512)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)
    a0 = par / par.sum()
    best = a0.copy()
    bs = float(evaluate_sequence(best.tolist()))
    inits = [a0.copy()]
    j = np.arange(n, dtype=np.float64)
    cw = 0.5 * (1 + np.cos(2 * np.pi * j / n))
    if cw.sum() > 1e-10:
        inits.append(cw / cw.sum())
    gb = np.exp(-0.5 * ((j - 0.5 * n) / (0.2 * n)) ** 2)
    if gb.sum() > 1e-10:
        inits.append(gb / gb.sum())
    for init in inits:
        if time.monotonic() > dl - 120:
            break
        a = init.copy()
        for tv in [3.0, 1.0, 0.4, 0.15, 0.06]:
            if time.monotonic() > dl - 60:
                break
            a = _lbgf(a, tv, n, 300)
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < bs:
                best, bs = a.copy(), sc
    best, bs = _ls(best, bs, rng, n, dl - 5.0)
    return [float(v) for v in best]

Actually, the autocorrelation R(t) = (f*f)(t) (using the same f, not f(-t)). For nonnegative f on [0,1], R(t) for t in [-1,1] is the cross-correlation. The FT of R is |F(ω)|^2 (where F is the FT of f).

But a Gaussian on [0,1] truncated would have some edge effects. A cosine window (1+cos) is smoother and might work better.

import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _p2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _smax(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    L = _p2(2 * n - 1)
    C = ifft(fft(a, L) * fft(a, L)).real[:2 * n - 1]
    mv, _ = _smax(C, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    L = _p2(3 * n)
    A = fft(a, L)
    C = ifft(A * A).real[:2 * n - 1]
    mv, p = _smax(C, tau)
    P = fft(p, L)
    T1 = ifft(fft(a[::-1], L) * P).real[n - 1:2 * n - 1]
    T2 = ifft(A * P).real[:n]
    return (2.0 * n / (S * S)) * (T1 + T2) - (4.0 * n * mv / (S ** 3)) * np.ones(n)


def _lbgf(a, tau, n, mx):
    a = a.copy()
    S = a.sum()
    if S > 1e-10:
        a /= S
    try:
        r = minimize(_obj, a, args=(tau,), jac=_grad, method='L-BFGS-B',
                     bounds=[(0.0, 1000.0)] * n,
                     options={'maxiter': mx, 'gtol': 1e-8, 'maxls': 20, 'ftol': 1e-15})
        if r.fun < _obj(a, tau) - 1e-12:
            nx = np.clip(r.x, 0, 1000)
            ns = nx.sum()
            if ns > 1e-10:
                nx /= ns
            return nx
    except Exception:
        pass
    return a

        elif st == 1:

        elif st == 2:

            if S < 1e-8:
