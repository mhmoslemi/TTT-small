    if n < 2 or p.sum() < 0.01:
        n = 1024
        p = np.ones(n)

    S0 = np.sqrt(2.0 * n)
    conv_size = 1 << (2 * n - 1).bit_length()
    grad_size = 1 << (3 * n - 2).bit_length()
    i = np.arange(n, dtype=np.float64)

    # Initializations
    rc = np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * i / n)), 0.005)
    sb = np.abs(np.sin(np.pi * (i + 0.5) / n)) ** 0.5 + 0.01
    gc = np.exp(-3.0 * ((i - 0.45 * n) / (0.45 * n)) ** 2) + 0.01
    cands = [
        p / p.sum() * S0,
        rc / rc.sum() * S0,
        sb / sb.sum() * S0,
        gc / gc.sum() * S0,
    ]

    best = cands[0].copy()
    best_score = _proxy(best)
    for cnd in cands[1:]:
        s = _proxy(cnd)
        if s < best_score:
            best_score = s
            best = cnd.copy()

    a = best.copy()

    # Adam state
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = 0.04
    beta = 10.0
    stall = 0
    phase = 0  # 0: main, 1: refine
    phase_start = time.monotonic()

    while time.monotonic() < deadline:
        t += 1

        # Softmax weights
        c = _fft_conv(a, a)
        cmax = c.max()
        wm = np.exp(beta * (c - cmax))
        ws = wm.sum()
        if ws > 1e-15:
            w = wm / ws
        else:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0

        # Gradient via FFT
        grad = _grad_fft(a, w, n, grad_size)
        grad -= grad.mean()

        # Adam update
        m_buf = b1 * m_buf + (1.0 - b1) * grad
        v_buf = b2 * v_buf + (1.0 - b2) * grad * grad
        m_hat = m_buf / (1.0 - b1 ** t)
        v_hat = v_buf / (1.0 - b2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        sa = a_new.sum()
        a_new = a_new / sa * S0 if sa > 1e-12 else a.copy()

        # Accept/reject based on proxy
        c2 = _fft_conv(a_new, a_new).max()
        if c2 > cmax * 1.003:
            lr *= 0.75
        else:
            a = a_new
            lr = min(lr * 1.005, 0.12)

        # Exact evaluation periodically
        if t % 200 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best = a.copy()
                stall = 0
            else:
                stall += 1

        # Schedule: increase beta, track progress
        if t % 400 == 0:
            beta = min(beta * 1.12, 300.0)

        # Stall recovery
        if stall >= 25:
            a = best * (1.0 + 0.002 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            sa2 = a.sum()
            a = a / sa2 * S0 if sa2 > 1e-12 else best.copy()
            beta = max(beta * 0.5, 5.0)
            lr = 0.04
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            stall = 0

        # Phase transition: after 70% of budget, do multi-start refinement
        frac = (time.monotonic() - phase_start) / max(1e-6, float(budget_s) - 5.0)
        if phase == 0 and frac > 0.70:
            phase = 1
            beta = 50.0
            lr = 0.02
            m_buf.fill(0)
            v_buf.fill(0)
            t = 0
            stall = 0

        # In refinement phase, more aggressive beta
        if phase == 1 and t % 200 == 0:
            beta = min(beta * 1.2, 500.0)

        if time.monotonic() > deadline - 12.0:
            break

    # Final: try the current iterate
    sc = float(evaluate_sequence(a.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best = a.copy()

    # One more: try exact peak descent from best
    if time.monotonic() < deadline - 5.0:
        c_best = _fft_conv(best, best)
        k_star = int(np.argmax(c_best))
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        if lo <= hi:
            sg[lo:hi + 1] = 2.0 * best[k_star - hi:k_star - lo + 1][::-1]
        sg -= sg.mean()
        for alpha in [0.001, 0.003, 0.01, 0.03]:
            a_try = best - alpha * sg
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st > 1e-12:
                a_try = a_try / st * S0
            sc_try = _proxy(a_try)
            if sc_try < best_score:
                best = a_try.copy()
                best_score = sc_try

    result = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    if not result or sum(result) < 0.01:
        result = [1.0] * n
    for x in result:
        if not np.isfinite(x):
            return [1.0] * n
    return result
