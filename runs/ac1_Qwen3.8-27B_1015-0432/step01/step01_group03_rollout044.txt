import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m

def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = np.convolve(a, a)
    mv, _ = _smooth_max(c, tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = np.convolve(a, a)
    mv, p = _smooth_max(c, tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def _local_search(bv, sc, rng, n, dl):
    while time.monotonic() < dl:
        cd = bv.copy()
        r = rng.random()
        if r < 0.3:
            # Window multiplicative
            w = max(1, n // 80)
            s = int(rng.integers(0, n - w))
            cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.03)))
        elif r < 0.5:
            # Frequency perturbation
            L = _fft_pow2(n)
            af = fft(cd, L)
            idx = rng.integers(0, L, size=max(4, L // 16))
            af[idx] *= (1.0 + rng.normal(0, 0.015, size=idx.size)).astype(np.complex128)
            cd = np.clip(ifft(af).real[:n], 0, 1000)
        elif r < 0.7:
            # Peak-targeted reduction
            c = np.convolve(cd, cd)
            pk = int(np.argmax(c))
            # The elements contributing to c[pk] are a[i] and a[pk-i] for valid i
            # Reduce elements that pair up at the peak
            center = pk // 2
            w = max(2, n // 40)
            s = max(0, center - w)
            e = min(n, center + w)
            cd[s:e] *= 0.97
        elif r < 0.85:
            # Add small random component
            cd = cd + rng.normal(0, 0.002 * cd.max(), n)
        else:
            # Mirror perturbation
            center = n // 2
            w = max(1, n // 50)
            s = int(rng.integers(0, n - w))
            cd[s:s + w] += 0.01 * cd[max(0, 2*center - s - w):2*center - s]
        cd = np.clip(cd, 0.0, 1000.0)
        cs = cd.sum()

            sc = s2
    return bv, sc
