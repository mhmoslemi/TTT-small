        if (it + 1) % 100 == 0:
            sc = _fast_score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = f.copy()
                no_imp = 0
            else:
                no_imp += 1
            
            if no_imp >= 800:
                f = best_f * np.exp(0.04 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_imp = 0
                beta = min(150.0, beta * 1.4)
                eta = max(0.008, eta * 0.85)
        
        # Adaptive schedule
        if (it + 1) % 2000 == 0:
            beta = min(80.0, beta * 1.015)
            eta = max(0.004, eta * 0.997)
    
    return best_f, best_sc


def _local_refine(f, deadline, rng):
    n = f.size
    best_f = f.copy()
    best_sc = _fast_score(f)
    
    scales = [0.15, 0.08, 0.04, 0.02, 0.01]
    for sc in scales:
        if time.monotonic() > deadline - 5.0:
            break
        for _ in range(60):
            if time.monotonic() > deadline - 5.0:
                break
            trial = best_f * np.exp(sc * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            s = _fast_score(trial)
            if np.isfinite(s) and s < best_sc - 1e-12:
                best_sc = s
                best_f = trial.copy()
    
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    n = len(height_sequence_1)
    
    # Build initializations
    inits = []
    pp = _init_parent(n, height_sequence_1)
    if pp is not None:
        inits.append(pp)
    inits.append(np.ones(n) / n)
    inits.append(_init_hann(n))
    inits.append(_init_pow(n, 0.3))
    inits.append(_init_pow(n, 0.7))
    inits.append(_init_pow(n, 1.2))
    inits.append(_init_pow(n, 1.8))
    inits.append(_init_ramp(n, False))
    inits.append(_init_ramp(n, True))
    inits.append(_init_pow(n, 0.5, sym=False))
    inits.append(_init_pow(n, 1.0, sym=False))
    inits.append(_init_bimodal(n, rng, 0.25))
    inits.append(_init_bimodal(n, rng, 0.4))
    
    # Select best init
    best_f = None
    best_score = np.inf
    for init in inits:
        sc = _fast_score(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = init.copy()
    
    if best_f is None:
        best_f = np.ones(n)
        best_score = 2.0
    
    # Phase 1: gradient optimization from best init
    f_opt, sc_opt = _gradient_opt(best_f, deadline - 30.0, rng, beta0=15.0)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
    
    # Phase 2: local refinement
    if time.monotonic() < deadline - 20.0:
        f_opt, sc_opt = _local_refine(best_f, deadline - 10.0, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
    
    # Phase 3: second gradient pass with different params if time allows
    if time.monotonic() < deadline - 25.0:
        f_opt, sc_opt = _gradient_opt(best_f * np.exp(0.03 * rng.standard_normal(n)),
                                       deadline - 15.0, rng, beta0=40.0)
        f_opt = np.maximum(f_opt, 1e-20)
        f_opt /= f_opt.sum()
        sc_opt = _fast_score(f_opt)
        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
    
    # Phase 4: final local polish
    if time.monotonic() < deadline - 8.0:
        f_opt, sc_opt = _local_refine(best_f, deadline - 3.0, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
    
    # Final validation
    result = [float(x) for x in best_f]
    final_score = evaluate_sequence(result)
    if not np.isfinite(final_score):
        fallback = np.ones(n)
        result = [float(x) for x in fallback]
    
    return result
