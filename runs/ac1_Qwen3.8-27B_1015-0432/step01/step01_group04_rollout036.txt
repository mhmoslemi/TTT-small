    s = float(a.sum())
    if s < 0.01:
        return np.inf
    L = 2 * n - 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.maximum(np.abs(c), 0.0)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s))

    z = np.exp(alpha * (c - max_c))

    m = 1
    while m < x:
        m <<= 1
