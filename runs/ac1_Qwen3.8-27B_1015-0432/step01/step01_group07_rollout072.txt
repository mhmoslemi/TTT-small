import numpy as np
import time


def _fft_conv(a, size):
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)


def _softgrad(a, conv, T, size):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return None
    cm = float(np.max(conv))
    ec = np.exp(T * (conv - cm))
    tot = ec.sum()
    if tot < 1e-300:
        return None
    W = ec / tot
    sm = cm + np.log(tot) / T
    ff = np.fft.rfft(a[::-1], size)
    fw = np.fft.rfft(W, size)
    ga = np.fft.irfft(fw * ff, size)[n - 1:2 * n - 1]
    return 2.0 * n * ga / (S * S) - 4.0 * n * sm / (S * S * S)
