import numpy as np
import time


def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def subgradient(a, conv, n, top_k, wpower):
    gr = np.zeros(n, dtype=np.float64)
    m = conv.size
    if top_k >= m:
        idx = np.arange(m)
        wts = np.ones(m)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        med = np.median(conv)
        ex = np.maximum(0.0, conv[idx] - med)
        mx = ex.max()
        wts = (ex / mx) ** wpower if mx > 1e-30 else np.ones(top_k) * 0.1
    sw = wts.sum()
    if sw < 1e-30:
        wts = np.ones_like(wts)
        sw = wts.size
    for j in range(len(idx)):
        k = int(idx[j])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            gr[lo:hi + 1] += wts[j] * a[k - hi:k - lo + 1][::-1]
    gr /= sw
    return gr


def norm1(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a / s


def make_init(kind, n, rng, parent):

    elif kind == 2:
        f1 = rng.uniform(0.03, 0.25)
        f2 = rng.uniform(0.2, 0.7)

        x = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2 * np.pi * f1 * x + p1) + 0.25 * np.cos(2 * np.pi * f2 * x + p2)
        a = parent * np.clip(a, 0.05, 1.95)
    elif kind == 3:
        blk = max(4, n // 60)
        nb = (n + blk - 1) // blk
        a = np.repeat(np.clip(rng.uniform(0.4, 1.6, nb), 0.05, None), blk)[:n]
    else:
        a = parent + 0.05 * parent * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n)
        s = n
    return a / s


def optimize(a, n, fsz, best_sc, deadline, rng, tk, wp):
    v1 = np.zeros(n)
    v2 = np.zeros(n)
    b1, b2, ad_eps = 0.8, 0.999, 1e-8
    lr = 0.5
    prev_m = np.inf
    stall = 0
    local_best = None
    local_sc = best_sc
    t = 0
    base_m = None
    for it in range(999999):
        if time.monotonic() > deadline:
            break
        conv = fft_convolve(a, fsz)[:2 * n - 1]
        M = float(conv.max())
        if base_m is None:
            base_m = M
        if M < prev_m - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_m = M
        if stall > 6000 and it > 2000:
            break
        g = subgradient(a, conv, n, tk, wp)
        t += 1
        v1 = b1 * v1 + (1 - b1) * g
        v2 = b2 * v2 + (1 - b2) * g * g
        u = v1 / (1 - b1 ** t) / (np.sqrt(v2 / (1 - b2 ** t)) + ad_eps)
        a = a - lr * u
        a = np.clip(a, 0.0, 1000.0)
        a = norm1(a)
        if a is None:
            break
        lr = max(0.02, lr * 0.99992)
        if (it & 1023) == 0:

    if local_best is None:
        local_best = a.copy()

    return local_best, local_sc
