import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = np.sum(e)
    return cm + tau * np.log(s), e / s


def _obj(a, tau, L):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e12
    mv, _ = _smooth_max(np.convolve(a, a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau, L):

    S = a.sum()
    if S < 1e-10:
        return np.zeros_like(a)
    mv, p = _smooth_max(np.convolve(a, a), tau)

    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv) / (S ** 3)
