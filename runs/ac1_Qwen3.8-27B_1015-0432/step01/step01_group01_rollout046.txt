def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def grad_peaks(a, conv, k, temp):
    n = a.size
    M = float(conv[:n].max())
    idx = np.argpartition(conv[:n], -k)[-k:]
    w = np.exp((conv[idx] - M) / temp)
    ws = w.sum()
    if ws < 1e-14:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for i, kk in enumerate(idx):
        kk = int(kk)
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        if lo <= hi:
            grad[lo:hi + 1] += w[i] * a[kk - hi:kk - lo + 1][::-1]
    return grad
