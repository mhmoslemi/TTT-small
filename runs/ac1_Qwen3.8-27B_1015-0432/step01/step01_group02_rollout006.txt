import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)[:2 * n - 1]


def _compute_h(a, idx, weights, n):
    h = np.zeros(n)
    for j in range(len(idx)):
        kj = int(idx[j])
        wj = float(weights[j])
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += wj * a[kj - np.arange(lo, hi + 1)]
    return h


def _make_perturbed(a, rng, strength, n):
    b = a.copy()
    w = max(2, n // (15 + int(strength * 10)))
    st = int(rng.integers(0, max(1, n - w)))
    b[st:st + w] *= np.exp(rng.normal(0.0, strength, size=w))
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 1e-12:
        b /= s
    else:
        b = a.copy()
    return b
