import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _subgrad_weighted(a, conv, peaks, S):
    n = len(a)
    M = float(np.max(conv))
    g = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += a[k - jj]
    g = g / len(peaks)
    g = (g * S - 2.0 * M) / S
    g -= g.mean()
    return g
