import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _conv_self(a):
    n = a.size
    L = _fft_pow2(2 * n - 1)
    A = fft(a, L)
    return ifft(A * A).real[:2 * n - 1]


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv_self(a)
    mv, _ = _smooth_max(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv_self(a)
    mv, p = _smooth_max(c, tau)
    L = _fft_pow2(3 * n - 2)
    pr = p[::-1]
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _local_search(bv, sc, rng, n, dl):
    while time.monotonic() < dl:
        cd = bv.copy()
        strategy = int(rng.integers(0, 4))
        if strategy == 0:
            w = max(1, n // int(rng.integers(20, 80)))
            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.03)))
        elif strategy == 1:
            w = max(1, n // int(rng.integers(3, 10)))
            s = int(rng.integers(0, max(1, n - w)))

        elif strategy == 2:
            cd *= float(np.exp(rng.normal(0.0, 0.008)))
        else:
            w = max(2, n // 20)
            s = int(rng.integers(0, max(1, n - w)))
            cd[s:s + w] = cd[s:s + w] * 0.88 + cd[s + 1:s + w + 1] * 0.12
        np.clip(cd, 0.0, 1000.0, out=cd)
        if cd.sum() < 1e-8:
            continue
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc:
            bv = cd.copy()
            sc = s2

            idx = rng.integers(0, n)

            s2 = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(s2) and s2 < sc:
                bv = cd.copy()
                sc = s2
    return bv, sc


def _run_lbfgs(cur, tau, n, bounds, dl, max_iter):
    remaining = dl - time.monotonic()
    if remaining < 15.0:
        return cur, False
    maxfun = min(max_iter * 5, int(remaining * 50))
    try:
        res = minimize(
            _obj, cur, args=(tau,), jac=_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'gtol': 1e-11,
                     'maxls': 50, 'ftol': 1e-14,
                     'maxfun': maxfun})
        nx = np.clip(res.x, 0.0, 1000.0)
        nxs = nx.sum()
        if nxs > 1e-8:
            nx = nx / nxs
        return nx, True
    except Exception:
        return cur, False


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(10.0, float(budget_s) - 20.0)

    for tv in taus:

        elif tv >= 0.01:
            mi = 400

            mi = 500

        if not ok:

        if np.isfinite(sc) and sc < best_sc:

            best_sc = sc

            stall = 0
        else:
            stall += 1
        prev_sc = sc

    return [float(v) for v in best]
