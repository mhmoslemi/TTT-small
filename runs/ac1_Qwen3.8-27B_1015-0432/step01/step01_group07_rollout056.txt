import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    conv = _fft_conv(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)
