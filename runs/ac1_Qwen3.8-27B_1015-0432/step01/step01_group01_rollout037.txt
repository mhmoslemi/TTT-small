```python
def project_simplex(v, S):
    """Project v onto {a >= 0, sum a = S}."""
    n = len(v)
    # Binary search for tau
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho = np.arange(1, n+1)
    cond = sorted_v - (cumsum - S) / rho > 0
    if not cond.any():
        return np.full(n, S / n)
    rho = np.max(np.where(cond)[0]) + 1
    tau = (cumsum[rho-1] - S) / rho
    return np.maximum(v - tau, 0)
```
