```python
    # Find tau: sum(max(v - tau, 0)) = target
```

```python
from scipy.sparse import lil_matrix
from scipy.optimize import linprog

def lp_direction(f, M, n):
    # Solve: max sum(g) s.t. (f*g)[x] <= M for all x, g >= 0
    # A is (2n-1) x n, A[x,i] = f[x-i] if 0<=x-i<n
    c_len = 2*n - 1
    # Build sparse A
    rows = []
    cols = []
    vals = []
    for x in range(c_len):
        lo = max(0, x - n + 1)
        hi = min(n - 1, x)
        for i in range(lo, hi + 1):
            rows.append(x)
            cols.append(i)
            vals.append(f[x - i])
    A = lil_matrix((c_len, n))
    for x in range(c_len):
        lo = max(0, x - n + 1)
        hi = min(n - 1, x)
        for i in range(lo, hi + 1):
            A[x, i] = f[x - i]
    A = A.tocsr()
    # LP: max sum(g) s.t. A @ g <= M, g >= 0
    # scipy linprog: min c @ g s.t. A_ub @ g <= b_ub, g >= 0 (A_ub and bounds)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    A_ub = A  # A @_g <= M
    b_ub = np.full(c_len, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'time_limit': 30})
    if res.status == 0:
        g = res.x
    else:
        g = f.copy()
    s = g.sum()
    if s < 1e-14:
        g = f.copy()
        s = 1.0
    return g / s
```

import numpy as np

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def simplex_project(v, target):
    lo = v.min()
    hi = v.max()
    if hi - lo < 1e-15:
        return v * (target / max(v.sum(), 1e-30))
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.maximum(v - mid, 0.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.maximum(v - (lo + hi) * 0.5, 0.0)


def subgradient(a, conv, n, top_k):
    c_len = 2 * n - 1
    if top_k >= c_len:
        top_k = c_len
    idx = np.argpartition(conv[:c_len], -top_k)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    grad *= 2.0 / float(top_k)
    return grad, idx


def optimize_run(a, n, fs, s0, deadline, top_k, beta, lr0):
    best = a.copy()
