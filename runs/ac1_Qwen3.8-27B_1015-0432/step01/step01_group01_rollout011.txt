def build_conv_matrix(f, n):
    """Build (2n-1) x n matrix B such that B @ g = convolve(f, g)."""
    # f has length n, g has length n
    # B[k, j] = f[k-j] if 0<=k-j<n and 0<=j<n, else 0
    m = 2*n - 1
    # Use diagonal structure
    # For each offset d = k-j (from -(n-1) to n-1):
    #   B[d+j, j] = f[d] for valid j
    # This is a band matrix with n diagonals
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for d in range(-(n-1), n):
        # d = k - j, so k = d + j
        # Need 0 <= j < n and 0 <= k < m and 0 <= d < n (i.e., 0 <= k-j < n)
        # d ranges from 0 to n-1 for f[d] to be valid
        if d < 0:
            # k = d + j, need 0 <= d+j < m and 0 <= j < n
            # d is negative, so k = d + j < j < n <= m, ok
            # Need d + j >= 0, so j >= -d = |d|
            j_start = -d
            j_end = n  # exclusive
        else:
            # d >= 0, k = d + j
            # Need d < n (for f[d] to exist) - always true since d <= n-1
            # Need 0 <= j < n and 0 <= d+j < m
            # d+j < m = 2n-1, so j < 2n-1-d. Since j < n and d >= 0, this is fine.
            j_start = 0
            j_end = n - d  # since d+j < 2n-1 means j < 2n-1-d, but also j < n
            # Actually j < min(n, 2n-1-d). For d >= 0, 2n-1-d >= n-1, so min is n (if d=0) or n-d (if d>0)... 
            # 2n-1-d >= n iff d <= n-1, which is always true. So j < n.
            j_end = n
        j_arr = np.arange(j_start, j_end)
        k_arr = d + j_arr
        length = len(j_arr)
        if length > 0:
            rows[idx:idx+length] = k_arr
            cols[idx:idx+length] = j_arr

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def _build_B(f, n):
    m = 2 * n - 1
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    cv = np.arange(n, dtype=np.int32)
    idx = 0
    for k in range(m):
        lo = 0 if k <= n - 1 else k - n + 1
        hi = k if k <= n - 1 else n - 1
        cnt = hi - lo + 1
        rows[idx:idx + cnt] = k
        cols[idx:idx + cnt] = cv[lo:hi + 1]
        data[idx:idx + cnt] = f[k - hi:k - lo + 1][::-1]
        idx += cnt
    return csr_matrix((data[:idx], (rows[:idx], cols[:idx])), shape=(m, n))


def _subgrad(a, conv, tk):
    n = a.size
    idx = np.argpartition(conv, -tk)[-tk:]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    return g / max(1, len(idx))


def _renorm(a):
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a / s
