import time


def _conv(a):
    n = len(a)
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _eval_c(c, n):
    return 2.0 * n * c.max()


def _grad_peak(a, idx, weights, n):
    h = np.zeros(n)
    for j in range(len(idx)):
        kj = int(idx[j])
        wj = weights[j]
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += wj * a[kj - np.arange(lo, hi + 1)]
    return h
