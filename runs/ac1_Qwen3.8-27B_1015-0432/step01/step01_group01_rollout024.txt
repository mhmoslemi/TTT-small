    x = np.arange(n, dtype=np.float64)
    if kind == 0:
        a = parent.copy()
    elif kind == 1:
        # Raised cosine
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * (x / n - 0.5)))
    elif kind == 2:
        # Gaussian
        sigma = rng.uniform(0.12, 0.30)
        t = (x - n / 2.0) / (n * sigma)
        a = np.exp(-0.5 * t * t)
    elif kind == 3:
        # Multi-frequency
        a = np.ones(n)
        for _ in range(int(rng.integers(2, 5))):
            f = rng.uniform(0.5, n * 0.4)
            ph = rng.uniform(0, 2 * np.pi)
            a *= (1.0 + 0.25 * np.cos(2.0 * np.pi * f * x / n + ph))
        a = np.clip(a, 0.01, None)
    elif kind == 4:
        # Blend of parent and raised cosine
        a = 0.5 * parent + 0.5 * (0.5 * (1.0 + np.cos(2.0 * np.pi * (x / n - 0.5))))
    else:
        # Smoothed random
        a = np.abs(rng.standard_normal(n))
        a = np.convolve(a, np.ones(7) / 7, mode='same')
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n)
        s = n
    return a / s


def optimize(a0, n, fft_size, deadline, top_k=64, lr=0.5, seed=0):
    """Adam subgradient descent on max-autocorrelation with sum=1 constraint."""
    a = np.clip(a0, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return a0
    a = a / s
    rng = np.random.default_rng(seed)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-10
    t = 0
    best_a = a.copy()
    best_M = np.inf
    stall = 0
    prev_M = np.inf
    
    while time.monotonic() < deadline:
        t += 1
        conv = fft_convolve(a, fft_size)
        M = float(conv.max())
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        # Slow down patience as we go (allow more iterations early)
        patience_limit = 2000 + 300 * (t // 1000)
        if stall > patience_limit:
            break
            
        # Compute subgradient
        if t % 200 < 50:  # vary top_k periodically
            k_eff = top_k
        else:
            k_eff = max(16, top_k // 2)
        grad = soft_grad(a, conv, k_eff)
        grad -= grad.mean()
        
        # Adam update
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        lr_t = lr * (1.0 + t) ** (-0.5)  # 1/sqrt(t) decay
        a_new = a - lr_t * m / (np.sqrt(v) + eps)
        a_new = norm_nonneg(a_new, 1.0)
        if a_new is None:
            a = best_a.copy()
            lr *= 0.5
            m = np.zeros(n)
            v = np.zeros(n)
            continue
        a = a_new
    
    return best_a

import numpy as np

def fft_convolve(a, fft_size):
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def soft_grad(a, conv, top_k):
    n = a.size
    M = conv.max()
    idx = np.argpartition(conv, -top_k)[-top_k:]
    cv = conv[idx]
    spread = M - cv
    temp = max(1e-12, 0.12 * (spread.max() + 1e-12))
    wts = np.exp(-spread / temp)
    wsum = wts.sum()
    if wsum < 1e-15:
        wts = np.ones(top_k) / top_k
    else:
        wts /= wsum
    grad = np.zeros(n)
    for i in range(top_k):
        k = int(idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += wts[i] * a[k - hi:k - lo + 1][::-1]
    return grad
