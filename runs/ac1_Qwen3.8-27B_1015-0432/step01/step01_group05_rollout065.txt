import numpy as np
import time

def _autocorr(a):
    n = len(a)
    nc = 2 * n - 1
    sz = 1 << (nc - 1).bit_length()
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:nc]

def _score_from_c(c, a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * len(a) * c.max() / (s * s)

def _gradient_fft(a, w):
    n = len(a)
    L = 1 << (3 * n).bit_length()
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    cc = np.fft.irfft(fa * np.conj(fw), L)
    idx = np.concatenate((np.array([0], dtype=np.int64), np.arange(L - 1, L - n, -1)))
    grad = 2.0 * cc[idx]
    grad -= grad.mean()
    return grad
