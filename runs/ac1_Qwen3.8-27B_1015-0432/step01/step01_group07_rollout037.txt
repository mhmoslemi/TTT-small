    n = len(a)
    S = float(a.sum())
    if S < 0.01:
        return None
    c = _fft_conv(a)
    m = float(c.max())
    if T > 1e-15:
        w = np.exp((c - m) / T)
        ws = float(w.sum())
        p = w / ws
        Ms = m + T * float(np.log(ws))
    else:
        p = np.zeros(len(c), dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        Ms = m
    # cross-correlation (p star a)[i] = sum_j p[j+i]*a[j]  via  (p_rev * a)[2n-2-i]
    pr = p[::-1]
    sz = 1
    while sz < 3 * n - 2:
        sz <<= 1
    cor = np.fft.irfft(np.fft.rfft(pr, sz) * np.fft.rfft(a, sz), sz)[:3 * n - 2]
    cross = 2.0 * cor[2 * n - 2 - np.arange(n)]
    return (2.0 * n / (S * S)) * cross - 4.0 * n * Ms / (S * S * S)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = a.size
    if n < 2:
        return [1.0]
    S0 = float(a.sum())
    if S0 < 0.01:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / S0

    best = a.copy()
    best_sc = _score(a)

    M0 = float(np.max(_fft_conv(a)))
    T = M0 * 0.5
    T_floor = M0 * 1e-6
    alpha = 1.0
    stall = 0

    while time.monotonic() < dl:
        g = _grad(a, T)
        if g is None:
            break
        f0 = 2.0 * n * float(np.max(_fft_conv(a)))

        improved = False
        tb = f0
        ta = a.copy()
        for k in range(10):
            step = alpha * (0.5 ** k)
            at = np.clip(a - step * g, 0.0, 1000.0)
            St = float(at.sum())
            if St < 0.01:
                continue
            at = at / St
            ft = 2.0 * n * float(np.max(_fft_conv(at)))
            if ft < tb:
                tb = ft
                ta = at
                improved = True

        if improved:
            a = ta
            alpha = min(50.0, alpha * 1.05)
            stall = 0
        else:
            stall += 1
            alpha *= 0.7
            if stall > 30:
                s = max(float(a.max()), 1.0 / n) * 0.3
                a = np.clip(a + rng.normal(0.0, s, n), 0.0, 1000.0)
                Sn = float(a.sum())
                a = a / Sn if Sn >= 0.01 else np.ones(n) / n
                T = min(M0 * 0.5, T * 3.0)
                alpha = 1.0
                stall = 0

        sc = 2.0 * n * float(np.max(_fft_conv(a)))
        if sc < best_sc:
            best_sc = sc
            best = a.copy()
        T = max(T_floor, T * 0.999)

    rem = dl - time.monotonic()
    if rem > 3.0:
        pdl = time.monotonic() + rem * 0.85
        while time.monotonic() < pdl:
            s = max(float(best.max()), 1.0 / n) * 0.02
            bt = np.clip(best + rng.normal(0.0, s, n), 0.0, 1000.0)
            St = float(bt.sum())
            if St < 0.01:
                continue
            bt = bt / St
            st = _score(bt)
            if st < best_sc:
                best_sc = st
                best = bt

    fin = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fin):
        best = np.ones(n, dtype=np.float64)
    return [float(x) for x in best]
