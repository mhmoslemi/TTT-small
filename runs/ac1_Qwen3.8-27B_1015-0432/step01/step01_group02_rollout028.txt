```python
def _project(a, n):
    # Project onto {a >= 0, sum(a) = 1}
    # a - t >= 0 and sum(a - t) = 1 => n*t = sum(a) - 1 => t = (sum(a)-1)/n (if all a_j > t)
    # More generally, find t such that sum(max(a_j - t, 0)) = 1
    lo, hi = a.min() - 1, a.max()
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.maximum(a - mid, 0).sum()
        if s > 1:
            lo = mid
        else:
            hi = mid
    a_proj = np.maximum(a - (lo+hi)/2, 0)
    a_proj /= a_proj.sum()
    return a_proj
```

from scipy.sparse import toeplitz as _sp_toeplitz

def _fft_conv(a):
    n = len(a)
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N)

def _proj_unit(a):
    """Project onto {x>=0, sum(x)=1} via bisection on threshold."""
    lo = a.min() - 1.0
    hi = a.max() + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.maximum(a - mid, 0.0).sum()
        if s > 1.0:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    x = np.maximum(a - t, 0.0)
    x /= x.sum()
    return x

    r_row = np.zeros(n)
