import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb)[:n + m - 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def make_init_from_parent(h):
    p = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1000.0)
    if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
        return p / p.sum()
    return None

def make_init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    h = np.maximum(h, 1e-10)
    return h / h.sum()

def make_init_gauss(n, sigma_frac):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    g = np.exp(-0.5 * ((x - center) / (sigma_frac * n)) ** 2)
    return g / g.sum()

    c = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    c = np.maximum(c, 1e-10)
    return c**power / (c**power).sum()
