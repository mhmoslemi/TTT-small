        if st >= sl:
            break
        g = subgrad(a, cv[:nlim], tk)
        g -= g.mean()
        v = bet * v + (1.0 - bet) * g
        a = a - lr * v
        a = proj(a)
        if a is None:
            break
        lr = max(0.01, lr * 0.99995)
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s)) - 8.0

    par = np.clip(height_sequence_1.astype(np.float64), 0.0, 1000.0)
    if par.size < 2 or par.ndim != 1 or not np.all(np.isfinite(par)) or par.sum() < 0.01:
        par = np.ones(1024, dtype=np.float64)
    n = par.shape[0]
    nlim = 2 * n - 1
    fsize = 1 << nlim.bit_length()
    sa = max(float(par.sum()), 1.0)
    pn = par / par.sum()

    cv0 = fft_autocorr(pn, fsize)
    bs = 2.0 * n * float(cv0[:nlim].max())
    ba = pn.copy()
    es = float(evaluate_sequence(par.tolist()))
    if np.isfinite(es) and es < bs:
        bs = es

    phases = [
        (2, 0.50, 0.70, 3000),
        (4, 0.35, 0.75, 4000),
        (8, 0.25, 0.80, 5000),
        (16, 0.15, 0.85, 6000),
        (32, 0.08, 0.90, 8000),
        (64, 0.04, 0.93, 10000),
    ]

    inits = [init_seq(k, n, rng, pn) for k in range(5)]
    for i, ini in enumerate(inits):
        if time.monotonic() > dl - 40.0:
            break
        a = ini.copy()
        for (tk, lr0, be, sl) in phases:
            if time.monotonic() > dl - 40.0:
                break
            a = descent(a, n, fsize, tk, lr0, be, dl - 40.0, sl)
            cv = fft_autocorr(a, fsize)
            sc = 2.0 * n * float(cv[:nlim].max())
            if sc < bs:
                bs = sc
                ba = a.copy()
        if time.monotonic() > dl - 80.0:
            break
        for (tk, lr0, be, sl) in phases[-3:]:
            if time.monotonic() > dl - 40.0:
                break
            a = descent(a, n, fsize, tk, lr0 * 0.7, min(0.97, be + 0.02), dl - 40.0, sl)
            cv = fft_autocorr(a, fsize)
            sc = 2.0 * n * float(cv[:nlim].max())
            if sc < bs:
                bs = sc
                ba = a.copy()

    base = ba.copy()
    while time.monotonic() < dl - 3.0:
        c = base.copy()
        w = max(2, n // 200)
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            c[st:e] *= float(np.exp(rng.normal(0.0, 0.03)))
        c = proj(c)
        if c is None:
            continue
        cv = fft_autocorr(c, fsize)
        sc = 2.0 * n * float(cv[:nlim].max())
