        w[int(np.argmax(conv))] = 1.0
    else:

    while N < 3 * n - 2:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    R = np.fft.irfft(np.conjugate(W) * A, N)
    idx = np.mod(-np.arange(n), N)
    grad = 2.0 * R[idx]
    return grad, M, conv


def _exact_subgrad(a, k_star):
    n = len(a)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    g = np.zeros(n)
    if js <= je:
        jj = np.arange(js, je + 1)
        g[jj] = 2.0 * a[k_star - jj]
    return g
