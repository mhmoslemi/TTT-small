        if time.monotonic() >= deadline:
            break
        fa, conv = fft_conv_self(a, fs)
        M = peak_of(conv, n)
        if not np.isfinite(M):
            break
        if M < best_M - 1e-14:

        if M < prev_M - 1e-15:
            stall = 0
        else:
            stall += 1

        if s <= 1e-30:

        return None
    return a / s


def fft_conv_self(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return fa, np.fft.irfft(fa * fa, n=fs)


def peak_of(conv, n):
    L = 2 * n - 1
    if L > conv.size:
        L = conv.size
    if L <= 0:
        return 0.0
    m = float(np.max(conv[:L]))
    if not np.isfinite(m):
        return np.inf
    return max(0.0, m)


def gen_init(kind, n, rng):
    if kind == 0:
        a = np.abs(rng.standard_normal(n))
    elif kind == 1:
        f = rng.uniform(0.05, 0.4)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        a = 1.0 + 0.5 * np.sin(2.0 * np.pi * f * np.arange(n) + ph)
    elif kind == 2:
        a = np.abs(rng.standard_normal(n)) ** 2
    elif kind == 3:
        f = rng.uniform(0.1, 0.6)
        c = rng.uniform(0.3, 0.8)
        m = f * np.arange(-n // 2, n // 2)
        a = np.where(np.abs(m) < c, 1.5, 0.3)
        a *= 1.0 + 0.1 * rng.standard_normal(n)
    else:
        k = int(rng.integers(2, max(3, n // 20)))
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * k * np.arange(n) / float(n))
        a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def run_opt(start, n, fs, deadline, top_k, lr, beta, rng, max_iters, stall_limit):
    a = norm1(start)
    if a is None:
        a = np.ones(n, dtype=np.float64) / float(n)
    top_k = max(1, int(top_k))
    L = 2 * n - 1

    fa, conv = fft_conv_self(a, fs)
    best_M = peak_of(conv, n)
    best = a.copy()
    prev_M = best_M
    stall = 0
    v = np.zeros(n, dtype=np.float64)
    w = np.zeros(fs, dtype=np.float64)

    for _ in range(max_iters):
        if time.monotonic() >= deadline:
            break

        fa, conv = fft_conv_self(a, fs)
        M = peak_of(conv, n)
        if not np.isfinite(M):
            break

        if M < best_M - 1e-14:
            best_M = M
            best = a.copy()

        if M < prev_M - 1e-15:
            stall = 0
        else:
            stall += 1
        prev_M = M

        if stall > stall_limit:
            a = best.copy()
            a *= np.exp(0.05 * rng.standard_normal(n))
            a += 1.0e-3 / float(n)
            a = norm1(a)
            if a is None:
                break
            v[:] = 0.0
            stall = 0
            prev_M = np.inf
            continue

        if top_k >= L:
            idx = np.arange(L, dtype=np.int64)
        else:
            idx = np.argpartition(conv[:L], -top_k)[-top_k:]

        vals = conv[:L][idx]
        vals = np.maximum(vals, 0.0)
        s = float(vals.sum())
        if s <= 1e-30:
            break

        vw = vals * vals
        s2 = float(vw.sum())
        if s2 <= 1e-30:
            vw = vals
            s2 = s

        weights = vw / s2
        w.fill(0.0)
        w[idx] = weights

        wf = np.fft.rfft(w, n=fs)
        g = np.fft.irfft(np.conj(fa) * wf, n=fs)[:n]
        g = np.maximum(g, 0.0)
        g -= float(g.mean())
        gnorm = float(np.sqrt(g.dot(g)))

        if gnorm <= 1e-18:
            a = best.copy()
            a *= np.exp(0.06 * rng.standard_normal(n))
            a += 1.0e-3 / float(n)
            a = norm1(a)
            if a is None:
                break
            v[:] = 0.0
            stall = 0
            prev_M = np.inf
            continue

        v *= beta
        v += (1.0 - beta) * g
        v -= float(v.mean())
        vnorm = float(np.sqrt(v.dot(v)))
        if vnorm <= 1e-18:
            break

        anorm = float(np.sqrt(a.dot(a)))
        step = lr * anorm / vnorm
        if not np.isfinite(step) or step <= 0.0:
            step = 1e-6
        if step > 0.5:
            step = 0.5

        a = a - step * v
        a = np.maximum(a, 0.0)
        a = norm1(a)
        if a is None:
            break

        lr = max(0.008, lr * 0.99997)

    return best, best_M


def consider_direction(cand, n, fs, best, best_score):
    c = norm1(cand)
    if c is None or not np.isfinite(c).all():
        return best, best_score

    fa, conv = fft_conv_self(c, fs)
    m = peak_of(conv, n)
    if not np.isfinite(m):
        return best, best_score

    sc_est = 2.0 * float(n) * m
    if sc_est >= best_score - 1e-9:
        return best, best_score

    sc = float(evaluate_sequence(c.tolist()))
    if np.isfinite(sc) and sc < best_score:
        return c.copy(), sc
    return best, best_score


def retarget_work(work, work_M, best_seq, best_score, n):
    bn = norm1(best_seq)
    if bn is None:
        return work, work_M
    bm = float(best_score) / (2.0 * float(n))
    if np.isfinite(bm) and bm < work_M - 1e-14:
        return bn.copy(), bm
    return work, work_M


def polish_direction(start, n, fs, deadline, rng):
    a = norm1(start)
    if a is None:
        a = np.ones(n, dtype=np.float64) / float(n)

    _, conv = fft_conv_self(a, fs)
    best = a.copy()
    best_M = peak_of(conv, n)
    block = max(2, n // 60)

    while time.monotonic() < deadline - 2.0:
        cand = best.copy()
        cnt = int(rng.integers(1, 4))
        for _ in range(cnt):
            hi = max(1, n - block + 1)
            st = int(rng.integers(0, hi))
            e = min(n, st + block)
            factor = float(np.exp(rng.normal(0.0, 0.04)))
            if not np.isfinite(factor):
                factor = 1.0
            cand[st:e] *= factor

        cand = norm1(cand)
        if cand is None:
            continue

        _, conv2 = fft_conv_self(cand, fs)
        M2 = peak_of(conv2, n)
        if M2 < best_M - 1e-14:
            best = cand
            best_M = M2

    return best, best_M


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).reshape(-1)
    except Exception:
        parent = np.ones(1024, dtype=np.float64)

    if parent.size < 2:
        parent = np.ones(1024, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.sum() < 1e-8:
        parent = np.ones(parent.size, dtype=np.float64)

    n = int(parent.size)
    fs = next_pow2(3 * n - 2)
    const = np.ones(n, dtype=np.float64) / float(n)

    pn = norm1(parent)
    if pn is None:
        pn = const.copy()

    best_seq = const.copy()
    best_score = float(evaluate_sequence(const.tolist()))
    st = float(evaluate_sequence(pn.tolist()))
    if np.isfinite(st) and st < best_score:
        best_seq = pn.copy()
        best_score = st

    work = best_seq.copy()
    work_M = best_score / (2.0 * float(n))

    t0 = time.monotonic()
    total = max(60.0, float(budget_s) - 12.0)
    dl = t0 + total
    rng = np.random.default_rng(seed)

    parent_end = min(dl, t0 + min(450.0, 0.35 * total))
    if time.monotonic() < parent_end - 5.0:
        r, rm = run_opt(pn, n, fs, parent_end, 48, 0.13, 0.85, rng, 500000, 4500)
        best_seq, best_score = consider_direction(r, n, fs, best_seq, best_score)
        if np.isfinite(rm) and rm < work_M - 1e-14:
            work = r.copy()
            work_M = rm
        work, work_M = retarget_work(work, work_M, best_seq, best_score, n)

    random_end = min(dl, t0 + min(950.0, 0.72 * total))
    for kind in range(4):
        if time.monotonic() >= random_end - 10.0:
            break
        init = norm1(gen_init(kind, n, rng))
        if init is None:
            continue
        remaining = max(0.0, random_end - time.monotonic())
        run_dl = min(random_end, time.monotonic() + min(160.0, max(20.0, remaining / 4.0)))
        if run_dl <= time.monotonic() + 5.0:
            continue
        r, rm = run_opt(init, n, fs, run_dl, 24, 0.18, 0.82, rng, 300000, 3000)
        best_seq, best_score = consider_direction(r, n, fs, best_seq, best_score)
        if np.isfinite(rm) and rm < work_M - 1e-14:
            work = r.copy()
            work_M = rm
        work, work_M = retarget_work(work, work_M, best_seq, best_score, n)

    refine_end = min(dl, t0 + min(1050.0, 0.90 * total))
    if time.monotonic() < refine_end - 20.0:
        r, rm = run_opt(work, n, fs, refine_end, 96, 0.05, 0.90, rng, 300000, 8000)
        best_seq, best_score = consider_direction(r, n, fs, best_seq, best_score)
        if np.isfinite(rm) and rm < work_M - 1e-14:
            work = r.copy()
            work_M = rm
        work, work_M = retarget_work(work, work_M, best_seq, best_score, n)

    polish_end = dl - 2.0
    if time.monotonic() < polish_end - 5.0:
        r, rm = polish_direction(work, n, fs, polish_end, rng)
        best_seq, best_score = consider_direction(r, n, fs, best_seq, best_score)
        if np.isfinite(rm) and rm < work_M - 1e-14:
            work = r.copy()
            work_M = rm
        work, work_M = retarget_work(work, work_M, best_seq, best_score, n)

    final = norm1(work)
    if final is None:
        final = norm1(best_seq)
    if final is None:
        final = const.copy()
    final = np.clip(final, 0.0, 1000.0)
    if not np.isfinite(final).all() or final.sum() < 0.01:
        final = const.copy()

    sc = float(evaluate_sequence(final.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_seq = final.copy()
        best_score = sc

    out = norm1(best_seq)
    if out is None:
        out = const.copy()
    out = np.clip(out, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.ones(max(1, n), dtype=np.float64)
        out /= float(out.size)

    return [float(x) for x in out]
