def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:

def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s <= 0:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = c[:L]
    c = np.maximum(c, 0)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s))

    lr = 0.5
    L = 2 * n - 1
    a = np.exp(y - y.max())

        mc = c.max()
        s = a.sum()
        sc = 2.0 * n * mc / (s * s)

            best_sc = sc
            no_imp = 0
        else:

        beta = min(200.0, beta * (1.0001 if no_imp > 100 else 1.0))
        
        # Soft-max weights on peaks
        z = np.exp(beta * (c - mc) / max(mc, 1e-12))
        zsum = z.sum()
        if zsum < 1e-15 or not np.isfinite(zsum):
            z = np.zeros(L)
            z[np.argmax(c)] = 1.0
        else:
            z /= zsum

        gd = float(np.dot(a, g))
        grad_y = a * (g - gd)

        m = b1 * m + (1 - b1) * grad_y
        v = b2 * v + (1 - b2) * grad_y * grad_y
        t1 = 1 - b1**t
        t2 = 1 - b2**t

        vb = v / t2

        y = y - y.mean()

            v.fill(0)
            t = 0
            lr = max(0.1, lr * 0.5)
            beta = 50.0
            no_imp = 0

            break

        sct = ct.sum()

        T = np.fft.rfft(ct, N)
        a2 = np.fft.irfft(np.sqrt(T), N)
