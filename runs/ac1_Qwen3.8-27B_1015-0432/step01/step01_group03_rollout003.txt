def _pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p


def _conv_fft(a, L):
    return ifft(fft(a, L) * fft(a, L)).real[:2 * len(a) - 1]


def _smobj(a, tau, L):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c = _conv_fft(a, L)
    cm = c.max()
    e = np.exp((c - cm) / tau)
    es = e.sum()
    sm = cm + tau * np.log(es)
    p = e / es
    af = a[::-1]

    s = cv[n - 1:2 * n - 1]
    obj = 2.0 * n * sm / (S * S)
    g = (4.0 * n * s) / (S * S) - (2.0 * n * sm) / (S * S * S)

        try:
            res = minimize(
                _obj_only, best, args=(0.005, L), jac=_grad_only,
                method='L-BFGS-B', bounds=bounds,
                options={'maxiter': 200, 'gtol': 1e-10,
                         'maxls': 20, 'ftol': 1e-15})
            if np.all(np.isfinite(res.x)):
                nx = np.clip(res.x, 0.0, 1000.0)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
        except Exception:
            pass

    # Phase 4: Extended local search (bulk of remaining time)
    t_end4 = dl - 3.0
    if time.monotonic() < t_end4:
        best, best_sc = _local_search(best.copy(), best_sc, rng, n, L, t_end4)

    # Final validation
    if not np.all(np.isfinite(best)) or len(best) == 0:
        best = a0
    final_sc = float(evaluate_sequence([float(v) for v in best]))
    if not np.isfinite(final_sc):
        best = a0
    return [float(v) for v in best]
