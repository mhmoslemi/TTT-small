import numpy as np
import time

def _fft_conv(a):
    """Full linear convolution using FFT."""
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = v.size
    # Bisection on threshold
    # x_i = max(0, v_i - t), sum x_i = 1
    # Sort v in descending order
    u = np.sort(v)[::-1]
    # Find rho: largest index where u_r - (sum_{i<=r} u_i)/r > 0
    cumsum = np.cumsum(u)
    w = 1.0 / np.arange(1, n + 1)
    cond = u - cumsum * w > 0
    if not cond.any():
        # All equal or degenerate
        return np.ones(n) / n
    rho = np.argmax(~cond)  # first False
    t = (cumsum[rho] - 1.0) / (rho + 1)
    return np.maximum(v - t, 0.0)
