import numpy as np
import time

def _np2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def _sc(a, N):
    n = a.size
    s = float(a.sum())
    if s < 1e-15:
        return np.inf
    an = a / s
    A = np.fft.rfft(an, N)
    b = np.fft.irfft(A * A, N)
    mb = float(b[:2 * n - 1].max())
    return 2.0 * n * mb if np.isfinite(mb) else np.inf

def _fmt(v, n, N):
    v = np.maximum(v, 0.0)
    s = float(v.sum())
    if s < 1e-10:
        return np.ones(n) / n
    return v / s

    P = np.clip(P.astype(np.float64), 0.0, 1000.0)
    if P.sum() < 0.01:

    N = _np2(2 * n - 1)
    L = 2 * n - 1
    rng = np.random.default_rng(seed)

    # Build initializations
    cands = [P / P.sum(), np.ones(n) / n]
    x = np.linspace(0.0, 1.0, n)
    for fn in (lambda t: np.sin(np.pi * t),
               lambda t: np.cos(np.pi * (t - 0.5)) ** 2,
               lambda t: np.exp(-3.0 * (t - 0.5) ** 2),
               lambda t: (1.0 - t) ** 1.5,
               lambda t: np.maximum(0.0, 1.0 - 4.0 * (t - 0.5) ** 2),
               lambda t: 1.0 / (1.0 + 2.0 * t * t)):
        v = np.maximum(fn(x), 0.0)
        if v.sum() > 1e-10:
            cands.append(v / v.sum())

    best_a, best_sc = cands[0].copy(), np.inf
    for v in cands:
        s = _sc(v, N)
        if np.isfinite(s) and s < best_sc:

    a = best_a.copy()

    # Adam state

    va = np.zeros(n)
    ta = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = 0.15
    no_imp = 0
    K = 10  # number of top lags for soft gradient

    while time.monotonic() < dl:
        # Self-convolution via FFT
        A = np.fft.rfft(a, N)
        b = np.fft.irfft(A * A, N)[:L]
        if not np.all(np.isfinite(b)):
            break
        mb = float(b.max())
        if mb <= 0:
            break

        # Soft gradient over top K lags
        order = np.argpartition(b, -K)[-K:] if K < L else np.arange(L)
        top = b[order]
        w = np.exp(0.5 * (top - top.max()))
        ws = float(w.sum())
        if ws > 1e-30:
            w /= ws
        else:
            w = np.zeros(K)
            w[int(top.argmax())] = 1.0

                continue

            no_imp = 0

        best_list = [1.0 / n] * n

    return best_list
