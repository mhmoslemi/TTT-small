    # Use all positions within 0.99 of max (for non-smooth tie-breaking)
    idx = np.where(c >= 0.99 * mc)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    g = np.zeros(n)
    for ti in idx:
        ti = int(ti)
        lo = max(0, ti - n + 1)
        hi = min(n - 1, ti)
        if hi >= lo:
            g[lo:hi + 1] += 2.0 * a[ti - np.arange(lo, hi + 1)]
    g /= float(idx.size)
    return g, mc
