import numpy as np
import time

def _fft_full_conv(a, b):
    total = a.size + b.size - 1
    sz = 1
    while sz < total:
        sz <<= 1
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz), sz)[:total]

def _fft_self_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _norm_a(a, n):
    s = a.sum()
    if s < 1e-12:
        return np.ones(n)
    return a * n / s

def _sc_from_corr(corr, n):
    return 2.0 * n * corr.max() / (n * n)

def _make_grad(corr, a, tau_frac):
    n = a.size
    cmax = corr.max()
    spread = cmax - corr.mean() + 1e-12
    tau = tau_frac * spread
    w = np.exp((corr - cmax) / max(tau, 1e-10))

    grad -= grad.mean()
    return grad
