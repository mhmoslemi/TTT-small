    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _solve_block_lp(a, s, b, n, conv_full):
    """Solve LP for block [s, s+b) given current a and conv_full."""
    a_fixed = a  # linearization: use current values for "other" indices
    
    # Compute in-block conv using a_fixed
    block = a_fixed[s:s+b]
    size_ba = 1
    while size_ba < n + b - 1:
        size_ba <<= 1
    fb = np.fft.rfft(block, size_ba)
    fa = np.fft.rfft(a_fixed, size_ba)
    ba = np.fft.irfft(fb * fa, size_ba)[:n + b - 1]

    rest = np.maximum(rest, 0.0)

    else:
