import numpy as np

def _fft_conv(a, b=None):
    """Full convolution of a with itself (or with b if given)."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = 1
    while size < na + nb - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:na + nb - 1]

    if S < 0.01:

    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)
