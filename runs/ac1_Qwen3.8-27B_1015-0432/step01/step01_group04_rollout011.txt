def npow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def pp(a):
    """Project to nonnegative, normalize sum=1."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s <= 0:
        a = np.ones(a.size)
        s = a.size

def scf(a, N):
    """Fast score via FFT."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:2*n-1]
    c = np.maximum(c, 0.0)
    return float(2.0*n*c.max()/(s*s))
