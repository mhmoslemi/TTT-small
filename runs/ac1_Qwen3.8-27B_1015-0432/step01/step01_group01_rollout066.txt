def _normalize1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)

def _lag_grad(a, n, k):
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    g = np.zeros(n)
    if lo <= hi:

    return g

def _grad_topk(a, conv, n, top_k):

    g = np.zeros(n)
