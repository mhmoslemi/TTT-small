import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def compute_grad_centered(f, beta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), 0.0
    w = np.exp(beta * (b / M - 1.0))
    w /= w.max()
    gc = fft_conv(w, f[::-1])
    grad = gc[n - 1:2 * n - 1].copy()
    grad = np.maximum(grad, 0.0)
    # Center: remove component proportional to f
    f2 = np.dot(f, f)
    if f2 > 1e-30:
        rayleigh = np.dot(grad, f) / f2
        grad_c = grad - rayleigh * f
        grad_c = np.maximum(grad_c, 0.0)
        if grad_c.max() > 1e-15:
            grad = grad_c
    return grad, M
