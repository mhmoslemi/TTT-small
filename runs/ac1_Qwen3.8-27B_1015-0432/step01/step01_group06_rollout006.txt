import numpy as np
import time

def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa, sz)[:2 * n - 1]
    return np.maximum(c, 0.0)

def _quick_score(a):
    corr = _fft_conv(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * a.size * corr.max() / (s * s)

def _topk_grad(corr, a, K):
    n = a.size
    m = 2 * n - 1
    K = min(K, m)
    top = np.zeros(m, dtype=np.float64)
    top[-K:] = np.sort(corr)[-K:]
    sel = np.where(corr >= top[-1] - 1e-12)[0]
    if sel.size == 0:
        sel = np.array([np.argmax(corr)])
    weights = corr[sel] - corr[sel].min() + 1e-12
    weights = weights ** 3
    weights /= weights.sum()
    g = np.zeros(n)
    for idx, w in zip(sel, weights):
        ki = int(idx)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        g[lo:hi + 1] += w * a[ki - np.arange(lo, hi + 1)]
    return g

def _gen_variety(n, rng, variant, phase):
    j = np.arange(n)
    if variant == 0:
        return 0.5 + 0.5 * np.cos(2.0 * np.pi * j * (j + 1) / n + phase)
    elif variant == 1:
        return np.abs(np.sin(np.pi * j / n + phase * 0.3)) ** 0.4
    elif variant == 2:
        return np.abs(np.cos(2.0 * np.pi * j * 0.61803398874989 / n + phase))
    elif variant == 3:
        return rng.random(n)
    elif variant == 4:
        blk = max(2, n // 400)
        return np.repeat(rng.random(n // blk + 2), blk)[:n]
    elif variant == 5:
        return 0.5 + 0.3 * np.sin(2.0 * np.pi * j * 3.0 / n + phase) + 0.2 * np.sin(2.0 * np.pi * j * 7.0 / n)
    else:
        return np.ones(n)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(0.0, float(budget_s) - 8.0)

    # Load parent

        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

            base = par * n / par.sum()
        else:
            raise ValueError
    except Exception:
        n = 1024
        base = np.ones(n)

    # Try multiple initializations
    best = base.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_sc):
        best = np.ones(n)
        best_sc = eval_seq_val(best)

    phase = rng.random()
    for v in range(7):
        cand = _gen_variety(n, rng, v, phase * v)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-10:
            continue
        cand = cand * n / s
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = cand.copy()

    a = best.copy()
    v_mom = np.zeros(n)
    mu = 0.75
    lr = 0.4
    K = 48
    it = 0
    stuck = 0
    tau = np.inf

    while time.monotonic() < dl:
        it += 1
        # Adaptive K: increase when stuck
        K_eff = min(128, K + stuck // 50)

        corr = _fft_conv(a)
        sc = 2.0 * n * corr.max() / (a.sum() ** 2)

        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Periodic exact check
        if it % 150 == 0:
            sc_ex = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_ex) and sc_ex < best_sc - 1e-13:

                best = a.copy()

        g = _topk_grad(corr, a, K_eff)
        g -= g.mean()  # remove mean (scale invariance)

        # Nesterov momentum
        v_mom = mu * v_mom + g
        vn = np.linalg.norm(v_mom)
        if vn < 1e-15:

        d = -v_mom / vn

        improved = False
        for f in (1.0, 0.6, 0.36, 0.216, 0.13, 0.078):

            a_new = np.clip(a_new, 0.0, 1000.0)

            if sn < 1e-10:

            a_new = a_new * n / sn
            sc_new = _quick_score(a_new)
            if sc_new < sc - 1e-14:
                a = a_new
                improved = True
                break

        if improved:
            lr = min(lr * 1.015, 4.0)
        else:
            lr = max(lr * 0.72, 0.005)

        # Escape strategies
        if stuck > 300:
            # Structured restart
            v_ = int(rng.integers(0, 7))
            ph = rng.random()
            a_alt = _gen_variety(n, rng, v_, ph)
            # Blend with current to smooth transition
            t_blend = min(0.7, 0.1 * stuck / 300.0)
            a_alt = (1.0 - t_blend) * a + t_blend * a_alt
            a_alt = np.clip(a_alt, 0.0, 1000.0)

                best_sc = sc_alt
            a = a_alt.copy()
            v_mom[:] = 0.0
            lr = 0.4
            stuck = 0
        elif stuck > 80:
            # Mild noise
            noise_scale = 0.02 * max(a.std(), 0.05)
            a = np.clip(a + rng.normal(0.0, noise_scale, n), 0.0, 1000.0)
            s = a.sum()
            a = a * n / s if s > 1e-10 else np.ones(n)
            v_mom[:] = 0.0
            stuck = 0

    # Final validation
    out = np.clip(best, 0.0, 1000.0)
    sc_final = float(evaluate_sequence([float(x) for x in out]))
    if not np.isfinite(sc_final) or sc_final > best_sc + 0.01:
        out = np.ones(n)
    return [float(x) for x in out]
