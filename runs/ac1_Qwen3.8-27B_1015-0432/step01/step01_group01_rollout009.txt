def fft_conv(a, fs):

def proj_simplex(a, s):
    """Project onto {x>=0, sum(x)=s}"""
    x = np.sort(a)[::-1]
    cumsum = np.cumsum(x)
    idx = np.nonzero(x * np.arange(1, len(x)+1) > (cumsum - s))[0]
    if len(idx) == 0:
        return np.full_like(a, s / len(a))
    rho = idx[-1]
    theta = (cumsum[rho] - s) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)

def subgrad_multi(a, conv, top_k):
    """Subgradient from top_k peak lags of full convolution."""
    n = a.size
    peaks = np.argpartition(conv, -top_k)[-top_k:]

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    return grad / top_k
