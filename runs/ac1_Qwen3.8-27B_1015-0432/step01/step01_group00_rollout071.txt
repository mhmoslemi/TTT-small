def fft_conv(a, b):

    x = np.arange(n, dtype=np.float64)
    h = np.exp(-((x - c) / w) ** 2) + np.exp(-((x - (n - 1 - c)) / w) ** 2)
    h = np.maximum(h, 1e-12)
    return h / h.sum()


def compute_gradient(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    wb = b / M
    wb = wb - wb.max()
    w = np.exp(beta * wb)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    f_rev = f[::-1]
    gc = fft_conv(w, f_rev)
    grad = gc[n - 1:2 * n - 1]
    return grad


def run_gradient_opt(f, deadline, rng, start_beta, start_eta):
    n = f.size
    f = f.copy()
    beta = start_beta
    eta = start_eta
    best_local = f.copy()
    best_local_score = fast_score(f)
    no_improve = 0
    restart_count = 0

    for it in range(5000000):
        if time.monotonic() > deadline - 3.0:
            break

        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break

        grad = compute_gradient(f, b, beta)
        gmax = grad.max()
        if gmax < 1e-30:
            break

        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)
        f = f / f.sum()

        if it % 80 == 0:
            sc = fast_score(f)
            if sc < best_local_score - 1e-9:
                best_local_score = sc
                best_local = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 400:
                sc2 = best_local_score
                f = best_local.copy()
                if restart_count < 20:
                    pert_scale = 0.03 + 0.03 * restart_count
                    f = f * np.exp(pert_scale * rng.standard_normal(n))
                    f = np.maximum(f, 1e-20)
                    f = f / f.sum()
                    restart_count += 1
                no_improve = 0
                eta = min(eta * 1.3, 0.5)
                beta = max(3.0, beta * 0.85)
                if restart_count > 20:
                    break

        if it % 500 == 0 and it > 0:
            beta = min(80.0, beta * 1.04)
            eta = max(0.005, eta * 0.995)

    return best_local, best_local_score


def run_peak_shaving(f, deadline, rng):
    n = f.size
    f = f.copy()
    best = f.copy()
    best_sc = fast_score(f)

    for it in range(20000):
        if time.monotonic() > deadline - 3.0:
            break

        b = fft_conv(f, f)
        M = b.max()
        k_star = int(b.argmax())

        contrib = np.zeros(n)
        lo = max(0, k_star - (n - 1))
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] += f[idx]
                contrib[idx] += f[j]

        cmax = contrib.max()
        if cmax < 1e-30:
            break

        pen = contrib / cmax
        f = f * np.exp(-0.15 * pen * pen)
        f = np.maximum(f, 1e-20)
        f = f / f.sum()

        if it % 100 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-9:
                best_sc = sc
                best = f.copy()

    return best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0
    n = len(height_sequence_1)

    inits = []
    pp = make_init_parent(n)
    if pp is not None:
        inits.append(pp)
    inits.append(np.ones(n) / n)
    inits.append(make_init_hann(n))
    inits.append(make_init_cosine(0.5, n))
    inits.append(make_init_cosine(1.0, n))
    inits.append(make_init_bimodal(n, rng))
    inits.append(make_init_bimodal(n, rng))

    best_f = None
    best_score = np.inf
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = init.copy()

    if best_f is None:
        best_f = np.ones(n)
        best_score = 2.0

    # Phase 2: main gradient optimization
    f = best_f.copy()
    for attempt in range(6):
        if time.monotonic() > deadline - 30.0:
            break
        f_opt, sc_opt = run_gradient_opt(f, deadline - 5.0, rng,
                                         start_beta=8.0, start_eta=0.1)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
        f = best_f * np.exp(0.04 * rng.standard_normal(n))
        f = np.maximum(f, 1e-20)
        f = f / f.sum()

    # Phase 3: peak shaving
    if time.monotonic() < deadline - 15.0:
        b = fft_conv(best_f, best_f)
        f_shave, sc_shave = run_peak_shaving(best_f, deadline - 3.0, rng)
        if sc_shave < best_score:
            best_score = sc_shave
            best_f = f_shave.copy()
            f_opt, sc_opt = run_gradient_opt(f_shave, deadline - 5.0, rng,
                                             start_beta=12.0, start_eta=0.06)
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()

    # Phase 4: fine-tuning with small perturbations
    if time.monotonic() < deadline - 5.0:
        for _ in range(30):
            if time.monotonic() > deadline - 3.0:
                break
            trial = best_f * np.exp(0.01 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial = trial / trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()

    # Final score with evaluator
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0] * n
        final_sc = evaluate_sequence(result)

    return result
