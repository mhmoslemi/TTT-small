def _sw_grad(a, c, T):
    """Softmax-weighted gradient of max(c)."""
    n = a.size
    mx = c.max()
    w = np.exp((c - mx) / T)
    w /= w.sum()
    # Only process lags with significant weight
    thresh = 1e-10
    active = w > thresh
    wc = w[active]
    kc = np.where(active)[0]
    g = np.zeros(n)
    for ki, wi in zip(kc, wc):
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        if lo <= hi:
            g[lo:hi + 1] += wi * a[ki - hi:ki - lo + 1][::-1]
    return g
