But for nonnegative sequences, the constraint is tighter. C[n-1] = sum a[i]*a[n-1-i]. For a uniform sequence, all C[k] are triangular (peaking at n-1). To flatten C, we need structure.

def _fc(a, fs):
    """Full autoconvolution via FFT; output length fs >= 2*len(a)-1."""
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)

def _sg(a, lags):
    """Normalised subgradient: mean of dC[k]/da over given lags."""
    n = a.size
    g = np.zeros(n)
    for k in lags:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    m = len(lags)
    if m:
        g /= m
    return g
