import numpy as np
import time

def _fft_conv(a):
    n = a.size
    M = 1
    while M < 2 * n - 1:
        M <<= 1
    fa = np.fft.rfft(a, M)
    return np.fft.irfft(fa * fa, M)[:2 * n - 1]

def _score(corr, n):
    return 2.0 * n * corr.max() / (n * n)

def _subgrad(corr, a, n, K):
    M = corr.size
    K = min(K, M)
    thresh = np.partition(corr, M - K)[M - K]
    idx = np.where(corr >= thresh)[0]
    if idx.size == 0:
        idx = np.array([np.argmax(corr)])
    g = np.zeros(n)
    for k in idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            partner = k - j
            if 0 <= partner < n:
                g[j] += a[partner]
    g /= idx.size
    g -= g.mean()
    return g

def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)

    if s < 1e-10:

    return a * a.size / s

        a = np.abs(np.sin(np.pi * j / n))
    elif kind == 3:
        a = rng.random(n)
    elif kind == 4:
        a = np.abs(np.cos(2.0 * np.pi * j / (n + 1)))
    elif kind == 5:
        blk = max(2, n // 200)
        a = np.repeat(rng.random(n // blk + 2) * 0.8 + 0.1, blk)[:n]
    else:
        a = 0.3 + 0.7 * np.abs(np.sin(3.14 * j * 1.3 / n))
    return _normalize(a)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        n = par.size
    except Exception:
        n = 1024
    if n < 2:
        n = 1024

    best = None
    best_sc = np.inf
    inits = []
    try:
        if par.size == n and par.sum() > 1e-10:
            inits.append(_normalize(par.copy()))
    except Exception:
        pass
    for k in range(7):
        inits.append(_gen_struct(n, rng, k))

    for a_try in inits:
        sc = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a_try.copy()
    if best is None:
        best = np.ones(n)
        best_sc = float(evaluate_sequence(best.tolist()))

    a = best.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step_t = 1
    stuck = 0

    while time.monotonic() < deadline:
        corr = _fft_conv(a)
        sc = _score(corr, n)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 0 and stuck % 200 == 0:
            sc_ex = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc_ex) and sc_ex < best_sc - 1e-12:
                best_sc = sc_ex
                best = a.copy()
                stuck = 0

        g = _subgrad(corr, a, n, 64)
        step_t += 1
        lr = 0.7 / np.sqrt(step_t + 1.0)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** step_t)
        v_hat = v / (1 - beta2 ** step_t)
        d = -m_hat / (np.sqrt(v_hat) + eps)

        ok = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03):
            a_new = _normalize(a + lr * scale * d)
            sc_new = _score(_fft_conv(a_new), n)
            if sc_new < sc - 1e-14:
                a = a_new
                ok = True
                break

        if not ok:
            lr_try = lr * 5.0
            for scale in (1.0, 0.5, 0.25):
                a_new = _normalize(a + lr_try * scale * d)
                sc_new = _score(_fft_conv(a_new), n)
                if sc_new < sc - 1e-14:
                    a = a_new
                    ok = True
                    break

        if stuck > 150:
            nm = int(rng.integers(0, 7))
            a_alt = _gen_struct(n, rng, nm)
            a_alt = _normalize(a_alt + 0.3 * a)
            sc_alt = _score(_fft_conv(a_alt), n)
            if sc_alt < sc:
                a = a_alt
                m[:] = 0.0
                v[:] = 0.0
                step_t = 0
                stuck = 0
            elif stuck > 400:
                a = _normalize(a + rng.normal(0, 0.05 * max(a.std(), 0.1), n))
                m[:] = 0.0
                v[:] = 0.0
                step_t = 0
                stuck = 0

    out = _normalize(best)
    return [float(x) for x in out]
