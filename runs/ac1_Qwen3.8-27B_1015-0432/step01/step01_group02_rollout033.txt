import numpy as np
import time


def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _sc(a, n):
    c = _conv(a, n)
    s = a.sum()
    if s < 1e-9:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _overlap(c, a, n, fraction=0.90, topK=12):
    pmax = c.max()
    if pmax < 1e-18:
        return None
    thresh = pmax * fraction
    idx = np.where(c >= thresh)[0]
    if len(idx) < 3:

        idx = np.argpartition(c, -kk)[-kk:]
    h = np.zeros(n)

        kj = int(kj)
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += a[kj - np.arange(lo, hi + 1)]
    return h
