    """Generate diverse initializations"""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Hann window
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1 - np.cos(2*np.pi*x / n))

    # Power law (various exponents)
    for pwr in [0.5, 1.0, 1.5, 2.0, 3.0]:
        d = np.minimum(x, n-1-x) + 1.0
        pw = 1.0 / d**pwr
        inits.append(pw / pw.sum())

    r = x.astype(np.float64)
    inits.append(r / r.sum())
    r2 = (n-1-x).astype(np.float64)
