import numpy as np
import time

def next_pow2(x):
    x = max(1, int(x))
    m = 1
    while m < x:
        m <<= 1
    return m

def fft_conv(a, N):
    n = a.size
    L = 2 * n - 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:L]

def score_fast(a, N):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = fft_conv(a, N)
    c = np.clip(c, 0, None)
    return float(2.0 * n * c.max() / (s * s))

def proj(a, n):
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones(n) / n
    return a

def init_two_peak(n):
    j = np.linspace(0, 1, n)
    a = np.exp(-12 * j * j) + np.exp(-12 * (1 - j) * (1 - j))
    a += 0.005
    a /= a.sum()
    return a

def init_w_shape(n):
    j = np.linspace(0, 1, n)
    a = np.exp(-15 * j * j) + np.exp(-15 * (1 - j) * (1 - j))
    a += 2.0 * np.exp(-6 * (j - 0.5) * (j - 0.5))
    a += 0.003
    a /= a.sum()
    return a

def init_rising(n):
    j = np.linspace(0, 1, n)
    a = j * j + np.exp(-8 * (1 - j) * (1 - j))
    a += 0.01
    a /= a.sum()
    return a

def spectral_step(a, alpha, N, mix):
    n = a.size
    c = fft_conv(a, N)
    c = np.clip(c, 0, None)
    cs = c.sum()
    if cs < 1e-300:
        return a.copy()
    cn = c / cs
    ca = cn ** alpha
    sa = ca.sum()
    if sa < 1e-300:
        return a.copy()
    ca /= sa
    T = np.fft.rfft(ca, N)
    T = np.clip(T, 0, None)
    A = np.fft.rfft(a, N)
    An = A * np.sqrt(T)
    an = np.fft.irfft(An, N)[:n]
    an = np.where(np.isfinite(an), an, 0.0)
    an = (1.0 - mix) * a + mix * an
    return proj(an, n)

def run_spectral(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_sc = score_fast(a, N)
    cur = a.copy()
    cur_sc = best_sc
    it = 0
    stall = 0
    while time.monotonic() < deadline:
        alpha = max(0.25, 0.85 - 0.0015 * it)
        mix = max(0.25, 0.85 - 0.002 * it)
        cand = spectral_step(cur, alpha, N, mix)
        sc = score_fast(cand, N)
        if np.isfinite(sc):
            if sc < best_sc - 1e-12:
                best = cand.copy()
                best_sc = sc
                stall = 0
            else:
                stall += 1
            if sc <= cur_sc * 1.04:
                cur = cand
                cur_sc = sc
            else:
                cand2 = 0.7 * cur + 0.3 * spectral_step(cur, alpha, N, 0.3)
                cand2 = proj(cand2, n)
                sc2 = score_fast(cand2, N)
                if np.isfinite(sc2) and sc2 < cur_sc:
                    cur = cand2
                    cur_sc = sc2
                    if sc2 < best_sc:
                        best = cand2.copy()
                        best_sc = sc2
                        stall = 0
        it += 1
        if stall > 120:
            cur = proj(best * (1.0 + 0.04 * rng.normal(0, 1, n)), n)
            cur_sc = score_fast(cur, N)
            stall = 0
            it = max(0, it - 40)
    return best, best_sc

def run_log_adam(a, n, N, deadline, seed):

    a_pos = np.maximum(a, 1e-300)

    stall = 0
    L = 2 * n - 1
    while time.monotonic() < deadline:
        ys = y - y.max()
        a_s = np.exp(ys)
        a_s /= a_s.sum()
        c = fft_conv(a_s, N)
        c = np.clip(c, 0, None)
        mc = float(c.max())
        sc = float(2.0 * n * mc)
        if np.isfinite(sc) and sc < best_sc:
            best = a_s.copy()
            best_sc = sc
            stall = 0
        else:
            stall += 1
        if stall > 400:
            y = np.log(np.maximum(best, 1e-10))
            y -= y.mean()
            y = np.clip(y + 0.04 * rng.normal(0, 1, n), -20, 20)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            b1c = 1.0
            b2c = 1.0
            lr = max(0.008, lr * 0.5)
            stall = 0
            continue
        k_star = int(np.argmax(c))
        i_lo = max(0, k_star - n + 1)
        i_hi = min(n - 1, k_star)
        G = np.zeros(n)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            G[idx] = a_s[k_star - idx]
        # Soft weighting: include near-max lags
        c_thresh = mc * 0.95
        near_lags = np.where(c >= c_thresh)[0]
        for kl in near_lags:
            if kl == k_star:
                continue
            lo2 = max(0, kl - n + 1)
            hi2 = min(n - 1, kl)
            if lo2 <= hi2:
                idx2 = np.arange(lo2, hi2 + 1)
                w = c[kl] / mc
                G[idx2] += w * a_s[kl - idx2]
        grad = a_s * G
        grad -= grad @ a_s
        grad *= 2.0 * n
        t += 1
        b1c *= b1
        b2c *= b2
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mb = m / (1.0 - b1c)
        vb = v / (1.0 - b2c)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.04, 0.04)
        y = y - step
        y -= y.mean()
        y = np.clip(y, -20, 20)
    return best, best_sc
