    """Minimize f on [a,b] using golden section search. f must be unimodal."""
    gr = (np.sqrt(5) - 1) / 2  # golden ratio ~0.618

    d = 2.0 * (x - v) * a_seg
    if two_k_in:
        d[two_k_off] = x * x - v * v
    sm = float((base_seg + d).max())
    fm = sm if sm > unaff_max else unaff_max
    ns = sum_rest + x
    if ns < 1e-10:
        return 1e10
    return fm / (ns * ns)

        stall = 0

            order = rng.permutation(n)

                if m_end + 1 < 2 * n - 1:
