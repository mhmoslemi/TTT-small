def _fft_conv(a):

    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)

    m = len(b)

    result = np.fft.irfft(fa * fb_flip, size)

    return result[n-1:n-1+m]

def _soft_grad(c, beta):
    """Compute soft-max phi and its gradient contribution.
    c: full convolution (length 2n-1)
