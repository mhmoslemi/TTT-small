import numpy as np
import time

def _fft_self_conv(a, L):
    n = a.size
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]

def _smooth_grad(corr, a, n, L, T):
    cmax = corr.max()
    w = np.exp(corr / T - cmax / T)
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    else:
        w /= ws
    a_rev = a[::-1]
    Wf = np.fft.rfft(w, L)
    Af = np.fft.rfft(a_rev, L)
    cw = np.fft.irfft(Wf * Af, L)
    grad = (2.0 / T) * cw[n - 1:2 * n - 1]
    grad -= grad.mean()
    return grad

def _normalize(a, n):
    sn = a.sum()
    if sn < 1e-12:
        return np.ones(n)
    return a * n / sn
