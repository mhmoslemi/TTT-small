```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def fft_cross_corr(a, b):
    # a: length p >= q: length b
    # returns (a ⋆ b)[k] = sum_j a[j+k]*b[j] for k=0..p-q
    p, q = len(a), len(b)
    L = 1 << (p + q - 1).bit_length()
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(np.conj(A) * B, L)[:p - q + 1]

def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))
```

def optimize_smoothed(f, deadline, rng, beta_start=5.0, beta_end=500.0, iters_max=200000):
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_score = fast_score(f)
    no_improve = 0
    
    for it in range(iters_max):
        if time.monotonic() > deadline - 5:
            break
        
        # Adaptive beta: start smooth, end sharp (approximating max)
        progress = min(1.0, it / iters_max)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        C = fft_conv(f, f)
        M = C.max()
        
        # Compute smooth subgradient
        # w[k] = exp(beta * C[k] / M)
        beta_C = beta * C / M
        beta_C -= beta_C.max()  # for numerical stability
        w = np.exp(beta_C)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w /= w_sum

        corr = fft_cross_corr(w, f)  # length: (2n-1) - n + 1 = n

        # Actually E_smooth should approximate M (the max). Since w = exp(beta*C/M)/sum(exp(...)),
        # E = M * sum(exp(beta*(C-M)/M)) / (2n-1) * (2n-1) ... 
        # Let's just use M as the smooth approximation to max.

        # Multiplicative update
        gmax = grad.max()
        gmin = grad.min()
        if gmax - gmin < 1e-30:
            break
        
        # Normalize gradient
        grad_norm = np.abs(grad).max()
        if grad_norm < 1e-30:
            break
        
        # Step size
        eta = 0.1 * max(0.01, 1.0 - progress)
        
        f_new = f * np.exp(-eta * grad / grad_norm)
        f_new = np.maximum(f_new, 1e-20)
        f_new /= f_new.sum()
        
        # Check score
        if it % 100 == 0:
            sc = fast_score(f_new)

                no_improve = 0

                f_new /= f_new.sum()
                no_improve = 0

    B = np.fft.rfft(b, L)
    return np.fft.irfft(np.conj(A) * B, L)[:p - q + 1]

def score_fast(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    C = fft_conv(f, f)
    return 2.0 * len(f) * C.max() / (s * s)

def make_init_parent(hs):
    p = np.clip(np.asarray(hs, dtype=np.float64), 0.0, 1000.0)
    if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
        return p / p.sum()
    return None

def make_init_cosine(n):
    x = np.arange(n, dtype=np.float64)
    h = np.cos(np.pi * x / (n - 1) / 2.0) ** 2
    h = np.maximum(h, 1e-10)
    return h / h.sum()

def make_init_ramp(n):
    x = np.arange(n, dtype=np.float64)
    h = x + 1.0
    return h / h.sum()

def make_init_bimodal(n, rng):
    c1 = rng.integers(1, n // 2)
    c2 = n - c1 + rng.integers(-n // 4, n // 4 + 1)
    c2 = min(max(c2, n // 4), n - 1)
    s1 = n // 8
    s2 = n // 8
    h = np.exp(-((np.arange(n) - c1) / s1) ** 2) + np.exp(-((np.arange(n) - c2) / s2) ** 2)
    h = np.maximum(h, 1e-10)
    return h / h.sum()

def run_smoothed_descent(f, deadline, rng, beta0=3.0, beta1=200.0, max_iters=100000):
    n = f.size
    f = f.copy()
    best_f = f.copy()
    best_sc = score_fast(f)
    no_improve = 0
    t_inner = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 3.0:
            break
        
        # Adaptive beta
        frac = min(1.0, (it + 1) / max_iters)
        beta = beta0 * (beta1 / beta0) ** frac
        
        C = fft_conv(f, f)
        M = C.max()
        if M < 1e-30:
            break
        
        # Weight vector
        bc = beta * C / M
        bc -= bc.max()
        w = np.exp(bc)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(C))
            w[:n] = 1.0 / n
        else:
            w /= ws
        
        # Smooth subgradient: corr[i] = sum_k w[k]*f[k-i] (valid cross-corr)
        corr = fft_cross_corr(w, f)  # length n
        
        # Score gradient direction (up to positive scale)
        sum_f = f.sum()
        grad = corr - (M / sum_f) * f
        
        # Multiplicative weight update
        gmax = np.abs(grad).max()
        if gmax < 1e-30:
            break
        
        eta = 0.15 * (1.0 - 0.5 * frac)
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        
        t_inner += 1
        if t_inner % 50 == 0:
            sc = score_fast(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 2000:
                # Perturb
                scale = 0.05 + 0.1 * rng.random()
                f = f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improve = 0
    
    return best_f, best_sc

w has length 2n-1 (from conv(f,f) where f has length n).
f has length n.

fft_cross_corr(w, f):
- a = w (length 2n-1), b = f (length n)
- p = 2n-1, q = n
- output length: p - q + 1 = (2n-1) - n + 1 = n ✓
- output[k] = sum_{j=0}^{n-1} w[j+k] * f[j] for k = 0, ..., n-1

So corr[k] = sum_{j=0}^{n-1} w[j+k] * f[j]

This is: for the k-th "column" of the gradient, we're computing the weighted sum of f with weights w[k], w[k+1], ..., w[k+n-1].

The subgradient of max(C)[k_max] with respect to f[i] is 2*f[k_max - i] (for valid k_max - i).

The smoothed version: sum_k w[k] * 2*f[k-i] (for valid k-i) = 2 * sum_{k: 0<=k-i<n} w[k]*f[k-i] = 2 * sum_{j: 0<=j<n} w[i+j]*f[j] = 2*corr[i].

Yes! So the smoothed subgradient of max is 2*corr, and the score gradient is:
dS/df[i] ∝ 2*corr[i] - 2*M*f[i]/sum_f

So grad = corr - M*f/sum_f ✓ (dropping the factor of 2 and positive constant).

One thing I notice: the parent program's `run_lp_update` function computes a specific direction based on the 1/f weights. This is actually solving a simplified LP (the "diagonal" approximation). Let me include a similar step.

The LP: max sum(g) s.t. conv(f,g) <= M, g >= 0.
A simple approximation: g[i] = 1 / max(f[0:n]) ... no.

Actually, the constraint conv(f,g)[k] <= M for all k means:
For each k: sum_{i in valid range for k} f[i] * g[k-i] <= M

A sufficient condition: for all i, f[i] * g[i] <= M/(n-1) (rough upper bound).
Better: g[i] <= M / (sum of f[j] for j where k-j = i for some k) = M / (sum of all f) = M/(sum_f).

So g = M/(sum_f) * ones(n) is feasible (barely). The LP will do better.

A greedy approach: start with g = 0, and for each i (in some order), set g[i] as large as possible while maintaining feasibility.
