def propose_candidate(seed=42, budget_s=1100, **kw):

        p = np.ones(n)

    rc = np.maximum(rc, 0.01)

    hn = 0.5 - 0.5 * np.cos(2.0 * np.pi * idx / n)
    hn = np.maximum(hn, 0.01)
    inits.append(hn / hn.sum() * S0)
    # Power-law bump
    t = (idx + 0.5) / n - 0.5
