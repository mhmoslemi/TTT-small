def fft_conv(a, size):
    fa = np.fft.rfft(a, n=size)
    return np.fft.irfft(fa * fa, n=size)

def topk_grad(a, conv, k):
    """Subgradient of max(conv) using top-k lags. O(k*n)."""
    n = a.size
    idx = np.argpartition(conv, -k)[-k:]
    grad = np.zeros(n)
    for lag in idx:
        lag = int(lag)
        lo = max(0, lag - (n - 1))
        hi = min(n - 1, lag)
        if lo <= hi:
            grad[lo:hi+1] += 2.0 * a[lag - np.arange(lo, hi+1)]
    return grad / k

    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)

    return a, m, v
