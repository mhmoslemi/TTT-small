def _fft_xconv(a, b, N):
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    return np.fft.irfft(fa * fb, n=N)

    a_rev = a[::-1]
    fa = np.fft.rfft(a, n=CN)
    fr = np.fft.rfft(a_rev, n=CN)
    fp = np.fft.rfft(p, n=CN)
