import numpy as np
import time


def fft_conv(a, N):
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)


def subgrad(a, conv, frac, kmax):
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n)
    idx = np.where(conv >= M * frac)[0]
    if len(idx) > kmax:
        idx = np.argpartition(conv, -kmax)[-kmax:]
    elif len(idx) < 3:
        idx = np.argpartition(conv, -3)[-3:]
    g = np.zeros(n)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    g /= len(idx)
    return g


def psimp(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a / s

        a = np.clip(a, 0.01, None)
    r = psimp(a)
