        return 1e10, np.zeros(n)
    L1 = _fp2(2 * n - 1)
    fa = fft(a, L1)
    b = ifft(fa * fa).real[:2*n-1]
    cm = b.max()
    if tau < 1e-10:
        m = (b == cm)
        p = m.astype(np.float64) / m.sum()
        M = cm
    else:
        e = np.exp((b - cm) / tau)
        s = e.sum()
        M = cm + tau * np.log(s)
        p = e / s
    q = p[::-1]
    L2 = _fp2(3 * n - 2)
    cv = ifft(fft(q, L2) * fft(a, L2)).real
    c = cv[2*n-2 : n-2 : -1]
    obj = 2.0 * n * M / (S * S)
    grad = (4.0 * n / (S * S)) * c - 4.0 * n * M / (S * S * S)
    return obj, grad
