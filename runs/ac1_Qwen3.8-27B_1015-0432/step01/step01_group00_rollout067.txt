import numpy as np
import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def init_parent():
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = np.maximum(p, 0.0)
    if p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
        return p / p.sum()
    return None


def init_hann(n):
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    h = np.maximum(h, 1e-12)
    return h / h.sum()


def init_powlaw(n, p):
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    pw = 1.0 / d ** p
    return pw / pw.sum()


def init_perturb(n, rng, scale):
    u = 1.0 + scale * rng.standard_normal(n)
    u = np.maximum(u, 0.01)
    u = 0.5 * (u + u[::-1])
    return u / u.sum()


def grad_direction(f, b, beta):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.exp(beta * b / M)
    wmax = w.max()
    if wmax > 0:
        w /= wmax
    w_rev = w[::-1]
    gc = fft_conv(f, w_rev)
    grad = 2.0 * gc[n - 1:2 * n - 1][::-1]
    return np.maximum(grad, 0.0)


def opt_loop(f, deadline, rng):
    n = f.size
    f = f.copy()
    best_f = f.copy()
    best_sc = fast_score(f)
    beta = 25.0
    eta = 0.10
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 8.0:
        b = fft_conv(f, f)
        grad = grad_direction(f, b, beta)
        gm = grad.max()
        if gm < 1e-30:
            break

        f_new = f * np.exp(-eta * grad / gm)
        f_new = np.maximum(f_new, 1e-18)
        f_new /= f_new.sum()
        f = f_new
        it += 1

        if it % 150 == 0:
            sc = fast_score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > 1500:
                f = best_f * np.exp(0.06 * rng.standard_normal(n))
                f = np.maximum(f, 1e-18)
                f /= f.sum()
                beta = 25.0
                eta = 0.10
                no_improve = 0

            if it > 300 and it % 1500 == 0:
                beta = min(400.0, beta * 1.06)
                eta = max(0.004, eta * 0.965)

    return best_f, best_sc


def peak_shave(f, deadline, rng):
    n = f.size
    best_f = f.copy()
    best_sc = fast_score(f)

    for _ in range(60):
        if time.monotonic() > deadline - 6.0:
            break
        b = fft_conv(best_f, best_f)
        M = b.max()
        top = np.argpartition(b, -15)[-15:]
        mask = b[top] > 0.85 * M
        if not mask.any():
            break
        active = top[mask]
        pressure = np.zeros(n)
        for k in active:
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            for j in range(jlo, jhi + 1):
                idx = k - j
                if 0 <= idx < n:
                    pressure[idx] += best_f[j]
        pm = pressure.max()
        if pm < 1e-30:
            break
        for t in (0.03, 0.07, 0.15, 0.3):
            cand = best_f * np.exp(-t * pressure / pm)
            cand = np.maximum(cand, 1e-18)
            cand /= cand.sum()
            sc = fast_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_f = cand.copy()
                break

        sc = fast_score(trial)

            best_f = trial.copy()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 18.0
    n = len(height_sequence_1)

    inits = []
    pp = init_parent()
    if pp is not None:
        inits.append(pp)
    inits.append(init_hann(n))
    inits.append(init_powlaw(n, 0.3))
    inits.append(init_powlaw(n, 0.7))
    inits.append(init_powlaw(n, 1.2))
    inits.append(init_powlaw(n, 2.0))
    for _ in range(2):
        inits.append(init_perturb(n, rng, 0.25))
    for _ in range(2):
        inits.append(init_perturb(n, rng, 0.5))

    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_f = init.copy()

        best_sc = 2.0

    phase_budget = (deadline - t0) * 0.55

        best_sc = sc1
        best_f = f1.copy()

    d2 = t0 + max(30.0, (deadline - t0) * 0.85)

        best_sc = sc3
        best_f = f3.copy()

    result = [float(x) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
