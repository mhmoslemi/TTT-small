import numpy as np
import time


def _conv(a):
    n = len(a)
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real


def _cross_corr(w, a):
    n = len(a)
    N = 1
    needed = len(w) + n - 1
    while N < needed:
        N <<= 1
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(np.conj(W) * A, N).real[:n]


def _score(c, n):
    return 2.0 * n * c.max()
