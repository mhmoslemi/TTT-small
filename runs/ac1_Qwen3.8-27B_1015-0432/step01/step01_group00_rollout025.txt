def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def sym(f):
    return 0.5 * (f + f[::-1])


def norm1(f):
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.size
    return f / s


def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max() / (s * s)
