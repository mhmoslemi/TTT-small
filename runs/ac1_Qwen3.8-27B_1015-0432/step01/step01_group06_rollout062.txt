import numpy as np

def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa, sz)
    return c[:2 * n - 1]

def _smooth_grad(conv, a, n, T):
    mc = conv.max()
    exp_c = np.exp(T * (conv - mc))
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_c / s
    sz = 1
    while sz < 3 * n - 2:
        sz <<= 1
    W = np.fft.rfft(w, sz)
    A = np.fft.rfft(a, sz)
    corr = np.fft.irfft(W * np.conj(A), sz)[:n]
