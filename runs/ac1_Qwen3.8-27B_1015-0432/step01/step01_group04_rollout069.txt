import numpy as np

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def project_positive(a, fb):
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    fb = np.asarray(fb, dtype=np.float64).reshape(-1).copy()
    if a.size != fb.size:
        a = fb.copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = fb.copy()
        s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(max(1, a.size))
    return a / max(float(a.sum()), 0.01)


def score_fft(a, N):
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.maximum(np.where(np.isfinite(c), c, 0.0), 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def spec_step(a, N, alpha, mix, use_phase):
    n = a.size
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = np.where(np.isfinite(c[:L]), c[:L], 0.0)
    c = np.maximum(c, 0.0)
    cs = c.sum()
    if cs <= 0.0:
        return a
    c = c / cs
    ct = np.power(c + 1e-20, alpha)
    sct = ct.sum()
    if sct <= 0.0:
        return a
    ct = ct / sct
    T = np.fft.rfft(ct, N)
    T = np.maximum(np.where(np.isfinite(T), T, 0.0), 0.0)
    A2 = np.sqrt(T)
    if use_phase:

    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.maximum(np.where(np.isfinite(a2), a2, 0.0), 0.0)
    s2 = a2.sum()
    if s2 <= 0.0:
        return a
    a_norm = a / a.sum()
    return project_positive((1.0 - mix) * a_norm + mix * a2 / s2, a)
