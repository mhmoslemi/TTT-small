        # Let j' = n-1-j, k' = n-1-k-j. Then b[2n-2-k] = sum_j f[n-1-j']*f[n-1-k-j']
        # = sum_j f_rev[j']*f_rev[k-j'] = (f_rev*f_rev)[k] = b_rev[k]
        # So b[2n-2-k] = b_rev[k] = (f_rev*f_rev)[k]. This is NOT the same as b[k] in general!
        # Unless f is symmetric.
        
        # So in general, w is NOT symmetric, and we need both terms.
        # sum_k w[k] * f[i-k] = sum_j w[i-j] * f[j] (letting j = i-k, k = i-j)
        # = (w_rev * f)[i] where w_rev[j] = w[2n-2-j]
        # Hmm, this is the convolution of the reversed w with f, at position i.
        
        # Let me just compute both terms:
        f_rev = f[::-1].copy()
        grad1 = fft_conv(w, f_rev)[n-1:2*n-1]  # sum_k w[k]*f[k-i]
        # For the second term: sum_k w[k]*f[i-k]
        # = sum_j w[i-j]*f[j] for valid i-j in [0,2n-2]
        # Pad w to be a sequence indexed 0..2n-2, then this is (w * f)[i] where we treat w as the longer sequence
        # Actually: (f * w)[i] = sum_j f[j]*w[i-j] which is what we want!
        grad2_raw = fft_conv(f, w)  # length n + (2n-1) - 1 = 3n-2
        grad2 = grad2_raw[0:n]  # positions 0 to n-1
        
        grad = grad1 + grad2
        grad = np.maximum(grad, 0.0)
        
        # Actually wait, let me double-check. The gradient of the smooth max is:
        # d/d a_i [sum_k (w_k/M) * b_k] where w_k = exp(beta * b_k / M) / sum
        # But M also depends on a, so this is more complex.
        # For simplicity, let's just use the raw gradient without the M derivative.
        # This is a common approximation that works well in practice.
        
        gmax = grad.max()
        if gmax < 1e-30:
            break
        
        # Multiplicative update
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        
        # Periodic scoring
        if iter_count % 100 == 0:
            sc = fast_score(f)
            if sc < best_local_score - 1e-9:
                best_local_score = sc
                best_local = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 500:
                # Restart perturbation
                f = best_local * np.exp(0.05 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improve = 0
                eta = min(eta * 1.5, 0.3)
                beta = max(5.0, beta * 0.8)
                if no_improve > 2000:
                    break
            
        # Adapt parameters
        if iter_count % 500 == 0:
            beta = min(100.0, beta * 1.02)
            eta = max(0.01, eta * 0.99)
    
    return best_local, best_local_score
