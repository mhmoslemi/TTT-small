def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _grad_and_val(a, tau, N_conv, N_corr):
    n = a.size
    # Convolution
    A = fft(a, N_conv)
    b = ifft(A * A).real[:2*n-1]

    E = np.exp((b - bmax) / tau)

    M = bmax + tau * np.log(E_sum)

    Pa = fft(p, N_corr)

    S = a.sum()
    grad = (4.0*n / S**2) * r - 4.0*n*M / S**3
