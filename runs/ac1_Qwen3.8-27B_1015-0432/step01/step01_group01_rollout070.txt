import numpy as np
import time


def fft_self_conv(a, fs):
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)


def fft_cross_conv(a, b, fs):
    return np.fft.irfft(np.fft.rfft(a, fs) * np.fft.rfft(b, fs), fs)


def make_init(kind, n, rng, parent):
    if kind == 0:
        a = parent / max(parent.sum(), 1e-14)
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        f = rng.uniform(0.08, 0.35)
        ph = rng.uniform(0, 2 * np.pi)
        a = np.clip(1.0 + 0.55 * np.sin(2 * np.pi * f * np.arange(n) + ph), 0.02, None)
    elif kind == 3:
        a = np.abs(rng.standard_normal(n)) ** 1.5
    elif kind == 4:
        block = max(4, n // 80)
        a = np.repeat(rng.uniform(0.2, 1.6, (n // block) + 1), block)[:n]
    elif kind == 5:

        h = n // 3
        a = np.where((m < h) | (m > 2 * h), 1.4, 0.3)
    elif kind == 6:
        f = rng.uniform(0.05, 0.2)
        m = np.arange(n)
        a = np.clip(1.0 + 0.4 * np.cos(2 * np.pi * f * m) + 0.2 * np.cos(4 * np.pi * f * m), 0.02, None)
    else:
        a = np.ones(n) * (0.5 + 0.5 * np.cos(2 * np.pi * np.arange(n) / n))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(n)
        s = n
    return a / s


def optimize_smooth(a, n, fs, deadline, p_start, p_end, n_iter, lr0, mom):
    a = a.copy()
    v = np.zeros(n)
    for it in range(n_iter):
        if time.monotonic() > deadline - 8:
            break
        b = fft_self_conv(a, fs)
        mb = float(b.max())
        if mb < 1e-16:
            break
        t = it / max(1, n_iter - 1)
        p = p_start + (p_end - p_start) * t
        lr = lr0 * (0.02 / lr0) ** t
        w = np.exp(p * (b - mb) / mb)
        sw = float(w.sum())
        if sw < 1e-14:
            w = np.ones_like(b)
            sw = len(b)
        w /= sw
        grad = 2.0 * fft_cross_conv(w, a, fs)[:n]
        v = mom * v + (1.0 - mom) * grad
        v -= v.mean()
        a = np.clip(a - lr * v, 0.0, None)
        s = a.sum()
        if s < 1e-14:
            break
        a /= s
    return a

    p2_start = p_end * 0.5
    p2_end = p_end * 3.0
    n2 = max(500, n_iter // 3)
    a = optimize_smooth(a, n, fs, deadline, p2_start, p2_end, n2, lr0 * 0.3, 0.9)
    return a
