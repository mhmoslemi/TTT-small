            top_contrib = np.argsort(contribs)[::-1][:max(1, n // 50)]
            for j in top_contrib:
                trial[j] *= (1.0 - 0.01 * rng.random() * 0.5)
            # Compensate: increase a spread subset
            n_comp = len(top_contrib)
            comp_idx = rng.integers(0, n, size=n_comp)
            for j in comp_idx:
                trial[j] += cur[j] * 0.005 * rng.random()
            trial = np.clip(trial, 0.0, 1000.0)
            if trial.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(trial.tolist()))
            if np.isfinite(sc) and sc < best_sc:

                best_sc = sc
                cur = trial.copy()
