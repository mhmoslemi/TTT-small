import numpy as np
import time

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def score_fft(a, N):
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if s <= 0:
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    return 2.0 * n * mc / (s * s)

def project(a, ref):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.asarray(ref, dtype=np.float64).copy()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
    return a
