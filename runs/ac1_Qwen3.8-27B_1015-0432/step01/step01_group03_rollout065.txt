import numpy as np
import time
from numpy.fft import fft, ifft

def _fp2(sz):
    m = 1
    while m < sz:
        m <<= 1
    return m

def _exact_score(a):
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    L = _fp2(2 * n - 1)
    fa = fft(a, L)
    b = ifft(fa * fa).real[:2 * n - 1]
    return float(2.0 * n * b.max() / (S * S))

def _surrogate_grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    L1 = _fp2(2 * n - 1)
    fa = fft(a, L1)
    b = ifft(fa * fa).real[:2 * n - 1]
    cm = b.max()
    if tau < 1e-10:
        mask = (b == cm)
        p = mask.astype(np.float64) / mask.sum()
        M = cm
    else:
        e = np.exp((b - cm) / tau)
        s = e.sum()
        M = cm + tau * np.log(s)
        p = e / s
    q = p[::-1]
    L2 = _fp2(3 * n - 2)
    cv = ifft(fft(q, L2) * fft(a, L2)).real
    c = cv[2 * n - 2 : n - 2 : -1]
    obj = 2.0 * n * M / (S * S)
    grad = (4.0 * n / (S * S)) * c - 4.0 * n * M / (S * S * S)
    return obj, grad
