import numpy as np

from numpy.fft import fft, ifft


def _fft_len(n):
    m = 1
    while m < 2 * n - 1:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    A = fft(a, _fft_len(n))
    return ifft(A * A).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(max(s, 1e-300)), e / s

    S = a.sum()
    if S < 1e-10:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

    S = a.sum()
    if S < 1e-10:

    cv = ifft(fft(a, L) * fft(p[::-1], L)).real
    g = cv[:n][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / S**3)
