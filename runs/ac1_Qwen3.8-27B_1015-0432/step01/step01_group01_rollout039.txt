import numpy as np
import time

def fft_conv(a, b=None, fft_size=None):
    if b is None:
        b = a
    if fft_size is None:
        fft_size = 1 << (2 * len(a) - 1).bit_length()
    fa = np.fft.rfft(a, n=fft_size)
    fb = np.fft.rfft(b, n=fft_size)
    return np.fft.irfft(fa * fb, n=fft_size)

def subgradient_from_peaks(a, conv, threshold_frac):
    """Compute subgradient of max(conv) using all lags above threshold."""
    n = len(a)
    M = conv.max()
    if M < 1e-30:
        return np.zeros(n), M
    active = np.where(conv >= threshold_frac * M)[0]
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += a[k-hi:k-lo+1][::-1]
    grad /= len(active)
    return grad, M

def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-14:
        a = np.ones(len(a)) / len(a)
        return a
    return a / s

def optimize(a, n, fft_size, deadline, label=""):
    """Subgradient descent with backtracking on peak of a*a."""
    a = normalize(a.copy())
    v = np.zeros(n)
    beta = 0.85
    lr = 0.3
    best_peak = np.inf
    best_a = a.copy()
    prev_M = np.inf
    stall = 0
    threshold = 0.98
    
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        conv = fft_conv(a, None, fft_size)
        M = conv.max()
        if M < best_peak:
            best_peak = M
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        if M >= prev_M:
            lr *= 0.95
        else:
            lr = min(lr * 1.01, 0.5)
        prev_M = M
        if stall > 3000:
            break
        grad, M = subgradient_from_peaks(a, conv, threshold)
        v = beta * v + (1 - beta) * grad
        v -= v.mean()  # preserve sum
        a = a - lr * v
        a = normalize(a)
        # Periodic smoothing
        if it % 5000 == 0 and it > 0:
            kernel_size = 3
            kernel = np.array([0.25, 0.5, 0.25])
            a = np.convolve(a, kernel, mode='same')
            a = normalize(a)
    
    return best_a, 2 * n * best_peak
