import time

def fft_conv(a, b, size=None):
    n, m = len(a), len(b)
    if size is None:
        size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))

def optimize(f, deadline, rng, fft_size):
    n = f.size
    f = f.copy()
    f /= f.sum()  # normalize to sum 1
    
    best_f = f.copy()
    best_score = fast_score(f)
    
    beta = 50.0
    eta = 0.1
    no_improve = 0
    check_interval = 100
    
    for it in range(2000000):
        if time.monotonic() > deadline:
            break
        
        # Compute autocorrelation
        F = np.fft.rfft(f, fft_size)
        power = F * np.conj(F)
        b = np.fft.irfft(power, fft_size)[:2*n-1]
        M = b.max()
        
        # Compute gradient
        w = np.exp(beta * b / M - beta)  # subtract beta for numerical stability
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(b)
            w[n-1] = 1.0
        else:
            w /= w_sum
        
        f_r = f[::-1]
        W = np.fft.rfft(w, fft_size)
        Fr = np.fft.rfft(f_r, fft_size)
        gc = np.fft.irfft(W * Fr, fft_size)
        grad = gc[n-1:2*n-1]
        grad = np.maximum(grad, 0)
        gmax = grad.max()

        # Multiplicative update
        f = f * np.exp(-eta * grad / gmax)
        f = np.maximum(f, 1e-20)

        f /= s

        if it % check_interval == 0:

            if sc < best_score - 1e-9:

                no_improve = 0

    n, m = len(a), len(b)

    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:

    F = np.fft.rfft(f, fft_sz)
    b = np.fft.irfft(F * F.conjugate(), fft_sz)[:2*len(f)-1]
    return float(2 * len(f) * b.max() / (s * s))

    return f / s

def gradient_step(f, beta, eta, fft_sz):
    n = f.size
    # Compute autocorrelation
    F = np.fft.rfft(f, fft_sz)
    P = F * F.conjugate()
    b = np.fft.irfft(P, fft_sz)[:2*n-1]
    M = b.max()

    w = np.exp(beta * (b - M) / M)

    if ws < 1e-30:

        w[n-1] = 1.0
    else:
        w /= ws

    f_r = f[::-1]
    W = np.fft.rfft(w, fft_sz)

    gc = np.fft.irfft(W * Fr, fft_sz)
