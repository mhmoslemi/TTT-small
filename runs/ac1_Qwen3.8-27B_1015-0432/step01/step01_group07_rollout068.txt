def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _multi_subgrad(a, conv, n, M, S, frac=0.92):

    add = 4.0 * n / (S * S)

        je = min(n - 1, ki)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += add * a[ki - jj]
    return g
