    """Find g >= 0 maximizing sum(g) subject to (a*g)[k] <= M for active lags."""
    m = len(active_lags)
    if m == 0:
        return np.ones(n) / n

    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
    rows = []
    cols = []

        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(a[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(m, n))

    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return np.ones(n) / n
