    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.size < 2 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(1024)
    n = parent.size
    s0 = float(parent.sum())
    best = parent.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    # Phase 1: Multi-start gradient descent (~55% of budget)
    n_starts = 6
    p1_end = time.monotonic() + 0.55 * (float(budget_s) - 16.0)
    configs = [(16, 2.0), (32, 3.0), (64, 3.0), (32, 2.5), (48, 4.0), (16, 2.0)]
    for k in range(n_starts):
        if time.monotonic() > p1_end - 10.0:
            break
        t_lim = min(p1_end - 5.0, time.monotonic() + (p1_end - time.monotonic()) / (n_starts - k))
        if t_lim <= time.monotonic():
            break
        a = make_init(k, n, rng, parent)
        tk, al = configs[k]
        a = run_opt(a, n, 1 << (2 * n - 1).bit_length(), t_lim, tk, al, rng)
        cand = np.clip(a * s0, 0.0, 1000.0)
        if cand.sum() >= 0.01:
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = cand
                best_sc = sc

    # Phase 2: Refine best with larger top_k (~25% of budget)
    p2_end = min(t_end - 100.0, time.monotonic() + 0.25 * (float(budget_s) - 16.0))
    if p2_end > time.monotonic() + 10.0:
        a = best / s0
        fs = 1 << (2 * n - 1).bit_length()
        for k, (tk, al) in enumerate([(64, 3.0), (128, 4.0), (64, 5.0)]):
            if time.monotonic() > p2_end - 5.0:
                break
            lim = min(p2_end - 3.0, time.monotonic() + (p2_end - time.monotonic()) / (3 - k))
            a = run_opt(a, n, fs, lim, tk, al, rng)
            cand = np.clip(a * s0, 0.0, 1000.0)
            if cand.sum() >= 0.01:
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cand
                    best_sc = sc
                    a = best / s0

    # Phase 3: Polish with random smooth perturbations (remaining time)
    base = best.copy()
    while time.monotonic() < t_end - 2.0:
        cand = base.copy()
        nb = int(rng.integers(1, 4))
        for _ in range(nb):
            w = max(2, int(n / int(rng.integers(50, 150))))
            st = int(rng.integers(0, max(1, n - w)))
            e = min(n, st + w)
            seg_t = np.arange(e - st) / max(1, e - st - 1)
            amp = rng.normal(0.0, 0.015)
            cand[st:e] *= (1.0 + amp * np.sin(np.pi * seg_t))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = cand
            base = cand
            best_sc = sc

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = parent
    return [float(x) for x in out]
