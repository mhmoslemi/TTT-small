            nblk = int(rng.integers(1, 4))
            for _ in range(nblk):
                w = max(2, int(n * rng.uniform(0.005, 0.05)))
                st = int(rng.integers(0, max(1, n - w)))
                cand[st:st + w] *= float(np.exp(rng.normal(0, 0.04)))
            cand = np.clip(cand, 0.0, 1000.0)
            if cand.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cand
                base = cand

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
    return [float(x) for x in out]
