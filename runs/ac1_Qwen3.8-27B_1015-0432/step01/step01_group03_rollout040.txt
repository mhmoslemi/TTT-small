    return sc

    best = a.copy()
    best_sc = _eval(best)
    for _ in range(n_steps):
        if not np.all(np.isfinite(a)):
            break
        S = a.sum()
        if S < 1e-8:
            break

        sg = np.zeros(a.size)

                best_sc = sc
                a = a_new.copy()
                break

def _coord_descent(a, n_sweeps=10):
    n = a.size
    conv = np.convolve(a, a).copy()
    best = a.copy()
    best_sc = _eval(best)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            old = a[i]
            S_rest = a.sum() - old
            if S_rest < 1e-8:
                continue
            kstar = int(np.argmax(conv))
            
            # Determine L, Q, C for kstar
            mi = kstar - i  # k-i
            if 0 <= mi < n:
                if mi == i:  # k = 2i
                    C = conv[kstar] - old * old
                    Q = 1.0
                    L = 0.0
                else:
                    C = conv[kstar] - 2.0 * old * a[mi]
                    L = 2.0 * a[mi]
                    Q = 0.0
            else:
                C = conv[kstar]
                L = 0.0
                Q = 0.0
            
            # Minimize 2n*(L*x + Q*x^2 + C) / (x + S_rest)^2 over x >= 0
            # (ignoring the 2n factor as it doesn't affect argmin)
            if Q == 0 and L == 0:
                # F is constant in x (or decreasing if C/(x+S)^2 is decreasing)
                # f(x) = C/(x+S)^2, which is decreasing in x
                # So optimal x is... well, we want to minimize, so large x.
                # But we're bounded by 1000.
                # Actually if L=0 and Q=0, conv_kstar doesn't depend on a[i].
                # But S does, so F = 2n*C/(x+S_rest)^2 which is decreasing in x.
                # So we want x as large as possible, but that would make other conv values large.
                # Skip this coordinate.
                continue
            elif Q == 0:
                # f(x) = (L*x + C)/(x+S)^2
                # f'(x) = [L(x+S)^2 - (Lx+C)*2(x+S)]/(x+S)^4
                # = [L(x+S) - 2(Lx+C)]/(x+S)^3
                # = [Lx + LS - 2Lx - 2C]/(x+S)^3
                # = [LS - Lx - 2C]/(x+S)^3
                # Setting to 0: x = (LS - 2C)/L = S - 2C/L
                x_opt = S_rest - 2.0 * C / L if L != 0 else 0.0
                if x_opt < 0:
                    x_opt = 0.0
            else:
                # f(x) = (x^2 + L*x + C)/(x+S)^2
                # f'(x) = 2*[x(x+S) - (x^2+Lx+C)]/(x+S)^3
                # = 2*[x^2+xS - x^2-Lx-C]/(x+S)^3
                # = 2*[x(S-L) - C]/(x+S)^3
                # Setting to 0: x = C/(S-L)
                if S_rest - L > 0:
                    x_opt = C / (S_rest - L)
                else:
                    x_opt = 0.0
                if x_opt < 0:
                    x_opt = 0.0
            
            x_opt = min(x_opt, 1000.0)
            
            if abs(x_opt - old) < 1e-12:
                continue
            
            # Update convolution
            delta = x_opt - old
            # conv_k changes for k in [i, i+n-1]
            lo = i
            hi = min(i + n - 1, 2 * n - 2)
            for k in range(lo, hi + 1):
                mi2 = k - i
                if 0 <= mi2 < n:
                    conv[k] += 2.0 * delta * a[mi2] if mi2 != i else (x_opt*x_opt - old*old)
            
            a[i] = x_opt
            
            # Check improvement
            # (We could check every few coordinates to save time)
            if i % 100 == 0 or i == n - 1:
                sc = 2.0 * n * conv.max() / (a.sum() ** 2)
                if sc < best_sc - 1e-10:
                    best_sc = sc
                    best = a.copy()
                    improved = True
        
        if not improved:
            break
        if time.monotonic() > dl - 30:
            break
    
    return best, best_sc

    m = 1
    while m < size:
        m <<= 1
    return m

def _smooth_max_data(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj_fn(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    conv = np.convolve(a, a)
    mv, _ = _smooth_max_data(conv, tau)
    return 2.0 * n * mv / (S * S)

def _grad_fn(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return np.zeros(n)
    conv = np.convolve(a, a)
    mv, p = _smooth_max_data(conv, tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 1)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]

import time

from numpy.fft import fft, ifft

def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m

def _smooth_max_data(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj_fn(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    conv = np.convolve(a, a)
    mv, _ = _smooth_max_data(conv, tau)
    return 2.0 * n * mv / (S * S)

    n = a.size
    S = a.sum()

    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
