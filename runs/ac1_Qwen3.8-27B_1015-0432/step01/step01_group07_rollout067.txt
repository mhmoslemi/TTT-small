    n = len(z)
    w = np.sort(z)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    k = int(np.sum(w > (cs - S) / idx))
    if k == 0:
        return np.full(n, S / n, dtype=np.float64)
    return np.maximum(z - (cs[k - 1] - S) / k, 0.0)

def _soft_grad(a, cv, m, eps, rsz, n):
    sc = cv[:m] / eps
    mx = sc.max()
    ex = np.exp(sc - mx)
    p = ex / ex.sum()
    fa = np.fft.rfft(a, rsz)
    fp = np.fft.rfft(p, rsz)
    g = np.fft.irfft(fp * np.conj(fa), rsz)[:n]

def _ffc(a, sz):

    return np.fft.irfft(fa * fa, sz)

    w = np.sort(z)[::-1]
    cs = np.cumsum(w)
    idx = np.arange(1, n + 1, dtype=np.float64)
    k = int(np.sum(w > (cs - S) / idx))
    if k == 0:
        return np.full(n, S / n, dtype=np.float64)
    return np.maximum(z - (cs[k - 1] - S) / k, 0.0)


def _soft_g(a, cv, m, eps, rsz, n):
    sc = cv[:m] / eps
    mx = sc.max()
    ex = np.exp(sc - mx)
    p = ex / ex.sum()
    fa = np.fft.rfft(a, rsz)
    fp = np.fft.rfft(p, rsz)
    return np.fft.irfft(fp * np.conj(fa), rsz)[:n]

    cv = _ffc(a, csz)[:m]
    best_mx = float(cv.max())
    best_a = a.copy()
    eps = 0.02
