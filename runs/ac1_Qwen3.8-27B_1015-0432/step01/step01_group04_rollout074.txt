            cur = best * (1.0 + 0.05 * rng.normal(0, 1, n))
            cur = np.maximum(cur, 0.0)
            cur_sc = fft_score(cur, N)
    return best, best_sc

def do_adam(a, n, Ns, Nc, deadline, seed):
    rng = np.random.default_rng(seed)
    y = np.log(np.maximum(a, 1e-15))
    y = y - y.mean()
    y = np.clip(y, -40.0, 40.0)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    b1c, b2c, t = 1.0, 1.0, 0
    lr = 0.05
    best = a.copy()
    best_sc = fft_score(a, Ns)
    L = 2 * n - 1
    no_imp = 0
    while time.monotonic() < deadline:
        ym = y - y.max()
        e = np.exp(np.clip(ym, -500.0, 500.0))
        a2 = e / e.sum()
        A = np.fft.rfft(a2, Ns)
        c = np.fft.irfft(A * A, Ns)[:L]
        c = np.maximum(c, 0.0)
        mc = c.max()
        sc = float(2.0 * n * mc / (a2.sum() ** 2))
        if np.isfinite(sc) and sc < best_sc:
            best = a2.copy()
            best_sc = sc
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 1500:
            y = np.log(np.maximum(best, 1e-15))
            y += rng.normal(0.0, 0.3, n)
            y = y - y.mean()
            y = np.clip(y, -40.0, 40.0)
            m.fill(0.0)
            v.fill(0.0)
            b1c, b2c, t = 1.0, 1.0, 0
            lr = max(0.003, lr * 0.5)
            no_imp = 0
            continue
        alpha = 5.0 + 50.0 * min(1.0, t / 8000.0)
        if no_imp > 400:
            alpha += 40.0
        z = np.exp(np.clip(alpha * (c - mc), -800.0, 0.0))
        zs = z.sum()
        if zs < 1e-300:
            z = np.zeros(L)
            z[int(c.argmax())] = 1.0
        else:
            z /= zs
        Ac = np.fft.rfft(a2, Nc)
        Zc = np.fft.rfft(z, Nc)
        corr = np.fft.irfft(Ac * np.conj(Zc), Nc)
        G = 2.0 * corr[:n]
        G = np.where(np.isfinite(G), G, 0.0)
        gd = a2 * (G - a2.dot(G))
        t += 1
        m = b1 * m + (1.0 - b1) * gd
        v = b2 * v + (1.0 - b2) * gd * gd
        b1c *= b1
        b2c *= b2
        mb = m / (1.0 - b1c)
        vb = v / (1.0 - b2c)
        step = lr * mb / (np.sqrt(vb) + 1e-8)
        np.clip(step, -0.3, 0.3, out=step)
        y -= step
        y -= y.mean()
        y = np.clip(y, -40.0, 40.0)
    return best, best_sc

def do_local(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_sc = fft_score(a, N)
    while time.monotonic() < deadline:
        w = int(rng.integers(max(2, n // 200), max(3, n // 10)))
        st = int(rng.integers(0, n - w + 1))
        cand = best.copy()
        cand[st:st + w] *= np.exp(rng.normal(0.0, 0.15))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = fft_score(cand, N)
        if np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
    return best, best_sc
