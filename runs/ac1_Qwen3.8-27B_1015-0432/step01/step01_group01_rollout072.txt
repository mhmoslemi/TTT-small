def fft_conv(a, sz):
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)


def subgrad_peaks(a, conv, peaks):
    n = len(a)
    grad = np.zeros(n)
    for k in peaks:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi + 1] += a[k - hi:k - lo + 1][::-1]
    return grad

    a = np.clip(a, 0.0, None)
    s = a.sum()

    return a / s
