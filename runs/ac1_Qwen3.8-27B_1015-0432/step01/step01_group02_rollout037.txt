import numpy as np
import time


def _conv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real[:N]


def _run_opt(a, n, Nfft, deadline, rng, alpha0=0.10, thresh=0.6):
    best_sc = np.inf
    best_a = a.copy()
    alpha = alpha0
    stuck = 0
    it = 0
    while time.monotonic() < deadline:
        c = _conv(a, n)
        sc = 2.0 * n * c.max()
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        cmax = c.max()
        if cmax < 1e-15:
            break
        w = np.clip(c / cmax - thresh, 0.0, 1.0) / (1.0 - thresh)
        W = np.fft.rfft(w, Nfft)
        A = np.fft.rfft(a, Nfft)
        h = np.fft.irfft(W * np.conj(A), Nfft).real[:n]
        hmax = h.max()
        if hmax > 1e-15:
            a *= np.exp(-alpha * h / hmax)
        a = np.maximum(a, 1e-18)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a /= s
        it += 1
        if it % 200 == 0:
            wsz = max(2, n // 400)
            st = int(rng.integers(0, n - wsz))
            a[st:st + wsz] = np.maximum(
                a[st:st + wsz] * np.exp(
                    0.2 * rng.standard_normal(wsz)), 1e-18)
            a /= a.sum()
        if stuck > 400:
            alpha *= 0.8
            stuck = 0
            if alpha < 0.003:
                for _ in range(5):
                    wsz = max(2, n // 40)
                    st = int(rng.integers(0, n - wsz))
                    a[st:st + wsz] = np.maximum(
                        a[st:st + wsz] * np.exp(
                            0.5 * rng.standard_normal(wsz)), 1e-18)
                a /= a.sum()
                alpha = alpha0
    return best_a, best_sc
