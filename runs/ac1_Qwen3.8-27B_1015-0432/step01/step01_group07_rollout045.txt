import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    sv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (sv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (sv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _subgrad_topk(a, conv, K, temp):
    n = len(a)
    topk = np.argsort(conv)[::-1][:K]
    M = conv[0]
    diffs = M - conv[topk]
    w = np.exp(-diffs / (temp * M + 1e-30))
    w = w / w.sum()
    g = np.zeros(n)
    for ki in range(K):
        k = int(topk[ki])
        if w[ki] < 1e-12:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            seg = a[k - je: k - js + 1]
            g[js: je + 1] += w[ki] * 2.0 * seg[::-1]
    return g


def _score_simplex(a, n):
    conv = _fft_conv(a)
    return float(np.max(conv)), conv

            best_M = M_t

    return best_a, best_M
