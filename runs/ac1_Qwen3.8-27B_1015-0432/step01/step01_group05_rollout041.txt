def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _score(a):

    if s < 0.01:
        return np.inf
    return 2.0 * len(a) * _fft_conv(a, a).max() / s**2

    cx = c - c.max()

def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def _score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * len(a) * _fft_conv(a, a).max() / (s * s)

def _grad(a, beta):
    n = len(a)
    c = _fft_conv(a, a)
    cx = c - c.max()
    w = np.exp(beta * cx)

    else:
        w /= ws
    h = np.correlate(w, a, mode='valid')
