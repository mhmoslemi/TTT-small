import numpy as np
import time

def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2*n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2*n - 1]

def _score(a, conv):
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * len(a) * conv.max() / (S * S))

def _rescale(a, tgt):
    s = a.sum()
    if s < 1e-15:
        return np.full(a.shape, tgt / a.size)
    return a * (tgt / s)

def _sg(a, conv, S, n, K, sharp):
    M = float(conv.max())
    if M < 1e-15:
        return np.zeros(n)
    kk = min(K, len(conv))
    tp = np.argpartition(conv, len(conv) - kk)[-kk:]
    tp = tp[np.argsort(conv[tp])[::-1]]
    tv = conv[tp].astype(float)
    w = np.exp((tv[0] - tv) * sharp / M)
    w /= w.sum()
    g = np.zeros(n)
    for i in range(kk):
        ki = int(tp[i])
        wi = w[i]
        Mi = tv[i]
        gm = 4.0 * n * wi / (S * S)
        g -= gm * Mi / S
        js = max(0, ki - n + 1)
        je = min(n - 1, ki)
        if js <= je:
            m = np.arange(js, je + 1)
            g[js:je+1] += gm * a[ki - m]
    return g
