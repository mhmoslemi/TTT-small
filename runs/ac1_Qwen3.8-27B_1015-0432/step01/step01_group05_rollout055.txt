def build_conv_matrix(a):
    n = len(a)
    # A is (2n-1) x n, A[i,k] = a[i-k] if 0<=i-k<n
    rows = []
    cols = []

        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            rows.extend(np.full(len(ks), i).tolist())
            cols.extend(ks.tolist())
            data.extend(a[i - ks].tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

```python
def build_conv_matrix(a):
    n = len(a)
    # For each i, valid k range is [max(0,i-n+1), min(n-1,i)]
    i_vals = np.arange(2*n-1)
    k_lo = np.maximum(0, i_vals - n + 1)
    k_hi = np.minimum(n-1, i_vals)
    ns = k_hi - k_lo + 1  # number of nonzeros per row
    total = ns.sum()
    
    # Build row indices (repeated)
    rows = np.repeat(i_vals, ns)
    # Build col indices: for row i, cols are k_lo[i], k_lo[i]+1, ..., k_hi[i]
    # Total positions: 0, 1, ..., total-1
    # For row i, its positions start at cumsum(ns)[:i]
    starts = np.concatenate([[0], np.cumsum(ns)[:-1]])
    col_offsets = np.arange(total) - np.repeat(starts, ns)
    cols = np.repeat(k_lo, ns) + col_offsets
    
    # Data: for position (i, k), value is a[i-k]
    idx = np.repeat(i_vals, ns) - cols
    data = a[idx]
    
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    return A
```

    rng = np.random.default_rng(seed)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = a.size
    if n < 2:
        n = 256
        a = np.ones(n)
    S = a.sum()
    if S < 0.01:
        a = np.ones(n)
        S = float(n)

    best = a.copy()
    best_s = float(evaluate_sequence(best.tolist()))

    # --- Phase 1: LP-based direction search ---
    max_lp_iters = 15
    for it in range(max_lp_iters):
        if time.monotonic() > deadline - 300:
            break
        M = _fft_conv(a, a).max()
        A_mat = _build_conv_mat(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        g_upper = 10.0 * M / n
        bounds = [(0.0, g_upper)] * n
        res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 30.0})
        if res.status != 0 or res.x is None:
            break
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-10:
            break
        g = g * S / gs
        for t_b in (0.1, 0.05, 0.02, 0.01):
            a_try = (1 - t_b) * a + t_b * g
            a_try = np.maximum(a_try, 0.0)
            s_try = float(evaluate_sequence(a_try.tolist()))
            if np.isfinite(s_try) and s_try < best_s:
                best = a_try.copy()
                best_s = s_try
                a = a_try
                break
        # Check if progress is stalled
        if time.monotonic() > deadline - 200:
            break

    # --- Phase 2: Gradient refinement ---
    a = best.copy()
    S0 = a.sum()
    soft_beta = 50.0
    lr = 3.0 / n
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t_adam = 0
    stall = 0
    best_phase2 = best_s

    while time.monotonic() < deadline - 15:
        t_adam += 1
        c_arr = _fft_conv(a, a)
        cmax = c_arr.max()
        w = np.exp(soft_beta * (c_arr - cmax))
        wsum = w.sum()
        if wsum < 1e-300:
            break
        w /= wsum
        grad = _fft_grad(a, w, n)
        # Center the gradient (scale invariance)
        grad -= (grad * a).sum() / max((a * a).sum(), 1e-30) * a

        m_buf = 0.9 * m_buf + 0.1 * grad
        v_buf = 0.999 * v_buf + 0.001 * grad * grad
        m_hat = m_buf / (1.0 - 0.9 ** t_adam)
        v_hat = v_buf / (1.0 - 0.999 ** t_adam)
        a = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.maximum(a, 0.0)
        sa = a.sum()
        if sa < 1e-12:
            a = best.copy()
            break
        a = a / sa * S0

        if t_adam % 30 == 0:
            s_try = float(evaluate_sequence(a.tolist()))
            if np.isfinite(s_try) and s_try < best_s:
                best = a.copy()
                best_s = s_try
                stall = 0
                best_phase2 = s_try
            else:
                stall += 1
                if stall >= 20:
                    soft_beta = min(soft_beta * 1.5, 300.0)
                    lr = max(lr * 0.7, 0.5 / n)

                        a = best.copy()
