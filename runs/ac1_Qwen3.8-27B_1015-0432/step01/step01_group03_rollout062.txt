import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _pow2(m):
    r = 1
    while r < m:
        r <<= 1
    return r


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10
    L = _pow2(2 * n - 1)
    fa = fft(a, L)
    c = ifft(fa * fa).real[:2 * n - 1]
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    mv = cm + tau * np.log(s)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)
    L = _pow2(3 * n - 2)
    fa = fft(a, L)
    c = ifft(fa * fa).real[:2 * n - 1]
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    p = e / s
    mv = cm + tau * np.log(s)
    fp = fft(p[::-1], L)
    h = ifft(fa * fp).real[:2 * n - 1]
    g = 2.0 * h[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3)) * np.ones(n)


def _dual_blend(a, tau, n):
