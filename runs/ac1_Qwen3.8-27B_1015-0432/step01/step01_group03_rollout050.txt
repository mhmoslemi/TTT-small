import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _conv_fft(a):
    n = a.size
    L = _fft_pow2(2 * n)
    return ifft(fft(a, L) * fft(a, L)).real[:2 * n - 1]


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv_fft(a)
    mv, _ = _smooth_max(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv_fft(a)
    mv, p = _smooth_max(c, tau)
    pr = p[::-1]
    L2 = _fft_pow2(3 * n)
    cv = ifft(fft(a, L2) * fft(pr, L2)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

    return float(2 * n * c.max() / (S * S))
