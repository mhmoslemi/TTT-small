import numpy as np

def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa, sz)
    return c[:2 * n - 1]


def _score(c, n, s):
    return 2.0 * n * c.max() / (s * s)


def _subgrad(c, a, n, K):
    K = min(K, 2 * n - 1)
    top = np.argpartition(c, -K)[-K:]
    vals = c[top]
    mx = vals.max()
    if mx < 1e-30:
        return np.zeros(n)
    w = (vals / mx) ** 3
    w /= w.sum()
    g = np.zeros(n)
    for idx in range(K):
        k = int(top[idx])
        wi = w[idx]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = np.arange(lo, hi + 1)
            g[seg] += wi * a[k - seg]
    g -= g.mean()
    return g
