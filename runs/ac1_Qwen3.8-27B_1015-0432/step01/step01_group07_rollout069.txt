import numpy as np
import time


def _fft_conv(a, b=None):
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def _soft_grad(a, conv, beta):
    n = len(a)
    M = float(conv.max())
    ks = int(np.argmax(conv))
    sh = beta * (conv - M)
    with np.errstate(over='ignore'):
        w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300 or not np.isfinite(ws):
        w = np.zeros(2 * n - 1)
        w[ks] = 1.0
    else:
        w = w / ws
    g = _fft_conv(a[::-1], w)[n - 1:2 * n - 1].copy()
    g -= g.mean()
    return g, M


def _simplex_proj(x):
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-12:
        return np.ones(len(x)) / len(x)
    return x / s
