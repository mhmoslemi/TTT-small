    e = np.exp((c - cm) / tau)
    s = np.sum(e)

import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _npo2(sz):
    m = 1
    while m < sz:
        m <<= 1
    return m

def _fc(a, b):
    n = a.size + b.size - 1
    L = _npo2(n)
    return ifft(fft(a, L) * fft(b, L)).real[:n]

def _smax(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = np.sum(e)
    return cm + tau * np.log(s), e / s

def _of(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros_like(a)
    c = _fc(a, a)
    mv, p = _smax(c, tau)
    val = 2.0 * n * mv / (S * S)
    af = a[::-1]
    L = _npo2(3 * n - 2)
    fp = fft(p, L)
    g2 = ifft(fp * fft(a, L)).real[:n]
    g1 = ifft(fp * fft(af, L)).real[n-1:2*n-1]
    gsm = g1 + g2
    grad = 2.0 * n * (S * gsm - 2.0 * mv * a) / (S**3)
    return val, grad

def _sc(a):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return np.inf
    L = _npo2(2 * n - 1)
    c = ifft(fft(a, L) * fft(a, L)).real[:2*n-1]
    return 2.0 * n * np.max(c) / (S * S)

def _ls(bv, sc, rng, dl):
    n = bv.size
    while time.monotonic() < dl:
        cd = bv.copy()
        w = max(2, min(n, int(np.exp(rng.normal(np.log(max(2, n//25)), 0.7)))))
        s = int(rng.integers(0, n - w + 1))
        sig = 0.01 + 0.15 * rng.random()
        cd[s:s+w] *= np.exp(rng.normal(0.0, sig, w))
        if rng.random() < 0.25:
            cd = np.maximum(0.0, cd + rng.normal(0.0, 0.003 * max(cd.max(), 1e-10), n))
        ns = cd.sum()
        if ns < 1e-8:
            continue
        cd /= ns
        s2 = _sc(cd)
        if s2 < sc:
            bv = cd
            sc = s2
    return bv, sc

    if ps < 1e-8:

        ps = float(n)
    a0 = par / ps
    
    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    cur = a0.copy()
    
    taus = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

        if time.monotonic() > dl - 200.0:

        try:
            res = minimize(_of, cur, args=(tv,), jac=True,
                          method='L-BFGS-B',
