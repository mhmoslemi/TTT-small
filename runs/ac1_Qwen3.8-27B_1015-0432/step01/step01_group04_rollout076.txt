import numpy as np
import time
import math

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64).ravel()
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (cssv - 1.0) / rho
    idx = int(np.argmax(cond)) if np.any(cond) else 0
    theta = (cssv[idx] - 1.0) / (idx + 1)
    a = np.maximum(v - theta, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(n, dtype=np.float64) / n
    return a / s

def fft_conv(a, N):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * a.size - 1
    return np.maximum(c[:L], 0.0)

def fft_xcorr(p, a, N2):
    P = np.fft.rfft(p, N2)
    A = np.fft.rfft(a, N2)
    g = np.fft.irfft(P * np.conj(A), N2)
    return g[:a.size]

def softmax_y(y):

    s = e.sum()
    if s < 1e-30 or not np.isfinite(s):
        return np.ones(y.size, dtype=np.float64) / y.size
