Wait, I want to double-check the gradient computation. For peak at position k in `conv(a,a)`:
`conv(a,a)[k] = sum_i a[i] * a[k-i]` where i ranges over valid indices.

The derivative with respect to `a[j]` is `a[k-j]` (if `k-j` is valid, i.e., `0 <= k-j < n`).

So the gradient for peak k is: `grad[j] += w_k * a[k-j]` for `j` in `[max(0, k-n+1), min(n-1, k)]`.

In my code:

Let's verify: for `j_i` in `[lo, hi]`, we add `w * a[k - j_i]`. When `j_i = lo = max(0, k-n+1)`:
- If `k >= n-1`: `lo = k-n+1`, so `k - lo = k - (k-n+1) = n-1`. Valid.
- If `k < n-1`: `lo = 0`, so `k - lo = k`. Valid (k < n).

When `j_i = hi = min(n-1, k)`:
- If `k >= n-1`: `hi = n-1`, so `k - hi = k - n + 1`. Valid (>= 0).
- If `k < n-1`: `hi = k`, so `k - hi = 0`. Valid.

So the gradient is correctly computed as `grad[j] = sum_{k in peaks} w_k * a[k-j]` for valid k-j. This is the derivative of `conv(a,a)[k]` w.r.t. `a[j]`, weighted by `w_k`. ✓

import numpy as np
import time

def _fconv(a):
    n = len(a)
    N = 2 * n - 1
    return np.fft.irfft(np.fft.rfft(a, N) ** 2)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    dl = time.monotonic() + max(0.0, float(budget_s) - 15.0)
    n = len(height_sequence_1)

    a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1e6)
    if a0.sum() < 1e-9:
        a0 = np.ones(n)
    a0 /= a0.sum()

    x = np.linspace(0, 1, n)
    cands = [a0]
    cands.append(np.maximum(np.sin(np.pi * x), 1e-8))
    cands.append(np.maximum(np.sin(np.pi * x) ** 2, 1e-8))
    cands.append(np.exp(-5 * (x - 0.5) ** 2))
    cands.append(0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x))

    ba, bs = None, 1e18
    for c in cands:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 1e-9:
            continue
        c /= s
        v = 2.0 * n * _fconv(c).max()
        if v < bs:
            bs, ba = v, c.copy()

    a = ba.copy()
    g = np.zeros(n)
    beta = 0.7
    alpha = 0.1
    K = 30
    stuck, it = 0, 0

    while time.monotonic() < dl:
        cv = _fconv(a)
        v = 2.0 * n * cv.max()
        if v < bs - 1e-13:
            bs, ba, stuck = v, a.copy(), 0
        else:
            stuck += 1

        idx = np.argpartition(cv, -K)[-K:]
        pv = cv[idx]
        pm = pv.max()
        w = pv / pm if pm > 0 else np.ones(K)

        h = np.zeros(n)
        for j in range(K):
            k = int(idx[j])
            lo, hi = max(0, k - n + 1), min(n - 1, k)
            h[lo:hi + 1] += w[j] * a[k - np.arange(lo, hi + 1)]

        g = beta * g + (1.0 - beta) * h
        gm = g.max()
        if gm > 1e-15:
            a *= np.exp(-alpha * g / gm)
        a = np.maximum(a, 1e-14)
        a /= a.sum()
        it += 1

        if it % 50 == 0:
            pw = max(2, n // 200)
            ps = int(rng.integers(0, n - pw))
            a[ps:ps + pw] *= np.clip(rng.normal(1.0, 0.3, pw), 0.1, 5.0)
            a = np.maximum(a, 1e-14)
            a /= a.sum()

        if stuck > 600:
            alpha *= 0.88
            if alpha < 0.01:
                for _ in range(max(3, n // 400)):
                    bw = int(rng.integers(n // 50, n // 4 + 1))
                    b0 = int(rng.integers(0, n - bw))
                    a[b0:b0 + bw] *= np.clip(rng.normal(1.0, 1.0, bw), 0.05, 10.0)
                a = np.maximum(a, 1e-14)
                a /= a.sum()
                g[:] = 0.0
                alpha, stuck = 0.1, 0
        elif stuck > 150:
            alpha = max(0.015, alpha * 0.97)

    res = [float(val) for val in ba]
    if not np.isfinite(evaluate_sequence(res)):
        res = [1.0 / n] * n
    return res
