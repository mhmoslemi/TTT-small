from numpy.fft import fft, ifft

_L = 1
_BD = None

def _npow2(x):
    m = 1
    while m < x:
        m <<= 1
    return m

def _init(n):
    global _L, _BD
    _L = _npow2(3 * n - 2)
    _BD = [(0.0, 1000.0)] * n

def _conv(a):
    return (ifft(fft(a, _L) * fft(a, _L)).real)[:2 * a.size - 1]

def _fscore(a):
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    S = a.sum()
    if S < 0.01:
        return 1e10
    return 2.0 * n * _conv(a).max() / (S * S)

def _smmax(a, tau):
    n = a.size
    c = _conv(a)
    cm = c.max()
    e = np.exp((c - cm) / tau)
    s = e.sum()
    p = e / s
    mv = cm + tau * np.log(s)
    pr = p[::-1]
    g = 2.0 * (ifft(fft(a, _L) * fft(pr, _L)).real)[n - 1: 2 * n - 1]
    return mv, p, g

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _, _ = _smmax(a, tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    mv, _, g = _smmax(a, tau)
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S * S * S)) * a

def _subg_topk(a, c, K):
    n = a.size
    top = np.argsort(c)[-K:]
    g = np.zeros(n)
    for ki in top:
        ki = int(ki)
        lo = max(0, ki - n + 1)
        hi = min(n - 1, ki)
        m = np.arange(lo, hi + 1)
        g[m] += (2.0 / K) * a[ki - m]
    return g

def _do_accept(cand, best, best_sc):
    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < best_sc:
        return cand.copy(), sc
    return best, best_sc

def _lbfgs(cur, tv, n, maxit, gtol):
    res = minimize(_obj, cur, args=(tv,), jac=_grad,
                   method='L-BFGS-B', bounds=_BD,
                   options={'maxiter': maxit, 'gtol': gtol,
                            'maxls': 20, 'ftol': 1e-14})
    nx = np.clip(res.x, 0.0, 1000.0)
    ns = nx.sum()
    if ns > 1e-8:
        nx = nx / ns
    return nx

def _refine_active(a, n, dl):
    ba = np.clip(a, 0.0, 1000.0).copy()
    bs = float(evaluate_sequence(ba.tolist()))
    K = 3
    lr = 1.0
    stall = 0
    for _ in range(20000):
        if time.monotonic() > dl:
            break
        c = _conv(ba)
        g = _subg_topk(ba, c, K)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        for alpha in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            cand = np.clip(ba - alpha * g, 0.0, 1000.0)
            cs = cand.sum()
            if cs < 0.01:
                continue
            cand = cand / cs
            fc = _fscore(cand)
            if fc < bs - 1e-12:
                ba = cand
                bs = fc
                improved = True
                break
        if improved:
            stall = 0
            lr = min(lr * 1.03, 8.0)
        else:
            stall += 1
            lr *= 0.6
            if stall > 80:
                break
    return ba, bs

def _stoch(a, rng, n, dl):
    ba = a.copy()
    bs = float(evaluate_sequence(ba.tolist()))
    wbase = max(1, n // 60)
    cnt = 0
    while time.monotonic() < dl:
        cd = ba.copy()
        w = max(1, min(n, int(wbase * rng.uniform(0.3, 3.0))))
        s = int(rng.integers(0, max(1, n - w)))
        cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.05)))
        r = rng.random()
        if r < 0.2:
            cd *= float(np.exp(rng.normal(0.0, 0.015)))
        elif r < 0.32:
            w2 = int(rng.integers(1, min(5, n)))
            s2 = int(rng.integers(0, max(1, n - w2)))
            cd[s2:s2 + w2] = ba[s2:s2 + w2] * rng.uniform(0.05, 3.0, w2)
        elif r < 0.40:
            w3 = max(1, n // 30)
            s3 = int(rng.integers(0, max(1, n - w3)))
            cd[s3:s3 + w3] = ba[s3:s3 + w3] * 0.0
        cd = np.clip(cd, 0.0, 1000.0)
        cnt += 1
        if cnt % 8 == 0:
            fc = _fscore(cd)
            if fc < bs - 1e-12:
                ba = cd.copy()
                bs = fc
                wbase = max(1, int(wbase * 0.92))
            else:
                wbase = min(n // 12, int(wbase * 1.04) + 1)
    return ba, bs
