        _c = _conv(ic)

        if sc < best_score:
            best_score = sc
            best_a = ic.copy()
            a = ic.copy()

    alpha = 0.15
    stuck = 0
    iters = 0
    temp = 0.03

    while time.monotonic() < deadline:
        c = _conv(a)
        score = 2.0 * n * c.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        c_max = c.max()
        if c_max <= 1e-15:
            break

        shifted = (c - c_max) / (c_max * temp + 1e-30)
        weights = np.exp(shifted)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.zeros(len(c))
            weights[int(np.argmax(c))] = 1.0
        else:
            weights /= w_sum

        h = _grad_peaks(weights, a)
        h_max = h.max()
        if h_max > 1e-15:
            a *= np.exp(-alpha * h / h_max)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = best_a.copy()
        else:
            a /= s

        iters += 1

        if iters % 50 == 0:
            ws = int(rng.integers(max(2, n // 600), max(3, n // 8)))
            st = int(rng.integers(0, n - ws))
            a[st:st + ws] *= float(np.exp(rng.normal(0.0, 0.25)))
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a /= s

        if iters % 200 == 0:
            temp = 0.02 + 0.03 * (1.0 - min(stuck / 500.0, 1.0))

        if stuck > 400:
            alpha = max(0.03, alpha * 0.9)
            if alpha <= 0.035:
                ws = int(rng.integers(n // 12, max(n // 12 + 1, n // 4)))
                st = int(rng.integers(0, max(1, n - ws)))
                a[st:st + ws] *= float(np.exp(rng.normal(0.0, 0.9)))
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-12:
                    a /= s
                alpha = 0.15
                stuck = 0
        elif stuck == 0:
            alpha = min(0.28, alpha * 1.015)

    result = [float(x) for x in best_a]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs) or fs > 1e4:
        result = [1.0 / n] * n
    return result
