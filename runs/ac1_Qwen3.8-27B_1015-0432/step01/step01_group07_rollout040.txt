import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _fast_score(a):
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf
    n = len(a)
    return 2.0 * n * float(np.max(_fft_conv(a))) / (S * S)


def _smooth_grad(a, T):
    n = len(a)
    S = float(np.sum(a))
    conv = _fft_conv(a)
    mc = float(np.max(conv))
    e = np.exp((conv - mc) / T)
    es = float(np.sum(e))
    p = e / es
    LSE = mc + T * np.log(es)
    a_rev = a[::-1]
    size = 1
    while size < 3 * n:
        size <<= 1
    fp = np.fft.rfft(p, size)
    fa = np.fft.rfft(a_rev, size)
    h = np.fft.irfft(fp * fa, size)[n - 1:n - 1 + n]
    grad = 2.0 * n * (S * h - 2.0 * LSE * a) / (S * S * S)
    return grad, LSE, conv


def _optimize(a0, deadline, rng):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / max(float(np.sum(a)), 0.01)
    T = 1.0 / (5.0 * n)
    T_min = 1.0 / (500.0 * n)
    lr = 0.5
    stall = 0
    best = a.copy()
    best_sc = _fast_score(best)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        if it % 2000 == 0 and time.monotonic() > deadline - 5:
            break
        grad, LSE, conv = _smooth_grad(a, T)
        gs = float(np.max(np.abs(grad)))
        if gs < 1e-18:
            break
        step = lr * float(np.max(a)) / gs
        improved = False
        for f in (1.0, 0.5, 0.25, 0.125, 2.0, 4.0):
            a_t = np.maximum(a - f * step * grad, 0.0)
            St = float(np.sum(a_t))
            if St < 0.01:
                continue
            a_t = a_t / St
            sc = _fast_score(a_t)
            if sc < best_sc:
                best_sc = sc
                best = a_t.copy()
                a = a_t
                improved = True
                break
        if improved:
            stall = 0
            T = max(T * 0.995, T_min)
            lr = min(lr * 1.01, 2.0)
        else:
            stall += 1
            if stall == 10:
                lr *= 0.6
            elif stall == 25:
                T = max(T * 0.5, T_min)
            elif stall == 50:
                a = np.maximum(best + rng.normal(0, 0.01 / n, n), 0.0)
                a = a / max(float(np.sum(a)), 0.01)
                lr = 0.5
                T = 1.0 / (5.0 * n)
                stall = 0
            elif stall > 80:
                break
    return best, best_sc
