- `k*` ranges from 0 to 2n-2
- `a[k*-i]` is valid when `0 <= k*-i < n`, i.e., `k*-n+1 <= i <= k*`
- For `i` outside this range, `a[k*-i] = 0`

So `d[i] = a[k*-i]` if `max(0,k*-n+1) <= i <= min(n-1,k*)` else 0.

This can be computed as:
- Reverse a: `ar[i] = a[n-1-i]`
- `d[i] = ar[n-1-k*+i]` if `0 <= n-1-k*+i < n` else 0

- If `shift >= 0`: `d[shift:n] = ar[0:n-shift]`, rest is 0
- If `shift < 0`: `d[0:n+shift] = ar[-shift:n]`, rest is 0

This is O(n) with no extra allocation needed (just a shift operation).

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def softmax_fn(y):

def score_fft(a, N):

        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0:
        return np.inf
    L = 2 * n - 1
    N = max(N, next_pow2(L))
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    return float(2.0 * n * mc / (s * s)) if np.isfinite(mc) else np.inf

def shift_vec(a, kstar, n):
    """Compute d[i] = a[kstar-i] with zero-padding."""
    ar = a[::-1]
    sh = n - 1 - kstar
    d = np.zeros(n, dtype=np.float64)
    if sh >= 0:
        cnt = n - sh
        if cnt > 0:
            d[sh:] = ar[:cnt]
    else:
        cnt = n + sh
        if cnt > 0:
            d[:cnt] = ar[-sh:]
    return d

def spectral_iter(a, N, alpha):
    n = a.size
    L = 2 * n - 1
    N = max(N, next_pow2(L))
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 1e-300, 0.0)
    cs = float(c.sum())
    if cs <= 0 or not np.isfinite(cs):
        return a.copy()
    c = c / cs
    ct = np.power(c, alpha)
    sct = float(ct.sum())
    if not np.isfinite(sct) or sct <= 0:
        return a.copy()
    ct = ct / sct
    T = np.fft.rfft(ct, N)
    # take sqrt in frequency: |A_new| = sqrt(FFT(ct))
    mag = np.sqrt(np.maximum(T.real, 0.0))
    phase = np.angle(T)
    Anew = mag * np.exp(1j * phase)
    a2 = np.fft.irfft(Anew, N)[:n]
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    return np.maximum(a2, 0.0)
