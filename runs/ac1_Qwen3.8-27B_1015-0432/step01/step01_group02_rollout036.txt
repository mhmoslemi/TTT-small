def _fftconv(a, n):
    N = 2 * n - 1
    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real

def _score(a, n):
    c = _fftconv(a, n)
    return 2.0 * n * c.max(), c

def _fftconv(a, n):

    F = np.fft.rfft(a, N)
    return np.fft.irfft(F * F, N).real
