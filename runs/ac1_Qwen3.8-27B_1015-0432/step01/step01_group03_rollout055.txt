    ps = float(par.sum())
    if ps < 1e-8:
        par = np.ones(n, dtype=np.float64)
        ps = float(n)
    a0 = par / ps
    L = _fft_pow2(3 * n - 2)
    cur = a0.copy()
    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    taus = [3.0, 1.5, 1.0, 0.6, 0.35, 0.2, 0.12, 0.07, 0.04, 0.025, 0.015, 0.01, 0.006, 0.004, 0.002]
    bnds = [(0.0, 1000.0)] * n

    for tv in taus:
        if time.monotonic() > dl - 120.0:
            break
        maxit = max(30, min(400, int((dl - 120.0 - time.monotonic()) / (30.0 / max(1, len(taus) - 1)))))
        try:
            res = minimize(
                _eval_inner, cur, args=(tv, n, L), method='L-BFGS-B',
                jac=True, bounds=bnds,
                options={'maxiter': maxit, 'gtol': 1e-10, 'ftol': 1e-15, 'maxls': 40}
            )
            if res.x is not None:
                nx = np.clip(res.x, 0.0, 1000.0)
                ns = float(nx.sum())
                if ns > 1e-8:
                    nx /= ns
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                if res.x is not None:
                    cur = nx.copy()
        except Exception:
            pass

    if time.monotonic() < dl - 30.0:
        try:
            res2 = minimize(
                _eval_inner, cur, args=(0.001, n, L), method='L-BFGS-B',
                jac=True, bounds=bnds,
                options={'maxiter': 300, 'gtol': 1e-11, 'ftol': 1e-15, 'maxls': 40}
            )
            if res2.x is not None:
                nx = np.clip(res2.x, 0.0, 1000.0)
                ns = float(nx.sum())
                if ns > 1e-8:
                    nx /= ns
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
        except Exception:
            pass

    best, best_sc = _local_search(best, best_sc, rng, n, L, dl - 5.0)

    if time.monotonic() < dl:
        s_final = float(evaluate_sequence(best.tolist()))
        if np.isfinite(s_final):
            best_sc = s_final
        else:
            best = a0.copy()
            best_sc = float(evaluate_sequence(best.tolist()))

    return [float(v) for v in best]
