def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 8.0
    n = len(height_sequence_1)
    inits = make_inits(n, rng)
    best_f = None
    best_score = np.inf
    for init in inits:
        sc = fast_score(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = init.copy()
    if best_f is None:
        best_f = sym(np.ones(n)) / n
        best_score = 2.0
    f = best_f.copy()
    while time.monotonic() < deadline - 60.0:
        f_opt, sc_opt = mwdescent(f, deadline - 60.0, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
        f = best_f.copy()
        for _ in range(6):
            if time.monotonic() > deadline - 60.0:
                break
            trial = best_f * np.exp(0.03 * rng.standard_normal(n))
            trial = sym(np.maximum(trial, 1e-20))
            trial /= trial.sum()
            sc = fast_score(trial)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = trial.copy()
    if time.monotonic() < deadline - 30.0:
        f_opt, sc_opt = coord_polish(best_f, deadline - 10.0)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
        f2 = sym(np.maximum(best_f, 1e-15)).copy()
        f2 /= f2.sum()
        sc2 = mwdescent(f2, deadline - 10.0, rng)
        if sc2[1] < best_score:
            best_score = sc2[1]
            best_f = sc2[0].copy()
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [float(x) for x in (sym(np.ones(n)) / n)]
    return result
