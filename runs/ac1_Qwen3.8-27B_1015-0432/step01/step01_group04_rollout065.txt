import numpy as np
import time


def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, 2 * N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def clip_norm(a, ref):

        s = float(a.sum())
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    return a / s
