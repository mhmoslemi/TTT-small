```python
    g = np.zeros(n)
    for idx, kk in enumerate(top_lags):
        kk = int(kk)
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += w[idx] * 2.0 * a[kk - jj]
    
    return g
```

But the loop over K lags is O(K*n) which is fine for small K.

Now, the "useful" direction is g projected onto the orthogonal complement of a (and 1):
g_useful = g - <g,a>/<a,a> * a - <g,1>/<1,1> * 1

But since we normalize after each step, the 1-component is automatically removed. We only need to worry about the a-component if we want to be efficient.

Actually, the a-component (proportional to a) just scales a uniformly, which is removed by normalization. So we don't need to explicitly remove it. The normalization handles it.

OK let me just write the final clean implementation. I'll use:
- Multi-peak subgradient (top 5 lags, weighted by value^2)
- No symmetrization (to avoid the fixed-point issue)
- Momentum (EMA of gradients)
- Adaptive step size with 4 trial values
- Periodic perturbation on stall
- FFT-based convolution

```python
import numpy as np
import time


def _fft_conv(a, size):
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * len(a) - 1]


def _subgrad_multi(a, conv, K, p):
    n = len(a)
    order = np.argsort(conv)[::-1]
    top = order[:K]
    tv = conv[top]
    w = tv ** p
    ws = float(w.sum())
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    g = np.zeros(n)
    for idx in range(K):
        kk = int(top[idx])
        js = max(0, kk - n + 1)
        je = min(n - 1, kk)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += w[idx] * 2.0 * a[kk - jj]
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.1, float(budget_s) - 12.0)

    n = a.size

    else:

    M = float(np.max(conv))

    mom = np.zeros(n)
    mu = 0.4
    stall = 0

        k_star = int(np.argmax(conv))
        M = float(conv[k_star])
        cur_score = 2.0 * n * M

        g = _subgrad_multi(a, conv, K, p)

        # Remove component proportional to a (irrelevant after normalization)
        aa_dot = float(a @ g)
        a_norm2 = float(a @ a)
        if a_norm2 > 1e-30:
            g = g - (aa_dot / a_norm2) * a

        # Momentum
        mom = mu * mom + (1.0 - mu) * g

        # Line search
        best_trial = cur_score
        best_trial_a = a.copy()

        for tlr in (lr, lr * 0.5, lr * 0.1, lr * 2.5):

            a_t = np.clip(a_t, 0.0, None)

            if st < 1e-10:
```
