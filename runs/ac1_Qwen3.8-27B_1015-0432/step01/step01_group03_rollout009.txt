def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m

def _convolve_fft(a, b):
    L = _fft_pow2(len(a) + len(b) - 1)
    return ifft(fft(a, L) * fft(b, L)).real[:len(a)+len(b)-1]

def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smooth_max(_convolve_fft(a, a), tau)
    return 2.0 * n * mv / (S * S)

    S = a.sum()

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from numpy.fft import fft, ifft

def _fft_pow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a, b=None):
    if b is None:
        b = a
    L = _fft_pow2(len(a) + len(b) - 1)
    return ifft(fft(a, L) * fft(b, L)).real[:len(a)+len(b)-1]

def _smax(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
