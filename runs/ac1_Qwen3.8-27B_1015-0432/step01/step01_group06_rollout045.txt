import time

def _fft_conv(a):
    n = len(a)
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _normalize(a, n):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n)
    return a * n / s

def _gen_init(n, rng, mode):

    if mode == 0:
        a = np.ones(n)

def _compute_grad(a, conv, n, K):
    m = len(conv)
    if K < m:
        idx = np.argpartition(conv, -K)[-K:]
    else:
        idx = np.arange(m)
    thresh = conv[idx].min()
    active = np.where(conv >= thresh - 1e-14)[0]
    if len(active) > 256:
        active = active[np.argsort(conv[active])[-256:]]
    g = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
    cnt = float(len(active))
    if cnt > 0:
        g /= cnt
    g -= g.mean()
    return g
