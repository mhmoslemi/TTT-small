    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        c = cov_full(a, N)
        cur_sc = 2.0 * n * float(c.max())
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        g = grad_topk(a, c, k=5)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        improved = False
        for f in (1.0, 0.5, 0.25, 0.1, 0.04):
            step = lr * f / gn
            a_try = proj(a - step * g)
            s_try = float(cov_full(a_try, N).max())
            if s_try < float(c.max()) * (1 - 1e-8):
                a = a_try
                improved = True
                break
        if not improved:
            lr *= 0.7
            if lr < 0.02:
                a = proj(a + 0.005 * rng.standard_normal(n))
                lr = 0.3
        if it % 30 == 29:
            alpha = 0.65 + 0.25 * min(1.0, it / 500.0)
            a_s = spectral_fn(a, N, alpha)
            a_try = proj(0.7 * a + 0.3 * a_s)
            s_try = float(cov_full(a_try, N).max()) * 2.0 * n
            if s_try < best_sc:
                a = a_try
                best_sc = s_try
                best = a.copy()
                no_improve = 0
            a_s2 = spectral_c(a, N, alpha)
            a_try2 = proj(0.6 * a + 0.4 * a_s2)
            s_try2 = float(cov_full(a_try2, N).max()) * 2.0 * n
            if s_try2 < best_sc:
                a = a_try2
                best_sc = s_try2
                best = a.copy()
                no_improve = 0
        if no_improve > 80:
            a = proj(a + 0.02 * rng.standard_normal(n))
            lr = 0.5
            no_improve = 0
        it += 1
    return best, best_sc / (2.0 * n)

def run_spectral_only(a, N, deadline, seed):
    rng = np.random.default_rng(seed)
    n = a.size
    a = proj(a)
    best = a.copy()
    best_sc = sc(a, N)
    it = 0
    no_improve = 0
    while time.monotonic() < deadline:
        alphas = (0.6, 0.65, 0.7, 0.75, 0.8)
        found = False
        for alpha in alphas:
            for method in (0, 1):
                if method == 0:
                    a_new = spectral_fn(a, N, alpha)
                else:
                    a_new = spectral_c(a, N, alpha)
                for mix in (0.3, 0.5, 0.7):
                    a_try = proj((1 - mix) * a + mix * a_new)
                    s_try = sc(a_try, N)
                    if s_try < best_sc:
                        a = a_try
                        best_sc = s_try
                        best = a.copy()
                        no_improve = 0
                        found = True
        if not found:
            no_improve += 1
            a = proj(a + 0.01 * rng.standard_normal(n))
            no_improve += 1
        if no_improve > 40:
            break
        it += 1
    return best, best_sc

def make_inits(n, N, parent, seed):
    rng = np.random.default_rng(seed)
    ones = np.ones(n) / n
    inits = [proj(parent.copy()), ones.copy()]
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    for freq in (1, 2, 3, 5):
        for amp in (0.5, 0.8, 1.0):
            v = amp * np.sin(2.0 * np.pi * freq * (x - center) / n) / np.sqrt(n)
            inits.append(proj(ones + v))
    for alpha in (0.5, 0.6, 0.7):
        inits.append(proj(spectral_fn(ones, N, alpha)))
        inits.append(proj(spectral_c(ones, N, alpha)))
    for _ in range(3):
        y = np.zeros(n)
        for kk in range(1, 6):
            y += np.sin(2.0 * np.pi * kk * x / n + rng.normal() * 2.0) / np.sqrt(n)
        inits.append(proj(ones + 0.3 * y))
    return inits
