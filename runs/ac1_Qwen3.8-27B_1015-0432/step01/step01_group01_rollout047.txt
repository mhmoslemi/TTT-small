import numpy as np
import time

def fft_convolve(a, L):
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)

def compute_grad(x, conv, T, L):
    """Compute soft gradient using softmax weights."""
    # Softmax weights over convolution lags
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / T)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(conv.size)
        w[conv.argmax()] = 1.0
    else:
        w /= s
    # Cross-correlation: grad[i] = sum_j w[i+j] * x[j]
    x_rev = x[::-1]
    # Full correlation via FFT
    X = np.fft.rfft(x_rev, L)
    W = np.fft.rfft(w, L)
    corr = np.fft.irfft(X * W, L)
    n = x.size
    return corr[n-1:2*n-1]

def project_simple(x, target_sum):
    """Project to {x >= 0, sum(x) = target_sum} with clipping and scaling."""
    x = np.clip(x, 0.0, None)
    s = x.sum()
    if s < 1e-15:
        return np.full(x.size, target_sum / x.size)
    return x * (target_sum / s)

def adam_step(x, grad, m, v, t, lr, beta1, beta2, target_sum):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    return project_simple(x_new, target_sum), m, v
