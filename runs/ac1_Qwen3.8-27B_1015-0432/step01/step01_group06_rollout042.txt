def _fft_conv(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

    order = np.argsort(corr)[-K:]
    thresh = corr[order[0]]
    active = np.where(corr >= thresh)[0]
