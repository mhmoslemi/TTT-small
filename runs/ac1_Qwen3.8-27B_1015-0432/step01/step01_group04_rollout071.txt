def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def normalize(a, fb):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.asarray(fb, dtype=np.float64).copy()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
    return a / s

def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s <= 0:
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))

def compute_grad(a, z, N):
    n = a.size

    g = conv2[n - 1:2 * n - 1][::-1].copy()
    g = np.where(np.isfinite(g), g, 0.0)
    return g
