import numpy as np
import time

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s <= 1e-6 or not np.isfinite(s):
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    mc = float(c.max())
    return 2.0 * n * mc / (s * s)

def grad_fft(a, c, n, Ng):
    L = 2 * n - 1
    mc = float(c.max())
    beta = 60.0
    w = np.exp(np.clip((c - mc) * beta, -500.0, 0.0))
    ws = float(w.sum())
    if ws < 1e-300:
        w = np.zeros(L)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws
    w_rev = w[::-1]
    Wr = np.fft.rfft(w_rev, Ng)
    Ap = np.fft.rfft(a, Ng)
    cf = np.fft.irfft(Wr * Ap, Ng)
    g = cf[2 * n - 2 : n - 2 : -1].copy()
    return np.nan_to_num(g, nan=0.0, posinf=0.0, neginf=0.0)

def spectral_init(a, N, n, alpha):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    cs = float(c.sum())
    if cs < 1e-300:
        return a.copy()
    ct = (c / cs) ** alpha
    ct /= ct.sum()
    T = np.fft.rfft(ct, N)
    T_pos = np.abs(T)
    a2 = np.fft.irfft(np.sqrt(T_pos), N)[:n]
    a2 = np.maximum(a2, 0.0)
    s2 = float(a2.sum())
    if s2 < 1e-10:
        return a.copy()
    return a2 / s2

            best_sc = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:2 * n - 1]
        c = np.maximum(c, 0.0)
        g_a = grad_fft(a, c, n, Ng)

        g_y -= float(g_y.mean())
