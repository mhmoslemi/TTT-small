import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def sc(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def mupdate(f, beta, eta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()
    w = np.exp(beta * (b - M) / M)
    g = 2.0 * fft_conv(w, f[::-1])[n - 1:2 * n - 1]
    g = np.maximum(g, 0.0)
    gm = g.max()
    if gm < 1e-30:
        return f
    f2 = f * np.exp(-eta * g / gm)
    f2 = np.maximum(f2, 1e-20)
    return f2 / f2.sum()

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def sc(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def mupdate(f, beta, eta):
    n = f.size
    b = fft_conv(f, f)
    M = b.max()

    g[g < 0.0] = 0.0
    gm = g.max()

        return f
    f2 = f * np.exp(-eta * g / gm)
    f2[f2 < 1e-20] = 1e-20
