import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    n = len(a)
    return 2.0 * n * float(np.max(_fft_conv(a))) / (S * S)


def _grad_multi(a, conv, S, M, topk=3, sharp=25.0):
    n = len(a)
    idx = np.argsort(conv)[::-1][:topk]
    wts = np.exp(sharp * (conv[idx] - M))
    wts /= wts.sum()
    g = np.full(n, -4.0 * n * M / (S * S * S))
    for w_i, k_i in zip(wts, idx):
        js = max(0, k_i - n + 1)
        je = min(n - 1, k_i)
        if js <= je:
            jj = np.arange(js, je + 1)
            g[jj] += w_i * 4.0 * n * a[k_i - jj] / (S * S)
    return g
