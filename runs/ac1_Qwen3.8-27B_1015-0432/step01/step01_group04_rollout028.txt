def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:

    return m

def project_pos(a, fb):
    a = np.asarray(a, dtype=np.float64).copy()
    fb = np.asarray(fb, dtype=np.float64).copy()
    if a.size != fb.size:
        a = fb.copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)

    if not np.isfinite(s) or s < 0.01:
        a = np.clip(fb, 0.0, 1000.0)
        s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:

    return a

def fft_score(a, N):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf, 0, 0.0
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    mc = float(c.max())
    ki = int(c.argmax())
    sc = 2.0 * n * mc / (s * s)
    if not np.isfinite(sc):
        return np.inf, 0, 0.0
    return sc, ki, mc

def spectral_step(a, n, N, alpha, shift_peak):
    s = float(a.sum())
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:2*n-1]
    c = np.clip(c, 0.0, None)
    cs = float(c.sum())
    if cs < 1e-30:
        return a.copy()
    c = c / cs
    # Suppress the peak
    if shift_peak:
        cmax = float(c.max())
        c = c - 0.3 * cmax
        c = np.clip(c, 1e-15, None)
        c = c / c.sum()
    ct = np.power(c, alpha)
    ct = ct / ct.sum()
    T = np.fft.rfft(ct, N)
    T = np.clip(T, 1e-30, None)
    A2 = np.sqrt(T)
    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    return project_pos(a2, a)

    kf = np.arange(N // 2 + 1, dtype=np.float64)
    
    m1 = np.zeros(n)
    v1 = np.zeros(n)
    b1c = 1.0
    b2c = 1.0
    t = 0
    lr = 0.3
    best_a = a0.copy()
    best_sc = fft_score(a0, N)[0]
    no_imp = 0
    
    while time.monotonic() < deadline:
        a = np.exp(y - y.max())
        a = a / a.sum()
        a = np.clip(a, 1e-300, 1000.0)
        
        s = float(a.sum())
        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:L]
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.clip(c, 0.0, None)
        mc = float(c.max())
        ki = int(c.argmax())
        sc = 2.0 * n * mc / (s * s)
        
        if np.isfinite(sc):
            if sc < best_sc:
                best_a = a.copy()
                best_sc = sc
                no_imp = 0
            else:
                no_imp += 1
        
        if not np.isfinite(sc) or mc < 1e-30:
            break
        
        # Gradient of conv[ki] w.r.t. a
        phase = np.cos(2.0 * np.pi * ki * kf / N)
        gc = np.fft.irfft(A * (2.0 * phase), N)[:n]
        gc = np.where(np.isfinite(gc), gc, 0.0)
        
        # Gradient of score
        ga = 2.0 * n * gc / (s * s) - 4.0 * n * mc / (s**3)
        
        # Chain rule to y: da/dy_j = a_j(1-a_j), da_i/dy_j = -a_i*a_j
        ga_dot_a = float(ga @ a)
        gy = a * (ga - ga_dot_a)
        gy = np.where(np.isfinite(gy), gy, 0.0)
        
        t += 1
        b1c *= 0.9
        b2c *= 0.999
        m1 = 0.9 * m1 + 0.1 * gy
        v1 = 0.999 * v1 + 0.001 * gy * gy
        mb = m1 / (1.0 - b1c)
        vb = v1 / (1.0 - b2c)
        
        eff_lr = lr / (1.0 + 0.005 * max(0, no_imp - 200))
        step = eff_lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.8, 0.8)
        y = y - step
        y -= y.mean()
        y = np.clip(y, -40.0, 40.0)
        
        if no_imp > 800 and no_imp % 100 == 0:
            y = np.log(np.clip(best_a, 1e-15, 1000.0))
            y -= y.mean()
            y += 0.2 * rng.standard_normal(n)
            y = np.clip(y, -40.0, 40.0)
            m1.fill(0.0)
            v1.fill(0.0)
            b1c = 1.0
            b2c = 1.0
            t = 0
            no_imp = 0
            lr = max(0.03, lr * 0.5)
    
    return best_a, best_sc

def local_search(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = a.copy()
    best_sc = fft_score(a, N)[0]
    lo = max(1, n // 400)
    hi = max(lo + 1, n // 6)
    while time.monotonic() < deadline:
        w = int(rng.integers(lo, hi + 1))
        st = int(rng.integers(0, n - w + 1))
        cand = best.copy()
        f = float(np.exp(rng.normal(0.0, 0.12)))
        cand[st:st+w] *= f
        cand = project_pos(cand, best)
        sc = fft_score(cand, N)[0]
        if np.isfinite(sc) and sc < best_sc:
            best = cand
            best_sc = sc
    return best, best_sc
