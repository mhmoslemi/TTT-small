    m = 1
    while m < s:
        m <<= 1
    return m


def _cconv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _xconv(a, b):
    L = _fpow2(len(a) + len(b) - 1)
    return ifft(fft(a, L) * fft(b, L)).real[:len(a) + len(b) - 1]


def _smax(c, tau):
    cm = c.max()
    if tau < 1e-14:
        p = np.zeros(c.size)
        p[int(np.argmax(c))] = 1.0
        return cm, p
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_cconv(a), tau)
    return 2.0 * n * mv / (S * S)
