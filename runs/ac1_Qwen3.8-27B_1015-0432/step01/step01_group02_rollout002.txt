    return np.fft.irfft(F * F, N).real


def _grad_from_peaks(a, c, n, idx, weights):
    h = np.zeros(n)
    for j in range(len(idx)):
        kj = int(idx[j])
        wj = float(weights[j])
        lo = max(0, kj - n + 1)
        hi = min(n - 1, kj)
        if hi >= lo:
            h[lo:hi + 1] += wj * a[kj - np.arange(lo, hi + 1)]
    return h


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)
    n = len(height_sequence_1)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-9:
        a = np.ones(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_score = 2.0 * n * _conv(a, n).max()
    best_a = a.copy()

    t = np.linspace(0.0, 1.0, n)
    inits = [
        np.maximum(np.sin(np.pi * t), 1e-12),
        np.maximum(np.sin(np.pi * t) ** 2, 1e-12),
        np.ones(n),
        np.maximum(1.0 - 2.0 * np.abs(t - 0.5), 1e-12),
        np.exp(-5.0 * (t - 0.5) ** 2),
        0.5 + 0.5 * np.cos(np.pi * t),
    ]
    for ic in inits:
        ic = np.maximum(ic, 1e-12)
        ic /= ic.sum()
        sc = 2.0 * n * _conv(ic, n).max()
        if sc < best_score:
            best_score = sc
            best_a = ic.copy()
            a = ic.copy()

    alpha = 0.10
    peak_thresh_frac = 0.82
    peak_cap = 35
    mom_decay = 0.92
    mom_buf = np.zeros(n)
    iters = 0
    stuck = 0

    while time.monotonic() < deadline:
        c = _conv(a, n)
        score = 2.0 * n * c.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        cmax = c.max()
        thresh = peak_thresh_frac * cmax
        idx = np.where(c >= thresh)[0]
        if len(idx) > peak_cap:
            idx = np.argpartition(c, -peak_cap)[-peak_cap:]
        if len(idx) < 3:
            idx = np.argpartition(c, -7)[-7:]

        peaks = c[idx]
        pmax = peaks.max()
        if pmax > 1e-15:
            weights = (peaks / pmax) ** 1.8
            weights /= weights.sum()
        else:
            weights = np.ones(len(idx)) / len(idx)

        h = _grad_from_peaks(a, c, n, idx, weights)
        mom_buf = mom_decay * mom_buf + (1.0 - mom_decay) * h
        h_use = mom_buf

        h_max = h_use.max()
        if h_max > 1e-15:
            a *= np.exp(-alpha * h_use / h_max)

        a = np.maximum(a, 1e-12)
        a /= a.sum()
        iters += 1

        if iters % 120 == 0:
            eps = 0.04 + min(0.06, 0.01 * stuck / 200.0)
            noise = np.abs(rng.normal(0.0, 1.0, n))
            noise /= noise.sum() * n
            a = (1.0 - eps) * a + eps * noise
            a = np.maximum(a, 1e-12)
            a /= a.sum()
            mom_buf *= 0.4

        if iters % 600 == 0:
            k = max(3, n // 250)
            kernel = np.ones(k) / k
            a = np.convolve(a, kernel, mode='same')
            a = np.maximum(a, 1e-12)
            a /= a.sum()

        if stuck > 600:
            alpha *= 0.82
            peak_thresh_frac = min(0.95, peak_thresh_frac + 0.02)
            if alpha < 0.025:
                w = int(rng.integers(max(10, n // 20), max(20, n // 3)))
                st = int(rng.integers(0, n - w))
                a[st:st + w] *= float(np.exp(rng.normal(0.0, 0.65)))
                a = np.maximum(a, 1e-12)
                a /= a.sum()
                alpha = 0.10
                peak_thresh_frac = 0.82
                mom_buf = np.zeros(n)
                stuck = 0

    result = [float(x) for x in best_a]
    fs = float(evaluate_sequence(result))
    if not np.isfinite(fs) or fs > 2.5:
        result = [1.0] * n
        fs = float(evaluate_sequence(result))
    return result
