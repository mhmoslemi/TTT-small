    g /= float(active.size)
    return g

def _norm1(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s

def _gen_init(kind, n, rng):
    if kind == 0:
        a = np.ones(n)
    elif kind == 1:
        a = np.abs(rng.standard_normal(n))
    elif kind == 2:
        f = rng.uniform(0.05, 0.95)
        p = rng.uniform(0, 2 * np.pi)
        a = np.clip(1 + 0.5 * np.sin(2 * np.pi * f * np.arange(n) + p), 0.01, None)
    elif kind == 3:
        a = np.abs(rng.standard_normal(n)) ** 2
    elif kind == 4:
        a = np.exp(-((np.arange(n) - 0.5 * n) ** 2) / (2.0 * (0.25 * n) ** 2))
    elif kind == 5:
        m = int(np.clip(rng.integers(2, n), 2, n - 1))
        a = 1.0 + 0.8 * np.cos(2 * np.pi * m * np.arange(n) / n)
        a = np.clip(a, 0.01, None)
    else:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(0.1, 0.5) * np.arange(n))
        a = np.clip(a, 0.01, None)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n)
        s = n
    return a / s

def _optimize(a, n, fs, rng, deadline, frac=0.85, max_iter=80000):
    v = np.zeros(n)
    beta = 0.80
    lr = 0.35
    prev_M = np.inf
    stall = 0
    best_a = a.copy()
    best_M = np.inf
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        conv = _fft_conv(a, fs)
        M = float(np.max(conv[:2 * n - 1]))
        if M < best_M:
            best_M = M
            best_a = a.copy()
        prev_M = M
        stall = 0 if M < prev_M - 1e-18 else stall + 1
        if stall > 2500:
            break
        a_n = a - beta * v
        g = _grad(a_n, conv, n, frac)
        g -= g.mean()
        v = beta * v + (1.0 - beta) * g
        a_new = a - lr * v
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-15:
            break
        a = a_new / s
        lr *= 0.99985
        if lr < 0.008:
            lr = 0.008
    return a, best_a, best_M

def _polish(base, n, s0, best_score, rng, deadline):
    best = base.copy()
    ws = max(3, n // 80)
    while time.monotonic() < deadline:
        cand = best.copy()
        for _ in range(int(rng.integers(1, 4))):
            st = int(rng.integers(0, n - ws))
            cand[st:st + ws] *= float(np.exp(rng.normal(0.0, 0.025)))
        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc
    return best, best_score

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t_end = time.monotonic() + max(0.0, float(budget_s) - 10.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.size < 2 or parent.ndim != 1 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(1024)
    n = int(parent.size)
    s0 = float(parent.sum())
    fs = 1 << (2 * n - 1).bit_length()
    a0 = parent / s0

    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    def _try(a_normed):
        nonlocal best, best_score
        cand = np.clip(a_normed * s0, 0.0, 1000.0)
        if cand.sum() < 0.01:
            return
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = cand.copy()
            best_score = sc

    # Phase 1: optimize from parent
    if time.monotonic() < t_end - 200:
        _, ba, _ = _optimize(a0.copy(), n, fs, rng, t_end - 500, frac=0.85)
        _try(ba)

    # Phase 2: diverse starts
    n_starts = 6
    for k in range(n_starts):
        remain = t_end - time.monotonic()
        if remain < 80:
            break
        per_start = min(150.0, (t_end - 100.0 - time.monotonic()) / (n_starts - k))
        dl = time.monotonic() + per_start
        a_init = _gen_init(k, n, rng)
        _, ba, _ = _optimize(a_init, n, fs, rng, dl, frac=0.80 + 0.03 * k)
        _try(ba)

    # Phase 3: second pass from current best (normalized)
    remain = t_end - time.monotonic()
    if remain > 100:
        a_cur = np.clip(best, 0, 1000) / s0
        _, ba, _ = _optimize(a_cur, n, fs, rng, t_end - 120, frac=0.92)
        _try(ba)

    # Phase 4: polish
    if time.monotonic() < t_end - 5:
        best, best_score = _polish(best, n, s0, best_score, rng, t_end - 3.0)

    out = np.clip(best, 0.0, 1000.0)
    if out.size < 1 or not np.isfinite(out).all() or out.sum() < 0.01:
        out = np.clip(parent, 0.0, 1000.0)
    return [float(x) for x in out]
