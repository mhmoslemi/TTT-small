        idx = int(idx)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        if lo <= hi:
            g[lo:hi + 1] += a[idx - np.arange(lo, hi + 1)]
            cnt += 1
    if cnt > 0:
        g /= cnt
    g -= g.mean()
    return g

def _mk_shapes(n, rng):
    j = np.arange(n, dtype=np.float64)
    pi = math.pi
    shapes = [
        np.ones(n),
        1.0 + 0.5 * np.cos(2.0 * pi * j / n),
        1.0 + 0.5 * np.cos(4.0 * pi * j / n),
        np.abs(np.sin(pi * j / n)),
        1.0 + 0.3 * np.cos(2.0 * pi * j / n) + 0.2 * np.cos(6.0 * pi * j / n),
        1.0 + 0.3 * np.cos(3.0 * pi * j / n) + 0.2 * np.cos(7.0 * pi * j / n),
        1.0 + 0.3 * np.sin(2.0 * pi * j / n) + 0.2 * np.sin(5.0 * pi * j / n),
        1.0 + 0.25 * np.cos(2.0 * pi * j * j / n),
        np.repeat(rng.random(max(2, n // 200) + 1), max(2, n // 200))[:n],
        rng.beta(2.0, 2.0, n),
    ]
    out = []
    for s in shapes:
        s = np.clip(s, 0.0, 1000.0)
        ss = s.sum()
        if ss < 1e-10:
            s = np.ones(n)
            ss = float(n)
        out.append(s * n / ss)
    return out

def _try_improve(a, n, corr, sc, lr, v, k, dl):
    g = _subgrad(corr, a, n, k)
    gnorm = np.linalg.norm(g)
    if gnorm < 1e-14:
        return a, v, lr, sc, True
    v = 0.85 * v + g if np.dot(v, v) > 0 else g
    beta = 0.88
    v = beta * v + g
    vn = np.linalg.norm(v)
    if vn < 1e-14:
        return a, v, lr, sc, True
    d = -v / vn
    for f in (1.0, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04):
        a_new = a + lr * f * d
        a_new = np.clip(a_new, 0.0, 1000.0)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = np.ones(n)
            sn = float(n)
        a_new = a_new * n / sn
        c2 = _fft_conv(a_new)
        sc2 = 2.0 * n * c2.max() / (n * n)
        if sc2 < sc - 1e-13:
            return a_new, v, min(lr * 1.04, 4.0), sc2, False
        if time.monotonic() > dl:
            break
    return a, v, max(lr * 0.72, 0.0005), sc, False

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 10.0)

    try:
        par = np.asarray(height_sequence_1, dtype=np.float64)
        par = np.clip(par, 0.0, 1000.0)
        n = par.size
    except Exception:
        par = None
        n = 1024
    if n < 2:
        n = 1024

    best = None
    best_sc = np.inf
    starts = []
    if par is not None and par.sum() > 1e-10:
        starts.append(par * n / par.sum())
    starts.extend(_mk_shapes(n, rng))

    for s in starts:
        s = np.clip(s, 0.0, 1000.0)
        if s.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(s.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = s.copy()

    if best is None:
        best = np.ones(n)
        best_sc = 2.0

    a = best.copy()
    v = np.zeros(n)
    lr = 0.5
    stuck = 0
    it = 0
    phase = 0

    while time.monotonic() < dl - 5.0:
        it += 1

        if it < 800:
            k = 16
        elif it < 2500:
            k = 8
        elif it < 6000:
            k = 4
        elif it < 12000:
            k = 2
        else:
            k = 1

        corr = _fft_conv(a)
        sc = 2.0 * n * corr.max() / (n * n)

        if sc < best_sc - 1e-12:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 300 == 0:
            kec = float(evaluate_sequence(a.tolist()))
            if np.isfinite(kec) and kec < best_sc:
                best_sc = kec
                best = a.copy()
                stuck = 0

        if time.monotonic() > dl - 3.0:
            break

        a, v, lr, sc, stuck_hard = _try_improve(a, n, corr, sc, lr, v, k, dl)

        if stuck_hard:
            break

        improved_internal = sc < 2.0 * n * corr.max() / (n * n) + 1e-13
        if not improved_internal and lr < 0.001 and stuck > 50:
            if stuck > 2000:
                alt = _mk_shapes(n, rng)
                idx = rng.integers(0, len(alt))
                a = 0.4 * best + 0.6 * alt[idx]
                a = np.clip(a, 0, 1000)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a * n / s2
                v[:] = 0.0
                lr = 0.5
                stuck = 0
            elif stuck > 500:
                a = 0.5 * (a + a[::-1])
                a += rng.normal(0.0, 0.03, n)
                a = np.clip(a, 0, 1000)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a * n / s2
                v[:] = 0.0
                lr = 0.5
                stuck = 0
            elif stuck > 150:
                a += rng.normal(0.0, 0.04, n)
                a = np.clip(a, 0, 1000)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a * n / s2
                v[:] = 0.0
                stuck = 0

    # Final polish: k=1, small steps
    if time.monotonic() < dl - 4.0:
        a = best.copy()
        v2 = np.zeros(n)
        for _ in range(1000):
            if time.monotonic() > dl - 4.0:
                break
            corr = _fft_conv(a)
            sc = 2.0 * n * corr.max() / (n * n)
            g = _subgrad(corr, a, n, 1)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break
            v2 = 0.9 * v2 + g
            vn2 = np.linalg.norm(v2)
            if vn2 < 1e-14:
                break
            d = -v2 / vn2
            done = True
            for f in (0.05, 0.03, 0.015, 0.008, 0.004, 0.002):
                a2 = a + f * d
                a2 = np.clip(a2, 0, 1000)
                sn = a2.sum()
                if sn < 1e-10:
                    continue
                a2 = a2 * n / sn
                c2 = _fft_conv(a2)
                sc2 = 2.0 * n * c2.max() / (n * n)
                if sc2 < sc - 1e-14:
                    a = a2
                    done = False
                    break
            if done:
                break
        sc_fin = 2.0 * n * _fft_conv(a).max() / (n * n)
        if sc_fin < best_sc:
            best = a.copy()
            best_sc = sc_fin

    out = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in out]
