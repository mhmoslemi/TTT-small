import numpy as np
import time


def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = np.maximum(c[:2 * n - 1], 0.0)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def compute_grad(a, w, N):
    n = a.size
    V = np.fft.rfft(w[::-1], N)
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(V * A, N)
    return C[2 * n - 2 - np.arange(n)]


def project(a, n):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s
    return a
