6. Project to simplex

`a[i] <- a[i] * (1 - eta * (p[i] - mean(p)))`

        u /= u.sum()

    for init in init_list[1:]:

        if np.isfinite(s) and s < best_s:

            best = init.copy()
    
    a = best.copy()

    eta = 0.8
    beta_ws = 15.0

    eval_interval = 200

            c = np.maximum(c, 0)
            w = np.exp(beta_ws * (c - c.max()))
            w /= w.sum()

            pr = np.maximum(pr, 0.0)
            a = a * np.exp(-eta * (pr - pr.mean()))
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            t_iter += 1

        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best = a.copy()

        eta *= 0.97

            a = np.maximum(a, 1e-15)
            a /= a.sum()
            beta_ws = 15.0
            eta = 0.8
    
    # Final
    fs = float(evaluate_sequence(a.tolist()))
    if np.isfinite(fs) and fs < best_s:
        best = a.copy()
    
    result = [float(x) for x in np.clip(best, 0.0, 1000.0)]
    if not result or sum(result) < 0.01:
        result = [1.0] * n
    return result
