def fft_conv(a, fsz):
    fa = np.fft.rfft(a, n=fsz)
    return np.fft.irfft(fa * fa, n=fsz)

def grad_topk(a, cv, tk):
    n = a.size
    idx = np.argpartition(cv, -tk)[-tk:]
    g = np.zeros(n)
    for kk in idx:

        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)

            g[lo:hi + 1] += a[kk - hi:kk - lo + 1][::-1]
    g /= tk
    return g
