def _fft_conv(a):
    n = len(a)
    N = 2 * n - 1
    Nf = 1
    while Nf < N:
        Nf *= 2
    F = np.fft.rfft(a, Nf)
    out = np.fft.irfft(F * F, Nf)[:N]
    return out.real


def _score(a):
    n = len(a)
    c = _fft_conv(a)
    s = a.sum()
    if s < 1e-12:
        return 1e18
    return 2.0 * n * c.max() / (s * s)
