            z /= zs

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:

def fft_score(a, N):
    n = a.size
    s = float(a.sum())
    if s < 1e-300:
        return np.inf
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = np.maximum(c[:2 * n - 1], 0.0)
    return float(2.0 * n * float(np.max(c)) / (s * s))
