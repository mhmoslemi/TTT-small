import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]


def fast_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2 * len(f) * b.max() / (s * s))


def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones_like(f) / f.size
    return f / s


def compute_grad(f, b, betas):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return None
    f_r = f[::-1]
    fft_fr = np.fft.rfft(f_r, 1 << (n + n - 2).bit_length())
    grad = np.zeros(n)
    for beta in betas:
        w = np.exp(beta * b / M)
        wmx = w.max()
        if wmx > 1e-30:
            w /= wmx
        sz = 1 << (len(w) + n - 2).bit_length()
        gc = np.fft.irfft(np.fft.rfft(w, sz) * fft_fr)[:len(w) + n - 1]
        grad += gc[n - 1:2 * n - 1]
    grad = np.maximum(grad, 0.0)
    return grad
