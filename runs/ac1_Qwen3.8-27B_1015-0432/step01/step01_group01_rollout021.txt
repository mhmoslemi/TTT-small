import numpy as np

def fft_conv(a, fs):
    fa = np.fft.rfft(a, n=fs)
    return np.fft.irfft(fa * fa, n=fs)


def proj_simplex(v, s):
    n = v.size
    if s <= 0:
        return np.zeros(n)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(50):
        tau = (lo + hi) * 0.5
        if float(np.maximum(v - tau, 0.0).sum()) > s:
            lo = tau
        else:
            hi = tau
    return np.maximum(v - (lo + hi) * 0.5, 0.0)
