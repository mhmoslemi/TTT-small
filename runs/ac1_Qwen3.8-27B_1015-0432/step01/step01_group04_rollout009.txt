                if np.isfinite(s2) and s2 <= sc(cur, N)[0] * 1.05:
                    cur = cand
            _, _, c = sc(cur, N)
    return ba, bs


def rand_ls(a, n, N, dead, seed):
    rng = np.random.default_rng(seed)
    ba = proj(a, a)
    bs = sc(ba, N)[0]
    lo = max(1, n // 400)
    hi = max(lo + 1, n // 6)
    while time.monotonic() < dead:
        w = int(rng.integers(lo, hi + 1))
        st = int(rng.integers(0, n - w + 1))
        cd = ba.copy()
        cd[st:st + w] *= float(np.exp(rng.normal(0, 0.12)))
        cd = proj(cd, ba)
        s2 = sc(cd, N)[0]
        if np.isfinite(s2) and s2 < bs:
            bs = s2
            ba = cd
    return ba, bs


def make_inits(n, seed, par, N):
    rng = np.random.default_rng(seed)
    ones = np.ones(n) / n
    out = [par.copy(), ones]
    x = np.linspace(0, 1, n)
    for A in (1, 2, 4, 8):
        out.append(np.exp(-A * x) / np.exp(-A * x).sum())
        out.append(np.exp(A * (x - 0.5)**2) / np.exp(A * (x - 0.5)**2).sum())
    j = np.arange(n, dtype=np.float64)
    for _ in range(3):
        yv = np.zeros(n)
        for k in range(1, 6):
            ph = rng.uniform(0, 2 * np.pi)
            yv += np.cos(2 * np.pi * k * j / n + ph) / k
        out.append(np.exp(0.5 * yv - 0.5 * yv.max()) / np.exp(0.5 * yv - 0.5 * yv.max()).sum())
    return out
