def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()

    return float(2 * len(f) * b.max() / (s * s))

    n = len(f)

    # A[k,j] = f[j] if 0 <= k-j <= n-1, else 0
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        lo = max(0, k-(n-1))
        hi = min(n-1, k)
        js = np.arange(lo, hi+1)
        rows.extend([k]*len(js))
        cols.extend(js.tolist())

        if gs > 1e-10:
