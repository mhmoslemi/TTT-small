import numpy as np
import time


def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def score_fft(a, N):
    n = a.size
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    c = c[:L]
    c = np.maximum(0, c)
    mc = float(c.max())
    if not np.isfinite(mc):
        return np.inf
    return 2.0 * n * mc / (s * s)


def softmax_pos(y):
    y = y - float(y.max())
    e = np.exp(np.clip(y, -700, 700))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(y) / len(y)
    return e / s


def project_pos(a, fb):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        a = np.asarray(fb, dtype=np.float64).copy()
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        a = np.ones(len(fb), dtype=np.float64)
        s = float(a.size)
    return a / s


def spectral_step(a, N, alpha):
    n = a.size
    L = 2 * n - 1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L].copy()
    c = np.maximum(0, c)
    cs = float(c.sum())
    if cs <= 0:
        return a.copy()
    c = c / cs
    c = np.power(c + 1e-18, alpha)
    sc = float(c.sum())
    if sc <= 0:
        return a.copy()
    c = c / sc
    T = np.fft.rfft(c, N)
    T = np.maximum(T, 0)
    A2 = np.sqrt(T)
    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.maximum(0, a2)
    s2 = float(a2.sum())
    if s2 <= 0 or not np.isfinite(s2):
        return a.copy()
    return a2 / s2

        sc = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_sc:
            best = cand.copy()
            best_sc = sc
    return best, best_sc

            y = 0.7 * y + 0.3 * y_sp
            y = y - float(y.mean())
            y = np.clip(y, -30, 30)
