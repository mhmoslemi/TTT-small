def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def project_nonneg_sum(a, s=1.0):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    t = float(a.sum())
    if t < 0.01 or not np.isfinite(t):
        return np.ones(a.size) / a.size
    return a * (s / t)

def auto_corr(a, N):
    """Full autocorrelation via FFT. Returns array of length 2n-1."""
    n = a.size
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A.conj(), N)  # For real a, A.conj() = A reversed, but let's be careful
    # Actually for real a: autocorrelation = irfft(|A|^2)
    # But conv(a,a) = irfft(A*A) for real a
    # Let's use irfft(A*A) to get conv(a,a)
    c = np.fft.irfft(A * A, N)
    c = c[:L]
    c = np.maximum(c, 0.0)
    return c

def score_from_corr(c, n):
    """Score = 2n * max(c) / (sum(a))^2, but if we normalize sum(a)=1, it's just 2n*max(c)."""
    mc = float(c.max())
    return 2.0 * n * mc

def score_fft(a, N=None):
    n = a.size
    if N is None:
        N = next_pow2(2 * n - 1)
    A = np.fft.rfft(a, N)
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
        A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:L]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(c.max()) / (s * s)

import numpy as np
import time

def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m

def proj_simplex(a):
    """Project a to the probability simplex {x >= 0, sum(x) = 1}."""
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-30:
        return np.ones(a.size) / a.size
    return a / s

def softmax(y):
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - y.max()
    e = np.exp(np.clip(y, -700, 700))
    s = e.sum()
    if s < 1e-30:
        return np.ones(y.size) / y.size
    return e / s

def score_norm(a, N):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    A = np.fft.rfft(a, N)
    L = 2*n - 1
    c = np.fft.irfft(A*A, N)[:L]
    c = np.maximum(c, 0.0)
    return 2.0 * n * float(c.max())

def exact_score(a):
    try:
        return float(evaluate_sequence(a.tolist()))
    except Exception:
        return np.inf

def compute_fft_grad(a, w, n, Ng):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    w has length 2n-1. Returns array of length n."""
    A = np.fft.rfft(a, Ng)
    W = np.fft.rfft(w, Ng)
    conv_aw = np.fft.irfft(A * W, Ng)
    w_rev = w[::-1]
    Wr = np.fft.rfft(w_rev, Ng)
    conv_awr = np.fft.irfft(A * Wr, Ng)
    # grad[i] = conv_awr[2n-2-i] + conv_aw[i]
    t1 = conv_awr[n-1:2*n-1][::-1]  # conv_awr[2n-2], ..., conv_awr[n-1]
    t2 = conv_aw[:n]
    return t1 + t2

    y = np.log(np.maximum(a, 1e-30))
    y = y - y.mean()
    
    best = a.copy()
    best_sc = score_norm(a, N)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.5
    beta1, beta2 = 0.9, 0.999
    no_improve = 0
    best_beta = 30.0
    
    while time.monotonic() < deadline:
        a = softmax(y)
        A = np.fft.rfft(a, N)
        L = 2*n - 1
        c = np.fft.irfft(A*A, N)[:L]
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        sc = 2.0 * n * max_c
        
        if sc < best_sc - 1e-12:
            best = a.copy()
            best_sc = sc
            no_improve = 0
        else:
            no_improve += 1

            a = proj_simplex(best + 0.02 * rng.standard_normal(n))
            y = np.log(np.maximum(a, 1e-30))
            y = y - y.mean()
            m.fill(0); v.fill(0); t = 0
            no_improve = 0
            best_beta = 30.0

def spectral_step(a, n, N, alpha, mix):
    """One spectral power iteration step."""
    A = np.fft.rfft(a, N)
    L = 2*n - 1
    c = np.fft.irfft(A*A, N)[:L]
    c = np.maximum(c, 0.0)
    cs = c.sum()
    if cs < 1e-30:
        return a.copy()
    c = c / cs
    c = np.power(c + 1e-18, alpha)
    cs2 = c.sum()
    if cs2 < 1e-30:
        return a.copy()
    c = c / cs2
    T = np.fft.rfft(c, N)
    T = np.maximum(T, 0.0)  # Ensure non-negative for sqrt
    A2 = np.sqrt(T)
    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.real(a2)
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    a_new = (1.0 - mix) * a + mix * np.maximum(a2, 0.0)
    return proj_simplex(a_new)

def random_search(a_best, n, N, deadline, seed):
    """Random block scaling search."""
    rng = np.random.default_rng(seed)
    a = a_best.copy()
    best_sc = score_norm(a, N)
    L = 2*n - 1
    while time.monotonic() < deadline:
        # Random block modification
        width = rng.integers(max(2, n//500), max(3, n//20))
        start = rng.integers(0, n - width)
        cand = a.copy()
        factor = np.exp(rng.standard_normal() * 0.1)
        cand[start:start+width] *= factor
        cand = proj_simplex(cand)
        sc = score_norm(cand, N)
        if sc < best_sc:
            a = cand
            best_sc = sc
    return a, best_sc
