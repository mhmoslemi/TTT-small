import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:need]


def _np_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _multi_subgrad(a, conv, S, topK, tau):
    n = len(a)
    M = float(conv.max())
    order = np.argsort(conv)[::-1]
    kk = order[:topK]
    vals = conv[kk]
    w = np.exp((vals - M) / tau)
    wsum = float(w.sum())
    if wsum <= 0.0:
        w = np.full(len(kk), 1.0 / len(kk))
    else:
        w = w / wsum
    # base term: sum_k w_k ( -4 n conv[k] / S^3 ) / a_i  for every i
    base_sum = -4.0 * n / (S * S * S) * float(np.dot(w, vals))
    g = np.full(n, base_sum)
    for cnt in range(len(kk)):
        k = int(kk[cnt])
        wk = float(w[cnt])
        js = k - (n - 1)
        if js < 0:
            js = 0
        je = k
        if je > n - 1:
            je = n - 1
        if js > je:
            continue
        ii = np.arange(js, je + 1)
        g[ii] += (4.0 * n / (S * S)) * wk * a[k - ii]
    return g


def _try_step(a, base_lr):
    n = len(a)
    S0 = float(np.sum(a))
    if S0 < 0.01:
        return a, np.inf, 0.0
    conv = _fft_conv(a)
    M = float(conv.max())
    tau = max(1e-9, 0.08 * M)
    g = _multi_subgrad(a, conv, S0, 16, tau)
    if not np.all(np.isfinite(g)):
        g = np.full(n, -4.0 * n * M / (S0 ** 3))
    cur = 2.0 * n * M / (S0 * S0)
    best = cur
    best_a = a.copy()
    best_lr = 0.0
    gnorm = float(np.sqrt(np.dot(g, g)))
    if gnorm < 1e-12:
        return best_a, cur, 0.0
    g = g / gnorm
    for mult in (1.0, 0.4, 0.15, 2.2, 4.0):
        step = base_lr * mult
        cand = np.clip(a + 0.5 * step * g, 0.0, 1000.0)
        sc = _np_score(cand)
        if np.isfinite(sc) and sc < best:
            best = sc
            best_a = cand
            best_lr = step
    if best_lr <= 0.0:
        # pure subgradient direction (no normalization push)
        for mult in (1.0, 0.4, 0.15):
            step = base_lr * mult
            cand = np.clip(a - step * g, 0.0, 1000.0)
            sc = _np_score(cand)
            if np.isfinite(sc) and sc < best:
                best = sc
                best_a = cand
                best_lr = step
    return best_a, best, best_lr
