import numpy as np
import time

def _fft_autocorr(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]

def _project_nn(a, target_sum):
    n = a.size
    if n == 0:
        return a
    x = np.sort(a)[::-1]
    cs = np.cumsum(x)
    rho = np.arange(1, n + 1)
    cand = (target_sum - cs) / rho
    ok = cand >= x
    if not np.any(ok):
        return np.full(n, target_sum / n)
    idx = np.max(np.where(ok)[0])
    alpha = cand[idx]
    return np.maximum(a - alpha, 0.0)

def _grad_active(c, a, K):
    n = a.size
    if K >= c.size:
        active = np.arange(c.size)
    else:
        active = np.argpartition(c, -K)[-K:]
    g = np.zeros(n)
    for k in active:
        k = int(k)
        if k < n:
            g[:k + 1] += 2.0 * a[k::-1]
        else:
            lo = k - n + 1
            g[lo:] += 2.0 * a[n - 1:lo - 1:-1]
    g /= len(active)
    return g

def _score_internal(c, n, s):
    if s < 1e-10:
        return 1e30
    return 2.0 * n * c.max() / (s * s)
