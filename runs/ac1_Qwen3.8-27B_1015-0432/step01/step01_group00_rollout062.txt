def fft_conv(a, b):
    n, m = len(a), len(b)

`conv[wc, ar][k] = sum_i w[i] * a_r[k-i] = sum_i w[i] * a[n-1-k+i]`

For k = n-1+j (j = 0, ..., n-1):
`= sum_i w[i] * a[n-1-(n-1+j)+i] = sum_i w[i] * a[i-j]`

This is the cross-correlation `(w ⋆ a)[j] = sum_i w[i] * a[i-j]`.

And the gradient at a[j] should involve:
`sum_k w[k] * a[k-j]` (for k-j in [0,n-1])
+ `sum_k w[k] * a[j-k]` (for j-k in [0,n-1])

The first term: `sum_m w[m+j] * a[m]` where m = k-j. This is like the cross-correlation of w and a at lag -j... hmm.

The second term: `sum_m w[k] * a[m]` where m = j-k, so k = j-m. This is `sum_m w[j-m] * a[m]`. This is the cross-correlation of w and a at lag j.

So the gradient is: `cross_corr(w, a, lag=j) + cross_corr(w, a, lag=-j)`

For a symmetric w (which it is, since w = exp(beta * conv/M) and conv is symmetric), this is `2 * cross_corr(w, a, lag=|j|)`.

This is the cross-correlation at lag -j: `sum_i w[i] * a[i-j]`.

```python
def spectral_shaping(a, compress, deadline):
    n = len(a)
    N = 1 << (2*n-1).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    med = np.median(mag)
    # Compress peaks above median
    new_mag = np.where(mag > med, med + (mag - med) * compress, mag)
    new_mag[0] = mag[0]  # preserve DC
    A_new = new_mag * (A / np.maximum(mag, 1e-30))
    a_new = np.fft.irfft(A_new, N)[:n]
    a_new = np.maximum(a_new, 0)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new / a_new.sum()
```

Where `compress` in (0, 1) controls how aggressively peaks are compressed. compress=0 means all peaks set to median (very flat), compress=1 means no change.

import time


def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]


def qscore(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))


def _norm(a):
    s = a.sum()
    return a / s if s > 1e-15 else a


def _sym(a):
    return 0.5 * (a + a[::-1])


def init_parent():
    p = np.clip(np.asarray(height_sequence_1, dtype=float), 0, 1000)
    if p.size > 0 and p.sum() > 1e-10 and np.isfinite(p).all():
        return _norm(_sym(p))
    return None


def init_uniform(n):
    return np.ones(n) / n


def init_cos_taper(n, pwr):
    x = np.arange(n, dtype=float) / max(n - 1, 1)
    t = 2.0 * x - 1.0
    c = np.cos(0.5 * np.pi * t)
    h = c ** pwr + 1e-8
    return _norm(h)


def init_gauss(n, sf):
    x = np.arange(n, dtype=float)
    c = (x - n / 2.0) / max(1.0, n * sf)
    return _norm(np.exp(-0.5 * c * c) + 1e-8)


def spect_compress(a, frac):
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    mag = np.abs(A)
    if mag.size < 3:
        return a.copy()
    med = np.median(mag[1:])
    new_mag = np.where(mag > med, med + (mag - med) * frac, mag)
    new_mag[0] = mag[0]
    safe = np.maximum(mag, 1e-30)
    A2 = (new_mag / safe) * A
    a2 = np.fft.irfft(A2, N)[:n]
    a2 = np.maximum(a2, 0)
    if a2.sum() < 1e-10:
        return a.copy()
    return _norm(a2)

    n = len(a)
    b = fft_conv(a, a)
    M = b.max()
    if M < 1e-30:
        return a
    w = np.exp(beta * b / M)
    w /= w.max()
    gc = fft_conv(w, a[::-1])
    gr = gc[n - 1:2 * n - 1]
    gr = np.maximum(gr, 0)
    gm = gr.max()
    if gm < 1e-30:
        return a
    a2 = a * np.exp(-eta * gr / gm)
    return _norm(np.maximum(a2, 1e-20))
