def subgradient(w, f, n):
    """Compute sg[j] = sum_k w[k] * f[k-j] using FFT."""
    f_flipped = f[::-1]
    N_fft = 1  # next power of 2 >= 3n-2
    while N_fft < 3*n - 2:

import numpy as np
import time


def _next_pow2(m):
    p = 1
    while p < m:
        p *= 2
    return p


def _conv_fft(f, N_fft):
    F = np.fft.rfft(f, N_fft)
    return np.fft.irfft(F * F, N_fft)


def _sg_fft(w, f, N_fft):
    fr = f[::-1]
    W = np.fft.rfft(w, N_fft)
    FR = np.fft.rfft(fr, N_fft)
    full = np.fft.irfft(W * FR, N_fft)
    n = len(f)
    return full[n - 1:n - 1 + n]


def _normalize(x, S):
    x = np.clip(x, 1e-15, None)
    s = x.sum()
    if s < 1e-30:
        return np.full_like(x, S / len(x))
    return x / s * S
