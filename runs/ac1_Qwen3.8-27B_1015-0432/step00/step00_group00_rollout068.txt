import time


def _nfft(sz):
    r = 1
    while r < sz:
        r *= 2
    return r


def _fft_conv_self(a):
    n = len(a)
    sz = 2 * n - 1
    m = _nfft(sz)
    A = np.fft.rfft(a, m)
    return np.fft.irfft(A * A, m)[:sz]


def _fft_grad(a, w):
    n = len(a)
    sz = 3 * n - 1
    m = _nfft(sz)
    B = np.fft.rfft(a[::-1], m)
    W = np.fft.rfft(w, m)
    return 2.0 * np.fft.irfft(B * W, m)[n - 1:2 * n - 1]
