    n = 2 * len(a) - 1

    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:n]

    n = len(f)
    cols = np.repeat(np.arange(n), n)
    rows = cols + np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def self_conv(a):
    n = 2*len(a)-1
    sz = 1 << (n-1).bit_length()
    F = np.fft.rfft(a, sz)
    return np.fft.irfft(F*F, sz)[:n]

    n = len(f)
    cols = np.repeat(np.arange(n), n)
    rows = cols + np.tile(np.arange(n), n)
    data = np.tile(f, n)
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
