def build_T(a, n):
    nn = 2*n - 1
    # For each column j, the nonzeros are at rows j through min(nn-1, j+n-1)
    # Values are a[0], a[1], ...
    row_starts = np.arange(n)  # j
    row_ends = np.minimum(nn - 1, row_starts + n - 1)
    lengths = row_ends - row_starts + 1
    total = lengths.sum()
    rows = np.zeros(total, dtype=int)
    cols = np.zeros(total, dtype=int)
    vals = np.zeros(total, dtype=float)
    offset = 0
    for j in range(n):
        ln = lengths[j]
        rows[offset:offset+ln] = np.arange(row_starts[j], row_starts[j]+ln)
        cols[offset:offset+ln] = j
        vals[offset:offset+ln] = a[:ln]
        offset += ln
    from scipy.sparse import csr_matrix
    return csr_matrix((vals, (rows, cols)), shape=(nn, n))

    rho_arr = np.where(mask)[0]
    if rho_arr.size == 0:
        return np.full(n, s / n)
    rho = rho_arr[-1]
    tau = (cs[rho] - s) / (rho + 1)
    return np.maximum(v - tau, 0.0)


def _subgrad(a, c, n, cap=80):
    cmax = c.max()
    if cmax <= 0:
        return np.zeros(n)
    thr = 1e-4 * cmax
    act = np.where(c >= cmax - thr)[0]
    if act.size > cap:
        act = act[np.argsort(c[act])[-cap:]]
    g = np.zeros(n)
    for m in act:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        j0 = m - hi
        j1 = m - lo
        g[lo:hi + 1] += 2.0 * a[j0:j1 + 1][::-1]
    g /= float(act.size)
    return g


def _lp_dir(a, n, M, tl=6.0):
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except ImportError:
        return None
    nn = 2 * n - 1
    if n > 5000:
        return None
    rp, cp, vp = [], [], []
    for j in range(n):
        kl = j
        kh = min(nn - 1, j + n - 1)
        ln = kh - kl + 1
        rp.append(np.arange(kl, kl + ln, dtype=np.int32))
        cp.append(np.full(ln, j, dtype=np.int32))
        vp.append(a[:ln])
    rows = np.concatenate(rp)
    cols = np.concatenate(cp)
    vals = np.concatenate(vp)
    del rp, cp, vp
    T = csr_matrix((vals, (rows, cols)), shape=(nn, n))
    del rows, cols, vals
    t0 = time.monotonic()
    rem = tl - (time.monotonic() - t0)
    if rem <= 0:
        del T
        return None
    res = linprog(-np.ones(n), A_ub=T, b_ub=np.full(nn, M),
                  method='highs',
                  options={'time_limit': rem, 'presolve': True})
    del T
    if not res.success:
        return None
    g = np.asarray(res.x, dtype=np.float64)
    gs = g.sum()
    if gs < 1e-12:
        return None
    return g / gs
