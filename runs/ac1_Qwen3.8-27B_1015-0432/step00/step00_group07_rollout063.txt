import numpy as np
import time

def _conv(a):
    n = len(a)
    # FFT-based full convolution
    m = 2 * n - 1
    fa = np.fft.rfft(a, 4 * n)  # zero-pad to avoid circular artifact
    c = np.fft.irfft(fa * fa)[:m]
    return c

def _score(a):
    n = len(a)
    c = _conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def _subgrad(a, k_top=5):
    n = len(a)
    c = _conv(a)
    m_max = c.max()
    # Find top-k positions
    # Use a threshold to include all near-max
    thresh = m_max * 0.999
    idx = np.where(c >= thresh)[0]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(c))])
    if len(idx) > k_top:
        # Take the top k
        vals = c[idx]
        top_idx = idx[np.argsort(vals)[-k_top:]]
        idx = top_idx
    grad = np.zeros(n)
    for m in idx:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            # grad[lo:hi+1] += 2 * a[m-hi : m-lo+1] reversed
            grad[lo:hi+1] += 2.0 * a[m-hi:m-lo+1][::-1]
    grad /= max(1, len(idx))
    return grad, c, m_max

    best_no_improve = 0

        grad, c, m_max = _subgrad(a, top_k)

        a_max = a.max()
        if a_max > 0:
