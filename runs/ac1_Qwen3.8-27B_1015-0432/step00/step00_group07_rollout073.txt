def _conv_fwd(conv, a, j, delta):
    n = len(a)
    lo = j
    hi = min(2*n - 1, j + n)
    conv[lo:hi] += 2.0 * delta * a[lo - j:hi - j]
    if 2*j < 2*n - 1:
        conv[2*j] += delta * delta

def _conv_bwd(conv, a, j, delta):
    n = len(a)
    lo = j
    hi = min(2*n - 1, j + n)
    conv[lo:hi] -= 2.0 * delta * a[lo - j:hi - j]
    if 2*j < 2*n - 1:
        conv[2*j] += delta * delta

def _score_from_conv(conv, S, n):
    return 2.0 * n * conv.max() / (S * S)

def _make_two_level(n, ratio, frac):
    m = int(n * frac)
    if m < 1:
        m = 1
    if m >= n:
        m = n - 1
    a = np.empty(n)
    a[:m] = ratio
    a[m:] = 1.0
    return a

    return np.exp(-alpha * (t - beta)**2)
