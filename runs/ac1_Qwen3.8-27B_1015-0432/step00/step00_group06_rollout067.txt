import numpy as np
import time


def _proj_simplex(v, S):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1.0) > (cssv - S)
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.full(n, S / n)
    rho = idx[-1]
    theta = (cssv[rho] - S) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smooth_grad(a, eps):
    n = a.size
    c = np.convolve(a, a)
    M = float(c.max())
    cs = c / eps - M / eps
    p = np.exp(cs)
    p /= p.sum()
    a_r = a[::-1]
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    P = np.fft.rfft(p, n=L)
    Ar = np.fft.rfft(a_r, n=L)
    h = np.fft.irfft(P * Ar, n=L)
    grad = (2.0 / eps) * h[n - 1:2 * n - 2]
    return grad, M


def _whiten(a, S):
    n = a.size
    c = np.convolve(a, a)
    M = float(c.max())
    L = 1
    while L < 3 * n - 1:
        L <<= 1
    A = np.fft.rfft(a, n=L)
    tgt = np.zeros(L)
    tgt[:2 * n - 1] = M
    T = np.fft.rfft(tgt, n=L)
    lam = 0.001 * M
    G = T / (A + lam)
    g = np.fft.irfft(G, n=L)[:n]
    g = np.maximum(g, 0.0)
    gs = g.sum()
    if gs < 1e-12:
        return np.full(n, S / n)
    return g * S / gs
