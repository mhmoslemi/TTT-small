import numpy as np

import time


def _fft_conv_len1(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A)[:2 * n - 1]


def _grad_at_lags(a, lags, weights, n):
    g = np.zeros(n)
    i = np.arange(n)
    for k, w in zip(lags, weights):
        r1 = k - i
        m1 = (r1 >= 0) & (r1 < n)
        if m1.any():
            g[m1] += w * a[r1[m1]]
        r2 = i - k
        m2 = (r2 >= 0) & (r2 < n)
        if m2.any():
            g[m2] += w * a[r2[m2]]
    g -= a.mean()
    return g
