def _project_simplex(v, s):

    if not np.all(np.isfinite(v)):
        v=np.where(np.isfinite(v), v, 0.0)
    u=np.sort(v)[::-1]
    css=np.cumsum(u)

def _pow2(m):

def _project_simplex(v, s):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    w = np.arange(1, n + 1, dtype=np.float64)
    cond = u > (css - s) / w
    idx = np.nonzero(cond)[0]
    rho = idx[-1] if idx.size else n - 1
    theta = (css[rho] - s) / float(rho + 1)
    p = np.maximum(v - theta, 0.0)
    for _ in range(3):
        p = np.clip(p, 0.0, 1000.0)
        ss = float(p.sum())
        if ss < 1e-12:
            p = np.full(n, s / float(n), dtype=np.float64)
            break
        scale = s / ss
        if abs(scale - 1.0) <= 1e-12:
            break
        p *= scale
    ss = float(p.sum())
    if ss < 1e-12:
        p = np.full(n, s / float(n), dtype=np.float64)
    else:
        p *= s / ss
    return np.clip(p, 0.0, 1000.0)


def _fft_data(a, N, L):
    A = np.fft.rfft(a, N)
    C = np.fft.irfft(A * A, N)[:L].real
    M = float(np.max(C)) if C.size else 0.0
    if not np.isfinite(M):
        M = 0.0
    return A, C, M


def _direct_score(a, n, N, L):
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    _, C, M = _fft_data(a, N, L)
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (S * S))


def _make_sinusoid(n, k, eps, phase):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / float(n)
    return 1.0 + eps * np.sin(2.0 * np.pi * k * t + phase)


def _make_linear(n, delta):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    t = np.arange(n, dtype=np.float64) / float(n - 1)
    return 1.0 + delta * (2.0 * t - 1.0)

def _gradient_active(x, C, n, tol, K):
    M = float(np.max(C))
    active = np.nonzero(C >= M - tol)[0]
    if active.size == 0:
        active = np.array([int(np.argmax(C))], dtype=np.int64)
    if active.size > K:
        return None
    g = np.zeros(n, dtype=np.float64)
    for m in active:
        m = int(m)
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi + 1] += x[m - hi:m - lo + 1][::-1]
    return (2.0 / float(active.size)) * g


def _gradient_softmax(x, A, C, n, N, L, scale):
    M = float(np.max(C))
    w = np.exp((C - M) / max(1e-12, scale))
    ws = float(w.sum())
    if not np.isfinite(ws) or ws <= 0.0:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(C))] = 1.0
    else:
        p = w / ws
    P = np.fft.rfft(p[::-1], N)
    G = np.fft.irfft(A * P, N)
    return 2.0 * G[L - n:L][::-1]

                    best_score = sx
                stagnant = 0
            elif stagnant > 140:
                rc, sc = _prepare(_random_candidate(rng, n), n), _direct_score(_prepare(_random_candidate(rng, n), n), n, N, L)
                sc2 = sc
                if np.isfinite(sc2) and sc2 < 1.10 * best_score:
                    x = rc
                else:
                    x = best.copy()
                A, C, M = _fft_data(x, N, L)
                stagnant = 0

    out = _prepare(best, n)

    if n <= 30000 and time.monotonic() < deadline:
        try:
            ev_out = float(evaluate_sequence(out.tolist()))
            if not np.isfinite(ev_out):
                out = np.ones(n, dtype=np.float64)
            else:
                par = _prepare(parent_arr, n)
                ev_par = float(evaluate_sequence(par.tolist()))
                if np.isfinite(ev_par) and ev_par < ev_out:
                    out = par
        except Exception:
            pass

    Sout = float(out.sum())
    if Sout < 0.01:
        out = np.ones(n, dtype=np.float64)

    return [float(v) for v in out]
