For the "state of the art" approach mentioned (LP-based), the idea is:
- Fix the current f and its max autocorrelation M
- Find the best g (maximizing sum(g)) subject to f*g <= M (element-wise) and g >= 0
- This LP is linear in g for fixed f
- Update f <- (1-t)*f + t*g

The LP: maximize sum(g_j) subject to (f*g)_k <= M for all k, g_j >= 0.

The constraint (f*g)_k = sum_j f_j * g_{k-j} <= M for k = 0, ..., 2n-2.

This is a linear program in g with 2n-1 constraints and n variables. The constraint matrix is the convolution matrix of f, which is a banded matrix (each row has at most n nonzeros, but in total it's a lower-triangular-ish structure).

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0
    
    # Initialize
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a.ndim != 1 or a.size == 0 or a.sum() < 0.01:
        a = np.ones(1024, dtype=np.float64)
    n = a.size
    
    best = a.astype(np.float64).tolist()
    best_score = float(evaluate_sequence(best))
    
    a = a / a.sum()
    eta = 0.12
    it = 0
    no_improve = 0
    check_every = 50
    
    while time.monotonic() < deadline:
        c = np.convolve(a, a)
        c_max = c.max()
        if c_max <= 1e-18:
            break
        
        # Smoothed subgradient
        w = c / c.sum()
        s = np.correlate(w, a, 'full')[:n]
        s_max = s.max()
        if s_max <= 1e-18:
            break
        
        # Multiplicative update
        a = a * (1.0 - eta * s / s_max)
        a = np.maximum(a, 0.0)
        ssum = a.sum()
        if ssum > 1e-15:
            a /= ssum
        else:
            a = np.ones(n) / n
        
        it += 1
        
        if it % check_every == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a.tolist()
                best_score = score
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= 8:
                eta = max(0.005, eta * 0.8)
                no_improve = 0
            
            # Occasional random perturbation
            if it % 2000 == 0:
                noise = np.exp(rng.normal(0, 0.05, n))
                a = a * noise
                a = np.maximum(a, 0)
                a /= a.sum()
    
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best = a.tolist()
    
    return best
