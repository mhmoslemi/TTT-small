import numpy as np

def fft_convolve(a):
    """Full linear convolution via FFT, O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:size]


def fast_score(a):
    """AC1 score using FFT convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * fft_convolve(a).max() / (s * s))


def make_gradient_step(a, p, alpha):
    """Reduce peak contributors at lag p while preserving sum."""
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    pj = p - np.arange(n)
    mask = (pj >= 0) & (pj < n)

        g[mask] = a[pj[mask]]
    gm = g - g.mean()
    peak = g.max()
    if peak <= 0:

    step = alpha * peak
    return a - step * gm
