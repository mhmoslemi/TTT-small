def _fft_conv_max(a):
    n = len(a)
    size = 1
    while size < 2 * n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    return float(np.max(b[:2 * n - 1]))
