def _lp_step(f, M, active_lags, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_lags, g >= 0."""
    K = len(active_lags)
    if K == 0:
        return None
    # Pre-compute row lengths
    lens = np.empty(K, dtype=np.int64)
    for i, k in enumerate(active_lags):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        lens[i] = jmax - jmin + 1
    total = int(lens.sum())
    if total == 0:
        return None
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    ptr = 0
    for i, k in enumerate(active_lags):
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        js = np.arange(jmin, jmax + 1)
        rows[ptr:ptr+len(js)] = i
        cols[ptr:ptr+len(js)] = js
        vals[ptr:ptr+len(js)] = f[k - js]

        g = _lp_direction(f, M, active, n)
        hit = False

        if g is not None and g.sum() > 1e-10:
            g = g * S / g.sum()
            for t in (0.4, 0.6, 0.8, 1.0, 0.2, 0.3, 0.5):
                fn = (1.0 - t) * f + t * g
                fn = np.clip(fn, 0.0, 1000.0)
                sc = float(evaluate_sequence(fn.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = fn.copy()
                    best_score = sc
                    f = fn.copy()
                    no_improve = 0
                    hit = True
                    break

        if not hit:
            no_improve += 1

        # Escape from plateaus via random block perturbations
        if no_improve >= 15:
            no_improve = 0
            improved_esc = False
            for _ in range(120):
                if time.monotonic() > deadline:
                    break
                ft = _block_perturb(best, rng, n)
                sc = float(evaluate_sequence(ft.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = ft.copy()
                    best_score = sc
                    f = ft.copy()
                    improved_esc = True
                    break
            if not improved_esc:
                f = best.copy()

    return [float(x) for x in best]
