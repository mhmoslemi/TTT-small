import numpy as np
import time


def _normalize_to_uniform(arr, n):
    s = np.sum(arr)
    if s < 0.01:
        return np.ones(n, dtype=np.float64)
    return np.clip(np.full(n, s / n, dtype=np.float64), 0.0, 1000.0)


def _try_perturb(base, rng, n):
    width = max(1, n // 200)
    start = int(rng.integers(0, max(1, n - width)))
    candidate = base.copy()
    candidate[start:start + width] *= float(np.exp(rng.normal(0.0, 0.04)))
