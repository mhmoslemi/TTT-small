            w = int(rng.integers(max(3, n // 600), n // 5))
            st = int(rng.integers(0, n - w))
            a[st:st + w] *= float(np.exp(rng.normal(0.0, step)))
        a = np.clip(a, 1e-6, 1000.0)
        sc = fast_score(a)
        if sc < best_sc:
            best = a
            best_sc = sc
            stall = 0
            step = max(0.005, step * 0.985)
        else:
            stall += 1
            if stall > 250:
                a = best.copy()
                m = max(5, n // 15)
                idx = rng.choice(n, m, replace=False)
                a[idx] *= np.exp(rng.normal(0.0, 0.25, m))
                a = np.clip(a, 1e-6, 1000.0)
                sc = fast_score(a)
                if sc < best_sc:
                    best = a
                    best_sc = sc
                    stall = 0
                    step = 0.04
                else:
                    stall = 0
                    step = min(0.12, step * 1.04)
        if it % 20000 == 0:
            vsc = float(evaluate_sequence(best.tolist()))
            if np.isfinite(vsc):
                best_sc = vsc
            if time.monotonic() > t_end - 10.0:
                break

    # Final verification with exact evaluator
    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        return [1.0 / n] * n
    return [float(x) for x in best]
