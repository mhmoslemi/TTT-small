import numpy as np
import time
import math


def _grad_at_lag(a, m):
    """Gradient of c[m]=convolve(a,a)[m] w.r.t. a. O(n)."""
    n = a.size
    lo = max(0, m - n + 1)
    hi = min(n - 1, m)
    if lo > hi:
        return np.zeros(n)
    g = np.zeros(n)
    s = m - hi
    g[lo:hi + 1] = 2.0 * a[s:s + hi - lo + 1][::-1]
    return g


def _subgrad_direction(a, tol=0.999):
    """Return averaged gradient over active lags, and the max value."""
    c = np.convolve(a, a)
    M = float(c.max())
    if M <= 0:
        return np.zeros(a.size), 0.0
    active = c >= tol * M
    am = np.where(active)[0]
    g = np.zeros(a.size)
    for m in am:
        g += _grad_at_lag(a, int(m))
    g /= len(am)
    return g, M
