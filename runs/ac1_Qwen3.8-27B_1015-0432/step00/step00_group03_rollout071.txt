    return rows_idx, cols_idx


def compute_score(f):
    """Compute AC1 score efficiently."""
    n = len(f)
    C = np.convolve(f, f)
    M = float(C.max())
    s = float(f.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * M / (s * s)


def solve_lp_for_g(f, M, rows_idx, cols_idx, time_limit):
    """Solve LP: max sum(g) s.t. convolve(f,g)[k] <= M for all k, g>=0."""
    n = len(f)
    f_safe = f + 1e-8  # avoid zero columns (unbounded LP)
    vals = np.tile(f_safe, n)
    A = scipy.sparse.csr_matrix((vals, (rows_idx, cols_idx)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': time_limit})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size

    # Build initial candidates
    inits = [parent.copy()]
    if n > 2:
        theta = 2.0 * np.pi * np.arange(n) / (n - 1)
        for amp in [1.0, 0.99, 0.7]:
            inits.append(1.0 + amp * np.sin(theta))
        inits.append(1.0 + 0.99 * np.cos(theta))
        # Multi-frequency
        inits.append(1.0 + 0.6 * np.sin(theta) + 0.3 * np.sin(2.0 * theta))
    inits.append(np.clip(parent * (1.0 + 0.15 * rng.standard_normal(n)), 0.0, 1000.0))

    # Select best initial
    best = inits[0].copy()
    best_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(best_score):
        best = np.ones(n)
        best_score = float(evaluate_sequence(best.tolist()))
    for c in inits[1:]:
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = c.copy()

    # Precompute LP matrix indices
    rows_idx, cols_idx = build_sparse_conv(n)

    # LP-based iterative refinement
    f = best.copy()
    t_blend = 0.25
    stall = 0

    while time.monotonic() < deadline:
        C = np.convolve(f, f)
        M = float(C.max())
        s_f = float(f.sum())
        if M < 1e-15 or s_f < 0.01:
            break

        remaining = deadline - time.monotonic()
        if remaining < 10.0:
            break
        lp_time = min(35.0, max(5.0, remaining * 0.6))

        g = solve_lp_for_g(f, M, rows_idx, cols_idx, lp_time)

        if g is None or g.sum() < 1e-12:
            # Fallback gradient step
            k_star = int(C.argmax())
            idx = k_star - np.arange(n)
            valid = (idx >= 0) & (idx < n)
            coup = np.zeros(n)
            coup[valid] = f[idx[valid]]
            f = f - 0.003 * coup
            f = np.maximum(f, 0.0)
            if f.sum() < 0.01:
                break
            stall += 1
            if stall > 15:
                break
            continue

        # Rescale g to match sum of f
        g = g * (s_f / g.sum())
        g = np.clip(g, 0.0, 1000.0)

        # Blend
        new_f = (1.0 - t_blend) * f + t_blend * g
        new_score = float(evaluate_sequence(new_f.tolist()))

        if np.isfinite(new_score) and new_score < best_score:
            best_score = new_score
            best = new_f.copy()
            f = new_f.copy()
            stall = 0
            t_blend = min(0.45, t_blend * 1.04)
        else:
            f = new_f.copy()
            t_blend = max(0.08, t_blend * 0.8)
            stall += 1
            if stall > 25:
                # Diversify restart
                m = int(rng.integers(1, max(2, n // 3)))
                amp = rng.uniform(0.5, 1.0)
                theta = 2.0 * np.pi * m * np.arange(n) / (n - 1)
                f = 1.0 + amp * np.sin(theta)
                f = np.clip(f, 0.0, 1000.0)
                stall = 0

    final = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final):
        best = parent
    return [float(x) for x in best]
