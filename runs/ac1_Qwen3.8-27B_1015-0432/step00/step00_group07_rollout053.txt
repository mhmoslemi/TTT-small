    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    except:
        parent = np.ones(256, dtype=np.float64)
    
    n = parent.size
    if n == 0 or not np.isfinite(parent).all():
        n = 256
        parent = np.ones(n, dtype=np.float64)
    
    # Initial candidates
    candidates = []
    candidates.append(parent.copy())
    candidates.append(np.ones(n, dtype=np.float64))
    
    # Periodic pattern: blocks of 1s and 0s
    for period in [2, 3, 4, 5, 7, 10]:
        pattern = np.zeros(n)
        for i in range((n // period) * period + 1):
            if i % period < (period + 1) // 2:
                pattern[i] = 1.0
        if pattern.sum() > 0:
            candidates.append(pattern.copy())
    
    # Tapered: high at edges
    x = np.linspace(0, 1, n)
    candidates.append(0.5 + 0.5 * np.cos(2 * np.pi * x))
    candidates.append(np.abs(x - 0.5) * 2)  # V-shaped
    
    # Evaluate all candidates, keep best
    best = None
    best_score = np.inf

        cand = np.clip(cand, 0.0, 1000.0)
        if cand.sum() < 0.01:
            continue

        if np.isfinite(score) and score < best_score:
            best_score = score
            best = cand.copy()
