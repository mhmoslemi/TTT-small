def fft_convolve(a):
    """Compute linear convolution using FFT. O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:

    fa = np.fft.rfft(a, size)
    fb = np.fft.irfft(fa * fa, size)
    return fb[:2*n-1]
