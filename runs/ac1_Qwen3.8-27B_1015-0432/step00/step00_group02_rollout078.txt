def _clean_candidate(cand):
    a = np.asarray(cand, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return None
    if not np.isfinite(a).all():
        return None
    return np.clip(a, 0.0, 1000.0)


def _consider(candidate, best, best_score, deadline):
    if time.monotonic() >= deadline:
        return best, best_score
    a = _clean_candidate(candidate)
    if a is None:
        return best, best_score
    try:
        sc = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best, best_score
    if np.isfinite(sc) and sc < best_score:
        return a, sc
    return best, best_score


def _rescale_sum(cand, n):
    s = float(np.sum(cand))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64)
    cand = cand * (float(n) / s)
    cand = np.clip(cand, 0.0, 1000.0)
    if float(np.sum(cand)) <= 0.0:
        return np.ones(n, dtype=np.float64)
    return cand


def _perturbed_candidate(n, idx, amp, rng, mode):
    if n < 2:
        return np.ones(n, dtype=np.float64)
    amp = float(amp)
    if not np.isfinite(amp) or amp <= 0.0:
        return np.ones(n, dtype=np.float64)
    mode = int(mode) % 3

    if mode == 0:
        m = int(rng.integers(1, n // 2 + 1))
        phase = float(rng.random() * (2.0 * np.pi))
        x = np.cos((2.0 * np.pi * float(m) / float(n)) * idx + phase)
        cand = 1.0 + amp * (x - float(np.mean(x)))
    elif mode == 1:
        x = rng.standard_normal(n)
        cand = 1.0 + amp * (x - float(np.mean(x)))
    else:
        lo = max(1, n // 2000)
        hi = max(lo + 1, n // 50)
        if hi > n:
            hi = n
        width = int(rng.integers(lo, hi + 1))
        span = max(1, n - width)
        start = int(rng.integers(0, span))
        factor = float(np.exp(amp * float(rng.standard_normal()) * 3.0))
        cand = np.ones(n, dtype=np.float64)
        cand[start:start + width] *= factor
        if rng.random() < 0.5:
            half = (n + 1) // 2
            cand[:half] = 0.5 * (cand[:half] + cand[n - 1:n - half - 1:-1])

    cand = np.clip(cand, 0.0, 1000.0)
    return _rescale_sum(cand, n)


def _scale_candidates(n):

            best_score = sc
        else:
            return [float(x) for x in uniform]

        attempts += 1

    return [float(x) for x in out]
