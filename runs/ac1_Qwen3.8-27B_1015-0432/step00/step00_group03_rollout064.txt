def _gradient_of_max_conv(f, ff, max_active=40):
    n = len(f)
    M = ff.max()
    tol = 0.002 * M
    active = np.where(ff >= M - tol)[0]
    if len(active) > max_active:
        active = np.argpartition(ff, -max_active)[-max_active:]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        lo = k - j_hi
        hi = k - j_lo
        grad[j_lo:j_hi + 1] += 2.0 * f[lo:hi + 1][::-1]
    return grad, M

    v_hat = v / (1.0 - b2 ** t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, step
