import numpy as np
import time

def compute_conv_fft(a):
    """Compute full linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for FFT
    fsize = 1
    while fsize < 2 * n:
        fsize *= 2
    fa = np.fft.rfft(a, fsize)
    C = np.fft.irfft(fa * fa, fsize)[:2 * n - 1]
    return C

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    n = len(height_sequence_1)
    a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # Try a structured initial perturbation
    # Raised cosine taper
    if n > 10:
        t = np.arange(n) / (n - 1)
        taper = 0.3 + 0.7 * np.cos(np.pi * (t - 0.5)) ** 2  # or some other shape
        # Actually let's use a smooth bump
        taper = 0.5 + 0.5 * np.cos(2 * np.pi * t)  # [1, 0, 1] shape
        # Or: high in middle
        taper = 0.5 - 0.5 * np.cos(2 * np.pi * t)  # [0, 1, 0] shape
        
        a_test = a * taper
        a_test = np.clip(a_test, 0.0, 1000.0)
        s = float(evaluate_sequence(a_test.tolist()))
        if np.isfinite(s) and s < best_score:
            best = a_test
            best_score = s
            a = a_test
    
    # Main optimization loop
    eta = 0.05
    stall = 0
    
    while time.monotonic() < deadline:
        if stall > 1000:
            # Random restart from best
            a = best * (1.0 + 0.1 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            stall = 0
            eta = 0.05
            continue
        
        # Compute convolution (using FFT for speed)
        C = compute_conv_fft(a)
        k_star = int(np.argmax(C))
        M = C[k_star]
        sum_a = np.sum(a)
        
        if M < 1e-15:
            break
