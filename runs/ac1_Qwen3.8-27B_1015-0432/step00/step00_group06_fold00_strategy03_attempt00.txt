        def crossover_uniform(parent1, parent2, n):
            # We do uniform crossover: for each gene, with 50% probability take from parent1, else parent2.
            child1 = np.zeros(n, dtype=np.float64)
            child2 = np.zeros(n, dtype=np.float64)
            for i in range(n):
                if np.random.rand() < 0.5:
                    child1[i] = parent1[i]
                    child2[i] = parent2[i]
                else:
                    child1[i] = parent2[i]
                    child2[i] = parent1[i]
            return child1, child2

        def mutate_individual(individual, n, mutation_rate=0.01):
            if np.random.rand() < mutation_rate:
                # Choose a block to mutate: say 10 consecutive indices.
                idx = np.random.randint(0, n-10)
                # Generate new values: we'll replace it with a random array in [0,1000]
                block_length = 10
                individual[idx:idx+block_length] = np.random.uniform(0, 1000, block_length)
            return individual
