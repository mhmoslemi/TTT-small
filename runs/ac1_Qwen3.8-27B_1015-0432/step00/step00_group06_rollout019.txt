import time

def fast_convolve(a):
    """Compute auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2 for efficient FFT
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    C = np.fft.irfft(A * A, fft_size)
    return C[:size]

def compute_score(a):
    """Compute AC1 score using FFT (for speed during search)."""
    n = len(a)

    max_c = np.max(c)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_c / (s * s)
