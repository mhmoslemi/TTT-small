def compute_conv_max_and_grad(a):
    """Compute (a*a), its max, argmax set, and gradient of the score."""
    n = len(a)
    conv = np.convolve(a, a)
    max_val = conv.max()
    # Find all indices where conv equals max (within tolerance)
    K0 = np.where(np.abs(conv - max_val) < 1e-10)[0]
    S = a.sum()
    # Gradient of M = max(conv)
    # dM/da_j = (1/|K0|) * sum_{k in K0} 2*a_{k-j} * [0<=k-j<n]
    dM = np.zeros(n)
    for k in K0:
        # a_{k-j} for j=0..n-1, valid when 0<=k-j<n i.e. k-n+1<=j<=k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        dM[lo:hi+1] += 2 * a[k - (np.arange(lo, hi+1))]

import numpy as np
import time

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        n = v.size
        return np.full(n, 1.0/n)
    return v / s

def _make_init(n, rng, mode):
    """Create a non-uniform non-negative initialization, normalized to sum=1."""
    if mode == 0:
        # Cosine modulation
        t = np.arange(n)
        a = 1.0 + 0.4 * np.cos(2.0 * np.pi * t / n)
    elif mode == 1:
        # Sine modulation
        t = np.arange(n)
        a = 1.0 + 0.4 * np.sin(2.0 * np.pi * t / n)
    elif mode == 2:
        # Random smooth
        a = 1.0 + 0.3 * rng.standard_normal(n)
        # Smooth it
        for _ in range(5):
            a = np.concatenate([a, a[:1]]) * 0.5 + np.concatenate([a[1:], a[-1:]]) * 0.5
    elif mode == 3:
        # Two-scale: periodic blocks
        p = max(2, n // 40)
        t = np.arange(n) % p
        a = 1.0 + 0.5 * np.cos(2.0 * np.pi * t / p)
    else:
        # Raised cosine window
        t = np.arange(n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / n))
    a = np.maximum(a, 0.01)  # avoid exact zeros
    a /= a.sum()
    return a

def _pgd_step(a, lr, rng):
    """One projected gradient step to minimize 2n*max(a*a) on simplex."""
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    # Find active set (indices where conv ≈ max)
    K0 = np.where(conv >= max_val - 1e-12)[0]
    # Gradient of max(a*a):
    # d/da_j of (a*a)[k] = 2*a[k-j] for 0<=k-j<n
    dM = np.zeros(n)
    for k in K0:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # a[k-j] for j in [lo, hi] means a[k-hi..k-lo]
        dM[lo:hi+1] += 2.0 * a[k - np.arange(lo, hi + 1)]
    dM /= len(K0)
    # Gradient of g = 2n*max(a*a) (ignoring the constraint for now)
    grad = 2.0 * n * dM
    # Step
    a_new = a - lr * grad
    a_new = _project_simplex(a_new)
    return a_new, max_val
