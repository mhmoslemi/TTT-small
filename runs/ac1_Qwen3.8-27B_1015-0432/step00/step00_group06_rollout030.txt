import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _subgrad(a, k):
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] = 2.0 * a[k - np.arange(lo, hi + 1)]
    return g


def _normalize(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        a[:] = 1.0
        s = a.size
    return a / s
