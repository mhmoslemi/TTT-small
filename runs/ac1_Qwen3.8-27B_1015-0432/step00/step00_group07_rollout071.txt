def _score(a, n):
    c = np.convolve(a, a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * c.max() / (s * s))


def _make_periodic(n, period, duty):
    a = np.zeros(n)
    seg = np.zeros(period)
    seg[:duty] = 1.0
    reps = n // period + 1
    a = np.tile(seg, reps)[:n]
    return a


def _make_triangle_taper(n):
    x = np.linspace(0.0, 1.0, n)
    return np.abs(x - 0.5) * 2.0


def _make_cosine_taper(n):
    x = np.linspace(0.0, 1.0, n)
    return 0.5 * (1.0 + np.cos(np.pi * x))


def _block_perturb(base, rng, step):
    n = base.size
    cand = base.copy()
    width = max(1, int(rng.integers(1, max(2, n // 10))))
    start = int(rng.integers(0, max(1, n - width)))
    stop = min(n, start + width)
    kind = rng.random()
    if kind < 0.55:
        cand[start:stop] *= float(np.exp(rng.normal(0.0, step)))
    elif kind < 0.8:
        cand[start:stop] += float(rng.normal(0.0, step * base[start:stop].mean()))
    else:
        cand[start:stop] = 0.0
    return np.clip(cand, 0.0, 1000.0)


def _gradient_perturb(base, rng, n):
    cand = base.copy()

    if total > 1e-12:

        scale = float(rng.uniform(0.7, 0.95))
        cand[idx] *= scale
        cand[partners] *= scale
    return np.clip(cand, 0.0, 1000.0)
