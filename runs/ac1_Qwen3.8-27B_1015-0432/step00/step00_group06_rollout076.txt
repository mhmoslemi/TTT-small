def peak_penalty(x, idx, w):
    n=x.size
    p=np.zeros(n,dtype=np.float64)
    for kk, ww in zip(idx,w):
        if ww <= 0: continue
        k=int(kk)
        lo=k-n+1
        if lo<0: lo=0
        hi=k
        if hi>=n: hi=n-1

    return p

def pick_peaks_exp(c, top_k, exponent):
    L = c.size
    top_k = int(max(1, min(int(top_k), L)))
    if top_k >= L:
        idx = np.arange(L)
    else:

    vals = c[idx].astype(np.float64, copy=False)

    if ws <= 1e-300:

    else:
        w = w / ws

def update_pop(pop, score, vec, max_pop):
    if not np.isfinite(score) or len(vec)==0:
        return pop
    if len(pop) < max_pop:
        pop.append((float(score), vec.copy()))
        return pop
    worst=0
    for i in range(1, len(pop)):
        if pop[i][0] > pop[worst][0]:
            worst=i
    if float(score) < pop[worst][0]:
        pop[worst] = (float(score), vec.copy())
    return pop

def sanitize(a):

def conv_score(a, size):
    n = int(a.size)
    s = float(a.sum())
    if s <= 1e-300:
        return np.inf, None
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    L = 2 * n - 1
    c = c[:L]
    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (s * s)), c


def cross_conv(a, b, size):
    n = int(a.size)

def pick_peaks_exp(c, top_k, exponent):

def shape_beta(x, a, b):
    return np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0) + 1e-300

def shape_pow_tri(x, c, p, q):
    left = x <= c
