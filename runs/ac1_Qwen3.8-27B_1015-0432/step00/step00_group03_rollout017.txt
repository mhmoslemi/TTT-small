import numpy as np
import time

def _npo2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_conv(x, y):
    n = len(x) + len(y) - 1
    N = _npo2(n)
    return np.fft.irfft(np.fft.rfft(x, N) * np.fft.rfft(y, N), N)[:n]

def _grad_smoothed_max(a, T, N):
    """Gradient of T*logsumexp(autocorr/T) w.r.t. a, using precomputed N."""
    n = len(a)
    c = _fft_conv(a, a)
    cmax = c.max()
    exps = np.exp((c - cmax) / T)
    p = exps / exps.sum()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    V = np.fft.irfft(A * np.conj(P), N)
    g = np.empty(n)
    g[0] = V[0]
    if n > 1:
        g[1:] = V[N-1:N-n:-1]
    return g / T, p, cmax

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    deadline = time.monotonic() + budget_s - 3.0
    rng = np.random.default_rng(seed)
    n = len(height_sequence_1)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.size == 0 or not np.all(np.isfinite(parent)):
        return [1.0]
    nfft = _npo2(4 * n)
    
    # Build initializations
    idx = np.arange(n, dtype=np.float64)
    hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * idx / max(1, n - 1)))
    
    inits = [parent.copy(), hann.copy()]
    for k in range(4):
        inits.append(hann * (0.5 + 0.5 * rng.random(n)))
    # Bimodal
    center = n / 2.0
    sigma = n / 6.0
    bim = np.exp(-0.5 * ((idx - center + n / 3.0) / sigma) ** 2) + np.exp(-0.5 * ((idx - center - n / 3.0) / sigma) ** 2)
    inits.append(bim.copy())
    
    best = parent.copy()
    best_sc = float(evaluate_sequence(best.tolist()))
    
    for init in inits:
        a = np.clip(init, 0.0, 1000.0)
        if a.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = a.copy()
    
    # Gradient descent
    a = best.copy()
    S0 = a.sum()
    c_init = _fft_conv(a, a)
    T = max(1e-6, c_init.max() * 0.2)
    lr = S0 / n * 0.5
    v = np.zeros(n)
    mu = 0.9
    step = 0
    stale = 0
    eval_every = max(1, 5000 // n * 5)
    
    while time.monotonic() < deadline:
        g, p, cmax = _grad_smoothed_max(a, T, nfft)
        v = mu * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        a = a_new
        step += 1
        
        if step % eval_every == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = a.copy()
                stale = 0
            else:
                stale += 1
                lr *= 0.85
                if stale >= 40:
                    a = np.maximum(best * (1.0 + 0.1 * rng.standard_normal(n)), 0.0)
                    v[:] = 0.0
                    lr = a.sum() / n * 0.5
                    T = max(1e-6, _fft_conv(a, a).max() * 0.2)
                    stale = 0
            T = max(c_init.max() * 0.005, T * 0.995)
        
        if time.monotonic() > deadline - 5.0:
            break
