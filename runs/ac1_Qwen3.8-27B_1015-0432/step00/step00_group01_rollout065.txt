import numpy as np
import time


def make_power_ramp(n, p, peak=1000.0):
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx / n) ** p
    if a.size and a.max() > 0:
        a = a / a.max() * peak
    return a


def block_perturb(base, rng):
    out = base.copy()
    n = out.size
    width = int(rng.integers(max(1, n // 400), max(2, n // 20)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    mode = rng.random()
    if mode < 0.5:
        out[start:stop] *= float(np.exp(rng.normal(0.0, 0.25)))
    elif mode < 0.75:
        out[start:stop] += float(rng.uniform(-1.0, 1.0)) * out[start:stop].mean()
    else:
        lo = out[start:stop].min()
        hi = out[start:stop].max()
        out[start:stop] = (lo + hi) / 2.0 + rng.uniform(-0.6, 0.6) * max(1e-9, hi - lo)
    return np.clip(out, 0.0, 1000.0)


def fast_score(a):
    n = a.size
    if n == 0:
        return np.inf
    conv = np.fft.irfft(np.fft.rfft(a) ** 2, 2 * n - 1)
    mb = float(np.max(conv))
    sa = float(np.sum(a))
    if sa < 0.01:
        return np.inf
    return 2.0 * n * mb / (sa * sa)
