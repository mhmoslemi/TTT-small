def _make_symmetric(half, n):
    """Build a palindromic sequence from its first half."""
    if n % 2 == 0:
        return np.concatenate([half, half[::-1]])
    else:
        return np.concatenate([half, half[-2::-1]])


def _parametric_shapes(n, rng):
    """Generate a list of diverse parametric initializations."""
    t = np.linspace(0.0, 1.0, n)
    u = 2.0 * t - 1.0  # u in [-1, 1]
    shapes = []
    shapes.append(np.ones(n))
    shapes.append(1.0 - u * u)
    shapes.append(np.clip(1.0 - np.abs(u), 0.0, 1.0))
    shapes.append(np.clip(1.0 - np.abs(u), 0.0, 1.0) ** 2)
    shapes.append(np.clip(1.0 - np.abs(u), 0.0, 1.0) ** 3)
    shapes.append(np.exp(-4.0 * u * u))
    shapes.append(np.exp(-9.0 * u * u))
    shapes.append(np.exp(-16.0 * u * u))
    shapes.append(((1.0 + np.cos(np.pi * u)) / 2.0) ** 2)
    shapes.append(((1.0 + np.cos(np.pi * u)) / 2.0) ** 4)
    shapes.append(np.clip(1.0 - u ** 4, 0.0, 1.0))
    # Stochastic smooth shapes

        s = np.clip(1.0 - np.abs(u) ** p, 0.0, 1.0)
        shapes.append(s * (0.5 + 0.5 * np.cos(a1 * np.pi * u)))
    # Ensure non-negative
    return [np.clip(s, 1e-8, 1000.0) for s in shapes]
