import numpy as np
import time

TARGET = 1.50274349


def _pow2(m):
    x = 1
    while x < m:
        x <<= 1
    return x


def _norm(p, eps=1e-12):
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, eps)
    s = p.sum()
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(p.size, dtype=np.float64) / max(1, p.size)
    return p / s


def _tolist(p):
    out = []
    for x in p:
        out.append(float(x))
    return out


def _score_fft(p, n, L):
    full = 2 * n - 1
    s = p.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.inf
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:full]
    mc = float(np.max(c))
    if mc < 0.0:
        mc = 0.0
    return float(2.0 * n * mc / (s * s))


def _make_basis(n, d):
    T = np.zeros((d, n), dtype=np.float64)
    if d <= 0:
        return T
    x = np.linspace(-1.0, 1.0, n)
    prev = np.ones(n, dtype=np.float64)
    curr = x.copy()
    idx = 0
    M = 2 * d - 1
    for m in range(1, M + 1):
        if (m & 1) and idx < d:
            T[idx] = curr
            idx += 1
        if m < M:
            nxt = 2.0 * x * curr - prev
            prev, curr = curr, nxt
    return T


def _p_from_theta(theta, basis, n):
    if theta.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    b = basis @ theta
    p = 1.0 + np.tanh(b)
    return _norm(p)


def _p_from_exp(theta, basis, n):
    if theta.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    b = basis @ theta
    b = np.clip(b, -4.0, 4.0)
    p = np.exp(b)
    return _norm(p)


def _add_candidate(top, score, p, theta, K):
    if not np.isfinite(score):
        return
    item = (
        float(score),
        np.asarray(p, dtype=np.float64).copy(),
        np.asarray(theta, dtype=np.float64).copy() if theta is not None else None
    )
    if len(top) < K:
        top.append(item)
        return
    worst = 0
    worst_sc = float(top[0][0])

        sc = float(top[i][0])
        if sc > worst_sc:
            worst_sc = sc
            worst = i

        top[worst] = item

    return arr


def _mirror_step_score(p, n, L, tau, beta):
    full = 2 * n - 1
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:full]
    z = float(n) * c
    zmax = float(np.max(z))
    if not np.isfinite(zmax) or zmax <= 0.0:
        return p, zmax

    arg = (z - zmax) / tau
    arg = np.clip(arg, -700.0, 0.0)
    w = np.exp(arg)
    ws = w.sum()
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.zeros(full, dtype=np.float64)
        w[int(np.argmax(z))] = 1.0
    else:
        w /= ws

    wr = w[::-1]
    W = np.fft.rfft(wr, L)

    c2 = np.fft.irfft(F2 * F2, L)[:full]
    z2 = float(float(n) * np.max(c2))
    return q, z2

def _lag_vector(p, k):
    n = p.size
    h = np.zeros(n, dtype=np.float64)
    start = max(0, k - (n - 1))
    stop = min(n, k + 1)
    if stop > start:
        j0 = k - (stop - 1)
        j1 = k - start
        h[start:stop] = p[j0:j1 + 1][::-1]
    return h

                break

        if not improved:
            got = False
            for kk in lags:
                if time.monotonic() > deadline - 1.0:
                    break
                h1 = _lag_vector(p, int(kk))
                h1m = float(np.max(h1))
                if not np.isfinite(h1m) or h1m <= 0.0:
                    continue
                for eta in (0.03, 0.015, 0.08):
                    if time.monotonic() > deadline - 1.0:
                        break
                    q = p * np.exp(-eta * h1 / h1m)
                    q = _norm(q)
                    qs = evaluate_sequence(q.tolist())
                    if np.isfinite(qs) and qs < sc:
                        p = q
                        sc = qs
                        improved = True
                        got = True
                        break
                if improved:
                    break
            if not improved:
                break

        if sc <= TARGET:
            break
    return p, sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    if parent.size == 0:
        parent = np.ones(256, dtype=np.float64)

    n = int(parent.size)
    L = _pow2(3 * n - 2)
    parent_p = _norm(parent.copy())

    best_p = parent_p.copy()
    best_score = _score_fft(best_p, n, L)
    if not np.isfinite(best_score):
        best_score = np.inf

    top = []
    _add_candidate(top, best_score, best_p, None, 12)

    d = min(40, max(12, n // 150))
    basis = _make_basis(n, d)
    rng = np.random.default_rng(seed)

    # Structured one-mode starts.
    modes = min(d, 14)
    amps = (0.2, 0.4, 0.7, 1.1, 1.6, 2.3, 3.2)
    for j in range(modes):
        for amp in amps:
            for sgn in (1.0, -1.0):
                if time.monotonic() > deadline:
                    break
                th = np.zeros(d, dtype=np.float64)
                th[j] = sgn * amp
                p = _p_from_theta(th, basis, n)
                sc = _score_fft(p, n, L)
                if np.isfinite(sc):
                    _add_candidate(top, sc, p, th, 12)
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()

    # Random parameterized candidates.
    rand_deadline = min(deadline, start + 130.0)
    for it in range(18000):
        if time.monotonic() > rand_deadline:
            break
        scale = 0.30 + 1.70 * min(1.0, float(it) / 2500.0)
        th = rng.normal(size=d) * scale
        if rng.random() < 0.30:
            mask = rng.random(d) < 0.5
            th[~mask] = 0.0
        if rng.random() < 0.35:
            p = _p_from_exp(th, basis, n)
        else:
            p = _p_from_theta(th, basis, n)
        sc = _score_fft(p, n, L)
        if np.isfinite(sc):
            _add_candidate(top, sc, p, th, 12)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()

    # Coordinate refinement in the smooth basis.
    local_deadline = min(deadline, start + 350.0)
    top_sorted = _sort_top(top)
    tried_theta = 0
    for _, _, th in top_sorted:
        if tried_theta >= 4 or time.monotonic() > local_deadline:
            break
        if th is None:
            continue
        p, sc, th = _theta_local(th, basis, n, L, rng, local_deadline)
        if np.isfinite(sc):
            _add_candidate(top, sc, p, th, 12)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
        tried_theta += 1

    # Mirror-descent refinement on all promising sequences.
    top_sorted = _sort_top(top)
    for _, p0, _ in top_sorted[:6]:
        for tau in (0.25, 0.15, 0.09, 0.055, 0.032, 0.020, 0.012):
            if time.monotonic() > deadline - 25.0:
                break
            p1, sc = _mirror_iterate(p0, n, L, tau, min(deadline - 25.0, time.monotonic() + 120.0))
            if np.isfinite(sc):
                _add_candidate(top, sc, p1, None, 12)
                if sc < best_score:
                    best_score = sc
                    best_p = p1.copy()
            if best_score <= TARGET:
                break

    # Final exact polish.
    final_best = np.inf
    final_candidates = []
    final_candidates.append(best_p)
    top_sorted = _sort_top(top)
    for _, p1, _ in top_sorted[:4]:
        final_candidates.append(p1)
    final_candidates.append(parent_p)

    for p1 in final_candidates:
        if time.monotonic() > deadline - 2.0:
            break
        p2, sc = _exact_polish(p1, deadline - 2.0, rounds=28)
        if np.isfinite(sc) and sc < final_best:
            final_best = sc
            best_p = p2.copy()
            if sc <= TARGET:
                break

    # Authoritative final check and safe fallback.
    final_sc = evaluate_sequence(best_p.tolist())
    if not np.isfinite(final_sc):
        final_sc = evaluate_sequence(parent_p.tolist())
    if not np.isfinite(final_sc):
        return _tolist(np.ones(n, dtype=np.float64))

    return _tolist(best_p)
