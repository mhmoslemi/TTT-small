import numpy as np
import time

def fft_convolve_square(a):
    """Compute a*b where b=a (full convolution) using FFT. Returns array of length 2*len(a)-1."""
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]

def softmax(x, beta):
    """Stable softmax with temperature 1/beta applied."""
    x = beta * x
    x -= np.max(x)
    e = np.exp(x)
    return e / np.sum(e)

def correlate_valid(W, P, N):
    """Compute R[i] = sum_j W[i+j]*P[j] for i=0..len(P)-1 using FFT."""
    # W has length 2n-1, P has length n
    # Need N >= len(W) + len(P) - 1 = 3n - 2
    corr = np.fft.irfft(np.fft.rfft(W, N) * np.conj(np.fft.rfft(P, N)), N)
    n = len(P)
    return corr[:n]

def ac1_of_p(p):
    """Compute 2n * max(conv(p,p)) for probability vector p."""
    n = len(p)
    b = fft_convolve_square(p)
    return 2.0 * n * np.max(b)

    N_conv = 1
    while N_conv < 2 * n - 1:

    N_corr = 1

    lr = 1.0
