import time

def make_raised_cosine(n):
    """Raised cosine bump, normalized to sum=1."""
    x = np.arange(n)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    return a / a.sum()

def make_plateau(n, frac=0.8):
    """Plateau with cosine edges."""
    x = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    half_width = n * frac / 2.0
    w = n * (1 - frac) / 2.0
    a = np.ones(n)
    # Left edge
    left_mask = x < center - half_width
    if np.any(left_mask):
        t = (x[left_mask] - (center - half_width - w)) / w
        a[left_mask] = 0.5 * (1 + np.cos(np.pi * t))
    # Right edge
    right_mask = x > center + half_width
    if np.any(right_mask):
        t = ((center + half_width + w) - x[right_mask]) / w
        a[right_mask] = 0.5 * (1 + np.cos(np.pi * t))
    return a / a.sum()

def make_block_perturbation(base, rng, block_frac=0.05):
    """Perturb a random block."""
    n = len(base)
    a = base.copy()
    width = max(1, int(n * block_frac * (0.5 + rng.random())))
    start = int(rng.integers(0, n - width))
    # Multiply by a random factor
    factor = np.exp(rng.normal(0, 0.3, size=width))
    a[start:start+width] *= factor
    np.clip(a, 0, 1000, out=a)
    return a

def gradient_step(a, alpha):
    n = len(a)
    # Compute convolution
    conv = np.convolve(a, a)  # length 2n-1
    # Find max
    M = conv.max()
    S = a.sum()
    # Find active lags
    active = np.where(conv >= M * (1 - 1e-10))[0]
    # Subgradient: average of gradients at active lags
    # (a*a)[k] derivative wrt a_j is 2*a_{k-j} if valid
    grad = np.zeros(n)
    for k in active:
        # (grad_k)_j = 2*a_{k-j} if 0 <= k-j < n
        j_start = max(0, k - (n-1))
        j_end = min(n-1, k)
        idx = k - np.arange(j_start, j_end+1)
        grad[j_start:j_end+1] += 2 * a[idx]
    grad /= len(active)
    # Gradient of F = 2nM/S^2
    # dF/da_j = (2n/S^2) * (grad_M)_j - (4nM/S^3)
    F = 2*n*M/S**2
    grad_F = (2*n/S**2) * grad - (4*n*M/S**3)
    # Projected gradient step
    a_new = a - alpha * grad_F
    np.clip(a_new, 0, 1000, out=a_new)
    return a_new
