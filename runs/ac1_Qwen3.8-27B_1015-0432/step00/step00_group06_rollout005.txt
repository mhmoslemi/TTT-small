        best_seq, best_score, bd = refine_alt(n, rng, hard_deadline, best_seq, best_score)

    s_final = score_seq(best_seq)
    if np.isfinite(s_final) and s_final < best_score:
        best_score = s_final
    if not np.isfinite(best_score) or not np.isfinite(best_seq).all():
        best_seq = build_alt(n, 0.0001)
        best_score = score_seq(best_seq)

    out = [float(max(0.0, min(1000.0, v))) for v in best_seq]
    if not out or any(not np.isfinite(x) for x in out):
        out = [1.0] * n
    return out
