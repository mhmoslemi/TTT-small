    A = np.fft.rfft(a, fft_size)
    B = np.fft.irfft(A * A, fft_size)
    return B[:size]

        sn = a.sum()

            a = a / sn

            lr *= 0.5
            continue

                best = eval_seq.copy()
                best_score = s
                patience = 0
            else:
                patience += 1
                if patience > max_patience:
                    break

                a = a + noise
                a = np.maximum(a, 0.0)
                sn = a.sum()
                if sn > 0:
                    a = a / sn
                patience = 0
                lr = max(1.0 / (n * 4), lr * 0.7)

    # Final return
    result = np.clip(best, 0.0, 1000.0)
    if result.sum() < 0.01:
        result = parent.copy()
    return [float(x) for x in result]

import numpy as np

def fft_convolve(a):
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eval_score(seq):
    return float(evaluate_sequence([float(x) for x in seq]))

    hi = min(n - 1, k_star)
    if hi >= lo:
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]
    return g
