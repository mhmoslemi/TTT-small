    n = len(a)
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:total]

import numpy as np
import time


def conv_full_fft(a):
    """Full convolution a*a via FFT, O(n log n)."""
    n = len(a)
    total = 2 * n - 1
    size = 1 << (total - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:total]


def compute_subgrad(c, f, n, top_peaks=3):
    """Subgradient direction from top peaks of c."""
    cmax = float(np.max(c))
    thresh = cmax - 0.15 * (cmax - float(np.mean(c)))
    peak_mask = c >= thresh
    n_peaks = int(np.sum(peak_mask))
    if n_peaks == 0:
        peak_mask = np.zeros(len(c), dtype=bool)
        peak_mask[int(np.argmax(c))] = True
    if n_peaks <= top_peaks:
        idx = np.where(peak_mask)[0]
    else:
        idx = np.where(peak_mask)[0]
        idx = idx[np.argsort(c[idx])[-top_peaks:]]
    grad = np.zeros(n)
    for K in idx:
        K = int(K)
        lo = max(0, K - n + 1)
        hi = min(n - 1, K)
        js = np.arange(lo, hi + 1)
        grad[js] += f[K - js]
    return grad
