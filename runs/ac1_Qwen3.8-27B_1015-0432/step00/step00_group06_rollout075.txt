```
def fft_full_conv(a):
    n = a.size
    m = 2*n - 1
    fsize = 1 << (m - 1).bit_length()
    fa = np.fft.rfft(a, n=fsize)
    return np.fft.irfft(fa * fa, n=fsize)[:m]
```

```
def fast_score(a):
    s = a.sum()
    if s <= 1e-12: return np.inf
    c = fft_full_conv(a)
    cmax = c.max()
    return 2.0 * a.size * cmax / (s*s)
```

```
def gradient_lags(a, lags):
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    # for each lag k, grad[j]+=a[k-j] if valid
    # can vectorize per k.
    for k in lags:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            # j in [lo, hi], a index = k - j, ranges k-hi to k-lo reversed
            grad[lo:hi+1] += a[k-hi : k-lo+1][::-1]
    return 2.0 * grad
```
This is O(K n) memory O(n). For K=4, okay.

```python
import numpy as np
import time

TARGET_SUM = 16.0
SCORE_DEN = TARGET_SUM * TARGET_SUM


def next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_raw(a, target=TARGET_SUM):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    if a.size == 0:
        a = np.ones(256, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1e6)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def quick_normalize(a, target=TARGET_SUM):
    a = np.asarray(a, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(a.size, target / max(1, a.size), dtype=np.float64)
    return a * (target / s)


def conv_max(a, fsize):
    n = a.size
    m = 2 * n - 1
    fa = np.fft.rfft(a, fsize)
    c = np.fft.irfft(fa * fa, fsize)
    mx = float(c[:m].max())
    if not np.isfinite(mx) or mx < 0.0:
        return 0.0
    return mx


def fast_score(a, fsize):
    return (2.0 * a.size * conv_max(a, fsize)) / SCORE_DEN


def score_and_grad(a, fsize, gamma):
    n = a.size
    m = 2 * n - 1
    fa = np.fft.rfft(a, fsize)
    c = np.fft.irfft(fa * fa, fsize)[:m]
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 0.0, np.zeros(n, dtype=np.float64)
    target = cmax * (1.0 - gamma)
    w = c - target
    w[w < 0.0] = 0.0
    if float(w.sum()) <= 1e-300:
        w.fill(0.0)
        w[int(np.argmax(c))] = 1.0
    fw = np.fft.rfft(w[::-1], fsize)
    gfull = np.fft.irfft(fa * fw, fsize)
    g = 2.0 * gfull[n - 1:2 * n - 1][::-1]
    g = np.maximum(g, 0.0)
    return cmax, g


def try_exact(a):
    try:
        val = evaluate_sequence(a.tolist())
    except Exception:
        val = np.inf
    if np.isfinite(val):
        return float(val)
    return float("inf")


def is_nearly_symmetric(a):
    return bool(np.allclose(a, a[::-1], rtol=1e-7, atol=1e-12))


def get_parent(n):
    fallback = np.full(n, TARGET_SUM / max(1, n), dtype=np.float64)
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        return fallback
    if p.ndim != 1:
        p = p.reshape(-1)
    if p.size != n:
        return fallback
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    return normalize_raw(p, TARGET_SUM)


def random_perturb(a, rng, n, strength):
    a2 = np.maximum(np.asarray(a, dtype=np.float64).copy(), 0.0)
    a2 += 1e-7 * TARGET_SUM / float(n)
    nb = max(8, min(64, n // 64))
    for b in range(nb):
        width = max(1, n // (nb * 3))
        start = int(rng.integers(0, max(1, n - width + 1)))
        a2[start:start + width] *= float(np.exp(rng.normal(0.0, strength)))
    freq = int(rng.integers(1, max(2, n // 50) + 1))
    phase = float(rng.uniform(0.0, 6.283185307179586))
    amp = float(rng.uniform(0.0, strength))
    if amp > 0.0:
        idx = np.arange(n)
        a2 *= np.exp(amp * np.cos((2.0 * 3.141592653589793 * freq * idx) / float(n) + phase))
    return quick_normalize(a2, TARGET_SUM)


def start_linear(n):
    x = np.arange(n, dtype=np.float64) + 1.0
    return quick_normalize(x, TARGET_SUM)


def start_pair_asym(n, rng):
    m = n // 2
    a = np.zeros(n, dtype=np.float64)
    if m > 0:
        side = rng.random(m) < 0.5
        w = rng.lognormal(0.0, 0.8, m)
        a[:m] = np.where(side, w, 0.0)
        idx = n - 1 - np.arange(m)
        a[idx] += np.where(~side, w, 0.0)
    if n % 2 == 1:
        a[n // 2] = rng.lognormal(0.0, 0.8)
    return quick_normalize(a, TARGET_SUM)


def start_smooth_asym(n, rng):
    idx = np.arange(n, dtype=np.float64)
    x = 1.0 + 0.6 * (idx / max(1.0, float(n - 1)))
    for r in range(6):
        f = int(rng.integers(1, max(2, n // 8) + 1))
        p = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(-0.6, 0.6))
        x += amp * np.cos((2.0 * np.pi * f * idx) / float(n) + p)
    x = np.clip(x, 0.0, None)
    if float(x.sum()) <= 1e-12:
        x = np.ones(n, dtype=np.float64)
    return quick_normalize(x, TARGET_SUM)


def start_gap(n, rng):
    a = np.ones(n, dtype=np.float64)
    ng = int(rng.integers(1, 4))
    for g in range(ng):
        q = int(n * float(rng.uniform(0.05, 0.25)))
        if q < 1:
            q = 1
        if q >= n:
            q = max(1, n // 2)
        s = int(rng.integers(0, max(1, n - q + 1)))
        a[s:s + q] = 0.0
    return quick_normalize(a, TARGET_SUM)


def optimize_trajectory(a0, stop_at, seed, score_fsize, grad_fsize):
    rng = np.random.default_rng(seed)
    n = int(a0.size)
    a = quick_normalize(np.asarray(a0, dtype=np.float64).copy(), TARGET_SUM)
    cur = fast_score(a, score_fsize)
    best_a = a.copy()
    best_score = cur
    eta = 0.15
    gamma = 0.15
    stall = 0
    it = 0

    while time.monotonic() < stop_at:
        cmax, g = score_and_grad(a, grad_fsize, gamma)
        cur = (2.0 * n * cmax) / SCORE_DEN
        if np.isfinite(cur) and cur < best_score - 1e-9:
            best_score = cur
            best_a = a.copy()
            stall = 0
            gamma = max(0.03, gamma * 0.92)
        else:
            stall += 1
            if stall > 40:
                gamma = min(0.35, gamma + 0.03)

        d = g - float(np.mean(g))
        sd = float(np.std(d))
        if not np.isfinite(sd) or sd < 1e-14:
            base = best_a if best_score <= cur else a
            a = random_perturb(base, rng, n, 0.7)
            cur = fast_score(a, score_fsize)
            it += 1
            continue

        u = d / sd
        best_cand = None
        best_cand_s = cur
        best_factor = 1.0

        for factor in (0.5, 1.0, 2.0):
            e = eta * factor
            logf = np.clip(-e * u, -50.0, 50.0)
            cand = quick_normalize(a * np.exp(logf), TARGET_SUM)
            cs = fast_score(cand, score_fsize)
            if cs < best_cand_s:
                best_cand_s = cs
                best_cand = cand
                best_factor = factor

        scale = max(float(np.std(a)), (TARGET_SUM / float(n)) * 0.5)
        lin = np.maximum(a - eta * scale * u, 0.0)
        cand = quick_normalize(lin, TARGET_SUM)
        cs = fast_score(cand, score_fsize)
        if cs < best_cand_s:
            best_cand_s = cs
            best_cand = cand
            best_factor = 1.0

        if best_cand is not None and best_cand_s < cur - 1e-12:
            a = best_cand
            cur = best_cand_s
            if best_factor > 1.5:
                eta = min(0.8, eta * 1.35)
            elif best_factor < 1.0:
                eta = max(0.03, eta * 0.85)
            else:
                eta = min(0.8, eta * 1.08)
            stall = 0
        else:
            eta = max(0.03, eta * 0.8)
            if stall > 80:
                base = best_a if rng.random() < 0.6 else a
                a = random_perturb(base, rng, n, 0.9)
                cur = fast_score(a, score_fsize)
                if cur < best_score:
                    best_score = cur
                    best_a = a.copy()
                eta = 0.15
                gamma = 0.18
                stall = 0
            elif it % 15 == 0:
                a2 = random_perturb(a, rng, n, 0.12)
                s2 = fast_score(a2, score_fsize)
                if s2 < cur:
                    a = a2
                    cur = s2
        it += 1

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    try:
        p_arr = np.asarray(height_sequence_1, dtype=np.float64)
        if p_arr.ndim != 1:
            p_arr = p_arr.reshape(-1)
        n = int(p_arr.size)
    except Exception:
        n = 256
    if n <= 0:
        n = 256

    m = 2 * n - 1
    score_fsize = next_pow2(m)
    grad_fsize = next_pow2(3 * n - 2)

    now = time.monotonic()
    budget = max(60.0, float(budget_s))
    hard_deadline = now + budget - 20.0
    search_deadline = hard_deadline - 40.0
    if search_deadline < now + 30.0:
        search_deadline = hard_deadline - 5.0

    parent = get_parent(n)
    global_best_a = parent.copy()
    global_best_score = try_exact(global_best_a)

    uniform = np.full(n, TARGET_SUM / max(1, n), dtype=np.float64)
    us = try_exact(uniform)
    if np.isfinite(us) and us < global_best_score:
        global_best_score = us
        global_best_a = uniform.copy()

    rng0 = np.random.default_rng(seed + 1)
    starts = []
    if not is_nearly_symmetric(parent):
        starts.append(parent)
    starts.append(random_perturb(parent, rng0, n, 0.8))
    starts.append(start_pair_asym(n, rng0))
    starts.append(start_linear(n))
    starts.append(start_smooth_asym(n, rng0))
    starts.append(start_gap(n, rng0))

    scored = []
    for i, a in enumerate(starts):
        if a.size != n:
            a = np.full(n, TARGET_SUM / max(1, n), dtype=np.float64)
        sc = fast_score(a, score_fsize)
        if not np.isfinite(sc):
            sc = 1e300
        scored.append((sc, i, a))
    scored.sort()

    max_traj = min(5, len(scored))
    for ti in range(max_traj):
        if time.monotonic() > search_deadline:
            break
        _, _, a0 = scored[ti]
        remaining = search_deadline - time.monotonic()
        if remaining < 90.0:
            break
        left = max_traj - ti
        slot = remaining / left * 0.8
        if slot < 90.0:
            slot = min(remaining, 90.0)
        if slot > 400.0:
            slot = 400.0
        stop = time.monotonic() + slot
        if stop > search_deadline:
            stop = search_deadline

        best_a, best_sc = optimize_trajectory(a0, stop, seed + 17 + ti, score_fsize, grad_fsize)
        if time.monotonic() > search_deadline:
            break
        esc = try_exact(best_a)
        if np.isfinite(esc) and esc < global_best_score:
            global_best_score = esc
            global_best_a = best_a.copy()

    out = np.asarray(global_best_a, dtype=np.float64)
    if out.size != n:
        out = np.full(n, TARGET_SUM / max(1, n), dtype=np.float64)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if not np.isfinite(s) or s < 0.01:
        out = np.full(n, TARGET_SUM / max(1, n), dtype=np.float64)
    return [float(x) for x in out]
```
