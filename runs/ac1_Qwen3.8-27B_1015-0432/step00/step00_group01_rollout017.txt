        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel().copy()
        parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        parent = np.ones(64, dtype=np.float64)

    n = parent.size
    if n < 4:
        parent = np.ones(128, dtype=np.float64)
        n = 128

    # Initial candidates
    candidates = [parent.copy()]
    # Constant sequence (achieves theoretical min of 2.0)
    c1 = np.ones(n, dtype=np.float64)
    candidates.append(c1)
    # Mean of parent
    pm = parent.mean()
    if pm > 0.01:
        candidates.append(np.full(n, pm, dtype=np.float64))
    # Symmetrized parent
    candidates.append(symmetrize(parent))
    # Smoothed parent
    candidates.append(smooth_step(parent))
    # Geometric taper
    tidx = np.arange(n, dtype=np.float64)
    taper = np.cos(np.pi * (tidx - (n - 1) / 2.0) / (n - 1 + 1e-9))
    candidates.append(np.clip(taper, 0.0, None))

    # Evaluate all initial candidates
    best = None
    best_score = np.inf
    for c in candidates:
        c = np.clip(c, 0.0, 1000.0)
        if c.sum() < 0.01:
            continue
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = c.copy()

    if best is None:
        best = np.ones(n, dtype=np.float64)
        best_score = 2.0

    # Local search
    stall = 0
    max_stall = 200
    while time.monotonic() < deadline and stall < max_stall:
        candidate = best.copy()
        mode = rng.random()

        if mode < 0.4:
            # Multiplicative block perturbation
            nblocks = int(rng.integers(1, max(2, n // 400)))
            for _ in range(nblocks):
                w = int(rng.integers(max(1, n // 200), max(2, n // 6)))
                st = int(rng.integers(0, n - w))
                f = float(np.exp(rng.normal(0.0, 0.08)))
                candidate[st:st + w] *= f
        elif mode < 0.6:
            # Symmetrize after perturbation
            w = int(rng.integers(max(1, n // 50), max(2, n // 4)))
            st = int(rng.integers(0, n - w))
            candidate[st:st + w] *= float(np.exp(rng.normal(0.0, 0.1)))
            candidate = symmetrize(candidate)
        elif mode < 0.8:
            # Smoothing pass
            candidate = smooth_step(best)
            candidate += rng.normal(0, 0.01 * max(1.0, candidate.mean()), size=n)
        else:
            # Blend with constant
            t = float(rng.uniform(0.05, 0.3))
            candidate = (1.0 - t) * best + t * np.full(n, best.mean(), dtype=np.float64)
            if best.mean() < 0.01:
                candidate = (1.0 - t) * best + t * np.ones(n, dtype=np.float64)

        candidate = np.clip(candidate, 0.0, 1000.0)
        if candidate.sum() < 0.01:
            stall += 1
            continue

        # Fast score check
        s = fast_score(candidate)
        if s < best_score - 1e-12:
            s_exact = float(evaluate_sequence(candidate.tolist()))
            if np.isfinite(s_exact) and s_exact < best_score:
                best = candidate.copy()
                best_score = s_exact
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        # Check deadline inside loop
        if time.monotonic() >= deadline:
            break

    # Final verification
    if time.monotonic() < deadline:
        final_score = float(evaluate_sequence(best.tolist()))
        if np.isfinite(final_score):
            best_score = min(best_score, final_score)

    return [float(x) for x in best]
