import numpy as np
import time

def compute_conv(a):
    """Compute linear convolution a*a via FFT. O(n log n)."""
    n = len(a)
    N = 2 * n - 1
    # Round up to next power of 2 for FFT efficiency
    Nfft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, Nfft)
    fc = fa * fa  # Product in frequency domain
    result = np.fft.irfft(fc, Nfft)[:N]
    return result

def subgradient(a, S):
    """Compute subgradient of max_{k in S} (a*a)_k w.r.t. a.
    sg[i] = 2 * sum_{k in S} a[k-i] (for valid k-i)
    This equals 2 * convolve(indicator_S, a) truncated to len(a).
    """
    n = len(a)
    N = 2 * n - 1
    Nfft = 1 << (N - 1).bit_length()
    # Indicator of S, zero-padded to N

    eS[S] = 1.0

    fa = np.fft.rfft(a, Nfft)
    sg_full = np.fft.irfft(feS * fa, Nfft)[:N]
    return 2.0 * sg_full[:n]
