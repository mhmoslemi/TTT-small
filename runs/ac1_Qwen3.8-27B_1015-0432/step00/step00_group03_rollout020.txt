import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

        n = 256

    f = f / f.sum()
    
    best = f.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    
    # FFT size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    
    lr = 0.05
    K = 5  # number of active lags to consider
    
    for it in range(2000000):
        if time.monotonic() > deadline:
            break
        
        # Full convolution via FFT
        F = np.fft.irfft(np.fft.rfft(f, N) ** 2)[:2 * n - 1]
        
        # Top-K lags
        if K < 2 * n - 1:
            top_idx = np.argpartition(F, -K)[-K:]
        else:
            top_idx = np.arange(2 * n - 1)

        # Soft weights
        beta = 100.0 / max(M, 1e-15)
        weights = np.exp(beta * (F[top_idx] - M))
        wsum = weights.sum()
        if wsum > 0:
            weights /= wsum
        else:
            weights = np.ones_like(top_idx) / len(top_idx)
        
        # Gradient: weighted sum of circular-shifted f (clipped to linear)
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if weights[i] < 1e-6:
                continue

        f *= np.exp(-lr * grad)
        S = f.sum()
        if S < 1e-15:
            f = np.ones(n) / n
        else:
            f /= S
        
        # Periodic scoring

                best_score = score
