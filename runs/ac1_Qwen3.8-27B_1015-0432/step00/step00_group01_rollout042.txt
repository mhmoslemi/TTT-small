import time

def _fft_conv(a):
    n = a.size
    size = 1
    while size < 2 * (2 * n - 1):
        size <<= 1
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _fast_score(a):
    n = a.size
    C = _fft_conv(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * C.max() / (s * s)

    if r <= 0.0 or r >= 1.0:
        return float('inf')
    lr = np.log(r)
    k = np.arange(n, dtype=np.float64)

import numpy as np
import time


def _fft_conv(a):
    n = a.size
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _fast_score(a):
    n = a.size
    C = _fft_conv(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * C.max() / (s * s)


def _geo_score(n, r):
    if r <= 0.0 or r >= 1.0:
        return float('inf')
    lr = np.log(r)
    k = np.float64(np.arange(n))
    f1 = (k + 1.0) * np.exp(k * lr)
    k2 = np.float64(np.arange(n, 2 * n - 1))
    f2 = (2.0 * n - 1.0 - k2) * np.exp(k2 * lr)
    mc = max(f1.max(), f2.max())
    tot = (1.0 - r ** n) / (1.0 - r)
    return 2.0 * n * mc / (tot * tot)


def _make_geometric(n, r):
    a = np.exp(np.float64(np.arange(n)) * np.log(float(r)))
    a *= 1000.0 / a[0]
    return np.clip(a, 0.0, 1000.0)


def _make_symmetric_geo(n, r):
    mid = (n - 1) / 2.0
    a = np.exp(np.float64(r) ** 0 - np.abs(np.float64(np.arange(n)) - mid) * np.log(1.0 / r))
    a = (1.0 / r) ** np.abs(np.float64(np.arange(n)) - mid)
    a *= 1000.0 / a.max()
    return np.clip(a, 0.0, 1000.0)
