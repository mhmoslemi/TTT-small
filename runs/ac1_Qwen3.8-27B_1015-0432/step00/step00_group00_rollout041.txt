```python
def compute_gradient(a, w):
    """g[j] = 2 * sum_m a[m] * w[j+m]"""
    n = len(a)
    b = a[::-1]  # reversed
    # full conv of b (len n) and w (len 2n-1) has length 3n-1
    size = 3*n - 1
    nfft = 1
    while nfft < size:
        nfft *= 2
    B = np.fft.rfft(b, nfft)
    W = np.fft.rfft(w, nfft)
    conv = np.fft.irfft(B * W, nfft)[:size]
    # We need conv[n-1 : 2n-1] (n elements)
    return 2.0 * conv[n-1:2*n-1]
```

```python
def soft_max_descent(a, n_iters, tau_start, tau_end, lr, deadline):
    n = len(a)
    a = a.copy().astype(np.float64)
    a /= a.sum()  # normalize to sum 1
    best_a = a.copy()
    best_score = np.inf
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        tau = tau_start * (tau_end/tau_start)**(it/max(1,n_iters-1))
        
        c = fft_convolve(a)  # length 2n-1
        max_c = c.max()
        
        # Soft-max weights
        # To avoid overflow, subtract max
        w = np.exp((c - max_c) / tau)
        w /= w.sum()
        
        # Gradient
        g = compute_gradient(a, w)
        
        # Projection: we want to descend in a while keeping a >= 0 and sum = 1
        # Simple approach: a_new = a - lr * g, then project
        a_new = a - lr * g
        a_new = np.maximum(a_new, 0)
        a_new /= a_new.sum()
        
        # Check if improved (using exact conv)
        c_new = fft_convolve(a_new)
        if c_new.max() < max_c:
            a = a_new
        else:
            # Halve step size (simple backtracking)
            lr = max(lr * 0.5, 1e-10)
        
        # Track best
        if c_new.max() < best_score:
            best_score = c_new.max()
            best_a = a.copy()
        
        if time.monotonic() > deadline:
            break
```
