        np.clip(gs, 0.0, 1000.0, out=gs)
        if gs.sum() < 0.01:
            break
        s = _sc(gs, n)
        if np.isfinite(s) and s < bs:
            bs, best = s, gs.copy()
            alpha = max(alpha * 0.995, 0.3)
    # --- Phase 2: multi-peak subgradient descent with periodic GS refresh ---
    eta = 0.05
    for it in range(200000):
        if time.monotonic() > dl:
            break
        cv = np.convolve(best, best)
        k = int(np.argmax(cv))
        sg = np.zeros(n)
        lo, hi = max(0, k - n + 1), min(n - 1, k)
        sg[lo:hi + 1] = best[k - np.arange(lo, hi + 1, dtype=np.int64)]
        cv2 = cv.copy()
        cv2[k] = -1.0
        k2 = int(np.argmax(cv2))
        if cv2[k2] > 0.9 * cv[k]:
            lo2, hi2 = max(0, k2 - n + 1), min(n - 1, k2)
            sg[lo2:hi2 + 1] += 0.5 * best[k2 - np.arange(lo2, hi2 + 1, dtype=np.int64)]
        c = best - eta * sg + eta * 0.01 * rng.standard_normal(n)
        np.clip(c, 0.0, 1000.0, out=c)
        s = _sc(c, n)
        if np.isfinite(s) and s < bs:
            best, bs = c, s
            eta = min(eta * 1.002, 0.25)
        else:
            eta = max(eta * 0.998, 1e-4)
            if eta < 3e-4:
                c = best * (1.0 + 0.1 * rng.standard_normal(n))
                np.clip(c, 0.0, 1000.0, out=c)
                s = _sc(c, n)
                if np.isfinite(s) and s < bs:
                    best, bs = c, s
                eta = 0.05
        if it > 0 and it % 5000 == 0:
            F = np.fft.rfft(best, N)
            mags = np.abs(F[1:]) + 1e-15
            F[1:] = F[1:] * (mags.mean() / mags) ** 0.5
            g2 = np.fft.irfft(F, N)[:n].real
            np.clip(g2, 0.0, 1000.0, out=g2)
            s2 = _sc(g2, n)
            if np.isfinite(s2) and s2 < bs:
                best, bs = g2, s2
    fs = evaluate_sequence(best.tolist())
    if not np.isfinite(fs) or fs > 2.0 + 1e-6:
        best = a0.copy()
    return [float(x) for x in best]
