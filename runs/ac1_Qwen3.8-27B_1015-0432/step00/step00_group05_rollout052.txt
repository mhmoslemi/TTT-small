import numpy as np
import time


def _fft_conv(a):
    """Full auto-convolution via FFT, O(n log n)."""
    n = a.size
    sz = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2 * n - 1]


def _score(a):
    n = a.size
    b = _fft_conv(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (s * s)


def _b2_grad(a):
    """∇_a of sum_k b[k]^2 where b = convolve(a,a)."""
    n = a.size
    b = _fft_conv(a)

    corr = np.fft.irfft(np.conj(np.fft.rfft(a, sz)) * np.fft.rfft(b, sz), sz)
    return 4.0 * corr[:n]
