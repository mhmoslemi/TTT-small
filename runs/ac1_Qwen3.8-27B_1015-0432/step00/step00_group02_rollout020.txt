import numpy as np
import time


def fft_conv(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def p_norm_gradient(a, b, p):
    n = len(a)
    if p == 1:
        w = np.ones_like(b)
    else:
        w = np.maximum(b, 0.0) ** (p - 1)
    size = 1
    while size < len(w) + n - 1:
        size <<= 1
    fw = np.fft.rfft(w, size)
    fa = np.fft.rfft(a, size)
    corr = np.fft.irfft(fw * np.conj(fa), size)
    return 2.0 * p * corr[:n]


def project_simplex(v, target=1.0):
    n = len(v)
    # Project onto {x>=0, sum(x)=target} via water-filling
    v = np.asarray(v, dtype=np.float64)
    lo = float(v.min()) - target
    hi = float(v.max()) + target
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.maximum(v - 0.5 * (lo + hi), 0.0)
    s = float(x.sum())
    if s > 1e-15:
        x *= target / s
    else:
        x.fill(target / n)
    return x

    cos_w = np.maximum(cos_w, 1e-6)
    shapes.append(cos_w / cos_w.sum())
    quad = (t / n) ** 2 + 1e-4
    shapes.append(quad / quad.sum())
    inv_quad = 1.0 / ((t / n) ** 2 + 0.1)
    shapes.append(inv_quad / inv_quad.sum())
    return shapes


def optimize_p_norm(a, p, lr, max_iters, deadline):
    n = len(a)
    a = a.copy()
    prev_obj = np.sum(np.maximum(fft_conv(a, a), 0.0) ** p)
    stuck = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fft_conv(a, a)
        b = np.maximum(b, 0.0)
        grad = p_norm_gradient(a, b, p)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-18:
            break
        step = lr * a.sum() ** 2
        a_new = a - step * grad / gnorm
        a_new = project_simplex(a_new, 1.0)
        b_new = np.maximum(fft_conv(a_new, a_new), 0.0)
        obj_new = np.sum(b_new ** p)
        if obj_new < prev_obj * (1.0 - 1e-12):
            a = a_new
            prev_obj = obj_new
            stuck = 0
            lr = min(lr * 1.04, 5.0 / n)
        else:
            lr = max(lr * 0.6, 1e-8 / n)
            stuck += 1
            if stuck > 200:
                break
    return a


def block_refine(best, best_score, rng, deadline, n):
    max_w = max(2, n // 30)
    min_w = max(1, n // 500)
    for _ in range(2000):
        if time.monotonic() > deadline:
            break
        c = best.copy()
        w = int(rng.integers(min_w, min(max_w, n)))
        start = int(rng.integers(0, n - w + 1))
        scale = float(np.exp(rng.normal(0.0, 0.08)))
        c[start:start + w] *= scale
        score = float(evaluate_sequence(c.tolist()))
        if np.isfinite(score) and score < best_score:
            best = c
            best_score = score
    return best, best_score


def lp_direction(a, b, max_b, n):
    # Approximate LP: g[i] ~ 1/(a[i]+eps) weighted to reduce peak
    eps = max_b * 0.1 + 1e-8
    g = 1.0 / (a + eps)
    g = project_simplex(g, 1.0)
    return g


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 12.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0 or not np.isfinite(parent).all():
        parent = np.ones(256, dtype=np.float64)
    n = len(parent)

    s = float(parent.sum())
    if s < 0.01:
        parent_norm = np.ones(n) / n
    else:
        parent_norm = parent / s

    best = parent_norm.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    # Try multiple initializations
    shapes = init_shapes(n, rng)
    shapes.append(parent_norm)

    for shp in shapes:
        if time.monotonic() > deadline:
            break
        a = shp.copy()

        # Progressive p-norm optimization
        for p in [2, 3, 4, 6, 8, 12, 16]:
            if time.monotonic() > deadline:
                break
            lr = 0.5 / (p * n)
            max_iters = max(5, int(120.0 / (n / 7000.0)))
            a = optimize_p_norm(a, p, lr, max_iters, deadline)
            if time.monotonic() > deadline:
                break

        # LP-based refinement
        if time.monotonic() < deadline - 5.0:
            b = np.maximum(fft_conv(a, a), 0.0)
            max_b = float(b.max())
            g = lp_direction(a, b, max_b, n)
            for t_frac in [0.3, 0.5, 0.7]:
                a_mix = (1.0 - t_frac) * a + t_frac * g
                a_mix = project_simplex(a_mix, 1.0)
                sc = float(evaluate_sequence(a_mix.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best = a_mix.copy()
                    best_score = sc
                if sc < best_score:
                    a = a_mix

        # Evaluate current
        sc = float(evaluate_sequence(a.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best = a.copy()
            best_score = sc

    # Final block perturbation refinement
    best, best_score = block_refine(best, best_score, rng, deadline, n)

    # Mirror-symmetrize check
    if time.monotonic() < deadline - 5.0:
        a_sym = 0.5 * (best + best[::-1])
        s_sym = float(a_sym.sum())
        if s_sym > 0.01:
            a_sym = a_sym / s_sym
            sc_sym = float(evaluate_sequence(a_sym.tolist()))
            if np.isfinite(sc_sym) and sc_sym < best_score:
                best = a_sym
                best_score = sc_sym

    # Ensure validity
    final_score = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_score) or final_score >= 2.0:
        # Fallback to parent
        best = parent_norm
        final_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(final_score):
            best = np.ones(n) / n
            final_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
