import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + budget_s - 5.0
    n = len(np.asarray(height_sequence_1))
    
    # Initialize
    p = np.ones(n, dtype=np.float64) / n
    p += rng.normal(0, 0.005, n)
    p = np.maximum(p, 0)
    p /= p.sum()
    
    best_p = p.copy()
    best_score = float(evaluate_sequence(p.tolist()))
    
    fft_size = 2 * n
    T = 0.01  # Temperature for soft-max
    lr = 0.5

        F = np.fft.rfft(p, fft_size)
        conv = np.fft.irfft(F * F)[:2*n-1]
