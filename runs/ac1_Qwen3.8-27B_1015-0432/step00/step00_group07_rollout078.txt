import numpy as np
import time


def subgradient_dir(a, tol_frac):
    n = len(a)
    c = np.convolve(a, a)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n)
    active = c >= M * (1.0 - tol_frac)
    n_act = max(1, int(active.sum()))
    h = active.astype(np.float64)
    conv = np.convolve(a[::-1], h)
    corr = conv[n - 1:2 * n - 1]
    return (2.0 / n_act) * corr


def do_step(a, v, step, tol_frac, rng):
    n = len(a)
    g = subgradient_dir(a, tol_frac)
    v = 0.5 * v - step * g
    a_new = a + v
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = np.full(n, 1.0 / n)
    else:
        a_new *= np.sqrt(2.0 * n) / s
    return a_new, v


def compute_score(a):
    c = np.convolve(a, a)
    M = c.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0 * len(a) * M / (S * S)
