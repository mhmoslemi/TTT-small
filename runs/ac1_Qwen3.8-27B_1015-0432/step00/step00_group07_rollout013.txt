def _smooth_perturb(base, n, rng, low_width, hi_width, sigma):
    """Apply one O(n)-memory smooth multiplicative perturbation."""
    candidate = base.copy()
    width = int(rng.integers(max(1, low_width), max(2, hi_width)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    noise = np.cumsum(rng.normal(0.0, sigma, stop - start))
    candidate[start:stop] = np.maximum(0.0, candidate[start:stop] * np.exp(noise))
    return np.minimum(candidate, 1000.0)
