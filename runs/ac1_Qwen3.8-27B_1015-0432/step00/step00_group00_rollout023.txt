import numpy as np
import time

def project_simplex(v):
    """Project v onto {a >= 0, sum(a) = 1}."""
    n = len(v)
    # Bisection on tau
    lo, hi = v.min(), v.max()
    for _ in range(50):
        tau = 0.5 * (lo + hi)
        proj = np.maximum(v - tau, 0.0)
        s = proj.sum()
        if abs(s - 1.0) < 1e-12:
            break
        if s > 1.0:
            lo = tau
        else:
            hi = tau
    return np.maximum(v - tau, 0.0)
