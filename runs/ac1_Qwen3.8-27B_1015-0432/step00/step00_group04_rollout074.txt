    it = 0

    while time.monotonic() < dl:
        it += 1
        corr = _fft_self_conv(a)
        sc = _internal_score(corr, n)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best = a.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 300 == 0:
            sc_exact = float(evaluate_sequence(a.tolist()))
            if sc_exact < best_sc - 1e-10:
                best_sc = sc_exact
                best = a.copy()
                stuck = 0

        g = _active_subgrad(corr, a, n, 32)
        v = 0.7 * v + g
        vn = np.linalg.norm(v)
        if vn < 1e-12:
            break
        d = -v / vn

        ok = False
        for f in (1.0, 0.5, 0.25, 0.125, 0.0625):
            a_new = np.clip(a + lr * f * d, 0.0, 1000.0)
            sn = a_new.sum()
            if sn < 1e-10:
                a_new = np.ones(n)
            else:
                a_new = a_new * n / sn
            sc_new = _internal_score(_fft_self_conv(a_new), n)
            if sc_new < sc - 1e-10:
                a = a_new
                ok = True
                break

        if ok:
            lr = min(lr * 1.03, 5.0)
        else:
            lr = max(lr * 0.75, 0.002)
            if stuck > 600:
                nm = int(rng.integers(1, 6))
                a_alt = _gen_init(n, rng, nm)
                sc_alt = float(evaluate_sequence(a_alt.tolist()))
                if np.isfinite(sc_alt) and sc_alt < best_sc:
                    best = a_alt.copy()
                    best_sc = sc_alt
                a = a_alt.copy()
                v[:] = 0.0
                lr = 0.5
                stuck = 0
            elif stuck > 200:
                a = np.clip(a + rng.normal(0.0, 0.03 * max(a.std(), 0.1), n), 0.0, 1000.0)
                sn = a.sum()
                a = a * n / sn if sn > 1e-10 else np.ones(n)
                v[:] = 0.0
                stuck = 0

    out = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in out]
