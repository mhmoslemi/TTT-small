def _score_and_grad(f, n):
    """Return (score, gradient) using the same formula as evaluate_sequence."""
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    max_b = float(conv[k_star])
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf'), None
    score = 2.0 * n * max_b / (S * S)

import numpy as np
import time


def _smooth_noise(n, rng, sigma, iters):
    noise = rng.normal(0.0, 1.0, size=n)
    for _ in range(iters):
        noise = (noise + np.roll(noise, 1)) * 0.5
        noise = (noise + np.roll(noise, -1)) * 0.5
    return noise * sigma


def _score_and_grad(f, n):
    f = np.clip(f, 0.0, 1000.0)
    conv = np.convolve(f, f)
    k_star = int(np.argmax(conv))
    max_b = float(conv[k_star])
    S = float(np.sum(f))
    if S < 0.01:
        return float('inf'), None
    score = 2.0 * n * max_b / (S * S)
    base = -4.0 * n * max_b / (S ** 3)
    grad = np.full(n, base, dtype=np.float64)
    lo = max(0, k_star - n + 1)
    hi = min(n, k_star + 1)
    if hi > lo:
        js = np.arange(lo, hi, dtype=np.int64)
        kmjs = k_star - js
        valid = (kmjs >= 0) & (kmjs < n)
        if np.any(valid):
            grad[js[valid]] = 4.0 * n * (f[kmjs[valid]] * S - max_b) / (S ** 3)
    return score, grad
