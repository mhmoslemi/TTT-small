        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        c = np.arange(lo, hi)
        rows_all.append(np.full(hi - lo, idx, dtype=np.int32))
        cols_all.append(c)
        vals_all.append(a[k - c])
    return csr_matrix(
        (np.concatenate(vals_all), (np.concatenate(rows_all), np.concatenate(cols_all))),
        shape=(len(k_lags), n),
    )


def simplex_clip(a, target_sum):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s > 1e-12:
        a *= target_sum / s
    else:
        a = np.full_like(a, target_sum / len(a))
    return a


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.ones(n)
    target_sum = np.sqrt(2.0 * n)
    a = simplex_clip(a, target_sum)

    fft_len = 1
    while fft_len < 2 * n - 1:
        fft_len <<= 1

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # Phase 1: LP-based iterations
    M = float(fft_conv(a, fft_len).max())
    K_cons = 300
    t_mix = 0.15
    stall = 0

    for it in range(500):
        if time.monotonic() > deadline - 80:
            break
        if M < 1e-8:
            break

        c = fft_conv(a, fft_len)
        mx = float(c.max())
        k_lags = np.argsort(c)[-K_cons:]
        A_sub = build_sub_lp(a, n, k_lags)

        try:
            res = linprog(
                -np.ones(n), A_ub=A_sub, b_ub=np.full(K_cons, mx),
                bounds=(0.0, np.inf), method="highs",
                options={"presolve": True},
            )
        except Exception:
            break

        if not res.success or res.x is None:
            break
        if not np.isfinite(res.x).all():
            break

        b = np.maximum(res.x, 0.0)
        sb = float(b.sum())
        if sb < 1e-8:
            break
        b *= target_sum / sb

        a = (1.0 - t_mix) * a + t_mix * b
        a = simplex_clip(a, target_sum)
        new_M = float(fft_conv(a, fft_len).max())

        if it % 3 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        if new_M >= mx * (1.0 - 1e-5):
            stall += 1
        else:
            stall = 0

        M = new_M
        t_mix = max(0.03, t_mix * 0.97)

        if stall > 50:
            pert = rng.normal(0.0, 0.005, n) * target_sum
            a = simplex_clip(a + pert, target_sum)
            M = float(fft_conv(a, fft_len).max())
            stall = 0
            t_mix = 0.1

    # Phase 2: Subgradient polish on best
    if time.monotonic() < deadline - 10:
        a_work = best_a.copy()
        lr = 0.002 * target_sum
        for it2 in range(30000):
            if time.monotonic() > deadline - 3:
                break
            c = fft_conv(a_work, fft_len)
            top = np.argsort(c)[-6:]
            g = np.zeros(n)
            for k in top:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n, k + 1)
                g[lo:hi] += a_work[k - np.arange(lo, hi)]
            g *= 2.0 / 6.0
            a_work = a_work - lr * g
            a_work = simplex_clip(a_work, target_sum)
            lr *= 0.99997
            if it2 % 500 == 0 or it2 == 29999:
                sc = float(evaluate_sequence(a_work.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_work.copy()

    sc = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(sc):
        best_a = np.ones(n)

    return [float(x) for x in best_a]
