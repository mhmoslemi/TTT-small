```python
import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog

def compute_conv(a):
    """Compute linear convolution using FFT."""
    n = len(a)
    size = 2 * n  # next power of 2 >= 2n-1
    fa = np.fft.rfft(a, size)
    fb = np.fft.irfft(fa * fa, size)[:2*n-1]
    return fb

def lp_step(a, M, max_iter=100):
    """Solve LP: max sum(b) s.t. (a*b)_k <= M, b >= 0."""
    n = len(a)
    # Constraint matrix: A[k, j] = a[j] if 0 <= k-j <= n-1
    # This is a Toeplitz-like matrix
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n-1)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    return res.x if res.success else None
```
