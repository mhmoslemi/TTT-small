```python
def local_search(a, rng, deadline):
    n = len(a)
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    
    while time.monotonic() < deadline:
        candidate = best.copy()
        move_type = rng.random()
        
        if move_type < 0.4:
            # Block scaling
            width = int(rng.integers(max(1, n//100), n//5 + 1))
            start = int(rng.integers(0, n - width))
            factor = np.exp(rng.normal(0, 0.3))
            candidate[start:start+width] *= factor
        elif move_type < 0.7:
            # Anti-symmetric perturbation
            width = int(rng.integers(n//10, n//2))
            start = 0
            end = min(n, n - start)
            # Add anti-symmetric perturbation to first 'width' elements
            pert = rng.normal(0, 0.1, width)
            # Make it anti-symmetric relative to center
            # a[i] += pert[i], a[n-1-i] -= pert[i]
            candidate[:width] += pert
            candidate[n-width:] -= pert[::-1]
        elif move_type < 0.85:
            # Coordinate push-pull: decrease a high-value element, increase a low one
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            amt = 0.1 * best[i]
            candidate[i] -= amt
            candidate[j] += amt
        else:
            # Smooth perturbation
            candidate = best * np.exp(rng.normal(0, 0.05, n))
        
        candidate = np.clip(candidate, 0.01, 1000)
        score = evaluate_sequence(candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
    
    return best, best_score
```

```python
import numpy as np
import time


def _fast_score(a):
    """Same formula as evaluate_sequence, using numpy for speed."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * n * b.max() / (s * s))


def _parametric_init(n, rng):
    """Generate a collection of structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    # Mild increasing ramp
    inits.append(1.0 + 0.8 * x)
    # Stronger ramp
    inits.append(0.5 + 1.5 * x)
    # Quadratic (U-shape, asymmetric offset)
    inits.append(0.5 + (x - 0.35) ** 2 * 2.0)
    # Cosine bump
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    # Half-wave
    inits.append(0.3 + 0.7 * np.sin(np.pi * x) ** 2)
    # Gentle S-curve
    inits.append(1.0 + 0.6 * np.tanh(3.0 * (x - 0.5)))
    # Two-bump asymmetric
    inits.append(0.5 + 0.5 * np.sin(4.0 * np.pi * x - 0.5) ** 2)
    # Linear decreasing
    inits.append(2.0 - 1.0 * x)
    return inits


def _local_search(a_best, best_score, rng, deadline):
    """Diverse local search with time budget."""
    n = a_best.size
    iter_count = 0
    while time.monotonic() < deadline:
        c = a_best.copy()
        r = rng.random()
        if r < 0.25:
            # Block multiplicative scaling
            w = int(rng.integers(max(1, n // 80), max(2, n // 4)))
            s = int(rng.integers(0, max(1, n - w)))
            c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.25)))
        elif r < 0.45:
            # Low-frequency smooth perturbation
            freq = int(rng.integers(1, 5))
            phase = rng.uniform(0, 2 * np.pi)

            c *= (1.0 + 0.08 * mask)
        elif r < 0.60:
            # Anti-symmetric nudge (breaks symmetry)
            k = int(rng.integers(n // 4, n - 1))
            delta = 0.05 * a_best.std()
            c[:k] += delta * np.linspace(1, 0, k)
            c[n - k:] -= delta * np.linspace(0, 1, k)[::-1]
        elif r < 0.75:
            # Gaussian bump (add or subtract)
            center = rng.uniform(0, n)
            width = rng.uniform(n // 20, n // 4)
            sig = np.exp(-0.5 * ((np.arange(n) - center) / width) ** 2)
            amp = rng.normal(0, 0.1) * a_best.mean()
            c += amp * sig
        elif r < 0.88:

            amt = abs(a_best[i] - a_best[j]) * rng.uniform(0.1, 0.5)
            c[i] -= amt if a_best[i] >= a_best[j] else -amt
            c[j] += amt if a_best[i] >= a_best[j] else -amt
        else:
            # Random walk per element (small)
            c += rng.normal(0, 0.02 * a_best.mean(), n)

        np.clip(c, 0.01, 1000.0, out=c)
        score = _fast_score(c)
        iter_count += 1

def _gradient_refine(a, lr, steps, rng, deadline):
    """Simple projected-gradient steps on the peak lag."""
    n = a.size
    best = a.copy()
    best_score = _fast_score(best)
    for _ in range(steps):
        if time.monotonic() > deadline:
            break
        b = np.convolve(best, best)
        k_max = int(np.argmax(b))
        # Gradient of b[k_max] w.r.t. a[i]: 2*a[k_max-i] for valid i
        lo = max(0, k_max - n + 1)
        hi = min(n - 1, k_max)
        if lo > hi:
            continue
        # Indices i in [lo, hi]: gradient = 2 * a[k_max - i]
        idx = np.arange(lo, hi + 1)
        g = 2.0 * best[k_max - idx]
        # Objective gradient (simplified, ignoring sum-dependent term for speed)
        s = best.sum()
        grad_obj = (2.0 * n / (s * s)) * g
        # Subtract mean to preserve sum approximately
        grad_obj -= grad_obj.mean()
        # Projected gradient descent
        c = best - lr * grad_obj
        # Handle non-gradient components (zero outside [lo,hi])
        np.clip(c, 0.01, 1000.0, out=c)
        sc = _fast_score(c)
        if np.isfinite(sc) and sc < best_score:
            best = c
            best_score = sc
            lr *= 0.95
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best, best_score
```
