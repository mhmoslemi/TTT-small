import numpy as np
import time

def _fft_convolve(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]

def _score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-6:
        return np.inf
    b = _fft_convolve(a)
    M = np.max(b)
    return 2.0 * n * M / (s * s)

def _gradient_step(a, lr):
    """One subgradient step to reduce max of autocorrelation."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-6:
        return a
    # Work with normalized sequence
    a_norm = a / s
    b = _fft_convolve(a_norm)
    k_star = int(np.argmax(b))
    # Gradient: g[m] = 2 * a_norm[k_star - m] if 0 <= k_star - m < n
    # Update: a_norm[m] -= lr * g[m]
    # For m where k_star - m is in [0, n-1]: subtract lr * 2 * a_norm[k_star - g]
    # This is a reversed slice of a_norm
    new_a = a_norm.copy()
    lo = max(0, k_star - n + 1)  # smallest m such that k_star - m <= n - 1
    hi = min(n - 1, k_star)      # largest m such that k_star - m >= 0
    if lo <= hi:
        # a_norm[k_star - m] for m in [lo, hi]
        # = a_norm[k_star - hi : k_star - lo + 1] reversed
        idx_lo = k_star - hi
        idx_hi = k_star - lo
        new_a[lo:hi+1] -= 2.0 * lr * a_norm[idx_lo:idx_hi+1][::-1]
    new_a = np.maximum(new_a, 0.0)
    s_new = np.sum(new_a)
    if s_new < 1e-6:
        return a  # fallback
    new_a /= s_new
    return new_a

def _fft_conv_square(a):
    """Compute full auto-convolution of a using FFT. O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]
