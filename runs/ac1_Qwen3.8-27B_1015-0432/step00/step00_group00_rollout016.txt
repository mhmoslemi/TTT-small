            score = 2.0 * n * mx
            g = _subgrad(a, c)
            gn = np.linalg.norm(g)
            if gn < 1e-14:
                break

            gu = g / gn
            amax = np.max(a)
            improved = False
            for alpha in [lr, lr * 0.5, lr * 0.2, lr * 0.05]:
                step = alpha * amax
                a_new = a - step * gu
                a_new = _proj_simplex(a_new)
                s_new, _ = _score_fft(a_new, n)
                if s_new < score - 1e-9:
                    a = a_new
                    if s_new < best_score - 1e-7:
                        # verify with exact evaluator
                        a_orig = a * n
                        a_orig = np.clip(a_orig, 0.0, 1000.0)
                        ex = float(evaluate_sequence(a_orig.tolist()))
                        if np.isfinite(ex) and ex < best_score:
                            best_a = a_orig.copy()
                            best_score = ex
                    stall = 0
                    improved = True
                    lr = min(lr * 1.03, 1.5)
                    break
            if not improved:
                lr *= 0.6
                stall += 1

            # Periodic random kick to escape plateaus
            if stall > 30 and it % 5 == 0 and stall < 200:
                kick = rng.normal(0.0, 0.05 * amax, size=n)
                a_kicked = _proj_simplex(a + kick)
                sk, _ = _score_fft(a_kicked, n)
                if sk < score:
                    a = a_kicked
                stall = 0

            if stall > 300:
                break

    # Final verification and return
    final_s = float(evaluate_sequence(best_a.tolist()))
    if not np.isfinite(final_s) or final_s > par_score:
        return [float(v) for v in parent]
    return [float(v) for v in np.clip(best_a, 0.0, 1000.0)]
