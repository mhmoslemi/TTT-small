def compute_subgradient(a, b, tol=1e-6):
    """Compute subgradient of max(convolve(a,a)) wrt a."""
    n = len(a)
    M = b.max()
    # Find indices where b is close to max
    K_star = np.where(b >= M - tol * M)[0]
    # For each peak k*, contribution to a[i] is 2*a[k*-i] for valid i
    # a[i] -= lr * (mean_{k*} 2*a[k*-i])
    grad = np.zeros(n)

        grad[i_start:i_end+1] += 2.0 * a[idx]
    grad /= len(K_star)

def _fft_xcorr(b, a, L):
    """Linear cross-correlation c[i]=sum_k b[k]*a[k-i] for i=0..len(a)-1."""
    B = np.fft.rfft(b, L)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(B * np.conj(A), L)
    return c[:len(a)]
