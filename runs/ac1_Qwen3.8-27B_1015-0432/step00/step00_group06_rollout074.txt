import numpy as np
import time


def next_pow2(x):
    m = 1
    x = max(1, int(x))
    while m < x:
        m <<= 1
    return m


def project_positive(a, fallback):
    a = np.asarray(a, dtype=np.float64).copy()
    fb = np.asarray(fallback, dtype=np.float64).copy()
    if a.size != fb.size:
        a = fb.copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = fb
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(max(1, a.size), dtype=np.float64)
        s = float(a.size)
    return a / s


def stable_softmax(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return y.copy()
    y = np.where(np.isfinite(y), y, 0.0)
    y = y - np.max(y)
    e = np.exp(np.clip(y, -700.0, 700.0))
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(y.size, dtype=np.float64) / float(y.size)
    return e / s


def score_fft(a, N):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n <= 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    L = 2 * n - 1
    if N < L:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    if c.size < L:
        return np.inf
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    mc = float(np.max(c))
    if not np.isfinite(mc):
        return np.inf
    return float(2.0 * n * mc / (s * s))


def spectral_raw(a, n, N, alpha):
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    L = 2 * n - 1
    c = c[:L].astype(np.float64, copy=True)
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    cs = float(c.sum())
    max_c = float(c.max()) if c.size else 0.0
    s = float(a.sum())
    if cs <= 0.0 or not np.isfinite(cs) or s <= 0.0:
        return a.copy(), np.inf
    c = c / cs
    ct = np.power(c + 1e-18, alpha)
    sct = float(ct.sum())
    if not np.isfinite(sct) or sct <= 0.0:
        ct = c
    else:
        ct = ct / sct
    T = np.fft.rfft(ct, N)
    A2 = np.sqrt(T)
    a2 = np.fft.irfft(A2, N)
    a2 = np.real(a2[:n])
    a2 = np.where(np.isfinite(a2), a2, 0.0)
    sc = float(2.0 * n * max_c / (s * s))
    return a2, sc

def run_adam(y, n, N, deadline, seed):

    v = np.zeros(n, dtype=np.float64)

    beta2 = 0.999
    b1p = 1.0
    b2p = 1.0
    t = 0
    lr = 0.1

    best = a.copy()

    L = 2 * n - 1
    no_improve = 0
    while time.monotonic() < deadline:
        a = stable_softmax(y)

        c = np.fft.irfft(A * A, N)
        c = c[:L].astype(np.float64, copy=True)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        s = float(a.sum())
        score = float(2.0 * n * max_c / (s * s))
        if np.isfinite(score) and score < best_score:
            best = a.copy()
            best_score = score
            no_improve = 0
        else:
            no_improve += 1
        if not np.isfinite(score) or max_c <= 0.0:
            break
        if no_improve > 3000:
            y = y_from_vector(best)
            y = y + rng.normal(0.0, 0.1, n)

            v.fill(0.0)
            b1p = 1.0
            b2p = 1.0
            t = 0
            lr = max(0.01, lr * 0.5)
            no_improve = 0
            continue
        alpha_coeff = 5.0 + 300.0 * min(1.0, t / 20000.0)

        alpha = alpha_coeff / max(max_c, 1e-12)

        b2p *= beta2
        mb = m / (1.0 - b1p)

        step = lr * mb / (np.sqrt(vb) + 1e-8)
        step = np.clip(step, -0.1, 0.1)
        y = y - step
        y = y - float(np.mean(y))
        y = np.clip(y, -30.0, 30.0)

def random_local_search(a, n, N, deadline, seed):
    rng = np.random.default_rng(seed)
    best = project_positive(a, a)
    best_score = score_fft(best, N)
    if n <= 1:
        return best, best_score
    lo = max(1, n // 500)
    hi = max(lo + 1, n // 8)
    if hi > n:
        hi = n
    while time.monotonic() < deadline:
        width = int(rng.integers(lo, hi + 1))
        start = int(rng.integers(0, n - width + 1))
        cand = best.copy()
        factor = float(np.exp(rng.normal(0.0, 0.15)))
        cand[start:start + width] *= factor
        if rng.random() < 0.25:
            block = cand[start:start + width].copy()
            block.reverse()
            cand[start:start + width] = block
        cand = project_positive(cand, best)
        sc = score_fft(cand, N)
        if np.isfinite(sc) and sc < best_score:
            best = cand
            best_score = sc
    return best, best_score

def choose_initializations(inits, N, keep):
    order = []

    seen = []
    for _, i in order:
        if i not in seen:

    return seen
