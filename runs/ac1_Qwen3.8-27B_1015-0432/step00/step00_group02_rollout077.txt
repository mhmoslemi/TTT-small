import time

def eval_score(a):
    """Fast evaluation using numpy (avoids Python list overhead)."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2 * n * b.max() / (s * s)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    # Get parent
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
    n = parent.size

    best = parent.copy()
    best_score = eval_score(best)
    
    # Try various parametric initializations
    inits = [parent]
    # Two-block
    for ratio in [0.5, 2, 3, 5]:
        m = n // 2
        a = np.empty(n)
        a[:m] = 1.0
        a[m:] = ratio
        inits.append(a)
    # Power laws (decreasing)
    x = np.arange(1, n+1, dtype=np.float64)
    for p in [-0.5, 0.5, 1.0, 1.5, 2.0]:
        inits.append(x ** p)
    # Cosine decay
    inits.append(1.0 + np.cos(np.pi * np.arange(n) / n))
    # Exponential decay
    inits.append(np.exp(-2.0 * np.arange(n) / n))
    
    for a in inits:
        s = eval_score(a)
        if s < best_score:
            best_score = s
            best = a.copy()
    
    # Local search
    T = 0.5  # initial temperature
    no_improve = 0
    while time.monotonic() < deadline:
        a = best.copy()
        # Random perturbation
        mode = rng.integers(0, 4)
        if mode == 0:  # Block scale
            w = int(rng.integers(1, max(2, n // 10)))
            s = int(rng.integers(0, n - w + 1))
            a[s:s+w] *= math.exp(rng.normal(0, 0.1))
        elif mode == 1:  # Add bump
            center = int(rng.integers(0, n))
            width = int(rng.integers(n // 50, n // 10))
            sigma = width * 0.5
            idx = np.arange(max(0, center - 3*width), min(n, center + 3*width))
            a[idx] += 0.1 * np.max(best) * np.exp(-0.5 * ((idx - center) / sigma)**2)
        elif mode == 2:  # Global tilt
            a += rng.normal(0, 0.05) * np.max(best) * (np.arange(n) / n - 0.5)
        else:  # Small noise
            a += rng.normal(0, 0.02, n) * np.max(best)
        
        a = np.clip(a, 0, 1000)
        s = eval_score(a)
        
        if s < best_score:
            best_score = s
            best = a
            no_improve = 0
        else:
            # Metropolis accept
            if rng.random() < math.exp(-(s - best_score) / T):
                best = a
                best_score = s
            no_improve += 1
        
        # Cool down
        if no_improve > 100:
            T *= 0.95
            no_improve = 0
    
    # Final: use exact evaluator
    final_score = float(evaluate_sequence(best.tolist()))
    return [float(x) for x in best]
