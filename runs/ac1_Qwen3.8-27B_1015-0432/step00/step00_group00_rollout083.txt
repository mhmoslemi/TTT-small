def _fft_autocorr(a):
    n = int(a.size)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    L = 2*n - 1
    m = 1
    while m < L:
        m <<= 1
    fa = np.fft.rfft(a, m)
    c = np.fft.irfft(fa * fa, m)[:L]
    return c

def _softmax_param(u, S):
    u = np.asarray(u, dtype=np.float64).ravel()
    if not np.all(np.isfinite(u)):
        return np.full_like(u, S/u.size), None
    u = u - float(np.max(u))
    e = np.exp(u)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full_like(u, S/u.size), None
    return S * e / s, e

def _unit_soft(x, scale):
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
       return np.ones(0)
    y = x - float(np.max(x))
    y = y / max(1e-8, float(scale))

    e=np.exp(y)
    s=e.sum()
    if s<=0 or not np.isfinite(s):
       return np.full_like(y, 1.0/y.size)
    return e/s

def _prob_grad(w, a):
    n = int(a.size)
    if n == 1:
       return np.array([2.0 * float(np.dot(w, a))]) # w length 1
    L = 2*n - 1
    pr = w[::-1]
    m = 1
    while m < L + n - 1:
       m <<= 1
    fw = np.fft.rfft(pr, m)
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fw * fa, m)
    stop = max(-1, L - 1 - n)
    return 2.0 * conv[L - 1 : stop : -1]

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    m = 1
    while m < x:
        m <<= 1
    return m


def _parent_array():
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        arr = np.ones(256, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        arr = np.ones(256, dtype=np.float64)
    return arr


def _normalize(a, S):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([float(S)], dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.full(a.size, float(S) / float(a.size), dtype=np.float64)
    a = a * (float(S) / s)
    return np.clip(a, 0.0, 1000.0)


def _fft_autocorr(a, N=None, return_fa=False):
    n = int(a.size)
    L = 2 * n - 1
    if N is None:
        N = _next_pow2(L)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    if return_fa:
        return c, fa
    return c


def _score_from_c(c, n, s):
    if s <= 1e-12:
        return float("inf")
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx <= 0.0:
        return float("inf")
    return float(2.0 * n * mx / (s * s))


def _score_array(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0 or not np.all(np.isfinite(a)):
        return float("inf")
    s = float(a.sum())
    if s <= 1e-12:
        return float("inf")
    c = _fft_autocorr(a)
    return _score_from_c(c, n, s)

def _softmax_param(u, S):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(u)):
        return np.full(u.size, float(S) / float(u.size), dtype=np.float64)
    u = u - float(np.max(u))
    e = np.exp(u)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(u.size, float(S) / float(u.size), dtype=np.float64)
    return float(S) * e / s


def _weights_from_c(c_norm, tau, scale):
    c_norm = np.asarray(c_norm, dtype=np.float64)
    if c_norm.size == 0:
        return np.array([], dtype=np.float64)
    x = (c_norm - float(tau)) / max(1e-8, float(scale))
    mx = float(np.max(x))
    if not np.isfinite(mx):
        return np.full(x.shape, 1.0 / float(x.size), dtype=np.float64)
    e = np.exp(x - mx)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.shape, 1.0 / float(x.size), dtype=np.float64)
    return e / s


def _prob_grad(w, a, N, fa=None):
    n = int(a.size)
    if n == 1:
        return np.array([2.0 * float(np.dot(w, a))], dtype=np.float64)
    L = 2 * n - 1
    pr = w[::-1]
    fw = np.fft.rfft(pr, N)
    if fa is None:
        fa = np.fft.rfft(a, N)
    conv = np.fft.irfft(fw * fa, N)
    stop = max(-1, L - 1 - n)
    return 2.0 * conv[L - 1:stop:-1]


def _make_exp(n, S, lam, alp):
    if n == 1:
        return np.array([float(S)], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if abs(alp - 1.0) < 1e-12:
        w = np.exp(-lam * t)
    else:
        w = np.exp(-lam * np.power(t, alp))
    return _normalize(w, S)


def _make_linear(n, S, c):
    if n == 1:
        return np.array([float(S)], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    w = 1.0 + c * t
    w = np.maximum(w, 1e-12)
    return _normalize(w, S)


def _make_power(n, S, p):
    if n == 1:
        return np.array([float(S)], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    w = np.power(1.0 - t, p)
    w = np.maximum(w, 1e-12)
    return _normalize(w, S)


def _make_modexp(n, S, lam, eps, freq, phase):
    if n == 1:
        return np.array([float(S)], dtype=np.float64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    env = np.exp(-lam * t)
    mod = 1.0 + eps * np.cos(2.0 * np.pi * freq * t + phase)
    w = env * np.maximum(mod, 0.05)
    return _normalize(w, S)

    if not np.isfinite(final_score):
        safe_best = np.full(n, S / float(n), dtype=np.float64)
        final_score = _score_exact(safe_best)

    if not np.isfinite(final_score):
        return [1.0]

    return [float(x) for x in safe_best]
