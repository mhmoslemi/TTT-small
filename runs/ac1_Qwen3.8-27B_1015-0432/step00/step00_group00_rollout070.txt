def fft_convolve(a):
    """Full linear convolution via FFT. O(n log n)."""
    n = a.size
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    fa = np.fft.rfft(a, N)
    out = np.fft.irfft(fa * fa, N)[:size]
    return np.maximum(out, 0.0)
