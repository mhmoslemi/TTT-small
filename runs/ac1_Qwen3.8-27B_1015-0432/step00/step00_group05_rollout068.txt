from scipy.fft import rfft, irfft
import time


def fft_autocorr(a, fsize, n):
    A = np.zeros(fsize)
    A[:n] = a
    C = irfft(rfft(A, n=fsize), n=fsize)
    return C[:2 * n - 1]


def subgrad_max_c(a, K, n):
    mu = np.zeros(n)
    for k in K:
        if k < n:
            mu[:k + 1] += a[k::-1]
        else:
            mu[k - n + 1:] += a[n - 1:(k - n):-1]
    mu *= 2.0 / len(K)
    return mu

import numpy as np
from scipy.fft import rfft, irfft
import time


def fft_conv_self(a, fsize, n):
    buf = np.zeros(fsize)
    buf[:n] = a
    spec = rfft(buf, n=fsize)
    out = irfft(spec * spec, n=fsize)
    return out[:2 * n - 1]


def compute_subgrad(a, K, n):
    mu = np.zeros(n)
    for k in K:
        if k < n:
            mu[:k + 1] += a[k::-1]
        else:
            mu[k - n + 1:] += a[n - 1:(k - n):-1]
    mu *= 2.0 / len(K)
    return mu
