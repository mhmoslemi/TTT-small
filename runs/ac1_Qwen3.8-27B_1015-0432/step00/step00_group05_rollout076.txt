    if p.ndim != 1 or p.size < 1 or not np.isfinite(p).all():
        p = np.ones(4498)
    n = p.size
    x = np.arange(n, dtype=np.float64)
    inv_n = 1.0 / n

    inits = [
        p / p.sum(),
        1.01 - x * inv_n,
        0.5 + 0.5 * np.cos(2.0 * np.pi * x * inv_n),
        np.exp(-((x - 0.3 * n) ** 2) / (0.15 * n) ** 2)
            + np.exp(-((x - 0.7 * n) ** 2) / (0.15 * n) ** 2),
        np.where(x < 0.6 * n, 1.5, 0.5),
        1.0 + 0.5 * np.sin(4.0 * np.pi * x * inv_n),
        1.0 + 0.5 * np.cos(2.0 * np.pi * x / 7.0),
        x * inv_n + 0.01,
        np.exp(-3.0 * x * inv_n),
    ]

    best, best_sc = None, np.inf
    for c in inits:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            continue
        c = c / s
        sc = ac1(c)
        if np.isfinite(sc) and sc < best_sc:
            best_sc, best = sc, c.copy()

    if best is None:
        best = np.ones(n) * inv_n
        best_sc = 2.0

    bl = best.tolist()
    if not np.isfinite(evaluate_sequence(bl)):
        return [1.0] * 100

    a = best.copy()
    cur = best_sc
    T = a.sum()
    step = 0.02
    stall = 0

    for it in range(500000):
        if time.monotonic() >= t_end:
            break
        conv = fast_conv(a)
        mx = conv.max()
        thr = mx * 0.97
        act = np.where(conv >= thr)[0]
        cap = 150
        if len(act) > cap:
            act = np.argpartition(conv, -cap)[-cap:]
        act = act[np.argsort(conv[act])[::-1]]

        grad = np.zeros(n)
        for k in act:
            ki = int(k)
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            grad[lo:hi + 1] += 2.0 * a[lo:hi + 1][::-1]
        grad /= max(1, len(act))
        gt = grad - grad.mean()

        ok = False
        for _ in range(4):
            a2 = np.maximum(a - step * gt, 0.0)
            s2 = a2.sum()
            if s2 < 1e-15:
                step *= 0.5
                continue
            a2 *= T / s2
            sc2 = ac1(a2)
            if np.isfinite(sc2) and sc2 < cur:
                a, cur = a2, sc2
                if cur < best_sc:
                    best_sc, best = cur, a.copy()
                stall = 0
                ok = True
                break
            step *= 0.5
        if not ok:
            stall += 1
            if stall > 100:
                a2 = a * np.exp(rng.normal(0.0, 0.04, n))
                a2 = np.maximum(a2, 0.0)
                s2 = a2.sum()
                if s2 > 1e-15:
                    a2 *= T / s2
                    sc2 = ac1(a2)
                    if np.isfinite(sc2) and sc2 < cur:
                        a, cur = a2, sc2
                        if cur < best_sc:
                            best_sc, best = cur, a.copy()
                stall = 0
                step = min(0.05, step * 2.0 + 0.01)

        if it % 500 == 0:
            v = evaluate_sequence(a.tolist())
            if np.isfinite(v) and v < best_sc:
                best_sc, best = v, a.copy()

    fl = best.tolist()
    fv = evaluate_sequence(fl)
    if np.isfinite(fv):
        return fl
    return [1.0] * 100
