import numpy as np
import time

def _score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(convolve(a,a))"""
    b = np.convolve(a, a)
    return 2.0 * len(a) * float(b.max())

def _gradient_peak(a, k):
    """Gradient of b_k w.r.t. a (not projected)."""
    n = len(a)
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi+1] = a[k-hi : k-lo+1][::-1]
    return 2.0 * g

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0
    
    a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(256)
        s = a.sum()
    a = a / s
    n = len(a)
    
    best = a.copy()
    best_score = _score_normalized(best)
    
    lr = 0.05
    momentum = np.zeros(n)
    beta = 0.9
    
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        
        b = np.convolve(a, a)
        k_star = int(np.argmax(b))
        peak = b[k_star]
        score = 2.0 * n * peak
        
        if score < best_score:
            best_score = score
            best = a.copy()
        
        # Gradient (top-3 peaks for smoothness)
        top_k = np.argsort(b)[-3:]
        grad = np.zeros(n)
        for kk in top_k:
            g = _gradient_peak(a, int(kk))
            grad += g
        grad /= 3.0
        
        # Project to maintain sum
        grad -= grad.mean()
        
        # Momentum step
        momentum = beta * momentum + (1-beta) * grad
        a_new = a - lr * momentum
        a_new = np.clip(a_new, 0.0, None)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a  # revert
        else:
            a_new /= s_new
            a = a_new
        
        if it % 200 == 0:
            lr = max(0.001, lr * 0.95)
    
    # Final check
    final = [float(x) for x in best]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs):
        # Fallback: uniform
        final = [1.0/n] * n
    return final
