    # Bimodal: heavy at both ends
    t = np.arange(n, dtype=np.float64)
    bi = np.concatenate([np.arange(n // 2, 0, -1, dtype=np.float64),
                        np.ones(max(1, n - 2 * (n // 2)), dtype=np.float64),
                        np.arange(1, n - (n // 2), dtype=np.float64)])[:n]
    bi /= bi.sum()
    init_list.append(bi)

    for init in init_list:
        ci = _fft_self_conv(init)
        if ci.max() < best_cmax:
            best = init.copy()
            best_cmax = ci.max()

    a = best.copy()
    best_score = 2.0 * n * best_cmax

    # Mirror descent with adaptive step
    eta = 1.0
    stall = 0
    beta = 30.0  # soft-max temperature
    it = 0

    while time.monotonic() < deadline:
        it += 1
        c = _fft_self_conv(a)
        cmax = c.max()

        g = _subgradient(c, a, beta)

        # Multiplicative weights update
        scale = -eta * g
        scale -= np.max(scale)  # numerical stability
        a_new = a * np.exp(scale)
        a_new = np.maximum(a_new, 1e-30)
        a_new /= a_new.sum()

        c_new = _fft_self_conv(a_new)
        cmax_new = c_new.max()

        if cmax_new < cmax - 1e-14:
            a = a_new
            stall = 0
            eta = min(50.0, eta * 1.06)
            if cmax_new < best_cmax:
                best = a.copy()
                best_cmax = cmax_new
                best_score = 2.0 * n * best_cmax
        else:
            stall += 1
            eta *= 0.75
            if stall > 300:
                # Random perturbation to escape plateau
                a = a + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                eta = 1.0
                beta = 15.0
                stall = 0
            elif stall > 100:
                beta = max(5.0, beta * 0.9)  # widen active set

    # Final check
    result = best.tolist()
    sc = evaluate_sequence(result)
    if not np.isfinite(sc):
        result = (np.ones(n) / n).tolist()
    return result
