import numpy as np
import time


def _score_fast(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return float(2.0 * a.size * b.max() / (s * s))


def _gradient_step(a, lr, rng):
    n = a.size
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    b = np.convolve(a, a)
    kstar = int(np.argmax(b))
    max_b = b[kstar]
    coef1 = 2.0 * n / (s * s)
    coef2 = 4.0 * n * max_b / (s * s * s)
    grad = -coef2 * a
    j_lo = max(0, kstar - n + 1)
    j_hi = min(n - 1, kstar)
    kj = kstar - np.arange(j_lo, j_hi + 1)
    grad[j_lo:j_hi + 1] += coef1 * a[kj]
    best_trial = _score_fast(a)
    best_new = a.copy()
    best_lr = 0.0
    for t in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
        a_new = np.clip(a - t * grad, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = _score_fast(a_new)
        if sc < best_trial:
            best_trial = sc
            best_new = a_new
            best_lr = t
    return best_new, best_trial


def _block_search(best, best_score, rng, deadline, iters):
