import numpy as np
import time


def fft_autocorr(a):
    n = a.size
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def make_init(n, style, rng):
    x = np.arange(n, dtype=np.float64) / max(1, n - 1)
    s = style % 8
    if s == 0:
        return np.exp(-2.0 * x)
    if s == 1:
        return np.maximum(1.0 - 0.95 * x, 0.02)
    if s == 2:
        return np.exp(-4.0 * x) + 0.05
    if s == 3:
        return 1.0 / (1.0 + 3.0 * x)
    if s == 4:
        return np.maximum((1.0 - x) ** 1.5, 0.01)
    if s == 5:
        return np.exp(-1.5 * x) * (1.0 + 0.3 * np.sin(4.0 * np.pi * x))
    if s == 6:
        return np.exp(-0.5 * x) * (1.0 + 0.5 * x)
    return np.ones(n)


def compute_grad(a, k_star, m, s, top_k, conv):
    n = a.size
    grad = np.full(n, -2.0 * m / (s ** 3), dtype=np.float64)
    peaks = np.argpartition(-conv, top_k)[:top_k]
    w = np.zeros(top_k)
    w[:] = 1.0 / top_k
    for pk, wt in zip(peaks, w):
        pk = int(pk)
        val = conv[pk]
        i_lo = max(0, pk - n + 1)
        i_hi = min(n - 1, pk)
        base = -2.0 * val * wt / (s ** 3)
        grad += base
        if i_lo <= i_hi:

            seg_end = pk - i_lo + 1
            seg = a[seg_start:seg_end][::-1]
            grad[i_lo:i_hi + 1] += 2.0 * seg * wt / (s ** 2)
    return grad
