    while N < 2*n - 1:

        b = np.fft.ifft(X * X).real

        b = convolve_fast(a)
        S = a.sum()
        M = b.max()
        score = 2*n*M/S**2

            best = a.copy()
            best_score = score

        tol = 1e-6 * M

        grad = np.zeros(n)
        for k in max_idx:

            segment = a[k-hi:k-lo+1][::-1]

        grad /= len(max_idx)  # average subgradient

        # Remove mean component (project to sum-zero subspace)
        grad -= grad.mean()
