    n = len(a)
    conv = np.convolve(a, a)
    k_max = np.argmax(conv)
    threshold = conv[k_max] * (1 - n_hot_frac)  # consider top 10%
    hot = np.where(conv >= threshold)[0]
    
    # Compute pressure at each position
    p = np.zeros(n)
    for k in hot:

    # Zero-sum update
    p = p - np.mean(p)
    a_new = a - alpha * p
    a_new = np.clip(a_new, 0, 1000)
    return a_new
