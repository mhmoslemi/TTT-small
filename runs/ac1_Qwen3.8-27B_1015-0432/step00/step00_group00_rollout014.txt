    deadline = time.monotonic() + max(0.0, float(budget_s) - 8.0)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size
    if n < 2 or not np.isfinite(parent).all() or parent.sum() < 0.01:
        parent = np.ones(256, dtype=np.float64)
        n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    idx = np.arange(n, dtype=np.float64)
    # Diverse initializations to escape the uniform saddle
    inits = []
    for freq in [1, 2, 3, 5, 8]:
        a = parent * (1.0 + 0.2 * np.sin(2.0 * np.pi * freq * idx / n))
        inits.append(np.clip(a, 0.0, 1000.0))
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / max(1, n - 1)))
    inits.append(a * (parent.max() / max(a.max(), 1e-9)))
    a = 1.0 - 0.3 * np.cos(2.0 * np.pi * idx / max(1, n - 1))
    inits.append(a * (parent.mean() / a.mean()))
    inits.append(np.clip(parent * np.exp(rng.normal(0, 0.12, n)), 0.0, 1000.0))
    # Gradient descent from each init
    for init in inits:
        phase_deadline = deadline - 3.0
        a_opt, f_opt = gd_run(init, n, phase_deadline, rng)
        if f_opt < best_score:
            best = a_opt
            best_score = f_opt
    # Random perturbation polish
    polish_deadline = deadline
    for _ in range(20000):
        if time.monotonic() > polish_deadline:
            break
        cand = block_perturb(best, rng)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand
            best_score = s
    # Final verification
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        best = parent
        fs = float(evaluate_sequence(best.tolist()))
    return [float(x) for x in best]
