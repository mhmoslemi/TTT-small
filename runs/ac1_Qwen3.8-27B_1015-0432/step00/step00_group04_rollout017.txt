            sc = 2.0 * nlen * np.max(np.clip(fft_convolve(a, a), 0.0, 10000.0))
            if sc < best_score:
                best_a = a.copy()
                best_score = sc
                stall = 0
            else:
                stall += 1
        else:
            stall += 1
        if stall > 250:
            pw = max(2, nlen // 40)
            while stall > 250:
                s0 = rng.integers(0, max(1, nlen - pw))
                a_p = a.copy()
                a_p[s0:s0 + pw] *= np.exp(rng.normal(0.0, 0.25, pw))
                if a_p[s0] < 0.001:
                    a_p[s0] = 0.001
                sp = a_p.sum()
                if sp > 0.01:
                    a_p = a_p / sp
                    try:
                        ps = float(evaluate_sequence(a_p.tolist()))
                    except Exception:
                        ps = 1e10
                    if ps < best_score + 0.35:
                        a = a_p
                        stall = 0
                        break
            if stall > 250:
                fac *= 0.85
                tau *= 0.85
                stall = 0
        elif it % 600 == 0:
            fac = min(fac * 1.05, 6.0)
            tau *= 0.998
    for _ in range(3):
        if time.monotonic() > deadline - 5.0:
            break
        sc = float(evaluate_sequence(best_a.tolist()))
        if sc < best_score:
            best_score = sc
    if not np.isfinite(best_score) or best_score > 2.0001:
        try:
            fu = np.ones(nlen, dtype=np.float64)
            fu = fu / np.sum(fu)
            us = float(evaluate_sequence(fu.tolist()))
            if np.isfinite(us) and us < best_score:
                best_a = fu.copy()
                best_score = us
        except Exception:
            pass
    return [float(x) for x in best_a]
