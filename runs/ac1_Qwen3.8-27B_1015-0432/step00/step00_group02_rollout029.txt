```python
def simplex_project(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    sorted_v = np.sort(v)[::-1]
    sv = np.cumsum(sorted_v)
    rho = np.where(sorted_v > 0)[0]
    cs = (sv[rho] - target) / (rho + 1)
    cond = (sorted_v[rho] - cs) >= 0
    if not cond.any():
        return np.full(n, target / n)
    t = cs[cond[-1]] if cond[-1] else 0
    return np.maximum(v - t, 0)
```

```python
def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = _next_pow2(n)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
```

```python
        A = np.fft.rfft(a, bsz)
        b = np.maximum(np.fft.irfft(A * A, bsz), 0.0)
        mb = b.max()

        # Softmax weights on lags
        ps = np.exp(alpha * (b - mb))
        ps /= ps.sum()

        # Gradient g[j] = 2 * sum_m ps[m+j]*a[m]  via cross-correlation trick
        ar = a[::-1]
        Ar = np.fft.rfft(ar, csv)
        Pr = np.fft.rfft(ps, csv)
        cv = np.fft.irfft(Ar * Pr, csv)
        g = 2.0 * cv[n - 1:2 * n - 1]
        g -= g.mean()
        gn = np.linalg.norm(g)

        if gn < 1e-14:
            stag += 1
            if stag > 100:
                a = best * (1.0 + 0.25 * rng.standard_normal(n))
                a = np.clip(a, 0.0, None)
                ss = a.sum()
                if ss > 0.01:
                    a = a / ss
                stag = 0
                alpha = max(20.0, alpha * 0.5)
            continue

        d = -g / gn
        d -= d.mean()

        # Backtracking line search (cheap FFT checks)
        improved = False
        for s in (0.3, 0.05, 0.005):
            an = _proj(a + s * d)
            An = np.fft.rfft(an, bsz)
            bn = np.maximum(np.fft.irfft(An * An, bsz), 0.0)
            if bn.max() < mb * 1.002:
                a = an
                improved = True
                break
        if not improved:
            a = _proj(a + 0.0005 * d)

        # Periodic exact scoring
        if it % 8 == 0:
            sc = float(evaluate_sequence(a.tolist()))
            if sc < best_score:
                best_score = sc
                best = a.copy()
                stag = 0
            else:
                stag += 8

        # Restart if stuck
        if stag > 600:
            a = best * (1.0 + 0.3 * rng.standard_normal(n))
            a = np.clip(a, 0.0, None)
            ss = a.sum()
            if ss > 0.01:
                a = a / ss
            stag = 0
            alpha = max(20.0, alpha * 0.5)

        alpha = min(2000.0, alpha * 1.001)

    # Final exact evaluation
    fs = float(evaluate_sequence(best.tolist()))
    if fs < best_score:
        best_score = fs
    return [float(v) for v in best]
```
