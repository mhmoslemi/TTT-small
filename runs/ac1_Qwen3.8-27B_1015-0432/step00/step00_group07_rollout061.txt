def build_constraint_matrix(a, n):
    # Returns sparse (2n-1) x n matrix
    n_rows = 2*n - 1
    data_parts = []
    row_parts = []
    col_parts = []
    for j in range(n):
        r_start = j
        r_end = min(n_rows, j + n)
        cnt = r_end - r_start
        data_parts.append(np.full(cnt, a[j]))
        row_parts.append(np.arange(r_start, r_end))
        col_parts.append(np.full(cnt, j))
    data = np.concatenate(data_parts)
    rows = np.concatenate(row_parts)
    cols = np.concatenate(col_parts)
    return sparse.csc_matrix((data, (rows, cols)), shape=(n_rows, n))
