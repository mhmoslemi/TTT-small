import numpy as np
import time


def _fft_conv(a):
    """Full linear convolution via FFT, O(n log n)."""
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score_fast(a):
    """Fast scale-invariant score via FFT (may differ from exact by ~1e-10)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _fft_conv(a)
    return float(2.0 * n * conv.max() / (s * s))


def _grad_max_constraint(a, k_star):
    """Subgradient of score at current argmax k_star (score treated as max_conv/(sum)^2 * 2n)."""
    n = len(a)
    s = a.sum()
    conv = _fft_conv(a)
    mc = conv[k_star]
    # d(score)/d(a[j]) = 2n/S^2 * 2*a[k*-j]  -  4n*mc/S^3 * a[j]
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    ii = np.arange(js, je + 1)
    grad[ii] = 2.0 * a[k_star - ii]
    c1 = 2.0 * n / (s * s)
    c2 = 4.0 * n * mc / (s * s * s)
    grad *= c1
    grad -= c2 * a
    return grad
