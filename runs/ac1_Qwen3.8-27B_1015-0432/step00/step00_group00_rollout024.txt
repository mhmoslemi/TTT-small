import time

def fft_convolve(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = fa * fa
    result = np.fft.irfft(fb, size)
    return result[:2*n-1]

def compute_objective(b, sum_a):
    max_b = b.max()
    n = (len(b) + 1) // 2

import numpy as np
import time


def fft_conv_sq(a):
    n = len(a)
    m = 1 << max(1, (2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    return np.fft.irfft(fa * fa, m)[:2 * n - 1]


def fast_obj(b, sa):
    n = (len(b) + 1) // 2
    if sa < 0.01:
        return np.inf
    return 2.0 * n * b.max() / (sa * sa)


def subgrad_peak(a, k):
    n = len(a)
    g = np.zeros(n)
    lo1, hi1 = max(0, k - n + 1), min(n - 1, k)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] = a[k - hi1:k - lo1 + 1][::-1]
    lo2, hi2 = max(0, k), min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += a[:hi2 - lo2 + 1]
    return g


def init_gauss(n, w):
    j = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    return np.exp(-0.5 * ((j - c) / max(w, 1.0)) ** 2)


def init_cosine(n):
    return 0.5 * (1.0 + np.cos(np.pi * np.arange(n, dtype=np.float64) / max(n - 1, 1)))


def normalize_sum(a, target):
    s = a.sum()
    if s < 1e-12:
        a[:] = target / len(a)
    else:
        a *= target / s
    return a


def sg_step(a, lr, target):
    sa = a.sum()
    if sa < 1e-12:
        a[:] = target / len(a)
        return a
    b = fft_conv_sq(a)
    kb = b.max()
    ks = int(np.argmax(b))
    gpk = subgrad_peak(a, ks)
    a = a - lr * gpk
    a = np.clip(a, 0.0, 1000.0)
    a = normalize_sum(a, target)
    return a
