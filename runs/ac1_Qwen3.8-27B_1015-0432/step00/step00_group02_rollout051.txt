def pressure_vec(f, ks):
    n = f.size
    p = np.zeros(n)
    i0 = max(0, ks - (n - 1))
    i1 = min(n - 1, ks)
    p[i0:i1 + 1] = f[ks - i1:ks - i0 + 1][::-1]
    return p


def sg_update(f, ks, alpha):
    p = pressure_vec(f, ks)
    mp = p.max()
    if mp < 1e-30:
        return f.copy()
    return np.clip(f * (1.0 - alpha * p / mp), 0.0, 1000.0)


def block_pert(f, rng, wlo, whi, sig):
    n = f.size
    w = int(rng.integers(wlo, min(whi, n) + 1))
    s = int(rng.integers(0, n - w))
    g = f.copy()
    g[s:s + w] *= float(np.exp(rng.normal(0.0, sig)))
    return np.clip(g, 0.0, 1000.0)
