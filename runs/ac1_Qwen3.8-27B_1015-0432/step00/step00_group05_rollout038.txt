import numpy as np
import time

def compute_score_fft(a):
    """Compute the AC1 score using FFT for convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    # Full convolution via FFT
    size = 2 * n - 1
    fsize = 1
    while fsize < size:
        fsize *= 2
    fa = np.zeros(fsize, dtype=np.complex128)
    fa[:n] = a
    fa_cap = fa.conjugate()  # for autocorrelation
    # Actually, for conv(a, a), just use fa * fa
    spectrum = np.fft.rfft(fa, fsize)
    conv = np.fft.irfft(spectrum * spectrum, fsize)[:size]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fsize = 1
    while fsize < size:
