import numpy as np
import time

def fast_conv(a):
    n = a.size
    size = 2**int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    fa = fa * fa
    out = np.fft.irfft(fa, size)[:2 * n - 1]
    return out

def partner_vec(f, kstar, n):
    p = np.zeros(n)
    jmin = max(0, kstar - n + 1)
    jmax = min(n - 1, kstar)
    p[jmin:jmax + 1] = f[kstar - jmax:kstar - jmin + 1][::-1]
    return p

def normalize(f, S):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.full_like(f, S / f.size)
    return f * (S / s)

def mk_perturb(f, rng):
    n = f.size

    if rng.random() < 0.25:
        f = 0.5 * (f + f[::-1])
    return f
