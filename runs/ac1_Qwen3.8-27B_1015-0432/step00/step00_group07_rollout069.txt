def _norm(a, n):
    s = a.sum()
    if s < 1e-15:
        a[:] = 1.0
        return
    a *= np.sqrt(2.0 * n) / s


def _score(a, n):
    size = 2 * n
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)
    M = float(b[:2 * n - 1].max())
    S = float(a.sum())
    if S < 0.01:
        return np.inf
    return 2.0 * n * M / (S * S)


def _grad_shape(a, n, j_star):
    lo = max(0, j_star - n + 1)
    hi = min(n - 1, j_star)
    g = np.zeros(n)
    if lo <= hi:
        s0 = max(0, j_star - (n - 1))
        s1 = min(n - 1, j_star) + 1
        g[lo:hi + 1] = a[s0:s1][::-1]
    g -= g.mean()
    return g
