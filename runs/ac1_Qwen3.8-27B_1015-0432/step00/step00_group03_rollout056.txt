            sc = float(evaluate_sequence(ev.tolist()))
            if sc < best_score:
                best_score = sc
                best = ev.copy()
    f_norm = best.copy()
    total_b = best.sum()
    if total_b > 1e-12:
        f_norm = f_norm / total_b
    f_norm, _ = local_refine(f_norm, rng, deadline)
    ev = np.clip(f_norm * 100.0, 0.0, 1000.0)
    if ev.sum() > 0.01:
        sc = float(evaluate_sequence(ev.tolist()))
        if sc < best_score:
            best = ev
            best_score = sc
    if time.monotonic() < deadline - 1.0:
        for _ in range(5000):
            if time.monotonic() > deadline:
                break
            pert = best * (1.0 + rng.normal(0, 0.02, n))
            pert = np.clip(pert, 0.0, 1000.0)
            if pert.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(pert.tolist()))
            if sc < best_score:
                best = pert
                best_score = sc
    return [float(x) for x in best]
