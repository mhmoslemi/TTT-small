    """Compute full convolution using FFT. O(n log n)."""
    n = len(a)

def _compute_score(a):
    """Fast AC1 score for a nonnegative array."""
    n = a.size
    S = a.sum()
    if S < 0.01:
        return np.inf
    b = np.convolve(a, a)
    max_b = b.max()
    return float(2.0 * n * max_b / (S * S))


def _make_candidate_ramp(n, slope, offset):
    """Linear ramp: a[i] = offset + slope * i/(n-1)."""
    t = np.linspace(0.0, 1.0, n)
    return np.maximum(offset + slope * t, 0.01)


def _make_candidate_clump(n, frac_left, frac_right, h_left, h_right):
    """Two-clump with possibly different heights."""
    c = np.zeros(n)
    m1 = max(1, int(n * frac_left))
    m2 = max(1, int(n * frac_right))
    c[:m1] = h_left
    c[n - m2:] = h_right
    return np.maximum(c, 0.01)


def _make_candidate_cos(n, freq, amp, asym):
    """Cosine-modulated with asymmetric envelope."""
    i = np.arange(n, dtype=np.float64)
    t = i / n
    env = 1.0 + asym * (2.0 * t - 1.0)
    return np.maximum(env * (1.0 + amp * np.cos(2.0 * np.pi * freq * t)), 0.01)
