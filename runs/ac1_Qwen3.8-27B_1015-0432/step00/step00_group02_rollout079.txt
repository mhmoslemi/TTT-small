import numpy as np
import time


def _fft_len(n):
    L = 1
    m = 2 * n - 1
    while L < m:
        L <<= 1
    return L


def _norm(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None
    return a / s


def _score_norm(a, L, n):
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    b = b[:2 * n - 1]
    b = np.maximum(b, 0.0)
    peak = float(b.max())
    if b.size == 0 or not np.isfinite(peak):
        return np.inf
    return 2.0 * float(n) * peak


def _consider(cand, L, n, best, best_score):
    out = _norm(cand)
    if out is None:
        return best, best_score
    sc = _score_norm(out, L, n)
    if np.isfinite(sc) and sc < best_score:
        return out, sc
    return best, best_score


def _soft_weights(b, K, rho):
    m = b.size
    peak = float(b.max())
    if peak <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    z = (b - peak) * (K / peak)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.full(m, 1.0 / m, dtype=np.float64)
    else:
        p = w / s
    if rho > 0.0:
        p = (1.0 - rho) * p + rho / m
        p /= p.sum()
    return p


def _cross_corr(p, A, n, L):
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(A * np.conj(P), L)
    c = c[:n]
    return np.maximum(c, 0.0)


def _lin_cand(a, c, alpha):
    d = c - c.mean()
    sc = float(np.mean(np.abs(d)))
    if sc <= 1e-15:
        return a.copy()
    a2 = a - alpha * d
    a2 = np.maximum(a2, 0.0)
    s = float(a2.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return a.copy()
    return a2 / s


def _mirror_cand(a, c, alpha):
    d = c - c.mean()
    sc = float(np.mean(np.abs(d)))
    if sc <= 1e-15:
        return a.copy()
    lf = np.clip(-alpha * d / sc, -4.0, 4.0)
    a2 = a * np.exp(lf)
    s = float(a2.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return a.copy()
    return a2 / s


def _block_cand(a, rng, sigma):
    n = a.size
    a2 = a.copy()
    for _ in range(3):
        width = int(rng.integers(1, max(2, n // 80)))
        if width > n:
            width = n
        start = int(rng.integers(0, n - width + 1))
        factor = float(np.exp(rng.normal(0.0, sigma)))
        a2[start:start + width] *= factor
    out = _norm(a2)
    return out if out is not None else a.copy()


def _frame_cand(a, frames, rng, scale):
    if frames is None or frames.size == 0:
        return a.copy()
    coeff = rng.normal(0.0, 1.0, frames.shape[0])
    v = frames @ coeff
    sd = float(np.std(v))
    if sd <= 1e-12:
        out = _norm(a)
        return out if out is not None else a.copy()
    v = v / sd
    a2 = a * (1.0 + scale * v)
    a2 = np.clip(a2, 1e-12, None)
    out = _norm(a2)
    return out if out is not None else a.copy()


def _mass_cand(a, k, rng):
    n = a.size
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    cnt = hi - lo + 1
    if cnt < 1:
        return None
    ns = int(min(cnt, 256))
    idx = lo + rng.integers(0, cnt, ns)
    prod = a[idx] * a[k - idx]
    imax = int(np.argmax(prod))
    imin = int(np.argmin(prod))
    if imax == imin:
        return None
    pmax = float(prod[imax])
    pmin = float(prod[imin])
    if pmax <= pmin * (1.0 + 1e-12):
        return None
    donor = int(idx[imax])
    rec = int(idx[imin])
    frac = float(rng.uniform(0.05, 0.40))
    delta = frac * float(a[donor])
    if delta <= 0.0 or delta >= float(a[donor]):
        return None
    a2 = a.copy()
    a2[donor] -= delta
    a2[rec] += delta
    out = _norm(a2)
    return out


def _pair_cand(a, k, rng):
    n = a.size
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    cnt = hi - lo + 1
    if cnt < 2:
        return None

    prod = a[idx] * a[partner]

    t = int(partner[i])

        return None
    a2 = a.copy()
    a2[src] -= delta
    a2[dst] += delta
    out = _norm(a2)
    return out
