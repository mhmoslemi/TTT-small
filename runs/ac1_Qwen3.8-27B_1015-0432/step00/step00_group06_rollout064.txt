import numpy as np
import time


def build_power_shape(n, p):
    x = np.arange(n, dtype=np.float64)
    base = ((x + 1.0) / float(n)) ** p
    base = base / np.max(base)
    a = 1000.0 * base
    a = a + 1e-4
    return np.clip(a, 0.0, 1000.0)


def build_mixed_shape(n, p, freq, rng):
    x = np.arange(n, dtype=np.float64)
    base = ((x + 1.0) / float(n)) ** p
    base = base / np.max(base)
    mod = 1.0 + 0.3 * np.sin(2.0 * np.pi * freq * x / float(n))
    a = 1000.0 * base * mod
    rng = None if rng is None else rng
    return np.clip(a, 0.0, 1000.0)


def block_perturb(base, rng):
    n = base.size
    width = int(rng.integers(max(1, n // 300), max(2, n // 8)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand = base.copy()
    factor = float(np.exp(rng.normal(0.0, 0.15)))
    cand[start:stop] *= factor
    if rng.random() < 0.5:
        cand = cand * float(np.exp(rng.normal(0.0, 0.05)))
    return np.clip(cand, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = 0
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 0
    if n < 1:
        n = 256

    best = None
    best_score = np.inf

    # Parametric non-symmetric power shapes (all within [0, 1000]).
    for p in (1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0):
        a = build_power_shape(n, p)
        s = float(evaluate_sequence(a.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = a.copy()
        if time.monotonic() > deadline:
            break

    # A few sine-modulated variants for diversity.
    for p in (1.5, 2.0, 2.5):
        for freq in (1, 2, 3):
            a = build_mixed_shape(n, p, freq, rng)
            s = float(evaluate_sequence(a.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = a.copy()

    if best is None:
        best = build_power_shape(n, 2.0)
        best_score = float(evaluate_sequence(best.tolist()))
        if not np.isfinite(best_score):
            best = np.ones(n, dtype=np.float64)
            best_score = float(evaluate_sequence(best.tolist()))

    # Local multiplicative refinement to polish the best shape.
    tries = 0
    while time.monotonic() < deadline:
        tries += 1
        cand = block_perturb(best, rng)
        if cand.sum() < 0.01:
            continue
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best = cand

    out = [float(v) for v in np.clip(best, 0.0, 1000.0)]
    sc = float(evaluate_sequence(out))
    if (not np.isfinite(sc)) or sc < 0.01:
        out = [float(v) for v in np.ones(n, dtype=np.float64)]
    return out
