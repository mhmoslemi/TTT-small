  ```python
  import numpy as np
  import time
  
  def compute_grad(a, top_k=10):
      """Compute subgradient of 2*n*max(conv(a,a))/(sum(a))^2."""
      n = len(a)
      c = np.convolve(a, a)  # length 2n-1
      max_b = c.max()
      sum_a = a.sum()
      if sum_a < 0.01:
          return None
      # Top-k indices
      k = min(top_k, len(c))
      idx = np.argsort(c)[-k:]
      weights = np.zeros(len(c))
      for i in idx:
          weights[i] = 1.0 / k
      # Subgradient of max(c) w.r.t. a[i]:
      # d c[m] / d a[i] = a[m-i] if 0 <= m-i < n
      # So g[i] = sum_m weights[m] * a[m-i] (if valid)
      g = np.zeros(n)
      for m in idx:
          for i in range(n):
              j = m - i
              if 0 <= j < n:
                  g[i] += weights[m] * a[j]
      # Full gradient of score:
      # df/da_i = 2n/(sum_a)^2 * (g[i] - 2*max_c/sum_a)
      grad = 2.0 * n / (sum_a**2) * (g - 2.0 * max_b / sum_a)
      return grad
  ```
