import numpy as np
import time
from scipy.fft import fft, ifft


def fft_self_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = a.size
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    A = fft(a, fs)
    return ifft(A * A)[:size].real


def init_step(n, lo, hi):
    """Two-level step function."""
    a = np.full(n, lo)
    a[n // 3:] = hi
    return a


def init_ramp(n):
    t = np.linspace(0.0, 1.0, n)
    return 1.0 + 2.0 * t


def init_bump(n, center_frac, width_frac, height):
    center = int(center_frac * n)

    a[np.minimum(0, center - half):center - half + 1] = np.minimum(0, center - half)  # no-op safety
    lo = max(0, center - half)
    hi = min(n, center + half + 1)
    a[:] = 1.0
    a[lo:hi] = height
    return a
