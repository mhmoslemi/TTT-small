def ippa_step(p, eta):
    c = np.convolve(p, p)
    max_c = c.max()
    # Find all lags within some tolerance of max
    active = np.where(c >= max_c * 0.999)[0]
    # Subgradient w.r.t. p[j]
    s = np.zeros_like(p)
    for k in active:
        # c[k] = sum_i p[i]*p[k-i], so dc[k]/dp[j] = p[k-j] if valid
        lo = max(0, k - (len(p)-1))
        hi = min(k, len(p)-1)
        for j in range(lo, hi+1):
            s[j] += p[k-j]
    s /= len(active)
    # Update: p - eta*s, projected to simplex
    p_new = p - eta * s
    # Project to simplex (nonnegative, sum=1)
    p_new = np.maximum(p_new, 0)
    p_new = p_new / p_new.sum()
    return p_new

4. Periodically verify with evaluate_sequence.
5. Additional random perturbations to escape local optima.

Let me also think about the step size η. For the subgradient method on a max function, η should decrease. But for the constrained version (simplex), we can use a fixed small η and just check for improvement.

def fast_self_conv(p):
    """Compute full self-convolution of p using FFT. O(n log n)."""
    n = len(p)
    out_len = 2 * n - 1
    size = 1
    while size < out_len:
        size <<= 1
    fp = np.fft.rfft(p, size)
    return np.fft.irfft(fp * fp, size)[:out_len]

def project_simplex(v):
    """Project v onto the probability simplex (nonnegative, sum=1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-30:
        return np.ones_like(v) / len(v)
    return v / s
