```python
def subgradient_step(a, n, eta):
    # a is normalized to sum=1
    # Find argmax of autocorrelation
    ac = np.convolve(a, a)  # length 2n-1
    k_star = np.argmax(ac)
    
    # Subgradient: for each i, the contribution to ac[k_star] from a[i] is a[k_star - i] (if valid)
    # We want to decrease a[i] weighted by a[k_star - i]
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = a[j]
    
    # a_new[i] = a[i] - eta * grad[i], then project to simplex
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    
    return a_new
```

def _full_conv_sq(a):
    n = a.size
