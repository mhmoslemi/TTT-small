def convolve_fast(a, b):
    """FFT-based linear convolution, O(n log n) memory."""
    n_a, n_b = len(a), len(b)
    size = 1
    while size < n_a + n_b - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    fc = fa * fb
    result = np.fft.irfft(fc, size)
    return result[:n_a + n_b - 1]

import time


def fft_conv(a, b):
    """Linear convolution via FFT. O(n log n) time and memory."""
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]


def lp_direction(f, M, n, max_inner):
    """Approximate max sum(g) s.t. conv(f,g)<=M, g>=0 via multiplicative updates."""
    g = np.ones(n)
    eta = 1.0 / max(M, 1e-15)
    f_rev = f[::-1]
    for _ in range(max_inner):
        fg = fft_conv(f, g)
        viol = np.maximum(0.0, fg - M)
        vmax = np.max(viol)
        if vmax < 1e-12:
            break
        cross = fft_conv(viol, f_rev)
        grad = cross[n - 1:2 * n - 1]
        grad = np.maximum(0.0, grad)
        g *= np.exp(-np.minimum(eta * grad / max(vmax, 1e-15), 50.0))
        sg = np.sum(g)
        if sg < 1e-15:
            g = np.ones(n)
            sg = n
            break
    return g
