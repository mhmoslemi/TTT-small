def _block_perturb(base, rng):
    n = base.size
    width = int(rng.integers(max(2, n // 200), max(3, n // 10)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    out = base.copy()
    out[start:stop] *= float(np.exp(rng.normal(0.0, 0.10)))
    return np.clip(out, 0.0, 1000.0)


def _local_move(base, rng):
    n = base.size
    out = base.copy()
    i = int(rng.integers(0, n))
    j = int(rng.integers(0, n))
    amt = float(rng.uniform(0.01, 0.5)) * np.mean(base)
