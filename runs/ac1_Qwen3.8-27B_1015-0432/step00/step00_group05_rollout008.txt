import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:2 * n - 1]


def _score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    M = float(np.max(_fft_conv(a)))
    return 2.0 * n * M / (S * S)


def _subgrad(a, k_star):
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return None
    conv = _fft_conv(a)
    M = float(conv[k_star])
    base = -4.0 * n * M / (S * S * S)
    g = np.full(n, base)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    if js <= je:
        jj = np.arange(js, je + 1)
        g[jj] += 4.0 * n * a[k_star - jj] / (S * S)
    return g, conv
