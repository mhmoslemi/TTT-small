Wait, I need to be more careful with the gradient. Let me re-derive:

(a*a)[k] = ∑_{j=0}^{n-1} a[j] * a[k-j] (with a[i]=0 outside [0,n-1])

∂/∂a[i] (a*a)[k]:
- From term j=i: a[k-i]
- From term j=k-i: a[i] (if k-i ≠ i)
- If k-i = i (i.e., k is even and i = k/2): only one term, derivative is a[i]
- If k-i ≠ i: derivative is a[k-i] + a[k-i] = 2*a[k-i]... wait no.

Let me be very careful. (a*a)[k] = ∑_{j} a[j] · a[k-j].

∂/∂a[i]: 
- a[j] appears when j=i: contributes a[k-i]
- a[k-j] appears when k-j=i, i.e., j=k-i: contributes a[j] = a[k-i]

If i ≠ k-i (i.e., 2i ≠ k): both terms contribute, total = a[k-i] + a[k-i] = 2a[k-i]
If i = k-i (i.e., 2i = k, k even): only one term (j=i=k-i), total = a[i] = a[k/2]

So ∂(a*a)[k]/∂a[i] = 2·a[k-i] if 2i≠k, a[k-i] if 2i=k, 0 if k-i outside [0,n-1].

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def build_conv_structure(n):
    rl, cl = [], []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            js = np.arange(lo, hi + 1)
            rl.append(np.full(len(js), k, dtype=np.int32))
            cl.append(js.astype(np.int32))
    return np.concatenate(rl), np.concatenate(cl)


def lp_step(a, M, rows, cols, tlim):
    n = len(a)
    idx = rows - cols
    A = sp.csr_matrix((a[idx], (rows, cols)), shape=(2 * n - 1, n))
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': tlim})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def peak_grad(a, alpha):
    n = len(a)
    conv = np.convolve(a, a)
    mx = float(np.max(conv))
    ak = np.where(np.abs(conv - mx) < 1e-12 * max(1.0, abs(mx)))[0]
    grad = np.zeros(n)
    for k in ak:
        lo = max(0, int(k) - n + 1)
        hi = min(int(k), n - 1)
        idx = np.arange(lo, hi + 1)
        grad[idx] += 2.0 * a[int(k) - idx]
        if int(k) % 2 == 0:
            mid = int(k) // 2
            if lo <= mid <= hi:
                grad[mid] -= a[mid]
    if len(ak) > 0:
        grad /= len(ak)
    new_a = np.maximum(a - alpha * grad, 0.0)
    return new_a, mx

        c[s:s + w] *= float(np.exp(rng.normal(0.0, 0.18)))
