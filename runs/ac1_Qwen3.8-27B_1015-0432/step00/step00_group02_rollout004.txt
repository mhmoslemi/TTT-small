import numpy as np
import time

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def ac1_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    ac = fft_convolve(a, a)
    return float(2 * n * ac.max() / (s * s))

def subgrad_step(a, K, lr):
    """One subgradient step to reduce the top-K autocorrelation peaks."""
    n = len(a)
    ac = fft_convolve(a, a)
    # Find top K lags
    if K == 1:
        peaks = [int(np.argmax(ac))]
    else:
        peaks = np.argsort(ac)[-K:].tolist()
    
    # Compute subgradient
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += a[k - js]  # factor of 2 is implicit (both sides)

    else:

import numpy as np
import time

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]

def ac1_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    ac = fft_convolve(a, a)
    return float(2.0 * n * ac.max() / (s * s))

def topk_subgrad(a, K, topk_frac):
    """Return top-K subgradient of max autocorrelation."""
    n = len(a)
    ac = fft_convolve(a, a)
    K = min(K, 2*n - 1)
    if K == 1:
        idx = int(np.argmax(ac))
        peaks = [idx]
    else:
        peek = ac.copy()
        peaks = []
        for _ in range(K):
            k = int(np.argmax(peek))
            peaks.append(k)
            peek[k] = -1.0
    grad = np.zeros(n)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        grad[js] += a[k - js]
    return grad, ac

def optimize_local(a0, max_iters, deadline):

    s = a.sum()
    if s > 0:

    best_a = a.copy()
    best_sc = ac1_score(a)

    K = 5

            break

        a = a - lr * grad
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-15:
