import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def convolve(a, b):
    """Linear convolution using FFT for efficiency."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def make_toeplitz_sparse(f):
    """Create the (2n-1) x n sparse Toeplitz matrix for linear convolution with f."""
    n = len(f)
    # Row k has nonzeros at columns j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        for j in range(j_start, j_end+1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    # Actually: j_lo = max(0, k-n+1), j_hi = min(n-1, k), count = j_hi - j_lo + 1
    total_nnz = 0
    for k in range(m):
        j_lo = max(0, k - n + 1)
        cnt = min(n - 1, k) - j_lo + 1
        indptr[k + 1] = total_nnz + cnt
        total_nnz += cnt
    indices = np.empty(total_nnz, dtype=np.int64)
    pos = 0
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        indices[pos:pos+cnt] = np.arange(j_lo, j_hi + 1)
        pos += cnt
    return indptr, indices

def make_conv_matrix_csr(f, indptr, indices):
    data = f[indptr[1:] - ... ]  # need to be careful
    # For row k, the entries are at columns indices[indptr[k]:indptr[k+1]]
    # The values are f[k - indices[indptr[k]:indptr[k+1]]]
    # So data = f[arange(m)[:, None] - indices]... but that's dense.
    # Better:
    data = np.empty_like(indices, dtype=np.float64)
    for k in range(len(indptr) - 1):
        sl = slice(indptr[k], indptr[k+1])
        data[sl] = f[k - indices[sl]]
    return sparse.csr_matrix((data, indices, indptr), shape=(len(f)*2-1, len(f)))

import numpy as np
from scipy import sparse

def linear_convolution(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n]


def build_conv_indices(n):
    m = 2 * n - 1
    parts_r = []
    parts_c = []
    for k in range(m):
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        cnt = j_hi - j_lo + 1
        parts_r.append(np.full(cnt, k, dtype=np.int64))
        parts_c.append(np.arange(j_lo, j_hi + 1, dtype=np.int64))
    return np.concatenate(parts_r), np.concatenate(parts_c)


def make_conv_matrix(f, rows, cols):
    n = len(f)
    return sparse.csr_matrix((f[rows - cols], (rows, cols)), shape=(2 * n - 1, n))
