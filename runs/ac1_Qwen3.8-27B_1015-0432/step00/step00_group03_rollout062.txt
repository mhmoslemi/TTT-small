import numpy as np
import time
from scipy import sparse
from scipy.optimize import linprog


def build_conv_matrix(a, n):
    """Build sparse (2n-1) x n matrix: A[k,i] = a[k-i] if 0<=k-i<n."""
    offsets = list(range(n))
    diags = [np.full(n, a[d]) for d in range(n)]
    return sparse.diags(diags, offsets=offsets, shape=(2 * n - 1, n), format='csr')


def solve_direction(a, M, n, deadline):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    A = build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    t_left = deadline - time.monotonic()
    tl = max(2.0, min(20.0, t_left * 0.6))
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'time_limit': tl})
    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None
