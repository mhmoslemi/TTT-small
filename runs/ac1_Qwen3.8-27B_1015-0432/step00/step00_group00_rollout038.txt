```python
import numpy as np
import time

def fft_convolve_full(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficiency
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    conv = np.fft.irfft(fa * fa, fsize)[:size]
    return conv

def simplex_project(a):
    """Project onto the probability simplex (nonnegative, sum=1)."""
    n = len(a)
    u = np.sort(a)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(a - theta, 0)
```

```python
def fft_conv(a):
    n = len(a)
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def proj_simplex(v):
    """Project v onto {x: x >= 0, sum(x) = 1}."""
    n = len(v)

    theta = (cssv[rho] - 1) / (rho + 1.0)
```

```python
import numpy as np
import time

def _fft_conv_full(a, b):
    """Full linear convolution via FFT. O(n log n)."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```
