def fast_autocorr(a):
    n = a.size
    size = 2 * n - 1
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)


def subgrad_at_lag(a, k):
    n = a.size
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(k, n - 1)
    if js <= je:
        m = np.arange(js, je + 1)
        g[m] = 2.0 * a[k - m]
    return g
