import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fft_pow2(size):
    m = 1
    while m < size:
        m <<= 1
    return m


def _smooth_max(c, tau):
    cm = np.max(c)
    e = np.exp((c - cm) / tau)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smooth_max(np.convolve(a, a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    mv, p = _smooth_max(np.convolve(a, a), tau)
    pr = p[::-1]
    L = _fft_pow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _local_search(bv, sc, rng, n, deadline):
    while time.monotonic() < deadline:
        cd = bv.copy()
        w = max(1, n // 80)
        s = int(rng.integers(0, n - w))
        cd[s:s + w] *= float(np.exp(rng.normal(0.0, 0.04)))
        if rng.random() < 0.3:
            cd *= float(np.exp(rng.normal(0.0, 0.01)))
        cd = np.clip(cd, 0.0, 1000.0)
        s2 = float(evaluate_sequence(cd.tolist()))
        if np.isfinite(s2) and s2 < sc:
            bv = cd.copy()
            sc = s2
    return bv, sc

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 5.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256, dtype=np.float64)
    n = par.size
    ps = par.sum()
    if ps < 1e-8:
        par = np.ones(n, dtype=np.float64)
        ps = n
    a0 = par / ps

    best = a0.copy()
    best_sc = float(evaluate_sequence(best.tolist()))

    cur = a0.copy()
    taus = [2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]

    for tv in taus:
        if time.monotonic() > dl - 60.0:
            break
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B',
                           bounds=[(0.0, 1000.0)] * n,
                           options={'maxiter': 200, 'gtol': 1e-8,
                                     'maxls': 30, 'ftol': 1e-14})
            if res.fun < _obj(cur, tv) - 1e-12:
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = nx / nx.sum()
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
        except Exception:
            pass

    best, best_sc = _local_search(best, best_sc, rng, n, dl - 3.0)
    return [float(v) for v in best]
