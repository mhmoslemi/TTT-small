def fast_conv(h):
    """Compute np.convolve(h, h) using FFT. Returns array of length 2n-1."""
    n = len(h)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    fh = np.fft.rfft(h, fft_size)
    return np.fft.irfft(fh * fh, fft_size)[:size]

def fast_eval(h):
    """Fast version of evaluate_sequence using FFT."""
    h = np.maximum(0.0, np.minimum(1000.0, h))

    if S < 0.01:
