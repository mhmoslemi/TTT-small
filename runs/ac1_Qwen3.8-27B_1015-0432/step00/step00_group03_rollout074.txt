import numpy as np
import time

def _fast_len(m):
    m = max(1, int(m))
    n = 1
    while n < m:
        n <<= 1
    return n

def _conv_self(x, N):
    F = np.fft.rfft(x, N)
    h = np.fft.irfft(F * F, N)
    h = h[:2*x.size-1]
    if h.size:
        h = np.maximum(h, 0.0)
    return h, F

def _corr_weights(p, Ff, N, n):
    q = np.fft.irfft(np.fft.rfft(p, N) * np.conjugate(Ff), N)
    return np.maximum(q[:n], 0.0)

def _score_from_h(h, n, S):
    mh = float(np.max(h)) if h.size else 0.0
    if S <= 0.0 or not np.isfinite(mh):
        return float('inf')
    return float(2.0 * n * mh / (S * S))

def _quick_score(x, N):
    S = float(np.sum(x))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    h, _ = _conv_self(x, N)
    return _score_from_h(h, x.size, S)

def _normalize_positive(x, S):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    inf_mask = ~np.isfinite(arr)
    if inf_mask.any():
        arr = arr.copy()
        arr[inf_mask] = 0.0
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000000.0)
    floor = 1e-6 * float(S) / float(arr.size)
    arr = arr + floor
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones_like(arr)
        s = float(arr.size)
    arr = arr * (float(S) / s)

import numpy as np
import time


def _fast_len(m):
    m = max(1, int(m))
    n = 1
    while n < m:
        n <<= 1
    return n


def _conv_self(x, N):
    F = np.fft.rfft(x, N)
    h = np.fft.irfft(F * F, N)
    h = h[:2 * x.size - 1]
    if h.size:
        h = np.maximum(h, 0.0)
    return h, F


def _corr_weights(p, Ff, N, n):
    q = np.fft.irfft(np.fft.rfft(p, N) * np.conjugate(Ff), N)
    return np.maximum(q[:n], 0.0)


def _score_from_h(h, n, S):
    mh = float(np.max(h)) if h.size else 0.0
    if S <= 0.0 or not np.isfinite(mh):
        return float("inf")
    return float(2.0 * n * mh / (S * S))


def _quick_score(x, N):
    S = float(np.sum(x))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    h, _ = _conv_self(x, N)
    return _score_from_h(h, x.size, S)


def _normalize_positive(x, S):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    bad = ~np.isfinite(arr)
    if bad.any():
        arr = arr.copy()
        arr[bad] = 0.0
    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1.0e6)
    floor = 1.0e-6 * float(S) / float(arr.size)
    arr = arr + floor
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 0.0:
        arr = np.ones_like(arr)
        s = float(arr.size)
    return arr * (float(S) / s)


def _bank_key(item):
    return item[1]


def _bank_add(bank, arr, score):
    if not np.isfinite(score):
        return
    bank.append((arr.copy(), float(score)))
    if len(bank) > 32:
        worst_i = 0
        worst_s = bank[0][1]
        for i in range(1, len(bank)):
            if bank[i][1] > worst_s:
                worst_s = bank[i][1]
                worst_i = i
        bank.pop(worst_i)


def _perturb(f, S, rng, sigma):
    g = f * np.exp(rng.normal(0.0, float(sigma), f.size))
    return _normalize_positive(g + 1.0e-4 * f, S)


def _param_search(n, S, N, rng, deadline, bank):

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    start = time.monotonic()
    budget = max(0.0, float(budget_s))
    return_deadline = start + max(0.0, budget - 3.0)
    search_deadline = start + max(0.0, budget - 30.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    if parent.ndim != 1 or parent.size == 0:
        parent = np.ones(256, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    n = int(parent.size)

    S = 100.0
    N = _fast_len(2 * n - 1)
    bank = []

    p0 = _normalize_positive(parent, S)
    s0 = _quick_score(p0, N)
    _bank_add(bank, p0, s0)

    if n == 1:
        return [float(x) for x in p0]

    if time.monotonic() < search_deadline:
        param_deadline = min(search_deadline, start + 120.0)
        if param_deadline > start:
            best_arr, _ = _param_search(n, S, N, np.random.default_rng(int(seed) + 1), param_deadline, bank)
        else:
            best_arr = p0
    else:
        best_arr = p0

    if bank:
        ordered = sorted(bank, key=_bank_key)
        best_arr = ordered[0][0]

    if time.monotonic() < search_deadline - 1.0:
        _optimize(best_arr, S, N, search_deadline, np.random.default_rng(int(seed) + 2), bank)

    candidates = [p0]
    if bank:
        ordered = sorted(bank, key=_bank_key)
        limit = 4
        for i in range(min(limit, len(ordered))):
            candidates.append(ordered[i][0])

    best_list = [float(x) for x in p0]
    best_exact = float("inf")

    for arr in candidates:
        if time.monotonic() > return_deadline:
            break
        a = np.asarray(arr, dtype=np.float64)
        if a.size != n:
            continue
        a = np.clip(a, 0.0, 1000.0)
        if not np.all(np.isfinite(a)):
            continue
        if float(a.sum()) < 0.01:
            a = np.full(n, 1.0)
        lst = [float(x) for x in a]
        sc = float(evaluate_sequence(lst))
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_list = lst

    if not np.isfinite(best_exact):
        best_list = [1.0] * n

    return best_list
