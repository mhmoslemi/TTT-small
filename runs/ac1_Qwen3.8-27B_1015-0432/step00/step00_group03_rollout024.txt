def _build_conv_mat(f, n):
    """Sparse (2n-1) x n matrix C s.t. C @ g = full_convolve(f, g)."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    vals = np.empty(total, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        c = hi - lo + 1
        rows[idx:idx+c] = k
        ia = np.arange(lo, hi + 1)
        cols[idx:idx+c] = ia
        vals[idx:idx+c] = f[k - ia]
        idx += c
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

    """Solve LP: max sum(g) s.t. C@f... wait, C@ g <= M, g >= 0.

                  options={'time_limit': 30, 'presolve': True})
