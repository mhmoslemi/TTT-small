import numpy as np
import time


def conv_self(a):
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n < 256:
        c = np.convolve(a, a, mode="full")
    else:
        target = 2 * n - 1
        size = 1
        while size < target:
            size <<= 1
        f = np.fft.rfft(a, size)
        c = np.fft.irfft(f * f, n=target)
    if c.size and c.min() < 0.0:
        c = np.maximum(c, 0.0)
    return c


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0 or not np.all(np.isfinite(a)):
        return None
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def score_normalized(p):
    c = conv_self(p)
    if c.size == 0:
        return float("inf")
    return float(2.0 * p.size * float(c.max()))


def append_candidate(cands, max_keep, p, score):
    if p is None or not np.isfinite(score):
        return
    cands.append((float(score), p))
    while len(cands) > max_keep:
        worst = 0
        for i in range(1, len(cands)):
            if cands[i][0] > cands[worst][0]:
                worst = i
        del cands[worst]


def pick_top(triples, k):
    res = []
    for _ in range(min(k, len(triples))):
        bi = 0
        bv = triples[0][0]
        for i in range(1, len(triples)):
            if triples[i][0] < bv:
                bv = triples[i][0]
                bi = i
        res.append(triples.pop(bi))
    return res


def range_x(n):
    return (np.arange(n, dtype=np.float64) + 0.5) / float(n)

    x = range_x(n)
    return 1.0 + a * x

    x = range_x(n)
    t = x * (1.0 - x)
    return 1.0 + a * x + b * t + c * t * (2.0 * x - 1.0)

    x = range_x(n)
    z = a * x + b * x * x
    z = z - float(np.max(z))
    return np.exp(z)


def gen_random_smooth(n, rng):
    x = range_x(n)

        amp = float(rng.uniform(-1.0, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z = z + amp * np.cos(2.0 * np.pi * k * x + ph)
    z = z - float(np.max(z))
    return np.exp(0.8 * z)


def perturb_normalize(p, rng, amp):
    if p.size == 0:
        return p
    spread = float(np.max(p) - np.mean(p))
    if not np.isfinite(spread) or spread <= 1e-300:
        spread = 1.0 / float(p.size)
    q = p + amp * spread * rng.standard_normal(p.size)
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return p.copy()
    return q / s


def gradient_step(p, count, step_frac):
    n = int(p.size)
    c = conv_self(p)
    if c.size == 0:
        return None, c, 0.0
    m = float(c.max())
    if not np.isfinite(m) or m <= 0.0:
        return None, c, m

    count = int(count)
    if count <= 0:
        count = 1
    if count >= c.size:
        active = np.argsort(c)[-count:]
    else:
        active = np.argpartition(c, -count)[-count:]

    g = np.zeros(n, dtype=np.float64)
    w = float(active.size)
    for kk in active:
        k = int(kk)
        lo = k - (n - 1)
        if lo < 0:
            lo = 0
        hi = k
        if hi > n - 1:
            hi = n - 1
        if hi >= lo:
            start = k - lo
            stop = k - hi - 1
            g[lo:hi + 1] += 2.0 * p[start:stop:-1]
    g /= max(1.0, w)

    d = g - float(np.mean(g))
    scale = float(np.max(np.abs(d)))
    if not np.isfinite(scale) or scale <= 1e-300:
        return None, c, m

    eta = step_frac * (float(np.mean(p)) + 1e-300) / scale
    q = p - eta * d
    q = np.clip(q, 0.0, None)
    s = float(q.sum())
    if not np.isfinite(s) or s <= 1e-300:
        return None, c, m
    return q / s, c, m
