    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Next power of 2
    import numpy as np
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a, fsize)
    C = A * A
    c = np.fft.irfft(C, fsize)[:size]
    return c

def _fft_conv(a, fsize):
    """Full linear self-convolution via FFT."""
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)[:2*len(a)-1]

def _grad(a, p, fsize):
    """Gradient of <p, c> w.r.t. a, where c = conv(a,a)."""
    n = len(a)
