```python
def fft_convolve(x):
    n = len(x)
    N = 1
    while N < 2*n - 1:
        N *= 2
    X = np.fft.rfft(x, N)
    return np.fft.irfft(X * X, N)[:2*n-1]

def simplex_proj(v, s=1.0):
    # Project v onto {x >= 0, sum(x) = s}
    n = len(v)
    if n == 0:
        return v
    # Binary search for the offset
    v_sorted = np.sort(v)[::-1]
    csum = np.cumsum(v_sorted)
    rho = np.nonzero(v_sorted * np.arange(1, n+1) > (csum - s))[0]
    if len(rho) == 0:
        return np.full(n, s/n)
    rho = rho[-1]
    theta = (csum[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```
