```python
import numpy as np
import time

def compute_autocorr(a):
    """Compute linear autocorrelation using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    # Pad to next power of 2 for efficient FFT
    fsize = 1
    while fsize < size:
        fsize *= 2
    A = np.fft.rfft(a, fsize)
    B = np.fft.irfft(A * A, fsize)[:size]
    return B

def autocorr_gradient_at(a, k):

    n = len(a)
```

This is a feasibility LP (or minimax LP) with $n$ variables and $2n-1$ constraints. For $n = 3809$, this is manageable with a modern LP solver.

```python
import numpy as np
import time


def _autocorr(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    F = np.fft.rfft(a, fsize)
    return np.fft.irfft(F * F, fsize)[:size]


def _grad_lag(a, k):
    n = a.size
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        g[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] += a[np.arange(lo2, hi2 + 1) - k]
    return g

    best_peak = float(np.max(_autocorr(best)))

    # --- main loop ---

    top = min(3, 2 * n - 1)
    stall = 0

    while time.monotonic() < deadline:

        peak = float(b.max())

        # Weighted gradient over top lags
        idx = np.argpartition(-b, top)[:top]
        g = np.zeros(n)
        for k in idx:
            w = b[k] / b[idx[0]] if b[idx[0]] > 0 else 0.0
            g += w * _grad_lag(a, int(k))

        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        g /= gn

        # Remove component along 1 (tangent to simplex)
        g -= g.mean()
        g /= max(np.linalg.norm(g), 1e-15)

        # Backtracking
        alpha = lr
        ok = False
        for _ in range(12):
            a_new = a - alpha * g
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-12:
                alpha *= 0.5
                continue
            a_new /= s
            p_new = float(np.max(_autocorr(a_new)))
            if p_new < peak * (1.0 - 1e-7):
                a = a_new
                ok = True
                break
            alpha *= 0.5

        if ok:
            stall = 0
            if p_new < best_peak * (1.0 - 1e-5):
                best_peak = p_new
                best = a.copy()
                # verify with evaluator every so often
                sc = float(evaluate_sequence(a.tolist()))
                if sc < best_score:
                    best_score = sc
            lr = min(lr * 1.03, 20.0 / n)
        else:
            stall += 1
            lr *= 0.85
            if stall > 150:
                # escape: perturb
                a = best + rng.normal(0.0, 0.02, n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                lr = 2.0 / n
                stall = 0
            elif stall > 50:
                a = best + rng.normal(0.0, 0.005, n)
                a = np.maximum(a, 0.0)
                a /= a.sum()

    # --- final check ---
    sc = float(evaluate_sequence(best.tolist()))
    if sc < best_score:
        best_score = sc
```
