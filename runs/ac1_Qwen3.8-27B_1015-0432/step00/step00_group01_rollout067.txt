def _score(a):
    n = len(a)
    b = np.convolve(a, a)
    max_b = np.max(b)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def _subgrad_step(a, peak_idx, lr):
    n = len(a)
    grad = np.zeros(n)
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    for j in range(lo, hi + 1):
        k = peak_idx - j
        grad[j] += a[k]
    norm = np.linalg.norm(grad)
    if norm > 1e-15:
        grad /= norm
    return a - lr * grad
