import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
from scipy.sparse import coo_matrix, csr_matrix
import time

def build_conv_constraint_matrix(a, lags):
    """Build sparse constraint matrix for convolution constraints."""
    n = len(a)
    rows, cols, vals = [], [], []
    for idx, k in enumerate(lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            rows.extend([idx] * (hi - lo))
            cols.extend(range(lo, hi))
            vals.extend(a[lo:hi])
    if not rows:
        return csr_matrix((len(lags), n))
    M = coo_matrix((vals, (rows, cols)), shape=(len(lags), n))
    return M.tocsr()

def lp_direction(a, M, K, deadline):
    """Solve reduced LP to find direction."""
    n = len(a)
    conv = fftconvolve(a, a)
    K = min(K, 2*n - 1)
    
    # Start with top K lags
    top_k = np.argpartition(conv, -K)[-K:]
    
    for _ in range(3):  # iterate to add violated constraints
        A = build_conv_constraint_matrix(a, top_k)
        c_obj = -np.ones(n)
        b_ub = M * np.ones(len(top_k))
        
        remaining = deadline - time.monotonic()
        if remaining < 5:
            return None
        
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                     bounds=[(0, None)] * n, method='highs',
                     options={'time_limit': min(30, remaining * 0.3),
                              'presolve': True})
        
        if not res.success or res.x is None:
            return None
        
        d = np.maximum(res.x, 0)
        
        # Check all constraints
        conv_d = fftconvolve(a, d)
        viol = np.where(conv_d > M * 1.001)[0]

            return d
        else:

            new_lags = np.unique(np.concatenate([top_k, viol]))
            if len(new_lags) > K:

                    keep.add(v)
