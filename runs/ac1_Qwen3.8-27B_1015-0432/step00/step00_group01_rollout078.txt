    b = np.asarray(a, dtype=np.float64).ravel()

                cand = np.clip(cand, 0.0, 1000.0)

                if s > 1e-12:

            cand = _exp_norm(logbase + 0.02 * rng.standard_normal(n))
            if cand is not None:

            cur_score = float(scores[idx])

            eta = 0.1

            continue

            best = best_cand.copy()
            best_score = best_cand_sc
            eta = min(0.8, eta * 1.08)
            beta = min(0.5 * n2, beta * 1.05)

        diff = best_cand_sc - cur_score
        accept = False
        if diff < 0.0:
            accept = True
        elif temp > 1e-7:
            val = -diff / max(temp, 1e-12)
            if val < -50.0:
                val = -50.0
            if val > 0.0:
                val = 0.0
            if float(rng.random()) < float(np.exp(val)):
                accept = True

        if accept:
            cur = best_cand.copy()
            cur_score = best_cand_sc
            reject = 0
        else:
            reject += 1
            if reject > 200:
                idx = int(rng.integers(len(arrs)))
                cur = arrs[idx].copy()
                cur_score = float(scores[idx])
                eta = 0.08 + 0.1 * float(rng.random())

            if cur is None:
