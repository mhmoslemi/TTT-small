        c_tr = fft_convolve(a_trial, a_trial)
        tr_score = 2.0 * n * c_tr.max() / (sa * sa)

        if tr_score < cur_score:
            a = a_trial
            cur_score = tr_score
            lr = min(lr * 1.03, 3.0)
        else:
            lr *= 0.55
            if lr < 1e-12:
                lr = 0.2
                T = max(T * 0.5, 0.05)
            continue

        if it % 300 == 0 and it > 0:
            T = max(T * 0.85, 0.02)

        if stall_count > 30 and it % 200 == 50:
            a = a + 0.01 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
            cur_score = 2.0 * n * np.max(fft_convolve(a, a)) / np.sum(a) ** 2
            T = max(T * 1.5, 1.0)

    s = float(evaluate_sequence(a.tolist()))
    if np.isfinite(s) and s < best_score:
        best = a.copy()
        best_score = s

    if not np.isfinite(best_score) or best_score > 2.0:
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    return [float(v) for v in best]
