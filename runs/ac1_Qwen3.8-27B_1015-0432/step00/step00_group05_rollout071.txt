```python
def build_A(f):
    n = len(f)
    # First half: k=0..n-1, j=0..k
    rows1 = np.concatenate([np.full(k+1, k) for k in range(n)])
    cols1 = np.concatenate([np.arange(k+1) for k in range(n)])
    # Second half: k=n..2n-2, j=k-n+1..n-1
    rows2 = np.concatenate([np.full(2*n-1-k, k) for k in range(n, 2*n-1)])
    cols2 = np.concatenate([np.arange(k-n+1, n) for k in range(n, 2*n-1)])
    rows = np.concatenate([rows1, rows2]).astype(np.int32)
    cols = np.concatenate([cols1, cols2]).astype(np.int32)
    vals = f[cols].astype(np.float64)
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
def _fft_conv(h):
    n = len(h)
    size = 2 * n - 1
    m = 1
    while m < size:
        m <<= 1
    fh = np.fft.rfft(h, m)
    return np.fft.irfft(fh * fh, m)[:size]
```

```python
        if np.isfinite(fb_s):
            return fb
        return [1.0] * max(1, min(n, 64))
    return cand
```
