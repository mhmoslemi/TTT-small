import numpy as np
import scipy.sparse
import scipy.optimize
import time


def build_conv_matrix(a, n):
    """Build sparse (2n-1) x n matrix A where A[k,j] = a[k-j] if 0<=k-j<n."""
    # Diagonal d (0<=d<n) is constant a[d], length n
    diags = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    return scipy.sparse.diags(diags, offsets, shape=(2*n-1, n), format='csr')

        return None
