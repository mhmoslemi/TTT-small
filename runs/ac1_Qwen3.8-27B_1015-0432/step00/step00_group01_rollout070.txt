import time

def _fft_conv_sq(a):
    n = len(a)
    sz = 1
    while sz < 2*n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa * fa, sz)[:2*n-1]

def _score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-6:
        return np.inf
    b = _fft_conv_sq(a)
    return 2.0 * n * b.max() / (s * s)

    n = len(a)
    s = a.sum()

    fa = np.fft.rfft(a, sz)
    return np.fft.irfft(fa*fa, sz)[:2*n-1]

    n = len(a)
    s = a.sum()
    if s < 1e-6:
        return np.inf
    b = _conv_sq(a)
    return 2.0*n*b.max()/(s*s)

    while sz < 2 * n - 1:
