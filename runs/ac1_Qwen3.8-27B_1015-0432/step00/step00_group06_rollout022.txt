def _exp_smooth_grad(a, b, tau):
    """Gradient of smooth max with temperature tau."""
    n = a.size
    # Shift b for numerical stability
    bshift = b - b.max()
    weights = np.exp(tau * bshift)
    weights /= weights.sum()
    # grad = sum_k weights[k] * 2 * a_reversed_at_k
    grad = np.zeros(n)
    for k in range(len(b)):
        w = weights[k]
        if w < 1e-12:

        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        grad[lo:hi] += w * a[k - lo:k - hi + 1][::-1]
    grad *= 2.0
    return grad
