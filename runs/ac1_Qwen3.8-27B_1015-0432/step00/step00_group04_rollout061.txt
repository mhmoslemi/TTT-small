def simplex_project(x, target_sum=1.0):
    """Project x onto {a >= 0, sum(a) = target_sum}."""
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s < 1e-15:
        n = x.size
        return np.full(n, target_sum / n)
    return x * (target_sum / s)

def grad_peak(a, k_star):
    """Gradient of C[k_star] w.r.t. a where C=convolve(a,a)."""
    n = a.size
    grad = np.zeros(n)
    i_lo = max(0, k_star - n + 1)
    i_hi = min(n - 1, k_star)

        idx_hi = n - 1 - k_star + i_hi
        grad[i_lo:i_hi+1] = 2.0 * a_rev[idx_lo:idx_hi+1]
    return grad

def simplex_step(a, g, lr):
    """Projected gradient step onto {x>=0, sum=1}."""
    a_new = a - lr * g
    np.maximum(a_new, 0.0, out=a_new)
    s = a_new.sum()
    if s < 1e-15:
        a_new.fill(1.0 / a_new.size)
    else:
        a_new /= s
    return a_new

    """Return (max, argmax) of convolve(a,a)."""
    C = np.convolve(a, a)
    M = C.max()
    k = int(C.argmax())

C[k] = sum_i a[i]*a[k-i]. Note that C[n-1] = sum_i a[i]*a[n-1-i].

We know sum a = sum b = S. By Cauchy-Schwarz: a·b ≤ |a||b| = |a|². But we want a LOWER bound.

The minimum of a·b over non-negative a, b with fixed sums S is 0 (achieved when a and b have disjoint support). But a and b are the same sequence (reversed), so they can't have fully disjoint support.

The minimum of a·b where b = a_reversed: this is the minimum of sum a[i]*a[n-1-i].

For n=2: a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]. Min with a[0]+a[1]=S: 0 (when one is 0).
C[1] = 2*a[0]*a[1]. Can be 0.

Min of 2a[0]a[2]+a[1]² with a[0]+a[1]+a[2]=S: set a[0]=a[2]=0, a[1]=S: C[2]=S².
