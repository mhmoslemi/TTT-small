    if order == 1:
        return t + 1.0

        return t * t + 1.0
    elif order == 3:
        return t * t * t + 1.0
    else:
        return np.cos(np.pi * t / (2.0 * n)) ** 2 + 0.01


def _project(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:

    else:
        a = np.ones_like(a) / a.size
    return a
