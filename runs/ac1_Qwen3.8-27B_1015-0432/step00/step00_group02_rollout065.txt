def _conv_peak(a):
    n = a.size
    sz = 1
    while sz < 2 * n - 1:
        sz <<= 1
    fa = np.fft.rfft(a, sz)
    c = np.fft.irfft(fa * fa, sz)[:2 * n - 1]
    return c.max()
