import numpy as np
import time


def _fft_conv(a):
    """Full linear self-convolution via FFT, O(n log n)."""
    n = len(a)
    m = 2 * n - 1
    fs = 1 << (m - 1).bit_length()
    af = np.fft.rfft(a, fs)
    return np.maximum(np.fft.irfft(af * af, fs)[:m], 0.0)


def _ac1(a):
    """Fast AC1 score (matches evaluator for nonnegative finite a)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * _fft_conv(a).max() / (s * s)


def _subgrad(b, a, n):
    """Subgradient of max(b) w.r.t. a using active lags."""
    max_b = b.max()
    tol = max_b * 1e-3 + 1e-15
    active = np.where(b >= max_b - tol)[0]
    g = np.zeros(n)
    for k in active:

    g /= len(active)
