def next_pow2(m):
    if m<=1: return 1

def fft_convolve(x,y):
    m=len(x)+len(y)-1
    size=next_pow2(m)

def active_gradient(a, c, K=8, q=4.0):
    n=a.size
    L=len(c)
    if K>L:
       idx=np.arange(L)

import numpy as np
import time


def next_power_of_two(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fft_convolve(x, y):
    out_len = int(x.size + y.size - 1)
    if out_len <= 0:
        return np.zeros(0, dtype=np.float64)
    size = next_power_of_two(out_len)
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    return np.fft.irfft(fx * fy, size)[:out_len]


def self_conv(a):
    out_len = int(2 * a.size - 1)
    size = next_power_of_two(out_len)
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa * fa, size)[:out_len]


def clean_normalize(x):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        arr = np.ones(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-12:
        arr = np.ones(arr.size, dtype=np.float64)
        s = float(arr.sum())
    return arr / s, s


def normalize_score(x):
    arr = np.asarray(x, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return None, float("inf")
    arr = arr / s

    m = float(np.max(c))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return arr, 2.0 * float(arr.size) * m


def active_direction(a, c, K=20, q=3.0):
    n = int(a.size)
    L = int(c.size)
    if n <= 0 or L <= 0:
        return np.zeros(max(1, n), dtype=np.float64)
    K = int(min(K, L))
    if K <= 0:
        return np.zeros(n, dtype=np.float64)

    if K < L:
        idx = np.argpartition(c, -K)[-K:]
    else:
        idx = np.arange(L)

    vals = c[idx]
    mx = float(np.max(vals))
    if not np.isfinite(mx) or mx <= 0.0:
        return np.zeros(n, dtype=np.float64)

    cn = vals / mx
    w = cn ** (q - 1.0)
    ws = float(np.sum(w))
    if not np.isfinite(ws) or ws <= 0.0:
        w = np.full(K, 1.0 / K, dtype=np.float64)
    else:
        w = w / ws

    G = np.zeros(n, dtype=np.float64)
    for pos in range(K):
        k = int(idx[pos])
        wi = float(w[pos])
        if wi == 0.0:
            continue
        jlo = k - n + 1
        if jlo < 0:
            jlo = 0
        jhi = k if k < n else n - 1
        if jlo <= jhi:
            start = k - jhi
            stop = k - jlo + 1
            G[jlo:jhi + 1] += 2.0 * wi * a[start:stop][::-1]
    return G


def smooth_direction(a, c, p=4.0):
    n = int(a.size)
    if n <= 0 or c.size <= 0:
        return np.zeros(max(1, n), dtype=np.float64)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return np.zeros(n, dtype=np.float64)
    cn = c / M
    w = p * (cn ** (p - 1.0))
    w = np.where(np.isfinite(w), w, 0.0)
    r = fft_convolve(a, w[::-1])
    G = 2.0 * r[n - 1:2 * n - 1][::-1]
    return G


def combined_gradient(a, c, use_smooth, p_val, K=20, q=3.0):
    n = int(a.size)
    z = np.zeros(n, dtype=np.float64)
    Ga = active_direction(a, c, K=K, q=q)
    maxa = float(np.max(np.abs(Ga))) if Ga.size else 0.0
    if not np.isfinite(maxa) or maxa <= 0.0:
        maxa = 0.0

    Gs = z
    maxs = 0.0
    if use_smooth or maxa <= 1e-12:
        Gs = smooth_direction(a, c, p=p_val)
        maxs = float(np.max(np.abs(Gs))) if Gs.size else 0.0
        if not np.isfinite(maxs) or maxs <= 0.0:
            maxs = 0.0

    if maxa > 0.0 and maxs > 0.0:
        G = 0.35 * (Ga / maxa) + 0.65 * (Gs / maxs)
    elif maxa > 0.0:
        G = Ga / maxa
    elif maxs > 0.0:
        G = Gs / maxs
    else:
        G = z
    G = np.where(np.isfinite(G), G, 0.0)
    return G


def random_move(a, rng, strength=0.2):
    n = int(a.size)
    if n <= 0:
        return np.ones(1, dtype=np.float64)
    mode = int(rng.integers(0, 4))
    st = float(strength)

    if mode == 0:
        sig = st * float(rng.uniform(0.4, 2.0))
        v = a * np.exp(rng.normal(0.0, sig, n))
    elif mode == 1:
        sig = st * float(rng.uniform(0.4, 2.0))
        v = a + (sig / float(max(1, n))) * rng.standard_normal(n)
    elif mode == 2:
        v = a.copy()
        nblocks = int(rng.integers(4, 13))
        block = max(1, n // max(1, nblocks))
        for _ in range(nblocks):
            length = int(rng.integers(1, block + 1))
            if length > n:
                length = n
            if length <= 0:
                continue
            start = int(rng.integers(0, n - length + 1))
            end = start + length
            if end > start:
                v[start:end] *= float(np.exp(rng.normal(0.0, 0.4)))
    else:
        t = np.linspace(0.0, 1.0, n)
        f = int(rng.integers(1, 7))
        ph = float(rng.uniform(0.0, 6.283185307179586))
        mix = float(rng.uniform(0.2, 0.9))
        w = 1.0 + 0.7 * np.sin(2.0 * np.pi * float(f) * t + ph)
        w = np.where(w > 0.01, w, 0.01)
        v = (1.0 - mix) * a + mix * w

    v, _ = clean_normalize(v)
    return v


def add_start(starts, v):
    x, _ = clean_normalize(v)
    if x.size > 0 and np.isfinite(x).all():
        starts.append(x)


def make_starts(parent, n, rng):
    starts = []
    add_start(starts, parent)

    t = np.linspace(0.0, 1.0, n)
    for eps in (0.2, 0.5, 1.0, -0.5):
        add_start(starts, 1.0 + float(eps) * (2.0 * t - 1.0))
    for beta in (1.0, 2.5, -1.0, -2.5):
        add_start(starts, np.exp(float(beta) * t))

    for f in (1, 2, 3):
        for ph in (0.0, 1.5707963267948966):
            add_start(starts, 1.0 + 0.30 * np.sin(2.0 * np.pi * float(f) * t + ph))
            add_start(starts, 1.0 + 0.20 * np.cos(2.0 * np.pi * float(f) * t + ph))

    for _ in range(2):
        add_start(starts, 1.0 + 0.30 * rng.standard_normal(n))

    x = 2.0 * (rng.random(n) < 0.5).astype(float) - 1.0
    add_start(starts, 1.0 + 0.25 * x)

    add_start(starts, (rng.random(n) < 0.5).astype(float))
    add_start(starts, rng.dirichlet(np.full(n, 1.0)))
    add_start(starts, np.ones(n, dtype=np.float64))
    return starts


def refine_sequence(a, rng, deadline):
    best, _ = clean_normalize(a)
    n = int(best.size)
    if n <= 0:
        return np.ones(1, dtype=np.float64)

    c = self_conv(best)
    m = float(np.max(c))
    best_score = 2.0 * n * m if np.isfinite(m) and m >= 0.0 else 2.0 * float(n)
    if not np.isfinite(best_score) or best_score <= 0.0:
        best = np.ones(n, dtype=np.float64) / n
        best_score = 2.0 * float(n)

    momentum = np.zeros(n, dtype=np.float64)
    vel_squared = np.zeros(n, dtype=np.float64)
    t = 0
    lr = 0.05
    fail = 0
    accept = 0

    while time.monotonic() < deadline:
        c = self_conv(best)
        m = float(np.max(c))
        cur = 2.0 * n * m if np.isfinite(m) and m >= 0.0 else float("inf")
        if np.isfinite(cur) and cur < best_score - 1e-12:
            best_score = cur
        if not np.isfinite(best_score) or best_score <= 1.005:
            break

        if best_score > 1.9:
            p_val = 4.0
        elif best_score > 1.7:
            p_val = 6.0
        elif best_score > 1.55:
            p_val = 8.0
        else:
            p_val = 6.0

        use_smooth = (fail == 0) or (fail % 4 == 0) or (accept % 5 == 0)
        G = combined_gradient(best, c, use_smooth, p_val)
        maxG = float(np.max(np.abs(G))) if G.size else 0.0
        if not np.isfinite(maxG) or maxG <= 1e-14:
            fail += 1
            if fail > 150:
                break
            continue

        G = G / maxG
        gu = best * G
        gm = float(np.mean(gu))
        if np.isfinite(gm):
            gu = gu - gm
        gu = np.where(np.isfinite(gu), gu, 0.0)
        maxgu = float(np.max(np.abs(gu)))
        if not np.isfinite(maxgu) or maxgu <= 1e-14:
            fail += 1
            if fail > 150:
                break
            continue

        b1 = 0.9
        b2 = 0.999
        t_next = t + 1
        test_mom = b1 * momentum + (1.0 - b1) * gu
        test_sq = b2 * vel_squared + (1.0 - b2) * (gu * gu)

        bias1 = 1.0 - (b1 ** t_next)
        bias2 = 1.0 - (b2 ** t_next)
        denom = np.sqrt(test_sq / max(bias2, 1e-12)) + 1e-8
        step = -lr * (test_mom / max(bias1, 1e-12)) / denom
        step = np.where(np.isfinite(step), step, 0.0)
        np.clip(step, -2.0, 2.0, out=step)

        improved = False
        for alpha in (1.0, 0.5):
            if time.monotonic() >= deadline:
                break
            cand = best * np.exp(alpha * step)
            x, sc = normalize_score(cand)
            if x is not None and np.isfinite(sc) and sc < best_score - 1e-12:
                best = x
                best_score = sc
                momentum = test_mom
                vel_squared = test_sq
                t = t_next
                improved = True
                break

        if improved:
            accept += 1
            fail = 0
            lr = min(0.20, lr * 1.05)
        else:
            fail += 1
            lr = max(0.001, lr * 0.72)

            if (fail in (4, 8, 16, 32, 64)) or (fail % 20 == 0):
                strength = 0.10 + min(1.0, 0.03 * float(fail))
                cand = random_move(best, rng, strength)
                x, sc = normalize_score(cand)
                if x is not None and np.isfinite(sc) and sc < best_score - 1e-12:
                    best = x
                    best_score = sc
                    momentum = np.zeros(n, dtype=np.float64)
                    vel_squared = np.zeros(n, dtype=np.float64)
                    t = 0
                    lr = 0.05
                    fail = 0
                else:
                    fail += 5

            if fail > 150:
                break

    return best


def exact_score_array(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return float("inf")
    return float(evaluate_sequence(arr.astype(float).tolist()))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 1.0:
        budget = 1100.0
    deadline = t0 + max(1.0, budget - 8.0)
    worker_deadline = deadline - 2.0

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = np.ones(256, dtype=np.float64)

    parent, _ = clean_normalize(parent)
    n = int(parent.size)
    if n < 1:
        n = 1
        parent = np.ones(1, dtype=np.float64)

    starts = make_starts(parent, n, rng)
    if not starts:
        starts = [parent]

    best, _ = clean_normalize(parent)
    best_x, best_score = normalize_score(best)
    if best_x is None or not np.isfinite(best_score):
        best_x = np.ones(n, dtype=np.float64) / n
        best_score = 2.0 * float(n)
    best = best_x

    fallback = best.copy()
    fallback_score = float("inf")
    if n <= 8000 and worker_deadline - time.monotonic() > 3.0:
        try:
            fallback_score = float(evaluate_sequence(fallback.astype(float).tolist()))
        except Exception:
            fallback_score = float("inf")

    for idx, st in enumerate(starts):
        now = time.monotonic()
        if now >= worker_deadline:
            break
        left = max(1, len(starts) - idx)
        reserve = max(10.0, 0.15 * (worker_deadline - now))
        search_deadline = max(now + 0.1, worker_deadline - reserve)
        slice_time = max(0.25, (search_deadline - now) / float(left))
        sub_deadline = min(search_deadline, now + slice_time)
        if sub_deadline <= now:
            continue

        cand = refine_sequence(st, rng, sub_deadline)
        x, sc = normalize_score(cand)
        if x is not None and np.isfinite(sc) and sc < best_score:
            best = x
            best_score = sc

        if best_score < 1.20:
            break

    now = time.monotonic()
    if now < worker_deadline - 3.0:
        final_sub_deadline = worker_deadline - max(2.0, 0.05 * (worker_deadline - now))
        cand = refine_sequence(best, rng, final_sub_deadline)
        x, sc = normalize_score(cand)
        if x is not None and np.isfinite(sc) and sc < best_score:
            best = x
            best_score = sc

    final = np.clip(best, 0.0, 1000.0).astype(float)
    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = np.ones(n, dtype=np.float64)

    if n <= 8000 and worker_deadline - time.monotonic() > 2.0:
        try:
            fscore = float(evaluate_sequence(final.astype(float).tolist()))
            if not np.isfinite(fscore):
                final = np.ones(n, dtype=np.float64)
            elif np.isfinite(fallback_score) and fscore > fallback_score + 1e-9:
                final = np.clip(fallback, 0.0, 1000.0).astype(float)
        except Exception:
            pass

    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if not np.isfinite(s) or s < 0.01:
        final = np.ones(max(1, n), dtype=np.float64)

    return final.tolist()
