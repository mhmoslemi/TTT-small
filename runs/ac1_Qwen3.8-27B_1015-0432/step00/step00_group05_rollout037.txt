def fast_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for efficiency
    m = 1
    while m < size:
        m *= 2
    fa = np.fft.rfft(a, m)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, m)[:size]
    return result

            if cand.sum() < 0.01:
                continue
            s = _score(cand)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = cand

    # Final verification
    final = [float(max(0.0, min(1000.0, x))) for x in best]
    fs = float(evaluate_sequence(final))
    if not np.isfinite(fs) or fs > 2.0:
        final = [float(x) for x in parent]
        fs2 = float(evaluate_sequence(final))
        if not np.isfinite(fs2):
            final = [1.0] * n
    return final
