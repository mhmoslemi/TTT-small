    for c in cands:

        if c.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = c.copy()

    # Soft-max subgradient descent
    a = best_a.copy()
    a = a / a.sum() if a.sum() > 0 else np.ones(n) / n
    mu = 1.0 / (20.0 * n)  # smoothing scale relative to b values (~1/n)
    lr = 0.05
    stall = 0
    eval_count = 0

    while time.monotonic() < deadline:
        if stall > 200:
            break

        b = autocorr(a)
        b_shift = b - b.max()
        w = np.exp(b_shift / mu)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum

        # Gradient: g[m] = 2 * sum_k w[k] * a[k-m], computed as cross-correlation
        g = 2.0 * np.correlate(w, a, mode='valid')

        a_new = a - lr * g
        a_new = simplex_project(a_new, 1.0)

        # Periodic evaluation
        eval_count += 1
        if eval_count % 30 == 0:
            sc = float(evaluate_sequence(a_new.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                stall = 0
            else:
                stall += 1
                if stall > 50:
                    lr *= 0.7
                    mu *= 0.9
                if stall > 150:
                    # Restart from best with perturbation
                    a = best_a.copy()
                    a += rng.normal(0.0, 1e-4, n)
                    a = simplex_project(a, 1.0)
                    stall = 0
            # Anneal mu
            mu *= 0.995
            if mu < 1.0 / (1000.0 * n):
                mu = 1.0 / (1000.0 * n)

        a = a_new

        # Random kicks
        if rng.random() < 0.02:
            a += rng.normal(0.0, 5e-5, n)
            a = simplex_project(a, 1.0)

    # Final phase: local block search from best
    while time.monotonic() < deadline:
        cand = best_a.copy()
        width = int(rng.integers(max(1, n // 100), max(2, n // 8)))
        start = int(rng.integers(0, max(1, n - width)))
        stop = min(n, start + width)
        cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.04)))
        cand = np.clip(cand, 0.0, 1000.0)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = cand.copy()

    # Validate and return
    final = np.clip(best_a, 0.0, 1000.0)
    if final.sum() < 0.01:
        final = np.ones(n) / n
    final_score = float(evaluate_sequence(final.tolist()))
    if not np.isfinite(final_score):
        final = np.ones(n) / n
    return [float(v) for v in final]
