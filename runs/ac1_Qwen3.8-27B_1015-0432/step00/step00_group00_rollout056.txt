import numpy as np
import time

def _fft_convolve(a):
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)
    return c[:2 * n - 1]

def _conv_grad(k, a):
    n = a.size
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = k - np.arange(hi + 1 - lo, -1, -1)
        grad[lo:hi + 1] = 2.0 * a[idx]
    return grad

def _perturb(a, rng, scale):
    n = a.size
    nfreq = min(32, max(4, n // 100))
    freqs = rng.integers(1, n // 2 + 1, size=nfreq)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=nfreq)
    amplitudes = rng.uniform(0.0, 1.0, size=nfreq) * scale / nfreq
    t = np.arange(n)
    d = np.zeros(n)
    for f, p, amp in zip(freqs, phases, amplitudes):
        d += amp * np.cos(2.0 * np.pi * float(f) * t / float(n) + p)
    return a + d

def _score(a):
    n = a.size
    c = _fft_convolve(a)
    return 2.0 * float(n) * float(c.max()) / (float(a.sum()) ** 2)
