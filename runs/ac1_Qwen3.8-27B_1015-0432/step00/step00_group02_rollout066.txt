def _subgrad_at_lag(a, k, n):
    """Compute gradient of conv(a,a)[k] w.r.t. a for a non-negative lag."""
    grad = np.zeros(n, dtype=np.float64)
    if k < 0:
        return grad
    if k >= n:
        k = 2 * n - 2 - k
    if k < 0:
        return grad
    lo = 0
    hi = min(n - 1, k)
    grad[lo:hi + 1] += a[k:lo - 1:-1]
    lo2 = max(0, k)
    hi2 = n - 1
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] += a[0:hi2 - lo2 + 1]
    return grad


def _normalize(a, target_sum):
    s = a.sum()
    if s < 1e-15:
        return a * 0.01
    return a * (target_sum / s)
