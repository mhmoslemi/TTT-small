    fb = np.fft.rfft(b, size)

def _fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1 << (n - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

    """Subgradient of b_k w.r.t. x at index i: 2*x[k-i] if valid."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if hi >= lo:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * x[k - idx]
    return grad

    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)

def _fft_convolve(a, b):
    n_out = len(a) + len(b) - 1
    size = 1 << (n_out - 1).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n_out]


def _smooth_grad(x, beta, fft_size):
    n = len(x)
