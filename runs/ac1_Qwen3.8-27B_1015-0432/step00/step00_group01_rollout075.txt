import numpy as np

def _subgrad_peak(a, k_star):
    """Subgradient of max(conv(a,a)) at the active coordinate k_star."""
    n = a.size
    s = np.zeros(n)
    lo1 = max(0, k_star - n + 1)
    hi1 = min(n - 1, k_star)
    if lo1 <= hi1:
        s[lo1:hi1 + 1] += a[k_star - lo1:k_star - hi1 - 1:-1]
    lo2 = max(0, k_star)
    hi2 = min(n - 1, k_star + n - 1)
    if lo2 <= hi2:
        s[lo2:hi2 + 1] += a[lo2 - k_star:hi2 - k_star + 1]
    return s


def _boltzmann_update(a, s, eta):
    """Redistribute mass away from high-subgradient positions."""
    log_w = np.where(a > 0, np.log(a + 1e-300), -900.0)
    log_w -= eta * s
    log_w -= log_w.max()
    w = np.exp(log_w)
    tw = w.sum()
    if tw < 1e-300:
        return a.copy()
    a_new = w / tw
    return a_new


def _conv_peak(a):
    c = np.convolve(a, a, mode='full')
    return c.max()
