import numpy as np
import time

def subgrad_step(a, k_star, eta):
    n = a.size
    a_new = a.copy()
    lo = max(0, k_star - (n - 1))
    hi = min(k_star, n - 1)
    if lo <= hi:
        a_new[lo:hi+1] -= eta * 2.0 * a[k_star - hi : k_star - lo + 1][::-1]
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 1e-30:
        a_new /= s
    else:
        a_new[:] = 1.0 / n
    return a_new
