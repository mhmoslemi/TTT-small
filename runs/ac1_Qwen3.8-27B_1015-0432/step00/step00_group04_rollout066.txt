import time
import numpy as np


def _valid(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return False
    if not np.isfinite(a).all():
        return False
    if a.sum() < 0.01:
        return False
    return True


def _norm(a):
    a = np.clip(np.asarray(a, dtype=np.float64), 0.0, None)
    s = a.sum()
    if s <= 1e-12:
        n = a.size
        a = np.ones(n)
        s = float(n)
    return a / s


def _peak_grad(a):
    """Return (maxconv, subgradient vector) of self-convolution peak. O(n) mem."""
    c = np.convolve(a, a)
    k = int(np.argmax(c))
    n = a.size
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    partner = k - np.arange(lo, hi + 1)
    g = np.zeros(n, dtype=np.float64)
    g[lo:hi + 1] = 2.0 * a[partner]
    if k % 2 == 0:
        m = k // 2
        if lo <= m <= hi:
            g[m] = a[m]
    return float(c[k]), g


def _template(i, n):
    t = np.arange(n, dtype=np.float64)
    x = t / float(n)
    base = 1.0
    if i == 0:
        w = base + 0.65 * (x - 0.5)
    elif i == 1:
        w = base + 0.65 * np.sin(2.0 * np.pi * x + 0.7)
    elif i == 2:
        w = base + 0.65 * np.sin(4.0 * np.pi * x + 1.3)
    elif i == 3:
        w = base + 0.55 * (2.0 * x - 1.0) * (1.0 - x)
    else:
        w = base + 0.45 * np.cos(2.0 * np.pi * x) - 0.25 * x
    return _norm(np.clip(w, 1e-6, None))


def _subgrad(a, eta, iters, deadline):
    """Projected subgradient descent on max self-convolution, fixed sum=1."""
    b = a.copy()
    for _ in range(iters):
        if time.monotonic() > deadline:
            break
        _, g = _peak_grad(b)
        b = b - eta * g
        b = np.clip(b, 0.0, None)
        s = b.sum()
        if s <= 1e-12:
            b = a.copy()
        else:
            b = b / s
        eta *= 0.997
    return b, eta

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 8.0)

    # Parent as guaranteed-valid fallback.
    parent = None
    try:
        p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if _valid(p):
            parent = _norm(p)
    except Exception:
        parent = None
    if parent is None:
        n = int(rng.integers(64, 512))
        parent = _norm(rng.random(n))

    n = parent.size
    best = parent.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    work = parent.copy()

    # Free scaling for the math: fixed sum=1, score invariant.
    scoref = float(evaluate_sequence(work.tolist()))
    best_work = work.copy()
    best_work_score = scoref

    # Template search window (early, cheap).
    switch_at = t0 + 0.10 * max(0.0, float(budget_s) - 8.0)
    ti = 0
    while time.monotonic() < switch_at and ti < 6:
        cand = _template(ti, n)
        s = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(s) and s < best_score:
            best = cand.copy()
            best_score = s
        if s < best_work_score:
            work = cand.copy()
            best_work_score = s
        ti += 1

    # Subgradient refinement on the current best structure.
    eta = 0.03
    while time.monotonic() + 2.0 < deadline:
        work, eta = _subgrad(work, eta, 120, deadline - 2.0)
        if np.isfinite(work).all() and work.size == n:
            s = float(evaluate_sequence(work.tolist()))
            if np.isfinite(s) and s < best_work_score:
                best_work_score = s
                best_work = work.copy()
            if np.isfinite(s) and s < best_score:
                best_score = s
                best = work.copy()
        # occasional random asymmetric nudge to escape plateaus
        if rng.random() < 0.12:
            w = work.copy()
            j = int(rng.integers(0, n))
            w[j] *= float(np.exp(rng.normal(0.0, 0.35)))
            w = _norm(w)
            s = float(evaluate_sequence(w.tolist()))
            if np.isfinite(s) and s < best_work_score:

                best_work_score = s

                    best_score = s

    if best_score > best_work_score:

        best_score = best_work_score

    # Headroom check

    return [float(x) for x in cand_list]
