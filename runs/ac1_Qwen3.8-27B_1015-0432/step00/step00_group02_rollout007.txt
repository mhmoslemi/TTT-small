        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

    # Final sanity check: ensure best is valid
    if not np.isfinite(best_score) or best_score > 2.0 + 1e-6:
        best = uniform.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Ensure non-empty, finite
    if best.size == 0 or not np.isfinite(best).all():
        best = np.ones(128, dtype=np.float64)

    return [float(x) for x in best]
