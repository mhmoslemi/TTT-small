But the sum of a' is still S (since c is scaled to have sum S). So the new score is 2n * M_new / S^2. If M_new < M, the score decreases.

The LP finds b with large sum (so c has components that are "spread out"), which should help keep (c*c) low.

One more consideration: the LP constraint is T*b <= M*1. But T is the convolution matrix of a with b. If a has large values, T has large entries, and the constraint is tighter, making b smaller. Conversely, if a has small values, b can be larger. This creates a natural "equalization" effect.

def build_conv_matrix(a):
    """Build sparse matrix T where (T @ b)_k = (a*b)_k (full convolution)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        jlo = max(0, k - n + 1)
        jhi = min(n - 1, k)
        rng = jhi - jlo + 1
        rows.extend([k] * rng)
        cols.extend(range(jlo, jhi + 1))
        vals.extend([a[k - j] for j in range(jlo, jhi + 1)])
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def lp_descent(a, time_limit=15):
    """One LP-based descent step. Returns (new_a, improved)."""
    n = len(a)
    S = np.sum(a)
    M = np.max(np.convolve(a, a))
    T = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
        if res.status != 0 or res.x is None:
            return a, False
        b = np.maximum(res.x, 0)
        b_sum = np.sum(b)
        if b_sum < 1e-12:
            return a, False
        c = b * (S / b_sum)
        # Try several step sizes
        for t in [0.15, 0.1, 0.05, 0.03]:
