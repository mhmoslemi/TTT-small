def fft_convolve(a, b):
    """Linear convolution via FFT, O(n log n)."""
    m = len(a) + len(b) - 1
    fa = np.fft.rfft(a, m)
    fb = np.fft.rfft(b, m)
    return np.fft.irfft(fa * fb, m)
