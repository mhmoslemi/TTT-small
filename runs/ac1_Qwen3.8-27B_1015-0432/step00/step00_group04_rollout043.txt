    """Reduce the top peaks of the autocorrelation."""
    conv = np.convolve(a, a)
    max_c = conv.max()
    
    # Find top peaks
    top_k = np.argsort(conv)[-num_peaks:]
    
    for k in top_k:

        j = np.arange(n)
        idx = k - j
        valid = (idx >= 0) & (idx < n)
