        np.maximum(0.0, 1.0 - np.abs(idx - mid) / (n / 2.0)),
        np.exp(-4.0 * ((idx - mid) / max(1, n / 2.0)) ** 2),
        1.0 / (1.0 + np.exp(-0.01 * (idx - mid))),  # smooth step
    ]

    # A few random multiplicative perturbations of the parent
    for _ in range(30):
        noise = rng.normal(0.0, 0.04, n)
        structured.append(np.clip(parent * np.exp(noise), 0.0, 1000.0))

    # Alternating-like pattern (still non-negative)
    structured.append(1.0 + 0.05 * np.cos(4.0 * np.pi * idx / n))
    structured.append(1.0 + 0.05 * np.sin(2.0 * np.pi * 3.0 * idx / n))

    for c in structured:
        c = np.clip(np.asarray(c, dtype=np.float64), 0.0, 1000.0)
        if not np.isfinite(c).all() or c.sum() < 0.01:
            continue
        s = float(evaluate_sequence(c.tolist()))
        if np.isfinite(s) and s < best_score:
            best = c.copy()
            best_score = s

    # --- Local block search (early exit if no improvement in 60 tries) ---
    no_improve = 0
    while time.monotonic() < deadline and no_improve < 60:
        w = max(2, int(n * rng.uniform(0.01, 0.15)))
        start = int(rng.integers(0, max(1, n - w + 1)))
        stop = min(n, start + w)
        candidate = best.copy()
        scale = float(np.exp(rng.normal(0.0, 0.02)))
        candidate[start:stop] *= scale
        # Occasionally mirror-symmetrize
        if rng.random() < 0.2:
            candidate = 0.5 * (candidate + candidate[::-1])
        candidate = np.clip(candidate, 0.0, 1000.0)

        s = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(s) and s < best_score - 1e-12:
            best = candidate
            best_score = s
            no_improve = 0
        else:
            no_improve += 1

    # --- Final sanity check ---
    best = np.clip(best, 0.0, 1000.0)
    fin = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fin) or fin >= best_score:
        # Fall back to the parent
        best = np.clip(parent, 0.0, 1000.0)
        best_score = float(evaluate_sequence(best.tolist()))

    return [float(x) for x in best]
