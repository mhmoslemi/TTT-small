    s = np.sum(a)
    conv = fft_convolve_nonneg(a)
    M = conv[k_star]

import numpy as np
import time

def auto_convolve(a):
    """Full auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)
    return result[:size]

def compute_subgrad(a, conv, k_star):
    """Subgradient of 2n*max(conv)/(sum)^2 at a, active at lag k_star."""
    n = len(a)
    s = np.sum(a)
    M = conv[k_star]
    i_lo = max(0, k_star - (n - 1))
    i_hi = min(n - 1, k_star)
    g = np.zeros(n)
    i_range = np.arange(i_lo, i_hi + 1)
    j_range = k_star - i_range
    g[i_range] = 2.0 * a[j_range]
    si = k_star // 2

import numpy as np
import time


def _auto_conv(a):
    """Full auto-convolution via zero-padded FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fs = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fs)
    return np.fft.irfft(fa * fa, fs)[:size]


def _smoothed_subgrad(a, conv, w):
    """Softmax-weighted subgradient of the max-of-conv term. O(n log n)."""
    n = len(a)
    fs = 1 << (3 * n - 2 - 1).bit_length()
    fw = np.fft.rfft(w, fs)
    farev = np.fft.rfft(a[::-1], fs)
    g_full = np.fft.irfft(fw * farev, fs)
    g = 2.0 * g_full[n - 1:2 * n - 1]
    # Self-correction for the dominant even lag
    k_star = int(np.argmax(conv))
    si = k_star // 2
    if k_star % 2 == 0 and 0 <= si < n:
        g[si] -= w[k_star] * a[si]
    return g, k_star


def _score_direct(conv, s):
    """Score = 2n * max(conv) / s^2."""
    n = s.size if hasattr(s, 'size') else len(s)
    return float(2.0 * n * conv.max() / (s * s))


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    n = a.size

    s0 = a.sum()
    if s0 < 0.01:
        a = np.ones(n) / n
    else:
        a = a / s0

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    # Break symmetry of the uniform parent
    a = a + 1e-5 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    lr = 5e-4
    no_improve = 0
    T_scale = 0.005

    while time.monotonic() < deadline:
        conv = _auto_conv(a)
        M = float(conv.max())
        s = float(a.sum())
        if s < 1e-12:
            break

        # Softmax weights over lags (smoothed max subgradient)
        T = max(M * T_scale, 1e-12)
        logits = (conv - M) / T
        logits = np.clip(logits, -50.0, 0.0)
        w = np.exp(logits)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            int(np.argmax(conv))
            w[int(np.argmax(conv))] = 1.0
            w_sum = 1.0
        w /= w_sum

        g, k_star = _smoothed_subgrad(a, conv, w)

        # Gradient of the scale-invariant score: 2n(g*s - 2M*a)/s^3
        grad = 2.0 * n * (g * s - 2.0 * M * a) / (s ** 3)
