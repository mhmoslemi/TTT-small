import numpy as np
import time

def _ac1_score(a, n):
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    C = np.convolve(a, a)
    return float(2.0 * n * np.max(C) / (s * s))

def _cmax_grad(a, C, n, tol=1e-6):
    maxC = np.max(C)
    active = np.where(C >= maxC - tol)[0]
    g = np.zeros(n, dtype=np.float64)
    for m in active:
        lo = max(0, m - n + 1)
        hi = min(n - 1, m)
        if lo <= hi:
            g[lo:hi+1] += a[m-hi:m-lo+1][::-1]
    return g

def _make_cos_taper(n, eps):

    if n > 1:
        x = 1.0 + eps * np.cos(np.pi * t / (n - 1))
    else:
        x = np.ones(1)
    return np.clip(x, 0.0, 1000.0)
