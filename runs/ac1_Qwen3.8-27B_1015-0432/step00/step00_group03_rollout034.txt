- When i=k-j: term is a[k-j]*a[j], derivative w.r.t. a[j] is a[k-j]

If j = k-j (i.e., k=2j): the term is a[j]^2, derivative is 2*a[j]... wait, that's the same as 2*a[k-j] = 2*a[j].

So in all cases: d(conv(a,a)[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n and 0 <= j < n, else 0.

import numpy as np
import time

def _conv_sq_fft(a):
    """Compute conv(a,a) via FFT. O(n log n)."""
    n = a.size
    L = 2 * n  # next power of 2 or just 2n
    X = np.fft.rfft(a, L)
    return np.fft.irfft(X * X, L)

def _subgrad_step(a, eta, k_active):
    """One subgradient step for minimizing max conv."""
    n = a.size
    grad = np.zeros(n)
    for k in k_active:
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            # grad[j] += 2*a[k-j] for j in [lo, hi]
            js = np.arange(lo, hi + 1)
            grad[js] += 2.0 * a[k - js]
    # average over active constraints
    grad /= len(k_active)
    a_new = a - eta * grad
    a_new = np.maximum(0, a_new)
    s = a_new.sum()
    if s > 1e-12:
        a_new /= s
    else:
        a_new = a  # revert
    return a_new
