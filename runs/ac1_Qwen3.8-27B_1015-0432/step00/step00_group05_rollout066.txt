import numpy as np
import time


def fft_convolve(a, fft_size):
    """Full convolution of a with a via FFT. O(n log n), linear storage."""
    fa = np.fft.rfft(a, n=fft_size)
    return np.fft.irfft(fa * fa, n=fft_size)


def top_k_subgradient(a, conv, top_k):
    """Sum of shifted/reversed copies of a over the top_k convolution lags.
    This is a (sub)gradient of the peak of the autocorrelation. O(top_k * n)."""
    n = a.size
    idx = np.argpartition(conv, -top_k)[-top_k:]
    grad = np.zeros(n, dtype=np.float64)
    for k in idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            seg = a[k - hi:k - lo + 1]
            grad[lo:hi + 1] += seg[::-1]
    grad /= float(top_k)
    return grad


def normalize_nonneg(a, target_sum):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-14:
        return None
    return a * (target_sum / s)
