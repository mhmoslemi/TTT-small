import numpy as np
import time


def fft_convolve(a):
    """Full linear self-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 1
    target = 2 * n - 1
    while size < target:
        size *= 2
    A = np.fft.rfft(a, size)
    out = np.fft.irfft(A * A, size)[:target]
    np.maximum(out, 0.0, out=out)
    return out


def compute_grad(b, a, n, k_list):
    """Averaged subgradient of max over positions in k_list."""
    g = np.zeros(n)
    for k in k_list:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            g[lo:hi + 1] += 2.0 * a[lo:hi + 1][::-1]
    g /= len(k_list)
    return g


def try_init(n, rng, parent):

    t = np.arange(n, dtype=np.float64)
    base = parent.copy()
    # Pick a random frequency and phase sinusoidal perturbation
    freq = int(rng.integers(1, max(2, n // 4)))

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 1e-12:
        a /= s
    else:
        a = np.ones(n) / n
    return a

        # Learning rate decay
