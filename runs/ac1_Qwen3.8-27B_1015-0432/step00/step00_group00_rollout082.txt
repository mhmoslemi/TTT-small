import numpy as np
import time


def _fft_size_for(n):
    need = 3 * int(n) - 2
    if need < 2:
        need = 2
    size = 1
    while size < need:
        size <<= 1
    return size


def _self_convolution(a, fft_size, fa=None):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if fa is None:
        fa = np.fft.rfft(a, fft_size)
    out = np.fft.irfft(fa * fa, fft_size)
    return out[: 2 * n - 1].astype(np.float64, copy=False)


def _correlation_weights(a, w, fft_size, fa=None):
    n = a.size
    if n == 1:
        return np.array([float(w[0] * a[0])], dtype=np.float64)
    if fa is None:
        fa = np.fft.rfft(a, fft_size)
    fw = np.fft.rfft(w[::-1], fft_size)
    y = np.fft.irfft(fa * fw, fft_size)
    return y[2 * n - 2 : n - 2 : -1].astype(np.float64, copy=False)


def _softnorm_weights(c, beta):
    m = float(np.max(c))
    L = c.size
    if not np.isfinite(m):
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    e = np.exp(beta * (c - m))
    tot = float(e.sum())
    if not np.isfinite(tot) or tot <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    return e / tot


def _softnorm_value(c, beta):
    m = float(np.max(c))
    if not np.isfinite(m):
        return 0.0
    if beta <= 1.0e-12:
        return m
    e = np.exp(beta * (c - m))
    tot = float(e.sum())
    if not np.isfinite(tot) or tot <= 0.0:
        return m
    return m + float(np.log(tot)) / float(beta)


def _normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1.0e-300:
        n = a.size if a.size > 0 else 4806
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s

def _flat_frequency_initials(n):
    N = 1
    while N < 4 * n + 16:
        N <<= 1
    L = 2 * n - 1
    b = np.zeros(N, dtype=np.float64)

    A = np.sqrt(B)

        inv[:n][::-1].copy(),
        np.abs(inv[:n]).copy(),
        np.roll(inv, n // 2)[:n].copy(),
        np.roll(inv, -n // 2)[:n].copy(),

        g *= 2.0 * float(n)
        g -= g.mean()

        gmax = float(np.max(np.abs(g)))
        if not np.isfinite(gmax) or gmax <= 1.0e-20:
            break
        g /= gmax

        if cur > best_score + 0.05:
            a = best.copy()
            fa = np.fft.rfft(a, fft_size)
            conv = _self_convolution(a, fft_size, fa)
            conv = np.maximum(conv, 0.0)
            cur = 2.0 * float(n) * float(np.max(conv))
            fail = 0
            it += 1
            it_this += 1
            continue

        typ = max(1.0 / float(n), 0.1 * float(np.max(a)))
        accepted = False

        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            cand = a - factor * lr * typ * g
            if it % 90 == 80:
                cand *= np.exp(rng.normal(0.0, 0.02, n))
            cand = np.maximum(cand, 0.0)
            cs = float(cand.sum())
            if not np.isfinite(cs) or cs <= 1.0e-300:
                continue
            cand /= cs

            fa2 = np.fft.rfft(cand, fft_size)
            conv2 = _self_convolution(cand, fft_size, fa2)
            conv2 = np.maximum(conv2, 0.0)
            F2 = _softnorm_value(conv2 * float(n), beta)

            if np.isfinite(F2) and F2 < F - 1.0e-13:
                a = cand
                fa = fa2
                conv = conv2
                cur = 2.0 * float(n) * float(np.max(conv2))
                accepted = True
                if cur < best_score:
                    best = a.copy()
                    best_score = cur
                break

        if accepted:
            lr = min(max_lr, lr * 1.15)
            fail = 0
        else:
            lr = max(min_lr, lr * 0.65)
            fail += 1
            if fail > 30:
                scale = float(rng.uniform(0.03, 0.15))
                cand = best * np.exp(rng.normal(0.0, scale, n))
                cand = _normalize_nonnegative(cand)
                fa = np.fft.rfft(cand, fft_size)
                conv = _self_convolution(cand, fft_size, fa)
                conv = np.maximum(conv, 0.0)
                cur = 2.0 * float(n) * float(np.max(conv))
                a = cand
                if cur < best_score:
                    best = a.copy()
                    best_score = cur
                fail = 0
                lr = 0.4

        it += 1
        it_this += 1

    if cur < best_score:
        best = a.copy()
        best_score = cur
    return best, best_score


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        parent = None

    if parent is None or parent.ndim != 1 or parent.size == 0:
        parent = np.ones(4806, dtype=np.float64)

    parent = np.where(np.isfinite(parent), parent, 0.0)
    parent = np.clip(parent, 0.0, 1000.0)
    n = parent.size

    parent_exact = float(evaluate_sequence([float(v) for v in parent]))
    parent_norm = _normalize_nonnegative(parent)
    fft_size = _fft_size_for(n)

    candidates = []
    s0 = _score_normalized(parent_norm, n, fft_size)
    candidates.append((s0, parent_norm))

    for a0 in _initial_candidates(n, rng):
        if time.monotonic() >= deadline - 20.0:
            break
        if a0.size != n:
            continue
        arr = _normalize_nonnegative(a0)
        sc = _score_normalized(arr, n, fft_size)
        if np.isfinite(sc):
            candidates.append((sc, arr))

    if n <= 20000:
        for a0 in _flat_frequency_initials(n):
            if time.monotonic() >= deadline - 20.0:
                break
            if a0.size != n:
                continue
            arr = _normalize_nonnegative(a0)
            sc = _score_normalized(arr, n, fft_size)
            if np.isfinite(sc):
                candidates.append((sc, arr))

    if not candidates:
        candidates = [(0.0, parent_norm)]

    candidates.sort(key=_score_key)

    best_search = parent_norm.copy()
    best_search_score = s0
    if not np.isfinite(best_search_score):
        best_search_score = float("inf")

    remaining = deadline - time.monotonic()
    if remaining > 200.0:
        top_k = min(4, len(candidates))
    elif remaining > 60.0:
        top_k = min(2, len(candidates))
    else:
        top_k = min(1, len(candidates))

    for i in range(top_k):
        now = time.monotonic()
        if now >= deadline - 10.0:
            break
        if i == top_k - 1:
            stage_end = deadline - 3.0
        else:
            span = max(0.0, deadline - 3.0 - now)
            stage_end = min(deadline - 3.0, now + max(30.0, span / float(top_k - i)))

        a_best, s_best = _refine(candidates[i][1], n, fft_size, stage_end, rng)
        if np.isfinite(s_best) and s_best < best_search_score:
            best_search = a_best.copy()
            best_search_score = s_best

    best_search = _normalize_nonnegative(best_search)
    search_exact = float(evaluate_sequence([float(v) for v in best_search]))

    fallback = parent if np.isfinite(parent_exact) else _normalize_nonnegative(parent)

    if np.isfinite(search_exact) and (
        not np.isfinite(parent_exact) or search_exact <= parent_exact
    ):
        return [float(v) for v in best_search]
    return [float(v) for v in fallback]
