import numpy as np
import time


def _fft_conv(a, fsize):
    A = np.fft.rfft(a, fsize)
    return np.fft.irfft(A * A, fsize)


def _grad_term(a, p, fsize):
    n = a.size
    Ae = np.zeros(fsize)
    Ae[:n] = a
    Pe = np.zeros(fsize)
    Pe[:2*n-1] = p
    return np.fft.irfft(np.fft.rfft(Ae).conj() * np.fft.rfft(Pe), fsize)[:n]


def _score_fft(a, fsize):
    n = a.size
    c = _fft_conv(a, fsize)[:2*n-1]
    s = a.sum()
    if s < 1e-15:
        return np.inf, c
    return 2.0 * n * c.max() / (s * s), c


def _softmax(c, t):
    cm = c.max()
    p = np.exp((c - cm) / t)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros_like(c)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= ps
    return p
