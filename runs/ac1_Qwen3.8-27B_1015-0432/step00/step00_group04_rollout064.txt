import numpy as np
import time


def power_ramp(n, p):
    t = np.arange(1, n + 1, dtype=np.float64) / float(n)
    return t ** p * 1000.0


def try_ramp(n, p, q=0.0):
    base = power_ramp(n, p)
    if q != 0.0:
        t = np.arange(1, n + 1, dtype=np.float64) / float(n)
        base = base * (t ** q)
        base = np.clip(base, 0.0, 1000.0)
    return base

def block_tilt(arr, rng, sigma=0.10):
    n = arr.size
    wlo = max(1, n // 500)
    whi = max(2, n // 8)
    width = int(rng.integers(wlo, whi + 1))
    start = int(rng.integers(0, n - width + 1))
    stop = min(n, start + width)
    out = arr.copy()
    out[start:stop] *= float(np.exp(rng.normal(0.0, sigma)))
    return np.clip(out, 0.0, 1000.0)


def tail_tilt(arr, rng, cmax=0.5):
    n = arr.size
    out = arr.copy()
    t = np.arange(n, dtype=np.float64) / float(n)
    c = float(rng.uniform(-1.0, 1.0)) * cmax
    out = out * np.exp(c * t)
    return np.clip(out, 0.0, 1000.0)
