```python
import numpy as np
import time

def fft_convolve(a, b):
    """Full convolution via FFT, O(n log n)."""
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def compute_score_and_grad(a):
    """Return (score, k_max, grad) where grad is gradient of score w.r.t. a."""
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return float('inf'), 0, np.zeros(n)
    b = fft_convolve(a, a)
    M = np.max(b)
    k_max = np.argmax(b)
    score = 2.0 * n * M / (S * S)
    # Gradient of score
    grad = -2.0 * n * M / (S * S * S) * np.ones(n)
    # Add 4n*a[k_max - j]/S^2 for valid j
    j_lo = max(0, k_max - n + 1)
    j_hi = min(n - 1, k_max)
    if j_lo <= j_hi:
        idx = k_max - np.arange(j_lo, j_hi + 1)
        grad[j_lo:j_hi+1] += 4.0 * n * a[idx] / (S * S)
    return score, k_max, grad
```
