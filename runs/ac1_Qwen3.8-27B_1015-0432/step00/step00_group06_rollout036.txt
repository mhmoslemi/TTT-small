import numpy as np
import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = parent.size if parent.size > 0 else 256
    if not np.isfinite(parent).all():
        parent = np.ones(n)
    
    def eval_score(arr):
        return float(evaluate_sequence(arr.tolist()))
    
    best = parent.copy()
    best_score = eval_score(best)
    
    # Phase 1: Parameterized families
    idx = np.arange(n, dtype=np.float64)
    
    for m_frac in [0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7]:
        m = int(m_frac * n)
        for alpha in np.linspace(0.01, 0.8, 40):
            cand = np.ones(n)
            cand[:m] = 1.0 + alpha
            cand[m:] = 1.0 - alpha
            cand = np.clip(cand, 0.0, 1000.0)
            s = eval_score(cand)
            if s < best_score:
                best = cand.copy()
                best_score = s
    
    # Smooth step variants
    for m_frac in [0.3, 0.4, 0.5, 0.6]:
        m = int(m_frac * n)
        for width_frac in [0.02, 0.05, 0.1, 0.2]:
            width = max(1, int(width_frac * n))
            for alpha in np.linspace(0.01, 0.8, 30):
                # Smooth step
                t = (idx - m) / width
                sig = 1.0 / (1.0 + np.exp(-5.0 * t))  # sigmoid
                cand = 1.0 + alpha * (sig - 0.5)
                cand = np.clip(cand, 0.0, 1000.0)
                s = eval_score(cand)
                if s < best_score:
                    best = cand.copy()
                    best_score = s
    
    # Phase 2: Local search
    # Multiplicative perturbations
    for _ in range(50000):
        if time.monotonic() > deadline:
            break
        
        pert_width = int(rng.integers(max(1, n // 200), n // 10))
        start = int(rng.integers(0, n - pert_width))
        
        # Random smooth perturbation on a block
        t_local = np.linspace(0, 2*np.pi, pert_width)
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(-0.3, 0.3)
        pert = 1.0 + amp * np.sin(freq * t_local)
        
        cand = best.copy()
        cand[start:start+pert_width] *= pert
        cand = np.clip(cand, 0.0, 1000.0)
        
        s = eval_score(cand)
        if s < best_score:
            best = cand
            best_score = s
    
    return [float(x) for x in best]
