```python
import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as so
from scipy.signal import fftconvolve

def build_T(f):
    """Return sparse matrix T such that T @ b = f * b (full conv)."""
    n = f.size
    # (2n-1) x n matrix
    # T[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Each column j has nonzeros at rows j, j+1, ..., j+n-1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # T[k, j] = f[k - j] for k in [j, j+n-1]
        ks = np.arange(j, j + n)
        vs = f[range(n)]  # f[k-j] for k in [j, j+n-1]
        rows.append(ks)
        cols.append(np.full(n, j))
        vals.append(vs)
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    vals = np.concatenate(vals)
    T = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T
```

```python
def dense_conv_mat(f):
    """Build dense matrix T such that T @ b = f * b (full convolution)."""
    n = f.size
    kf = np.arange(2 * n - 1)[:, None]
    kfj = np.arange(n)[None, :]

    T = np.zeros((2 * n - 1, n))
    np.copyto(T, np.zeros_like(T), where=False)
```

```python
import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as so
from scipy.signal import fftconvolve


def max_conv_of(f):
    return float(np.max(fftconvolve(f, f, mode='full')))


def sparse_conv_matrix(f):
    n = f.size
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n))
        col_parts.append(np.full(n, j, dtype=np.int64))
        val_parts.append(f)
    rows = np.concatenate(row_parts).astype(np.int64)
    cols = np.concatenate(col_parts).astype(np.int64)
    vals = np.concatenate(val_parts).astype(np.float64)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def linprog_max_b(f, M, time_limit):
    n = f.size
    try:
        T = sparse_conv_matrix(f)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        res = so.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'time_limit': time_limit, 'presolve': 'default'})
        if res is not None and res.x is not None:
            return res.x
    except Exception:
        pass
    return None
```

```python
    n = f.size
    row_parts = []
    col_parts = []

        row_parts.append(np.arange(j, j + n))
```

```python
def max_conv_of(f):
    return float(np.max(fftconvolve(f, f, mode='full')))


def sparse_conv_matrix(f):
    """Build T of shape (2n-1) x n so that T @ b == f * b (full convolution)."""
    n = f.size
    row_parts = []
    col_parts = []
    val_parts = []
    for j in range(n):
        row_parts.append(np.arange(j, j + n))
        col_parts.append(np.full(n, j, dtype=np.int64))
        val_parts.append(f)
    rows = np.concatenate(row_parts).astype(np.int64)
    cols = np.concatenate(col_parts).astype(np.int64)
    vals = np.concatenate(val_parts).astype(np.float64)
    return sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
```
