import numpy as np
import time

def convolve_fft(a):
    """Compute np.convolve(a,a) via FFT. O(n log n)."""
    n = a.size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(fa * fa, fft_size)[:size]
    return result

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 10.0)
    
    # Initialize
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).astype(np.float64)
    if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
        a = np.ones(256, dtype=np.float64)
    n = a.size
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64)
    a = a / s  # normalize to sum 1
    
    best = a.copy()
    best_peak = convolve_fft(best).max()
    
    # Main optimization loop
    lr = 0.02
    for step in range(500000):
        if time.monotonic() > deadline:
            break
        
        c = convolve_fft(a)
        peak = c.max()
        
        if peak < best_peak:
            best = a.copy()
            best_peak = peak
            # Slightly increase lr on improvement
            lr = min(lr * 1.02, 0.4)
        else:
            lr = max(lr * 0.98, 1e-5)
        
        k_star = int(np.argmax(c))
        
        # Subgradient: g[j] = a[k_star - j] for valid j
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        
        # Apply gradient step (in-place for memory efficiency)
        # Store the values first since we're modifying a
        g_vals = a[k_star - j_hi : k_star - j_lo + 1][::-1].copy()
        a[j_lo:j_hi+1] -= lr * g_vals
        
        # Project: clamp to [0, inf) and renormalize
        np.maximum(a, 0.0, out=a)
        s = a.sum()
        if s < 1e-30:
            a[:] = 1.0 / n
        else:
            a /= s
        
        # Every 500 steps, add small noise to escape local minima
        if step % 500 == 499:
            noise = 0.002 * rng.standard_normal(n)
            a += noise
            np.maximum(a, 0.0, out=a)
            a /= a.sum()
        
        # Every 5000 steps, try a more aggressive perturbation
        if step % 5000 == 4999:
            # Scale a random subset
            mask = rng.random(n) < 0.1
            a[mask] *= np.exp(0.05 * rng.standard_normal(mask.sum()))
            np.maximum(a, 0.0, out=a)
            a /= a.sum()
    
    # Final: evaluate and return
    # The objective is scale-invariant, so sum=1 is fine
    # But ensure sum >= 0.01 for evaluator
    if best.sum() < 0.01:
        best = best * (1.0 / best.sum())
    
    result = [float(x) for x in best]
    score = evaluate_sequence(result)
    if np.isfinite(score):
        return result
    return [1.0 / n] * n
