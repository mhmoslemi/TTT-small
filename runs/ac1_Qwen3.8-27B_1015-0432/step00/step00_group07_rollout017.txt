import numpy as np
import time


def ac1_vec(a):
    """Vectorized AC1 score matching the evaluator for valid inputs."""
    n = len(a)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * float(b.max()) / (s * s)


def gd_step(a, beta, eta):
    """One softmax-weighted projected gradient descent step."""
    b = np.convolve(a, a)
    shifted = beta * (b - b.max())
    w = np.exp(shifted)
    ws = w.sum()
    if ws > 0 and np.isfinite(ws):
        w /= ws
    else:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    C = np.correlate(w, a, mode='valid')
    grad = 2.0 * C
    a_new = a - eta * grad
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s < 0.01:
        return a, np.inf
    a_new = a_new / s
    return a_new, ac1_vec(a_new)


def optimize_gd(a, deadline, rng):

    best = best / s

    beta = 5.0
