def _fast_score(a):
    n = a.size
    s = a.sum()
    if s < 0.01:

    inits.append(1.0 + 0.5 * (1.0 - (2.0 * t - 1.0) ** 2))
    # Semicircle
    inits.append(1.0 + 0.3 * np.sin(np.pi * t))
    # Exponential ramp
    inits.append(np.exp(0.5 * t))
    inits.append(np.exp(-0.5 * t))
    # Low-freq cosine modulation
    for f in (1, 2, 3):
        inits.append(1.0 + 0.15 * np.cos(2.0 * np.pi * f * t))
        inits.append(1.0 + 0.15 * np.sin(2.0 * np.pi * f * t))
    # Random smooth (Gaussian-differentiated)
    for _ in range(6):
        noise = np.cumsum(rng.normal(0, 1, n))
        noise -= noise.mean()
        inits.append(1.0 + 0.2 * noise / max(1.0, np.std(noise)))
    # Power ramp
    inits.append(1.0 + t ** 3)
    inits.append(1.0 + 0.3 * t ** 0.5)
    return inits


def _perturb(a, rng, n, alpha):
    j = np.arange(n, dtype=np.float64)
    t = j / (n - 1)
    r = rng.random()
    c = a.copy()
    if r < 0.20:
        # Sinusoidal (preserves sum)
        f = int(rng.integers(1, max(2, n // 300)))
        ph = rng.uniform(0, 2 * np.pi)
        amp = alpha * a.max() * rng.uniform(0.3, 1.0)
        c += amp * np.sin(2 * np.pi * f * t + ph)
    elif r < 0.40:
        # Cosine modulation
        f = int(rng.integers(1, max(2, n // 500)))
        ph = rng.uniform(0, 2 * np.pi)
        amp = alpha * a.max() * rng.uniform(0.3, 1.0)
        c += amp * np.cos(2 * np.pi * f * t + ph)
    elif r < 0.55:
        # Linear ramp perturbation
        slope = rng.normal(0, alpha * 0.8)
        c += slope * (2 * t - 1)
    elif r < 0.70:
        # Block multiplicative
        w = int(rng.integers(max(2, n // 400), max(4, n // 20)))
        st = int(rng.integers(0, max(1, n - w)))
        c[st:st + w] *= float(np.exp(rng.normal(0, alpha * 1.5)))
    elif r < 0.85:
        # Gaussian bump
        center = rng.uniform(0, n)
        width = rng.uniform(n / 50, n / 8)
        bump = np.exp(-0.5 * ((j - center) / width) ** 2)
        c += alpha * a.max() * rng.uniform(-1, 1) * bump
    else:
        # Multiplicative noise
        c *= (1.0 + rng.normal(0, alpha * 0.3, n))
    return np.clip(c, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size < 2 or not np.all(np.isfinite(parent)):
        parent = np.ones(256, dtype=np.float64)
    n = parent.size

    # Phase 1: diverse initializations
    inits = _make_inits(parent, n, rng)
    best = None
    best_score = 1e30
    for s in inits:
        s = np.clip(s, 0.0, 1000.0)
        if s.sum() < 0.01:
            continue
        sc = _fast_score(s)
        if sc < best_score:
            best_score = sc
            best = s.copy()

    # Confirm with exact evaluator
    exact = float(evaluate_sequence(best.tolist()))
    if np.isfinite(exact):
        best_score = exact
    else:
        best = parent.copy()
        best_score = float(evaluate_sequence(best.tolist()))

    # Phase 2: local search
    alpha = 0.12
    no_improve = 0
    step = 0
    while time.monotonic() < deadline:
        step += 1
        if no_improve > 3000:
            break
        if step % 500 == 0:
            alpha = max(0.02, alpha * 0.9)

        c = _perturb(best, rng, n, alpha)
        cp = np.clip(c, 0.0, 1000.0)
        if cp.sum() < 0.01:
            no_improve += 1
            continue

        sc = _fast_score(cp)
        if sc < best_score * (1.0 - 1e-7):
            exact = float(evaluate_sequence(cp.tolist()))
            if np.isfinite(exact) and exact < best_score:
                best = cp.copy()
                best_score = exact
                no_improve = 0
                alpha = min(0.3, alpha * 1.08)
            else:
                no_improve += 1
        else:
            no_improve += 1

    # Final safety: verify returned sequence
    out = [float(x) for x in best]
    final_sc = float(evaluate_sequence(out))
    if not np.isfinite(final_sc) or final_sc > 1e20:
        out = [float(1.0)] * n
    return out
