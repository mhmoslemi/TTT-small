import numpy as np
import time
import math

EPS = 1e-15


def normalize_sequence(p):
    p = np.asarray(p, dtype=np.float64)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    p = np.minimum(p, 1000.0)
    n = p.size
    if n == 0:
        return np.array([1.0], dtype=np.float64)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-12:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return p / s


def fft_length(n):
    L = 3 * n - 2
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def conv_square(p, m=None, P=None):
    n = p.size
    if m is None:
        m = fft_length(n)
    if P is None:
        P = np.fft.rfft(p, m)
    c = np.fft.irfft(P * P, m)
    return c[:2 * n - 1]


def fast_score(p, m=None):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    s = float(p.sum())
    if n == 0 or not math.isfinite(s) or s <= 1e-12:
        return float("inf")
    if m is None:
        m = fft_length(n)
    P = np.fft.rfft(p, m)
    c = conv_square(p, m, P)
    mx = float(np.max(c))
    if not math.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx / (s * s))


def gradient_weights(p, w, m, P):
    n = p.size
    wr = w[::-1]
    W = np.fft.rfft(wr, m)
    c = np.fft.irfft(W * P, m)
    return 2.0 * n * c[2 * n - 2::-1]


def multiplicative_update(p, H, step):
    z = -float(step) * H
    z = np.clip(z, -8.0, 8.0)
    q = p * np.exp(z)
    q = np.maximum(q, EPS)
    s = float(q.sum())
    if not math.isfinite(s) or s <= 1e-12:
        return normalize_sequence(p)
    return q / s


def add_candidate(starts, p, m, k=4):
    p = normalize_sequence(p)
    score = fast_score(p, m)
    if not math.isfinite(score):
        return starts
    pos = len(starts)
    for i in range(len(starts)):
        if score < starts[i][0]:
            pos = i
            break
    starts.insert(pos, (score, p))
    if len(starts) > k:
        starts.pop()
    return starts


def sort_score_pairs(items):
    for i in range(1, len(items)):
        item = items[i]
        j = i - 1
        while j >= 0 and items[j][0] > item[0]:
            items[j + 1] = items[j]
            j -= 1
        items[j + 1] = item
    return items


def cosine_candidate(idx, a, phi):
    n = idx.size
    p = 1.0 + a * np.cos(math.pi * idx / n + phi)
    return np.maximum(p, 0.0)


def two_term_candidate(idx, rho, phi):
    n = idx.size
    theta = math.pi * idx / n
    c = 1.0 + rho * np.exp(1j * (phi + theta))
    return c.real * c.real + c.imag * c.imag


def trig_candidate(idx, rng, K):

            break

    for a in (0.7, 0.9, 1.1):
        for phi in (0.4, 0.9, 1.4, 1.9, 2.4, 2.9):
            if time.monotonic() > deadline:
                break
            starts = add_candidate(starts, cosine_candidate(idx, a, phi), m)
        if time.monotonic() > deadline:
            break

    max_trials = 10000
    for _ in range(max_trials):
        if time.monotonic() > deadline:
            break
        starts = add_candidate(starts, trig_candidate(idx, rng, 5), m)

    if not starts:
        p = np.full(n, 1.0 / n, dtype=np.float64)
        starts = [(fast_score(p, m), p)]
    return starts


def refine_candidate(p0, rng, deadline):
    p0 = normalize_sequence(p0)
    n = p0.size
    m = fft_length(n)

    best_p = p0.copy()
    best_score = fast_score(best_p, m)
    current_p = p0.copy()
    current_score = best_score

    targets = (0.85, 0.78, 0.74, 0.72, 0.70, 0.68, 0.66)
    max_target_iters = 8000
    step = 0.25

    for T in targets:
        if time.monotonic() > deadline:
            break
        if current_score > best_score + 0.03:
            current_p = best_p.copy()
            current_score = best_score

        fails = 0
        for _ in range(max_target_iters):
            if time.monotonic() > deadline:
                break

            P = np.fft.rfft(current_p, m)
            d = conv_square(current_p, m, P)
            umax = float(np.max(d))
            if not math.isfinite(umax):
                break
            u = n * d
            score = 2.0 * n * umax

            if score < best_score:
                best_score = score
                best_p = current_p.copy()

            tmp = 30.0 * (u - umax)
            wsmooth = np.exp(tmp)
            wsum = float(np.sum(wsmooth))
            if not math.isfinite(wsum) or wsum <= 0.0:
                wsmooth = np.full_like(u, 1.0 / u.size)
            else:
                wsmooth /= wsum

            e = u - T
            pos = e > 1e-10
            if pos.any():
                se = float(np.sum(e[pos]))
                if se > 0.0:
                    wex = np.zeros_like(u)
                    wex[pos] = e[pos] / se
                    w = 0.6 * wex + 0.4 * wsmooth
                else:
                    w = wsmooth
            else:
                w = wsmooth

            ws = float(np.sum(w))
            if not math.isfinite(ws) or ws <= 0.0:
                w = wsmooth
            else:
                w /= ws

            G = gradient_weights(current_p, w, m, P)
            dot = float(np.dot(current_p, G))
            H = G - dot
            maxh = float(np.max(np.abs(H)))
            if not math.isfinite(maxh) or maxh <= 1e-14:
                break
            H /= maxh

            cand = multiplicative_update(current_p, H, step)
            cand_score = fast_score(cand, m)

            if math.isfinite(cand_score) and cand_score < score - 1e-12:
                current_p = cand
                current_score = cand_score
                step = min(0.7, step * 1.07)
                fails = 0
            else:
                step *= 0.5
                fails += 1
                if fails > 20:
                    noise = np.clip(rng.normal(size=n), -3.0, 3.0)
                    base = best_p + 1e-8 * rng.random(size=n)
                    current_p = base * np.exp(noise)
                    current_p = normalize_sequence(current_p)
                    current_score = fast_score(current_p, m)
                    if math.isfinite(current_score) and current_score < best_score:
                        best_score = current_score
                        best_p = current_p.copy()
                    step = 0.25
                    fails = 0

    return normalize_sequence(best_p), best_score


def exact_score(p):
    arr = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1000.0)
    try:
        return float(evaluate_sequence([float(x) for x in arr]))
    except Exception:
        return float("inf")


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    hard = t0 + max(1.0, budget - 25.0)

    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except Exception:
        parent = None

    if parent is None or parent.size == 0:
        n = 256
    else:
        n = int(parent.size)

    idx = np.arange(n, dtype=np.float64)
    rng = np.random.default_rng(int(seed))
    m = fft_length(n)

    search_deadline = min(hard - 25.0, t0 + min(180.0, 0.25 * max(1.0, budget)))
    if search_deadline < t0 + 0.1:
        search_deadline = t0 + 0.1

    starts = run_initial_search(idx, parent, rng, search_deadline, m)
    starts = sort_score_pairs(starts)

    refined = []
    refine_hard = hard - 15.0
    n_starts = min(4, len(starts))
    for rank in range(n_starts):
        if time.monotonic() > refine_hard:
            break
        rem = max(0.0, refine_hard - time.monotonic())
        sub_deadline = time.monotonic() + max(2.0, rem - 3.0)
        if sub_deadline > refine_hard:
            sub_deadline = refine_hard

        score, p = starts[rank]
        fp, fs = refine_candidate(p, rng, sub_deadline)
        if math.isfinite(fs):
            refined.append((fs, fp))

    if not refined:
        if starts:
            best_score, best_p = starts[0]
        else:
            best_p = normalize_sequence(np.full(n, 1.0 / n, dtype=np.float64))
            best_score = fast_score(best_p, m)
    else:
        best_p, best_score = min(refined, key=lambda x: x[0])

    if time.monotonic() < hard - 8.0:
        candidates = sort_score_pairs(list(refined if refined else starts))
        exact_best = None
        for _score, p in candidates[:3]:
            es = exact_score(p)
            if math.isfinite(es):
                if exact_best is None or es < exact_best[0]:
                    exact_best = (es, p)
        if exact_best is not None:
            best_p = normalize_sequence(exact_best[1])

    best_p = normalize_sequence(best_p)
    out = [float(x) for x in best_p]
    if not out:
        out = [1.0]
    return out
