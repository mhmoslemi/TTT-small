```python
def fft_convolve(a):
    """Compute full convolution a*a via FFT."""
    n = len(a)
    size = 2*n - 1

    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    b = np.fft.irfft(fa * fa, fft_size)
    return b[:size]
```

```python
def build_toeplitz_sparse(f, n):
    """Build the (2n-1) × n Toeplitz matrix for convolving with f."""
    rows, cols, vals = [], [], []
    for j in range(n):
        for k in range(2*n - 1):
            i = k - j
            if 0 <= i < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
```

```python
def build_conv_toeplitz(f):
    """Sparse (2n-1) x n matrix T where (T @ g) = convolve(f, g)."""
    n = len(f)
    m = 2*n - 1
    # Column j has nonzeros at rows j, j+1, ..., j+n-1
    # T[k, j] = f[k-j] for k-j in [0, n-1]
    rows = []
    cols = []
    data = []
    for j in range(n):
        k_start = j
        k_end = min(j + n, m)
        rows.extend(range(k_start, k_end))
        cols.extend([j] * (k_end - k_start))
        data.extend(f[:k_end - k_start])
    return sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
```
