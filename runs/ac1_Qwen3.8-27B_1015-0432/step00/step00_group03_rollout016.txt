```python
from scipy import sparse
from scipy.optimize import linprog
import time

def build_conv_matrix(f):
    """Build the (2n-1) x n convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    # For each column j, non-zeros at rows j..j+n-1 with values f[0], f[1], ..., f[n-1]
    js = np.arange(n)
    # Column j has entries at rows j+offset for offset in 0..n-1
    # We can build this as a banded matrix
    # Actually, let's use the fact that it's a Toeplitz-like matrix
    row_idx = []
    col_idx = []
    data = []
    for j in range(n):
        row_idx.extend(range(j, j+n))
        col_idx.extend([j]*n)
        data.extend(f.tolist())
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
    return A
```
