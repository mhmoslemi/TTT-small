    n = len(a)
    b = np.convolve(a, a, mode='full')  # or FFT
    k_star = np.argmax(b)
    # Gradient of b[k_star] w.r.t. a[j]:
    # d/db: a[k_star - j] + a[j - k_star] (with zero extension)
    grad = np.zeros(n)
    for j in range(n):
        val = 0
        idx1 = k_star - j
        if 0 <= idx1 < n:
            val += a[idx1]
        idx2 = j - k_star
        if 0 <= idx2 < n:
            val += a[idx2]
        grad[j] = val
    # This is equivalent to: grad = a_rolled_right(k_star) + a_rolled_left(k_star) (padded)
    
    # Project: subtract mean to maintain sum
    grad = grad - np.mean(grad)
    # Clip: avoid making a negative
    a_new = a - lr * grad
    a_new = np.clip(a_new, 0, None)
    # Renormalize
    if a_new.sum() < 0.01:
        return a
    a_new = a_new * (n / a_new.sum())  # fix sum = n
    return a_new
