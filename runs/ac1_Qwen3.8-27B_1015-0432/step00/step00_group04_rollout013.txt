import time


def _grad_conv_lag(a, k):
    """dC(k)/da for linear convolution C = a*a, at lag k. O(n)."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        js = np.arange(j_min, j_max + 1)
        g[js] = 2.0 * a[k - js]
    return g


def _fast_obj(a):

    n = a.size
    S = a.sum()
    if S < 0.01:
        return float('inf')
    C = np.convolve(a, a, mode='full')

import numpy as np
import time


def _grad_conv(a, k):
    """dC(k)/da for linear convolution C=a*a at lag k. O(n)."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    j0 = max(0, k - n + 1)
    j1 = min(n - 1, k)
    if j0 <= j1:
        js = np.arange(j0, j1 + 1)
        g[js] = 2.0 * a[k - js]
    return g


def _obj(a):
    """AC1 objective value."""
    n = a.size
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    C = np.convolve(a, a, mode='full')
    return 2.0 * n * float(C.max()) / (S * S)
