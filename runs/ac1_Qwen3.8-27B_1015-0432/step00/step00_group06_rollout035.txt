import numpy as np
import time

def _fft_grad(c, a, S):
    """Compute cross-correlation grad[i] = sum_m c[m+i]*a[m] for i=0..n-1."""
    Cf = np.fft.fft(c, n=S)
    Af = np.fft.fft(a, n=S)
    return np.fft.ifft(Cf * np.conj(Af)).real[:len(a)]

            stall = 0

                a = np.maximum(a, 0)

                stall = 0
                lr = 0.5
                if time.monotonic() > deadline - 10:
