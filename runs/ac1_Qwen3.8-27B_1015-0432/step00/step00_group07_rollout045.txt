def block_perturb(best, rng, n, sigma):

    c = best.copy()
    lo = max(1, n // 200)
    hi = max(lo + 1, n // 8)

    """Move toward uniform (the minimizer) by blending factor t."""
    mean = best.mean()
    return np.clip((1.0 - t) * best + t * mean, 0.0, 1000.0)
