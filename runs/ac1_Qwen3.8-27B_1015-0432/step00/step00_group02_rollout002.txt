def compute_score_fftleast(a):
    """Compute 2n*max(autocorr)/sum^2 using FFT, O(n log n)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    # Linear autocorrelation via FFT
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < 2 * n:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    pacf = np.fft.irfft(fa * fa.real, fft_size)[:size]
    max_b = pacf.max()

def _fftleast_score(a, fft_size):
