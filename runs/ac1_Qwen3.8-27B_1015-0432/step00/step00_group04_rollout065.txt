import numpy as np
import time
import scipy.sparse as sp
import scipy.optimize as opt


def quick_score(f):
    s = float(np.sum(f))
    if s < 0.01:
        return np.inf
    n = f.shape[0]
    conv = np.convolve(f, f)
    return 2.0 * n * float(np.max(conv)) / (s * s)


def make_struct(n):
    nnz = n * n
    rows = np.empty(nnz, dtype=np.int32)
    cols = np.empty(nnz, dtype=np.int32)
    ar = np.arange(n, dtype=np.int32)
    for j in range(n):
        rows[j * n:(j + 1) * n] = j + ar
        cols[j * n:(j + 1) * n] = j
    return rows, cols


def build_conv_mat(f, rows, cols):
    n = f.shape[0]
    data = np.tile(f, n)
    return sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()


def lp_solve(f, M, rows, cols):
    n = f.shape[0]
    A = build_conv_mat(f, rows, cols)
    Afull = sp.vstack([A, -sp.eye(n, dtype=np.float64)], format='csr')
    b = np.empty(3 * n - 1, dtype=np.float64)
    b[:2 * n - 1] = M
    b[2 * n - 1:] = 0.0
    res = opt.linprog(-np.ones(n, dtype=np.float64), A_ub=Afull, b_ub=b,
                      bounds=[(0.0, None)] * n, method='highs')
    if res.status == 0 and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1e-12:
            return g
    return None


def block_pass(f, n_blocks, scales):
    n = f.shape[0]
    bs = max(1, n // n_blocks)
    best = f.copy()
    best_s = quick_score(best)
    for b in range(n_blocks):
        lo = b * bs
        hi = min(n, (b + 1) * bs)
        for sc in scales:
            trial = best.copy()
            trial[lo:hi] *= sc
            s = quick_score(trial)
            if s < best_s:
                best = trial
                best_s = s
    return best, best_s


def random_perturb(f, rng, frac, sigma):
    n = f.shape[0]
    sel = rng.random(n) < frac
    cnt = int(sel.sum())
    if cnt == 0:
        return f
    trial = f.copy()
    trial[sel] *= np.exp(rng.normal(0.0, sigma, cnt))
    return np.clip(trial, 0.0, 1000.0)

        f = np.ones(128, dtype=np.float64)

    f = f * (1.0 + 0.03 * np.sin(4.0 * np.pi * idx / n))
