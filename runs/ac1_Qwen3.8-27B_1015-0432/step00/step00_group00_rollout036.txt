import numpy as np
import time

def fft_convolve(x):
    """Compute linear convolution x*x using FFT."""
    n = len(x)
    size = 1
    while size < 2*n-1:
        size *= 2
    X = np.fft.rfft(x, size)
    return np.fft.irfft(X * X, size)[:2*n-1]

def compute_score_fast(x):
    """Fast score computation using FFT."""
    conv = fft_convolve(x)
    max_b = np.max(conv)
    sum_a = np.sum(x)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * len(x) * max_b / (sum_a * sum_a)

def active_set_gradient(x, k_star, top_k=5):

    conv = fft_convolve(x)

def rfft_conv(x):
    n = len(x)
    size = 1
    while size < 2*n - 1:

    return np.fft.irfft(X * X, size)[:2*n-1]
