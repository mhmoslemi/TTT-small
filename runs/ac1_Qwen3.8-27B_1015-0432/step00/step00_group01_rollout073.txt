import math

def _finite(a):
    return np.isfinite(a).all()


def _clip(a):
    return np.clip(a, 0.0, 1e4)


def _peak_lags(conv, L):
    m = conv.size
    if m <= L:
        return np.arange(m)
    return np.argpartition(-conv, L)[:L]


def _subgrad_dir(a, L):
    """Subgradient direction for the scale-free objective max(conv)/sum^2."""
    n = a.size
    conv = np.convolve(a, a)
    S = float(a.sum())
    mx = float(conv.max())
    act = _peak_lags(conv, L)
    sb = np.zeros(n)
    for k in act:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = k - np.arange(lo, hi + 1)
        sb[lo:hi + 1] += a[idx]
    sb /= max(1, len(act))
    h = (S * sb - 2.0 * mx) / (S * S)
    return h, mx

    best_score = float("inf")
    for a in inits:
        a = _clip(a)
        if a.size == 0 or not _finite(a):
            continue
        sc = _score(a)
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best = a.copy()

    if best is None:
        best = np.ones(n)
        best_score = _score(best)

    local = best.copy()
    local_score = best_score
    noimp = 0
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        frac = min(1.0, (time.monotonic() - t0) / max(1.0, budget_s - 6.0))
        stuck = min(1.0, noimp / 300.0)

        h, _mx = _subgrad_dir(local, 16)
        step = 0.01 + 0.06 * stuck
        sigma = 0.03 + 0.30 * stuck

        cands = []
        for st in (0.3 * step, step, 3.0 * step):
            cands.append(_clip(local - st * h))
        for _ in range(2):
            cands.append(_perturb(local, rng, sigma))
        cands.append(_tilt(local, rng, 0.05 + 0.5 * stuck))
        if stuck > 0.6 and rng.random() < 0.2:
            cands.append(_perturb(best, rng, 0.6))

        improved = False
        for cand in cands:
            if cand.size == 0 or not _finite(cand) or cand.sum() < 0.01:
                continue
            sc = _score(cand)
            if math.isfinite(sc) and sc < local_score - 1e-12:
                local = cand.copy()
                local_score = sc
                improved = True
                if sc < best_score:
                    best_score = sc
                    best = cand.copy()
                break
        noimp = 0 if improved else noimp + 1

    out = [float(v) for v in _clip(best)]
    if not out or any((math.isnan(v) or math.isinf(v) or v < 0.0) for v in out):
        out = [1.0] * n
    # Ensure a definitely valid, non-degenerate return.
    cur = _score(np.asarray(out))
    if not math.isfinite(cur):
        out = [float(v) for v in np.ones(n)]
    return out
