import numpy as np
import time


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + float(budget_s) - 5.0

    idx = np.arange(1, n + 1, dtype=np.float64)
    best = None
    best_score = np.inf

    # Try several power-law ramps: a[i] = 1000*(i+1)^p / n^p
    # Score for p=1 is ~4/3, well below target 1.5027
    for p in [1.0, 1.3, 1.5, 1.8, 2.0, 2.5, 3.0]:
        if time.monotonic() > deadline:
            break
        candidate = 1000.0 * (idx ** p) / (n ** p)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate.copy()
            best_score = score

    # Fallback: pure linear ramp (guaranteed score ≈ 4/3)
    if best is None:
        best = 1000.0 * idx / n
        best_score = float(evaluate_sequence(best.tolist()))

    # Light local search: small multiplicative block perturbations
    step = 0.03
    while time.monotonic() < deadline:
        width = max(5, n // 200)
        start = int(rng.integers(0, max(1, n - width)))
        noise = rng.normal(0.0, step, size=width)
        candidate = best.copy()
        candidate[start:start + width] *= np.exp(noise)
        candidate = np.clip(candidate, 0.0, 1000.0)
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best = candidate
            best_score = score
        else:
            step = max(0.005, step * 0.95)

    return [float(x) for x in best]
