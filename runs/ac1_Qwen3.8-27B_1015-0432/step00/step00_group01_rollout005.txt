def subgradient_step(a, lr, peak_idx):
    n = len(a)
    grad = np.zeros(n)
    # The peak is at lag peak_idx
    # (a*a)[peak_idx] = sum_j a[j] * a[peak_idx - j]
    # d/dx a[j] of this = a[peak_idx - j] (if valid)
    for j in range(n):
        k = peak_idx - j
        if 0 <= k < n:
            grad[j] = a[k]
    # Also, changing a[j] affects (a*a)[j+offset] for other lags
    # But for the subgradient of the MAX, we only need the active constraint
    # Normalize gradient
    norm = np.linalg.norm(grad)
    if norm > 0:
        grad /= norm
    return a - lr * grad
