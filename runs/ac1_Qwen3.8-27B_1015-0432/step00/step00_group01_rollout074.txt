import numpy as np
from scipy.optimize import linprog
from scipy.signal import fftconvolve
from scipy.sparse import coo_matrix, csr_matrix

def _conv_mat(a, lags):
    n = len(a)
    r, c, v = [], [], []
    for i, k in enumerate(lags):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            r += [i] * (hi - lo)
            c += list(range(lo, hi))
            v += list(a[k - hi + 1: k - lo + 1][::-1])
    if not r:
        return csr_matrix((len(lags), n), dtype=np.float64)
    return coo_matrix((v, (r, c)), shape=(len(lags), n), dtype=np.float64).tocsr()


def _lp_dir(a, M, K, tlim):
    n = len(a)
    conv = fftconvolve(a, a)
    k = min(K, 2 * n - 1)
    lags = np.argpartition(conv, -k)[-k:]
    for _ in range(3):
        A = _conv_mat(a, lags)
        res = linprog(-np.ones(n), A_ub=A, b_ub=M * np.ones(len(lags)),
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': max(1.0, tlim)})
        if not res.success or res.x is None:
            return None
        d = np.maximum(res.x, 0.0)
        if d.sum() < 1e-8:
            return None
        cd = fftconvolve(a, d)
        viol = np.where(cd > M * 1.001)[0]
        if len(viol) == 0:
            return d
        lags = np.unique(np.concatenate([lags, viol]))
        if len(lags) > K:
            order = np.argsort(cd[lags])[::-1]
            lags = lags[order[:K]]
    return d

        at[s2:s2 + w2] *= np.exp(rng.normal(0.0, 0.25))
    return np.clip(at, 0.0, None)
