    """Find coordinates that contribute most to c[k]."""
    n = len(a)
    # c[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    if i_lo > i_hi:
        return np.array([], dtype=int)
    idx = np.arange(i_lo, i_hi + 1)
    idx2 = k - idx
    contrib = a[idx] * a[idx2]
    total = contrib.sum()
    if total <= 0:
        return idx
    # Find top contributors
    n_top = max(1, int(len(idx) * top_frac))
    sorted_idx = idx[np.argsort(contrib)[::-1][:n_top]]
    return sorted_idx

import time

def _quick_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * np.convolve(a, a).max() / (s * s)

def _block_perturb(a, rng):
    n = a.size
    w = int(rng.integers(max(1, n // 100), max(2, n // 5)))
    s = int(rng.integers(0, n - w + 1))
    e = min(n, s + w)
    f = float(np.exp(rng.normal(0.0, 0.15)))
    c = a.copy()
    c[s:e] *= f
    return np.clip(c, 0.0, 1000.0)

def _smooth_step(a, mean_val, alpha):
    return np.clip((1 - alpha) * a + alpha * mean_val, 0.0, 1000.0)

def _hot_reduce(a, rng):
    """Reduce values at the lag-0 peak contributors."""
    n = a.size
    b = np.convolve(a, a)
    k_peak = int(np.argmax(b))
    i_lo = max(0, k_peak - n + 1)
    i_hi = min(n - 1, k_peak)
    if i_lo > i_hi:
        return a
    # Find the single most contributing index pair
    idx = np.arange(i_lo, i_hi + 1)
    other = k_peak - idx
    contrib = a[idx] * a[other]
    i_max = idx[int(np.argmax(contrib))]
    # Try reducing a[i_max] a bit
    c = a.copy()
    reduce_by = 0.05 * a[i_max]
    c[i_max] = max(0.0, a[i_max] - reduce_by)
    # Also try reducing the partner
    j_max = k_peak - i_max
    if j_max != i_max and 0 <= j_max < n:
        c[j_max] = max(0.0, a[j_max] - reduce_by)
    return np.clip(c, 0.0, 1000.0)

Actually, let me reconsider. The `_quick_score` also calls `np.convolve`, so every iteration is O(n log n) (FFT-based) or O(n²). For n=2535, O(n log n) ≈ 30000 operations, which is very fast. We can do millions of iterations in the budget.

Let me also add a "gradient step" approach: compute the gradient of the max autocorrelation term w.r.t. each a_i, and take a step in the negative gradient direction (projected onto nonnegativity).

For the peak at lag k*, the "gradient" of c_{k*} w.r.t. a_i is:
∂c_{k*}/∂a_i = a[k*-i] + a[i] (if both indices are valid) [counting both forward and backward]

Wait, c_{k*} = sum_j a[j]*a[k*-j]. The gradient w.r.t. a_i appears in two terms: when j=i (giving a[k*-i]) and when k*-j=i, i.e., j=k*-i (giving a[i]). So if i ≠ k*-i, the gradient is a[k*-i] + a[i]. If i = k*-i (i.e., 2i = k*), the gradient is 2*a[i].

import numpy as np
import time

def _quick_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    if n == 0:
        return float('inf')
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * np.convolve(a, a).max() / (s * s))

def _block_perturb(a, rng):
    n = a.size
    w = int(rng.integers(max(1, n // 150), max(2, n // 4)))
    s = int(rng.integers(0, n - w + 1))
    e = min(n, s + w)
    f = float(np.exp(rng.normal(0.0, 0.12)))
    c = a.copy()
    c[s:e] *= f
    if rng.random() < 0.3:
        c = 0.5 * (c + c[::-1])
    return np.clip(c, 0.0, 1000.0)

def _hot_reduce(a):
    n = a.size
    b = np.convolve(a, a)
    kp = int(np.argmax(b))
    lo = max(0, kp - n + 1)
    hi = min(n - 1, kp)
    if lo > hi:
        return a
    idx = np.arange(lo, hi + 1)
    contrib = a[idx] * a[kp - idx]
    top = max(1, len(idx) // 5)
    hot = idx[np.argsort(contrib)[::-1][:top]]
    c = a.copy()
    for i in hot:
        c[i] *= 0.97
    return np.clip(c, 0.0, 1000.0)

def _gradient_step(a, learn_rate):
    n = a.size
    b = np.convolve(a, a)
    kp = int(np.argmax(b))
    lo = max(0, kp - n + 1)
    hi = min(n - 1, kp)
    if lo > hi:
        return a
    idx = np.arange(lo, hi + 1)
    grad = np.zeros(n)
    g = a[kp - idx]
    np.add.at(grad, idx, g)
    np.add.at(grad, kp - idx, g)
    if (2 * idx == kp).any():
        for i in idx[2 * idx == kp]:
            grad[i] += a[i]
    c = a - learn_rate * grad
    return np.clip(c, 0.0, 1000.0)

def _smooth_toward_mean(a, alpha):
    m = a.mean()
    if m < 1e-9:
        m = 1.0
    return np.clip((1.0 - alpha) * a + alpha * m, 0.0, 1000.0)

def _two_block(a, rng):
    n = a.size
    split = int(rng.integers(n // 4, 3 * n // 4 + 1))
    f1 = rng.uniform(0.5, 2.0)
    f2 = rng.uniform(0.5, 2.0)
    c = a.copy()
    if split > 0:
        c[:split] *= f1
    if split < n:
        c[split:] *= f2
    return np.clip(c, 0.0, 1000.0)
