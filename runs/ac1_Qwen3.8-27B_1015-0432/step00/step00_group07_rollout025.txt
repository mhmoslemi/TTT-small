import numpy as np
import time

def eval_local(a):
    """Fast local evaluation matching the authoritative evaluator."""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a)
    return float(2.0 * a.size * c.max() / (s * s))

def smooth_delta(n, rng, nfreq):
    """Zero-mean smooth random vector via low-frequency DFT components."""
    nfft = 1 << (2 * n - 1).bit_length()
    half = nfft // 2 + 1
    spec = np.zeros(half)
    kmax = min(nfreq, half - 1)
    if kmax > 0:
        spec[1:kmax + 1] = rng.normal(0.0, 1.0, kmax)
    d = np.fft.irfft(spec, nfft)[:n]
    d -= d.mean()
    return d
