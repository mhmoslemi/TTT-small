def _eval(a):
    return float(evaluate_sequence(a.tolist()))


def _perturb_blocks(base, rng, n):
    c = base.copy()
    n_blocks = int(rng.integers(1, 4))
    for _ in range(n_blocks):
        w_lo = max(1, n // 500)
        w_hi = max(w_lo + 1, n // 15)
