def fast_score(a, n):
    conv = np.convolve(a, a)
    s = float(a.sum())
    if s < 0.01:
        return 1e18
    return float(2.0 * n * conv.max()) / (s * s)


def peak_update(a, conv, n, peaks, lr):
    for p in peaks:
        p = int(p)

        i_max = min(p, n - 1)
        if i_min > i_max:
            continue
