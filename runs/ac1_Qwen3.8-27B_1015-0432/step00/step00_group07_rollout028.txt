import time

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    
    n = len(height_sequence_1)
    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    
    # Initialize with uniform
    base = np.ones(n, dtype=np.float64)
    best = base.copy()
    best_score = evaluate_sequence(best.tolist())
    
    # Try parent
    ps = evaluate_sequence(parent.tolist())
    if np.isfinite(ps) and ps < best_score:
        best = parent.copy()
        best_score = ps
    
    # Precompute templates
    x = np.arange(n, dtype=np.float64)
    if n > 1:
        x_norm = 2.0 * x / (n - 1) - 1.0  # linear, -1 to 1
    else:
        x_norm = np.zeros(n)
    
    templates = [
        x_norm,                          # linear ramp
        -(x_norm ** 2),                  # inverted parabola (peaks at center)
        x_norm ** 2 - 1.0,              # U-shape (peaks at edges)
        np.sin(np.pi * x / max(1, n-1)), # sine arch
        np.cos(np.pi * x / max(1, n-1)), # cosine
    ]
    # Remove zeros from templates
