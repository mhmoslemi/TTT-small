import numpy as np
import time

def linear_conv(a):
    n = len(a)
    size = 1 << (2*n-2).bit_length()
    fa = np.fft.rfft(a, size)
    return np.fft.irfft(fa*fa, size)[:2*n-1]

def init_ramps(n, rng):
    inits = []
    # Linear ramp
    x = np.arange(1, n+1, dtype=np.float64)
    inits.append(x / x.sum())
    # Shifted ramp (zero at start)
    x0 = np.arange(n, dtype=np.float64)
    if x0.sum() > 0:
        inits.append(x0 / x0.sum())
    # Power ramps
    for p in [0.5, 1.5, 2.0]:
        xp = x**p
        if xp.sum() > 0.01:
            inits.append(xp / xp.sum())
    # Exponential ramps
    for alpha in [1.0, 2.0, 3.0]:
        xe = np.exp(alpha * np.linspace(0, 1, n))
        inits.append(xe / xe.sum())
    # Reversed ramp
    inits.append((x[::-1]) / x.sum())
    # Hanning (for comparison)
    t = np.linspace(0, np.pi, n)
    h = 0.5*(1-np.cos(t))
    inits.append(h / h.sum())
    return inits
