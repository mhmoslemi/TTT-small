import numpy as np
import time

def fft_convolve(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    N = 1
    while N < size:
        N *= 2
    fa = np.fft.rfft(a, N)
    fa2 = fa * fa
    result = np.fft.irfft(fa2, N)[:size]
    return result

def compute_gradient(a, M):

    c = np.sum(a) ** 2  # placeholder, will compute properly
