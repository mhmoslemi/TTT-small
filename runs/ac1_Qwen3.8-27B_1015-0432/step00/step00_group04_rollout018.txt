import numpy as np
import time


def _full_pressure(a, w):
    """Compute sum_k w[k]*a[k-i] for i=0..n-1 via one FFT convolution."""
    n = len(a)
    rev_w = w[::-1]
    full = np.convolve(a, rev_w)
    return full[n - 1:2 * n - 1][::-1]


def propose_candidate(seed=42, budget_s=1100, **kwargs):

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000).astype(np.float64).copy()
    n = a.size
    if n < 2 or not np.isfinite(a).all():
        a = np.ones(256, dtype=np.float64)
        n = a.size

    target_sum = np.sqrt(2.0 * n)
    s = a.sum()
    if s < 1e-6:
        a[:] = 1.0
        s = float(n)
    a *= target_sum / s

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))

    T = 50.0
    eta = 0.06
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = np.convolve(a, a)
        cmax = conv.max()
        if cmax < 1e-14:
            break

        # Soft-max weights peaked at high-autocorrelation lags
        w = np.exp((conv - cmax) / T)
        ws = w.sum()
        if ws < 1e-30:
            break
        w /= ws

        # Gradient of soft-max autocorrelation w.r.t. each a[i]
        pressure = _full_pressure(a, w)
        pressure = np.maximum(pressure, 0.0)
        pmax = pressure.max()
        if pmax < 1e-14:
            break
        pressure /= pmax

        # Multiplicative update: reduce high-pressure, boost low-pressure
        mp = pressure.mean()
        a_new = a * np.exp(-eta * (pressure - mp))
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 0.01:
            a_new = a.copy()
        else:
            a_new *= (a.sum()) / s_new

            a = a_new
            no_improve = 0
            T = max(T * 0.97, 2.0)

            eta *= 0.96
            T *= 1.02

                T = 30.0
                no_improve = 0
