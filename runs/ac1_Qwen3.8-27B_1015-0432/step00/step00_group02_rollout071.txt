import numpy as np

def fft_convolve(a):
    """Compute full convolution a*a using FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)[:size]
    return result

def score_fast(a):
    """Compute the AC1 score using FFT convolution."""
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a)
    max_b = np.max(c)
    return float(2 * n * max_b / (s * s))
