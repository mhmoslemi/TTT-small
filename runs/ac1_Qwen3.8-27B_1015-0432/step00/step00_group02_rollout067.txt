    n = len(a) + len(b) - 1

def fft_conv1d(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb)[:n]

def fast_score(a):
    n = len(a)
    sa = a.sum()
    if sa < 1e-15:
        return float('inf')
    conv = fft_conv1d(a, a)
    return 2.0 * n * conv.max() / (sa * sa)

    cands.append((1.0 + 4.0 * (1.0 - t) ** 2) / (1.0 + 4.0 * (1.0 - t) ** 2).sum())
    cands.append((1.0 + 2.0 * np.sin(np.pi * t)) / (1.0 + 2.0 * np.sin(np.pi * t)).sum())
    cands.append((1.0 - 0.5 * np.sin(np.pi * t)) / (1.0 - 0.5 * np.sin(np.pi * t)).sum())
    rv = rng.random(n)
    cands.append(rv / rv.sum())
    rv2 = rng.chisquare(2.0, n)
    cands.append(rv2 / rv2.sum())

    for c in cands:

        if c.sum() > 1e-15:

        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = c.copy()

    # Main gradient-based local search
    a = best_a.copy()
    lr = 0.3
    v = np.zeros(n)
    beta = 0.85
    stuck = 0
    m_lags = 200

    while time.monotonic() < deadline:
        conv = fft_conv1d(a, a)
        peak = conv.max()
        thresh = peak * 0.95
        top_idx = np.where(conv >= thresh)[0]
        if len(top_idx) > m_lags:
            w = conv[top_idx]
            w = w / w.sum()
            order = np.argsort(w)[-m_lags:]
            top_idx = top_idx[order]
            w = w[order]
        else:
            w = np.ones(len(top_idx)) / max(1, len(top_idx))

        grad = np.zeros(n)
        for j in range(len(top_idx)):

            lo = max(0, k - n + 1)

            v *= 0.5

                best_a = a_new.copy()
                a = a_new.copy()
                stuck = 0
                lr = min(lr * 1.03, 3.0)

                    a = np.clip(a, 0.0, None)

                    lr = 0.3
                    v = np.zeros(n)
                    stuck = 0
                else:

                    if lr < 0.02:
                        lr = 0.3
                        v = np.zeros(n)
                        stuck = 0

        if stuck == 0 and stuck % 5000 == 0:
            s_val = float(evaluate_sequence(best_a.tolist()))
            if np.isfinite(s_val) and s_val < best_score:
                best_score = s_val

    # Final verify
    fs = float(evaluate_sequence([float(x) for x in best_a]))
    if not np.isfinite(fs):
        fb = np.ones(n) / n
        return [float(x) for x in fb]
    return [float(x) for x in best_a]
