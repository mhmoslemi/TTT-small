import numpy as np

def fft_convolve(a):
    """Compute full convolution using FFT. O(n log n)."""
    n = len(a)
    size = 1
    while size < 2*n - 1:
        size *= 2
    A = np.fft.rfft(a, size)
    B = A * A
    return np.fft.irfft(B, size)[:2*n-1]

def compute_gradient_at_peak(a, b, k_star):
    """Compute subgradient of b[k_star] w.r.t. a.
    b[k] = sum_j a[j]*a[k-j], so db[k]/da[i] = a[k-i] for valid range."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            g[i] = a[j]
    return g

soft_max(b) ≈ log(sum(b[k]^(1/T))) for small T, or sum(b[k] * exp(b[k]/T) / sum(exp(b[j]/T))) for temperature T.

d(soft_max)/da[i] = sum_k w[k] * a[k-i] where w[k] = exp(b[k]/T) / sum_j exp(b[j]/T)

This is a weighted sum of shifted versions of a. Can be computed as:
d(soft_max)/da = correlate_weights * a where weights are the softmax weights reversed.

Actually, this is a convolution! `sum_k w[k] * a[k-i]` = `(w * a)[i]` where * denotes convolution.

def compute_score(b, s, n):
    return 2.0 * n * b.max() / (s * s)


def softmax_grad(A, size, T, n):
    B = A * A
    b = np.fft.irfft(B, size)[:2 * n - 1]
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w /= w.sum()
    V = np.fft.rfft(w, size)
    G = np.fft.irfft(np.conj(A) * V, size)[:n]
    return G, b


def perturb_block(a, rng, n):
    width = int(rng.integers(max(1, n // 80), max(2, n // 6) + 1))
    start = int(rng.integers(0, n - width))
    a[start:start + width] *= np.exp(rng.normal(0.0, 0.12))
    return a


def perturb_block_big(a, rng, n):
    width = int(rng.integers(max(1, n // 20), max(2, n // 3) + 1))
    start = int(rng.integers(0, n - width))
    a[start:start + width] *= np.exp(rng.normal(0.0, 0.6))
    return a


def normalize_nonneg(a, n):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return None
    return a * (n / s)


def make_init(idx, n, rng):
    inits = []
    base = 1.0 + 0.9 * np.cos(2.0 * np.pi * idx / n)
    base = np.clip(base, 0.01, None)
    inits.append(base * (n / base.sum()))
    q = 1.0 + 0.7 * (2.0 * idx / n - 1.0) ** 3
    q = np.clip(q, 0.01, None)
    inits.append(q * (n / q.sum()))
    sin = 1.0 + 0.5 * np.sin(4.0 * np.pi * idx / n) * np.cos(2.0 * np.pi * idx / n)
    sin = np.clip(sin, 0.01, None)
    inits.append(sin * (n / sin.sum()))
    r = np.abs(rng.normal(1.0, 0.4, n))
    inits.append(r * (n / r.sum()))
    return inits


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    parent = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if parent.ndim != 1 or parent.size == 0:
        parent = np.ones(256, dtype=np.float64)
    n = len(parent)
    idx = np.arange(n, dtype=np.float64)
    size = 1 << (2 * n - 1).bit_length()

    # Multi-start initialization
    p = parent.copy()
    ps = p.sum()
    inits = make_init(idx, n, rng)
    if ps >= 0.01:
        inits.insert(0, p * (n / ps))

    best = np.ones(n)
    best_score = np.inf
    for init in inits:
        b, _, _ = fft_convolve(init)
        sc = compute_score(b, init.sum(), n)
        if sc < best_score:
            best_score = sc
            best = init.copy()

    chk = evaluate_sequence(best.tolist())
    if np.isfinite(chk) and chk < best_score:
        best_score = chk

    # Phase 1: Soft-max gradient descent with annealing
    a = best.copy()
    T = float(n) * 0.5
    T_min = 1.0
    lr = 0.4
    stagnant = 0

    while time.monotonic() < deadline:
        A = np.fft.rfft(a, size)
        G, b = softmax_grad(A, size, T, n)
        s = a.sum()
        sc = compute_score(b, s, n)

        if sc < best_score - 1e-10:
            best_score = sc
            best = a.copy()
            stagnant = 0
        else:
            stagnant += 1

        gn = np.linalg.norm(G)
        if gn > 1e-14:
            a = a - lr * G / gn
        a = normalize_nonneg(a, n)
        if a is None:
            a = best.copy()

        T = max(T_min, T * 0.997)
        lr = max(0.05, lr * 0.999)

        if rng.random() < 0.04:
            a = perturb_block_big(a, rng, n)
            a = normalize_nonneg(a, n)
            if a is None:
                a = best.copy()

        if stagnant > 2000:
            a = best.copy()
            a = perturb_block_big(a, rng, n)
            a = normalize_nonneg(a, n)
            if a is None:
                a = best.copy()
            stagnant = 0
            T = max(T, 10.0)

    # Phase 2: Fine block perturbations
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        a = best.copy()
        a = perturb_block(a, rng, n)
        a = normalize_nonneg(a, n)
        if a is None:
            continue
        b, _, _ = fft_convolve(a)
        sc = 2.0 * n * b.max() / n
        if sc < best_score - 1e-12:
            best_score = sc
            best = a.copy()

    # Final validation
    result = [float(x) for x in best]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [float(x) for x in parent]
        fs = evaluate_sequence(result)
        if not np.isfinite(fs):
            result = [1.0] * n
    return result
