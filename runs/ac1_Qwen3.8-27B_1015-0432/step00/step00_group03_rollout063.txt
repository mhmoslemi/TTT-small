def build_conv_matrix(f, n):
    """Build sparse (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
