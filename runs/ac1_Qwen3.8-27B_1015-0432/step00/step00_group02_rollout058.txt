import numpy as np
import time

def convolve_self(a):
    """Compute linear self-convolution via FFT. O(n) memory."""
    n = len(a)
    size = 2 * n - 1
    S = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, S)
    c = np.fft.irfft(fa * fa, S)[:size]
    return np.maximum(c, 0.0)  # numerical safety

def peak_gradient(a, c, K=5):
    """Subgradient of top-K sum of c. O(K*n) time, O(n) memory."""
    n = len(a)
    if K >= len(c):
        ks = np.arange(len(c))
    else:
        ks = np.argpartition(c, -K)[-K:]
    grad = np.zeros(n)
    for k in ks:
        k = int(k)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            grad[jmin:jmax + 1] += 2.0 * a[k - jmax:k - jmin - 1:-1]
    return grad / len(ks)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(0.0, float(budget_s) - 5.0)
    
    n = len(height_sequence_1)
    a = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / s
    
    c = convolve_self(a)
    best_score = float(evaluate_sequence(a.tolist()))
    best = a.copy()
    
    eta = 1.0
    K = 5
    eval_interval = 5
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = convolve_self(a)
        cmax = c.max()
        if cmax < 1e-15:
            break
        
        grad = peak_gradient(a, c, K)
        grad_scale = np.maximum(cmax, 1e-15)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad / grad_scale)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            a = best.copy()
        else:
            a = a / s
        
        # Periodic perturbation for diversity
        if it % 200 == 0:
            noise = rng.normal(0.0, 0.05, n)
            a = a * np.exp(noise)
            a = a / a.sum()
        
        # Evaluate
        if it % eval_interval == 0:
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best = a.copy()
        
        # Slowly reduce step size
        if it % 100 == 0:
            eta *= 0.98
    
    # Final check
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best = a.copy()
    
    return [float(x) for x in best]

    w = np.exp(lam * (c - cmax) / cmax)
    w = w / w.sum()

    S = 1 << (3 * n - 3).bit_length()
    A = np.fft.rfft(a, S)
    W = np.fft.rfft(w, S)
    R = np.fft.irfft(np.conj(A) * W, S)
    return 2.0 * R[:n]

import numpy as np
import time


def conv_self(a):
    n = len(a)
    size = 2 * n - 1
    S = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, S)
    c = np.fft.irfft(fa * fa, S)[:size]
    return np.maximum(c, 0.0)


def smooth_grad(a, c, lam=80.0):
    n = len(a)
    cmax = c.max()
    if cmax < 1e-18:
        return np.zeros(n)
    w = np.exp(lam * (c - cmax) / cmax)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    S = 1 << (3 * n - 3 + 1 - 1).bit_length()
    A = np.fft.rfft(a, S)
    W = np.fft.rfft(w, S)
    R = np.fft.irfft(np.conj(A) * W, S)
    return 2.0 * R[:n]


def topk_grad(a, c, K=5):
    n = len(a)
    ks = np.argpartition(c, -K)[-K:]
    grad = np.zeros(n)
    for k in ks:
        k = int(k)
        jmin = max(0, k - n + 1)
        jmax = min(n - 1, k)
        if jmin <= jmax:
            grad[jmin:jmax + 1] += 2.0 * a[k - jmax:k - jmin - 1:-1]
    return grad / len(ks)
