import numpy as np
import time


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    n = len(height_sequence_1)

    # Phase 1: soft-gradient descent from one-sided exponential
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-1.3 * j / n)
    a /= a.sum()

    best_a = a.copy()
    best_score = float(evaluate_sequence(best_a.tolist()))

    kappa = 0.15
    lr = 0.08

    for it in range(30000):
        if time.monotonic() > deadline - 60.0:
            break
        conv = np.convolve(a, a)
        shifted = (conv - conv.max()) / kappa
        weights = np.exp(shifted)
        ws = weights.sum()
        if ws < 1e-300:
            break
        weights /= ws
        grad = np.correlate(weights, a, mode='valid')
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        stepped = False
        for trial in (lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05):
            new_a = a - trial * grad
            new_a = np.maximum(new_a, 0.0)
            s = new_a.sum()
            if s < 1e-12:
                continue
            new_a /= s
            score = float(evaluate_sequence(new_a.tolist()))
            if score < best_score:
                best_score = score
                best_a = new_a.copy()
                a = new_a
                stepped = True
                break
            if trial < lr:
                a = new_a
                stepped = True
                break
        if not stepped:
            lr *= 0.8
        kappa *= 0.9995

    # Phase 2: sparse LP refinement (matrix-free via sparse Hankel)
    try:
        import scipy.sparse as sp
        import cvxpy as cp

        def build_hankel_sparse(a_vec):
            nn = len(a_vec)
            rows = []
            cols = []
            vals = []
            for k in range(2 * nn - 1):
                jlo = max(0, k - nn + 1)
                jhi = min(k, nn - 1)
                for jj in range(jlo, jhi + 1):
                    rows.append(k)
                    cols.append(jj)
                    vals.append(a_vec[k - jj])
            return sp.csr_matrix(
                (vals, (rows, cols)), shape=(2 * nn - 1, nn))

        T = build_hankel_sparse(a)
        b_var = cp.Variable(n, nonneg=True)
        objective = cp.Maximize(cp.sum(b_var))
        constraints = [T @ b_var <= best_score / (2.0 * n) * (1.0 ** 2)]
        prob = cp.Problem(objective, constraints)

                        best_a = mixed.copy()
                        a = mixed

    for it in range(10000):

            break
        conv = np.convolve(a, a)
