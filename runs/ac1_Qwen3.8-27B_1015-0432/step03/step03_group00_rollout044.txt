        return 1e10
    c = _conv(a)
    mv, _ = _smax(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    L = _np2(3 * n)
    fa = fft(a, L)
    fp = fft(p, L)
    aflip = a[::-1]
    faf = fft(aflip, L)
    conv_part = ifft(fa * fp).real[:n]
    corr_full = ifft(faf * fp).real
    corr_part = corr_full[n - 1:2 * n - 1]
    gM = corr_part + conv_part
    return (2.0 * n / (S * S)) * gM - (2.0 * n * mv / (S ** 3))


def _eval(a):
    s = float(evaluate_sequence(a.tolist()))
    return s if np.isfinite(s) else 1e18


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 20.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(6733)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)

    sc = np.sqrt(2.0 * n)
    a0 = par / par.sum() * sc
    best = a0.copy()
    best_sc = _eval(best)
    cur = a0.copy()
    x = np.arange(n, dtype=np.float64)

    # Cosine and noise perturbations
    for amp in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        if time.monotonic() > dl - 400:
            break
        for fr in [1, 2, 3, 5, 7]:
            shift = n * rng.uniform()
            c = a0 * (1.0 + amp * np.cos(2.0 * np.pi * fr * (x + shift) / n))
            c = np.clip(c, 0, None)
            if c.sum() > 0.01:
                c = c / c.sum() * sc
                s = _eval(c)
                if s < best_sc:
                    best_sc = s
                    best = c.copy()
                    cur = c.copy()
    for _ in range(6):
        if time.monotonic() > dl - 400:
            break
        nz = cur * np.exp(0.04 * rng.standard_normal(n))
        nz = np.clip(nz, 0, None)
        if nz.sum() > 0.01:
            nz = nz / nz.sum() * sc
            s = _eval(nz)
            if s < best_sc:
                best_sc = s
                best = nz.copy()
    cur = best.copy()

    # L-BFGS-B with soft-max annealing
    bl = [(0.0, 1000.0)] * n
    taus = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tv in taus:
        if time.monotonic() > dl - 120:
            break
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': 2000, 'gtol': 1e-14,
                                    'maxls': 50, 'ftol': 1e-20,
                                    'maxfun': 5000, 'iprint': 0})
            nx = np.clip(res.x, 0.0, 1000.0)
            sn = nx.sum()
            if sn > 1e-8:
                nx = nx / sn * sc
            s2 = _eval(nx)
            if s2 < best_sc:
                best_sc = s2
                best = nx.copy()
            cur = nx.copy()
        except Exception:
            pass

    # Targeted peak reduction
    if time.monotonic() < dl - 40:
        cur = best.copy()
        for _rd in range(15):
            if time.monotonic() > dl - 40:
                break
            c = _conv(cur)
            top_k = 8
            idx = np.argsort(c)[-top_k:]
            improved = False
            for jp in idx:
                lo = max(0, jp - (n - 1))
                hi = min(n - 1, jp)
                ks = np.arange(lo, hi + 1)
                prods = cur[ks] * cur[jp - ks]
                tc = ks[np.argsort(prods)[-8:]]
                for frac in [0.98, 0.95, 0.92, 0.88, 0.83, 0.78]:
                    trial = cur.copy()
                    trial[tc] *= frac
                    trial = np.clip(trial, 0, None)
                    if trial.sum() > 0.01:
                        s2 = _eval(trial)
                        if s2 < best_sc:
                            best_sc = s2
                            best = trial.copy()
                            cur = trial.copy()
                            improved = True
                            break
            if not improved:
                break

    # Second L-BFGS-B pass at very low temperature
    if time.monotonic() < dl - 50:
        cur = best.copy()
        for tv in [0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]:
            if time.monotonic() > dl - 50:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 1500, 'gtol': 1e-15,
                                        'maxls': 40, 'ftol': 1e-21,
                                        'maxfun': 3000, 'iprint': 0})
                nx = np.clip(res.x, 0.0, 1000.0)
                sn = nx.sum()
                if sn > 1e-8:
                    nx = nx / sn * sc
                s2 = _eval(nx)
                if s2 < best_sc:
                    best_sc = s2
                    best = nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    # Random local search
    while time.monotonic() < dl - 5:
        cd = cur.copy()
        nw = int(rng.integers(3, max(4, n // 40)))
        sidx = int(rng.integers(0, n - nw))
        cd[sidx:sidx + nw] *= np.exp(rng.normal(0.0, 0.04, nw))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            s2 = _eval(cd)
            if s2 < best_sc:
                best_sc = s2
                best = cd.copy()
                cur = cd.copy()
        if rng.random() < 0.06:
            cur = best.copy()

    # Final cleanup: zero tiny values
    thresh = 0.001 * best.max()
    trial = best.copy()
    trial[trial < thresh] = 0.0
    if trial.sum() > 0.01:
        s2 = _eval(trial)
        if s2 < best_sc:
            best_sc = s2
            best = trial.copy()

    if not np.isfinite(best_sc):
        best = a0.copy()

    return [float(v) for v in best]
