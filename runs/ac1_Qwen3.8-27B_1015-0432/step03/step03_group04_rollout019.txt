    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def lp_weights(f, peaks, b, n):
    """Solve LP for optimal subgradient weights."""
    K = len(peaks)
    # A: n x K, A[j, k] = 2 * f[peaks[k] - j] if 0 <= peaks[k]-j < n
    # We want: A^T lambda >= 1 (component-wise)
    # Equivalently: -A^T lambda <= -1
    # But A is n x K... that's large. Let's build it column by column.
    # Column k: A[:, k] = 2 * f[peaks[k] - j] for valid j
    # This is a shifted version of f.
    
    # Actually, for efficiency, let's just build A as a (n, K) array
    # Each column k is 2*f shifted by peaks[k]
    A = np.zeros((n, K))
    for k_idx in range(K):
        p = peaks[k_idx]
        # A[j, k_idx] = 2*f[p-j] if 0 <= p-j < n
        j_min = max(0, p - n + 1)
        j_max = min(n - 1, p)
        if j_min <= j_max:
            A[j_min:j_max+1, k_idx] = 2.0 * f[p - j_max:p - j_min + 1][::-1]
    
    # Normalize: we want sum_k lambda_k * A[j,k] >= 1
    # -> -sum_k lambda_k * A[j,k] <= -1 for all j
    # Objective: minimize sum_k lambda_k * (1 - b[peaks[k]]/M)  (prefer higher peaks)
    # Or just minimize sum_k lambda_k * b[peaks[k]] / M
    
    c = b[peaks] / b[peaks].max()
    # Constraints: -A^T @ lambda <= -1 (n constraints)
    #              -lambda <= 0 (K constraints)
    #              sum lambda = 1
    # (K-1) <= sum lambda <= 1
    # Or: sum lambda = 1 as equality
    
    # Using scipy linprog: min c^T x s.t. A_ub x <= b_ub, A_eq x = b_eq, bounds
    A_ub = np.vstack([-A.T, -np.eye(K)])  # (n + K) x K
    b_ub = np.concatenate([-np.ones(n), np.zeros(K)])
    A_eq = np.array([[1.0] * K])
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(0, None)] * K, method='highs',
                  options={'maxiter': 10000})
    if res.success:
        w = res.x
        w = np.maximum(w, 0)
        w = w / w.sum()
        return w
    return None
