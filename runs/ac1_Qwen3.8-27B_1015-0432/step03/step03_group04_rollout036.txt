import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, top_frac, max_pk):
    n = f.shape[0]
    M = b.max()
    # Use softmax-like weighting over top peaks
    thresh = M * (1.0 - top_frac)
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(b)])
    # Softmax weighting
    w = b[peaks]
    w = np.exp((w - w.max()) / (M * 0.01 + 1e-30))
    w /= w.sum()
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, max_iter=500000, sym=False):
    n = f.shape[0]
    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.15
    no_improv = 0
    restarts = 0
    top_frac_start = 0.01
    top_frac_end = 0.001
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
