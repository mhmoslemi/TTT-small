import numpy as np
import scipy.sparse
import scipy.optimize
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    n = len(f)
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def build_conv_matrix(f):
    """Build sparse matrix A such that (A @ g)[k] = (f * g)[k] for k=0,...,2n-2"""
    n = len(f)
    L = 2 * n - 1
    # A[k, m] = f[k-m] if 0 <= k-m <= n-1, else 0
    # Using COO format
    rows_list = []
    cols_list = []
    vals_list = []
    for m in range(n):
        ks = range(m, min(m + n, L))
        ds = range(0, min(n, L - m))
        rows_list.extend(ks)
        cols_list.extend([m] * len(ds))
        vals_list.extend(f[:len(ds)])
    A = scipy.sparse.coo_matrix((vals_list, (rows_list, cols_list)), shape=(L, n))
    return A

# Helper functions (top-level, no closures/lambdas)

def fc(a, b):
    """Fast FFT-based convolution."""
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def sc(f):
    """Score function."""
    n = f.size
    b = fc(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def norm(f, tgt):
    """Normalize to target sum, non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.size) / f.size * tgt
    return f / s * tgt

def bmat(f):
    """Build sparse convolution matrix A: (A@g)[k] = (f*g)[k]."""
    n = f.size
    L = 2 * n - 1
    diags = [np.full(n, f[d]) for d in range(n)]
    off = -np.arange(n, dtype=np.int32)
    return scipy.sparse.dia_matrix((diags, off), shape=(L, n)).tocsr()
