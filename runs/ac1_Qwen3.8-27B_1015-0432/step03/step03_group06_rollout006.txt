def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def optimize(f, deadline, rng, sym=True):
    n = f.shape[0]
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.05
    no_improv = 0
    restarts = 0
    it = 0
    t_curr = 0.08
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-15:
            break
        t_curr = max(0.002, 0.08 * (0.996 ** it))
        rel = b / M
        w = np.exp((rel - 1.0) / t_curr)
        ws = w.sum()
        if ws < 1e-10:
            w = np.zeros(2 * n - 1)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws
        c = fft_conv(f, w[::-1])
        g = 2.0 * c[n - 1:2 * n - 1][::-1]
        g = np.maximum(g, 0.0) + 1e-10
        gmax = g.max()
        if gmax < 1e-20:
            break
        gn = g / gmax
        best_eta_t = eta
        best_sc_t = np.inf
        best_tr = f.copy()
        for te in [eta * 0.5, eta, eta * 1.5, eta * 2.5]:
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-15)
            if sym:
                tr = 0.5 * (tr + tr[::-1])
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc_t:
                best_sc_t = sc
                best_eta_t = te
                best_tr = tr
        f = best_tr
        if best_eta_t > eta * 1.05:
            eta = min(0.25, eta * 1.1)
        elif best_eta_t < eta * 0.95:
            eta = max(0.005, eta * 0.87)
        if it % 10 == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 300:
                scl = 0.02 + 0.05 * min(1.0, restarts / 12.0)
                f = best_f * np.exp(scl * rng.standard_normal(n))
                f = np.maximum(f, 1e-15)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = normalize(f)
                no_improv = 0
                restarts += 1
                eta = 0.05
                t_curr = 0.08
    return best_f, best_sc
