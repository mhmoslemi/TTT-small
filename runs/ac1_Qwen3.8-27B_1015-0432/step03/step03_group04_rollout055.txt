def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    n = len(f)
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * n * b.max() / (s * s))

def compute_subgrad(f, b, mode='peak', max_pk=3000, tau=None):
    n = f.shape[0]
    L = 2 * n - 1
    if mode == 'softmax':
        bt = b / tau
        bt -= bt.max()
        w = np.exp(bt)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(L)
            w[np.argmax(b)] = 1.0
        else:
            w /= ws
    else:
        M = b.max()
        thresh = M * 0.995
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > max_pk:
            idx = np.argsort(b[peaks])[-max_pk:]
            peaks = peaks[idx]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        w = np.zeros(L)
        wp = b[peaks]
        w[peaks] = wp / wp.sum()
    # g[j] = 2 * conv(w, f[::-1])[n-1+j]
    c = fft_conv(w, f[::-1])
    g = 2.0 * c[n-1:2*n-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False, max_iter=500000, use_softmax=False):

    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.08
    no_improv = 0
    restarts = 0
    tau = 0.01  # softmax temperature
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        
        if use_softmax:
            # Adaptive temperature
            tau_eff = max(0.001, tau * (M / 1.0))
            g = compute_subgrad(f, b, mode='softmax', tau=tau_eff)
        else:
            g = compute_subgrad(f, b, mode='peak', max_pk=3000)
        
        gmax = g.max()
        if gmax < 1e-28:
            break
        
        # Multiplicative update
        f_new = f * np.exp(-eta * g / gmax)
        f_new = np.maximum(f_new, 1e-20)
        if sym:
            f_new = 0.5 * (f_new + f_new[::-1])
        f_new /= f_new.sum()
        
        # Simple line search (up to 3 halvings)
        sc_new = get_score(f_new)
        if sc_new > best_sc:
            # Halve step
            f_new = f * np.exp(-eta * 0.5 * g / gmax)
            f_new = np.maximum(f_new, 1e-20)
            if sym:
                f_new = 0.5 * (f_new + f_new[::-1])
            f_new /= f_new.sum()
            sc_new = get_score(f_new)
            if sc_new > best_sc:
                f_new = f  # Revert
                sc_new = best_sc
        
        f = f_new
        eta *= 0.998
        
        if it % 50 == 0:
            sc = get_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:

            if no_improv > 500:

                no_improv = 0
                restarts += 1
                eta = min(0.25, eta * 1.3)
    return best_f, best_sc

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    n = f.shape[0]
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * n * b.max() / (s * s))

def subgrad(f, b, max_pk=5000, tau=None):
    n = f.shape[0]
    L = 2 * n - 1
    if tau is not None and tau > 0:
        bt = (b - b.max()) / tau
        w = np.exp(bt)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(L)
            w[np.argmax(b)] = 1.0
        else:
            w /= ws
    else:
        M = b.max()
        peaks = np.where(b >= 0.995 * M)[0]
        if len(peaks) > max_pk:
            idx = np.argsort(b[peaks])[-max_pk:]
            peaks = peaks[idx]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        w = np.zeros(L)
        wp = b[peaks]
        w[peaks] = wp / wp.sum()
    c = fft_conv(w, f[::-1])
    g = 2.0 * c[n-1:2*n-1]
    return g

def optimize(f, deadline, rng, sym=False, max_iter=1000000, mode='peak'):
    n = f.shape[0]
    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.06
    no_improv = 0
    restarts = 0
    check_interval = 30
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        
        if mode == 'softmax':
            tau = max(0.005 * M, M * 0.001)
            g = subgrad(f, b, tau=tau)
        else:
            g = subgrad(f, b, max_pk=5000)
        
        gmax = g.max()
        if gmax < 1e-28:
            break
        
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        
        if (it + 1) % check_interval == 0:
            sc = get_score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 400:
                scale = 0.03 + 0.08 * min(1.0, restarts / 12.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                restarts += 1
                eta = min(0.20, eta * 1.25)
        eta *= 0.9985
    
    return best_f, best_sc

Actually, I think the key issue might be that the subgradient of the max function is not very informative. The max is a single point (or a few points), and the subgradient only tells us to decrease f at the locations contributing to those points. A better approach might be to use a smoothed max (like log-sum-exp) which gives a true gradient that includes contributions from all points.

The log-sum-exp approach: `LSE(b, tau) = tau * log(sum(exp(b/tau)))`. The gradient is `softmax(b/tau)`. As tau -> 0, this approaches the max. For finite tau, it's a smooth upper bound on the max.

But we want to MINIMIZE the max, not the LSE. Minimizing the LSE is a proxy for minimizing the max. The LSE is convex in b (hence in f), so gradient descent should work.

The gradient of LSE w.r.t. f_j is: `sum_k softmax_k * db_k/df_j = sum_k softmax_k * 2*f[k-j]` (for valid indices).

This is exactly the softmax subgradient I described earlier. So using softmax weighting with a small tau is equivalent to gradient descent on the LSE approximation.

The question is: what tau to use? Too small: the gradient is concentrated on the max point (like the true subgradient). Too large: the gradient is spread out and less focused. An adaptive tau would be ideal.

One heuristic: set tau so that the "effective number of active points" (1/sum(softmax^2)) is around 100-1000. This can be done iteratively.
