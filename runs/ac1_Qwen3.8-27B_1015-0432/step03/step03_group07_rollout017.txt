            for amp in amps:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = compute_score(trial)
                if sc < best_sc:
                    best_sc = sc
                    best_f = trial.copy()
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = np.ones(n) / n
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
