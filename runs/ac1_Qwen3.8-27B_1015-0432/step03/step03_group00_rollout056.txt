        s = _eval(ini)
        if s < best_sc:
            best_sc = s
            best = ini.copy()
            cur = ini.copy()

    # Phase 2: L-BFGS-B with geometrically decreasing tau
    bl = [(0.0, 100.0)] * n
    taus = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
            0.001, 0.0005, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5]
    stall = 0
    for tv in taus:
        if time.monotonic() > dl - 80:
            break
        remaining = dl - time.monotonic()
        iters = max(150, min(2000, int(remaining / 18)))
        prev_sc = best_sc
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': iters, 'gtol': 1e-14,
                                    'maxls': 40, 'ftol': 1e-17,
                                    'maxfun': iters * 5})
            nx = np.clip(res.x, 0.0, 100.0)
            if nx.sum() > 1e-8:
                nx = nx / nx.sum() * sc
            s2 = _eval(nx)
            if s2 < best_sc:
                best_sc = s2
                best = nx.copy()
                stall = 0
            else:
                stall += 1
            cur = nx.copy()
        except Exception:
            pass
        if stall >= 3:
            cur = best.copy() + 0.01 * best.max() * rng.standard_normal(n)
            cur = np.clip(cur, 0, None)
            cur = cur / cur.sum() * sc
            stall = 0

    # Phase 3: second L-BFGS-B pass from best with tiny tau
    if time.monotonic() < dl - 50:
        cur = best.copy()
        bl2 = [(0.0, 100.0)] * n
        for tv in [1e-5, 5e-6, 2e-6]:
            if time.monotonic() > dl - 20:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl2,
                               options={'maxiter': 800, 'gtol': 1e-15,
                                        'maxls': 30, 'ftol': 1e-19})
                nx = np.clip(res.x, 0.0, 100.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * sc
                s2 = _eval(nx)
                if s2 < best_sc:
                    best_sc = s2
                    best = nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    # Phase 4: subgradient steps on the true max
    if time.monotonic() < dl - 20:
        cur = best.copy()
        step = 0.03 * cur.max()
        for it in range(400):
            if time.monotonic() > dl - 10:
                break
            S = cur.sum()
            c = _conv(cur)
            idx = int(np.argmax(c))
            g2 = np.zeros(n)
            lo = max(0, idx - (n - 1))
            hi = min(n - 1, idx)
            ks = np.arange(lo, hi + 1)
            vals = cur[idx - ks]
            g2[ks] = vals
            mask = (idx != 2 * ks)
            g2[ks[mask]] += vals[mask]
            grad = (2.0 * n / (S * S)) * g2 - (4.0 * n * c[idx] / (S ** 3))
            improved = False
            for shrink in [1.0, 0.5, 0.25, 0.1, 0.05]:
                trial = cur - step * shrink * grad
                trial = np.clip(trial, 0, None)
                if trial.sum() < 1e-8:
                    continue
                trial = trial / trial.sum() * sc
                s2 = _eval(trial)
                if s2 < best_sc:
                    best_sc = s2
                    best = trial.copy()
                    cur = trial.copy()
                    improved = True
                    break
                elif s2 < best_sc + 0.002:
                    cur = trial.copy()
                    break
            if not improved:
                step *= 0.97
            if it % 40 == 0:
                cur = best.copy()

    # Phase 5: top-lag targeted adjustment
    if time.monotonic() < dl - 10:
        cur = best.copy()
        c = _conv(cur)
        top_k = 3
        idxs = np.argsort(c)[-top_k:]
        for jp in idxs:
            lo = max(0, jp - (n - 1))
            hi = min(n - 1, jp)
            ks = np.arange(lo, hi + 1)
            prods = cur[ks] * cur[jp - ks]
            tc = ks[np.argsort(prods)[-6:]]
            for frac in [0.85, 0.7, 0.5, 0.4]:
                trial = cur.copy()
                trial[tc] *= frac
                trial = np.clip(trial, 0, None)
                if trial.sum() > 0.01:
                    trial = trial / trial.sum() * sc
                    s2 = _eval(trial)
                    if s2 < best_sc:
                        best_sc = s2
                        best = trial.copy()
                        cur = trial.copy()
                        break

    # Phase 6: truncation sweep
    for tf in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
        if time.monotonic() > dl - 5:
            break
        trial = best.copy()
        trial[trial < tf * best.max()] = 0.0
        if trial.sum() > 0.01:
            s = _eval(trial)
            if s < best_sc:
                best_sc = s
                best = trial.copy()

    # Phase 7: stochastic local search
    if time.monotonic() < dl - 3:
        cur = best.copy()
        while time.monotonic() < dl - 2:
            cd = cur.copy()
            nw = int(rng.integers(3, max(4, n // 15)))
            s = int(rng.integers(0, max(1, n - nw)))
            cd[s:s + nw] *= np.exp(rng.normal(0.0, 0.04, nw))
            if rng.random() < 0.1:
                cd *= np.exp(rng.normal(0.0, 0.005))
            cd = np.clip(cd, 0, None)
            if cd.sum() > 0.01:
                cd = cd / cd.sum() * sc
                s2 = _eval(cd)
                if s2 < best_sc:
                    best_sc = s2
                    best = cd.copy()
                    cur = cd.copy()
            if rng.random() < 0.06:
                cur = best.copy()

    if not np.isfinite(best_sc):
        best = a0.copy()
    best = np.clip(best, 0.0, 1000.0)
    out = [float(v) for v in best]
    if not out or not all(np.isfinite(v) for v in out):
        out = [1.0] * 10
    return out
