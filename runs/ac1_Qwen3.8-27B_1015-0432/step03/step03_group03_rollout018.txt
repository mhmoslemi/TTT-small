        wave = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        inits.append(norm_pos(np.maximum(wave, 0.01)))
    
    # U-shape
    u = 0.3 + 0.7 * (1.0 - np.sin(np.pi * x / (n - 1)))
    inits.append(norm_pos(u))
    
    # Two lobe
    two_lobe = np.ones(n)
    half = n // 2
    t1 = np.arange(half) / max(1, half - 1)
    t2 = np.arange(n - half) / max(1, n - half)
    two_lobe[:half] = 1.5 - np.cos(np.pi * t1)
    two_lobe[half:] = 1.5 - np.cos(np.pi * t2)
    inits.append(norm_pos(np.maximum(two_lobe, 0.01)))
    
    # Random
    r = np.ones(n) + 0.5 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(norm_pos(r))
    
    # Multi-wave
    mw = np.ones(n)
    for freq in [2.0, 3.0]:
        mw = mw * (0.7 + 0.3 * np.cos(2.0 * np.pi * freq * x / n))
    inits.append(norm_pos(np.maximum(mw, 0.01)))
    
    return inits
