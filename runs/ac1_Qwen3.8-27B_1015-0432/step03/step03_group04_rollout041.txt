def compute_grad_dense(f, b, T):
    """Compute gradient using all lags with temperature-based weighting."""
    n = f.shape[0]
    L = 2 * n - 1
    if T < 1e-10:
        # Peak-based
        k = np.argmax(b)
        h = np.zeros(L)
