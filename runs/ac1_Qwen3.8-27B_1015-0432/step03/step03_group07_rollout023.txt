    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def ac1_score(f):
    S = f.sum()
    if S < 1e-10:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max() / (S * S)

def grad_from_weights(f, w):
    n = f.size
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.12, sym=False):
    n = f.size
    f = np.maximum(f.copy(), 1e-25)
    f /= f.sum()
    best_f = f.copy()
    best_sc = ac1_score(f)
    eta = eta0
    no_improv = 0
    L = 2 * n - 1
    it = 0
    restart_count = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 6000.0)
        frac = 0.94 + 0.055 * prog
        spread = max(1.0 - frac, 1e-9)
        w = np.maximum(b - M * frac, 0.0)
        w = (w / (M * spread)) ** 2
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(L)
            w[np.argmax(b)] = 1.0
            wsum = 1.0
        w /= wsum
        g = grad_from_weights(f, w)
        gmax = g.max()
        if gmax < 1e-30:
            break
        ratio = np.minimum(eta * g / gmax, 3.0)
        f = f * np.exp(-ratio)
        f = np.maximum(f, 1e-25)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        it += 1
        if it % 15 == 0:
            sc = ac1_score(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 220:
                restart_count += 1
                amp = 0.02 + 0.012 * min(1.0, restart_count / 10.0)
                r = restart_count % 4
                if r == 0:
                    pert = rng.standard_normal(n)
                elif r == 1:
                    Ff = np.fft.rfft(best_f)
                    Ff[1:-1] *= 1.0 + 0.5 * rng.standard_normal(len(Ff) - 2)
                    pert = np.fft.irfft(Ff, n) - best_f
                elif r == 2:
                    BL = max(1, n // 60)
                    pert = np.zeros(n)
                    for bi in range(0, n - BL, BL):
                        pert[bi:bi + BL] = 0.3 * rng.standard_normal(BL)
                else:
                    x = np.arange(n, dtype=np.float64)
                    pert = rng.standard_normal(n) * (0.4 + 1.2 * x / n)
                f = best_f * np.exp(amp * pert / (np.abs(pert).max() + 1e-10))
                f = np.maximum(f, 1e-25)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                eta = min(0.5, 0.10 * (1.15 ** restart_count))
        eta *= 0.9992
    return best_f, best_sc

def gen_inits(n, p, rng):
    inits = []
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    cw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits.append(cw / cw.sum())
    tri = np.maximum(1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / (n - 1), 0.0)
    inits.append(tri / tri.sum())
    ramp_up = x + 1.0
    inits.append(ramp_up / ramp_up.sum())
    ramp_dn = n - x
    inits.append(ramp_dn / ramp_dn.sum())
    for alpha in [0.5, 1.0, 2.0]:
        pw = (x + 1.0) ** (-alpha)
        inits.append(pw / pw.sum())
    for frac, bw in [(0.3, 0.1), (0.5, 0.15), (0.7, 0.1), (0.25, 0.2), (0.75, 0.2)]:
        ctr = frac * (n - 1)
        sig = bw * n
        g = np.exp(-((x - ctr) ** 2) / (2.0 * sig ** 2))
        inits.append(g / g.sum())
    for _ in range(3):
        r = np.maximum(1.0 + 0.4 * rng.standard_normal(n), 0.01)
        Fr = np.fft.rfft(r)
        Fr[int(len(Fr) * 0.3):] = 0
        r = np.fft.irfft(Fr, n)
        r = np.maximum(r, 0.01)
        inits.append(r / r.sum())
    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 30.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = gen_inits(n, p, rng)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = ac1_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    dl1 = min(deadline, t0 + 380.0)
    f1, sc1 = optimize(best_f.copy(), dl1, rng)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    if time.monotonic() < deadline - 320:
        scores = [ac1_score(init) for init in inits]
        order = np.argsort(scores)
        tried = set()
        for idx in order:
            if len(tried) >= 3:
                break
            if np.allclose(inits[idx], best_f, atol=1e-6):
                continue
            tried.add(idx)
            dl_i = min(deadline - 180, time.monotonic() + 100.0)
            fi, sci = optimize(inits[idx].copy(), dl_i, rng)
            if sci < best_sc:
                best_sc, best_f = sci, fi.copy()
    if time.monotonic() < deadline - 80:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = np.maximum(fs, 1e-25)
        fs /= fs.sum()
        sc_s = ac1_score(fs)
        if sc_s < best_sc:
            best_f, best_sc = fs.copy(), sc_s
        dl2 = min(deadline - 60, time.monotonic() + 200.0)
        f2, sc2 = optimize(fs.copy(), dl2, rng, sym=True, eta0=0.06)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2.copy()
    if time.monotonic() < deadline - 50:
        dl3 = deadline - 10.0
        f3, sc3 = optimize(best_f.copy(), dl3, rng, eta0=0.03)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()
    if time.monotonic() < deadline - 5:
        for i in range(2000):
            if time.monotonic() > deadline - 3:
                break
            for amp in [0.001, 0.003, 0.007, 0.015]:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-25)
                trial /= trial.sum()
                sc = ac1_score(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
