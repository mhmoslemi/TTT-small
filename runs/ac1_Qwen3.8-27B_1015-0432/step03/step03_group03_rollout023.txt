import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def soft_grad(f, beta):
    """Compute gradient of smooth max approximation of conv(f,f)."""
    n = f.shape[0]
    b = fft_conv(f, f)
    M = b.max()
    # Softmax weights (numerically stable)
    eb = np.exp(beta * (b - M))
    w = eb / eb.sum()
    L = 2 * n - 1
    h = np.zeros(L)
    h[:] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0), M, w

def hard_grad(f, peaks, weights):
    """Compute subgradient at specific peaks."""
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    best_f = f.copy()

    eta = eta0
    no_improv = 0
    momentum = np.zeros(n)

            break
        
        if use_soft:

        else:
            b = fft_conv(f, f)

            if len(peaks) > 200:

                peaks = idx[:200]
            vals = b[peaks]
            ww = vals / vals.max()
            g = hard_grad(f, peaks, ww)

                no_improv = 0

import numpy as np
import time

# Helper functions (top-level, no closures)

def fft_conv(a, b):
    """FFT-based convolution."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    """Project to non-negative and normalize to sum 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def sym_proj(f):
    """Project to symmetric."""
    f = 0.5 * (f + f[::-1])
    return norm_pos(f)

def obj_val(f):
    """Compute the objective value."""
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, beta=None):
    """Compute gradient/subgradient of max(conv(f,f)).
    If beta is None, use hard peaks. Otherwise use softmax."""
    n = f.shape[0]
    b = fft_conv(f, f)
    M = b.max()
    if beta is not None:
        # Soft max weights
        eb = np.exp(beta * (b - M))
        w = eb / eb.sum()
    else:
        # Hard peaks: top fraction
        thresh = M * 0.95
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 300:
            idx = np.argsort(b)[::-1]
            peaks = idx[:300]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        w = np.zeros(2 * n - 1)
        vals = b[peaks]
        w[peaks] = vals / vals.max()
    L = 2 * n - 1
    h = np.zeros(L)
    h[:] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0), b, M

def optimize(f, deadline, rng, eta0, beta, sym, max_iter):
    """Main optimization loop with multiplicative weights."""
    n = f.shape[0]
    f = norm_pos(f.copy())
    if sym:

    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    momentum = np.zeros(n)

        gmax = g.max()
        if gmax < 1e-25:

        mmax = momentum.max()
        if mmax < 1e-25:

        f_new = np.maximum(f_new, 1e-20)
        if sym:

        f = norm_pos(f_new)

            if sc < best_sc - 1e-13:
