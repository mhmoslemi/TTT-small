import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def cross_corr(f, w):
    """e[i] = sum_k f[k] * w[i+k], length n."""
    n = len(f)
    L = len(w)
    size = 1 << (n + L - 1).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    return np.fft.irfft(np.conj(F) * W, size)[:n]

    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    w = np.zeros(L)
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 6000.0)
        frac = 0.996 - 0.048 * prog
        thresh = M * frac
        # Soft weights with linear ramp

        w[:] = np.clip((b - thresh) / rw, 0.0, 1.0)
        n_act = (w > 0).sum()
        if n_act == 0:

            w[:] = 0.0
            w[idx] = 1.0
        elif n_act > max_peaks:
            top = np.argpartition(b, -max_peaks)[-max_peaks:]
            wm = np.zeros(L)

            w[:] = wm

        f *= np.exp(-eta * e / emax)

                f /= f.sum()
                patience = 0
                eta = min(0.4, eta * 1.3)
        if it > 200:
            eta *= 0.9996
    return best_f, best_sc

import numpy as np

def fft_conv(a, b):
    na, nb = len(a), len(b)
    sz = 1 << (na + nb - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:na + nb - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def cross_corr(f, w):
    n = len(f)
    L = len(w)
    sz = 1 << (n + L - 1).bit_length()
    F = np.fft.rfft(f, sz)
    W = np.fft.rfft(w, sz)
    return np.fft.irfft(np.conj(F) * W, sz)[:n]

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.10, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    w = np.zeros(L)
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 6000.0)
        frac = 0.996 - 0.048 * prog
        thresh = M * frac
        ramp = M * 0.008 + 1e-30
        w[:] = np.clip((b - thresh) / ramp, 0.0, 1.0)
        if (w > 0).sum() == 0:
            w[:] = 0.0
            w[np.argmax(b)] = 1.0
        elif (w > 0).sum() > max_peaks:
            top = np.argpartition(b, -max_peaks)[-max_peaks:]
            tmp = np.zeros(L)
            tmp[top] = w[top]
            w[:] = tmp
            if w.sum() < 1e-30:
