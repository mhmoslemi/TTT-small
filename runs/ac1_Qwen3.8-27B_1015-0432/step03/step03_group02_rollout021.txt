        g = compute_grad_dir(f, b, frac, max_peaks)
        if g is None:
            break
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        
        if use_line_search:
            # Try a few step sizes
            cands = [eta, eta * 0.5, eta * 1.5, eta * 0.25]
            best_step_sc = np.inf
            best_step_f = None
            for ts in cands:
                trial = f * np.exp(-ts * g / gmax)
                np.maximum(trial, 1e-30, out=trial)
                trial /= trial.sum()
                sc = score(trial)
                if sc < best_step_sc:
                    best_step_sc = sc
                    best_step_f = trial.copy()
            f = best_step_f
            eta *= 0.997
        else:
            f *= np.exp(-eta * g / gmax)
            np.maximum(f, 1e-30, out=f)
            f /= f.sum()
            eta *= 0.9995
        
        it += 1
        if it % 20 == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > 500:
                # Restart with perturbation
                scert = 0.02 + 0.03 * min(1.0, patience / 1000.0)
                f = best_f * np.exp(scert * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.4, eta * 1.5)
    return best_f, best_sc

def symmetric_fine_tune(f, deadline, rng):
    """Fine-tune with symmetric perturbations and gradient steps."""
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    # Make symmetric
    fsym = 0.5 * (f + f[::-1])
    fsym /= fsym.sum()
    scs = score(fsym)
    if scs < best_sc:
        best_sc = scs
        best_f = fsym
    
    scale = 0.004
    for it in range(500):
        if time.monotonic() > deadline:
            break
        for _ in range(6):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.95
    return best_f, best_sc
