    for w1, w2 in [(1.0, 0.5), (1.5, 0.7), (2.0, 1.0)]:
        s1 = np.exp(-((x - c1) ** 2) / (2.0 * (0.12 * n) ** 2)) * w1
        s2 = np.exp(-((x - c2) ** 2) / (2.0 * (0.12 * n) ** 2)) * w2
        shapes = s1 + s2
        inits.append(shapes / shapes.sum())
    for _ in range(4):
        r = np.maximum(1.0 + 0.4 * rng.standard_normal(n), 0.01)
        inits.append(r / r.sum())
    # Find best init
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = nscore(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    # Phase 1: wide search
    dl1 = min(deadline, t0 + 300.0)
    f1, sc1 = optimize(best_f.copy(), dl1, rng, top_frac=0.95, eta0=0.15)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    # Phase 2: from a different init
    if time.monotonic() < deadline - 250:
        dl2 = min(deadline, t0 + 550.0)
        alt = 0.5 * (best_f + best_f[::-1])
        alt /= alt.sum()
        f2, sc2 = optimize(alt.copy(), dl2, rng, top_frac=0.97, eta0=0.10)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2.copy()
    # Phase 3: fine-tune
    if time.monotonic() < deadline - 80:
        dl3 = deadline - 15.0
        f3, sc3 = optimize(best_f.copy(), dl3, rng, top_frac=0.99, eta0=0.06)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()
    # Phase 4: local perturbations
    if time.monotonic() < deadline - 5:
        for _ in range(800):
            if time.monotonic() > deadline - 3:
                break
            for amp in [0.001, 0.003, 0.008, 0.015]:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = nscore(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
