    tri = np.maximum(n - 1 - np.abs(x - (n - 1) / 2.0), 0.5) ** 0.7
    inits.append(tri / tri.sum())
    gc = np.exp(-2.0 * ((x - (n - 1) / 2.0) / (n / 4.0)) ** 2)
    gc += 0.05
    inits.append(gc / gc.sum())
    g2 = np.exp(-3.0 * ((x - 0.25 * n) / (n / 8.0)) ** 2) + np.exp(-3.0 * ((x - 0.75 * n) / (n / 8.0)) ** 2)
    g2 += 0.03
    inits.append(g2 / g2.sum())
    cw = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    cw += 0.1
    inits.append(cw / cw.sum())
    r = np.maximum(rng.standard_normal(n) + 0.6, 0.01)
    inits.append(r / r.sum())
    best_f = None
    best_sc = np.inf
    for init in inits:
        init = np.maximum(init, 0.0)
        init /= init.sum()
        sc = ac1(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    dl1 = min(deadline, t0 + 320.0)
    f1, sc1 = optimize(best_f.copy(), dl1, rng)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    if time.monotonic() < deadline - 320:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        sc_s = ac1(fs)
        if sc_s < best_sc:
            best_sc, best_f = sc_s, fs.copy()
        dl2 = min(deadline, t0 + 680.0)
        f2, sc2 = optimize(fs.copy(), dl2, rng, sym=True)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2.copy()
    if time.monotonic() < deadline - 80:
        dl3 = deadline - 12.0
        f3, sc3 = optimize(best_f.copy(), dl3, rng)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()
    if time.monotonic() < deadline - 4:
        for _ in range(800):
            if time.monotonic() > deadline - 2:
                break
            for amp in [0.001, 0.004, 0.01, 0.025]:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = ac1(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
