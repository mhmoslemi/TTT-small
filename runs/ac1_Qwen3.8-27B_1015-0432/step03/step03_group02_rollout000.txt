def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.12, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    f = f / max(f.sum(), 1e-15)
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 8000.0)
        frac = 0.995 - 0.05 * prog
        thresh = M * frac
        active = b >= thresh
        cnt = active.sum()
        h = np.zeros(L)
        if cnt > max_peaks:
            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
            h[idx] = b[idx]
        elif cnt >= 1:
            h[active] = b[active]
        else:
            h[np.argmax(b)] = 1.0
        hs = h.sum()
        if hs < 1e-30:
            break
        h /= hs
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
                eta = min(0.30, eta * 1.03)
            else:
                patience += 1
                eta = max(0.03, eta * 0.98)
            if patience > 1200:
                amt = 0.025 + 0.03 * min(1.0, patience / 2500.0)
                f = best_f * np.exp(amt * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.40, eta * 1.5)
    return best_f, best_sc

def fine_tune(f, deadline, rng, steps=300):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    for i in range(steps):
        if time.monotonic() > deadline:
            break
        for _ in range(10):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.93
    return best_f, best_sc
