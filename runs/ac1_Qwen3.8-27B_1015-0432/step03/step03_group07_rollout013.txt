    g = 2.0 * c[n-1:2*n-1][::-1]
    return g, b, M

def optimize(f, deadline, rng, max_iter=200000):
    n = f.size
    f = np.maximum(f.copy(), 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.1
    no_improv = 0
    restarts = 0
    beta = 0.02
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        g, b, M = compute_grad(f, beta)
        gmax = g.max()
        if gmax < 1e-30:
            break
        # Multiplicative update
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        # Periodic exact evaluation
        if it % 20 == 0:
            sc = score(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            # Adapt parameters
            beta = max(0.005, 0.02 * 0.9995**it)
            eta *= 0.998
            # Restart if stuck
            if no_improv > 200:
                amp = 0.03 + 0.02 * min(1.0, restarts / 5.0)
                pert = rng.standard_normal(n)
                # Add some frequency-domain perturbation
                Ff = np.fft.rfft(best_f)
                frac = rng.standard_normal(len(Ff))
                Ff[1:] *= (1.0 + 0.3 * frac)
                Ff[0] = max(Ff[0], 0)
                fp = np.fft.irfft(Ff, n)
                fp = np.maximum(fp, 0)
                if fp.sum() < 1e-10:
                    fp = best_f
                else:
                    fp /= fp.sum()
                f = (1 - amp) * best_f + amp * fp
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improv = 0
                restarts += 1
                eta = min(0.3, 0.1 * (1.15 ** restarts))
    return best_f, best_sc
