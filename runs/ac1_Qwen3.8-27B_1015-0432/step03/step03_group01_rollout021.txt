    taus = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    for tv in taus:
        if time.monotonic() > dl - 80.0:
            break
        rem = dl - time.monotonic()
        iters = min(600, max(50, int(rem * 10)))
        if iters < 20:
            break
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bounds_list,
                           options={'maxiter': iters, 'gtol': 1e-15,
                                    'maxls': 30, 'ftol': 1e-20})
            best, best_sc, cur = _try_update(res.x, best, best_sc, scale)
        except Exception:
            pass

    # Symmetrize and re-optimize
    if time.monotonic() < dl - 60.0:
        sym = 0.5 * (cur + cur[::-1])
        sym = _normalize(np.clip(sym, 0.0, 1000.0), scale)
        sc = float(evaluate_sequence(sym.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = sym.copy()
            best_sc = sc
            cur = sym.copy()
        for tv in [0.01, 0.005, 0.002, 0.001]:
            if time.monotonic() > dl - 60.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 300, 'gtol': 1e-15,
                                        'maxls': 25, 'ftol': 1e-20})
                best, best_sc, cur = _try_update(res.x, best, best_sc, scale)
            except Exception:
                pass

    # Targeted local search: perturb segments near peak lags
    if time.monotonic() < dl - 20.0:
        wmax = max(2, n // 30)
        for _ in range(3000):
            if time.monotonic() > dl - 20.0:
                break
            cd = best.copy()
            c = _conv(cd)
            kp = int(np.argmax(c))
            wp1 = int(rng.integers(1, wmax + 1))
            center = int(np.clip(kp // 2, 0, n - wp1))
            s1 = int(np.clip(center + int(rng.normal(0, wp1)), 0, n - wp1))
            cd[s1:s1 + wp1] *= float(np.exp(rng.normal(0.0, 0.04)))
            if rng.random() < 0.4:
                wp2 = int(rng.integers(1, wmax + 1))
                s2 = int(rng.integers(0, n - wp2))
                cd[s2:s2 + wp2] *= float(np.exp(rng.normal(0.0, 0.04)))
            if rng.random() < 0.1:
                cd *= float(np.exp(rng.normal(0.0, 0.008)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc

    # Final ultra-low-temperature polish
    if time.monotonic() < dl - 5.0:
        cur = best.copy()
        for tv in [0.0003, 0.0001, 5e-05, 2e-05]:
            if time.monotonic() > dl - 5.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 400, 'gtol': 1e-16,
                                        'maxls': 20, 'ftol': 1e-22})
                best, best_sc, cur = _try_update(res.x, best, best_sc, scale)
            except Exception:
                pass

    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = a0
    return [float(v) for v in best]
