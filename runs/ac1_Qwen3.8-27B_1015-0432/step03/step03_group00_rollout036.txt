import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smmax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    mv = cm + tau * np.log(e.sum())
    return mv, e / e.sum()

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return 1e10
    mv, _ = _smmax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-10:
        return np.zeros(n)
    c = _conv(a)
    mv, p = _smmax(c, tau)
    L = _fpow2(3 * n - 2)
    r = ifft(np.conj(fft(a, L)) * fft(p, L)).real[:n]
    return (4.0 * n / (S ** 3)) * (S * r - mv)
