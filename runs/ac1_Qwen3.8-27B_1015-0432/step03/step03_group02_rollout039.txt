import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def run_opt(f, deadline, rng, max_peaks=300, eta_init=0.12, check_iv=25, sym=True):
    n = f.shape[0]

        return f, np.inf
    f = f / s
    if sym:
        f = 0.5 * (f + f[::-1]); f /= f.sum()
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0

        b = fft_conv(f, f)
        M = b.max()

        frac = 0.9985 - 0.050 * prog
        thresh = M * frac

            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
            h.fill(0.0)
            h[idx] = b[idx] - thresh
        elif cnt >= 1:
            h.fill(0.0)
            h[active] = b[active] - thresh

        hw = h.max()
        if hw < 1e-30:
            h.fill(0.0)
            h[np.argmax(b)] = 1.0
