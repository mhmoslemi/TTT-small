import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * (b.max() / (s * s))

def subgrad_from_peaks(f, peaks, weights):
    """Compute 2*sum_k w_k * shift_{p_k}(f) projected to [0,n)."""
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = c[n-1:2*n-1][::-1]
    return 2.0 * np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.1, max_iter=500000):
    n = f.shape[0]
    f = np.maximum(f, 0)
    s = f.sum()
    f = f / s  # normalize to sum=1
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    momentum = np.zeros(n)
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        b = fft_conv(f, f)
        M = b.max()
        
        # Select peaks: use all above 99.9% of max, cap at n peaks
        thresh = M * 0.995
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > n:
            # Use top n peaks
            top = np.argsort(b)[::-1][:n]
            peaks = top
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        
        # Weight by how close to max
        vals = b[peaks]
        w = (vals - M + 1) / (vals.max() - M + 1)  # or just uniform
        # Actually, uniform weights are the standard subgradient
        w = np.ones(len(peaks)) / len(peaks)
        
        g = subgrad_from_peaks(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-30:
            break
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * g
        mmax = momentum.max()
        
        # Multiplicative update
        f = f * np.exp(-eta * momentum / mmax)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()
        
        # Learning rate decay (slower!)
        eta *= 0.9999  # much slower decay
        
        # Check improvement every 20 iterations

                best_f = f.copy()
                no_improv = 0

                momentum = np.zeros(n)
                no_improv = 0
                eta = max(eta, 0.05)
    
    return best_f, best_sc

        t3 = min(deadline - 180.0, t0 + 900.0)
        f3, sc3 = prop_sg(best_f.copy(), t3, rng, sym=False)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3

    if time.monotonic() < deadline - 80.0:
        t4 = min(deadline - 15.0, t0 + 1050.0)
        f4, sc4 = run_sg(best_f.copy(), t4, rng, sym=False, eta0=0.02, T_ramp=50000)
        if sc4 < best_sc:
            best_sc, best_f = sc4, f4

    for _ in range(300):
        if time.monotonic() > deadline - 5.0:
            break
        trial = best_f * np.exp(0.004 * rng.standard_normal(n))
        trial = trial / trial.sum()
        sc = obj_val(trial)
        if sc < best_sc:
            best_sc = sc
            best_f = trial.copy()

    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or fs > 2.0:
        result = [1.0 / n] * n
    return result
