    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def peak_grad(f, peaks, w):
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    return np.maximum(2.0 * c[n - 1:2 * n - 1][::-1], 0.0)

def optimize(f, deadline, rng, eta0, n_peaks, sym):
    n = f.shape[0]
    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    restart = 0
    mom = np.zeros(n)

    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        frac = max(0.88, 0.99 - 0.04 * restart)
        peaks = np.where(b >= M * frac)[0]
        if len(peaks) > n_peaks:
            peaks = np.argsort(b)[::-1][:n_peaks]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])

        w = (b[peaks] / M) ** 3
        g = peak_grad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:
            break

        mom = 0.82 * mom + 0.18 * g
        mmax = mom.max()
        if mmax < 1e-25:
            break

        best_trial_sc = best_sc
        best_trial_f = f
        for te in (eta, eta * 0.4, eta * 2.5):
            cand = f * np.exp(-te * mom / mmax)
            cand = np.maximum(cand, 1e-18)
            if sym:
                cand = 0.5 * (cand + cand[::-1])
            cand = norm_pos(cand)
            sc = obj_val(cand)
            if sc < best_trial_sc:
                best_trial_sc = sc
                best_trial_f = cand

        f = best_trial_f
        if best_trial_sc < best_sc - 1e-13:
            best_sc = best_trial_sc
            best_f = f.copy()
            no_improv = 0
        else:
            no_improv += 1

        if no_improv > 400:
            sc2 = 0.02 + 0.04 * min(1.0, restart / 6.0)
            f = best_f * np.exp(sc2 * rng.standard_normal(n))
            f = np.maximum(f, 1e-12)
            if sym:
                f = 0.5 * (f + f[::-1])
            f = norm_pos(f)
            mom = np.zeros(n)
            no_improv = 0
            restart += 1
            eta = min(0.35, eta * 1.35)

        eta *= 0.9997

    return best_f, best_sc

def gen_inits(n, rng):
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(norm_pos(p.copy()))
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.4, 0.7, 1.2, 2.0]:
        inits.append(norm_pos(d ** pw))

    inits.append(norm_pos(0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))))
    for fq in [2.0, 3.0, 5.0]:
        inits.append(norm_pos(0.5 + 0.5 * np.cos(2.0 * np.pi * fq * x / n) + 0.1))

    r = np.ones(n) + 0.25 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(norm_pos(r))

    wt = 0.15
    m = max(1, int(n * wt))
    ft = np.ones(n)
    t = np.arange(m) / m
    ft[:m] = 0.5 * (1.0 + np.cos(np.pi * t))
    ft[n - m:] = 0.5 * (1.0 + np.cos(np.pi * t))[::-1]
    inits.append(norm_pos(ft))

    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    inits = gen_inits(n, rng)
    best_f = None
    best_sc = np.inf
    for ini in inits:
        sc = obj_val(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    t1 = min(deadline, t0 + 280.0)
    f1, s1 = optimize(best_f.copy(), t1, rng, 0.08, 150, True)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()

    if time.monotonic() < deadline - 200.0:
        t2 = min(deadline - 200.0, t0 + 580.0)
        f2, s2 = optimize(best_f.copy(), t2, rng, 0.05, 250, True)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()

    if time.monotonic() < deadline - 100.0:
        t3 = min(deadline - 80.0, t0 + 880.0)
        f3, s3 = optimize(best_f.copy(), t3, rng, 0.025, 400, True)
        if s3 < best_sc:
            best_sc, best_f = s3, f3.copy()

    if time.monotonic() < deadline - 15.0:
        for _ in range(300):
            if time.monotonic() > deadline - 5.0:
                break
            tr = best_f * np.exp(0.003 * rng.standard_normal(n))
            tr = 0.5 * (tr + tr[::-1])
            tr = norm_pos(tr)
            sc = obj_val(tr)
            if sc < best_sc:
                best_sc = sc
                best_f = tr.copy()

    result = [float(min(1000.0, max(0.0, x))) for x in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
