from numpy.fft import fft, ifft

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax_val(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    return cm + tau * np.log(e.sum()), e / e.sum()

def _grad_from_p(a, p):
    n = a.size
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    return 2.0 * cv[n - 1:2 * n - 1][::-1]

    sm, p = _smax_val(c, tau)
    g = _grad_from_p(a, p)
