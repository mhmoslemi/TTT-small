```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def get_grad(f, b, tf, max_active):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.maximum(b - M * tf, 0.0)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= ws
    if max_active > 0 and (w > 0).sum() > max_active:
        idx = np.flatnonzero(w)
        part = np.argpartition(w[idx], -max_active)[-max_active:]
        w2 = np.zeros(L)
        w2[idx[part]] = w[idx[part]]
        w2 /= w2.sum()
        w = w2
    c = fft_conv(f, w[::-1])
    g = c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False, max_active=1000):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    f /= f.sum()
    best_f = f.copy()
    best_sc = nscore(f)
    eta = 0.14
    no_improv = 0
    rc = 0
    for it in range(1000000):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        prog = min(1.0, it / 8000.0)
        tf = 0.95 + 0.045 * prog
        g = get_grad(f, b, tf, max_active)
        gmax = g.max()
        if gmax < 1e-30:
            break
        ratio = np.minimum(eta * g / gmax, 3.0)
        f = f * np.exp(-ratio)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 10 == 0:
            sc = nscore(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 250:
                amp = 0.025 + 0.02 * min(1.0, rc / 10.0)
                pt = rc % 4
                if pt == 0:
                    pert = rng.standard_normal(n)
                elif pt == 1:
                    Ff = np.fft.rfft(best_f)
                    Ff[1:] *= 1.0 + 0.4 * rng.standard_normal(len(Ff) - 1)
                    pert = np.fft.irfft(Ff, n)
                    pert -= pert.mean()
                elif pt == 2:
                    pert = rng.standard_normal(n) * np.sin(np.pi * np.arange(n) / n)
                else:
                    blk = max(1, n // 16)
                    pert = np.zeros(n)
                    ci = rng.integers(0, n // blk)
                    pert[ci * blk:(ci + 1) * blk] = rng.standard_normal(blk)
                f = best_f * np.exp(amp * pert / (np.abs(pert).max() + 1e-10))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                rc += 1
                eta = min(0.5, 0.14 * (1.15 ** rc))
        eta *= 0.9993
    return best_f, best_sc

def polish(f, deadline, rng):
    n = f.shape[0]
    best = f.copy()
    best_sc = nscore(best)
    x = np.arange(n)
    while time.monotonic() < deadline:
        idx = rng.integers(0, n, size=20)
        improved = False
        for i in idx:
            if time.monotonic() > deadline:
                break
            b0 = best[i]
            for mult in [0.96, 1.04]:
                trial = best.copy()
                trial[i] = b0 * mult
                trial /= trial.sum()
                sc = nscore(trial)
                if sc < best_sc:
                    best_sc = sc
                    best = trial.copy()
                    b0 = best[i]
                    improved = True
        if not improved and rng.random() < 0.05:
            trial = best * np.exp(0.001 * rng.standard_normal(n))
