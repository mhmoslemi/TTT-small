    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = []
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))
    x = np.arange(n, dtype=np.float64)
    inits.append(normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))))
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.0, 1.5, 2.5]:
        inits.append(normalize(d ** pw))
    inits.append(normalize(np.ones(n)))
    r = np.ones(n) * (0.7 + 0.6 * rng.random(n))
    r = 0.5 * (r + r[::-1])
    inits.append(normalize(r))
    best_f, best_sc = None, np.inf
    for init in inits:
        sc = score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    f_s = normalize(0.5 * (best_f + best_f[::-1]))
    sc_s = score(f_s)
    if sc_s < best_sc:
        best_sc, best_f = sc_s, f_s
    ph1_end = min(deadline, t0 + budget_s * 0.30)
    fo, sco = mirror_optimize(f_s, ph1_end, rng, sym=True)
    if sco < best_sc:
        best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 80.0:
        ph2_end = min(deadline, t0 + budget_s * 0.55)
        fo, sco = mirror_optimize(best_f.copy(), ph2_end, rng, sym=False)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 60.0:
        fs = normalize(0.5 * (best_f + best_f[::-1]))
        ph3_end = min(deadline, t0 + budget_s * 0.75)
        fo, sco = mirror_optimize(fs, ph3_end, rng, sym=True)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 30.0:
        ph4_end = min(deadline, t0 + budget_s * 0.90)
        fo, sco = local_shave(best_f.copy(), ph4_end, rng)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 5.0:
        for _ in range(300):
            if time.monotonic() > deadline - 2.0:
                break
            tr = best_f * np.exp(0.002 * rng.standard_normal(n))
            tr = np.maximum(tr, 0.0)
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()
    result = [float(v) for v in best_f]
    ev = evaluate_sequence(result)
    if not np.isfinite(ev):
        result = [1.0 / n] * n
    return result
