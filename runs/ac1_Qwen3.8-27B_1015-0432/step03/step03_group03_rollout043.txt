import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks, weights):
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

    restarts = 0
    mom = np.zeros(n)

        if time.monotonic() > deadline:
            break

        M = b.max()

        peaks = np.where(b >= M * frac)[0]

            peaks = np.argsort(b)[::-1][:300]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        w = (b[peaks] / M) ** 5.0
        g = np.maximum(subgrad(f, peaks, w), 0.0)
        gm = g.max()
        if gm < 1e-25:
