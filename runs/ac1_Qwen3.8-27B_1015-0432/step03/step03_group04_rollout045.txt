    mu = 0.85
    
    while True:
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        g = compute_grad(f, b, beta_c, K, sym)
        gmax = g.max()
        if gmax < 1e-28:
            break
        g = g / gmax
        momentum = mu * momentum + (1 - mu) * g
        mmax = momentum.max()
        if mmax < 1e-28:
            break
        momentum = momentum / mmax
        f = f * np.exp(-eta * momentum)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        eta *= decay
        
        no_improv += 1
        if no_improv >= check_every:
            no_improv = 0
            sc = get_score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
            else:
                restarts += 1
                if restarts > 1:
                    scale = 0.02 + 0.06 * min(1.0, (restarts - 1) / 8.0)
                    f = best_f * np.exp(scale * rng.standard_normal(n))
                    f = np.maximum(f, 1e-20)
                    if sym:
                        f = 0.5 * (f + f[::-1])
                    f /= f.sum()
                    momentum *= 0.1
                    eta = min(0.20, eta0 * (1.0 + 0.15 * min(restarts, 10)))
                else:
                    # First "no improvement" might just be noise
                    pass
    
    return best_f, best_sc
