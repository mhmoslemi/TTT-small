import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_subgrad(f, peaks, weights):
    """Compute exact subgradient of max_j (f*f)[j] w.r.t. f."""
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[peaks] = weights
    # term2[i] = sum_j h[j]*f[i-j] = (h*f)[i]
    c2 = fft_conv(h, f)  # length 3n-2, we want indices 0..n-1
    t2 = c2[:n]
    # term1[i] = sum_j h[j]*f[j-i] = (h*f_r)[n-1+i] where f_r = f[::-1]
    c1 = fft_conv(h, f[::-1])  # length 3n-2
    t1 = c1[n-1:2*n-1]
    return np.maximum(t1 + t2, 0.0)

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_subgrad(f, peaks, weights):
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[peaks] = weights
    c = fft_conv(h, f[::-1])
    return np.maximum(2.0 * c[n - 1:2 * n - 1], 0.0)

def run_opt(f, deadline, rng, eta0, n_peaks, sym, max_iter, beta_w):

    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    restarts = 0
    momentum = np.zeros(n)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        frac = 0.995 - 0.018 * min(1.0, it / 4000.0)
        thresh = M * frac
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > n_peaks:
            idx = np.argsort(b)[::-1]
            peaks = idx[:n_peaks]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])

        vals = b[peaks]
        vrange = vals.max() - vals.min()
        if vrange < 1e-30:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = np.exp(-beta_w * (vals.max() - vals) / vrange)
            w = w / w.sum()

        g = compute_subgrad(f, peaks, w)

        if gmax < 1e-25:

        f = norm_pos(f)

        if it % 25 == 0:

            if sc < best_sc - 1e-10:

                best_f = f.copy()
                no_improv = 0

                no_improv += 1

                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-15)

                    f = 0.5 * (f + f[::-1])
                f = norm_pos(f)
                momentum = np.zeros(n)
                no_improv = 0
                restarts += 1
                eta = min(0.3, eta * 1.45)

        eta *= 0.9997

    return best_f, best_sc
