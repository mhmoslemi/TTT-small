import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def cross_corr(a, v):
    """Compute g_j = sum_k v_k * a_{k-j} for j=0..n-1"""
    n = len(a)
    L = len(v)  # should be 2n-1
    vrev = v[::-1]
    c = fft_conv(a, vrev)
    return c[n-1:2*n-1][::-1]

def compute_grad_full(a, b, tau):
    """Compute gradient using soft-max weights over all peaks"""
    n = len(a)
    # Soft weights
    b_max = b.max()
    b_min = b.min()
    # Shift to avoid overflow
    shifted = (b - b_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        # Fallback: hard max
        w = np.zeros_like(b)
        w[b == b_max] = 1.0 / (b == b_max).sum()
        w_sum = 1.0
    w /= w_sum

    cc = cross_corr(a, w)

    n = f.shape[0]
    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    if sym:
        f = 0.5 * (f + f[::-1])
        f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.08
    no_improv = 0
    restarts = 0
    b = fft_conv(f, f)
    M = b.max()
    recompute_every = 15
    for it in range(500000):
        if time.monotonic() > deadline:
            break
        if it % recompute_every == 0 or it < 3:
            b = fft_conv(f, f)
            M = b.max()
        if mode == 'smooth':
            tau_frac = 0.12 - 0.11 * min(1.0, it / 50000.0)
            g = grad_smooth(f, b, max(tau_frac * M, M * 0.005))
        else:
            tf = 0.999 - 0.045 * min(1.0, it / 30000.0)
            g = grad_active(f, b, max(tf, 0.94))
        gmax = g.max()
        if gmax < 1e-28:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 60 == 0:
            sc = get_score(f, b)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 1200:
                scale = 0.03 + 0.06 * min(1.0, restarts / 12.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                b = fft_conv(f, f)
                M = b.max()
                no_improv = 0
                restarts += 1
                eta = 0.08
        eta *= 0.9992
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = []
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.0, 2.0]:
        inits.append(d ** pw / (d ** pw).sum())
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.01
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())
    best_f = inits[0]
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    ph1_end = min(deadline, t0 + 400.0)
    f1, s1 = optimize(best_f.copy(), ph1_end, rng, sym=False, mode='smooth')
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()
    ph2_end = min(deadline, t0 + 800.0)
    if time.monotonic() < deadline - 150.0:
        f2, s2 = optimize(best_f.copy(), ph2_end, rng, sym=False, mode='active')
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()
    ph3_end = min(deadline, t0 + 1040.0)
    if time.monotonic() < deadline - 60.0:
        fsym = 0.5 * (best_f + best_f[::-1])
        fsym /= fsym.sum()
        f3, s3 = optimize(fsym, ph3_end, rng, sym=True, mode='smooth')
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()
    if time.monotonic() < deadline - 4.0:
        for _ in range(200):
            if time.monotonic() > deadline - 2.0:
                break
            trial = best_f.copy()
            trial *= np.exp(0.004 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()
    result = best_f.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    fs_check = evaluate_sequence(result_list)
    if not np.isfinite(fs_check):
        result_list = [1.0 / n] * n
    return result_list
