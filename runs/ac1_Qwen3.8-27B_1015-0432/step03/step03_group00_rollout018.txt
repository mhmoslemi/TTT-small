    cm = c.max()
    d = np.clip((c - cm) / tau, -50.0, 0.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    n = a.size
    c = _conv(a, L)
    mv, w = _smax(c, tau)
    W = fft(w, L)
    A = fft(a, L)
    h = ifft(W * np.conj(A)).real[:n]
    return mv, 2.0 * h, c

            break
        c = _conv(cur)
        top_idx = np.argsort(c)[-6:]
        improved = False
        for jp in top_idx:
            lo = max(0, jp - (n - 1))
            hi = min(n - 1, jp)
            ks = np.arange(lo, hi + 1)
            prods = cur[ks] * cur[jp - ks]
            order = np.argsort(prods)[::-1]
            for frac in [0.94, 0.88, 0.82]:
                for kcount in [3, 5, 8]:
                    tc = ks[order[:kcount]]
                    trial = cur.copy()
                    trial[tc] *= frac
                    trial = np.clip(trial, 0.0, None)
                    if trial.sum() > 0.01:
                        if _try_update(trial):
                            cur[:] = best
                            improved = True
                            break
                if improved:
                    break
        if not improved:
            break

    # Phase 5: Random perturbation polish
    while time.monotonic() < dl - 5.0:
        cd = cur.copy()
        nw = int(rng.integers(3, max(4, n // 40)))
        s = int(rng.integers(0, n - nw))
        cd[s:s+nw] *= np.exp(rng.normal(0.0, 0.04, nw))
        if rng.random() < 0.1:
            cd *= np.exp(rng.normal(0.0, 0.01))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            _try_update(cd)
            if best_sc < 1.505:
                break
        if rng.random() < 0.05:
            cur[:] = best

    # Final threshold cleanup
    thresh = 0.003 * best.max()
    clean = best.copy()
    clean[clean < thresh] = 0.0
    sc_clean = float(evaluate_sequence(clean.tolist()))
    if np.isfinite(sc_clean) and sc_clean < best_sc:
        best = clean
        best_sc = sc_clean

    if not np.isfinite(best_sc):
        best = cur.copy()
        best_sc = float(evaluate_sequence(best.tolist()))

    out = [float(v) for v in best]
    return out
