def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

import numpy as np
import time

def _fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def _score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = _fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def _make_inits(n, rng, parent):
    inits = []
    p = np.clip(np.asarray(parent, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    for pw in (0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        d = np.minimum(x, n - 1.0 - x) + 1.0
        inits.append(d ** pw / (d ** pw).sum())
    for k in (1, 2, 3):
        h = 0.5 * (1.0 - np.cos(2.0 * np.pi * k * (x + 0.5) / n))
        inits.append(h / h.sum())
    cx = (n - 1) / 2.0
    for sg in (0.15, 0.25, 0.4, 0.6):
        g = np.exp(-0.5 * ((x - cx) / (sg * n)) ** 2)
        inits.append(g / g.sum())
    for _ in range(4):
        r = 1.0 + 0.4 * rng.standard_normal(n)
        np.maximum(r, 0.01, out=r)
        inits.append(r / r.sum())
    return inits

def _optimize(f, end_time, rng, n):
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    bf, bs = f.copy(), _score(f)
    L = 2 * n - 1
    eta, gamma0, gamma1 = 0.12, 3.0, 25.0
    pat, it = 0, 0
    while time.monotonic() < end_time:
        b = _fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 4000.0)
        gam = gamma0 + (gamma1 - gamma0) * prog
        w = np.exp(-gam * (M - b) / (M + 1e-30))
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(L)
            w[np.argmax(b)] = 1.0
        else:
            w /= ws
        c = _fft_conv(f, w[::-1])
        g = c[n - 1 : 2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gm = g.max()
        if gm < 1e-30:
            break
        f *= np.exp(-eta * g / gm)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        if it % 25 == 0:
            sc = _score(f)
            if sc < bs - 1e-12:
                bs, bf, pat = sc, f.copy(), 0
            else:
                pat += 1
            if pat > 400:
                pv = 0.02 + 0.03 * min(1.0, pat / 800.0)
                f = bf * np.exp(pv * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                pat, eta = 0, min(0.35, eta * 1.3)
            if it > 300:
                eta *= 0.9995
    return bf, bs

def _polish(f, end_time, rng, n):
    bf, bs = f.copy(), _score(f)
    scale = 0.005
    for _ in range(200):
        if time.monotonic() > end_time:

        for _ in range(5):
            d = rng.standard_normal(n)
            tr = bf * np.exp(scale * d)
            np.maximum(tr, 1e-30, out=tr)
            tr /= tr.sum()
            sc = _score(tr)
            if sc < bs - 1e-13:
                bs, bf = sc, tr.copy()
        scale *= 0.9
    return bf, bs

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    n = len(height_sequence_1)
    inits = _make_inits(n, rng, height_sequence_1)
    bf, bs = inits[0].copy(), _score(inits[0])
    for v in inits[1:]:
        sc = _score(v)
        if sc < bs:
            bs, bf = sc, v.copy()
    pe1 = min(dl, t0 + 350.0)
    f1, s1 = _optimize(bf.copy(), pe1, rng, n)
    if s1 < bs:
        bs, bf = s1, f1
    if time.monotonic() < dl - 120.0:
        f2, s2 = _optimize(bf.copy(), min(dl, t0 + 750.0), rng, n)
        if s2 < bs:
            bs, bf = s2, f2
    if time.monotonic() < dl - 80.0:

            if time.monotonic() > dl - 30.0:

            pv = 0.01 + 0.015 * r
            fr = bf * np.exp(pv * rng.standard_normal(n))

            fr, sr = _optimize(fr, dl - 40.0, rng, n)
            if sr < bs:

    if time.monotonic() < dl - 5.0:

    result = [float(v) for v in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result

            d = rng.standard_normal(n)
            tr = bf * np.exp(sc * d)
            np.maximum(tr, 1e-30, out=tr)
            tr /= tr.sum()
            v = _score(tr)
            if v < bs - 1e-13:
                bs, bf = v, tr.copy()
        sc *= 0.92
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 30.0
    n = len(height_sequence_1)
    inits = _make_inits(n, rng, height_sequence_1)
    bf, bs = inits[0].copy(), _score(inits[0])
    for v in inits[1:]:
        sc = _score(v)
        if sc < bs:
            bs, bf = sc, v.copy()
    pe1 = min(dl, t0 + 350.0)
    f1, s1 = _optimize(bf.copy(), pe1, rng, n, 0.15, 1.0, 20.0)
    if s1 < bs:
        bs, bf = s1, f1
    if time.monotonic() < dl - 120.0:
        f2, s2 = _optimize(bf.copy(), min(dl, t0 + 750.0), rng, n, 0.06, 8.0, 60.0)
        if s2 < bs:
            bs, bf = s2, f2
    if time.monotonic() < dl - 80.0:
        for r in range(4):
            if time.monotonic() > dl - 30.0:
                break
            pv = 0.01 + 0.015 * r
            fr = bf * np.exp(pv * rng.standard_normal(n))
            np.maximum(fr, 1e-30, out=fr)
            fr /= fr.sum()
            fr, sr = _optimize(fr, dl - 40.0, rng, n, 0.10, 3.0, 40.0)
            if sr < bs:
                bs, bf = sr, fr
    if time.monotonic() < dl - 5.0:
        bf, bs = _polish(bf.copy(), dl - 3.0, rng, n)
    result = [float(v) for v in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
