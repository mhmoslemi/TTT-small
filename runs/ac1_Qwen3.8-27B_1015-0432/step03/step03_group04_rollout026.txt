            bsc = sc
            bf = ini.copy()

    # Phase 1: Asymmetric warm-start from parent
    f1, s1 = opt(bf.copy(), min(dl, t0 + 420.0), rng, sym=False)
    if s1 < bsc:
        bsc, bf = s1, f1.copy()

    # Phase 2: Symmetric refinement
    if time.monotonic() < dl - 250.0:
        fs = 0.5 * (bf + bf[::-1])
        fs /= fs.sum()
        f2, s2 = opt(fs, min(dl, t0 + 850.0), rng, sym=True)
        if s2 < bsc:
            bsc, bf = s2, f2.copy()

    # Phase 3: Free refinement
    if time.monotonic() < dl - 100.0:
        f3, s3 = opt(bf.copy(), dl - 30.0, rng, sym=False)
        if s3 < bsc:
            bsc, bf = s3, f3.copy()

    # Phase 4: Local perturbations
    if time.monotonic() < dl - 3.0:
        for _ in range(200):
            if time.monotonic() > dl - 1.0:
                break
            tr = bf.copy()
            tr *= np.exp(0.004 * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-15)
            tr /= tr.sum()
            sc = get_score(tr)
            if sc < bsc:
                bsc, bf = sc, tr.copy()
        # Try Laplacian-smoothed variants
        for la in [0.02, 0.05, 0.1]:
            if time.monotonic() > dl - 1.0:
                break
            tr = laplap_smooth(bf.copy(), la)
            tr = np.maximum(tr, 1e-15)
            tr /= tr.sum()
            sc = get_score(tr)
            if sc < bsc:
                bsc, bf = sc, tr.copy()

    result = bf.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    out = [float(v) for v in result]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs):
        out = [1.0 / n] * n
    return out
