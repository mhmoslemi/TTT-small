        fb = 1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        inits.append(fb / fb.sum())
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.01
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    best_f = inits[0]
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: Main asymmetric
    ph1 = min(deadline, t0 + 340.0)
    f1, s1 = optimize(best_f.copy(), ph1, rng)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()

    # Phase 2: Symmetric
    if time.monotonic() < deadline - 150.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ph2 = min(deadline, t0 + 760.0)
        f2, s2 = optimize(fs, ph2, rng, sym=True)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()

    # Phase 3: Free refinement
    if time.monotonic() < deadline - 50.0:
        f3, s3 = optimize(best_f.copy(), deadline - 15.0, rng)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()

    # Phase 4: Local perturbations
    if time.monotonic() < deadline - 4.0:
        for _ in range(80):
            if time.monotonic() > deadline - 2.0:
                break
            trial = best_f.copy()
            trial *= np.exp(0.003 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = np.clip(best_f.copy(), 0.0, 1000.0)
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    fs = evaluate_sequence(result_list)
    if not np.isfinite(fs):
        result_list = [1.0 / n] * n
    return result_list
