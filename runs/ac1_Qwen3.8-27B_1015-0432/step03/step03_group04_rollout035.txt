import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def subgrad(f, b, K, temp):
    n = f.size
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    pi = np.argpartition(b, -K)[-K:]
    w = np.exp((b[pi] - M) / (M * temp + 1e-30))
    w /= w.sum()
    g = np.zeros(n)
    for c in range(K):
        k = int(pi[c])
        wc = w[c]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            g[lo:hi+1] += wc * f[k-hi:k-lo+1][::-1]
        lo2 = max(0, k - hi)
        hi2 = min(n - 1, k - lo)
        if lo2 <= hi2:
            g[lo2:hi2+1] += wc * f[k-hi2:k-lo2+1][::-1]
    return g
