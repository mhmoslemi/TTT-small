        if sym:
            f_new = 0.5 * (f_new + f_new[::-1])
        f_new /= f_new.sum()
        sc = get_score(f_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = f_new
        eta_p *= 0.999
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.4, 1.0]:
        inits.append(d ** pw / (d ** pw).sum())
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.01
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    best_f = inits[0]
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: free optimization
    ph1_end = min(deadline, t0 + 420.0)
    f1, s1 = optimize(best_f.copy(), ph1_end, rng, sym=False, max_pk=25, eta0=0.06)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()

    # Phase 2: symmetric optimization
    if time.monotonic() < deadline - 200.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        f2, s2 = optimize(fs, min(deadline, t0 + 820.0), rng, sym=True, max_pk=20, eta0=0.05)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()

    # Phase 3: free refinement from best
    if time.monotonic() < deadline - 60.0:
        f3, s3 = optimize(best_f.copy(), min(deadline, t0 + 1050.0), rng, sym=False, max_pk=15, eta0=0.03)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()

    # Phase 4: final micro-polish
    if time.monotonic() < deadline - 5.0:
        eta_final = 0.003
        for _ in range(2000):
            if time.monotonic() > deadline - 2.0:
                break
            b = fft_conv(best_f, best_f)
            M = b.max()
            peaks = select_peaks(b, M, 0.9999, 5)
            g = compute_grad(best_f, b, n, peaks, sharpness=0.999)
            gmax = g.max()
            if gmax < 1e-30:
                break
            f_new = best_f * np.exp(-eta_final * g / gmax)
            f_new = np.maximum(f_new, 1e-20)
            f_new /= f_new.sum()
            sc = get_score(f_new)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_f = f_new
            eta_final *= 0.9995

    # Local perturbations
    if time.monotonic() < deadline - 3.0:
        for _ in range(40):
            if time.monotonic() > deadline - 1.0:
                break
            trial = best_f.copy()
            trial *= np.exp(0.004 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = best_f.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    fs = evaluate_sequence(result_list)
    if not np.isfinite(fs):
        result_list = [1.0 / n] * n
    return result_list
