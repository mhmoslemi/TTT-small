Also, I want to make sure the weighted peaks are computed correctly. The key formula:

For each active peak position k, the contribution to b[k] is sum_i a[i]*a[k-i]. To reduce b[k], we want to reduce a[i] for i that contribute to b[k]. The "gradient" for coordinate i is:

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def make_init(n, rng, kind):
    x = np.arange(n, dtype=np.float64)
    if kind == 0:
        return np.ones(n) / n
    if kind == 1:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        return d ** 0.5 / (d ** 0.5).sum()
    if kind == 2:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        return d / d.sum()
    if kind == 3:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        return d ** 1.5 / (d ** 1.5).sum()
    if kind == 4:
        h2 = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1.0)))
        return h2 / h2.sum()
    # Symmetric random
    amp = 0.1 * (kind - 3)
    half = (n + 1) // 2
    r = np.ones(half) + amp * rng.standard_normal(half)
    np.maximum(r, 0.05, out=r)
    idx = np.minimum(np.arange(n), n - 1 - np.arange(n))
    return r[idx] / r[idx].sum()

def run_opt(f, deadline, rng, max_peaks, eta0, check_iv, pat_lim):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta_cur = eta0
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 50000.0)
        frac = 0.996 - 0.04 * prog
        thresh = M * frac
        active = b >= thresh
        cnt = active.sum()
        h = np.zeros(L)
        if cnt > max_peaks:
            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
            h[idx] = b[idx] - thresh
        elif cnt >= 1:
            h[active] = b[active] - thresh
        else:
            h[np.argmax(b)] = 1.0
        hmax = h.max()
        if hmax < 1e-30:
            break
        h /= hmax
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta_cur * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
                if patience >= pat_lim:
                    amp = 0.01 + 0.02 * min(1.0, (patience - pat_lim) / 150.0)
                    f = best_f * np.exp(amp * rng.standard_normal(n))
                    np.maximum(f, 1e-30, out=f)
                    f /= f.sum()
                    patience = 0
        if it > 500:
            eta_cur *= 0.9999999
    return best_f, best_sc
