def _p2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _p2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]
