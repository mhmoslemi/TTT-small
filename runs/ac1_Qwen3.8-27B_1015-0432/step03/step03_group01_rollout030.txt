                best_sc = sc

    # Local perturbation search
    if time.monotonic() < dl - 15:
        wmax = max(2, n // 25)
        cur = best.copy()
        for it in range(8000):
            if time.monotonic() > dl - 15:
                break
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            w1 = int(rng.integers(1, wmax + 1))
            ctr = min(pk, n - 1)
            lo = max(0, ctr - w1)
            hi = min(n - w1, ctr)
            s1 = int(rng.integers(lo, hi + 1)) if hi >= lo else 0
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.02, 0.03)))
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.02, 0.03)))
            if rng.random() < 0.1:
                cd *= float(np.exp(rng.normal(0.0, 0.005)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(_tolist(cd)))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.05:
                cur = best.copy()

    # Symmetrization check
    if time.monotonic() < dl - 10:
        sym = 0.5 * (best + best[::-1])
        sym = _norm(np.clip(sym, 0.0, 1000.0), scale)
        sc = float(evaluate_sequence(_tolist(sym)))
        if np.isfinite(sc) and sc < best_sc:
            best = sym.copy()
            best_sc = sc

    # Final polish
    if time.monotonic() < dl - 4:
        cur = best.copy()
        for tv in [1e-5, 5e-6, 2e-6]:
            if time.monotonic() > dl - 4:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 500, 'gtol': 1e-15,
                                        'maxls': 20, 'ftol': 1e-20})
                cur = np.clip(res.x, 0.0, 1000.0)
                cur = _norm(cur, scale)
                best, best_sc = _upd(cur, best, best_sc, scale)
            except Exception:
                pass

    best = np.clip(best, 0.0, 1000.0)
    if not np.all(np.isfinite(best)) or best.sum() < 0.01:
        best = np.ones(n)
    return _tolist(best)
