import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    s = f.sum()
    if s < 1e-10:
        return 1e30
    b = fft_conv(f, f)
    return float(2.0 * f.size * b.max() / (s * s))

def grad_softmax(f, b, temp_frac=0.005):
    """Gradient using softmax weighting of peaks."""
    n = f.size
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    temp = M * temp_frac
    scaled = b / temp
    scaled -= scaled.max()
    h = np.exp(scaled)
    h_sum = h.sum()
    if h_sum < 1e-15:
        h = np.zeros(L)
        h[int(np.argmax(b))] = 1.0
    else:
        h /= h_sum
    # Zero out negligible weights for efficiency
    cutoff = h.max() * 1e-8
    h[h < cutoff] = 0.0
    h_sum2 = h.sum()
    if h_sum2 > 0:
        h /= h_sum2
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def grad_peaks(f, b, frac=0.999, power=10, max_pk=2000):
    """Gradient using top-peak selection with power weighting."""
    n = f.size
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    L = 2 * n - 1
    w = np.maximum(b[peaks] - thresh, 1e-30) ** power
    w /= w.sum()
    h = np.zeros(L)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

        if gmax < 1e-28:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:

                    f /= f.sum()
                    no_improv = 0

        eta *= 0.999
    return best_f, best_sc
