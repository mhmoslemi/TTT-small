import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n+m-1]

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks, weights):
    n = f.size
    L = 2*n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n-1:2*n-1][::-1]

            trial = best_f * np.exp(scale * rng.standard_normal(n))
            trial = np.maximum(trial, 0.0)
            trial = norm1(trial)
            sc = obj_val(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()
    
    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
