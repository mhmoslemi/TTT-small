                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = normalize(f)
                no_improv = 0
                eta = 0.05
                if time.monotonic() > deadline:
                    break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    inits.append(normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))))
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.3, 1.0, 2.0]:
        inits.append(normalize(d ** pw))
    r = np.ones(n) + 0.5 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(normalize(r))
    base = normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1))))
    inits.append(normalize(base * (1.0 + 0.3 * rng.standard_normal(n))))
    # Anti-symmetric structure
    asym = np.maximum(base - 0.5 * d / d.max(), 0.01)
    inits.append(normalize(asym))
    best_f, best_sc = None, np.inf
    for init in inits:
        sc = score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    # Phase 1: broad asymmetric (50%)
    ph1 = t0 + budget_s * 0.50
    fo, sco = optimize(best_f.copy(), ph1, rng, n_peak_max=200)
    if sco < best_sc:
        best_sc, best_f = sco, fo.copy()
    # Phase 2: focused (25%)
    if time.monotonic() < deadline - 40.0:
        ph2 = t0 + budget_s * 0.75
        fo, sco = optimize(best_f.copy(), ph2, rng, n_peak_max=80)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    # Phase 3: asymmetric polish
    if time.monotonic() < deadline - 20.0:
        fo, sco = optimize(best_f.copy(), deadline - 10.0, rng, n_peak_max=40)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    # Symmetric variant check
    if time.monotonic() < deadline - 15.0:
        fs = normalize(0.5 * (best_f + best_f[::-1]))
        scs = score(fs)
        if scs < best_sc:
            best_sc, best_f = scs, fs.copy()
    # Tiny local perturbations
    if time.monotonic() < deadline - 3.0:
        for _ in range(400):
            if time.monotonic() > deadline - 1.0:
                break
            tr = best_f * np.exp(0.002 * rng.standard_normal(n))
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()
    # Final: try anti-symmetric projection
    if time.monotonic() < deadline - 2.0:
        anti = np.maximum(best_f - 0.5 * (best_f + best_f[::-1]), 0.0)
        if anti.sum() > 1e-10:
            af = normalize(anti)
            s_a = score(af)
            if s_a < best_sc:
                best_sc, best_f = s_a, af
    result = [float(v) for v in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
