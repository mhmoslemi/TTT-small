    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])
    L = 2 * n - 1
    h = np.zeros(L)
    wv = b[peaks]
    h[peaks] = wv / wv.sum()
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False, max_iter=600000):
    n = f.shape[0]
    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    if sym:
        f = 0.5 * (f + f[::-1])
        f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.07
    tau = 12.0
    no_improv = 0
    restarts = 0
    eval_period = 20

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        tau = 12.0 + 0.015 * it
        if it % 4000 < 2000:
            g = smooth_grad(f, b, M, tau)
        else:
            frac = 0.9995 - 0.003 * min(1.0, it / 5000.0)
            g = hard_grad(f, b, frac, 300)
        gmax = g.max()
        if gmax < 1e-28:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()

        if it % eval_period == 0:
            sc = get_score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 500:
                scale = 0.02 + 0.09 * min(1.0, restarts / 6.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                restarts += 1
                eta = min(0.22, eta * 1.3)
                if restarts > 8:
                    break
        eta *= 0.9987

    return best_f, best_sc

def lp_hull(f, M, time_limit=60.0):
    import scipy.optimize as sco
    n = f.shape[0]
    t_end = time.monotonic() + time_limit
    b = fft_conv(f, f)
    active = np.argsort(b)[-60:]
    active.sort()
    g_sol = None
    for _ in range(40):
        if time.monotonic() > t_end:
            break
        A = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            A[i, js] = f[k - js]
        try:
            res = sco.linprog(np.full(n, -1.0), A_ub=A,
                              b_ub=np.full(len(active), M),
                              bounds=[(0, None)] * n,
                              method='highs',
                              options=dict(time_limit=3.0))
            if res.status != 0:
                break
            g_sol = res.x.copy()
        except Exception:
            break
        if g_sol is None or g_sol.sum() < 1e-15:
            break
        fg = fft_conv(f, g_sol)
        viol = fg > M * (1.0 + 1e-7)
        if not viol.any():
            return g_sol
        vidx = np.where(viol)[0]
        if len(vidx) > 40:
            vidx = vidx[np.argsort(fg[vidx])[-40:]]
        active = np.unique(np.concatenate([active, vidx]))
        if len(active) > 150:
            tv = fg[active]
            active = active[np.argsort(tv)[-150:]]
            active.sort()
    return g_sol

def lp_update(f, deadline_left):
    b = fft_conv(f, f)
    M = b.max()
    g = lp_hull(f, M, time_limit=min(40.0, deadline_left * 0.3))
    if g is None or g.sum() < 1e-10:
        return f, get_score(f)
    t_blend = 0.3
    fn = (1.0 - t_blend) * f + t_blend * g / g.sum() * f.sum()
    fn = np.maximum(fn, 1e-20)
    fn /= fn.sum()
    return fn, get_score(fn)

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 22.0
    n = len(height_sequence_1)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in (0.4, 1.0, 2.0):
        inits.append(d ** pw / (d ** pw).sum())

    best_f = inits[0]
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: free optimization
    ph1 = min(deadline, t0 + 480.0)
    f1, s1 = optimize(best_f.copy(), ph1, rng, sym=False)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()

    # Phase 2: symmetric
    if time.monotonic() < deadline - 250.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ph2 = min(deadline, t0 + 820.0)
        f2, s2 = optimize(fs, ph2, rng, sym=True)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()

    # Phase 3: LP hulling blend
    if time.monotonic() < deadline - 120.0:
        dl_left = deadline - time.monotonic()
        fn, sn = lp_update(best_f.copy(), dl_left)
        if sn < best_sc:
            best_sc, best_f = sn, fn.copy()

    # Phase 4: final free refinement
    if time.monotonic() < deadline - 40.0:
        f4, s4 = optimize(best_f.copy(), deadline - 12.0, rng, sym=False)
        if s4 < best_sc:
            best_sc, best_f = s4, f4.copy()

    # Phase 5: local perturbations
    if time.monotonic() < deadline - 4.0:
        for _ in range(80):
            if time.monotonic() > deadline - 2.0:
                break
            trial = best_f * np.exp(0.004 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc, best_f = sc, trial.copy()

    result = np.clip(best_f, 0.0, 1000.0)
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    fv = evaluate_sequence(result_list)
    if not np.isfinite(fv):
        result_list = [1.0 / n] * n
    return result_list
