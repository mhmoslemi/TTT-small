def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def ac1(f):
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    c = fft_conv(f, f)
    return float(2.0 * n * c.max() / (s * s))

def norm1(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / len(f)
    return f / s

def subgrad(f, peaks, weights):
    n = len(f)
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    conv = fft_conv(f, h[::-1])
    g = 2.0 * conv[n-1:2*n-1][::-1]
    return np.maximum(g, 0.0)
