import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, tau, max_peaks=600):
    """Compute soft-weighted subgradient of max(conv(f,f))."""
    n = f.shape[0]
    b = fft_conv(f, f)
    M = b.max()
    shifted = tau * (b - M)
    p = np.exp(shifted)
    ps = p.sum()
    if ps < 1e-30:
        pk = np.argmax(b)
        p = np.zeros_like(b)
        p[pk] = 1.0
    else:
        p /= ps
        if len(p) > max_peaks:
            top = np.argsort(p)[::-1][:max_peaks]
            pf = np.zeros_like(p)
            pf[top] = p[top]
            pf /= pf.sum()
            p = pf
    # grad_j = 2 * sum_k p_k * f_{k-j}
    h = p.copy()
    c = fft_conv(f, h[::-1])
    grad = 2.0 * c[n - 1:2 * n - 1][::-1]
    return grad, b, M

def optimize(f, deadline, rng, eta0=0.04, tau_init=50, tau_end=5, max_it=200000, sym=False):

    if s < 1e-15:
        return f, np.inf

    bf = f.copy()
    bs = score(f)
    eta = eta0
    tau = tau_init
    ni = 0
    for it in range(max_it):
        if time.monotonic() > deadline:

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n+m-1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def soft_grad(f, tau, K):
    n = f.shape[0]
    b = fft_conv(f, f)
    M = b.max()
    z = np.clip(tau * (b - M), -700, 0)
    p = np.exp(z)
    ps = p.sum()
    if ps < 1e-30:
        p = np.zeros_like(b)
        p[int(np.argmax(b))] = 1.0
    else:
        p /= ps
        if p.size > K:
            idx = np.argpartition(-p, K)[:K]
            p2 = np.zeros_like(p)
            p2[idx] = p[idx]
            s2 = p2.sum()
            if s2 > 1e-30:
                p = p2 / s2
    c = fft_conv(f, p[::-1])
    return 2.0 * c[n-1:2*n-1][::-1]
