                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > 50:
                amp = 0.015 + 0.01 * min(1.0, patience / 300.0)
                f = best_f * np.exp(amp * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.30, eta * 1.3)
            if it > 200:
                eta *= 0.9985
    return best_f, best_sc

def fine_tune(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    for i in range(500):
        if time.monotonic() > deadline:
            break
        for _ in range(5):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.93
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    x = np.arange(n, dtype=np.float64)
    inits = []
    inits.append(p / p.sum())
    inits.append(np.ones(n, dtype=np.float64) / n)
    for pw in (0.5, 1.0, 1.5):
        d = np.minimum(x, n - 1.0 - x) + 1.0
        v = d ** pw
        inits.append(v / v.sum())
    for a in (0.1, 0.25, 0.5):
        r = np.ones(n) + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())
    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()
    pe1 = min(deadline, t0 + 280.0)
    f1, s1 = run_opt(best_f.copy(), pe1, rng, max_peaks=250, eta_init=0.12)
    if s1 < best_sc:
        best_sc, best_f = s1, f1
    if time.monotonic() < deadline - 120.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = np.maximum(fs, 0.0)
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        pe2 = min(deadline, t0 + 650.0)
        f2, s2 = run_opt(best_f.copy(), pe2, rng, max_peaks=350, eta_init=0.06)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 40.0:
        for k in range(10):
            if time.monotonic() > deadline - 20.0:
                break
            amp = 0.01 + 0.006 * k
            fr = best_f * np.exp(amp * rng.standard_normal(n))
            np.maximum(fr, 1e-30, out=fr)
            fr /= fr.sum()
            dl_k = min(deadline - 15.0, time.monotonic() + 80.0)
            fr, sr = run_opt(fr, dl_k, rng, max_peaks=350, eta_init=0.06)
            if sr < best_sc:
                best_sc, best_f = sr, fr
    if time.monotonic() < deadline - 3.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 1.0, rng)
    result = [float(v) for v in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
