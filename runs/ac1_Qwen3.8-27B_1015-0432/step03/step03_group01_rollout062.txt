    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau, ktop=0):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    if ktop > 0 and ktop < len(d):
        kth = np.partition(d, -ktop)[-ktop]
        mask = d >= kth
        e = np.where(mask, np.exp(d), 0.0)
    else:
        e = np.exp(d)
    s = e.sum()
    if s < 1e-30:
        return cm, np.zeros_like(c)
    return cm + tau * np.log(s), e / s

def _obj_grad(a, tau, ktop=0):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _conv(a)
    mv, p = _smax(c, tau, ktop)
    ar = a[::-1]
    L = _fpow2(len(p) + n - 1)
    cv = ifft(fft(p, L) * fft(ar, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1]
    obj = 2.0 * n * mv / (S * S)
    grad = (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
    return obj, grad

def _obj_fn(a, tau, ktop=0):
    return _obj_grad(a, tau, ktop)[0]

def _grad_fn(a, tau, ktop=0):
    return _obj_grad(a, tau, ktop)[1]

def _norm(a, sc):
    s = a.sum()
    if s < 1e-8:
        return a.copy()
    return a / s * sc

def _ev(cand):
    try:
        sc = float(evaluate_sequence(cand.tolist() if isinstance(cand, np.ndarray) else cand))
        return sc if np.isfinite(sc) else float('inf')
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 18.0)

    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(512)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(256)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)

    scale = np.sqrt(2.0 * n)
    a0 = _norm(par, scale)
    best = a0.copy()
    best_sc = _ev(best)

    x = np.arange(n, dtype=np.float64)
    xm = 0.5 * (n - 1)
    dx = x - xm
    inits = [a0.copy()]

    for p in [1.0, 1.5, 2.0, 3.0, 4.0, 6.0, 8.0]:
        c = np.clip(np.cos(np.pi * dx / (n + 1)), 0, None) ** p
        inits.append(_norm(c, scale))
    for sig in [0.12, 0.18, 0.25, 0.33, 0.45, 0.6]:
        g = np.exp(-0.5 * (dx / (sig * n)) ** 2)
        inits.append(_norm(g, scale))
    rc = 0.5 * (1 + np.cos(2 * np.pi * dx / n))
    inits.append(_norm(rc, scale))
    pb = np.clip(1 - (dx / (0.5 * n)) ** 2, 0, None)
    inits.append(_norm(pb, scale))
    for w in [0.55, 0.7, 0.85]:
        bx = np.where(np.abs(dx) <= 0.5 * w * n, 1.0, 0.0).astype(float)
        k = max(1, int(0.05 * n))
        kern = np.exp(-0.5 * (np.arange(-k, k + 1) / (k / 3.0)) ** 2)
        kern /= kern.sum()
        bx = np.convolve(bx, kern, 'same')
        inits.append(_norm(bx, scale))
    for _ in range(3):
        r = rng.uniform(0, 1, n)
        k = max(1, int(0.03 * n))
        kern = np.exp(-0.5 * (np.arange(-k, k + 1) / (k / 3.0)) ** 2)
        kern /= kern.sum()
        r = np.convolve(r, kern, 'same')
        inits.append(_norm(r, scale))

    for inp in inits:
        sc = _ev(inp)
        if sc < best_sc:
            best = inp.copy()
            best_sc = sc
    cur = best.copy()
    bounds = [(0.0, 1000.0)] * n

    # Phase 1: broad optimization
    stage1 = [(8.0, 0, 200), (4.0, 0, 200), (2.0, 0, 250), (1.0, 0, 300), (0.5, 0, 300)]
    for tv, kt, mx in stage1:
        if time.monotonic() > dl - 300.0:
            break
        rem = dl - time.monotonic()
        iters = int(min(mx, rem * 12))
        if iters < 30:
            break
        try:
            res = minimize(_obj_fn, cur, args=(tv, kt), jac=_grad_fn,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': iters, 'gtol': 1e-14,
                                    'maxls': 50, 'ftol': 1e-20, 'maxfun': iters * 4})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _norm(nx, scale)
            sc = _ev(nx)
            if sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # Phase 2: focused optimization with top-k
    stage2 = [(0.3, 500, 400), (0.15, 300, 400), (0.08, 200, 500),
              (0.04, 150, 500), (0.02, 100, 500), (0.01, 80, 600),
              (0.005, 60, 600), (0.002, 50, 600), (0.001, 40, 700)]
    for tv, kt, mx in stage2:
        if time.monotonic() > dl - 100.0:
            break
        rem = dl - time.monotonic()
        iters = int(min(mx, rem * 14))
        if iters < 30:
            break
        try:
            res = minimize(_obj_fn, cur, args=(tv, kt), jac=_grad_fn,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': iters, 'gtol': 1e-15,
                                    'maxls': 40, 'ftol': 1e-20, 'maxfun': iters * 4})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _norm(nx, scale)
            sc = _ev(nx)
            if sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # Symmetrization
    if time.monotonic() < dl - 50.0:
        sym = 0.5 * (cur + cur[::-1])
        sym = _norm(np.clip(sym, 0, 1000.0), scale)
        sc = _ev(sym)
        if sc < best_sc:
            best = sym.copy()
            cur = sym.copy()
            best_sc = sc

    # Multi-restart
    if time.monotonic() < dl - 60.0:
        for _ in range(4):
            if time.monotonic() > dl - 60.0:
                break
            cd = best.copy() * np.exp(rng.normal(0, 0.02, n))
            cd = np.clip(cd, 0, 1000.0)
            cd = _norm(cd, scale)
            try:
                res = minimize(_obj_fn, cd, args=(0.02, 100), jac=_grad_fn,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 200, 'gtol': 1e-15,
                                        'maxls': 30, 'ftol': 1e-18})
                nx = np.clip(res.x, 0, 1000.0)
                nx = _norm(nx, scale)
                sc = _ev(nx)
                if sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                    cur = nx.copy()
            except Exception:
                pass

    # Local search
    if time.monotonic() < dl - 15.0:
        wmax = max(2, n // 40)
        for it in range(1500):
            if time.monotonic() > dl - 15.0:
                break
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            center = min(max(pk, n - 1), n - 1)
            w1 = int(rng.integers(1, wmax + 1))
            lo = max(0, center - w1)
            hi = min(n - w1, center)
            s1 = int(rng.integers(lo, hi + 1)) if hi >= lo else int(rng.integers(0, max(1, n - w1)))
            cd[s1:s1 + w1] *= np.exp(float(rng.normal(-0.03, 0.04)))
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, max(1, n - w2)))
            cd[s2:s2 + w2] *= np.exp(float(rng.normal(0.03, 0.04)))
            if rng.random() < 0.1:
                cd *= np.exp(float(rng.normal(0, 0.01)))
            cd = np.clip(cd, 0, 1000.0)
            if cd.sum() > 0.01:
                sc = _ev(cd)
                if sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.05:
                cur = best.copy()

    # Final polish
    if time.monotonic() < dl - 3.0:
        cur = best.copy()
        for tv, kt in [(0.002, 50), (0.001, 40), (5e-4, 30), (2e-4, 20)]:
            if time.monotonic() > dl - 3.0:
                break
            try:
                res = minimize(_obj_fn, cur, args=(tv, kt), jac=_grad_fn,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'gtol': 1e-16,
                                        'maxls': 25, 'ftol': 1e-22})
                nx = np.clip(res.x, 0, 1000.0)
                nx = _norm(nx, scale)
                sc = _ev(nx)
                if sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    final_sc = _ev(best)
    if not np.isfinite(final_sc):
        best = a0.copy()
    if not np.isfinite(_ev(best)):
        best = np.ones(n)
    return [float(v) for v in best]
