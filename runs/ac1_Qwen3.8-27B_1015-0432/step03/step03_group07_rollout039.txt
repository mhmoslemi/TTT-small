import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.maximum(b - M * thresh, 0.0)

    w = (w / (M * spread)) ** 2
    active = np.where(w > 1e-12)[0]
    if len(active) > max_active:
        idx = np.argsort(w[active])[::-1][:max_active]
        w2 = np.zeros(L)
        w2[active[idx]] = w[active[idx]]
        w = w2
    if w.sum() < 1e-30:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
