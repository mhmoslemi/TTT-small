            h[idx] = 1.0
        else:
            h.fill(0.0)
            h[active] = 1.0
        c = fft_conv(best_f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        trial = best_f * np.exp(-eta * g / gmax)
        np.maximum(trial, 1e-30, out=trial)
        trial /= trial.sum()
        sc = score(trial)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_f = trial.copy()
        eta *= 0.995
    # Random micro-perturbations
    scale = 0.003
    while time.monotonic() < deadline - 1.0:
        for _ in range(4):
            if time.monotonic() > deadline - 1.0:
                break
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.85
        if scale < 1e-6:
            break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()

    inits = [p.copy()]
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    for pw in [0.5, 1.0, 1.5, 2.5]:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        v = d ** pw
        inits.append(v / v.sum())
    for ph in [0.0, 0.5]:
        h2 = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + ph) / (n + 1.0)))
        inits.append(h2 / h2.sum())
    for a in [0.15, 0.3, 0.5]:
        r = np.ones(n) + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()

    phase1_end = min(deadline, t0 + 380.0)
    f1, s1 = run_opt(best_f.copy(), phase1_end, rng, max_peaks=200, eta_init=0.12)
    if s1 < best_sc:
        best_sc, best_f = s1, f1

    if time.monotonic() < deadline - 160.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        phase2_end = min(deadline, t0 + 720.0)
        f2, s2 = run_opt(best_f.copy(), phase2_end, rng, max_peaks=350, eta_init=0.07)
        if s2 < best_sc:
            best_sc, best_f = s2, f2

    if time.monotonic() < deadline - 120.0:
        for restart in range(3):
            if time.monotonic() > deadline - 50.0:
                break
            sc_r = 0.012 + 0.012 * restart
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            end_r = min(deadline - 50.0, time.monotonic() + 70.0)
            f_r, s_r = run_opt(f_r, end_r, rng, max_peaks=350, eta_init=0.09)
            f_s = 0.5 * (f_r + f_r[::-1])
            f_s /= f_s.sum()
            s_s = score(f_s)
            if s_s < s_r:
                s_r, f_r = s_s, f_s
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r

    if time.monotonic() < deadline - 10.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 5.0, rng)

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
