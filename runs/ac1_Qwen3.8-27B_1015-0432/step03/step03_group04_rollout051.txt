    s = f.sum()

        h[int(np.argmax(b))] = 1.0
    else:
        h /= hs
    conv = fft_conv(f, h[::-1])
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.08
    patience = 0
    total_patience = 800
    
    for it in range(max_iter):
        if time.monotonic() > deadline:

                        tr = 0.5 * (tr + tr[::-1])
                    tr /= tr.sum()
                    if get_score(tr) < best_sc:
                        f = tr.copy()
                        best_f = tr.copy()
                        best_sc = get_score(tr)
                        moved = True
                        break
                if not moved:
                    f = best_f * np.exp(0.08 * rng.standard_normal(n))
                    f = np.maximum(f, 1e-18)
                    if sym:
                        f = 0.5 * (f + f[::-1])
                    f /= f.sum()
                pat = 0
                eta = min(0.3, eta * 1.3)
        eta *= 0.998
    return best_f, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    p = p / p.sum()
    x = np.arange(n, dtype=np.float64)
    inits = [p]
    inits.append(np.ones(n) / n)
    hc = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(hc / hc.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in (0.5, 1.0, 1.5, 2.5):
        v = d ** pw
        inits.append(v / v.sum())

    best_f = inits[0].copy()
    best_sc = 1e30
    for ini in inits:
        sc = get_score(ini)
        if sc < best_sc:
            best_sc = sc
            best_f = ini.copy()

    # Phase 1: main asymmetric optimization with smooth gradient
    f1, s1 = run_opt(best_f.copy(), dl, rng, T0=1.0, T1=100.0, eta0=0.06, sym=False, max_it=400000)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()

    # Phase 2: symmetric optimization
    if time.monotonic() < dl - 50.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        f2, s2 = run_opt(fs, dl, rng, T0=2.0, T1=120.0, eta0=0.05, sym=True, max_it=200000)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()

    # Phase 3: free refinement with high T for peak focusing
    if time.monotonic() < dl - 10.0:
        f3, s3 = run_opt(best_f.copy(), dl - 5.0, rng, T0=5.0, T1=150.0, eta0=0.03, sym=False, max_it=200000)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()

    # Phase 4: local fine-tuning with tiny perturbations
    if time.monotonic() < dl - 2.0:
        for _ in range(300):
            if time.monotonic() > dl - 1.0:
                break
            tr = best_f * np.exp(0.002 * rng.standard_normal(n))
            tr = np.maximum(tr, 1e-18)
            tr /= tr.sum()
            sc = get_score(tr)
            if sc < best_sc:
                best_sc = sc
                best_f = tr.copy()

    result = best_f.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    out = [float(v) for v in result]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs):
        out = [1.0 / n] * n
    return out
