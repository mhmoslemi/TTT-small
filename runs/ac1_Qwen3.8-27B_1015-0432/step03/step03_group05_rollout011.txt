import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return float(2.0 * f.size * b.max() / (s * s))

def soft_grad(f, T_frac, n):
    b = fft_conv(f, f)
    bm = b.max()
    T = T_frac * bm if bm > 1e-30 else 1e-10
    w = np.exp(np.clip((b - bm) / T, -40, 0))
    W = w.sum()
    if W < 1e-30:
        w[0] = 1.0
        W = 1.0
    c = fft_conv(f, w[::-1])
    grad = 2.0 * c[n - 1:2 * n - 1][::-1]
    grad = (2.0 * n / T) * (grad / W)
    return grad, float(bm)

def hard_grad(f, n):
    b = fft_conv(f, f)
    M = b.max()
    thr = M * 0.9995
    pk = np.where(b >= thr)[0]
    if len(pk) < 3:
        pk = np.argsort(b)[-15:]
    elif len(pk) > 300:
        pk = np.argsort(b)[-300:]
    L = 2 * n - 1

    h[pk] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1] / len(pk)
    return g, float(M)
