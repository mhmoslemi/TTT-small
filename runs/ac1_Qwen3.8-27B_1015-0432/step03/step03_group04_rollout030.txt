            best_f = f.copy()
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in (0.3, 0.8, 1.5, 2.5):
        inits.append(d ** pw / (d ** pw).sum())
    for _ in range(4):
        r = np.abs(rng.standard_normal(n)) + 0.01
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())
    for k in (2, 3):
        h2 = 0.5 * (1.0 + np.cos(2.0 * np.pi * k * x / n))
        h2 = np.maximum(h2, 0.01)
        inits.append(h2 / h2.sum())
    best_f = inits[0]
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    ph1 = min(deadline, t0 + 380.0)
    f1, s1 = optimize(best_f.copy(), ph1, rng)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()
    if time.monotonic() < deadline - 90.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = np.maximum(fs, 1e-20)
        fs /= fs.sum()
        f2, s2 = optimize(fs, min(deadline, t0 + 760.0), rng, sym=True)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()
    if time.monotonic() < deadline - 50.0:
        f3, s3 = optimize(best_f.copy(), min(deadline, t0 + 960.0), rng)
        if s3 < best_sc:
            best_sc, best_f = s3, f3.copy()
    if time.monotonic() < deadline - 25.0:
        f4, s4 = flatten_pass(best_f.copy(), min(deadline, t0 + 1060.0), rng)
        if s4 < best_sc:
            best_sc, best_f = s4, f4.copy()
    if time.monotonic() < deadline - 2.0:
        for _ in range(300):
            if time.monotonic() > deadline - 1.0:
                break
            trial = best_f * np.exp(0.003 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc, best_f = sc, trial.copy()
    result = best_f.copy()
    if result.sum() < 0.01:
        result *= 0.02 / result.sum()
    result_list = [float(v) for v in result]
    fs = evaluate_sequence(result_list)
    if not np.isfinite(fs):
        result_list = [1.0 / n] * n
    return result_list
