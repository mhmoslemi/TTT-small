import numpy as np
import time

def fft_conv(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()  # Need at least n+m-1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    """Compute the AC1 score."""
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def subgrad_weighted(f, peaks, weights):
    """Compute weighted subgradient of max_k (f*f)[k]."""
    n = f.shape[0]
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += w * 2.0 * f[k-hi:k-lo+1][::-1]
    return np.maximum(grad, 0.0)

    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.05
    no_improv = 0
    restarts = 0
    it = 0
    while time.monotonic() < deadline and it < max_iter:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        # Adaptive threshold for peak selection
        width = 0.003 + 0.04 * min(1.0, it / 10000.0)
        thresh = M * (1.0 - width)
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 200:
            peaks = np.argsort(b)[-200:]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])
        # Weighted subgradient
        weights = (b[peaks] - thresh) / max(b[peaks].max() - thresh, 1e-30)
        weights = weights / weights.sum()
        g = subgrad_weighted(f, peaks, weights)
        gmax = g.max()
        if gmax < 1e-30:
            break
        gn = g / gmax
        # Adaptive step size

        best_sc_t = np.inf
        best_tr = f.copy()
        for te in [eta * 0.5, eta, eta * 1.5, eta * 2.0]:

    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.04
    no_improv = 0
    restarts = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        width = 0.003 + 0.025 * min(1.0, it / 6000.0)
        thresh = M * (1.0 - width)
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 120:
            idx = np.argsort(b[peaks])[-120:]
            peaks = peaks[idx]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])
        excess = b[peaks] - thresh
        w = excess / max(excess.sum(), 1e-30)
        g = subgrad_fft(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-30:
            break
        gn = g / gmax
        best_eta = eta
        best_sc_t = np.inf
        best_tr = f.copy()
        for te in [eta * 0.4, eta * 0.7, eta, eta * 1.6, eta * 2.5]:
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-20)
            if sym:
                tr = 0.5 * (tr + tr[::-1])
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc_t:
                best_sc_t = sc
                best_eta = te
                best_tr = tr
        f = best_tr
        if best_eta > eta * 1.25:
            eta = min(0.28, eta * 1.07)
        elif best_eta < eta * 0.7:
            eta = max(0.006, eta * 0.88)
        if it % 4 == 0:
            sc = score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 120:
                scl = 0.012 + 0.025 * min(1.0, restarts / 10.0)
                f = best_f * np.exp(scl * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = normalize(f)
                no_improv = 0
                restarts += 1
                eta = 0.04
                if restarts > 30:
                    break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = []
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    inits.append(normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))))
    inits.append(normalize(np.sin(np.pi * x / n) ** 2))
    d = np.minimum(x, n - 1.0 - x)
    inits.append(normalize(np.maximum(1.0 - 2.0 * d / n, 0.0) ** 2))
    r = np.ones(n) + 0.3 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(normalize(r))
    best_f, best_sc = None, np.inf
    for init in inits:
        sc = score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    f_s = normalize(0.5 * (best_f + best_f[::-1]))
    ph1_end = t0 + budget_s * 0.38
    fo, sco = optimize(f_s, ph1_end, rng, sym=True)
    if sco < best_sc:
        best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 80.0:
        ph2_end = min(deadline - 80.0, t0 + budget_s * 0.68)
        fo, sco = optimize(best_f.copy(), ph2_end, rng, sym=False)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 35.0:
        fs = normalize(0.5 * (best_f + best_f[::-1]))
        ph3_end = deadline - 20.0
        fo, sco = optimize(fs, ph3_end, rng, sym=True)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < deadline - 5.0:
        for _ in range(300):
            if time.monotonic() > deadline - 3.0:
                break
            tr = best_f * np.exp(0.0015 * rng.standard_normal(n))
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()
    result = [float(v) for v in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
