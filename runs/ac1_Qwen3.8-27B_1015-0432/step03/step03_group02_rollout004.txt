    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def optimize(f, deadline, rng, max_peaks=300, eta0=0.1, patience_max=80):
    """Main optimization loop with exponential update."""
    n = f.shape[0]
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta0
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        # Adaptive threshold: gradually tighten
        prog = min(1.0, it / 5000.0)
        frac = 0.99 - 0.06 * prog
        thresh = M * frac
        active = np.where(b >= thresh)[0]
        cnt = len(active)
        if cnt > max_peaks:
            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
        elif cnt >= 1:
            idx = active
        else:
            idx = np.array([np.argmax(b)])
        
        # Compute gradient direction
        h = np.zeros(L)
        h[idx] = 1.0
        c = fft_conv(f, h[::-1])
        g = c[n-1:2*n-1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        
        # Exponential update (mirror descent)
        f *= np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        
        # Check score periodically
        if it % 20 == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            
            if patience >= patience_max:
                # Restart with perturbation
                eps = 0.02 + 0.03 * min(1.0, patience / (2 * patience_max))
                f = best_f * np.exp(eps * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.3, eta * 1.5)
        
        # Decay eta
        if it > 200:
            eta *= 0.998
    
    # Final score check
    sc = score(best_f)
    if sc < best_sc:
        best_sc = sc
    return best_f, best_sc

def fine_tune(f, deadline, rng, n_steps=300):
    """Random perturbation refinement."""
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scales = [0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0001, 0.00005]

                sc = score(trial)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_f = trial.copy()
    return best_f, best_sc
