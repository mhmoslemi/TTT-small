Actually, this is exactly the problem of finding a nonnegative sequence whose aperiodic autocorrelation (or rather, whose self-convolution) is as flat as possible. This is related to "minimum redundancy sequences" and "Barker-like" codes but for continuous nonnegative values.

The optimal solution (without nonnegativity) would be related to Zadoff-Chu or similar sequences, but with nonnegativity it's different.

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def weighted_grad(f, b, thresh_frac, max_active=1200):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    spread = max(1.0 - thresh_frac, 1e-9)
    w = np.maximum(b - M * thresh_frac, 0.0)
    w = (w / (M * spread)) ** 2
    active = np.where(w > 1e-15)[0]
    if len(active) > max_active:
        idx = np.argpartition(w, -max_active)[-max_active:]
        w2 = np.zeros(L)
        w2[idx] = w[idx]
        w = w2
    if w.sum() < 1e-30:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

        if sym:

        if it % 10 == 0:

                no_improv = 0
