import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    needed = n + m - 1
    size = 1 << (needed - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:needed]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def peak_grad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.05, max_peaks=200, sym=False,
             max_iter=900000, evict=40, patience=300, wpow=2):

    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    momentum = np.zeros(n)
    rc = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        prog = min(1.0, it / 4000.0)
        thresh = M * (0.999 - 0.049 * prog)
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > max_peaks:
            peaks = peaks[np.argsort(b[peaks])[::-1][:max_peaks]]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])
        vals = b[peaks]
        w = (vals / vals.max()) ** wpow
        g = peak_grad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:
            break
        momentum = 0.85 * momentum + 0.15 * g
        mmax = momentum.max()

            break
        f = f * np.exp(-eta * momentum / mmax)
        f = np.maximum(f, 1e-20)
        if sym:

        f = norm_pos(f)
        if it % evict == 0:

                momentum = np.zeros(n)
                no_improv, rc = 0, rc + 1
                eta = min(0.2, eta * 1.4)
        eta *= 0.9998
    return best_f, best_sc
