import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def smooth_subgrad(f, b, T):
    n = f.shape[0]
    w = np.exp((b - b.max()) / T)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]
