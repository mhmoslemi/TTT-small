import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -40.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    L = _fpow2(3 * n - 2)
    # grad of smoothmax wrt a: 2 * cross_corr(a, p)
    bc = ifft(fft(a, L) * np.conj(fft(p, L))).real[:n]
    return 4.0 * n * bc / (S * S) - 4.0 * n * mv / (S * S * S)


def _normalize(a, scale):
    s = a.sum()
    if s < 1e-8:
        return a
    return a / s * scale


def _clip0(a):
    return np.clip(a, 0.0, 1000.0)

    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < best_sc:

        return sc
    return best_sc
