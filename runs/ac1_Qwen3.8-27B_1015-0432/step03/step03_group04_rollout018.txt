def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    s = f.sum()
    if s < 1e-10:
        return 1e30
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def compute_grad(f, b, k):
    n = len(f)
    Lc = 2 * n - 1

        peaks = np.arange(Lc)
    else:
        peaks = np.argpartition(b, -k)[-k:]
    w = b[peaks].astype(np.float64)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w = w / ws
    h = np.zeros(Lc)
    h[peaks] = w
    L = 1 << (3 * n - 1).bit_length()
    F = np.fft.rfft(f, L)
    H = np.fft.rfft(h, L)
    corr = np.fft.irfft(np.conj(F) * H, L)
    g = 2.0 * corr[:n]
    return np.maximum(g, 0.0)
