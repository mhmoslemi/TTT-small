def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _eval(a):
    return float(evaluate_sequence(a.tolist()))


def _normalize(a, sc):
    s = a.sum()
    if s < 1e-8:
        return np.ones(a.size) * sc / a.size
    return a / s * sc


def _peak_reduce(cur, n, best_sc, best, sc, dl):
    for _round in range(40):
        if time.monotonic() > dl - 60.0:
            break
        c = _conv(cur)
        top_k = min(5, 2 * n - 1)
        idx = np.argsort(c)[-top_k:]
        improved = False
        for jp in idx:
            lo = max(0, jp - (n - 1))
            hi = min(n - 1, jp)
            if hi < lo:
                continue
            ks = np.arange(lo, hi + 1)
            prods = cur[ks] * cur[jp - ks]
            tc = ks[np.argsort(prods)[-min(8, len(ks)):]]
            for frac in [0.97, 0.94, 0.90, 0.85, 0.78]:
                trial = cur.copy()
                trial[tc] *= frac
                trial = np.clip(trial, 0, None)
                if trial.sum() < 0.01:
                    continue
                s2 = _eval(trial)
                if np.isfinite(s2) and s2 < best_sc[0]:
                    best_sc[0] = s2
                    best[:] = trial
                    cur[:] = trial
                    improved = True
                    break
        if not improved:
            break
    return cur, best, best_sc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 20.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(6733)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)

    sc = np.sqrt(2.0 * n)
    x = np.arange(n, dtype=np.float64)

    inits = [par.copy()]
    inits.append((1.0 + np.cos(2.0 * np.pi * (x + 0.5) / n)) / 2.0)
    inits.append(np.sin(np.pi * (x + 0.5) / n) ** 2)
    ks = 41
    kh = np.exp(-0.5 * (np.arange(ks) - ks // 2) ** 2 / (ks // 3.0) ** 2)
    kh /= kh.sum()
    inits.append(np.convolve(par, kh, mode='same'))
    inits.append(par * (1.0 + 0.2 * np.cos(2.0 * np.pi * x / n)))
    inits.append(par * (1.0 + 0.3 * np.cos(4.0 * np.pi * x / n)))

    best = None
    best_sc = [np.inf]
    cur = None

    for ini in inits:
        c = np.clip(ini, 0, None)
        if c.sum() > 0.01:
            c = _normalize(c, sc)
            s2 = _eval(c)
            if np.isfinite(s2) and s2 < best_sc[0]:
                best_sc[0] = s2
                best = c.copy()
                cur = c.copy()

    if best is None:
        best = _normalize(par.copy(), sc)
        cur = best.copy()
        best_sc[0] = _eval(best)

    bl = [(0.0, 1000.0)] * n

    taus = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002,
            0.001, 0.0005, 0.0002, 0.0001, 5e-05, 2e-05]

    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 200.0:
            break
        miter = max(400, 2000 - i * 120)
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': miter, 'gtol': 1e-16,
                                    'maxls': 30, 'ftol': 1e-18,
                                    'maxfun': miter * 2})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                nx = _normalize(nx, sc)
            s2 = _eval(nx)
            if np.isfinite(s2) and s2 < best_sc[0]:
                best_sc[0] = s2
                best = nx.copy()
            cur = np.clip(res.x, 0.0, 1000.0)
        except Exception:
            pass

    cur, best, best_sc = _peak_reduce(cur.copy(), n, best_sc, best, sc, dl)

    if time.monotonic() < dl - 50.0:
        cur = best.copy()
        for tv in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 5e-05]:
            if time.monotonic() > dl - 50.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 500, 'gtol': 1e-17,
                                        'maxls': 25, 'ftol': 1e-19,
                                        'maxfun': 900})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = _normalize(nx, sc)
                s2 = _eval(nx)
                if np.isfinite(s2) and s2 < best_sc[0]:
                    best_sc[0] = s2
                    best = nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    cur, best, best_sc = _peak_reduce(cur.copy(), n, best_sc, best, sc, dl)

    if time.monotonic() < dl - 5.0:
        while time.monotonic() < dl - 3.0:
            cd = cur.copy()
            nw = int(rng.integers(1, max(2, n // 50)))
            s = int(rng.integers(0, n - nw))
            cd[s:s + nw] *= np.exp(rng.normal(0.0, 0.04, nw))
            if rng.random() < 0.06:
                p2 = int(rng.integers(0, n))
                w2 = int(rng.integers(1, max(2, n // 200)))
                cd[max(0, p2 - w2):min(n, p2 + w2)] *= np.exp(rng.normal(0, 0.03, min(n, p2 + w2) - max(0, p2 - w2)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                s2 = _eval(cd)
                if np.isfinite(s2) and s2 < best_sc[0]:
                    best_sc[0] = s2
                    best = cd.copy()
                    cur = cd.copy()
            if rng.random() < 0.05:
                cur = best.copy()

    if not np.isfinite(best_sc[0]):
        best = _normalize(par.copy(), sc)
        best_sc = [_eval(best)]

    return [float(v) for v in best]
