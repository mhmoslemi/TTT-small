def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, M, frac, max_pk, n):
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])
    w = b[peaks] - thresh
    w = np.maximum(w, M * 1e-4)
    w = w * w
    w /= w.sum()
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
