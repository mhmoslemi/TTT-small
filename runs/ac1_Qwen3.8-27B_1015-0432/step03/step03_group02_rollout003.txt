    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.10, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    h = np.zeros(L)
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 5000.0)
        frac = 0.996 - 0.04 * prog
        thresh = M * frac
        active = b >= thresh
        cnt = active.sum()
        if cnt > max_peaks:
            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
            h.fill(0.0)
            w = (b[idx] - thresh) / max(M - thresh, 1e-30)
            h[idx] = w
        elif cnt >= 1:
            h.fill(0.0)
            h[active] = 1.0
        else:
            h.fill(0.0)
            h[np.argmax(b)] = 1.0
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > 400:
                noise_sc = 0.02 + 0.03 * min(1.0, patience / 800.0)
                f = best_f * np.exp(noise_sc * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.30, eta * 1.25)
        if it > 200:
            eta *= 0.9994
    sc = score(f)
    if sc < best_sc:
        best_sc, best_f = sc, f
    return best_f, best_sc

def fine_tune(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    stuck = 0
    while time.monotonic() < deadline and scale > 1e-6:
        for _ in range(12):
            if time.monotonic() > deadline:
                break
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = trial.copy()
                stuck = 0
                break
        else:
            stuck += 1
        scale *= (0.85 if stuck >= 3 else 0.93)
    return best_f, best_sc

def smooth_step(f, sigma=3.0):
    n = len(f)
    k = int(2 * max(1, int(3 * sigma))) | 1
    x = np.arange(k) - k // 2
    ker = np.exp(-0.5 * (x / sigma) ** 2)
    ker /= ker.sum()
    out = np.convolve(f, ker, 'same')
    np.maximum(out, 1e-30, out=out)
    return out / out.sum()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 22.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()
    inits = [p]
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.0, 1.5, 2.0, 3.0]:
        v = d ** pw
        inits.append(v / v.sum())
    for ph in [0.0, 0.3, 0.7]:
        h2 = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + ph) / (n + 1.0)))
        inits.append(h2 / h2.sum())
    for a in [0.1, 0.25, 0.5]:
        r = 1.0 + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        k2 = 15
        ker = np.ones(k2) / k2
        r = np.convolve(r, ker, 'same')
        np.maximum(r, 0.01, out=r)
        inits.append(r / r.sum())
    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()
    phase1_end = min(deadline, t0 + 380.0)
    f1, s1 = run_opt(best_f.copy(), phase1_end, rng, max_peaks=180, eta_init=0.12, check_iv=18)
    if s1 < best_sc:
        best_sc, best_f = s1, f1
    if time.monotonic() < deadline - 140.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        fsm = smooth_step(fs, 2.0)
        sm = score(fsm)
        if sm < best_sc:
            best_sc, best_f = sm, fsm
        phase2_end = min(deadline, t0 + 720.0)
        f2, s2 = run_opt(best_f.copy(), phase2_end, rng, max_peaks=320, eta_init=0.07, check_iv=25)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 90.0:
        for restart in range(3):
            if time.monotonic() > deadline - 45.0:
                break
            sc_r = 0.012 + 0.012 * restart
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            f_r, s_r = run_opt(f_r, deadline - 45.0, rng, max_peaks=320, eta_init=0.08, check_iv=22)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r
    if time.monotonic() < deadline - 8.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 4.0, rng)
    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
