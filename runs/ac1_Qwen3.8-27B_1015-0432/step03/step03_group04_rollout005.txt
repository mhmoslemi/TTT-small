def fft_conv(a, b):
    """Full convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]

def get_score(f):
    """Compute AC1 upper bound."""
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, frac, max_pk):
    """Compute surrogate gradient of max autocorrelation."""
    n = f.shape[0]
    M = b.max()
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
