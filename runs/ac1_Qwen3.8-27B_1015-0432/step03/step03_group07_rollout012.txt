def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(a):
    n = len(a)
    b = fft_conv(a, a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * b.max() / (s * s)
