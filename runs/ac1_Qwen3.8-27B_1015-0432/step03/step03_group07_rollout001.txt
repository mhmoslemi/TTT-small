import numpy as np
import time

def fft_conv(a, b):
    na, nb = len(a), len(b)
    size = 1 << (na + nb - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:na + nb - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def compute_grad(f, b, temp):
    n = len(f)

        return np.zeros(n)
