def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def calc_score(b, n, s):
    if s < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def subgrad_peaks(f, peaks, wts):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = wts
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

    idx = np.argsort(b)[::-1][:P]

        jhi = min(n - 1, k)

            rl.append(np.full(len(jj), p))
            cl.append(jj)
