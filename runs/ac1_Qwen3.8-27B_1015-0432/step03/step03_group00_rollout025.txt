    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def _score(a):
    return float(evaluate_sequence(a.tolist()))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 20.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)):
        par = np.ones(6733)
    n = par.size
    if par.sum() < 1e-8:
        par = np.ones(n)

    sc = np.sqrt(2.0 * n)
    cur = par / par.sum() * sc
    best = cur.copy()
    best_sc = _score(best)

    x = np.arange(n, dtype=np.float64)
    # Diverse initializations
    for amp in [0.15, 0.35, 0.6]:
        sh = n * rng.uniform()
        c = cur * (1.0 + amp * np.cos(2.0 * np.pi * (x + sh) / n))
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            c = c / c.sum() * sc
            s = _score(c)
            if s < best_sc:
                best_sc = s
                best = c.copy()
                cur = c.copy()
    # Two-hump and anti-center
    for harm in [2, 3, 4]:
        c = 1.0 + 0.3 * np.cos(2.0 * np.pi * harm * x / n)
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            c = c / c.sum() * sc
            s = _score(c)
            if s < best_sc:
                best_sc = s
                best = c.copy()
                cur = c.copy()
    # Thinned parent
    for k in [2, 3]:
        c = cur.copy()
        c[k::k] *= 0.3
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            c = c / c.sum() * sc
            s = _score(c)
            if s < best_sc:
                best_sc = s
                best = c.copy()
                cur = c.copy()
    # Random noise on parent
    for _ in range(3):
        c = cur * np.exp(0.04 * rng.standard_normal(n))
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            c = c / c.sum() * sc
            s = _score(c)
            if s < best_sc:
                best_sc = s
                best = c.copy()
                cur = c.copy()

    bl = [(0.0, 1000.0)] * n
    # L-BFGS-B with decreasing tau
    taus = [5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tv in taus:
        if time.monotonic() > dl - 80.0:
            break
        remaining = dl - 80.0 - time.monotonic()
        iters = max(150, min(2000, int(remaining / 0.25)))
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': iters, 'gtol': 1e-15,
                                    'maxls': 50, 'ftol': 1e-18,
                                    'maxfun': iters * 4})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 1e-8:
                nx = nx / nx.sum() * sc
            s = _score(nx)
            if np.isfinite(s) and s < best_sc:
                best_sc = s
                best = nx.copy()
            cur = nx
        except Exception:
            pass

    # Peak shaving via explicit gradient on top peaks
    if time.monotonic() < dl - 40.0:
        for rnd in range(40):
            if time.monotonic() > dl - 40.0:
                break
            c = _conv(cur)
            top_k = 6
            idx = np.argsort(c)[-top_k:]
            improved = False
            for jp in sorted(idx, reverse=True):
                lo = max(0, jp - (n - 1))
                hi = min(n - 1, jp)
                ks = np.arange(lo, hi + 1)
                prods = cur[ks] * cur[jp - ks]
                tc = ks[np.argsort(prods)[-10:]]
                for frac in [0.97, 0.94, 0.90, 0.85, 0.78]:
                    trial = cur.copy()
                    trial[tc] *= frac
                    trial = np.clip(trial, 0, None)
                    if trial.sum() > 0.01:
                        s = _score(trial)
                        if np.isfinite(s) and s < best_sc:
                            best_sc = s
                            best = trial.copy()
                            cur = trial.copy()
                            improved = True
                            break
            if not improved:
                break

    # Local search with diverse perturbations
    while time.monotonic() < dl - 10.0:
        cd = cur.copy()
        nw = int(rng.integers(3, max(4, n // 15)))
        s0 = int(rng.integers(0, max(1, n - nw)))
        cd[s0:s0 + nw] *= np.exp(rng.normal(0.0, 0.06, nw))
        if rng.random() < 0.12:
            cd *= np.exp(rng.normal(0.0, 0.01))
        if rng.random() < 0.06:
            pos = rng.integers(0, n, 10)
            cd[pos] *= np.exp(rng.normal(0.0, 0.3, 10))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            s2 = _score(cd)
            if np.isfinite(s2) and s2 < best_sc:
                best_sc = s2
                best = cd.copy()
                cur = cd.copy()
        if rng.random() < 0.05:
            cur = best.copy()

    # Final cleanup
    thresh = 0.001 * max(best.max(), 1e-10)
    trial = best.copy()
    trial[trial < thresh] = 0.0
    if trial.sum() > 0.01:
        s = _score(trial)
        if np.isfinite(s) and s < best_sc:
            best_sc = s
            best = trial

    if not np.isfinite(best_sc):
        best = cur.copy()
        if not np.all(np.isfinite(best)):
            best = np.ones(n)

    return [float(v) for v in best]
