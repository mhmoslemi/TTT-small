    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, sym=False, max_it=500000, eta0=0.05, pf0=0.999, pf1=0.995, pmin=5, pmax=800):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        frac = pf0 + (pf1 - pf0) * min(1.0, it / 50000.0)
        thr = M * frac
        pk = np.where(b >= thr)[0]
        if len(pk) < pmin:
            pk = np.argsort(b)[-pmin:]
        elif len(pk) > pmax:
            pk = np.argsort(b)[-pmax:]
        g = subgrad(f, pk)
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 5 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 150:
                amp = 0.005 + 0.02 * min(1.0, ni / 1500.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                ni = 0
                eta = eta0
        if it > 20000:
            eta *= 0.99997
        eta = max(eta, 0.001)
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def lp_refine(f, deadline, max_lp=40, K=200, t0_step=0.08):
    n = f.shape[0]
    f = f.copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    t = t0_step
    bounds = [(0.0, None)] * n
    for lp_i in range(max_lp):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        k_sel = np.argsort(b)[-K:]
        A = np.zeros((K, n))
        for i, k in enumerate(k_sel):
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            A[i, jlo:jhi + 1] = f[k - jhi:k - jlo + 1][::-1]
        res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M),
                      bounds=bounds, method='highs',
                      options={'time_limit': 5, 'presolve': True})
        if not res.success:
            break
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g /= gs
        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        f_new /= f_new.sum()
        sc = score(f_new)
        if sc < bs - 1e-15:
            bs = sc
            bf = f_new.copy()
            f = f_new
            t = min(t * 1.3, 0.15)
        else:
            t *= 0.5
            if t < 0.005:
                t = t0_step
                # Perturb
                f = f * (1.0 + 0.01 * np.random.default_rng(lp_i * 1000 + 7).standard_normal(n))
                f = np.maximum(f, 0.0)
                f /= f.sum()
    return bf, bs
