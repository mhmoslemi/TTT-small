    it = 0

    while True:
        if time.monotonic() > deadline:
            break
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        T = T_base * (1.0 + 0.6 * min(1.0, no_improv / 200.0))
        if it % 80 == 0:
            T = T_base * 0.2
        g = soft_grad(f, b, T, topK=700)
        gmax = g.max()
        if gmax < 1e-28:
            eta *= 0.7
            if eta < 1e-5:
                break
            continue
        g /= gmax
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 20 == 0:
            sc = get_score(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
                eta = min(0.30, eta * 1.08)
            else:
                no_improv += 1
                eta *= 0.975
        if no_improv > 300:
            scale = 0.015 + 0.05 * min(1.0, restarts / 10.0)
            noise = rng.standard_normal(n)
            if sym:
                noise = 0.5 * (noise + noise[::-1])
            f = best_f * np.exp(scale * noise)
            f = np.maximum(f, 1e-20)
            if sym:
                f = 0.5 * (f + f[::-1])
            f /= f.sum()
            no_improv = 0
            restarts += 1
            eta = 0.08 + 0.03 * (restarts % 4)
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p = p / p.sum()
    x = np.arange(n, dtype=np.float64)
    inits = [p.copy()]
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.3, 0.7, 1.5]:
        inits.append(d ** pw / (d ** pw).sum())
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.02
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())
    for sig in [0.1, 0.2, 0.35]:
        gi = np.exp(-0.5 * ((x - (n - 1) / 2.0) / (sig * n)) ** 2)
        inits.append(gi / gi.sum())
    best_f = inits[0].copy()
    best_sc = get_score(inits[0])
    for init in inits[1:]:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    ph1_end = min(deadline, t0 + 480.0)
    f1, s1 = optimize(best_f.copy(), ph1_end, rng, sym=True)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()
    if time.monotonic() < deadline - 220.0:
        alt = inits[1].copy()
        f2, s2 = optimize(alt, min(deadline, t0 + 720.0), rng, sym=True)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()
    if time.monotonic() < deadline - 50.0:
        f3, s3 = optimize(best_f.copy(), deadline - 15.0, rng, sym=True)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()
    if time.monotonic() < deadline - 3.0:
        for _ in range(300):
            if time.monotonic() > deadline - 1.0:
                break
            trial = best_f.copy()
            trial *= np.exp(0.003 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial = 0.5 * (trial + trial[::-1])
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
