import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]


def eval_local(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-30:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)


def subgrad_weighted(f, w):
    n = f.shape[0]
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]
