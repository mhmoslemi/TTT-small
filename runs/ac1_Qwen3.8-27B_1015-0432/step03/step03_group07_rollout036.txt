import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def grad_gauss(f, b, beta):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.exp(-beta * np.maximum(M - b, 0.0) / M)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(L)
        w[np.argmax(b)] = 1.0
    else:
        w /= ws
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
