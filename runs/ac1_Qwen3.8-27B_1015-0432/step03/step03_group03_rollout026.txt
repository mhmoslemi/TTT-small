import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def compute_score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max() / (s * s)

def norm1(f):
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.size
    return f / s

def smooth_grad(f, b, tau):
    n = f.size
    L = 2 * n - 1
    shifted = b - b.max()
    w = np.exp(tau * shifted)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(L)
        w[np.argmax(b)] = 1.0
    else:
        w /= wsum
    h = np.zeros(L)
    h[:] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    n = f.size
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
