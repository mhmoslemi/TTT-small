    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv(a)
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    mv = cm + tau * np.log(s)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    mv = cm + tau * np.log(s)
    p = e / s
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
