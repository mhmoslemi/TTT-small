def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    s = f.sum()
    if s < 1e-10:
        return 1e30
    size = 1 << (2 * f.shape[0] - 2).bit_length()
    F = np.fft.rfft(f, size)
    b = np.fft.irfft(F * F, size)[:2 * f.shape[0] - 1]
    return float(2.0 * f.shape[0] * b.max() / (s * s))
