        inits.append(v / v.sum())
    for sg in [0.2, 0.35, 0.5, 0.7]:
        v = np.exp(-0.5 * ((x - 0.5) / sg) ** 2)
        inits.append(v / v.sum())
    for k in [1, 2, 3]:
        v = np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * k * x), 0.05)
        inits.append(v / v.sum())
    for a in [0.15, 0.35]:
        r = 1.0 + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        inits.append(r / r.sum())

    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()

    phase1_end = min(deadline, t0 + 320.0)
    f1, s1 = optimize(best_f.copy(), phase1_end, rng, max_peaks=220, eta_init=0.14)
    if s1 < best_sc:
        best_sc, best_f = s1, f1

    if time.monotonic() < deadline - 160.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        phase2_end = min(deadline, t0 + 620.0)
        f2, s2 = optimize(best_f.copy(), phase2_end, rng, max_peaks=380, eta_init=0.08)
        if s2 < best_sc:
            best_sc, best_f = s2, f2

    if time.monotonic() < deadline - 110.0:
        for restart in range(5):
            if time.monotonic() > deadline - 55.0:
                break
            sc_r = 0.015 + 0.012 * restart
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            f_r, s_r = optimize(f_r, deadline - 70.0, rng, max_peaks=350, eta_init=0.09)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r

    if time.monotonic() < deadline - 12.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 4.0, rng)

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
        final_sc = evaluate_sequence(result)
    return result
