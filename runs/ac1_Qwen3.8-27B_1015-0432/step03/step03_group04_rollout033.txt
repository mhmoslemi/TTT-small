    g = 2.0 * c[n - 1:2 * n - 1][::-1] / len(peaks)
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False):
    n = f.shape[0]
    f = np.maximum(f, 1e-20).astype(np.float64).copy()
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.06
    no_improv = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        sc = 2.0 * n * M
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f.copy()
            no_improv = 0
        else:
            no_improv += 1
        tf = max(0.95, 1.0 - 0.05 * min(1.0, it / 50000.0))
        g = compute_grad(f, b, tf, 500)
        gmax = g.max()
        if gmax < 1e-28:
            break
        f = f * np.exp(-eta * g / gmax)
        np.maximum(f, 1e-20, out=f)
        if sym:
            f += f[::-1]
            f *= 0.5
        f /= f.sum()
        it += 1
        if no_improv > 400:
            scale = 0.02 + 0.12 * min(1.0, no_improv / 2500.0)
            f = best_f * np.exp(scale * rng.standard_normal(n))
            np.maximum(f, 1e-20, out=f)
            if sym:
                f += f[::-1]
