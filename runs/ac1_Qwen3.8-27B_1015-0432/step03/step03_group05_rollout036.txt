def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subg(f, pk):
    n = f.shape[0]
    h = np.zeros(2 * n - 1)
    h[pk] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]
