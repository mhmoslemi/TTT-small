Delta (a*a)[k] ≈ 2*a[i]*(a[k-i] - a[i]) + 2*delta*a[k-i] (for small delta)

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    if size < 2:
        size = 2
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def run_opt(f, deadline, rng, max_active=256, eta_start=0.12):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = 2.0 * n * fft_conv(f, f).max()
    eta = eta_start
    patience = 0
    it = 0
    L = 2 * n - 1
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 8000.0)
        alpha = 0.10 + 0.05 * prog
        thresh = M * (1.0 - alpha)
        h = np.clip(b - thresh, 0.0, None)
        hsum = h.sum()
        if hsum < 1e-30:
            h = np.zeros(L)
            h[np.argmax(b)] = 1.0
        else:
            h /= hsum
        nz = (h > 0).sum()
        if nz > max_active:
            idx = np.where(h > 0)[0]
            top = idx[np.argpartition(h[idx], -max_active)[-max_active:]]
            h2 = np.zeros(L)
            h2[top] = 1.0
            h = h2
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        s = f.sum()
        f /= s
        it += 1
        if it % 20 == 0:
            sc = 2.0 * n * fft_conv(f, f).max() / (s * s)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
        if it > 200:
            eta *= 0.9996
        if patience > 50:
            nz2 = 0.02 + 0.03 * min(1.0, patience / 300.0)
            f = best_f * np.exp(nz2 * rng.standard_normal(n))
            np.maximum(f, 1e-30, out=f)
            f /= f.sum()
            patience = 0
            eta = 0.10
    return best_f, best_sc

def local_polish(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    step = 0.004
    iterations = 0
    while time.monotonic() < deadline and step > 1e-5:
        for _ in range(12):
            if time.monotonic() > deadline:
                break
            idx = rng.integers(0, n, size=max(1, n // 200))
            d = np.zeros(n)
            d[idx] = rng.standard_normal(len(idx))
            trial = best_f + step * d
            np.maximum(trial, 0.0, out=trial)
            if trial.sum() < 1e-10:
                continue
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = trial.copy()
                iterations = 0
            else:
                iterations += 1
        if iterations > 30:
            step *= 0.85
            iterations = 0
    return best_f, best_sc
