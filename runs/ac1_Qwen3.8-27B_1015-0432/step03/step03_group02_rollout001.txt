def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    c = np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))
    return c[:n + m - 1]
