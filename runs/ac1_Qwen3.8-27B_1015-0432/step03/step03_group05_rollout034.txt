            bs = sc
            bf = tr.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    for w in (0.05, 0.15, 0.3, 0.5, 0.8):
        v = np.exp(-0.5 * ((x - c) / (w * n)) ** 2)
        inits.append(v / v.sum())
    for pw in (0.3, 1.0, 2.0, 5.0):
        v = np.maximum(1.0 - np.abs(x - c) / (c + 1), 0.005) ** pw
        inits.append(v / v.sum())
    d = np.minimum(x, n - 1 - x).astype(float) + 1.0
    for pw in (0.5, 2.0):
        inits.append(d ** pw / (d ** pw).sum())
    r1 = 1.0 - x / max(n - 1, 1) + 0.01
    inits.append(r1 / r1.sum())
    r2 = x / max(n - 1, 1) + 0.01
    inits.append(r2 / r2.sum())
    q = (x / (n - 1)) ** 2 + 0.01
    inits.append(q / q.sum())
    q2 = (1.0 - x / (n - 1)) ** 2 + 0.01
    inits.append(q2 / q2.sum())
    vw = np.maximum(np.cos(np.pi * (x - c) / (2 * c + 1)), 0) ** 2
    vw = np.maximum(vw, 1e-8)
    inits.append(vw / vw.sum())
    rv = np.maximum(rng.standard_normal(n), 0.01)
    inits.append(rv / rv.sum())

    bf = None
    bs = np.inf
    for ini in inits:
        sc = score(ini)
        if sc < bs:
            bs = sc
            bf = ini.copy()

    p1 = min(dl, t0 + 380)
    f1, s1 = optimize(bf, p1, rng, sym=False, eta0=0.05)
    if s1 < bs:
        bs, bf = s1, f1

    if time.monotonic() < dl - 220:
        fs = 0.5 * (bf + bf[::-1])
        fs = np.maximum(fs, 0)
        ss = fs.sum()
        if ss > 1e-15:
            fs /= ss
        p2 = min(dl, t0 + 700)
        f2, s2 = optimize(fs, p2, rng, sym=True, eta0=0.03)
        if s2 < bs:
            bs, bf = s2, f2

    if time.monotonic() < dl - 100:
        f3, s3 = optimize(bf.copy(), min(dl, t0 + 960), rng, sym=False, eta0=0.015)
        if s3 < bs:
            bs, bf = s3, f3

    if time.monotonic() < dl - 25:
        f4, s4 = optimize(bf.copy(), dl - 10, rng, sym=False, eta0=0.004, max_it=200000)
        if s4 < bs:
            bs, bf = s4, f4

    if time.monotonic() < dl - 3:
        bf, bs = perturb_best(bf, bs, rng, dl - 2, count=60, amp=0.003)

    result = [float(v) for v in bf]
    fs8 = evaluate_sequence(result)
    if not np.isfinite(fs8):
        result = [1.0 / n] * n
    return result
