    m = 1

    L = _fpow2(2 * n - 1)

    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    L = _fp2(3 * n - 2)
    fc = ifft(fft(a, L) * fft(p[::-1], L)).real
    g = 2.0 * fc[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _norm(a, sc):
    s = a.sum()
    if s < 1e-8:
        return a.max() / a.sum() * sc if a.sum() > 0 else np.ones_like(a)
    return a / s * sc


def _upd(cand, best, bsc):
    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < bsc:
        best[:] = cand
        bsc = sc
    return bsc


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, budget_s - 18.0)
    try:
        par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    except Exception:
        par = np.ones(256)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)) or par.sum() < 1e-8:
        par = np.ones(512)
    n = par.size
    sc_ = np.sqrt(2.0 * n)
    a0 = _norm(par, sc_)
    best = a0.copy()
    bsc = float(evaluate_sequence(best.tolist()))
    x = np.arange(n, dtype=np.float64)
    inits = [a0]
    inits.append(np.ones(n) * sc_ / n)
    for amp in (0.3, 0.6, 0.9):
        for fr in (1.0, 3.0, 7.0):
            c = 1.0 + amp * np.cos(2.0 * np.pi * fr * x / n)
            c = np.clip(c, 0.0, None)
            if c.sum() > 0.01:
                inits.append(_norm(c, sc_))
    for sig in (0.08, 0.2, 0.4, 0.7, 1.2):
        for ctr in (0.35, 0.5, 0.7, 1.1):
            g = np.exp(-0.5 * ((x - ctr * n) / (sig * n)) ** 2)
            if g.sum() > 0.01:
                inits.append(_norm(g, sc_))
    for fr in (0.4, 0.7, 0.9):
        bx = np.zeros(n)
        s = max(0, int(n * (1.0 - fr) / 2.0))
        e = min(n, int(n * (1.0 + fr) / 2.0))
        bx[s:e] = 1.0
        if bx.sum() > 0.01:
            inits.append(_norm(bx, sc_))
    inits.append(_norm(np.clip(1.0 - np.abs(x - 0.4 * n) / (0.4 * n), 0.0, None), sc_))
    inits.append(_norm(np.clip(1.0 - (x / (n - 1 + 1e-9)) ** 2, 0.0, None), sc_))
    for _ in range(3):
        r = rng.uniform(0.0, 1.0, n)
        r = np.convolve(r, np.ones(13) / 13, mode='same')
        r = np.clip(r, 0.0, None)
        if r.sum() > 0.01:
            inits.append(_norm(r, sc_))
    bounds = [(0.0, 1000.0)] * n
    cur = a0.copy()
    for inp in inits:
        sc = float(evaluate_sequence(inp.tolist()))
        if np.isfinite(sc) and sc < bsc:
            bsc = sc
            best[:] = inp
            cur[:] = inp
    taus = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15,
            0.1, 0.07, 0.05, 0.03, 0.02, 0.012, 0.008, 0.005, 0.003, 0.0015]
    for tv in taus:
        if time.monotonic() > dl - 120.0:
            break
        rem = dl - time.monotonic()
        it = min(900, max(20, int(rem * 12)))
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': it, 'gtol': 1e-16,
                                    'maxls': 50, 'ftol': 1e-20})
            nx = np.clip(res.x, 0.0, 1000.0)
            if nx.sum() > 0.01:
                _upd(_norm(nx, sc_), best, bsc)
                cur = np.clip(res.x, 0.0, 1000.0)
        except Exception:
            pass
    if time.monotonic() < dl - 40.0:
        sym = 0.5 * (cur + cur[::-1])
        s2 = _norm(np.clip(sym, 0.0, 1000.0), sc_)
        bsc = _upd(s2, best, bsc)
        if np.isfinite(float(evaluate_sequence(s2.tolist()))):
            cur[:] = s2
    if time.monotonic() < dl - 20.0:
        wmx = max(3, n // 20)
        for _ in range(2000):
            if time.monotonic() > dl - 20.0:
                break
            cd = cur.copy()
            w1 = int(rng.integers(1, wmx + 1))
            s1 = int(rng.integers(0, max(1, n - w1)))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.05, 0.06)))
            w2 = int(rng.integers(1, wmx + 1))
            s2 = int(rng.integers(0, max(1, n - w2)))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.03, 0.04)))

    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs) or fs >= 2.0:
        a0 = par / par.sum() * sc_
        a0 = np.clip(a0, 0.0, 1000.0)
        return [float(v) for v in a0]
    return [float(v) for v in best]
