def fft_conv(a, b):

    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n+m-1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-30:
        return np.inf
    return float(2.0 * f.size * b.max() / (s * s))

def grad_peak(f, b, top_k, sharpness):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    idx = np.argpartition(b, L - top_k)[-top_k:]
    w = np.zeros(L)
    for k in idx:
        w[k] = (b[k] / M) ** sharpness
    if w.sum() < 1e-30:
        w[int(np.argmax(b))] = 1.0
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n-1:2*n-1][::-1]
    return np.maximum(g, 0.0)

def mm_run(f, deadline, rng, sym=False, eta0=0.12, max_ni=250):
    n = f.shape[0]
    f = np.maximum(f.astype(np.float64).copy(), 0.0)
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f /= s
    best_f = f.copy()
    best_sc = nscore(f)
    eta = eta0
    no_improv = 0
    rc = 0
    for it in range(300000):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        prog = min(1.0, it / 3000.0)
        tk = max(3, int(12 - 9 * prog))
        sharp = 6 + 30 * prog
        g = grad_peak(f, b, tk, sharp)
        gmax = g.max()
        if gmax < 1e-30:
            break
        ratio = np.minimum(eta * g / gmax, 2.5)
        f *= np.exp(-ratio)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 12 == 0:
            sc = nscore(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv >= max_ni:
                amp = 0.012 + 0.01 * min(1.0, rc / 6.0)
                r = rng.integers(3)
                if r == 0:
                    pert = rng.standard_normal(n)
                elif r == 1:
                    Ff = np.fft.rfft(best_f)
                    Ff[1:-1] *= 1.0 + 0.3 * rng.standard_normal(len(Ff) - 2)
                    pert = np.fft.irfft(Ff, n)
                    pert -= pert.mean()
                else:
                    x = np.arange(n, dtype=np.float64)
                    pert = rng.standard_normal(n) * np.sin(np.pi * x / n)
                f = best_f * np.exp(amp * pert / (np.abs(pert).max() + 1e-10))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                rc += 1
                eta = min(0.35, eta0 * (1.13 ** rc))
        eta *= 0.9993
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 0.01:
        p = np.ones(n)
    best_f = p.copy()
    best_sc = nscore(p)
    # Phase 1: main optimization from parent
    dl1 = min(deadline, t0 + 420.0)
    f1, sc1 = mm_run(p.copy(), dl1, rng)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    # Phase 2: symmetric refinement
    if time.monotonic() < deadline - 280:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        scs = nscore(fs)
        if scs < best_sc:
            best_sc, best_f = scs, fs.copy()
        dl2 = min(deadline, t0 + 750.0)
        f2, sc2 = mm_run(fs.copy(), dl2, rng, sym=True)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2.copy()
    # Phase 3: continue from best
    if time.monotonic() < deadline - 80:
        f3, sc3 = mm_run(best_f.copy(), deadline - 10.0, rng)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()
    # Phase 4: fine random polish
    if time.monotonic() < deadline - 5:
        for _ in range(400):
            if time.monotonic() > deadline - 3:
                break
            for amp in [0.001, 0.004, 0.01]:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                sc = nscore(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in np.maximum(best_f, 0.0)]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs) or len(result) == 0:
        result = [1.0 / n] * n
    return result
