import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def extract_cross(c, n):
    return c[n - 1:2 * n - 1][::-1].copy()

def run_opt(f, deadline, rng, eta_init=0.12, check_iv=20, mode="soft"):
    n = f.shape[0]

    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    eta = eta_init
    patience = 0
    it = 0
    L = 2 * n - 1
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        bmin = b.min()
        if M < 1e-30:
            break
        prog = min(1.0, it / 12000.0)
        if mode == "soft":
            frac = 0.998 - 0.03 * prog
            thresh = M * frac
            h = np.clip((b - thresh) / (M - thresh + 1e-30), 0.0, 1.0)
        else:
            frac = 0.995 - 0.04 * prog
            thresh = M * frac
            active = b >= thresh
            cnt = active.sum()
            if cnt > 400:
                idx = np.argpartition(b, -400)[-400:]
                h = np.zeros(L)
                h[idx] = 1.0
            else:
                h = active.astype(np.float64)
        c_pen = fft_conv(f, h[::-1])
        g_pen = extract_cross(c_pen, n)
        np.maximum(g_pen, 0.0, out=g_pen)
        if it % 4 == 0:
            vw = 0.15 * (1.0 - prog * 0.5)
            vthresh = bmin + 0.08 * (M - bmin)

        gmax = g_pen.max() + g_rew.max() + 1e-30
        f *= np.exp(-eta * (g_pen - g_rew) / gmax)

                best_f = f.copy()

                f /= f.sum()
                patience = 0
                eta = min(0.28, eta * 1.18)
        if it % 200 == 0:
            eta *= 0.9985
    return best_f, best_sc
