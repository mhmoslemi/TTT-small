                                   'maxls': 30, 'ftol': 1e-22})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _norm(nx, sc_)
            sc = float(evaluate_sequence(nx.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = nx.copy()
                best_sc = sc
            cur = nx.copy()
        except Exception:
            pass

    # Peak-targeted reduction
    if time.monotonic() < dl - 50.0:
        n_red = max(3, n // 80)
        for _rnd in range(300):
            if time.monotonic() > dl - 50.0:
                break
            c = _conv(cur)
            top_lags = np.argsort(c)[-3:][::-1]
            improved = False
            for lag in top_lags:
                lag = int(lag)
                cd = cur.copy()
                lo = max(0, lag - (n - 1))
                hi = min(n - 1, lag)
                idx1 = np.arange(lo, hi + 1)
                idx2 = lag - idx1
                contrib = cd[idx1] * cd[idx2]
                order = np.argsort(contrib)[-n_red:]
                for idx in order:
                    cd[idx] *= 0.92
                cd = np.clip(cd, 0.0, 1000.0)
                cd = _norm(cd, sc_)
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
                    improved = True
                    break
            if not improved and rng.random() < 0.3:
                cur = best.copy()

    # Random block perturbation search
    if time.monotonic() < dl - 25.0:
        wmax = max(2, n // 30)
        while time.monotonic() < dl - 25.0:
            cd = cur.copy()
            nb = int(rng.integers(1, 4))
            for _ in range(nb):
                w = int(rng.integers(1, wmax + 1))
                s = int(rng.integers(0, n - w))
                cd[s:s + w] *= float(np.exp(rng.normal(-0.03, 0.04)))
            if rng.random() < 0.06:
                cd *= float(np.exp(rng.normal(0.0, 0.01)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.04:
                cur = best.copy()

    # Final L-BFGS-B polish at very low temperature
    if time.monotonic() < dl - 4.0:
        cur = best.copy()
        for tv in (4e-4, 2e-4, 1e-4):
            if time.monotonic() > dl - 4.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                              method='L-BFGS-B', bounds=bounds_list,
                              options={'maxiter': 600, 'gtol': 1e-16,
                                       'maxls': 25, 'ftol': 1e-24})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _norm(nx, sc_)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs):
        best = a0
    return [float(v) for v in best]
