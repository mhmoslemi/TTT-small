```python
def project_simplex(v, s=1.0):
    # Project v onto {x: sum(x)=s, x>=0}
    n = len(v)
    v = np.clip(v, 0, None)
    if v.sum() == 0:
        return np.ones(n) / n * s
    # Find threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def simplex_project(v, s=1.0):
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    idx = np.where(cond)[0][-1]
    theta = (cssv[idx] - s) / (idx + 1)
    return np.maximum(v - theta, 0.0)

def max_conv(a):
    return fft_conv(a, a).max()

def compute_subgrad(a, c, tol_frac):
    n = len(a)
    M = c.max()
    if M < 1e-30:
        return np.zeros(n)
    peaks = np.where(c >= M * tol_frac)[0]
    w = c[peaks]
    w = w / w.sum()
    w_full = np.zeros(2 * n - 1)
    w_full[peaks] = w
    r = a[::-1]
    G = 2.0 * fft_conv(r, w_full)[n - 1:n - 1 + n]
    return G

def run_mirror(a, deadline, eta0=0.12, tol_frac=0.995, check_every=10, max_iter=500000):
    n = len(a)
    a = simplex_project(a.copy().astype(np.float64), 1.0)
    best_a = a.copy()
    best_mc = max_conv(a)
    no_improve = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = fft_conv(a, a)
        M = c.max()
        G = compute_subgrad(a, c, tol_frac)
        Gmax = G.max()
        if Gmax < 1e-25:
            break
        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * G / Gmax)
        a = simplex_project(a, 1.0)
        if (it + 1) % check_every == 0:
            mc = max_conv(a)
            if mc < best_mc - 1e-14:
                best_mc = mc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 3000:
                break
    return best_a, best_mc

def run_proj(a, deadline, eta0=0.08, tol_frac=0.999, check_every=8, max_iter=500000):
    n = len(a)
    a = simplex_project(a.copy().astype(np.float64), 1.0)
    best_a = a.copy()
    best_mc = max_conv(a)
    no_improve = 0
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = fft_conv(a, a)
        M = c.max()
        G = compute_subgrad(a, c, tol_frac)
        Gmax = G.max()
        if Gmax < 1e-25:
            break
        eta = eta0 / np.sqrt(it + 1)
        a = simplex_project(a - eta * G / Gmax, 1.0)
        if (it + 1) % check_every == 0:
            mc = max_conv(a)
            if mc < best_mc - 1e-14:
                best_mc = mc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 3000:
                break
    return best_a, best_mc
```
