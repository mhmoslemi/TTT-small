import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def sym(f):
    return 0.5 * (f + f[::-1])

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]

    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.clip(f, 0.0, None)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def sym(f):
    return 0.5 * (f + f[::-1])

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, b, frac, n_peaks):
    n = f.shape[0]
    M = b.max()
    peaks = np.where(b >= M * frac)[0]
    if len(peaks) > n_peaks:
        idx = np.argsort(b)[::-1]
        peaks = idx[:n_peaks]
    if len(peaks) < 1:
        peaks = np.array([np.argmax(b)])
    vals = b[peaks] / M
    w = vals ** 4
    w /= w.sum()
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0), M
