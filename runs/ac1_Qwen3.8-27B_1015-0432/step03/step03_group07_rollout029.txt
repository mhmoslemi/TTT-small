           lo = max(0, x - n + 1)
           hi = min(n - 1, x)
           g[lo:hi+1] += f[x-lo:x-hi+1][::-1]
       if g.max() > 0:
           return 2.0 * g / len(peaks)
       g[int(np.argmax(b))] = 1.0
       return g

   def optimize(f, deadline, rng, use_lp=True, max_lp_iters=300):
       n = f.size
       f = np.maximum(f.copy(), 0.0)
       f /= f.sum()
       best_f = f.copy()
       best_sc = score(f)
       it = 0
       while time.monotonic() < deadline and it < max_lp_iters:
           if use_lp:
               b = fft_conv(f, f)
               M = b.max()
               g = lp_dir(f, M)
               if g is None:
                   g = grad_dir(f, b)
                   f_new = np.maximum(f - 0.01 * g / (g.max() + 1e-15), 0)
                   f_new /= f_new.sum()
                   f = f_new
               else:
                   improved = False
                   for t in (1.0, 0.7, 0.5, 0.3, 0.1):
                       f_new = (1-t)*f + t*g
                       f_new = np.maximum(f_new, 0)
                       f_new /= f_new.sum()
                       sc = score(f_new)
                       if sc < best_sc * (1 + 1e-12):
                           best_sc = sc
                           best_f = f_new.copy()
                           f = f_new
                           improved = True
                           break
                   if not improved:
                       # Try gradient step
                       b = fft_conv(f, f)
                       g2 = grad_dir(f, b, 0.995)
                       f_new = np.maximum(f - 0.005 * g2 / (g2.max() + 1e-15), 0)
                       f_new /= f_new.sum()
                       sc = score(f_new)
                       if sc < best_sc:
                           best_sc = sc
                           best_f = f_new.copy()
               it += 1
           else:
               # Gradient-only mode
               b = fft_conv(f, f)
               g = grad_dir(f, b, 0.99)
               gmax = g.max()
               if gmax < 1e-30:
                   break
               f = np.maximum(f - 0.01 * g / gmax, 0)
               f /= f.sum()
               sc = score(f)
               if sc < best_sc:
                   best_sc = sc
                   best_f = f.copy()
               it += 1
       return best_f, best_sc

       rng = np.random.default_rng(seed)
       t0 = time.monotonic()

       p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
       
       # Generate initializations
       inits = []
       if p.sum() > 1e-10:
           inits.append(p / p.sum())
       inits.append(np.ones(n) / n)
       x = np.arange(n, dtype=np.float64)
       shapes = [0.5 * (1 + np.cos(2*np.pi*x/n)),
                 0.5 * (1 - np.cos(2*np.pi*x/(n+1)))]
       d = np.maximum(np.minimum(x, n-1-x), 0.5)
       for pw in (0.3, 0.7, 1.0, 1.5):
           shapes.append(d ** pw)
       inits += [s/s.sum() for s in shapes]
       
       # Pick best init
       best_f, best_sc = None, np.inf
       for init in inits:
           sc = score(init)
           if sc < best_sc:
               best_sc = sc
               best_f = init.copy()
