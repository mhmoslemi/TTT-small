```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, w):
    n = f.shape[0]
    L = 2 * n - 1
    sz = 1 << (n + L - 2).bit_length()
    F = np.fft.rfft(f, sz)
    W = np.fft.rfft(w, sz)
    h = np.fft.irfft(F * np.conj(W), sz)
    B = np.fft.irfft(W * F, sz)
    return h[:n] + B[:n]

def optimize(f, deadline, rng, sym=False, max_it=500000, eta0=0.04):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        # Soft-max weights
        frac = 0.999 - 0.006 * min(1.0, it / 40000.0)
        thr = M * frac
        pk = np.where(b >= thr)[0]
        if len(pk) < 3:
            pk = np.argsort(b)[-30:]
        elif len(pk) > 800:
            pk = np.argsort(b)[-800:]
        tau = M * 0.05
        if tau < 1e-10:
            tau = 1e-10
        w = np.exp((b - M) / tau)
        wsum = w.sum()
        w /= wsum
        g = subgrad(f, w)
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it %
