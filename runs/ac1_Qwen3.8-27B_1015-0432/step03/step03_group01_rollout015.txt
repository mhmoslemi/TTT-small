def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2*n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2*n-1]

def _smax_grad(c, tau):
    """Returns (smooth_max, gradient_of_smooth_max w.r.t. c)"""
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    p = e / s
    return cm + tau * np.log(s), p
