import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def compute_grad_weighted(f, w):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[:] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def compute_grad_peaks(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False, eta_init=0.08, momentum=0.5, peak_cap=600):
    n = f.shape[0]
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    eta = eta_init
    no_improv = 0
    it = 0
    g_smooth = None
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        beta = 3.0 + 12.0 * min(1.0, it / 3000.0)
        shifted = b - M
        w = np.exp(beta * shifted)
        wsum = w.sum()
        if wsum < 1e-30:
            idx = np.argwhere(b == M).ravel()
            g = compute_grad_peaks(f, idx)
        else:
            w /= wsum
            nz = np.where(w > 1e-12)[0]
            if len(nz) > peak_cap:
                top = np.argsort(w)[::-1][:peak_cap]
                w2 = np.zeros_like(w)
                w2[top] = w[top]
                w2 /= w2.sum()
                w = w2
            g = compute_grad_weighted(f, w)
        gmax = g.max()
        if gmax < 1e-30:
            break
        gn = g / gmax
        if g_smooth is None:
            g_smooth = gn.copy()
        else:
            g_smooth = momentum * g_smooth + (1.0 - momentum) * gn
            gmax_s = g_smooth.max()
            if gmax_s > 1e-30:
                gn = g_smooth / gmax_s
        best_step_sc = np.inf
        best_tr = f.copy()
        best_eta = eta
        for te in [eta * 0.4, eta * 0.7, eta, eta * 1.4, eta * 2.0, eta * 3.5]:
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-25)
            if sym:
                tr = 0.5 * (tr + tr[::-1])
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_step_sc:
                best_step_sc = sc
                best_eta = te
                best_tr = tr
        f = best_tr
        if best_eta > eta * 1.15:
            eta = min(0.5, eta * 1.25)
        elif best_eta < eta * 0.85:
            eta = max(0.005, eta * 0.65)
        if it % 6 == 0:
            sc = score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:

            if no_improv > 150:
                scl = 0.02 + 0.05 * min(1.0, no_improv / 300.0)
                f = best_f * np.exp(scl * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = normalize(f)
                g_smooth = None
                no_improv = 0
                eta = eta_init
    return best_f, best_sc
