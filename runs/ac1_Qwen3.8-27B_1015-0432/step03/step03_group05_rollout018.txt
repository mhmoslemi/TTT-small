    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, eta0=0.03, max_it=300000):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        frac = 0.9995 - 0.003 * min(1.0, it / 60000.0)
        thr = M * frac
        pk = np.where(b >= thr)[0]
        if len(pk) < 5:
            pk = np.argsort(b)[-50:]
        elif len(pk) > 1200:
            pk = np.argsort(b)[-1200:]
        g = grad_peaks(f, pk)
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        if it % 10 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 200:
                amp = 0.006 + 0.02 * min(1.0, ni / 1500.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f = 0.96 * f + 0.04 / n
                f /= f.sum()
                ni = 0
                eta = eta0
        if it > 20000:
            eta *= 0.99997
        eta = max(eta, eta0 * 0.06)
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def polish(f, deadline, rng, count=400):
    n = f.shape[0]
    bf = f.copy()
    bs = score(f)
    amp = 0.0015
    for i in range(count):
        if time.monotonic() > deadline:
            break
        tr = bf * np.exp(amp * rng.standard_normal(n))
        tr = np.maximum(tr, 0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = score(tr)
        if sc < bs:
            bs = sc
            bf = tr.copy()
    return bf, bs

def peak_target(f, deadline, rng, count=300):
    """Target only the single max peak with tiny steps."""
    n = f.shape[0]
    bf = f.copy()
    bs = score(f)
    for i in range(count):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        pk = np.array([np.argmax(b)])
        g = grad_peaks(f, pk)
        gm = g.max()
        if gm < 1e-30:
            break
        eta = 0.003
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        sc = score(f)
        if sc < bs:
            bs = sc
            bf = f.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 15.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    ramp = (x + 1.0)
    inits.append(ramp / ramp.sum())
    iramp = (n - x + 1.0)
    inits.append(iramp / iramp.sum())
    expo = np.exp(-0.8 * x / n)
    inits.append(expo / expo.sum())
    expo2 = np.exp(-0.8 * (n - 1 - x) / n)
    inits.append(expo2 / expo2.sum())
    edge = np.minimum(x, n - 1 - x).astype(float) + 1.0
    inits.append(edge / edge.sum())
    rv = np.maximum(rng.standard_normal(n), 0.02)
    inits.append(rv / rv.sum())

    bf = None
    bs = np.inf
    for ini in inits:
        sc = score(ini)
        if sc < bs:
            bs = sc
            bf = ini.copy()

    # Phase 1: Main asymmetric optimization (use parent, no symmetrize)
    p1 = min(dl, t0 + 450)
    f1, s1 = optimize(bf, p1, rng, eta0=0.035)
    if s1 < bs:
        bs, bf = s1, f1

    # Phase 2: Second asymmetric pass with different eta
    if time.monotonic() < dl - 200:
        p2 = min(dl, t0 + 780)
        f2, s2 = optimize(bf.copy(), p2, rng, eta0=0.02)
        if s2 < bs:
            bs, bf = s2, f2

    # Phase 3: Fine asymmetric refinement
    if time.monotonic() < dl - 100:
        p3 = min(dl, t0 + 960)
        f3, s3 = optimize(bf.copy(), p3, rng, eta0=0.01)
        if s3 < bs:
            bs, bf = s3, f3

    # Phase 4: Peak-targeting (single-peak gradient)
    if time.monotonic() < dl - 40:
        f4, s4 = peak_target(bf.copy(), dl - 20, rng, count=200)
        if s4 < bs:
            bs, bf = s4, f4

    # Final polish: tiny random perturbations
    if time.monotonic() < dl - 2:
        bf, bs = polish(bf, dl - 1, rng, count=500)

    result = [float(v) for v in bf]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
