def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_fn(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * float(f.size) * float(b.max()) / (s * s)

def make_grad(f, b, temp, max_act):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    thr = M * temp
    w = np.maximum(b - thr, 0.0)
    scale = M * (1.0 - temp) + 1e-30
    w = (w / scale) ** 2
    if w.sum() < 1e-20:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    nz = np.count_nonzero(w > 1e-15)

        w = w2
    c = fft_conv(f, w[::-1])
