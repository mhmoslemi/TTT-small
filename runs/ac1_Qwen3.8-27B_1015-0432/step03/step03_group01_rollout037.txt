            inits.append(_norm(pl, scale))
    # Two-bump asymmetric
    for r1 in (0.25, 0.4):
        for r2 in (0.6, 0.75):
            c = (np.exp(-((x - r1 * n) / (0.08 * n)) ** 2)
                 + 0.6 * np.exp(-((x - r2 * n) / (0.1 * n)) ** 2))
            inits.append(_norm(c, scale))
    # Random
    for _ in range(3):
        r = np.abs(rng.standard_normal(n))
        r = r / r.max()
        inits.append(_norm(r, scale))

    cur = a0.copy()
    for inp in inits:
        sc = _score(np.clip(inp, 0.0, 1000.0))
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best = np.clip(inp, 0.0, 1000.0).copy()
            cur = best.copy()

    # --- L-BFGS-B annealed optimization ---
    bl = [(0.0, 1000.0)] * n
    taus = [8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005,
            0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    for i, tv in enumerate(taus):
        if time.monotonic() > dl - 80.0:
            break
        rem = dl - time.monotonic()
        iters = min(1500, max(50, int(rem * 12)))
        if iters < 30:
            break
        try:
            res = minimize(_obj, cur, args=(tv,), jac=_grad,
                           method='L-BFGS-B', bounds=bl,
                           options={'maxiter': iters, 'gtol': 1e-15,
                                    'maxls': 50, 'ftol': 1e-25})
            nx = np.clip(res.x, 0.0, 1000.0)
            nx = _norm(nx, scale)
            sc = _score(nx)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = nx.copy()
            cur = nx.copy()
        except Exception:
            pass

    # --- Peak-leveling local search ---
    wmax = max(3, n // 20)
    for _ in range(1200):
        if time.monotonic() > dl - 20.0:
            break
        cd = cur.copy()
        c = _conv(cd)
        # Find top-3 peak lags
        topk = np.argpartition(c, -4)[-4:]
        pk = int(topk[np.argmax(c[topk])])
        # Reduce near peak contributor
        w1 = int(rng.integers(2, wmax + 1))
        ci = max(0, min(pk, n - 1))
        lo = max(0, ci - w1)
        hi = min(n - w1, ci)
        s1 = int(rng.integers(lo, hi + 1))
        cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.03, 0.04)))
        # Random perturbation elsewhere
        w2 = int(rng.integers(2, wmax + 1))
        s2 = int(rng.integers(0, n - w2))
        cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.03, 0.04)))
        if rng.random() < 0.1:
            cd *= float(np.exp(rng.normal(0.0, 0.01)))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            sc = _score(cd)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = cd.copy()
                cur = cd.copy()
        if rng.random() < 0.05:
            cur = best.copy()

    # --- Symmetrization (antipalindromic trick) ---
    if time.monotonic() < dl - 30.0:
        # Try making sequence more asymmetric to reduce center peak
        for shift in (0.3, 0.7):
            if time.monotonic() > dl - 30.0:
                break
            w = 0.5 * (1.0 + np.sin(2.0 * np.pi * shift * x / n))
            sv = np.clip(cur * w, 0.0, 1000.0)
            sv = _norm(sv, scale)
            sc = _score(sv)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = sv.copy()
                cur = sv.copy()

    # --- Final polish at ultra-low temperature ---
    if time.monotonic() < dl - 5.0:
        cur = best.copy()
        for tv in (1e-5, 5e-6):
            if time.monotonic() > dl - 5.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 2000, 'gtol': 1e-16,
                                        'maxls': 60, 'ftol': 1e-28})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _norm(nx, scale)
                sc = _score(nx)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best = nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    # --- Final verification ---
    final_sc = _score(best)
    if not np.isfinite(final_sc):
        best = np.ones(n) * (scale / n)
    return [float(max(0.0, min(1000.0, v))) for v in best]
