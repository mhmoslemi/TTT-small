        if it_eval:

            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
                # Reduce eta slightly on improvement
                eta *= 0.98
            else:
                no_improv += 1
                # Backtrack: revert if no improvement
                f = best_f.copy()
                eta *= 0.7
                if no_improv > 300:
                    # Restart
                    scale = 0.05 + 0.1 * min(1.0, restarts / 8.0)
                    f = best_f * np.exp(scale * rng.standard_normal(n))
                    f = np.maximum(f, 1e-20)
                    if sym:

                    f /= f.sum()
                    no_improv = 0
                    restarts += 1
                    eta = 0.05
        
        eta = max(0.001, eta * 0.9998)
    
    return best_f, best_sc
