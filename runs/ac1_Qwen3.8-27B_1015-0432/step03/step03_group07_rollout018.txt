def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    la = np.fft.rfft(a, size)
    lb = np.fft.rfft(b, size)
    return np.fft.irfft(la * lb)[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

    n = f.shape[0]

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def comp_grad(f, b, tf, ma=400):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    sp = max(1.0 - tf, 0.005)
    w = np.maximum(b - M * tf, 0.0)
    w = (w / (M * sp)) ** 2
    act = np.where(w > 1e-18)[0]
    if len(act) > ma:
        idx = act[np.argsort(w[act])[-ma:]]
        w2 = np.zeros(L)
        w2[idx] = w[idx]
        w = w2
    if w.sum() < 1e-30:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

                ni += 1
