- Compute q_k (exp and normalize)
- Compute gradient: 2 * (f * q_rev)[n-1:2n-1] (one more FFT conv)

So 2 FFT convolutions per iteration, same as before. The difference is in the weighting.

For the optimization, I'll use:
- Initially: larger T (smoother gradient, faster progress in the right direction)
- Later: smaller T (sharper gradient, converging to the true max)

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def compute_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def grad_soft(f, b, T):
    n = f.shape[0]
    L = 2 * n - 1
    # Softmax weights
    b_shifted = b - b.max()
    w = np.exp(b_shifted / T)
    w /= w.sum()
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n-1:2*n-1]

def run_opt(f, deadline, rng, sym=False, max_it=1000000):

    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = compute_score(f)
    ni = 0
    L = 2 * n - 1
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        # Temperature: start high, decrease
        T0 = M * 0.05
        T1 = M * 0.001
        T = T0 + (T1 - T0) * min(1.0, it / 50000.0)
        T = max(T, 1e-15)
        g = grad_soft(f, b, T)
        gm = g.max()
        if gm < 1e-30:
            break
        # Step size
        eta = 0.05 / (1.0 + it * 1e-6)
        f = f * np.exp(-eta * g / max(gm, 1e-30))
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
