def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def sc_of(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def wgrad(f, b, T):
    n = len(f)
    L = 2 * n - 1
    sb = b - b.max()
    w = np.exp(sb / T)
    tot = w.sum()
    if tot < 1e-30:
        pk = np.where(b >= b.max() * 0.999)[0]
        if len(pk) < 3:
            pk = np.argsort(b)[-20:]
        h = np.zeros(L)
        h[pk] = 1.0
        w = h
    else:

            bs = sc
            bf = tr.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    for pw in (0.5, 1.0, 2.0):
        inits.append(d ** pw)
    inits[-3] /= (d ** 0.5).sum()
    u = (x / max(n - 1, 1)) ** 2 + (1.0 - x / max(n - 1, 1)) ** 2 + 0.01
    inits.append(u)
    rv = np.maximum(rng.standard_normal(n), 0.01)
    inits.append(rv)
    c1 = np.exp(-((x - 0.25 * n) / (0.07 * n)) ** 2) + 0.01
    c2 = np.exp(-((x - 0.75 * n) / (0.07 * n)) ** 2) + 0.01
    inits.append(c1 + c2)
    chirp = np.abs(np.cos(np.pi * 4.0 * x / n + np.pi * 8.0 * (x / max(n, 1.0)) ** 2)) + 0.02
    inits.append(chirp)

    bf, bs = None, np.inf
    for ini in inits:
        ini = np.maximum(np.asarray(ini, dtype=np.float64), 0)
        s = ini.sum()
        if s < 1e-15:
            continue
        ini = ini / s
        sc = sc_of(ini)
        if sc < bs:
            bs = sc
            bf = ini.copy()

    if bf is None:
        bf = np.ones(n) / n
        bs = sc_of(bf)

    p1 = min(dl, t0 + 380)
    f1, s1 = run_opt(bf.copy(), p1, rng, eta0=0.035, sym=False)
    if s1 < bs:
        bs, bf = s1, f1

    if time.monotonic() < dl - 220:
        fs = 0.5 * (bf + bf[::-1])
        fs = np.maximum(fs, 0)
        fs /= fs.sum()
        p2 = min(dl, t0 + 680)
        f2, s2 = run_opt(fs, p2, rng, eta0=0.025, sym=True)
        if s2 < bs:
            bs, bf = s2, f2

    if time.monotonic() < dl - 160:
        p3 = min(dl, t0 + 920)
        f3, s3 = run_opt(bf.copy(), p3, rng, eta0=0.012, sym=False)
        if s3 < bs:
            bs, bf = s3, f3

    if time.monotonic() < dl - 35:
        f4, s4 = run_opt(bf.copy(), dl - 8, rng, eta0=0.004, max_it=400000)
        if s4 < bs:
            bs, bf = s4, f4

    bf, bs = perturb_local(bf, bs, rng, dl - 2, count=100, amp=0.002)

    result = [float(v) for v in bf]
    fs8 = evaluate_sequence(result)
    if not np.isfinite(fs8):
        result = [1.0 / n] * n
    return result
