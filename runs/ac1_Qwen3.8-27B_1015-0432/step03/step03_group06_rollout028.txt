import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def compute_score(f):
    n = f.shape[0]
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * b.max() / (s * s)

def optimize(f, deadline, rng, npeaks_base=50, npeaks_max=300):
    n = f.shape[0]
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = compute_score(f)
    eta = 0.08
    no_improv = 0
    restarts = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        
        # Select top-k peaks
        k = min(npeaks_max, npeaks_base + it // 100)
        # Use partition for efficiency
        if len(b) > k:
            # np.argpartition is O(n) average
            top_idx = np.argpartition(b, -k)[-k:]
            peaks = top_idx[np.argsort(b[top_idx][::-1])]  # sorted by value
            # Actually just use the top k indices
            peaks = top_idx
        else:

        h = np.zeros(2 * n - 1)

        c = fft_conv(f, h[::-1])

        g = np.maximum(g, 0.0)
        
        gmax = g.max()
        if gmax < 1e-30:
            break
        gn = g / gmax
        
        # Line search over a few step sizes
        best_tr = f.copy()
        best_sc_t = np.inf
        etas = [eta * 0.5, eta, eta * 1.5, eta * 2.0]
        for te in etas:
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-20)
            tr = normalize(tr)
            sc = compute_score(tr)
            if sc < best_sc_t:

                best_tr = tr
                best_eta = te
        
        f = best_tr
        
        # Adapt eta

            eta = min(0.5, eta * 1.2)

            eta = max(0.01, eta * 0.75)
