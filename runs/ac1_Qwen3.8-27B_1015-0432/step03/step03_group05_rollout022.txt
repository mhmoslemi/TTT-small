    if time.monotonic() < dl - 5:
        bf, bs = perturb(bf, bs, rng, dl - 2, cnt=300, amp=0.003)
        bf, bs = perturb(bf, bs, rng, dl - 2, cnt=200, amp=0.001)
    result = [float(v) for v in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
