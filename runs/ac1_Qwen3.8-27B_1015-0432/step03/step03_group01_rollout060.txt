def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    if s < 1e-300:
        return cm, np.zeros_like(c)
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    L1 = _fpow2(2 * n - 1)
    fa = fft(a, L1)
    c = ifft(fa * fa).real[:2 * n - 1]
    mv, p = _smax(c, tau)
    L2 = _fpow2(3 * n)
    fp = fft(p, L2)
    fa2 = fft(a, L2)
    gp = ifft(fp * np.conj(fa2)).real[:n]
    return (2.0 * n / (S * S)) * gp - (4.0 * n * mv / (S ** 3))


def _norm(a):
    s = a.sum()
    if s < 1e-8:
        return a.copy()
    return a / s


def _eval_up(x, best, best_sc):
    sc = float(evaluate_sequence(x.tolist()))
    if np.isfinite(sc) and sc < best_sc:
        best[:] = x
        best_sc = sc
    return best_sc

    for lo, hi in [(0.1, 0.9), (0.0, 0.8), (0.2, 1.0)]:
