def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def grad_soft(f, b, temp, nmax=600):

        w = wm

    c = fft_conv(f, h[::-1])

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def grad_soft(f, b, temp, nmax=600):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    w = np.exp((b - M) / temp)
    if nmax < L:
        idx = np.argpartition(w, -nmax)[-nmax:]
        w2 = np.zeros(L)
        w2[idx] = w[idx]
        w = w2
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]
