        nsc = _eval(sym)
        if np.isfinite(nsc) and nsc < best_sc:
            best = sym.copy()
            best_sc = nsc
            cur = sym.copy()
            for tv in [0.1, 0.05, 0.02, 0.01]:
                if time.monotonic() > dl - 100.0:
                    break
                try:
                    res = minimize(_obj, cur, args=(tv,), jac=_grad,
                                   method='L-BFGS-B', bounds=bounds_list,
                                   options={'maxiter': 400, 'gtol': 1e-15,
                                            'maxls': 30, 'ftol': 1e-22})
                    nx = np.clip(res.x, 0.0, 1000.0)
                    nx = _normalize(nx, scale)
                    best, best_sc = _try_update(nx, best, best_sc)
                    cur = nx.copy()
                except Exception:
                    pass

    # Restart from best with fresh random perturbations
    if time.monotonic() < dl - 80.0:
        cur = best.copy()
        for rnd in range(5):
            if time.monotonic() > dl - 80.0:
                break
            pert = cur * (1.0 + 0.05 * rng.standard_normal(n))
            pert = _normalize(np.clip(pert, 0.0, 1000.0), scale)
            for tv in [0.5, 0.1, 0.03, 0.01]:
                if time.monotonic() > dl - 80.0:
                    break
                try:
                    res = minimize(_obj, pert, args=(tv,), jac=_grad,
                                   method='L-BFGS-B', bounds=bounds_list,
                                   options={'maxiter': 300, 'gtol': 1e-14,
                                            'maxls': 25, 'ftol': 1e-20})
                    nx = np.clip(res.x, 0.0, 1000.0)
                    nx = _normalize(nx, scale)
                    best, best_sc = _try_update(nx, best, best_sc)
                    pert = nx.copy()
                except Exception:
                    break
            cur = best.copy()

    # Local perturbation search around best
    if time.monotonic() < dl - 30.0:
        wmax = max(4, n // 40)
        cur = best.copy()
        for _ in range(3000):
            if time.monotonic() > dl - 30.0:
                break
            cd = cur.copy()
            w1 = int(rng.integers(1, wmax + 1))
            s1 = int(rng.integers(0, n - w1))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.03, 0.06)))
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.03, 0.06)))
            if rng.random() < 0.1:
                cd *= float(np.exp(rng.normal(0.0, 0.015)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = _eval(cd)
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.08:
                cur = best.copy()

    # Final very low temperature polish
    if time.monotonic() < dl - 10.0:
        cur = best.copy()
        for tv in [0.0005, 0.0002, 0.0001]:
            if time.monotonic() > dl - 10.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 500, 'gtol': 1e-16,
                                        'maxls': 20, 'ftol': 1e-24})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _normalize(nx, scale)
                best, best_sc = _try_update(nx, best, best_sc)
                cur = nx.copy()
            except Exception:
                pass

    # Final validation
    final_sc = _eval(best)
    if not np.isfinite(final_sc) or final_sc >= 100.0:
        best = a0
        final_sc = _eval(best)
        if not np.isfinite(final_sc):
            best = _normalize(np.ones(n), scale)

    return [float(v) for v in best]
