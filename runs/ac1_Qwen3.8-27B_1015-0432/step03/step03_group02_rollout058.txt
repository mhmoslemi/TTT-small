def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

    """Compute gradient of sum_k h[k]*(f*f)[k] w.r.t. f."""
    c = fft_conv(f, h[::-1])
    n = len(f)
    g = c[n-1:2*n-1][::-1]
    return g
