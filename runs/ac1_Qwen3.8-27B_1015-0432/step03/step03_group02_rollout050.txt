def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    g_smooth = None
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 6000.0)
        # Soft active set: weight by excess above threshold
        frac = 0.95 - 0.25 * prog
        thresh = M * frac
        excess = np.maximum(0.0, b - thresh)
        h = excess ** alpha
        hmax = h.max()
        if hmax < 1e-15:
            h = np.zeros(L)
            h[int(np.argmax(b))] = 1.0
        else:
            h /= hmax
        # Subgradient
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        g /= gmax
        # Momentum
        if g_smooth is None:
            g_smooth = g.copy()
        else:
            g_smooth = 0.7 * g_smooth + 0.3 * g
        # Exponential update
        f *= np.exp(-eta * g_smooth)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    inits = [p / p.sum(), np.ones(n) / n]
    x = np.arange(n, dtype=np.float64)
    for pw in (0.5, 1.0, 2.0):
        d = np.minimum(x, n - 1.0 - x) + 1.0
        v = d ** pw
        inits.append(v / v.sum())
    for ph in (0.0, 0.5):
        h2 = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + ph) / (n + 1.0)))
        inits.append(h2 / h2.sum())
    for fr in (1, 2, 4):
        h3 = np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * fr * x / n), 0.01)
        inits.append(h3 / h3.sum())
    for _ in range(3):
        rv = np.maximum(rng.standard_normal(n), 0.05)
        inits.append(rv / rv.sum())
    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc, best_f = sc, v.copy()
    p1 = min(deadline, t0 + 350.0)
    f1, s1 = run_opt(best_f.copy(), p1, rng, 0.15, 2.0, 20, 400)
    if s1 < best_sc:
        best_sc, best_f = s1, f1
    if time.monotonic() < deadline - 250.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        p2 = min(deadline, t0 + 700.0)
        f2, s2 = run_opt(best_f.copy(), p2, rng, 0.08, 1.5, 15, 500)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 120.0:
        for r in range(4):
            if time.monotonic() > deadline - 60.0:
                break
            amp = 0.04 + 0.02 * r
            fr2 = best_f * np.exp(amp * rng.standard_normal(n))
            np.maximum(fr2, 1e-30, out=fr2)
            fr2 /= fr2.sum()
            fr2, sr = run_opt(fr2, deadline - 70.0, rng, 0.10, 2.0, 15, 300)
            if sr < best_sc:
                best_sc, best_f = sr, fr2
    if time.monotonic() < deadline - 15.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 5.0, rng, 200)
    result = [float(min(1000.0, max(0.0, v))) for v in best_f]
    fv = evaluate_sequence(result)
    if not np.isfinite(fv):
        result = [1.0 / n] * n
    return result
