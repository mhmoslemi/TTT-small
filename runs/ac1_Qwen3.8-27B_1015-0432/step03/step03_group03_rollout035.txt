    size = 1 << (n + m - 2).bit_length()

    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

    n = f.size
    b = fft_conv(f, f)
    bs = b / tau
    bs -= bs.max()
    e = np.exp(bs)

    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def xcorr(f, w):
    n = f.size
    L = 1 << (3 * n - 1).bit_length()
    F = np.fft.rfft(f, L)
    W = np.fft.rfft(w, L)
    return np.fft.irfft(F * np.conj(W), L)[:n]

def npos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    return f / s if s > 1e-15 else np.ones_like(f) / f.size

def sc(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * fft_conv(f, f).max() / (s * s)

def g_smooth(f, tau):
    b = fft_conv(f, f)
    x = b / tau
    x -= x.max()
    w = np.exp(x)
    w /= w.sum()
    g = 2.0 * xcorr(f, w)
    return np.maximum(g, 0.0), b.max()

def g_peak(f, K, fr):
    b = fft_conv(f, f)
    M = b.max()
    th = M * fr
    pk = np.where(b >= th)[0]
    if len(pk) > K:
        pk = np.argsort(b)[::-1][:K]
    if len(pk) < 1:
        pk = np.array([np.argmax(b)])
    w = np.zeros(2 * f.size - 1)
    w[pk] = b[pk] - th + 1e-30
    w /= w.sum()
    g = 2.0 * xcorr(f, w)
    return np.maximum(g, 0.0), M

def run(f, dl, rng, tau0, K, eta0, smooth, nit):
    n = f.size
    f = npos(f.copy())
    bf, bs = f.copy(), sc(f)
    eta, tau = eta0, tau0
    mom = np.zeros(n)
    ni = rc = 0
    for i in range(nit):
        if time.monotonic() > dl:
            break
        g, M = g_smooth(f, tau) if smooth else g_peak(f, K, 0.95)
        gm = g.max()
        if gm < 1e-30:
            break
        mom = 0.88 * mom + 0.12 * g
        mm = mom.max()
        if mm < 1e-30:
            break
        f = f * np.exp(-eta * mom / mm)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        if smooth:
            tau *= 0.99999
        if i % 50 == 0:
            s = sc(f)
            if s < bs - 1e-14:
                bs, bf, ni = s, f.copy(), 0
            else:
                ni += 1
            if ni > 500:
                f = bf * np.exp((0.02 + 0.015 * rc) * rng.standard_normal(n))
                f = npos(np.maximum(f, 1e-15))
                mom[:] = 0
                ni, rc = 0, rc + 1
                eta = min(0.25, eta * 1.3)
        eta *= 0.9998
    return bf, bs

def sym(f):
    return npos(0.5 * (f + f[::-1]))

def propose_candidate(seed=42, budget_s=1100, **kwargs):
