import time
from scipy.optimize import minimize, linprog

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = np.fft.fft(a, L)
    return np.fft.ifft(fa * fa).real[:2 * n - 1]

def _conv_cross(a, b):
    L = _fpow2(2 * a.size - 1)
    return np.fft.ifft(np.fft.fft(a, L) * np.fft.fft(b, L)).real[:2 * a.size - 1]

def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    return cm + tau * np.log(e.sum()), e / e.sum()

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

    n = a.size
    S = a.sum()

    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

            js = np.arange(lo, hi + 1)

        b_ub = np.zeros(K)
        A_eq = np.zeros((1, n + 1))

                          A_eq=A_eq, b_eq=[1.0],
