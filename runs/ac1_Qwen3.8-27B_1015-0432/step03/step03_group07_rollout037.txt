import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n+m-1]

def acscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def softmax_grad(f, b, alpha):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:

    T = alpha * M

    mask = logw > -30
    if not mask.any():
        idx = np.argmax(b)
        w = np.zeros(L)
        w[idx] = 1.0
    else:
        expw = np.exp(logw[mask])
        w = np.zeros(L)
        w[mask] = expw
        ws = w.sum()
        if ws < 1e-30:
            idx = np.argmax(b)
            w = np.zeros(L)
            w[idx] = 1.0
        else:
            w /= ws
    # Gradient
