def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_subgrad(f, peaks, S):
    n = f.size
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    A = np.fft.rfft(f, S)
    H = np.fft.rfft(h, S)
    G = np.fft.irfft(np.conj(A) * H, S)
    return G[:n].copy()
