import time
from scipy.fft import rfft, irfft

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return irfft(rfft(a, sz) * rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad_smooth(f, b, temp):
    """Compute subgradient using softmax weighting of peaks."""
    n = f.shape[0]
    L = 2 * n - 1
    # Softmax weights on b values
    M = b.max()
    w = np.exp((b - M) / temp)
    w /= w.sum()
    # Gradient of b[k] w.r.t. f
    # conv(f, w_rev) gives part of the gradient
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return g

    bs = score(f)

    temp = temp0
    ni = 0
