            cur = best + 0.001 * float(best_sum_safe(best)) * rng.normal(0, 1, n)
            cur = np.clip(cur, 0.0, 1000.0)
            cur = _normalize(cur, scale)

    final_sc = _eval(best)
    if not np.isfinite(final_sc):
        best = par / par.sum() * scale
    return [float(v) for v in best]


def float_best_sum_safe(a):
    s = float(np.sum(a))
    if s < 1e-8 or not np.isfinite(s):
        return 1.0
    return s
