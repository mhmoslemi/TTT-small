            thr = np.partition(prods, -tk)[-tk:].min()
            m = prods >= thr
            tr[ks[m]] *= frac
        tr = np.clip(tr, 0.0, None)
        if tr.sum() < 0.01:
            continue
        sc = float(evaluate_sequence(tr.tolist()))
        if sc < best_sc:
            best_sc = sc
            best_a = tr.copy()
    return best_a, best_sc
