import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, b, max_peaks):
    n = f.shape[0]
    M = b.max()
    thr = M * 0.99
    pk = np.where(b >= thr)[0]
    if len(pk) < 5:
        pk = np.argsort(b)[-30:]
    elif len(pk) > max_peaks:
        pk = np.argsort(b)[-max_peaks:]
    L = 2 * n - 1
    h = np.zeros(L)
    h[pk] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, sym=False, eta0=0.045, max_it=500000):
