def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()

def score(f):
    b = fft_conv(f, f)
    s = f.sum()

def lse(b, beta):

    if mx < 1e-30:

    e = np.exp(beta * (b - mx))
    s = e.sum()
    return mx + np.log(s) / beta, e / s
