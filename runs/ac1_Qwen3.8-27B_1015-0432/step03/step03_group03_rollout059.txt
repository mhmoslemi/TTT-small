    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.0, 2.0]:
        v = d ** pw
        inits.append(v / v.sum())
    h = 0.5 * (1.0 + np.cos(np.pi * x / (n - 1)))
    inits.append(h / h.sum())
    p2 = p * np.exp(0.04 * rng.standard_normal(n))
    inits.append(np.maximum(p2, 1e-15) / np.maximum(p2, 1e-15).sum())
    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    n = len(height_sequence_1)

    inits = gen_inits(n, rng, height_sequence_1)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = 2.0 * n * fft_conv(init, init).max()
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: Asymmetric aggressive
    t1 = min(deadline, t0 + 400.0)
    f1, sc1 = optimize(best_f.copy(), t1, rng, eta0=0.07, pmax=400, pmin=5)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()

    # Phase 2: Symmetric
    if time.monotonic() < deadline - 280.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = fs / fs.sum()
        t2 = min(deadline - 180.0, t0 + 720.0)
        f2, sc2 = optimize(fs, t2, rng, eta0=0.04, pmax=600, pmin=10, sym=True)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2.copy()

    # Phase 3: Asymmetric fine
    if time.monotonic() < deadline - 100.0:
        t3 = min(deadline - 30.0, t0 + 1020.0)
        f3, sc3 = optimize(best_f.copy(), t3, rng, eta0=0.025, pmax=800, pmin=20)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3.copy()

    # Phase 4: Micro-perturbations
    if time.monotonic() < deadline - 10.0:
        for _ in range(500):
            if time.monotonic() > deadline - 5.0:
                break
            trial = best_f * np.exp(0.002 * rng.standard_normal(n))
            trial = np.maximum(trial, 0.0)
            trial = trial / trial.sum()
            sc = 2.0 * n * fft_conv(trial, trial).max()
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()

    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
