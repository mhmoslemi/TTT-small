        w = weights[peaks]
        g = peak_grad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-18)
        f = norm1(f)
        if it % 20 == 0:
            sc = obj_val(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
        eta *= 0.998
    sc = obj_val(f)
    if sc < best_sc:
        best_sc = sc
        best_f = f.copy()
    return best_f, best_sc

def gen_inits(n, rng):
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(norm1(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.4, 0.8, 1.4, 2.2]:
        inits.append(norm1(d ** pw))
    h1 = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(norm1(h1))
    for beta in [0.4, 0.8, 1.2]:
        asym = np.exp(-beta * np.abs(2.0 * x / max(n - 1, 1) - 1.0))
        inits.append(norm1(asym))
    half = n // 2
    tl = np.ones(n)
    tl[:half] = 1.0 + 0.5 * np.cos(np.pi * np.arange(half) / max(half, 1))
    tl[half:] = 1.0 - 0.5 * np.cos(np.pi * np.arange(n - half) / max(n - half + 1, 1))
    inits.append(norm1(np.maximum(tl, 0.01)))
    r = np.ones(n) + 0.4 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(norm1(r))
    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = gen_inits(n, rng)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = obj_val(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    t1 = min(deadline - 350.0, t0 + 280.0)
    f1, sc1 = optimize(best_f.copy(), t1, rng, eta0=0.06, n_peaks=150)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    if time.monotonic() < deadline - 280.0:
        f2 = norm1(0.5 * (best_f + best_f[::-1]))
        t2 = min(deadline - 280.0, t0 + 530.0)
        f2o, sc2o = optimize(f2, t2, rng, eta0=0.04, n_peaks=200, sym=True)
        if sc2o < best_sc:
            best_sc, best_f = sc2o, f2o.copy()
    if time.monotonic() < deadline - 150.0:
        t3 = min(deadline - 150.0, t0 + 780.0)
        f3o, sc3o = optimize(best_f.copy(), t3, rng, eta0=0.025, n_peaks=300)
        if sc3o < best_sc:
            best_sc, best_f = sc3o, f3o.copy()
    if time.monotonic() < deadline - 60.0:
        t4 = min(deadline - 40.0, t0 + 950.0)
        f4o, sc4o = smooth_polish(best_f.copy(), t4, rng, iters=800)
        if sc4o < best_sc:
            best_sc, best_f = sc4o, f4o.copy()
    if time.monotonic() < deadline - 15.0:
        for _ in range(150):
            if time.monotonic() > deadline - 8.0:
                break
            for scale in [0.003, 0.008, 0.02]:
                trial = best_f * np.exp(scale * rng.standard_normal(n))
                trial = np.maximum(trial, 0.0)
                trial = norm1(trial)
                sc = obj_val(trial)
                if sc < best_sc:
                    best_sc, best_f = sc, trial
    result = [float(x) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
