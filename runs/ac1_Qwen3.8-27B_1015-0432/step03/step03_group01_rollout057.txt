    """Try to equalize the top m peaks of the convolution."""
    n = a.size

    # Build the linear system: for each peak j, sum_i (dc_j/da_i) * delta_i = target - c_j
    # dc_j/da_i = 2 * a[j-i] (with boundary handling)
    # This is equivalent to: 2 * (c_j convolved with delta) ... 
    # Actually dc_j/da = 2 * flip(a) shifted by j
    # So the j-th row of the Jacobian is 2 * a_reversed[j:j+n]
    # (with appropriate handling)

            w1 = int(rng.integers(2, wmax + 1))
            pos = int(np.clip(pk - (n - 1), 0, n - w1))
            cd[pos:pos + w1] *= float(np.exp(rng.normal(-0.03, 0.04)))
            w2 = int(rng.integers(2, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.02, 0.03)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                cd = _normalize(cd, scale)
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.05:
                cur = best.copy()

    # Final peak shave
    if time.monotonic() < dl - 2.0:
        fx = _peak_shave(best.copy(), m=10, lr=0.02)
        fx = _normalize(fx, scale)
        best, best_sc = _eval_upd(fx, best, best_sc)

    # Symmetrization check
    if time.monotonic() < dl - 1.0:
        sym = _normalize(np.clip(0.5 * (best + best[::-1]), 0, 1000.0), scale)
        best, best_sc = _eval_upd(sym, best, best_sc)

    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = _normalize(np.ones(n), scale)
    return [float(v) for v in best]
