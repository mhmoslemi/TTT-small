    f1, s1 = optimize(best_f.copy(), ph1, rng, eta0=0.12)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()

    if time.monotonic() < dl - 130.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        f2, s2 = optimize(fs, min(dl, t0 + 800.0), rng, sym=True, eta0=0.08)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()

    if time.monotonic() < dl - 40.0:
        f3, s3 = optimize(best_f.copy(), dl - 12.0, rng, eta0=0.04)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()

    if time.monotonic() < dl - 4.0:
        for _ in range(80):
            if time.monotonic() > dl - 2.0:
                break
            trial = best_f * np.exp(0.003 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = best_f.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    if not np.isfinite(evaluate_sequence(result_list)):
        result_list = [1.0 / n] * n
    return result_list
