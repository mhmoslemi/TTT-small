    d = np.minimum(x, 1.0 - x) + 0.05
    for pw in (0.5, 1.0, 2.0):
        inits.append(d ** pw / (d ** pw).sum())
    rv = np.maximum(rng.standard_normal(n), 0.05)
    inits.append(rv / rv.sum())
    rv2 = np.maximum(0.5 + 0.5 * rng.standard_normal(n), 0.01)
    inits.append(rv2 / rv2.sum())
    bf, bs = None, np.inf
    for ini in inits:
        sc = sct(ini)
        if sc < bs:
            bs, bf = sc, ini.copy()
    p1 = min(dl - 250, t0 + 400)
    f1, s1 = optimize(bf.copy(), p1, rng, sym=False, eta0=0.05)
    if s1 < bs:
        bs, bf = s1, f1
    if time.monotonic() < dl - 200:
        fs = np.maximum(0.5 * (bf + bf[::-1]), 0.0)
        ss = fs.sum()
        if ss > 1e-15:
            fs /= ss
        p2 = min(dl - 180, t0 + 750)
        f2, s2 = optimize(fs, p2, rng, sym=True, eta0=0.03)
        if s2 < bs:
            bs, bf = s2, f2
    if time.monotonic() < dl - 120:
        p3 = min(dl - 80, t0 + 960)
        f3, s3 = optimize(bf.copy(), p3, rng, sym=False, eta0=0.015)
        if s3 < bs:
            bs, bf = s3, f3
    if time.monotonic() < dl - 40:
        f4, s4 = optimize(bf.copy(), dl - 15, rng, sym=False, eta0=0.004, max_it=300000)
        if s4 < bs:
            bs, bf = s4, f4
    if time.monotonic() < dl - 3:
        bf, bs = polish(bf, bs, dl - 3, rng)
    result = [float(v) for v in bf]
    fs8 = evaluate_sequence(result)
    if not np.isfinite(fs8):
        result = [1.0] * n
        fs8 = evaluate_sequence(result)
    return result
