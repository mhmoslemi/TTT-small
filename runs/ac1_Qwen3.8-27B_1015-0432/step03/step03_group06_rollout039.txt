def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

    x = np.arange(n, dtype=np.float64)
    inits.append(normalize(0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))))
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in (0.5, 1.0, 2.0, 3.0):
        inits.append(normalize(d ** pw))
    xc = (n - 1) / 2.0
    for sig in (0.1, 0.2, 0.3):
        inits.append(normalize(np.exp(-0.5 * ((x - xc) / (sig * n)) ** 2)))
    r = np.ones(n) + 0.3 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(normalize(r))
    best_f, best_sc = None, np.inf
    for init in inits:
        sc = score(init)
        if sc < best_sc:
            best_sc, best_f = sc, init.copy()
    fs = normalize(0.5 * (best_f + best_f[::-1]))
    ph1 = min(dl, t0 + budget_s * 0.35)
    fo, sco = optimize(fs, ph1, rng, True)
    if sco < best_sc:
        best_sc, best_f = sco, fo.copy()
    if time.monotonic() < dl - 60.0:
        ph2 = min(dl, t0 + budget_s * 0.65)
        fo, sco = optimize(best_f.copy(), ph2, rng, False)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < dl - 30.0:
        fs2 = normalize(0.5 * (best_f + best_f[::-1]))
        fo, sco = optimize(fs2, dl - 10.0, rng, True)
        if sco < best_sc:
            best_sc, best_f = sco, fo.copy()
    if time.monotonic() < dl - 3.0:
        for _ in range(300):
            if time.monotonic() > dl - 1.5:
                break
            tr = best_f * np.exp(0.002 * rng.standard_normal(n))
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()
    result = [float(v) for v in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
