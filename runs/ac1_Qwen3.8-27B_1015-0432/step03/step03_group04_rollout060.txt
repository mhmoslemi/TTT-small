def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()

    s = f.sum()
    if s < 1e-10:

    return float(2.0 * f.size * b.max() / (s * s))

    n = f.shape[0]
    M = b.max()
    S = np.where(b >= threshold * M)[0]

    s = len(S)

    g *= 2.0 / s

        # b[k*] = sum_j f[j] f[k*-j]
        # To reduce b[k*], reduce f[j] for j where f[j]*f[k*-j] is large
        contrib = np.zeros(len(f))
        lo = max(0, k_star - len(f) + 1)
        hi = min(len(f) - 1, k_star)

        # Reduce f proportionally to contribution
        f *= np.exp(-eta * contrib / contrib.max())
        f = np.maximum(f, 1e-20)
        f /= f.sum()

        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
        else:
            f = best_f.copy()
            eta *= 0.9

import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def subgrad(f, b, thresh=0.995, max_pk=400):
    n = f.shape[0]
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    pk = np.where(b >= thresh * M)[0]
    if len(pk) > max_pk:
        pk = pk[np.argsort(b[pk])[-max_pk:]]
    s = max(1, len(pk))
    g = np.zeros(n)
    for k in pk:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            idx = k - np.arange(lo, hi + 1)
            g[lo:hi + 1] += f[idx]
    g *= 2.0 / s
    return g
