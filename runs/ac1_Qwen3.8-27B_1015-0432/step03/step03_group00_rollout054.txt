    cur, best_sc = _shave(cur, best_sc, min(dl - 80, t0 + budget_s - 60), 8000)
    s = _eval(cur)
    if np.isfinite(s) and s < best_sc:
        best_sc, best = s, cur.copy()
    else:
        cur = best.copy()

    # Phase 4: Final polish with tiny tau
    if time.monotonic() < dl - 45:
        for tau in [0.003, 0.001, 0.0005]:
            if time.monotonic() > dl - 30:
                break
            try:
                res = minimize(_obj, cur, args=(tau,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': 1200, 'gtol': 1e-17,
                                        'maxls': 50, 'ftol': 1e-20,
                                        'maxfun': 3500})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * sc
                s = _eval(nx)
                if np.isfinite(s) and s < best_sc:
                    best_sc, best = s, nx.copy()
                cur = nx.copy()
            except Exception:
                pass

    # Phase 5: Final peak shaving
    if time.monotonic() < dl - 10:
        cur = best.copy()
        cur, best_sc = _shave(cur, best_sc, dl - 5, 3000)
        s = _eval(cur)
        if np.isfinite(s) and s < best_sc:
            best_sc, best = s, cur.copy()

    # Ensure validity
    if not np.isfinite(best_sc):
        best = a0.copy()
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = a0.copy()

    return [float(v) for v in best]
