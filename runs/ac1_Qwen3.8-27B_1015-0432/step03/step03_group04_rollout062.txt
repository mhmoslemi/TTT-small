        progress = min(1.0, it / (max_iter * 0.5))
        alpha = alpha_hi + (alpha_lo - alpha_hi) * (progress ** 0.7)
        g = compute_grad(f, b, alpha)
        gmax = g.max()
        if gmax < 1e-25:
            break
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()

        if it % 25 == 0:
            sc = get_score(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 400:
                rest_scale = 0.04 + 0.08 * min(1.0, restart_count / 12.0)
                f = best_f * np.exp(rest_scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                restart_count += 1
                eta = min(0.22, eta * 1.35)
        eta *= 0.9985

    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 22.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.15, 0.4, 0.9, 1.8]:
        inits.append(d ** pw / (d ** pw).sum())
    u = 1.0 / (d * d + 1.0)
    inits.append(u / u.sum())
    v = 1.0 / (d + 0.3)
    inits.append(v / v.sum())
    for cf in [0.25, 0.33]:
        mu = cf * (n - 1.0)
        hump = np.exp(-((x - mu) ** 2 + (x - (n - 1.0 - mu)) ** 2) / (2.0 * (0.12 * n) ** 2))
        hump += 0.05
        inits.append(hump / hump.sum())
    for _ in range(4):
        r = np.abs(rng.standard_normal(n)) + 0.03
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    best_f = inits[0].copy()
    best_sc = 1e30
    for init in inits:
        sc = get_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: asymmetric main
    ph1 = min(deadline, t0 + 380.0)
    f1, s1 = optimize(best_f.copy(), ph1, rng)
    if s1 < best_sc:
        best_sc = s1
        best_f = f1.copy()

    # Phase 2: symmetric refinement
    if time.monotonic() < deadline - 90.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ph2 = min(deadline, t0 + 760.0)
        f2, s2 = optimize(fs, ph2, rng, sym=True)
        if s2 < best_sc:
            best_sc = s2
            best_f = f2.copy()

    # Phase 3: asymmetric re-refinement
    if time.monotonic() < deadline - 30.0:
        ph3 = deadline - 12.0
        f3, s3 = optimize(best_f.copy(), ph3, rng)
        if s3 < best_sc:
            best_sc = s3
            best_f = f3.copy()

    # Phase 4: tiny perturbations
    if time.monotonic() < deadline - 4.0:
        for _ in range(150):
            if time.monotonic() > deadline - 2.0:
                break
            trial = best_f.copy()
            trial *= np.exp(0.004 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-15)
            trial /= trial.sum()
            sc = get_score(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = best_f.copy()
    s = result.sum()
    if s < 0.01:
        result *= 0.02 / s
    result_list = [float(v) for v in result]
    fs = evaluate_sequence(result_list)
    if not np.isfinite(fs):
        result_list = [1.0 / n] * n
    return result_list
