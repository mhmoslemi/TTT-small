def fft_conv(a, b):

    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def pressure(f, b, frac, max_peaks):
    L = len(b)
    M = b.max()
    thresh = M * frac
    active = np.where(b >= thresh)[0]

    else:

    h[idx] = np.maximum(b[idx] - thresh, 0.0) + 1e-10
    n = len(f)
    c = fft_conv(f, h[::-1])
    g = c[n-1:2*n-1][::-1]
    np.maximum(g, 0.0, out=g)
    gm = g.max()
    if gm < 1e-30:
        return None
    return g / gm

    best_sc = score(f)

        if it % check_iv == 0:

            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:

            if patience > 600:

                f /= f.sum()
                patience = 0
                eta = min(0.30, eta * 1.3)

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return float('inf')
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def smooth_pressure(f, b, T):
    n = len(f)
    scaled = b - b.max()
    h = np.exp(scaled / T)
    S = h.sum()
    if S < 1e-30:
        h = np.zeros(len(b))
        h[np.argmax(b)] = 1.0
    else:
        h /= S
    c = fft_conv(f, h[::-1])
    g = c[n-1:2*n-1][::-1]
    np.maximum(g, 0.0, out=g)
    gm = g.max()
    if gm < 1e-30:
        return None
    return g / gm
