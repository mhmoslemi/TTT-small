import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n-1:2*n-1][::-1]

def soft_subgrad(f, b, temp):
    """Compute subgradient with softmax weighting over peaks."""
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    # Soft weights: w_k = exp(-(M - b_k)/temp)
    w = np.exp(-(M - b) / temp)
    w /= w.sum()
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n-1:2*n-1][::-1]

    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0  # no-improvement counter
    for it in range(max_it):

        b = fft_conv(f, f)
        M = b.max()

            elif len(pk) > 600:

            g = subgrad(f, pk)

            temp = 0.5 * M
            g = soft_subgrad(f, b, temp)
        
        gm = g.max()
        if gm < 1e-30:

        # Multiplicative update with adaptive step
        for attempt in range(4):

            f_new = np.maximum(f_new, 1e-20)

            f_new /= f_new.sum()

                if sc < bs - 1e-15:
