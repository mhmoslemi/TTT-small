def _fn2(m):
    r = 1
    while r < m:
        r <<= 1
    return r


def _conv(a):
    n = a.size
    L = _fn2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s
