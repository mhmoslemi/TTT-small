                if 0 <= j < n:
                    prod[i] = cd[i] * cd[j]
            cnt = max(2, (i_hi - i_lo + 1) // 5)
            top = np.argsort(prod)[-cnt:]
            for i in top:
                cd[i] *= (1.0 - 0.03 * rng.random())
            # Boost random other elements
            others = np.array([i for i in range(n) if i not in set(top.tolist())])
            if len(others) > 0:
                pick = rng.choice(others, size=min(cnt, len(others)), replace=False)
                for i in pick:
                    cd[i] *= (1.0 + 0.02 * rng.random())
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy(); best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.04:
                cur = best.copy()

    # Final ultra-low-temp polish
    if time.monotonic() < dl - 3:
        cur = best.copy()
        for tv in [5e-5, 2e-5]:
            if time.monotonic() > dl - 3:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds,
                               options={'maxiter': 400, 'gtol': 1e-16,
                                        'maxls': 20, 'ftol': 1e-24})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _norm(nx, scale)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy(); best_sc = sc; cur = nx.copy()
            except Exception:
                pass

    # Validate
    fs = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(fs) or len(best) == 0:
        best = np.ones(n)
    return [float(v) for v in best]
