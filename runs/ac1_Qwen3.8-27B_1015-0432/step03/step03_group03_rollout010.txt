import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def normalize(f):
    s = f.sum()
    if s < 1e-30:
        return np.ones_like(f) / f.shape[0]
    return f / s

def symmetrize(f):
    return 0.5 * (f + f[::-1])

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

    eta = eta0

            break
