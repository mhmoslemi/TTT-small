        best_sc, best_f = s2, f2
    e3 = min(deadline, t0 + 880.0)
    f3, s3 = run_direct(best_f.copy(), e3, rng, eta0=0.04)
    if s3 < best_sc:
        best_sc, best_f = s3, f3
    if time.monotonic() < deadline - 60:
        for r in range(3):
            if time.monotonic() > deadline - 30:
                break
            f_r = best_f * np.exp(0.01 * (1 + 0.5 * r) * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            fr, sr = run_opt(f_r, deadline - 40, rng, eta0=0.06, tau_start=0.03, tau_end=0.004)
            if sr < best_sc:
                best_sc, best_f = sr, fr
    if time.monotonic() < deadline - 8:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 5, rng)
    result = [float(v) for v in best_f]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
