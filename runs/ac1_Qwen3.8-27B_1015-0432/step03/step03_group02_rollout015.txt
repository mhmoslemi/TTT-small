import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb)[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.10):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 6000.0)
        frac = 0.995 - 0.05 * prog
        thresh = M * frac
        active = np.where(b >= thresh)[0]
        cnt = len(active)
        h = np.zeros(L)
        if cnt > max_peaks:
            idx = np.argpartition(b, -max_peaks)[-max_peaks:]
            h[idx] = 1.0
        elif cnt >= 1:
            h[active] = 1.0
        else:
            h[np.argmax(b)] = 1.0
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1]
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1
