import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks, weights=None):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    if weights is None:
        h[peaks] = 1.0
    else:
        h[peaks] = weights
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

    # Apply a simple smoothing using FFT
    n = g.size
    freqs = np.fft.rfftfreq(n, d=1.0)
    fft_g = np.fft.rfft(g)
    window = np.exp(-2.0 * np.pi**2 * freqs**2 * sigma**2)
    return np.fft.irfft(fft_g * window, n)

def optimize(f, deadline, rng, sym=False, max_it=500000, eta0=0.04, smooth_sigma=0):

    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0

        b_norm = b - M
