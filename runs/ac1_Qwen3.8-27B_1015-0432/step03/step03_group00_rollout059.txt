def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = np.fft.fft(a, L)
    return np.fft.ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = np.fft.ifft(np.fft.fft(a, L) * np.fft.fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _norm(a):

    s = a.sum()
    if s < 1e-8:

    return a / s * np.sqrt(2.0 * a.size)

    g = np.zeros(n)
