import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def obj_val(f):
    b = fft_conv(f, f)
    return float(b.max())

def compute_subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.1, n_peaks=100, smooth=5.0, max_iter=300000):
    n = f.shape[0]
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    best_f = f.copy()
    best_obj = obj_val(f)
    prev_obj = best_obj
    eta = eta0
    no_improv = 0
    restarts = 0
    momentum = np.zeros(n)
    k = int(4 * smooth) | 1
    xk = np.arange(k) - k // 2
    kern = np.exp(-0.5 * (xk / smooth) ** 2)
    kern /= kern.sum()

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        frac = 0.97 - 0.03 * min(1.0, it / 2000.0)
        thresh = M * frac
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > n_peaks:
            idx = np.argsort(b[peaks])[::-1][:n_peaks]
            peaks = peaks[idx]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        vals = b[peaks]
        w = (vals / M) ** 3
        g = compute_subgrad(f, peaks, w)
        if smooth > 0.5:
            g = np.convolve(g, kern, mode='same')
            g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-25:
            break
        momentum = 0.75 * momentum + 0.25 * g
        mmax = momentum.max()
        if mmax < 1e-25:

        t = eta / mmax
        f_new = f * np.exp(-t * momentum)
        f_new = np.maximum(f_new, 1e-20)
        f_new = f_new / f_new.sum()
        sc = obj_val(f_new)
        if sc < best_obj - 1e-14:
            f = f_new
            best_obj = sc
            best_f = f.copy()
            prev_obj = sc
            no_improv = 0
            eta = min(0.2, eta * 1.005)
        elif sc < prev_obj:
            f = f_new
            prev_obj = sc
            no_improv += 1
        else:
            no_improv += 1
            eta = max(0.005, eta * 0.93)
        if no_improv > 200:
            scale = 0.02 + 0.05 * min(1.0, restarts / 6.0)
            f = best_f * np.exp(scale * rng.standard_normal(n))

            momentum = np.zeros(n)
            prev_obj = obj_val(f)
            no_improv = 0
            restarts += 1
            eta = min(0.15, eta * 1.2)
        eta *= 0.99995
