import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return 2.0 * f.size * b.max() / (s * s)

def grad_peak(f, b, frac, K):
    n = f.shape[0]
    M = b.max()
    peaks = np.where(b >= M * frac)[0]
    if len(peaks) > K:
        peaks = peaks[np.argsort(b[peaks])][-K:]
    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])
    L = 2 * n - 1
    h = np.zeros(L)
    w = b[peaks]
    h[peaks] = w / w.sum()
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
