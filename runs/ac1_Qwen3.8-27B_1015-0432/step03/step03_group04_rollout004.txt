def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * len(f) * b.max() / (s * s))

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    s = f.sum()
    if s < 1e-10:
        return 1e30
    b = fft_conv(f, f)
    return float(2.0 * len(f) * b.max() / (s * s))

def compute_grad_sym(x, n, thresh_frac, max_pk=600):
    """Compute symmetric gradient for half-sequence x of length m=n//2."""
    m = len(x)
    a = np.concatenate([x, x[::-1]]) if n % 2 == 0 else np.concatenate([x, x[-2::-1]])
    b = fft_conv(a, a)
    M = b.max()
    thresh = M * thresh_frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    if len(peaks) < 1:
        peaks = np.array([np.argmax(b)])
    w = np.maximum(b[peaks] - thresh, M * 0.002)
    h = np.zeros(len(b))
    h[peaks] = w / w.sum()
    h = 0.5 * (h + h[::-1])
    c = fft_conv(a, h[::-1])
    g_full = 2.0 * c[n - 1:2 * n - 1][::-1]
