        if time.monotonic() < dl - 120.0:
            nb = 128
            bv = _blocks_from_seq(cur, nb)
            block_size = n // nb
            x_b = (np.arange(nb) + 0.5) * block_size
            x_f = np.arange(n) + 0.5 * block_size
            f_interp = interp1d(x_b, bv, kind='cubic', fill_value='extrapolate')
            a_interp = np.maximum(f_interp(x_f), 0.0)
            a_interp = _normalize(a_interp, scale)
            sc = float(evaluate_sequence(a_interp.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = a_interp.copy()
                best_sc = sc
                cur = a_interp.copy()

    # Phase 3: Peak-targeting local search
    if time.monotonic() < dl - 50.0:
        wmax = max(3, n // 40)
        for _ in range(3000):
            if time.monotonic() > dl - 50.0:
                break
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            w1 = int(rng.integers(1, wmax + 1))
            lo = max(0, min(pk, n - 1) - w1)
            hi = min(n - w1, max(0, min(pk, n - 1)))
            s1 = int(rng.integers(lo, hi + 1)) if hi >= lo else int(rng.integers(0, n - w1))
            cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.025, 0.035)))
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.025, 0.035)))
            if rng.random() < 0.1:
                cd *= float(np.exp(rng.normal(0.0, 0.01)))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                sc = float(evaluate_sequence(cd.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = cd.copy()
                    best_sc = sc
                    cur = cd.copy()
            if rng.random() < 0.06:
                cur = best.copy()

    # Phase 4: Symmetrize
    if time.monotonic() < dl - 40.0:
        sym = 0.5 * (cur + cur[::-1])
        sym = _normalize(np.clip(sym, 0.0, 1000.0), scale)
        sc = float(evaluate_sequence(sym.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = sym.copy()
            best_sc = sc
            cur = sym.copy()

    # Phase 5: Final polish at very low temperature
    if time.monotonic() < dl - 10.0:
        cur = best.copy()
        for tv in [0.003, 0.001, 3e-4]:
            if time.monotonic() > dl - 10.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 500, 'gtol': 1e-15,
                                        'maxls': 25, 'ftol': 1e-21})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _normalize(nx, scale)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Final symmetrization and validation
    best = 0.5 * (best + best[::-1])
    best = np.clip(best, 0.0, 1000.0)
    if best.sum() < 0.01:
        best = np.ones(n)
    best = _normalize(best, scale)
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = a0

    return [float(v) for v in best]
