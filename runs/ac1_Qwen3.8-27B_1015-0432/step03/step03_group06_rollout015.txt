import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym, k, wpow, mu0):
    n = f.shape[0]
    f = normalize(f.copy())
    best_f, best_sc = f.copy(), score(f)

        b = fft_conv(f, f)
        idx = np.argsort(b)[-k:]
        w = b[idx] ** wpow
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(k) / k
        else:
            w = w / ws
        g = subgrad(f, idx, w)
        gm = g.max()
        if gm < 1e-30:
            break
        mom = mu * mom + (1.0 - mu) * (g / gm)
        mn = mom / np.maximum(mom.max(), 1e-30)
        best_te, best_sct, best_tr = eta, np.inf, f.copy()
        for te in [eta * 0.25, eta * 0.5, eta, eta * 2.0, eta * 4.0]:
            tr = np.maximum(f * np.exp(-te * mn), 1e-20)
