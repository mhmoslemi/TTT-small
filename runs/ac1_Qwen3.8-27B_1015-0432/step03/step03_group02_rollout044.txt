    if s1 < best_sc:
        best_sc, best_f = s1, f1
    if time.monotonic() < deadline - 180.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = np.maximum(fs, 1e-30)
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        phase2_end = min(deadline, t0 + 700.0)
        f2, s2 = run_opt(best_f.copy(), phase2_end, rng, max_peaks=500, eta_init=0.06, weight_pow=2.0)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 120.0:
        fs = smooth1d(best_f, 3)
        fs = np.maximum(fs, 1e-30)
        fs /= fs.sum()
        ss = score(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        f2, s2 = run_opt(best_f.copy(), min(deadline, t0 + 900.0), rng, max_peaks=600, eta_init=0.04, weight_pow=3.0)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 80.0:
        for restart in range(3):
            if time.monotonic() > deadline - 40.0:
                break
            sc_r = 0.012 + 0.012 * restart
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            f_r, s_r = run_opt(f_r, deadline - 50.0, rng, max_peaks=600, eta_init=0.05, weight_pow=2.0)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r
    if time.monotonic() < deadline - 15.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 8.0, rng)
    if time.monotonic() < deadline - 3.0:
        best_f, best_sc = coord_polish(best_f.copy(), deadline - 1.0, rng)
    result = [float(min(1000.0, max(0.0, v))) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
