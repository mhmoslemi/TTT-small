from scipy.sparse import csr_matrix

def fft_conv(a, b):

    size = 1 << (n + m - 2).bit_length()

    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def make_conv_matrix(f):
    """Build sparse matrix A where A @ g = f * g (convolution)."""
    n = len(f)
    L = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(L):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(L, n))
    return A

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

    n = f.shape[0]
    L = 2 * n - 1

    if M < 1e-30:

    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    inits = []
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    c1 = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits.append(c1 / c1.sum())
    for cf in [0.3, 0.7]:
        g = np.exp(-((x - cf * (n - 1)) ** 2) / (2.0 * (0.12 * n) ** 2))
        inits.append(g / g.sum())
    for _ in range(4):
        r = np.maximum(1.0 + 0.4 * rng.standard_normal(n), 0.01)
        inits.append(r / r.sum())
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = nscore(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()
    dl1 = min(deadline, t0 + 380.0)
    f1, sc1 = optimize(best_f.copy(), dl1, rng, 3.0, 15.0)
    if sc1 < best_sc:
        best_sc, best_f = sc1, f1.copy()
    if time.monotonic() < deadline - 350:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        sc_s = nscore(fs)
        if sc_s < best_sc:
            best_sc, best_f = sc_s, fs.copy()
        dl2 = min(deadline, t0 + 750.0)
        f2, sc2 = optimize(fs.copy(), dl2, rng, 5.0, 25.0, sym=True, mu=0.9)
        if sc2 < best_sc:
            best_sc, best_f = sc2, f2.copy()
    if time.monotonic() < deadline - 100:
        dl3 = deadline - 20.0
        f3, sc3 = optimize(best_f.copy(), dl3, rng, 10.0, 50.0, mu=0.7)
        if sc3 < best_sc:
            best_sc, best_f = sc3, f3.copy()
    if time.monotonic() < deadline - 8:
        amps = [0.001, 0.003, 0.01]
        while time.monotonic() < deadline - 5:
            amp = amps[rng.integers(3)]
            trial = best_f * np.exp(amp * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-20)
            trial /= trial.sum()
            sc = nscore(trial)
            if sc < best_sc:
                best_sc, best_f = sc, trial.copy()
    result = [float(v) for v in best_f]
    fc = evaluate_sequence(result)
    if not np.isfinite(fc):
        result = [1.0 / n] * n
    return result
