import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, b, beta, thr_frac, max_peaks):
    n = f.shape[0]
    M = b.max()
    if M < 1e-30:
        return np.zeros(n), 0.0
    thr = M * thr_frac
    pk = np.where(b >= thr)[0]
    if len(pk) > max_peaks:
        pk = np.argsort(b)[-max_peaks:]
    if len(pk) < 3:
        pk = np.argsort(b)[-30:]
    w = np.exp(beta * (b[pk] - M) / M)
    w /= w.sum()
    L = 2 * n - 1
    h = np.zeros(L)
    h[pk] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return g, M

def optimize(f, deadline, rng, beta, thr_frac, max_peaks, eta0, max_it=400000):
    n = f.shape[0]
    f = np.maximum(f.copy(), 1e-30)
    f /= f.sum()
    bf = f.copy()
    bs = score(f)
    eta = eta0
    stall = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        g, M = compute_grad(f, b, beta, thr_frac, max_peaks)
        gm = g.max()
        if gm < 1e-30:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        if it % 15 == 0:
            sc = score(f)
            if sc < bs - 1e-12:
                bs = sc
                bf = f.copy()
                stall = 0
            else:
                stall += 1
            if stall > 500:
                f = bf * np.exp(0.015 * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)
                f /= f.sum()
                eta = min(eta * 1.3, 0.08)
                stall = 0
        if it % 3000 == 0 and it > 0:
            eta *= 0.96
        if eta < 0.002:
            eta = 0.002
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def perturb_polish(bf, bs, rng, dl, count=100):
    n = bf.shape[0]
    for i in range(count):

            break
