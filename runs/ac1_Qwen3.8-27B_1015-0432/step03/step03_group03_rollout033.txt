def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    ii = gen_inits(n, rng)
    bf, bs = None, np.inf
    for init in ii:
        sc = obj_val(init)
        if sc < bs:
            bs, bf = sc, init.copy()
    t1 = min(dl - 200.0, t0 + 380.0)
    f1, s1 = optimize(bf.copy(), t1, rng, 0.05, 0.02, False)
    if s1 < bs:
        bs, bf = s1, f1.copy()
    if time.monotonic() < dl - 150.0:
        f2 = norm_pos(0.5 * (bf + bf[::-1]))
        t2 = min(dl - 100.0, t0 + 680.0)
        f2o, s2o = optimize(f2, t2, rng, 0.04, 0.015, True)
        if s2o < bs:
            bs, bf = s2o, f2o.copy()
    if time.monotonic() < dl - 80.0:
        t3 = min(dl - 40.0, t0 + 1020.0)
        f3o, s3o = optimize(bf.copy(), t3, rng, 0.03, 0.012, False)
        if s3o < bs:
            bs, bf = s3o, f3o.copy()
    if time.monotonic() < dl - 10.0:
        for _ in range(100):
            if time.monotonic() > dl - 5.0:
                break
            tr = norm_pos(bf * np.exp(0.003 * rng.standard_normal(n)))
            sc = obj_val(tr)
            if sc < bs:
                bs, bf = sc, tr.copy()
    out = [float(x) for x in bf]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs) or not out:
        out = [1.0 / n] * n
    return out
