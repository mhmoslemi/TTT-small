def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    s = f.sum()
    if s < 1e-10:
        return 1e30
    b = fft_conv(f, f)
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad_sym(f, b, thresh_frac, max_pk):
    """Compute gradient for symmetric sequence using peak-weighted approach."""
    n = f.shape[0]
    M = b.max()
    thresh = M * thresh_frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(b)])
    L = 2 * n - 1
    h = np.zeros(L)
    w = b[peaks] - thresh  # weight by excess above threshold
    w = np.maximum(w, 0)
    if w.sum() < 1e-30:
        w = np.ones(len(peaks))
    h[peaks] = w / w.sum()

    # Use FFT: c = conv(f, h_flipped), take appropriate slice
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
