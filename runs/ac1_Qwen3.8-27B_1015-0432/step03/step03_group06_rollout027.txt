import numpy as np
import time

def fft_conv(a, b):
    N = 1 << (len(a) + len(b) - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N))[:len(a)+len(b)-1]

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / s / s if s > 1e-15 else 1e9

def nrm(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    return f / s if s > 1e-15 else np.ones_like(f) / f.size

def sgrad(f, pk, wt):
    n = f.size
    h = np.zeros(2 * n - 1)
    h[pk] = wt
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n-1:2*n-1][::-1]
