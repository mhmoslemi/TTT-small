    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def grad_at_peaks(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, sym=False, eta0=0.04, topk=120):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    no_imp = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        if len(b) <= topk:
            pk = np.arange(len(b))
        else:
            pk = np.argpartition(b, -topk)[-topk:]
        g = grad_at_peaks(f, pk)
        gmax = g.max()
        if gmax < 1e-15:
            break
        g /= gmax
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-30)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 10 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                no_imp = 0
            else:
                no_imp += 1
            if no_imp > 180:
                amp = 0.01 + 0.05 * min(1.0, no_imp / 1500.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-30)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_imp = 0
                eta = eta0
        if it > 30000:
            eta *= 0.99992
        eta = max(eta, 0.001)
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def perturb(bf, bs, rng, dl, n_trials=70, amp=0.004):
    n = bf.shape[0]
    for i in range(n_trials):
        if time.monotonic() > dl:
            break
        a = amp * (0.4 + 0.6 * (i + 1) / n_trials)
        tr = bf * np.exp(a * rng.standard_normal(n))
        tr = np.maximum(tr, 0.0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = score(tr)
        if sc < bs:
            bs = sc
            bf = tr.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p /= p.sum()

    bf = p.copy()
    bs = score(p)

    # Phase 1: Aggressive optimization from parent
    d1 = min(dl, t0 + 400)
    f1, s1 = optimize(bf, d1, rng, eta0=0.05, topk=150)
    if s1 < bs:
        bs, bf = s1, f1

    # Phase 2: Symmetric refinement
    if time.monotonic() < dl - 250:
        fs = 0.5 * (bf + bf[::-1])
        fs = np.maximum(fs, 0.0)
        ss = fs.sum()
        if ss > 1e-15:
            fs /= ss
        sc_s = score(fs)
        if sc_s < bs:
            bs, bf = sc_s, fs
        d2 = min(dl, t0 + 720)
        f2, s2 = optimize(bf.copy(), d2, rng, sym=True, eta0=0.03, topk=100)
        if s2 < bs:
            bs, bf = s2, f2

    # Phase 3: Asymmetric fine-tune
    if time.monotonic() < dl - 120:
        d3 = min(dl, t0 + 930)
        f3, s3 = optimize(bf.copy(), d3, rng, eta0=0.015, topk=80)
        if s3 < bs:
            bs, bf = s3, f3

    # Phase 4: Ultra-fine polish
    if time.monotonic() < dl - 40:
        f4, s4 = optimize(bf.copy(), dl - 15, rng, eta0=0.006, topk=40)
        if s4 < bs:
            bs, bf = s4, f4

    # Final perturbation sweep
    bf, bs = perturb(bf, bs, rng, dl - 2, n_trials=60, amp=0.003)

    result = [float(v) for v in bf]
    fs8 = evaluate_sequence(result)
    if not np.isfinite(fs8):
        result = [1.0 / n] * n
    return result
