from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _fpow2(s):
    m = 1
    while m < s:

    return m

    return 2.0 * n * mv / (S * S)

def _gradient(a, tau):
    n = a.size
    S = a.sum()
    c = _fft_conv(a)
    mv, p = _smooth_max(c, tau)
    # gradient of smooth_max w.r.t. a: need gradient of c = fft_conv(a) w.r.t. a
    # d(c)_k / da_i = 2 * a_{k-i} (if 0<=k-i<=n-1)
    # So grad of smooth_max = sum_k p_k * 2 * a_{k-i} = 2 * (p_reversed * a)_i
    # where p_reversed is p flipped (since c has length 2n-1, p has length 2n-1)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    # Wait, let me re-derive. 
    # d smooth_max / da_i = sum_k p_k * d(c_k)/da_i = sum_k p_k * 2*a_{k-i}
    # = 2 * sum_k p_k * a_{k-i}
    # Let q = p (length 2n-1). Then sum_k q_k * a_{k-i} is a cross-correlation.
    # sum_k q_k * a_{k-i} for k=0..2n-2: let m=k-i, sum_m q_{m+i} * a_m
    # This is (q_shifted * a) where q_shifted is q shifted by i... 
    # Actually, if we define the cross-correlation: (q ⊛ a)_i = sum_k q_k a_{k-i}
    # This can be computed as: ifft(fft(q, L) * conj(fft(a, L))).real
    # But since a is real: conj(fft(a)) = fft(a[::-1] padded)... 
    # Simpler: (q ⊛ a)_i = sum_k q_k a_{k-i} = (flipped_q * a)_i where flipped_q = q[::-1]
    # (flipped_q * a)_i = sum_k flipped_q_k a_{i-k} = sum_k q_{len-1-k} a_{i-k}
    # Let j = len-1-k: sum_j q_j a_{i - len + 1 + j}
    # Hmm, this is getting confusing. Let me use the direct approach.
    
    # Actually, the standard result: if c = a * a (convolution), then
    # d(c_k)/da_i = a_{k-i} + a_{k-i} = 2a_{k-i} (for valid indices)
    # So d(smooth_max)/da = 2 * (p_reversed :convolution: a) but only the middle part
    
    # Let me just compute it as: g_i = 2 * sum_k p_k * a_{k-i} for 0<=k-i<n
    # = 2 * (p * a)_{n-1+i} ... no.
    # sum_{k=max(i,0)}^{min(2n-2, i+n-1)} p_k * a_{k-i}
    # Let j = k-i: sum_j p_{j+i} * a_j for j=0..n-1
    # This is the cross-correlation of p and a, evaluated at lag i.
    # = (ifft(fft(p, L) * conj(fft(a, L))).real)[i] ... but need to be careful with indexing.
    
    # Alternative: use the fact that the gradient of ||a*a||_inf w.r.t. a at the point where
    # the max is at k* is 2 * flip(a)[k*-i] = 2*a[k*-i].
    # For smooth max: gradient_i = 2 * sum_k p_k * a_{k-i} (when valid).
    
    # Computation: let's define r = p (length 2n-1). 
    # g_i = 2 * sum_{k=0}^{2n-2} r_k * a_{k-i} * [0<=k-i<=n-1]
    # = 2 * sum_{j=0}^{n-1} r_{j+i} * a_j (where j=k-i, so k=j+i)
    # = 2 * (r_shifted ⊛ a) where r_shifted[j+i] = r[j+i]
    # This is: g_i = 2 * (the i-th element of the cross-correlation of r and a)
    # Cross-correlation: (r ⊛ a)_i = sum_j r_{j+i} * a_j
    # = ifft(fft(r, L) * conj(fft(a, L))).real ... 
    # Actually: cross-correlation can be computed as:
    # (r ⊛ a)_i = sum_j r_{j+i} a_j = (r_reversed * a)_{(len_r-1) - i + ...}
    # This is getting confusing with indexing. Let me just use:
    # g_i = 2 * sum_j a_j * p_{j+i}
    # Let's compute h = ifft(fft(p_padded, L) * fft(a_padded, L)).real
    # where p_padded = p (length 2n-1, padded to L)
    # h_k = sum_j p_j * a_{k-j} (circular, but with enough padding it's linear)
    # We want g_i/2 = sum_j p_{j+i} * a_j = h_{...}
    # h_k = sum_j p_j a_{k-j}. Set k-j = m, j = k-m: h_k = sum_m p_{k-m} a_m
    # We want sum_j p_{j+i} a_j. Set j=m: sum_m p_{m+i} a_m.
    # From h_k = sum_m p_{k-m} a_m, we want k-m = m+i, i.e., k = 2m+i. That's not a simple shift.
    
    # OK let me try differently. Define q = p[::-1] (reversed p, length 2n-1).
    # Then the convolution q * a at position i: (q*a)_i = sum_j q_j a_{i-j}
    # = sum_j p_{2n-2-j} a_{i-j}
    # Let m = 2n-2-j: = sum_m p_m a_{i-2n+2+m}
    # For this to equal sum_j p_{j+i} a_j, we need 2n-2-j + i - j = ... this is getting too complicated.
    
    # Let me just use the formula from the previous program, which works:
    # pr = p[::-1], then convolve with a, take the middle part, reverse.
    # The previous program does:
    # pr = p[::-1]
    # cv = ifft(fft(a, L) * fft(pr, L)).real
    # g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    
    # Let me verify this formula. pr = p[::-1], so pr_j = p_{2n-2-j}.
    # cv_k = sum_j a_j pr_{k-j} = sum_j a_j p_{2n-2-k+j}
    # For the full linear convolution, k ranges 0 to 3n-4 (a has length n, pr has length 2n-1).
    # We take cv[n-1 : 2n-2] (n elements) and reverse.
    # cv[n-1+i] = sum_j a_j p_{2n-2-(n-1+i)+j} = sum_j a_j p_{n-1-i+j}
    # Reversed: g_i_orig = cv[n-2-i] for i=0..n-1 (after reversal, index i corresponds to original 2n-2-(n-1+i) = n-1-i)
    # Wait, the slice cv[n-1:2n-1] has elements cv[n-1], cv[n], ..., cv[2n-2]. After [::-1], element at position i is cv[2n-2-i].
    # cv[2n-2-i] = sum_j a_j p_{2n-2-(2n-2-i)+j} = sum_j a_j p_{i+j}
    # So g_i = 2 * sum_j a_j p_{i+j} = 2 * sum_j a_j p_{j+i}. ✓ This matches!
    
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def _normalize(a, scale):
    s = a.sum()
    if s < 1e-8:
        return a.copy()
    return a / s * scale
