import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def compute_grad(f, b, n_peaks=400, width=0.03):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    thresh = M * (1.0 - width)
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > n_peaks:
        idx = np.argsort(b[peaks])[-n_peaks:]
        peaks = peaks[idx]
    if len(peaks) < 1:
        peaks = np.array([np.argmax(b)])
    # Soft weights: normalized distance to max
    w = (b[peaks] - thresh) / (M - thresh + 1e-30)
    w = np.maximum(w, 0.1)  # minimum weight
    w = w / w.sum()
    h = np.zeros(L)
    h[peaks] = w
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, sym=False, fine=False):
    n = f.shape[0]
    f = normalize(f.copy())
    best_f = f.copy()
    best_sc = score(f)
    if fine:
        eta = 0.015
        n_peaks = 200
        width = 0.01
    else:
        eta = 0.06
        n_peaks = 400
        width = 0.04
    no_improv = 0
    restarts = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        # Adaptive width: start wide, narrow down
        w_adj = width * (1.0 + 0.5 * max(0, 1 - it / 2000.0))
        g = compute_grad(f, b, n_peaks=n_peaks, width=w_adj)
        gmax = g.max()
        if gmax < 1e-30:
            break
        gn = g / gmax
        # Line search over several step sizes
        best_eta = eta
        best_sc_t = np.inf
        best_tr = f.copy()
        candidates = [eta * 0.5, eta, eta * 1.5, eta * 2.0]
        for te in candidates:
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-20)
            if sym:
                tr = 0.5 * (tr + tr[::-1])
            tr = normalize(tr)
            sc = score(tr)
            if sc < best_sc_t:
                best_sc_t = sc
                best_eta = te
                best_tr = tr.copy()
        f = best_tr
        # Adapt step size
        if best_eta > eta * 1.05:
            eta = min(0.25, eta * 1.12)
        elif best_eta < eta * 0.95:
            eta = max(0.005, eta * 0.82)
        # Track best
        if best_sc_t < best_sc - 1e-12:
            best_sc = best_sc_t
            best_f = best_tr.copy()
            no_improv = 0
        else:
            no_improv += 1
        # Restart
        if no_improv > (150 if not fine else 300):
            scl = 0.03 + 0.05 * min(1.0, restarts / 6.0)
            f = best_f * np.exp(scl * rng.standard_normal(n))
            f = np.maximum(f, 1e-20)

            f = normalize(f)
