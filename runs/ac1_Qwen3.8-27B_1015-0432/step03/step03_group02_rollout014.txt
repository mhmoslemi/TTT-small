def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def get_gradient(f, b, M, L, n, alpha):
    """Compute gradient contribution using soft active-set weighting."""
    # Soft weight: peaks above threshold get weighted by excess
    thresh = M * (1.0 - alpha)
    w = np.zeros(L)
    above = b >= thresh
    cnt = above.sum()
    if cnt < 1:
        w[np.argmax(b)] = 1.0
    elif cnt <= 512:
        w[above] = (b[above] - thresh) / (M - thresh + 1e-30)
    else:
        idx = np.argpartition(b, -300)[-300:]
        w2 = np.zeros(L)
        w2[idx] = (b[idx] - thresh) / (M - thresh + 1e-30)
        w = w2
    wsum = w.sum()
    if wsum < 1e-30:
        return None
    w /= wsum
    # Gradient via circular-shifted convolution
    c = fft_conv(f, w[::-1])
    g = c[n - 1:2 * n - 1][::-1]
    np.maximum(g, 0.0, out=g)
    gmax = g.max()
    if gmax < 1e-30:
        return None
    return g / gmax

def optimize(f, deadline, rng, max_alpha=0.01, eta_init=0.12, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    L = 2 * n - 1
    eta = eta_init
    alpha = 0.06
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        g = get_gradient(f, b, M, L, n, alpha)
        if g is None:
            break
        f *= np.exp(-eta * g)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        # Sharpen active set over time
        alpha = max(max_alpha, alpha * 0.9985)
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                if sc < best_sc:
                    best_sc = sc
                    best_f = f.copy()
                patience = 0
            else:
                patience += 1
                eta *= 0.99
            if patience > 500:
                sc_r = 0.012 + 0.015 * min(1.0, patience / 800.0)
                f = best_f * np.exp(sc_r * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = eta_init * 0.6
                alpha = 0.05
        if it > 600:
            eta *= 0.999
    return best_f, best_sc

def fine_tune(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    for i in range(300):
        if time.monotonic() > deadline:
            break
        for _ in range(10):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.9
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)

    inits = []
    inits.append(p / p.sum())
    ps = 0.5 * (p + p[::-1])
    ps = np.maximum(ps, 0.01)
    inits.append(ps / ps.sum())
    x = np.arange(n, dtype=np.float64)
    for pw in [0.5, 1.0, 1.5]:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        v = d ** pw
        inits.append(v / v.sum())
    for _ in range(3):
        r = 1.0 + 0.25 * rng.standard_normal(n)
        np.maximum(r, 0.1, out=r)
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())

    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()

    # Phase 1: broad optimization
    phase1_end = min(deadline, t0 + 350.0)
    f1, s1 = optimize(best_f.copy(), phase1_end, rng, max_alpha=0.015, eta_init=0.12)
    if s1 < best_sc:
        best_sc, best_f = s1, f1

    # Phase 2: refine with sharper active set
    if time.monotonic() < deadline - 150.0:
        phase2_end = min(deadline, t0 + 700.0)
        f2, s2 = optimize(best_f.copy(), phase2_end, rng, max_alpha=0.005, eta_init=0.06)
        if s2 < best_sc:
            best_sc, best_f = s2, f2

    # Phase 3: targeted restarts
    if time.monotonic() < deadline - 100.0:
        for restart in range(4):
            if time.monotonic() > deadline - 40.0:
                break
            sc_r = 0.01 + 0.01 * restart
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            f_r, s_r = optimize(f_r, deadline - 35.0, rng, max_alpha=0.004, eta_init=0.05)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r

    # Phase 4: final fine-tune
    if time.monotonic() < deadline - 10.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 5.0, rng)

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
