import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def compute_grad(f, b, thresh):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    spread = max(1.0 - thresh, 1e-9)
    w = np.clip((b - thresh * M) / (M * spread), 0.0, None)
    w = w * w
    wsum = w.sum()
    if wsum < 1e-20:
        w = np.zeros(L)
        w[int(np.argmax(b))] = 1.0
    wr = f[::-1]
    c = fft_conv(wr, w)
    return np.maximum(2.0 * c[n-1:2*n-1], 0.0)
