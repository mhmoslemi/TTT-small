                    f = best_f * np.exp(ns * rng.standard_normal(n))
                    np.maximum(f, 1e-30, out=f)
                    f /= f.sum()
                    patience = 0
                    eta = 0.12
            if it > 300:
                eta = max(0.02, eta * 0.999)
    return best_f, best_sc

def fine_tune(f, deadline, rng, steps=300):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.003
    for i in range(steps):
        if time.monotonic() > deadline:
            break
        # Use subgradient for a few steps
        b = fft_conv(best_f, best_f)
        g = subgradient(best_f, b, 400, 80.0)
        if g is not None:
            gmax = g.max()
            if gmax > 1e-30:
                trial = best_f * np.exp(0.005 * g / gmax)
                # Wait, this INCREASES f where g is large. That's wrong.
                # We want to DECREASE f where g is large (peaks in convolution)
                trial = best_f * np.exp(-0.005 * g / gmax)
                np.maximum(trial, 1e-30, out=trial)
                trial /= trial.sum()
                sc = score(trial)
                if sc < best_sc - 1e-13:
                    best_sc = sc
                    best_f = trial.copy()
        # Random perturbation
        for _ in range(3):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score(trial)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.94
    return best_f, best_sc
