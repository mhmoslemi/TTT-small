import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def lp_grad(f):
    """LP-dual based gradient: continuous weighting of peaks via interval LP in O(n)."""
    n = len(f)
    y = np.zeros(2 * n - 1)
    running = 0.0
    for j in range(n - 1, -1, -1):
        vf = f[j] if f[j] > 1e-10 else 1e-10
        y[j] = max(0.0, 1.0 / vf - running)
        running = y[j] + running - y[j + n - 1]
    w = y[::-1]
    conv = fft_conv(f, w)
    g = conv[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def peak_grad(f):
    """Fallback: top-peak subgradient."""
    n = f.shape[0]
    b = fft_conv(f, f)
    M = b.max()
    pk = np.argsort(b)[-120:]
    L = 2 * n - 1
    h = np.zeros(L)
    h[pk] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, eta0=0.05, sym=False, max_it=500000):

    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        if it % 6 < 5:
            g = lp_grad(f)
        else:
            g = peak_grad(f)
        gm = g.max()
        if gm < 1e-30:
            break
        g = g / gm
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-14)

            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        if it % 8 == 0:
            sc = score(f)
            if sc < bs - 1e-13:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 150:
                amp = 0.004 + 0.025 * min(1.0, ni / 2000.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-14)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                ni = 0
                eta = eta0
        if it > 20000:
            eta *= 0.99995
        eta = max(eta, 0.003)
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def perturb(bf, bs, rng, dl, count=100, amp=0.003):
    n = bf.shape[0]
    for i in range(count):
        if time.monotonic() > dl:
            break
        tr = bf * np.exp(amp * rng.standard_normal(n))
        tr = np.maximum(tr, 0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = score(tr)
        if sc < bs:
            bs = sc
            bf = tr.copy()
    return bf, bs
