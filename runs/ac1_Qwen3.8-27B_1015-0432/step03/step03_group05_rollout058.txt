```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, sym=False, max_it=500000, eta0=0.04, 
             peak_frac_start=0.999, peak_frac_end=0.9999, min_peaks=5, max_peaks=300):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0  # no-improve counter
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        # Adaptive threshold
        t = min(1.0, it / 30000.0)
        frac = peak_frac_start + (peak_frac_end - peak_frac_start) * t
        thr = M * frac
        pk = np.where(b >= thr)[0]
        if len(pk) < min_peaks:
            pk = np.argsort(b)[-min_peaks:]
        elif len(pk) > max_peaks:
            pk = np.argsort(b)[-max_peaks:]
        g = subgrad(f, pk)
        gm = g.max()
        if gm < 1e-30:
            break
        # Multiplicative update
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f = f / f.sum()
        # Periodic scoring
        if it % 5 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 200:
                # Perturb
                amp = 0.005 + 0.01 * min(1.0, ni / 1000.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = f / f.sum()
                ni = 0
                eta = eta0
        # Learning rate decay
        if it > 20000:
            eta *= 0.99995
        eta = max(eta, 0.001)
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs
```

```python
        f2, s2 = run_opt(fs, p2, rng, True, 400000, 0.03, 0.999, 0.9999, 3, 200, 5, 250)
        if s2 < bs:
            bs, bf = s2, f2

    # Phase 3: fine asymmetric
    if time.monotonic() < dl - 150:
        p3 = min(dl, t0 + 920)
        f3, s3 = run_opt(bf.copy(), p3, rng, False, 400000, 0.015, 0.9995, 0.99999, 3, 100, 5, 300)
        if s3 < bs:
            bs, bf = s3, f3

    # Phase 4: ultra-fine polish
    if time.monotonic() < dl - 50:
        f4, s4 = run_opt(bf.copy(), dl - 15, rng, False, 400000, 0.005,
                         0.9999, 0.999999, 2, 30, 3, 400)
        if s4 < bs:
            bs, bf = s4, f4

    # Final micro-perturbation
    for i in range(80):
        if time.monotonic() > dl - 3:
            break
        for amp in (0.002, 0.005, 0.01):
            tr = bf * np.exp(amp * rng.standard_normal(n))
            tr = np.maximum(tr, 0)
            ts = tr.sum()
            if ts < 1e-15:
                continue
            tr /= ts
            sc = score(tr)
            if sc < bs:
                bs = sc
                bf = tr.copy()

    result = [float(v) for v in bf]
    fs_test = evaluate_sequence(result)
    if not np.isfinite(fs_test):
        result = [1.0 / n] * n
    return result
```
