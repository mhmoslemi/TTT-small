        T = M * (0.01 + 0.19 * np.exp(-it / 5000.0))
        w = soft_w(b, T)
        g = grad_fn(f, w)
        gm = g.max()
        if gm < 1e-25:
            break
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        eta *= 0.9999
        if it % 25 == 0:
            sc = get_score(f)
            if sc < bs - 5e-15:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 350:
                amp = 0.05 + 0.10 * min(1.0, nr / 10.0)
                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                ni = 0
                nr += 1
                eta = min(0.40, eta * 1.5)
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 20.0
    n = len(height_sequence_1)
    x = np.arange(n, dtype=np.float64)

    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + 0.5) / (n + 1)))
    inits.append(h / h.sum())
    dm = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in (0.5, 1.0, 2.0):
        tv = dm ** pw
        inits.append(tv / tv.sum())
    sv = np.sin(np.pi * (x + 0.5) / n)
    inits.append((sv * sv) / np.sum(sv * sv))
    for _ in range(2):
        rv = np.abs(rng.standard_normal(n)) + 0.05
        rv = 0.5 * (rv + rv[::-1])
        inits.append(rv / rv.sum())

    bf = inits[0]
    bs = 1e30
    for ini in inits:
        sc = get_score(ini)
        if sc < bs:
            bs = sc
            bf = ini.copy()

    e1 = min(dl, t0 + 350.0)
    f1, s1 = run_opt(bf.copy(), e1, rng)
    if s1 < bs:
        bs, bf = s1, f1

    if time.monotonic() < dl - 100.0:
        fs = 0.5 * (bf + bf[::-1])
        fs /= fs.sum()
        e2 = min(dl, t0 + 720.0)
        f2, s2 = run_opt(fs, e2, rng, sym=True)
        if s2 < bs:
            bs, bf = s2, f2

    if time.monotonic() < dl - 30.0:
        f3, s3 = run_opt(bf.copy(), dl - 8.0, rng)
        if s3 < bs:
            bs, bf = s3, f3

    if time.monotonic() < dl - 1.0:
        for _ in range(300):
            if time.monotonic() > dl - 0.5:
                break
            for am in (0.002, 0.008, 0.02):
                tr = bf * np.exp(am * rng.standard_normal(n))
                tr = np.maximum(tr, 1e-15)
                tr /= tr.sum()
                st = get_score(tr)
                if st < bs:
                    bs, bf = st, tr
                    break

    res = bf.copy()
    if res.sum() < 0.01:
        res *= 0.02 / res.sum()
    out = [float(v) for v in res]
    fs = evaluate_sequence(out)
    if not np.isfinite(fs):
        out = [1.0 / n] * n
    return out
