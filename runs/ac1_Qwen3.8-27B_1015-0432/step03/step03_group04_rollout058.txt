def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2 + 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, frac, max_pk=500):
    n = f.shape[0]
    M = b.max()
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]

    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]

The full convolution: `b[k] = sum_{i=0}^{n-1} a[i] * a[k-i]` for `k = 0, ..., 2n-2`.

The gradient of `b[k]` with respect to `a[j]`:

No wait. `b[k] = sum_i a[i] * a[k-i]`. 

∂b[k]/∂a[j] = (∂/∂a[j]) sum_i a[i] * a[k-i]

The terms that involve a[j]:
- i = j: a[j] * a[k-j] → contribution: a[k-j]
- k-i = j (i.e., i = k-j): a[k-j] * a[j] → contribution: a[k-j]

If j ≠ k-j (i.e., k ≠ 2j), these are two distinct terms, so ∂b[k]/∂a[j] = 2a[k-j].
If j = k-j (k = 2j), it's the same term counted once, so ∂b[k]/∂a[j] = 2a[j]... wait no.

If k = 2j, then the term i = j gives a[j]*a[j] = a[j]², and its derivative w.r.t. a[j] is 2a[j].

The second condition k-i = j means i = k-j = 2j-j = j, so it's the same term. So the total derivative is 2a[j].

So in all cases: ∂b[k]/∂a[j] = 2a[k-j] if 0 <= k-j < n, else 0.

The subgradient of f(a) = max_k b[k] at the active set K (where b is achieved) is:
`∇f = sum_{k∈K} w_k * ∇b[k]` where `w_k >= 0`, `sum w_k = 1`.

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2 + 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, frac, max_pk=500):
    n = f.shape[0]
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    thresh = M * frac
    peaks = np.where(b >= thresh)[0]
    if len(peaks) > max_pk:
        idx = np.argsort(b[peaks])[-max_pk:]
        peaks = peaks[idx]
    if len(peaks) == 0:
        peaks = np.array([np.argmax(b)])
    L = 2 * n - 1
    h = np.zeros(L)
    w = b[peaks]
    w = w - w.min() + 1e-10
    h[peaks] = w * w
    h_s = h.sum()
    if h_s < 1e-30:
        h[np.argmax(b)] = 1.0
    else:
        h /= h_s
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
