def _idx_active(c, frac=0.02, max_k=300):
    cm = c.max()
    if cm < 1e-15:
        return np.arange(c.size)
    thr = cm * (1.0 - frac)
    idx = np.where(c >= thr)[0]
    if idx.size == 0:
        idx = np.array([int(np.argmax(c))])
    if idx.size > max_k:
        order = np.argsort(c[idx])[::-1]
        idx = idx[order[:max_k]]
    return idx

def _obj(a, tau, idx):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv(a)
    ci = c[idx]
    cm = ci.max()
    d = np.maximum((ci - cm) / tau, -50.0)
    e = np.exp(d)
    smax = cm + tau * np.log(e.sum())
    return float(2.0 * n * smax / (S * S))

def _grad(a, tau, idx):
    n = a.size
    S = a.sum()
    c = _conv(a)
    ci = c[idx]
    cm = ci.max()
    d = np.maximum((ci - cm) / tau, -50.0)
    e = np.exp(d)
    ws = e / e.sum()
    smax = cm + tau * np.log(e.sum())
    p = np.zeros(2 * n - 1)
    p[idx] = ws
    q = p[::-1]
    L = _fpow2(3 * n)
    cv = ifft(fft(a, L) * fft(q, L)).real
    gc = 2.0 * cv[n - 1:2 * n - 1][::-1]
    g = (2.0 * n / (S * S)) * gc - (4.0 * n * smax / (S ** 3))
    return g

def _normalize(a, target):
    s = a.sum()
    if s < 1e-8:
        return np.ones_like(a) * target / a.size
    return a / s * target

def _eval(cand):

def _subgrad_polish(a, max_iter=200, eta0=0.02):
    n = a.size
    best_a = a.copy()
    best_sc = _eval(a)
    eta = eta0
    for it in range(max_iter):
        c = _conv(a)
        km = int(np.argmax(c))
        cm = c[km]
        S = a.sum()
        gc = np.zeros(n)
        jlo = max(0, km - n + 1)
        jhi = min(n - 1, km)
        js = np.arange(jlo, jhi + 1)
        gc[js] = 2.0 * a[km - js]
        g = (2.0 * n / (S * S)) * gc - (4.0 * n * cm / (S ** 3))
        a_new = np.clip(a - eta * g, 0.0, 1000.0)
        a_new = _normalize(a_new, np.sqrt(2.0 * n))
        sc = _eval(a_new)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.03, 0.05)
        else:
            a = best_a.copy()
            eta *= 0.75
            if eta < 1e-6:
                break
    return best_a, best_sc

def _make_inits(par, n, rng, scale):
    inits = []
    inits.append(par.copy())
    # Symmetrized
    sym = 0.5 * (par + par[::-1])
    sym = np.clip(sym, 0, None)
    if sym.sum() > 0.01:
        inits.append(_normalize(sym, scale))
    # Slight perturbations of parent
    for amp in [0.02, 0.05, 0.1]:
        p = par * np.exp(rng.normal(0, amp, n))
        p = np.clip(p, 0, 1000)
        if p.sum() > 0.01:
            inits.append(_normalize(p, scale))
    # Bimodal
    x = np.arange(n, dtype=np.float64)
    for gap in [0.2, 0.4, 0.6]:
        w = int(n * (1 - gap) / 2)
        b = np.zeros(n)
        b[:w] = 1.0
        b[n-w:] = 1.0
        if b.sum() > 0.01:
            inits.append(_normalize(b, scale))
    # Cosine bump
    for amp in [0.3, 0.6, 0.9]:
        c = 1.0 + amp * np.cos(2.0 * np.pi * x / n)
        c = np.clip(c, 0, None)
        if c.sum() > 0.01:
            inits.append(_normalize(c, scale))
    # Gaussian
    for sig in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*n) / (sig*n))**2)
        if g.sum() > 0.01:
            inits.append(_normalize(g, scale))
    return inits
