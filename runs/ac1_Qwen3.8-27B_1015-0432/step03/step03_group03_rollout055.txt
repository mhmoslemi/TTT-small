    size = 1 << (n + m - 2).bit_length()

    return result[:n + m - 1]

    n = len(a)
    conv = fft_conv(a, a)
    # Soft-max weights

    size = 1 << (n + m - 2).bit_length()

    s = f.sum()
    if s < 1e-15:
        return np.inf
    conv = fft_conv(f, f)
    return 2.0 * n * conv.max() / (s * s)

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

    n = len(a)
    conv = fft_conv(a, a)
    sh = conv - conv.max()

    w /= w.sum()
    p_full = fft_conv(a, w[::-1])
