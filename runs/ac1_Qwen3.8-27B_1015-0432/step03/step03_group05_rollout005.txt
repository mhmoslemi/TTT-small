import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score_fn(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def gradient_peaks(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def gradient_softmax(f, beta):
    n = f.shape[0]
    L = 2 * n - 1
    b = fft_conv(f, f)
    bm = b.max()
    w = np.exp(beta * (b - bm))
    w /= w.sum()
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]
