from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

    S = a.sum()
    if S < 1e-8:
        return 1e10
    L = _fpow2(2 * n - 1)

    c = ifft(fa * fa).real[:2 * n - 1]
