        h = np.zeros(L)
        h[np.argmax(weights)] = 1.0
    else:
        h = weights / weights.sum()
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
