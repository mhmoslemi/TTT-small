    half = n // 2
    two[:half] = 0.6 + 0.4 * np.cos(np.pi * np.arange(half) / max(half, 1))
    two[half:] = 0.6 - 0.4 * np.cos(np.pi * np.arange(n - half) / max(n - half + 1, 1))
    inits.append(normalize(np.maximum(two, 0.01)))

    return inits

def local_polish(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = eval_score(f)
    eps_schedule = [0.002, 0.001, 0.0005, 0.0002]

    for ep in eps_schedule:
        while time.monotonic() < deadline:
            improved = False
            for _ in range(300):
                if time.monotonic() > deadline:
                    break
                trial = best_f * np.exp(ep * rng.standard_normal(n))
                trial = normalize(trial)
                sc = eval_score(trial)
                if sc < best_sc:
                    best_sc = sc
                    best_f = trial.copy()
                    improved = True
            if not improved:
                break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 35.0
    n = len(height_sequence_1)

    inits = gen_inits(n, rng)
    best_f = inits[0]
    best_sc = eval_score(best_f)
    for init in inits[1:]:
        sc = eval_score(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    t1 = min(deadline - 400.0, t0 + 450.0)
    f1, sc1 = subgrad_optimize(best_f.copy(), t1, rng, n_peaks=120, eta0=0.08)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()

    if time.monotonic() < deadline - 250.0:
        t2 = min(deadline - 200.0, t0 + 750.0)
        f2, sc2 = subgrad_optimize(best_f.copy(), t2, rng, n_peaks=250, eta0=0.04)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2.copy()

    if time.monotonic() < deadline - 120.0:
        t3 = min(deadline - 80.0, t0 + 950.0)
        f3, sc3 = subgrad_optimize(best_f.copy(), t3, rng, n_peaks=400, eta0=0.015)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3.copy()

    if time.monotonic() < deadline - 20.0:
        fp, scp = local_polish(best_f.copy(), min(deadline - 10.0, t0 + budget_s - 45.0), rng)
        if scp < best_sc:
            best_sc = scp
            best_f = fp.copy()

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
