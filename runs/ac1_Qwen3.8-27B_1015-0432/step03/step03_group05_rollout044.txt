    inits.append(v / v.sum())
    v = 0.5 * (1.0 - np.cos(np.pi * x / (n - 1))) + 0.01
    inits.append(v / v.sum())
    bf, bs = inits[0], get_score(inits[0])
    for ini in inits[1:]:
        sc = get_score(ini)
        if sc < bs:
            bs, bf = sc, ini.copy()
    p1 = min(dl - 280, t0 + 400)
    f1, s1 = run_opt(bf, p1, rng, eta0=0.055)
    if s1 < bs:
        bs, bf = s1, f1
    if time.monotonic() < dl - 220:
        p2 = min(dl - 160, t0 + 680)
        f2, s2 = run_opt(bf.copy(), p2, rng, eta0=0.035, soft_alpha=200.0)
        if s2 < bs:
            bs, bf = s2, f2
    if time.monotonic() < dl - 120:
        fs = np.maximum(0.5 * (bf + bf[::-1]), 0)
        ss = fs.sum()
        if ss > 1e-15:
            fs /= ss
        p3 = min(dl - 60, t0 + 900)
        f3, s3 = run_opt(fs, p3, rng, sym=True, eta0=0.025)
        if s3 < bs:
            bs, bf = s3, f3
    if time.monotonic() < dl - 35:
        p4 = min(dl - 5, t0 + 1060)
        f4, s4 = run_opt(bf.copy(), p4, rng, eta0=0.006, max_pk=200)
        if s4 < bs:
            bs, bf = s4, f4
    if time.monotonic() < dl - 3:
        for _ in range(400):
            if time.monotonic() > dl:
                break
            for amp in (0.002, 0.005, 0.01):
                ft = bf * np.exp(amp * rng.standard_normal(n))
                ft = np.maximum(ft, 0)
                st = ft.sum()
                if st > 1e-15:
                    ft /= st
                    sc = get_score(ft)
                    if sc < bs:
                        bs, bf = sc, ft.copy()
    result = [float(v) for v in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
