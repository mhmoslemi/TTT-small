import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def optimize(f, deadline, rng, eta0=0.09, max_peaks=500, sym=False, temp0=0.08):

    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    temp = temp0
    no_improv = 0
    restart_count = 0
    momentum = np.zeros(n)

    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-25:
            break

        thresh = M * (0.55 + 0.35 * min(1.0, no_improv / 4000.0))
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > max_peaks:
            idx = np.argsort(b[peaks])[::-1]
            peaks = peaks[idx[:max_peaks]]
        if peaks.size < 1:
            peaks = np.array([np.argmax(b)])

        vals = b[peaks]
        w = np.exp((vals - M) / (temp * M + 1e-30))
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.ones_like(w) / len(w)
        else:
            w = w / wsum

        g = compute_subgrad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:

        momentum = 0.8 * momentum + 0.2 * g

                best_f = f.copy()
                no_improv = 0

                f = best_f * np.exp(scale * rng.standard_normal(n))

                if sym:

                f = norm_pos(f)
                momentum = np.zeros(n)
                no_improv = 0
                restart_count += 1
                eta = min(0.4, eta * 1.5)
                temp = max(0.01, temp * 0.7)

        eta *= 0.99975
        temp *= 1.002

    return best_f, best_sc
