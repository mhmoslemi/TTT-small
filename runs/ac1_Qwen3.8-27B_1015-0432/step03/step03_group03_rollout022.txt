def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def full_subgrad(f, b, beta):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    w = np.exp(-beta * (M - b) / M)
    w /= w.sum()  # normalize weights
    size = 1 << (3 * n - 1).bit_length()
    w_rev = w[::-1]
    c_full = np.fft.irfft(np.fft.rfft(f, size) * np.fft.rfft(w_rev, size))[:L]
    H = c_full[L-1:L-1-n:-1]
    g = 2.0 * H
    return np.maximum(g, 0.0)

    M = b.max()
