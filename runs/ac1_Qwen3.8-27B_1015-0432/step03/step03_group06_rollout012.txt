`0.01 * lap_smooth(f)` has magnitude O(f_max / n) for smooth, O(f_max) for spikes.

After normalization by gmax, the smoothing contribution is O(1/n) ≈ 0.0002 for smooth functions. That's very small. For spikes, it could be O(1), which would be significant.

I think 0.01 is fine. It mainly affects spiky components, which is what we want.

Let me go with 0.01.

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def compute_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def weighted_subgrad(f, b, alpha, top_k):
    n = len(f)
    L = 2 * n - 1
    b_max = b.max()
    if b_max < 1e-30:
        return np.zeros(n)
    idx = np.argsort(b)[-top_k:]
    w = np.zeros(L)
    w[idx] = (b[idx] / b_max) ** alpha
    h_rev = w[::-1]
    c = fft_conv(f, h_rev)
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def lap_smooth(f):
    n = len(f)
    if n < 3:
        return np.zeros(n)
    L = np.zeros(n)
    L[1:-1] = f[1:-1] - 0.5 * (f[:-2] + f[2:])
    L[0] = f[0] - f[1]
    L[-1] = f[-1] - f[-2]
    return L
