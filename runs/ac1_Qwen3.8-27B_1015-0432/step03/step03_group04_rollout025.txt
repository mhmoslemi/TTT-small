            tr /= tr.sum()
            sc = score(tr)
            if sc < best_sc:
                best_sc, best_f = sc, tr.copy()
    res = np.clip(best_f, 0.0, 1000.0)
    ls = res.sum()
    if ls < 0.01:
        res = np.full(n, 0.02 / n)
    rl = [float(v) for v in res]
    if not np.isfinite(evaluate_sequence(rl)):
        rl = [1.0 / n] * n
    return rl
