    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0  # no-improvement counter
    g_mom = np.zeros(n)  # momentum
    L = 2 * n - 1
    
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        
        # Adaptive peak selection
        frac = peak_frac_start + (peak_frac_end - peak_frac_start) * min(1.0, it / 100000.0)
        thr = M * frac
        pk = np.where(b >= thr)[0]
        
        # Cap number of peaks
        max_pk = 1000
        if len(pk) > max_pk:
            pk = pk[np.argsort(b[pk])[-max_pk:]]
        if len(pk) < 5:
            pk = np.argpartition(b, -20)[-20:]
        
        # Soft weights: proportional to (b/M) at active peaks
        w = np.zeros(L)
        w[pk] = (b[pk] / M) ** 2  # Square for sharper focus
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w /= w_sum
        
        # Compute gradient
        g = compute_grad(f, w)
        gmax = g.max()
        if gmax < 1e-30:
            break
        g = g / gmax
        
        # Momentum
        g_mom = 0.6 * g_mom + 0.4 * g
        g_use = g_mom
        gmax2 = g_use.max()
        if gmax2 > 1e-30:
            g_use = g_use / gmax2
        
        # Multiplicative update
        f = f * np.exp(-eta * g_use)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        
        # Score check every few iterations
        if it % 5 == 0:
            sc = score(f)
            if sc < bs - 1e-15:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 500:
                # Restart with perturbation
                f = bf * np.exp(0.005 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                g_mom = np.zeros(n)
                ni = 0
                eta = eta0
        
        # Learning rate schedule
        if it > 20000:
            eta *= 0.9999
        eta = max(eta, 0.002)
    
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def local_polish(f, deadline, max_rounds=50):
    """Try small adjustments to reduce the max peak."""
    n = f.shape[0]
    bf = f.copy()
    bs = score(f)
    for r in range(max_rounds):
        if time.monotonic() > deadline:
            break
        b = fft_conv(bf, bf)
        M = b.max()
        pk = np.where(b >= 0.99999 * M)[0]
        if len(pk) == 0:
            break
        # Try reducing elements that contribute most to the max peak
        # For peak at lag k, the contribution of f[j] is f[k-j]
        # So reduce f[j] where f[k-j] is large for k in pk
        contrib = np.zeros(n)
        for k in pk[:50]:  # Limit to top 50 peaks
            for j in range(n):
                if 0 <= k - j < n:
                    contrib[j] += bf[k - j]
                if 0 <= j - k < n:
                    contrib[j] += bf[j - k]
        # Reduce high-contribution elements slightly
        top = np.argsort(contrib)[-min(200, n // 10):]
        f_new = bf.copy()
        delta = 0.001 * bf[top].mean()
        f_new[top] -= delta
        f_new = np.maximum(f_new, 0)
        if f_new.sum() < 1e-15:
            break
        f_new /= f_new.sum()
        sc = score(f_new)
        if sc < bs:
            bs = sc
            bf = f_new.copy()
    return bf, bs

def perturb_best(bf, bs, rng, dl, count=80, amp=0.004):
    n = bf.shape[0]
    for i in range(count):
        if time.monotonic() > dl:
            break
        tr = bf * np.exp(amp * rng.standard_normal(n))
        tr = np.maximum(tr, 0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = score(tr)
        if sc < bs:
            bs = sc
            bf = tr.copy()
    return bf, bs

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()

def score(f):

    s = f.sum()

        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_grad(f, w):
    n = f.shape[0]
    c = fft_conv(f, w[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def compute_grad_exact(f, w):
    n = f.shape[0]
    c1 = fft_conv(f, w[::-1])
    term_a = c1[n - 1:2 * n - 1][::-1].copy()
    c2 = fft_conv(w, f)
    term_b = c2[0:n].copy()
    return term_a + term_b

             peak_w_start=0.998, peak_w_end=0.9995, max_pk=1000,
             score_interval=10, patience=50, restart_amp=0.006):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    ni = 0
    grad_fn = compute_grad_exact if use_exact_grad else compute_grad
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        frac = peak_w_start + (peak_w_end - peak_w_start) * min(1.0, it / 100000.0)
        thr = M * frac
        pk = np.where(b >= thr)[0]
        if len(pk) > max_pk:
            pk = pk[np.argsort(b[pk])[-max_pk:]]

        w /= ws
        g = grad_fn(f, w)
        gm = g.max()
        if gm < 1e-30:
            break
        g /= gm
        f = f * np.exp(-eta * g)
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        if it % score_interval == 0:

                    bs = sc
                    bf = f.copy()
                ni = 0

                    f = np.maximum(f, 1e-20)
                    f /= f.sum()
                    ni = 0
                    eta = min(eta * 0.5, eta0)
        if it > 15000:
            eta *= 0.99995
        eta = max(eta, 0.001)
    sc = score(f)
    if sc < bs:
        bs = sc
