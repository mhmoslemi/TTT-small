```python
def _peak_descent(f, alpha, num_steps):
    n = f.size
    S = f.sum()
    for _ in range(num_steps):
        c = _conv(f)
        jstar = int(np.argmax(c))
        # Gradient of c[jstar] w.r.t. f[k] is f[jstar-k] for valid k
        # jstar-k must be in [0, n-1], so k in [jstar-(n-1), jstar] intersect [0, n-1]
        kmin = max(0, jstar - (n-1))
        kmax = min(n-1, jstar)
        g = np.zeros(n)
        # g[k] = f[jstar-k]
        idx = np.arange(kmin, kmax+1)
        g[idx] = f[jstar - idx]
        # Subtract mean to maintain sum
        g = g - g.mean()
        # Update
        f_new = f - alpha * g
        # Project onto f >= 0, sum = S
        f_new = np.maximum(f_new, 0)
        if f_new.sum() > 0:
            f_new = f_new / f_new.sum() * S
        f = f_new
    return f
```
