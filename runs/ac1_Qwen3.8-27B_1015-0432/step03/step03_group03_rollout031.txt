        inits.append(norm_pos(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.0, 2.0]:
        inits.append(norm_pos(d ** pw))
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(norm_pos(h))
    wv = 0.5 + 0.5 * np.cos(2.0 * np.pi * 2.0 * x / n)
    inits.append(norm_pos(wv + 0.1))
    r = np.ones(n) + 0.2 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(norm_pos(r))
    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = gen_inits(n, rng)
    best_f, best_sc = inits[0], obj_val(inits[0])
    for init in inits[1:]:
        sc = obj_val(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    d1 = min(deadline - 350.0, t0 + 250.0)
    f1, s1 = run_opt(best_f.copy(), d1, rng, 0.10, 0.03, False)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()

    if time.monotonic() < deadline - 180.0:
        f2 = norm_pos(0.5 * (best_f + best_f[::-1]))
        d2 = min(deadline - 180.0, t0 + 550.0)
        f2o, s2o = run_opt(f2, d2, rng, 0.06, 0.015, True)
        if s2o < best_sc:
            best_sc, best_f = s2o, f2o.copy()

    if time.monotonic() < deadline - 100.0:
        d3 = min(deadline - 60.0, t0 + 850.0)
        f3o, s3o = run_opt(best_f.copy(), d3, rng, 0.04, 0.008, False)
        if s3o < best_sc:
            best_sc, best_f = s3o, f3o.copy()

    if time.monotonic() < deadline - 12.0:
        for _ in range(500):
            if time.monotonic() > deadline - 8.0:
                break
            tri = norm_pos(best_f * np.exp(0.003 * rng.standard_normal(n)))
            sc = obj_val(tri)
            if sc < best_sc:
                best_sc = sc
                best_f = tri.copy()

    result = [float(x) for x in best_f]
    fineval = evaluate_sequence(result)
    if not np.isfinite(fineval):
        result = [1.0 / n] * n
    return result
