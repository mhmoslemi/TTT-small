import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return 1e10
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def lp_mwu(f, M, n_iter=40, gamma=5.0, deadline=None):
    """Approximate LP: max sum(g) s.t. (f*g)_k <= M, g >= 0, via multiplicative weights."""
    n = len(f)
    L = 2 * n - 1
    y = np.ones(L) / L
    g = np.ones(n) / n
    eps = 1e-12
    size2 = 1 << (3 * n).bit_length()
    FF = np.fft.rfft(f, size2)
    for t in range(n_iter):
        if deadline is not None and time.monotonic() > deadline:
            break

        ys = y.sum()
        if ys < 1e-30 or not np.isfinite(ys):
