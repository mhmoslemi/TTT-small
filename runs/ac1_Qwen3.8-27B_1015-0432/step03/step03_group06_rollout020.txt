import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return irfft(rfft(a, size) * rfft(b, size), n + m - 1)

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def norm1(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def sym(f):
    return 0.5 * (f + f[::-1])

def ugrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    gf = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(gf, 0.0) * f

def optimize(f, deadline, rng, use_sym):
    n = f.shape[0]
    f = norm1(f.copy())
    if use_sym:
        f = norm1(sym(f))
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.05
    no_improv = 0
    restarts = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        frac = min(1.0, it / 3000.0)
        width = 0.10 * (1.0 - frac) + 0.008
        thresh = M * (1.0 - width)
        peaks = np.where(b >= thresh)[0]
        if len(peaks) > 120:
            peaks = peaks[np.argsort(b[peaks])[-120:]]
        if len(peaks) < 1:
            peaks = np.array([np.argmax(b)])
        gu = ugrad(f, peaks)
        gmax = gu.max()
        if gmax < 1e-25:
            break
        gn = gu / gmax
        best_eta = eta
        best_sc_t = np.inf
        best_tr = f.copy()
        for te in (eta * 0.5, eta, eta * 1.5, eta * 2.5):
            tr = f * np.exp(-te * gn)
            tr = np.maximum(tr, 1e-25)
            if use_sym:
                tr = sym(tr)
            tr = norm1(tr)
            sc = score(tr)
            if sc < best_sc_t:
                best_sc_t = sc
                best_eta = te
                best_tr = tr
        f = best_tr
        if best_eta > eta * 1.15:
            eta = min(0.2, eta * 1.12)
        elif best_eta < eta * 0.85:
            eta = max(0.003, eta * 0.7)
        if it % 8 == 0:
            sc = score(f)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 120 and time.monotonic() < deadline - 50:
                s = 0.008 + 0.04 * min(1.0, restarts / 10.0)
                noise = rng.standard_normal(n)

                    noise = sym(noise)

                else:
