def _np2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _np2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _obj(a, tau, n):
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv(a)
    cm = c.max()
    if tau < 1e-8:
        return 2.0 * n * cm / (S * S)
    d = (c - cm) / tau
    d = np.clip(d, -50.0, 0.0)
    sm = cm + tau * np.log(np.exp(d).sum())
