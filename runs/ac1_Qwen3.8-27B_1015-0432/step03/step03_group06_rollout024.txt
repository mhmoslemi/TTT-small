import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

def subgrad(f, peaks):
    n = len(f)
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    N = 1 << (n + L).bit_length()
    F = np.fft.rfft(f, N)
    H = np.fft.rfft(h, N)
    cc = np.fft.irfft(np.conj(F) * H, N)[:n]
    return np.maximum(cc, 0.0)

            break
        width = 0.015 + 0.04 * min(1.0, it / 5000.0)
        thresh = M * (1.0 - width)

        if len(peaks) > max_pk:

        g = subgrad(f, peaks)
        gmax = g.max()
        if gmax < 1e-30:

        gn = g / gmax
        best_et = eta

        for te in [eta * 0.4, eta, eta * 1.6, eta * 2.8]:
