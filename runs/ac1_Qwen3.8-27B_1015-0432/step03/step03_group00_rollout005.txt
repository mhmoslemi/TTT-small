import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -60.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))
