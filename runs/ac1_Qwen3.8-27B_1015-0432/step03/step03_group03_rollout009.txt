The dual constraint for j: sum_k lambda_k * f[k-j] >= 1

Now, lambda_k will be nonzero only for "important" k (where the constraint is tight in the primal). At optimality, by complementary slackness, lambda_k > 0 only if (f*g)[k] = M (the constraint is tight).

So the optimal lambda is supported on the set of peaks. We can restrict to a candidate set of peaks.

The simplified dual (restricted to candidate peaks P):
min M * sum_{k in P} lambda_k s.t. sum_{k in P} lambda_k * f[k-j] >= 1 for j=0,...,n-1, lambda_k >= 0

This has |P| variables and n constraints. For |P|=200, n=4806, this is very solvable.

The optimal solution gives us the weights w_k = lambda_k / sum(lambda) for the subgradient.

Let me implement this properly. The key question is: how often to solve the LP and how many peaks to include.

For the LP constraint matrix:
- Row j, column k_idx (where peaks[k_idx] = k): entry = f[k-j] if 0 <= k-j <= n-1, else 0
- For a peak at position k, it contributes to rows j where 0 <= k-j <= n-1, i.e., k-n+1 <= j <= k
- So each peak contributes to at most n rows (actually at most min(k+1, n, 2n-1-k) rows)

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def norm_pos(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones_like(f) / f.shape[0]
    return f / s

def obj_val(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def compute_subgrad(f, peaks, weights):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = weights
    c = fft_conv(f, h[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)

def solve_lp_weights(f, peaks, M, deadline):
    """Solve the dual LP to find optimal peak weights."""
    n = f.shape[0]
    npk = len(peaks)
    
    if npk < 2 or time.monotonic() > deadline:
        return np.ones(npk) / npk
    
    # Build sparse constraint matrix A (n x npk)
    # A[j, k_idx] = -f[peaks[k_idx] - j] if 0 <= peaks[k_idx] - j <= n-1
    rows = []
    cols = []
    vals = []
    
    for ki, pk in enumerate(peaks):
        j_lo = max(0, pk - n + 1)
        j_hi = min(n - 1, pk)
        if j_lo > j_hi:
            continue
        j_range = range(j_lo, j_hi + 1)
        # f[peaks[ki] - j] for j in j_range
        idx = pk - np.arange(j_lo, j_hi + 1)
        v = -f[idx]
        # Add non-zero entries
        nz = v != 0
        if nz.any():
            rows.extend((np.arange(j_lo, j_hi + 1))[nz].tolist())
            cols.extend(np.full(nz.sum(), ki).tolist())
            vals.extend(v[nz].tolist())
    
    if not vals:
        return np.ones(npk) / npk
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(n, npk))
    
    c_obj = -M * np.ones(npk)
    b_ub = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, 
                  bounds=[(0, None)] * npk, method='highs',
                  options={'time_limit': 30})
    
    if res.success and res.x is not None:
        w = np.maximum(res.x, 0.0)
        wsum = w.sum()
        if wsum > 1e-15:
            w = w / wsum
            return w
    return np.ones(npk) / npk

def optimize_lp(f, deadline, rng, eta0, n_peaks, sym, max_iter):
    n = f.shape[0]
    f = norm_pos(f.copy())
    best_f = f.copy()
    best_sc = obj_val(f)
    eta = eta0
    no_improv = 0
    restart_count = 0
    momentum = np.zeros(n)
    L = 2 * n - 1
    
    # Initial LP solve
    b = fft_conv(f, f)
    M = b.max()
    peaks = np.argsort(b)[::-1][:n_peaks]
    w = solve_lp_weights(f, peaks, M, deadline)
    
    lp_interval = 100  # Solve LP every 100 iterations
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        if it % lp_interval == 0:
            b = fft_conv(f, f)
            M = b.max()
            # Select peaks: top n_peaks by value
            peaks = np.argsort(b)[::-1][:n_peaks]
            w = solve_lp_weights(f, peaks, M, deadline)
        else:
            # Use previous peaks and weights
            pass
        
        # Compute subgradient
        g = compute_subgrad(f, peaks, w)
        gmax = g.max()
        if gmax < 1e-25:
            break
        
        # Momentum
        momentum = 0.75 * momentum + 0.25 * g
        mmax = momentum.max()
        if mmax < 1e-25:
            break
        
        # Multiplicative update
        f = f * np.exp(-eta * momentum / mmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f = norm_pos(f)
        
        # Evaluate
        if it % 25 == 0:
            sc = obj_val(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            
            if no_improv > 300:
                scale = 0.02 + 0.03 * min(1.0, restart_count / 5.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-15)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = norm_pos(f)
                momentum = np.zeros(n)
                no_improv = 0
                restart_count += 1
                eta = min(0.2, eta * 1.25)
        
        eta *= 0.9999
    
    return best_f, best_sc
