def _fpow2(s):
    m = 1
    while m < s:

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)

    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

        if i % 4 == 3 and time.monotonic() < dl - 60.0:
            sym = 0.5 * (cur + cur[::-1])
            sym = _normalize(np.clip(sym, 0.0, 1000.0), scale)
            sc = float(evaluate_sequence(sym.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = sym.copy()
                best_sc = sc
                cur = sym.copy()
        # Reheat: blend back toward best to escape
        if i % 5 == 4 and time.monotonic() < dl - 80.0:
            cur = 0.7 * cur + 0.3 * best
            cur = _normalize(np.clip(cur, 0.0, 1000.0), scale)

    # LP-inspired directional step
    if time.monotonic() < dl - 60.0:
        for _ in range(5):
            if time.monotonic() > dl - 60.0:
                break
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            S = cd.sum()
            M = c[pk]
            # Reduce elements that contribute to peak, increase others
            contrib = np.zeros(n)
            lo = max(0, pk - (n - 1))
            hi = min(n - 1, pk)
            for j in range(lo, hi + 1):
                contrib[j] = cd[j] * cd[pk - j]
            top_k = min(20, // 3) if n > 60 else 3
            idx = np.argsort(contrib)[::-1][:top_k]
            for j in idx:
                cd[j] *= 0.92
            quiet = np.argsort(contrib)[:top_k]
            for j in quiet:
                cd[j] *= 1.08
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() > 0.01:
                best2, sc2 = _update_best(cd, scale, best, best_sc)
                if sc2 < best_sc:
                    best = best2
                    best_sc = sc2
                    cur = best2.copy()
            # Follow up with short L-BFGS-B
            nx = _lbfgs_step(cur, 0.01, bounds_list, 100, dl)
            best, best_sc = _update_best(nx, scale, best, best_sc)
            cur = _normalize(np.clip(nx, 0.0, 1000.0), scale)

    # Local search: targeted perturbations near convolution peak
    if time.monotonic() < dl - 20.0:
        wmax = max(2, n // 30)
        for outer in range(20):
            if time.monotonic() > dl - 20.0:
                break
            found = False
            for _ in range(200):
                if time.monotonic() > dl - 20.0:
                    break
                cd = cur.copy()
                c = _conv(cd)
                pk = int(np.argmax(c))
                w1 = int(rng.integers(1, wmax + 1))
                center = int(np.clip(pk, 0, n - 1))
                s1 = int(rng.integers(max(0, center - w1), min(n - w1, center) + 1))
                cd[s1:s1 + w1] *= float(np.exp(rng.normal(-0.025, 0.035)))
                w2 = int(rng.integers(1, wmax + 1))
                s2 = int(rng.integers(0, n - w2))
                cd[s2:s2 + w2] *= float(np.exp(rng.normal(0.025, 0.035)))
                if rng.random() < 0.15:
                    cd *= float(np.exp(rng.normal(0.0, 0.008)))
                cd = np.clip(cd, 0.0, 1000.0)
                if cd.sum() > 0.01:
                    sc = float(evaluate_sequence(cd.tolist()))
                    if np.isfinite(sc) and sc < best_sc * (1.0 - 1e-9):
                        best = cd.copy()
                        best_sc = sc
                        cur = cd.copy()
                        found = True
            if not found and outer > 3:
                cur = best.copy()

    # Final high-precision polish
    if time.monotonic() < dl - 5.0:
        cur = best.copy()
        for tv in [5e-5, 2e-5, 1e-5]:
            if time.monotonic() > dl - 5.0:
                break
            nx = _lbfgs_step(cur, tv, bounds_list, 400, dl)
            best, best_sc = _update_best(nx, scale, best, best_sc)
            cur = _normalize(np.clip(nx, 0.0, 1000.0), scale)

    # Final verification
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = np.ones(n)
    return [float(v) for v in best]
