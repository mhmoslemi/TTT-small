                best_f = trial.copy()

    result = np.maximum(best_f, 0.0).copy()
    rs = result.sum()
    if rs < 0.01:
        result *= 0.02 / max(rs, 1e-30)
    result = np.minimum(result, 1000.0)
    result_list = [float(v) for v in result]
    fs = evaluate_sequence(result_list)
    if not np.isfinite(fs):
        result_list = [1.0 / n] * n
    return result_list
