        except Exception:
            pass

    # Symmetrization
    if time.monotonic() < dl - 50.0:
        sym = 0.5 * (cur + cur[::-1])
        sym = _normalize(np.clip(sym, 0.0, 1000.0), scale)
        sc = float(evaluate_sequence(sym.tolist()))
        if np.isfinite(sc) and sc < best_sc:
            best = sym.copy()
            best_sc = sc
            cur = sym.copy()

    # Focused peak shaving
    if time.monotonic() < dl - 20.0:
        wmax = max(2, n // 60)
        no_imp = 0
        for _ in range(3000):
            if time.monotonic() > dl - 20.0:
                break
            if no_imp > 400:
                cur = best.copy()
                no_imp = 0
            cd = cur.copy()
            c = _conv(cd)
            pk = int(np.argmax(c))
            # Shrink window near peak-contributing lag
            w1 = int(rng.integers(1, wmax + 1))
            hi = min(n - w1, max(0, min(pk, n - 1)))
            lo = max(0, min(pk, n - 1) - w1)
            s1 = int(rng.integers(lo, max(lo + 1, hi + 1)))
            cd[s1:s1 + w1] *= exp(rng.normal(-0.04, 0.05))
            # Grow complementary window
            w2 = int(rng.integers(1, wmax + 1))
            s2 = int(rng.integers(0, n - w2))
            cd[s2:s2 + w2] *= exp(rng.normal(0.04, 0.05))
            # Occasionally scale globally
            if rng.random() < 0.1:
                cd *= exp(rng.normal(0, 0.008))
            cd = np.clip(cd, 0.0, 1000.0)
            if cd.sum() < 0.01:
                continue
            sc = float(evaluate_sequence(cd.tolist()))
            if np.isfinite(sc) and sc < best_sc:
                best = cd.copy()
                best_sc = sc
                cur = cd.copy()
                no_imp = 0
            else:
                no_imp += 1
            if rng.random() < 0.05:
                cur = best.copy()

    # Final low-temperature polish
    if time.monotonic() < dl - 4.0:
        cur = best.copy()
        for tv in [0.001, 4e-4, 1.5e-4]:
            if time.monotonic() > dl - 4.0:
                break
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bounds_list,
                               options={'maxiter': 300, 'gtol': 1e-16,
                                        'maxls': 20, 'ftol': 1e-23})
                nx = np.clip(res.x, 0.0, 1000.0)
                nx = _normalize(nx, scale)
                sc = float(evaluate_sequence(nx.tolist()))
                if np.isfinite(sc) and sc < best_sc:
                    best = nx.copy()
                    best_sc = sc
                cur = nx.copy()
            except Exception:
                pass

    # Safety check
    final_sc = float(evaluate_sequence(best.tolist()))
    if not np.isfinite(final_sc):
        best = a0
    return [float(v) for v in best]
