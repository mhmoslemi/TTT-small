        row[lo:hi+1] = a[jp-lo:jp-hi+1]  # a[jp-i] for i in [lo,hi]

def _fft2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fft2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

        return 1e10
    mv, _ = _smax(_conv(a), tau)
