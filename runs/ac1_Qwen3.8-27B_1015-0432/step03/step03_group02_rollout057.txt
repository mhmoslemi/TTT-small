import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def compute_grad(f, c, beta, max_peaks):
    n = len(f)
    L = 2 * n - 1
    M = c.max()
    if M < 1e-30:
        return None
    w = np.exp(-beta * (M - c) / M)
    w -= w.min()
    wsum = w.sum()
    if wsum < 1e-30:
        h = np.zeros(L)
        h[np.argmax(c)] = 1.0
    else:
        if max_peaks < L:
            idx = np.argpartition(-w, max_peaks)[:max_peaks]
            mask = np.zeros(L, dtype=bool)
            mask[idx] = True
            w[~mask] = 0.0
            wsum = w.sum()
            if wsum < 1e-30:
                h = np.zeros(L)
                h[np.argmax(c)] = 1.0
            else:
                h = w / wsum
        else:
            h = w / wsum
    conv = fft_conv(f, h[::-1])
    g = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return g

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.10, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    best_f = f.copy()
    best_sc = score(f)
    eta = eta_init
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 10000.0)
        beta = 3.0 + 40.0 * prog
        g = compute_grad(f, b, beta, max_peaks)
        if g is None:
            break
        gmax = g.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        if it > 0 and it % 500 == 0:
            f = 0.5 * (f + f[::-1])
            f /= f.sum()
        it += 1
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > 1000:
                sc2 = 0.02 + 0.03 * min(1.0, patience / 2000.0)
                f = best_f * np.exp(sc2 * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.4, eta * 1.4)
        if it > 200:
            eta *= 0.9994
    return best_f, best_sc

def fine_tune(f, deadline, rng, steps=400):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    for i in range(steps):

            sc = score(trial)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.94
    return best_f, best_sc
