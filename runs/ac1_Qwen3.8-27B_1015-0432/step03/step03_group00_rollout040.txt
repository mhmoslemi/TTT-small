    m = 1
    while m < n:
        m <<= 1
    return m

    L = _epad(2 * a.size - 1)
    fa = np.fft.fft(a, L)
    return np.fft.ifft(fa * fa).real[:2 * a.size - 1]

    np.clip(d, -50.0, None, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s
