    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _accept(trial, sc, best, best_sc):
    s = float(evaluate_sequence(trial.tolist()))
    if np.isfinite(s) and s < best_sc[0]:
        best_sc[0] = s
        best[:] = trial
        return True
    return False


def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + max(1.0, float(budget_s) - 12.0)

    par = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if par.ndim != 1 or par.size == 0 or not np.all(np.isfinite(par)) or par.sum() < 1e-8:
        par = np.ones(6733)
    n = par.size
    sc = np.sqrt(2.0 * n)
    best = par / par.sum() * sc
    cur = best.copy()
    best_sc = [float(evaluate_sequence(best.tolist()))]
    bl = [(0.0, 100.0)] * n

    # Quick global perturbations
    x = np.arange(n, dtype=np.float64)
    for amp in [0.05, 0.1, 0.2, 0.35]:
        for _ in range(2):
            sh = n * rng.uniform()
            ph = 2.0 * np.pi * rng.uniform()
            c = best * (1.0 + amp * np.cos(2.0 * np.pi * (x + sh) / n + ph))
            c = np.clip(c, 0, None)
            if c.sum() > 0.01:
                c = c / c.sum() * sc
                _accept(c, sc, best, best_sc)
                cur /= cur.sum() * 0 + 1
                cur = best.copy()
        if time.monotonic() > dl - 900:
            break

    # Random multiplicative perturbations
    for _ in range(6):
        rz = best * np.exp(0.04 * rng.standard_normal(n))
        rz = np.clip(rz, 0, None)
        if rz.sum() > 0.01:
            rz = rz / rz.sum() * sc
            if _accept(rz, sc, best, best_sc):
                cur = rz.copy()
        if time.monotonic() > dl - 900:
            break

    # Main L-BFGS-B optimization with decreasing tau
    taus = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4]
    for ti, tv in enumerate(taus):
        if time.monotonic() > dl - 180:
            break
        # Two passes per tau for smaller taus
        n_pass = 2 if tv < 0.01 else 1
        for _p in range(n_pass):
            if time.monotonic() > dl - 180:
                break
            mx = 400 + 200 * ti
            try:
                res = minimize(_obj, cur, args=(tv,), jac=_grad,
                               method='L-BFGS-B', bounds=bl,
                               options={'maxiter': mx, 'gtol': 1e-14,
                                        'maxls': 20, 'ftol': 1e-16,
                                        'maxfun': mx * 3})
                nx = np.clip(res.x, 0.0, 1000.0)
                if nx.sum() > 1e-8:
                    nx = nx / nx.sum() * sc
                _accept(nx, sc, best, best_sc)
                cur = nx.copy()
            except Exception:
                pass

    # Peak shaving: reduce top-contributing coefficients at peak lags
    for _round in range(300):
        if time.monotonic() > dl - 40:
            break
        c = _conv(cur)
        top_k = min(6, 2 * n - 1)
        idx = np.argsort(c)[-top_k:]
        improved = False
        for jp in idx:
            lo = max(0, jp - (n - 1))
            hi = min(n - 1, jp)
            ks = np.arange(lo, hi + 1)
            prods = cur[ks] * cur[jp - ks]
            tc = ks[np.argsort(prods)[-8:]]
            for frac in [0.98, 0.95, 0.92, 0.88, 0.83]:
                trial = cur.copy()
                trial[tc] *= frac
                trial = np.clip(trial, 0, None)
                if trial.sum() > 0.01:
                    if _accept(trial, sc, best, best_sc):
                        cur = trial.copy()
                        improved = True
                        break
            if improved:
                break
        if not improved:
            # Try small L-BFGS-B nudge to escape
            if time.monotonic() < dl - 40:
                try:
                    res = minimize(_obj, cur, args=(2e-4,), jac=_grad,
                                   method='L-BFGS-B', bounds=bl,
                                   options={'maxiter': 100, 'gtol': 1e-13,
                                            'maxls': 10, 'ftol': 1e-15,
                                            'maxfun': 300})
                    nx = np.clip(res.x, 0.0, 1000.0)
                    if nx.sum() > 1e-8:
                        nx = nx / nx.sum() * sc
                    _accept(nx, sc, best, best_sc)
                    cur = nx.copy()
                except Exception:
                    pass
            else:
                break

    # Final random walk
    while time.monotonic() < dl - 4:
        cd = cur.copy()
        nw = int(rng.integers(2, max(3, n // 20)))
        s0 = int(rng.integers(0, n - nw))
        cd[s0:s0 + nw] *= np.exp(rng.normal(0.0, 0.05, nw))
        if rng.random() < 0.06:
            cd *= np.exp(rng.normal(0.0, 0.01))
        cd = np.clip(cd, 0.0, 1000.0)
        if cd.sum() > 0.01:
            cd = cd / cd.sum() * sc
            if _accept(cd, sc, best, best_sc):
                cur = cd.copy()
        if rng.random() < 0.05:
            cur = best.copy()

    # Final cleanup: zero out tiny values, re-evaluate
    thresh = 1e-4 * best.max()
    b2 = best.copy()
    b2[b2 < thresh] = 0.0
    if b2.sum() > 0.01:
        s = float(evaluate_sequence(b2.tolist()))
        if np.isfinite(s) and s <= best_sc[0]:
            best = b2

    # Safety: ensure valid return
    if not np.all(np.isfinite(best)) or best.sum() < 0.01:
        best = cur
        if not np.all(np.isfinite(best)) or best.sum() < 0.01:
            best = par / par.sum() * sc

    return [float(v) for v in best]
