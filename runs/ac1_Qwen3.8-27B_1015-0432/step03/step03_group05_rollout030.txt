```python
def compute_subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c1 = fft_conv(f, h[::-1])  # gives A in c1[n-1:2n-1][::-1]
    c2 = fft_conv(f, h)         # gives B in c2[0:n]
    A = c1[n-1:2*n-1][::-1]
    B = c2[0:n]
    return A + B
```
