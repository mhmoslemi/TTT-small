def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score_of(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * len(f) * b.max() / (s * s))

def grad_smooth(f, b, frac, max_pk):
    n = len(f)
    M = b.max()
