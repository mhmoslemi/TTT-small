    inits.append(r / r.sum())
    return inits

def propose_candidate(seed=42, budget_s=1100, **kw):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = gen_inits(n, rng)
    best_f = inits[0].copy()
    best_sc = obj(best_f)
    for fx in inits[1:]:
        sc = obj(fx)
        if sc < best_sc:
            best_sc = sc
            best_f = fx.copy()
    # Phase 1: fast convergence
    t1 = min(dl - 400.0, t0 + 380.0)
    f1, sc1 = optimize(best_f.copy(), t1, rng, eta0=0.09, n_peaks=180)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()
    # Phase 2: symmetric refinement
    if time.monotonic() < dl - 250.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs = fs / fs.sum()
        t2 = min(dl - 250.0, t0 + 700.0)
        f2, sc2 = optimize(fs, t2, rng, eta0=0.06, n_peaks=300, sym=True)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2.copy()
    # Phase 3: asymmetric fine polish with many peaks
    if time.monotonic() < dl - 120.0:
        t3 = min(dl - 60.0, t0 + 960.0)
        f3, sc3 = optimize(best_f.copy(), t3, rng, eta0=0.03, n_peaks=400)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3.copy()
    # Phase 4: micro local search
    if time.monotonic() < dl - 8.0:
        for _ in range(300):
            if time.monotonic() > dl - 5.0:
                break
            trial = best_f * np.exp(0.002 * rng.standard_normal(n))
            trial = np.maximum(trial, 1e-18)
            trial = trial / trial.sum()
            sc = obj(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()
    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
