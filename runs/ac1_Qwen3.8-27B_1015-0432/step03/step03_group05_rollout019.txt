    if time.monotonic() < dl - 40:
        f4, s4 = fine_polish(bf.copy(), dl - 15, rng)
        if s4 < bs:
            bs, bf = s4, f4
    if time.monotonic() < dl - 5:
        bf, bs = perturb_best(bf, bs, rng, dl - 2, count=80, amp=0.003)
    result = [float(v) for v in bf]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
