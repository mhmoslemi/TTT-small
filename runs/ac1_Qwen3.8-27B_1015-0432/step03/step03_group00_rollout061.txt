import numpy as np
import time
from numpy.fft import fft, ifft
from scipy.optimize import minimize


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _lse_grad(a, tau):
    n = a.size
    c = _conv(a)
    cm = c.max()
    d = np.clip((c - cm) / tau, -50.0, 0.0)
    e = np.exp(d)
    s = e.sum()
    p = e / s
    mv = cm + tau * np.log(s)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return mv, g


def _obj(a, tau, n):
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _lse_grad(a, tau)
    return 2.0 * n * mv / (S * S)


def _jac(a, tau, n):
    S = a.sum()
    mv, g = _lse_grad(a, tau)
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

    S = a.sum()

        return 1e10
    c = _conv(a)
    return float(2.0 * a.size * c.max() / (S * S))
