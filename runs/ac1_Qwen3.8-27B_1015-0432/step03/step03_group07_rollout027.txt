import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def nscore(f):
    S = f.sum()
    if S < 0.01:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * f.size * b.max() / (S * S)

def compute_grad(f, w, size_gc):
    F = np.fft.rfft(f, size_gc)
    W = np.fft.rfft(w, size_gc)
    G = np.fft.irfft(np.conj(W) * F, size_gc)
    return np.maximum(2.0 * G[:f.shape[0]], 0.0)
