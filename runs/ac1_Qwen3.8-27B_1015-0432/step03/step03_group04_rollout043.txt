import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):

    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

    s = f.sum()
    if s < 1e-10:

def _optimize(f0, deadline, rng, sym=False, max_iter=1000000):

    f = np.maximum(f0.copy(), 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = _score(f)
    eta = 0.05
    no_improv = 0
    restarts = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        
        b = _fft_conv(f, f)
        M = b.max()
        
        # Threshold: start very tight, widen slowly
        progress = min(1.0, it / 50000.0)
        frac = 0.9999 - 0.0399 * progress
        thresh = M * frac
        peaks = np.where(b >= thresh)[0]
        
        max_pk = 4000
        if len(peaks) > max_pk:
            idx = np.argsort(b[peaks])[-max_pk:]
            peaks = peaks[idx]
        if len(peaks) == 0:
            peaks = np.array([np.argmax(b)])
        
        w = b[peaks]
        wmin = w.min()
        w = (w - wmin + 1e-15) / (w - wmin + 1e-15).sum()

        h = np.zeros(len(b))
        h[peaks] = w
        c = _fft_conv(f, h[::-1])
        g = 2.0 * c[n-1:2*n-1][::-1]

        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-25:
            break
        
        f = f * np.exp(-eta * g / gmax)
        f = np.maximum(f, 1e-20)
        if sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()
        
        if it % 20 == 0:
            sc = _score(f)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1

                f /= f.sum()
                no_improv = 0
                restarts += 1
                eta = min(0.15, eta * 1.15)

Actually, I think the key improvement is in the ACTIVE SET. Let me think about this more carefully.

The gradient of the max function is only defined at the argmax. When multiple coordinates are near the max, we can use a smoothed max:
M_smooth(ε) = ε * log(sum_k exp(b[k]/ε))

The gradient of M_smooth is:
dM_smooth/da[i] = (1/S) * sum_k S_k * db[k]/da[i] where S_k = exp(b[k]/ε) / sum_j exp(b[j]/ε)

This is a softmax-weighted gradient. For small ε, it focuses on the max. For large ε, it averages over all k.

Using the softmax gradient, we don't need to explicitly select an active set. The softmax automatically weights the important peaks.

But computing the softmax over 2n-1 = 9611 values is fine (it's O(n)). And the correlation with the softmax weights requires one FFT convolution.

Wait, I just realized that the key issue with the softmax approach is that it computes the correlation with ALL 2n-1 values of w, not just the active peaks. The previous approach used a sparse h (only K nonzero entries), but the FFT cost is the same regardless of sparsity (it's still O(N log N) where N is the FFT size). The advantage of the softmax approach is:
