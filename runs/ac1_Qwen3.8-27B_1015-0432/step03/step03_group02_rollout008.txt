            vals = b[top] - thresh
            h[top] = vals / max(vals.max(), 1e-30)
        else:
            h.fill(0.0)
            h[mask] = (excess[mask] / max(excess[mask].max(), 1e-30)) ** 1.5
        c = fft_conv(f, h[::-1])
        g = c[n - 1:2 * n - 1][::-1].copy()
        np.maximum(g, 0.0, out=g)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        sc = 2.0 * n * M
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
        if patience > 400:
            mag = 0.02 + 0.04 * min(1.0, patience / 1200.0)
            f = best_f * np.exp(mag * rng.standard_normal(n))
            np.maximum(f, 1e-30, out=f)
            f /= f.sum()
            b = fft_conv(f, f)
            M = b.max()
            patience = 0
            eta = min(0.35, eta * 1.4)
        if it > 200:
            eta *= 0.9985
    return best_f, best_sc

def fine_tune(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    scale = 0.004
    stall = 0
    for i in range(600):
        if time.monotonic() > deadline:
            break
        improved = False
        for _ in range(10):
            d = rng.standard_normal(n)
            for frac in [0.5, 1.0, 2.0]:
                trial = best_f * np.exp(frac * scale * d)
                np.maximum(trial, 1e-30, out=trial)
                trial /= trial.sum()
                sc = score(trial)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = trial.copy()
                    improved = True
        if improved:
            scale *= 0.93
            stall = 0
        else:
            stall += 1
            scale *= 0.8
        if stall > 40:
            break
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 15.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    p /= p.sum()

    inits = [p.copy(), np.ones(n) / n]
    x = np.arange(n, dtype=np.float64)
    for pw in [0.5, 0.8, 1.0, 1.5, 2.0, 3.0]:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        inits.append(d ** pw / (d ** pw).sum())
    for ph in [0.0, 0.3, 0.5, 0.7]:
        h2 = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + ph) / (n + 1.0)))
        inits.append(h2 / h2.sum())
    for a in [0.1, 0.2, 0.4]:
        r = np.ones(n) + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        inits.append(r / r.sum())
    inits.append(0.5 * (p + p[::-1]) / p.sum())

    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc, best_f = sc, v.copy()

    end1 = min(deadline, t0 + 300.0)
    f, s1 = run_opt(best_f.copy(), end1, rng, max_peaks=128, eta_init=0.15)
    if s1 < best_sc:
        best_sc, best_f = s1, f

    if time.monotonic() < deadline - 300.0:
        end2 = min(deadline, t0 + 620.0)
        f, s2 = run_opt(best_f.copy(), end2, rng, max_peaks=300, eta_init=0.08)
        if s2 < best_sc:
            best_sc, best_f = s2, f

    if time.monotonic() < deadline - 200.0:
        end3 = min(deadline, t0 + 880.0)
        f, s3 = run_opt(best_f.copy(), end3, rng, max_peaks=600, eta_init=0.04)
        if s3 < best_sc:
            best_sc, best_f = s3, f

    if time.monotonic() < deadline - 80.0:
        for k in range(5):
            if time.monotonic() > deadline - 40.0:
                break
            mag = 0.015 + 0.015 * k
            fr = best_f * np.exp(mag * rng.standard_normal(n))
            np.maximum(fr, 1e-30, out=fr)
            fr /= fr.sum()
            fr, sr = run_opt(fr, deadline - 45.0, rng, max_peaks=300, eta_init=0.08)
            if sr < best_sc:
                best_sc, best_f = sr, fr

    if time.monotonic() < deadline - 10.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 5.0, rng)

    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
