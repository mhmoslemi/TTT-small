    rtri = 1.0 / (np.maximum(x, 0.5) * np.maximum(n - x, 0.5))
    inits.append(rtri / rtri.sum())
    bimod = np.exp(-((x - 0.25 * n)**2) / (0.04 * n**2)) + np.exp(-((x - 0.75 * n)**2) / (0.04 * n**2))
    inits.append(bimod / bimod.sum())
    cos_v = 0.5 + 0.5 * np.cos(2.0 * np.pi * x / (n + 1.0))
    inits.append(cos_v / cos_v.sum())
    for kr in range(3):
        r = np.ones(n) + 0.35 * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        inits.append(r / r.sum())
    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()
    phase1_end = min(deadline, t0 + 280.0)
    f1, s1 = run_opt(best_f.copy(), phase1_end, rng, max_peaks=200, eta_init=0.20)
    if s1 < best_sc:
        best_sc, best_f = s1, f1
    fs = 0.5 * (best_f + best_f[::-1])
    fs /= fs.sum()
    ss = score(fs)
    if ss < best_sc:
        best_sc, best_f = ss, fs
    if time.monotonic() < deadline - 160.0:
        phase2_end = min(deadline, t0 + 620.0)
        f2, s2 = run_opt(best_f.copy(), phase2_end, rng, max_peaks=350, eta_init=0.10)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 110.0:
        for rest in range(6):
            if time.monotonic() > deadline - 45.0:
                break
            sc_r = 0.02 + 0.012 * rest
            f_r = best_f * np.exp(sc_r * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            bl = 0.15 * min(1.0, rest * 0.3)
            st = np.exp(-((x - 0.5 * n)**2) / (0.08 * n * n))
            st /= st.sum()
            f_r = (1.0 - bl) * f_r + bl * st
            np.maximum(f_r, 1e-30, out=f_r)
            f_r /= f_r.sum()
            f_r, s_r = run_opt(f_r, deadline - 55.0, rng, max_peaks=350, eta_init=0.12)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r
    if time.monotonic() < deadline - 12.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 8.0, rng)
    result = [float(min(1000.0, max(0.0, v))) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
