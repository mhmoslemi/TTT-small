def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -60.0)
    e = np.exp(d)
    tot = e.sum()
    return cm + tau * np.log(tot), e / tot

def _obj_grad(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _conv(a)
    mv, p = _smax(c, tau)
    # Gradient: d(loss)/d(a[i]) = (4n/S^2)*corr[i] - (4n*mv/S^3)
    # corr[i] = sum_k p[k] * a[k-i]

    a_rev = a[::-1]

    return obj, g
