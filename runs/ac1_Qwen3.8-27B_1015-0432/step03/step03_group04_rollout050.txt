import time

# Precompute FFT size
def get_fft_size(n):
    return 1 << (2*n - 1).bit_length()

def fft_conv2(a, b, size):
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:len(a)+len(b)-1]

def get_score(f):
    b = fft_conv2(f, f, fft_size)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def optimize(f, deadline, rng, sym=True, max_iter=500000):
    n = f.shape[0]
    f = np.abs(f).copy()
    f = np.maximum(f, 1e-20)
    if sym:
        f = 0.5 * (f + f[::-1])
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = 0.05
    no_improv = 0
    eval_count = 0
    
    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        # Compute autocorrelation
        b = fft_conv2(f, f, fft_size)
        M = b.max()

        npeaks = min(n, 1000)
        # Get indices of top npeaks values
        peaks = np.argpartition(b, -npeaks)[-npeaks:]
        w = b[peaks]
        w = w / w.sum()

        g = np.maximum(g, 0.0)

        gmax = g.max()
