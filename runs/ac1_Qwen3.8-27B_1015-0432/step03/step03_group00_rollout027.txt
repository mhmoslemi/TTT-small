import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -50.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    c = _conv(a)
    mv, _ = _smax(c, tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _try_update(cand, best, best_sc):
    cand = np.clip(cand, 0.0, 1000.0)
    if cand.sum() < 0.01:
        return False
    s = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(s) and s < best_sc[0]:
        best_sc[0] = s
        best[:] = cand
        return True
    return False


def _peak_shave(cur, best, best_sc, n):
    c = _conv(cur)
    K = 8
    idx = np.argsort(c)[-K:]
    improved = False
    for jp in idx:
        lo = max(0, jp - (n - 1))
        hi = min(n - 1, jp)
        ks = np.arange(lo, hi + 1)
        prods = cur[ks] * cur[jp - ks]
        top3 = ks[np.argsort(prods)[-3:]]
        for frac in (0.96, 0.92, 0.88, 0.83):
            trial = cur.copy()
            trial[top3] *= frac
            if _try_update(trial, best, best_sc):
                improved = True
                break
        if improved:
            break
    return improved
