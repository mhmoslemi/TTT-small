Key insight for improvement: the parent uses a fixed number of top peaks. Instead, I should use ALL peaks above a certain threshold, with the threshold adapting over time. Early on, use a lower threshold (more peaks, smoother gradient). Late, use a higher threshold (fewer peaks, more focused).

import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-30:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

def compute_grad(f, peaks):
    n = len(f)
    g = f.astype(np.float64)
    for k in peaks:
        lo = max(0, k - n + 1)
        hi = min(k + 1, n)
        g[lo:hi] += f[k-hi+1:k-lo+1][::-1]
    g /= float(len(peaks))
    return g
