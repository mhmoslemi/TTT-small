import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = (c - cm) / tau
    np.maximum(d, -60.0, out=d)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _fpow2(3 * n - 2)
    fa = fft(a, L)
    fp = fft(pr, L)
    cv = ifft(fa * fp).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))

def _myscore(a):
    n = a.size
    S = a.sum()
    if S < 0.01:
        return 1e10
    c = _conv(a)
    return 2.0 * n * c.max() / (S * S)

def _lbfgs(a, tau, maxiter, deadline):
    n = a.size
    if time.monotonic() > deadline:
        return a
    bl = [(0.0, 100.0)] * n
    try:
        res = minimize(_obj, a, args=(tau,), jac=_grad,
                       method='L-BFGS-B', bounds=bl,
                       options={'maxiter': maxiter, 'gtol': 1e-14,
                                'maxls': 20, 'ftol': 1e-16,
                                'maxfun': maxiter * 3})
        out = np.clip(res.x, 0.0, 100.0)
        if out.sum() < 1e-8:
            return a
        return out
    except Exception:
        return a

def _refine_peaks(a, deadline):

    for _ in range(60):
        if time.monotonic() > deadline:
            break
        c = _conv(a)
        S = a.sum()
        sc0 = 2.0 * n * c.max() / (S * S)

        improved = False
        for jp in lags:
            if time.monotonic() > deadline:
                break
            lo = max(0, jp - n + 1)
            hi = min(n - 1, jp)
            ks = np.arange(lo, hi + 1)
            prods = a[ks] * a[jp - ks]
            n_tc = min(8, len(ks))

                if sc1 < sc0 - 1e-10:
                    a = trial
                    sc0 = sc1
                    improved = True
                    break
        if not improved:
            break
    return a
