    W = np.fft.rfft(w, N)
    F = np.fft.rfft(f, N)
    corr = np.fft.irfft(W * np.conj(F), N)
    g = 2.0 * corr[:n]
    return np.maximum(g, 0.0)

def do_optimize(f, deadline, rng, use_sym=False):
    n = len(f)
    f = np.maximum(f.astype(np.float64).copy(), 0.0)
    s = f.sum()
    if s < 1e-10:
        f[:] = 1.0 / n
    else:
        f /= s
    if use_sym:
        f = 0.5 * (f + f[::-1])
        f /= f.sum()
    best_f = f.copy()
    best_sc = sc(f)
    eta = 0.06
    no_improv = 0
    rc = 0

    for it in range(5000000):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        prog = min(1.0, it / 6000.0)
        frac = 0.96 + 0.035 * prog
        g = subgrad(f, b, frac)
        gmax = g.max()
        if gmax < 1e-30:
            break
        step = np.minimum(eta * g / gmax, 2.0)
        f = f * np.exp(-step)
        f = np.maximum(f, 1e-25)
        if use_sym:
            f = 0.5 * (f + f[::-1])
        f /= f.sum()

        if (it + 1) % 10 == 0:
            cur_sc = sc(f)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 450:
                amp = 0.006 + 0.018 * min(1.0, rc / 6.0)
                pert = rng.standard_normal(n)
                Fb = np.fft.rfft(best_f)
                Fp = Fb * (1.0 + 0.25 * rng.standard_normal(len(Fb)))
                fp = np.fft.irfft(Fp, n) - best_f
                f = best_f * (1.0 + amp * (pert + 0.4 * fp / (np.abs(fp).max() + 1e-10)))
                f = np.maximum(f, 1e-25)
                if use_sym:
                    f = 0.5 * (f + f[::-1])
                f /= f.sum()
                no_improv = 0
                rc += 1
                eta = min(0.35, 0.06 * (1.18 ** rc))
        eta *= 0.998
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 18.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    inits = []
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    shapes = []
    shapes.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x / n)))
    cx = (n - 1.0) / 2.0
    for sig in [0.25, 0.4, 0.6, 0.8]:
        shapes.append(np.exp(-((x - cx) ** 2) / (2.0 * (sig * n / 2.0) ** 2)))
    d = np.maximum(np.minimum(x, n - 1.0 - x), 0.5)
    for pw in [0.5, 1.5, 2.0]:
        shapes.append(d ** pw)
    for i_s in shapes:
        ss = i_s.sum()
        if ss > 1e-10:
            inits.append(i_s / ss)
    for _ in range(2):
        r = np.maximum(1.0 + 0.25 * rng.standard_normal(n), 0.01)
        inits.append(r / r.sum())

    best_f = None
    best_sc = np.inf
    for init in inits:
        cur = sc(init)
        if cur < best_sc:
            best_sc = cur
            best_f = init.copy()

    dl1 = min(deadline, t0 + 380.0)
    f1, s1 = do_optimize(best_f.copy(), dl1, rng)
    if s1 < best_sc:
        best_sc, best_f = s1, f1.copy()

    if time.monotonic() < deadline - 320:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = sc(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs.copy()
        dl2 = min(deadline, t0 + 720.0)
        f2, s2 = do_optimize(fs.copy(), dl2, rng, use_sym=True)
        if s2 < best_sc:
            best_sc, best_f = s2, f2.copy()

    if time.monotonic() < deadline - 80:
        dl3 = deadline - 12.0
        f3, s3 = do_optimize(best_f.copy(), dl3, rng)
        if s3 < best_sc:
            best_sc, best_f = s3, f3.copy()

    if time.monotonic() < deadline - 5:
        for _ in range(500):
            if time.monotonic() > deadline - 3:
                break
            for amp in [0.001, 0.003, 0.008, 0.015]:
                trial = best_f * np.exp(amp * rng.standard_normal(n))
                trial = np.maximum(trial, 1e-20)
                trial /= trial.sum()
                ts = sc(trial)
                if ts < best_sc:
                    best_sc, best_f = ts, trial.copy()
                    break

    result = [float(v) for v in best_f]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
