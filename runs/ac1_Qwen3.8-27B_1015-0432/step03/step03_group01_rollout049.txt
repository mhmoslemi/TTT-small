import numpy as np
import time
from scipy.optimize import minimize
from numpy.fft import fft, ifft


def _n2(s):
    m = 1
    while m < s:
        m <<= 1
    return m


def _conv(a):
    n = a.size
    L = _n2(2 * n - 1)
    fa = fft(a, L)
    return ifft(fa * fa).real[:2 * n - 1]


def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -60.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s


def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)


def _grad(a, tau):
    n = a.size
    S = a.sum()
    c = _conv(a)
    mv, p = _smax(c, tau)
    pr = p[::-1]
    L = _n2(3 * n - 2)
    cv = ifft(fft(a, L) * fft(pr, L)).real
    g = 2.0 * cv[n - 1:2 * n - 1][::-1]
    return (2.0 * n / (S * S)) * g - (4.0 * n * mv / (S ** 3))


def _norm(a, s):
    t = a.sum()
    if t < 1e-8:
        return a.copy()
    return a / t * s


def _score(a):
    return float(evaluate_sequence(a.tolist()))


def _waterfill(a, alpha, n_it):
    n = a.size
    c = _conv(a)
    S = a.sum()
    for _ in range(n_it):
        mv, p = _smax(c, 0.01 * max(1e-8, c.max()))
        pr = p[::-1]
        L = _n2(3 * n - 2)
        cv = ifft(fft(a, L) * fft(pr, L)).real
        h = 2.0 * cv[n - 1:2 * n - 1][::-1]

        target = h.mean()

        if a.sum() < 1e-8:
            return a
        a = a / a.sum() * S
        c = _conv(a)
    return a
