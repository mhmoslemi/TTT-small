def gen_inits(n, rng):
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))

    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)

    inits.append(normalize(x + 1.0))
    inits.append(normalize(n - x))
    inits.append(normalize((x + 1.0) ** 2))
    inits.append(normalize((n - x) ** 2))
    inits.append(normalize((x + 1.0) ** 0.5))

    inits.append(normalize(np.exp(-x / (n * 0.25))))
    inits.append(normalize(np.exp(-(n - 1.0 - x) / (n * 0.25))))
    inits.append(normalize(np.exp(-x / (n * 0.4)) + np.exp(-(n - 1.0 - x) / (n * 0.4))))

    inits.append(normalize(np.exp(-np.abs(x - 0.35 * n) / (n * 0.25))))
    inits.append(normalize(np.exp(-np.abs(x - 0.65 * n) / (n * 0.25))))

    half = n // 2
    ramp_left = x[:half] + 1.0
    tail_right = np.maximum(n - 1.0 - x[half:], 1.0) ** 2
    piece = np.concatenate([ramp_left, tail_right])
    inits.append(normalize(piece))

    r = 1.0 + 0.4 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    inits.append(normalize(r))

    return inits
