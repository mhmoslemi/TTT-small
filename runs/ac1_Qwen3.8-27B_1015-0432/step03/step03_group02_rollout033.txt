        if time.monotonic() > deadline:
            break
        for _ in range(8):
            d = rng.standard_normal(n)
            trial = best_f * np.exp(scale * d)
            np.maximum(trial, 1e-30, out=trial)
            trial /= trial.sum()
            sc = score_ct(trial)
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial.copy()
        scale *= 0.94
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    inits = [p / p.sum(), np.ones(n) / n]
    x = np.arange(n, dtype=np.float64)
    for pw in (0.4, 0.8, 1.2, 2.0, 3.0):
        d = np.minimum(x, n - 1.0 - x) + 1.0
        inits.append(d ** pw / (d ** pw).sum())
    for ph in (0.0, 0.25, 0.5):
        v = 0.5 * (1.0 - np.cos(2.0 * np.pi * (x + ph) / (n + 1.0)))
        inits.append(v / v.sum())
    for a in (0.1, 0.25, 0.4):
        r = np.ones(n) + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        r = 0.5 * (r + r[::-1])
        inits.append(r / r.sum())
    best_f = inits[0].copy()
    best_sc = score_ct(best_f)
    for v in inits[1:]:
        sc = score_ct(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()
    ph1 = min(deadline, t0 + 380.0)
    f1, s1 = run_opt(best_f.copy(), ph1, rng, eta_init=0.12)
    if s1 < best_sc:
        best_sc, best_f = s1, f1
    if time.monotonic() < deadline - 160.0:
        fs = 0.5 * (best_f + best_f[::-1])
        fs /= fs.sum()
        ss = score_ct(fs)
        if ss < best_sc:
            best_sc, best_f = ss, fs
        ph2 = min(deadline, t0 + 720.0)
        f2, s2 = run_opt(best_f.copy(), ph2, rng, eta_init=0.06, check_iv=15)
        if s2 < best_sc:
            best_sc, best_f = s2, f2
    if time.monotonic() < deadline - 90.0:
        for rr in range(8):
            if time.monotonic() > deadline - 40.0:
                break
            amp = 0.01 + 0.01 * rr
            fr = best_f * np.exp(amp * rng.standard_normal(n))
            np.maximum(fr, 1e-30, out=fr)
            fr /= fr.sum()
            fr, sr = run_opt(fr, deadline - 45.0, rng, eta_init=0.08)
            if sr < best_sc:
                best_sc, best_f = sr, fr
    if time.monotonic() < deadline - 12.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 6.0, rng)
    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
