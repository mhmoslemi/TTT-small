def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def nscore(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * f.size * b.max() / (s * s))

def lse_grad(f, b, beta):
    n = f.size
    M = b.max()
    bm = b - M
    bm = np.clip(bm, -40.0 / beta, 0.0)
    ek = np.exp(beta * bm)
    w = ek / ek.sum()
    c = fft_conv(f, w[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return g, M
