import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(f.shape[0]) / f.shape[0]
    return f / s

    n = f.shape[0]
    c = fft_conv(f, f)
    s = f.sum()
    return 2.0 * n * c.max() / (s * s), c
