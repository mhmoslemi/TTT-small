def optimize(f, deadline, rng, power=20, eta0=0.15):
    n = f.shape[0]
    f = np.maximum(f, 1e-20).astype(np.float64)
    f /= f.sum()
    best_f = f.copy()
    best_sc = get_score(f)
    eta = eta0
    no_improv = 0
    eval_count = 0
    
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        g = compute_subgrad(f, b, power)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f_new = f * np.exp(-eta * g / gmax)
        f_new = np.maximum(f_new, 1e-20)
        f_new /= f_new.sum()
        eval_count += 1
        if eval_count % 20 == 0:
            sc = get_score(f_new)
            if sc < best_sc - 1e-14:
                best_sc = sc
                best_f = f_new.copy()
                no_improv = 0
            else:
                no_improv += 1
            f = f_new
            if no_improv > 400:
                # Restart
                f = best_f * np.exp(0.05 * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improv = 0
                eta = min(0.4, eta * 1.3)
            eta *= 0.998
        else:
            f = f_new
    
    # Final check
    sc = get_score(f)
    if sc < best_sc:
        best_sc = sc
        best_f = f.copy()
    return best_f, best_sc
