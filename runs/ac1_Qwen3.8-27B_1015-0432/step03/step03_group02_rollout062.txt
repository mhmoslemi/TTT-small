```python
lo = max(0, k - n + 1)
hi = min(k, n - 1)
if lo <= hi:
    # i in [lo, hi], k-i in [k-hi, k-lo]
    grad[lo:hi+1] = 2.0 * best_f[k-hi:k-lo+1][::-1]
```

This is O(n) and vectorized. 

Now, the subgradient of the score:
score = 2n*M/S² where M = b[k*], S = sum(f)

```python
import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    s = f.sum()
    if s < 1e-15:
        return np.inf
    b = fft_conv(f, f)
    return 2.0 * len(f) * b.max() / (s * s)

def peak_pressure(f, b, max_peaks, frac):
    n = len(f)
    L = 2 * n - 1
    M = b.max()
    thresh = M * frac
    h = np.zeros(L)
    active = b >= thresh
    cnt = active.sum()
    if cnt > max_peaks:
        idx = np.argpartition(b, -max_peaks)[-max_peaks:]
        h[idx] = b[idx] - thresh
    elif cnt >= 1:
        h[active] = b[active] - thresh
    else:
        h[np.argmax(b)] = 1.0
    if h.sum() < 1e-30:
        idx = np.argpartition(b, -max_peaks)[-max_peaks:]
        h[idx] = np.maximum(b[idx] - thresh, 1e-10)
    g = fft_conv(f, h[::-1])[n - 1:2 * n - 1][::-1]
    np.maximum(g, 0.0, out=g)
    return g

def run_opt(f, deadline, rng, max_peaks=256, eta_init=0.12, check_iv=20):
    n = f.shape[0]
    f = np.maximum(f, 0.0).copy()
    f /= f.sum()
    best_f = f.copy()
    best_sc = score(f)
    eta = eta_init
    patience = 0
    it = 0
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        if M < 1e-30:
            break
        prog = min(1.0, it / 10000.0)
        frac = 0.985 - 0.04 * prog
        g = peak_pressure(f, b, max_peaks, frac)
        gmax = g.max()
        if gmax < 1e-30:
            break
        f *= np.exp(-eta * g / gmax)
        np.maximum(f, 1e-30, out=f)
        f /= f.sum()
        it += 1
        if it % check_iv == 0:
            sc = score(f)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            if patience > 500:
                sc2 = 0.015 + 0.03 * min(1.0, patience / 1200.0)
                f = best_f * np.exp(sc2 * rng.standard_normal(n))
                np.maximum(f, 1e-30, out=f)
                f /= f.sum()
                patience = 0
                eta = min(0.35, eta * 1.4)
            if it > 400:
                eta *= 0.9992
    return best_f, best_sc

def subgradient_tune(f, deadline, rng):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = score(f)
    step = 0.03
    stall = 0
    while time.monotonic() < deadline:
        b = fft_conv(best_f, best_f)
        k = int(np.argmax(b))
        M = b[k]
        S = best_f.sum()
        if S < 1e-15:
            break
        grad = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] = 2.0 * best_f[k - hi:k - lo + 1][::-1]
        lo2 = max(k, 0)
        hi2 = min(k + n - 1, n - 1)
        if lo2 <= hi2 and lo2 > k:
            grad[lo2:hi2 + 1] += 2.0 * best_f[lo2 - k:hi2 - k + 1]
        c1 = 2.0 * n / (S * S)
        c2 = 4.0 * n * M / (S * S * S)
        sg = c1 * grad - c2
        sg_norm = np.abs(sg).max()
        if sg_norm < 1e-30:
            break
        exponent = np.clip(-step * sg / sg_norm, -0.06, 0.06)
        trial = best_f * np.exp(exponent)
        np.maximum(trial, 1e-30, out=trial)
        trial /= trial.sum()
        sc = score(trial)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best_f = trial
            stall = 0
        else:
            stall += 1
            for _ in range(4):
                d = rng.standard_normal(n) * 0.004
                t2 = best_f * np.exp(d)
                np.maximum(t2, 1e-30, out=t2)
                t2 /= t2.sum()
                sc2 = score(t2)
                if sc2 < best_sc - 1e-14:
                    best_sc = sc2
                    best_f = t2.copy()
                    stall = 0
                    break
            step *= 0.96
            if stall > 250:
                break
    return best_f, best_sc
```
