def solve_direction(f, M, K=200):
    n = len(f)
    C = conv(f)  # length 2n-1
    # Get top K indices
    idx = np.argsort(C)[-K:]
    A = np.zeros((K, n))  # this is K*n = 200*6733 = 1.3M, fine
    for i, k in enumerate(idx):
        # A[i, j] = f[k-j] if 0 <= k-j < n
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[i, j] = f[k-j]

    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K, M), bounds=(0, None), method='highs')
    return res.x
