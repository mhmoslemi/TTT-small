    c = fft_conv(f, hr)
    return 2.0 * c[(2 * n - 2) - np.arange(n)]

def optimize(f0, deadline, rng, sym=False, eta0=0.05, check_every=12):
    n = f0.shape[0]
    L = 2 * n - 1
    f = np.maximum(f0.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0.copy(), np.inf
    f = f / s
    bf = f.copy()
    bs = score_of(f)
    eta = eta0
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        it += 1
        b = fft_conv(f, f)
        M = b.max()
        n_top = max(40, n // 80)
        thr = M * 0.9997
        idx1 = np.where(b >= thr)[0]
        idx2 = np.argsort(b)[-n_top:]
        pk = np.unique(np.concatenate((idx1, idx2)))
        h = np.zeros(L)
        w = b[pk]
        h[pk] = w / (w.sum() + 1e-30)
        sg = get_subgrad(f, h)
        gmax = sg.max()
        if gmax < 1e-30:
            break
        f = f * np.exp(-eta * sg / gmax)
        f = np.maximum(f, 1e-25)
        if sym:
            f = 0.5 * (f + f[::-1])
            f = np.maximum(f, 0)
        f = f / f.sum()
        stall += 1
        if stall >= check_every:
            sc = score_of(f)
            if sc < bs - 1e-12:
                bs = sc
                bf = f.copy()
                stall = 0
                if it % (check_every * 20) == 0:
                    eta *= 0.97
            else:
                stall += 1
                if stall > 400:
                    amp = 0.004 + 0.02 * min((stall - 400) / 4000.0, 1.0)
                    f = bf * np.exp(amp * rng.standard_normal(n))
                    f = np.maximum(f, 1e-25)
                    if sym:
                        f = 0.5 * (f + f[::-1])
                        f = np.maximum(f, 0)
                    f = f / f.sum()
                    stall = 0
                    eta = eta0 * 0.6
        eta = max(eta, 0.0008)
    sc = score_of(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def local_refine(bf, bs, deadline, rng):
    n = bf.shape[0]
    L = 2 * n - 1
    f = bf.copy()
    while time.monotonic() < deadline:
        b = fft_conv(f, f)
        M = b.max()
        k_max = int(b.argmax())
        n_top = max(20, n // 200)
        thr = M * 0.9999
        idx1 = np.where(b >= thr)[0]
        idx2 = np.argsort(b)[-n_top:]
        pk = np.unique(np.concatenate((idx1, idx2)))
        h = np.zeros(L)
        h[pk] = 1.0
        sg = get_subgrad(f, h)
        gmax = sg.max()
        if gmax < 1e-30:
            break
        eta_local = 0.003
        f = f * np.exp(-eta_local * sg / gmax)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()
        sc = score_of(f)
        if sc < bs - 1e-12:
            bs = sc
            bf = f.copy()
        else:
            f = bf.copy()
            break
    return bf, bs

def perturb_search(bf, bs, deadline, rng, count=200, amp=0.003):
    n = bf.shape[0]
    for i in range(count):
        if time.monotonic() > deadline:
            break
        tr = bf * np.exp(amp * rng.standard_normal(n))
        tr = np.maximum(tr, 0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = score_of(tr)
        if sc < bs - 1e-13:
            bs = sc
            bf = tr.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):

    inits.append(h_cos / h_cos.sum())

    for pw in (0.5, 1.0, 2.0):
        v = d ** pw
        inits.append(v / v.sum())
    rv = np.maximum(rng.standard_normal(n), 0.005)
    inits.append(rv / rv.sum())

    bf = None
    bs = np.inf
    for ini in inits:

        if sc < bs:
            bs = sc
            bf = ini.copy()

        if ss > 1e-15:

        f2, s2 = optimize(fs, p2, rng, sym=True, eta0=0.04)

    result = [float(v) for v in bf]
