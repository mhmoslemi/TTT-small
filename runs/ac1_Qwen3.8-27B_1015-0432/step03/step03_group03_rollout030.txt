        alpha_cur = max(10.0, alpha0 * np.exp(-0.0003 * it))
        w_full = lse_weights(b, alpha_cur)

        # Restrict to top peaks for efficiency
        thresh_idx = np.where(w_full > w_full.max() * 1e-4)[0]
        if len(thresh_idx) > max_peaks:
            top = np.argsort(w_full[thresh_idx])[::-1][:max_peaks]
            thresh_idx = thresh_idx[top]
        if len(thresh_idx) < 1:
            thresh_idx = np.array([np.argmax(b)])
        w = w_full[thresh_idx]
        w = w / w.sum()

        g = get_grad(f, thresh_idx, w)

        # Small flatness regularization to avoid over-concentration
        if flat_reg > 0:
            g = g + flat_reg * n * (f - 1.0 / n)

        gmax = g.max()
        if gmax < 1e-28:
            break

        momentum = 0.88 * momentum + 0.12 * g
        mmax = momentum.max()
        if mmax < 1e-28:
            break

        # Adaptive step: clamp effective step
        step = np.clip(eta * momentum / mmax, 0, 0.5)
        f = f * np.exp(-step)
        f = np.maximum(f, 1e-20)

        if sym:
            f = 0.5 * (f + f[::-1])
        f = f / f.sum()

        # Evaluate periodically
        if it % 25 == 0:
            sc = 2.0 * n * fft_conv(f, f).max()
            if sc < best_sc - 1e-11:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1

            if no_improv > 700:
                scale = 0.015 + 0.025 * min(1.0, restart_cnt / 10.0)
                pert = rng.standard_normal(n)
                if sym:
                    pert = 0.5 * (pert + pert[::-1])
                f = best_f * np.exp(scale * pert)
                f = np.maximum(f, 1e-15)
                if sym:
                    f = 0.5 * (f + f[::-1])
                f = f / f.sum()
                momentum = np.zeros(n)
                no_improv = 0
                restart_cnt += 1
                eta = min(0.25, eta * 1.3)

        eta *= 0.99985

    final_sc = 2.0 * n * fft_conv(best_f, best_f).max()
    if final_sc < best_sc:
        best_sc = final_sc
    return best_f, best_sc

def polish(f, deadline, rng, n_trials=400):
    n = f.shape[0]
    best_f = f.copy()
    best_sc = 2.0 * n * fft_conv(f, f).max()
    scales = [0.0005, 0.001, 0.003, 0.007, 0.015]
    for trial in range(n_trials):
        if time.monotonic() > deadline - 5:
            break
        scale = scales[trial % len(scales)]
        for attempt in range(3):
            if time.monotonic() > deadline - 5:
                break
            pert = rng.standard_normal(n)
            pert = 0.5 * (pert + pert[::-1])
            trial_f = best_f * np.exp(scale * pert)
            trial_f = np.maximum(trial_f, 0)
            trial_f = trial_f / trial_f.sum()
            sc = 2.0 * n * fft_conv(trial_f, trial_f).max()
            if sc < best_sc - 1e-13:
                best_sc = sc
                best_f = trial_f.copy()
    return best_f, best_sc

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    # Initialize from parent
    f = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_sc = obj_val(f)

    # Try a few quick structured inits
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.5, 1.5]:
        init = d ** pw
        init = init / init.sum()
        sc = 2.0 * n * fft_conv(init, init).max()
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: Asymmetric optimization
    t1 = min(deadline - 350, t0 + 350)
    f1, sc1 = optimize(best_f.copy(), t1, rng, eta0=0.06, alpha0=60.0,
                       max_peaks=200, sym=False, max_iter=200000, flat_reg=0.5e-4)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1

    # Phase 2: Symmetric optimization
    if time.monotonic() < deadline - 250:
        f2 = 0.5 * (best_f + best_f[::-1])
        f2 = f2 / f2.sum()
        t2 = min(deadline - 250, t0 + 650)
        f2o, sc2 = optimize(f2, t2, rng, eta0=0.05, alpha0=80.0,
                            max_peaks=300, sym=True, max_iter=200000, flat_reg=1e-4)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2o

    # Phase 3: Fine-tuning with sharper peaks
    if time.monotonic() < deadline - 120:
        t3 = min(deadline - 120, t0 + 880)
        f3o, sc3 = optimize(best_f.copy(), t3, rng, eta0=0.025, alpha0=120.0,
                            max_peaks=500, sym=True, max_iter=200000, flat_reg=0.0)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3o

    # Phase 4: Polish with random perturbations
    if time.monotonic() < deadline - 15:
        fp, scp = polish(best_f, deadline - 10, rng, n_trials=500)
        if scp < best_sc:
            best_sc = scp
            best_f = fp

    # Final: ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = np.ones(n) / n
    result = [float(v) for v in best_f]

    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n

    return result
