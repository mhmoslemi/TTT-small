                scale = 0.02 + 0.04 * min(1.0, no_improv / 2000.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if n_sym:
                    f = 0.5 * (f + f[::-1])
                f = norm_pos(f)
                log_f = np.log(np.maximum(f, 1e-30))
                m = np.zeros(n)
                v = np.zeros(n)
                no_improv = 0
    return best_f, best_obj

def gen_inits(n, rng):
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(norm_pos(p.copy()))
    inits.append(np.ones(n) / n)
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.3, 0.7, 1.5, 2.5]:
        inits.append(norm_pos(d ** pw))
    h1 = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n + 1)))
    inits.append(norm_pos(h1))
    h2 = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n + 1))) + 1e-12
    inits.append(norm_pos(h2))
    for freq in [2.0, 3.0, 5.0, 8.0]:
        wave = 0.5 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        inits.append(norm_pos(np.maximum(wave, 0.01)))
    r = np.ones(n) + 0.4 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    r = 0.5 * (r + r[::-1])
    inits.append(norm_pos(r))
    half = n // 2
    two_lobe = np.ones(n)
    two_lobe[:half] = 1.0 + 0.5 * np.cos(np.pi * np.arange(half) / max(half, 1))
    two_lobe[half:] = 1.0 + 0.5 * np.cos(np.pi * np.arange(n - half) / max(n - half + 1, 1))
    inits.append(norm_pos(np.maximum(two_lobe, 0.01)))
    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    inits = gen_inits(n, rng)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = obj_val(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    t_p1 = min(deadline - 400.0, t0 + 330.0)
    f1, sc1 = optimize(best_f.copy(), t_p1, rng, eta0=0.12, beta_start=3.0,
                       beta_end=10.0, max_peaks=1500)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()

    if time.monotonic() < deadline - 250.0:
        t_p2 = min(deadline - 250.0, t0 + 600.0)
        f2 = 0.5 * (best_f + best_f[::-1])
        f2 = norm_pos(f2)
        f2o, sc2o = optimize(f2, t_p2, rng, eta0=0.08, beta_start=5.0,
                             beta_end=18.0, max_peaks=2200, n_sym=True)
        if sc2o < best_sc:
            best_sc = sc2o
            best_f = f2o.copy()

    if time.monotonic() < deadline - 100.0:
        t_p3 = min(deadline - 100.0, t0 + 850.0)
        f3o, sc3o = optimize(best_f.copy(), t_p3, rng, eta0=0.05,
                             beta_start=8.0, beta_end=25.0, max_peaks=3000)
        if sc3o < best_sc:
            best_sc = sc3o
            best_f = f3o.copy()

    if time.monotonic() < deadline - 40.0:
        t_p4 = min(deadline - 40.0, t0 + 1020.0)
        f4o, sc4o = optimize(best_f.copy(), t_p4, rng, eta0=0.03,
                             beta_start=12.0, beta_end=35.0, max_peaks=4000)
        if sc4o < best_sc:
            best_sc = sc4o
            best_f = f4o.copy()

    if time.monotonic() < deadline - 15.0:
        for _ in range(200):
            if time.monotonic() > deadline - 10.0:
                break
            scale = 0.003 + 0.002 * rng.random()
            trial = best_f * np.exp(scale * rng.standard_normal(n))
            trial = norm_pos(trial)
            sc = obj_val(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = [float(x) for x in np.maximum(best_f, 0.0)]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
