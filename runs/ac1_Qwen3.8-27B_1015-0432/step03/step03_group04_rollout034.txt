def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    if size < 4:
        size = 4
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def get_score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return 1e30
    return float(2.0 * f.size * b.max() / (s * s))

def compute_grad(f, b, temp, max_pk):
    n = f.shape[0]
    L = 2 * n - 1

    h[peaks] = w
    c = fft_conv(h, f[::-1])
    g = 2.0 * c[n - 1:2 * n - 1]
    return np.maximum(g, 0.0)
