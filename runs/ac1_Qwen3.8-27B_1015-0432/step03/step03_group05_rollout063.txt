        if it % 15 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                ni = 0
            else:
                ni += 1
            if ni > 350:
                sp = 0.004 + 0.012 * min(1.0, ni / 4000.0)
                f = bf * np.exp(sp * rng.standard_normal(n))
                f = np.maximum(f, 1e-20)
                if sym:
                    f = (f + f[::-1]) * 0.5
                f = f / f.sum()
                ni = 0
                eta = eta0
        if it > 30000:
            eta *= 0.99998
        if eta < 0.0005:
            eta = 0.0005
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def polish(f, deadline, rng, sym=False):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f = f / s
    bf = f.copy()
    bs = score(f)
    for it in range(200000):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        thr = M * 0.99999
        pk = np.where(b >= thr)[0]
        if len(pk) < 2:
            pk = np.array([np.argmax(b)])
        if len(pk) > 300:
            pk = np.argsort(b)[-300:]
        w = np.zeros(b.shape)
        w[pk] = 1.0
        g = corr_grad(f, w / len(pk))
        gm = g.max()
        if gm < 1e-30:
            break
        eta = 0.0015
        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-20)
        if sym:
            f = (f + f[::-1]) * 0.5
        f = f / f.sum()
        if it % 50 == 0:
            sc = score(f)
            if sc < bs - 1e-15:
                bs = sc
                bf = f.copy()
    sc = score(f)
    if sc < bs:
        bs = sc
        bf = f.copy()
    return bf, bs

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    dl = t0 + budget_s - 25.0
    n = len(height_sequence_1)
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(p / p.sum())
    inits.append(np.ones(n, dtype=np.float64) / n)
    x = np.arange(n, dtype=np.float64)
    h = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    inits.append(h / h.sum())
    d = np.minimum(x, n - 1 - x) + 1.0
    for pw in (0.3, 0.7, 1.3, 2.0):
        v = d ** pw
        inits.append(v / v.sum())
    uu = (x / (n - 1.0)) ** 2 + (1.0 - x / (n - 1.0)) ** 2 + 0.01
    inits.append(uu / uu.sum())
    rv = np.maximum(rng.standard_normal(n), 0.01)
    inits.append(rv / rv.sum())
    bf, bs = None, np.inf
    for ini in inits:
        sc = score(ini)
        if sc < bs:
            bs = sc
            bf = ini.copy()
    f1, s1 = optimize(bf, min(dl, t0 + 400), rng, False, 300000, 0.04)
    if s1 < bs:
        bs, bf = s1, f1
    if time.monotonic() < dl - 300:
        fs = np.maximum((bf + bf[::-1]) * 0.5, 0)
        if fs.sum() > 1e-15:
            fs = fs / fs.sum()
            f2, s2 = optimize(fs, min(dl, t0 + 750), rng, True, 300000, 0.03)
            if s2 < bs:
                bs, bf = s2, f2
    if time.monotonic() < dl - 180:
        f3, s3 = optimize(bf.copy(), min(dl, t0 + 920), rng, False, 200000, 0.015)
        if s3 < bs:
            bs, bf = s3, f3
    if time.monotonic() < dl - 40:
        f4, s4 = polish(bf.copy(), dl - 10, rng, False)
        if s4 < bs:
            bs, bf = s4, f4
    if time.monotonic() < dl - 3:
        for i in range(300):
            if time.monotonic() > dl - 1:
                break
            tr = bf * np.exp(0.002 * rng.standard_normal(n))
            tr = np.maximum(tr, 0)
            ts = tr.sum()
            if ts < 1e-15:
                continue
            tr = tr / ts
            sc = score(tr)
            if sc < bs:
                bs, bf = sc, tr.copy()
    result = [float(v) for v in bf]
    if not np.isfinite(evaluate_sequence(result)):
        result = [1.0 / n] * n
    return result
