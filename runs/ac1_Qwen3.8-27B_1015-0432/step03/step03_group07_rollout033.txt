import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    return 2.0 * f.size * b.max() / (s * s)

def build_F(f):
    """Build sparse convolution matrix F where F @ g = conv(f, g)."""
    n = len(f)
    # T[k,j] = 1 if j <= k <= j+n-1
    diags = [np.ones(n) for _ in range(n)]
    offsets = list(range(n))
    T = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    F = T.multiply(f[np.newaxis, :]).tocsc()
    return F

def lp_step(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. F@g <= M, g >= 0."""
    n = len(f)
    F = build_F(f)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=F, b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'time_limit': time_limit, 'presolve': True})
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-10:
            return g / g.sum()
    return None

def grad_descent(f, deadline, rng, max_iters=100000):
    """Multiplicative gradient descent."""
    n = f.size
    f = f.copy()
    f = np.maximum(f, 1e-20)
    f /= f.sum()
    best_f = f.copy()
    best_sc = score(f)
    eta = 0.1
    no_improv = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        # Soft-max weight
        T = 0.05 * M
        w = np.exp((b - M) / T)
        w /= w.sum()
        # Gradient: contribution of each f[i] to weighted autocorrelation
        c = fft_conv(f, w[::-1])
        g = c[n-1:2*n-1]
        g = np.maximum(g, 1e-30)
        # Update
        f = f * np.exp(-eta * (g - g.mean()) / (g.std() + 1e-30))
        f = np.maximum(f, 1e-20)
        f /= f.sum()
        if it % 20 == 0:
            sc = score(f)
            if sc < best_sc:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1
            if no_improv > 200:
                pert = rng.standard_normal(n) * 0.05
                f = best_f * np.exp(pert)
                f = np.maximum(f, 1e-20)
                f /= f.sum()
                no_improv = 0
        eta *= 0.999
    return best_f, best_sc
