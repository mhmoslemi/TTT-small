import numpy as np

def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def ac1(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def soft_subgrad(f, b, beta, top_k):
    n = f.shape[0]
    L = 2 * n - 1
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)
    if top_k < L:
        idx = np.argpartition(b, -top_k)[-top_k:]
        w_full = np.zeros(L)
        vals = b[idx]
        w_full[idx] = np.exp(beta * (vals - M) / M)
    else:
        w_full = np.exp(beta * (b - M) / M)
    if w_full.sum() < 1e-30:
        w_full = np.zeros(L)
        w_full[int(np.argmax(b))] = 1.0
    c = fft_conv(f, w_full[::-1])
    g = 2.0 * c[n - 1:2 * n - 1][::-1]
    return np.maximum(g, 0.0)
