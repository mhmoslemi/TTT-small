from scipy.optimize import minimize
from numpy.fft import fft, ifft

def _fpow2(s):
    m = 1
    while m < s:
        m <<= 1
    return m

def _conv(a):
    n = a.size
    L = _fpow2(2 * n - 1)
    fa = fft(a, L)
    ifa = ifft(fa * fa).real
    return ifa[:2 * n - 1]

def _smax(c, tau):
    cm = c.max()
    d = np.maximum((c - cm) / tau, -50.0)
    e = np.exp(d)
    s = e.sum()
    return cm + tau * np.log(s), e / s

def _obj(a, tau):
    n = a.size
    S = a.sum()
    if S < 1e-8:
        return 1e10
    mv, _ = _smax(_conv(a), tau)
    return 2.0 * n * mv / (S * S)

    S = a.sum()
