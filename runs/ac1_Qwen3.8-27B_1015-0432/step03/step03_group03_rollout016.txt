        q[peaks] = 1.0
        g = compute_grad(f, q)
        g = np.maximum(g, 0.0)
        gmax = g.max()
        if gmax < 1e-30:
            break
        g = g / gmax

        beta1, beta2 = 0.9, 0.999
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * g * g
        m1_hat = m1 / (1 - beta1 ** t_adam)
        m2_hat = m2 / (1 - beta2 ** t_adam)

        step = eta * m1_hat / (np.sqrt(m2_hat) + 1e-10)
        step = np.minimum(step, 3.0)
        f = f * np.exp(-step)
        f = np.maximum(f, 1e-25)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s

        if it % 30 == 0:
            sc = score_of(f)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_f = f.copy()
                no_improv = 0
            else:
                no_improv += 1

            if no_improv > 500:
                scale = 0.04 + 0.04 * min(1.0, restart_count / 12.0)
                f = best_f * np.exp(scale * rng.standard_normal(n))
                f = np.maximum(f, 1e-12)
                f = normalize(f)
                m1[:] = 0
                m2[:] = 0
                t_adam = 0
                no_improv = 0
                restart_count += 1
                eta = min(0.25, eta * 1.25)

        eta *= 0.9997

    return best_f, best_sc

def gen_inits(n, rng):
    inits = []
    p = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    if p.sum() > 1e-10:
        inits.append(normalize(p.copy()))

    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1.0 - x) + 1.0
    for pw in [0.4, 1.0, 2.0]:
        inits.append(normalize(d ** pw))

    h1 = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / n))
    inits.append(normalize(h1 + 1e-12))

    for freq in [2, 4, 7]:
        wave = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        inits.append(normalize(wave + 0.05))

    r = np.ones(n) + 0.4 * rng.standard_normal(n)
    r = np.maximum(r, 0.01)
    inits.append(normalize(r))

    # Two-lobe structure
    half = n // 2
    two = np.ones(n)
    t1 = np.arange(half)
    t2 = np.arange(n - half)
    two[:half] = 0.8 + 0.4 * np.cos(np.pi * t1 / half)
    two[half:] = 0.8 - 0.4 * np.cos(np.pi * t2 / (n - half + 1))
    inits.append(normalize(np.maximum(two, 0.01)))

    return inits

def propose_candidate(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + budget_s - 25.0
    n = len(height_sequence_1)

    inits = gen_inits(n, rng)
    best_f = None
    best_sc = np.inf
    for init in inits:
        sc = score_of(init)
        if sc < best_sc:
            best_sc = sc
            best_f = init.copy()

    # Phase 1: coarse optimization
    t_p1 = min(deadline - 400.0, t0 + 300.0)
    f1, sc1 = multiplicative_optimize(best_f.copy(), t_p1, rng, eta0=0.08, n_peaks=150)
    if sc1 < best_sc:
        best_sc = sc1
        best_f = f1.copy()

    # Phase 2: refined with more peaks
    if time.monotonic() < deadline - 200.0:
        t_p2 = min(deadline - 180.0, t0 + 650.0)
        f2, sc2 = multiplicative_optimize(best_f.copy(), t_p2, rng, eta0=0.05, n_peaks=250)
        if sc2 < best_sc:
            best_sc = sc2
            best_f = f2.copy()

    # Phase 3: fine polishing
    if time.monotonic() < deadline - 100.0:
        t_p3 = min(deadline - 80.0, t0 + 950.0)
        f3, sc3 = multiplicative_optimize(best_f.copy(), t_p3, rng, eta0=0.03, n_peaks=300)
        if sc3 < best_sc:
            best_sc = sc3
            best_f = f3.copy()

    # Phase 4: local perturbation search
    if time.monotonic() < deadline - 20.0:
        for _ in range(200):
            if time.monotonic() > deadline - 10.0:
                break
            trial = best_f * np.exp(0.003 * rng.standard_normal(n))
            trial = normalize(trial)
            sc = score_of(trial)
            if sc < best_sc:
                best_sc = sc
                best_f = trial.copy()

    result = [float(max(0.0, min(1000.0, x))) for x in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
    return result
