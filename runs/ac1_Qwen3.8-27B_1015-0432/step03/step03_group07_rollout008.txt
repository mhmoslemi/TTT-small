def fft_conv(a, b):
    n, m = len(a), len(b)
    size = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size))[:n + m - 1]

def sc(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * f.size * b.max() / (s * s))

def grad_mc(f, b, thresh):
    n = f.shape[0]
    M = b.max()
    if M < 1e-30:
        return np.zeros(n)

    w = w * w
    if w.sum() < 1e-30:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    c = fft_conv(f, w[::-1])
