import numpy as np
import time

def fft_conv(a, b):
    n, m = len(a), len(b)
    sz = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, sz) * np.fft.rfft(b, sz))[:n + m - 1]

def score(f):
    b = fft_conv(f, f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * f.size * b.max() / (s * s)

def subgrad(f, peaks):
    n = f.shape[0]
    L = 2 * n - 1
    h = np.zeros(L)
    h[peaks] = 1.0
    c = fft_conv(f, h[::-1])
    return 2.0 * c[n - 1:2 * n - 1][::-1]

def optimize(f, deadline, rng, max_it=200000, eta0=0.05, peak_frac=0.995, max_peaks=400):
    n = f.shape[0]
    f = np.maximum(f.copy(), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f, np.inf
    f /= s
    bf = f.copy()
    bs = score(f)
    eta = eta0
    stalling = 0
    for it in range(max_it):
        if time.monotonic() > deadline:
            break
        b = fft_conv(f, f)
        M = b.max()
        thr = M * peak_frac
        pk = np.where(b >= thr)[0]
        if len(pk) < 2:
            pk = np.argsort(b)[-20:]
        elif len(pk) > max_peaks:
            idx = np.argsort(b)[::-1][:max_peaks]
            pk = idx
        g = subgrad(f, pk)
        gm = g.max()
        if gm < 1e-30:

        f = f * np.exp(-eta * g / gm)
        f = np.maximum(f, 1e-25)
        f /= f.sum()
        if it % 5 == 0:
            sc = score(f)
            if sc < bs - 1e-14:
                bs = sc
                bf = f.copy()
                stalling = 0
            else:
                stalling += 1
            if stalling > 400:

                f = bf * np.exp(amp * rng.standard_normal(n))
                f = np.maximum(f, 1e-25)

                stalling = 0

        sc = get_score(ini)
        if sc < bs:
            bs, bf = sc, ini.copy()

    p1 = min(dl, t0 + 400)
    f, s = opt(bf.copy(), p1, rng, eta0=0.06, max_peaks=250, pf=0.99, alpha=1.0)
    if s < bs:
        bs, bf = s, f

    if time.monotonic() < dl - 150:
        p2 = min(dl, t0 + 720)
        f, s = opt(bf.copy(), p2, rng, eta0=0.03, max_peaks=400, pf=0.994, alpha=2.0)
        if s < bs:
            bs, bf = s, f

    if time.monotonic() < dl - 80:
        p3 = min(dl - 50, t0 + 900)
        f, s = opt(bf.copy(), p3, rng, eta0=0.015, max_peaks=200, pf=0.997, alpha=3.0)
        if s < bs:
            bs, bf = s, f

    if time.monotonic() < dl - 15:
        f, s = polish(bf.copy(), dl - 5, rng)
        if s < bs:
            bs, bf = s, f

    for _ in range(60):
        if time.monotonic() > dl - 2:
            break
        tr = bf * np.exp(0.002 * rng.standard_normal(n))
        tr = np.maximum(tr, 0)
        ts = tr.sum()
        if ts < 1e-15:
            continue
        tr /= ts
        sc = get_score(tr)
        if sc < bs:
            bs, bf = sc, tr.copy()

    result = [float(v) for v in bf]
    fs = evaluate_sequence(result)
    if not np.isfinite(fs):
        result = [1.0 / n] * n
    return result
