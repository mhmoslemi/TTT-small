```python
def soft_grad(f, t):
    """Compute gradient of t * logsumexp(conv(f,f)/t) w.r.t. f"""
    n = f.shape[0]
    b = fft_conv(f, f)
    # Numerically stable logsumexp
    bm = b.max()
    weights = np.exp((b - bm) / t)
    W = weights.sum()
    # gradient: sum_k (w_k / W) * (2 * conv(f, e_k shifted))
    # Use FFT: weight the signal and convolve
    # grad_i = 2 * sum_k (w_k/W) * f_shifted
    # where f_shifted depends on k
    # This is: 2 * conv(w_reversed / W, f)[n-1:2n-1] reversed
    wr = weights[::-1] / W
    c = fft_conv(f, wr)
    return 2.0 * c[n-1:2*n-1][::-1]
```
