    x = np.arange(n, dtype=np.float64)
    for pw in [0.3, 0.6, 1.0, 1.5, 2.5]:
        d = np.minimum(x, n - 1.0 - x) + 1.0
        v = d ** pw
        inits.append(make_sym(v / v.sum()))
    for ph in [0.0, 0.3, 0.6]:
        h2 = 1.0 - np.cos(2.0 * np.pi * (x + ph) / n)
        inits.append(make_sym(np.maximum(h2, 0.01) / h2.sum()))
    for a in [0.08, 0.2, 0.4]:
        r = 1.0 + a * rng.standard_normal(n)
        np.maximum(r, 0.05, out=r)
        inits.append(make_sym(r))

    best_f = inits[0].copy()
    best_sc = score(best_f)
    for v in inits[1:]:
        sc = score(v)
        if sc < best_sc:
            best_sc = sc
            best_f = v.copy()

    phase1_end = min(deadline, t0 + 280.0)
    f1, s1 = run_opt(best_f.copy(), phase1_end, rng, max_peaks=100, eta_init=0.15, check_iv=10)
    if s1 < best_sc:
        best_sc, best_f = s1, f1

    if time.monotonic() < deadline - 120.0:
        phase2_end = min(deadline, t0 + 620.0)
        f2, s2 = run_opt(best_f.copy(), phase2_end, rng, max_peaks=200, eta_init=0.05, check_iv=15)
        if s2 < best_sc:
            best_sc, best_f = s2, f2

    if time.monotonic() < deadline - 50.0:
        for r in range(4):
            if time.monotonic() > deadline - 30.0:
                break
            amp = 0.012 + 0.012 * r
            f_r = best_f * np.exp(amp * rng.standard_normal(n))
            np.maximum(f_r, 1e-30, out=f_r)
            f_r = make_sym(f_r)
            sub_dl = min(deadline - 10.0, t0 + 1060.0)
            f_r, s_r = run_opt(f_r, sub_dl, rng, max_peaks=300, eta_init=0.04, check_iv=15)
            if s_r < best_sc:
                best_sc, best_f = s_r, f_r

    if time.monotonic() < deadline - 8.0:
        best_f, best_sc = fine_tune(best_f.copy(), deadline - 4.0, rng, steps=200)

    best_f = make_sym(best_f)
    result = [float(v) for v in best_f]
    final_sc = evaluate_sequence(result)
    if not np.isfinite(final_sc):
        result = [1.0 / n] * n
        final_sc = evaluate_sequence(result)
    return result
