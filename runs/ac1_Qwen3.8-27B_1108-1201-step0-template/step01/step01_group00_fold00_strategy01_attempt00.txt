def gaussian_basis(n_length, sigma=0.5, mu=0.75, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    t = np.arange(n_length, dtype=float)
    gaussian = rng.standard_normal(n_length) * sigma + mu
    gaussian = np.clip(gaussian, 0.01, 1000.00)
    return gaussian

def wave_packet(n_length, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    t = np.arange(n_length, dtype=float)
    # Generate a wave that tries to have cancellation in convolution
    freq_start, freq_end = 0.1, 0.9
    freq = freq_start + (freq_end - freq_start) * rng.random()
    # Wave equation with amplitude and frequency modulation
    amplitude = rng.uniform(0.5, 1.5)
    angle = freq * 2 * np.pi * t / n_length
    gaussian_sigma = rng.uniform(0.3, 0.8)
    gaussian = amplitude * np.exp(-(t - n_length/2)**2 / (2* gaussian_sigma**2))
    return gaussian + amplitude * np.cos(angle)  # Combine with a phase

def heuristic_scramble(sequence, rng=None):
    # Randomly permute the sequence to potentially disrupt order in convolution
    if rng is None:
        rng = np.random.default_rng()
    shuffled = sequence.copy()
    shuffled_indices = rng.permutation(len(shuffled))
    return [shuffled[i] * rng.uniform(0.8, 1.2) for i in shuffled_indices]
