def compute_subgradient(a, n, top_indices, c):
    """Compute the subgradient for the full convolution for the given active top indices.
    We are only given the top k indices, so we have to consider the respective lags for each active index.
    Parameters:
        a: normalized vector (numpy array)
        n: length of the sequence
        top_indices: active indices (from 0 to 2n-2)
        c: full convolution result (numpy array of length 2n-1)
    
    Returns:
        gradient vector (numpy array) which is the derivative of the max of the convolution terms
    """
    grad = np.zeros(n, dtype=np.float64)
    top_vals = c[top_indices]
    weights = top_vals / (np.sum(top_vals) if np.sum(top_vals) > 0.0 else 1.0)
    for w_val, k in zip(weights, top_indices):
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            # Take a slice of a indices from lo to hi (this is the part that contributes to the convolution at index k)
            slice_a = a[lo: hi+1]
            # Compute the reversed vector to correlate with the lag k
            reversed_slice = slice_a[::-1]
            grad[lo: hi+1] += w_val * 2.0 * reversed_slice
    return grad
