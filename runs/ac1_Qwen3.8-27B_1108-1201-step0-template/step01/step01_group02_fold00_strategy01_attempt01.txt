        current_score = evaluate_sequence(current_sequence)

        # Key procedure: get all positions and peppering weights
        # Apply perturbation pattern over the entire sequence length
        try:
            improved_seq = try_perturbation(current_sequence, n=len(current_sequence))
            current_score = score_with_adjustments(improved_seq, len(improved_seq))
            if current_score < best_score:
                best_sequence = improved_seq.copy()
                best_score = current_score
        except Timout or OptimalNotFurtherImproved:
            # Final convolution pass if improvement stuck
            new_sequence = do_final_convolution(current_sequence)
            best_sequence = new_sequence.copy()
            best_score = evaluate_sequence(new_sequence)
            break

    return best_sequence

# Helper functions
def initial_raised_cosine(n):
    t = np.arange(n, dtype=np.float64)
    return np.concatenate([0.5 * (1 + np.cos(2*pi*(t - center)/scale)), np.zeros(pad_length)])

def adjust_weights(seq, temp_factor):
    # Small, distributed adjustments
    new_seq = seq.copy() * (1 + 0.25 * temp_factor)
    # Hybr something for stronger boundaries
    center_vals = new_seq[int(len(seq)/4):int(len(seq)*3/4)]
    center_vals = center_vals * (0.5 + 0.1 * temp_factor * random_sign())
    return normalize(new_seq)

def try_perturbation(base_seq, n):
    # Random walk type adjustments
    new_seq = np.zeros(n)
    sweep = np.linspace(0, 1, n)
    points = np.exp(np.random.rand(n) * 1.5)
    points = points * (2.0 * (0.5) * temp_factor)
    for i in range(n):
        if np.random.rand() > points:
            new_seq[i] = base_seq[i] + random.small_uniform_variation()
        else:
            # More significant change
            new_seq[i] = evaluate_updated_value(base_seq, i)
    return new_seq

def score_with_adjustments(candidate, n):
    # Numerical minimization via relative scaling
    # Could involve solving an eigenproblem here, but too heavy
    # Let's use a derivative-free local minimizer
    bounds = [(0.01, None) for _ in range(n)]
    options = {'maxiter': 100, 'disp': False}
    result = siopt.differential_evolution(
        optimize_sequence_score,
        bounds,
        popsize=15,
        mutation=0.6,
        recombination=0.7,
        seed=42,
        tol=0.0001,
        poloccoptimizer,
        options=options)
    return result.x, result.fun

def optimize_sequence_score(seq):
    # Check if sequence is admissible
    scores = evaluate_sequence(seq)
    if np.any(seq < 0) or np.any(seq > 1000): return float('inf')
    return scores

def do_final_convolution(seq):
    # Ensure non-empty with smaller factor
    convolved = np.convolve(seq, seq, mode='full')
    max_val = convolved.max()
    if max_val > min_peak_value * (np.exp(np.random.rand()) ): return None
        
    # Equivalent to previous final pass
    return adjust_peak(seq, enhancement_factor=0.7, bounds=(-0.5, 0.5))

# The final return value
return best_sequence

**Mathematical Foundation:**
The algorithm explicitly tracks the following:
1. Unixed constraint of non-negative coefficients
2. Convolution operation is defined using the `evaluate_sequence` function
3. The score 2*n/max(conv)/sum(a)² is minimized through gradient-based numerical methods

This approach differs from the previous one by focusing attention on the combinatorial structure of coefficient updates rather than stochastic initialization.
