import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(x, target_sum):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - np.sum(x)
        if abs(d) < 1e-14:
            break
        if d > 0:
            room = 1.0 - x
            mask = room > 1e-14
            if not np.any(mask):
                break
            x[mask] = np.minimum(x[mask] + d / np.sum(room[mask]), 1.0)
        else:
            room = x
            mask = room > 1e-14
            if not np.any(mask):
                break
            x[mask] = np.maximum(x[mask] + d / np.sum(room[mask]), 0.0)
    return x

def soft_weights(corr, temp):
    shifted = corr - np.max(corr)
    e = np.exp(shifted / temp)
    return e / np.sum(e)

def obj_and_grad(h, temp, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    p = soft_weights(corr, temp)
    obj = float(np.max(corr) + temp * np.log(np.sum(np.exp((corr - np.max(corr)) / temp))))
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-13:
                continue
            off = k - (n - 1)
            t1 = (1.0 - h[j - off]) if 0 <= j - off < n else 0.0
            t2 = -h[j + off] if 0 <= j + off < n else 0.0
            s += pk * (t1 + t2)
        grad[j] = s * dx
    return obj, grad

def opt_traj(h0, dx, t_deadline):
    n = len(h0)
    ts = n / 2.0
    h = project_feasible(h0.copy(), ts)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.04
    temp = 0.04
    stall = 0
    while time.time() < t_deadline:
        obj, grad = obj_and_grad(h, temp, dx)
        h_new = project_feasible(h - lr * grad, ts)
        c5n = compute_c5(h_new, dx)
        accept = c5n < best_c5 or (np.random.random() < np.exp(-(c5n - best_c5) / max(temp * 0.1, 1e-6)))
        if c5n < best_c5:
            best_c5, best_h = c5n, h_new.copy()
            stall = 0
        if accept:
            h = h_new
        else:
            stall += 1
            lr *= 0.97
            temp *= 0.99
            if stall > 150:
                h = project_feasible(h + 0.15 * np.random.randn(n), ts)
                stall = 0
                lr, temp = 0.04, 0.04
        if stall > 600:
            break
    return best_h, best_c5

def make_init(n, typ, rng):
    ts = n / 2.0
    if typ == 0:
        return project_feasible(np.ones(n) * 0.5, ts)
    if typ == 1:
        nb = int(rng.choice([4, 6, 8, 10, 12, 16, 20]))
        hs = n // nb
        h = np.zeros(n)
        for b in range(nb):
            if rng.random() < 0.5:
                h[b*hs:(b+1)*hs] = 1.0
        return project_feasible(h, ts)
    if typ == 2:
        t = np.linspace(0, 2, n, endpoint=False)
        freqs = rng.uniform(1, 6, 4)
        phs = rng.uniform(0, 2*np.pi, 4)
        h = 0.5 + 0.35 * np.sin(2*np.pi*freqs[0]*t + phs[0]) + 0.2 * np.sin(2*np.pi*freqs[1]*t + phs[1])
        return project_feasible(h, ts)
    if typ == 3:
        h = rng.exponential(0.4, n) + 0.2
        return project_feasible(h, ts)
    if typ == 4:
        h = np.zeros(n)
        h[rng.choice(n, size=n//2, replace=False)] = 1.0
        return project_feasible(h, ts)
    if typ == 5:
        h = np.zeros(n)
        for s in range(2):
            start = int(rng.uniform(0, n - n//4))
            h[start:start+n//4] = 1.0 + s * 0.3
        return project_feasible(h, ts)
    return project_feasible(np.ones(n) * 0.5, ts)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()
    tl = t0 + max(budget_s - 5, 10)

    n = 80
    dx = 2.0 / n
    ts = n / 2.0

    if 'initial_h_values' in dir() and initial_h_values is not None:
        h_init = np.asarray(initial_h_values, dtype=float).copy()
        if len(h_init) != n:
            from scipy.interpolate import interp1d
            xo = np.linspace(0, 2, len(h_init), endpoint=False)
            xn = np.linspace(0, 2, n, endpoint=False)
            h_init = interp1d(xo, h_init, kind='linear', fill_value='extrapolate')(xn)
    else:
        h_init = np.ones(n) * 0.5
    h_init = project_feasible(h_init, ts)
    best_h, best_c5 = h_init.copy(), compute_c5(h_init, dx)

    # Phase 1: optimize from parent
    if time.time() < tl - 60:
        h1, c51 = opt_traj(h_init, dx, min(tl - 60, t0 + 300))
        if c51 < best_c5:
            best_h, best_c5 = h1, c51

    # Phase 2: diverse restarts
    init_types = [4, 1, 2, 5, 3, 1, 4, 2, 0, 5, 3, 4, 1, 2]
    for r in range(len(init_types)):
        if time.time() > tl - 30:
            break
        elapsed = time.time() - t0
        remaining = tl - time.time()
        phase_time = min(remaining * 0.7, 80)
        if phase_time < 10:
            break
        if r == 0:
            hr = project_feasible(best_h + 0.1 * rng.randn(n), ts)
        elif r == 1:
            hr = project_feasible(best_h + 0.3 * rng.randn(n), ts)
        else:
            hr = make_init(n, init_types[r % len(init_types)], rng)
        ho, c5o = opt_traj(hr, dx, time.time() + phase_time)
        if c5o < best_c5:
            best_h, best_c5 = ho, c5o

    # Phase 3: refine best with scipy
    if time.time() < tl - 15:
        for attempt in range(4):
            if time.time() > tl - 10:
                break
            x0 = project_feasible(best_h + 0.03 * rng.randn(n), ts)
            def sobj(x):
                o, _ = obj_and_grad(x, 0.005, dx)
                return o
            try:
                res = minimize(sobj, x0, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n,
                               options={'maxiter': 200, 'maxfun': 2000})
                hs = project_feasible(res.x, ts)
                c5s = compute_c5(hs, dx)
                if c5s < best_c5:
                    best_h, best_c5 = hs, c5s
            except Exception:
                pass

    # Final cleanup
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
