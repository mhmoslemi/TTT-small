def fix_h(h):
    x = np.asarray(h, dtype=float).copy()
    n = x.size
    target = 0.5 * n
    x = np.clip(x, 0.0, 1.0)
    # iterative proportional correction
    for _ in range(100):
        err = target - x.sum()
        if abs(err) < 1e-13:
            break
        if err > 0:
            mask = x < 1.0
            slack = (1.0 - x)[mask].sum()
        else:
            mask = x > 0.0
            slack = x[mask].sum()
        if slack <= 1e-15:
            break
        x[mask] = np.clip(x[mask] + err / slack, 0.0, 1.0)
    # small uniform adjustments over eligible
    for _ in range(1000):
        err = target - x.sum()
        if abs(err) < 1e-12:
            break
        if err > 0:
            mask = x < 1.0 - 1e-14
            if not mask.any(): break
            x[mask] = np.clip(x[mask] + err / mask.sum(), 0.0, 1.0)
        else:
            mask = x > 1e-14
            if not mask.any(): break
            x[mask] = np.clip(x[mask] + err / mask.sum(), 0.0, 1.0)
    # final last tweak
    for _ in range(20):
        err = target - x.sum()
        if abs(err) < 1e-12:
            break
        if err > 0:
            idx = int(np.argmin(x))
        else:
            idx = int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + err, 0.0, 1.0))
    return x

def smooth_c5(h, a):
    x = np.asarray(h, dtype=float)
    n = x.size
    dx = 2.0 / n
    q = 1.0 - x
    m = 2*n - 1
    T = np.empty(m)
    # neg lags index k-1 for negative -k
    for k in range(1,n):
        L = n-k
        T[k-1] = dx * np.dot(x[k:], q[:L])
    T[n-1] = dx * np.dot(x, q)
    for k in range(1,n):
        L = n-k
        T[n-1+k] = dx * np.dot(x[:L], q[k:])
    mn = float(np.max(T))
    e = np.exp(a * (T - mn))
    Z = float(np.sum(e))
    w = e / Z
    val = mn + np.log(Z) / a
    g = np.zeros(n)
    w0 = w[n-1]
    if w0 > 0.0:
        g += w0 * dx * (1.0 - 2.0*x)

import numpy as np
import time
import math
from scipy import optimize as opt
from scipy import sparse as sp
try:
    from scipy.optimize import milp, LinearConstraint, Bounds
except Exception:
    milp = LinearConstraint = Bounds = None

def true_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def fix_h(h):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    n = x.size
    target = 0.5 * n
    for _ in range(200):
        err = target - x.sum()
        if abs(err) < 1e-12:
            break

import numpy as np
import time
import math
from scipy import optimize as opt
from scipy import sparse as sp
try:
    from scipy.optimize import milp, LinearConstraint, Bounds
except Exception:
    milp = LinearConstraint = Bounds = None

def true_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def fix_h(h):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    n = x.size
    target = 0.5 * n
    for _ in range(200):
        err = target - x.sum()
        if abs(err) < 1e-12:
            break
        if err > 0.0:
            m = x < 1.0
        else:
            m = x > 0.0
        if not m.any():
            break
        x[m] = np.clip(x[m] + err / m.sum(), 0.0, 1.0)
    err = target - x.sum()
    if abs(err) > 1e-12:
        idx = int(np.argmin(x)) if err > 0.0 else int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + err, 0.0, 1.0))
    return x

def smooth_c5(h, a):
    x = np.asarray(h, dtype=float)
    n = x.size
    dx = 2.0 / n
    q = 1.0 - x
    T = np.empty(2 * n - 1)
    for k in range(1, n):
        L = n - k
        T[k - 1] = dx * np.dot(x[k:], q[:L])
    T[n - 1] = dx * np.dot(x, q)
    for k in range(1, n):
        L = n - k
        T[n - 1 + k] = dx * np.dot(x[:L], q[k:])
    mn = float(np.max(T))
    e = np.exp(a * (T - mn))
    Z = float(np.sum(e))
    w = e / Z
    val = mn + float(np.log(Z)) / a
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 0.0:
        g += w0 * dx * (1.0 - 2.0 * x)
    for k in range(1, n):
        L = n - k
        wn = w[k - 1]
        if wn > 0.0:
            g[k:] += wn * dx * (1.0 - x[:L])
            g[:L] -= wn * dx * x[k:]
        wp = w[n - 1 + k]
        if wp > 0.0:
            g[:L] += wp * dx * (1.0 - x[k:])
            g[k:] -= wp * dx * x[:L]
    return float(val), g

def smooth_fun(h, a):
    return smooth_c5(h, a)[0]

def smooth_jac(h, a):
    return smooth_c5(h, a)[1]

def sum_constraint(x):
    return float(x.sum() - 0.5 * x.size)

def sum_jac(x):
    return np.ones_like(x)

def polish_slsqp(h, deadline, maxiter=60):
    x = fix_h(h)
    best_x = x.copy()
    best_c = true_c5(x)
    n = x.size
    if n <= 1 or time.time() >= deadline:
        return best_x, best_c
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac}
    cur_x = best_x.copy()
    cur_c = best_c
    for a in (20.0, 80.0, 300.0):
        if time.time() > deadline - 5.0:
            break
        if n > 80:
            mit = min(25, int(maxiter))
        elif n > 40:
            mit = min(45, int(maxiter))
        else:
            mit = min(70, int(maxiter))
        if mit < 5:
            mit = 5
        try:
            res = opt.minimize(smooth_fun, cur_x, args=(a,), jac=smooth_jac,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': mit, 'ftol': 1e-9, 'disp': False})
        except Exception:
            res = None
        if res is not None and getattr(res, 'x', None) is not None:
            y = fix_h(res.x)
            c = true_c5(y)
            if c < cur_c - 1e-12:
                cur_x = y
                cur_c = c
                if c < best_c:
                    best_c = c
                    best_x = y
    return best_x, best_c

def binary_project(b, target):
    b = (b > 0.5).astype(float)
    target = int(target)
    while b.sum() < target:
        z = np.flatnonzero(b == 0.0)
        if z.size == 0:
            break
        b[z[len(z) // 2]] = 1.0
    while b.sum() > target:
        o = np.flatnonzero(b == 1.0)
        if o.size == 0:
            break
        b[o[len(o) // 2]] = 0.0
    return b

def binary_refine(b, deadline, seed):
    b = binary_project(b, b.size // 2)
    cur = true_c5(b)
    rng = np.random.default_rng(seed)
    evals = 0
    while time.time() < deadline - 2.0 and evals < 20000:
        ones = np.flatnonzero(b == 1.0)
        zeros = np.flatnonzero(b == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        found = False
        if ones.size * zeros.size <= 8000:
            perm_q = rng.permutation(zeros)
            for p in rng.permutation(ones):
                if found:
                    break
                for q in perm_q:
                    evals += 1
                    if evals >= 20000:
                        found = True
                        break
                    b[p] = 0.0
                    b[q] = 1.0
                    c = true_c5(b)
                    if c < cur - 1e-12:
                        cur = c
                        found = True
                        break
                    b[p] = 1.0
                    b[q] = 0.0
            if not found:
                break
        else:
            for _ in range(2000):
                p = int(rng.choice(ones))
                q = int(rng.choice(zeros))
                evals += 1
                b[p] = 0.0
                b[q] = 1.0
                c = true_c5(b)
                if c < cur - 1e-12:
                    cur = c
                    found = True
                    break
                b[p] = 1.0
                b[q] = 0.0
            if not found:
                break
    return b, cur

def pair_var_idx(n, i, j):
    if i > j:
        i, j = j, i
    return i * (n - 1) - (i * (i - 1) // 2) + (j - i - 1)

def _append_row(data, I, J, row, coeffs):
    for c, v in coeffs:
        data.append(float(v))
        I.append(int(row))
        J.append(int(c))

def milp_binary(n, time_limit, t_max=None):
    if milp is None or LinearConstraint is None or Bounds is None:
        return None
    if n < 8 or n % 2 != 0:
        return None
    target = n // 2
    nvar = n + 1 + n * (n - 1) // 2
    t_idx = n
    p0 = n + 1
    data = []
    I = []
    J = []
    lb = []
    ub = []
    row = 0
    for d in range(1, n):
        L = n - d
        for shift in (0, d):
            coeffs = [(t_idx, -1.0)]
            for i in range(L):
                coeffs.append((i + shift, 1.0))
                coeffs.append((p0 + pair_var_idx(n, i, i + d), -1.0))
            _append_row(data, I, J, row, coeffs)
            lb.append(-np.inf)
            ub.append(0.0)
            row += 1
    off = 0
    for i in range(n - 1):
        for j in range(i + 1, n):
            pv = p0 + off
            _append_row(data, I, J, row, [(pv, 1.0), (i, -1.0)])
            lb.append(-np.inf)
            ub.append(0.0)
            row += 1
            _append_row(data, I, J, row, [(pv, 1.0), (j, -1.0)])
            lb.append(-np.inf)
            ub.append(0.0)
            row += 1
            _append_row(data, I, J, row, [(i, 1.0), (j, 1.0), (pv, -1.0)])
            lb.append(-np.inf)
            ub.append(1.0)
            row += 1
            off += 1
    _append_row(data, I, J, row, [(i, 1.0) for i in range(n)])
    lb.append(float(target))
    ub.append(float(target))
    row += 1
    A = sp.csr_matrix((data, (I, J)), shape=(row, nvar))
    c = np.zeros(nvar)
    c[t_idx] = 1.0
    lbound = np.zeros(nvar)
    ubound = np.ones(nvar)
    if t_max is None:
        t_upper = float(n)
    else:
        t_upper = float(max(0, int(t_max)))
        if int(t_max) >= n:
            t_upper = float(n)
    if lbound.size:
        ubound[t_idx] = min(float(n), t_upper)
    lbound[0] = 1.0
    ubound[0] = 1.0
    integ = np.zeros(nvar)
    integ[:n] = 1.0
    cons = LinearConstraint(A, np.asarray(lb, dtype=float), np.asarray(ub, dtype=float))
    bnd = Bounds(lbound, ubound)
    try:
        res = milp(c, constraints=cons, integrality=integ, bounds=bnd,
                   options={'time_limit': float(max(1.0, time_limit))})
    except Exception:
        return None
    x = getattr(res, 'x', None)
    if x is None:
        return None
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x[:n])):
        return None
    b = np.where(x[:n] > 0.5, 1.0, 0.0)
    b = binary_project(b, target)
    if b[0] != 1.0:
        b[0] = 1.0
        if b.sum() > target:
            o = np.flatnonzero(b == 1.0)
            o = o[o != 0]
            if o.size:
                b[o[len(o) // 2]] = 0.0
    c5 = true_c5(b)
    status = int(getattr(res, 'status', -1))
    return (b, c5, status, str(getattr(res, 'message', '')))

import numpy as np
import time
import math
from scipy import optimize as opt
from scipy import sparse as sp

try:
    from scipy.optimize import milp, LinearConstraint, Bounds
except Exception:
    milp = LinearConstraint = Bounds = None


def true_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def fix_h(h):
    x = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = 0.5 * x.size
    for _ in range(200):
        err = target - x.sum()
        if abs(err) < 1e-12:
            break
        if err > 0.0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not mask.any():
            break
        x[mask] = np.clip(x[mask] + err / mask.sum(), 0.0, 1.0)
    err = target - x.sum()
    if abs(err) > 1e-12:
        idx = int(np.argmin(x)) if err > 0.0 else int(np.argmax(x))
        x[idx] = float(np.clip(x[idx] + err, 0.0, 1.0))
    return x


def smooth_c5(h, a):
    x = np.asarray(h, dtype=float)
    n = x.size
    dx = 2.0 / n
    q = 1.0 - x
    T = np.empty(2 * n - 1)
    for k in range(1, n):
        L = n - k
        T[k - 1] = dx * np.dot(x[k:], q[:L])
    T[n - 1] = dx * np.dot(x, q)
    for k in range(1, n):
        L = n - k
        T[n - 1 + k] = dx * np.dot(x[:L], q[k:])
    mn = float(np.max(T))
    e = np.exp(a * (T - mn))
    Z = float(np.sum(e))
    w = e / Z
    val = mn + float(np.log(Z)) / a
    g = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 0.0:
        g += w0 * dx * (1.0 - 2.0 * x)
    for k in range(1, n):
        L = n - k
        wn = w[k - 1]
        if wn > 0.0:
            g[k:] += wn * dx * (1.0 - x[:L])
            g[:L] -= wn * dx * x[k:]
        wp = w[n - 1 + k]
        if wp > 0.0:
            g[:L] += wp * dx * (1.0 - x[k:])
            g[k:] -= wp * dx * x[:L]
    return float(val), g


def smooth_fun(h, a):
    return smooth_c5(h, a)[0]


def smooth_jac(h, a):
    return smooth_c5(h, a)[1]


def sum_constraint(x):
    return float(x.sum() - 0.5 * x.size)


def sum_jac(x):
    return np.ones_like(x)


def polish_slsqp(h, deadline, maxiter=60):
    x = fix_h(h)
    best_x = x.copy()
    best_c = true_c5(x)
    n = x.size
    if n <= 1 or time.time() >= deadline:
        return best_x, best_c
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": sum_constraint, "jac": sum_jac}
    cur_x = best_x.copy()
    cur_c = best_c
    for a in (20.0, 80.0, 300.0):
        if time.time() > deadline - 5.0:
            break
        base = 25 if n > 80 else (45 if n > 40 else 70)
        mit = max(5, min(base, int(maxiter)))
        try:
            res = opt.minimize(
                smooth_fun, cur_x, args=(a,), jac=smooth_jac,
                method="SLSQP", bounds=bounds, constraints=cons,
                options={"maxiter": mit, "ftol": 1e-9, "disp": False}
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            y = fix_h(res.x)
            c = true_c5(y)
            if c < cur_c - 1e-12:
                cur_x = y
                cur_c = c
                if c < best_c:
                    best_x = y
                    best_c = c
    return best_x, best_c


def binary_project(b, target):
    b = (b > 0.5).astype(float)
    target = int(target)
    while b.sum() < target:
        z = np.flatnonzero(b == 0.0)
        if z.size == 0:
            break
        b[z[len(z) // 2]] = 1.0
    while b.sum() > target:
        o = np.flatnonzero(b == 1.0)
        if o.size == 0:
            break
        b[o[len(o) // 2]] = 0.0
    return b


def binary_refine(b, deadline, seed):
    b = binary_project(b, b.size // 2)
    cur = true_c5(b)
    rng = np.random.default_rng(seed)
    evals = 0
    while time.time() < deadline - 2.0 and evals < 20000:
        ones = np.flatnonzero(b == 1.0)
        zeros = np.flatnonzero(b == 0.0)
        if ones.size == 0 or zeros.size == 0:
            break
        found = False
        perm_q = rng.permutation(zeros)
        for p in rng.permutation(ones):
            if found:
                break
            for q in perm_q:
                evals += 1
                if evals >= 20000:
                    found = True
                    break
                b[p] = 0.0
                b[q] = 1.0
                c = true_c5(b)
                if c < cur - 1e-12:
                    cur = c
                    found = True
                    break
                b[p] = 1.0
                b[q] = 0.0
        if not found:
            break
    return b, cur


def pair_var_idx(n, i, j):
    if i > j:
        i, j = j, i
    return i * (n - 1) - (i * (i - 1) // 2) + (j - i - 1)


def mip_add_row(data, I, J, lb, ub, row, coeffs, lo, hi):
    for c, v in coeffs:
        data.append(float(v))
        I.append(int(row))
        J.append(int(c))
    lb.append(lo)
    ub.append(hi)
    return row + 1


def milp_binary(n, time_limit, t_max=None):
    if milp is None or LinearConstraint is None or Bounds is None:
        return None
    if n < 8 or n % 2 != 0:
        return None
    target = n // 2
    nvar = n + 1 + n * (n - 1) // 2
    t_idx = n
    p0 = n + 1
    data = []
    I = []
    J = []
    lb = []
    ub = []
    row = 0
    inf = float("inf")

    for d in range(1, n):
        L = n - d
        for shift in (0, d):
            coeffs = [(t_idx, -1.0)]
            for i in range(L):
                coeffs.append((i + shift, 1.0))
                coeffs.append((p0 + pair_var_idx(n, i, i + d), -1.0))
            row = mip_add_row(data, I, J, lb, ub, row, coeffs, -inf, 0.0)

    off = 0
    for i in range(n - 1):
        for j in range(i + 1, n):
            pv = p0 + off
            row = mip_add_row(data, I, J, lb, ub, row, [(pv, 1.0), (i, -1.0)], -inf, 0.0)
            row = mip_add_row(data, I, J, lb, ub, row, [(pv, 1.0), (j, -1.0)], -inf, 0.0)
            row = mip_add_row(data, I, J, lb, ub, row, [(i, 1.0), (j, 1.0), (pv, -1.0)], -inf, 1.0)
            off += 1

    row = mip_add_row(data, I, J, lb, ub, row, [(i, 1.0) for i in range(n)], float(target), float(target))

    A = sp.csr_matrix((data, (I, J)), shape=(row, nvar))
    c = np.zeros(nvar)
    c[t_idx] = 1.0

    lbound = np.zeros(nvar)
    ubound = np.ones(nvar)
    t_upper = float(n) if t_max is None else float(max(0, int(t_max)))
    ubound[t_idx] = min(float(n), t_upper)
    lbound[0] = 1.0
    ubound[0] = 1.0

    integ = np.zeros(nvar)
    integ[:n] = 1.0

    try:
        res = milp(
            c,
            constraints=LinearConstraint(A, np.asarray(lb, dtype=float), np.asarray(ub, dtype=float)),
            integrality=integ,
            bounds=Bounds(lbound, ubound),
            options={"time_limit": float(max(1.0, time_limit))},
        )
    except Exception:
        return None

    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=float)
    if not np.all(np.isfinite(x[:n])):
        return None

    b = np.where(x[:n] > 0.5, 1.0, 0.0)
    b = binary_project(b, target)
    if b[0] != 1.0:
        b[0] = 1.0
        if b.sum() > target:
            o = np.flatnonzero(b == 1.0)
            o = o[o != 0]
            if o.size:
                b[o[len(o) // 2]] = 0.0

    return b, true_c5(b), int(getattr(res, "status", -1)), str(getattr(res, "message", ""))


def run(seed=42, budget_s=1100, **kwargs):

        best_h = fix_h(np.full(70, 0.5, dtype=float))
        best_c = true_c5(best_h)

    if int(best_h.size) != best_n:
