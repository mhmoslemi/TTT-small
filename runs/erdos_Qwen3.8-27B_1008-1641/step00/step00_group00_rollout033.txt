∂C_smooth/∂h_i = sum_k p_k * ∂COR_k/∂h_i * dx
where p_k = exp(COR_k * dx / T) / sum_j exp(COR_j * dx / T)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def upsample(h, n_new):
    n = len(h)
    idx = np.minimum((np.arange(n_new) * n) // n_new, n - 1)
    return project_feasible(h[idx])

def coord_descent(h, max_rounds=30, time_limit=30.0):
    n = len(h)
    t0 = time.time()
    h = project_feasible(h.astype(float).copy())
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            for val in [0.0, 1.0]:
                if abs(h[i] - val) < 1e-8:
                    continue
                save = h.copy()
                h[i] = val
                h = project_feasible(h)
                c = compute_c5(h)
                if c < cur - 1e-12:
                    cur = c
                    improved = True
                else:
                    h = save
        if not improved:
            break
    return h, cur

def sim_anneal(h, dur, seed, t_hot=0.025, t_cold=0.0005):
    n = len(h)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    h = project_feasible(h.astype(float).copy())
    cur = compute_c5(h)
    best = h.copy()
    best_c5 = cur
    while time.time() - t0 < dur:
        f = min(1.0, (time.time() - t0) / dur)
        T = t_hot * (t_cold / t_hot) ** f
        save = h.copy()
        i = rng.integers(n)
        h[i] = rng.uniform(0.0, 1.0)
        h = project_feasible(h)
        nc = compute_c5(h)
        d = nc - cur
        if d < 0:
            cur = nc
            if cur < best_c5:
                best_c5 = cur
                best = h.copy()
        elif T > 1e-14:
            log_p = -min(d / max(T, 1e-14), 50.0)
            if rng.random() < np.exp(log_p):
                cur = nc
                if cur < best_c5:
                    best_c5 = cur
                    best = h.copy()
            else:
                h = save
        else:
            h = save
    return best, best_c5

def random_init(n, rng):
    h = rng.binomial(1, 0.5, size=n).astype(float)
    target = n / 2.0
    diff = target - np.sum(h)
    if diff > 0.01:
        idx0 = np.where(h == 0)[0]
        rng.shuffle(idx0)
        k = int(np.ceil(diff))
        h[idx0[:k]] = 1.0
    elif diff < -0.01:
        idx1 = np.where(h == 1)[0]
        rng.shuffle(idx1)
        k = int(np.ceil(-diff))
        h[idx1[:k]] = 0.0
    rem = target - np.sum(h)
    if abs(rem) > 1e-10:
        if rem > 0:
            idx0 = np.where(h == 0)[0]
            if len(idx0) > 0:
                h[idx0[0]] = min(rem, 1.0)
        else:
            idx1 = np.where(h == 1)[0]
            if len(idx1) > 0:
                h[idx1[0]] = max(1.0 + rem, 0.0)
    return project_feasible(h)
