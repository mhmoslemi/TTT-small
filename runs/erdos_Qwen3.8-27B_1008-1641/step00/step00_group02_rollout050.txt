import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(c, beta):
    c_max = np.max(c)
    return float(c_max + np.log(np.exp((c - c_max) * beta).sum()) / beta)

def fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)

def make_binary(n, rng):
    h = np.zeros(n)
    half = n // 2
    perm = rng.permutation(n)
    h[perm[:half]] = 1.0
    return h

def local_search_binary(h, budget=30.0, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < budget:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c > best_c5 + 1e-15:
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = c
    return h, best_c5

def coord_descent(h, budget=20.0, target=None):
    if target is None:
        target = len(h) / 2.0
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    improved = True
    while improved and time.time() - t0 < budget:
        improved = False
        for i in range(n):
            old = h[i]
            for trial in [0.0, 1.0, 0.5, np.clip(old + 0.1, 0, 1), np.clip(old - 0.1, 0, 1)]:
                if trial == old:
                    continue
                h[i] = trial
                h = fix_sum(h, target)
                c = compute_c5(h)
                if c < best_c5:
                    best_c5 = c
                    improved = True
                else:
                    h[i] = old
                    h = fix_sum(h, target)
    return h, best_c5

def slsqp_optimize(h0, n, budget=60.0):
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(c, 40.0) * dx

    def jac(x):
        x = np.clip(x, 0.0, 1.0)
        c = np.correlate(x, 1.0 - x, mode='full')
        c_max = np.max(c)
        w = np.exp((c - c_max) * 40.0)
        w /= w.sum()
        # gradient of correlation wrt x
        g = np.zeros(n)
        idx = np.arange(len(c))
        for k in range(len(c)):
            wk = w[k] * 40.0 * dx
            lag = k - (n - 1)
            if lag >= 0:
                # c[k] = sum_{i=0}^{n-1-lag} x[i+lag]*(1-x[i])
                for i in range(n - lag):
                    g[i + lag] += wk * (1.0 - x[i])
                    g[i] -= wk * x[i + lag]
            else:
                # c[k] = sum_{i=-lag}^{n-1} x[i-lag+lag-(-lag)]*(1-x[...])
                # For full correlation: c[k] = sum_i a[i]*b[i-(k-(n-1))]
                # Simpler: just use the definition
                # correlate(a,v,'full')[k] = sum_i a[i+k-n+1]*v[i] for valid i
                sh = k - (n - 1)  # negative
                for i in range(n + sh, n):
                    # a[i] * v[i - (n-1-k)] = a[i] * v[i - (n-1-k)]
                    j = i - (n - 1 - k)
                    if 0 <= j < n:
                        g[i] += wk * (1.0 - x[j])
                        g[j] -= wk * x[i]
        return g

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    t0 = time.time()
    max_iter = max(50, int((budget - (time.time() - t0)) * 50))
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=jac,
                       options={'maxiter': min(max_iter, 500), 'ftol': 1e-12})
        h_opt = fix_sum(res.x, target)
        return h_opt, compute_c5(h_opt)
    except Exception:
        return h0, compute_c5(h0)

    start_t = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')

    def remaining():
        return max(0.0, budget_s * 0.88 - (time.time() - start_t))

    # Phase 1: SLSQP from multiple starts for each n
    for n in ns_to_try:
        if remaining() < 10:

        dx = 2.0 / n
        target = n / 2.0
        time_per_n = min(120.0, remaining() / max(1, len(ns_to_try)))
        n_restarts = max(1, int(time_per_n / 20))
        per_restart = time_per_n / n_restarts

        for r in range(n_restarts):
            if remaining() < 5:

            if r == 0 and h_init is not None and n == len(h_init):
                h0 = h_init.copy()
            elif r == 1 and n % 2 == 0:

            elif r == 2 and h_init is not None:

            else:
