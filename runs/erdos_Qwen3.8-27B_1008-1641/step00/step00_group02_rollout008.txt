import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def obj_func(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = alpha * (corr - m)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (m + log_sum / alpha) * dx


def constrain_sum(h, target_sum):
    return np.sum(h) - target_sum


def project_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    h = h + diff / len(h)
    return np.clip(h, 0.0, 1.0)
