from scipy.optimize import minimize, differential_evolution

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-15:
            break
        active = (h < 1.0 - 1e-14) if diff > 0 else (h > 1e-14)
        if not np.any(active):
            break
        h[active] += diff / np.sum(active)
        h = np.clip(h, 0.0, 1.0)
    return h


def make_smooth_obj(dx, T):

        return softmax_max(corr, T)
    return obj
