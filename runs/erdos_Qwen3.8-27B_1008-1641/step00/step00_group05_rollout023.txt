    best_h = h.copy()

        zi = np.where(h == 0)[0]
        pi = oi[rng.integers(len(oi))]
        pj = zi[rng.integers(len(zi))]

        h[pj] = 1.0
        new = compute_c5(h, n)

            cur = new
            if new < best:
                best = new
                best_h = h.copy()

            h[pi] = 1.0
            h[pj] = 0.0

    return h

def sa_loop(h, n, rng, t_end, temp_start=0.03, temp_floor=0.001):
    temp = temp_start
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    cur_c5 = best_c5
    while time.time() < t_end:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            break
        pi = oi[rng.integers(len(oi))]
        pj = zi[rng.integers(len(zi))]
        h[pi] = 0.0
        h[pj] = 1.0
        new_c5 = compute_c5(h, n)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        d = new_c5 - cur_c5
        if d <= 0.0 or (temp > temp_floor and rng.random() < np.exp(-d / temp)):
            cur_c5 = new_c5
        else:
            h[pi] = 1.0
            h[pj] = 0.0
        temp = max(temp * 0.999995, temp_floor)
    return best_h, best_c5

def coord_descent(h, n, t_end):
    for _ in range(300):
        if time.time() > t_end:
            break
        improved = False
        oi = list(np.where(h == 1)[0])
        zi = list(np.where(h == 0)[0])
        for pos in oi:
            if time.time() > t_end:
                return h, compute_c5(h, n)
            if h[pos] != 1.0:
                continue
            cur_val = compute_c5(h, n)
            best_j = -1
            best_v = cur_val
            for j in zi:
                if h[j] != 0.0:
                    continue
                h[pos] = 0.0
                h[j] = 1.0
                v = compute_c5(h, n)
                h[pos] = 1.0
                h[j] = 0.0
                if v < best_v:
                    best_v = v
                    best_j = j
            if best_j >= 0:
                h[pos] = 0.0
                h[best_j] = 1.0
                improved = True
        if not improved:
            break
    return h, compute_c5(h, n)

def nonbinary_refine(h, n, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    oi = list(np.where(h == 1)[0])
    zi = list(np.where(h == 0)[0])
    for i in oi:
        if time.time() > t_end:
            break
        for j in zi:
            if time.time() > t_end:
                break
            for eps in [0.05, 0.1, 0.15, 0.2, 0.3]:
                h[i] = 1.0 - eps
                h[j] = eps
                v = compute_c5(h, n)
                if v < best_c5:
                    best_c5 = v
                    best_h = h.copy()
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    t_limit = t0 + budget_s - 15

    n = len(initial_h_values)
    h = make_binary(initial_h_values, n, rng)

    best_h = h.copy()
    best_c5 = compute_c5(h, n)

    # Phase 1: Main SA (50% of time)
    p1_end = t0 + (t_limit - t0) * 0.50
    best_h, best_c5 = sa_loop(h, n, rng, p1_end, temp_start=0.03)

    # Phase 2: Coordinate descent (15%)
    p2_end = t0 + (t_limit - t0) * 0.65
    h_cd = best_h.copy()
    h_cd, c5_cd = coord_descent(h_cd, n, p2_end)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: Second SA pass with lower temp (15%)
    p3_end = t0 + (t_limit - t0) * 0.80
    h3 = best_h.copy()
    h3, c5_3 = sa_loop(h3, n, rng, p3_end, temp_start=0.015, temp_floor=0.0005)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    # Coord descent again
    h_cd2 = best_h.copy()
    h_cd2, c5_cd2 = coord_descent(h_cd2, n, p3_end + 5)
    if c5_cd2 < best_c5:
        best_c5 = c5_cd2
        best_h = h_cd2.copy()

    # Phase 4: Random restarts (15%)
    p4_end = t_limit
    while time.time() < p4_end - 5:
        h4 = best_h.copy()
        n_flips = int(rng.integers(4, 14))
        for _ in range(n_flips):
            oi = np.where(h4 == 1)[0]
            zi = np.where(h4 == 0)[0]
            h4[oi[rng.integers(len(oi))]] = 0.0
            h4[zi[rng.integers(len(zi))]] = 1.0
        t_sub = min(time.time() + 8.0, p4_end - 3)
        h4, c5_4 = sa_loop(h4, n, rng, t_sub, temp_start=0.012, temp_floor=0.0005)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Non-binary refinement (last 5%)
    p5_end = t_limit
    best_h, best_c5 = nonbinary_refine(best_h, n, p5_end)

    # Final clip and validate
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h, n)
    return best_h, final_c5, n
