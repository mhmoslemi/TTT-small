def _project_fix(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = 1.0 - h[mask]
            tr = np.sum(room)
            if tr < 1e-13:
                break
            step = min(1.0, diff / tr)
            h[mask] += step * room
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            room = h[mask]
            tr = np.sum(room)
            if tr < 1e-13:
                break
            step = min(1.0, -diff / tr)
            h[mask] -= step * room
    return np.clip(h, 0.0, 1.0)
