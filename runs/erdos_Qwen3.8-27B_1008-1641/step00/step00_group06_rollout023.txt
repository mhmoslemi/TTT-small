import numpy as np
from scipy.optimize import minimize
from functools import partial
import time

_g_target = 0.0

def _sum_eq(h):
    return np.sum(h) - _g_target

def _sum_eq_jac(h):
    return np.ones(len(h))

def _correlate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr)), corr

def _lse(x, alpha):
    m = np.max(x)
    a = alpha * (x - m)
    if np.max(a) > 700:
        return float(m)
    return float(m + np.log(np.sum(np.exp(a))) / alpha)

def _smooth_obj(h, dx, alpha):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return _lse(corr, alpha)

def _enforce(h, target):
    h = np.asarray(h, dtype=float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        adj = (target - np.sum(h)) / len(h)
        if abs(adj) < 1e-14:
            break
        h = h + adj
    h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_step(h0, n, dx, target, alpha, max_iter):
    global _g_target
    _g_target = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac}]
    obj_fn = partial(_smooth_obj, dx=dx, alpha=alpha)
    try:
        res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
        return res.x if res.x is not None else h0.copy()
    except Exception:
        return h0.copy()

def _coord_descend(h, n, dx, target, steps=30):
    best = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
    h = h.copy()
    for s in range(steps):
        improved = False
        for i in range(n):
            for delta in [0.01, 0.05, 0.1, 0.2, -0.01, -0.05, -0.1, -0.2]:
                h_new = h.copy()
                h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
                h_new = _enforce(h_new, target)
                c5, _ = _correlate_c5(h_new, dx)
                if c5 < best - 1e-10:
                    best = c5
                    h = h_new
                    improved = True
                    break
        if not improved:
            break
    return h, best

def _random_feasible(n, target, rng):
    h = np.zeros(n)
    if target == int(target):
        idx = rng.choice(n, size=int(target), replace=False)
        h[idx] = 1.0
    else:

        frac = target - full
        idx = rng.choice(n, size=full, replace=False)
        h[idx] = 1.0
        h_full = np.ones(n)
        h_full[idx] = 0.0
        frac_idx = np.where(h_full > 0)[0]
        h[frac_idx[rng.randint(len(frac_idx))]] = frac
    return h
