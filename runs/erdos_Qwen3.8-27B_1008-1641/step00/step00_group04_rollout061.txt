    n = len(x0)
    rng = np.random.default_rng(seed)
    popsize = max(4, int(4 + 3 * np.log(n)))
    best_x = x0.copy()
    best_f = objective(x0)
    mean = x0.copy()
    cov = np.eye(n) * sigma**2
