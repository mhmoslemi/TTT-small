def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = 0.5 * n
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    if s > 1e-12 and s < n - 1e-12:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    s = h.sum()

    if abs(r) > 1e-10:

                else np.where(h > 1e-10)[0])
        for i in idxs:

            h[i] += np.sign(r) * step
            r -= np.sign(r) * step
