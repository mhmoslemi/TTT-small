def _binary_flip(h, n, max_iters=50000, T=1.0):
    h = _project(h.copy(), n)
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    extra = n / 2.0 - target
    ones = h_bin.sum()
    if ones > target:
        idx1 = np.where(h_bin == 1)[0]
        for i in range(int(ones - target)):
            h_bin[idx1[i]] = 0.0
    elif ones < target:
        idx0 = np.where(h_bin == 0)[0]
        for i in range(int(target - ones)):
            h_bin[idx0[i]] = 1.0
    if extra > 0 and extra < 1:
        idx0 = np.where(h_bin == 0)[0]
        if len(idx0) > 0:
            h_bin[idx0[0]] = extra
    best = _c5_val(h_bin, n)
    rng = np.random.default_rng(1)
    for it in range(max_iters):
        i = rng.integers(0, n)
        trial = h_bin.copy()
        trial[i] = 1.0 - trial[i]
        t_sum = trial.sum()
        target_sum = n / 2.0
        if abs(t_sum - target_sum) > 1e-10:
            diff = target_sum - t_sum
            if abs(diff) < 1.0:
                mask = (trial > 0) & (trial < 1)
                if mask.any():
                    trial[mask[0]] += diff
                else:
                    if diff > 0:
                        m = trial < 1
                    else:
                        m = trial > 0
                    if m.any():
                        trial[m] += diff / m.sum()
            else:
                continue
        trial = np.clip(trial, 0, 1)
        c = _c5_val(trial, n)
        if c < best or rng.random() < np.exp(-max(0, c - best) / T):
            h_bin = trial
            best = c
        T *= 0.9999
    return h_bin, best

def _nelder_mead(h, n, maxiter=8000):
    global _global_n
    _global_n = n
    res = minimize(_obj_fn, h.copy(), method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-8, 'fatol': 1e-10,
                            'adaptive': True})
    h_opt = _project(res.x, n)
    return h_opt, _c5_val(h_opt, n)

def _upsample(h, n_from, n_to):
    x_old = np.linspace(0, 2, n_from, endpoint=False)
    x_new = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return _project(h_new, n_to)

def run(seed=42, budget_s=1100, **kwargs):
    global _global_n
    t0 = time.time()
    def elapsed():
        return time.time() - t0
    def remaining():
        return budget_s - elapsed()

    best_h = None
    best_c5 = float('inf')
    best_n = 45

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)
    n0 = len(h0)
    h0 = _project(h0, n0)
    best_h = h0.copy()
    best_c5 = _c5_val(h0, n0)

    rng = np.random.default_rng(seed)

    # Phase 1: Coordinate descent on initial resolution
    if remaining() > 20:
        h_cd, c_cd = _coord_descent(best_h, n0, max_rounds=3000, T=0.05)
        if c_cd < best_c5:
            best_c5, best_h, best_n = c_cd, h_cd, n0

    # Phase 2: Nelder-Mead on initial resolution
    _global_n = n0
    if remaining() > 30:
        h_nm, c_nm = _nelder_mead(best_h, n0, maxiter=6000)
        if c_nm < best_c5:
            best_c5, best_h, best_n = c_nm, h_nm, n0

    # Phase 3: Binary flip search on n0
    if remaining() > 15:
        h_bf, c_bf = _binary_flip(best_h, n0, max_iters=80000, T=0.05)
        h_bf = _project(h_bf, n0)
        c_bf = _c5_val(h_bf, n0)
        if c_bf < best_c5:
            best_c5, best_h, best_n = c_bf, h_bf, n0
        # Also try random binary start
        for _ in range(3):
            if remaining() < 10:
                break
            h_rand = (rng.random(n0) > 0.5).astype(float)
            h_rand = _project(h_rand, n0)
            h_bf2, c_bf2 = _binary_flip(h_rand, n0, max_iters=40000, T=0.04)
            h_bf2 = _project(h_bf2, n0)
            c_bf2 = _c5_val(h_bf2, n0)
            if c_bf2 < best_c5:
                best_c5, best_h, best_n = c_bf2, h_bf2, n0

    # Phase 4: Multi-resolution Nelder-Mead
    _global_n = best_n
    resolutions = [max(n0, 90), 180, 360]
    for n_target in resolutions:
        if remaining() < 40:
            break
        h_up = _upsample(best_h, len(best_h), n_target)
        # Coordinate descent warm-up
        h_cd2, c_cd2 = _coord_descent(h_up, n_target, max_rounds=1500, T=0.03)
        if c_cd2 < best_c5:
            best_c5, best_h, best_n = c_cd2, h_cd2, n_target
        # Nelder-Mead
        _global_n = n_target
        h_nm2, c_nm2 = _nelder_mead(h_cd2, n_target, maxiter=5000)
        if c_nm2 < best_c5:
            best_c5, best_h, best_n = c_nm2, h_nm2, n_target
        # Binary on this resolution
        if remaining() > 20:
            h_bf3, c_bf3 = _binary_flip(h_cd2, n_target, max_iters=50000, T=0.04)
            h_bf3 = _project(h_bf3, n_target)
            c_bf3 = _c5_val(h_bf3, n_target)
            if c_bf3 < best_c5:
                best_c5, best_h, best_n = c_bf3, h_bf3, n_target

    # Phase 5: Multiple random restarts with Nelder-Mead at best_n
    if remaining() > 30:
        n_ws = best_n
        _global_n = n_ws
        for _ in range(6):
            if remaining() < 25:
                break
            h_r = best_h + rng.normal(0, 0.08, n_ws)
            h_r = _project(h_r, n_ws)
            h_r_cd, _ = _coord_descent(h_r, n_ws, max_rounds=500, T=0.04)
            h_nm3, c_nm3 = _nelder_mead(h_r_cd, n_ws, maxiter=5000)
            if c_nm3 < best_c5:
                best_c5, best_h, best_n = c_nm3, h_nm3, n_ws

    # Phase 6: Final polish
    if remaining() > 20:
        _global_n = best_n
        h_fin, c_fin = _nelder_mead(best_h, best_n, maxiter=10000)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin
        h_fin = _project(h_fin, best_n)
        best_c5 = _c5_val(h_fin, best_n)
        best_h = h_fin

    # Ensure exact sum constraint
    best_h = _project(best_h, best_n)
    best_c5 = _c5_val(best_h, best_n)

    return (best_h, best_c5, best_n)
