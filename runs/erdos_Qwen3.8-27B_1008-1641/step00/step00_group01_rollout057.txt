import numpy as np
from scipy.optimize import minimize, differential_evolution
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_simplex(h, target_sum):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target_sum - s
        if abs(d) < 1e-15:
            break
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_max(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_smooth_max(h, dx, m):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    mx = float(np.max(corr))
    shifted = np.clip(m * (corr - mx), -700.0, 700.0)
    return float(mx + np.log(np.sum(np.exp(shifted))) / m)

def sum_eq(h, target):
    return np.sum(h) - target

def proj_obj(h, target, dx):
    hp = project_simplex(h, target)
    return obj_max(hp, dx)

def slsqp_optimize(h, target, dx, m, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    constr = [{'type': 'eq', 'fun': partial(sum_eq, target=target)}]
    res = minimize(partial(obj_smooth_max, dx=dx, m=m), h,
                   method='SLSQP', bounds=bounds, constraints=constr,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_simplex(res.x, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    best_h = None
    best_c5 = float('inf')

    h_init = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)

    n_list = []
    if h_init is not None:
        n_list.append(len(h_init))
    for nn in [47, 49, 45, 48, 50, 51]:
        if nn not in n_list:
            n_list.append(nn)

    for ni, n in enumerate(n_list):
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        dx = 2.0 / n
        target = n / 2.0

        if h_init is not None and n == len(h_init):
            h0 = project_simplex(h_init, target)
        else:
            h0 = np.full(n, 0.5) + rng.uniform(-0.3, 0.3, n)
            h0 = project_simplex(h0, target)

        # Stage 1: SLSQP with smooth max m=100
        rem = budget_s - (time.time() - t0)
        if rem < 10:
            break
        try:
            h_opt = slsqp_optimize(h0, target, dx, 100.0, 3000, 1e-16)
            c5v = compute_c5(h_opt)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt.copy()
        except Exception:
            pass

        # Stage 2: Nelder-Mead on projected true max
        rem = budget_s - (time.time() - t0)
        if rem < 15:
            break
        try:
            res = minimize(partial(proj_obj, target=target, dx=dx), h0,
                           method='Nelder-Mead',
                           options={'maxiter': 10000, 'xatol': 1e-13,
                                    'fatol': 1e-15, 'adaptive': True})
            h_opt = project_simplex(res.x, target)
            c5v = compute_c5(h_opt)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt.copy()
        except Exception:
            pass

        # Stage 3: DE global search (primary n only)
        if ni == 0 and (budget_s - (time.time() - t0)) > 200:

            if diff > 0:

            else:
                idx = int(np.argmin(best_h))
            best_h[idx] += diff
            best_h = np.clip(best_h, 0.0, 1.0)

    if best_h is None:
        best_h = project_simplex(np.full(47, 0.5), 23.5)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
