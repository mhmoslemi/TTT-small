    max_c = np.max(corr)
    weights = np.exp(beta * (corr - max_c))

        if weights[m] < 1e-8:
            continue
        w = weights[m]
        k = m - (n - 1)
        # Contribution from (1-h[j-k]) where 0 <= j-k < n
        # j ranges from max(0,k) to min(n-1, n-1+k)
        j_lo = max(0, k)
        j_hi = min(n, n + k)  # exclusive
        # j-k ranges from j_lo-k to j_hi-1-k
        if j_hi > j_lo:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - k : j_hi - 1 - k])
        # Contribution from -h[j+k] where 0 <= j+k < n
        # j ranges from max(0,-k) to min(n-1, n-1-k)
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_hi2 > j_lo2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + k : j_hi2 - 1 + k]
    return grad
