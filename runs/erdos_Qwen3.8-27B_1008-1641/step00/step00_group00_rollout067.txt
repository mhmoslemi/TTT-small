def _project(v, ts):
    """Project v onto {h: 0<=h<=1, sum(h)=ts} via bisection on threshold."""
    lo, hi = -10.0, 10.0
    for _ in range(70):
        tau = 0.5 * (lo + hi)
        s = np.clip(v - tau, 0.0, 1.0).sum()
        if s > ts:
            lo = tau
        else:
            hi = tau
