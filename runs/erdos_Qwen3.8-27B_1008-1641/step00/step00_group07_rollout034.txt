    o = 1.0 - x
    n_lags = 2*n - 1
    corr = np.zeros(n_lags)
    # For efficiency, compute all correlations using FFT
    # Or just use np.correlate
    corr = np.correlate(x, o, mode='full')
    
    # Compute gradient matrix: G[j, k] = ∂c_k/∂x_j
    G = np.zeros((n, n_lags))
    for k in range(n_lags):
        ko = k - (n - 1)
        # ∂c_k/∂x_j = (1-x[j-ko]) * [0<=j-ko<n] - x[j+ko] * [0<=j+ko<n]

        G[ma, k] += o[a[ma]]
        b = idx + ko
        mb = (b >= 0) & (b < n)
        G[mb, k] -= x[b[mb]]
