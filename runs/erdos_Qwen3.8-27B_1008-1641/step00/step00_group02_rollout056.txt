from scipy.optimize import minimize

def c5_eval(h, dx):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def soft_c5(h, dx, T):
    h = np.clip(h, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def project_sum(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def make_initial(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    target = n / 2.0
    h[:int(target)] = 1.0
    if target != int(target):
        h[int(target)] = target - int(target)
    rng.shuffle(h)

def gradient_step(h, dx, T, lr, n):
