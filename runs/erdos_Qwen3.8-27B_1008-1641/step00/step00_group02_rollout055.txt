```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def smooth_obj(x, params):
    n, T = params
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = n / 2.0 - np.sum(x)
    if h[n - 1] < -0.01 or h[n - 1] > 1.01:
        return 1e8
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    lse = m + np.log(np.sum(np.exp(scaled - m)))
    return (dx / T) * lse


def exact_obj(x, params):
    n = params
    h = np.empty(n)
    h[:n - 1] = x
    h[n - 1] = n / 2.0 - np.sum(x)
    if h[n - 1] < -0.01 or h[n - 1] > 1.01:
        return 1e8
    return compute_c5(h)


def try_lbfgb(x0, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(smooth_obj, x0, args=(n, T),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = np.append(res.x, n / 2.0 - np.sum(res.x))
        if h_opt[n - 1] >= 0 and h_opt[n - 1] <= 1:
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return None, float('inf')


def try_exact(x0, n, maxiter, ftol):
    bounds = [(0.0, 1.0)] * (n - 1)
    try:
        res = minimize(exact_obj, x0, args=(n,),
                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_opt = np.append(res.x, n / 2.0 - np.sum(res.x))
        if h_opt[n - 1] >= 0 and h_opt[n - 1] <= 1:
            return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return None, float('inf')


def slsqp_obj(x, params):
    n = params
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.max(corr) * dx


def slsqp_con(x, params):
    n = params
    return np.sum(x) - n / 2.0


def slsqp_con_jac(x, params):
    n = params
    return np.ones(n)


def try_slsqp(x0, n, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac, 'args': (n,)}]
    try:
        res = minimize(slsqp_obj, x0, args=(n,),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.fun < 1e6:
            h_opt = res.x
            diff = n / 2.0 - np.sum(h_opt)
            if abs(diff) < 0.01:
                h_opt += diff / n
                np.clip(h_opt, 0.0, 1.0, out=h_opt)
                return h_opt, compute_c5(h_opt)
    except Exception:
        pass
    return None, float('inf')


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 45
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n_free = n - 1

    # Phase 1: Annealed smooth optimization from initial point
    temps = [5.0, 15.0, 50.0, 150.0, 400.0, 1000.0]
    for T in temps:
        if time.time() - start_time > budget_s * 0.15:
            break
        x0 = best_h[:n_free].copy()
        h_opt, c5v = try_lbfgb(x0, n, T, 600, 1e-15)
        if h_opt is not None and c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    # Phase 2: SLSQP polish on full vector
    if time.time() - start_time < budget_s * 0.2:
        h_opt, c5v = try_slsqp(best_h.copy(), n, 500)
        if h_opt is not None and c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    # Phase 3: Many random local searches with varying strategy
    while time.time() - start_time < budget_s * 0.78:
        r = rng.random()
        if r < 0.25:
            scale = 0.005 + 0.03 * rng.random()
            x0 = best_h[:n_free] + rng.normal(0, scale, n_free)
        elif r < 0.5:
            scale = 0.03 + 0.08 * rng.random()
            x0 = best_h[:n_free] + rng.normal(0, scale, n_free)
        elif r < 0.7:
            scale = 0.08 + 0.15 * rng.random()
            x0 = best_h[:n_free] + rng.normal(0, scale, n_free)
        elif r < 0.85:
            # Structured: sinusoidal perturbation
            f = rng.uniform(0.5, 5.0)
            amp = 0.05 + 0.1 * rng.random()
            phase = rng.uniform(0, 2 * np.pi)
            x0 = best_h[:n_free] + amp * np.sin(2 * np.pi * f * np.arange(n_free) / n_free + phase)
        else:
            # Fresh random with binary-like structure
            x0 = 0.5 + 0.4 * rng.choice([-1, 1], size=n_free)
            x0 += rng.normal(0, 0.1, n_free)

        x0 = np.clip(x0, 0.0, 1.0)
        last = n / 2.0 - np.sum(x0)
        if last < -0.05 or last > 1.05:
            # Share correction among all coords
            adj = np.clip(last, 0.0, 1.0) - last
            x0 -= adj / n_free
            x0 = np.clip(x0, 0.0, 1.0)
            last2 = n / 2.0 - np.sum(x0)
            if last2 < -0.02 or last2 > 1.02:
                continue

        T = 30.0 + 600.0 * rng.random() ** 0.7
        h_opt, c5v = try_lbfgb(x0, n, T, 300, 1e-14)
        if h_opt is not None and c5v < best_c5 - 1e-15:
            best_h, best_c5 = h_opt, c5v
            # Immediate SLSQP polish after improvement
            if time.time() - start_time < budget_s * 0.75:
                h2, c5b = try_slsqp(h_opt.copy(), n, 200)
                if h2 is not None and c5b < best_c5:
                    best_h, best_c5 = h2, c5b

    # Phase 4: Final exact polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        for i in range(4):
            if time.time() - start_time > budget_s - 10:
                break
            x0 = best_h[:n_free].copy()
            if i > 0:
                x0 += rng.normal(0, 0.003, n_free)
                x0 = np.clip(x0, 0, 1)
            h_opt, c5v = try_exact(x0, n, 3000, 1e-16)
            if h_opt is not None and c5v < best_c5:
                best_h, best_c5 = h_opt, c5v
        # Final SLSQP
        if time.time() - start_time < budget_s - 10:
            h_opt, c5v = try_slsqp(best_h.copy(), n, 1000)
            if h_opt is not None and c5v < best_c5:
                best_h, best_c5 = h_opt, c5v

    # Enforce integral constraint precisely
    diff = n / 2.0 - np.sum(best_h)
    if abs(diff) > 1e-14:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            best_h[mask] += diff / cnt
        else:
            best_h += diff / n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
