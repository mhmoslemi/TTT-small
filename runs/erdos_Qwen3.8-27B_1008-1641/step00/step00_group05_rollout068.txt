import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(vals, t):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(t * (vals - m)))) / t

def obj_grad(x, t):
    n = len(x)
    h = x
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * (2.0 / n)  # multiply by dx
    val = smooth_max(vals, t)
    weights = np.exp(t * (vals - np.max(vals)))
    sw = np.sum(weights)
    weights /= sw
    
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        lag = k - nm1
        w = weights[k]
        if w < 1e-15:
            continue
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            j = np.arange(n)
            v1 = (j - lag >= 0) & (j - lag < n)
            v2 = (j + lag >= 0) & (j + lag < n)
            grad[v1] += w * (1.0 - h[j[v1] - lag])
            grad[v2] += w * (-h[j[v2] + lag])
    return val, grad

    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) > 1e-10:
        # Add/subtract to reach target
        diff = target - s
        if diff > 0:
            # Need to increase some entries
            order = np.argsort(1.0 - h)  # prioritize entries close to 1... no, close to 0
            order = np.argsort(h)  # entries with smallest values first (most room to increase)
            for i in order:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
                if diff < 1e-12:
                    break
        else:
            order = np.argsort(-h)  # entries with largest values first (most room to decrease)
            for i in order:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
                if -diff < 1e-12:
                    break
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize

_temp = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def smooth_max(vals, t):
    m = np.max(vals)
    return m + np.log(np.sum(np.exp(t * (vals - m)))) / t

def objective_and_grad(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full") * dx
    val = smooth_max(corr, t)
    m = np.max(corr)
    ew = np.exp(t * (corr - m))
    weights = ew / np.sum(ew)
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        wk = weights[k] * dx
        if wk < 1e-18:
            continue
        lag = k - nm1
        if lag == 0:
            grad += wk * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - x[:n - lag])
            grad[:n - lag] += wk * (-x[lag:])
        else:
            P = -lag
            grad[:n - P] += wk * (1.0 - x[P:])
            grad[P:] += wk * (-x[:n - P])
    return val, grad

def obj_fn(x):
    return objective_and_grad(x, _temp)[0]

def grad_fn(x):
    return objective_and_grad(x, _temp)[1]

def sum_c(x):
    return np.sum(x) - len(x) / 2.0

def sum_j(x):

def project_h(h, n):
    h = h.copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for i in order:
            if diff > 0:
                a = min(diff, 1.0 - h[i])
                h[i] += a
                diff -= a
            else:
                a = min(-diff, h[i])
                h[i] -= a
                diff += a
            if abs(diff) < 1e-12:
                break
    np.clip(h, 0.0, 1.0, out=h)
    return h

    h_f = np.interp(xf, xc, h_c)
