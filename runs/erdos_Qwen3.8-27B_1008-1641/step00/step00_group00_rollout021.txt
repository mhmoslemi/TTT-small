        if time.time() > deadline:
            break
        _, grad = soft_obj_grad(h, dx, alpha)
        h_new = project_sum(h - step * grad, target)
        new_c = compute_c5(h_new, dx)
        if new_c < cur - 1e-10:
            h = h_new
            cur = new_c
            stall = 0
            step = max(0.002, step * 0.997)
        else:
            stall += 1
            step *= 0.92
            if step < 1e-5:
                h = project_sum(h + rng.standard_normal(n) * 0.06, target)
                cur = compute_c5(h, dx)
                step = 0.03
                stall = 0
        if cur < best[0]:
            best[0] = cur
            best[1] = h.copy()
            best[2] = n
    return h, best

def coordinate_refine(h, n, dx, target, deadline, best):
    h = h.copy()
    cur = compute_c5(h, dx)
    if cur < best[0]:
        best[0] = cur
        best[1] = h.copy()
        best[2] = n
    cands = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]
    for sweep in range(4):
        if time.time() > deadline:
            break
        for j in range(n):
            old = h[j]
            for c in cands:
                h[j] = c
                hp = project_sum(h, target)
                v = compute_c5(hp, dx)
                if v < best[0] - 1e-10:
                    h = hp.copy()
                    best[0] = v
                    best[1] = h.copy()
            h[j] = best[1][j]
    return best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s * 0.93
    best = [np.inf, None, 0]

    # Phase 1: optimize from parent construction
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        dx0 = 2.0 / n0
        tgt0 = n0 / 2.0
        h0 = project_sum(h0, tgt0)
        v0 = compute_c5(h0, dx0)
        if v0 < best[0]:
            best[0] = v0
            best[1] = h0.copy()
            best[2] = n0
        h0, best = gradient_descent(h0, n0, dx0, tgt0, 60.0, 4000, deadline, rng, best)
        # Second pass with higher alpha
        h0, best = gradient_descent(h0, n0, dx0, tgt0, 120.0, 3000, deadline, rng, best)
        best = coordinate_refine(best[1], n0, dx0, tgt0, deadline, best)
    except NameError:
        pass

    # Phase 2: random starts at various n
    n_list = [45, 48, 50, 54, 60, 64, 72, 80, 100]
    if len(best) > 2 and best[2] > 0:
        if best[2] not in n_list:
            n_list.insert(1, best[2])
    for n in n_list:
        if time.time() > deadline - 15:
            break
        dx = 2.0 / n
        tgt = n / 2.0
        n_re = 4
        for r in range(n_re):
            if time.time() > deadline - 5:
                break
            h_s = gen_start(n, rng)
            h_s = project_sum(h_s + rng.standard_normal(n) * 0.15, tgt)
            h_s, best = gradient_descent(h_s, n, dx, tgt, 70.0, 2000, deadline, rng, best)
            if time.time() > deadline - 5:
                break
            h_s, best = gradient_descent(h_s, n, dx, tgt, 150.0, 1500, deadline, rng, best)

    # Phase 3: fine-tune the global best
    if best[1] is not None and time.time() < deadline - 5:
        n_b = best[2]
        dx_b = 2.0 / n_b
        tgt_b = n_b / 2.0
        best = coordinate_refine(best[1], n_b, dx_b, tgt_b, deadline, best)
        # One more round of gradient with very high alpha
        best = gradient_descent(best[1], n_b, dx_b, tgt_b, 200.0, 2000, deadline, rng, best)
        best = coordinate_refine(best[1], n_b, dx_b, tgt_b, deadline, best)

    h_final = project_sum(best[1], best[2] / 2.0)
    dx_f = 2.0 / best[2]
    c5_final = compute_c5(h_final, dx_f)
    if c5_final < best[0]:
        best[0] = c5_final
    return (best[1], best[0], best[2])
