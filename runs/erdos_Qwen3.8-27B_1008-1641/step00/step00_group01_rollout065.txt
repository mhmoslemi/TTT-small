from scipy.optimize import minimize
from scipy.special import logsumexp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_c5(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(logsumexp(alpha * corr) / alpha * dx)


def fix_sum(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-14:

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    try:
        h = fix_sum(np.array(initial_h_values, dtype=float),
                     len(initial_h_values) / 2.0)
    except Exception:
        h = np.zeros(47)
        h[:23] = 1.0
        h[23] = 0.5

    n = len(h)
    ts = n / 2.0
    h = fix_sum(h, ts)
    best = h.copy()
    bc = compute_c5(best)
    print(f"init  n={n} C5={bc:.8f}")
    bounds = [(0.0, 1.0)] * n

    best, bc = slsqp_phase(best, ts, [10, 30, 100, 300, 1000], bounds, t0, budget_s)
    print(f"p1 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best, bc = rand_phase(best, ts, [0.3, 0.15, 0.08, 0.04], seed, t0, budget_s)
    print(f"p2 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best, bc = coord_phase(best, ts, n, t0, budget_s, seed + 100)
    print(f"p3 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best, bc = slsqp_phase(best, ts, [2000, 10000, 50000], bounds, t0, budget_s)
    print(f"p4 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best, bc = coord_phase(best, ts, n, t0, budget_s, seed + 200)
    print(f"p5 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best, bc = rand_phase(best, ts, [0.003, 0.001, 0.0005], seed + 300, t0, budget_s)
    print(f"p6 C5={bc:.8f} t={time.time()-t0:.0f}s")

    best = np.clip(best, 0.0, 1.0)
    bc = compute_c5(best)

    try:
        v = float(evaluate_erdos_solution(best))
        bc = min(bc, v)
    except Exception:
        pass

    print(f"DONE  C5={bc:.8f} t={time.time()-t0:.0f}s")
    return (best, bc, n)
