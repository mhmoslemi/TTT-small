    corr = np.correlate(h, 1 - h, mode='full') * dx

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_c5_vec(h)))

def grad_c5(h, T=100.0):

    n = len(h)
    dx = 2.0 / n
    c5_vec = compute_c5_vec(h)
    max_val = np.max(c5_vec)
    shifted = T * (c5_vec - max_val)
    exp_s = np.exp(shifted)
    weights = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    s_vals = np.arange(-(n-1), n)
    for j in range(n):
        v1 = (j + s_vals >= 0) & (j + s_vals < n)
        if np.any(v1):
            grad[j] += np.sum(weights[v1] * (1.0 - h[j + s_vals[v1]]))
        v2 = (j - s_vals >= 0) & (j - s_vals < n)
        if np.any(v2):
            grad[j] -= np.sum(weights[v2] * h[j - s_vals[v2]])
    return grad * dx, max_val

def targeted_swap(h, c5_vec, rng):

    n = len(h)

    # where h[i+offset]=1 and h[i]=0 (these are the bad pairs)

    # - Change h[i] from 0 to 1 (for some i where h[i+offset]=1)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_c5_vec(h)))

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-8
        else:
            mask = h > 1e-8
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - np.sum(h)
            if abs(diff2) > 1e-12:
                h += diff2 / len(h)
                h = np.clip(h, 0.0, 1.0)
    return h
