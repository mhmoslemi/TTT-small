```python
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project(x, target_sum):
    """Project x onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    # Use bisection on the water-filling level
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo+hi)/2, 0, 1)

def smooth_obj(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    scaled = corr * t
    m = scaled.max()
    return (m + np.log(np.exp(scaled - m).sum())) / t * dx

def smooth_grad(x, t):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    scaled = corr * t
    m = scaled.max()
    w = np.exp(scaled - m)
    w /= w.sum()
    
    shifts = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for i in range(n):
        v1 = (i + shifts >= 0) & (i + shifts < n)
        v2 = (i - shifts >= 0) & (i - shifts < n)
        t1 = np.sum(w[v1] * (1.0 - x[i + shifts[v1]]))
        t2 = np.sum(w[v2] * x[i - shifts[v2]])
        grad[i] = dx * (t1 - t2)
    return grad

def projected_gradient(h, target_sum, n_iter=200, t=200.0, lr=None):
    h = h.copy()
    n = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        g = smooth_grad(h, t)
        f = smooth_obj(h, t)
        # Adaptive step size
        if lr is None:
            lr = 1.0 / (it + 50)
        h_new = project(h - lr * g, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # Backtracking line search
        step_lr = lr
        while c5_new >= f and step_lr > 1e-6:
            step_lr *= 0.5
            h_new = project(h - step_lr * g, target_sum)
            c5_new = compute_c5(h_new)
        h = h_new
        if it % 50 == 0:
            print(f"  PG iter {it}: C5={best_c5:.6f}")
    return best_h, best_c5

def slsqp_optimize(h, target_sum, t=200.0, maxiter=200):
    n = len(h)
    def obj(x):
        return smooth_obj(x, t)
    def grad(x):
        return smooth_grad(x, t)
    def con_sum(x):
        return x.sum() - target_sum
    def jac_con(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': con_sum, 'jac': jac_con}]
    bounds = [(0, 1)] * n
    
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-10})
    h_opt = np.clip(res.x, 0, 1)
    # Fix sum constraint
    if abs(h_opt.sum() - target_sum) > 1e-6:
        h_opt = project(h_opt, target_sum)
    c5 = compute_c5(h_opt)
    return h_opt, c5
```
