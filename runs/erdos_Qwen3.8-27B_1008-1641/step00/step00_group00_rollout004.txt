import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum_box(v, target_sum):
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(v - mid, 0.0, 1.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def subgrad_per_unit(h, L, n):
    """Compute partial derivative of corr[L] w.r.t each h[i]"""
    phi = np.zeros(n)
    if L >= 0:
        if L == 0:
            phi[:] = 1.0 - 2.0 * h
        else:
            phi[L:] += 1.0
            phi[L:] -= h[:n - L]
            phi[:n - L] -= h[L:]
    else:
        Lp = -L
        if Lp == 0:
            phi[:] = 1.0 - 2.0 * h
        else:
            phi[:n - Lp] += 1.0
            phi[:n - Lp] -= h[Lp:]
            phi[Lp:] -= h[:n - Lp]
    return phi

def subgrad_optimize(h, dx, target_sum, n, max_iter=50000, t0=None, time_budget=60):
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    lr = 0.05
    stall = 0
    for it in range(max_iter):
        if it % 200 == 0 and time.time() - t0 > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        L = max_idx - (n - 1)
        grad = subgrad_per_unit(h, L, n)
        grad -= np.mean(grad)
        h_new = project_sum_box(h - lr * grad, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.5)
            stall = 0
        else:
            lr *= 0.85
            h = best_h.copy()
            stall += 1
            if stall > 300:
                break
    return best_h, best_c5

def local_pair_optimize(h, dx, target_sum, n, max_rounds=200, t0=None, time_budget=120):
    best_c5 = compute_c5(h, dx)
    improved_any = True
    for rnd in range(max_rounds):
        if time.time() - t0 > time_budget:
            break
        if not improved_any:
            break
        improved_any = False
        alpha = 0.05 if rnd < 50 else 0.01
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] < 1e-9 or h[j] > 1.0 - 1e-9:
                    continue
                for sign in (1, -1):
                    delta = alpha * min(max(h[i if sign > 0 else j], 0),
                                        max(1.0 - h[j if sign > 0 else i], 0))
                    if delta < 1e-10:
                        continue
                    h[i] -= sign * delta
                    h[j] += sign * delta
                    c5_new = compute_c5(h, dx)
                    if c5_new < best_c5 - 1e-12:
                        best_c5 = c5_new
                        improved_any = True
                    else:
                        h[i] += sign * delta
                        h[j] -= sign * delta
    return h, best_c5
