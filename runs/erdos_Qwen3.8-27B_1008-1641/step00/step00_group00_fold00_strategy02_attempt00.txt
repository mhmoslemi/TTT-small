def compute_c5(vector):
    """Compute the C5 value given a vector representation of the step function"""
    P = len(vector)
    dx = 2.0 / P
    target_sum = P / 2.0
    
    # Project to [0,1] and fixed sum
    vector_projected = project_vector(vector.copy(), target_sum)
    
    # Compute the autocorrelation with the complement function (1 - h(x))
    x = vector_projected
    y = 1.0 - x
    corr = np.correlate(x, y, mode='full')
    # The full mode gives correlation values from lag -P+1 to P-1 (indexed from 0 to 2P-1)
    # The maximum correlation value (plus 1 for the 0 lag) corresponds to the maximum overlap
    c5_val = np.max(corr) * dx
    return c5_val
