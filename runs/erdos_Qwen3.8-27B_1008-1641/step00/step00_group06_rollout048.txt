import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def grad_corr_component(h, m):
    n = len(h)
    g = np.zeros(n)
    j_min = max(0, m - n + 1)
    j_max = min(n - 1, m)
    if j_min <= j_max:
        idx = np.arange(j_min, j_max + 1)
        g[idx] = 1.0 - 2.0 * h[m - idx]
    return g

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)

def nlp_wrapper(n, target_sum):
    def obj(v):
        return v[n]
    def jac(v):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    def c_eq(v):
        return np.sum(v[:n]) - target_sum
    def c_ineq(v):
        return v[n] - compute_c5(v[:n])
    return obj, jac, c_eq, c_ineq

def run_slsqp(h0, max_iter=400, ftol=1e-12):
    n = len(h0)
    c5 = compute_c5(h0)
    x0 = np.concatenate([h0, [c5]])
    bounds = [(0.0, 1.0)] * n + [(c5 - 0.05, 1.0)]
    obj, jac, c_eq, c_ineq = nlp_wrapper(n, n / 2.0)
    cons = [{'type': 'eq', 'fun': c_eq}, {'type': 'ineq', 'fun': c_ineq}]
    res = minimize(obj, x0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    h = np.clip(res.x[:n], 0.0, 1.0)
    h = project_feasible(h, n / 2.0)
    return h, compute_c5(h)

    h = h0.copy()
    n = len(h)
    best = compute_c5(h)
    best_h = h.copy()

        g = np.zeros(n)
        for k in top_idx:
            g += grad_corr_component(h, k)
        g /= float(n_top)
        lr = 0.3 / (1.0 + 0.02 * it)
        h_new = h - lr * g
        h_new = project_feasible(h_new, n / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best - 1e-12:
            best = c5
            best_h = h_new.copy()
            stall_count = 0

                h_new = project_feasible(h_new, n / 2.0)
                stall_count = 0
        h = h_new
    return best_h, best
