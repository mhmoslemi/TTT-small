    cor = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(cor)

    # Softmax weights
    weights = np.exp(shifted - np.max(shifted))
    weights /= np.sum(weights)

def project_to_feasible(x, n, target_sum):
    """Project x onto {h: sum(h)=target_sum, 0<=h<=1}"""
    lo, hi = -np.max(x) - 1, 1 - np.min(x) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x + mid, 0, 1)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
        if abs(s - target_sum) < 1e-12:
            break
    h = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fix any small numerical error
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-15) & (h < 1 - 1e-15)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    cor = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(cor) * dx)

def smooth_obj(h, dx, T):
    cor = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(cor)
    shifted = T * (cor - m)
    shifted -= np.max(shifted)
    log_sum = np.max(shifted) + np.log(np.sum(np.exp(shifted)))
    return (m + log_sum / T) * dx

def smooth_grad(h, dx, T):
    n = len(h)
    cor = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(cor)
    shifted = T * (cor - m)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= np.sum(w)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        delta = k - n + 1
        wk = w[k]
        for j in range(n):
            g = 0.0
            jm = j - delta
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jp = j + delta
            if 0 <= jp < n:
                g -= h[jp]
            grad[j] += wk * g
    grad *= dx
    return grad

def project_feasible(x, n, target_sum):
    lo, hi = -np.max(x) - 0.5, 1.0 - np.min(x) + 0.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
        if abs(s - target_sum) < 1e-13:
            break
    h = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def projected_grad_descent(h_init, n, dx, target_sum, max_iter, T_start, seed, time_limit):
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    T = T_start
    step = 0.02
    best_local = compute_c5(h, dx)
    best_h = h.copy()
    t_check = time.time()
    for it in range(max_iter):
        if it % 50 == 0 and it > 0:
            if time.time() > time_limit:
                break
        grad = smooth_grad(h, dx, T)
        h_try = project_feasible(h - step * grad, n, target_sum)
        f_try = smooth_obj(h_try, dx, T)
        f_cur = smooth_obj(h, dx, T)
        if f_try < f_cur:
            h = h_try
            step = min(step * 1.05, 0.1)
        else:
            step *= 0.5
            if step < 1e-8:
                step = 0.01
        if it % 30 == 0:
            T = max(T * 0.85, 5.0)
        if it % 10 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_local:
                best_local = c5
                best_h = h.copy()
    c5_final = compute_c5(h, dx)
    if c5_final < best_local:
        best_local = c5_final
        best_h = h.copy()
    return best_h, best_local

def slsqp_refine(h_init, n, dx, target_sum, T, time_limit):
    def obj(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_obj(h, dx, T)
    def obj_grad(x):
        h = np.clip(x, 0.0, 1.0)
        return smooth_grad(h, dx, T)
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_init, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': 200, 'ftol': 1e-12})
    if res.success or res.fun < smooth_obj(h_init, dx, T):
        h = np.clip(res.x, 0.0, 1.0)
        h = project_feasible(h, n, target_sum)
        return h
    return h_init.copy()

def coord_descent(h_init, n, dx, target_sum, max_rounds, time_limit):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() > time_limit:
                break
            old_val = h[j]
            # Vary h[j], adjust one other coordinate to maintain sum
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                new_val = old_val + delta
                if new_val < 0 or new_val > 1:
                    continue
                h[j] = new_val
                # Adjust another coordinate
                diff = delta
                # Spread adjustment
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                # Find coordinate with room
                best_k = -1
                for k in range(n):
                    if not mask[k]:
                        continue
                    adj = h[k] - diff
                    if 0 <= adj <= 1:
                        best_k = k
                        break
                if best_k < 0:
                    h[j] = old_val
                    continue
                h[best_k] -= diff
                c5_try = compute_c5(h, dx)
                if c5_try < best_c5 - 1e-10:
                    best_c5 = c5_try
                    improved = True
                else:
                    h[j] = old_val
                    h[best_k] += diff
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)
    time_limit = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
        n = len(h_best)
    else:
        n = 45
        h_best = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    h_best = project_feasible(h_best, n, target_sum)
    best_c5 = compute_c5(h_best, dx)

    # Phase 1: Projected gradient with annealing
    for attempt in range(6):
        if time.time() > time_limit - 60:
            break
        h_start = h_best.copy()
        if attempt > 0:
            noise = np.random.RandomState(seed + attempt * 137).randn(n) * 0.05
            h_start = project_feasible(h_start + noise, n, target_sum)
        h_new, c5_new = projected_grad_descent(h_start, n, dx, target_sum,
                                               max_iter=800, T_start=30.0,
                                               seed=seed + attempt, time_limit=time_limit)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()

    # Phase 2: SLSQP refinement from best
    if time.time() < time_limit - 30:
        h_slq = slsqp_refine(h_best, n, dx, target_sum, T=40.0, time_limit=time_limit)
        c5_slq = compute_c5(h_slq, dx)
        if c5_slq < best_c5:
            best_c5 = c5_slq
            h_best = h_slq.copy()

    # Phase 3: Coordinate descent polish
    if time.time() < time_limit - 10:
        h_cd, c5_cd = coord_descent(h_best, n, dx, target_sum,
                                    max_rounds=20, time_limit=time_limit)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            h_best = h_cd.copy()

    # Phase 4: Another round of gradient + polish
    if time.time() < time_limit - 30:
        h_new, c5_new = projected_grad_descent(h_best, n, dx, target_sum,
                                               max_iter=500, T_start=50.0,
                                               seed=seed + 999, time_limit=time_limit)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_best = h_new.copy()
        if time.time() < time_limit - 10:
            h_slq2 = slsqp_refine(h_best, n, dx, target_sum, T=60.0, time_limit=time_limit)
            c5_s2 = compute_c5(h_slq2, dx)
            if c5_s2 < best_c5:
                best_c5 = c5_s2
                h_best = h_slq2.copy()

    # Final verification and repair
    h_best = np.clip(h_best, 0.0, 1.0)
    h_best = project_feasible(h_best, n, target_sum)
    best_c5 = compute_c5(h_best, dx)

    return (h_best, best_c5, n)
