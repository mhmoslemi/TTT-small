import numpy as np
import time

def evaluate(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(500):
        err = h.sum() - target
        if abs(err) < 1e-15:
            break
        if err > 0:
            mask = h > 1e-15
        else:
            mask = h < 1.0 - 1e-15
        if not mask.any():
            break
        h = np.clip(h - (err / mask.sum()) * mask, 0.0, 1.0)
    return h

def hill_climb(h, n, dx, rng, t_end, step0=0.04):
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    step = step0
    stall = 0
    while time.time() < t_end:
        h_try = best_h + rng.standard_normal(n) * step
        h_try = project(h_try, n)
        c5 = evaluate(h_try, dx)
        if c5 < best_c5 - 1e-14:
            best_h = h_try
            best_c5 = c5
            stall = 0
            step = min(0.08, step * 1.006)
        else:
            stall += 1
            if stall > 80:
                step = max(0.012, step * 2.2)
                stall = 0
    return best_h, best_c5

def coord_polish(h, n, dx, rng, t_end):
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    deltas = np.array([0.005, 0.01, 0.02, -0.005, -0.01, -0.02,
                       0.002, -0.002, 0.015, -0.015])
    while time.time() < t_end:
        improved = False
        idx = rng.permutation(n)
        for i in idx:
            if time.time() >= t_end:
                break
            for d in deltas:
                h_try = best_h.copy()
                h_try[i] = np.clip(h_try[i] + d, 0.0, 1.0)
                h_try = project(h_try, n)
                c5 = evaluate(h_try, dx)
                if c5 < best_c5 - 1e-15:
                    best_h = h_try
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
