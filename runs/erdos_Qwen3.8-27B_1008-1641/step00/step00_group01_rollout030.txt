import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return (2.0 / n) * np.max(np.correlate(h, 1 - h, mode='full'))

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(5):
        lo, hi = -1.5, 1.5
        for _ in range(70):
            mid = (lo + hi) * 0.5
            s = np.sum(np.clip(h - mid, 0.0, 1.0))
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
        if abs(h.sum() - target_sum) < 1e-13:
            return h
    return h

def smooth_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    sc = corr / T
    sc -= sc.max()
    ec = np.exp(sc)
    w = ec / ec.sum()
    g = np.zeros(n)
    omh = 1.0 - h
    for lag in range(-(n - 1), n):
        wk = w[lag + n - 1]
