For a random h with mean 1/2, the expected value of each correlate entry is approximately 0 (since E[h] = E[1-h] = 1/2, so E[h(1-h)] = 1/4, and the sum over n terms gives n/4, times dx = 2/n gives (2/n)(n/4) = 1/2). Wait, that's not right.

Actually: correlate(h, 1-h)[k] = sum_i h[i] * (1-h)[i+k-n+1] (with appropriate bounds)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    return float(v_max + np.log(np.exp((vals - v_max) * beta).sum()) / beta)

def project(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        if not np.any(mask):
            break
        h[mask] += diff / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    return h

def make_objective(dx, target_sum, beta):
    def obj(p):
        h = project(p, target_sum)
        return soft_c5(h, dx, beta)
    return obj

def lbbfgs_optimize(h, n, dx, target, beta, max_iter, rng):
    def obj(p):
        h = project(p, target)
        return soft_c5(h, dx, beta)
    res = minimize(obj, h, method='L-BFGS-B',
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-13})
    h_new = project(res.x, target)
    return h_new, compute_c5(h_new, dx)

def coord_descent(h, n, dx, target, t_end, rng):
    best = compute_c5(h, dx)
    scales = [0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
    for scale in scales:
        if time.time() > t_end:
            break
        for sweep in range(3):
            if time.time() > t_end:
                break
            improved = True
            while improved:
                improved = False
                if time.time() > t_end:
                    break
                for i in range(n):
                    for j in range(i + 1, n):
                        for dv in [scale, -scale]:
                            if 0.0 <= h[i] + dv <= 1.0 and 0.0 <= h[j] - dv <= 1.0:
                                h[i] += dv
                                h[j] -= dv
                                c5_try = compute_c5(h, dx)
                                if c5_try < best - 1e-14:
                                    best = c5_try
                                    improved = True
                                else:
                                    h[i] -= dv
                                    h[j] += dv
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = np.inf
    best_h = None
    best_n = 47

    def try_push(h, c5, n):
        nonlocal best_c5, best_h, best_n
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n

    # Get initial values
    h_init = None
    n_init = 47
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)

    # Phase 1: Optimize at initial n
    if h_init is not None:
        n = n_init
        dx = 2.0 / n
        target = n / 2.0
        h = project(h_init, target)
        c5 = compute_c5(h, dx)
        try_push(h, c5, n)
        print(f"Initial n={n}: C5={c5:.6f}")

        # L-BFGS-B with various betas and restarts
        t_lbfgs_end = t0 + budget_s * 0.4
        for beta in [20.0, 40.0, 80.0, 150.0]:
            for trial in range(4):
                if time.time() > t_lbfgs_end:
                    break
                if trial == 0:
                    h0 = best_h.copy()
                else:
                    h0 = project(best_h + rng.randn(n) * (0.08 - 0.015 * trial), target)
                h_new, c5_new = lbbfgs_optimize(h0, n, dx, target, beta, 400, rng)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
                    best_n = n
                    print(f"  LBB n={n} beta={beta} t{trial}: C5={c5_new:.6f}")

        # Coordinate descent polishing
        t_cd_end = t0 + budget_s * 0.75
        best_h, best_c5 = coord_descent(best_h, best_n, 2.0 / best_n,
                                         best_n / 2.0, t_cd_end, rng)
        best_n = best_n
        print(f"After CD n={best_n}: C5={best_c5:.6f}")

    # Phase 2: Try nearby n values for improvement
    t_phase2_end = t0 + budget_s * 0.9
    if time.time() < t_phase2_end:
        for n_try in [45, 49, 51, 47, 53, 55]:
            if time.time() > t_phase2_end:
                break
            dx_t = 2.0 / n_try
            target_t = n_try / 2.0
            # Interpolate or create initial guess
            if best_h is not None and n_try != len(best_h):
                x_old = np.linspace(0, 2, len(best_h), endpoint=False)
                x_new = np.linspace(0, 2, n_try, endpoint=False)
                h_guess = np.interp(x_new, x_old, best_h)
            elif best_h is not None:
                h_guess = best_h.copy()
            else:
                h_guess = np.full(n_try, 0.5)
            h_guess = project(h_guess, target_t)
            c5_g = compute_c5(h_guess, dx_t)
            # Quick L-BFGS-B
            h_new, c5_new = lbbfgs_optimize(h_guess, n_try, dx_t, target_t, 60.0, 200, rng)
            h_new, c5_new2 = coord_descent(h_new, n_try, dx_t, target_t,
                                            min(time.time() + 30, t_phase2_end), rng)
            try_push(h_new, c5_new2, n_try)
            if c5_new2 < 0.381:
                print(f"  n={n_try}: C5={c5_new2:.6f}")

    # Final: ensure constraints are met
    dx_f = 2.0 / best_n
    best_h = project(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h, dx_f)
    print(f"FINAL: n={best_n}, C5={best_c5:.6f}, time={time.time()-t0:.1f}s")

    return (best_h, best_c5, best_n)
