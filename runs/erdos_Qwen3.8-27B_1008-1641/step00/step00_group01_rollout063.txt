        starts.append(fix_sum(h, _SUM_TARGET))

    # Phase 1: L-BFGS-B with warm temperature
    _SM = 0.05
    _MU = 200.0
    for h_s in starts:
        if time.time() - t0 > budget_s * 0.50:
            break
        if np.sum(h_s) < 1e-8:
            continue
        try:
            r = minimize(pen_obj, h_s, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': 400, 'ftol': 1e-12})
        except Exception:
            continue
        h_new = fix_sum(r.x, _SUM_TARGET)
        c, _ = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new

    # Phase 2: L-BFGS-B with cold temperature from best
    _SM = 0.005
    _MU = 1000.0
    h_s = best_h.copy()
    try:
        r = minimize(pen_obj, h_s, method='L-BFGS-B',
                     bounds=bounds, options={'maxiter': 600, 'ftol': 1e-14})
        h_new = fix_sum(r.x, _SUM_TARGET)
        c, _ = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new
    except Exception:
        pass

    # Phase 2b: restart with slightly perturbed best
    if time.time() - t0 < budget_s * 0.65:
        for _ in range(3):
            hp = best_h + 0.015 * rng.randn(n)
            hp = fix_sum(np.clip(hp, 0, 1), _SUM_TARGET)
            try:
                r = minimize(pen_obj, hp, method='L-BFGS-B',
                             bounds=bounds, options={'maxiter': 300, 'ftol': 1e-13})
                h_new = fix_sum(r.x, _SUM_TARGET)
                c, _ = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new
            except Exception:
                pass
            if time.time() - t0 > budget_s * 0.65:
                break

    # Phase 3: SLSQP with exact equality constraint
    _SM = 0.002
    cons = [{'type': 'eq', 'fun': sum_eq}]
    slsqp_starts = [best_h.copy()]
    if len(starts) > 1:
        slsqp_starts.append(starts[1].copy())
    for h_s in slsqp_starts:
        if time.time() - t0 > budget_s * 0.78:
            break
        try:
            r = minimize(slsqp_obj, h_s, method='SLSQP', bounds=bounds,
                         constraints=cons,
                         options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = fix_sum(r.x, _SUM_TARGET)
            c, _ = compute_c5(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new
        except Exception:
            pass

    # Phase 4: Local exchange moves (preserves sum exactly)
    deltas_coarse = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    deltas_fine = [0.008, -0.008, 0.003, -0.003, 0.001, -0.001, 0.0005, -0.0005]
    for it in range(60):
        if time.time() - t0 > budget_s - 15:
            break
        improved = False
        dl = deltas_coarse if it < 35 else deltas_fine
        attempts = 3000 if it < 40 else 2000
        for _ in range(attempts):
            i, j = int(rng.randint(0, n)), int(rng.randint(0, n))
            if i == j:
                continue
            d = float(rng.choice(dl))
            hb = best_h.copy()
            hb[i] += d
            hb[j] -= d
            if 0.0 <= hb[i] <= 1.0 and 0.0 <= hb[j] <= 1.0:
                c, _ = compute_c5(hb)
                if c < best_c5 - 1e-13:
                    best_h = hb
                    best_c5 = c
                    improved = True
        if not improved:
            break

    # Final verification and sum fix
    best_h = fix_sum(best_h, _SUM_TARGET)
    best_c5, _ = compute_c5(best_h)
    return best_h, best_c5, n
