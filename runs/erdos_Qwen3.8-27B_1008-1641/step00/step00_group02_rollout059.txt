import time

    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def objective_smooth(h, n, penalty, T):
    """Smooth objective using log-sum-exp + penalty"""
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    # Log-sum-exp of corr
    c = corr / T
    c_max = np.max(c)
    smooth_max = T * (c_max + np.log(np.sum(np.exp(c - c_max))))
    sum_pen = penalty * (float(np.sum(h)) - n / 2.0) ** 2

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5_from_h(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * (2.0 / n)


def objective_smooth(h, n, penalty, T):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    c = corr / T
    c_max = np.max(c)
    smooth_max = T * (c_max + np.log(np.sum(np.exp(c - c_max))))
    s = float(np.sum(h)) - n / 2.0
    return smooth_max + penalty * s * s


def objective_direct(h, n, penalty):
    corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    s = float(np.sum(h)) - n / 2.0
    return float(np.max(corr)) + penalty * s * s


def fix_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = n / 2.0 - float(np.sum(h))
    if abs(diff) > 1e-14:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += diff / float(np.sum(mask))
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = n / 2.0 - float(np.sum(h))
        if abs(diff2) > 1e-14:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    return h


def generate_initial(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n + 1)[:-1]
    h = np.full(n, 0.5)
    for freq, amp in [(1, 0.3), (2, 0.18), (3, 0.1), (5, 0.06)]:
        phase = rng.uniform(0.0, 2.0 * np.pi)
        h += amp * np.cos(2.0 * np.pi * freq * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    total = float(np.sum(h))
    if total > 1e-6:
        h *= (n / 2.0) / total
    h = fix_h(h, n)
    return h


def pairwise_search(h, n, time_limit, seed):

    stall = 0
    while time.time() < deadline:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        h_new = h_cur.copy()
        h_new[i], h_new[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5_from_h(h_new)
        if c5_new < c5_cur - 1e-9:
            h_cur = h_new
            c5_cur = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 8000:
                break
    return h_cur, c5_cur


def optimize_for_n(n, h_start, time_limit, seed):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bounds = [(0.0, 1.0)] * n
    penalty = 500.0
    h_cur = h_start.copy()
    c5_cur = compute_c5_from_h(h_cur)
    deadline = t0 + time_limit
    rounds = 0

    while time.time() < deadline - 10:
        rounds += 1
        if c5_cur > 0.39:
            T = 0.08
        elif c5_cur > 0.381:
            T = 0.03
        else:
            T = 0.008

        res = minimize(
            objective_smooth, h_cur,
            args=(n, penalty, T),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-13, 'gtol': 1e-8}
        )
        h_new = fix_h(res.x, n)
        c5_new = compute_c5_from_h(h_new)

        if c5_new < c5_cur - 1e-9:
            h_cur = h_new
            c5_cur = c5_new
        else:
