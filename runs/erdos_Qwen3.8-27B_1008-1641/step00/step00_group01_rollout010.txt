import numpy as np
import scipy.optimize as opt
import time

def project_to_feasible(x, n):
    """Project x onto {h : 0<=h[i]<=1, sum(h)=n/2}"""
    h = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to smallest values (those with room)
        order = np.argsort(h)
        for i in order:
            room = 1.0 - h[i]
            add = min(room, diff)
            h[i] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    else:
        # Need to decrease sum: reduce largest values
        order = np.argsort(-h)
        for i in order:
            room = h[i]
            rem = min(room, -diff)
            h[i] -= rem
            diff += rem
            if abs(diff) < 1e-12:
                break
    return h

    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

import numpy as np
import scipy.optimize as opt
from functools import partial
import time


def project_feasible(x, n):
    h = np.clip(np.asarray(x, dtype=float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) < 1e-15:
        return h
    idx = np.argsort(h) if diff > 0 else np.argsort(-h)
    for i in idx:
        if diff > 0:
            step = min(diff, 1.0 - h[i])
        else:
            step = min(-diff, h[i])
        h[i] += step if diff > 0 else -step
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
    return h


def c5_val(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def c5_obj(x, dx, n):
    return c5_val(project_feasible(x, n), dx)


def smooth_c5(x, dx, n, alpha):
    h = project_feasible(x, n)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = corr.max()
    return float(m + np.log(np.mean(np.exp((corr - m) * alpha))) / alpha)


def _sum_eq(x, target):
    return x.sum() - target


def _jac_ones(x, n):
    return np.ones(n)


def nm_refine(h, dx, n, maxiter):
    res = opt.minimize(c5_obj, h, args=(dx, n), method='Nelder-Mead',
                       options={'maxiter': maxiter, 'xatol': 1e-13,
                                'fatol': 1e-15, 'adaptive': True})
    h_out = project_feasible(res.x, n)
    return h_out, c5_val(h_out, dx)


def powell_refine(h, dx, n, maxiter):
    res = opt.minimize(c5_obj, h, args=(dx, n), method='Powell',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'xtol': 1e-13,
                                'ftol': 1e-15})
    h_out = project_feasible(res.x, n)
