from scipy.optimize import minimize, Bounds, LinearConstraint
import time

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / len(h)
    return float(np.max(corr) * dx)

def smooth_max_obj(h, alpha):
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / len(h)
    c = corr * dx
    c_max = np.max(c)
    return c_max + np.log(np.sum(np.exp(alpha * (c - c_max)))) / alpha

def smooth_max_jac(h, alpha):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c = corr * dx
    c_max = np.max(c)
    exp_c = np.exp(alpha * (c - c_max))
    weights = exp_c / np.sum(exp_c)
    
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-14:
            continue
        k = j - (n - 1)
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            m = -k
            grad[m:] += w * (1.0 - h[:n-m])
            grad[:n-m] -= w * h[m:]
    return grad

def project_to_feasible(h, n):
    """Project h to satisfy sum(h) = n/2 and 0<=h<=1"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    # Simple: distribute uniformly among non-bound variables
    free = (h > 1e-8) & (h < 1.0 - 1e-8)
    if np.sum(free) > 0:
        h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)

        diff = target - np.sum(h)

        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)

        lambda x: smooth_max_obj(x, alpha),
        h0.copy(),

        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
    )
    if res.success or res.fun < smooth_max_obj(h0, alpha):

    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # cell centers

    h_new = np.clip(h_new, 0, 1)
    # Final adjustment
    diff = target - np.sum(h_new)
    h_new += diff / n_new
    return h_new

        if res.x is not None and np.all(np.isfinite(res.x)):
            hx = np.clip(res.x, 0.0, 1.0)
            cv = compute_c5(hx)
            if cv < smooth_max_obj(h0, alpha) + 0.01:
                return hx, cv
    except Exception:
        pass
    return h0, compute_c5(h0)


def interpolate_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    h_init = np.array(initial_h_values, dtype=float)
    h_init = np.clip(h_init, 0.0, 1.0)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: progressive local optimization on initial
    for alpha in [30.0, 80.0, 150.0, 300.0, 600.0]:
        if time_left() < 40:
            break
        h_opt, cv = local_optimize(best_h, len(best_h), alpha=alpha, maxiter=200)
        if cv < best_c5 - 1e-7:
            best_c5 = cv
            best_h = h_opt.copy()

    # Phase 2: try different resolutions
    for n_try in [80, 90, 100, 120, 140, 160]:
        if time_left() < 150:
            break
        if n_try == n_init:
            continue
        h_interp = interpolate_h(best_h, n_try)
        for alpha in [50.0, 150.0, 300.0]:
            if time_left() < 80:
                break
            h_opt, cv = local_optimize(h_interp, n_try, alpha=alpha, maxiter=120)
            if cv < best_c5 - 1e-7:
                best_c5 = cv
                best_h = h_opt.copy()

    # Phase 3: random restarts at best resolution
    n_best = len(best_h)
    restarts = 0
    while time_left() > 80 and restarts < 40:
        restarts += 1
        scale = 0.03 + 0.12 * (restarts / 40.0)
        h_pert = best_h + rng.randn(n_best) * scale
        h_pert = make_feasible(h_pert, n_best)

        if restarts % 7 == 0:
            h_rand = rng.rand(n_best)
            h_rand = h_rand * (n_best / (2.0 * max(np.sum(h_rand), 1e-9)))
            h_pert = make_feasible(h_rand, n_best)

        alpha_use = 100.0 if restarts < 20 else 200.0
        h_opt, cv = local_optimize(h_pert, n_best, alpha=alpha_use, maxiter=100)
        if cv < best_c5 - 1e-7:
            best_c5 = cv
            best_h = h_opt.copy()

    # Phase 4: final fine-tuning with high alpha
    for alpha in [300.0, 600.0, 1000.0]:
        if time_left() < 25:
            break
        h_opt, cv = local_optimize(best_h, len(best_h), alpha=alpha, maxiter=250)
        if cv < best_c5 - 1e-9:
            best_c5 = cv
            best_h = h_opt.copy()

    # Final feasibility check
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
