    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_simplex_box(h, target_sum):
    """Project h onto {0 <= x_i <= 1, sum(x) = target_sum}"""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        # Need to increase sum: add to elements with most room
        room = 1.0 - h
        order = np.argsort(-room)  # most room first
        for i in order:
            add = min(diff, room[i])
            h[i] += add
            diff -= add
            if diff <= 0:
                break
    else:

        diff = -diff
        order = np.argsort(-h)  # most value first
        for i in order:
            sub = min(diff, h[i])
            h[i] -= sub
            diff -= sub
            if diff <= 0:

        # = sum(h) - sum_A h[i] + |A| + (n - |A|)*t

            if s2 >= 0 and e2 <= n:
                block = trial[start:start+block_len].copy()
                trial[start:s2] = trial[s2:e2]
                trial[s2:e2] = block
        else:
            # Reset a few random positions
            k = rng.integers(2, 6)
            idx = rng.choice(n, size=k, replace=False)
            trial[idx] = rng.uniform(0, 1, size=k)
    else:
        # Coarse: large changes
        if r < 0.3:
            # Half-reset
            k = n // 2
            idx = rng.choice(n, size=k, replace=False)
            trial[idx] = rng.uniform(0, 1, size=k)
        elif r < 0.6:
            # Random anneal: mix with random
            alpha = rng.uniform(0.3, 0.7)
            trial = alpha * h + (1 - alpha) * rng.uniform(0, 1, n)
            # Bias toward sum constraint
            trial = trial * (target_sum(trial, len(trial)) / np.sum(trial)) if np.sum(trial) > 0 else h
        else:
            # Large perturbation on contiguous block
            block_len = rng.integers(n // 4, n // 2)
            start = rng.integers(0, n - block_len + 1)
            trial[start:start+block_len] = rng.uniform(0, 1, size=block_len)
    return trial

def target_sum(h, n):
    return n * 0.5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = np.full(45, 0.5)
    
    n = len(h_init)
    target = n / 2.0
    h_current = project_box_sum(h_init, target)
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)
    
    no_improve = 0
    max_no_improve = 2000
    phase = 0
    it_count = 0
    
    while time.time() - t0 < budget_s - 5:
        elapsed = time.time() - t0
        # Update phase based on elapsed time
        if elapsed < budget_s * 0.35:
            phase = 0
        elif elapsed < budget_s * 0.7:
            phase = 1
        else:
            phase = 2
        
        trial = do_perturbation(h_current, rng, phase)
        trial = project_box_sum(trial, target)
        trial_c5 = compute_c5(trial)
        
        if trial_c5 <= best_c5 + 1e-15:
            if trial_c5 < best_c5:
                best_h = trial.copy()
                best_c5 = trial_c5
                no_improve = 0
                max_no_improve = int(max(500, max_no_improve * 1.05))
            h_current = trial
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Reset from best with perturbation
                h_current = best_h + rng.normal(0, 0.1, n)
                h_current = project_box_sum(h_current, target)
                no_improve = 0
                max_no_improve = int(max(2000, max_no_improve * 1.2))
        
        it_count += 1
        
        # Periodic: try coordinate-wise optimization every 5000 iters
        if it_count % 5000 == 0 and time.time() - t0 < budget_s - 20:
            # For each coordinate, try to find better value via local search
            for coord in range(n):
                old_val = h_current[coord]
                # Try a few candidate values
                for candidate in [0.0, 1.0, old_val * 0.5, 1.0 - old_val,
                                  max(0, old_val - 0.1), min(1, old_val + 0.1)]:
                    h_try = h_current.copy()
                    h_try[coord] = candidate
                    h_try = project_box_sum(h_try, target)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try.copy()
                        best_c5 = c5_try
                        h_current = h_try.copy()
        
        # If we've been at best for a while, try a fresh start
        if no_improve > 50000 and phase >= 1:
            h_current = best_h + rng.normal(0, 0.3, n)
            h_current = project_box_sum(h_current, target)
            no_improve = 0
            max_no_improve = 2000
    
    # Final: try to refine at higher resolution if time allows
    remaining = budget_s - (time.time() - t0)
    if remaining > 60 and best_c5 < 0.381:
        n2 = n * 2
        h2 = interpolate_to_n(best_h, n2)
        h2 = project_box_sum(h2, target2 := n2 / 2.0)
        c5_2 = compute_c5(h2)
        
        if c5_2 <= best_c5 + 0.001:
            target2 = n2 / 2.0
            t2 = time.time()
            for _ in range(min(20000, int(remaining * 3000))):
                if time.time() - t2 > remaining - 5:
                    break
                trial2 = do_perturbation(h2, rng, 0)
                trial2 = project_box_sum(trial2, target2)
                c5_try = compute_c5(trial2)
                if c5_try < c5_2:
                    best_h = trial2.copy()
                    best_c5 = c5_2 = c5_try
                    h2 = trial2
    
    # Ensure final feasibility
    n_final = len(best_h)
    best_h = project_box_sum(best_h, n_final / 2.0)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_final
