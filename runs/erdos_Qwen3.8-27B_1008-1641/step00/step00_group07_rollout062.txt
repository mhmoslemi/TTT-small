    """Project v onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(v)
    lo, hi = -1.0, 2.0
    for _ in range(100):

        h = np.clip(v - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0, 1)
    # Fix tiny numerical error
    diff = target - h.sum()
    if abs(diff) < 1e-8:
        # distribute among non-bound entries
        free = (h > 1e-6) & (h < 1-1e-6)
        if free.sum() > 0:
            h[free] += diff / free.sum()
        else:
            h += diff / n
    return h

def compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def smooth_obj_hess(h, dx, T):
    """Smooth max of correlations, returns (value, gradient)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    shifted = np.clip((corr - m) / T, -60, 0)
    w = np.exp(shifted)
    ws = w.sum()
    val = m + T * np.log(ws) if ws > 1e-300 else m
    if ws < 1e-300:
        return val, np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        L = n - abs(lag)
        if L <= 0:
            continue
        if lag >= 0:
            grad[:L] += wk * (1.0 - h[lag:lag + L])
            grad[lag:lag + L] -= wk * h[:L]
        else:
            l = -lag
            grad[l:l + L] += wk * (1.0 - h[:L])
            grad[:L] -= wk * h[l:l + L]
    grad *= dx
    return val, grad

        return m, np.zeros(n)
    w /= ws
    val = m + T * np.log(ws)
    grad = np.zeros(n)
    active = np.where(w > 1e-9)[0]
    for k in active:
        wk = w[k]
        lag = k - (n - 1)
        L = n - abs(lag)
        if lag >= 0:
            grad[:L] += wk * (1.0 - h[lag:lag + L])
            grad[lag:lag + L] -= wk * h[:L]
        else:
            l = -lag
            grad[l:l + L] += wk * (1.0 - h[:L])
            grad[:L] -= wk * h[l:l + L]
    grad *= dx
    return val, grad

def project_feasible(v, target):
    n = len(v)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        if free.any():
            h[free] += diff / free.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def fix_sum_exact(h, target):
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    idx = np.argsort(h)
    if diff > 0:
        for i in range(len(idx) - 1, -1, -1):
            room = 1.0 - h[idx[i]]
            add = min(room, diff)
            h[idx[i]] += add
            diff -= add
            if diff < 1e-14:
                break
    else:
        for i in range(len(idx)):
            room = h[idx[i]]
            sub = min(room, -diff)
            h[idx[i]] -= sub
            diff += sub
            if -diff < 1e-14:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(80, 0.5)

    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    h0 = project_feasible(h0, target)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    T_schedule = [0.1, 0.05, 0.03, 0.01, 0.005, 0.002, 0.001, 0.0005]

    for Ti, T in enumerate(T_schedule):
        if tl() < 60:
            break
        n_rest = 2 + (Ti % 3)
        for r in range(n_rest):
            if tl() < 20:
                break
            if r == 0:
                x0 = best_h.copy()
            elif r == 1:
                x0 = best_h + 0.005 * np.random.randn(n)
                x0 = project_feasible(x0, target)
            else:
                x0 = best_h + (0.01 + 0.01 * r) * np.random.randn(n)
                x0 = project_feasible(x0, target)

            Tloc = T
            one_arr = np.ones(n)

            def f(x):
                v, _ = smooth_obj_and_grad(x, dx, Tloc)
                return v

            def g(x):
                _, gg = smooth_obj_and_grad(x, dx, Tloc)
                return gg

            def con_fun(x):
                return x.sum() - target

            def con_jac(x):
                return one_arr

            cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
            maxiter = 400 if Ti < 4 else 600
            try:
                res = minimize(f, x0, jac=g, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': maxiter, 'ftol': 1e-14})
                x_opt = project_feasible(res.x, target)
            except Exception:
                x_opt = project_feasible(x0, target)

            c5 = compute_c5(x_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x_opt.copy()

    # Fine-tuning at very low temperature with small perturbations
    if tl() > 50:
        T_fine = 0.0001
        one_arr2 = np.ones(n)
        for attempt in range(15):
            if tl() < 20:
                break
            x0 = best_h + 0.002 * np.random.randn(n)
            x0 = project_feasible(x0, target)
            Tloc = T_fine

            def f2(x):
                v, _ = smooth_obj_and_grad(x, dx, Tloc)
                return v

            def g2(x):
                _, gg = smooth_obj_and_grad(x, dx, Tloc)
                return gg

            cons2 = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                     'jac': lambda x: one_arr2}]
            try:
                res = minimize(f2, x0, jac=g2, method='SLSQP',
                              bounds=bounds, constraints=cons2,
                              options={'maxiter': 300, 'ftol': 1e-14})
                x_opt = project_feasible(res.x, target)
                c5 = compute_c5(x_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x_opt.copy()
            except Exception:
                pass

    # One more round at even lower T if time permits
    if tl() > 40:
        T_fine2 = 0.00001
        one_arr3 = np.ones(n)
        for attempt in range(10):
            if tl() < 15:
                break
            x0 = best_h + 0.001 * np.random.randn(n)
            x0 = project_feasible(x0, target)
            Tloc = T_fine2

            def f3(x):
                v, _ = smooth_obj_and_grad(x, dx, Tloc)
                return v

            def g3(x):
                _, gg = smooth_obj_and_grad(x, dx, Tloc)
                return gg

            cons3 = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                     'jac': lambda x: one_arr3}]
            try:
                res = minimize(f3, x0, jac=g3, method='SLSQP',
                              bounds=bounds, constraints=cons3,
                              options={'maxiter': 200, 'ftol': 1e-15})
                x_opt = project_feasible(res.x, target)
                c5 = compute_c5(x_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x_opt.copy()
            except Exception:
                pass

    # Final exact sum fix and verification
    best_h = project_feasible(best_h, target)
    best_h = fix_sum_exact(best_h, target)
    best_c5 = compute_c5(best_h, dx)

    return (best_h, best_c5, n)
