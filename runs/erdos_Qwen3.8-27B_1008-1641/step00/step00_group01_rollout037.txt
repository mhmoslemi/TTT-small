            h_try = project(h_try, tgt)
            c_try = c5_of(h_try)
            if c_try < c5c - 1e-12:
                h = h_try
                c5c = c_try
                if c_try < best_c:
                    best_c = c_try
                    best_h = h.copy()
        return best_h, best_c

    # Start from initial
    h_best = initial_h_values.astype(np.float64).copy()
    n_init = len(h_best)
    h_best = project(h_best, n_init * 0.5)
    c5_best = c5_of(h_best)

    # Phase 1: long SA on initial (40% of budget)
    dl1 = t0 + budget_s * 0.40
    h_best, c5_best = optimize(h_best, c5_best, dl1, aggressive=True)

    # Phase 2: try other sizes (50% of budget total for this)
    sizes = [46, 48, 50, 54, 60, 80, 100]
    per_size = budget_s * 0.06
    for ns in sizes:
        if time.time() - t0 > budget_s * 0.90:
            break
        # Interpolate from best
        xo = np.linspace(0, 1, len(h_best), endpoint=False)
        xn = np.linspace(0, 1, ns, endpoint=False)
        h_try = np.interp(xn, xo, h_best)
        h_try = project(h_try, ns * 0.5)
        c5_try = c5_of(h_try)
        # Also try binary init for even n
        if ns % 2 == 0 and ns <= 100:
            h_bin = rng.permutation(ns).astype(np.float64)
            h_bin2 = np.zeros(ns)
            h_bin2[:ns//2] = 1.0
            c5_bin = c5_of(h_bin2)
            if c5_bin < c5_try:
                h_try = h_bin2.copy()
                c5_try = c5_bin
        dl = min(t0 + budget_s * 0.90, time.time() + per_size)
        h_try, c5_try = optimize(h_try, c5_try, dl, aggressive=True)
        if c5_try < c5_best:
            h_best = h_try
            c5_best = c5_try

    # Final polish on best
    h_best, c5_best = optimize(h_best, c5_best, t0 + budget_s * 0.97, aggressive=False)

    # Ensure constraints
    n_final = len(h_best)
    h_best = project(h_best, n_final * 0.5)
    # Verify
    assert np.all(h_best >= 0) and np.all(h_best <= 1)
    assert abs(h_best.sum() - n_final * 0.5) < 1e-8
    c5_final = c5_of(h_best)
    if c5_final > c5_best + 1e-10:
        c5_final = c5_best  # report the best we actually achieved

    return (h_best, c5_final, n_final)
