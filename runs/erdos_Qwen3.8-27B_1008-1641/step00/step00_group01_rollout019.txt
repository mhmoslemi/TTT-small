import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    e = np.exp(shifted)
    return m + np.log(np.sum(e)) / beta

class ObjFn:
    def __init__(self, beta, dx):
        self.beta = beta
        self.dx = dx
    def __call__(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return smooth_max(corr, self.beta) * self.dx

class CstFn:
    def __init__(self, n, target):
        self.n = n
        self.target = target
        self.ones = np.ones(n)
    def fun(self, h):
        return h.sum() - self.target
    def jac(self, h):
        return self.ones

def project_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        # Second-pass water-filling if still off
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            if mask.sum() > 0:
                h[mask] += diff / mask.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, dx, target, rng, t_limit, max_iter=200000):
    c5 = compute_c5(h)
    delta = 0.03
    t_start = time.time()
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        if (it + 1) % 200 == 0:
            delta = max(0.001, 0.001 + 0.04 * (1.0 - it / max_iter))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] >= 0.005 and h[j] <= 0.995:
            h[i] -= delta
            h[j] += delta
            if h[i] < 0:
                h[i] = 0
                h[j] -= delta
            elif h[j] > 1:
                h[j] = 1
                h[i] += delta
            new_c5 = compute_c5(h)

                h[i] = min(max(h[i], 0), 1)
                h[j] = min(max(h[j], 0), 1)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    per_n_slsqp = max(8.0, time_alloc / len(n_list))

    for n in n_list:
        if time.time() > deadline:

        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        cst = CstFn(n, target)
        constraints = [{'type': 'eq', 'fun': cst.fun, 'jac': cst.jac}]

        # Build initial

            h_cur = np.array(initial_h_values, dtype=float).copy()
        elif n % 2 == 0:

            h_cur[n // 2:] = 1.0
        else:

            h_cur[n // 2:] = 1.0
            h_cur[n // 2] = 0.5
        h_cur = project_feasible(h_cur, target)

        betas = [8, 25, 80, 250, 700]

                               constraints=constraints,
