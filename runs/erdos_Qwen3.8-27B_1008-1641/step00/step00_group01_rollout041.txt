        target = n / 2.0
        dx = 2.0 / n
        def de_obj(x):
            h_r = np.clip(x, 0.0, 1.0)
            pen = 500.0 * (np.sum(h_r) - target) ** 2
            return smooth_obj(h_r, 80.0, dx) + pen
        try:
            de_res = differential_evolution(de_obj, [(0.0, 1.0)] * n, seed=seed,
                                            maxiter=80, popsize=15, tol=1e-10,
                                            polish=False, mutation=(0.4, 1.0),
                                            recombination=0.8, init='latinhypercube')
            h_new = repair(de_res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
                best_h, best_c5 = refine_multi_pass(best_h, n, start_time, budget_s, max_passes=4)
                best_n = len(best_h)
        except Exception:
            pass

    # Phase 3: Try different resolutions
    remaining = budget_s - (time.time() - start_time)
    for n_try in [31, 39, 47, 55, 63, 71, 83, 95, 101]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 20:
            break
        if n_try == best_n:
            continue
        x_old = np.linspace(0.0, 2.0, best_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_try, endpoint=False)
        h_start = np.interp(x_new, x_old, best_h)
        h_start = repair(h_start, n_try)
        h_opt, c5_opt = refine_multi_pass(h_start, n_try, start_time, budget_s, max_passes=4)
        if c5_opt < best_c5:
            best_h, best_c5, best_n = h_opt, c5_opt, n_try

    # Phase 4: Final high-precision polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        best_h, best_c5 = local_optimize(best_h, best_n, maxiter=1500, alpha=200.0)

    final_h = repair(best_h, len(best_h))
    final_c5 = compute_c5(final_h)
    return (final_h, final_c5, len(final_h))
