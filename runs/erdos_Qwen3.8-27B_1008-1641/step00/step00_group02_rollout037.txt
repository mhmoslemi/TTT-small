from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx, corr * dx

def project_to_feasible(h, n):
    """Project h to {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    for _ in range(100):
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if abs(s - target) < 1e-10:
            break
        h = h - (s - target) / n
    return np.clip(h, 0, 1)

def smooth_c5(h, n, tau=50):
    """Smoothed C5 using log-sum-exp"""
    dx = 2.0 / n

    corr = np.clip(corr, -10, 10)
    m = np.max(corr)

def analytic_grad_c5(h, n, tau=200):
    dx = 2.0 / n
    n_lags = 2*n - 1
    # Compute all f_k
    f = np.zeros(n_lags)
    # f[m] = dx * sum_n h[n+k]*(1-h[n]) where k = m-(n-1)
    for m in range(n_lags):
        k = m - (n-1)
        # sum over n where 0<=n<=n-1 and 0<=n+k<=n-1
        n_start = max(0, -k)
        n_end = min(n-1, n-1-k)
        if n_start <= n_end:
            f[m] = dx * np.sum(h[n_start+k:n_end+k+1] * (1-h[n_start:n_end+1]))
    
    # Softmax weights
    f_max = np.max(f)
    p = np.exp(tau * (f - f_max))
    p /= p.sum()
    
    # Gradient: sum_k p_k * df_k/dh[j]
    grad = np.zeros(n)
    for j in range(n):
        for m in range(n_lags):
            k = m - (n-1)
            # df_k/dh[j] = dx * [I(0<=j-k<=n-1)*(1-h[j-k]) - I(0<=j+k<=n-1)*h[j+k]]
            val = 0.0
            if 0 <= j - k <= n-1:
                val += (1 - h[j-k])
            if 0 <= j + k <= n-1:
                val -= h[j+k]
            grad[j] += p[m] * dx * val
    
    return grad, f_max

def project_sum_box(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(h)
    for _ in range(200):
        # Bisection on lambda
        lo, hi = -10, 10
        for _ in range(60):
            mid = (lo + hi) / 2
            h_proj = np.clip(h - mid, 0, 1)
            s = np.sum(h_proj)
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h_proj = np.clip(h - (lo+hi)/2, 0, 1)

            break
        # Fallback: equal adjustment
        diff = target_sum - np.sum(h_proj)
        h_proj += diff / n
        h_proj = np.clip(h_proj, 0, 1)
    return h_proj

import numpy as np
from scipy.optimize import minimize
import time

def project_sum_box(h, target_sum):
    n = len(h)
    # Bisection on lambda: find λ s.t. sum(clip(h-λ, 0, 1)) = target_sum
    lo, hi = -5.0, 5.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    # Tiny correction (should be essentially exact)
    diff = target_sum - np.sum(h_proj)
    if abs(diff) > 1e-12:
        h_proj += diff / n
        h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), corr

def smooth_objective(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    m = np.max(corr)
    return m + np.log(np.mean(np.exp(tau * (corr - m)))) / tau

    nz = 2 * n - 1

            if 0 <= j + k <= n - 1:
                v -= h[j + k]

    if t_left_total > 5:
        n_pert = max(3, int(t_left_total / 2))
        for _ in range(n_pert):
            if time.time() - t_start > budget_s - 3:
                break
            pert = best_h + np.random.randn(n) * 0.05
            t_sub = min(2.0, (budget_s - time.time() - t_start) / 2)
            if t_sub < 0.5:
                break
            h_new, c5_new = optimize_pg(pert, n, 200, 0.01, 5000, t_sub)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    return best_h, best_c5, n
