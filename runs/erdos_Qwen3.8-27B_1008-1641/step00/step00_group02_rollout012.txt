import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            rooms = 1.0 - h
            s = rooms.sum()
            if s < 1e-15:
                break
            h += diff * rooms / s
        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += diff * h / s
        np.clip(h, 0.0, 1.0, out=h)
    return h

def nm_obj(x, n):
    return compute_c5(repair(x, n))

def coord_descent(h, n, rng, max_rounds=500, time_limit=None, t0=None):
    best_c5 = compute_c5(h)
    steps_big = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02]
    steps_small = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for rnd in range(max_rounds):
        if time_limit and time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            all_steps = steps_big if rnd < 200 else steps_small
            for s in all_steps:
                h_try = h.copy()
                h_try[j] += s
                h_try = repair(h_try, n)
                c = compute_c5(h_try)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5
