    if s > target + 1e-13:
        lo, hi = 0.0, h.max()
        for _ in range(70):

            if np.clip(h - mid, 0, 1).sum() > target:

            else:
                hi = mid
        h = np.clip(h - 0.5*(lo+hi), 0, 1)

import numpy as np
import time

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    s = h.sum()
    if s > target + 1e-13:
        lo, hi = 0.0, h.max()
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            if np.clip(h - mid, 0, 1).sum() > target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h - 0.5 * (lo + hi), 0, 1)
    elif s < target - 1e-13:
        lo, hi = 0.0, 1.0 - h.min()
        for _ in range(70):
            mid = 0.5 * (lo + hi)
            if np.clip(h + mid, 0, 1).sum() < target:
                lo = mid
            else:
                hi = mid
        h = np.clip(h + 0.5 * (lo + hi), 0, 1)
    diff = h.sum() - target
    if abs(diff) > 1e-13:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full").max() * (2.0 / n)

def run_cma(h0, rng, t0, deadline):
    n = len(h0)
    h0 = project_feasible(h0, n)
    f_best = compute_c5(h0)
    x_best = h0.copy()

    mean = h0.copy()
    sigma = 0.04
    C = np.eye(n)
    pc = np.zeros(n)
    ps = np.zeros(n)

    lam = max(20, 4 + int(3.0 * np.log(n)))
    mu = lam // 2
    w_arr = np.log(mu / 2.0 + 1.0) - np.arange(mu)
    w_arr /= w_arr.sum()
    mu_w = w_arr.sum()**2 / (w_arr**2).sum()

    c_sig = (1 + 2 * min(1.0, (mu_w - 1) / (n + 1 + 2 * mu_w)) + mu_w / n) / (n + 3 + mu_w)
    d_sig = 1 + 2 * max(0.0, np.sqrt((mu_w - 1) / (n + 1) - 1) - 1) + c_sig
    c_c = (4 + mu_w / n) / (n + 4 + 2 * mu_w / n)
    c_1 = 2.0 / ((n + 1.3)**2 + mu_w)
    c_mu = min(1.0 - c_1, 2.0 * ((mu_w - 2) / (n + 2)**2 + 1) / (n + 2 + mu_w) - c_1)
    chi_n = np.sqrt(n) * (1 - 1.0 / (4 * n) + 1.0 / (21 * n * n))

    no_improve = 0
    for gen in range(200000):
        if time.time() - t0 > deadline:
            break

        ev, evec = np.linalg.eigh(C)
        ev = np.maximum(ev, 1e-14)
        d = np.sqrt(ev)

        Y = rng.standard_normal((lam, n))
        W = (Y @ evec) * d
        X = mean[None, :] + sigma * (W @ evec.T)

        Xf = np.empty((lam, n))
        F = np.empty(lam)
        for i in range(lam):
            Xf[i] = project_feasible(X[i], n)
            F[i] = compute_c5(Xf[i])

        bi = int(np.argmin(F))
        if F[bi] < f_best:
            f_best = F[bi]
            x_best = Xf[bi].copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            sigma *= 0.8
            no_improve = 0

        I = np.argsort(F)
        Im = I[:mu]
        xw = (w_arr[:, None] * Xf[Im]).sum(0)

        v = (xw - mean) / max(sigma, 1e-12)
        Cinv_sqrt_v = evec @ ((evec.T @ v) / d)
        ps = (1 - c_sig) * ps + np.sqrt(c_sig * (2 - c_sig) * mu_w) * Cinv_sqrt_v
        norm_ps = np.linalg.norm(ps)
        denom = np.sqrt(1.0 - (1 - c_sig)**(2 * (gen + 1)))
        h_sig = 1.0 if (denom > 1e-15 and norm_ps / denom < (1.4 + 2.0 / (n + 1)) * chi_n) else 0.0

        norm_v = np.linalg.norm(Cinv_sqrt_v)
        if chi_n > 1e-15:
            sigma *= np.exp(c_sig / d_sig * (norm_v / chi_n - 1))
        sigma = min(max(sigma, 1e-12), 0.5)

        pc = (1 - c_c) * pc + h_sig * c_c * v
        delta_h = (1 - h_sig) * c_c * (2 - c_c)

        D_mat = Xf[Im] - mean[None, :]
        rank_mu = (w_arr[:, None, None] * D_mat[:, :, None] * D_mat[:, None, :]).sum(0)
        C = (1 + c_1 + c_mu - mu_w * c_mu) * C + c_1 * (np.outer(pc, pc) + delta_h * C) + (c_mu / max(sigma, 1e-10)**2) * rank_mu
        C = 0.5 * (C + C.T)

        min_ev_chk = np.linalg.eigvalsh(C).min()
        if min_ev_chk < 1e-10:
            C += (1e-10 - min_ev_chk) * np.eye(n)

        mean = project_feasible(xw, n)

    return x_best, f_best, n
