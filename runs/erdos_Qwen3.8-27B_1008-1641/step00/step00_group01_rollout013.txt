    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    km = int(np.argmax(corr))
    s = km - (n - 1)
    g = np.zeros(n)
    for m in range(n):
        c = 0.0
        if 0 <= m - s < n:
            c += (1.0 - h[m - s])
        if 0 <= m + s < n:
            c += (-h[m + s])
        g[m] = c
    return corr[km] * dx, g * dx

def fix_sum(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        m = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(m.sum())
        if cnt == 0:
            m = h < 1.0
            cnt = int(m.sum())
            if cnt == 0:
                break
        h[m] += d / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _s_fun(x):
    return np.sum(x) - _target

def _s_jac(x):
    return np.ones(_n)

def slsqp_solve(xs, bounds, maxiter=300, ftol=1e-13):
    cons = [{'type': 'eq', 'fun': _s_fun, 'jac': _s_jac}]
    return minimize(c5_grad, xs, method='SLSQP', bounds=bounds,
                    constraints=cons, jac=True,
                    options={'maxiter': maxiter, 'ftol': ftol})

def binary_project(h, target):
    n = len(h)
    sorted_idx = np.argsort(-h)
    result = np.zeros(n)
    assigned = 0
    remaining = target
    for i in range(n):
        idx = sorted_idx[i]
        if remaining >= 1.0 - 1e-9:
            result[idx] = 1.0
            remaining -= 1.0
        elif remaining > 1e-9:
            result[idx] = remaining
            remaining = 0.0
    if remaining < -1e-9:
        for i in range(n):
            idx = sorted_idx[i]
            if result[idx] > 0:
                dec = min(result[idx], -remaining)
                result[idx] -= dec
                remaining += dec
                if abs(remaining) < 1e-9:
                    break
    return fix_sum(result, target)

def run(seed=42, budget_s=1100, **kwargs):
    global _n, _target, _rng
    t0 = time.time()
    _rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.zeros(47)
        h0[:23] = 1.0
        h0[23] = 0.5

    n = len(h0)
    _n = n
    _target = n / 2.0
    target = _target
    bounds = [(0.0, 1.0)] * n

    best_h = fix_sum(h0.copy(), target)
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    # Phase 1: SLSQP from initial
    try:
        r = slsqp_solve(best_h.copy(), bounds, maxiter=500)
        if r.fun < best_c5:
            best_h = fix_sum(r.x, target)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass

    # Phase 2: iterative local search with restarts
    sigma = 0.15
    n_trials = 0
    stagnation = 0

    while elapsed() < budget_s - 30:
        # Generate starting point
        roll = _rng.random()
        if n_trials == 0:
            xs = best_h.copy()
        elif roll < 0.5:
            xs = best_h + _rng.normal(0, sigma, n)
        elif roll < 0.7:
            xs = h0 + _rng.normal(0, sigma * 1.5, n)
        elif roll < 0.85:
            xs = binary_project(best_h, target)
            # slight perturbation of binary
            xs = xs + _rng.normal(0, 0.05, n)
        else:
            # random binary start
            xs = np.zeros(n)
            perm = _rng.permutation(n)
            trial_sum = 0.0
            for i in range(n):
                if trial_sum + 1.0 <= target + 1e-9:
                    xs[perm[i]] = 1.0
                    trial_sum += 1.0
            if target - trial_sum > 0:
                xs[perm[n - 1]] = target - trial_sum
            xs = fix_sum(xs, target)

        xs = fix_sum(xs, target)

        if elapsed() > budget_s - 20:
            break

        # Local optimization
        try:
            r = slsqp_solve(xs, bounds, maxiter=250)
            cand = fix_sum(r.x, target)
            c = compute_c5(cand)
            if c < best_c5 - 1e-12:
                best_h = cand
                best_c5 = c
                stagnation = 0
                sigma = max(0.03, sigma * 0.95)
            else:
                stagnation += 1
                sigma = min(0.6, sigma * 1.08)
        except Exception:
            pass

        # Coordinate descent polish
        if elapsed() < budget_s - 30:
            for _ in range(3):
                improved = False
                for i in range(n):
                    for di in [0.02, -0.02, 0.05, -0.05]:
                        trial = best_h.copy()
                        trial[i] += di
                        # compensate on random other index
                        j = int(_rng.integers(n))
                        if j == i:
                            continue
                        trial[j] -= di
                        trial = fix_sum(trial, target)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-12:
                            best_h = trial
                            best_c5 = c
                            improved = True
                            break
                    if improved:
                        break
                if not improved:
                    break

        n_trials += 1

    # Final ensure
    best_h = fix_sum(best_h, target)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
