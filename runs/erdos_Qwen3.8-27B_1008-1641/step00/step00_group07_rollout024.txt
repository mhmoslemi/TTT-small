def _c5(h):

    h = np.clip(h, 0.0, 1.0)

        w = (1.0 - h) if d > 0 else h.copy()

        if s < 1e-14:
            break
        h = np.clip(h + d * w / s, 0.0, 1.0)
        d = tgt - h.sum()
    return h

    mom = np.zeros(n)
    temp = 0.002
    t0 = time.time()
    stuck = 0

    while time.time() - t0 < budget:
        corr = _corr(h)
        scale = 2.0 / n
        c5_cur = float(np.max(corr)) * scale

        top = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for idx in top:
            grad += _grad_lag(h, idx - (n - 1))
        grad /= float(top_k)

        mom = beta * mom - lr * grad
        # Limit step size to prevent divergence
        step = mom
        if np.max(np.abs(step)) > 0.3:
            step = step * 0.3 / np.max(np.abs(step))

        h_new = _repair(h + step, tgt)
        c5_new = _c5(h_new)

        delta = c5_new - c5_cur
        if delta < 0 or (temp > 1e-12 and rng.random() < np.exp(delta / temp)):
            h = h_new
            if c5_new < best - 1e-13:
                best = c5_new
                h_best = h.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1

        if stuck > 300:
            pert = rng.uniform(0.01, 0.05) * rng.randn(n)
            h = _repair(h_best + pert, tgt)
            mom = np.zeros(n)
            lr = max(lr * 0.6, 0.0005)
            temp = max(temp * 0.5, 1e-10)
            stuck = 0

        temp = max(temp * 0.99997, 1e-12)
        lr = max(lr * 0.99999, 0.0003)

    return h_best, best

def _pair_swap_phase(h0, n, tgt, budget, rng):
    h = h0.copy()
    best = _c5(h)
    h_best = h.copy()
    t0 = time.time()
    trial_size = 0.02
    it = 0

    while time.time() - t0 < budget:
        batch = 2000
        t_batch = time.time()
        while time.time() - t_batch < 0.5 and it < 2000000:
            i, j = rng.choice(n, 2, replace=False)
            dm = rng.uniform(-trial_size, trial_size)
            h_t = h_best.copy()
            h_t[i] = np.clip(h_t[i] + dm, 0.0, 1.0)
            h_t[j] = np.clip(h_t[j] - dm, 0.0, 1.0)
            h_t = _repair(h_t, tgt)
            c = _c5(h_t)
            if c < best - 1e-13:
                best = c
                h_best = h_t
            it += 1

        if it % 20000 < 5000:
            trial_size = max(trial_size * 0.8, 0.001)

    return h_best, best

def _coord_sweep(h0, n, tgt, budget, rng):
    h = h0.copy()
    best = _c5(h)
    h_best = h.copy()
    t0 = time.time()
    deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.003, -0.003]

    while time.time() - t0 < budget:
        improved = False
        for j in range(n):
            if time.time() - t0 > budget:
                break
            orig = h[j]
            best_v = orig
            best_c = best
            for dv in deltas:
                h[j] = np.clip(orig + dv, 0.0, 1.0)
                h_r = _repair(h.copy(), tgt)
                c = _c5(h_r)
                if c < best_c:
                    best_c = c
                    best_v = h_r[j]
            if best_v != orig:
                h[j] = best_v
                h = _repair(h, tgt)
                if best_c < best:
                    best = best_c
                    h_best = h.copy()
                improved = True
            else:
                h[j] = orig
        if not improved:
            break

    return h_best, best

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(80, 0.5)

    n = len(h0)
    tgt = n / 2.0
    h0 = _repair(h0, tgt)

    # Phase 1: Main subgradient (55%)
    b1 = (budget_s - 20) * 0.55
    h1, c1 = _subgrad_phase(h0, n, tgt, b1, rng)

    # Phase 2: Refined subgradient with smaller lr (20%)
    elapsed = time.time() - t0
    b2 = (budget_s - 20) * 0.20
    h2, c2 = _subgrad_phase(h1, n, tgt, b2, rng, top_k=3, lr0=0.001, beta=0.75)
    if c2 < c1:
        h1, c1 = h2, c2

    # Phase 3: Pair swaps (15%)
    remaining = budget_s - 20 - (time.time() - t0)
    b3 = max(remaining * 0.5, 5)
    h3, c3 = _pair_swap_phase(h1, n, tgt, b3, rng)
    if c3 < c1:
        h1, c1 = h3, c3

    # Phase 4: Coordinate sweep for final polish
    remaining = budget_s - 10 - (time.time() - t0)
    if remaining > 5:
        h4, c4 = _coord_sweep(h1, n, tgt, remaining, rng)
        if c4 < c1:
            h1, c1 = h4, c4

    # Final verification and repair
    h1 = _repair(h1, tgt)
    c_final = _c5(h1)
    if c_final < c1:
        c1 = c_final

    return h1, c1, n
