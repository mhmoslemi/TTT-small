    return np.max(corr) * dx

            lo = mid

    corr = np.correlate(h, 1 - h, mode='full')
    m = np.max(corr)
    return (m/tau + np.log(np.sum(np.exp(tau*(corr - m)))))/tau * dx

    return float(np.max(corr) * dx)

    cur_sum = np.sum(h)
    if abs(cur_sum - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def compute_grad_active(h, active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        d = eps if h[i] < 0.9999 else -eps
        h_p = h.copy()
        h_p[i] += d
        corr_p = np.correlate(h_p, 1.0 - h_p, mode='full')
        grad[i] = (corr_p[active] - corr[active]) / d
    return grad, corr

def subgradient_step(h, target, dx, lr):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    active = int(np.argmax(corr))
    # Use near-active set for smoother gradient
    threshold = corr[active] - 0.01
    active_set = np.where(corr >= threshold)[0]
    if len(active_set) == 1:
        grad, _ = compute_grad_active(h, active)
    else:
        grad = np.zeros(n)
        for a in active_set:
            g_a, _ = compute_grad_active(h, int(a))
            grad += g_a
        grad /= len(active_set)
    h_new = h - lr * grad * dx
    return project_feasible(h_new, target)
