import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    c = np.correlate(h, 1 - h, mode="full")
    return float(np.max(c) * dx)

def project_onto_constraint(h, n):
    """Project h onto {x : sum(x)=n/2, 0≤x≤1}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = np.sum(h)
    if np.abs(s - target) < 1e-15:
        return h
    # Scale
    if s > 0:
        h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h)
    if diff > 1e-12:
        # Need to add more; add to entries with most room
        order = np.argsort(1.0 - h)
        for idx in order:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
            if diff <= 1e-12:
                break
    elif diff < -1e-12:
        # Need to remove; remove from entries with most value
        order = np.argsort(h)
        for idx in order:
            rem = min(-diff, h[idx])
            h[idx] -= rem
            diff += rem
            if diff >= -1e-12:
                break
    return h

def subgradient_c5(h, n):
    """Compute subgradient of max_k c[k] w.r.t. h."""
    c = np.correlate(h, 1 - h, mode="full")
    k_max = int(np.argmax(c))
    # k_max is index in the full correl output of length 2n-1
    # lag = k_max - (n-1)  (since full mode lags go from -(n-1) to n-1)
    lag = k_max - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        # Term: [0 ≤ i-lag ≤ n-1]
        if 0 <= i - lag <= n - 1:
            g += 1.0 - h[i - lag]
        # Term: -h[i+lag] if 0 ≤ i+lag ≤ n-1
        if 0 <= i + lag <= n - 1:
            g -= h[i + lag]
        # Term: -h[i-lag] if 0 ≤ i-lag ≤ n-1
        if 0 <= i - lag <= n - 1:
            g -= h[i - lag]
        grad[i] = g
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.ones(47) * (2.0/47)
    
    n = len(h)
    dx = 2.0 / n
    h = project_onto_constraint(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Subgradient descent with adaptive step size
    step = 0.01
    n_iter = 0
    while time_left() > 60 and n_iter < 50000:
        n_iter += 1
        grad = subgradient_c5(h, n)
        # Project gradient onto the sum constraint (remove component in direction of ones)
        grad_mean = np.mean(grad)
        grad_proj = grad - grad_mean
        h_new = h - step * grad_proj
        h_new = project_onto_constraint(h_new, n)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            step = min(step * 1.05, 0.05)
        else:
            h = h_new  # still move, but don't update best
            step = max(step * 0.95, 0.001)
        # Anneal
        step *= (1 - n_iter * 1e-6)
        step = max(step, 0.0001)
    
    # Phase 2: Nelder-Mead polish
    if time_left() > 30:
        def obj(x):
            hx = project_onto_constraint(x, n)
            return compute_c5(hx, dx)
        res = minimize(obj, best_h, method='Nelder-Mead',
                      options={'maxiter': 2000, 'xatol': 1e-12, 'fatol': 1e-14,
                               'adaptive': True})
        h_nm = project_onto_constraint(res.x, n)
        c5_nm = compute_c5(h_nm, dx)
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm.copy()
    
    # Phase 3: Try with refined n_points
    # ... (if time remains)
    
    best_h = project_onto_constraint(best_h, n)
    best_c5 = compute_c5(best_h, dx)
    return best_h, best_c5, n
