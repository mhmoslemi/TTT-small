def _project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(-h)
        remaining = diff
        for idx in order:
            if remaining < 1e-14:
                break
            r = min(h[idx], remaining)
            h[idx] -= r
            remaining -= r
    else:

        remaining = -diff

            if remaining < 1e-14:
                break
            r = min(1.0 - h[idx], remaining)

            remaining -= r

def _eval(h, dx):
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def _subgrad(h, k, n):
    g = np.zeros(n)
    if k >= 0:
        for i in range(n - k):
            g[i] += 1.0 - h[i + k]
            g[i + k] -= h[i]
    else:
        kp = -k
        for i in range(kp, n):
            g[i] += 1.0 - h[i - kp]
            g[i - kp] -= h[i]
    return g

        temp = (c_best - 0.37) * 0.1 + 0.005
        d3 = t0 + 0.95 * budget_s
        h_p = _perturb(h_best, n, target, rng, randint(5, 10) if False else rng.randint(5, 10))
        h_p, cp = _sa_local_search(h_p, dx, rng, d3, temp)
        if cp < c_best:
            h_best, c_best = h_p, cp

    # Phase 4: Final polish from multiple nearby starts
    if time.time() < t0 + budget_s - 5:
        for ps in [2, 3, 5]:
            if time.time() > t0 + budget_s - 5:
                break
            h_p = _perturb(h_best, n, target, rng, ps)
            h_p, cp = _local_search(h_p, dx, rng, t0 + budget_s - 2)
            if cp < c_best:
                h_best, c_best = h_p, cp

    # Final: ensure clean binary, feasible
    h_best = _project_sum(np.round(h_best), target)
    c_final = _eval(h_best, dx)
    return (h_best, c_final, n)
