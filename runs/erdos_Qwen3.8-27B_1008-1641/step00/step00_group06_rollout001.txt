def pair_local_search(h, off, data, pairs, dx, deadline, reserve, max_passes, rng):
    h = np.asarray(h,float).copy()
    target = h.size/2.0
    h = project_box_sum(h,target)
    q = np.correlate(h,1-h,'full')
    c = dx*q.max()
    best_h = h.copy(); best_c=c
    for p in range(max_passes):
        if time.time() > deadline - reserve: break
        order = rng.permutation(len(pairs))
        improved=False
        for pi in order:
            a,b = pairs[pi]
            if abs(h[a]-h[b]) < 1e-14: continue
            ev = evaluate_paired_flip(h,q,a,b,data,off,dx)
            if ev is None: continue
            h2,q2,c2=ev
            if c2 < best_c - 1e-12:
                h=h2; q=q2; best_c=c2; best_h=h.copy(); improved=True; break
        if not improved:
            break
        # optional if c2 direct mismatch? no
    # recompute best q/c exactly
    bq = np.correlate(best_h,1-best_h,'full'); bc=dx*bq.max()
    return best_h, bc, bq
```

def project_box_sum(x, target):
    h = np.array(x, dtype=float, copy=True)
    target = float(target)
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    deficit = target - s
    n = h.size
    for _ in range(n + 5):
        if deficit > 1e-14:
            mask = h < 1.0 - 1e-14
            if not np.any(mask): break
            caps = 1.0 - h[mask]
            inc = deficit / float(mask.sum())
            if inc <= caps.min():
                h[mask] += inc
                break
            else:
                add = float(caps.min())
                if add <= 0.0: break
                h[mask] += add
                deficit -= add * float(mask.sum())
        elif deficit < -1e-14:
            mask = h > 1e-14
            if not np.any(mask): break
            floors = h[mask]
            dec = (-deficit) / float(mask.sum())
            if dec <= floors.min():
                h[mask] -= dec
                break
            else:
                sub = float(floors.min())
                if sub <= 0.0: break
                h[mask] -= sub
                deficit += sub * float(mask.sum())
        else:
            break
    deficit = target - float(np.sum(h))
    if abs(deficit) > 0.0:
        if deficit > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += deficit / float(mask.sum())
            h = np.clip(h,0,1)
    return h
```

import time
import numpy as np
import scipy.optimize as so


def direct_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    q = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(q) * (2.0 / n))


def project_box_sum(x, target):
    h = np.array(x, dtype=float, copy=True)
    if h.size == 0:
        return h
    target = float(target)
    bad = ~np.isfinite(h)
    if np.any(bad):
        h[bad] = 0.0
    h = np.clip(h, 0.0, 1.0)
    deficit = target - float(np.sum(h))
    for _ in range(h.size + 5):
        if deficit > 1e-14:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            caps = 1.0 - h[mask]
            inc = deficit / float(mask.sum())
            if inc <= caps.min():
                h[mask] += inc
                break
            add = float(caps.min())
            if add <= 0.0:
                break
            h[mask] += add
            deficit -= add * float(mask.sum())
        elif deficit < -1e-14:
            mask = h > 1e-14
            if not np.any(mask):
                break
            floors = h[mask]
            dec = (-deficit) / float(mask.sum())
            if dec <= floors.min():
                h[mask] -= dec
                break
            sub = float(floors.min())
            if sub <= 0.0:
                break
            h[mask] -= sub
            deficit += sub * float(mask.sum())
        else:
            break
    deficit = target - float(np.sum(h))
    if abs(deficit) > 0.0:
        if deficit > 0.0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            h[mask] += deficit / float(mask.sum())
            h = np.clip(h, 0.0, 1.0)
    deficit = target - float(np.sum(h))
    if abs(deficit) > 1e-15:
        interior = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if interior.size:
            h[int(interior[interior.size // 2])] += deficit
            h = np.clip(h, 0.0, 1.0)
    return h


def grad_from_weights(h, p):
    n = h.size
    off = n - 1
    g = np.zeros(n)
    if n == 0:
        return g
    g += p[off] * (1.0 - 2.0 * h)
    for t in range(1, n):
        pp = p[off + t]
        pn = p[off - t]
        if pp + pn <= 0.0:
            continue
        a = h[:-t]
        b = h[t:]
        ml = n - t
        if pp != 0.0:
            g[:ml] += -pp * b
            g[t:] += pp * (1.0 - a)
        if pn != 0.0:
            g[:ml] += pn * (1.0 - b)
            g[t:] += -pn * a
    return g


def smooth_value(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    q = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(q))
    dx = 2.0 / n
    if tau <= 1e-12:
        return m * dx
    z = (q - m) / tau
    e = np.exp(z)
    s = float(np.sum(e))
    return (m + tau * np.log(s)) * dx


def smooth_value_grad(h, tau):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0, np.zeros(0)
    q = np.correlate(h, 1.0 - h, mode="full")
    off = n - 1
    m = float(np.max(q))
    dx = 2.0 / n
    if tau <= 1e-12:
        mask = q >= (m - 1e-12)
        p = np.zeros(q.shape, dtype=float)
        p[mask] = 1.0 / float(mask.sum())
        val = m * dx
    else:
        z = (q - m) / tau
        e = np.exp(z)
        s = float(np.sum(e))
        val = (m + tau * np.log(s)) * dx
        p = e / s
    grad = grad_from_weights(h, p) * dx
    return val, grad


def smooth_func(x, tau):
    return smooth_value(x, tau)


def smooth_jac(x, tau):
    return smooth_value_grad(x, tau)[1]


def sum_eq(x):
    return float(np.sum(x) - 0.5 * x.size)


def project_gradient_search(h0, tau, max_iters, deadline, reserve, rng):
    h = project_box_sum(h0, 0.5 * h0.size)
    if not np.all(np.isfinite(h)):
        return h0.copy()
    step_h = 0.1
    for _ in range(max_iters):
        if time.time() > deadline - reserve:
            break
        f, grad = smooth_value_grad(h, tau)
        if not np.isfinite(f) or not np.all(np.isfinite(grad)):
            break
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-10:
            break
        alpha = step_h / gnorm
        ok = False
        for _ in range(12):
            y = project_box_sum(h - alpha * grad, 0.5 * h.size)
            if not np.all(np.isfinite(y)):
                alpha *= 0.5
                continue
            fy = smooth_value(y, tau)
            if np.isfinite(fy) and fy + 1e-12 < f:
                h = y
                f = fy
                ok = True
                break
            alpha *= 0.5
        if not ok:
            step_h *= 0.6
            if step_h < 1e-4:
                break
    return h


def slsqp_refine(h0, tau, maxiter, deadline, reserve):
    if time.time() > deadline - reserve:
        return h0.copy()
    h = project_box_sum(h0, 0.5 * h0.size)
    n = h.size
    if n == 0:
        return h
    try:
        bounds = [(0.0, 1.0)] * n
        constraints = [{'type': 'eq', 'fun': sum_eq}]
        res = so.minimize(
            smooth_func,
            h,
            args=(tau,),
            method='SLSQP',
            jac=smooth_jac,
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
        )
        if res.x is not None:
            h2 = project_box_sum(res.x, 0.5 * n)
            if np.all(np.isfinite(h2)):
                return h2
    except Exception:
        pass
    return h


def build_swap_indices(n):
    off = n - 1
    data = []
    empty = np.empty(0, dtype=int)
    for p in range(n):
        if p > 0:
            dist = p - np.arange(p, dtype=int)
            lp = off + dist
            ln = off - dist
        else:
            lp = empty.copy()
            ln = empty.copy()
        r = n - p - 1
        if r > 0:
            dist = 1 + np.arange(r, dtype=int)
            rp = off + dist
            rn = off - dist
        else:
            rp = empty.copy()
            rn = empty.copy()
        data.append((lp, ln, rp, rn))
    return off, data


def build_pairs(n):
    return [(i, j) for i in range(n) for j in range(i + 1, n)]


def apply_single_to_q(q, h, p, d, data, off):
    lp, ln, rp, rn = data[p]
    if lp.size:
        v = h[:p]
        q[lp] += d * (1.0 - v)
        q[ln] -= d * v
    if rp.size:
        v = h[p + 1:]
        q[rp] -= d * v
        q[rn] += d * (1.0 - v)
    q[off] += d * (1.0 - 2.0 * h[p] - d)


def evaluate_paired_flip(h, q, a, b, data, off, dx):
    A = float(h[a])
    B = float(h[b])
    if abs(A - B) < 1e-15:
        return None
    d = B - A
    h2 = h.copy()
    q2 = q.copy()
    apply_single_to_q(q2, h2, a, d, data, off)
    h2[a] = B
    apply_single_to_q(q2, h2, b, -d, data, off)
    h2[b] = A
    c = float(np.max(q2)) * dx
    return h2, q2, c


def pair_local_search(h0, off, data, pairs, dx, deadline, reserve, max_passes, rng):
    h = project_box_sum(h0, 0.5 * h0.size)
    if not np.all(np.isfinite(h)):
        q = np.zeros(2 * h0.size - 1)
        return h0.copy(), direct_c5(h0), q
    n = h.size
    q = np.correlate(h, 1.0 - h, mode="full")
    c = float(np.max(q)) * dx
    best_h = h.copy()
    best_c = c
    accepted = 0
    for _ in range(max_passes):
        if time.time() > deadline - reserve:
            break
        order = rng.permutation(len(pairs))
        improved = False
        for idx in order:
            a, b = pairs[int(idx)]
            if abs(h[a] - h[b]) < 1e-14:
                continue
            ev = evaluate_paired_flip(h, q, a, b, data, off, dx)
            if ev is None:
                continue
            h2, q2, c2 = ev
            if c2 < best_c - 1e-12:
                h = h2
                q = q2
                best_c = c2
                best_h = h.copy()
                improved = True
                accepted += 1
                if accepted % 25 == 0:
                    q = np.correlate(h, 1.0 - h, mode="full")
                break
        if not improved:
            break
    bq = np.correlate(best_h, 1.0 - best_h, mode="full")
    bc = float(np.max(bq)) * dx
    return best_h, bc, bq


def random_perturb(h, m, rng, target, transfer_prob):
    hp = h.copy()
    n = hp.size
    if n == 0:
        return hp
    for _ in range(m):
        a = int(rng.randint(0, n))
        b = int(rng.randint(0, n))
        if a == b:
            continue
        A = float(hp[a])
        B = float(hp[b])
        if abs(A - B) <= 1e-14:
            continue
        if rng.rand() < transfer_prob:
            lo = max(-A, B - 1.0)
            hi = min(1.0 - A, B)
            if hi > lo + 1e-14:
                d = rng.uniform(lo, hi)
                hp[a] = A + d
                hp[b] = B - d
        else:
            hp[a] = B
            hp[b] = A
    return project_box_sum(hp, target)


def binarize_by_rank(h, target):
    h = np.asarray(h, dtype=float)
    n = h.size
    out = np.zeros(n)
    if n == 0:
        return out
    counts1 = int(np.floor(target + 1e-12))
    half = int(round((target - counts1) * 2.0))
    if counts1 > n:
        counts1 = n
    order = np.argsort(-h, kind='mergesort')
    if counts1 > 0:
        out[order[:counts1]] = 1.0
    if half > 0:
        out[order[counts1:counts1 + half]] = 0.5
    return out


def random_binary(n, target, rng):
    out = np.zeros(n)
    if n == 0:
        return out
    counts1 = int(np.floor(target + 1e-12))
    half = int(round((target - counts1) * 2.0))
    idx = np.arange(n, dtype=int)
    rng.shuffle(idx)
    if counts1 > 0:
        out[idx[:counts1]] = 1.0
    if half > 0:
        out[idx[counts1:counts1 + half]] = 0.5
    return out


def make_initial(n, rng=None):
    if n <= 0:
        return np.array([], dtype=float)
    x = np.linspace(0.0, 2.0, int(n), endpoint=False)
    h = 0.5 + 0.45 * np.sin(np.pi * x)
    if rng is not None:
        h += 0.04 * np.sin(2.7 * x + 0.35 * rng.rand() * 10.0)
    return project_box_sum(h, 0.5 * n)


def resample_linear(h, m):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0 or m <= 0:
        return np.array([], dtype=float)
    if n == m:
        return h.copy()
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, int(m))
    return np.interp(x_new, x_old, h)


def finalize_h(h):

    return h

    if time.time() <= deadline - max(8.0, 0.04 * budget):
        for _ in range(4):
            if time.time() > deadline - max(8.0, 0.04 * budget):
                break
            hr = random_binary(n, target, rng)
            hr = finalize_h(hr)
            c0 = direct_c5(hr)
            if np.isfinite(c0) and c0 < best_opt_c + 0.02:
                hloc, cloc, _ = pair_local_search(
                    hr, off, data, pairs, dx, deadline,
                    max(5.0, 0.03 * budget), 35, rng
                )
                if np.isfinite(cloc):
                    if cloc < best_opt_c:
                        best_opt_c = cloc
                        best_opt = hloc.copy()
                    if cloc < best_c:
                        best_c = cloc
                        h_best = hloc.copy()

    if best_opt is not None and best_opt_c < best_c:
        best_c = best_opt_c
        h_best = best_opt.copy()

    h_best = finalize_h(h_best)
    final_c = direct_c5(h_best)
    if not np.isfinite(final_c):
        h_best = project_box_sum(h_orig, 0.5 * h_orig.size) if h_orig.size else make_initial(95, rng)
        final_c = direct_c5(h_best)

    return np.asarray(h_best, dtype=float), float(final_c), int(h_best.size)
