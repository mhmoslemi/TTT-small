import numpy as np
import time
from scipy.optimize import minimize as sp_min

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        for i in range(n):
            if abs(diff) < 1e-14:
                break
            if diff > 0 and h[i] < 1.0:
                a = min(diff, 1.0 - h[i]); h[i] += a; diff -= a
            elif diff < 0 and h[i] > 0.0:
                a = min(-diff, h[i]); h[i] -= a; diff += a
    return h

def lse_max(vals, T):
    v = vals * T
    m = v.max()
    if m > 700:
        m = 700.0
    return float((m + np.log(np.exp(v - m).sum())) / T)

def slsqp_refine(h0, n, T_list, time_limit):
    cur = project(h0, n).copy()
    best_c = evaluate_c5(cur, n)
    dx_val = 2.0 / n
    for T in T_list:
        if time.time() > time_limit:
            break
        def f_obj(x):
            x = np.clip(x, 0.0, 1.0)
            c = np.correlate(x, 1.0 - x, mode='full') * dx_val
            return lse_max(c, T)
        def g_obj(x):
            g = np.zeros(n)
            eps = 1e-8
            f0 = f_obj(x)
            for i in range(n):
                xp = x.copy(); xp[i] += eps
                xm = x.copy(); xm[i] -= eps
                g[i] = (f_obj(xp) - f_obj(xm)) / (2.0 * eps)
            return g
        try:
            res = sp_min(f_obj, cur, method='SLSQP', jac=g_obj,
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq',
                                       'fun': lambda x: x.sum() - n / 2.0,
                                       'jac': lambda x: np.ones(n)}],
                         maxiter=300, tol=1e-12)
            cur = project(res.x, n)
            c = evaluate_c5(cur, n)
            if c < best_c:
                best_c = c
        except Exception:
            pass
    return cur, best_c

def es_optimize(h0, n, time_limit, rng):
    best_h = h0.copy()
    best_f = evaluate_c5(best_h, n)
    lam_n, mu_n = 18, 9
    sigma = 0.25
    mean = best_h.copy()
    stall = 0
    gen = 0
    while time.time() < time_limit and sigma > 1e-7:
        off = np.array([project(mean + sigma * rng.standard_normal(n), n)
                        for _ in range(lam_n)])
        fits = np.array([evaluate_c5(x, n) for x in off])
        srt = np.argsort(fits)
        if fits[srt[0]] < best_f:
            best_f = float(fits[srt[0]])
            best_h = off[srt[0]].copy()
            stall = 0
        else:
            stall += 1
        mean = off[srt[:mu_n]].mean(axis=0)
        if stall > 25:
            sigma *= 0.85
            stall = 0
        if stall > 150:
            mean = best_h + 0.15 * rng.standard_normal(n)
            sigma = 0.2
            stall = 0
        gen += 1
        if gen % 500 == 0 and time.time() > time_limit:
            break
    return best_h, best_f

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.default_rng(seed)

        h_r = project(best_h + pert, n)

            if c_t2 < best_c5:
                best_c5, best_h, n = c_t2, h_t2, n_try

    # Final validation
    best_h = project(best_h, n)
    best_c5 = evaluate_c5(best_h, n)
    assert n == len(best_h)
    assert abs(best_h.sum() - n / 2.0) < 1e-8
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    return (best_h, best_c5, n)
