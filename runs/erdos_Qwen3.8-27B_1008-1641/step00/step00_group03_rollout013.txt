def compute_c5(h, dx):
    """Compute C5 = max_k correlate(h, 1-h, full) * dx"""
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx, corr * dx

def compute_gradient(h, dx):
    """Compute gradient of smoothed max or max of g_k"""
    n = len(h)
    grad = np.zeros(n)
    corr = np.correlate(h, 1 - h, mode="full")
    vals = corr * dx
    k_max = np.argmax(vals)
    s = k_max - (n - 1)
    # g_s(h) = dx * sum_{i valid} h[i]*(1-h[i+s])
    # dg_s/dh[i] = dx * [(1-h[i+s])*1_{i+s valid} - h[i-s]*1_{i-s valid}]
    if s >= 0:
        # h[i+s] valid for i in [0, n-1-s]
        for i in range(n - s):
            grad[i] += dx * (1 - h[i + s])
        # h[i-s] valid for i in [s, n-1]
        for i in range(s, n):
            grad[i] -= dx * h[i - s]
    else:
        s_abs = -s
        # h[i+s] = h[i-s_abs] valid for i in [s_abs, n-1]
        for i in range(s_abs, n):
            grad[i] += dx * (1 - h[i - s_abs])
        # h[i-s] = h[i+s_abs] valid for i in [0, n-1-s_abs]
        for i in range(n - s_abs):
            grad[i] -= dx * h[i + s_abs]
    return grad

def project_onto_feasible(h, n):
    """Project h onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0

    # Adjust to meet sum constraint
    diff = target - np.sum(h)
    # Distribute the adjustment to elements that can absorb it
    if diff > 0:
        mask = h < 1 - 1e-12
        count = np.sum(mask)
        if count > 0:
            step = min(diff / count, 1 - h[mask])
            # This is approximate; for exact projection we'd need more work
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
    elif diff < 0:
        mask = h > 1e-12
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0, 1)
    # Final correction for floating point
    h = np.clip(h, 0, 1)
    correction = target - np.sum(h)
    if abs(correction) > 1e-10:
        # Distribute evenly among elements not at boundary
        mask = (h > 1e-8) & (h < 1 - 1e-8)
        if np.sum(mask) > 0:
            h[mask] += correction / np.sum(mask)
            h = np.clip(h, 0, 1)
    return h

    v = np.max(val)
    return v + np.log(np.sum(np.exp(val - v))) / T

import numpy as np
from scipy.optimize import minimize
import time

def correlate_vals(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_c5(h, dx):
    return float(np.max(correlate_vals(h, dx)))

def grad_one_shift(h, dx, k, n):
    s = k - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += dx * (1.0 - h[s:])
        g[s:] -= dx * h[:n - s]
    else:
        sa = -s
        g[sa:] += dx * (1.0 - h[:n - sa])
        g[:n - sa] -= dx * h[sa:]
    return g

def grad_smooth_max(h, dx, T, n):
    vals = correlate_vals(h, dx)
    Tv = T * vals
    m = Tv.max()
    ev = np.exp(Tv - m)
    w = ev / ev.sum()
    g = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for k in active:
        g += w[k] * grad_one_shift(h, dx, k, n)
    return g

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
