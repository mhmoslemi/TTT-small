import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_descent(h, n, rng, max_sweeps=30, grid_size=51):
    best_val = evaluate_c5(h, n)
    for sweep in range(max_sweeps):
        improved = False
        order = list(range(n))
        rng.shuffle(order)
        for j in order:
            old = h[j]
            best_t = old
            candidates = np.unique(np.concatenate([np.linspace(0, 1, grid_size), [old, old - 0.05, old + 0.05]]))
            for t in candidates:
                t = float(np.clip(t, 0.0, 1.0))
                trial = h.copy()
                trial[j] = t
                trial = project_feasible(trial, n)
                v = evaluate_c5(trial, n)
                if v < best_val - 1e-13:
                    best_val = v
                    best_t = t
                    improved = True
            h[j] = best_t
            h = project_feasible(h, n)
            best_val = evaluate_c5(h, n)
        if not improved:
            break
    return h, best_val

def nm_objective(x, n):
    xc = np.clip(x, 0.0, 1.0)
    pen = 200.0 * (np.sum(xc) - n / 2.0) ** 2
    return evaluate_c5(xc, n) + pen

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        best_h = project_feasible(np.array(initial_h_values, dtype=float), n)
    else:
        n = 45
        best_h = np.full(n, 0.5)

    best_score = evaluate_c5(best_h, n)

    def remaining():
        return budget_s - (time.time() - t0) - 3.0

    # Phase 1: Coordinate descent from initial (most important)
    h_work = best_h.copy()
    while remaining() > 80:
        h_work, sc = coord_descent(h_work, n, rng, max_sweeps=15, grid_size=81)
        if sc < best_score:
            best_h = h_work.copy()
            best_score = sc
        else:
            h_work = best_h + rng.normal(0, 0.03, n)
            h_work = project_feasible(h_work, n)

    # Phase 2: Nelder-Mead fine-tuning
    if remaining() > 40:
        res = minimize(nm_objective, best_h.copy(), method='Nelder-Mead',
                       options={'maxiter': 5000, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
        cand = project_feasible(res.x, n)
        sc = evaluate_c5(cand, n)
        if sc < best_score:
            best_h = cand
            best_score = sc
        if remaining() > 30:
            h_work = best_h.copy()
            h_work, sc2 = coord_descent(h_work, n, rng, max_sweeps=20, grid_size=101)
            sc2 = evaluate_c5(h_work, n)
            if sc2 < best_score:
                best_h = h_work
                best_score = sc2

    # Phase 3: Random restarts from structured patterns
    if remaining() > 40:
        attempts = 0
        while remaining() > 15 and attempts < 8:
            attempts += 1
            if rng.rand() < 0.5:
                h_try = (rng.rand(n) > 0.5).astype(float)
            else:
                h_try = np.full(n, 0.5)
                mask = rng.rand(n) > 0.5
                h_try[mask] = 1.0
                h_try[~mask] = 0.0
            h_try = project_feasible(h_try, n)
            h_try, sc = coord_descent(h_try, n, rng, max_sweeps=20, grid_size=51)
            sc = evaluate_c5(h_try, n)
            if sc < best_score:
                best_h = h_try.copy()
                best_score = sc

    # Phase 4: Final polish with very fine grid
    if remaining() > 10:
        h_work = best_h.copy()
        h_work, sc = coord_descent(h_work, n, rng, max_sweeps=30, grid_size=121)

        if sc < best_score:
            best_h = h_work
            best_score = sc

    best_h = project_feasible(best_h, n)
    best_score = evaluate_c5(best_h, n)
    return best_h, best_score, n
