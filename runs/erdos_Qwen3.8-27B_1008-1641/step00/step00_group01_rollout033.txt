```python
import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def time_left(t0, budget):
    return budget - (time.time() - t0)


def project_h(h, target):
    h = h.astype(float).copy()
    for _ in range(1000):
        h = np.clip(h, 0.0, 1.0)
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = float(free.sum())
        if cnt == 0.0:
            h += d / len(h)
        else:
            h[free] += d / cnt
    h = np.clip(h, 0.0, 1.0)
    return h


def soft_obj(h, tau, target, pen_w):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = c.max()
    scaled = np.clip((c - m) / tau, -60.0, 60.0)
    sm = m + tau * np.log(np.sum(np.exp(scaled)))
    pen = pen_w * (h.sum() - target) ** 2
    return sm + pen


def slsqp_optimize(h, n, target, tau, pen_w, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(soft_obj, h, args=(tau, target, pen_w),
                   method='SLSQP', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_h(res.x, target)


def sa_optimize(h_init, n, target, t0, budget, max_iters):
    h = h_init.copy()
    c5_cur = compute_c5(h)
    c5_best_local = c5_cur
    h_best_local = h.copy()
    T = 0.002
    for i in range(max_iters):
        if i % 1000 == 0 and time_left(t0, budget) < 60:
            break
        idx = np.random.randint(n)
        h_new = h.copy()
        h_new[idx] += np.random.randn() * 0.07
        h_new = project_h(h_new, target)
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_cur
        if delta < 0 or np.random.random() < np.exp(-max(delta / T, -500)):
            h, c5_cur = h_new, c5_new
            if c5_cur < c5_best_local:
                h_best_local = h.copy()
                c5_best_local = c5_cur
        T = max(T * 0.999995, 0.00003)
    return h_best_local, c5_best_local


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=float)
    else:
        h_best = np.ones(50) * 0.5

    n = len(h_best)
    target = n / 2.0
    h_best = project_h(h_best, target)
    c5_best = compute_c5(h_best)

    # Phase 1: SLSQP with annealed tau
    for tau in [0.05, 0.02, 0.005, 0.001, 0.0003]:
        if time_left(t0, budget_s) < 30:
            break
        h = slsqp_optimize(h_best, n, target, tau, 80.0, 800)
        c5 = compute_c5(h)
        if c5 < c5_best:
            h_best, c5_best = h, c5

    # Phase 2: Simulated annealing
    h_sa, c5_sa = sa_optimize(h_best, n, target, t0, budget_s, 200000)
    if c5_sa < c5_best:
        h_best, c5_best = h_sa, c5_sa

    if time_left(t0, budget_s) < 60:
        h_final = project_h(h_best, target)
        return (h_final, compute_c5(h_final), n)

    # Phase 3: Multi-start with diverse initializations
    for r in range(60):
        if time_left(t0, budget_s) < 30:

        mode = r % 3
        if mode == 0:

        elif mode == 1:
            h_r = np.random.rand(n) * 0.6 + 0.2
        else:

        h_r = project_h(h_r, target)
        h_s = slsqp_optimize(h_r, n, target, 0.001, 200.0, 400)
        c5_s = compute_c5(h_s)
        if c5_s < c5_best:
            h_best, c5_best = h_s, c5_s

    if time_left(t0, budget_s) < 45:
        h_final = project_h(h_best, target)
        return (h_final, compute_c5(h_final), n)

    # Phase 4: Interpolate to larger n and optimize
    for n_new in [60, 80, 120]:
        if time_left(t0, budget_s) < 45:

        x_o = np.linspace(0, 2.0, n, endpoint=False)

    # Final polish

    return (h_best, c5_best, n)
```
