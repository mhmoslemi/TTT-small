```
def compute_lag_data(h, grad=True):
    N=h.size; L=2*N-1
    c=np.empty(L)
    G=np.zeros((N,L)) if grad else None
    for lag in range(-(N-1),N):
       idx=lag+N-1
       a0=max(0,-lag); a1=min(N,N-lag)
       a=np.arange(a0,a1); b=a+lag
       ha=h[a]; hb=h[b]
       c[idx]=np.dot(ha,1-hb)
       if grad:
          G[a,idx]+=1-hb; G[b,idx]-=ha
    return c,G,float(c.max())
```

```python
import time
import math
import numpy as np
from scipy import optimize as sopt

_SMOOTH_TAU = 40.0


def c5_unscaled(h):
    h = np.asarray(h, dtype=float)
    if h.size < 2:
        return float(np.dot(h, 1.0 - h))
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def c5_scaled(h):
    return c5_unscaled(h) * (2.0 / h.size)


def project_box_mean(v, target):
    v = np.clip(np.asarray(v, dtype=float), 0.0, 1.0)
    if v.size == 0:
        return v
    if abs(float(v.sum()) - target) < 1e-14:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.clip(v - mid, 0.0, 1.0).sum())
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)


def enforce_exact_sum(h, target):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    if h.size == 0:
        return h
    h = project_box_mean(h, target)
    for _ in range(5):
        diff = target - float(h.sum())
        if diff == 0.0:
            break
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    if h.size > 1:
        val = target - float(np.sum(h[:-1]))
        if 0.0 <= val <= 1.0:
            h[-1] = val
            diff = target - float(h.sum())
            if diff != 0.0:
                idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
                h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h


def default_start(n, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    s = np.zeros(n)
    for f in (1, 2, 3, 5, 7):
        s += rng.uniform(0.3, 1.0) * np.sin(2.0 * math.pi * f * x + rng.uniform(0.0, 2.0 * math.pi))
    h = 0.5 + 0.5 * np.tanh(2.5 * s)
    return enforce_exact_sum(h, 0.5 * n)


def resize_h(h, n):
    m = h.size
    if m == n:
        return enforce_exact_sum(h, 0.5 * n)
    x = np.linspace(0.0, 1.0, m, endpoint=False)
    y = np.linspace(0.0, 1.0, n, endpoint=False)
    hh = np.interp(y, x, np.r_[h, h[0]])
    return enforce_exact_sum(hh, 0.5 * n)


def compute_lag_data(h, grad=True):
    N = h.size
    L = 2 * N - 1
    c = np.empty(L, dtype=float)
    G = np.zeros((N, L), dtype=float) if grad else None
    for lag in range(-(N - 1), N):
        idx = lag + N - 1
        a0 = max(0, -lag)
        a1 = min(N, N - lag)
        a = np.arange(a0, a1)
        b = a + lag
        ha = h[a]
        hb = h[b]
        c[idx] = np.dot(ha, 1.0 - hb)
        if grad:
            G[a, idx] += 1.0 - hb
            G[b, idx] -= ha
    return c, G, float(c.max())


def line_min_subgrad(c, slopes, lo, hi, iters=28):
    if hi <= lo:
        return 0.5 * (lo + hi)
    if hi - lo < 1e-14:
        return 0.0
    for _ in range(iters):
        mid = 0.5 * (lo + hi)
        y = c + slopes * mid
        m = float(y.max())
        active = y >= m - 1e-11
        smax = float(slopes[active].max())
        smin = float(slopes[active].min())
        if smax < 0.0:
            lo = mid
        elif smin > 0.0:
            hi = mid
        else:
            return mid
        if hi - lo < 1e-13:
            return 0.5 * (lo + hi)
    return 0.5 * (lo + hi)


def local_pair_ref(h, max_seconds, rng):
    n = h.size
    target = 0.5 * n
    h = enforce_exact_sum(h, target).copy()
    best = c5_unscaled(h)
    tol = 1e-8 / (2.0 / n)
    t0 = time.time()
    c, G, stale = compute_lag_data(h, True)
    acc = 0
    while max_seconds > 0.0:
        if time.time() - t0 > max_seconds:
            break
        order = np.arange(n)
        rng.shuffle(order)
        changed = False
        for ii in range(n - 1):
            i = int(order[ii])
            for jj in range(ii + 1, n):
                if time.time() - t0 > max_seconds:
                    break
                j = int(order[jj])
                slopes = G[i] - G[j]
                lo = -h[i]
                hi = 1.0 - h[j]
                if hi - lo < 1e-13:
                    continue
                d = line_min_subgrad(c, slopes, lo, hi)
                if abs(d) < 1e-13:
                    continue
                if float((c + slopes * d).max()) + tol >= stale:
                    continue
                cand = h.copy()
                cand[i] = float(np.clip(h[i] + d, 0.0, 1.0))
                cand[j] = float(np.clip(h[j] - d, 0.0, 1.0))
                tv = c5_unscaled(cand)
                if tv + tol < best:
                    cand = enforce_exact_sum(cand, target)
                    tv = c5_unscaled(cand)
                    if tv + tol < best:
                        h = cand
                        best = tv
                        changed = True
                        acc += 1
                        if acc >= 5:
                            c, G, stale = compute_lag_data(h, True)
                            best = c5_unscaled(h)
                            acc = 0
            if time.time() - t0 > max_seconds:
                break
        if not changed:
            break
        c, G, stale = compute_lag_data(h, True)
        best = c5_unscaled(h)
        acc = 0
    return enforce_exact_sum(h, target), c5_scaled(h)


def sum_eq_fun(h):
    return float(h.sum() - 0.5 * h.size)


def sum_eq_jac(h):
    return np.ones_like(h)


def smooth_obj(h):
    c, _, _ = compute_lag_data(h, False)
    tau = _SMOOTH_TAU
    m = float(c.max())
    z = np.exp(tau * (c - m))
    return float((m + math.log(float(z.sum())) / tau) * (2.0 / h.size))


def smooth_jac(h):
    c, G, mc = compute_lag_data(h, True)
    tau = _SMOOTH_TAU
    z = np.exp(tau * (c - mc))
    w = z / float(z.sum())
    return (w @ G) * (2.0 / h.size)


def slsqp_refine(h, max_seconds, rng):
    global _SMOOTH_TAU
    n = h.size
    if max_seconds < 4.0 or n > 80:
        return enforce_exact_sum(h, 0.5 * n), c5_scaled(h)
    h = enforce_exact_sum(h, 0.5 * n)
    best = c5_scaled(h)
    t0 = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": sum_eq_fun, "jac": sum_eq_jac}]
    maxiter = int(min(50, max(12, int(max_seconds))))
    for tau in (25.0, 70.0, 160.0):
        if time.time() - t0 > max_seconds:
            break
        _SMOOTH_TAU = tau
        try:
            res = sopt.minimize(
                smooth_obj,
                h,
                method="SLSQP",
                jac=smooth_jac,
                bounds=bounds,
                constraints=cons,
                options={"maxiter": maxiter, "ftol": 1e-12},
            )
            if res.x is not None:
                cand = enforce_exact_sum(res.x, 0.5 * n)
                v = c5_scaled(cand)
                if v + 1e-12 < best:
                    h = cand
                    best = v
        except Exception:
            pass
    _SMOOTH_TAU = 40.0
    return enforce_exact_sum(h, 0.5 * h.size), c5_scaled(h)


def raw_h_from_t(t):
    n = t.size
    h = 0.5 + t
    h -= (float(h.sum()) - 0.5 * n) / n
    np.clip(h, 0.0, 1.0, out=h)
    diff = float(h.sum()) - 0.5 * n
    if abs(diff) > 1e-5:
        h -= diff / n
        np.clip(h, 0.0, 1.0, out=h)
    if abs(float(h.sum()) - 0.5 * n) > 1e-4:
        h = project_box_mean(h, 0.5 * n)
    return h


def de_objective(t):
    h = raw_h_from_t(t)
    return c5_unscaled(h) * (2.0 / t.size)


def de_optimize(dim, max_seconds, rng, init_t=None):
    if dim < 2 or max_seconds < 2.0:
        return np.zeros(dim) if init_t is None else init_t.copy()
    pop_size = int(min(140, max(40, 3 * dim)))
    pop = rng.uniform(-0.5, 0.5, size=(pop_size, dim))
    if init_t is not None:
        pop[0] = np.clip(init_t, -0.5, 0.5)
        if pop_size > 1:
            pop[1] = np.clip(-init_t, -0.5, 0.5)
    vals = np.empty(pop_size)
    for i in range(pop_size):
        vals[i] = de_objective(pop[i])
    best_i = int(vals.argmin())
    t0 = time.time()
    no_imp = 0
    it = 0
    F = 0.8
    CR = 0.8
    while time.time() - t0 < max_seconds:
        it += 1
        imp = False
        for i in range(pop_size):
            if time.time() - t0 > max_seconds:
                break
            a = int(rng.integers(0, pop_size))
            while a == i:
                a = int(rng.integers(0, pop_size))
            b = int(rng.integers(0, pop_size))
            while b == i or b == a:
                b = int(rng.integers(0, pop_size))
            d = int(rng.integers(0, pop_size))
            while d == i or d == a or d == b:
                d = int(rng.integers(0, pop_size))
            mut = pop[a] + F * (pop[b] - pop[d])
            tri = pop[i].copy()
            mask = rng.random(dim) < CR
            tri[mask] = mut[mask]
            np.clip(tri, -0.5, 0.5, out=tri)
            v = de_objective(tri)
            if v < vals[i]:
                pop[i] = tri
                vals[i] = v
                imp = True
                if v < vals[best_i]:
                    best_i = i
        if not imp:
            no_imp += 1
            if no_imp > 12:
                break
        else:
            no_imp = 0
        if it % 20 == 0:
            for idx in range(pop_size):
                if vals[idx] > vals[best_i] + 0.005 and rng.random() < 0.4:
                    new = pop[best_i] + rng.normal(0.0, 0.1, size=dim)
                    np.clip(new, -0.5, 0.5, out=new)
                    v = de_objective(new)
                    pop[idx] = new
                    vals[idx] = v
                    if v < vals[best_i]:
                        best_i = idx
    return pop[best_i].copy()


def refine_candidate(h, max_seconds, rng):
    h = enforce_exact_sum(h, 0.5 * h.size)
    best = c5_scaled(h)
    t0 = time.time()
    n = h.size

    sl = min(50.0, 0.2 * max_seconds)
    if n <= 70 and sl > 4.0:
        h2, b2 = slsqp_refine(h, sl, rng)
        if b2 + 1e-12 < best:
            h, best = h2, b2

    rem = max(0.0, max_seconds - (time.time() - t0))
    la = min(160.0, 0.5 * rem) if rem > 10.0 else rem
    if la > 1.0:
        h2, b2 = local_pair_ref(h, la, rng)
        if b2 + 1e-12 < best:
            h, best = h2, b2

    rem = max(0.0, max_seconds - (time.time() - t0))
    sl2 = min(30.0, 0.25 * rem)
    if n <= 70 and sl2 > 4.0:
        h2, b2 = slsqp_refine(h, sl2, rng)
        if b2 + 1e-12 < best:
            h, best = h2, b2

    rem = max(0.0, max_seconds - (time.time() - t0))
    if rem > 1.0:
        h2, b2 = local_pair_ref(h, rem, rng)
        if b2 + 1e-12 < best:
            h, best = h2, b2

    h = enforce_exact_sum(h, 0.5 * h.size)
    return h, c5_scaled(h)


def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    rng = np.random.default_rng(seed)
    budget = float(budget_s)
    reserve = min(25.0, 0.05 * budget)
    deadline = start + max(1.0, budget - reserve)

    parent = globals().get("initial_h_values")
    if parent is not None:
        parent = np.asarray(parent, dtype=float).copy()
        if parent.size < 2:
            parent = default_start(45, rng)
    else:
        parent = default_start(45, rng)

    initial_h = enforce_exact_sum(parent, 0.5 * parent.size)
    initial_c = c5_scaled(initial_h)
    best_h = initial_h.copy()
    best_c = initial_c

    avail = deadline - time.time()
    if avail > 15.0:
        alloc = min(280.0, 0.45 * max(0.0, budget - reserve), max(0.0, avail - 5.0))
        h2, b2 = refine_candidate(best_h, alloc, rng)
        if b2 + 1e-12 < best_c:
            best_h, best_c = h2, b2

    avail = deadline - time.time()
    if avail > 60.0 and best_h.size <= 120:
        dim = int(best_h.size)
        init_t = np.clip(best_h - 0.5, -0.5, 0.5)
        de_alloc = min(220.0, 0.3 * max(0.0, budget - reserve), 0.35 * max(0.0, avail))
        tbest = de_optimize(dim, de_alloc, rng, init_t)
        h_de = enforce_exact_sum(0.5 + tbest, 0.5 * dim)
        rem = deadline - time.time()
        if rem > 10.0:
            h_de, b_de = refine_candidate(h_de, min(140.0, 0.35 * rem), rng)
            if b_de + 1e-12 < best_c:
                best_h, best_c = h_de, b_de
        else:
            v = c5_scaled(h_de)
            if v + 1e-12 < best_c:
                best_h, best_c = h_de, v

    for nr in (32, 44, 64, 80):
        if time.time() > deadline:
            break
        if nr == int(best_h.size):
            continue
        h_r = resize_h(parent, nr)
        alloc = min(25.0, max(4.0, 0.08 * max(0.0, deadline - time.time())))
        h_r, b_r = refine_candidate(h_r, alloc, rng)
        if b_r + 1e-12 < best_c:
            best_h, best_c = h_r, b_r

    rem = deadline - time.time()
    if rem > 4.0:
        h2, b2 = local_pair_ref(best_h, min(20.0, 0.75 * rem), rng)
        if b2 + 1e-12 < best_c:
            best_h, best_c = h2, b2

    best_h = enforce_exact_sum(best_h, 0.5 * best_h.size)
    best_c = float(c5_scaled(best_h))

    if (not math.isfinite(best_c)) or (best_c > initial_c + 1e-12):
        best_h = initial_h.copy()
        best_c = initial_c

    return best_h, float(best_c), int(best_h.size)
```
