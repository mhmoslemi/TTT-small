```python
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    return np.max(corr) * dx

def grad_for_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] -= h[:n-s]
    else:
        t = -s
        grad[t:] += (1.0 - h[:n-t])
        grad[:n-t] -= h[t:]
    return grad

def subgradient_max(corr, h):
    n = len(h)
    max_val = np.max(corr)
    # Use all entries within tolerance of max
    active = np.where(corr >= max_val - 1e-6)[0]
    grad = np.zeros(n)
    for idx in active:
        s = idx - (n - 1)
        grad += grad_for_shift(h, s)
    grad /= len(active)
    return grad

def project_feasible(h, n):
    """Project onto {x : 0<=x[i]<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Bisection on shift
    lo, hi = h.min() - 1.0, h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0, 1)
    # Fix numerical issues - adjust to get exact sum
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(result)  # least constrained first
        if diff > 0:
            for i in idx[::-1]:
                add = min(diff, 1.0 - result[i])
                result[i] += add
                diff -= add
                if diff <= 0: break
        else:
            for i in idx:
                sub = min(-diff, result[i])
                result[i] -= sub
                diff += sub
                if diff >= 0: break
    return result

def projected_gradient_step(h, lr, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    grad = subgradient_max(corr, h)
    # Step
    new_h = h - lr * grad
    new_h = project_feasible(new_h, n)
    return new_h, max_corr
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def grad_for_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        t = -s
        if t < n:
            grad[t:] += (1.0 - h[:n-t])
            grad[:n-t] -= h[t:]
    return grad

    n = len(h)
    max_val = np.max(corr)

    grad = np.zeros(n)
    for idx in active:
        s = int(idx) - (n - 1)
        grad += grad_for_shift(h, s)
    grad /= float(len(active))
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def grad_for_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] -= h[:n-s]
    else:
        t = -s
        if t < n:
            grad[t:] += (1.0 - h[:n-t])
            grad[:n-t] -= h[t:]
    return grad

def smooth_grad(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = np.max(corr)
    ex = np.exp(alpha * (corr - mc))
    w = ex / np.sum(ex)
    grad = np.zeros(n)
    active = np.where(w > 1e-5)[0]
    for idx in active:
        s = int(idx) - (n - 1)
        grad += w[idx] * grad_for_shift(h, s)
    return grad

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 1.5
    hi = h.max() + 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-9:
        idx = np.argsort(result)
        if diff > 0:
            for i in idx[::-1]:
                add = min(diff, 1.0 - result[i])
                if add > 0:
                    result[i] += add
                    diff -= add
                if diff <= 1e-9:
                    break
        else:
            for i in idx:
                sub = min(-diff, result[i])
                if sub > 0:
                    result[i] -= sub
                    diff += sub
                if diff >= -1e-9:
                    break
    return result

def interpolate_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x_src = np.linspace(0.0, 2.0, n_src)
    x_dst = np.linspace(0.0, 2.0, n_dst)
    return np.interp(x_dst, x_src, h_src)

def optimize_pg(h, time_limit, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    lr = 0.04
    no_improve = 0
    alpha = 30.0

    while time.time() < time_limit:
        grad = smooth_grad(h, alpha)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad = grad / gn * gn  # keep as is
        
        accepted = False
        lr_t = lr
        for _ in range(12):
            new_h = h - lr_t * grad
            new_h = project_feasible(new_h, n)
            new_c5 = compute_c5(new_h)
            if new_c5 < cur_c5 - 1e-13:
                h = new_h
                cur_c5 = new_c5
                accepted = True
                break
            lr_t *= 0.5
        if not accepted:
            # Try random perturbation
            if no_improve > 200:
                h = project_feasible(h + 0.01 * rng.randn(n), n)
                cur_c5 = compute_c5(h)
                no_improve = 0
                alpha = max(15.0, alpha * 0.8)
                continue
            break

        if cur_c5 < best_c5 - 1e-10:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
            lr = min(lr * 1.04, 0.25)
            alpha = min(alpha * 1.01, 200.0)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.7
                alpha = max(10.0, alpha * 0.85)
            if no_improve > 800:
                break

    return best_h, best_c5

def make_phase_h(n, rng):
    """Create a structured initial condition with some pattern."""
    h = np.zeros(n)
    num_blocks = rng.randint(3, 10)
    boundaries = sorted(rng.choice(n, num_blocks, replace=False))
    boundaries = [0] + list(boundaries)
    for i in range(len(boundaries)):
        start = boundaries[i]
        end = boundaries[i+1] if i + 1 < len(boundaries) else n
        if rng.rand() > 0.4:
            h[start:end] = 1.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    time_limit = start_time + budget_s - 5.0
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    # Phase 1: Use initial_h_values
    init = globals().get('initial_h_values', None)
    if init is not None:
        init = np.array(init, dtype=float).copy()
        n_init = len(init)
        init = project_feasible(init, n_init)
        c5_init = compute_c5(init)
        best_h = init.copy()
        best_c5 = c5_init
        best_n = n_init

        if time.time() < time_limit:
            h_opt, c5_opt = optimize_pg(init, min(start_time + 300, time_limit), rng)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = len(h_opt)

    # Phase 2: Multi-resolution optimization
    if time.time() < time_limit:
        for n_try in [50, 60, 70, 80, 100, 140]:
            if time.time() > time_limit:
                break
            h = interpolate_to_n(best_h, n_try)
            h = project_feasible(h, n_try)
            sub_limit = min(time.time() + 150, time_limit)
            h_opt, c5_opt = optimize_pg(h, sub_limit, rng)
            if h_opt is not None and c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = len(h_opt)

    # Phase 3: Random and structured restarts
    while time.time() < time_limit:
        n_try = int(rng.choice([50, 70, 100, 120]))
        mode = rng.randint(3)
        if mode == 0:
            h = rng.rand(n_try)
        elif mode == 1:
            h = make_phase_h(n_try, rng)
        else:
            # Interpolate best and perturb
            h = interpolate_to_n(best_h, n_try)
            h = h + 0.1 * rng.randn(n_try)
        h = project_feasible(h, n_try)

        restart_limit = min(time.time() + 40, time_limit)
        h_opt, c5_opt = optimize_pg(h, restart_limit, rng)
        if h_opt is not None and c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = len(h_opt)

    # Phase 4: Final polish on best solution
    if time.time() < time_limit and best_h is not None:
        sub_limit = min(time.time() + 120, time_limit)
        h_opt, c5_opt = optimize_pg(best_h.copy(), sub_limit, rng)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = len(h_opt)

    if best_h is None:
        n = 70
        best_h = np.full(n, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n

    # Final verification
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(best_n)
```
