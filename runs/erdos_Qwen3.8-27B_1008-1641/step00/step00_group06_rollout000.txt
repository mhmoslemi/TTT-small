    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode='full')

def smooth_jacobian(x):
    base = np.asarray(x, dtype=float)
    if base.size == 0: return np.array([])
    work = base.copy()
    grad = np.empty(base.size, dtype=float)
    eps = 1e-6
    for i in range(base.size):
        orig = float(work[i])
        if orig < 0.0: orig = 0.0
        elif orig > 1.0: orig = 1.0
        if orig <= eps:
            hp = min(orig + eps, 1.0)
            work[i] = orig
            f0 = smooth_objective(work)
            work[i] = hp
            f1 = smooth_objective(work)
            grad[i] = (f1 - f0) / max(hp - orig, eps)
        elif orig >= 1.0 - eps:
            lp = max(orig - eps, 0.0)
            work[i] = lp
            f0 = smooth_objective(work)
            work[i] = orig
            f1 = smooth_objective(work)
            grad[i] = (f1 - f0) / max(orig - lp, eps)

            work[i] = orig - eps
            f0 = smooth_objective(work)
            work[i] = orig + eps
            f1 = smooth_objective(work)
            grad[i] = (f1 - f0) / (2.0 * eps)
        work[i] = float(base[i])
    return grad

    if base.size == 0:
        return base.copy()

```python
import time
import numpy as np
from scipy import optimize

_G_N = 95
_G_TARGET = 47.5
_G_BETA = 10.0


def project_feasible(v, target=None):
    if target is None:
        target = _G_TARGET
    x = np.asarray(v, dtype=float)
    if not np.all(np.isfinite(x)):
        x = np.where(np.isfinite(x), x, 0.5)
    x = np.clip(x, 0.0, 1.0).copy()
    diff = float(target - x.sum())
    for _ in range(30):
        if abs(diff) <= 1e-12 * max(1.0, float(target)):
            break
        if diff > 0.0:
            mask = x < 1.0
            if not mask.any():
                break
            add = min(float((1.0 - x[mask]).min()), diff / float(mask.sum()))
            if add <= 0.0:
                break
            x[mask] += add
        else:
            mask = x > 0.0
            if not mask.any():
                break
            rem = min(float(x[mask].min()), -diff / float(mask.sum()))
            if rem <= 0.0:
                break
            x[mask] -= rem
        diff = float(target - x.sum())

    diff = float(target - x.sum())
    if abs(diff) > 1e-14:
        if diff > 0.0:
            order = np.argsort(1.0 - x)
            for idx in order:
                step = min(float(1.0 - x[idx]), diff)
                x[idx] += step
                diff -= step
                if abs(diff) <= 1e-15:
                    break
        else:
            order = np.argsort(x)
            for idx in order:
                step = min(float(x[idx]), -diff)
                x[idx] -= step
                diff += step
                if abs(diff) <= 1e-15:
                    break
    return np.clip(x, 0.0, 1.0)


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    if h.size == 0 or not np.all(np.isfinite(h)):
        return 1e100
    dx = 2.0 / float(h.size)
    vals = np.correlate(h, 1.0 - h, mode="full")
    val = dx * float(np.max(vals))
    if not np.isfinite(val):
        return 1e100
    return val


def smooth_objective(x):
    x = np.asarray(x, dtype=float)
    if x.size == 0 or not np.all(np.isfinite(x)):
        return 1e100
    beta = float(_G_BETA)
    if beta <= 0.0:
        return exact_c5(x)
    c = np.correlate(x, 1.0 - x, mode="full")
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e100
    q = (c - m) * beta
    s = float(np.sum(np.exp(q)))
    if s <= 0.0 or not np.isfinite(s):
        return 1e100
    return (2.0 / float(x.size)) * (m + np.log(s) / beta)


def smooth_jacobian(x):
    base = np.asarray(x, dtype=float)
    if base.size == 0:
        return base.copy()
    work = base.copy()
    grad = np.zeros(base.size, dtype=float)
    eps = 1e-6
    for i in range(base.size):
        orig = float(work[i])
        if orig < 0.0:
            orig = 0.0
        elif orig > 1.0:
            orig = 1.0
        if orig <= eps:
            hp = min(orig + eps, 1.0)
            work[i] = orig
            f0 = smooth_objective(work)
            work[i] = hp
            f1 = smooth_objective(work)
            val = (f1 - f0) / max(hp - orig, eps)
        elif orig >= 1.0 - eps:
            lp = max(orig - eps, 0.0)
            work[i] = lp
            f0 = smooth_objective(work)
            work[i] = orig
            f1 = smooth_objective(work)
            val = (f1 - f0) / max(orig - lp, eps)
        else:
            work[i] = orig - eps
            f0 = smooth_objective(work)
            work[i] = orig + eps
            f1 = smooth_objective(work)
            val = (f1 - f0) / (2.0 * eps)
        work[i] = float(base[i])
        if np.isfinite(val):
            grad[i] = val
    return grad


def sum_constraint(x):
    return float(np.sum(x) - _G_TARGET)


def polish_slsqp(h, beta, deadline, maxiter=30, ftol=1e-11):
    global _G_BETA
    h0 = project_feasible(h)
    val0 = exact_c5(h0)
    if time.time() > deadline - 2.0:
        return h0, val0
    _G_BETA = float(beta)
    try:
        res = optimize.minimize(
            smooth_objective,
            h0,
            method="SLSQP",
            jac=smooth_jacobian,
            bounds=[(0.0, 1.0) for _ in range(h0.size)],
            constraints=({"type": "eq", "fun": sum_constraint},),
            options={"maxiter": int(maxiter), "ftol": float(ftol)},
        )
        cand = project_feasible(np.asarray(res.x, dtype=float))
    except Exception:
        cand = h0
    val = exact_c5(cand)
    if not np.isfinite(val):
        return h0, val0
    return cand, val


def random_feasible(rng, mode):
    n = int(_G_N)
    target = float(_G_TARGET)
    if mode == 0:
        x = rng.uniform(0.0, 1.0, n)
    elif mode == 1:
        x = 0.5 + 0.5 * rng.standard_normal(n)
    elif mode == 2:
        x = np.zeros(n)
        k = int(target)
        rem = float(target - k)
        if rem > 1.0 - 1e-12:
            k += 1
            rem = 0.0
        k = max(0, min(n, k))
        idx = rng.permutation(n)
        if k > 0:
            x[idx[:k]] = 1.0
        if rem > 1e-12 and k < n:
            x[idx[k]] = rem
        return x
    elif mode == 3:
        alpha = 0.6180339887
        f = np.arange(n, dtype=float) * alpha
        f -= np.floor(f)
        x = (f < 0.5).astype(float)
    elif mode == 4:
        t = np.arange(n, dtype=float) / float(n)
        m1 = int(rng.integers(1, max(2, n // 3 + 1)))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        x = 0.5 + 0.45 * np.sin(2.0 * np.pi * m1 * t + p1)
    else:
        t = np.arange(n, dtype=float) / float(n)
        m1 = int(rng.integers(1, max(2, n // 3 + 1)))
        m2 = int(rng.integers(1, max(2, n // 2 + 1)))
        p1 = float(rng.uniform(0.0, 2.0 * np.pi))
        p2 = float(rng.uniform(0.0, 2.0 * np.pi))
        x = (
            0.5
            + 0.35 * np.sin(2.0 * np.pi * m1 * t + p1)
            + 0.20 * np.sin(2.0 * np.pi * m2 * t + p2)
        )
    return project_feasible(x)


def de_objective(x):
    return exact_c5(project_feasible(x))


def pair_local_search(h, deadline, rng, max_evals=50000):
    h = project_feasible(h)
    f = exact_c5(h)
    n = int(h.size)
    if n < 2:
        return h, f
    evals = 0
    ds = (0.005, 0.02, 0.1, 0.3, 0.7, 1.0)
    while evals < max_evals and time.time() < deadline:
        order = np.argsort(h)
        k = max(1, n // 2)
        lows = order[:k]
        highs = order[n - k:]
        for _ in range(60):
            if evals >= max_evals or time.time() >= deadline:
                break
            if rng.random() < 0.25:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
            else:
                i = int(highs[rng.integers(len(highs))])
                j = int(lows[rng.integers(len(lows))])
            if i == j:
                continue
            hi = float(h[i])
            hj = float(h[j])
            for d in ds:
                md1 = min(hi, 1.0 - hj)
                if md1 > 1e-8:
                    dd = min(float(d), md1)
                    if dd > 1e-6:
                        cand = h.copy()
                        cand[i] -= dd
                        cand[j] += dd
                        cand = project_feasible(cand)
                        val = exact_c5(cand)
                        evals += 1
                        if val < f - 1e-13:
                            h = cand
                            f = val
                            break
                md2 = min(hj, 1.0 - hi)
                if md2 > 1e-8:
                    dd = min(float(d), md2)
                    if dd > 1e-6:
                        cand = h.copy()
                        cand[i] += dd
                        cand[j] -= dd
                        cand = project_feasible(cand)
                        val = exact_c5(cand)
                        evals += 1
                        if val < f - 1e-13:
                            h = cand
                            f = val
                            break
                if evals >= max_evals:
                    break
    return h, f


def run(seed=42, budget_s=1100, **kwargs):
    global _G_N, _G_TARGET

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 42
    try:
        budget = float(budget_s)
    except Exception:
        budget = 1100.0
    if budget < 30.0:
        budget = 30.0

    t0 = time.time()
    if budget > 300.0:
        deadline = t0 + budget - 120.0
    else:
        deadline = t0 + budget * 0.75
    stop = deadline - 10.0

    rng = np.random.default_rng(seed_int)
    best = None
    best_val = 1e100

    if "initial_h_values" in globals():
        try:
            p = np.asarray(initial_h_values, dtype=float).ravel()
            if p.size >= 10 and np.all(np.isfinite(p)):
                _G_N = int(p.size)
                _G_TARGET = 0.5 * float(_G_N)
                cand = project_feasible(p)
                best = cand
                best_val = exact_c5(cand)
        except Exception:
            pass

    if best is None:
        _G_N = 95
        _G_TARGET = 47.5
        best = random_feasible(rng, 0)
        best_val = exact_c5(best)
    else:
        best = project_feasible(best)
        best_val = exact_c5(best)

    primary = ((2.0, 20), (5.0, 30), (10.0, 30), (20.0, 30), (40.0, 30), (80.0, 25), (160.0, 20))
    for beta, mi in primary:
        if time.time() > stop:
            break
        cand, val = polish_slsqp(best, beta, deadline, mi, 1e-11)
        if val < best_val - 1e-13:
            best, best_val = cand, val

    if time.time() < stop - 30.0:
        try:
            bnd = [(0.0, 1.0) for _ in range(_G_N)]
            res = optimize.differential_evolution(
                de_objective,
                bnd,
                seed=seed_int + 77,
                maxiter=80,
                popsize=3,
                tol=1e-4,
                polish=False,
                init="latinhypercube",
            )
            cand = project_feasible(np.asarray(res.x, dtype=float))
            val = exact_c5(cand)
            if val < best_val + 0.02:
                for beta, mi in ((10.0, 25), (40.0, 25)):
                    c2, v2 = polish_slsqp(cand, beta, deadline, mi, 1e-10)
                    if v2 < val:
                        cand, val = c2, v2
                    if v2 < best_val - 1e-13:
                        best, best_val = c2, v2
        except Exception:
            pass

    starts = []
    for p in (0.35, 0.6, 1.4, 2.2, 4.0):
        starts.append(project_feasible(np.power(best, p)))
    for amp in (0.02, 0.05, 0.1, 0.2, 0.4, 0.7):
        starts.append(project_feasible(best + amp * rng.standard_normal(_G_N)))
    ref = random_feasible(rng, 3)
    for w in (0.2, 0.4, 0.6, 0.8):
        starts.append(project_feasible(w * best + (1.0 - w) * ref))
    for mode in (2, 3, 4, 5, 1, 0, 2, 3, 4, 1, 0, 5, 2, 3, 4):
        starts.append(random_feasible(rng, mode))

    pool = []
    for x in starts:
        if time.time() > stop:
            break
        xv = exact_c5(x)
        if xv >= best_val + 0.03:
            continue
        for beta, mi in ((5.0, 15), (20.0, 20), (50.0, 15)):
            cand, val = polish_slsqp(x, beta, deadline, mi, 1e-10)
            if val < xv:
                x, xv = cand, val
            if val < best_val - 1e-13:
                best, best_val = cand, val
        if xv < best_val + 0.005:
            if len(pool) < 8:
                pool.append((x, xv))
            else:
                worst = 0
                for i in range(1, len(pool)):
                    if pool[i][1] < pool[worst][1]:
                        worst = i
                if xv < pool[worst][1]:
                    pool[worst] = (x, xv)

    for x, xv in pool:
        if time.time() > stop - 5.0:
            break
        if np.max(np.abs(x - best)) < 1e-8 and abs(xv - best_val) < 1e-8:
            continue
        for beta, mi in ((40.0, 20), (80.0, 20)):
            cand, val = polish_slsqp(x, beta, deadline, mi, 1e-10)
            if val < xv:
                x, xv = cand, val
            if val < best_val - 1e-13:
                best, best_val = cand, val

    if time.time() < stop - 5.0:
        cand, val = pair_local_search(best, min(deadline, stop), rng, 50000)
        if val < best_val - 1e-13:
            best, best_val = cand, val

    final = ((40.0, 30), (80.0, 25), (160.0, 20), (320.0, 15))
    for beta, mi in final:
        if time.time() > stop:
            break
        cand, val = polish_slsqp(best, beta, deadline, mi, 1e-12)
        if val < best_val - 1e-13:
            best, best_val = cand, val

    best = project_feasible(best)
    best_val = exact_c5(best)
    return best, float(best_val), int(best.size)
```
