import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_polish(h, target_sum, max_iter=2000):
    n = len(h)
    best_val = compute_c5(h)
    step_sizes = [0.005, 0.01, 0.02, 0.05, 0.1, -0.005, -0.01, -0.02, -0.05]
    for _ in range(max_iter):
        improved = False
        for i in range(n):
            for d in step_sizes:
                h2 = h.copy()
                h2[i] = np.clip(h2[i] + d, 0.0, 1.0)
                h2 = project_feasible(h2, target_sum)
                v = compute_c5(h2)
                if v < best_val - 1e-13:
                    h = h2
                    best_val = v
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_val

            if res.x is not None:
