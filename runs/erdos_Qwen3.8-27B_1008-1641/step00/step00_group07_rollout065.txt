    h = project_feasible(x, target_sum)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = corr.max()
    return m + np.log(np.sum(np.exp((corr - m) * beta))) / beta

def coord_descent(h, target_sum, max_passes=20, deadline=None):
    n = len(h)
    best = compute_c5(h)[0]
    for p in range(max_passes):

        for m in range(n):
            # Candidates for h[m]
            cands = set()
            cands.add(0.0)
            cands.add(1.0)
            cands.add(h[m])
            for d in (0.1, -0.1, 0.3, -0.3):
                cands.add(h[m] + d)
            for c in cands:
                c = np.clip(c, 0.0, 1.0)
                if c == h[m]:
                    continue
                ht = h.copy()
                ht[m] = c
                ht = project_feasible(ht, target_sum)
                c5, _ = compute_c5(ht)
                if c5 < best - 1e-10:
                    h = ht
                    best = c5
                    improved = True
        if not improved:
            break
    return h, best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_feasible(h, target_sum):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(diff / cnt, 1.0 - h[mask].min())
            h[mask] += step
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            step = min(-diff / cnt, h[mask].min())
            h[mask] -= step
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, target_sum, beta):
    h = project_feasible(x, target_sum)
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = corr.max()
    shifted = (corr - m) * beta
    shifted -= shifted.max()
    return m + np.log(np.sum(np.exp(shifted))) / beta

def coord_descent(h, target_sum, max_passes=20, deadline=None):
    n = len(h)
    best, _ = compute_c5(h)
    deltas = [0.05, -0.05, 0.1, -0.1, 0.2, -0.2, 0.5, -0.5]
    for p in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for m in range(n):
            cands = {0.0, 1.0, h[m]}
            for d in deltas:
                c = h[m] + d
                if 0.0 <= c <= 1.0:
                    cands.add(c)
            for c in cands:
                if abs(c - h[m]) < 1e-10:
                    continue
                ht = h.copy()
                ht[m] = c
                ht = project_feasible(ht, target_sum)
                c5, _ = compute_c5(ht)
                if c5 < best - 1e-12:
                    h = ht
                    best = c5
                    improved = True
        if not improved:
            break
    return h, best

    best, _ = compute_c5(h)
    deltas = [0.01, -0.01, 0.02, -0.02, 0.04, -0.04, 0.08, -0.08]
