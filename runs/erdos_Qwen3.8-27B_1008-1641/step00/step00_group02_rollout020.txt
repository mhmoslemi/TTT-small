    m = np.max(corr)

    n = len(h)
    lo, hi = -1.0, 2.0
    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:

        else:
            hi = mid

def grad_c5(h, dx):

    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)

    grad[mask_pos] += dx * (1.0 - h[idx_pos[mask_pos]])
    grad[mask_neg] -= dx * h[idx_neg[mask_neg]]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_capped(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on lambda."""
    lo, hi = -1.5, 2.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def grad_c5(h, dx):
    """Analytical gradient of C5 using the active (max) correlation shift."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    js = np.arange(n)
    idx_p = js + s
    idx_m = js - s
    mask_p = (idx_p >= 0) & (idx_p < n)
    mask_m = (idx_m >= 0) & (idx_m < n)
    grad = np.zeros(n)
    grad[mask_p] += dx * (1.0 - h[idx_p[mask_p]])
    grad[mask_m] -= dx * h[idx_m[mask_m]]
    return grad, s

def smooth_max_gradient(h, dx, top_k=5):
    """Gradient using weighted average of top-k active shifts for smoother landscape."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    weights = np.exp(40.0 * (corr - m))
    weights /= np.sum(weights)
    # Focus on top shifts
    top_idx = np.argsort(weights)[-top_k:]
    w = weights[top_idx]
    w /= np.sum(w)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        s = k - (n - 1)
        js = np.arange(n)
        idx_p = js + s
        idx_m = js - s
        mask_p = (idx_p >= 0) & (idx_p < n)
        mask_m = (idx_m >= 0) & (idx_m < n)
        grad[mask_p] += wi * dx * (1.0 - h[idx_p[mask_p]])
        grad[mask_m] -= wi * dx * h[idx_m[mask_m]]
    return grad

def projected_gradient_descent(h, dx, target_sum, max_iter, time_limit, t0):
    """PGD with momentum and backtracking line search."""
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    momentum = np.zeros(n)
    beta_mom = 0.85
    lr = 0.04
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        c5_now = compute_c5(h, dx)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
        
        grad, _ = grad_c5(h, dx)
        # Occasionally use smooth gradient for better direction
        if it % 5 == 4:
            grad_s = smooth_max_gradient(h, dx, top_k=5)
            # Normalize to similar scale
            if np.linalg.norm(grad_s) > 1e-10:
                grad = 0.5 * grad + 0.5 * grad_s * (np.linalg.norm(grad) / np.linalg.norm(grad_s))
        
        momentum = beta_mom * momentum + grad
        direction = -momentum
        
        # Backtracking line search
        step = lr
        improved = False
        for _ in range(15):
            h_new = project_capped(h + step * direction, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new < c5_now - 1e-14:
                h = h_new
                improved = True
                break
            step *= 0.5
        
        if improved:
            lr = min(lr * 1.08, 0.4)
        else:
            lr *= 0.75
            if lr < 1e-11:
                h = best_h + np.random.normal(0, 0.015, n)
                h = project_capped(h, target_sum)
                momentum = np.zeros(n)
                lr = 0.04
    
    return best_h, best_c5

def nelder_mead_refine(best_h, dx, target_sum, t0, time_limit):
    """Nelder-Mead local refinement."""
    n = len(best_h)
    
    def obj(h_raw):
        if time.time() - t0 > time_limit:
            return 1e10
        hp = project_capped(h_raw, target_sum)
        return compute_c5(hp, dx)
    
    res = minimize(obj, best_h, method='Nelder-Mead',
                  options={'maxiter': 2000, 'xatol': 1e-11, 'fatol': 1e-13, 'adaptive': True})
    if res.fun < compute_c5(best_h, dx) - 1e-12:
        hc = project_capped(res.x, target_sum)
        c5c = compute_c5(hc, dx)
        if c5c < compute_c5(best_h, dx):
            return hc, c5c
    return best_h, compute_c5(best_h, dx)

def fine_perturbation(best_h, dx, target_sum, t0, time_limit):
    """Small random perturbations around the best solution."""
    n = len(best_h)
    best_c5 = compute_c5(best_h, dx)
    for _ in range(2000):
        if time.time() - t0 > time_limit:
            break
        scale = 0.002 + 0.003 * np.random.rand()
        h_try = best_h + np.random.normal(0, scale, n)
        h_try = project_capped(h_try, target_sum)
        c5_try = compute_c5(h_try, dx)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            # Additional refinement from new point
            for _2 in range(10):
                h_t2 = best_h + np.random.normal(0, 0.001, n)
                h_t2 = project_capped(h_t2, target_sum)
                c5_t2 = compute_c5(h_t2, dx)
                if c5_t2 < best_c5:
                    best_c5 = c5_t2
                    best_h = h_t2
    return best_h, best_c5

def try_n_points(h_init, n_new, target_sum_new):
    """Interpolate initial solution to a new grid size."""
    n_old = len(h_init)
    x_old = np.linspace(0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0, 1.0, n_new, endpoint=False)
    h_inter = np.interp(x_new, x_old, h_init, period=1.0)
    return np.clip(h_inter, 0, 1)
