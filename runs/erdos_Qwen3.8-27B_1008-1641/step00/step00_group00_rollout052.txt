        else:
            stagnant += 1
            if stagnant > 60:
                stagnant = 0

    # Phase 3: Final polish with higher p and more iterations
    if time.time() - t0 < budget_s - 15:
        _G_p = 32
        h_new, c5_new = do_slsqp(best_h, 3000, 1e-15, True)
        if h_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        _G_p = 16

    # Final validation
    best_h = project_feasible(best_h, S)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
