from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def repair_h(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = ts - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_nm(h0, dx, ts, iters=10000):
    n = len(h0)
    def obj(x):
        x = np.clip(x, 0, 1)
        return evaluate_h(x, dx) + 1e4 * (x.sum() - ts) ** 2
    r = minimize(obj, h0, method='Nelder-Mead',
                 options={'maxiter': iters, 'xatol': 1e-8, 'fatol': 1e-10, 'adaptive': True})
    return repair_h(r.x, ts)

def subgrad_step(h, dx, ts, lr):
    """One subgradient step: use the active lag's gradient."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_star = np.argmax(corr)
    # Numerical gradient of corr[k_star] w.r.t. h
    grad = np.zeros(n)
    eps = 1e-5
    for i in range(n):
        hp = h.copy(); hp[i] = min(1.0, h[i] + eps)
        hm = h.copy(); hm[i] = max(0.0, h[i] - eps)
        cp = np.correlate(hp, 1.0 - hp, mode="full")[k_star]
        cm = np.correlate(hm, 1.0 - hm, mode="full")[k_star]
        grad[i] = (cp - cm) / (2 * eps)
    # Projected step
    h_new = h - lr * grad
    h_new = repair_h(h_new, ts)
    return h_new, k_star

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)

def repair_h(h, ts):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        d = ts - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-14) & (h < 1.0 - 1e-14)
        if not m.any():
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_grad(k_idx, h):
    n = len(h)
    p = k_idx - (n - 1)
    m = np.arange(n)
    mask1 = (m - p >= 0) & (m - p < n)
    mask2 = (m + p >= 0) & (m + p < n)
    grad = np.zeros(n)
    grad[mask1] = 1.0 - h[m[mask1] - p]
    grad[mask2] -= h[m[mask2] + p]
    return grad

def nm_opt(h0, dx, ts, iters=10000):
    def obj(x):
        x = np.clip(x, 0, 1)
        return evaluate_h(x, dx) + 1e4 * (x.sum() - ts) ** 2
    r = minimize(obj, h0, method='Nelder-Mead',
                 options={'maxiter': iters, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
    return repair_h(r.x, ts)

def subgrad_opt(h0, dx, ts, max_iter=50000, deadline=None):
    n = len(h0)
    h = h0.copy()
    best = evaluate_h(h, dx)
    best_h = h.copy()
    lr = 0.008
    stall = 0
    for it in range(max_iter):
        if deadline and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        mx = corr.max()
        k_stars = np.where(corr > mx - 1e-10)[0]
        g = np.zeros(n)
        for ks in k_stars:
            g += corr_grad(ks, h)
        g /= len(k_stars)
        g -= g.mean()
        h_new = repair_h(h - lr * g, ts)
        c_new = evaluate_h(h_new, dx)
        if c_new < best:
            best = c_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
        else:
            h = h_new
            stall += 1
            if stall > 300:
                lr *= 0.85
                stall = 0
        if it % 5000 == 0 and lr < 0.001:
            h = best_h.copy()
            lr = 0.004
    return best_h

        if c < best_c5:

        return c

    # Primary: optimize from initial_h_values
