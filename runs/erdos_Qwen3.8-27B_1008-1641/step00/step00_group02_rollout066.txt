    """Compute subgradient of max active correlation w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    shifts_present = np.where(active_mask)[0]
    
    for j in shifts_present:
        s = j - (n - 1)
        # c_s = sum_i h[i]*(1-h[i-s]) for valid i, i-s in [0,n)

            # -h[m-s] for m in [sp, n-1]
            grad[sp:] -= h[:n-sp]
