import numpy as np
from scipy.optimize import minimize

_TARGET = 0.0


def _clean(h):
    h = np.asarray(h, dtype=float).reshape(-1)
    if h.size == 0:
        return h
    h = np.where(np.isfinite(h), h, 0.5)
    return np.clip(h, 0.0, 1.0)


def official_c(h):
    h = _clean(h)
    n = len(h)
    if n == 0:
        return 1.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / float(n)))


def count_max(h):
    h = _clean(h)
    if len(h) == 0:
        return 1.0
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def custom_c(h):
    h = _clean(h)
    n = len(h)
    if n == 0:
        return np.empty(0)
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    c = np.empty(2 * n - 1)
    for idx in range(2 * n - 1):
        lag = idx - (n - 1)
        if lag >= 0:
            m = n - lag
            if m <= 0:
                c[idx] = 0.0
            else:
                c[idx] = float(pref[m] - pref[0] - np.dot(h[:m], h[lag:]))
        else:
            r = -lag
            m = n - r
            if m <= 0:
                c[idx] = 0.0
            else:
                c[idx] = float(pref[n] - pref[r] - np.dot(h[r:], h[:m]))
    return c


def make_weights(c, beta):
    c = np.asarray(c, dtype=float)
    if c.size == 0:
        return c
    mx = float(np.max(c))
    if beta is None or float(beta) <= 0.0:
        w = (c >= mx - 1e-9).astype(float)
        s = float(np.sum(w))
        if s <= 0.0:
            w[np.argmax(c)] = 1.0
            return w
        return w / s
    b = float(beta)
    z = np.exp(b * (c - mx))
    s = float(np.sum(z))
    if s <= 0.0 or not math.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return z / s


def weighted_gradient(h, w):

    z = np.exp(b * (c - mx))
    s = float(np.sum(z))
    if s <= 0.0 or not math.isfinite(s):
        return mx
    return float(mx + math.log(s) / b)


def soft_jac(x, beta):
    h = _clean(x)
    c = custom_c(h)
    w = make_weights(c, beta)
    return weighted_gradient(h, w)


def sum_constraint(x):
    return float(np.sum(x) - _TARGET)


def project_sum_box(v, S):
    h = np.asarray(v, dtype=float).reshape(-1).copy()
    n = len(h)
    if n == 0:
        return h
    h = np.where(np.isfinite(h), h, 0.5)
    S = float(S)
    if S < 0.0:
        S = 0.0
    if S > float(n):
        S = float(n)
    lo = -float(np.max(h)) - 1.0
    hi = 1.0 - float(np.min(h)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(h + mid, 0.0, 1.0)))
        if s < S:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    for _ in range(10):
        d = S - float(np.sum(h))
        if abs(d) <= 1e-13:
            break
        if d > 0.0:
            idx = np.nonzero(h < 1.0 - 1e-15)[0]
            if idx.size == 0:
                break
            room = float(np.sum(1.0 - h[idx]))
            if room <= 0.0:
                break
            frac = min(1.0, abs(d) / room)
            h[idx] += (1.0 - h[idx]) * frac
        else:
            idx = np.nonzero(h > 1e-15)[0]
            if idx.size == 0:
                break
            room = float(np.sum(h[idx]))
            if room <= 0.0:
                break
            frac = min(1.0, abs(d) / room)
            h[idx] -= h[idx] * frac
    if n > 1:
        last = S - float(np.sum(h[:n - 1]))
        if 0.0 <= last <= 1.0:
            h[n - 1] = float(last)
    h = np.clip(h, 0.0, 1.0)
    return h


def random_binary(n, rng):
    n = int(n)
    h = np.zeros(n)
    if n <= 0:
        return h
    t = n // 2
    if t > 0:
        perm = rng.permutation(n)
        h[perm[:t]] = 1.0
    return h


def random_feasible(n, rng, center=None, scale=0.1):
    n = int(n)
    if center is None:
        h = rng.uniform(size=n)
    else:
        h = _clean(center) + float(scale) * rng.standard_normal(n)
    return project_sum_box(h, float(n) / 2.0)


def default_h(n):
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    h = 0.5 + 0.45 * np.sin(2.0 * np.pi * x)
    return project_sum_box(h, float(n) / 2.0)


def resample(h, m):
    h = _clean(h)
    n = len(h)
    m = int(m)
    if m <= 0:
        return np.empty(0)
    if n == 0:
        return project_sum_box(np.full(m, 0.5), float(m) / 2.0)
    if n == m:
        return project_sum_box(h, float(m) / 2.0)
    old = (np.arange(n) + 0.5) * (2.0 / float(n))
    new = (np.arange(m) + 0.5) * (2.0 / float(m))
    hnew = np.interp(new, old, h, left=float(h[0]), right=float(h[-1]))
    return project_sum_box(hnew, float(m) / 2.0)


def get_initial_h():
    try:
        raw = initial_h_values
    except NameError:
        raw = None
    if raw is not None:
        try:
            arr = np.asarray(raw, dtype=float).reshape(-1)
        except Exception:
            arr = None
        if arr is not None and arr.size > 0:
            n = int(arr.size)
            arr = np.where(np.isfinite(arr), arr, 0.5)
            if n % 2 == 1:
                m = 2 * (n // 2)
                if m < 4:
                    m = 70
                arr = resample(arr, m)
                n = m
            return project_sum_box(arr, float(n) / 2.0)
    return project_sum_box(default_h(70), 35.0)


def slsqp_refine(x0, beta, maxiter):
    n = len(x0)
    if n == 0:
        return x0
    global _TARGET
    S = float(n) / 2.0
    x0 = project_sum_box(x0, S)
    _TARGET = S
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': sum_constraint},)
    maxiter = int(max(1, maxiter))
    try:
        res = minimize(
            soft_obj, x0, args=(float(beta),), jac=soft_jac,
            method="SLSQP", bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-11, 'iprint': 0}
        )
        x = res.x
    except Exception:
        x = x0
    return project_sum_box(x, _TARGET)


def slsqp_rounds(h_best, c_best, betas, maxiter, time_limit, seed, max_calls=24):
    h_best = _clean(h_best)
    n = len(h_best)
    best_h = h_best.copy()
    best_c = float(c_best)
    S = float(n) / 2.0
    rng = np.random.default_rng(seed)
    t0 = time.time()
    starts = [project_sum_box(best_h, S)]
    for scale in (0.02, 0.05, 0.10, 0.20):
        starts.append(random_feasible(n, rng, best_h, scale))
    for _ in range(2):
        rb = random_binary(n, rng)
        starts.append(project_sum_box(0.5 * best_h + 0.5 * rb, S))
    starts.append(random_feasible(n, rng))
    calls = 0
    for beta in betas:
        if calls >= max_calls or time.time() - t0 > time_limit:
            break
        for si in range(len(starts)):
            if calls >= max_calls or time.time() - t0 > time_limit:
                break
            x0 = starts[si].copy()
            if float(beta) >= 100.0 and si > 0:
                x0 = project_sum_box(0.7 * best_h + 0.3 * x0, S)
            x1 = slsqp_refine(x0, float(beta), int(maxiter))
            c1 = official_c(x1)
            calls += 1
            if math.isfinite(c1) and c1 < best_c - 1e-12:
                best_h = x1
                best_c = c1
                starts[0] = project_sum_box(best_h, S)
                if len(starts) < 40:
                    starts.append(project_sum_box(best_h + 0.01 * rng.standard_normal(n), S))
    return best_h, best_c


def binary_polish(h, time_limit, seed, max_passes=12):
    h = _clean(h)
    n = len(h)
    if n <= 1 or n % 2 != 0 or time_limit <= 0:
        return h.copy(), official_c(h)
    target = n // 2
    h = np.rint(h).astype(float)
    s = int(round(float(np.sum(h))))
    if s > target:
        idx = np.nonzero(h > 0.5)[0]
        for i in idx[:s - target]:
            h[i] = 0.0
    elif s < target:
        idx = np.nonzero(h < 0.5)[0]
        for i in idx[:target - s]:
            h[i] = 1.0
    best = count_max(h)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    stall = 0
    while time.time() - t0 < time_limit and stall < max_passes:
        ones = np.nonzero(h > 0.5)[0]
        zeros = np.nonzero(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        total = int(ones.size * zeros.size)
        if total <= 20000:
            pair_list = []
            for i in ones:
                ii = int(i)
                for j in zeros:
                    pair_list.append((ii, int(j)))
            order = rng.permutation(len(pair_list))
            pair_list = [pair_list[k] for k in order]
        else:
            pair_list = []
            for _ in range(3000):
                ii = int(ones[rng.integers(ones.size)])
                jj = int(zeros[rng.integers(zeros.size)])
                pair_list.append((ii, jj))
        improved = False
        for i, j in pair_list:
            if time.time() - t0 > time_limit:
                break
            if h[i] > 0.5 and h[j] < 0.5:
                oi = float(h[i])
                oj = float(h[j])
                h[i] = 0.0
                h[j] = 1.0
                c = count_max(h)
                if c < best - 1e-12:
                    best = c
                    improved = True
                else:
                    h[i] = oi
                    h[j] = oj
        if improved:
            stall = 0
        else:
            stall += 1
    return h, float(best) * (2.0 / float(n))


def binary_rounds(h_best, c_best, n, time_limit, seed, max_starts=3):
    h_best = _clean(h_best)
    n = int(n)
    if n <= 0 or n % 2 != 0:
        return h_best, float(c_best), np.zeros(n), 1.0
    rng = np.random.default_rng(seed)
    t0 = time.time()
    best_h = h_best.copy()
    best_c = float(c_best)
    extra_h = random_binary(n, rng)
    extra_c = official_c(extra_h)
    for s in range(max_starts):
        if time.time() - t0 > time_limit:
            break
        left = max(1.0, (time_limit - (time.time() - t0)) / float(max_starts - s))
        rb = random_binary(n, rng)
        hb, cb = binary_polish(rb, left, int(seed) + 1000 + s, max_passes=12)
        if cb < extra_c:
            extra_h = hb.copy()
            extra_c = cb
        if cb < best_c - 1e-12:
            best_h = hb.copy()
            best_c = cb
    return best_h, best_c, extra_h, extra_c


def local_pair_polish(h, time_limit, seed, max_evals=30000):
    h = _clean(h)
    n = len(h)
    if n < 2 or time_limit <= 0:
        return h, official_c(h)
    S = float(n) / 2.0
    h = project_sum_box(h.copy(), S)
    best = official_c(h)
    if not math.isfinite(best) or best <= 1e-8:
        return h, best
    rng = np.random.default_rng(seed)
    t0 = time.time()
    evals = 0
    stall = 0
    top = 12 if n <= 100 else 10
    fracs = (0.02, 0.06, 0.15, 0.35, 0.80)
    while time.time() - t0 < time_limit and evals < max_evals and stall < 25:
        cvals = custom_c(h)
        w = make_weights(cvals, 250.0)
        g = weighted_gradient(h, w)
        order = np.argsort(g)
        low = order[:top]
        high = order[-top:]
        pairs = []
        for i in low:
            ii = int(i)
            for j in high:
                jj = int(j)
                if ii != jj:
                    pairs.append((ii, jj))
        for _ in range(20):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj:
                pairs.append((ii, jj))
                pairs.append((jj, ii))
        improved = False
        for ii, jj in pairs:
            if evals >= max_evals:
                break
            dmax = min(float(h[ii]), 1.0 - float(h[jj]))
            if dmax <= 1e-10:
                continue
            local_best = best
            local_h = None
            for fr in fracs:
                if evals >= max_evals:
                    break
                d = float(fr) * dmax
                if d <= 1e-11:
                    continue
                ht = h.copy()
                ht[ii] += d
                ht[jj] -= d
                if (ht[ii] < -1e-12 or ht[ii] > 1.0 + 1e-12 or
                        ht[jj] < -1e-12 or ht[jj] > 1.0 + 1e-12):
                    continue
                c = official_c(ht)
                evals += 1
                if c < local_best - 1e-13:
                    local_best = c
                    local_h = ht
            if local_h is not None and local_best < best - 1e-12:
                h = local_h
                best = local_best
                improved = True
                break
        if improved:
            stall = 0
        else:
            stall += 1
    h = project_sum_box(h, S)
    best = official_c(h)
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    budget = float(budget_s)

    best_h = get_initial_h()
    n = int(len(best_h))
    best_c = official_c(best_h)

    if n % 2 == 1:
        m = 2 * (n // 2)
        if m < 2:
            m = 70
        best_h = resample(best_h, m)
        n = int(len(best_h))
        best_c = official_c(best_h)

    rem = budget - (time.time() - t_start)
    extra_h = np.zeros(n)
    extra_c = 1.0
    if rem > 60 and n % 2 == 0:
        best_h, best_c, extra_h, extra_c = binary_rounds(
            best_h, best_c, n, min(30.0, rem * 0.10), int(seed) + 11, max_starts=2
        )
        rem = budget - (time.time() - t_start)
        if rem > 30 and extra_c < best_c + 0.05 and len(extra_h) == n:
            temp = slsqp_refine(extra_h, 60.0, 90 if n <= 90 else 40)
            tc = official_c(temp)
            if tc < best_c - 1e-12:
                best_h = temp
                best_c = tc

    rem = budget - (time.time() - t_start)
    if rem > 60:
        maxit = 90 if n <= 90 else 60
        best_h, best_c = slsqp_rounds(
            best_h, best_c, (5.0, 12.0, 30.0, 80.0, 200.0, 500.0),
            maxit, min(320.0, rem * 0.45), int(seed) + 23, max_calls=20
        )

    rem = budget - (time.time() - t_start)
    if rem > 60:
        me = 60000 if n <= 100 else 20000
        best_h, best_c = local_pair_polish(
            best_h, min(90.0, rem * 0.25), int(seed) + 37, max_evals=me
        )

    rem = budget - (time.time() - t_start)
    up2 = min(280, n * 2)
    if up2 > n and rem > 200:
        h2 = resample(best_h, up2)
        c2 = official_c(h2)
        if c2 < best_c - 1e-12:
            best_h = h2
            best_c = c2
        maxit2 = 40 if up2 <= 160 else 25
        h2o, c2o = slsqp_rounds(
            h2, c2, (10.0, 30.0, 100.0, 300.0),
            maxit2, min(220.0, rem * 0.45), int(seed) + 53, max_calls=10
        )
        if c2o < best_c - 1e-12:
            best_h = h2o
            best_c = c2o
        cand_h = h2o if c2o < c2 else h2
        rem = budget - (time.time() - t_start)
        if rem > 60:
            me2 = 40000 if up2 <= 160 else 15000
            h2p, c2p = local_pair_polish(
                cand_h, min(60.0, rem * 0.25), int(seed) + 71, max_evals=me2
            )
            if c2p < best_c - 1e-12:
                best_h = h2p
                best_c = c2p

    rem = budget - (time.time() - t_start)
    cur_n = int(len(best_h))
    up3 = min(560, cur_n * 2)
    if up3 > cur_n and rem > 350:
        h3 = resample(best_h, up3)
        c3 = official_c(h3)
        if c3 < best_c - 1e-12:
            best_h = h3
            best_c = c3
        maxit3 = 25 if up3 <= 280 else 15
        h3o, c3o = slsqp_rounds(
            h3, c3, (20.0, 80.0, 300.0),
            maxit3, min(150.0, rem * 0.40), int(seed) + 89, max_calls=6
        )
        if c3o < best_c - 1e-12:
            best_h = h3o
            best_c = c3o

    rem = budget - (time.time() - t_start)
    if rem > 50:
        nfin = int(len(best_h))
        maxitf = 120 if nfin <= 90 else (60 if nfin <= 160 else 25)
        best_h, best_c = slsqp_rounds(
            best_h, best_c, (250.0, 700.0, 1500.0),
            maxitf, min(120.0, rem * 0.35), int(seed) + 101, max_calls=6
        )

    rem = budget - (time.time() - t_start)
    if rem > 15:
        mef = 60000 if len(best_h) <= 100 else 25000
        best_h, best_c = local_pair_polish(
            best_h, min(70.0, rem * 0.60), int(seed) + 103, max_evals=mef
        )

    safe_h = np.asarray(best_h, dtype=float).copy()
    safe_c = float(best_c)

    best_h = project_sum_box(best_h, float(len(best_h)) / 2.0)
    final_c = official_c(best_h)

    if not math.isfinite(final_c) or final_c > safe_c + 1e-9:
        best_h = safe_h
        final_c = safe_c

    best_h = project_sum_box(best_h, float(len(best_h)) / 2.0)
    final_c = official_c(best_h)

    if not math.isfinite(final_c):
        best_h = safe_h
        best_h = project_sum_box(best_h, float(len(best_h)) / 2.0)
        final_c = official_c(best_h)

    return best_h, float(final_c), int(len(best_h))
