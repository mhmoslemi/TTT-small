                    best_h, best_score = pgd(best_h, n, 3000, 0.01, top_k=3, decay=0.99995)
            except Exception:
                pass

    # Phase 3: Random restarts
    remaining = budget_s - (time.time() - t0)
    n_restarts = max(2, int(remaining / 28))
    for r in range(n_restarts):
        if time.time() - t0 > budget_s - 35:
            break
        # Gaussian perturbation restart
        scale = 0.05 + 0.15 * rng.random()
        h_p = best_h + rng.randn(n) * scale
        h_p = project_onto_constraint(h_p, target_sum)
        hk = 2 + rng.randint(0, 4)
        hd = 0.9999 + rng.random() * 0.0001
        lr_v = 0.015 + 0.02 * rng.random()
        hr, sr = pgd(h_p, n, 2500, lr_v, top_k=hk, decay=hd)
        if sr < best_score:
            best_score, best_h = sr, hr.copy()

        # Binary-structured restart
        if time.time() - t0 < budget_s - 35:
            h_b = random_binary_start(n, rng)
            # Smooth it slightly
            h_b = h_b * 0.7 + best_h * 0.3
            h_b = project_onto_constraint(h_b, target_sum)
            hr, sr = pgd(h_b, n, 2000, 0.02, top_k=3, decay=0.9999)
            if sr < best_score:
                best_score, best_h = sr, hr.copy()

    # Phase 4: Final fine-grained refinement
    if budget_s - (time.time() - t0) > 8:
        best_h, best_score = pgd(best_h, n, 6000, 0.005, top_k=5, decay=0.99997)

    # Phase 5: One more scipy polish
    if budget_s - (time.time() - t0) > 15:
        try:
            res = minimize(smooth_obj, best_h, args=(n, 300),
                           method='L-BFGS-B', options={'maxiter': 100, 'ftol': 1e-14})
            hr = project_onto_constraint(res.x, target_sum)
            sr = evaluate_h(hr, dx)
            if sr < best_score:
                best_score, best_h = sr, hr.copy()
        except Exception:
            pass

    best_h = project_onto_constraint(best_h, target_sum)
    best_score = evaluate_h(best_h, dx)
    return best_h, best_score, n
