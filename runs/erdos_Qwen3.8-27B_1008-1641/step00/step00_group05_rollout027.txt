import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(h - lam, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-14:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if np.any(mask):
            proj[mask] += diff / np.count_nonzero(mask)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def local_search(h, n_iter, start_temp, end_temp, seed=42):
    rng = np.random.default_rng(seed)
    n = len(h)
    h = h.copy()

    h = project_feasible(h)

    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    for it in range(n_iter):
        # Annealing schedule
        t = it / max(n_iter - 1, 1)
        temp = start_temp * (end_temp / start_temp) ** t

        j = rng.integers(n)
        if i == j:

        delta = direction * rng.uniform(0.01, 0.2)

        new_j = h[j] - delta

        h[j] = new_j
