            if time.time() - start_time < budget_s - 30:
                h3, c5_3 = projected_gradient(h_cur, ts_new, n_new, rng,
                                              max_iter=15000, start_time=start_time,
                                              budget_s=budget_s, reserve=25)
                if c5_3 < c5_cur:
                    c5_cur = c5_3
                    h_cur = h3
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = h_cur.copy()
                    best_n = n_new

    # Phase 4: Final SLSQP polish
    if time.time() - start_time < budget_s - 20:
        final_n = len(best_h)
        final_ts = final_n / 2.0
        m_final = 60.0
        res_x = try_slmq(best_h, final_ts, m_final, 200)
        if res_x is not None:
            h_try = project_feasible(res_x, final_ts)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Final projection
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
