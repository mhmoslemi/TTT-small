def project_to_simplex(h, target_sum):
    """Project h to [0,1]^n with sum = target_sum using iterative method."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not mask.any():
                break
            h[mask] -= diff / mask.sum()
        else:
            mask = h < 1
            if not mask.any():
                break
            h[mask] += (-diff) / mask.sum()
        h = np.clip(h, 0.0, 1.0)

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = eval_c5(best_h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Projected subgradient descent
    lr = 0.015
    stuck = 0
    best_c5_p1 = best_c5
    for it in range(15000):
        if elapsed() > budget_s * 0.55:
            break
        grad = grad_max_corr(best_h)
        h_try = best_h - lr * grad
        h_try = project_feasible(h_try, target_sum)
        c5_try = eval_c5(h_try, dx)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            best_c5_p1 = c5_try
            stuck = 0
        else:
            stuck += 1
            lr *= 0.98
        if stuck > 100:
            best_h = project_feasible(best_h + rng.normal(0, 0.04, n_points), target_sum)
            lr = 0.015
            stuck = 0
        lr = max(lr, 0.001)

    # Phase 2: Coordinate descent
    deltas_list = [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]
    no_improve_cd = 0
    for it in range(200):
        if elapsed() > budget_s * 0.80:
            break
        improved_f = False
        for j in range(n_points):
            if elapsed() > budget_s * 0.80:
                break
            c5_cur = best_c5
            for d in deltas_list:
                h_t = best_h.copy()
                h_t[j] += d
                h_t = project_feasible(h_t, target_sum)
                c5_t = eval_c5(h_t, dx)
                if c5_t < c5_cur - 1e-10:
                    best_h = h_t
                    best_c5 = c5_t
                    c5_cur = c5_t
                    improved_f = True
        if not improved_f:
            no_improve_cd += 1
            if no_improve_cd >= 3:
                break
        else:
            no_improve_cd = 0

    # Phase 3: Fine coordinate descent
    fine_deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02]
    for it in range(50):
        if elapsed() > budget_s * 0.90:
            break
        imp = False
        for j in range(n_points):
            for d in fine_deltas:
                h_t = best_h.copy()
                h_t[j] += d
                h_t = project_feasible(h_t, target_sum)
                c5_t = eval_c5(h_t, dx)
                if c5_t < best_c5 - 1e-10:
                    best_h = h_t
                    best_c5 = c5_t
                    imp = True
                    break
        if not imp:
            break

    # Phase 4: Random restarts with short local search
    remaining = budget_s * 0.95 - elapsed()
    n_rest = min(8, int(remaining / 15.0))
    for r in range(n_rest):
        if elapsed() > budget_s * 0.96:
            break
        h_r = project_feasible(best_h + rng.normal(0, 0.06, n_points), target_sum)
        lr_r = 0.008
        for it in range(2000):
            if elapsed() > budget_s * 0.97:
                break
            grad = grad_max_corr(h_r)
            h_t = project_feasible(h_r - lr_r * grad, target_sum)
            c5_t = eval_c5(h_t, dx)
            if c5_t < eval_c5(h_r, dx):
                h_r = h_t
                lr_r = min(lr_r * 1.02, 0.05)
            else:
                lr_r *= 0.95
        if eval_c5(h_r, dx) < best_c5:
            best_h = h_r
            best_c5 = eval_c5(best_h, dx)

    # Final: one more coordinate pass
    if elapsed() < budget_s * 0.97:
        for j in range(n_points):
            for d in [-0.03, -0.01, -0.005, 0.005, 0.01, 0.03]:
                h_t = best_h.copy()
                h_t[j] += d
                h_t = project_feasible(h_t, target_sum)
                c5_t = eval_c5(h_t, dx)
                if c5_t < best_c5 - 1e-10:
                    best_h = h_t
                    best_c5 = c5_t
                    break

    best_h = project_feasible(best_h, target_sum)
    best_c5 = eval_c5(best_h, dx)
    return best_h, best_c5, n_points
