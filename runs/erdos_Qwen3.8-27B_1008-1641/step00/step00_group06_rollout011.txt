import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)


def fix_sum(h, target):
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = (h > 0) & (h < 1)
        if mask.any():
            h[mask] += diff / np.sum(mask)
        else:
            break
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    t_start = time.time()

    if initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.ones(95) * 0.5
    n = len(h_best)
    target = n / 2.0
    h_best = fix_sum(h_best, target)
    c5_best = compute_c5(h_best)

    bounds = [(0.0, 1.0)] * n

        return compute_c5(x)

    def con_fun(x):
        return np.sum(x) - target

    def con_jac(x):
        return np.ones(n)

    constraints = {'type': 'eq', 'fun': con_fun, 'jac': con_jac}
