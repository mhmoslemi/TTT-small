        else:
            break

    # Phase 2: DE on n0
    if tl() > 80:
        n = n0
        bounds = [(0.0, 1.0)] * n
        res = differential_evolution(
            lambda x: obj_de(x, n), bounds,
            seed=seed, maxiter=800, popsize=30,
            tol=1e-10, mutation=(0.5, 1.2), crossover=0.8, polish=False
        )
        h_de = repair(res.x, n)
        c_de = compute_c5(h_de, n)
        if c_de < best_c5:
            best_h, best_c5, best_n = h_de, c_de, n
        if tl() > 30:
            h_r, c_r = local_opt(h_de, n)
            if c_r < best_c5:
                best_h, best_c5 = h_r, c_r

    # Phase 3: Try n=48 (even, pure 0-1 possible)
    if tl() > 60:
        n48 = 48
        for attempt in range(8):
            if tl() < 20:
                break
            h_rand = make_balanced_01(n48, rng)
            c_r = compute_c5(h_rand, n48)
            if c_r < best_c5:
                best_h, best_c5, best_n = h_rand, c_r, n48
        if tl() > 40:
            res = differential_evolution(
                lambda x: obj_de(x, n48), [(0.0, 1.0)] * n48,
                seed=seed + 7, maxiter=400, popsize=20,
                tol=1e-10, mutation=(0.6, 1.2), crossover=0.7, polish=False
            )
            h_de = repair(res.x, n48)
            c_de = compute_c5(h_de, n48)
            if c_de < best_c5:
                best_h, best_c5, best_n = h_de, c_de, n48
            if tl() > 20:
                h_r, c_r = local_opt(h_de, n48)
                if c_r < best_c5:
                    best_h, best_c5 = h_r, c_r

    # Phase 4: Perturbation + local refine on current best
    n = best_n
    for _ in range(30):
        if tl() < 8:
            break
        scale = 0.15
        h_pert = best_h + rng.randn(n) * scale
        h_pert = repair(h_pert, n)
        h_r, c_r = local_opt(h_pert, n, maxiter=3000)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r
            scale = max(0.03, scale * 0.7)
        else:
            scale = min(0.3, scale * 1.1)

    # Phase 5: Second DE with different params
    if tl() > 40:
        n = best_n
        res = differential_evolution(
            lambda x: obj_de(x, n), [(0.0, 1.0)] * n,
            seed=seed + 100, maxiter=500, popsize=25,
            tol=1e-11, mutation=(0.3, 0.8), crossover=0.9, polish=False
        )
        h_de = repair(res.x, n)
        c_de = compute_c5(h_de, n)
        if c_de < best_c5:
            best_h, best_c5 = h_de, c_de
        if tl() > 15:
            h_r, c_r = local_opt(best_h, n)
            if c_r < best_c5:
                best_h, best_c5 = h_r, c_r

    # Phase 6: Final tight local optimization
    if tl() > 5:
        h_r, c_r = local_opt(best_h, best_n, maxiter=10000)
        if c_r < best_c5:
            best_h, best_c5 = h_r, c_r

    # Ensure exact constraint satisfaction
    best_h = repair(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
