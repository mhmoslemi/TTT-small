def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(values, tau):
    v = np.asarray(values, dtype=float)
    m = np.max(v)
    if m < 0 and m < -100*tau:
        return m
    return m + tau * np.log(np.sum(np.exp((v - m) / tau)))

def enforce_constraint(h, n):
    """Ensure 0<=h<=1 and sum(h) = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Distribute among non-boundary elements
    mask = (h > 1e-6) & (h < 1.0 - 1e-6)
    n_mask = np.sum(mask)
    if n_mask == 0:
        # All at boundary, use all elements
        mask = np.ones(n, dtype=bool)
        n_mask = n
    h[mask] += diff / n_mask
    h = np.clip(h, 0.0, 1.0)
    # If still off, do a second pass
    diff2 = target - np.sum(h)
    if abs(diff2) > 1e-10:
        mask2 = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask2) > 0:
            h[mask2] += diff2 / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def smooth_max(v, tau):
    m = np.max(v)
    return m + tau * np.log(np.sum(np.exp((v - m) / tau)))

def fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    mask = (h > 1e-4) & (h < 1.0 - 1e-4)
    if np.sum(mask) > 0 and abs(diff) > 1e-12:
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff2 = n / 2.0 - np.sum(h)
    if abs(diff2) > 1e-10:
        mask2 = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask2) > 0:
            h[mask2] += diff2 / np.sum(mask2)
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_lspace(h, n, tau):
    hc = np.clip(h, 0.0, 1.0)
    dx = 2.0 / n
    corr = np.correlate(hc, 1.0 - hc, mode='full') * dx
    return smooth_max(corr, tau) + 5.0 * (np.sum(hc) - n / 2.0) ** 2

# Global workspace for coordinate descent sub-objective
_G_H = None
_G_I = 0
_G_N = 47

def _cd_subobj(x):
    global _G_H, _G_I, _G_N
    ht = _G_H.copy()
    ht[_G_I] = x
    diff = _G_N / 2.0 - np.sum(ht)
    ot = np.ones(_G_N, dtype=bool)
    ot[_G_I] = False
    ht[ot] -= diff / (_G_N - 1)
    ht = np.clip(ht, 0.0, 1.0)
    return compute_c5(ht, _G_N)

def coord_descent(h, n, max_rounds, budget_check):
    global _G_H, _G_I, _G_N
    hcd = h.copy()
    _G_N = n
    for _ in range(max_rounds):
        if not budget_check():
            break
        for i in range(n):
            if not budget_check():
                break
            _G_H = hcd
            _G_I = i
            try:
                r = minimize_scalar(_cd_subobj, bounds=(0.0, 1.0), method='bounded',
                                    options={'xatol': 1e-7})
                hcd[i] = r.x
            except Exception:
                hcd[i] = hcd[i]
            # Maintain sum constraint
            diff = n / 2.0 - np.sum(hcd)
            if abs(diff) > 1e-12:
                ot = np.ones(n, dtype=bool)
                ot[i] = False
                hcd[ot] -= diff / (n - 1)
                hcd = np.clip(hcd, 0.0, 1.0)
    return hcd

def is_prime(p):
    if p < 2:
        return False
    for q in range(2, int(p ** 0.5) + 1):
        if p % q == 0:
            return False
    return True

def legendre_h(n):
    leg = np.zeros(n)
    for i in range(1, n):
        leg[i] = 1.0 if pow(i, (n - 1) // 2, n) == 1 else -1.0
    h_leg = (leg + 1.0) / 2.0
    return h_leg
