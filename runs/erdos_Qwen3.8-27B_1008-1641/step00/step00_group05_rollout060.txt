import numpy as np
import time

def proj(h, ts):
    """Project h onto {0<=x<=1, sum(x)=ts}"""
    if abs(h.sum() - ts) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        m = (lo + hi) * 0.5
        if np.clip(h + m, 0, 1).sum() < ts:
            lo = m
        else:
            hi = m
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def c5_eval(h):
    return np.correlate(h, 1.0 - h, "full").max() * (2.0 / len(h))

def subgrad(h, ak, n):
    s = ak - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        g[-s:] += 1.0 - h[:n + s]
        g[:n + s] -= h[-s:]
    return g

def do_subgrad(h, n, ts, t0, tb):
    bh = h.copy()
    bc = c5_eval(h)
    lr = 0.1
    for _ in range(2000000):
        if time.time() - t0 > tb:
            break
        corr = np.correlate(h, 1.0 - h, "full")
        ak = int(np.argmax(corr))
        g = subgrad(h, ak, n)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        h = proj(h - lr * g / gn, ts)
        c = c5_eval(h)
        if c < bc:
            bc = c
            bh = h.copy()
        lr *= 0.99998
    return bh, bc

def do_perturb(h, n, ts, rng, t0, tb, sigma, kmin, kmax):
    bh = h.copy()
    bc = c5_eval(h)
    for _ in range(2000000):
        if time.time() - t0 > tb:
            break
        ht = bh.copy()
        k = int(rng.integers(kmin, kmax))
        idx = rng.choice(n, k, replace=False)
        ht[idx] += rng.normal(0, sigma, k)
        ht = proj(ht, ts)
        c = c5_eval(ht)
        if c < bc:
            bc = c
            bh = ht
    return bh, bc

def do_binary(h, n, ts, rng, t0, tb):

    for _ in range(2000000):
        if time.time() - t0 > tb:

        ht = proj(ht, ts)
        c = c5_eval(ht)
        if c < bc:
            bc = c
            bh = ht
    return bh, bc

def do_transport(h, n, ts, rng, t0, tb, eps):
    """Pair transport: move mass from one coordinate to another"""
    bh = h.copy()
    bc = c5_eval(h)
    for _ in range(2000000):
        if time.time() - t0 > tb:
            break
        ht = bh.copy()
        i, j = int(rng.integers(0, n)), int(rng.integers(0, n))
        if i == j:
            continue
        d = eps * (1.0 if rng.random() < 0.5 else -1.0)
        if ht[i] - d < 0 or ht[j] + d > 1:
            continue
        ht[i] -= d
        ht[j] += d
        ht = proj(ht, ts)
        c = c5_eval(ht)
        if c < bc:
            bc = c
            bh = ht
    return bh, bc
