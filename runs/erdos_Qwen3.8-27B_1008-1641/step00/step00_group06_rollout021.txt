            if tot < 1e-15:

    n_targets = [49, 65, 81, 97, 113, 129, 161, 193, 241]
    if best_h is not None:
        n0 = len(best_h)
        if n0 not in n_targets:
            n_targets.insert(0, n0)
        n_targets.remove(n0)
        n_targets.insert(0, n0)

    for n in n_targets:
        if time_left() < 40:
            break
        styles = ['quin', 'block', 'rand']
        h0_best = best_h.copy() if (best_h is not None and len(best_h) == n) else None

        for style in styles:
            if time_left() < 20:
                break
            if h0_best is not None and style == 'quin':
                h0 = h0_best.copy()
            else:
                h0 = random_init(n, rng, style=style)

            for alpha in [50.0, 120.0]:
                if time_left() < 10:
                    break
                h_g, c5_g = grad_opt(h0, n, t0, budget_s, alpha=alpha, lam=150.0, max_iter=250)
                if time_left() > 20:
                    h_g, c5_g2 = local_search(h_g, t0, budget_s, max_secs=min(10, time_left() * 0.08))
                    c5_g = min(c5_g, c5_g2)
                if c5_g < best_c5:
                    best_c5 = c5_g
                    best_h = h_g.copy()
                    h0_best = h_g.copy()
                h0 = h_g

    # Phase 4: Final polish - more local search + gradient on best
    if time_left() > 20 and best_h is not None:
        h_fin, c5_fin = local_search(best_h, t0, budget_s, max_secs=min(30, time_left() * 0.3))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
        n_fin = len(best_h)
        if time_left() > 15:
            h_g, c5_g = grad_opt(best_h, n_fin, t0, budget_s, alpha=200.0, lam=500.0, max_iter=300)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g

    # Final verification
    if best_h is None:
        best_h = np.ones(49) * (50.0 / 49.0)
        best_h = fix_sum(np.clip(best_h, 0, 1), 49)
        best_c5 = c5_value(best_h)

    best_c5 = c5_value(best_h)
    return best_h, best_c5, len(best_h)
