    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        # Use water-filling style correction
        deficit = target - s
        if deficit > 0:
            mask = h < 1.0

        else:
            mask = h > 0.0
            if not np.any(mask):
                break
            h[mask] += deficit / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        if len(idx) > 0:

            h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):

        diff = target - s
        if abs(diff) < 1e-14:

        else:

            break

        h = np.clip(h, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

    m = np.max(corr)
    return dx * (m + T * np.log(np.sum(np.exp((corr - m) / T))))

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode="full")

            if c < best:
                best = c
                best_h = h.copy()
        except Exception:
            pass
    return best_h, best

def combi_refine(h0, n, rng, time_limit):
    h_bin = (h0 >= 0.5).astype(float)
    target = n // 2
    while int(h_bin.sum()) > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.integers(len(idx))]] = 0
    while int(h_bin.sum()) < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.integers(len(idx))]] = 1
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    ones_l = list(np.where(best_h == 1)[0])
    zeros_l = list(np.where(best_h == 0)[0])
    deadline = time.time() + time_limit
    stall = 0
    while time.time() < deadline and stall < 80:
        h_cur = best_h.copy()
        ones_l = list(np.where(h_cur == 1)[0])
        zeros_l = list(np.where(h_cur == 0)[0])
        improved = False
        remaining_t = max(0.1, deadline - time.time())
        n_trials = int(min(30000, max(2000, remaining_t * 300)))
        for _ in range(n_trials):
            i = ones_l[rng.integers(len(ones_l))]
            j = zeros_l[rng.integers(len(zeros_l))]
            h_cur[i] = 0.0
            h_cur[j] = 1.0
            c = compute_c5(h_cur)
            if c < best_c5 - 1e-12:
                best_c5 = c
                best_h = h_cur.copy()
                ones_l[i] = j
                zeros_l[j] = i
                improved = True
            else:
                h_cur[i] = 1.0
                h_cur[j] = 0.0
        if improved:
            stall = 0
        else:
            stall += 1
    return best_h, best_c5

def make_starts(n, rng, k):
    starts = []
    for i in range(k):
        if i == 0:
            starts.append(np.full(n, 0.5))
        elif i == 1:
            h = np.zeros(n)
            h[::2] = 1.0
            starts.append(h)
        elif i == 2:
            h = np.zeros(n)
            h[:n//2] = 1.0
            starts.append(h)
        elif i == 3:
            h = np.zeros(n)
            h[::3] = 1.0
            h[1::3] = 1.0
            starts.append(h)
        else:
            starts.append(rng.random(n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    best_h = project_feasible(h_init, n)
    best_c5 = compute_c5(best_h)

    # Phase 1: optimize from initial
    h1, c1 = lgbfgs_run(best_h, n, max_iter=600)
    if c1 < best_c5:
        best_h, best_c5 = h1, c1

    # Phase 2: multi-start
    combi_reserved = 120
    deadline_multi = t0 + budget_s - combi_reserved
    starts = make_starts(n, rng, 4)
    si = 0
    while time.time() < deadline_multi:
        if si < len(starts):
            hp = starts[si]
        else:
            scale = 0.4 / (1.0 + si * 0.15)
            hp = best_h + rng.normal(0, scale, n)
        hp = project_feasible(hp, n)
        h2, c2 = lgbfgs_run(hp, n, max_iter=300)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2
        si += 1

    # Phase 3: try binarizing and combinatorial
    remaining = max(10, budget_s - (time.time() - t0) - 5)
    hc, cc = combi_refine(best_h, n, rng, remaining)
    if cc < best_c5:
        best_h, best_c5 = hc, cc

    # Phase 4: one more L-BFGS from best (continuous)
    if time.time() - t0 < budget_s - 10:
        h4, c4 = lgbfgs_run(best_h, n, max_iter=300)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Final
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
