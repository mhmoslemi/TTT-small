def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get initial values
    if 'initial_h_values' in dir() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)

    else:

    best_h = parent.copy()
    
    # Try multiple n values and starting points
    n_candidates = [n_parent]
    for n in range(20, 200, 5):

                break

            elif attempt == 1:

                if needed > 0:
                    zeros = np.where(h0 == 0)[0]
                    for idx in zeros[:int(needed)]:
                        h0[idx] = 1.0
            else:

                if n == n_parent:

                else:

                h0 *= target_sum / max(np.sum(h0), 1e-10)
                h0 = np.clip(h0, 0, 1)

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    # Water filling
    diff = target_sum - s
    if diff > 0:
        # Need to increase sum
        order = np.argsort(-h)  # largest first (least room to increase)
        for i in order:
            if diff <= 1e-10:
                break
            room = 1.0 - h[i]
            inc = min(room, diff)
            h[i] += inc
            diff -= inc
    else:
        # Need to decrease sum
        order = np.argsort(h)  # smallest first (least room to decrease)
        diff = -diff
        for i in order:
            if diff <= 1e-10:
                break
            room = h[i]
            dec = min(room, diff)
            h[i] -= dec
            diff -= dec
    return h
