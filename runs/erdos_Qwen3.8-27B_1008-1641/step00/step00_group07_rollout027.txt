import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(x, target_sum):
    """Project x onto {0<=y[i]<=1, sum(y)=target_sum}"""
    # Bisection on lambda
    lo, hi = -np.max(np.abs(x)) - 1, np.max(np.abs(x)) + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0, 1)
        s = np.sum(y)
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2, 0, 1)
    # Final correction: adjust the element closest to boundary
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-12:
        # Find adjustable elements
        adjustable = (y > 1e-12) & (y < 1 - 1e-12)
        if np.any(adjustable):
            y[adjustable] += diff / np.sum(adjustable)
            y = np.clip(y, 0, 1)
    return y

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return np.max(c) * dx, c

def subgradient(h, n, s):
    """Compute subgradient of correlation at shift s.
    s = k - (n-1), range [-(n-1), n-1]
    c[k] = sum_{j valid} h[j] * (1-h[j+s])
    valid: max(0,-s) <= j <= min(n-1, n-1-s)
    """
    j_lo = max(0, -s)
    j_hi = min(n - 1, n - 1 - s)
    grad = np.zeros(n)
    # First term: (1-h[i+s]) for i in [j_lo, j_hi]
    for i in range(j_lo, j_hi + 1):
        grad[i] += (1 - h[i + s])
    # Second term: -h[i-s] for i-s in [j_lo, j_hi], i in [0, n-1]
    # i-s in [j_lo, j_hi] => i in [j_lo+s, j_hi+s]
    i_lo2 = max(0, j_lo + s)
    i_hi2 = min(n - 1, j_hi + s)
    for i in range(i_lo2, i_hi2 + 1):
        grad[i] -= h[i - s]
    return grad

import time

def project_feasible(x, target_sum):

    for _ in range(80):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0, 1)
        if np.sum(y) > target_sum:

        if np.any(adj):
            y[adj] += diff / np.sum(adj)
    return np.clip(y, 0, 1)

    return np.correlate(h, 1.0 - h, mode="full")

def c5_val(h):
    n = len(h)
    return np.max(compute_c(h)) * (2.0 / n)

def subgrad_for_shift(h, n, s):
    j_lo = max(0, -s)
    j_hi = min(n - 1, n - 1 - s)
    g = np.zeros(n)
    g[j_lo:j_hi+1] += (1.0 - h[j_lo+s:j_hi+1+s])
    i2_lo = max(0, j_lo + s)
    i2_hi = min(n - 1, j_hi + s)
    g[i2_lo:i2_hi+1] -= h[i2_lo-s:i2_hi+1-s]
    return g

    c = compute_c(h)
    scaled = c * dx / eps

import numpy as np
import time

def project_feasible(x, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0, 1))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0, 1)
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-10:
        adj = (y > 1e-12) & (y < 1.0 - 1e-12)
        if np.any(adj):
            y[adj] += diff / np.sum(adj)
    return np.clip(y, 0.0, 1.0)

def compute_c(h):
    return np.correlate(h, 1.0 - h, mode="full")

def c5_val(h):
    n = len(h)
    return float(np.max(compute_c(h))) * (2.0 / n)

def subgrad_for_shift(h, n, s):
    j_lo = max(0, -s)
    j_hi = min(n - 1, n - 1 - s)
    g = np.zeros(n)
    g[j_lo:j_hi + 1] += (1.0 - h[j_lo + s:j_hi + 1 + s])
    i2_lo = max(0, j_lo + s)
    i2_hi = min(n - 1, j_hi + s)
    g[i2_lo:i2_hi + 1] -= h[i2_lo - s:i2_hi + 1 - s]
    return g

def smooth_grad(h, n, eps):
    dx = 2.0 / n
    c = compute_c(h)
    scaled = c * dx / eps
    mx = np.max(scaled)
    w = np.exp(scaled - mx)
    wsum = np.sum(w)
    w /= wsum
    obj = (mx + np.log(wsum)) * eps
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] > 1e-6:
            s = k - (n - 1)
            grad += w[k] * subgrad_for_shift(h, n, s)
    return obj, grad * dx

def subgrad_refine(h, n, target, max_iters, t0, budget_s):
    lr = 0.004
    for it in range(max_iters):
        if time.time() - t0 > budget_s - 5:
            break
        c = compute_c(h)
        s_star = int(np.argmax(c)) - (n - 1)
        g = subgrad_for_shift(h, n, s_star)
        h_new = project_feasible(h - lr * g, target)
        cv_new = float(np.max(compute_c(h_new))) * (2.0 / n)
        cv_old = float(np.max(c)) * (2.0 / n)
        if cv_new < cv_old:
            h = h_new
            lr = min(lr * 1.03, 0.03)
        else:
            lr *= 0.85
            if lr < 0.0003:
                h = project_feasible(h + np.random.randn(n) * 0.004, target)
                lr = 0.004
    return h

def coord_descent(h, n, target, t0, budget_s, best_c5, best_h):

        if time.time() - t0 > budget_s - 3:
            break
        improved = False

            saved = h[j]

                h[j] = np.clip(saved + delta, 0.0, 1.0)
                h = project_feasible(h, target)
                cv = c5_val(h)

                    improved = True

            h[j] = np.clip(h[j] if abs(h[j] - saved) < 1e-12 else saved, 0, 1)
        if not improved:
            break
    return h, best_c5, best_h
